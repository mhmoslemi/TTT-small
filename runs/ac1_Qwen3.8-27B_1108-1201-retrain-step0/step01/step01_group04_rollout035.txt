We want to minimize: score = 2n * max(conv(p,p)) / (sum(p))^2

For non-negative p. The score is scale-invariant, so fix sum(p) = 1.

score = 2n * max_k C[k] where C[k] = (p*p)[k].

The key observation: for non-negative p, C[k] >= 0 for all k. The maximum is typically at the center (k = n-1) for "spread out" sequences, or at the edges (k=0 or k=2n-2) for "peaked" sequences.

The LP best-response approach:
- Given current p, compute M = max_k C[k].
- Find g >= 0 that maximizes sum(g) subject to (p*g)[k] <= M for all k.
- Blend: p_new = (1-t)p + t*g (normalized).
- If score improves, accept; else reduce t.

This is essentially a "convex optimization" step in the direction of g, which is the best feasible direction.

The convergence theory: this is related to the "alternating direction method" or "best response dynamics" in game theory. For concave games, best response dynamics converge to a Nash equilibrium. Here, the "game" is between p and the constraint set, and the best response g is the Nash equilibrium of a related zero-sum game.

In practice, this method converges quickly (often in < 100 iterations) to a local optimum.

"""
LP-driven search for low max-autocorrelation non-negative step functions.
Uses iterative LP best-response with mirror-descent polishing.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len
from scipy.sparse import csr_matrix


def _fft_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def conv_self(p):
    """Full self-convolution of 1D vector p."""
    n = len(p)
    if n == 0:
        return np.empty(0)
    L = _fft_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1]


def compute_score(p):
    """Compute the evaluation score for a non-negative sequence."""
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if s <= 1e-12:
        return np.inf
    c = conv_self(p)
    cmax = float(np.max(c)) if len(c) > 0 else 0.0
    return float(2.0 * n * cmax / (s * s))


def normalize(p):
    """Normalize to sum 1, non-negative."""
    p = np.asarray(p, dtype=np.float64)
    p = np.maximum(p, 0.0)
    p = np.where(np.isfinite(p), p, 0.0)
    s = float(p.sum())
    if s <= 1e-12:
        return np.full(len(p), 1.0/len(p))
    return p / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = u * idx > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0/n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-300:
        return np.full(n, 1.0/n)
    return out / s


def make_geometric(n, r):
    """Geometric sequence p[i] = r^i, normalized."""
    p = np.power(r, np.arange(n, dtype=np.float64))
    return normalize(p)


def make_tapered(n, alpha, beta):
    """Tapered: p[i] = (i+1)^alpha * (n-i)^beta, normalized."""
    i = np.arange(n, dtype=np.float64)
    p = np.power(i + 1.0, alpha) * np.power(n - i, beta)
    return normalize(np.maximum(p, 1e-30))


def make_linear_taper(n, a):
    """Linear taper: p[i] = 1 - a*i/(n-1), clamped to non-negative."""
    i = np.arange(n, dtype=np.float64)
    p = 1.0 - a * i / max(1, n-1)
    return normalize(np.maximum(p, 0.0))


def make_power_taper(n, a):
    """Power taper: p[i] = (1 - i/(n-1))^a."""
    i = np.arange(n, dtype=np.float64)
    x = 1.0 - i / max(1, n-1)
    p = np.power(np.maximum(x, 1e-10), a)
    return normalize(p)


def make_bimodal(n, mu1, mu2, sigma):
    """Two Gaussian bumps."""
    x = np.linspace(0, 1, n)
    p = np.exp(-((x-mu1)**2)/(2*sigma**2)) + np.exp(-((x-mu2)**2)/(2*sigma**2))
    return normalize(p)


def lp_best_response(p, M, budget_s=10.0):
    """
    Solve LP: max sum(g) s.t. conv(p,g) <= M, g >= 0.
    Returns (g, success).
    """
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    if n < 2:
        return p, False
    
    # Build constraint matrix A where A[k,j] = p[k-j] if 0 <= k-j < n
    # We only need the "first half" by symmetry, but let's do all for correctness.
    # Actually, for the LP we need all 2n-1 constraints.
    
    # Use the structure: A[k,j] = p[k-j] if 0 <= k-j < n else 0
    # This is a "convolution matrix" (Toeplitz-like).
    
    # For efficiency, build as sparse.
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            rows.append(k)
            cols.append(j)
            data.append(p[k - j])
    
    # Actually, for n=700, this matrix has n*(2n-1) - O(n^2) nonzeros ≈ n^2.
    # That's 490,000 nonzeros for n=700. Should be fine for HiGHS.
    
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs', options={'disp': False, 'time_limit': budget_s})
    except Exception:
        return p, False
    
    if result is None or not result.success:
        return p, False
    
    g = result.x
    g = np.maximum(g, 0.0)
    s = g.sum()
    if s <= 1e-12:
        return p, False
    g = g / s
    return g, True


def lp_optimize(p_init, max_iters=5000, time_limit=100.0):
    """
    Iterative LP best-response optimization.
    """
    p = normalize(p_init)
    n = len(p)
    start = time.monotonic()
    best_p = p.copy()
    best_score = compute_score(p)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        c = conv_self(p)
        M = float(np.max(c))
        current_score = 2.0 * n * M  # since sum(p) = 1
        
        if not np.isfinite(current_score):
            break
        
        # Solve LP for best response
        g, success = lp_best_response(p, M, budget_s=5.0)
        if not success:
            break
        
        # Line search on t
        accepted = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05, 0.02]:
            p_new = normalize((1.0 - t) * p + t * g)
            sc = compute_score(p_new)
            if np.isfinite(sc) and sc < current_score:
                p = p_new
                if sc < best_score:
                    best_score = sc
                    best_p = p.copy()
                accepted = True
                break
        
        if not accepted:
            # Try with slightly relaxed M
            M_relaxed = M * 1.01
            g, success = lp_best_response(p, M_relaxed, budget_s=5.0)
            if success:
                for t in [1.0, 0.5, 0.2, 0.1]:
                    p_new = normalize((1.0 - t) * p + t * g)
                    sc = compute_score(p_new)
                    if np.isfinite(sc) and sc < current_score:
                        p = p_new
                        if sc < best_score:
                            best_score = sc
                            best_p = p.copy()
                        accepted = True
                        break
        
        if not accepted:
            # Perturb and retry
            p = normalize(p + 0.01 * np.random.default_rng(it).normal(0, 1/n, n))
            if time.monotonic() - start > time_limit * 0.8:
                break
        
        if it % 50 == 0:
            print(f"  LP iter {it}: score={best_score:.10f} cur={current_score:.10f}")
    
    return best_p, best_score


def mirror_descent_polish(p, time_limit=30.0, T_init=0.01):
    """
    Mirror descent (multiplicative updates) polishing step.
    """
    p = normalize(np.asarray(p, dtype=np.float64))
    n = len(p)
    start = time.monotonic()
    best_p = p.copy()
    best_score = compute_score(p)
    T = T_init
    
    for it in range(100000):
        if time.monotonic() - start > time_limit:
            break
        
        c = conv_self(p)
        cmax = float(np.max(c))
        if cmax <= 0:
            break
        
        # Softmax weights over the convolution
        T_cur = max(T * cmax, 1e-10)
        diff = (c - cmax) / T_cur
        w = np.exp(diff)
        w /= w.sum()
        
        # Subgradient: SG[j] = 2 * sum_k w[k] * p[k-j] (for valid k-j)
        # This is 2 * (w * p) centered appropriately.
        # Actually: SG[j] = 2 * sum_k w[k] * p[k-j] where 0 <= k-j < n
        # Let's compute h = convolve(w, p) and extract the right part.
        # conv(w, p)[k] = sum_i w[i] * p[k-i]
        # We want: for each j, sum_k w[k] * p[k-j]
        # This is conv(w_reversed, p)[j] where w_reversed[i] = w[n-1-i]... 
        # Actually it's correlation: sum_k w[k]*p[k-j] = (w * p)[j] in the correlation sense.
        # In FFT terms: it's ifft(fft(w) * conj(fft(p)))... 
        # Let me just use np.correlate or manual computation.
        
        # Actually: sum_k w[k] * p[k-j] for fixed j.
        # Let's define this as the "cross-correlation" of w and p at lag j.
        # In numpy: this is np.correlate(w, p, mode='full')[n-1+j]
        # Or we can compute via FFT.
        
        L = _fft_len(n)
        Fw = fft(w, L)
        Fp = fft(p, L)
        # We want h[j] = sum_k w[k] * p[k-j]
        # = ifft(FFT(w) * FFT(reversed_p)) ... hmm
        # Let's think: ifft(FFT(w) * FFT(p))[m] = sum_k w[k]*p[m-k] (circular)
        # We want sum_k w[k]*p[k-j]. Let m = k-j, so k = m+j:
        # sum_m w[m+j]*p[m] = cross-correlation.
        # In terms of FFT: ifft(FFT(w_reversed) * FFT(p)) where w_reversed[k] = w[n-1-k]
        # Hmm, let me just use a direct approach for now.
        
        # Actually, the simplest: h = np.correlate(w, p, mode='full')
        # h[m] = sum_k w[k] * p[k + m - (n-1)]  ... this is getting confusing.
        
        # Let me just compute it directly using the convolution structure.
        # SG[j] = 2 * sum_k w[k] * p[k-j] (where 0 <= k-j < n, i.e., j <= k < j+n)
        # = 2 * sum_{k=j}^{j+n-1} w[k] * p[k-j] (with w[k]=0 for k >= 2n-1)
        
        # This is the same as: if I define the full convolution F = conv(w, p) (length 2n-1+... no)
        # Let me think again.
        # C[k] = sum_i p[i] * p[k-i], grad C[k][j] = 2*p[k-j] (if 0 <= k-j < n)
        # SG = sum_k w[k] * grad C[k] = sum_k w[k] * 2*p[k-j] (vectorized over j)
        # SG[j] = 2 * sum_k w[k] * p[k-j] (where 0 <= k-j < n)
        
        # Let's compute this via FFT:
        # Define v[m] = p[m] for 0<=m<n, else 0.
        # SG[j]/2 = sum_k w[k]*v[k-j] = sum_k w[k]*v[k-j]
        # Let l = k-j: = sum_l w[l+j]*v[l] = (w_shifted * v)[l] summed...
        # This is the cross-correlation: (w ⋆ v)[j] = sum_l w[l+j]*v[l]
        # In numpy: np.correlate(w, v, 'full')[n-1+j] ... not quite.
        
        # Simplest: use the fact that this is ifft(FFT(w) * conj(FFT(v))) in circular case.
        # For linear case, do it with zero-padding.
        
        Fw_full = fft(w, 3*n)  # enough padding
        Fv_full = fft(p, 3*n)
        h = np.real(ifft(Fw_full * np.conj(Fv_full)))
        # h[m] = sum_k w[k+m] * p[k] (circular, but with enough padding it's linear)
        # We want SG[j]/2 = sum_k w[k] * p[k-j] = sum_l w[l+j] * p[l] = h[j]
        
        SG = 2.0 * h[:n]
        SG = np.maximum(SG, 0.0)
        
        # Mirror descent step: p_new[j] = p[j] * exp(-lr * SG[j]) / Z
        lr = 1.0 / max(float(np.max(SG)), 1e-10) * 0.1
        log_p_new = np.log(np.maximum(p, 1e-300)) - lr * SG
        log_p_new -= np.max(log_p_new)  # for numerical stability
        p_new = np.exp(log_p_new)
        s = p_new.sum()
        if s <= 1e-12 or not np.isfinite(s):
            break
        p_new /= s
        
        sc = compute_score(p_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p_new.copy()
            p = p_new
            T = max(T * 0.99, 1e-4)
        else:
            T = max(T * 0.9, 1e-5)
            # Try smaller step
            lr = lr * 0.5
            log_p_new = np.log(np.maximum(p, 1e-300)) - lr * SG
            log_p_new -= np.max(log_p_new)
            p_new = np.exp(log_p_new)
            s = p_new.sum()
            if s > 1e-12 and np.isfinite(s):
                p_new /= s
                sc = compute_score(p_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_p = p_new.copy()
                    p = p_new
            if time.monotonic() - start > time_limit:

        if it % 200 == 0:

def interpolate_sequence(p, new_n):
    """Interpolate a sequence to a new length."""
    p = np.asarray(p, dtype=np.float64)
    old_n = len(p)
    if old_n == new_n:
        return p.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    p_new = np.interp(x_new, x_old, p)
    return normalize(p_new)
