"""
FISTA-based optimization with adaptive log-sum-exp smoothing for minimizing
the normalized peak autocorrelation of a non-negative sequence.
"""

import numpy as np
import time

def _project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def _fista(a_init, deadline, tau0=10.0, tau_max=500.0, eta=None, tau_rate=1.02, damp=1.0):
    """
    FISTA with adaptive tau for smooth max approximation.
    """
    n = len(a_init)
    x = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(x)
    if S > 1e-10:
        x /= S
    else:
        x = np.ones(n) / n
    
    z = x.copy()
    t = 1.0
    best = x.copy()
    best_score = _score(best)
    
    if eta is None:
        eta = 0.4
    
    tau = tau0
    no_improve = 0
    
    iter_count = 0
    while time.monotonic() < deadline:
        # Compute gradient at z
        c = np.convolve(z, z)
        c_max = np.max(c)
        exp_vals = np.exp(tau * (c - c_max))
        p = exp_vals / np.sum(exp_vals)
        
        # Gradient = 2 * convolve(p, z_rev)[n-1:n-1+n]
        z_rev = z[::-1]
        conv_pz = np.convolve(p, z_rev, mode='full')
        grad = 2.0 * conv_pz[n-1:n-1+n]
        grad -= np.mean(grad)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = damp * (t - 1.0) / t_new
        y = z + momentum * (z - x)
        t = t_new
        
        # Gradient step + projection
        x_new = y - eta * grad
        x_new = _project_simplex(x_new, 1.0)
        
        # Evaluate
        score = _score(x_new)
        if score < best_score - 1e-12:
            best = x_new.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
        
        x, z = z, x_new
        
        # Adapt tau
        if no_improve > 50:
            tau = min(tau * tau_rate, tau_max)
            no_improve = 0
            t = 1.0  # Reset momentum when changing tau
        
        iter_count += 1
    
    return best, best_score

def _subgradient(a_init, deadline, M=10, eta_init=None):
    """Subgradient descent (from previous approach, slightly modified)."""
    n = len(a_init)
    if eta_init is None:
        eta_init = 0.1 / np.sqrt(n)
    
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    
    best = a.copy()
    best_score = _score(best)
    eta = eta_init
    no_improve = 0
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 200:
            eta *= 0.8
            no_improve = 0
            if eta < 1e-12:
                break
        
        # Weighted gradient from top M peaks
        if M >= len(b):
            k_idx = np.arange(len(b))
            b_sel = b
        else:
            k_idx = np.argpartition(b, -M)[-M:]
            b_sel = b[k_idx]
        
        wts = np.exp(-3.0 * (b_max - b_sel) / max(b_max, 1e-15))
        wts /= np.sum(wts)
        
        # Compute gradient
        grad = np.zeros(n)
        for ki in range(len(k_idx)):
            if wts[ki] < 1e-10:
                continue
            k = k_idx[ki]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                grad[idx] += wts[ki] * 2.0 * a[k - idx]
        
        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        a = a - eta * grad / grad_norm
        a = _project_simplex(a, 1.0)
    
    return best, best_score

"""
Multi-start FISTA with adaptive log-sum-exp smoothing for minimizing
peak autocorrelation of non-negative sequences. Exploits problem structure
with anti-symmetric constructions. Includes subgradient polish.
"""

import numpy as np
import time

# Helper functions at top level (no closures, no lambdas)

def _project_simplex(v, target=1.0):
    """Project vector onto probability simplex {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def _score_normalized(a):
    """Compute 2*n*max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c)

def _fista_smooth(a_init, deadline, tau_start=5.0, tau_max=200.0, 
                   eta_init=0.5, damp=0.9, patience=100):
    """
    FISTA with log-sum-exp smooth max approximation.
    """
    n = len(a_init)
    x = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(x)
    if S > 1e-10:
        x /= S
    else:
        x = np.ones(n) / n
    
    z = x.copy()
    t = 1.0
    best = x.copy()
    best_score = _score_normalized(best)
    eta = eta_init
    tau = tau_start
    no_improve = 0
    tau_rate = 1.05
    
    while time.monotonic() < deadline:
        # Compute smooth gradient
        c = np.convolve(z, z)
        c_max = np.max(c)
        # Avoid overflow
        exp_vals = np.exp(tau * (c - c_max))
        p = exp_vals / np.sum(exp_vals)
        
        # Gradient via convolution
        z_rev = z[::-1]
        conv_pz = np.convolve(p, z_rev, mode='full')
        grad = 2.0 * conv_pz[n-1:n-1+n]
        grad -= np.mean(grad)
        
        # Scale eta with tau (L grows with tau)
        eta_eff = eta / max(1.0, tau / tau_start)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = damp * (t - 1.0) / t_new
        y = z + momentum * (z - x) if momentum != 0 else z.copy()
        t = t_new
        
        # Gradient step + projection
        x_new = y - eta_eff * grad
        x_new = _project_simplex(x_new, 1.0)
        
        # Evaluate
        score = _score_normalized(x_new)
        if score < best_score - 1e-12:
            best = x_new.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
        
        x, z = z, x_new
        
        # Adapt tau and restart
        if no_improve >= patience:
            tau = min(tau * tau_rate, tau_max)
            t = 1.0
            no_improve = 0
        
        # Safety: if eta is too small, break
        if eta_eff < 1e-10:
            break
    
    return best, best_score

def _subgradient_polish(a_init, deadline, M=20, eta_init=None, cool_rate=0.9):
    """Subgradient descent polish."""
    n = len(a_init)
    if eta_init is None:
        eta_init = 0.05 / np.sqrt(n)
    
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    
    best = a.copy()
    best_score = _score_normalized(best)
    eta = eta_init
    no_improve = 0
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max
        
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 500:
            eta *= cool_rate
            no_improve = 0
            if eta < 1e-14:
                break
        
        # Weighted gradient from top M peaks
        if M >= len(b):
            k_idx = np.arange(len(b))
            b_sel = b
        else:
            k_idx = np.argpartition(b, -M)[-M:]
            b_sel = b[k_idx]
        
        b_max_sel = np.max(b_sel)
        wts = np.exp(-2.0 * (b_max_sel - b_sel) / max(b_max_sel - np.min(b_sel), 1e-15))
        wts /= np.sum(wts)
        
        # Compute gradient efficiently using convolution
        w_rev = wts[::-1]
        conv_wa = np.convolve(a, w_rev, mode='full')
        grad = 2.0 * conv_wa[n-1:n-1+n]
        grad -= np.mean(grad)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        a = a - eta * grad / grad_norm
        a = _project_simplex(a, 1.0)
    
    return best, best_score

def _make_two_level(n, eps):
    """Two-level anti-symmetric sequence."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def _make_smooth(n, eps):
    """Smooth anti-symmetric: 1 + eps*cos(pi*i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def _make_quadratic(n, eps):
    """Quadratic profile: 1 + eps*(2i/(n-1) - 1)."""
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / (n - 1) - 1.0
    a = 1.0 + eps * t
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def _perturb(a, rng, scale=0.05):
    """Add smooth random perturbation."""
    n = len(a)
    freqs = rng.integers(1, max(2, n // 4), size=min(5, n // 2))
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    if np.max(np.abs(noise)) > 1e-15:
        noise /= np.max(np.abs(noise))
    a_new = a + scale * noise
    a_new = _project_simplex(a_new, 1.0)
    return a_new
