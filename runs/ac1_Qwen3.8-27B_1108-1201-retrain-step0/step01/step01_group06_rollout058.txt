"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive mirror descent
with log-sum-exp smoothing, momentum, and multi-start strategy.
Explores multiple sequence lengths and diverse initializations.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_sgrad(w, a):
    """
    Compute the subgradient of max(conv) w.r.t. a.
    Result: sg[j] = 2 * sum_k w[k] * a[k-j] (valid range)
    Only processes active weights for efficiency.
    """
    n = len(a)
    sg = np.zeros(n)
    thresh = w.max() * 0.001
    active_idx = np.where(w > thresh)[0]
    for k_int in active_idx:
        k = int(k_int)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]
    return 2.0 * sg


def optimize_adaptive(a0, time_limit, t_init=20.0, t_final=500.0,
                      eta0=0.5, beta1=0.9, beta2=0.95):
    """
    Adaptive mirror descent with log-sum-exp smoothing, momentum, and
    per-coordinate adaptive step sizes.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    n = len(a)
    a_sum = a.sum()
    a = a / a_sum
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    # Adam-like moments
    m = np.zeros(n)
    v = np.ones(n)  # initialize to 1 to avoid division by zero early on
    
    t0 = time.monotonic()
    current_t = t_init
    step = 0
    patience = 0
    last_best_time = t0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        now = time.monotonic()
        
        # Compute gradient
        conv = fast_convolve(a)
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Softmax weights (log-sum-exp approximation)
        scaled = current_t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Subgradient
        sg = compute_sgrad(w, a)
        
        # Center the gradient (for the softmax parameterization)
        sg_centered = sg - np.sum(a * sg)
        
        # Adam-like update
        m = beta1 * m + (1.0 - beta1) * sg_centered
        v = beta2 * v + (1.0 - beta2) * (sg_centered ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        
        # Adaptive step size
        eta = eta0 / (np.sqrt(step + 1))
        log_factor = -eta * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Clip to avoid extreme updates
        log_factor = np.clip(log_factor, -5.0, 5.0)
        
        # Multiplicative update (mirror descent)
        a = a * np.exp(log_factor)
        a = np.maximum(a, 1e-15)
        a_sum = a.sum()
        if a_sum < 1e-10:
            a = np.ones(n) / n
        else:
            a /= a_sum
        
        # Evaluate
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        
        # Track best
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
            last_best_time = now
        else:
            patience += 1
        
        # Temperature schedule: increase over time
        progress = (now - t0) / time_limit
        current_t = t_init + (t_final - t_init) * (progress ** 2)
        
        # If stuck, try a perturbation
        if patience > 100:
            # Add noise
            noise = np.random.randn(n) * 0.01
            a = a + a * noise
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            patience = 0
            # Reset moments
            m *= 0.5
            v *= 0.5
            # Increase temperature to escape
            current_t = min(current_t * 2, 1000)
    
    return best_a, best_obj


def generate_initializations(n, rng):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive mirror descent with
log-sum-exp smoothing, Nesterov momentum, per-coordinate adaptive step sizes,
multi-start over diverse initializations and sequence lengths, plus a final
polishing phase with sharpened temperature and random perturbations.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_sgrad(w, a, threshold_frac=0.001):
    """
    Compute subgradient of max(conv) w.r.t. a given softmax weights w.
    Returns sg[j] = 2 * sum_k w[k] * a[k-j] (valid indices only).
    """
    n = len(a)
    sg = np.zeros(n)
    thresh = w.max() * threshold_frac
    active_idx = np.where(w > thresh)[0]
    for k_int in active_idx:
        k = int(k_int)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]
    return 2.0 * sg


def get_softmax_weights(conv, t):
    """Compute softmax weights over convolution values."""
    max_conv = conv.max()
    if max_conv < 1e-15:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w, max_conv
    scaled = t * conv / max_conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-12:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    return w, max_conv


def optimize_core(a0, time_limit, t_init=15.0, t_final=400.0,
                  eta0=0.6, beta1=0.9, beta2=0.95):
    """
    Adaptive mirror descent (multiplicative) with log-sum-exp smoothing,
    Nesterov momentum, and per-coordinate adaptive step sizes.
    Operates on the probability simplex (a >= 0, sum(a) = 1).
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / a_sum

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    # Momentum and adaptive variance
    m = np.zeros(n)
    v = np.ones(n)
    vel = np.zeros(n)  # Nesterov velocity

    t0 = time.monotonic()
    current_t = t_init
    step = 0
    patience = 0
    max_patience = 150

    while time.monotonic() - t0 < time_limit:
        step += 1
        now = time.monotonic()

        # Compute convolution and gradient
        conv = fast_convolve(a)
        w, max_conv = get_softmax_weights(conv, current_t)
        if max_conv < 1e-15:
            break

        sg = compute_sgrad(w, a)
        # Center for softmax parameterization
        sg_centered = sg - np.sum(a * sg)

        # Nesterov momentum
        vel = beta1 * vel - eta0 * sg_centered
        a_lookahead = a * np.exp(np.clip(vel, -5.0, 5.0))
        a_lookahead = np.maximum(a_lookahead, 1e-15)
        a_lookahead /= a_lookahead.sum()

        # Recompute gradient at lookahead point
        conv_la = fast_convolve(a_lookahead)
        w_la, _ = get_softmax_weights(conv_la, current_t)
        sg_la = compute_sgrad(w_la, a_lookahead)
        sg_la_centered = sg_la - np.sum(a_lookahead * sg_la)

        # Adam-like adaptive step
        m = beta1 * m + (1.0 - beta1) * sg_la_centered
        v = beta2 * v + (1.0 - beta2) * (sg_la_centered ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        eta = eta0 / (1.0 + 0.01 * step)
        log_factor = -eta * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_factor = np.clip(log_factor, -3.0, 3.0)

        # Multiplicative update
        a_new = a * np.exp(log_factor)
        a_new = np.maximum(a_new, 1e-15)
        a_sum = a_new.sum()
        if a_sum < 1e-10:
            a = np.ones(n) / n
        else:
            a = a_new / a_sum

        # Evaluate
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()

        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Temperature schedule: quadratic ramp
        progress = min(1.0, (now - t0) / time_limit)
        current_t = t_init + (t_final - t_init) * (progress ** 1.5)

        # Escape from local minima
        if patience >= max_patience:
            patience = 0
            # Perturb
            noise_scale = 0.05 * (1.0 + patience / max_patience)
            a = a + a * np.random.randn(n) * noise_scale
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            # Reset moments partially
            m *= 0.3
            v = np.maximum(v * 0.5, 1e-6)
            vel *= 0.3
            # Sharpen temperature to escape plateau
            current_t = min(current_t * 2.0, 2000.0)
            max_patience = int(max_patience * 0.8)
            if max_patience < 30:
                max_patience = 30

    return best_a, best_obj


def polish(a0, time_limit, t_values=None):
    """
    Final polishing phase: run optimization with very high temperatures
    and small steps to precisely equalize the top peaks.
    """
    if t_values is None:
        t_values = [200.0, 500.0, 1000.0, 2000.0]

    a = a0.copy()
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    time_per = time_limit / len(t_values)

    for t_val in t_values:
        if time.monotonic() - t0 > time_limit - 2:
            break
        t_alloc = min(time_per, time_limit - (time.monotonic() - t0) - 1)
        if t_alloc < 1:

                                       eta0=0.3, beta1=0.85, beta2=0.9)
        if obj_opt < best_obj - 1e-10:
            best_obj = obj_opt
            best_a = a_opt.copy()

    return best_a, best_obj


def generate_initializations(n, rng):

        e /= e.sum()

        se = np.exp(-np.abs(x - mid) / (width * n))
        se /= se.sum()
        inits[f'sym_exp_{int(width*100)}'] = se

    # Gaussians
    for sf in [0.1, 0.2, 0.35]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g

    # Bimodal (edges)
    bi = np.zeros(n)
    width = max(1, n // 4)
    bi[:width] = 1.0
    bi[-width:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Cosine bump
    cos_b = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_b /= cos_b.sum()
    inits['cosine'] = cos_b

    # Random (Dirichlet)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Multi-frequency sine (squared for non-negativity)

    sine /= sine.sum()

    alt = np.maximum(alt, 0.01)
    alt /= alt.sum()
    inits['alternating'] = alt

    return inits
