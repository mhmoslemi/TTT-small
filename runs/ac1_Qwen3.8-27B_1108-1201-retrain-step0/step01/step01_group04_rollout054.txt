    L = _fft_len(n)
    Fp = fft(p, L)
    Fq = fft(q, L)
    c = np.real(ifft(Fp * Fq))
    return c[:2 * n - 1]


def full_conv_pp(p):
    """Full linear self-convolution of p."""
    n = len(p)
    L = _fft_len(n)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))
    return c[:2 * n - 1]


def internal_score(p):
    """Compute 2*n*max(conv(p,p))/sum(p)^2."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if s < 1e-12:
        return np.inf
    c = full_conv_pp(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s < 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def solve_best_response_lp(p, n_constraints_max=3000, max_cg_iters=40):
    """
    Solve: min_q max_k (p*q)(k) s.t. q>=0, sum(q)=1
    using constraint generation. Returns (q, M_value).
    """
    n = len(p)
    num_total = 2 * n - 1

    # Initialize working set with top lags
    c_init = full_conv_pp(p)
    cmax_init = float(np.max(c_init))
    thresholds = [0.99, 0.95, 0.90, 0.80, 0.70]
    working_set = set()
    for th in thresholds:
        working_set.update(np.where(c_init > th * cmax_init)[0].tolist())
    working_set.add(0)
    working_set.add(n - 1)
    working_set.add(2 * n - 2)
    working_set.add((2 * n - 2) // 2)
    if len(working_set) > n_constraints_max:
        sorted_ws = sorted(working_set, key=lambda k: c_init[k], reverse=True)
        working_set = set(sorted_ws[:n_constraints_max])

    q = None
    M_val = np.inf

    for cg_iter in range(max_cg_iters):
        ws = sorted(working_set)
        n_c = len(ws)

        # Build sparse constraint matrix
        rows_data = []
        rows_idx = []
        row_ptr = [0]
        for k in ws:
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                val = p[k - j]
                if val > 1e-15:
                    rows_data.append(val)
                    rows_idx.append(j)
            rows_data.append(-1.0)
            rows_idx.append(n)
            row_ptr.append(len(rows_data))

        if len(rows_data) == 0:
            return None, np.inf

        A_ub = csr_matrix((rows_data, rows_idx, row_ptr),
                          shape=(n_c, n + 1), dtype=np.float64)
        b_ub = np.zeros(n_c, dtype=np.float64)
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        c_obj = np.zeros(n + 1, dtype=np.float64)
        c_obj[n] = 1.0
        bounds = [(0, None)] * (n + 1)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'disp': False, 'presolve': True})
        except Exception:
            return None, np.inf

        if res is None or res.status != 0:
            return None, np.inf

        q = np.asarray(res.x[:n], dtype=np.float64)
        M_val = float(res.x[n])
        q = np.maximum(q, 0.0)
        s_q = float(q.sum())
        if s_q < 1e-12:
            return None, np.inf
        q = q / s_q

        # Check for violated constraints
        c_full = full_conv_pq(p, q)
        violations = np.where(c_full > M_val + 1e-6 * max(1.0, M_val))[0]
        if len(violations) == 0:
            break

        # Add worst violators
        viol_amounts = c_full[violations] - M_val
        n_add = min(20, len(violations))
        worst = violations[np.argsort(viol_amounts)[::-1][:n_add]]
        working_set.update(worst.tolist())

        if len(working_set) > n_constraints_max * 2:
            c_ws = c_full[list(working_set)]
            order = np.argsort(c_ws)[::-1]
            working_set = set([list(working_set)[i] for i in order[:n_constraints_max]])

    return q, M_val


def iterated_best_response(p0, max_iters, budget_s, label=""):
    """
    Iterate best response: p_{t+1} = best_response(p_t).
    Track the best score (of the q iterates).
    """
    p = safe_normalize(np.asarray(p0, dtype=np.float64))
    n = len(p)
    best_p = p.copy()
    best_score = internal_score(p)
    current_score = best_score

    end_time = time.monotonic() + max(0.0, budget_s)
    prev_M = np.inf
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > end_time - 5.0:
            break

        q, M_val = solve_best_response_lp(p,
                                          n_constraints_max=max(2000, n),
                                          max_cg_iters=30)
        if q is None:
            break

        # Try different damping factors
        best_candidate = None
        best_candidate_score = np.inf
        for alpha in (1.0, 0.7, 0.5, 0.3):
            cand = project_simplex((1.0 - alpha) * p + alpha * q)
            sc = internal_score(cand)
            if np.isfinite(sc) and sc < best_candidate_score:
                best_candidate_score = sc
                best_candidate = cand

        if best_candidate is None:
            break

        # Track best
        if best_candidate_score < best_score:
            best_score = best_candidate_score
            best_p = best_candidate.copy()
            stall_count = 0
        else:
            stall_count += 1

        current_score = best_candidate_score
        p = best_candidate

        if it % 20 == 0:
            print("  %s iter=%d score=%.8f M=%.8f stall=%d" %
                  (label, it, best_score, M_val, stall_count), flush=True)

        if stall_count > 80 and prev_M - M_val < 1e-10 * max(1.0, prev_M):
            # Converged
            break
        prev_M = M_val

        # If score plateaus, try a perturbation
        if stall_count > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            p = project_simplex(p + rng.normal(0, 0.01 / n, n))

    return best_p, best_score


def make_geometric_growing(n, r):
    """p_j proportional to r^j with r > 1 (growing)."""
    x = np.arange(n, dtype=np.float64)
    p = np.power(r, x)
    return safe_normalize(p)


def make_geometric_decaying(n, r):
    """p_j proportional to r^j with r < 1 (decaying)."""
    x = np.arange(n, dtype=np.float64)
    p = np.exp(-r * x)
    return safe_normalize(p)


def make_two_exponential(n, a1, a2):
    """Mixture of two exponentials."""
    x = np.arange(n, dtype=np.float64)
    p = np.exp(-a1 * x) + np.exp(-a2 * (n - 1 - x))
    return safe_normalize(p)


def make_power_asym(n, alpha, beta):
    """p_j proportional to (j+1)^alpha * (n-j)^beta."""
    j = np.arange(1, n + 1, dtype=np.float64)
    p = np.power(j, alpha) * np.power(n + 1 - j, beta)
    return safe_normalize(np.maximum(p, 0))


def generate_non_symmetric_inits(n, rng):
    """Generate diverse non-symmetric initial shapes."""
    inits = []
    rs = [1.01, 1.03, 1.05, 1.08, 1.12, 1.2, 1.4, 1.8, 2.5]
    for r in rs:
        inits.append(make_geometric_growing(n, r))
    for a in [0.01, 0.03, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        inits.append(make_geometric_decaying(n, a))
    for a1 in [0.05, 0.1, 0.3, 0.8]:
        for a2 in [0.05, 0.1, 0.3, 0.8]:
            if a1 != a2:
                inits.append(make_two_exponential(n, a1, a2))
    for alpha in [0.5, 1.0, 2.0, 4.0]:
        for beta in [-0.5, 0.0, 0.5, -1.0]:
            inits.append(make_power_asym(n, alpha, beta))
    # Random smooth
    for _ in range(8):
        x = np.linspace(0, 1, n)
        f1 = int(rng.integers(1, max(2, n // 50)))
        f2 = int(rng.integers(1, max(2, n // 25)))
        p1 = float(rng.uniform(0, 2 * np.pi))
        p2 = float(rng.uniform(0, 2 * np.pi))
        amp1 = float(rng.uniform(0.3, 1.5))
        amp2 = float(rng.uniform(0.1, 0.6))
        bias = float(rng.uniform(-1.0, 1.0))
        p = 1.0 + amp1 * np.sin(2 * np.pi * f1 * x + p1) + \
            amp2 * np.sin(2 * np.pi * f2 * x + p2) + bias
        p = np.maximum(p, 0.01)
        inits.append(safe_normalize(p))
    return inits


def gradient_step(p, T_factor, lr, rng=None):
    """
    One gradient step on the smoothed max surrogate.
    Returns (new_p, surrogate_value).
    """
    n = len(p)
    L = _fft_len_grad(n)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]
    cmax = float(np.max(c))
    if cmax < 1e-30:
        return p, 0.0

    T = max(T_factor * cmax, 1e-15)
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = float(e.sum())
    if s < 1e-30 or not np.isfinite(s):
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        s = 1.0
    w = e / s
    surr = cmax + T * np.log(s)

    # Gradient: grad_j = 2 * sum_k w_k * p_{k-j}
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    # Extract: h[j] = sum_k p[k] * w[k-j] for valid... need to be careful
    # Actually h = ifft(fft(p) * fft(w flipped))
    # This gives circular convolution of p and reversed w
    # Let me compute directly for correctness
    # grad_j = 2 * sum_{k: 0<=k-j<=n-1} w[k] * p[k-j]
    # = 2 * sum_{j'} w[j+j'] * p[j'] where j' = k-j, 0<=j'<=n-1, 0<=j+j'<=2n-2

    # Use FFT-based approach:
    # Define w_ext of length 2n-1 (the convolution lags)
    # grad = 2 * (p * w_reversed) where * is the appropriate cross-correlation
    # Actually: grad_j = 2 * sum_{k=0}^{2n-2} w[k] * p[k-j] (with p out of range = 0)
    # This is (p * w_flipped)_j where w_flipped[k] = w[k] and we do:
    # (p * w_flipped)(j) = sum_k p[k] * w_flipped[j-k] ... no this isn't right either.

    # Let me just do: grad_j = 2 * sum_{k=max(j,0)}^{min(2n-2,j+n-1)} w[k] * p[k-j]
    # = 2 * sum_{j'=0}^{n-1} w[j+j'] * p[j'] (with w out of [0,2n-2] = 0)
    # This is the correlation of w and p, which is ifft(fft(p) * conj(fft(w)))
    # But we need to be careful about the shift.

    # Simpler: just compute it via the h vector
    # h[m] = sum_j p[j] * w_rev[m-j] where w_rev = w[::-1]
    # w_rev[m-j] = w[(2n-2)-(m-j)] = w[2n-2-m+j]
    # So h[m] = sum_j p[j] * w[2n-2-m+j]
    # Let l = 2n-2-m, then h[2n-2-l] = sum_j p[j] * w[l+j]
    # We want grad_j/2 = sum_{j'} w[j+j'] * p[j']
    # Set l = j: grad_j/2 = sum_{j'} p[j'] * w[j+j'] = h[2n-2-j]
    # So grad_j = 2 * h[2n-2-j]

    if n == 1:
        grad = 2.0 * np.array([h[0]])
    else:
        grad = 2.0 * h[2 * n - 2 :: -1][:n]

    grad = np.maximum(grad, 0.0)  # gradient of max is nonnegative for nonnegative p
    grad = np.where(np.isfinite(grad), grad, 0.0)

    new_p = project_simplex(p - lr * grad)
    return new_p, surr


def gradient_polish(p0, budget_s, label=""):
    """
    Polish using gradient descent on the smoothed max surrogate.
    """
    p = safe_normalize(np.asarray(p0, dtype=np.float64))
    n = len(p)
    best_p = p.copy()
    best_score = internal_score(p)
    current_score = best_score

    end_time = time.monotonic() + max(0.0, budget_s)
    T_factor = 0.5
    lr = 0.5 / n
    rng = np.random.default_rng(12345)
    stall = 0
    it = 0

    while time.monotonic() < end_time - 2.0:
        it += 1
        # Cool temperature
        T_factor = max(0.005, 0.5 * (0.995 ** it))
        # Adjust learning rate
        lr_scale = max(0.01, 1.0 * (0.99 ** (it // 100)))
        base_lr = lr_scale / max(1.0, np.max(np.abs(2.0 * p)))

        accepted = False
        for attempt in range(3):
            try_lr = base_lr * (0.5 ** attempt)
            if try_lr < 1e-10:
                continue
            new_p, surr = gradient_step(p, T_factor, try_lr)
            sc = internal_score(new_p)
            if np.isfinite(sc):
                if sc < best_score:
                    best_score = sc
                    best_p = new_p.copy()
                    stall = 0
                elif sc < current_score:
                    stall = min(stall, 5)
                else:
                    stall += 1
                current_score = sc
                p = new_p
                accepted = True
                break

        if not accepted:
            stall += 5
            # Try random perturbation
            if stall > 30:
                sig = max(0.001, 0.05 / n) * (stall // 30)
                p = project_simplex(p + rng.normal(0, sig, n))
                current_score = internal_score(p)
                stall = max(0, stall - 10)

        if it % 500 == 0:
            print("  %s grad iter=%d best=%.8f T=%.4f lr=%.6f" %
                  (label, it, best_score, T_factor, base_lr), flush=True)

        if best_score < 1.499 and stall > 200:
            break

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for nonnegative step function minimizing evaluate_sequence score.
    Uses iterated best-response (LP with constraint generation) + gradient polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(10.0, float(budget_s) - 15.0)

    best_p_global = None
    best_score_global = np.inf

    def try_record(p, sc):
        nonlocal best_p_global, best_score_global
        if p is None or len(p) == 0:
            return
        p = safe_normalize(p)
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score_global:
            best_score_global = float(sc)
            best_p_global = p.copy()
            print("  NEW BEST: score=%.8f n=%d" % (sc, len(p)), flush=True)

    # Try to use parent sequence
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 64:
            parent = safe_normalize(parent_arr)
            print("Using parent sequence, n=%d, score=%.8f" %
                  (parent.size, internal_score(parent)), flush=True)
    except Exception:
        parent = None

    # Sizes to try
    sizes = [1024, 2048, 4096]

    # Filter sizes based on available time
    remaining = deadline - time.monotonic()
    if remaining < 600:
        sizes = [1024, 2048]
    if remaining < 300:
        sizes = [1024]

    for n in sizes:
        if time.monotonic() > deadline - 60.0:
            break

        print("\n=== Size n=%d ===" % n, flush=True)

        # Generate initial shapes
        inits = generate_non_symmetric_inits(n, rng)

        # Also add parent if appropriate size
        if parent is not None:
            if parent.size == n:
                inits.insert(0, parent.copy())
            else:
                # Resample parent to size n
                if parent.size > n:
                    # Downsample by averaging
                    n_chunks = parent.size // n
                    p_resampled = parent[:n * n_chunks].reshape(n, n_chunks).mean(axis=1)
                else:
                    # Upsample by interpolation
                    x_old = np.linspace(0, 1, parent.size)
                    x_new = np.linspace(0, 1, n)
                    p_resampled = np.interp(x_new, x_old, parent)
                inits.insert(0, safe_normalize(p_resampled))

        # Score all inits
        scored = []
        for init in inits:
            init = safe_normalize(init)
            sc = internal_score(init)
            if np.isfinite(sc):
                scored.append((sc, init))
        scored.sort(key=lambda x: x[0])

        print("  Best init: score=%.8f (out of %d inits)" %
              (scored[0][0], len(scored)), flush=True)

        # Run best-response iteration on top inits
        top_k = min(4, len(scored))
        for i in range(top_k):
            if time.monotonic() > deadline - 60.0:
                break
            sc_init, p_init = scored[i]
            try_record(p_init, sc_init)

            # Time budget for best response
            remaining_br = min(400.0, (deadline - time.monotonic()) * 0.4)
            if remaining_br < 30:
                break

            print("  Running best response on init %d (score=%.8f)..." %
                  (i, sc_init), flush=True)
            p_br, sc_br = iterated_best_response(
                p_init, max_iters=200, budget_s=remaining_br,
                label="n=%d_i%d" % (n, i))
            try_record(p_br, sc_br)

            # Gradient polish on the best from best-response
            if time.monotonic() < deadline - 30.0:
                remaining_g = min(120.0, (deadline - time.monotonic()) * 0.3)
                if remaining_g > 10:
                    p_g, sc_g = gradient_polish(
                        p_br, budget_s=remaining_g,
                        label="n=%d_i%d_g" % (n, i))
                    try_record(p_g, sc_g)

    # Final polish on the global best
    if best_p_global is not None and time.monotonic() < deadline - 10.0:
        remaining_final = deadline - 10.0 - time.monotonic()
        if remaining_final > 20:
            print("\n=== Final polish ===", flush=True)
            p_final, sc_final = gradient_polish(
                best_p_global, budget_s=remaining_final, label="FINAL")
            try_record(p_final, sc_final)

    if best_p_global is None:
        best_p_global = make_geometric_growing(1024, 1.05)
        best_score_global = internal_score(best_p_global)

    # Final validation
    best_p_global = np.asarray(best_p_global, dtype=np.float64)
    best_p_global = np.where(np.isfinite(best_p_global), best_p_global, 0.0)
    best_p_global = np.maximum(best_p_global, 0.0)
    s = float(best_p_global.sum())
    if s < 1e-12:
        best_p_global = np.full(max(1, len(best_p_global)), 1.0)
        s = float(best_p_global.sum())
    best_p_global = best_p_global / s
    best_p_global = np.minimum(best_p_global, 1000.0)

    elapsed = time.monotonic() - start_time
    print("\nDone in %.1fs. Best score=%.8f, n=%d" %
          (elapsed, best_score_global, len(best_p_global)), flush=True)

    # Verify with evaluate_sequence
    try:
        val = evaluate_sequence([float(x) for x in best_p_global])
        print("evaluate_sequence verification: %.8f" % val, flush=True)
        if not np.isfinite(val) or val < best_score_global:
            # If our internal score was wrong, still return the candidate
            pass
    except Exception as e:
        print("Verification error: %s" % str(e), flush=True)

    return [float(x) for x in best_p_global]
