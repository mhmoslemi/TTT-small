"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses smoothed gradient descent (log-sum-exp) with Nesterov momentum,
structured "flat-half" initialization, multi-scale search, and coordinate refinement.
"""

import numpy as np
import time


def compute_flat_half(n):
    """Compute sequence whose first n auto-convolution values are all equal to 1."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        s = 0.0
        for j in range(1, k):
            s += a[j] * a[k - j]
        a[k] = (1.0 - s) / 2.0
        if a[k] < 0.0:
            a[k] = 0.0
    return a


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def norm1(a):
    """Normalize sequence to sum to 1."""
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a / S


def smooth_grad(a, tau):
    """Compute gradient of smooth max of conv(a,a) using log-sum-exp with temperature tau."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp((c - c_max) / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[int(c.argmax())] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(w[i:i + n], a)
    return grad


def hard_grad(a):
    """Compute subgradient at the argmax of conv(a,a)."""
    n = len(a)
    c = np.convolve(a, a)
    m_star = int(c.argmax())
    grad = np.zeros(n)
    for i in range(n):
        k_minus_i = m_star - i
        if 0 <= k_minus_i < n:
            grad[i] = 2.0 * a[k_minus_i]
    return grad


def optimize_smooth(a0, deadline):
    """
    Smoothed gradient descent with Nesterov momentum and cosine-annealed temperature.
    Returns (best_sequence, best_score).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = norm1(a)

    best_score = compute_obj(a)
    best_a = a.copy()

    v = np.zeros(n)
    mu = 0.9

    total_iters = 6000
    tau_start = 5.0
    tau_end = 0.05
    lr_start = 0.4
    lr_end = 0.003

    for it in range(total_iters):
        if time.monotonic() > deadline - 25.0:
            break

        frac = it / total_iters
        tau = tau_end + (tau_start - tau_end) * 0.5 * (1.0 + np.cos(np.pi * frac))
        lr = lr_end + (lr_start - lr_end) * 0.5 * (1.0 + np.cos(np.pi * frac))

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        a_look = norm1(a_look)

        if tau < 0.1:

        else:

        g = g - g.mean()

        a = a + v
