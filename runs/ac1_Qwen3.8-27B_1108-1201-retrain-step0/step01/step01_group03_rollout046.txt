def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize to have sum = target (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def softmax_gradient_step(a, lr, tau):
    """One step of soft-max gradient descent on peak autocorrelation."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    
    # grad[j] = 2 * sum_k a[k] * w[j+k] for valid k
    grad = np.zeros(n)
    for j in range(n):
        k_max = min(n, 2 * n - 1 - j)
        if k_max > 0:
            grad[j] = 2.0 * np.dot(a[:k_max], w[j:j + k_max])
    
    # Project out the mean (fixed-sum constraint)
    grad -= grad.mean()
    
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = normalize(a, S_t)
    return a


def hard_gradient_step(a, lr):
    """One step of hard-max (subgradient) descent."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    
    c = np.convolve(a, a)
    # Find all positions where c is close to max
    M = c.max()
    close_to_max = c >= M * 0.999
    
    grad = np.zeros(n)
    for k in np.where(close_to_max)[0]:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * a[k - j]
    grad /= len(np.where(close_to_max)[0])
    
    # Project out the mean
    grad -= grad.mean()
    
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = normalize(a, S_t)
    return a
