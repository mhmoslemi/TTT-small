def build_conv_matrix(p):
    n=len(p)
    rows=[]; cols=[]; data=[]
    for k in range(2*n-1):
        jlo = 0 if k < n-1 else k-(n-1)
        jhi = n-1 if k >= n-1 else k
        # j in [jlo, jhi]; A[k,j]=p[k-j]
        js = np.arange(jlo, jhi+1)
        data.extend(p[k - js].tolist())  # but p is array; k-js is array of indices
        rows.extend([k]*(jhi-jlo+1))
        cols.extend(js.tolist())
    A=csr_matrix((data,(rows,cols)), shape=(2*n-1,n))
    return A

def project_simplex(v):
    v=np.asarray(v, dtype=float).ravel()
    n=v.size
    if n==0: return v
    v=np.where(np.isfinite(v), v, 0.0)
    u=np.sort(v)[::-1]
    cssum=np.cumsum(u)
    idx=np.arange(1, n+1, dtype=float)
    cond=(u*idx) > (cssum-1.0)
    if not np.any(cond):
        return np.full(n, 1.0/n)
    k=int(np.count_nonzero(cond)-1)
    theta=(cssum[k]-1.0)/(float(k+1))
    out=np.maximum(v-theta, 0.0)
    s=float(np.sum(out))
    if (not np.isfinite(s)) or s<=1e-300:
        return np.full(n, 1.0/n)
    return out/s

def build_conv_matrix(p):
    p=np.asarray(p, dtype=float)
    n=p.size
    rows=[]; cols=[]; data=[]
    for k in range(2*n-1):
        jlo = 0 if k < (n-1) else (k-(n-1))
        jhi = (n-1) if k >= (n-1) else k

def make_random_smooth(L, rng):
    x=np.linspace(0,1,L)
    p=np.ones(L)
    a=rng.uniform(-1.0,1.0); b=rng.uniform(-1.0,1.0); c=rng.uniform(-1.0,1.0); d=rng.uniform(-1.0,1.0)
    p = 1.0 + 0.5*a + 0.35*b*np.sin(2*math.pi*(1.0+rng.uniform(0,4.0))*x + rng.uniform(0,2*math.pi))
    p = p + 0.25*c*np.sin(2*math.pi*(2.0+rng.uniform(0,8.0))*x + rng.uniform(0,2*math.pi))
    p = p + 0.18*d*np.sin(2*math.pi*(3.0+rng.uniform(0,15.0))*x + rng.uniform(0,2*math.pi))
    # endpoint taper
    el=rng.uniform(-2.5,2.5); er=rng.uniform(-2.5,2.5)
    p = p * np.power(1e-4+x, el) * np.power(1e-4+1.0-x, er)
    p=np.maximum(p, 0.02)
    return p

def shape_from_theta(theta, n, x, M, eps=1e-6):
    th=np.asarray(theta, dtype=float).ravel()
    dim=2*M+3
    if th.size<dim: th=np.concatenate([th, np.zeros(dim-th.size)])
    else: th=th[:dim].copy()
    if not np.all(np.isfinite(th)): th=np.nan_to_num(th)
    logp=np.full(n, float(th[0]))
    idx=1
    for m in range(1, M+1):
        a=float(th[idx]); b=float(th[idx+1]); idx+=2
        ang=2*math.pi*m*x
        logp = logp + a*np.cos(ang) + b*np.sin(ang)
    eL=float(th[idx]); eR=float(th[idx+1]); idx+=2
    logp = logp + eL*np.log(x+eps) + eR*np.log(1.0-x+eps)
    p=np.exp(logp - float(np.max(logp)))
    s=float(np.sum(p))
    if (not np.isfinite(s)) or s<=1e-12:
        return np.full(n, 1.0/n)
    return p/s

Score = 2*n*max(p*p) / (sum p)^2, scale-invariant -> optimize on the probability

Method:
  * expressive asymmetric low-dim init: log p = Fourier partial sum + independent

def full_conv(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    la = a.size; lb = b.size
    if la == 0 or lb == 0:
        return np.empty(0)
    L = _nfl(la + lb - 1)
    fa = _fft(a, L)
    fb = _fft(b, L)
    r = _ifft(fa * fb)
    return r.real[:la + lb - 1]


def conv_sq(p):
    return full_conv(p, p)


def normalize(p):
    p = np.asarray(p, dtype=float).ravel()
    if p.size == 0:
        return np.empty(0)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 1e-12:
        return np.full(p.size, 1.0 / p.size)
    return p / s


def score_of(p):
    p = np.asarray(p, dtype=float).ravel()
    n = p.size
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if (not np.isfinite(s)) or s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if (not np.isfinite(mx)) or mx < 0.0:
        mx = 0.0
    return float(2.0 * n * mx / (s * s))


def project_simplex(v):
    v = np.asarray(v, dtype=float).ravel()
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=float)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(np.sum(out))
    if (not np.isfinite(s)) or s <= 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def cross_w_p(w, p):
    n = int(p.size)
    r = full_conv(w, p[::-1])
    if r.size < 2 * n - 1:
        tmp = np.zeros(2 * n - 1)
        tmp[:r.size] = r
        r = tmp
    return r[n - 1:2 * n - 1]


def softmax_stable(c, T):
    c = np.asarray(c, dtype=float)
    if c.size == 0:
        return np.zeros(0)
    if (T is None) or (not np.isfinite(T)) or T <= 0:
        w = np.zeros(c.size)
        w[int(np.argmax(c))] = 1.0
        return w
    cx = float(np.max(c))
    e = np.exp((c - cx) / T)
    s = float(np.sum(e))
    if (not np.isfinite(s)) or s <= 1e-300:
        w = np.zeros(c.size)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


# ---------------- LP best response ----------------

def build_conv_matrix(p):
    p = np.asarray(p, dtype=float)
    n = p.size
    rows = []; cols = []; data = []
    nm = n - 1
    for k in range(2 * n - 1):
        jlo = 0 if k < nm else k - nm
        jhi = nm if k >= nm else k
        if jhi < jlo:
            continue
        js = np.arange(jlo, jhi + 1)
        vals = p[k - js]
        data.extend(vals.tolist())
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    if (n > LP_MAX) or (n < 8):
        return None
    c = conv_sq(p)
    M = float(np.max(c))
    if (not np.isfinite(M)) or M <= 1e-300:
        return None
    l0 = time.monotonic()
    try:
        A = build_conv_matrix(p)
        b = np.full(2 * n - 1, M, dtype=float)
        cobj = -np.ones(n)
        res = linprog(cobj, A_ub=A, b_ub=b, bounds=(0.0, None), method='highs',
                      options={'disp': False, 'time_limit': max(0.05, min(float(tcap), 3.0))})
    except Exception:
        return None
    if res is None:
        return None
    if getattr(res, 'status', 1) not in (0, 2, 3):
        return None
    g = np.asarray(getattr(res, 'x', None), dtype=float)
    if (g is None) or (g.size != n) or (not np.all(np.isfinite(g))):
        return None
    g = np.maximum(g, 0.0)
    s = float(np.sum(g))
    if s <= 1e-12:
        return None
    g = g / s
    cur = score_of(p)
    best = None
    bestsc = cur
    for t in (1.0, 0.75, 0.5, 0.35, 0.2, 0.1, 0.05, 0.02):
        q = project_simplex((1.0 - t) * p + t * g)
        sc = score_of(q)
        if np.isfinite(sc) and sc < bestsc:
            bestsc = sc
            best = q
    return best if (best is not None and bestsc < cur - 1e-12) else None


# ---------------- initial shapes ----------------

def make_two_gauss(x, mu1, sig1, w1, mu2, sig2, w2):
    p = w1 * np.exp(-((x - mu1) ** 2) / (2.0 * max(1e-4, float(sig1) ** 2)))
    p = p + w2 * np.exp(-((x - mu2) ** 2) / (2.0 * max(1e-4, float(sig2) ** 2)))
    return p


def make_random_smooth(L, rng):
    x = np.linspace(0.0, 1.0, L)
    a = rng.uniform(-1.0, 1.0)
    b = rng.uniform(-1.0, 1.0)
    c = rng.uniform(-1.0, 1.0)
    d = rng.uniform(-1.0, 1.0)
    p = 1.0 + 0.5 * a
    p = p + 0.35 * b * np.sin(2.0 * math.pi * (1.0 + rng.uniform(0, 4.0)) * x + rng.uniform(0, 2 * math.pi))
    p = p + 0.25 * c * np.sin(2.0 * math.pi * (2.0 + rng.uniform(0, 8.0)) * x + rng.uniform(0, 2 * math.pi))
    p = p + 0.18 * d * np.sin(2.0 * math.pi * (3.0 + rng.uniform(0, 15.0)) * x + rng.uniform(0, 2 * math.pi))
    el = rng.uniform(-2.5, 2.5)
    er = rng.uniform(-2.5, 2.5)
    p = p * np.power(1e-4 + x, el) * np.power(1e-4 + 1.0 - x, er)
    p = np.maximum(p, 0.02)
    return p


def flat_convol_seq(nn):
    b = [1.0] * nn
    if nn < 2:
        return np.array(b)
    for k in range(1, nn):
        s = 0.0
        for i in range(1, k):
            s += b[i] * b[k - i]
        b[k] = (1.0 - s) / 2.0
    return np.array(b)


def make_windows(L, rng):
    x = np.linspace(0.0, 1.0, L)
    wins = []
    wins.append(np.full(L, 1.0))
    for s in (0.5, 1.0, 1.7, 2.6, 4.0, 6.0, 9.0, 13.0, 18.0, 26.0, 40.0):
        wins.append(np.exp(s * x))
        wins.append(np.exp(-s * x))
    for aa in (0.4, 0.8, 1.5, 2.5, 4.0, 7.0, 12.0, 20.0):
        wins.append(np.power(1e-4 + x, aa))
        wins.append(np.power(1e-4 + 1.0 - x, aa))
    pairs = [(0.2, 2.0), (0.3, 3.5), (0.5, 5.0), (0.5, 8.0), (1.0, 3.0), (2.0, 0.3),
             (3.5, 0.3), (5.0, 0.5), (8.0, 0.5), (0.8, 1.8), (1.8, 0.8),
             (0.4, 6.0), (6.0, 0.4), (2.0, 5.0), (5.0, 2.0)]
    for (aa, bb) in pairs:
        wins.append(np.power(1e-4 + x, aa) * np.power(1e-4 + 1.0 - x, bb))
    gset = [(0.30, 0.10, 1.0, 0.70, 0.07, 1.0), (0.20, 0.07, 1.0, 0.78, 0.09, 1.0),
            (0.50, 0.16, 2.0, 0.12, 0.05, 1.0), (0.50, 0.16, 2.0, 0.88, 0.05, 1.0),
            (0.30, 0.06, 2.0, 0.65, 0.08, 1.0), (0.75, 0.06, 1.0, 0.45, 0.16, 1.0),
            (0.45, 0.20, 1.0, 0.55, 0.05, 2.0)]
    for (mu1, sig1, w1, mu2, sig2, w2) in gset:
        wins.append(make_two_gauss(x, mu1, sig1, w1, mu2, sig2, w2))
    for _ in range(12):
        wins.append(make_random_smooth(L, rng))
    try:
        fb = flat_convol_seq(L)
        fb = np.maximum(fb, 0.0)
        if float(np.sum(fb)) > 1e-9:
            wins.append(fb)
    except Exception:
        pass
    out = [normalize(w) for w in wins]
    return [w for w in out if w.size == L]


# ---------------- low-dim shape parametrisation ----------------

_G_N = 0
_G_X = None
_G_M = 0
_G_EPS = 1e-6


def shape_from_theta(theta, n, x, M, eps=1e-6):
    th = np.asarray(theta, dtype=float).ravel()
    dim = 2 * M + 3
    if th.size < dim:
        th = np.concatenate([th, np.zeros(dim - th.size)])
    else:
        th = th[:dim].copy()
    if not np.all(np.isfinite(th)):
        th = np.nan_to_num(th)
    logp = np.full(n, float(th[0]))
    idx = 1
    for m in range(1, M + 1):
        a = float(th[idx]); b = float(th[idx + 1]); idx += 2
        ang = 2.0 * math.pi * m * x
        logp = logp + a * np.cos(ang) + b * np.sin(ang)
    eL = float(th[idx]); eR = float(th[idx + 1])
    logp = logp + eL * np.log(x + eps) + eR * np.log(1.0 - x + eps)
    p = np.exp(logp - float(np.max(logp)))
    s = float(np.sum(p))
    if (not np.isfinite(s)) or s <= 1e-12:
        return np.full(n, 1.0 / n)
    return p / s


def _shape_objective(theta):
    if _G_X is None:
        return 1e18
    try:
        p = shape_from_theta(theta, int(_G_N), _G_X, int(_G_M), _G_EPS)
        sc = score_of(p)
        return sc if np.isfinite(sc) else 1e18
    except Exception:
        return 1e18


def shape_global(n, budget, rng, M):
    global _G_N, _G_X, _G_M, _G_EPS
    _G_N = int(n)
    _G_X = np.linspace(0.0, 1.0, n)
    _G_M = int(M)
    _G_EPS = 1e-6
    dim = 2 * M + 3
    end = time.monotonic() + max(5.0, float(budget))
    best_p = None
    best_sc = np.inf
    best_th = None
    starts = [np.zeros(dim)]
    tt = np.zeros(dim); tt[dim - 2] = 1.2; tt[dim - 1] = 1.2; starts.append(tt)
    tt2 = np.zeros(dim); tt2[0] = 1.0; tt2[dim - 2] = 0.0; tt2[dim - 1] = 2.0; starts.append(tt2)
    for _ in range(6):
        starts.append(rng.uniform(-2.2, 2.2, dim))
    for st in starts:
        if time.monotonic() > end - 30:
            break
        p = shape_from_theta(st, n, _G_X, M, _G_EPS)
        sc = score_of(p)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_p = p.copy()
            best_th = st.copy()
    if _HAVE_MINIMIZE and (time.monotonic() < end - 40):
        bounds = [(-4.0, 4.0)] * dim
        try:
            res = differential_evolution(_shape_objective, bounds, maxiter=450, popsize=12,
                                         tol=0.2, seed=int(rng.integers(0, 2147483646)),
                                         polish=False, init='latinhypercube')
        except Exception:
            res = None
        if res is not None and np.isfinite(getattr(res, 'fun', np.inf)):
            p = shape_from_theta(res.x, n, _G_X, M, _G_EPS)
            sc = score_of(p)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_p = p.copy()
                best_th = np.asarray(res.x, dtype=float).copy()
        nmseeds = []
        if best_th is not None:
            nmseeds.append(best_th.copy())
        nmseeds.append(starts[0].copy())
        for _ in range(3):
            nmseeds.append(rng.uniform(-2.2, 2.2, dim))
        for st in nmseeds:
            if time.monotonic() > end - 4:
                break
            try:
                r2 = minimize(_shape_objective, st, method='Nelder-Mead',
                              options={'maxiter': 250, 'xatol': 1e-5, 'fatol': 1e-7})
            except Exception:
                continue
            if r2 is not None and np.isfinite(getattr(r2, 'fun', np.inf)):
                p = shape_from_theta(r2.x, n, _G_X, M, _G_EPS)
                sc = score_of(p)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_p = p.copy()
                    best_th = np.asarray(r2.x, dtype=float).copy()
    else:
        cur = np.zeros(dim) if best_th is None else best_th.copy()
        curp = shape_from_theta(cur, n, _G_X, M, _G_EPS)
        cursc = score_of(curp)
        if np.isfinite(cursc) and cursc < best_sc:
            best_sc = cursc
            best_p = curp.copy()
            best_th = cur.copy()
        for _ in range(20000):
            if time.monotonic() > end - 2:
                break
            j = int(rng.integers(0, dim))
            cand = cur.copy()
            cand[j] += float(rng.normal(0.0, 0.4))
            p = shape_from_theta(cand, n, _G_X, M, _G_EPS)
            sc = score_of(p)
            if np.isfinite(sc) and sc < cursc:
                cur = cand
                curp = p
                cursc = sc
                if sc < best_sc:
                    best_sc = sc
                    best_p = p.copy()
                    best_th = cur.copy()
    return best_p, (float(best_sc) if np.isfinite(best_sc) else np.inf), best_th


# ---------------- local polish ----------------

def _as_valid(p):
    p = np.asarray(p, dtype=float).ravel()
    if p.size == 0 or not np.all(np.isfinite(p)):
        return None
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 1e-9:
        return None
    return p / s


def scored_sort(item):
    return item[0]


def polish(p0, budget, rng, use_lp, alpha0, tag):
    p = normalize(p0)
    n = int(p.size)
    if n == 0:
        return p0, np.inf
    start = time.monotonic()
    end = start + max(1.0, float(budget))
    c = conv_sq(p)
    best_p = p.copy()
    best_sc = score_of(p)
    cur_sc = best_sc
    vel = np.zeros(n)
    alpha = float(alpha0)
    it = 0
    stall = 0
    lp_budget = (0.09 * float(budget)) if use_lp else 0.0
    lp_used = 0.0
    while time.monotonic() < end - 0.2:
        it += 1
        prev_best = best_sc
        c = conv_sq(p)
        cx = float(np.max(c)) if c.size else 0.0
        T = alpha * cx
        if (not np.isfinite(T)) or T <= 0:
            T = 1e-9
        w = softmax_stable(c, T)
        g = 2.0 * cross_w_p(w, p)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        beta = 0.0 if (alpha > 0.25 and it < 40) else 0.4
        newvel = beta * vel + (1.0 - beta) * g
        base = 0.8 / n
        gnm = float(np.max(np.abs(g))) if g.size else 0.0
        if (not np.isfinite(gnm)) or gnm <= 1e-300:
            gnm = 1.0
        bstep = base / gnm
        besttrial = None
        bestq = None
        for lf in (0.4, 0.15, 0.06, 0.024, 0.01):
            q = project_simplex(p - (bstep * lf) * newvel)
            sc = score_of(q)
            if (besttrial is None) or (np.isfinite(sc) and sc < besttrial):
                besttrial = sc
                bestq = q
        accepted = False
        if (besttrial is not None) and np.isfinite(besttrial) and (besttrial < cur_sc - 1e-12 or besttrial < best_sc):
            p = bestq
            vel = newvel
            cur_sc = besttrial
            accepted = True
            alpha = max(0.004, alpha * 0.997)
        else:
            alpha = max(0.003, alpha * 0.5)
        # exact peak cycling over the near-max cluster
        c = conv_sq(p)
        cx = float(np.max(c)) if c.size else 0.0
        order = np.argsort(c)[::-1]
        chosen = []
        for kk in order[:16]:
            kii = int(kk)
            if c[kii] >= 0.5 * cx:
                chosen.append(kii)
                if len(chosen) >= 6:
                    break
        moved = False
        for kk in chosen:
            v = np.zeros(n)
            ii = kk - np.arange(n)
            valid = (ii >= 0) & (ii < n)
            v[valid] = 2.0 * p[ii[valid]]
            vmax = float(np.max(v)) if v.size else 0.0
            if (not np.isfinite(vmax)) or vmax <= 1e-300:
                continue
            v = v / vmax
            basek = 1.0 / n
            bfr = None
            bqp = None
            for lf in (0.3, 0.7, 1.5, 3.0, 6.0, 12.0, 24.0, 36.0):
                q = project_simplex(p - basek * lf * v)
                sc = score_of(q)
                if (bfr is None) or (np.isfinite(sc) and sc < bfr):
                    bfr = sc
                    bqp = q
            if (bfr is not None) and np.isfinite(bfr) and bfr < cur_sc - 1e-12:
                p = bqp
                cur_sc = bfr
                moved = True
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_p = p.copy()
                c = conv_sq(p)
                cx = float(np.max(c)) if c.size else 0.0
        # periodic LP best response
        if use_lp and n <= LP_MAX and (lp_budget - lp_used > 0.4) and (it % 100 == 0):
            rem = end - time.monotonic()
            tcap = min(1.0, max(0.1, lp_budget - lp_used, rem * 0.15))
            l0 = time.monotonic()
            q = lp_best_response(p, tcap)
            lp_used += (time.monotonic() - l0)
            if q is not None:
                sc = score_of(q)
                if np.isfinite(sc) and sc < cur_sc - 1e-12:
                    p = q
                    cur_sc = sc
                    if sc < best_sc:
                        best_sc = sc
                        best_p = p.copy()
        if best_sc < prev_best - 1e-14:
            stall = 0
        else:
            stall += 1
        if (not accepted) and (not moved):
            if stall > 1000 and alpha < 0.03:
                alpha = 0.22
                p = project_simplex(p + rng.normal(0.0, 0.02 / n, n))
                vel = np.zeros(n)
                cur_sc = score_of(p)
            elif stall > 2000:
                p = project_simplex(0.6 * best_p + 0.45 * p + rng.normal(0.0, 0.01 / n, n))
                vel = np.zeros(n)
                cur_sc = score_of(p)
                alpha = 0.1
        if it % 500 == 0:
            print("    %-9s it=%-5d cur=%.9f best=%.9f alpha=%.4f rem=%.0fs"
                  % (tag, it, cur_sc, best_sc, alpha, max(0.0, end - time.monotonic())), flush=True)
        if best_sc < 1.5 and stall > 7000:
            break
    best_p = normalize(best_p)
    s_col = score_of(best_p)
    return best_p, (float(best_sc) if np.isfinite(best_sc) else s_col)

def propose_candidate(seed=42, budget_s=2350, **kwargs):
    try:
        rng = np.random.default_rng(seed)
        t0 = time.monotonic()
        total = float(budget_s)
        if (not np.isfinite(total)) or (total <= 0):
            total = 2350.0
        total = min(total, 2380.0)
        deadline = t0 + max(120.0, total - 40.0)

        parent = None
        try:
            pa = np.asarray(height_sequence_1, dtype=float)
            pa = np.where(np.isfinite(pa), pa, 0.0)
            pa = np.maximum(pa, 0.0)
            if (pa.ndim == 1) and (pa.size >= 32) and (float(pa.sum()) > 0):
                parent = pa
        except Exception:
            parent = None

        gbest_p = None
        gbest_sc = np.inf

        # candidate lengths
        lst = [700]
        for L in (600, 800, 900, 1100, 1400, 1800, 2400, 3200, 4000):
            if (L not in lst) and (250 <= L <= 4200):
                lst.append(L)
        if parent is not None:
            Lp = int(parent.size)
            if (250 <= Lp <= 4200) and (Lp not in lst):
                lst.append(Lp)
        lst = [L for L in lst if L <= 3800]
        main_n = 700

        # window probe
        probe = {}
        for L in lst:
            wins = make_windows(L, rng)
            bw = None
            bws = np.inf
            for wp in wins:
                v = _as_valid(wp)
                if v is None:
                    continue
                sc = score_of(v)
                if np.isfinite(sc) and sc < bws:
                    bws = sc
                    bw = v.copy()
            probe[L] = (bws, bw)
            print("[probe] n=%d  win_best=%.6f  (t=%.0fs)" % (L, bws, time.monotonic() - t0), flush=True)

        mainws = probe.get(main_n, (np.inf, None))[0]
        alt_n = None
        altw = np.inf
        for L in lst:
            if L == main_n:
                continue
            wv = probe.get(L, (np.inf, None))[0]
            if wv < altw:
                altw = wv
                alt_n = L

        # shape search
        orderL = [main_n]
        use_alt = (alt_n is not None) and (altw < mainws)
        if use_alt:
            orderL.append(alt_n)
        Mdeg = 7
        shape_res = {}
        for L in orderL:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            if L == main_n:
                sb = min(230.0, rem * 0.30)
            else:
                sb = min(90.0, rem * 0.16)
            sp, ss, st = shape_global(L, max(20.0, sb), rng, Mdeg)
            if sp is not None:
                shape_res[L] = (sp, ss)
                v = _as_valid(sp)
                if v is not None:
                    sc = score_of(v)
                    if np.isfinite(sc) and sc < gbest_sc:
                        gbest_sc = sc
                        gbest_p = v.copy()
            print("[shape] n=%d  shape_best=%.6f" % (L, ss), flush=True)

        # polish top inits per length
        rem_after = max(0.0, deadline - time.monotonic())
        final_reserve = max(150.0, 0.20 * rem_after)
        polish_total = max(0.0, rem_after - final_reserve - 20.0)
        shares = {}
        if use_alt:
            shares[main_n] = 0.72 * polish_total
            shares[alt_n] = 0.28 * polish_total
        else:
            shares[main_n] = polish_total
        for L in orderL:
            BL = shares.get(L, 0.0)
            if (deadline - time.monotonic()) < 50:
                break
            use_lp = (L <= LP_MAX)
            inits = []
            if L in shape_res:
                sp, ss = shape_res[L]
                inits.append(sp)
            inits.extend(make_windows(L, rng))
            if (parent is not None) and (int(parent.size) == L):
                inits.append(parent)
            scored = []
            for ip in inits:
                v = _as_valid(ip)
                if v is None:
                    continue
                sc = score_of(v)
                if np.isfinite(sc):
                    scored.append((sc, v.copy()))
            scored.sort(key=scored_sort)
            tops = scored[: (3 if L == main_n else 2)]
            k = max(1, len(tops))
            per = BL / float(k)
            for i in range(k):
                if (deadline - time.monotonic()) < 25:
                    break
                per_i = max(15.0, min(per, (deadline - time.monotonic()) - 2.0))
                if per_i < 10:
                    per_i = 15.0
                ip, is0 = tops[i]
                pp, ps = polish(ip, per_i, rng, use_lp, alpha0=0.5, tag=("n%dJ%d") % (L, i))
                v = _as_valid(pp)
                if v is not None:
                    sc = score_of(v)
                    if np.isfinite(sc) and sc < gbest_sc:
                        gbest_sc = sc
                        gbest_p = v.copy()
                print("[polish n=%d J%d] %.6f  global=%.6f" % (L, i, ps, gbest_sc), flush=True)

        # final long squeeze on the global incumbent
        if (gbest_p is not None) and ((deadline - time.monotonic()) > 60):
            rem = deadline - time.monotonic()
            fb = min(rem - 20.0, max(120.0, 0.60 * rem))
            pp, ps = polish(gbest_p, fb, rng, (gbest_p.size <= LP_MAX), alpha0=0.02, tag="FINAL")
            v = _as_valid(pp)
            if v is not None:
                sc = score_of(v)
                if np.isfinite(sc) and sc < gbest_sc:
                    gbest_sc = sc
                    gbest_p = v.copy()
            print("[FINAL] score=%.9f  n=%d" % (gbest_sc, (int(gbest_p.size) if gbest_p is not None else -1)), flush=True)

        if gbest_p is None:
            gbest_p = normalize(np.exp(-1.0 * np.linspace(0, 1, 700)))
        out = np.asarray(gbest_p, dtype=float).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.maximum(out, 0.0)
        s = float(out.sum())
        if s <= 1e-9:
            out = np.full(max(1, out.size), 1.0 / max(1, out.size))
            s = 1.0
        out = out / s
        out = np.minimum(out, 1000.0)
        print("[propose] returning n=%d  score=%.9f" % (out.size, score_of(out)), flush=True)
        return [float(x) for x in out]
    except Exception as e:
        print("PROPOSE_ERROR %r" % (e,), flush=True)
        onset = np.where(np.isfinite(np.exp(-1.0 * np.linspace(0, 1, 700))), np.exp(-1.0 * np.linspace(0, 1, 700)), 0.0)
        s = float(onset.sum())
        if s > 0:
            onset = onset / s
        else:
            onset = np.full(700, 1.0 / 700)
        return [float(x) for x in onset]
