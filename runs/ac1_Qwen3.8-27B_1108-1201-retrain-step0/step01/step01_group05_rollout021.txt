def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def run_smooth_descent(a_init, n, S, max_time, rng, T_init, T_final, top_k_fallback=10):
    """Projected gradient descent with log-sum-exp smoothing and annealing."""
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    T = T_init
    
    # Backtracking line search parameters
    c1 = 1e-4
    rho = 0.5
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        
        # Anneal temperature
        progress = t / max(t, 1)  # not great
        T = T_init * (T_final / T_init) ** min(1.0, t / 200.0)
        
        # Compute smooth gradient
        c_max = c.max()
        scaled = T * (c - c_max)
        # Clip to avoid overflow
        scaled = np.clip(scaled, -100, 100)
        exp_c = np.exp(scaled)
        weights = exp_c / exp_c.sum()
        
        grad = compute_gradient_smooth(a, weights, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        
        # Step size with backtracking
        lr = S / (n * gnorm)  # initial step
        for _ in range(20):
            a_trial = project_simplex(a - lr * grad, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_max - c1 * lr * gnorm:
                a = a_trial
                break
            lr *= rho
        
        # Periodic restart
        if t % 500 == 0 and rng.random() < 0.15:
            noise = rng.normal(0, 0.05 * S / n, n)
            a = project_simplex(a + noise, S)
    
    return best_local, best_score


def run_mirror_v2(a_init, n, S, max_time, rng, eta_init, top_k):
    """Improved mirror descent with adaptive step size."""
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    eta = eta_init
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        
        # Use top-k for gradient
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-15:
            break
        weights_full = np.zeros_like(c)
        weights_full[top_indices] = top_vals / total
        
        grad = compute_gradient_smooth(a, weights_full, 1.0)
        gmax = np.abs(grad).max()
        if gmax < 1e-12:
            break
        grad = grad / gmax
        
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.full(n, S / n)
        else:
            a = a_new * (S / s_new)
        
        # Adaptive eta
        eta = eta_init / (1.0 + 0.01 * t)
    
    return best_local, best_score

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Strategy:
  1. Exploit scale invariance: normalize sum(a) = sqrt(2n) so objective = max(conv).
  2. Use multiple optimization methods: projected gradient with adaptive step,

  4. Extensive multi-round polishing of the best candidate.
  5. FFT-based convolution for speed (O(n log n) per step).

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, c, top_k):
    """Subgradient using top-k active lags (equal weight). Returns normalized grad."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    grad = np.zeros(n, dtype=np.float64)
    for k in top_indices:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= len(top_indices)
    return grad, top_indices


def compute_weighted_subgradient(a, c, top_k):
    """Subgradient using top-k active lags (weight proportional to c[k])."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-15:
        return np.zeros(n)
    grad = np.zeros(n, dtype=np.float64)
    for i, k in enumerate(top_indices):
        k = int(k)
        w = top_vals[i] / total
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def compute_smooth_gradient_fft(a, c, T):
    """Smooth gradient using log-sum-exp weights, computed via FFT."""
    n = len(a)
    c_max = c.max()
    scaled = T * (c - c_max)
    scaled = np.clip(scaled, -50, 50)
    exp_c = np.exp(scaled)
    weights = exp_c / exp_c.sum()
    
    # grad[m] = 2 * sum_l a[l] * weights[m+l] for m=0..n-1
    # Use FFT cross-correlation
    fsize = 1 << (3 * n - 1).bit_length()
    A = fft(a, fsize)
    W = fft(weights, fsize)
    corr = np.real(ifft(np.conj(A) * W))
    grad = 2.0 * corr[:n]
    return grad

    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    eta = eta_init
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    
    # Phase 1: Smooth (large T) for global progress
    # Phase 2: Sharp (small T / top-k) for local refinement
    T_current = 10.0  # start smooth
    T_target = 100.0  # end sharp
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        
        # Anneal temperature
        fraction = min(1.0, t / 300.0)
        T_current = T_current + (T_target - T_current) * 0.01
        
        if T_current < 30:
            # Use smooth gradient
            grad = compute_smooth_gradient_fft(a, c, T_current)
        else:
            # Switch to top-k
            grad = compute_weighted_subgradient(a, c, 6)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        
        a_norm = np.linalg.norm(a)
        L = 2.0 * a_norm
        step = S / (np.sqrt(n) * L)
        
        a_trial = project_simplex(a - step * grad * gnorm, S)
        c_trial = fft_convolve_self(a_trial)
        
        if c_trial.max() < cur_max:
            a = a_trial
        else:
            half_step = project_simplex(a - 0.5 * step * grad * gnorm, S)
            c_half = fft_convolve_self(half_step)
            if c_half.max() < cur_max:
                a = half_step
            else:
                # Try a small random perturbation
                a = project_simplex(a - 0.1 * step * grad * gnorm + 
                                   0.01 * (S/n) * rng.standard_normal(n), S)
    
    return best_local, best_score


def run_multi_topk(a_init, n, S, max_time, rng):
    """Cycle through different top-k values to escape local minima."""
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    topk_cycle = [3, 5, 8, 12, 20, 30]
    topk_idx = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        top_k = topk_cycle[topk_idx % len(topk_cycle)]
        if t % 50 == 0:
            topk_idx += 1
        
        a, s_cur = run_projected_grad(a, n, S, 2.0, rng, top_k=top_k, 
                                       use_weighted=True, lr_scale=1.0,
                                       noise_every=200, noise_prob=0.2, noise_scale=0.05)
        if s_cur < best_score:
            best_score = s_cur
            best_local = a.copy()
    
    return best_local, best_score


def make_initializations(rng, n):
    """Create diverse initializations of length n."""
    inits = []
    
    # Given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and given.sum() > 0:
            inits.append(("given", np.clip(given, 0, 1000)))
            # Smoothed
            kernel = np.ones(5) / 5.0
            smooth = np.convolve(np.clip(given, 0, 1000), kernel, mode='same')
            inits.append(("given_smooth", smooth))
            # Perturbed
            pert = np.clip(given, 0, 1000) * (1 + 0.2 * rng.standard_normal(n))
            inits.append(("given_perturb", np.maximum(pert, 0)))
            # Dilated / shifted
            shifted = np.roll(np.clip(given, 0, 1000), n // 7)
            inits.append(("given_shift", shifted))
    except Exception:
        pass
    
    # Random uniform
    inits.append(("rand", rng.random(n) + 0.05))
    
    # Beta distribution
    inits.append(("beta", rng.beta(2, 3, n)))
    
    # Cosine window
    x = np.linspace(0, np.pi, n)
    inits.append(("cos_window", 0.3 + 0.7 * np.cos(x) ** 2))
    
    # Multi-sine
    tt = np.linspace(0, 1, n)
    sig = np.ones(n)
    for f in (3, 7, 17, 31):
        sig += (0.3 / f) * np.sin(2 * np.pi * f * tt + rng.random() * 2 * np.pi)
    inits.append(("multisine", np.abs(sig)))
    
    # Two-level (Barker-inspired)
    base = (rng.random(n) > 0.4).astype(float)
    base = 0.3 + 0.7 * base
    inits.append(("bilevel", base))
    
    # Exponential from center
    center = n / 2.0
    x = np.abs(np.arange(n) - center) / center
    inits.append(("exp_center", np.exp(-x * rng.uniform(1.0, 4.0))))
    
    # Periodic with noise
    period = n // 4
    pat = np.sin(2 * np.pi * np.arange(n) / period)
    inits.append(("periodic", np.abs(pat) + 0.3))
    
    return inits


def polish_sequence(a, n, S, time_budget, rng):
    """Multi-round polishing of a single sequence."""
    a = np.clip(a.copy(), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    
    # Round 1: Projected gradient with small top_k
    seg = min(time_budget * 0.3, time_budget)
    a, s1 = run_projected_grad(a, n, S, seg, rng, top_k=3, use_weighted=True,
                                lr_scale=0.5, noise_every=500, noise_prob=0.05, noise_scale=0.02)
    print("    Polish R1 (top3): %.6f" % s1)
    if s1 < best_score:
        best_score, best_local = s1, a.copy()
    
    # Round 2: Mirror descent
    time_left = time_budget - (time.monotonic() - start) - 5
    if time_left > 3:
        a, s2 = run_mirror_v2(best_local, n, S, min(time_left * 0.4, 200), rng,
                              eta_init=0.3, top_k=8)
        print("    Polish R2 (mirror): %.6f" % s2)
        if s2 < best_score:
            best_score, best_local = s2, a.copy()
    
    # Round 3: Smooth combined
    time_left = time_budget - (time.monotonic() - start) - 5
    if time_left > 3:
        a, s3 = run_smooth_combined(best_local, n, S, min(time_left * 0.3, 200), rng)
        print("    Polish R3 (smooth): %.6f" % s3)
        if s3 < best_score:
            best_score, best_local = s3, a.copy()
    
    # Round 4: Very fine projected gradient
    time_left = time_budget - (time.monotonic() - start) - 3
    if time_left > 2:
        a, s4 = run_projected_grad(best_local, n, S, min(time_left * 0.5, 300), rng,
                                    top_k=3, use_weighted=True,
                                    lr_scale=0.2, noise_every=2000, noise_prob=0.0, noise_scale=0.0)
        print("    Polish R4 (fine): %.6f" % s4)
        if s4 < best_score:
            best_score, best_local = s4, a.copy()
    
    # Round 5: Multi-topk cycling
    time_left = time_budget - (time.monotonic() - start) - 2
    if time_left > 2:
        a, s5 = run_multi_topk(best_local, n, S, min(time_left * 0.5, 200), rng)
        print("    Polish R5 (multi-topk): %.6f" % s5)
        if s5 < best_score:
            best_score, best_local = s5, a.copy()
    
    return best_local, best_score


def propose_candidate(seed=42, budget_s=2400):
    """Search for a non-negative sequence minimizing 2*n*max(conv)/(sum^2)."""
    rng = np.random.default_rng(seed)
    start_global = time.monotonic()
    deadline = start_global + budget_s - 15.0
    
    # Determine which lengths to try
    n_options = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 128:
            n_options.append(given.size)
    except Exception:
        pass
    for n_try in (2048, 4096, 8192):
        if n_try not in n_options:
            n_options.append(n_try)
    
    print("Lengths to try: %s" % str(n_options))
    print("Total budget: %.0f s" % budget_s)
    
    global_best = np.inf
    global_a = None
    results = []
    
    # Allocate time: 60% exploration, 35% polish, 5% buffer
    explore_time = 0.60 * (budget_s - 15)
    polish_time = 0.35 * (budget_s - 15)
    
    # Phase 1: Explore
    print("\n=== PHASE 1: EXPLORATION ===")
    explore_deadline = start_global + explore_time
    
    total_inits = 0
    for n in n_options:
        if time.monotonic() > explore_deadline:
            break
        inits = make_initializations(rng, n)
        S = np.sqrt(2.0 * float(n))
        total_inits += len(inits)
    
    explore_deadline = start_global + explore_time
    per_init_time = explore_time / max(1, total_inits)
    
    for n in n_options:
        if time.monotonic() > explore_deadline:
            break
        inits = make_initializations(rng, n)
        S = np.sqrt(2.0 * float(n))
        
        for name, arr in inits:
            if time.monotonic() > explore_deadline:
                break
            arr = np.clip(arr.reshape(-1).astype(np.float64), 0, 1000)
            if arr.sum() < 1e-10:
                continue
            
            t0 = time.monotonic()
            time_left = min(per_init_time * 1.5, explore_deadline - t0)
            if time_left < 5:
                continue
            
            # Run 2-3 methods per init
            seg = time_left / 3.0
            
            a1, s1 = run_projected_grad(arr, n, S, seg, rng, top_k=5,
                                         use_weighted=True, lr_scale=1.0)
            if s1 < global_best:
                global_best = s1
                global_a = a1.copy()
                results.append((n, name, "grad", s1))
                print("  [%d] %s grad: %.6f (best)" % (n, name, s1))
            
            if time.monotonic() < explore_deadline - 5:
                a2, s2 = run_mirror_v2(a1, n, S, min(seg, explore_deadline - time.monotonic() - 5),
                                        rng, eta_init=0.8, top_k=5)
                if s2 < global_best:
                    global_best = s2
                    global_a = a2.copy()
                    results.append((n, name, "mirror", s2))
                    print("  [%d] %s mirror: %.6f (best)" % (n, name, s2))
            
            if time.monotonic() < explore_deadline - 5:
                a3, s3 = run_smooth_combined(a1 if s1 < s2 else a2, n, S,
                                              min(seg, explore_deadline - time.monotonic() - 5), rng)
                if s3 < global_best:
                    global_best = s3
                    global_a = a3.copy()
                    results.append((n, name, "smooth", s3))
                    print("  [%d] %s smooth: %.6f (best)" % (n, name, s3))
    
    # Sort results for reporting
    results.sort(key=lambda x: x[3])
    print("\nTop 5 found during exploration:")
    for r in results[:5]:
        print("  n=%d init=%s method=%s score=%.6f" % r)
    
    # Phase 2: Polish
    print("\n=== PHASE 2: POLISHING ===")
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        print("Polishing sequence of length %d, current best: %.6f" % (n, global_best))
        
        polish_deadline = deadline
        time_for_polish = polish_deadline - time.monotonic() - 10
        if time_for_polish > 10:
            global_a, global_best = polish_sequence(global_a, n, S, time_for_polish, rng)
    
    # Phase 3: Try a different length if time remains
    print("\n=== PHASE 3: FINAL CHECKS ===")
    time_left = deadline - time.monotonic() - 10
    if time_left > 30 and global_a is not None:
        # Try to improve with a different method on the best sequence
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        
        # Attempt multi-topk refinement
        a_final, s_final = run_multi_topk(global_a, n, S, min(time_left * 0.5, 300), rng)
        print("  Multi-topk final: %.6f" % s_final)
        if s_final < global_best:
            global_best = s_final
            global_a = a_final.copy()
    
    # Final verification
    if global_a is not None:
        final = np.clip(global_a, 0, 1000)
        if final.sum() < 0.01:
            final = np.ones(256)
        verified = evaluate_sequence(final.tolist())
        print("\nFinal verified score: %.6f" % verified)
        if np.isfinite(verified) and verified < global_best:
            global_best = verified
            global_a = final
    else:
        global_a = np.ones(256)
        verified = evaluate_sequence(global_a.tolist())
        print("Fallback verified: %.6f" % verified)
    
    elapsed = time.monotonic() - start_global
    print("\nTotal time: %.0f s. Best score: %.6f" % (elapsed, global_best))
    
    return [float(x) for x in np.clip(global_a, 0, 1000)]
