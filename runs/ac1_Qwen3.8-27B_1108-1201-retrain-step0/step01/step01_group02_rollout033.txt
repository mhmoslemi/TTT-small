```python
    a = np.maximum(0, a)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    for level in n_lse_levels:
        elapsed = time.monotonic() - start
        if elapsed > time_limit:
            break
        # For each level, do several gradient steps
        for it in range(200):
            elapsed = time.monotonic() - start
            if elapsed > time_limit:
                break
            
            c = np.convolve(a, a)
            # Log-sum-exp smoothing of max
            scaled = level * c
            max_c = scaled.max()
            lse = max_c + np.log(np.sum(np.exp(scaled - max_c)))
            score_approx = lse / level  # approximates max(c)
            
            # Gradient of lse w.r.t. a
            # d(lse)/da[i] = (1/level) * sum_m exp(level*c[m]) * dc[m]/da[i] / sum_m exp(level*c[m])
            # dc[m]/da[i] = a[m-i] (with bounds checking)
            
            weights = np.exp(scaled - max_c)
            weights /= weights.sum()
            
            # Gradient: for each i, grad[i] = sum_m weights[m] * a[m-i]
            # This is like a weighted average of shifted versions of a
            grad = np.zeros(n)
            for m in range(2*n-1):
                if weights[m] < 1e-15:
                    continue
                # c[m] = sum_i a[i] * a[m-i] where 0<=m-i<n
                # dc[m]/da[i] = a[m-i] if 0 <= m-i < n, else 0

                # For i in [lo, hi], dc[m]/da[i] = a[m-i]
                # So grad[i] += weights[m] * a[m-i]

                # a[j] for j in [m-hi, m-lo]

                grad[lo:hi+1] += weights[m] * a[j_lo:j_hi+1][::-1]
            
            # Normalize gradient to have zero sum (to maintain sum(a)=1)
            grad -= grad.mean()
            
            # Gradient descent step
            lr = 0.01  # learning rate
            a_new = a - lr * grad
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s < 1e-15:

            a_new /= s

            a = a_new
```

Actually, here's a cleaner approach inspired by the **multiplicative weight update** (used in algorithms for min-cut and other problems):

This multiplicative update has nice properties: it naturally maintains non-negativity, and it's related to the exponential family / maximum entropy principle.

```python
def multiplicative_update(a, beta, eta, n_iters):
    n = len(a)
    L = 1
    while L < 3*n:
        L *= 2
    for it in range(n_iters):
        c = np.convolve(a, a)
        # Compute weights
        scaled = beta * c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        # Compute gradient (cross-correlation of w and a)
        w_pad = np.zeros(L)
        w_pad[:len(w)] = w
        a_pad = np.zeros(L)
        a_pad[:n] = a
        grad = np.real(np.fft.ifft(np.fft.fft(w_pad) * np.conj(np.fft.fft(a_pad))))[:n]
        # Multiplicative update
        a = a * np.exp(-eta * grad)
        a = np.maximum(0, a)
        s = a.sum()
        if s < 1e-15:
            break
        a /= s
    return a
```

```python
def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def compute_gradient(a, weights, next_pow2):
    """Compute gradient of sum_m weights[m]*c[m] w.r.t. a, using FFT."""
    n = len(a)
    L = next_pow2(3 * n)
    w_pad = np.zeros(L)
    w_pad[:len(weights)] = weights
    a_pad = np.zeros(L)
    a_pad[:n] = a
    grad_full = np.real(np.fft.ifft(np.fft.fft(w_pad) * np.conj(np.fft.fft(a_pad))))
    return grad_full[:n]

def next_pow2(n):
    m = 1
    while m < n:
        m *= 2
    return m

def lse_gradient_descent(a, time_limit, verbose=True):
    """
    Minimize max(conv(a,a)) using log-sum-exp smoothed gradient descent.
    """
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    L = next_pow2(3 * n + 10)
    best_a = a.copy()
    best_score = score_sequence(a)
    start = time.monotonic()
    
    # Schedule: start with small beta (smooth), increase to large beta (sharp)
    n_phases = 5
    for phase in range(n_phases):
        beta = 10.0 * (phase + 1) ** 1.5  # Increasing sharpness
        n_iters = max(50, int(200 * (1.0 / (phase + 1)**0.5)))
        for it in range(n_iters):
            elapsed = time.monotonic() - start
            if elapsed > time_limit:
                return best_a, best_score
            
            c = np.convolve(a, a)
            c_max = c.max()
            scaled = beta * (c - c_max)

            if w_sum < 1e-30:
```

```python
"""
FFT-accelerated optimization for minimizing peak auto-convolution of non-negative sequences.
Uses log-sum-exp smoothed gradient descent with adaptive step size, multiplicative updates,
active-set refinement, and multi-start with diverse initializations and lengths.
The key innovation is computing gradients via FFT in O(n log n) per step, enabling
thousands of iterations within the time budget.
"""

import numpy as np
import time


def score_sequence(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    """Clip to non-negative and normalize to unit sum."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def next_pow2(n):
    """Next power of 2 >= n."""
    m = 1
    while m < n:
        m *= 2
    return m


def compute_gradient_fft(a, weights, L):
    """
    Compute gradient g[i] = sum_m weights[m] * a[m-i] using FFT.
    This is the cross-correlation of weights and a.
    """
    n = len(a)
    w_pad = np.zeros(L)
    w_pad[:len(weights)] = weights
    a_pad = np.zeros(L)
    a_pad[:n] = a
    grad_full = np.real(np.fft.ifft(np.fft.fft(w_pad) * np.conj(np.fft.fft(a_pad))))
    return grad_full[:n]


def make_init_raised_cosine(n):
    """Raised cosine initialization."""
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def make_init_gaussian(n, sigma_frac=0.3):
    """Gaussian initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def make_init_blackman(n):
    """Blackman window initialization."""
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def make_init_random_smooth(n, rng, n_bumps=None):
    """Random smooth initialization (sum of random Gaussians)."""
    if n_bumps is None:
        n_bumps = int(rng.integers(2, max(3, n // 100)))
    a = np.zeros(n)
    t = np.arange(n, dtype=np.float64)
    for _ in range(n_bumps):
        center = float(rng.uniform(0.1, 0.9)) * (n - 1)
        width = float(rng.uniform(0.05, 0.25)) * n
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return make_init_raised_cosine(n)
    return a / s


def make_init_asymmetric_bump(n, rng):
    """Asymmetric bump - two bumps of different heights."""
    a = np.zeros(n)

    c2 = float(rng.uniform(0.6, 0.8)) * n

def lse_gradient_descent(a, time_limit, verbose=False, beta_start=5.0, beta_end=500.0,
                         n_phases=8, lr_scale=0.03):
    """
    Log-sum-exp smoothed gradient descent with adaptive beta schedule.
    Beta increases over time to sharpen the approximation to max.
    """
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    L = next_pow2(3 * n + 10)
    best_a = a.copy()
    best_score = score_sequence(a)
    start = time.monotonic()

    for phase in range(n_phases):
        frac = (phase + 1) / n_phases
        beta = beta_start * (beta_end / beta_start) ** frac
        n_iters = max(30, int(300 // (1.0 + 0.5 * phase)))
        lr = lr_scale / (1.0 + 0.3 * phase)

        for it in range(n_iters):
            elapsed = time.monotonic() - start
            if elapsed > time_limit:
                return best_a, best_score

            c = np.convolve(a, a)
            c_max = c.max()
            scaled = beta * (c - c_max)
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                break
            w /= w_sum

            grad = compute_gradient_fft(a, w, L)
            grad -= grad.mean()

            a_new = a - lr * grad
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s < 1e-15:
                a = best_a.copy()
                break
            a_new /= s

            new_score = score_sequence(a_new)
            if new_score < best_score - 1e-15:
                best_score = new_score
                best_a = a_new.copy()

            a = a_new

            if verbose and it % 30 == 0:
                print(f"        LSE-GD phase {phase}/{n_phases}, iter {it}: score={score_sequence(a):.6f}, beta={beta:.1f}")

    return best_a, best_score


def multiplicative_update(a, time_limit, verbose=False, betas=None):
    """
    Multiplicative (exponential) update: a[i] *= exp(-eta * grad[i]).
    Naturally maintains non-negativity.
    """
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    L = next_pow2(3 * n + 10)

    best_score = score_sequence(a)
    start = time.monotonic()

    if betas is None:
        betas = [5, 10, 20, 50, 100, 200, 400]

    for beta in betas:
        for it in range(150):
            elapsed = time.monotonic() - start
            if elapsed > time_limit:
                return best_a, best_score

            c = np.convolve(a, a)
            scaled = beta * c
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                break
            w /= w_sum

            grad = compute_gradient_fft(a, w, L)
            grad -= grad.mean()

            eta = 1.0 / (1.0 + beta * 0.05)
            a = a * np.exp(-eta * grad)
            a = np.maximum(0, a)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
                break

            sc = score_sequence(a)

                best_score = sc
                best_a = a.copy()

        B = A * A  # FFT of convolution (up to scaling)
        b = np.real(np.fft.ifft(B))[:2 * n - 1]
        b_mean = b.mean()

        # Error: how non-flat is b?
        error = b - b_mean
        # Modify B to reduce error in time domain

        t4 = t_limit * 0.20  # Local search

        # Phase 1: LSE Gradient Descent
        a1, s1 = lse_gradient_descent(a, t1, verbose=False)
        print(f"  After LSE-GD: {s1:.6f}")

        # Phase 2: Multiplicative Update
        a2, s2 = multiplicative_update(a1, t2, verbose=False)
        print(f"  After MU: {s2:.6f}")

        # Phase 3: Active Set Refinement
        a3, s3 = active_set_refine(a2, t3, verbose=False)
        print(f"  After AS: {s3:.6f}")

        # Phase 4: Local Search
        a4, s4 = local_search(a3, t4, rng, verbose=False)
        print(f"  After LS: {s4:.6f}")

        # Track best
        final_score = min(s1, s2, s3, s4)
        if s4 <= s3:
            final_a = a4
        elif s3 <= s2:
            final_a = a3
        elif s2 <= s1:
            final_a = a2
        else:
            final_a = a1

        if final_score < best_score:
            best_score = final_score
            best_seq = final_a.copy()
            print(f"  *** NEW BEST: {best_score:.6f}, n={n}, elapsed={time.monotonic()-total_start:.1f}s")

    # === Final Polish Phase ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\n=== Final Polish (remaining={remaining:.0f}s) ===")

            # Try spectral refinement
            t_spec = remaining * 0.15
            a_sp, s_sp = spectral_refine(best_seq, t_spec, verbose=True)
            if s_sp < best_score:
                best_score = s_sp
                best_seq = a_sp
                print(f"  After spectral: {best_score:.6f}")

            # More LSE-GD with fine-tuned parameters
            remaining2 = deadline - time.monotonic()
            if remaining2 > 30:
                a_gd, s_gd = lse_gradient_descent(best_seq, remaining2 * 0.35,
                                                  beta_start=100.0, beta_end=2000.0,
                                                  n_phases=6, lr_scale=0.01, verbose=True)
                if s_gd < best_score:
                    best_score = s_gd
                    best_seq = a_gd
                    print(f"  After fine LSE-GD: {best_score:.6f}")

            # Active set with more peaks
            remaining3 = deadline - time.monotonic()
            if remaining3 > 20:
                a_as, s_as = active_set_refine(best_seq, remaining3 * 0.3, n_active=10, verbose=True)
                if s_as < best_score:
                    best_score = s_as
                    best_seq = a_as
                    print(f"  After fine AS: {best_score:.6f}")

            # Final local search
            remaining4 = deadline - time.monotonic()
            if remaining4 > 15:
                a_ls, s_ls = local_search(best_seq, remaining4 * 0.8, rng, verbose=True)
                if s_ls < best_score:
                    best_score = s_ls
                    best_seq = a_ls
                    print(f"  After final LS: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = make_init_raised_cosine(500)
        best_seq = normalize_nonneg(best_seq)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Final verification
    print(f"\nFinal result: score={best_score:.6f}, n={len(best_seq)}")
    verified = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified via evaluate_sequence: {verified:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")

    return [float(x) for x in best_seq]
```
