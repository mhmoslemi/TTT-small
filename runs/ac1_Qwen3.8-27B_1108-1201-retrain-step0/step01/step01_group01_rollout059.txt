"""
Subgradient descent with full subgradient, adaptive step size, and local search
to minimize the normalized autocorrelation peak of a non-negative sequence.
"""

import numpy as np
import time


def _fft_convolve_same(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient(a, c, beta=100.0):
    """
    Compute the weighted subgradient of max(c) using soft assignment.
    mu[k] = exp(beta * c[k]) / sum exp(beta * c[j])
    g[i] = 2 * sum_k mu[k] * a[k-i] (valid k-i)
    Then center: g -= mean(g) (for simplex constraint)
    """
    # Soft max weights
    c_max = np.max(c)
    log_mu = beta * (c - c_max)
    # For numerical stability
    log_mu -= np.max(log_mu)
    mu = np.exp(log_mu)
    mu_sum = np.sum(mu)
    if mu_sum < 1e-300:
        # Fallback: uniform over top peaks
        top_k = np.argmax(c)
        mu = np.zeros(len(c))
        mu[top_k] = 1.0
    else:
        mu /= mu_sum
    
    # Compute g[i] = 2 * sum_k mu[k] * a[k-i]
    # This is cross-correlation: g[i] = 2 * (mu ⋆ a)[i]
    # where (mu ⋆ a)[i] = sum_k mu[k] * a[k-i]
    # = ifft(FFT(mu) * conj(FFT(a)))
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    mu_f = np.fft.rfft(mu, size)
    a_f = np.fft.rfft(a, size)
    # Cross-correlation: ifft(FFT(mu) * conj(FFT(a)))
    # This gives sum_k mu[k] * conj(a[k-i]) = sum_k mu[k] * a[k-i] (since a is real)
    g = np.fft.irfft(mu_f * np.conj(a_f), size)[:2 * n - 1]
    # We want g[i] for i in [0, n-1]
    # The cross-correlation gives (mu ⋆ a)[i] = sum_k mu[k] * a[k-i] for i in [-(n-1), 2n-2]
    # We want i in [0, n-1], which corresponds to indices [0, n-1] in the output
    # Wait, the output of ifft has length size. The cross-correlation at lag i is at position (i mod size).
    # For i >= 0, it's at position i. For i < 0, it's at position size + i.
    # We want i in [0, n-1], so we take positions [0, n-1].
    g = g[:n] * 2.0
    
    # Center for simplex constraint
    g -= np.mean(g)
    return g


def _compute_objective(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    c = _fft_convolve_same(a)
    return 2.0 * n * np.max(c)


def subgradient_descent(a, deadline, beta=100.0, f_lower=1.49, patience=200):
    """
    Subgradient descent to minimize 2n*max(c) subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a /= S
    
    best_a = a.copy()
    best_f = _compute_objective(a)
    no_improve = 0
    eta_base = 1.0
    
    while time.monotonic() < deadline:
        c = _fft_convolve_same(a)
        f_val = 2.0 * n * np.max(c)
        
        if f_val < best_f - 1e-10:
            best_f = f_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= patience:
            break
        
        # Compute subgradient
        g = _subgradient(a, c, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Adaptive step size
        eta = (f_val - f_lower) / (2.0 * n * g_norm)
        # Cap the step size
        eta = min(eta, 0.5 / g_norm)
        
        a_new = a - eta * g
        a_new = _project_simplex(a_new, 1.0)
        
        # Check if the step was valid
        c_new = _fft_convolve_same(a_new)
        f_new = 2.0 * n * np.max(c_new)
        
        if f_new < f_val:
            a = a_new
        else:
            # Halve step size and retry
            a_new = a - 0.5 * eta * g
            a_new = _project_simplex(a_new, 1.0)
            c_new = _fft_convolve_same(a_new)
            f_new = 2.0 * n * np.max(c_new)
            if f_new < f_val:
                a = a_new
            else:
                no_improve += 5
    
    return best_a, best_f

    and transfer mass to a random position.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    c = _fft_convolve_same(a)
    best_f = 2.0 * n * np.max(c)
    
    iterations = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        if n_iter_budget is not None and iterations >= n_iter_budget:
            break
        iterations += 1
        
        # Pick a peak position (weighted by value)
        c_max = np.max(c)
        # Use top few peaks
        top_k = np.argpartition(c, -10)[-10:]
        top_k = top_k[c[top_k] > 0.8 * c_max]  # Only peaks near the max
        if len(top_k) == 0:
            top_k = np.array([np.argmax(c)])
        
        k = top_k[rng.integers(len(top_k))]
        
        # Find the pair (i, k-i) with largest contribution to c[k]
        # c[k] = sum_i a[i]*a[k-i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        products = a[idx] * a[k - idx]
        max_idx_local = np.argmax(products)
        i = idx[max_idx_local]
        j = k - i
        if j < 0 or j >= n:

            a_new = a.copy()
            a_new[i] -= delta
            a_new[r] += delta

"""
Multi-phase optimization combining subgradient descent with full soft-subgradient,
adaptive Polyak step size, and targeted local search (peak-reduction transfers)
to minimize the normalized autocorrelation peak 2n*max(conv(a,a))/sum(a)^2
for non-negative sequences of length 6121.
"""

import numpy as np
import time


def _fft_convolve_same(a):
    """Full convolution of a with itself via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _soft_subgradient(a, c, beta):
    """
    Compute soft subgradient of max(c).
    mu[k] = exp(beta*(c[k]-max_c)) / sum
    g[i] = 2 * sum_k mu[k] * a[k-i]  (valid indices)
    Centered for simplex constraint.
    """
    n = len(a)
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    shifted -= np.max(shifted)
    mu = np.exp(shifted)
    mu_sum = np.sum(mu)
    if mu_sum < 1e-300:
        idx = np.argmax(c)
        mu = np.zeros(len(c))
        mu[idx] = 1.0
    else:
        mu /= mu_sum

    size = 1 << (2 * n - 1).bit_length()
    mu_f = np.fft.rfft(mu, size)
    a_f = np.fft.rfft(a, size)
    # Cross-correlation: (mu * a_rev) at appropriate lags
    # g[i] = 2 * sum_k mu[k] * a[k-i]
    # = 2 * ifft(FFT(mu) * conj(FFT(a)))[:n]
    g = np.fft.irfft(mu_f * np.conj(a_f), size)[:n] * 2.0

    g -= np.mean(g)
    return g


def _compute_obj_normalized(a):
    """Compute 2n*max(c) for normalized a (sum=1)."""
    n = len(a)
    c = _fft_convolve_same(a)
    return 2.0 * n * np.max(c)


def _subgradient_optimize(a, deadline, beta=200.0, f_lower=1.49, patience=150, max_steps=None):
    """
    Subgradient descent minimizing 2n*max(c) s.t. a>=0, sum(a)=1.
    Uses full soft-subgradient and Polyak-like adaptive step size.
    Returns (best_a, best_objective).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a /= S

    best_a = a.copy()
    c = _fft_convolve_same(a)
    best_f = 2.0 * n * np.max(c)
    no_improve = 0
    step = 0

    while time.monotonic() < deadline:
        if max_steps is not None and step >= max_steps:
            break
        step += 1

        f_val = 2.0 * n * np.max(c)

        if f_val < best_f - 1e-9:
            best_f = f_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            break

        g = _soft_subgradient(a, c, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Polyak step size with cap
        gap = f_val - f_lower
        eta = gap / (2.0 * n * g_norm)
        eta = min(eta, 1.0 / (n * g_norm))
        eta = max(eta, 1e-8 / g_norm)

        a_new = a - eta * g
        a_new = _project_simplex(a_new, 1.0)

        c_new = _fft_convolve_same(a_new)
        f_new = 2.0 * n * np.max(c_new)

        if f_new <= f_val:
            a = a_new
            c = c_new
        else:
            # Try smaller step
            a_new2 = a - 0.5 * eta * g
            a_new2 = _project_simplex(a_new2, 1.0)
            c_new2 = _fft_convolve_same(a_new2)
            f_new2 = 2.0 * n * np.max(c_new2)
            if f_new2 <= f_val:
                a = a_new2
                c = c_new2
            else:
                no_improve += 3

    return best_a, best_f


def _local_search(a, deadline, rng, temperature=0.0, max_no_improve=10000):
    """
    Targeted local search: identify peak-contributing pairs and transfer mass.
    Uses simulated annealing acceptance when temperature > 0.
    Returns (best_a, best_obj, n_evals).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    c = _fft_convolve_same(a)
    best_a = a.copy()
    best_f = 2.0 * n * np.max(c)
    cur_f = best_f
    no_improve = 0
    n_evals = 0
    t = temperature
    t_min = temperature * 0.1
    cool_rate = 0.9995

    while time.monotonic() < deadline:
        if no_improve > max_no_improve:
            break
        n_evals += 1

        c_max = np.max(c)

        # Weight sample from top peaks
        top_count = min(20, len(c))
        top_idx = np.argpartition(c, -top_count)[-top_count:]
        top_idx = top_idx[c[top_idx] > 0.85 * c_max]
        if len(top_idx) == 0:
            top_idx = np.array([np.argmax(c)])

        k = int(top_idx[rng.integers(len(top_idx))])

        # Valid range for i in c[k] = sum_i a[i]*a[k-i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        pairs = a[idx] * a[k - idx]

        # Pick the pair with max product (with some randomness)
        if rng.random() < 0.3:
            # Random pair weighted by product
            w = pairs / (np.sum(pairs) + 1e-30)
            i = int(idx[rng.choice(len(idx), p=w)])
        else:
            i = int(idx[np.argmax(pairs)])

        j = k - i
        if j < 0 or j >= n:
            continue

        # Determine source (the one with larger value)
        if i == j:
            src = i
        elif a[i] >= a[j]:
            src = i
        else:
            src = j

        # Choose target position: prefer low-peak-involvement positions
        # Generate a few candidates
        r_candidates = rng.integers(n, size=5)
        best_r = -1
        best_score_r = np.inf
        for r in r_candidates:
            r = int(r)
            if r == src:
                continue
            # Score: how much does adding to r affect peaks?
            # Approximate: sum over top peaks of a[r] * a[k'-r]
            score = 0.0
            for kp in top_idx[:5]:
                kr = int(kp - r)
                if 0 <= kr < n:
                    score += a[kr]
            if score < best_score_r:
                best_score_r = score
                best_r = r

        if best_r < 0:
            continue

        # Transfer amount
        delta = 0.005 * a[src]
        if delta > a[src] * 0.4:
            delta = a[src] * 0.4
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[best_r] += delta

        c_new = _fft_convolve_same(a_new)
        f_new = 2.0 * n * np.max(c_new)
        n_evals += 1

        # Acceptance
        if f_new <= cur_f or (temperature > 0 and rng.random() < np.exp(-(f_new - cur_f) / max(t, 1e-10) / (2.0 * n))):
            a = a_new
            c = c_new
            cur_f = f_new
            if f_new < best_f:
                best_f = f_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        t = max(t * cool_rate, t_min)

    return best_a, best_f, n_evals


def _swap_local_search(a, deadline, rng, max_no_improve=20000):
    """
    Local search via swaps: exchange a[i] and a[j] for promising pairs.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    c = _fft_convolve_same(a)
    best_a = a.copy()
    best_f = 2.0 * n * np.max(c)
    no_improve = 0
    n_evals = 0

    # Precompute: for each position, its "peak involvement"
    while time.monotonic() < deadline:
        if no_improve > max_no_improve:
            break
        n_evals += 1

        c_max = np.max(c)
        top_count = min(10, len(c))
        top_idx = np.argpartition(c, -top_count)[-top_count:]
        top_idx = top_idx[c[top_idx] > 0.9 * c_max]
        if len(top_idx) == 0:
            top_idx = np.array([np.argmax(c)])

        # Source: position with high peak involvement
        # Compute peak involvement for all positions
        peak_inv = np.zeros(n)
        for kp in top_idx:
            lo = max(0, kp - n + 1)
            hi = min(n - 1, kp)
            for ii in range(lo, hi + 1):
                jj = kp - ii
                if 0 <= jj < n:
                    peak_inv[ii] += a[ii] * a[jj]
                    if jj != ii:
                        peak_inv[jj] += a[ii] * a[jj]

        # Pick source from high peak involvement
        src_candidates = np.argsort(peak_inv)[-min(50, n):]
        src = int(src_candidates[rng.integers(len(src_candidates))])

        # Target: position with low peak involvement
        tgt_candidates = np.argsort(peak_inv)[:min(50, n)]
        tgt = int(tgt_candidates[rng.integers(len(tgt_candidates))])

        if src == tgt:
            continue
        if abs(a[src] - a[tgt]) < 1e-10:
            continue

        # Try the swap
        a_new = a.copy()
        a_new[src], a_new[tgt] = a[tgt], a[src]

        c_new = _fft_convolve_same(a_new)
        f_new = 2.0 * n * np.max(c_new)

        if f_new < best_f - 1e-12:
            a = a_new
            c = c_new
            best_f = f_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_a, best_f, n_evals
