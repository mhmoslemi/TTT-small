```python
import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def lp_direction(f, M):
    """Find g that maximizes sum(g) subject to g >= 0 and (f*g)[k] <= M for all k."""
    n = len(f)
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) -> minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_i f[i] * g[k-i] for valid i, k-i
    # For a given k, the coefficient of g[j] in (f*g)[k] is f[k-j] (if 0 <= k-j < n)
    # So A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # g[j] contributes to (f*g)[k] for k = j, j+1, ..., j+n-1
        # with coefficient f[k-j]
        for k in range(j, j+n):
            A[k, j] = f[k-j]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g[j] >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return None

def optimize_lp(a_init, time_limit, rng):
    """Optimize using LP-based direction finding."""
    a = a_init.copy()
    S = np.sqrt(2 * len(a))
    a = a / np.sum(a) * S
    best = a.copy()
    best_score = local_score(a)
    n = len(a)
    t0 = time.monotonic()
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        M = np.max(c)
        
        g = lp_direction(a, M)
        if g is None:
            break
        
        # Rescale g to match normalization
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum * S
        
        # Convex combination
        t = 0.1 + 0.3 * (1 - (time.monotonic() - t0) / time_limit)
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, 1000)
        
        score = local_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
        
        a = a_new
        step += 1
        
        if step % 10 == 0:
            print(f"    LP step {step}: score={score:.6f}, best={best_score:.6f}")
    
    return best, best_score
```

"""
Multi-phase optimization for minimizing the peak of auto-convolution of
non-negative sequences. Combines LP-based direction finding, gradient descent
with soft-max, and local search polishing. Uses 1/sqrt(i) structured
initializations inspired by integral approximations of flat convolutions.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score (same formula as evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def soft_max_and_weights(x, tau):
    """Compute soft-max value and weights."""
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    value = x_max + tau * np.log(sum_exp)
    return value, weights


def grad_soft_max_conv(a, tau):
    """Compute soft-max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_and_weights(c, tau)
    # grad[j] = 2 * sum_k w[k] * a[k-j] (if 0<=k-j<n)
    # = 2 * conv(a_reversed, w)[j + n - 1]
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return value, g, c


def make_inv_sqrt_init(n, rng, noise_level=0.1):
    """Initialize with 1/sqrt shape (motivated by flat convolution integral)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(idx)
    a += noise_level * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_inv_sqrt_sym_init(n, rng, noise_level=0.05):
    """Symmetric 1/sqrt initialization."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a_left = 1.0 / np.sqrt(idx[:n // 2 + 1])
    a_right = a_left[::-1]
    if n % 2 == 0:
        a = np.concatenate([a_left[:-1], a_right])
    else:
        a = np.concatenate([a_left, a_right[1:]])
    a += noise_level * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_chirp_init(n, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    alpha = 0.3 + 0.3 * rng.random()
    a = 1.0 + alpha * np.cos(np.pi * t * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    n_tones = min(8, n // 3)
    for k in range(1, n_tones + 1):
        amp = 0.3 / (k + 0.5)
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng):
    """Random positive initialization."""
    a = 1.0 + 0.5 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def optimize_gradient(n, a_init, time_limit, rng):
    """Adam gradient descent with annealed soft-max."""
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S
    theta = np.log(a / S + 1e-10)
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_a = a.copy()
    best_score = local_score(a)
    tau_start = 20.0 / n
    tau_end = 0.1 / n
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)

        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)

        # Normalize constraint: proj gradient onto tangent of simplex
        inner = np.dot(a, g_a) / S
        g_theta = softmax_normalize(theta) * (g_a - inner)

        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        lr = 0.5 / np.sqrt(step + 100)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        a_new = softmax_normalize(theta) * S
        score = local_score(a_new)

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            best_theta = theta.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500 and step > 200:
            # Restart with perturbation
            theta = best_theta + 0.1 * rng.standard_normal(n)
            no_improve = 0

        if step % 500 == 0:
            print(f"      grad step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}")

    return best_a, best_score


def lp_direction(f, M):
    """Find g maximizing sum(g) subject to g>=0 and (f*g)[k] <= M."""
    n = len(f)
    c_obj = -np.ones(n)

    # Build constraint matrix: (f*g)[k] = sum_j A[k,j] * g[j] <= M
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = j + n
        A[start:end, j] = f[:n]

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'disp': False, 'presolve': True})
    if res.success:
        return res.x
    return None


def optimize_lp(a_init, time_limit, rng):
    """LP-based optimization: iteratively find direction via LP."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S
    best = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        M = np.max(c)

        g = lp_direction(a, M)
        if g is None:
            break

        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum * S

        # Adaptive step size
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        t_base = 0.3 * (1.0 - 0.5 * elapsed_frac)
        t = t_base * (0.8 + 0.4 * rng.random())
        t = min(t, 0.6)

        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, 1000)

        score = local_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
            stall = 0
            a = a_new
        else:
            stall += 1
            # Try smaller step
            t2 = t * 0.3
            a_try = (1 - t2) * a + t2 * g
            a_try = np.clip(a_try, 0, 1000)
            score_try = local_score(a_try)
            if score_try < local_score(a):
                a = a_try
            else:
                # Perturb to escape
                a = a * np.exp(0.02 * rng.standard_normal(n))
                a = np.clip(a, 0.01, 1000)

        it += 1
        if it % 20 == 0:
            print(f"      LP it={it}, score={local_score(a):.6f}, best={best_score:.6f}")

        if stall > 30:
            break

    return best, best_score


def local_search_polish(a_init, time_limit, rng):
    """Fine-grained local search with multiple perturbation types."""
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()

    it = 0
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.30:
            # Targeted peak reduction
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            # Find top contributors to peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            n_decrease = max(1, top_k // 2)
            dec_idx = top_idx[:n_decrease]
            inc_idx = top_idx[n_decrease:]
            candidate[dec_idx] *= np.exp(rng.normal(-0.06, 0.02, size=len(dec_idx)))
            if len(inc_idx) > 0:
                candidate[inc_idx] *= np.exp(rng.normal(0.06, 0.02, size=len(inc_idx)))
        elif r < 0.50:
            # Block modification
            width = max(1, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        elif r < 0.65:
            # Multi-point swap
            n_pts = max(2, n // 16)
            idx = rng.choice(n, size=2 * n_pts, replace=False)
            dc = rng.normal(0.0, 0.05, size=n_pts)
            candidate[idx[:n_pts]] *= np.exp(dc)
            candidate[idx[n_pts:]] *= np.exp(-dc)
        elif r < 0.80:
            # Partial symmetrization
            sym = 0.5 * (current + current[::-1])
            anti = 0.5 * (current - current[::-1])
            mix = rng.uniform(0.0, 0.3)
            candidate = sym + mix * anti
            candidate = candidate * np.exp(rng.normal(0.0, 0.01, size=n))
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.015, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / 0.001):
            current = candidate
            current_score = new_score

        it += 1
        if it % 200 == 0:
            print(f"      polish it={it}, score={current_score:.6f}, best={best_score:.6f}")

    return best, best_score


def resample_sequence(a, new_n):
    """Resample a sequence to a new length via interpolation."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: gradient optimization, LP-based direction finding,
    and local search polishing across multiple sequence lengths.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            s = local_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()
        except Exception as e:
            print(f"Failed to process provided sequence: {e}")

    # Phase 1: Quick gradient exploration across lengths
    print("\n=== Phase 1: Gradient exploration ===")
    phase1_time = time_budget * 0.25
    phase1_deadline = t_global + phase1_time

    lengths = [128, 192, 256, 320, 384, 512, 768]
    candidates = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 10:
            break
        time_per_init = min(8.0, (phase1_deadline - time.monotonic()) / 4.0)
        if time_per_init < 2.0:
            break

        inits = [
            ("inv_sqrt", make_inv_sqrt_init(n, rng, 0.05)),
            ("inv_sqrt_sym", make_inv_sqrt_sym_init(n, rng, 0.03)),
            ("chirp", make_chirp_init(n, rng)),
            ("multi_tone", make_multi_tone_init(n, rng)),
        ]

        # Also try resampled provided sequence
        if start_seq is not None:
            inits.append(("provided_resample", resample_sequence(start_seq, n)))

        for name, a0 in inits:
            if time.monotonic() > phase1_deadline - 5:
                break
            a0 = np.clip(a0, 0.01, 1000.0)
            a_best, score = optimize_gradient(n, a0, time_per_init, rng)
            if score < best_score:
                best_score = score
                best_seq = a_best.copy()
                print(f"  n={n}, init={name}: NEW BEST {score:.6f}")
            else:
                print(f"  n={n}, init={name}: {score:.6f}")
            candidates.append((score, n, a_best))

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"\nPhase 1 best: {best_score:.6f}")

    # Phase 2: LP-based optimization on top candidates
    print("\n=== Phase 2: LP optimization ===")
    phase2_time = min(time_budget * 0.45, deadline - time.monotonic())
    phase2_deadline = t_global + phase1_time + phase2_time

    # Select top candidates (diverse lengths)
    selected = []
    used_lengths = set()
    for score, n, a in candidates:
        if len(selected) >= 4:
            break
        if n not in used_lengths:
            selected.append((score, n, a))
            used_lengths.add(n)

    # Also add the provided sequence if it's good
    if start_seq is not None and len(start_seq) not in used_lengths:
        selected.append((local_score(start_seq), len(start_seq), start_seq.copy()))

    # Also try a new 1/sqrt init at a medium length
    if len(selected) < 5:
        n_med = 384
        if n_med not in used_lengths:
            selected.append((local_score(make_inv_sqrt_init(n_med, rng, 0.02)),
                            n_med, make_inv_sqrt_init(n_med, rng, 0.02)))

    for score, n, a0 in selected:
        if time.monotonic() > phase2_deadline - 15:
            break
        time_left = phase2_deadline - time.monotonic()
        time_per = min(time_left / (len(selected) + 1), 120.0)
        if time_per < 5.0:
            break

        print(f"  LP on n={n}, starting score={score:.6f}, time={time_per:.0f}s")
        a_best, lp_score = optimize_lp(a0, time_per, rng)
        if lp_score < best_score:
            best_score = lp_score
            best_seq = a_best.copy()
            print(f"    *** LP NEW BEST: {best_score:.6f} at n={n} ***")
        else:
            print(f"    LP result: {lp_score:.6f}")

    # Phase 3: Gradient refinement on best
    print(f"\n=== Phase 3: Gradient refinement ===")
    phase3_time = min(time_budget * 0.15, deadline - time.monotonic() - 30)
    if phase3_time > 5 and best_seq is not None:
        n = len(best_seq)
        time_per = min(30.0, phase3_time / 2)
        print(f"  Refining n={n}, score={best_score:.6f}")
        a_best, ref_score = optimize_gradient(n, best_seq, time_per, rng)
        if ref_score < best_score:
            best_score = ref_score
            best_seq = a_best.copy()
            print(f"    *** Refinement NEW BEST: {best_score:.6f} ***")

        # Also try a different init at the same length
        a_alt = make_inv_sqrt_init(n, rng, 0.02)
        a_best2, score2 = optimize_gradient(n, a_alt, time_per, rng)
        if score2 < best_score:
            best_score = score2
            best_seq = a_best2.copy()
            print(f"    *** Alt init NEW BEST: {best_score:.6f} ***")

    # Phase 4: LP on the current best
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print(f"\n=== Phase 4: LP on best ===")
        time_left = deadline - time.monotonic() - 20
        time_per = min(60.0, time_left / 2)
        if time_per > 5:
            n = len(best_seq)
            print(f"  LP on best n={n}, score={best_score:.6f}")
            a_best, lp_score = optimize_lp(best_seq, time_per, rng)
            if lp_score < best_score:
                best_score = lp_score
                best_seq = a_best.copy()
                print(f"    *** LP Polish NEW BEST: {best_score:.6f} ***")

    # Phase 5: Final local search polish
    print(f"\n=== Phase 5: Local search polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        polish_time = min(60.0, deadline - time.monotonic() - 5)
        if polish_time > 3:
            print(f"  Polishing, time={polish_time:.0f}s")
            polished, polish_score = local_search_polish(best_seq, polish_time, rng)
            if polish_score < best_score:
                best_score = polish_score
                best_seq = polished.copy()
                print(f"  After polish: {best_score:.6f}")

    # Final verification
    print(f"\n=== FINAL: score={best_score:.6f} ===")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Verify with evaluate_sequence
        try:
            verified = evaluate_sequence(final_list)
            print(f"Verified score: {verified:.6f}, n={len(final_list)}")
        except Exception as e:
            print(f"Verification error: {e}")
            verified = best_score
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
