"""
Soft-max gradient descent for minimizing peak auto-convolution of non-negative sequences.
Uses logarithmic-barrier-style soft-max to smooth the max, with softmax parameterization
to enforce non-negativity and normalization. Combines multiple restarts, annealing,
and local search polishing.

import numpy as np
import time


def score_seq(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def grad_a_from_w(a, w):
    """
    Compute grad of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    grad[i] = 2 * sum_k w[k] * a[k-i] (with a out-of-range = 0)
    Using Hankel matrix: M[j,i] = w[j+i], then grad = 2*M@a
    """
    n = len(a)
    # w has length 2n-1
    # M[j, i] = w[j + i] for j=0..n-1, i=0..n-1
    # grad[j] = 2 * sum_i w[j+i] * a[i]
    idx = np.arange(n)
    # Instead of forming the full matrix, use a more efficient approach
    # M @ a = [sum_i w[i] * a[i], sum_i w[i+1] * a[i], ..., sum_i w[i+n-1] * a[i]]
    # This is like a sliding dot product
    # Can be computed with FFT or just as a matrix-vector product
    # For n up to ~10000, forming the matrix is fine (100M entries = 800MB... too much for n=10000)
    # For n=5000: 25M entries = 200MB, borderline
    # For n=1000: 1M entries = 8MB, fine
    # Let me use a block approach or FFT for large n
    
    if n <= 3000:
        M = w[idx[:, None] + idx[None, :]]
        grad = 2.0 * (M @ a)
    else:
        # FFT-based computation
        # grad[j] = 2 * sum_i a[i] * w[j+i]
        # = 2 * cross_correlation(a, w) at lag j (with appropriate convention)
        # cross_correlation(a, w)[j] = sum_i a[i] * w[i+j]
        # Using FFT: = ifft(fft(a) * conj(fft(w))) with appropriate handling
        # Actually: sum_i a[i] * w[i+j] = (a * flip(w))[j + len(w) - 1] ... 
        # Let me use np.correlate
        # np.correlate(w, a, 'full')[k] = sum_i w[i+k] * conj(a[i]) = sum_i w[i+k] * a[i] (real)
        # So we want np.correlate(w, a, 'full')[j] for j = 0, ..., n-1
        corr = np.correlate(w, a, 'full')
        # corr has length (2n-1) + n - 1 = 3n-2
        # corr[k] = sum_i w[i+k] * a[i]
        # We want k = j for j = 0, ..., n-1
        # But the 'full' output starts at k = -(n-1)
        # In numpy, 'full' correlate output index 0 corresponds to k = -(len(a)-1) = -(n-1)
        # So corr[numpy_idx] corresponds to k = numpy_idx - (n-1)
        # We want k = j, so numpy_idx = j + n - 1
        grad = 2.0 * corr[(n-1):(2*n-1)]
    
    return grad


def soft_max_weights(conv, T):
    """Compute soft-max weights from convolution values."""
    # Numerically stable
    conv_shifted = conv - conv.max()
    w = np.exp(conv_shifted / T)
    w /= w.sum()
    return w

def lp_refine(a_init, time_limit, verbose=True):
    """LP-based refinement (from previous approach)."""
    start_time = time.monotonic()
    a = normalize(a_init.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)

    it = 0

        elapsed = time.monotonic() - start_time
        if elapsed > time_limit:
            break
        
        conv = np.convolve(a, a)
        M = conv.max()
        score = score_seq(a)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 30:
            break
        
        remaining = time_limit - elapsed
        if remaining < 10.0:
            break
        
        # Solve LP: maximize sum(b) subject to conv(a,b) <= M, b >= 0
        # Build constraint matrix
        n_rows = 2 * n - 1
        # A @ b <= M where A is the convolution matrix
        # A[k, j] = a[k-j] (valid indices)
        # conv(a,b)[k] = sum_j a[k-j] * b[j]
        
        timeout = max(5.0, min(60.0, remaining * 0.3))
        
        try:
            # Build sparse matrix
            col_idx = np.repeat(np.arange(n), n)
            row_idx = np.concatenate([np.arange(j, j + n) for j in range(n)])
            data = np.tile(a, n)
            A = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))
            
            c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
            b_ub = np.full(n_rows, M)
            bounds = [(0, None)] * n
            
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': timeout})

                g = np.maximum(0, res.x)
                sum_g = g.sum()
                if sum_g <= 1.0 + 1e-10:
                    break
                g = g / sum_g
                alpha = 0.3
                a_new = (1 - alpha) * a + alpha * g
                a_new = normalize(a_new)
                a = a_new
        except Exception:
            break
    
    final_score = score_seq(best_a)
    if final_score < best_score:
        best_score = final_score

        print(f"    LP done: best={best_score:.6f}, iters={it}, "

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key insight: optimal sequences are asymmetric (peak of convolution is off-center).
Combines soft-max gradient descent, LP-based refinement, and local search with
multiple asymmetric initializations and adaptive time allocation.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def score_seq(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_asymmetric(n, shift_frac=0.1):
    """Gaussian centered off-center for asymmetric init."""
    t = np.arange(n, dtype=np.float64)
    center = n * (0.5 - shift_frac)
    width = n * 0.25
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    return np.maximum(0, a)


def initial_ramp(n, power=2.0):
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    return t ** power + 0.01


def initial_dual_bump(n):
    t = np.arange(n, dtype=np.float64)
    c1 = n * 0.2
    c2 = n * 0.7
    w = n * 0.12
    a = np.exp(-0.5 * ((t - c1) / w) ** 2) + 0.7 * np.exp(-0.5 * ((t - c2) / w) ** 2)
    return a


def initial_trapezoid(n, asym=0.3):
    """Trapezoidal window with asymmetry."""
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    rise_end = 0.2 + asym * 0.1

    a[int(fall_start * n):] = np.linspace(1.0, 0.3, n - int(fall_start * n))
    return np.maximum(0.01, a)


def soft_max_weights(conv, T):
    """Compute numerically stable soft-max weights."""
    conv_shifted = conv - conv.max()
    exponents = conv_shifted / T
    max_exp = exponents.max()
    w = np.exp(exponents - max_exp)
    s = w.sum()
    if s < 1e-300:
        idx = np.argmax(conv)
        w = np.zeros_like(conv)
        w[idx] = 1.0
        return w
    return w / s


def compute_grad(a, w):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    grad[m] = 2 * sum_j a[j] * w[j+m]
    """
    n = len(a)
    if n <= 5000:
        idx = np.arange(n)
        M = w[idx[None, :] + idx[:, None]]
        grad = 2.0 * (M @ a)
    else:
        corr = np.correlate(w, a, 'full')
        grad = 2.0 * corr[(n - 1):(2 * n - 1)]
    return grad

        peak = conv.max()
        T = max(T_frac * peak, 1e-12)

    it = 0

            break

        conv = np.convolve(a, a)
        M = conv.max()
        score = score_seq(a)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 25:
            break

        # Build LP: max sum(b) s.t. A@b <= M, b >= 0
        # where A is the convolution operator: (A@b)[k] = sum_j a[k-j]*b[j]
        timeout = max(5.0, min(90.0, remaining * 0.4))

        try:
            n_rows = 2 * n - 1
            col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
            row_parts = []
            for j in range(n):
                lo = max(0, j)
                hi = min(n, n_rows - j)
                row_parts.append(np.arange(lo, hi, dtype=np.int32))
            row_idx = np.concatenate(row_parts)
            data_parts = []
            for j in range(n):
                lo = max(0, j)
                hi = min(n, n_rows - j)
                data_parts.append(a[lo:hi])
            data = np.concatenate(data_parts)

            A_mat = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))

            c_vec = -np.ones(n)
            b_ub = np.full(n_rows, M)
            bounds = [(0.0, None)] * n

            res = linprog(c_vec, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': timeout})

            if res.success and res.x is not None:
                g = np.maximum(0.0, res.x)
                sum_g = g.sum()
                if sum_g <= 1.0 + 1e-8:
                    if verbose:
                        print(f"    LP converged at it {it}, sum_g={sum_g:.8f}")
                    break
                g = g / sum_g
                alpha = 0.25
                a_new = (1 - alpha) * a + alpha * g
                a = normalize(a_new)
            else:
                break
        except Exception as e:
            if verbose:
                print(f"    LP error: {e}")
            break

        it += 1

    final_score = score_seq(best_a)
    if final_score < best_score:
        best_score = final_score

    if verbose:
        print(f"    LP done: best={best_score:.6f}, iters={it}, time={time.monotonic()-start_time:.1f}s")

    return best_a, best_score


def local_search_refine(a_init, time_limit, rng, verbose=True):
    """Random perturbation local search for final polish."""
    start_time = time.monotonic()
    a = normalize(a_init.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)

    if verbose:
        print(f"    LS start: score={best_score:.6f}")

    accept = 0
    while time.monotonic() - start_time < time_limit:
        # Multi-scale perturbation
        scale = rng.choice(['fine', 'medium', 'coarse'])
        if scale == 'fine':
            width = max(1, int(rng.integers(1, max(2, n // 50))))
            sigma = 0.05
        elif scale == 'medium':
            width = max(2, int(rng.integers(n // 50, max(3, n // 10))))
            sigma = 0.15
        else:
            width = max(3, int(rng.integers(n // 10, max(4, n // 3))))
            sigma = 0.3

        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, sigma, stop - start))

        # Occasionally try mirror or shift
        r = rng.random()
        if r < 0.1:
            candidate = 0.5 * (candidate + candidate[::-1])
        elif r < 0.15:
            shift = int(rng.integers(-n // 10, n // 10 + 1))
            candidate = np.roll(candidate, shift)
            candidate = np.maximum(0, candidate)

        candidate = normalize(candidate)
        s = score_seq(candidate)
        if s < best_score:
            best_score = s
            best_a = candidate.copy()
            a = candidate
            accept += 1
            if verbose and accept % 200 == 0:
                print(f"    LS accept {accept}: score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, accepts={accept}")

    return best_a, best_score


def get_initializations(n, rng):
    """Generate diverse initializations for a given length."""
    inits = []
    inits.append(('rc', initial_raised_cosine(n)))
    inits.append(('asym10', initial_asymmetric(n, 0.1)))
    inits.append(('asym20', initial_asymmetric(n, 0.2)))
    inits.append(('asym30', initial_asymmetric(n, 0.3)))
    inits.append(('ramp2', initial_ramp(n, 2.0)))
    inits.append(('ramp1', initial_ramp(n, 1.0)))
    inits.append(('dual', initial_dual_bump(n)))
    inits.append(('trap', initial_trapezoid(n, 0.3)))

    # Random bump inits
    for k in range(3):
        a = np.zeros(n)
        n_bumps = rng.integers(2, 5)
        for _ in range(n_bumps):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.25)
            amp = rng.uniform(0.5, 1.0)
            t = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
        inits.append((f'rand{k}', np.maximum(0.01, a)))

    return inits


def propose_candidate(budget_s=2400):
    """
    Main search: minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    Uses soft-max gradient descent, LP refinement, and local search with
    multiple asymmetric initializations and adaptive time allocation.
    """
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30  # safety margin

    # Try to get previous best
    prev_seq = None
    try:
        prev_seq = np.array(height_sequence_1, dtype=np.float64)
    except (NameError, UnboundLocalError, TypeError):
        pass

    best_score = float('inf')
    best_seq = None
    rng = np.random.default_rng(2024)

    # If we have a previous sequence, resample it to different lengths
    def resample_to(prev, n):
        if len(prev) == n:
            return prev.copy()
        from scipy.interpolate import interp1d
        old_t = np.linspace(0, 1, len(prev))
        new_t = np.linspace(0, 1, n)
        f = interp1d(old_t, prev, kind='cubic', fill_value=0, bounds_error=False)
        return np.maximum(0, f(new_t))

    # Build config list
    configs = []
    for n in [500, 800, 1000, 1500, 2000]:
        for name, init in get_initializations(n, rng):
            configs.append((n, name, init))
        if prev_seq is not None:
            configs.append((n, 'prev', resample_to(prev_seq, n)))

    # Prioritize: previous best first, then shorter sequences, then diverse inits
    def config_priority(cfg):
        n, name, _ = cfg
        if name == 'prev':
            return (0, n)
        if name in ('asym10', 'asym20', 'asym30', 'ramp1', 'ramp2'):
            return (1, n)
        return (2, n)

    configs.sort(key=config_priority)

    # Time allocation strategy
    processed = set()
    for n, name, init in configs:
        remaining = deadline - time.monotonic()
        if remaining < 25.0:
            break

        # Skip if we already processed this (n, init_type)
        key = (n, name)
        if key in processed:
            continue
        processed.add(key)

        # Allocate time
        if n <= 500:
            gd_t = min(100.0, remaining * 0.35)
            lp_t = min(80.0, remaining * 0.25)
            ls_t = min(30.0, remaining * 0.1)
        elif n <= 1000:
            gd_t = min(120.0, remaining * 0.35)
            lp_t = min(60.0, remaining * 0.2)
            ls_t = min(30.0, remaining * 0.1)
        else:
            gd_t = min(150.0, remaining * 0.4)
            lp_t = 0.0  # LP too slow for large n
            ls_t = min(30.0, remaining * 0.1)

        # Scale down if too much
        total_alloc = gd_t + lp_t + ls_t
        if total_alloc > remaining * 0.85:
            scale = remaining * 0.85 / total_alloc
            gd_t *= scale
            lp_t *= scale
            ls_t *= scale

        if gd_t < 10 and lp_t < 5:
            continue

        if remaining > 60:
            print(f"\n--- n={n}, init={name} | gd={gd_t:.0f}s lp={lp_t:.0f}s ls={ls_t:.0f}s | "
                  f"remaining={remaining:.0f}s ---")

        a = normalize(init)
        cur_score = score_seq(a)

        # Phase 1: Gradient descent
        if gd_t > 10:
            a, cur_score = gradient_descent_refine(
                a, gd_t, verbose=(remaining > 120),
                seed=hash(key) % 100000,
                T_frac_start=0.1, T_frac_end=0.003
            )

        # Phase 2: LP refinement
        if lp_t > 15 and cur_score < best_score + 0.1:
            a, cur_score = lp_refine(a, lp_t, verbose=(remaining > 60))

        # Phase 3: Local search
        if ls_t > 5:
            a, cur_score = local_search_refine(a, ls_t, rng, verbose=False)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"  *** NEW BEST: {best_score:.6f} (n={n}, {name}) "
                  f"[t={time.monotonic()-total_start:.0f}s]")

    # Final polishing phase
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n=== FINAL POLISH (remaining={remaining:.0f}s) ===")

            # Local search
            if remaining > 60:
                t_ls = min(remaining * 0.25, 120)
                a_p, s_p = local_search_refine(best_seq, t_ls, rng, verbose=True)
                if s_p < best_score:
                    best_score = s_p
                    best_seq = a_p

            # LP on best
            remaining2 = deadline - time.monotonic()
            if remaining2 > 30 and len(best_seq) <= 1500:
                t_lp = min(remaining2 * 0.35, 120)
                a_p2, s_p2 = lp_refine(best_seq, t_lp, verbose=True)
                if s_p2 < best_score:
                    best_score = s_p2
                    best_seq = a_p2

            # Another GD round
            remaining3 = deadline - time.monotonic()
            if remaining3 > 40:
                t_gd = min(remaining3 * 0.5, 150)
                a_p3, s_p3 = gradient_descent_refine(
                    best_seq, t_gd, verbose=True, seed=99999,
                    T_frac_start=0.05, T_frac_end=0.001
                )
                if s_p3 < best_score:
                    best_score = s_p3
                    best_seq = a_p3

            # Final local search
            remaining4 = deadline - time.monotonic()
            if remaining4 > 20:
                a_p4, s_p4 = local_search_refine(
                    best_seq, min(remaining4 * 0.6, 60), rng, verbose=False
                )
                if s_p4 < best_score:
                    best_score = s_p4
                    best_seq = a_p4

    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_seq(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Convert to list of floats for return
    result = [float(x) for x in best_seq]

    # Verify
    verify = evaluate_sequence(result)
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.6f}, verified={verify:.6f}, n={len(result)}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s / {budget_s}s")
    print(f"{'='*60}")

    return result
