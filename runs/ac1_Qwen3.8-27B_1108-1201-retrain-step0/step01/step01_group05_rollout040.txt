Minimize 2*n*max(self-convolution)/sum^2 over non-negative sequences.

Key insight: Monotone power-law ramps a[i] = (i+1)^p give much better results
than symmetric/random sequences because anti-correlation at the center lag
reduces the peak. Asymptotic objective for integer p: 2*((p+1)!)^2/(2p+1)!.
We sweep exponents, pick the best, then refine with subgradient optimization.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_objective_fast(a):
    """Compute 2*n*max_b/sum^2 using FFT."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def make_power_ramp(n, p, max_val=1000.0):
    """Create a[i] = scale * (i+1)^p with max value = max_val."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    scale = max_val / a[-1]
    return a * scale


def make_truncated_ramp(n, p, max_val=1000.0, trunc_frac=0.1):
    """Power ramp with the top truncated for a flatter peak."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    # Truncate: cap at some fraction of max
    cap = max_val * trunc_frac
    a = np.minimum(a, cap)
    # Ensure the top values are at cap
    a[a > cap] = cap
    scale = 1.0  # already bounded by cap
    # Actually just scale so max is max_val
    a = a / a.max() * max_val
    return a


def compute_subgradient(a, top_indices, c, k_weight):
    """Weighted subgradient using top-k active lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total <= 1e-30:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # dc[k]/da[m] = 2*a[k-m] for valid m
            # a[k-m] for m in [lo, hi] means k-m in [k-hi, k-lo]
            seg = a[k - hi : k - lo + 1][::-1]
            grad[lo:hi + 1] += w_val * 2.0 * seg
    return grad

    b = a / S
    best_b = b.copy()
    c = fft_convolve_self(b)
    best_score = c.max()
    start = time.monotonic()
    lr = lr_init if lr_init is not None else 0.1 / n

    best_a = a.copy()

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(b)
        cur = c.max()
        if cur < best_score:

            best_b = b.copy()

        grad = compute_subgradient(b, top_indices, c, 1.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        grad = grad / gnorm

        b = np.maximum(b, 0.0)

        for run_i in range(n_runs):
            if time.monotonic() > deadline - 5.0:
                break
            a_cur, score = subgradient_refine(
                best_seq, n, per_run, top_k=20,
                lr_init=0.5 / max(1, n // 100), lr_decay=0.9995
            )
            a_cur = np.clip(a_cur, 0.0, 1000.0)
            obj = compute_objective_fast(a_cur)
            if obj < best_overall:
                best_overall = obj
                best_seq = a_cur.copy()
                print(f"  refinement run {run_i}: obj={obj:.6f}")

    # Phase 3: Also try more aggressive shapes
    time_left = deadline - time.monotonic()
    if time_left > 10.0 and best_seq is not None:
        print(f"Phase 3: Trying shaped variants (budget={time_left:.1f}s)")
        n = len(best_seq)

        # Try: power ramp with soft cap
        for cap_frac in [0.3, 0.5, 0.7]:
            i = np.arange(1, n + 1, dtype=np.float64)
            for p_try in [2.0, 2.5, 3.0]:
                a_raw = i ** p_try
                cap_val = a_raw.max() * cap_frac
                a_raw = np.minimum(a_raw, cap_val)
                # Smooth transition: blend near the cap
                # Actually just use the truncated version
                a_try = a_raw / a_raw.max() * 1000.0
                obj = compute_objective_fast(a_try)
                if obj < best_overall:
                    best_overall = obj
                    best_seq = a_try.copy()
                    print(f"  cap={cap_frac}, p={p_try}: obj={obj:.6f}")

        # Try: two-sided ramp (increase then decrease) - usually worse but check
        for mid_frac in [0.5, 0.6, 0.7]:
            mid = int(n * mid_frac)
            a_try = np.zeros(n)
            a_try[:mid] = np.arange(1, mid + 1) ** 2
            tail = n - mid
            a_try[mid:] = (np.arange(tail, 0, -1)) ** 2
            a_try = a_try / a_try.max() * 1000.0
            obj = compute_objective_fast(a_try)
            if obj < best_overall:
                best_overall = obj
                best_seq = a_try.copy()
                print(f"  two-sided mid={mid_frac}: obj={obj:.6f}")

    # Phase 4: Final verification and return
    if best_seq is None:
        best_seq = make_power_ramp(8192, 2.0)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = make_power_ramp(8192, 2.0)

    # Final authoritative check
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final score: {final_score:.6f} (n={len(best_seq)})")

    if not np.isfinite(final_score):
        # Fallback
        best_seq = make_power_ramp(8192, 2.0)
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Fallback score: {final_score:.6f}")

    return [float(x) for x in best_seq]
