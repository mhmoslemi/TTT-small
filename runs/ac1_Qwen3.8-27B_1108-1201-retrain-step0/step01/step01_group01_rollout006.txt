```python
"""
Multi-start gradient optimization with momentum, adaptive soft-max gradient,
backtracking line search, and periodic diversification to minimize the
normalized autocorrelation peak of a non-negative sequence.
"""

import numpy as np
import time


def _score_normalized(a):
    """Score for a sequence normalized to sum=1: 2*n*max(conv(a,a))."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _full_score(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (S * S)


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _soft_max_grad(a, b, task=30, temperature=None):
    """
    Compute soft-max of top-task b values and gradient w.r.t. a.
    Returns (soft_max_value, gradient).
    """
    n = len(a)
    if task >= len(b):
        k_idx = np.arange(len(b))
    else:
        k_idx = np.argpartition(b, -task)[-task:]
    
    b_sel = b[k_idx]
    b_max_sel = np.max(b_sel)
    
    if temperature is None:
        # Adaptive temperature
        b_sorted = np.sort(b_sel)[::-1]
        if len(b_sorted) > 1 and b_sorted[0] > b_sorted[-1]:
            temperature = (b_sorted[0] - b_sorted[-1]) / 3.0
        else:
            temperature = max(b_sorted[0], 1e-10) * 0.1
    
    exp_vals = np.exp((b_sel - b_max_sel) / temperature)
    sum_exp = np.sum(exp_vals)
    wts = exp_vals / sum_exp
    
    soft_val = b_max_sel + temperature * np.log(sum_exp)
    
    # Gradient: sum_k w_k * 2*a[k-i] for valid i
    grad = np.zeros(n)
    for idx in range(len(k_idx)):
        w = wts[idx]
        if w < 1e-10:
            continue
        k = k_idx[idx]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += w * 2.0 * a[k - i_range]
    
    return soft_val, grad


def _optimize_momentum(a_init, deadline, rng, init_lr=None, momentum=0.9):
    """
    Momentum gradient descent with backtracking line search on the simplex.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = _score_normalized(best_a)
    
    if init_lr is None:
        lr = 1.0 / np.sqrt(n)
    
    v = np.zeros(n)  # momentum
    task = 30
    patience = 0
    max_patience = 500
    
    start_time = time.monotonic()
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        cur_score = 2.0 * n * np.max(b)
        
        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            break
        
        # Compute gradient
        soft_val, grad = _soft_max_grad(a, b, task=task)
        
        # Center gradient (for sum constraint)
        grad = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v - lr * grad

def _coordinate_descent_polish(a, deadline, rng, n_iter=10000):
    """
    Coordinate descent: try swapping mass between pairs of coordinates.
    """
    n = len(a)
    a = a.copy()
    score = _score_normalized(a)
    
    # Pre-compute some indices for efficiency
    # For each pair (i, j), the change in b_k when we move mass from i to j:
    # Δb_k = δ*(a[k-j] - a[k-i]) for the relevant k
    
    # Strategy: pick the current max peak k*, then find (i,j) that best reduces b_{k*}
    # while not increasing other peaks too much
    
    best_a = a.copy()
    best_score = score
    
    elapsed_start = time.monotonic()
    
    for iteration in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        b = np.convolve(a, a)
        k_max = np.argmax(b)
        
        # Gradient of b_{k_max}: grad[i] = 2*a[k_max - i] for valid i
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        grad = np.zeros(n)
        if lo <= hi:
            i_range = np.arange(lo, hi + 1)
            grad[i_range] = 2.0 * a[k_max - i_range]
        
        # Find best pair (i, j): decrease a_i and increase a_j
        # to reduce b_{k_max}: want grad[j] < grad[i] (move mass from high grad to low grad)
        # But also need to not increase other peaks too much
        
        # Simple heuristic: find i with highest grad (among coordinates with a_i > 0)
        # and j with lowest grad
        i_best = np.argmax(grad * (a > 1e-10))
        j_best = np.argmin(grad)
        
        if i_best == j_best:
            continue
        
        # How much can we move?
        max_move = a[i_best]
        
        # Try a few step sizes
        for frac in [0.1, 0.05, 0.01]:
            delta = frac * max_move
            if delta <= 0:
                continue
            a_trial = a.copy()
            a_trial[i_best] -= delta
            a_trial[j_best] += delta
            a_trial = _project_simplex(a_trial, 1.0)
            b_trial = np.convolve(a_trial, a_trial)
            score_trial = 2.0 * n * np.max(b_trial)
            if score_trial < score - 1e-12:
                a = a_trial
                score = score_trial
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                break
    
    return best_a, best_score


def _make_anti_symmetric(n, profile='linear'):
    """Create an anti-symmetric starting point."""
    i = np.arange(n, dtype=np.float64)
    if profile == 'linear':
        a = 1.0 + 0.5 * (1 - 2*i/(n-1))  # linear ramp from 1.5 to 0.5
    elif profile == 'cosine':
        a = 1.0 + 0.5 * np.cos(np.pi * i / (n-1))
    elif profile == 'quadratic':
        t = i / (n-1)
        a = 1.0 + 0.5 * (2*t - 1)**2  # U-shaped
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)
```
