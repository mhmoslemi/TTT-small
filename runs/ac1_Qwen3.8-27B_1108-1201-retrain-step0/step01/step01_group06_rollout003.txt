"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive mirror descent
with periodic restarts, multi-length exploration, and coordinate descent
polishing. Starts from previous best when available.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def get_subgradient(conv, a, n, t):
    """Compute the smoothed subgradient of max(conv) w.r.t. p."""
    max_conv = conv.max()
    if max_conv < 1e-15:
        return np.zeros(n)
    scaled = t * conv / max_conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-12:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    sg = np.zeros(n)
    thresh = w.max() * 0.001
    active = np.where(w > thresh)[0]
    
    for k_val in active:
        k = int(k_val)
        # conv[k] = sum_j a[j] * a[k-j] where 0<=j<n, 0<=k-j<n
        # j ranges from max(0, k-n+1) to min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            # d(conv[k])/d(a[j]) = a[k-j] for j in [j_lo, j_hi]
            # so sg[j] += w[k] * a[k-j]
            indices = np.arange(j_lo, j_hi + 1)
            sg[indices] += w[k] * a[k - indices]
    
    # Also, by symmetry, d(conv[k])/d(a[j]) = a[k-j] covers it
    # But wait, conv[k] = sum_j a[j]*a[k-j], so d/d(a[j]) = a[k-j]
    # This accounts for both "sides" since when we vary a[j], it appears
    # in conv[k] as both a[j]*a[k-j] AND a[k-j]*a[j] (same term)
    # So we only need to add once.
    
    return sg


def mirror_descent_step(a, sg, eta):
    """Multiplicative update: a -> a * exp(-eta * sg_centered), renormalize."""
    sg_c = sg - sg.mean()
    log_update = -eta * sg_c
    log_update -= log_update.max()  # numerical stability
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones_like(a)
        s = len(a)
    a_new /= s
    return a_new


def coordinate_polish(a, max_iters=200):
    """Coordinate descent: try shifting mass from peak-contributing to non-peak positions."""
    n = len(a)
    best = a.copy()
    best_obj = 2.0 * n * fast_convolve(best).max()
    
    for iteration in range(max_iters):
        conv = fast_convolve(best)
        peak = conv.max()
        if peak < 1e-15:
            break
        
        # Find the peak location
        peak_idx = int(np.argmax(conv))
        
        # For each position j, compute how much increasing it would hurt the peak
        # d(conv[peak_idx])/d(a[j]) = a[peak_idx - j] if valid
        j_lo = max(0, peak_idx - n + 1)
        j_hi = min(n - 1, peak_idx)
        
        damage = np.zeros(n)
        if j_lo <= j_hi:
            for j in range(j_lo, j_hi + 1):
                damage[j] = best[peak_idx - j]
        
        # Move mass from high-damage to low-damage positions
        sorted_idx = np.argsort(damage)
        src = sorted_idx[-n//10:]  # top 10% most damaging
        dst = sorted_idx[:n//10]   # bottom 10% least damaging
        
        # Shift a small amount
        total_src = best[src].sum()
        if total_src < 1e-15:
            break
        frac = 0.05  # move 5% of source mass
        amount = total_src * frac
        amount_per = amount / len(src)
        
        candidate = best.copy()
        candidate[src] -= amount_per
        np.maximum(candidate, 0, out=candidate)
        candidate[dst] += amount_per / len(dst)
        s = candidate.sum()
        candidate /= s
        
        new_conv = fast_convolve(candidate)
        new_peak = new_conv.max()
        
        if new_peak < peak - 1e-12:
            best = candidate
            best_obj = new_peak
    
    return best


def optimize_one(a0, time_limit, t_start=20.0, t_end=200.0, eta0=0.5, restart_patience=60):
    """
    Run adaptive mirror descent on a single sequence.
    Returns (best_a, best_obj_normalized).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    eta = eta0
    patience = 0
    t0 = time.monotonic()
    elapsed = 0.0
    total_time = time_limit
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / total_time)
        
        # Temperature schedule: increase over time
        current_t = t_start + (t_end - t_start) * progress
        
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Compute subgradient
        sg = get_subgradient(conv, a, n, current_t)
        
        # Mirror descent step
        a = mirror_descent_step(a, sg, eta)
        
        # Evaluate
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= restart_patience:
            # Restart with perturbation
            noise = np.random.default_rng().dirichlet(np.ones(n) * 0.3)
            mix = min(0.3, patience / (2.0 * restart_patience))
            a = (1 - mix) * best_a + mix * noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-12:
                a /= s
            conv = fast_convolve(a)
            patience = 0
            eta = min(eta0, eta * 1.5)  # reset to be more aggressive
    
    # Final coordinate polish
    if time.monotonic() - t0 < time_limit - 2:
        polished = coordinate_polish(best_a, max_iters=50)
        pol_obj = 2.0 * len(polished) * fast_convolve(polished).max()
        if pol_obj < best_obj:
            best_a = polished
            best_obj = pol_obj
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Multi-length, multi-start adaptive mirror descent

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive mirror descent
with projected-gradient hybrid, periodic restarts, multi-length search,
and coordinate descent polishing. Employs Aggressive Temperature Annealing
and Zero-Revival to escape local minima.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_normalized_obj(a):
    """Compute 2n * max(conv) for sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def get_smoothed_subgradient(conv, a, n, t):
    """Smoothed subgradient via log-sum-exp weighting."""
    max_conv = conv.max()
    if max_conv < 1e-15:
        return np.zeros(n)
    scaled = t * conv / max_conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-12:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    thresh = w.max() * 0.0005
    active = np.where(w > thresh)[0]

    for ki in active:
        k = int(ki)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            sg[idx] += w[ki] * a[k - idx]

    return sg


def mirror_update(a, sg, eta):
    """Multiplicative (mirror) descent update with centering."""
    sg_c = sg - sg.mean()
    log_f = -eta * sg_c
    log_f -= log_f.max()
    a_new = a * np.exp(log_f)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones_like(a)
    else:
        a_new /= s
    return a_new


def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    tau = (cssv[rho[-1]] - 1) / (rho[-1] + 1.0)
    return np.maximum(v - tau, 0.0)


def projected_grad_step(a, sg, eta):
    """Projected gradient descent step."""
    q = a - eta * sg
    return project_simplex(q)


def coordinate_polish(a, max_steps=100):
    """Coordinate descent: shift mass away from peak contributors."""
    n = len(a)
    best = a.copy() / a.sum()
    best_peak = fast_convolve(best).max()

    for _ in range(max_steps):
        conv = fast_convolve(best)
        peak = conv.max()
        if peak < 1e-15:
            break
        peak_idx = int(np.argmax(conv))

        j_lo = max(0, peak_idx - n + 1)
        j_hi = min(n - 1, peak_idx)
        damage = np.zeros(n)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            damage[idx] = best[peak_idx - idx]

        # Also consider neighboring peaks
        sorted_peaks = np.argsort(conv)[::-1][:5]
        for pk in sorted_peaks[1:]:
            pk = int(pk)
            j_lo2 = max(0, pk - n + 1)
            j_hi2 = min(n - 1, pk)
            if j_lo2 <= j_hi2:
                idx2 = np.arange(j_lo2, j_hi2 + 1)
                damage[idx2] += conv[pk] / peak * best[pk - idx2]

        frac = max(3, n // 32)
        sorted_idx = np.argsort(damage)
        dst = sorted_idx[:frac]
        src = sorted_idx[-frac:]

        total_src = best[src].sum()
        if total_src < 1e-15:
            break
        amt = total_src * 0.03 / len(src)

        candidate = best.copy()
        candidate[src] -= amt
        np.maximum(candidate, 0, out=candidate)
        candidate[dst] += amt
        candidate /= candidate.sum()

        new_peak = fast_convolve(candidate).max()
        if new_peak < best_peak - 1e-13:
            best = candidate
            best_peak = new_peak

    return best

                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan n={n:3d} {name:8s} score={score:.6f}")

    # Phase 2: Deep optimization (50% of budget)
    print(f"Phase 2: Deep opt. Best so far: {best_score:.6f}")
    scan_results.sort(key=lambda x: x[0])

    top = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top) < 12:
            top.append((n, name, a_opt))
            seen.add(key)

    if best_sequence is not None:
        top.append((len(best_sequence), 'BEST', best_sequence.copy()))

    remaining2 = deadline - time.monotonic() - 25
    t_per_deep = max(10.0, remaining2 / max(1, len(top)))

    for i, (n, name, a0) in enumerate(top):
        if time.monotonic() > deadline - 25:
            break
        t_total = min(t_per_deep * 2.5, deadline - time.monotonic() - 25)
        if t_total < 5:
            break

        # Try several (t_init, t_max, eta, hybrid) combos
        params_list = [
            (10, 100, 0.6, False),
            (30, 250, 0.4, True),
            (50, 500, 0.3, False),
            (80, 600, 0.2, True),
        ]
        t_single = t_total / len(params_list)

        for t_i, t_m, eta_i, hybrid in params_list:
            if time.monotonic() > deadline - 25:
                break
            if t_single < 3:
                continue
            a_opt, _ = optimize_sequence(a0, time_limit=t_single,
                                         t_init=t_i, t_max=t_m,
                                         eta_init=eta_i, use_hybrid=hybrid)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep n={n:3d} {name:8s} t={t_i:3d}-{t_m:3d} score={score:.6f}")

    # Phase 3: Final concentrated polish (remaining budget)
    if best_sequence is not None:
        remaining3 = deadline - time.monotonic() - 10
        if remaining3 > 15:
            print(f"Phase 3: Final polish, {remaining3:.0f}s remaining")
            n_best = len(best_sequence)

            polish_params = [
                (100, 800, 0.15, False),
                (200, 1000, 0.1, True),
                (300, 1200, 0.08, False),
                (50, 400, 0.25, True),
            ]
            t_single = remaining3 / len(polish_params)

            for t_i, t_m, eta_i, hybrid in polish_params:
                if time.monotonic() > deadline - 10:
                    break
                if t_single < 3:
                    break
                a_opt, _ = optimize_sequence(best_sequence, time_limit=t_single,
                                             t_init=t_i, t_max=t_m,
                                             eta_init=eta_i, use_hybrid=hybrid)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Pol  n={n_best:3d} t={t_i:3d}-{t_m:3d} score={score:.6f}")
                t_single = (deadline - time.monotonic() - 10) / 2.0

            # One last coordinate polish
            if time.monotonic() < deadline - 5:
                polished = coordinate_polish(best_sequence, max_steps=50)
                score = float(evaluate_sequence(polished.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = polished
                    print(f"  CoordPolish score={score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL: score={best_score:.6f} n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
