import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = mu - (cv - S) / rho > 0
    if not np.any(cond):
        return np.full(n, S / n, dtype=np.float64)
    rho_val = int(np.max(np.where(cond)[0]) + 1)
    theta = (cv[rho_val - 1] - S) / rho_val
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_for_lag(a, k):
    """Gradient of b[k] = sum_i a[i]*a[k-i] with respect to a.
    Result: grad[m] = 2*a[k-m] if 0 <= k-m < n, else 0."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        # For m in [lo, hi]: grad[m] = 2*a[k-m]
        # k-m ranges from k-hi to k-lo (decreasing as m increases)
        grad[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad

    """Weighted sum of gradients for specified lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    for w_val, k in zip(weights, top_indices):
        if w_val < 1e-15:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def softmax_weights(c, temperature):
    """Compute softmax weights over the convolution values."""
    c_max = c.max()
    exp_c = np.exp((c - c_max) / temperature)
    total = exp_c.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_c / total


def run_adam_softmax(a_init, n, S, max_time, rng, lr_init, topo_k_start, temp_start, temp_end):
    """Adam with softmax-smoothed subgradient and adaptive temperature."""
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n, dtype=np.float64)
        s = n
    a = a * (S / s)

    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8

    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    no_improve_count = 0

    while time.monotonic() - start < max_time:
        t += 1

        # Adaptive temperature: anneal from temp_start to temp_end
        progress = min(1.0, t / max(1, int(max_time * 10)))
        temp = temp_start * (temp_end / temp_start) ** progress

        # Adaptive top-k: grow from small to large
        top_k = max(topo_k_start, int(topo_k_start + (50 - topo_k_start) * progress))

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Compute softmax weights
        sw = softmax_weights(c, temp)
        # Focus on top-k for efficiency (zero out small weights)
        top_indices = np.argsort(sw)[::-1][:top_k]
        top_sw = sw[top_indices]
        top_sw = top_sw / top_sw.sum()

        grad = compute_subgradient_topk(a, c, top_indices, top_sw)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm

        # Adam update
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad ** 2)
        m_hat = m_buf / (1.0 - beta1 ** t)
        v_hat = v_buf / (1.0 - beta2 ** t)

        lr = lr_init * (0.7 ** (float(t) / 300.0))
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

        # Periodic perturbation to escape local minima
        if no_improve_count > 500 and rng.random() < 0.05:
            a = project_simplex(a + rng.normal(0.0, 0.02 * S / np.sqrt(n), n), S)
            no_improve_count = 0

    return best_local, best_score


def run_mirror_softmax(a_init, n, S, max_time, rng, eta_init, topo_k_start, temp_start, temp_end):
    """Mirror descent (multiplicative weights) with softmax subgradient."""
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n, dtype=np.float64)
    else:
        a = a * (S / s)

    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    t = 0

    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, t / max(1, int(max_time * 10)))
        temp = temp_start * (temp_end / temp_start) ** progress
        top_k = max(topo_k_start, int(topo_k_start + (30 - topo_k_start) * progress))

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()

        sw = softmax_weights(c, temp)
        top_indices = np.argsort(sw)[::-1][:top_k]
        top_sw = sw[top_indices]
        top_sw = top_sw / top_sw.sum()

        grad = compute_subgradient_topk(a, c, top_indices, top_sw)
        gmax = np.abs(grad).max()
        if gmax < 1e-12:
            break
        grad = grad / gmax

        eta = eta_init * max(0.05, 1.0 / (1.0 + 0.003 * t))
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a * (S / s)

    return best_local, best_score


def mass_redistribution_polish(a, n, S, max_time, rng):
    """Local search: transfer mass between positions to reduce max convolution."""
    a = np.clip(a.copy(), 0.0, 1000.0)
    s = a.sum()
    a = a * (S / s)

    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    step_size = S / (100 * n)
    no_improve = 0

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                step_size *= 0.8
                no_improve = 0

        # Find the lag with maximum value
        k_max = int(np.argmax(c))

        # Find best (i, j) pair to transfer mass
        # We want to decrease b[k_max]: decrease a[i] where a[k_max - i] is large,
        # or increase a[j] where it helps keep other lags down
        # Simple heuristic: find i that maximizes contribution to b[k_max]
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo > hi:
            continue

        # Contribution of a[m] to b[k_max] is proportional to a[k_max - m]
        # We want to move mass out of positions where this is high
        contrib = np.zeros(n)
        contrib[lo:hi + 1] = a[lo:hi + 1] * a[k_max - hi:k_max - lo + 1][::-1]

        # Source: position with highest contribution
        i = int(np.argmax(contrib))
        # Destination: position that minimizes increase in max
        # Approximate: position with lowest a value (to keep balance)
        candidates = np.where(a < np.median(a))[0]
        if len(candidates) == 0:
            candidates = [0]
        j = int(rng.choice(candidates))

        if i == j:
            continue

        transfer = min(step_size, a[i] * 0.1)
        if transfer < 1e-15:
            continue

        a_trial = a.copy()
        a_trial[i] -= transfer
        a_trial[j] += transfer
        if a_trial[i] < 0:
            continue

        c_trial = fft_convolve_self(a_trial)
        if c_trial.max() < cur:
            a = a_trial

    return best_local, best_score


def coordinate_descent_polish(a, n, S, max_time, rng):
    """Coordinate descent: adjust pairs of elements to reduce max convolution."""
    a = np.clip(a.copy(), 0.0, 1000.0)
    s = a.sum()
    a = a * (S / s)

    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    no_improve = 0
    step_frac = 0.05

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                step_frac *= 0.9
                no_improve = 0
                if step_frac < 0.001:
                    break

        # Pick random pair
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        amount = step_frac * max(a[i], a[j])
        if amount < 1e-15:
            continue

        # Try transferring from i to j
        a_trial = a.copy()
        a_trial[i] -= amount
        a_trial[j] += amount
        if a_trial[i] < 0:
            # Try reverse
            a_trial = a.copy()
            a_trial[j] -= amount
            a_trial[i] += amount
            if a_trial[j] < 0:
                continue

        c_trial = fft_convolve_self(a_trial)
        if c_trial.max() < cur:
            a = a_trial
        else:
            # Try smaller step
            amount2 = amount * 0.3
            a_trial = a.copy()
            a_trial[i] -= amount2
            a_trial[j] += amount2
            if a_trial[i] >= 0:
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur:
                    a = a_trial

    return best_local, best_score


def make_chirp_init(n, rng, rate=1.0):
    """Chirp-inspired initialization: 1 + eps*cos(2*pi*rate*k^2/n)."""
    k = np.arange(n, dtype=np.float64)
    eps = 0.3 + 0.2 * rng.random()
    a = 1.0 + eps * np.cos(2.0 * np.pi * rate * k * k / n)
    return np.maximum(a, 0.1)


def make_smooth_init(n, rng):
    """Smooth multi-frequency initialization."""
    k = np.arange(n, dtype=np.float64) / n
    a = (1.0
         + 0.2 * rng.random() * np.sin(2 * np.pi * int(rng.integers(2, 6)) * k)
         + 0.1 * rng.random() * np.cos(2 * np.pi * int(rng.integers(5, 12)) * k)
         + 0.05 * np.sin(2 * np.pi * int(rng.integers(10, 20)) * k))
    return np.maximum(a, 0.05)


def make_banded_init(n, rng):
    """Banded structure: alternating high/low with smooth envelope."""
    k = np.arange(n, dtype=np.float64)
    period = int(rng.integers(4, 16))
    envelope = 1.0 + 0.3 * np.sin(2 * np.pi * k / (n * 0.3))
    band = 1.0 + 0.25 * np.cos(2 * np.pi * k / period)
    return np.maximum(envelope * band + 0.1 * rng.standard_normal(n) * 0.1, 0.05)


def normalize_seq(a, n, S):
    """Clamp and normalize to sum S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(n, S / n, dtype=np.float64)
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 10.0)

    global_best = np.inf
    global_a = None

    # Try multiple lengths
    lengths = [8192, 16384]

    # Collect initializations
    all_inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 16 and np.sum(np.clip(given, 0, 1000)) > 0:
            for n in lengths:
                if given.size == n:
                    all_inits.append(("given", n, np.clip(given, 0, 1000)))
                else:
                    # Resample to desired length
                    idx = (np.arange(n) * given.size // n).astype(int)
                    all_inits.append(("given_resample", n, np.clip(given[idx], 0, 1000)))
    except Exception:
        pass

    for n in lengths:
        S = np.sqrt(2.0 * n)
        all_inits.append(("chirp", n, make_chirp_init(n, rng)))
        all_inits.append(("chirp2", n, make_chirp_init(n, rng, rate=2.0)))
        all_inits.append(("smooth", n, make_smooth_init(n, rng)))
        all_inits.append(("banded", n, make_banded_init(n, rng)))
        all_inits.append(("uniform", n, np.ones(n) * (1.0 + 0.1 * rng.standard_normal(n))))
        all_inits.append(("random", n, rng.random(n) + 0.1))

    n_inits = len(all_inits)
    usable = max(1.0, deadline - time.monotonic())
    explore_budget = 0.65 * usable
    polish_budget = 0.30 * usable
    verify_budget = usable - explore_budget - polish_budget

    per_init = explore_budget / max(1, n_inits)

    print("Starting search: %d initializations, explore=%.0fs polish=%.0fs" %
          (n_inits, explore_budget, polish_budget))

    for idx, (name, n, arr) in enumerate(all_inits):
        if time.monotonic() > deadline - polish_budget - 5.0:
            break

        S = np.sqrt(2.0 * n)
        a_norm = normalize_seq(arr, n, S)
        t0 = time.monotonic()
        seg = max(0.5, per_init)

        # Method 1: Adam with softmax subgradient
        left = min(seg, deadline - time.monotonic() - polish_budget - 5.0)
        if left > 1.0:
            a1, s1 = run_adam_softmax(a_norm, n, S, left, rng,
                                       lr_init=0.8 * S / n,
                                       topo_k_start=5,
                                       temp_start=S * 0.5,
                                       temp_end=S * 0.02)
            if s1 < global_best:
                global_best = s1
                global_a = a1.copy()
                print("  [%d/%d] %s n=%d: adam_soft=%.6f (global best)" %
                      (idx, n_inits, name, n, s1))
            else:
                print("  [%d/%d] %s n=%d: adam_soft=%.6f" % (idx, n_inits, name, n, s1))

        # Method 2: Mirror descent with softmax (if time remains)
        left = min(seg * 0.6, deadline - time.monotonic() - polish_budget - 5.0)
        if left > 1.0 and global_a is not None and len(global_a) == n:
            a2, s2 = run_mirror_softmax(a1, n, S, left, rng,
                                         eta_init=0.3,
                                         topo_k_start=5,
                                         temp_start=S * 0.3,
                                         temp_end=S * 0.01)
            if s2 < global_best:
                global_best = s2
                global_a = a2.copy()
                print("  [%d/%d] %s mirror: %.6f (NEW global best!)" %
                      (idx, n_inits, name, s2))

    print("After exploration: global_best=%.6f" % global_best)

    # Polish phase
    left = deadline - time.monotonic()
    if left > 3.0 and global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)

        # Phase A: Adam polish with finer parameters
        seg_a = left * 0.4
        if seg_a > 1.0:
            a_pol, s_pol = run_adam_softmax(global_a, n, S, seg_a, rng,
                                             lr_init=0.3 * S / n,
                                             topo_k_start=10,
                                             temp_start=S * 0.05,
                                             temp_end=S * 0.005)
            if s_pol < global_best:
                global_best = s_pol
                global_a = a_pol.copy()
            print("Polish A (adam): %.6f" % s_pol)

        # Phase B: Mass redistribution
        left = deadline - time.monotonic()
        seg_b = left * 0.35
        if seg_b > 1.0:
            a_mr, s_mr = mass_redistribution_polish(global_a, n, S, seg_b, rng)
            if s_mr < global_best:
                global_best = s_mr
                global_a = a_mr.copy()
            print("Polish B (mass_redist): %.6f" % s_mr)

        # Phase C: Coordinate descent
        left = deadline - time.monotonic()
        seg_c = max(1.0, left * 0.5)
        if seg_c > 1.0:
            a_cd, s_cd = coordinate_descent_polish(global_a, n, S, seg_c, rng)
            if s_cd < global_best:
                global_best = s_cd
                global_a = a_cd.copy()
            print("Polish C (coord_descent): %.6f" % s_cd)

    # Final verification
    if global_a is None:
        global_a = np.ones(8192, dtype=np.float64)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    result = global_a.tolist()
    ver = evaluate_sequence(result)
    print("Final verified score: %.6f" % ver)

    if not np.isfinite(ver):
        # Fallback
        result = np.ones(8192, dtype=np.float64).tolist()
        ver = evaluate_sequence(result)
        print("Fallback score: %.6f" % ver)

    return result
