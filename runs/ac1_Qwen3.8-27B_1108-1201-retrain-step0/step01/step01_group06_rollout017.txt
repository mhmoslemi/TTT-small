import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def lp_direction(f, M):
    """
    Find g that maximizes sum(g) subject to (f*g)_i <= M for all i, g >= 0.
    f is the current sequence, M is its max convolution value.
    """
    n = len(f)
    # (f*g)_i = sum_j f_j * g_{i-j} for valid j
    # This is a linear function of g.
    # For constraint i: sum_j f_j * g_{i-j} <= M
    # The coefficient of g_k in constraint i is f_{i-k} (when i-k is valid)
    
    # Build constraint matrix A_ub
    # For each i in range(2n-1):
    #   sum_k A_ub[i,k] * g_k <= M
    #   A_ub[i,k] = f[i-k] if 0 <= i-k < n, else 0
    
    # This is a (2n-1) x n matrix
    # A_ub[i,k] = f[i-k] if max(0, i-n+1) <= k <= min(n-1, i)
    
    n_constraints = 2 * n - 1
    
    # Build the matrix efficiently
    # For each k, it appears in constraints i = k, k+1, ..., k+n-1 (clamped to [0, 2n-2])
    # A_ub[i,k] = f[i-k]
    
    # Actually let me just build it directly
    A_ub = np.zeros((n_constraints, n))
    for i in range(n_constraints):
        k_start = max(0, i - n + 1)
        k_end = min(n - 1, i)
        A_ub[i, k_start:k_end+1] = f[i - k_end : i - k_start + 1][::-1]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    res = linprog(c, A_ub=A_ub, b_ub=np.full(n_constraints, M), bounds=[(0, None)]*n, method='highs')

from scipy.optimize import linprog, minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def build_conv_matrix(f):
    """Build the matrix A where A[i,k] = f[i-k] if 0<=i-k<len(f), else 0."""
    n = len(f)
    nc = 2 * n - 1
    A = np.zeros((nc, n))
    for k in range(n):
        # Column k: A[i,k] = f[i-k] for i in [k, k+n-1]
        i_start = k
        i_end = min(k + n, nc)  # exclusive
        A[i_start:i_end, k] = f[:i_end - i_start]
    return A

def lp_direction(f, M):
    """Find g maximizing sum(g) s.t. (f*g)_i <= M, g >= 0."""
    n = len(f)
    A_ub = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, 
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None

    """Compute log-sum-exp objective and gradient w.r.t. v (log-domain parameterization)."""
    # v -> a = softmax(v)

    a = a / a_sum

    shifted = t * (conv - max_c)

    obj = log_sum_exp / t

    # Convert to gradient w.r.t. v
    dot_grad_a = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot_grad_a)

            best_v = res.x.copy()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based mirror descent
with log-sum-exp polishing. The LP finds optimal descent directions by
maximizing mass subject to convolution constraints, while LSE provides
smooth gradient-based refinement. Multi-start with diverse initializations
and sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_conv_matrix(f):
    """Build matrix A where (A @ g)[i] = (f*g)[i] for linear convolution."""
    n = len(f)
    nc = 2 * n - 1
    A = np.zeros((nc, n))
    for k in range(n):
        i_start = k
        i_end = min(k + n, nc)
        A[i_start:i_end, k] = f[:i_end - i_start]
    return A


def lp_direction(f, M, timeout=5.0):
    """Find g maximizing sum(g) s.t. (f*g)_i <= M for all i, g >= 0."""
    n = len(f)
    A_ub = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def softmax_direction(f, conv, t=50.0):
    """Softmax-based direction as fallback when LP fails."""
    n = len(f)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.ones(n) / n
    scaled = t * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-12:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    thresh = w.max() * 0.001
    active_idx = np.where(w > thresh)[0]
    for k_int in active_idx:
        k = int(k_int)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            sg[j_start:j_end + 1] += w[k] * a_slice(f, k, j_start, j_end)

    sg_centered = sg - sg.mean()
    log_factor = -sg_centered
    log_factor -= log_factor.max()
    g = f * np.exp(log_factor)
    g = np.maximum(g, 0)
    s = g.sum()
    if s < 1e-12:
        return np.ones(n) / n
    return g / s


def a_slice(f, k, j_start, j_end):
    """Return f[k-j_end : k-j_start+1][::-1]."""
    return f[k - j_end:k - j_start + 1][::-1]


def lse_obj_and_grad(v, t, n):
    """Log-sum-exp of conv(softmax(v), softmax(v)) and gradient w.r.t. v."""
    v_shifted = v - v.max()
    a = np.exp(v_shifted)
    a_sum = a.sum()
    if a_sum < 1e-30:
        return 1e10, np.zeros(n)
    a = a / a_sum

    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = t * (conv - max_c)
    exp_vals = np.exp(shifted)
    log_sum = np.log(exp_vals.sum()) + t * max_c
    obj = log_sum / t

    w = exp_vals / exp_vals.sum()

    # Gradient w.r.t. a: grad_a[k] = 2 * sum_i w[i] * a[i-k] (for valid i-k)
    # = 2 * conv(a[::-1], w)[k + n - 1]
    a_rev = a[::-1]
    size = n + len(w) - 1
    fft_size = 1 << (size - 1).bit_length()
    AR = np.fft.rfft(a_rev, fft_size)
    WR = np.fft.rfft(w, fft_size)
    conv_aw = np.fft.irfft(AR * WR, fft_size)
    grad_a = 2.0 * conv_aw[n - 1:2 * n - 1]

    # Gradient w.r.t. v (softmax Jacobian)
    dot_ga = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot_ga)

    return obj, grad_v


def optimize_lse(v0, time_limit, t_start=20, t_end=500, n_rep=5):
    """L-BFGS-B on log-sum-exp with annealing."""
    n = len(v0)
    best_v = v0.copy()
    best_obj = lse_obj_and_grad(v0, t_start, n)[0]
    t0 = time.monotonic()

    for rep in range(n_rep):
        if time.monotonic() - t0 > time_limit:
            break
        frac = rep / max(1, n_rep - 1)
        t = t_start * (t_end / t_start) ** frac

        def fun(x):
            return lse_obj_and_grad(x, t, n)

        try:
            res = minimize(fun, v0, jac=True, method='L-BFGS-B',
                           options={'maxiter': 2000, 'gtol': 1e-10, 'maxls': 50})
            if res.fun < best_obj:
                best_obj = res.fun
                best_v = res.x.copy()
                v0 = res.x.copy()
        except Exception:
            pass

    return best_v, best_obj


def do_mirror_descent(f0, max_time, t_mix=0.1, use_lp=True, lp_timeout=3.0):
    """LP-based mirror descent on the probability simplex."""
    n = len(f0)
    f = np.maximum(f0, 0).astype(np.float64)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    patience = 0
    max_patience = 200
    current_t = t_mix

    while time.monotonic() - t0 < max_time:
        remaining = max_time - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        M = conv.max()
        if M < 1e-15:
            break

        g = None
        if use_lp:
            g = lp_direction(f, M, timeout=lp_timeout)

        if g is None or g.sum() < 1e-12:
            g = softmax_direction(f, conv, t=30.0)
        else:
            g = g / g.sum()

        f_new = (1.0 - current_t) * f + current_t * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-12:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / s

        conv = fast_convolve(f_new)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            patience = 0
        else:
            patience += 1

        f = f_new

        if patience >= max_patience:
            patience = 0
            current_t *= 0.5
            if current_t < 0.003:
                # Restart with fresh momentum
                current_t = 0.08
                max_patience = 100

    return best_f, best_obj


def do_fine_tune(a0, max_time):
    """Coordinate-wise fine tuning: adjust each element to reduce peak convolution."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / a_sum

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()

    while time.monotonic() - t0 < max_time:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > max_time - 1:
                return best_a, 2.0 * n * best_a.max() if False else best_obj

            old_val = a[j]
            if old_val < 1e-15:
                continue

            # Try reducing a[j] by factor
            for factor in [0.7, 0.5, 0.8, 1.2, 1.5]:
                a[j] = old_val * factor
                if a[j] < 0:
                    a[j] = 0
                a[j] = min(a[j], 1000.0)
                s = a.sum()
                if s < 1e-12:
                    a[j] = old_val
                    continue
                a_norm = a / s
                c = fast_convolve(a_norm)
                obj = 2.0 * n * c.max()
                if obj < best_obj - 1e-10:
                    best_obj = obj
                    best_a = a_norm.copy()
                    improved = True
                    a = a_norm.copy()
                    break
            a[j] = old_val  # restore if no improvement kept

        if not improved:
            break

    return best_a, best_obj
