"""
LP-based iterative refinement with first-order linearization for minimizing
peak auto-convolution of non-negative sequences. Combines improved LP directions,
gradient-based polishing, and multi-length search.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_hann(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * t / n))


def initial_triangle(n):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(t - center) / (n / 2.0))


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def build_conv_matrix_sparse(a):
    """Build sparse matrix where M @ b gives the linear convolution a * b."""
    n = len(a)
    n_rows = 2 * n - 1
    # For each output position k, the contributing indices are:
    # conv[k] = sum_i a[i] * b[k-i]
    # So row k has entries at columns j where j = k-i for valid i
    # i.e., row k has a[i] at column (k-i) for i where 0<=i<n and 0<=k-i<n
    row_idx = []
    col_idx = []
    data = []
    for k in range(n_rows):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            row_idx.append(k)
            col_idx.append(j)
            data.append(a[i])
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_improved(a, M, use_tight=True, timeout=30.0):
    """Solve improved LP: max sum(b) s.t. (a*b)[k] <= rhs[k], b >= 0"""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix_sparse(a)
    
    if use_tight:
        conv_a = np.convolve(a, a)
        rhs = 0.5 * (M + conv_a)
    else:
        rhs = np.full(n_rows, M)
    
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=rhs, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def gradient_step(a, a_hat, lr):
    """
    Gradient step to reduce peak autocorrelation.
    a: current sequence (normalized, sum=1)
    a_hat: "correlated" sequence where a_hat[i] = sum of a[j] for j that pair with i at peak lags
    lr: learning rate
    """
    # The subgradient of max_k (a*a)[k] w.r.t. a[i] is:
    # sum over peak lags k*: (a[i+k*] + a[i-k*]) (with boundary handling)
    # This is equivalent to (a * a_hat) where a_hat is the all-ones at peak lags
    # But more precisely, a_hat[i] = sum_{k in peaks} (a[i+k] + a[i-k])
    # which is 2*(a * a_delta)[i] where a_delta has 1s at peak lags
    # Actually it's: for each peak lag k*, the contribution to gradient[i] is a[i+k*] + a[i-k*]
    # So gradient = sum_{k* in peaks} (shift(a, k*) + shift(a, -k*))
    
    # a_hat computed by caller: a_hat[i] = sum of a[j] for all (i,j) pairs that contribute to peak lags
    # a_hat[i] = sum_{k in peaks} (a[i+k] + a[i-k]) (with a[outside] = 0)
    
    g = a - lr * a_hat
    g = np.maximum(0, g)
    s = g.sum()
    if s < 1e-15:
        return a
    return g / s


def compute_a_hat(a, peak_lags):
    """Compute the 'correlated' vector a_hat[i] = sum of a[j] for j pairing with i at peak lags."""
    n = len(a)
    a_hat = np.zeros(n)
    for k in peak_lags:
        # Contribution from lag k: a_hat[i] += a[i+k] (if valid) + a[i-k] (if valid)
        for i in range(n):
            j1 = i + k
            j2 = i - k
            if 0 <= j1 < n:
                a_hat[i] += a[j1]
            if 0 <= j2 < n:
                a_hat[i] += a[j2]
    return a_hat


def compute_a_hat_fast(a, peak_lags):
    """Faster computation using numpy shifts."""
    n = len(a)
    a_hat = np.zeros(n)
    for k in peak_lags:
        if k == 0:
            a_hat += 2 * a  # a[i+0] + a[i-0] = 2*a[i]
        else:
            # a[i+k] for valid i
            if k > 0:
                # shift right by k: a[i+k] is valid for i = 0, ..., n-1-k
                a_hat[:n-k] += a[k:]
                # shift left by k: a[i-k] is valid for i = k, ..., n-1
                a_hat[k:] += a[:n-k]
            else:
                k = -k
                a_hat[k:] += a[:n-k]
                a_hat[:n-k] += a[k:]
    return a_hat

def gradient_refine(a, time_limit_s=120.0, verbose=True):
    """Gradient-based refinement using subgradient of peak autocorrelation."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    Grad start: score={best_score:.6f}")
    
    lr = 0.1 / n  # Initial learning rate
    stagnant = 0
    
    while time.monotonic() - start_time < time_limit_s:
        conv = np.convolve(a, a)
        max_val = conv.max()
        score = 2.0 * n * max_val / (a.sum() ** 2)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            lr *= 0.95
        
        if stagnant > 200:
            break
        
        # Find peak lags (top few)
        threshold = max_val * 0.95
        peak_lags = set(np.where(conv >= threshold)[0].tolist())
        # Convert to offset from center
        center = n - 1
        peak_offsets = [k - center for k in peak_lags]
        # Remove duplicates (symmetric lags give same contribution)
        peak_offsets = list(set(peak_offsets))
        
        # Compute a_hat
        a_hat = compute_a_hat_fast(a, peak_offsets)
        
        # Gradient step
        a_new = gradient_step(a, a_hat, lr)
        a = a_new
    
    if verbose:
        print(f"    Grad done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=120.0, rng=None, verbose=True):
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LS start: score={best_score:.6f}")
    
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(rng.integers(max(1, n // 200), max(2, n // 6))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.15, size=stop - start))
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])

        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1

                print(f"    LS accept {accept_count}: score={best_score:.6f}")
    
    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

"""
Improved LP-based optimization for minimum peak auto-convolution of non-negative sequences.
Key improvements: tighter LP constraints via first-order linearization, asymmetric initializations,
subgradient polishing, multi-length search with adaptive time allocation, and backtracking line search.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_hann(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * t / (n - 1))) if n > 1 else np.ones(1)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_gaussian(n, sigma_frac=0.3, center_frac=0.5):
    t = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_exponential(n, beta=1.0):
    t = np.arange(n, dtype=np.float64)
    return np.exp(-beta * t / n)


def initial_ramp(n, gamma=1.0):
    t = np.arange(n, dtype=np.float64)
    return 1.0 + gamma * t / (n - 1) if n > 1 else np.ones(1)


def initial_two_bump(n, rng, asym=0.3):
    t = np.arange(n, dtype=np.float64)
    c1 = (0.5 - asym) * n
    c2 = (0.5 + asym) * n
    w1 = 0.15 * n
    w2 = 0.25 * n
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + 0.7 * np.exp(-0.5 * ((t - c2) / w2) ** 2)
    return np.maximum(0, a)


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def build_conv_matrix_sparse(a):
    n = len(a)
    n_rows = 2 * n - 1
    row_idx = []
    col_idx = []
    data = []
    for k in range(n_rows):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            i = k - j
            row_idx.append(k)
            col_idx.append(j)
            data.append(a[i])
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp(a, conv_a, M, use_tight, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix_sparse(a)
    if use_tight:
        rhs = 0.5 * (M + conv_a)
    else:
        rhs = np.full(n_rows, M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=rhs, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def compute_gradient_vector(a, peak_lags):
    n = len(a)
    grad = np.zeros(n)
    for K in peak_lags:
        j_min = max(0, K - (n - 1))
        j_max = min(n - 1, K)
        if j_min <= j_max:
            m_vals = K - np.arange(j_min, j_max + 1)
            grad[j_min:j_max + 1] += a[m_vals]
    return 2.0 * grad


def lp_refine(a, time_limit_s=600.0, max_iters=2000, verbose=True, tight=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}, tight={tight}")
    stagnant = 0
    it = 0
    alpha = 0.3
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            alpha = min(0.5, alpha * 1.05)
            if verbose and (it % 200 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}, alpha={alpha:.3f}")
        else:
            stagnant += 1
            alpha = max(0.02, alpha * 0.98)
        if stagnant > 80:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(20.0, remaining * 0.15))
        g = solve_lp(a, conv, M, use_tight=tight, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-12:
            a = a_new
        else:
            a2 = (1.0 - alpha * 0.3) * a + alpha * 0.3 * g
            a2 = normalize_nonneg(a2)
            a = a2
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def gradient_refine(a, time_limit_s=60.0, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    Grad start: score={best_score:.6f}")
    lr = 0.5 / n
    stagnant = 0
    while time.monotonic() - start_time < time_limit_s:
        conv = np.convolve(a, a)
        max_val = conv.max()
        score = 2.0 * n * max_val / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            lr *= 0.97
        if stagnant > 300:
            break
        threshold = max_val * 0.92
        peak_indices = np.where(conv >= threshold)[0]
        if len(peak_indices) > 20:
            top_idx = np.argsort(conv[peak_indices])[-20:]
            peak_lags = peak_indices[top_idx].tolist()
        else:
            peak_lags = peak_indices.tolist()
        grad_vec = compute_gradient_vector(a, peak_lags)
        a_new = a - lr * grad_vec
        a_new = np.maximum(0.0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    if verbose:
        print(f"    Grad done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

    if verbose:

    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(rng.integers(max(1, n // 300), max(2, n // 5))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.12, size=stop - start))
        if rng.random() < 0.2:
            shift = int(rng.integers(1, max(2, n // 10)))
            candidate = np.maximum(0, candidate + shift * 0.01 * candidate[:n - shift] - shift * 0.01 * candidate[shift:])

        sc = score_sequence(candidate)
        if sc < best_score:

            accept_count += 1
            if verbose and accept_count % 200 == 0:

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
