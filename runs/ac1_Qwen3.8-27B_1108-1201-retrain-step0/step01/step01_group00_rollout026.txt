"""
Multi-phase optimization for minimizing peak self-convolution of non-negative sequences.
Uses L-BFGS-B with annealed soft-max, multi-start, and coordinate descent polish.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (S * S)

def soft_max_value_and_grad(c, tau):
    c_max = c.max()
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    weights = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    return value, weights

def obj_grad(theta, tau, n):
    a = np.exp(theta - theta.max())
    S = a.sum()
    c = fft_convolve(a, a)
    value, w = soft_max_value_and_grad(c, tau)
    w_rev = w[::-1]
    corr = fft_convolve(a, w_rev)
    g = corr[n-1:2*n-1][::-1]
    obj = 2.0 * n * value / (S * S)
    grad = 2.0 * n * a * (S * g - 2.0 * value) / (S * S)
    return obj, grad

def l_bfgs_optimize(a_init, n, tau_seq, max_iter_per_tau=200, time_limit=None):
    """Run L-BFGS with a sequence of temperatures."""
    theta = np.log(a_init + 1e-15)
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    t0 = time.monotonic()
    
    for tau in tau_seq:
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        result = minimize(
            lambda th: obj_grad(th, tau, n)[0],
            theta,

"""
Multi-phase optimization minimizing peak self-convolution of non-negative step functions.
Combines L-BFGS-B with annealed soft-max smooth approximation, gradient-guided simulated
annealing polish, and adaptive temperature scheduling. Uses FFT-based convolution and
analytical gradients for efficiency.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score (scale-invariant)."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (S * S)


def soft_max_value_and_weights(c, tau):
    """Compute smooth max value and softmax weights."""
    c_max = c.max()
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    weights = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    return value, weights


def obj_grad_smooth(theta, tau, n):
    """
    Compute smooth objective and gradient w.r.t. theta.
    a[j] = exp(theta[j]), objective = 2n * soft_max_tau(c) / S^2
    """
    a = np.exp(theta - theta.max())
    S = a.sum()
    if S < 1e-30:
        return np.inf, np.zeros(n)
    c = fft_convolve(a, a)
    value, w = soft_max_value_and_weights(c, tau)

    # g[j] = sum_k w[k] * a[k-j] (cross-correlation)
    w_rev = w[::-1]
    corr = fft_convolve(a, w_rev)
    g = corr[n - 1:2 * n - 1][::-1]

    obj = 2.0 * n * value / (S * S)
    grad = 2.0 * n * a * (S * g - 2.0 * value) / (S * S)

    return obj, grad


def run_lbfgs(a_init, n, tau_list, max_iter, time_limit):
    """Run L-BFGS-B with a sequence of temperatures. Returns (best_a, best_score)."""
    theta = np.log(a_init + 1e-15)
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    t0 = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - t0 > time_limit:
            break

        def fun(th):
            return obj_grad_smooth(th, tau, n)[0]

        def jac(th):
            return obj_grad_smooth(th, tau, n)[1]

        try:
            result = minimize(
                fun, theta, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxfun': max_iter * 3, 'iprint': -1}
            )
            theta = result.x
        except Exception:
            pass

        a = np.exp(theta - theta.max())
        a = np.clip(a, 1e-15, None)
        a = a / a.sum()
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        theta = np.log(a + 1e-15)

    return best_a, best_score

    n = len(best_a)

    # === Phase 1: Adaptive L-BFGS on best candidate ===
    print(f"\n=== Phase 1: Adaptive L-BFGS (n={n}) ===")
    p1_time = min(900.0, max(60.0, (deadline - time.monotonic()) * 0.4))
    a_opt, score_opt = adaptive_lbfgs_schedule(best_a, n, p1_time, rng)
    if score_opt < best_score:
        best_score = score_opt
        best_a = a_opt.copy()
        print(f"  Phase 1 result: {best_score:.6f}")
    else:
        print(f"  Phase 1: no improvement, best={best_score:.6f}")

    # === Phase 2: Gradient-guided simulated annealing ===
    print(f"\n=== Phase 2: Gradient-guided SA ===")
    p2_time = min(600.0, max(60.0, (deadline - time.monotonic()) * 0.3))
    a_sa, score_sa = gradient_guided_sa(best_a, n, p2_time, rng)
    if score_sa < best_score:
        best_score = score_sa
        best_a = a_sa.copy()
        print(f"  Phase 2 result: {best_score:.6f}")
    else:
        print(f"  Phase 2: no improvement, best={best_score:.6f}")

    # === Phase 3: Final L-BFGS refinement ===
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 3: Final L-BFGS refinement ===")
        p3_time = min(500.0, max(30.0, deadline - time.monotonic() - 20.0))

        # Run on the current best
        tau_list_final = np.exp(np.linspace(np.log(5.0/n), np.log(0.003/n), 15))
        a_opt3, score_opt3 = run_lbfgs(best_a, n, tau_list_final, 300, p3_time * 0.5)
        if score_opt3 < best_score:
            best_score = score_opt3
            best_a = a_opt3.copy()
            print(f"  Phase 3a: {best_score:.6f}")

        # One more SA polish
        if time.monotonic() < deadline - 30:
            p3b_time = max(10.0, deadline - time.monotonic() - 20.0)
            a_sa2, score_sa2 = gradient_guided_sa(best_a, n, p3b_time, rng)
            if score_sa2 < best_score:
                best_score = score_sa2
                best_a = a_sa2.copy()
                print(f"  Phase 3b: {best_score:.6f}")

    # === Phase 4: Try a different-length sequence if time permits ===
    if time.monotonic() < deadline - 120:
        print(f"\n=== Phase 4: Alternative length exploration ===")
        for alt_n in [320, 384]:
            if time.monotonic() > deadline - 60:
                break
            # Resample best to new length
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, alt_n)
            a_alt = np.interp(x_new, x_old, best_a * best_a.sum())
            a_alt = np.clip(a_alt, 1e-6, None)
            a_alt = a_alt / a_alt.sum()

            tau_list_alt = np.exp(np.linspace(np.log(5.0/alt_n), np.log(0.01/alt_n), 10))
            time_alt = min(60.0, max(15.0, deadline - time.monotonic() - 30.0))
            a_alt_opt, score_alt = run_lbfgs(a_alt, alt_n, tau_list_alt, 200, time_alt)

            print(f"  n={alt_n}: score={score_alt:.6f}")
            if score_alt < best_score:
                best_score = score_alt
                best_a = a_alt_opt.copy()
                n = alt_n
                print(f"  *** New best with n={alt_n}: {best_score:.6f} ***")

    # Final: prepare output
    final_a = best_a.copy()
    final_a = np.clip(final_a, 0.0, 1000.0)

    # Ensure sum is not too small (evaluate_sequence rejects sum < 0.01)
    if final_a.sum() < 1.0:
        final_a *= 1.0 / max(final_a.sum(), 1e-10)

    final_list = [float(x) for x in final_a]
    final_score = evaluate_sequence(final_list)
    print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")

    return final_list
