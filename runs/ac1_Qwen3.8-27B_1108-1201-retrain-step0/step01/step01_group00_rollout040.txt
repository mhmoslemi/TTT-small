    fw = np.fft.rfft(w, N)

    g = 2.0 * r[:n]
    
    return value, g, c


def local_score_fast(a):
    """Fast local computation of the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def optimize_v2(n, theta_init, time_limit, rng, init_temp=1.0):
    """
    Improved gradient-based optimization with:
    - Cyclic temperature schedule
    - Adaptive learning rate
    - Periodic re-randomization to escape local minima
    """
    S = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf
    plateau_count = 0
    restart_count = 0
    max_restarts = 20
    
    # Cyclic temperature: oscillates between low and high
    tau_min = 0.1 / n
    tau_max = 10.0 / n
    cycle_length = 2000  # steps per cycle
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0 - t0
        elapsed_frac = (time.monotonic() - t0) / time_limit
        
        # Cyclic temperature with overall downward trend
        cycle_pos = (step % cycle_length) / cycle_length
        # Triangle wave: 0 -> 1 -> 0
        tri = 1 - abs(2 * cycle_pos - 1)
        # Overall trend: start high, end low
        trend = 1.0 - 0.8 * min(1.0, elapsed_frac * 1.5)
        # Combined: oscillate within a band that shrinks over time
        tau_center = tau_min + (tau_max - tau_min) * trend * 0.5
        tau_amplitude = (tau_max - tau_min) * trend * 0.5
        tau = tau_center + tau_amplitude * (tri - 0.5) * 2
        tau = max(tau, tau_min)
        
        a = softmax_normalize(theta) * S
        value, g_a, c = gradient_soft_max_conv(a, tau)
        
        # Gradient w.r.t. theta: g_theta = a * (g_a - <a, g_a> / S)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)
        
        # Adam update
        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        
        # Adaptive learning rate
        base_lr = 5.0
        lr = base_lr / (1.0 + 0.1 * step)
        
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Evaluate
        a_check = softmax_normalize(theta) * S
        score = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (S * S)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
            plateau_count = 0
        else:
            plateau_count += 1
        
        # Periodic restart if stuck
        if plateau_count > 5000 and restart_count < max_restarts:
            # Add noise to escape
            theta = best_theta + rng.normal(0, 0.5 / n, size=n)
            momentum1 = np.zeros(n)
            momentum2 = np.zeros(n)
            step = 0
            plateau_count = 0
            restart_count += 1
            print(f"      Restart #{restart_count} at score={best_score:.6f}")
        
        if step % 5000 == 0:
            print(f"      step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")
    
    return best_theta, best_score


def coordinate_descent_polish(a_init, time_limit, rng):
    """
    Coordinate descent: try adjusting individual elements to reduce the peak.
    More targeted than the previous simulated annealing.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    # Precompute convolution for efficiency
    conv = fft_convolve(current, current)
    
    while time.monotonic() - t0 < time_limit:
        # Find peak location
        peak_idx = int(np.argmax(conv))
        
        # Compute contributions to peak
        # conv[peak_idx] = sum_i a[i]*a[peak_idx-i]
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = current[i] * current[j]
        
        # Try reducing top contributors and increasing others
        top_k = max(3, n // 8)
        top_idx = np.argsort(contribs)[-top_k:]
        
        # For each top contributor, try a small reduction
        improved = False
        for idx in top_idx:
            for frac in [0.9, 0.8, 0.7, 0.5]:
                trial = current.copy()
                trial[idx] *= frac
                # Renormalize to keep sum constant
                trial = trial / np.sum(trial) * np.sum(current)
                trial_score = local_score_fast(trial)
                if trial_score < current_score - 1e-10:
                    current = trial
                    current_score = trial_score
                    conv = fft_convolve(current, current)
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try increasing a random non-peak contributor
            bottom_k = max(3, n // 8)
            bottom_idx = np.argsort(contribs)[:bottom_k]
            idx = int(rng.integers(0, len(bottom_idx)))
            target = bottom_idx[idx]
            for frac in [1.1, 1.2, 1.5, 2.0]:
                trial = current.copy()
                trial[target] *= frac
                trial = trial / np.sum(trial) * np.sum(current)
                trial_score = local_score_fast(trial)
                if trial_score < current_score - 1e-10:
                    current = trial
                    current_score = trial_score
                    conv = fft_convolve(current, current)
                    improved = True
                    break
        
        if not improved:
            # Try a random pair swap
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 != i2:
                trial = current.copy()
                delta = 0.01 * np.sum(current) / n
                trial[i1] -= delta
                trial[i2] += delta
                if trial[i1] >= 0:
                    trial_score = local_score_fast(trial)
                    if trial_score < current_score - 1e-10:
                        current = trial
                        current_score = trial_score
                        conv = fft_convolve(current, current)
        
        if current_score < best_score:
            best_score = current_score
            best = current.copy()
    
    return best, best_score


def targeted_peak_reduction(a_init, time_limit, rng):
    """
    Iteratively identify peak contributors and reduce them, while compensating
    by increasing non-contributing elements. Uses a more sophisticated strategy
    than simple coordinate descent.
    """
    current = a_init.astype(np.float64).copy()
    sum_a = np.sum(current)
    current_score = local_score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(current, current)
        peak_idx = int(np.argmax(conv))
        
        # Compute all contributions to peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = current[i] * current[j]
        
        # Strategy: reduce top contributors, increase bottom contributors
        # proportional to their current values
        top_f = max(5, n // 10)
        bottom_f = max(5, n // 10)
        top_idx = np.argsort(contribs)[-top_f:]
        bottom_idx = np.argsort(contribs)[:bottom_f]
        
        # Damped adjustment
        for strength in [0.05, 0.02, 0.01]:
            trial = current.copy()
            reduce_amount = strength * contribs[top_idx].sum() / max(1, current[top_idx].sum())
            trial[top_idx] -= reduce_amount * current[top_idx] / current[top_idx].max()
            increase_amount = reduce_amount * current[top_idx].sum() / max(1e-10, current[bottom_idx].sum())
            trial[bottom_idx] += increase_amount * current[bottom_idx] / (current[bottom_idx].max() + 1e-10)
            
            trial = np.clip(trial, 0.01, None)
            trial = trial / np.sum(trial) * sum_a
            
            trial_score = local_score_fast(trial)
            if trial_score < current_score - 1e-12:
                current = trial
                current_score = trial_score
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
                break
    
    return best, best_score

Multi-strategy optimization for minimizing the peak of auto-convolution of
non-negative sequences. Combines gradient-based optimization with cyclic
temperature annealing, top-K subgradient methods, targeted peak reduction,
and coordinate descent polish. Explores multiple initializations and lengths.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def soft_max_value_and_weights(c, tau):
    """Compute soft-max value and corresponding weights."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    return value, weights


def gradient_soft_max_conv(a, tau):
    """
    Compute gradient of soft_max(conv(a,a), tau) w.r.t. a using FFT.
    Formula: g = 2 * ifft(fft(w, N) * conj(fft(a, N)))[:n]
    where w = soft_max_weights(conv(a,a), tau), N >= 3n-2.
    """
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_value_and_weights(c, tau)

    N = 1
    while N < 3 * n - 2:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    r = np.fft.irfft(fw * np.conj(fa), N)
    g = 2.0 * r[:n]

    return value, g, c


def topk_subgradient(a, K=10):
    """
    Compute a subgradient of max(conv(a,a)) using the top-K peak positions.
    Returns (max_val, gradient).
    """
    n = len(a)
    c = fft_convolve(a, a)
    max_val = np.max(c)

    # Find top-K indices
    if len(c) <= K:
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argsort(c)[-K:]

    # Weight by how close to max they are
    weights = np.zeros_like(c)
    for idx in top_idx:
        weights[idx] = c[idx] / max_val if max_val > 0 else 0.0
    total_w = np.sum(weights)
    if total_w > 0:
        weights = weights / total_w

    # Compute gradient: g[j] = 2 * sum_k w[k] * a[k-j]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(weights, N)
    r = np.fft.irfft(fw * np.conj(fa), N)
    g = 2.0 * r[:n]

    return max_val, g, c


def local_score_fast(a):
    """Fast local computation of the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def optimize_cyclic(n, theta_init, time_limit, rng):
    """
    Gradient-based optimization with cyclic temperature schedule and Adam.
    The temperature oscillates to help escape local minima, with an overall
    downward trend over the total time budget.
    """
    S = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf
    plateau_count = 0
    restart_count = 0
    max_restarts = 30

    tau_min = 0.05 / n
    tau_max = 5.0 / n
    cycle_len = 3000

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit

        # Cyclic temperature with downward trend
        cycle_pos = (step % cycle_len) / cycle_len
        tri = 1.0 - abs(2.0 * cycle_pos - 1.0)
        trend = 1.0 - 0.85 * min(1.0, elapsed_frac * 1.2)
        tau_center = tau_min + (tau_max - tau_min) * trend * 0.5
        tau_amp = (tau_max - tau_min) * trend * 0.4
        tau = tau_center + tau_amp * (tri - 0.5) * 2.0
        tau = max(tau, tau_min)

        a = softmax_normalize(theta) * S
        value, g_a, c = gradient_soft_max_conv(a, tau)

        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        step += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (g_theta * g_theta)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)

        # Adaptive LR: start high, decay gently
        lr = 8.0 / (1.0 + 0.05 * np.sqrt(step))
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        a_check = softmax_normalize(theta) * S
        score = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (S * S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
            plateau_count = 0
        else:
            plateau_count += 1

        # Restart if stuck for too long
        if plateau_count > 8000 and restart_count < max_restarts:
            noise_scale = 0.3 + 0.3 * rng.random()
            theta = best_theta + rng.normal(0, noise_scale, size=n)
            momentum1 = np.zeros(n)
            momentum2 = np.zeros(n)
            step = 0
            plateau_count = 0
            restart_count += 1

        if step % 10000 == 0:
            print(f"        step={step}, score={score:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.2e}, t={time.monotonic()-t0:.1f}s, restarts={restart_count}")

    return best_theta, best_score


def optimize_topk(n, theta_init, time_limit, rng):
    """
    Optimization using top-K subgradient with backtracking line search.
    Different from the soft-max approach: uses a sparse gradient from the
    top-K peaks with adaptive step size.
    """
    S = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf
    step = 0
    lr = 1.0
    momentum = np.zeros(n)
    beta1 = 0.9
    stall = 0
    max_stall = 5000

    while time.monotonic() - t0 < time_limit:
        a = softmax_normalize(theta) * S
        max_val, g_a, c = topk_subgradient(a, K=8)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        # Normalize gradient
        g_norm = np.linalg.norm(g_theta)
        if g_norm < 1e-15:
            break
        g_theta = g_theta / g_norm

        # Momentum
        momentum = beta1 * momentum + (1.0 - beta1) * g_theta

        # Try step
        candidate = theta - lr * momentum
        a_cand = softmax_normalize(candidate) * S
        score_cand = 2.0 * n * np.max(fft_convolve(a_cand, a_cand)) / (S * S)

        a_cur = softmax_normalize(theta) * S
        score_cur = 2.0 * n * np.max(fft_convolve(a_cur, a_cur)) / (S * S)

        if score_cand < score_cur:
            theta = candidate
            lr = min(lr * 1.05, 10.0)
            stall = 0
        else:
            lr = max(lr * 0.7, 0.001)
            stall += 1

        step += 1
        if np.isfinite(score_cand) and score_cand < best_score:
            best_score = score_cand
            best_theta = candidate.copy()

        if stall > max_stall:
            # Restart with noise
            theta = best_theta + rng.normal(0, 0.5, size=n)
            momentum = np.zeros(n)
            lr = 1.0
            stall = 0

        if step % 5000 == 0:
            print(f"        topk step={step}, score={score_cur:.6f}, best={best_score:.6f}, "
                  f"lr={lr:.4f}, t={time.monotonic()-t0:.1f}s")

    return best_theta, best_score


def targeted_peak_reduction(a_init, time_limit, rng):
    """
    Iteratively reduce peak contributors and increase non-contributors.
    Uses a multi-scale approach: try large, medium, and small adjustments.
    """
    current = a_init.astype(np.float64).copy()
    sum_a = np.sum(current)
    current_score = local_score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(current, current)
        peak_idx = int(np.argmax(conv))

        # Compute contributions to peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = current[i] * current[j]

        total_contrib = np.sum(contribs)
        if total_contrib < 1e-15:
            break

        # Multi-scale adjustment
        improved = False
        scales = [0.1, 0.05, 0.02, 0.01]

        for scale in scales:
            trial = current.copy()

            # Reduce top contributors
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            reduce_frac = scale * contribs[top_idx]
            trial[top_idx] -= reduce_frac / (current[top_idx] + 1e-10) * current[top_idx] * 0.5

            # Increase bottom contributors to maintain sum
            bottom_k = max(3, n // 12)
            bottom_idx = np.argsort(contribs)[:bottom_k]
            total_reduced = np.sum(current[top_idx] - trial[top_idx])
            if total_reduced > 1e-10:
                bottom_sum = np.sum(current[bottom_idx])
                if bottom_sum > 1e-10:
                    trial[bottom_idx] += total_reduced * current[bottom_idx] / bottom_sum

            trial = np.clip(trial, 0.001, None)
            trial = trial / np.sum(trial) * sum_a
            trial_score = local_score_fast(trial)

            if trial_score < current_score - 1e-12:
                current = trial
                current_score = trial_score
                improved = True
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
                break

        if not improved:
            # Try random perturbation in a small neighborhood
            n_perturb = max(2, n // 20)
            idx = rng.choice(n, size=n_perturb, replace=False)
            trial = current.copy()
            trial[idx] *= np.exp(rng.normal(-0.02, 0.02, size=n_perturb))
            trial = np.clip(trial, 0.001, None)
            trial = trial / np.sum(trial) * sum_a
            trial_score = local_score_fast(trial)
            if trial_score < current_score - 1e-12:
                current = trial
                current_score = trial_score
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()

        iteration += 1
        if iteration % 200 == 0:
            print(f"      TPR iter={iteration}, score={current_score:.6f}, best={best_score:.6f}, "
                  f"t={time.monotonic()-t0:.1f}s")

    return best, best_score


def coordinate_polish(a_init, time_limit, rng):
    """
    Coordinate descent: try adjusting one or two elements at a time
    to reduce the convolution peak.
    """
    current = a_init.astype(np.float64).copy()
    sum_a = np.sum(current)
    current_score = local_score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(current, current)
        peak_idx = int(np.argmax(conv))

        # Find elements that contribute to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = current[i] * current[j]

        # Try adjusting the single most impactful element
        top_idx = np.argsort(contribs)[-5:]
        found = False
        for idx in top_idx:
            for frac in [0.85, 0.7, 0.5, 1.15, 1.3]:
                trial = current.copy()
                trial[idx] *= frac
                if trial[idx] < 0:
                    continue
                trial = trial / np.sum(trial) * sum_a
                trial_score = local_score_fast(trial)
                if trial_score < current_score - 1e-10:
                    current = trial
                    current_score = trial_score
                    found = True
                    break
            if found:
                break

        if not found:
            # Try a small random move
            i1 = int(rng.integers(0, n))
            delta = 0.005 * sum_a / n * rng.uniform(0.5, 2.0)
            trial = current.copy()
            trial[i1] -= delta
            if trial[i1] >= 0:
                # Distribute the removed mass to a random other position
                i2 = int(rng.integers(0, n))
                if i2 != i1:
                    trial[i2] += delta
                    trial = trial / np.sum(trial) * sum_a
                    trial_score = local_score_fast(trial)
                    if trial_score < current_score - 1e-10:
                        current = trial
                        current_score = trial_score

        if current_score < best_score:
            best_score = current_score
            best = current.copy()

    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Create a chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng):
    """Create a multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k, amp in [(2, 0.3), (3, 0.2), (5, 0.15), (8, 0.1), (13, 0.05)]:
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=2380, **kwargs):
    """
    Multi-phase search for minimizing the auto-convolution peak:
    Phase 1: Gradient optimization with cyclic temperature on provided sequence
    Phase 2: Top-K subgradient optimization on provided sequence
    Phase 3: Different initializations and lengths
    Phase 4: Targeted peak reduction polish
    Phase 5: Coordinate descent polish
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score_fast(start_seq)
        print(f"Starting sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    n_default = len(start_seq) if start_seq is not None else 256

    # Phase 1: Cyclic gradient optimization on provided sequence
    print("\n=== Phase 1: Cyclic gradient optimization ===")
    if start_seq is not None:
        n = n_default
        theta_init = np.log(start_seq / np.sum(start_seq) + 1e-10)
        avail = deadline - time.monotonic()
        t1 = max(120.0, min(900.0, avail * 0.35))
        t0 = time.monotonic()
        theta_best, score1 = optimize_cyclic(n, theta_init, t1, rng)
        a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n)
        actual_score = local_score_fast(a_best)
        print(f"  Phase 1 result: score={actual_score:.6f} (time={time.monotonic()-t0:.1f}s)")
        if actual_score < best_score:
            best_score = actual_score
            best_seq = a_best.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # Phase 2: Top-K subgradient optimization
    print("\n=== Phase 2: Top-K subgradient ===")
    if start_seq is not None:
        n = n_default
        theta_init = np.log(start_seq / np.sum(start_seq) + 1e-10)
        avail = deadline - time.monotonic()
        t2 = max(60.0, min(400.0, avail * 0.2))
        t0 = time.monotonic()
        theta_best, score2 = optimize_topk(n, theta_init, t2, rng)
        a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n)
        actual_score = local_score_fast(a_best)
        print(f"  Phase 2 result: score={actual_score:.6f} (time={time.monotonic()-t0:.1f}s)")
        if actual_score < best_score:
            best_score = actual_score
            best_seq = a_best.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # Phase 3: Different initializations
    print("\n=== Phase 3: Different initializations ===")
    avail = deadline - time.monotonic()
    if avail > 100:
        n = n_default
        t = np.arange(n, dtype=np.float64)

        init_list = []
        # Various chirps
        for alpha in [0.3, 0.6, 0.9]:
            init_list.append(make_chirp_init(n, alpha, rng))
        # Multi-tone
        init_list.append(make_multitone_init(n, rng))
        # Random with different spreads
        init_list.append(np.clip(1.0 + 0.2 * rng.standard_normal(n), 0.01, None))
        init_list.append(np.clip(1.0 + 0.5 * rng.standard_normal(n), 0.01, None))
        # Hamming-windowed
        window = 0.54 - 0.46 * np.cos(2.0 * np.pi * t / (n - 1))
        init_list.append(np.clip(window + 0.1 * rng.standard_normal(n), 0.01, None))

        t_per = max(30.0, min(120.0, avail / (len(init_list) + 1)))
        for i, a0 in enumerate(init_list):
            if time.monotonic() > deadline - 60:
                break
            theta_init = np.log(a0 / np.sum(a0) + 1e-10)
            t_use = min(t_per, deadline - time.monotonic() - 60)
            if t_use < 20:
                break
            theta_best, score_i = optimize_cyclic(n, theta_init, t_use, rng)
            a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n)
            actual_score = local_score_fast(a_best)
            print(f"  Init {i}: score={actual_score:.6f}")
            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Phase 4: Try other lengths
    print("\n=== Phase 4: Other lengths ===")
    avail = deadline - time.monotonic()
    if avail > 80:
        other_lengths = [200, 300, 350, 400, 512, 640]
        for n in other_lengths:
            if time.monotonic() > deadline - 60:
                break
            a0 = make_chirp_init(n, 0.5, rng)
            theta_init = np.log(a0 / np.sum(a0) + 1e-10)
            t_use = max(20.0, min(80.0, (deadline - time.monotonic() - 60) / 2.0))
            if t_use < 15:
                break
            theta_best, _ = optimize_cyclic(n, theta_init, t_use, rng)
            a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n)
            actual_score = local_score_fast(a_best)
            print(f"  n={n}: score={actual_score:.6f}")
            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Phase 5: Polish
    print(f"\n=== Phase 5: Polish (current best={best_score:.6f}) ===")
    if best_seq is not None:
        avail = deadline - time.monotonic()
        if avail > 30:
            # Targeted peak reduction
            t_pr = max(30.0, min(180.0, avail * 0.6))
            t0 = time.monotonic()
            polished, p_score = targeted_peak_reduction(best_seq, t_pr, rng)
            print(f"  After TPR: {p_score:.6f} (t={time.monotonic()-t0:.1f}s)")
            if p_score < best_score:
                best_score = p_score
                best_seq = polished.copy()

            # Coordinate descent
            avail2 = deadline - time.monotonic()
            if avail2 > 20:
                t_cd = max(20.0, min(120.0, avail2 * 0.7))
                t0 = time.monotonic()
                polished, p_score = coordinate_polish(best_seq, t_cd, rng)
                print(f"  After coord descent: {p_score:.6f} (t={time.monotonic()-t0:.1f}s)")
                if p_score < best_score:
                    best_score = p_score
                    best_seq = polished.copy()

    # Final verification
    print(f"\n=== Final: best_score={best_score:.6f} ===")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"Verified score: {final_score:.6f}, n={len(final_list)}")
        return final_list

    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
