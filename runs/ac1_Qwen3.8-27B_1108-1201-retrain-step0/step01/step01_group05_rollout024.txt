        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_init * (0.5 ** (t / 800.0))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Random perturbation to escape local minima
        if t - last_improve > 3000 and rng.random() < 0.1:
            x = x + rng.normal(0.0, 0.1, n)
            stuck_count += 1
            if stuck_count > 5:
                break

    return best_a, best_maxc


def make_chirp_init(n, alpha=0.8, rng=None):
    """Create a chirp-based non-negative initialization."""
    i = np.arange(n, dtype=np.float64)
    phase = 2.0 * np.pi * alpha * i * i / n
    a = 1.0 + 0.5 * np.cos(phase)
    a = np.maximum(a, 0.01)
    if rng is not None:
        a = a + 0.05 * rng.random(n)
    return a


def make_smooth_init(n, freq1=2, freq2=7, rng=None):
    """Create a smooth sinusoidal initialization."""
    t = np.linspace(0, 1, n)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * freq1 * t) + 0.15 * np.cos(2 * np.pi * freq2 * t)
    a = np.maximum(a, 0.01)
    if rng is not None:
        a = a + 0.05 * rng.random(n)
    return a


def make_random_init(n, rng):
    """Create a random initialization."""
    return 0.5 + rng.random(n)


def polish_best(a_best, n, budget, rng):
    """Final polishing with very low temperature and hybrid gradient."""
    if budget < 2.0:
        return a_best, np.inf
    S = np.sqrt(2.0 * n)
    best_a = np.asarray(a_best, dtype=np.float64)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.full(n, S / n)
    else:
        best_a = best_a * (S / s)

    c = fft_autocorr(best_a)
    best_maxc = c.max()

    x = a_to_x(best_a, S)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    T = 0.05
    lr = 0.3
    start = time.monotonic()

    while time.monotonic() - start < budget - 1.0:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / budget)
        T = 0.05 * (0.01 / 0.05) ** frac
        T = max(T, 0.01)

        a = softmax_make_a(x, S)
        c = fft_autocorr(a)
        cur_maxc = c.max()

        if cur_maxc < best_maxc:
            best_maxc = cur_maxc
            best_a = a.copy()

        # Hybrid: blend smooth and exact
        if t % 3 == 0:
            grad_s = compute_smooth_grad(a, c, T, S)
            grad_e = compute_exact_subgrad_x(a, c, S, top_k=5)
            grad = 0.5 * grad_s + 0.5 * grad_e
        else:
            grad = compute_smooth_grad(a, c, T, S)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr_cur = lr * (0.9995 ** t)
        x = x - lr_cur * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_maxc


def propose_candidate(seed=42, budget_s=2400):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    # Collect initializations with their lengths
    init_configs = []

    # Try to use the given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(given) > 0:
            init_configs.append(("given", given.copy(), len(given)))
    except Exception:
        pass

    # Chirp initializations at different lengths
    for n_try in (4096, 8192, 16384):
        init_configs.append(("chirp_%d" % n_try, make_chirp_init(n_try, 0.8, rng), n_try))
    for n_try in (8192, 16384):
        init_configs.append(("smooth_%d" % n_try, make_smooth_init(n_try, 3, 11, rng), n_try))
    for n_try in (8192, 16384):
        init_configs.append(("rand_%d" % n_try, make_random_init(n_try, rng), n_try))

    # Filter out very small
    init_configs = [(nm, arr, nn) for nm, arr, nn in init_configs if nn >= 1024]

    if not init_configs:
        n_default = 4096
        init_configs.append(("default", make_random_init(n_default, rng), n_default))

    # Phase 1: Explore each initialization
    n_configs = len(init_configs)
    explore_fraction = 0.70
    explore_budget = explore_fraction * (deadline - time.monotonic())
    per_config = explore_budget / max(1, n_configs)

    global_best_maxc = np.inf
    global_best_a = None
    global_best_n = 0
    results = []

    for idx, (name, arr, n) in enumerate(init_configs):
        if time.monotonic() > deadline - (1.0 - explore_fraction) * (deadline - time.monotonic() - explore_budget) - 5.0:
            pass

        if time.monotonic() > deadline - 60.0:
            break

        remaining_config = per_config - (time.monotonic() - (deadline - explore_budget - idx * per_config))
        # Simplified: just use a fraction of remaining time
        time_left = deadline - 60.0 - time.monotonic()
        configs_left = n_configs - idx
        alloc = min(per_config * 1.5, time_left / max(1, configs_left))
        alloc = max(5.0, alloc)

        if alloc < 5.0:
            break

        # Run log-domain Adam
        t0 = time.monotonic()
        a_res, maxc_res = run_log_adam(arr, n, alloc, rng, T_init=1.0, T_final=0.05,
                                       lr_init=0.8, use_hybrid=False)
        elapsed = time.monotonic() - t0
        results.append((name, n, maxc_res, a_res, elapsed))
        print("  %s (n=%d): maxc=%.6f in %.1fs" % (name, n, maxc_res, elapsed))

        # If we have time, try hybrid version on the result
        time_left2 = deadline - 60.0 - time.monotonic()
        if time_left2 > 15.0 and maxc_res < global_best_maxc:
            t1 = time.monotonic()
            a_res2, maxc_res2 = run_log_adam(a_res, n, min(alloc * 0.5, time_left2 - 5.0),
                                             rng, T_init=0.1, T_final=0.02,
                                             lr_init=0.4, use_hybrid=True, top_k=5)
            elapsed2 = time.monotonic() - t1
            if maxc_res2 < maxc_res:
                a_res, maxc_res = a_res2, maxc_res2
            results.append((name + "_hybrid", n, maxc_res, a_res, elapsed2))
            print("    %s hybrid: maxc=%.6f in %.1fs" % (name, maxc_res, elapsed2))

        if maxc_res < global_best_maxc:
            global_best_maxc = maxc_res
            global_best_a = a_res.copy()
            global_best_n = n

    print("\nPhase 1 best: maxc=%.6f (n=%d)" % (global_best_maxc, global_best_n))

    # Phase 2: Polish the best
    if global_best_a is not None:
        polish_budget = max(5.0, deadline - time.monotonic() - 5.0)
        if polish_budget > 3.0:
            a_polished, maxc_polished = polish_best(global_best_a, global_best_n, polish_budget, rng)
            if maxc_polished < global_best_maxc:
                global_best_a = a_polished
                global_best_maxc = maxc_polished
            print("Phase 2 polished: maxc=%.6f" % global_best_maxc)

    # Final verification
    if global_best_a is None:
        global_best_a = np.ones(1024, dtype=np.float64)

    global_best_a = np.clip(global_best_a, 0.0, 1000.0)
    if global_best_a.sum() < 0.01:
        global_best_a = np.ones(1024, dtype=np.float64)

    # Verify with authoritative function
    try:
        val = evaluate_sequence([float(x) for x in global_best_a])
        print("Final verified score: %.6f" % val if np.isfinite(val) else "Final verify: inf")
        if not np.isfinite(val):
            # Fallback
            global_best_a = np.ones(1024, dtype=np.float64)
    except Exception as e:
        print("Verification error: %s" % str(e))

    return [float(x) for x in global_best_a]
