```python
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)
```

```python
"""
Minimize 2*n*max_conv/sum^2 for non-negative sequences using LSE-smoothed
subgradient with FFT-accelerated gradient computation.

Key ideas:
- Use log-sum-exp smoothing of the max to get a smooth surrogate objective
- Compute the exact gradient of the LSE in O(n log n) via FFT
- Use Adam optimizer with simplex projection
- Anneal the LSE temperature for progressive sharpening
- Try multiple sequence lengths and diverse initializations

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n, dtype=np.float64)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_temperature(c, target_frac=0.5):
    """Adaptive temperature: set t so that the top fraction of lags are 'active'."""
    sorted_c = np.sort(c)[::-1]
    k = max(1, int(target_frac * len(c)))
    # Set t so that c[0] - c[k] ≈ 1 (top k are roughly equally weighted)
    gap = c[0] - sorted_c[k]
    if gap < 1e-10:
        return 10.0
    return max(1.0, min(50.0, 1.0 / gap))


def compute_lse_grad(a, c, t):
    """Compute gradient of (1/t)*log(sum(exp(t*c))) w.r.t. a.
    
    Returns grad of length n.
    """
    n = len(a)
    # Soft weights
    tc = t * c
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    Z = exp_tc.sum()
    w = exp_tc / Z  # length 2n-1
    
    # Cross-correlation part: x[m] = sum_j a[j] * w[m+j]
    # x[m] = conv(a[::-1], w)[m + n - 1]
    a_rev = a[::-1]
    conv_size = n + (2 * n - 1) - 1  # = 3n - 2
    fsize = 1 << (conv_size - 1).bit_length()
    A_rev = fft(a_rev, fsize)
    W = fft(w, fsize)
    conv_full = np.real(ifft(A_rev * W))[:conv_size]
    x = conv_full[n - 1: 2 * n - 1]  # length n
    
    # Windowed sum part: s[m] = sum_{k=m}^{m+n-1} w[k]
    csum = np.zeros(2 * n)
    csum[1:] = np.cumsum(w)
    s = csum[n: 2 * n] - csum[:n]  # length n
    
    grad = x + a * s
    return grad


def run_lse_adam(a_init, n, S, max_time, rng, t_init=1.0, t_max=50.0):
    """Adam optimizer with LSE-smoothed objective and simplex projection."""
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t_count = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.1 * S / n
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    
    start = time.monotonic()
    last_print = start
    
    while time.monotonic() - start < max_time:
        t_count += 1
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Adaptive temperature
        t = t_init + (t_max - t_init) * min(1.0, t_count / (max_time * 20))
        t = min(t, lse_temperature(c, 0.3))
        
        grad = compute_lse_grad(a, c, t)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)
        
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        
        # Learning rate decay
        lr = 0.1 * (S / n) * (0.5 ** (float(t_count) / 500.0))
        
        # Periodic restart with noise

            a = project_simplex(a + rng.normal(0.0, 0.005 * S / np.sqrt(n), n), S)
        
        if time.monotonic() - last_print > 30:
            print("  adam iter %d, maxc=%.6f, t=%.2f, lr=%.2e" % (t_count, cur, t, lr))
            last_print = time.monotonic()
    
    return best_local, best_score


def run_multiplicative(a_init, n, S, max_time, rng, t_init=1.0, t_max=30.0):
    """Multiplicative Weights (mirror descent) with LSE gradient."""
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    a = np.maximum(a, S / (10 * n))  # Ensure positivity for log

    best_local = a.copy()

    t_count = 0
    eta = 0.5

        t_count += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:

            best_local = a.copy()
```

The objective (normalized) is to minimize `max(c)` where `c = a * a` (self-convolution) and `sum(a) = sqrt(2n)`, `a >= 0`.

The LSE approximation is `f(a) = (1/t) * log(sum_k exp(t * c[k]))`.

The gradient is:
`grad[m] = sum_k w[k] * dc[k]/da[m]`

where `w[k] = exp(t*c[k]) / sum_k exp(t*c[k])`.

Now, `c[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]`.

`dc[k]/da[m]`:
- If `0 <= k-m < n`: the term `a[m] * a[k-m]` appears in the sum (when j=m).
  - If `k-m != m` (i.e., `k != 2m`): the term `a[k-m] * a[m]` also appears (when j=k-m).
  - So `dc[k]/da[m] = a[k-m] + a[m]`.
- If `k = 2m` (and `0 <= m < n`): `dc[k]/da[m] = 2*a[m]` (same as `a[k-m] + a[m] = a[m] + a[m]`).
- Otherwise: `dc[k]/da[m] = 0`.

So in all cases where `0 <= k-m < n`: `dc[k]/da[m] = a[k-m] + a[m]`.

Therefore:
`grad[m] = sum_{k: 0<=k-m<n} w[k] * (a[k-m] + a[m])`
`= sum_{k: 0<=k-m<n} w[k] * a[k-m] + a[m] * sum_{k: 0<=k-m<n} w[k]`
`= sum_{j=0}^{n-1} w[m+j] * a[j] + a[m] * sum_{j=0}^{n-1} w[m+j]`

```python
"""
Minimize 2*n*max_conv/sum^2 for non-negative sequences.

Algorithm:
- Normalize to sum = sqrt(2n) so objective = max(self-convolution).
- Use log-sum-exp (LSE) smoothing to get a smooth surrogate of max.
- Compute exact gradient of LSE in O(n log n) via FFT-based cross-correlation.
- Adam optimizer with simplex projection and adaptive LSE temperature.
- Multiply (mirror descent) as an alternative that respects non-negativity.
- Multiple sequence lengths (4096-32768) and diverse initializations.
- Polish best solution with high LSE temperature.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a (length n) via FFT, returns length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n, dtype=np.float64)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(a, c, t):
    """Exact gradient of (1/t)*log(sum_k exp(t*c[k])) w.r.t. a.
    
    c = a*a (self-convolution, length 2n-1).
    Returns gradient of length n.
    
    grad[m] = sum_j a[j]*w[m+j] + a[m]*sum_j w[m+j]
    where w[k] = exp(t*c[k]) / sum(exp(t*c[k])).
    """
    n = len(a)
    tc = t * c
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    Z = exp_tc.sum()
    w = exp_tc / Z  # length 2n-1

    # Part 1: x[m] = sum_{j=0}^{n-1} a[j] * w[m+j]
    # This is conv(a[::-1], w) at indices [n-1, 2n-2]
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # 3n - 2
    fsize = 1 << (conv_len - 1).bit_length()
    A_rev = fft(a_rev, fsize)
    W_mat = fft(w, fsize)
    conv_full = np.real(ifft(A_rev * W_mat))[:conv_len]
    x = conv_full[n - 1: 2 * n - 1]

    # Part 2: s[m] = sum_{k=m}^{m+n-1} w[k] via cumsum
    csum = np.zeros(2 * n, dtype=np.float64)
    csum[1:] = np.cumsum(w)
    s = csum[n: 2 * n] - csum[:n]

    return x + a * s


def compute_topk_grad(a, c, top_k=10):
    """Subgradient using top-k active lags (fallback for numerical issues)."""
    n = len(a)
    idx = np.argsort(c)[::-1][:top_k]
    vals = c[idx]
    total = vals.sum()
    if total < 1e-15:
        return np.zeros(n)
    weights = vals / total
    grad = np.zeros(n, dtype=np.float64)
    for w_val, k in zip(weights, idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def run_adam_lse(a_init, n, S, max_time, rng, t_init=2.0, t_max=80.0, lr_scale=1.0):
    """Adam with LSE-smoothed objective and simplex projection."""
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t_count = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.5 * (S / n) * lr_scale

    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    last_print = start

    while time.monotonic() - start < max_time:
        t_count += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Adaptive temperature based on the gap between top lags
        sorted_c = np.sort(c)[::-1]
        gap = c.max() - sorted_c[min(5, len(c) - 1)]
        t_adapt = max(t_init, min(t_max, 3.0 / max(gap, 1e-6)))
        # Also anneal over time
        t_anneal = t_init + (t_max - t_init) * min(1.0, t_count / max(1, int(max_time * 15)))
        t = min(t_adapt, t_anneal)

        grad = compute_lse_grad(a, c, t)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)

        lr = base_lr * (0.5 ** (t_count / 800.0))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)

        # Noise injection for exploration
        if t_count % 3000 == 0 and rng.random() < 0.08:
            noise = rng.normal(0.0, 0.01 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

        if time.monotonic() - last_print > 15:
            print("    adam t=%d maxc=%.6f t_lse=%.2f lr=%.2e" % (t_count, cur, t, lr))
            last_print = time.monotonic()

    return best_local, best_score


def run_multiply_lse(a_init, n, S, max_time, rng, t_init=2.0, t_max=60.0):
    """Multiplicative weights (mirror descent) with LSE gradient."""
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    a = np.maximum(a, S / (20.0 * n))
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    t_count = 0
    last_print = start

    while time.monotonic() - start < max_time:
        t_count += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        sorted_c = np.sort(c)[::-1]
        gap = c.max() - sorted_c[min(5, len(c) - 1)]
        t = min(t_max, max(t_init, 3.0 / max(gap, 1e-6)))
        t = min(t, t_init + (t_max - t_init) * min(1.0, t_count / max(1, int(max_time * 10))))

        grad = compute_lse_grad(a, c, t)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break
        grad = grad / gmax

        eta = 0.3 * max(0.02, 1.0 / (1.0 + 0.003 * t_count))
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15 * S / n)
        s = a_new.sum()
        if s < 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / s)

        if time.monotonic() - last_print > 15:
            print("    mult t=%d maxc=%.6f eta=%.4f" % (t_count, cur, eta))
            last_print = time.monotonic()

    return best_local, best_score


def hybrid_run(a_init, n, S, max_time, rng):
    """Run Adam then switch to multiplicative for the remainder."""
    t0 = time.monotonic()
    adam_time = max_time * 0.55
    a1, s1 = run_adam_lse(a_init, n, S, adam_time, rng)
    elapsed = time.monotonic() - t0
    left = max_time - elapsed - 0.5
    if left > 2.0:
        a2, s2 = run_multiply_lse(a1, n, S, left, rng)
        if s2 < s1:
            return a2, s2
    return a1, s1


def make_inits(rng, n):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.linspace(0, 1, n)

    inits.append(("uniform", np.ones(n, dtype=np.float64)))

    inits.append(("random", rng.random(n).astype(np.float64)))

    # Triangle
    tri = np.maximum(0.0, 1.0 - 2.0 * np.abs(x - 0.5))
    inits.append(("triangle", tri + 0.05))

    # Two bumps (mass at edges, valley in middle)
    w = 0.15
    b1 = np.exp(-0.5 * ((x - 0.2) / w) ** 2)
    b2 = np.exp(-0.5 * ((x - 0.8) / w) ** 2)
    inits.append(("two_bump", b1 + b2 + 0.02))

    # Sine window
    inits.append(("sine_win", np.sin(np.pi * x) ** 2 + 0.05))

    # Edge concentration
    inits.append(("edges", np.exp(-6 * x) + np.exp(-6 * (1 - x)) + 0.03))

    # Smooth random (low-pass filtered)
    nfreq = n // 2 + 1
    spec = rng.standard_normal(nfreq) / (1.0 + 3.0 * np.abs(np.arange(nfreq) / n) * 10)
    smooth = np.fft.irfft(spec * n, n)
    inits.append(("smooth", np.abs(smooth) + 0.1))

    # Step-like transitions (sharp features)
    steps = np.zeros(n)
    n_steps = 4
    for i in range(n_steps):
        start = i * n // n_steps
        end = (i + 1) * n // n_steps
        steps[start:end] = 0.5 + 0.5 * np.sin(np.pi * (x[start:end] - x[start]) / max(1, x[end] - x[start]))
    inits.append(("steps", steps + 0.05))

    return inits


def safe_norm(a, n):
    """Clip to [0,1000] and rescale to sum sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s < 1e-15:
        return np.full(n, S / n, dtype=np.float64)
    return a * (S / s)


def polish_sequence(a, budget, rng):
    """Polish a single sequence with high LSE temperature."""
    n = len(a)
    S = np.sqrt(2.0 * float(n))
    a = safe_norm(a, n)

    results = []
    half = budget / 2.0

    if half > 2.0:
        a1, s1 = run_adam_lse(a, n, S, half, rng, t_init=15.0, t_max=200.0, lr_scale=0.3)
        results.append((a1, s1))

    if budget - half > 2.0:
        a2, s2 = run_multiply_lse(a, n, S, budget - half - 1.0, rng, t_init=20.0, t_max=200.0)
        results.append((a2, s2))

    if not results:
        return a

    best_a = min(results, key=lambda x: x[1])
    return best_a[0]

    print("Total time: %.1fs" % (time.monotonic() - start_time))

    return [float(x) for x in global_a]
```
