def convolve(seq):
    """FFT-convolution faster than numpy"""
    n = len(seq)
    arr_full = np.zeros(2*n-1)
    for i in range(n):
        for j in range(n):
            k = i+j
            if k < 2*n-1:
                arr_full[k] += seq[i]*seq[j]
            # This is O(n^2), so we use FFT-style looping
    return arr_full

def generate_multi_tone_init(length):
    """Multiple sine waves with different frequencies"""

def softmax_normalize(theta):
    """Traditional softmax for posteriors"""
    score_val = theta / (np.sum(theta)[:,None])
    score_soft_val = np.exp(score_val) / np.sum(np.exp(score_val), axis=1, keepdims=True)
