"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Uses multi-phase Adam optimization with softmax-weighted subgradients,
starting from the previous best solution. Employs adaptive top-k 
and learning rate schedules. Adds LP-based direction finding for 
final polishing and perturbation-based restarts to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_softmax_subgradient(a, c, tau, top_k=None):
    """Compute softmax-weighted subgradient of max(c).
    
    Uses exp(c_k / tau) weighting over top-k lags (or all if top_k is None).
    """
    n = len(a)
    if top_k is not None and top_k < len(c):
        # Get top-k indices
        top_indices = np.argpartition(c, -top_k)[-top_k:]
        top_vals = c[top_indices]
    else:
        top_indices = np.arange(len(c))
        top_vals = c
    
    # Softmax weights
    shifted = top_vals - top_vals.max()
    weights = np.exp(shifted / tau)
    total = weights.sum()
    if total <= 1e-30:
        return np.zeros(n)
    weights = weights / total
    
    grad = np.zeros(n, dtype=np.float64)
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - lo:k - hi + 1][::-1]
    return grad


def compute_topk_subgradient(a, c, top_k):
    """Weighted subgradient using top-k lags with uniform weights."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    grad = np.zeros(n, dtype=np.float64)
    for k in top_indices:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += (2.0 / top_k) * a[k - lo:k - hi + 1][::-1]
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
    return a * (S / s)


def run_adam_v2(a_init, n, S, max_time, rng, lr_init, top_k_start, top_k_end, 
                tau, use_softmax=True):
    """Improved Adam with adaptive top-k and softmax subgradient."""
    a = safe_normalize(a_init, n)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    patience = 0
    best_seen = best_score
    
    while time.monotonic() - start < max_time:
        t += 1
        # Adaptive top-k: interpolate from start to end
        frac = min(1.0, t / (max_time * 60))  # roughly
        cur_top_k = int(top_k_start + (top_k_end - top_k_start) * frac)
        cur_top_k = max(1, min(cur_top_k, 2 * n - 1))
        
        # Learning rate with cosine decay
        progress = min(1.0, t / (max_time * 30))
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, lr_init * 0.01)
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            patience = 0
        else:
            patience += 1
        
        # Patience-based early stopping
        if patience > 5000 and t > 1000:
            break
        
        if use_softmax:
            grad = compute_softmax_subgradient(a, c, tau, cur_top_k)
        else:
            grad = compute_topk_subgradient(a, c, cur_top_k)
        
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-12:
            break
        grad = grad / gnorm
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

    return best_local, best_score


def run_mirror_v2(a_init, n, S, max_time, rng, eta_init, top_k, tau):
    """Improved mirror descent with softmax subgradient."""
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    eta = eta_init
    start = time.monotonic()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        grad = compute_softmax_subgradient(a, c, tau, top_k)
        gmax = np.abs(grad).max()
        if gmax <= 1e-12:
            break
        grad = grad / gmax
        
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / s)
        
        eta = eta_init * max(0.01, 1.0 / (1.0 + 0.002 * t))
    
    return best_local, best_score


def run_lp_polish(a_init, n, S, max_time, top_k=20):
    """LP-based direction finding for final polishing.
    
    At each iteration, solve: maximize sum(b) s.t. b >= 0, (a*b)_k <= M for all k
    where M = max(a*a). Then move toward b.
    """
    from scipy.optimize import linprog
    
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    M = c.max()
    best_score = M
    best_local = a.copy()
    start = time.monotonic()
    
    n_constraints = 2 * n - 1
    # We'll use only the top-k lags as constraints to keep LP tractable
    # Actually for n=8192, even 16383 constraints with 8192 vars is large
    # Let's use top_k constraints
    
    iters = 0
    while time.monotonic() - start < max_time:
        iters += 1
        c = fft_convolve_self(a)
        M = c.max()
        if M < best_score:
            best_score = M
            best_local = a.copy()
        
        # Get top-k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # Build LP: maximize sum(b) s.t. A*b <= M, b >= 0
        # A has shape (top_k, n)
        # A[i, j] = a[top_idx[i] - j] if 0 <= top_idx[i] - j < n else 0
        A = np.zeros((len(top_idx), n), dtype=np.float64)
        for i, k in enumerate(top_idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[i, lo:hi + 1] = a[k - lo:k - hi + 1][::-1]
        
        c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
        
        res = linprog(c_obj, A_ub=A, b_ub=np.full(len(top_idx), M), 
                      bounds=[(0, None)] * n, method='highs')
        
        if res.status != 0:
            break
        
        b = res.x
        s_b = b.sum()
        if s_b <= 1e-15:
            break
        
        # Rescale and interpolate
        b_scaled = b * (S / s_b)
        
        # Move toward b with diminishing step
        gamma = 0.5 / (1.0 + 0.1 * iters)
        a = (1.0 - gamma) * a + gamma * b_scaled
        a = np.maximum(a, 0.0)
        a = a * (S / a.sum())
        
        # Update M
        c = fft_convolve_self(a)
    
    return best_local, best_score

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
  1. Start from the previous best solution (height_sequence_1) as primary init.
  2. Use multi-phase Adam with softmax-weighted subgradients and adaptive top-k.
  3. Apply LP-based direction finding for final polishing (exploits the
     linear structure of the constraint (a*b)_k <= M).
  4. Cooldown cycles: small perturbations followed by short fine-tuning runs
     to escape shallow local minima.
  5. Try multiple sequence lengths (4096, 8192, 16384) to exploit the
     additional degrees of freedom.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_softmax_subgradient(a, c, tau, top_k):
    """Softmax-weighted subgradient of max(c) over top-k lags."""
    n = len(a)
    total_lags = len(c)
    k = min(top_k, total_lags)
    if k == total_lags:
        top_indices = np.arange(total_lags)
        top_vals = c
    else:
        top_indices = np.argpartition(c, -k)[-k:]
        top_vals = c[top_indices]
    
    shifted = top_vals - top_vals.max()
    weights = np.exp(shifted / max(tau, 1e-8))
    total_w = weights.sum()
    if total_w <= 1e-30:
        return np.zeros(n)
    weights = weights / total_w
    
    grad = np.zeros(n, dtype=np.float64)
    for w_val, k_idx in zip(weights, top_indices):
        k_i = int(k_idx)
        lo = max(0, k_i - n + 1)
        hi = min(n - 1, k_i)
        if lo <= hi:
            seg = a[lo:hi + 1][::-1]
            grad[lo:hi + 1] += w_val * 2.0 * seg
    return grad


def compute_topk_subgradient(a, c, top_k):
    """Uniform-weighted subgradient over top-k lags."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    grad = np.zeros(n, dtype=np.float64)
    inv_k = 2.0 / k
    for k_i in top_indices:
        k_val = int(k_i)
        lo = max(0, k_val - n + 1)
        hi = min(n - 1, k_val)
        if lo <= hi:
            grad[lo:hi + 1] += inv_k * a[lo:hi + 1][::-1]
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    return a * (S / s)


def run_adam(a_init, n, S, max_time, rng, lr_init, top_k_start, top_k_end,
             tau_init, tau_end, use_softmax=True, max_iters=None):
    """Adam with adaptive top-k, softmax subgradient, cosine LR decay."""
    a = safe_normalize(a_init, n)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    c = fft_convolve_self(a)
    best_score = float(c.max())
    best_local = a.copy()
    start = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - start < max_time and (max_iters is None or t < max_iters):
        t += 1
        frac = min(1.0, t / max(1.0, max_time * 20.0))
        cur_top_k = int(top_k_start + (top_k_end - top_k_start) * frac)
        cur_top_k = max(3, min(cur_top_k, 2 * n - 1))
        cur_tau = tau_init + (tau_end - tau_init) * frac
        
        progress = min(1.0, t / max(1.0, max_time * 15.0))
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, lr_init * 0.005)
        
        c = fft_convolve_self(a)
        cur = float(c.max())
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 3000 and t > 500:
            break
        
        if use_softmax:
            grad = compute_softmax_subgradient(a, c, cur_tau, cur_top_k)
        else:
            grad = compute_topk_subgradient(a, c, cur_top_k)
        
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-14:
            break
        grad /= gnorm
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        
        if t % 3000 == 0 and rng.random() < 0.1:
            noise = rng.normal(0.0, 0.01 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
    
    return best_local, best_score


def run_mirror(a_init, n, S, max_time, rng, eta_init, top_k, tau):
    """Mirror descent with softmax subgradient."""
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = float(c.max())
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = float(c.max())
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()
        
        grad = compute_softmax_subgradient(a, c, tau, top_k)
        gmax = np.abs(grad).max()
        if gmax <= 1e-14:
            break
        grad /= gmax
        
        a_new = a * np.exp(-eta_init * 0.5 * (1 + np.cos(np.pi * min(1.0, t / (max_time * 10.0)))) * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / s)
    
    return best_local, best_score


def run_lp_direction(a_init, n, S, max_time, n_constraints=10, max_iters=30):
    """LP-based direction finding: solve max sum(b) s.t. (a*b)_k <= M, b >= 0.
    
    Move toward the LP solution with diminishing step size.
    """
    from scipy.optimize import linprog
    
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = float(c.max())
    best_local = a.copy()
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start >= max_time:
            break
        
        c = fft_convolve_self(a)
        M = float(c.max())
        if M < best_score - 1e-10:
            best_score = M
            best_local = a.copy()
        
        top_idx = np.argsort(c)[::-1][:n_constraints]
        
        A = np.zeros((n_constraints, n), dtype=np.float64)
        for i, k in enumerate(top_idx):
            k_val = int(k)
            lo = max(0, k_val - n + 1)
            hi = min(n - 1, k_val)
            if lo <= hi:
                A[i, lo:hi + 1] = a[lo:hi + 1][::-1]
        
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(n_constraints, M, dtype=np.float64)
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs')
        except Exception:
            break
        
        if res.status != 0 or res.x is None:
            break
        
        b = res.x
        s_b = b.sum()
        if s_b <= 1e-15:
            break
        
        b_scaled = b * (S / s_b)
        gamma = 0.3 / (1.0 + 0.05 * it)
        a = (1.0 - gamma) * a + gamma * b_scaled
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a > 1e-15:
            a *= S / s_a
    
    return best_local, best_score


def cooldown_cycle(a_best, n, S, budget, rng, lr_scale, top_k, tau):
    """Perturb and re-optimize with small steps."""
    a = a_best.copy()
    noise_scale = 0.005 * S / np.sqrt(n)
    a = project_simplex(a + rng.normal(0.0, noise_scale, n), S)
    
    result_a, result_score = run_adam(
        a, n, S, budget, rng,
        lr_init=lr_scale * S / n,
        top_k_start=top_k, top_k_end=top_k * 2,
        tau_init=tau, tau_end=tau * 0.5
    )
    return result_a, result_score
