def surrogate_from_c(c, T, cmax=None):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0
    if cmax is None:
        cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return 0.0
    T = float(T)
    if not np.isfinite(T) or T <= 0.0:
        return float(cmax)
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return float(cmax)
    return float(cmax + T * np.log(s))


def conv_sq(p):

    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)

def gradient_and_c(p, T_factor):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:

    L = _fft_grad_len(n)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]

    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.zeros(n, dtype=np.float64), c.astype(np.float64, copy=False), 1.0, 0.0

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def project_simplex_symmetric(v, n):
    """Project onto symmetric simplex: a[j] = a[n-1-j], a >= 0, sum = 1."""
    half = (n + 1) // 2
    v_half = v[:half].copy()
    v_half = np.where(np.isfinite(v_half), v_half, 0.0)
    if n % 2 == 0:
        weights = np.ones(half)
        weights *= 2.0
    else:
        weights = np.ones(half)
        weights[-1] = 1.0
    # sum = sum(v_half * weights) = 1
    # Normalize: find theta such that sum(max(v_half - theta, 0) * weights) = 1
    # This is a weighted simplex projection.
    # Simple approach: iterate
    for _ in range(50):
        s = float(np.dot(v_half, weights))
        if abs(s - 1.0) < 1e-14:
            break
        delta = (s - 1.0) / float(weights.sum())
        v_half = v_half - delta
        v_half = np.maximum(v_half, 0.0)
        s2 = float(np.dot(v_half, weights))
        if s2 > 0:
            v_half = v_half / s2
        else:
            v_half = np.full(half, 1.0 / float(weights.sum()))
            break
    # Build full sequence
    p = np.zeros(n, dtype=np.float64)
    if n % 2 == 0:
        p[:half] = v_half
        p[half:] = v_half[::-1]
    else:
        p[:half] = v_half
        p[half + 1:] = v_half[:half][::-1]
    s = float(p.sum())
    if s > 1e-12:
        p /= s
    else:
        p = np.full(n, 1.0 / n)
    return p


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def make_raised_cosine(n, alpha=1.0):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = 0.5 + 0.5 * np.cos(np.pi * alpha * (x - 0.5))
    return safe_normalize(np.maximum(p, 0.0))


def make_gaussian(n, sigma_ratio=0.3, mu=0.5):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    sig = max(1e-4, sigma_ratio)
    p = np.exp(-((x - mu) ** 2) / (2.0 * sig ** 2))
    return safe_normalize(p)


def make_beta_dist(n, a, b):
    n = int(max(2, n))
    x = np.linspace(1e-6, 1.0 - 1e-6, n)
    p = np.power(x, a) * np.power(1.0 - x, b)
    return safe_normalize(np.maximum(p, 0.0))


def make_two_gauss(n, mu1, sig1, w1, mu2, sig2, w2):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = w1 * np.exp(-((x - mu1) ** 2) / (2.0 * max(1e-4, sig1) ** 2))
    p += w2 * np.exp(-((x - mu2) ** 2) / (2.0 * max(1e-4, sig2) ** 2))
    return safe_normalize(np.maximum(p, 0.0))


def make_sinc_taper(n, freq=1.0):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    t = (x - 0.5) * 2.0  # -1 to 1
    p = np.sinc(freq * t)
    p = np.maximum(p, 0.0)
    return safe_normalize(p)


def make_cheb_taper(n, deg=6, coefs=None):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    z = 2.0 * x - 1.0
    if coefs is None:
        coefs = np.array([0.0, 0.3, -0.15, 0.08, -0.04, 0.02, -0.01][:deg+1])
    coefs = np.asarray(coefs, dtype=np.float64)
    if coefs.size == 0:
        return safe_normalize(np.ones(n))
    T_nm = np.ones(n)
    T_n = z.copy()
    logp = np.zeros(n)
    for d in range(len(coefs)):
        if d == 0:
            T_d = np.ones(n)
        elif d == 1:
            T_d = z
        else:
            T_np = 2.0 * z * T_n - T_nm
            T_d = T_np
            T_nm = T_n
            T_n = T_np
        if d == 2:
            T_n = T_np
        logp += coefs[d] * T_d
    # Endpoint taper
    eps = 1e-4
    logp += 1.5 * np.log(x + eps) + 1.5 * np.log(1.0 - x + eps)
    m = float(np.max(logp))
    p = np.exp(logp - m)
    return safe_normalize(p)


def make_random_smooth(n, rng, strength=1.0):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0
    for _ in range(4):
        f = int(rng.integers(1, max(2, n // 50)))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.1, 0.5)) * strength
        p += amp * np.sin(2.0 * np.pi * f * x + ph)
    p = np.maximum(p, 0.02)
    return safe_normalize(p)

def reshape_sequence(p, target_n):
    """Reshape a sequence to a new length using interpolation."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    target_n = int(max(2, target_n))
    if n == target_n:
        return p
    if n < 2:
        return np.full(target_n, 1.0 / target_n)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, target_n)
    p_new = np.interp(x_new, x_old, p)
    return safe_normalize(p_new)


def lp_best_response(p, budget_s):
    if not _HAVE_LP:
        return None, np.inf
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 10 or n > LP_NMAX:
        return None, np.inf
    if budget_s <= 0.5:
        return None, np.inf

    c = conv_sq(p)
    M = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(M) or M <= 0.0:
        return None, np.inf

    f = p + 1e-14
    fs = float(f.sum())
    if not np.isfinite(fs) or fs <= 0.0:
        return None, np.inf
    f = f / fs

    end = time.monotonic() + max(0.1, budget_s)
    if end - time.monotonic() < 0.3:
        return None, np.inf

    try:
        mid = np.arange(n, dtype=np.int32)
        rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel().astype(np.int32)
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(f, n).astype(np.float64, copy=False)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs",
                      options={"disp": False})
    except Exception:
        return None, np.inf

    if res is None:
        return None, np.inf
    g = getattr(res, "x", None)
    if g is None:
        return None, np.inf
    g = np.asarray(g, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None, np.inf
    g = g / s
    return g, 0.0


def optimize_local(p0, rng, time_limit, label="", allow_lp=False, symmetric=False):
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    if n == 0:
        return p, np.inf

    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score
    prev_p = p.copy()

    end = time.monotonic() + max(0.0, float(time_limit))
    iter_no = 0
    since_improve = 0
    T_factor = 0.5
    T_cool = 0.998
    T_min = 0.002

    if allow_lp and n <= LP_NMAX:
        lp_budget = min(30.0, max(3.0, float(time_limit) * 0.1))
        g, _ = lp_best_response(p, lp_budget)
        if g is not None:
            for t_mix in (1.0, 0.7, 0.5, 0.3, 0.15):
                if symmetric:
                    q = project_simplex_symmetric((1.0 - t_mix) * p + t_mix * g, n)
                else:
                    q = project_simplex((1.0 - t_mix) * p + t_mix * g)
                sc = internal_score(q)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    current_score = sc
                    best_p = q.copy()
                    p = q.copy()
                    prev_p = q.copy()
                    since_improve = 0
                    break

    while time.monotonic() < end:
        iter_no += 1
        T_factor = max(T_min, T_factor * T_cool)
        if since_improve > 200:
            T_factor = max(T_min, T_factor * 0.95)

        g, c_cur, T, surr = gradient_and_c(p, T_factor)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)

        accepted = False
        pmax = float(np.max(p)) if n > 0 else 0.0
        p_scale = max(1.0 / float(n), 0.03 * pmax, 1e-12)
        gabs = float(np.max(np.abs(g))) if g.size else 0.0

        if gabs > 1e-300:
            base = p_scale / gabs
            lrs = (0.5 * base, 0.15 * base, 0.04 * base, 0.01 * base, 0.003 * base, 0.001 * base)
        else:
            lrs = (0.1, 0.05, 0.02, 0.005)

        momentum = 0.0
        if iter_no > 80:
            momentum = 0.3

        for lr in lrs:
            lr = float(lr)
            if not np.isfinite(lr) or lr <= 0.0:
                continue
            lr = min(max(lr, 1e-9), 10.0)
            v = p - lr * g
            if momentum > 0.0:
                v = v + momentum * (p - prev_p)
            if symmetric:
                pt = project_simplex_symmetric(v, n)
            else:
                pt = project_simplex(v)

            c_t = conv_sq(pt)
            cmax_t = float(np.max(c_t)) if c_t.size else 0.0
            s_t = float(pt.sum())
            score_t = float(2.0 * n * cmax_t / (s_t * s_t)) if s_t > 1e-12 else np.inf
            surr_t = surrogate_from_c(c_t, T, cmax_t)

            accept = False
            if np.isfinite(score_t) and np.isfinite(surr_t):
                if surr_t < surr - 1e-12 * max(1.0, abs(surr)):
                    accept = True
                elif score_t < best_score - 1e-12:
                    if surr_t < surr + max(1.0, surr):
                        accept = True
                elif score_t < current_score - 1e-12:
                    if surr_t < surr + 0.5 * max(1.0, surr):
                        accept = True

            if accept:
                if score_t < best_score:
                    best_score = score_t
                    best_p = pt.copy()
                    since_improve = 0
                else:
                    since_improve += 1
                current_score = score_t
                prev_p = p
                p = pt
                accepted = True
                break

        if not accepted:
            T_factor = max(T_min, T_factor * 0.6)

            if since_improve > 100:
                sigmas = (1.0 / n, 0.2 / n, 0.05 / n, 0.01 / n, 0.002 / n)
            else:
                sigmas = (0.1 / n, 0.02 / n, 0.005 / n, 0.001 / n)

            for sigma in sigmas:
                noise = rng.normal(0.0, float(sigma), n)
                if symmetric:
                    noise = 0.5 * (noise + noise[::-1])
                pt = project_simplex(p + noise)
                sc = internal_score(pt)
                if np.isfinite(sc) and (sc < best_score - 1e-12 or sc < current_score - 1e-12):
                    if sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        since_improve = 0
                    else:
                        since_improve += 1
                    current_score = sc
                    prev_p = p
                    p = pt
                    accepted = True
                    break

            if not accepted and n > 1 and np.all(np.isfinite(g)):
                hi = int(np.argmax(g))
                lo = int(np.argmin(g))
                if hi != lo and g[hi] > g[lo] + 1e-15:
                    for frac in (0.3, 0.1, 0.03, 0.008, 0.002):
                        delta = float(frac) * float(p[hi])
                        if delta <= 0.0:
                            continue
                        pt = p.copy()
                        pt[hi] -= delta
                        pt[lo] += delta
                        if symmetric:
                            pt2 = p.copy()
                            pt2[n - 1 - hi] -= delta
                            pt2[n - 1 - lo] += delta
                            pt = 0.5 * (pt + pt2)
                        pt = project_simplex(pt)
                        sc = internal_score(pt)
                        if np.isfinite(sc) and (sc < best_score - 1e-12 or sc < current_score - 1e-12):
                            if sc < best_score:
                                best_score = sc
                                best_p = pt.copy()
                                since_improve = 0
                            else:
                                since_improve += 1
                            current_score = sc
                            prev_p = p
                            p = pt
                            accepted = True
                            break

            if not accepted:
                since_improve += 1
                if since_improve > 500:
                    mix = 0.4 + 0.3 * float(rng.random())
                    p = project_simplex(mix * best_p + (1.0 - mix) * p + rng.normal(0.0, 0.5 / n, n))
                    current_score = internal_score(p)
                    prev_p = p
                    T_factor = 0.4
                    since_improve = 0
                elif since_improve > 150:
                    p = project_simplex(0.8 * best_p + 0.2 * p + rng.normal(0.0, 0.1 / n, n))
                    current_score = internal_score(p)
                    prev_p = p

        if allow_lp and n <= LP_NMAX and iter_no % 1500 == 0:
            remaining = end - time.monotonic()
            if remaining > 10.0:
                g, _ = lp_best_response(p, min(30.0, remaining))
                if g is not None:
                    for t_mix in (1.0, 0.7, 0.5, 0.3, 0.1):
                        if symmetric:
                            q = project_simplex_symmetric((1.0 - t_mix) * p + t_mix * g, n)
                        else:
                            q = project_simplex((1.0 - t_mix) * p + t_mix * g)
                        sc = internal_score(q)
                        if np.isfinite(sc) and sc < best_score:
                            best_score = sc
                            current_score = sc
                            best_p = q.copy()
                            p = q.copy()
                            prev_p = q.copy()
                            since_improve = 0
                            break

        if iter_no % 3000 == 0:
            print(label + " iter=%d best=%.10f cur=%.10f Tf=%.4f" % (iter_no, best_score, current_score, T_factor), flush=True)

        if best_score < 1.4980 and since_improve > 8000:
            break

    if not np.all(np.isfinite(best_p)) or best_p.size == 0:
        best_p = project_simplex(p)
    else:
        best_p = project_simplex(best_p)
    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def shape_objective_global(theta, n):
    """Evaluate a parametric shape."""
    n = int(max(16, n))
    x = np.linspace(0.0, 1.0, n)
    z = 2.0 * x - 1.0

    theta = np.asarray(theta, dtype=np.float64)
    deg = min(int(theta.size) - 2, 8)
    if theta.size < 3:
        theta = np.array([0.0, 0.0, 0.0])

    logp = np.zeros(n, dtype=np.float64)
    T_nm = np.ones(n)
    T_n = z.copy()
    for d in range(min(deg, 7)):
        if d == 0:
            T_d = np.ones(n)
        elif d == 1:
            T_d = z
        else:
            T_np = 2.0 * z * T_n - T_nm
            T_d = T_np
            T_nm = T_n
            T_n = T_np
        logp += theta[d] * T_d

    eps = 1e-4
    ew = theta[deg] if theta.size > deg else 1.0
    logp += ew * np.log(x + eps) + ew * np.log(1.0 - x + eps) + 1.0

    if not np.all(np.isfinite(logp)):
        return np.inf
    m = float(np.max(logp))
    p = np.exp(logp - m)
    p = safe_normalize(p)
    return internal_score(p)

                                T_d = T_np

                                T_n = T_np
                            logp += res.x[d] * T_d
                        eps = 1e-4
                        ew = res.x[7] if res.x.size > 7 else 1.0
                        logp += ew * np.log(x + eps) + ew * np.log(1.0 - x + eps) + 1.0
                        p = np.exp(logp - np.max(logp))
                        best_p = safe_normalize(p)
            except Exception:
                pass

        for x0 in x0_list:
            if time.monotonic() > end - 5.0:
                break
            try:
                res = minimize(lambda th: shape_objective_global(th, n), x0,
                               method="Nelder-Mead",
                               options={"maxiter": 300, "xatol": 1e-5, "fatol": 1e-7})
                if res is not None and np.isfinite(getattr(res, "fun", np.inf)):
                    if res.fun < best_score:

                        x = np.linspace(0.0, 1.0, n)
                        z = 2.0 * x - 1.0

            except Exception:

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a nonnegative step function with low evaluate_sequence score."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 15.0)

    # Try to use parent sequence as starting point
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and float(parent_arr.sum()) > 0.0:
            parent = parent_arr
    except Exception:
        parent = None

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0 or not np.all(np.isfinite(p)):
            return

        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()

    candidate_lengths = [1024, 2048, 3072, 4096, 5120, 6144, 8192]

        pn = int(parent.size)
        if pn not in candidate_lengths:
            candidate_lengths.append(pn)

    if deadline - time.monotonic() > 180.0:

                break
