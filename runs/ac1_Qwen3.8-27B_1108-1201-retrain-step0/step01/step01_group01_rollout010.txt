"""
Hybrid LP + subgradient + smooth optimization to minimize the normalized
autocorrelation peak of a non-negative sequence. Uses LP-based direction
finding combined with fast subgradient steps and final smooth polishing.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog

def _score(a):
    """Compute 2*n*max(convolve(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    p = a / S
    b = np.convolve(p, p)
    return 2.0 * n * np.max(b)

def _normalize(a):
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / S

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    w = u * np.arange(1, n + 1) - (cssv - target)
    rho = np.nonzero(w > 0)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _precompute_pattern(n):
    """Precompute row/col indices for the sparse convolution constraint matrix."""
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    for j in range(n):
        s = j * n
        rows[s:s+n] = np.arange(j, j + n, dtype=np.int32)
        cols[s:s+n] = j
    return rows, cols

def _solve_lp(f, M, rows, cols, time_limit=120):
    """Solve LP: max sum(g) s.t. B@ g <= M, g >= 0, where B[k,j] = f[k-j]."""
    n = len(f)
    n_rows = 2 * n - 1
    data = np.tile(f, n)
    B = sp.coo_matrix((data, (rows, cols)), shape=(n_rows, n)).tocsr()
    
    c = -np.ones(n)
    b_ub = np.full(n_rows, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=B, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'time_limit': time_limit})
        if result.success and result.x is not None:
            return result.x
    except Exception as e:
        print(f"LP failed: {e}")
    return None

def _subgradient_step(a, k_peaks, weights, eta):
    """One subgradient step towards reducing max convolution peak."""
    n = len(a)
    grad = np.zeros(n)
    for w, k in zip(weights, k_peaks):
        if w < 1e-10:
            continue
        # grad of b_k w.r.t. a[j] = a[k-j] + a[j-k] (with bounds)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        j = np.arange(j_lo, j_hi + 1)
        # a[k-j] for j in [j_lo, j_hi]: k-j ranges from k-j_hi to k-j_lo
        # k-j_hi = k - min(n-1,k) and k-j_lo = k - max(0,k-n+1)
        grad[j] += w * a[k - j]
        # a[j-k] for j in [j_lo, j_hi]: j-k ranges from j_lo-k to j_hi-k
        # Valid when j-k >= 0, i.e., j >= k
        j2_lo = max(j_lo, k)
        j2_hi = j_hi
        if j2_lo <= j2_hi:
            j2 = np.arange(j2_lo, j2_hi + 1)
            grad[j2] += w * a[j2 - k]
    
    # Center gradient (for sum constraint)
    grad -= np.mean(grad)
    
    a_new = a - eta * grad
    return _project_simplex(a_new, 1.0)

def _soft_max_objective(a, tau):
    """Smooth approximation of 2n * max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = np.max(b)
    # Subtract max for numerical stability
    log_sum = max_b / tau + np.log(np.mean(np.exp((b - max_b) / tau)))
    soft_max = tau * log_sum
    return 2.0 * n * soft_max

def _soft_max_grad(a, tau):
    """Gradient of the soft-max objective."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = np.max(b)
    weights = np.exp((b - max_b) / tau)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, range(len(b))):
        if w < 1e-10:
            continue
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        j = np.arange(j_lo, j_hi + 1)
        grad[j] += w * a[k - j]
        j2_lo = max(j_lo, k)
        if j2_lo <= j_hi:
            j2 = np.arange(j2_lo, j_hi + 1)
            grad[j2] += w * a[j2 - k]
    
    return 2.0 * n * grad

def _make_two_level(n, eps):
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    return _normalize(np.maximum(a, 0.0))

def _make_smooth_anti(n, eps):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    return _normalize(np.maximum(a, 1e-6))
