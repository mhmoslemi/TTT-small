def _proj_simplex(v, s=1.0):

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]

        return np.full(n, s / n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _fista_optimize(a0, deadline, max_iters=5000, tau_init=10.0, tau_final=100.0):
    """FISTA with smoothed objective and simplex projection."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S  # normalize to sum=1
    
    y = a.copy()
    t = 1.0
    best = a.copy()
    best_score = _compute_obj(a)
    
    lr_init = 0.5 / tau_final  # step size (conservative)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        tau = tau_init + (tau_final - tau_init) * (it / max_iters)
        lr = lr_init * (1.0 + 0.5 * np.cos(np.pi * it / max_iters))
        lr = max(lr, 0.001 / tau)
        
        # Compute gradient at y
        g = _smoothed_grad(y, tau)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a - lr * g + ((t - 1.0) / t_new) * (a - y)
        y = _proj_simplex(y, s=1.0)
        t = t_new
        
        # Periodic evaluation
        if (it + 1) % 200 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    # Final evaluation
    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    
    return best, best_score

def _lp_refine(a, M, deadline, max_iters=100):

        top_lags = np.argsort(c)[::-1][:top_k]
        
        # Build LP with only active constraints
        # Primal: max sum(g) s.t. conv(a_norm, g)[m] <= M for m in top_lags, g >= 0
        A_rows = []
        for m in top_lags:
            row = np.zeros(n)
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            for j in range(lo, hi + 1):
                row[j] = a_norm[m - j]
            A_rows.append(row)
        
        A = np.array(A_rows)
        b = np.full(len(top_lags), M)
        c_obj = -np.ones(n)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break
        
        # Blend
        improved = False
        for t_blend in [0.7, 0.5, 0.3, 0.2, 0.1]:
            a_new = (1.0 - t_blend) * a_norm + t_blend * g
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-12:
                continue
            a_new = a_new / S_new
            s_new = _compute_obj(a_new)
            # Compare with current
            c_curr = np.convolve(a_norm, a_norm)
            M_curr = float(c_curr.max())
            if s_new < 2.0 * n * M_curr:  # since sum=1, obj = 2*n*max_c
                a_norm = a_new
                improved = True
                break
        
        if not improved:
            break
    
    return a_norm

def _gradient_descent_simple(a0, deadline, max_iters=5000, lr_init=0.1):
    """Simple projected gradient descent on peak autocorrelation."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S
    
    best = a.copy()
    best_score = _compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * it / max_iters))
        lr = max(lr, 0.001)
        
        c = np.convolve(a, a)
        m_star = int(c.argmax())
        
        # Gradient of max (subgradient)
        grad = np.zeros(n)
        lo = max(0, m_star - n + 1)
        hi = min(n - 1, m_star)
        for j in range(lo, hi + 1):
            grad[j] = 2.0 * a[m_star - j]
        
        # Average over all max-achieving lags for a better subgradient
        # (use top few)
        top_lags = np.argsort(c)[::-1][:min(10, n)]
        if len(top_lags) > 1 and c[top_lags[0]] - c[top_lags[-1]] < 1e-6 * c[top_lags[0]]:
            grad = np.zeros(n)
            for m in top_lags:
                lo_m = max(0, m - n + 1)
                hi_m = min(n - 1, m)
                for j in range(lo_m, hi_m + 1):
                    grad[j] += a[m - j]
            grad /= len(top_lags)
        
        a = a - lr * grad
        a = _proj_simplex(a, s=1.0)
        
        if (it + 1) % 500 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    
    return best, best_score

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-accelerated FISTA with smoothed (log-sum-exp) objective, simplex projection,
active-set LP refinement, and diverse initializations across multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.special import logsumexp


def _fast_conv_sq(a):
    """Compute a * a (linear convolution) using FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[:2 * n - 1]


def _fast_cross_corr(a_rev, w):
    """Compute cross-correlation conv(a_rev, w) using FFT."""
    n = len(a_rev)
    m = len(w)
    N = 1
    while N < n + m - 1:
        N *= 2
    A = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(w, N)
    return np.fft.irfft(A * W, N)


def _smoothed_grad_fast(a, tau):
    """Compute gradient of (1/tau)*logsumexp(tau*conv(a,a)) using FFT."""
    n = len(a)
    c = _fast_conv_sq(a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()

    # grad[j] = 2 * sum_k a[k] * w[j+k] = 2 * conv(a[::-1], w)[j+n-1]
    a_rev = a[::-1]
    conv_result = _fast_cross_corr(a_rev, w)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad


def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = _fast_conv_sq(a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _proj_simplex(v, s=1.0):
    """Project v onto the simplex {x: sum(x)=s, x>=0}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.full(n, s / n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fista_optimize(a0, deadline, max_iters=5000, tau_init=5.0, tau_final=150.0):
    """FISTA with smoothed objective and simplex projection (FFT-accelerated)."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    y = a.copy()
    t_mom = 1.0
    best = a.copy()
    best_score = _compute_obj(a)
    a_prev = a.copy()

    stall_count = 0
    stall_threshold = 500

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = it / max_iters
        tau = tau_init + (tau_final - tau_init) * progress
        lr = 1.0 / (tau + 1.0)
        lr *= 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, 0.005 / tau)

        # Compute gradient at y
        g = _smoothed_grad_fast(y, tau)

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a - lr * g + ((t_mom - 1.0) / t_new) * (a - y)
        y = _proj_simplex(y, s=1.0)
        t_mom = t_new

        # Periodic evaluation
        if (it + 1) % 300 == 0:
            s = _compute_obj(a)
            if s < best_score - 1e-9:
                best_score = s
                best = a.copy()
                stall_count = 0
            else:
                stall_count += 300

        # Restart mechanism: if stalled, add perturbation
        if stall_count >= stall_threshold and (it + 1) % 100 == 0:
            noise = np.random.default_rng(it).random(n) * 0.1
            a = a + noise
            a = _proj_simplex(a, s=1.0)
            y = a.copy()
            t_mom = 1.0
            stall_count = 0

    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score


def _gradient_descent_simple(a0, deadline, max_iters=10000, lr_init=0.3):
    """Projected gradient descent on peak autocorrelation with averaging over top lags."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    best = a.copy()
    best_score = _compute_obj(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = it / max_iters
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, 0.002)

        c = _fast_conv_sq(a)

        # Use top K lags for averaged subgradient
        max_val = c.max()
        threshold = max_val - 1e-8 * max_val
        top_lags = np.where(c >= threshold)[0]
        if len(top_lags) == 0:
            top_lags = np.array([c.argmax()])
        # Cap at 50 lags for efficiency
        if len(top_lags) > 50:
            top_lags = np.argsort(c)[::-1][:50]

        # Compute averaged gradient
        a_rev = a[::-1]
        # For each lag m, gradient contribution is 2*a[m-j] for valid j
        # Use cross-correlation: for a single lag m, grad = 2 * a[m - j]
        # Which is 2 * conv(a[::-1], impulse_at_m)[j+n-1]
        # For multiple lags, sum over them
        w_single = np.zeros(2 * n - 1)
        for m in top_lags:
            w_single[m] = 1.0
        w_single /= len(top_lags)

        conv_result = _fast_cross_corr(a_rev, w_single)
        grad = 2.0 * conv_result[n - 1:2 * n - 1]

        a = a - lr * grad
        a = _proj_simplex(a, s=1.0)

        if (it + 1) % 500 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score


def _lp_refine(a, deadline, max_iters=80, active_fraction=0.5):
    """LP-based refinement using active set (top constraints)."""
    n = len(a)
    if n > 1200:
        return a.copy()

    a = np.asarray(a, dtype=np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best = a.copy()
    best_score = _compute_obj(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fast_conv_sq(a)
        M = float(c.max())
        n_total = 2 * n - 1

        # Select active constraints (top fraction)
        k_active = max(n, int(n_total * active_fraction))
        top_lags = np.argsort(c)[::-1][:k_active]

        # Build LP: max sum(g) s.t. conv(a, g)[m] <= M for m in top_lags, g >= 0
        A = np.zeros((k_active, n))
        for idx, m in enumerate(top_lags):
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            # A[idx, j] = a[m-j] for j in [lo, hi]
            js = np.arange(lo, hi + 1)
            A[idx, js] = a[m - js]

        b = np.full(k_active, M)
        c_obj = -np.ones(n)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'time_limit': 10})
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break

        # Try blending
        improved = False
        for t_blend in [0.8, 0.6, 0.4, 0.25, 0.15, 0.1]:
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-12:
                continue
            a_new = a_new / S_new
            s_new = _compute_obj(a_new)
            if s_new < best_score - 1e-10:
                a = a_new
                best = a_new.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            # Try a small gradient nudge to escape
            c = _fast_conv_sq(a)
            m_star = int(c.argmax())
            grad = np.zeros(n)
            lo = max(0, m_star - n + 1)
            hi = min(n - 1, m_star)
            for j in range(lo, hi + 1):
                grad[j] = 2.0 * a[m_star - j]
            a = a - 0.01 * grad
            a = _proj_simplex(a, s=1.0)
            s = _compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s

    return best


def _make_starts(n, rng):
    """Generate diverse starting sequences of length n, normalized to sum=1."""
    starts = []
    i = np.arange(n, dtype=np.float64)

    # Uniform
    starts.append(np.ones(n) / n)

    # Random
    for _ in range(4):
        x = rng.random(n)
        x = np.maximum(x, 1e-6)
        starts.append(x / x.sum())

    # Cosine modulations
    for f in [1, 2, 3, 5, 7, 11, 13]:
        x = 0.3 + 0.7 * np.abs(np.cos(2 * np.pi * f * i / n))
        starts.append(x / x.sum())

    # Multi-frequency
    for f1, f2 in [(1, 3), (2, 5), (3, 7), (1, 7)]:
        x = np.abs(np.sin(2 * np.pi * f1 * i / n)) * np.abs(np.cos(2 * np.pi * f2 * i / n)) + 0.05
        starts.append(x / x.sum())

    # Gaussian bumps
    for width in [0.15, 0.3, 0.5]:
        center = n / 2.0
        x = np.exp(-0.5 * ((i - center) / (n * width)) ** 2)
        starts.append(x / x.sum())

    # Block structure
    for block in [4, 8, 16, 32, 64]:
        if n % block == 0:
            nb = n // block
            x = np.tile(rng.random(nb) + 0.1, nb)
            starts.append(x / x.sum())

    # Triangular / tent
    center = n / 2.0
    x = 1.0 - np.abs(i - center) / center
    x = np.maximum(x, 0.01)
    starts.append(x / x.sum())

    # power-law
    x = (i + 1) ** 0.5
    starts.append(x / x.sum())
    x = (i + 1) ** -0.5
    starts.append(x / x.sum())

    return starts


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Uses FFT-accelerated FISTA with smoothed objective,
    gradient descent, and LP-based refinement across multiple sequence lengths
    and diverse starting points.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    print(f"Starting optimization, budget={budget_s}s, deadline in {float(budget_s)-12:.0f}s")

    # Try to use previous best
    prev_seq = None
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            prev_seq = gs.copy()
            print(f"Loaded previous sequence of length {len(prev_seq)}, "
                  f"score={_compute_obj(prev_seq):.6f}")
    except NameError:
        pass

    # Build task list
    tasks = []

    # Priority 1: Refine previous best with multiple methods
    if prev_seq is not None:
        n_prev = len(prev_seq)
        a0 = prev_seq.copy()
        a0 = a0 / a0.sum()
        tasks.append(('fista', a0.copy(), f"prev_fista_{n_prev}"))
        tasks.append(('grad', a0.copy(), f"prev_grad_{n_prev}"))
        tasks.append(('lp', a0.copy(), f"prev_lp_{n_prev}"))

        # Interpolated versions
        for n_try in sorted(set([max(64, n_prev // 2), n_prev * 2 if n_prev * 2 <= 2048 else n_prev])):
            if n_try != n_prev:
                x_new = np.interp(np.linspace(0, 1, n_try),
                                  np.linspace(0, 1, n_prev), prev_seq)
                x_new = np.maximum(x_new, 0)
                x_new = x_new / x_new.sum()
                tasks.append(('fista', x_new, f"prev_interp_{n_try}"))

    # Priority 2: Explore other lengths
    lengths = [256, 384, 512, 768, 1024, 1536, 2048]
    for n in lengths:
        starts = _make_starts(n, rng)
        # Pick diverse subset
        picks = [0, 1, len(starts) // 3, 2 * len(starts) // 3, len(starts) - 1]
        seen = set()
        for idx in picks:
            if idx not in seen and idx < len(starts):
                seen.add(idx)
                tasks.append(('fista', starts[idx].copy(), f"fista_{n}_s{idx}"))

    # Shuffle non-priority tasks
    n_priority = 0
    if prev_seq is not None:
        n_priority = min(len(tasks), 4 + len(lengths) * 0)  # first few are priority
    # Actually let's just keep the first (number from prev) tasks as priority
    prev_tasks = [t for t in tasks if t[2].startswith("prev")]
    other_tasks = [t for t in tasks if not t[2].startswith("prev")]
    rng.shuffle(other_tasks)
    tasks = prev_tasks + other_tasks

    print(f"Total tasks: {len(tasks)}")

    # Time allocation: give more time to promising tasks
    time_per_task = max(30.0, (deadline - t_start) / max(1, len(tasks)))

    for method, a0, label in tasks:
        if time.monotonic() > deadline - 30:
            break

        # Allocate time for this task
        task_deadline = min(deadline, time.monotonic() + time_per_task * 1.5)
        if time.monotonic() > task_deadline - 10:
            task_deadline = deadline

        t0 = time.monotonic()
        try:
            if method == 'fista':
                seq, score = _fista_optimize(a0, task_deadline, max_iters=8000,
                                             tau_init=5.0, tau_final=200.0)
            elif method == 'grad':
                seq, score = _gradient_descent_simple(a0, task_deadline,
                                                       max_iters=20000, lr_init=0.5)
                # Follow with FISTA
                if time.monotonic() < deadline - 20:
                    seq2, score2 = _fista_optimize(seq, deadline - 10, max_iters=3000,
                                                    tau_init=50.0, tau_final=200.0)
                    if score2 < score:
                        seq, score = seq2, score2
            elif method == 'lp':
                # Gradient first, then LP
                seq, score = _gradient_descent_simple(a0, task_deadline - 30,
                                                       max_iters=5000, lr_init=0.5)
                if time.monotonic() < deadline - 10:
                    seq2 = _lp_refine(seq, deadline - 5, max_iters=50)
                    score2 = _compute_obj(seq2)
                    if score2 < score:
                        seq, score = seq2, score2
            else:
                continue

            elapsed = time.monotonic() - t0
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [{label}] NEW BEST: score={best_score:.6f}, n={len(seq)}, "
                      f"time={elapsed:.1f}s")
            else:
                if time.monotonic() - t_start < 60:  # only print early on
                    print(f"  [{label}] score={score:.6f}, n={len(seq)}, time={elapsed:.1f}s")

        except Exception as e:
            print(f"  [{label}] Error: {e}")

    # Final intensive refinement of best sequence
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\nFinal refinement phase (current best={best_score:.6f})...")

        remaining_time = deadline - time.monotonic() - 5
        n_refine_passes = max(1, int(remaining_time / 30))

        for pass_i in range(n_refine_passes):
            if time.monotonic() > deadline - 5:
                break

            # FISTA with high tau
            t_pass = deadline - 5 - (n_refine_passes - 1 - pass_i) * 25
            seq, score = _fista_optimize(best_seq, t_pass, max_iters=5000,
                                         tau_init=80.0, tau_final=300.0)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  FISTA refine {pass_i}: score={best_score:.6f}")

            # LP refinement
            if time.monotonic() < deadline - 10 and len(best_seq) <= 1200:
                seq2 = _lp_refine(best_seq, deadline - 5, max_iters=40)
                score2 = _compute_obj(seq2)
                if score2 < best_score:
                    best_score = score2
                    best_seq = seq2.copy()
                    print(f"  LP refine {pass_i}: score={best_score:.6f}")

            # Gradient polish
            if time.monotonic() < deadline - 8:
                seq3, score3 = _gradient_descent_simple(best_seq, deadline - 5,
                                                        max_iters=5000, lr_init=0.05)
                if score3 < best_score:
                    best_score = score3
                    best_seq = seq3.copy()
                    print(f"  Grad refine {pass_i}: score={best_score:.6f}")

    # Final validation
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence validation: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
