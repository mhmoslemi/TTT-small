def compare_sequences(seq1, seq2, alpha=0.1, beta=0.9):
    """
    Compares two sequences using peak convolution and returns a weighted combination.
    alpha: weight for sequence changes (lower is more cautious)
    beta: used for normalization by length (higher values normalize better)

    conv1 = np.convolve(seq1, seq1, mode='full')
