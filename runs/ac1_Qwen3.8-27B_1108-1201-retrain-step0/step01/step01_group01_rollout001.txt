"""
Multi-stage optimization of non-negative sequence minimizing peak autocorrelation.
Uses log-sum-exp smoothing of the max function with L-BFGS-B and Adam optimizers,
T-annealing schedule, and multiple diverse starting points.
"""

import numpy as np
import time
from scipy.optimize import minimize


def _softmax(x):
    """Stable softmax."""
    x = np.asarray(x, dtype=np.float64)
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    s = np.sum(e)
    if s < 1e-300:
        return np.full(len(x), 1.0 / len(x))
    return e / s


def _full_conv(a):
    """Full convolution a * a using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 2 * n - 1
    fa = np.fft.fft(a, n=size)
    return np.real(np.fft.ifft(fa * fa, n=size))


def _score_from_a(a):
    """Compute score = 2*n*max(b) / (sum(a))^2 assuming a >= 0."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    b = _full_conv(a)
    return 2.0 * n * np.max(b) / (S * S)


def _obj_and_grad(params, T, n):
    """
    Compute smoothed objective and gradient w.r.t. log-space params.
    a = softmax(params), objective = 2*n*LSE(b/T) where b = conv(a,a).
    """
    a = _softmax(params)
    b = _full_conv(a)
    
    # Log-sum-exp for numerical stability
    b_max = np.max(b)
    scaled = (b - b_max) / T
    # Clip for numerical stability
    scaled = np.clip(scaled, -50, 50)
    log_sum = np.log(np.sum(np.exp(scaled)))
    lse = b_max + T * log_sum
    obj = 2.0 * n * lse
    
    # Softmax weights over b
    w = np.exp(scaled)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.full(2*n-1, 1.0/(2*n-1))
    else:
        w /= w_sum
    
    # grad_a[j] = 2 * sum_{i=0}^{n-1} a[i] * w[i+j]
    # Compute via FFT: convolve(a, w_reversed)
    size_cc = n + (2*n - 1) - 1  # 3n - 2
    fa_cc = np.fft.fft(a, n=size_cc)
    fw_cc = np.fft.fft(w[::-1], n=size_cc)
    cc = np.real(np.fft.ifft(fa_cc * fw_cc, n=size_cc))
    grad_a = 2.0 * cc[2*n-2 : n-2 : -1]
    
    # Gradient w.r.t. params (log-space)
    # d(a_j)/d(params_k) = a_j * (delta_jk - a_k)
    # grad_params_k = sum_j grad_a_j * a_j * (delta_jk - a_k)
    #               = a_k * (grad_a_k - sum_j grad_a_j * a_j)
    dot_ga_a = np.dot(grad_a, a)
    grad_params = a * (grad_a - dot_ga_a)
    
    return obj, grad_params


def _obj_wrapper(params, T, n):
    obj, _ = _obj_and_grad(params, T, n)
    return obj


def _grad_wrapper(params, T, n):
    _, grad = _obj_and_grad(params, T, n)
    return grad


def _lbfgs_step(params, T, n, maxiter=200):
    """One L-BFGS-B optimization step with given T."""
    result = minimize(
        _obj_wrapper, params,
        jac=_grad_wrapper,
        args=(T, n),
        method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50}
    )
    return result.x


def _adam_optimize(params, T, n, lr=0.01, n_steps=500, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer for the smoothed objective."""
    m = np.zeros_like(params)
    v = np.zeros_like(params)
    t = 0
    best_obj = _obj_wrapper(params, T, n)
    best_params = params.copy()
    
    for step in range(n_steps):
        t += 1
        obj, grad = _obj_and_grad(params, T, n)
        
        if obj < best_obj:
            best_obj = obj
            best_params = params.copy()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        params = params - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # No projection needed since we're in log-space
    
    return best_params


def _make_anti_symmetric(n, eps, rng):
    """Create anti-symmetric profile."""
    x = np.zeros(n)
    for i in range(n):
        x[i] = eps * (2 * i / (n - 1) - 1)
    a = np.exp(x)
    a /= np.sum(a)
    return a


def _make_random_smooth(n, rng, n_freqs=5):
    """Create a random smooth positive profile."""
    x = np.zeros(n)
    freqs = rng.integers(1, max(2, n // 10), size=n_freqs)

        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        x += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    a = np.exp(0.5 * x)
    a /= np.sum(a)
    return a
