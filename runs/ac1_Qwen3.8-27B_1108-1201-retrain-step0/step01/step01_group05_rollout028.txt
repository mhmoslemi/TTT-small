def project_simplex(v, S):

    if n == 0:
        return v

    total = top_vals.sum()

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.

Strategy:
- Normalize to sum=sqrt(2n), reducing objective to max(self-convolution).
- Use smooth log-sum-exp surrogate with adaptive temperature for the max.
- Compute full gradient via FFT (cross-correlation of a with softmax weights).
- Adam optimizer with simplex projection + mirror descent variants.
- Multi-phase: coarse softmax -> exact subgradient -> fine polish.
- Multiple initializations to avoid local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(n, S / n, dtype=np.float64)
    return a * (S / s)


def compute_grad_smooth(a, c, T):
    """
    Gradient of T*log(sum(exp(c_k/T))) w.r.t. a, computed via FFT.
    Returns grad of shape (n,).
    """
    n = len(a)
    c_shifted = c - c.max()
    exp_vals = np.exp(c_shifted / T)
    w = exp_vals / exp_vals.sum()
    # v[m] = sum_{l=0}^{n-1} a[l] * w[m+l] = conv(a, w_rev)[2n-2-m]
    w_rev = w[::-1]
    fsize = 1 << (3 * n).bit_length()
    A = fft(a, fsize)
    W = fft(w_rev, fsize)
    conv_result = np.real(ifft(A * W))
    v = conv_result[2 * n - 2 : n - 2 : -1]
    return 2.0 * v


def compute_grad_topk(a, c, top_k):
    """Weighted top-k subgradient via FFT."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total <= 1e-15:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    w_full[top_indices] = weights
    w_rev = w_full[::-1]
    fsize = 1 << (3 * n).bit_length()
    A = fft(a, fsize)
    W = fft(w_rev, fsize)
    conv_result = np.real(ifft(A * W))
    v = conv_result[2 * n - 2 : n - 2 : -1]
    return 2.0 * v


def run_smooth_adam(a_init, n, S, max_time, rng, T_frac_start=0.3, T_frac_end=0.005, lr_scale=0.5):
    """Adam with smooth softmax gradient, geometrically decreasing T."""
    a = normalize_seq(a_init, n, S)
    c = fft_convolve_self(a)
    c_range = max(1e-6, c.max() - c.min())
    T_start = T_frac_start * c_range
    T_end = T_frac_end * c_range

    m_adam = np.zeros(n, dtype=np.float64)
    v_adam = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    steps_per_sec = 600.0

    while time.monotonic() - start < max_time:
        t += 1
        progress = min(0.999, float(t) / (max_time * steps_per_sec))
        T = T_start * (T_end / T_start) ** progress

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        grad = compute_grad_smooth(a, c, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break

        m_adam = beta1 * m_adam + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad ** 2)
        m_hat = m_adam / (1.0 - beta1 ** t)
        v_hat = v_adam / (1.0 - beta2 ** t)

        lr = lr_scale * S / (n * gnorm)
        lr *= (1.0 / (1.0 + 0.0005 * t))
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

        if t % 4000 == 0 and rng.random() < 0.12:
            noise = rng.normal(0.0, 0.008 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_local, best_score

    t = 0
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        grad = compute_grad_topk(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break

        m_adam = beta1 * m_adam + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad ** 2)
        m_hat = m_adam / (1.0 - beta1 ** t)
        v_hat = v_adam / (1.0 - beta2 ** t)

        lr = lr_scale * S / (n * gnorm)
        lr *= (1.0 / (1.0 + 0.001 * t))
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

        if t % 8000 == 0 and rng.random() < 0.08:
            noise = rng.normal(0.0, 0.004 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_local, best_score


def run_mirror(a_init, n, S, max_time, rng, eta_init=0.3, top_k=20):
    """Mirror descent (multiplicative weights) with top-k gradient."""
    a = normalize_seq(a_init, n, S)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    t = 0

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        grad = compute_grad_topk(a, c, top_k)
        gmax = np.abs(grad).max()
        if gmax < 1e-12:
            break
        grad_normed = grad / gmax

        eta = eta_init / (1.0 + 0.003 * t)
        a_new = a * np.exp(-eta * grad_normed)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / s)

        if t % 5000 == 0 and rng.random() < 0.1:
            a = project_simplex(a + rng.normal(0.0, 0.005 * S / np.sqrt(n), n), S)

    return best_local, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.1, budget_s - 12.0)

    global_best = np.inf
    global_a = None
    global_n = 8192

    # Build initializations
    inits = []
    n_main = 8192

    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 16 and given.sum() > 0:
            n_main = len(given)
            inits.append(("given", np.clip(given, 0, 1000).copy()))
    except Exception:
        pass

    if not inits:
        n_main = 8192
        inits.append(("rand", np.maximum(rng.random(n_main), 0.05)))

    inits.append(("rand", np.maximum(rng.random(n_main), 0.05)))

    tt = np.linspace(0.0, 1.0, n_main)
    bell = np.exp(-3.0 * ((tt - 0.5) / 0.25) ** 2) + 0.1
    inits.append(("bell", bell))

    sine = 1.0 + 0.25 * np.sin(2 * np.pi * 2.0 * tt) + 0.1 * np.sin(2 * np.pi * 5.0 * tt)
    inits.append(("sine", sine))

    # A different length for diversity
    n_alt = 4096
    if n_alt != n_main:
        inits.append(("rand_%d" % n_alt, np.maximum(rng.random(n_alt), 0.05)))

    print("Inits: %s (budget=%.0fs)" % ([x[0] for x in inits], budget_s))

    # Time allocation
    total_avail = deadline - time.monotonic()
    p1_frac = 0.45
    p2_frac = 0.30
    p3_frac = 0.20
    p1_time = total_avail * p1_frac
    p2_time = total_avail * p2_frac
    p3_time = total_avail * p3_frac

    # Phase 1: Explore each init with smooth softmax Adam
    per_init = p1_time / len(inits)
    phase1_best_a = None
    phase1_best_score = np.inf

    for name, arr in inits:
        if time.monotonic() > deadline - p2_time - p3_time - 10.0:
            break
        n = len(arr)
        S = np.sqrt(2.0 * float(n))
        t_alloc = max(2.0, per_init)

        a1, s1 = run_smooth_adam(arr, n, S, t_alloc, rng,
                                  T_frac_start=0.3, T_frac_end=0.005, lr_scale=0.5)
        print("P1 %s (n=%d): maxc=%.6f" % (name, n, s1))

        if s1 < global_best:
            global_best = s1
            global_a = a1.copy()
            global_n = n

        if s1 < phase1_best_score:
            phase1_best_score = s1
            phase1_best_a = a1.copy()

    print("After P1: best=%.6f" % global_best)

    # Phase 2: Exact subgradient + mirror descent on best
    if time.monotonic() < deadline - p3_time - 10.0 and global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        t_avail_exact = p2_time * 0.6
        t_avail_mirror = p2_time * 0.4

        # Exact subgradient
        a2, s2 = run_exact_adam(global_a, n, S, t_avail_exact, rng, top_k=40, lr_scale=0.2)
        print("P2 exact (top-40): maxc=%.6f" % s2)
        if s2 < global_best:
            global_best = s2
            global_a = a2.copy()
            global_n = n

        # Mirror descent from current best
        if time.monotonic() < deadline - p3_time - 5.0:
            a3, s3 = run_mirror(global_a, n, S, max(1.0, t_avail_mirror), rng,
                                eta_init=0.4, top_k=30)
            print("P2 mirror: maxc=%.6f" % s3)
            if s3 < global_best:
                global_best = s3
                global_a = a3.copy()
                global_n = n

    # Phase 3: Fine polishing
    left = deadline - time.monotonic() - 8.0
    if left > 2.0 and global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))

        # Low-temperature softmax polish
        t_polish1 = left * 0.5
        a4, s4 = run_smooth_adam(global_a, n, S, max(1.0, t_polish1), rng,
                                  T_frac_start=0.05, T_frac_end=0.002, lr_scale=0.2)
        print("P3 low-T softmax: maxc=%.6f" % s4)
        if s4 < global_best:
            global_best = s4
            global_a = a4.copy()
            global_n = n

        # Final exact polish
        if time.monotonic() < deadline - 5.0:
            t_polish2 = max(1.0, deadline - time.monotonic() - 5.0)
            a5, s5 = run_exact_adam(global_a, n, S, t_polish2, rng, top_k=60, lr_scale=0.1)
            print("P3 exact (top-60): maxc=%.6f" % s5)
            if s5 < global_best:
                global_best = s5
                global_a = a5.copy()
                global_n = n

    # Finalize
    if global_a is None:
        global_a = np.ones(n_main, dtype=np.float64)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    n_final = len(global_a)
    c_final = fft_convolve_self(global_a)
    max_c = c_final.max()
    sum_a = global_a.sum()
    obj = 2.0 * n_final * max_c / (sum_a ** 2)
    print("FINAL: n=%d maxc=%.6f sum=%.4f obj=%.6f" % (n_final, max_c, sum_a, obj))

    ver = evaluate_sequence(global_a.tolist())
    print("Verified: %.6f" % ver)

    return [float(x) for x in global_a]
