def _score(a):
    """Compute the score for a normalized sequence (sum=1)."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)

def _grad_bk(a, k):
    """Compute gradient of b_k w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _mul_update(a, g, eta):
    """Multiplicative update: a_i *= exp(-eta * g_i), then normalize."""
    # Scale gradient for numerical stability
    g_max = np.max(np.abs(g))
    if g_max < 1e-15:
        return a
    log_a = np.log(a + 1e-30) - eta * g / g_max
    # Numerical stability: clip
    log_a = np.clip(log_a, -30, 30)
    new_a = np.exp(log_a)
    new_a = np.maximum(new_a, 0)
    s = np.sum(new_a)
    if s < 1e-30:
        return a
    new_a /= s
    return new_a

    Returns (smooth_max_value, weights) where weights_k = exp(beta*b_k)/sum(exp(beta*b_j)).

    b_max = np.max(b)
    shifted = beta * (b - b_max)
    # For numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    smooth_val = b_max + np.log(np.sum(exp_shifted)) / beta
    return smooth_val, w

def _grad_bk(a, k):
    """Gradient of b_k = sum_i a_i*a_{k-i} w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _make_starts(n, rng):
    """Generate diverse starting points."""
    starts = []
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Anti-symmetric two-level (various eps)
    for eps in [0.3, 0.5, 0.7]:
        a = np.ones(n)
        a[:n//2] = 1.0 + eps
        a[n//2:] = 1.0 - eps
        a = np.maximum(a, 0.01)
        starts.append(a / np.sum(a))
    
    # Smooth cosine
    for eps in [0.5, 0.8]:
        i = np.arange(n, dtype=np.float64)
        a = 1.0 + eps * np.cos(2 * np.pi * i / n)
        a = np.maximum(a, 0.01)
        starts.append(a / np.sum(a))
    
    # Exponential decay
    a = np.exp(-np.arange(n, dtype=np.float64) / (n * 0.3))
    starts.append(a / np.sum(a))
    a = np.exp(-np.arange(n-1, -1, -1, dtype=np.float64) / (n * 0.3))
    starts.append(a / np.sum(a))
    
    # Gaussian bump
    i = np.arange(n, dtype=np.float64)
    center = n / 2
    sigma = n / 4
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    starts.append(a / np.sum(a))
    
    # Random
    a = rng.random(n)
    starts.append(a / np.sum(a))
    
    # "V-shaped" (low in middle, high at edges) - opposite of what we want for center peak
    a = np.abs(np.arange(n, dtype=np.float64) - n/2) / (n/2)

    a = np.maximum(a, 0.01)
    starts.append(a / np.sum(a))
    
    return starts
