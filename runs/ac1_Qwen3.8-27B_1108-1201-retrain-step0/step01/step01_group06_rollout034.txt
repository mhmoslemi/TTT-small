Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. LP-based direction finding (solving max sum(g) s.t. f*g <= M)
2. Soft-max multiplicative gradient descent
3. Multi-start with diverse initializations and sequence lengths

import numpy as np
import scipy.optimize
import time

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def solve_lp_direction(f, M, epsilon=1e-10):
    """
    Solve LP: max sum(g_j) s.t. (f*g)_k <= M for all k, g_j >= 0.
    Returns optimal g (non-negative).
    """
    n = len(f)
    # Build constraint matrix efficiently
    # A has shape (2n-1, n), A[k,j] = f[k-j] if 0<=k-j<n
    # This is a banded Toeplitz-like matrix
    A = np.zeros((2 * n - 1, n))
    # For efficiency, use the structure
    for j in range(n):
        # f[j] appears in row k where k-j is in [0, n-1], i.e., k in [j, j+n-1]
        # But k is in [0, 2n-2], so effectively k in [j, min(j+n-1, 2n-2)]
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        A[k_start:k_end + 1, j] = f[:k_end - k_start + 1]
    
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M + epsilon)  # Slight slack to avoid numerical issues
    
    result = scipy.optimize.linprog(
        c, A_ub=A, b_ub=b,
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True}
    )
    
    if result.success:
        return np.maximum(result.x, 0.0)
    return np.ones(n) / n

def lp_gradient_hybrid(f0, time_limit, verbose=False):

    Hybrid: alternate between LP direction and gradient updates.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lp_t = 0.05  # LP blending parameter
    grad_eta = 0.3
    grad_temp = 40.0
    iteration = 0
    lp_interval = 5  # Do LP every 5 iterations
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        if iteration % lp_interval == 0:
            # LP step
            g = solve_lp_direction(f, max_conv)
            g_sum = g.sum()
            if g_sum > 1e-12:
                g /= g_sum
                # Check if g is meaningfully different from f
                cos_sim = np.dot(f, g) / (np.sqrt(np.dot(f, f)) * np.sqrt(np.dot(g, g)) + 1e-15)
                if cos_sim < 0.999:
                    f_new = (1 - lp_t) * f + lp_t * g
                    f_new = np.maximum(f_new, 0.0)
                    f_sum = f_new.sum()
                    if f_sum > 1e-12:
                        f_new /= f_sum
                        conv_new = fast_convolve(f_new)
                        obj_new = 2.0 * n * conv_new.max()
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_f = f_new.copy()
                            f = f_new
                            conv = conv_new
                else:
                    # Degenerate LP, use gradient instead
                    pass
        else:
            # Gradient step
            scaled = grad_temp * conv / max_conv
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()
            
            sg = np.zeros(n)
            thresh = w.max() * 0.001
            active_idx = np.where(w > thresh)[0]
            for k_int in active_idx:
                k = int(k_int)
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                if j_start <= j_end:
                    sg[j_start:j_end + 1] += w[k] * f[k - j_end:k - j_start + 1][::-1]
            
            sg_centered = sg - sg.mean()
            log_factor = -grad_eta * sg_centered
            log_factor -= log_factor.max()
            f = f * np.exp(log_factor)
            f = np.maximum(f, 0.0)
            f_sum = f.sum()
            if f_sum < 1e-12:
                f = np.ones(n) / n
            else:
                f /= f_sum

                best_f = f.copy()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. L-BFGS optimization on a soft-max smoothed objective with softmax parameterization
2. Multiplicative mirror descent (exponential gradient) with adaptive temperature
3. LP-based direction finding for escaping local minima
4. Multi-start with diverse initializations and sequence lengths (64-512)
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_score(a):
    """Compute the objective value for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_objective_and_grad(x, f, conv, t, n):
    """
    Compute soft-max smoothed objective and gradient w.r.t. x (where f = softmax(x_full)).
    x_full = [0, x[0], x[1], ..., x[n-2]]
    f is already computed from x.
    """
    scaled = conv / t
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    obj = t * (scaled.max() + np.log(sum_exp))

    w = exp_scaled / sum_exp

    # grad_f_j = (1/t) * sum_k w_k * f_{k-j}
    grad_f = np.zeros(n)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad_f[j_start:j_end + 1] += w[k] * f[k - j_end:k - j_start + 1][::-1]
    grad_f /= t

    # Chain rule for softmax parameterization
    dot = np.dot(grad_f, f)
    grad_x_full = f * (grad_f - dot)

    # x_full[0] is fixed to 0, so gradient w.r.t. x is grad_x_full[1:]
    return obj, grad_x_full[1:]


def lbfgs_optimize(f0, time_limit, t_init=0.05, verbose=False):
    """
    L-BFGS optimization using softmax parameterization.
    f0: initial non-negative sequence (will be normalized to sum=1).
    Returns (best_f, best_obj).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f /= s

    # Compute initial objective (exact max, not smoothed)
    conv = fast_convolve(f)
    best_obj_exact = compute_score(f)
    best_f = f.copy()

    # Convert to log-space parameterization
    f_safe = np.maximum(f, 1e-30)
    x_full = np.log(f_safe)
    x_full[0] = 0.0  # Fix for identifiability
    x0 = x_full[1:].copy()

    t0_time = time.monotonic()
    t = t_init
    max_conv = conv.max()

    # Scale t relative to the convolution scale
    t = max_conv * 0.1

    best_smoothed_obj = np.inf

    while time.monotonic() - t0_time < time_limit:
        # Define objective function for L-BFGS
        def obj_func(x_vec):
            x_full_local = np.empty(n)
            x_full_local[0] = 0.0
            x_full_local[1:] = x_vec
            # Normalize for numerical stability
            x_full_local -= x_full_local.max()
            f_local = np.exp(x_full_local)
            f_sum = f_local.sum()
            if f_sum < 1e-30:
                return 1e10, np.zeros(n - 1)
            f_local /= f_sum
            conv_local = fast_convolve(f_local)
            obj, grad = softmax_objective_and_grad(x_vec, f_local, conv_local, t, n)
            return obj, grad

        # Run L-BFGS
        def objective_only(x_vec):
            x_full_local = np.empty(n)
            x_full_local[0] = 0.0
            x_full_local[1:] = x_vec
            x_full_local -= x_full_local.max()
            f_local = np.exp(x_full_local)
            f_sum = f_local.sum()
            if f_sum < 1e-30:
                return 1e10
            f_local /= f_sum
            conv_local = fast_convolve(f_local)
            obj, _ = softmax_objective_and_grad(x_vec, f_local, conv_local, t, n)
            return obj

        # Use a few L-BFGS sub-iterations with decreasing temperature
        result = scipy.optimize.minimize(
            objective_only, x0,
            jac=lambda x_vec: softmax_objective_and_grad(
                x_vec,
                np.exp(np.concatenate([[0.0], x_vec]) - np.max(np.concatenate([[0.0], x_vec]))),
                fast_convolve(np.exp(np.concatenate([[0.0], x_vec]) - np.max(np.concatenate([[0.0], x_vec])))),
                t, n
            )[1],
            method='L-BFGS-B',
            options={'maxiter': 200, 'gtol': 1e-8, 'maxls': 20}
        )

        if result.success or result.fun < best_smoothed_obj:
            x_new = result.x
            x_full_new = np.empty(n)
            x_full_new[0] = 0.0
            x_full_new[1:] = x_new
            x_full_new -= x_full_new.max()
            f_new = np.exp(x_full_new)
            f_new /= f_new.sum()
            f_new = np.maximum(f_new, 0.0)
            f_sum = f_new.sum()
            if f_sum > 1e-15:
                f_new /= f_sum
                conv_new = fast_convolve(f_new)
                obj_new = compute_score(f_new)
                if obj_new < best_obj_exact:
                    best_obj_exact = obj_new
                    best_f = f_new.copy()
                    best_smoothed_obj = result.fun
                    x0 = x_new.copy()
                    f = f_new
                    conv = conv_new
                    if verbose:
                        print(f"    L-BFGS improved: obj={obj_new:.6f}")
                else:
                    # Update x0 even if exact obj didn't improve (smoothed did)
                    x0 = x_new.copy()
                    f = f_new
                    conv = conv_new
                    best_smoothed_obj = result.fun
            else:
                break
        else:
            # No improvement, decrease temperature and try again
            t *= 0.7
            if t < max_conv * 0.01:
                break

        # Update t for next iteration
        max_conv = conv.max()
        t = max_conv * 0.1

        if time.monotonic() - t0_time > time_limit * 0.8:
            break

    # Final: also try a few multiplicative gradient steps from best_f
    f = best_f.copy()
    conv = fast_convolve(f)
    t_grad = compute_score(f) / (2.0 * n) * 50  # temperature for gradient
    eta = 0.3

    for _ in range(50):
        if time.monotonic() - t0_time > time_limit:
            break
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        scaled = (t_grad * 2 * n) * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        sg = np.zeros(n)
        thresh = w.max() * 0.001
        active_idx = np.where(w > thresh)[0]
        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end + 1] += w[k] * f[k - j_end:k - j_start + 1][::-1]

        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        f = f * np.exp(log_factor)
        f = np.maximum(f, 0.0)
        f_sum = f.sum()
        if f_sum < 1e-15:
            break
        f /= f_sum
        conv = fast_convolve(f)
        obj = compute_score(f)
        if obj < best_obj_exact:
            best_obj_exact = obj
            best_f = f.copy()

    return best_f, best_obj_exact


def multiplicative_optimize(f0, time_limit, t_param=40.0, eta0=0.5):
    """
    Pure multiplicative mirror descent with adaptive parameters.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s

    conv = fast_convolve(f)
    best_obj = compute_score(f)
    best_f = f.copy()

    eta = eta0
    patience = 0
    max_patience = 80
    current_t = t_param
    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        scaled = current_t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        thresh = w.max() * 0.001
        active_idx = np.where(w > thresh)[0]
        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end + 1] += w[k] * f[k - j_end:k - j_start + 1][::-1]

        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        f = f * np.exp(log_factor)
        f = np.maximum(f, 0.0)
        f_sum = f.sum()
        if f_sum < 1e-12:
            f = np.ones(n) / n
        else:
            f /= f_sum

        conv = fast_convolve(f)
        current_obj = compute_score(f)

        if current_obj < best_obj - 1e-9:
            best_obj = current_obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            patience = 0
            eta *= 0.65
            if eta < 0.002:
                current_t *= 2.0
                eta = 0.4
                max_patience = 40
            else:
                current_t = np.clip(current_t * 1.2, 5.0, 500.0)

        # Periodic temperature annealing
        if iteration % 300 == 0:
            current_t = np.clip(current_t * 0.75, 5.0, 500.0)

def lp_direction_step(f, M, n):
    """
    Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Returns g or None if LP fails or is degenerate.
    """
    if n > 300:
        return None  # Too large for efficient LP

    # Build constraint matrix A (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        A[k_start:k_end + 1, j] = f[:k_end - k_start + 1]

    c = -np.ones(n)
    b = np.full(2 * n - 1, M * (1 + 1e-10))

    try:
        result = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': 5.0}
        )
        if result.success:
            g = np.maximum(result.x, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-15:
                return None
            g /= g_sum
            # Check if g is meaningfully different from f
            cos_sim = np.dot(f, g) / (np.sqrt(np.dot(f, f)) * np.sqrt(np.dot(g, g)) + 1e-15)
            if cos_sim > 0.9995:
                return None  # Degenerate
            return g
    except Exception:
        pass
    return None


def hybrid_optimize(f0, time_limit, use_lp=True):

                f_new /= f_new.sum()
                conv_new = fast_convolve(f_new)

                    best_obj = obj_new
                    best_f = f_new.copy()
                    f = f_new
                    conv = conv_new

                    continue
        # Gradient step

            conv = fast_convolve(f)
        else:
            f /= f_sum
            conv = fast_convolve(f)

                stagnation = 0
            else:

def generate_initializations(n, rng, parent=None):

    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian bumps with different widths
    for sf in [0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits.append(g)

    # Bimodal (edges)
    for wf in [0.15, 0.25, 0.35]:
        bi = np.zeros(n)
        w = max(1, int(wf * n))
        bi[:w] = 1.0
        bi[-w:] = 1.0
        bi /= bi.sum()
        inits.append(bi)

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    rc = np.maximum(rc, 0.01)
    rc /= rc.sum()
    inits.append(rc)

    # Power law (centered)
    pl = 1.0 / (1.0 + 0.3 * np.abs(x - n / 2))
    pl /= pl.sum()
    inits.append(pl)

    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)

    # Smooth random (low-pass filtered random)
    r = rng.random(n)
    # Simple low-pass: moving average
    kernel_size = max(3, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    r_smooth = np.convolve(r, kernel, mode='same')
    r_smooth = np.maximum(r_smooth, 0.01)
    r_smooth /= r_smooth.sum()
    inits.append(r_smooth)

    # Parent-derived
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        pp = np.maximum(pp, 0.0)
        pp_sum = pp.sum()
        if pp_sum > 1e-10:
            pp /= pp_sum
            inits.append(pp)
            # Perturbed parent
            noise_scale = 0.1
            perturbed = pp * (1 + noise_scale * (rng.random(n) - 0.5))
            perturbed = np.maximum(perturbed, 0.0)
            perturbed_sum = perturbed.sum()
            if perturbed_sum > 1e-10:
                perturbed /= perturbed_sum
                inits.append(perturbed)

    return inits


def run_optimization(seed, budget_s):
    """Main optimization driver with multi-phase strategy."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent = parent / parent.sum()
    except Exception:
        pass

    # Lengths to try
    lengths = [64, 96, 128, 160, 192, 256, 320]

    # Phase 1: Quick scan
    print("Phase 1: Quick scan...")
    phase1_deadline = time.monotonic() + min(500.0, (deadline - time.monotonic()) * 0.25)

    scan_results = []
    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        inits = generate_initializations(n, rng, parent)
        time_per_init = max(1.0, (phase1_deadline - time.monotonic() - 30) / max(1, len(inits)))

        for a0 in inits:
            if time.monotonic() > phase1_deadline - 30:
                break
            t_alloc = min(time_per_init, phase1_deadline - time.monotonic() - 30)
            if t_alloc < 0.5:
                break

            a_opt, obj_opt = multiplicative_optimize(a0, time_limit=t_alloc, t_param=30.0, eta0=0.5)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, a_opt))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    print(f"  Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Deep L-BFGS optimization on top candidates
    print("Phase 2: L-BFGS deep optimization...")
    scan_results.sort(key=lambda x: x[0])

    # Select diverse top candidates
    top_candidates = []
    seen_lengths = set()
    for score, n, a_opt in scan_results:
        if len(top_candidates) < 12:
            top_candidates.append((n, a_opt))
            seen_lengths.add(n)
        elif n not in seen_lengths and len(top_candidates) < 18:
            top_candidates.append((n, a_opt))
            seen_lengths.add(n)

    # Always include current best
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), best_sequence.copy()))

    phase2_deadline = time.monotonic() + min(1200.0, (deadline - time.monotonic()) * 0.55)
    n_candidates = len(top_candidates)
    time_per_candidate = max(5.0, (phase2_deadline - time.monotonic()) / max(1, n_candidates))

    for i, (n, a0) in enumerate(top_candidates):
        if time.monotonic() > phase2_deadline - 20:
            break
        t_alloc = min(time_per_candidate, phase2_deadline - time.monotonic() - 20)
        if t_alloc < 3.0:
            break

        # L-BFGS with adaptive temperature
        a_opt, obj_opt = lbfgs_optimize(a0, time_limit=t_alloc * 0.6, t_init=0.05)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  L-BFGS: n={n}, score={score:.6f}")

        # Also try hybrid (gradient + LP)
        if time.monotonic() < phase2_deadline - 20:
            t_remaining = phase2_deadline - time.monotonic() - 20
            a_opt, obj_opt = hybrid_optimize(a0, time_limit=t_remaining * 0.5, use_lp=(n <= 256))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Hybrid: n={n}, score={score:.6f}")

    print(f"  Phase 2 done. Best: {best_score:.6f}")

    # Phase 3: Final polish
    print("Phase 3: Final polish...")
    if best_sequence is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_best = len(best_sequence)

            # Multiple polish attempts with different parameters
            for attempt in range(5):
                if time.monotonic() > deadline - 10:
                    break
                t_alloc = min(remaining * 0.2, (deadline - 10 - time.monotonic()))
                if t_alloc < 3.0:
                    break

                # Vary temperature and method
                if attempt % 2 == 0:
                    a_opt, obj_opt = multiplicative_optimize(
                        best_sequence, time_limit=t_alloc,
                        t_param=20.0 + attempt * 30.0, eta0=0.3
                    )
                else:
                    a_opt, obj_opt = lbfgs_optimize(
                        best_sequence, time_limit=t_alloc, t_init=0.03
                    )

                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish {attempt}: score={score:.6f}")

                remaining = deadline - 10 - time.monotonic()

            # One more hybrid attempt
            if time.monotonic() < deadline - 15 and n_best <= 256:
                t_alloc = deadline - 15 - time.monotonic()
                if t_alloc > 5:
                    a_opt, obj_opt = hybrid_optimize(
                        best_sequence, time_limit=t_alloc, use_lp=True
                    )
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Final hybrid: score={score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return best_sequence


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Uses L-BFGS on soft-max smoothed objective combined
    with multiplicative mirror descent and LP-based direction finding
    to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    result = run_optimization(seed=seed, budget_s=budget_s)
    return [float(x) for x in result]
