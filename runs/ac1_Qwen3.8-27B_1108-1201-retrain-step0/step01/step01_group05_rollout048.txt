import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_gradient_topk(a, c, top_k, tau):
    """Compute approximate gradient using log-sum-exp weighting on top-k lags."""
    n = len(a)
    # Get top-k lags
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    
    # Compute softmax weights over top-k
    top_shifted = top_vals - top_vals.max()
    exp_top = np.exp(top_shifted / tau)
    weights = exp_top / exp_top.sum()
    
    # Compute gradient
    grad = np.zeros(n, dtype=np.float64)
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        # For j in [lo, hi]: grad[j] += w_val * 2 * a[k-j]
        # But if k = 2*j (j = k//2 and k is even), it should be w_val * a[j]
        grad[lo:hi+1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
        # Fix the diagonal term
        if k % 2 == 0:
            j_diag = k // 2
            if lo <= j_diag <= hi:
                grad[j_diag] -= w_val * a[j_diag]  # We added 2*a[j_diag], should be a[j_diag]
    return grad

def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
    return a * (S / s)

def run_smoothed_adam(a_init, n, S, max_time, rng, tau_init, tau_decay, lr_init, top_k, use_nesterov=True):
    """Adam with log-sum-exp smoothing and optional Nesterov momentum."""
    a = safe_normalize(a_init, n)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    y = a.copy()  # Nesterov lookahead
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        tau = tau_init * max(0.01, tau_decay ** t)
        
        if use_nesterov:
            # Nesterov: evaluate gradient at y
            c = fft_convolve_self(y)
            grad = compute_gradient_topk(y, c, top_k, tau)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr_init * max(0.01, (0.999 ** t))
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            y = a_new - beta1 * step  # Nesterov: y = a - lr*step - beta1*lr*grad... 
            # Actually, standard Nesterov: y = a + momentum * (a - prev_a), then grad at y
            # Let me use a simpler version: just use momentum on the gradient
            y = a_new
        else:
            c = fft_convolve_self(a)
            grad = compute_gradient_topk(a, c, top_k, tau)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr_init * max(0.01, (0.999 ** t))
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a = project_simplex(a_new, S)
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Random perturbation to escape local minima
        if t % 5000 == 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.02 * S / n, n), S)
    
    return best_local, best_score

    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_grad_weighted(a, weights, indices):
    """Compute gradient contribution from weighted lags.
    weights and indices are arrays of same length.
    grad[j] = sum_k w[k] * (2*a[k-j] if valid and k!=2j, a[j] if k==2j)
    """
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    for w_val, k in zip(weights, indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        seg_len = hi - lo + 1
        # a[k-hi : k-lo+1] reversed gives a[k-hi], a[k-hi+1], ..., a[k-lo]
        # We want: for j in [lo, hi], contribution = 2*a[k-j]
        # a[k-j] for j=lo is a[k-lo], for j=hi is a[k-hi]
        # So the array is a[k-hi : k-lo+1] and we need it reversed: a[k-lo], ..., a[k-hi]
        # Wait: a[k-hi], a[k-hi+1], ..., a[k-lo] is a[k-hi:k-lo+1]
        # We want for j=lo: a[k-lo], for j=lo+1: a[k-lo-1], ..., for j=hi: a[k-hi]
        # So it's a[k-hi : k-lo+1] in forward order assigned to grad[hi:lo-1:-1]
        # Or equivalently: a[k-lo : k-hi-1 : -1] assigned to grad[lo:hi+1]
        arr = a[k - hi : k - lo + 1][::-1]  # this is a[k-lo], a[k-lo-1], ..., a[k-hi]
        # Wait, a[k-hi:k-lo+1] = [a[k-hi], a[k-hi+1], ..., a[k-lo]]
        # Reversed: [a[k-lo], a[k-lo-1], ..., a[k-hi]]
        # We want grad[lo] += w*2*a[k-lo], grad[lo+1] += w*2*a[k-lo-1], ..., grad[hi] += w*2*a[k-hi]
        # So grad[lo:hi+1] += w*2*[a[k-lo], a[k-lo-1], ..., a[k-hi]]
        # Which is w*2*reversed(a[k-hi:k-lo+1])
        grad[lo:hi + 1] += w_val * 2.0 * arr
        # Fix diagonal: if k is even and k//2 is in [lo, hi], we added 2*a[k//2] but should add a[k//2]
        if k % 2 == 0:
            j_diag = k // 2
            if lo <= j_diag <= hi:
                grad[j_diag] -= w_val * a[j_diag]
    return grad


def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
    return a * (S / s)


def run_smoothed_adam(a_init, n, S, max_time, rng, tau_init=1.0, tau_min=0.01,
                       lr_init=0.1, top_k=20, name=""):
    """Adam with log-sum-exp smoothing and simplex projection."""
    a = safe_normalize(a_init, n)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    n_total_steps = max(1, int(max_time / 0.001))  # rough estimate

    while time.monotonic() - start < max_time:
        t += 1
        # Temperature schedule: exponential decay
        frac = min(1.0, t / (0.7 * n_total_steps))
        tau = max(tau_min, tau_init * np.exp(-3.0 * frac))

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Compute gradient: use top-k with softmax weighting
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        top_shifted = top_vals - top_vals.max()
        exp_top = np.exp(top_shifted / tau)
        total = exp_top.sum()
        if total < 1e-30:
            weights = np.full(top_k, 1.0 / top_k)
        else:
            weights = exp_top / total

        grad = compute_grad_weighted(a, weights, top_indices)
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-14:
            break
        grad = grad / gnorm

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Cosine learning rate schedule
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * min(1.0, t / n_total_steps)))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)

        # Periodic perturbation
        if t % 8000 == 0 and rng.random() < 0.15:
            a = project_simplex(a + rng.normal(0, 0.01 * S / n, n), S)

        if t % 20000 == 0:
            elapsed = time.monotonic() - start
            print("  %s t=%d tau=%.4f best=%.6f elapsed=%.1fs" % (name, t, tau, best_score, elapsed))

    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, eta_init=0.3, top_k=20, name=""):
    """Mirror descent / multiplicative weights."""
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    t = 0

    while time.monotonic() - start < max_time:
        t += 1
        tau = max(0.05, eta_init * 0.5 * (1.0 + np.cos(np.pi * min(1.0, t / max(1, int(max_time/0.001))))))

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        top_shifted = top_vals - top_vals.max()
        exp_top = np.exp(top_shifted / tau)
        total = exp_top.sum()
        if total < 1e-30:
            weights = np.full(top_k, 1.0 / top_k)
        else:
            weights = exp_top / total

        grad = compute_grad_weighted(a, weights, top_indices)
        gmax = np.abs(grad).max()
        if gmax <= 1e-14:
            break
        grad = grad / gmax

        eta = eta_init * max(0.05, 1.0 / (1.0 + 0.002 * t))
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        a = a_new * (S / s)

    return best_local, best_score


def run_polish(a_init, n, S, max_time, rng, name="polish"):
    """Intensive polishing with decreasing tau and fine steps."""
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()

    phases = [
        (max_time * 0.4, 0.1, 0.01, 30, 0.05),
        (max_time * 0.3, 0.05, 0.005, 50, 0.03),
        (max_time * 0.3, 0.02, 0.001, 100, 0.01),
    ]

    for phase_time, tau_init, tau_min, top_k, lr in phases:
        if time.monotonic() - start >= max_time:
            break
        a, s = run_smoothed_adam(a, n, S, phase_time, rng,
                                  tau_init=tau_init, tau_min=tau_min,
                                  lr_init=lr, top_k=top_k, name=name)
        if s < best_score:
            best_score = s
            best_local = a.copy()

    # Final fine-tuning: coordinate-wise adjustments
    time_left = max_time - (time.monotonic() - start)
    if time_left > 1.0:
        a = best_local.copy()
        ft_start = time.monotonic()
        step = S / (10 * n)
        while time.monotonic() - ft_start < time_left:
            improved = False
            c = fft_convolve_self(a)
            cur_max = c.max()
            # Try small moves on a few random coordinates
            trial_count = min(n // 10, 200)
            for _ in range(trial_count):
                j = rng.integers(0, n)
                for delta in [step, -step, 2*step, -2*step]:
                    a[j] = max(0, a[j] + delta)
                    s = a.sum()
                    if s > 0:
                        a_norm = a * (S / s)
                        c2 = fft_convolve_self(a_norm)
                        if c2.max() < cur_max - 1e-10:
                            a = a_norm
                            cur_max = c2.max()
                            improved = True
                            if cur_max < best_score:
                                best_score = cur_max
                                best_local = a.copy()
                    a[j] = max(0, a[j])  # restore
            a = best_local.copy()
            if not improved:
                step *= 0.7
                if step < S / (1000 * n):
                    break

    return best_local, best_score


def make_antisymmetric_init(n, alpha_scale=1.0, rng=None):
    """Create an anti-symmetric initialization."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)  # [0, 1]
    base = np.ones(n) * alpha_scale
    anti = (1.0 - 2.0 * t)  # linear anti-symmetry: +1 at start, -1 at end
    a = 1.0 + 0.5 * anti * alpha_scale
    # Add some smooth variation
    if rng is not None:
        a += 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    return a


def make_ramp_init(n, rng=None):
    """Linearly increasing sequence."""
    t = np.linspace(0, 1, n)
    a = 0.5 + 1.0 * t
    if rng is not None:
        a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def make_bump_init(n, center=0.3, width=0.15, rng=None):
    """Gaussian bump offset from center."""
    t = np.linspace(0, 1, n)
    a = 1.0 + 2.0 * np.exp(-0.5 * ((t - center) / width) ** 2)
    if rng is not None:
        a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 10.0)

    global_best = np.inf
    global_a = None

    # Collect initializations
    inits = []

    # Previous best
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64)
        given = given.reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("prev_best", np.clip(given, 0, 1000)))
    except Exception:
        pass

    # Anti-symmetric initializations at different lengths
    for n_try in (8192, 16384):
        inits.append(("antisym_%d" % n_try, make_antisymmetric_init(n_try, 0.3, rng)))
        inits.append(("ramp_%d" % n_try, make_ramp_init(n_try, rng)))
        inits.append(("bump_%d" % n_try, make_bump_init(n_try, 0.25, rng)))

    # Longer sequence
    inits.append(("antisym_32768", make_antisymmetric_init(32768, 0.3, rng)))

    n_inits = len(inits)
    usable = max(1.0, deadline - time.monotonic())
    explore_budget = 0.65 * usable
    polish_budget = usable - explore_budget

    per_init = explore_budget / max(1, n_inits)

    print("Starting search: %d initializations, explore=%.0fs, polish=%.0fs" %
          (n_inits, explore_budget, polish_budget))

    for idx, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - polish_budget - 5.0:
            break

        n = len(arr)
        if n < 64:
            continue
        S = np.sqrt(2.0 * float(n))
        left = per_init - (time.monotonic() - (deadline - explore_budget))
        left = max(2.0, min(left, per_init))

        print("Init %d/%d: %s (n=%d), time_budget=%.1fs" % (idx + 1, n_inits, name, n, left))
        t0 = time.monotonic()

        # Run smoothed Adam
        seg1 = left * 0.6
        a_best, s_best = run_smoothed_adam(arr, n, S, seg1, rng,
                                            tau_init=1.0, tau_min=0.01,
                                            lr_init=0.1 * S / n, top_k=20,
                                            name=name)
        if s_best < global_best:
            global_best = s_best
            global_a = a_best.copy()
            print("  %s: Adam found %.6f (new global best!)" % (name, s_best))

        # Try mirror descent from the Adam result
        seg2 = left * 0.3
        if time.monotonic() < deadline - polish_budget - 3.0:
            a_mir, s_mir = run_mirror_descent(a_best, n, S, seg2, rng,
                                               eta_init=0.5, top_k=20,
                                               name=name + "_mir")
            if s_mir < global_best:
                global_best = s_mir
                global_a = a_mir.copy()
                print("  %s: Mirror found %.6f (new global best!)" % (name, s_mir))
            elif s_mir < s_best:
                a_best = a_mir
                s_best = s_mir

        # Verify with evaluate_sequence
        if s_best < global_best + 0.01:
            ver = evaluate_sequence(np.clip(a_best, 0.0, 1000.0).tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = np.clip(a_best, 0.0, 1000.0)
                print("  %s: verified %.6f (new global best!)" % (name, ver))

        elapsed = time.monotonic() - t0
        print("  %s: done in %.1fs, best_so_far=%.6f" % (name, elapsed, global_best))

    # Phase 2: Polish the best
    left = deadline - time.monotonic()
    print("\nPolishing phase: %.1fs remaining, global_best=%.6f" % (left, global_best))

    if left > 5.0 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        a_pol, s_pol = run_polish(global_a, n_best, S_best, left * 0.7, rng,
                                   name="polish")
        if s_pol < global_best:
            global_best = s_pol
            global_a = a_pol.copy()
            print("  Polish Adam: %.6f" % s_pol)

        # Additional mirror descent polish
        time_left = deadline - time.monotonic()
        if time_left > 3.0:
            a_mir2, s_mir2 = run_mirror_descent(global_a, n_best, S_best, time_left * 0.5,
                                                 rng, eta_init=0.2, top_k=50,
                                                 name="polish_mir")
            if s_mir2 < global_best:
                global_best = s_mir2
                global_a = a_mir2.copy()
                print("  Polish Mirror: %.6f" % s_mir2)

        # Final verification
        time_left = deadline - time.monotonic()
        if time_left > 2.0:
            final_a = np.clip(global_a, 0.0, 1000.0)
            ver = evaluate_sequence(final_a.tolist())
            print("\nFinal verified score: %.6f" % ver)
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = final_a
            elif np.isfinite(ver):
                global_a = final_a

    # Final safety check
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("\nReturning sequence of length %d, best internal score = %.6f" %
          (len(global_a), global_best))
    return [float(x) for x in global_a]
