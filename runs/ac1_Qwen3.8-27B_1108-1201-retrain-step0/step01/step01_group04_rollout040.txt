"""
Step-function optimization using LP best-response iterations and local gradient refinement.
Key insight: asymmetric sequences (like linear ramps) can achieve scores well below 2,
approaching 4/3 for large n. LP best-response refines these toward the optimal minimax point.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

_FMT = "%.10f"

def conv_sq(p):
    """Full self-convolution of 1D vector p (length 2n-1)."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].copy()

def internal_score(p):
    """Score = 2n * max(conv) / sum^2, mirroring evaluate_sequence."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    return float(2.0 * n * cmax / (s * s))

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s

def safe_normalize(p):
    """Ensure nonnegative, finite, sum=1."""
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s

def build_sparse_A(n):
    """Build sparse matrix A where (A @ g)[k] = sum_j A[k,j]*g[j] = (p_identity * g)[k].
    Here A[k,j] = 1 if 0 <= k-j <= n-1, else 0. But we'll multiply by p directly
    in the LP, so A[k,j] = 1 for valid (k,j) pairs."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            m = j_end - j_start + 1
            rows.extend([k] * m)
            cols.extend(js.tolist())
    return rows, cols

def lp_solve(p, M):
    """Solve: max sum(g) s.t. (p*g)[k] <= M for all k, g >= 0.
    Returns normalized g or None."""
    n = int(p.size)
    if n <= 0 or n > 4000:
        return None
    try:
        # Build sparse constraint: A[k, j] = p[k-j] if 0 <= k-j < n
        rows = []
        cols = []
        data = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                js = np.arange(j_start, j_end + 1)
                m = j_end - j_start + 1
                rows.extend([k] * m)
                cols.extend(js.tolist())
                data.extend((p[k - js]).tolist())
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'disp': False})
        if res is not None and getattr(res, 'x', None) is not None:
            g = np.asarray(res.x, dtype=np.float64)
            g = np.where(np.isfinite(g), g, 0.0)
            g = np.maximum(g, 0.0)
            s = float(g.sum())
            if s > 1e-12:
                return g / s
    except Exception as e:
        print(f"  LP error: {e}", flush=True)
    return None

def gradient_smooth_max(p, T_factor=0.05):
    """
    Gradient of smoothed max of self-convolution w.r.t. p (on the simplex tangent).
    Uses softmax weighting of c_k values.
    """
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.zeros(0)
    L = next_fast_len(3 * n - 2)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]
    cmax = float(np.max(c)) if c.size else 0.0
    if cmax <= 0:
        return np.zeros(n)
    T = max(T_factor * cmax, 1e-300)
    diff = (c - cmax) / T
    diff = np.clip(diff, -50, 50)
    e = np.exp(diff)
    raw = float(e.sum())
    if raw <= 0 or not np.isfinite(raw):
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        raw = 1.0
    w = e / raw

    # grad_j = 2 * sum_k w_k * p[k-j] for 0 <= k-j < n
    # = 2 * cross-correlation of w (length 2n-1) with p (length n) at lag j
    # Using: (w * p_rev)[n-1+j] = sum_k w[k] * p_rev[n-1+j-k] = sum_k w[k] * p[k-j]
    # where p_rev[i] = p[n-1-i]
    L2 = next_fast_len((2 * n - 1) + n - 1)
    Fw = fft(w, L2)
    Fpr = fft(p[::-1], L2)
    h = np.real(ifft(Fw * Fpr))
    # h[k] = (w * p_rev)[k] for k = 0, ..., L2-1
    # We want h[n-1 : 2n-1] (indices n-1 to 2n-2)
    grad = 2.0 * h[n - 1: 2 * n - 1]
    if grad.size != n:
        tmp = np.zeros(n)
        m = min(n, grad.size)
        tmp[:m] = grad[:m]
        grad = tmp
    return grad

def local_optimize(p, time_budget=300.0, T_start=0.1, T_end=0.005):
    """Local optimization via projected gradient with adaptive step size."""
    p = safe_normalize(p)
    n = int(p.size)
    best_p = p.copy()
    best_sc = internal_score(p)
    start = time.monotonic()
    it = 0
    since_improve = 0

    while time.monotonic() - start < time_budget:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max(1.0, time_budget))
        T = T_start * (T_end / T_start) ** progress
        T = max(T, T_end)

        g = gradient_smooth_max(p, T)
        if not np.all(np.isfinite(g)) or np.max(np.abs(g)) < 1e-15:
            # Try random perturbation
            p = project_simplex(p + np.random.randn(n) * 0.5 / n)
            continue

        # Adaptive step sizes
        gm = float(np.max(np.abs(g)))
        base_lr = 1.0 / (n * gm + 1e-15)
        lrs = [base_lr * 1.0, base_lr * 0.3, base_lr * 0.1, base_lr * 0.03, base_lr * 0.01]

        accepted = False
        for lr in lrs:
            v = p - lr * g
            # Project away from gradient of sum constraint (stay on simplex)
            v = v - np.mean(v)
            q = project_simplex(v)
            sc = internal_score(q)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_p = q.copy()
                p = q
                since_improve = 0
                accepted = True
                break
            elif sc < internal_score(p) - 1e-14:
                p = q
                since_improve += 1
                accepted = True
                break

        if not accepted:
            since_improve += 1
            if since_improve > 200:
                # Random restart around best
                sig = 2.0 / n
                p = project_simplex(0.7 * best_p + 0.3 * p + np.random.randn(n) * sig)
                since_improve = 0
            elif since_improve > 50:
                p = project_simplex(p + np.random.randn(n) * (1.0 / n))

    return best_p, best_sc

def make_linear_ramp(n):
    """Increasing linear ramp, normalized to sum=1."""
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()

def make_power_ramp(n, alpha=2.0):
    """Power ramp: a_i = (i+1)^alpha, normalized."""
    a = np.power(np.arange(1, n + 1, dtype=np.float64), alpha)

def make_cosine_taper(n, beta=1.0):

    x = np.linspace(0, 1, n)

    return a / a.sum()

def make_quadratic(n):

            p_cur = best_p_global.copy()

            g_b = max(10.0, min(100.0, (deadline - 30.0 - time.monotonic()) * 0.3))
            p_g, sc_g = local_optimize(p_cur, time_budget=g_b)
            if sc_g < best_sc_global:
                best_sc_global = sc_g
                best_p_global = p_g.copy()
            print(f"  Round {round_i}: score={_FMT % best_sc_global}", flush=True)

    # Final validation
    if best_p_global is None:
        best_p_global = make_linear_ramp(700)
        best_sc_global = internal_score(best_p_global)

    best_p_global = np.asarray(best_p_global, dtype=np.float64)
    best_p_global = np.where(np.isfinite(best_p_global), best_p_global, 0.0)
    best_p_global = np.maximum(best_p_global, 0.0)
    s = float(best_p_global.sum())
    if s <= 1e-12:
        best_p_global = make_linear_ramp(700)
    else:
        best_p_global = best_p_global / s
    best_p_global = np.minimum(best_p_global, 1000.0)

    elapsed = time.monotonic() - t0
    print(f"\n=== FINAL: n={len(best_p_global)}, score={_FMT % best_sc_global}, time={elapsed:.1f}s ===", flush=True)

    return [float(x) for x in best_p_global]
