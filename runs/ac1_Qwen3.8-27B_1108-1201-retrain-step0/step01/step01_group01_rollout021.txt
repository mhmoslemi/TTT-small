import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csc_matrix

def compute_score(a):
    """Compute 2*n*max(convolve(a,a))/(sum(a))^2"""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (np.sum(a)**2)

def mirror_descent_step(a, eta, peaks, weights):
    """One mirror descent step."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(peaks, weights):
        # grad[i] += w * 2 * a[k-i] (for valid k-i)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += w * 2.0 * a[k - idx]
    # Update: a[i] *= exp(-eta * grad[i])
    a = a * np.exp(-eta * grad)
    # Renormalize
    a = a / np.sum(a)
    return a

    """Solve the LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    # Build sparse constraint matrix T (2n-1) x n
    # T[k, i] = 1 if 0 <= k-i < n

    cols = []

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
    T = csc_matrix((np.ones(len(rows)), (rows, cols)), shape=(2*n-1, n))

    res = linprog(c, A_ub=T, b_ub=M * np.ones(2*n-1), bounds=[(0, None)] * n, method='highs')

        parent = np.ones(6121)

    n = parent.size
    a0 = parent / parent.sum()

    global_best = a0.copy()
    global_best_score = _compute_score_normalized(a0)
    print(f"n={n}, Initial score: {global_best_score:.6f}")

    # === Phase 1: Multi-start mirror descent ===
    print("\n=== Phase 1: Multi-start Mirror Descent ===")
    starts = _make_diverse_starts(n, rng, a0)
    phase1_frac = 0.20
    phase1_time = phase1_frac * (deadline - time.monotonic())
    time_per_start = phase1_time / len(starts)

    for si, a_start in enumerate(starts):
        if time.monotonic() >= deadline - 30:
            break
        a = a_start.copy()
        start_deadline = min(time.monotonic() + time_per_start, deadline - 30)

        eta = 1.5
        best_local = a.copy()
        best_local_score = _compute_score_normalized(a)
        no_improve = 0
        iter_count = 0

        while time.monotonic() < start_deadline:
            c = np.convolve(a, a)
            c_max = float(np.max(c))
            score = 2.0 * n * c_max

            if score < best_local_score - 1e-12:
                best_local_score = score
                best_local = a.copy()
                no_improve = 0
                # Adapt eta: increase if making good progress
                if iter_count % 100 == 0 and iter_count > 0:
                    eta = min(eta * 1.02, 5.0)
            else:
                no_improve += 1
                if no_improve > 300:
                    eta *= 0.9
                    no_improve = 0
                    if eta < 0.05:
                        break

            iter_count += 1

            # Peak selection with exponential weighting
            n_peaks = 7
            if len(c) <= n_peaks:
                peaks = np.arange(len(c))
                weights = np.ones(len(c)) / len(c)
            else:
                peaks = np.argpartition(c, -n_peaks)[-n_peaks:]
                c_peaks = c[peaks]
                # Sharper weighting for more focused updates
                weights = np.exp(-4.0 * (c_max - c_peaks) / max(c_max, 1e-15))
                weights /= weights.sum()

            grad = _compute_gradient_at_peaks(a, peaks, weights)
            a = _mirror_descent_update(a, grad, eta)

            # Occasionally use anti-correlation heuristic
            if iter_count % 500 == 0:
                k_peak = int(np.argmax(c))
                a_alt = _anti_correlation_update(a, k_peak, n)
                # Blend
                a = 0.7 * a + 0.3 * a_alt
                a = np.maximum(a, 0)
                a = a / a.sum()

        if best_local_score < global_best_score:
            global_best_score = best_local_score
            global_best = best_local.copy()
            print(f"  Start {si+1}/{len(starts)}: NEW BEST! Score: {global_best_score:.6f}")
        else:
            print(f"  Start {si+1}/{len(starts)}: Score {best_local_score:.6f}")

    # === Phase 2: LP-based refinement ===
    print(f"\n=== Phase 2: LP Refinement (best={global_best_score:.6f}) ===")
    a = global_best.copy()
    lp_count = 0
    last_improve_lp = 0

    while time.monotonic() < deadline - 120:
        c = np.convolve(a, a)
        M = float(np.max(c))
        score = 2.0 * n * M

        if score < global_best_score - 1e-12:
            global_best_score = score
            global_best = a.copy()
            last_improve_lp = lp_count

        # Try LP
        t0 = time.monotonic()
        g = _solve_lp(a, M, n, time_limit=15)
        lp_time = time.monotonic() - t0

        if g is not None and float(np.sum(g)) > 1.0 + 1e-8:
            sum_g = float(np.sum(g))
            a_new = g / sum_g
            c_new = np.convolve(a_new, a_new)
            new_M = float(np.max(c_new))
            new_score = 2.0 * n * new_M

            if new_score < score:
                a = a_new
                lp_count += 1
                if new_score < global_best_score:
                    global_best_score = new_score
                    global_best = a.copy()
                    last_improve_lp = lp_count
                print(f"  LP {lp_count}: M {M:.8f} -> {new_M:.8f} (sum_g={sum_g:.6f}, t={lp_time:.1f}s) score={new_score:.6f}")
            else:
                print(f"  LP {lp_count}: no improvement (sum_g={sum_g:.6f}, t={lp_time:.1f}s)")
                # Fall back to mirror descent
                peaks = np.argpartition(c, -5)[-5:]
                c_peaks = c[peaks]
                weights = np.exp(-4.0 * (M - c_peaks) / max(M, 1e-15))
                weights /= weights.sum()
                grad = _compute_gradient_at_peaks(a, peaks, weights)
                a = _mirror_descent_update(a, grad, 0.8)
        else:
            print(f"  LP {lp_count}: failed or no gain (t={lp_time:.1f}s), using MD")
            # Mirror descent fallback
            n_peaks = 10
            peaks = np.argpartition(c, -n_peaks)[-n_peaks:]
            c_peaks = c[peaks]
            weights = np.exp(-4.0 * (M - c_peaks) / max(M, 1e-15))
            weights /= weights.sum()
            grad = _compute_gradient_at_peaks(a, peaks, weights)
            a = _mirror_descent_update(a, grad, 0.8)

        if lp_count - last_improve_lp > 30:
            print("  Stagnation detected, switching to Phase 3")
            break

    # === Phase 3: Intensive mirror descent polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"\n=== Phase 3: Final Polish ({remaining:.0f}s remaining, best={global_best_score:.6f}) ===")
        a = global_best.copy()
        eta = 0.5
        it = 0
        while time.monotonic() < deadline - 5:
            c = np.convolve(a, a)
            c_max = float(np.max(c))
            score = 2.0 * n * c_max

            if score < global_best_score - 1e-14:
                global_best_score = score
                global_best = a.copy()

            it += 1
            # Vary number of peaks for diversity
            n_peaks = 3 + (it % 5) * 2
            peaks = np.argpartition(c, -min(n_peaks, len(c)))[-min(n_peaks, len(c)):]
            c_peaks = c[peaks]
            weights = np.exp(-3.0 * (c_max - c_peaks) / max(c_max, 1e-15))
            weights /= weights.sum()

            grad = _compute_gradient_at_peaks(a, peaks, weights)
            a = _mirror_descent_update(a, grad, eta)

            # Periodic small perturbation to avoid stagnation
            if it % 5000 == 0:
                noise = rng.normal(0, 0.01, n)
                a = a + noise * a
                a = np.maximum(a, 0)
                a = a / a.sum()

    # Final: ensure valid output
    result = [float(max(0.0, x)) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"\nFINAL SCORE: {final_score:.6f}")
    print(f"Best internal score: {global_best_score:.6f}")
    return result
