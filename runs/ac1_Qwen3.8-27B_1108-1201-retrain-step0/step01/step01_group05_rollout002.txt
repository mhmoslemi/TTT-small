        step = lr * grad / gnorm
        a_proposed = project_simplex(a - step, S)
        
        # Quick check: did max increase? If so, halve the step
        c_check = fft_self_conv(a_proposed)
        if c_check.max() > cur_max * 1.05 and t > 50:
            step = lr * 0.25 * grad / gnorm
            a_proposed = project_simplex(a - step, S)
        
        a = a_proposed
        
        # Periodic exact subgradient refinement
        if t % 50 == 0 and beta > 20:
            es_grad, _ = exact_subgradient(c, a, 64)
            eg_norm = np.linalg.norm(es_grad)
            if eg_norm > 1e-14:
                a = project_simplex(a - 0.1 * S / np.sqrt(n) * es_grad / eg_norm * eg_norm / (S/np.sqrt(n)), S)
        
        # Random perturbation to escape local minima
        if t % 200 == 0 and rng.random() < 0.25:
            eps = rng.normal(0, 0.005 * S / n, n)
            a = project_simplex(a + eps, S)
        
        # Patience-based restart
        if patience > 500 and t > 1000:
            a = project_simplex(best_local + rng.normal(0, 0.01 * S / n, n), S)
            patience = 0
            lr = 0.5 * S / np.sqrt(n)
        
        # Progress logging
        if t % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    P1 t={t} elapsed={elapsed:.0f}s beta={beta:.1f} "
                  f"cur={cur_max:.6f} best={best_score:.6f} max-mean={cur_max-cur_mean:.4f}")
    
    return best_local, best_score


def run_phase2(a_init, n, S, max_time, rng):
    """Phase 2: Refined optimization with higher beta and exact subgradient.
    
    Alternates between smooth gradient (high beta) and exact top-k subgradient.
    Returns (best_a, best_maxc).
    """
    a = normalize_to_sum(a_init, n)
    start = time.monotonic()
    
    c = fft_self_conv(a)
    best_score = c.max()
    best_local = a.copy()
    t = 0
    lr = 0.15 * S / np.sqrt(n)
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_self_conv(a)
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        
        # Every 5 iterations: smooth gradient with high beta
        if t % 5 != 0:
            beta = 50.0 + 50.0 * (t / max(1, int(max_time * 5)))
            grad, _, _, _ = smooth_gradient(a, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step_scale = lr * (0.999 ** (t // 5))
            a = project_simplex(a - step_scale * grad / gnorm * gnorm / (S / np.sqrt(n)), S)
        else:
            # Exact subgradient with large top-k
            top_k = min(512, 2 * n - 1)
            es_grad, _ = exact_subgradient(c, a, top_k)
            eg_norm = np.linalg.norm(es_grad)
            if eg_norm < 1e-14:
                break
            step_scale = lr * 0.5 * (0.9995 ** (t // 5))
            a = project_simplex(a - step_scale * es_grad / eg_norm * eg_norm / (S / np.sqrt(n)), S)
        
        # Small perturbation
        if t % 300 == 0 and rng.random() < 0.15:
            a = project_simplex(a + rng.normal(0, 0.003 * S / n, n), S)
        
        if t % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    P2 t={t} elapsed={elapsed:.0f}s cur={cur_max:.6f} best={best_score:.6f}")
    
    return best_local, best_score


def run_phase3(a_init, n, S, max_time, rng):
    """Phase 3: Ultra-fine polish with very small steps and large top-k.
    
    Returns (best_a, best_maxc).
    """
    a = normalize_to_sum(a_init, n)
    start = time.monotonic()
    
    c = fft_self_conv(a)
    best_score = c.max()
    best_local = a.copy()
    t = 0
    base_lr = 0.05 * S / np.sqrt(n)
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_self_conv(a)
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        
        # Use very large top-k for precise gradient
        top_k = min(1024, 2 * n - 1)
        es_grad, _ = exact_subgradient(c, a, top_k)
        eg_norm = np.linalg.norm(es_grad)
        if eg_norm < 1e-15:
            break
        
        # Very small steps with slow decay
        scale = base_lr * (0.9999 ** t) / (S / np.sqrt(n))
        a = project_simplex(a - scale * es_grad, S)
        
        # Occasional smooth gradient step for broader view
        if t % 10 == 0:
            grad, _, _, _ = smooth_gradient(a, 100.0)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-14:
                a = project_simplex(a - scale * 0.5 * grad, S)
        
        if t % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    P3 t={t} elapsed={elapsed:.0f}s cur={cur_max:.7f} best={best_score:.7f}")
    
    return best_local, best_score


def propose_candidate(seed=42, budget_s=2400):
    """Search for a non-negative sequence minimizing 2*n*max_conv/sum^2."""
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + max(0.0, total_budget - 15.0)
    
    # Load initial sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given = np.clip(given, 0.0, 1000.0)
    except Exception:
        given = None
    
    global_best = np.inf
    global_a = None
    
    # Build initializations
    n_main = 8192
    S_main = np.sqrt(2.0 * n_main)
    
    inits = []
    if given is not None and given.size >= 256:
        inits.append(("given", given.copy()))
    
    # Diverse random initializations
    inits.append(("rand1", rng.random(n_main) * 0.2 + 0.8))
    
    # Smooth (low frequency) initialization
    tt = np.linspace(0, 1, n_main)
    inits.append(("smooth1", 1.0 + 0.3 * np.sin(2 * np.pi * 2.0 * tt)
                  + 0.15 * np.cos(2 * np.pi * 5.0 * tt)))
    inits.append(("smooth2", 1.0 + 0.4 * np.cos(2 * np.pi * 3.0 * tt)
                  + 0.1 * np.sin(2 * np.pi * 11.0 * tt)))
    
    # Windowed initialization (tapered edges reduce sidebar effects)
    window = 0.5 * (1 + np.cos(2 * np.pi * tt))
    window = window / window.sum() * n_main
    inits.append(("window", 0.5 + 0.5 * window))
    
    n_inits = len(inits)
    print(f"Starting search: {n_inits} initializations, budget={total_budget:.0f}s")
    print(f"Primary length: {n_main}, S={S_main:.4f}")
    
    # === Phase 1: Exploration (50% of time) ===
    explore_budget = 0.50 * (deadline - time.monotonic())
    per_init = explore_budget / n_inits
    
    for idx, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - 0.45 * (deadline - time.monotonic() + 0.001):
            break
        n = len(arr)
        S = np.sqrt(2.0 * n)
        seg = max(3.0, per_init)
        
        print(f"\n  Init [{name}] n={n}, time={seg:.0f}s")
        t0 = time.monotonic()
        
        a_res, score = run_phase1(arr, n, S, seg, rng)
        
        # Quick verify
        ver = evaluate_sequence(a_res.tolist())
        print(f"  Init [{name}] done in {time.monotonic()-t0:.0f}s: "
              f"maxc={score:.6f} verify={ver:.6f}")
        
        if score < global_best:
            global_best = score
            global_a = a_res.copy()
    
    print(f"\n  After Phase 1: best={global_best:.6f}")
    
    # === Phase 2: Focused refinement (30% of time) ===
    remaining = deadline - time.monotonic()
    refine_budget = 0.55 * remaining
    remaining_after_refine = remaining - refine_budget
    
    if global_a is not None and refine_budget > 5:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        
        print(f"\n  Phase 2: refining best (budget={refine_budget:.0f}s)")
        t0 = time.monotonic()
        a_ref, score_ref = run_phase2(global_a, n, S, refine_budget, rng)
        ver_ref = evaluate_sequence(a_ref.tolist())
        print(f"  Phase 2 done in {time.monotonic()-t0:.0f}s: "
              f"maxc={score_ref:.6f} verify={ver_ref:.6f}")
        
        if score_ref < global_best:
            global_best = score_ref
            global_a = a_ref.copy()
    
    # === Phase 3: Ultra-fine polish (20% of time) ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 5:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        polish_budget = remaining * 0.85
        
        print(f"\n  Phase 3: ultra-polish (budget={polish_budget:.0f}s)")
        t0 = time.monotonic()
        a_pol, score_pol = run_phase3(global_a, n, S, polish_budget, rng)
        ver_pol = evaluate_sequence(a_pol.tolist())
        print(f"  Phase 3 done in {time.monotonic()-t0:.0f}s: "
              f"maxc={score_pol:.7f} verify={ver_pol:.7f}")
        
        if score_pol < global_best:
            global_best = score_pol
            global_a = a_pol.copy()
    
    # === Final: try one more round if time permits ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 10:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        print(f"\n  Final squeeze (budget={remaining:.0f}s)")
        a_fin, score_fin = run_phase3(global_a, n, S, remaining * 0.9, rng)
        ver_fin = evaluate_sequence(a_fin.tolist())
        print(f"  Final: maxc={score_fin:.7f} verify={ver_fin:.7f}")
        if score_fin < global_best:
            global_best = score_fin
            global_a = a_fin.copy()
    
    # Ensure valid output
    if global_a is None:
        global_a = np.full(n_main, S_main / n_main, dtype=np.float64)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.full(n_main, S_main / n_main, dtype=np.float64)
    
    final_ver = evaluate_sequence(global_a.tolist())
    print(f"\n  FINAL RESULT: {final_ver:.7f} (length={len(global_a)})")
    
    return [float(x) for x in global_a]
