```python
def spectral_flatten(a, n_iter=100):
    """Flatten the power spectrum to reduce autocorrelation peak."""
    n = len(a)
    # Pad to 2n-1 for linear convolution
    N = 2 * n - 1
    A = np.fft.rfft(a, n=N)
    for _ in range(n_iter):
        # Compute current autocorrelation
        P = np.abs(A)**2
        # Target: flat spectrum
        # But we need to maintain non-negativity of a
        # Use a soft constraint: penalize deviation from flat
        target = P.sum() / len(P)
        # Update: reduce peaks in spectrum
        P_new = P * np.exp(-0.1 * (P - target) / target)
        # Redistribute
        P_new = P_new * (P.sum() / P_new.sum())
        # Get new a
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, n=N)[:n]
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        # Normalize
        if a_new.sum() > 0:
            a_new = a_new * a.sum() / a_new.sum()
        a = a_new
    return a
```

```python
def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def softmax_grad(a, tau=10.0):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    for j in range(n):
        # (a*a)[m] = sum_k a[k]*a[m-k], d/d a[j] = sum_m w[m] * 2*a[m-j] (for valid m-j)

        k_lo = max(0, 0-j)  # k = m-j >= 0, always true

            m_hi = j + k_hi

def lp_refine(a, M, n_max=2048):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > n_max:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix efficiently
    A = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = a[m - j]
    
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def lp_optimize(a, deadline, max_iters=100):

def spectral_refine(a, deadline, max_iters=200):
    """Try to flatten the power spectrum to reduce autocorrelation peak."""
    n = len(a)
    a = np.array(a, dtype=np.float64)
    a = np.maximum(a, 0)
    S = a.sum()
    
    best = a.copy()
    best_score = compute_obj(a)
    
    N = 2 * n - 1
    for it in range(max_iters):
        if time.monotonic() > deadline - 10:
            break
        
        A = np.fft.rfft(a, n=N)
        P = np.abs(A)**2
        P_mean = P.mean()
        
        # Reduce peaks in spectrum
        alpha = 0.1 + 0.4 * (it / max_iters)
        P_new = P * np.exp(-alpha * (P - P_mean) / (P_mean + 1e-10))
        P_new = P_new * (P.sum() / (P_new.sum() + 1e-10))
        
        # Reconstruct maintaining phase
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, n=N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            break
        a_new = a_new * S / a_new.sum()
        
        s_new = compute_obj(a_new)
        if s_new < best_score:
            best = a_new.copy()
            best_score = s_new
            a = a_new
    
    return best, best_score
```

```python
"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Combines multiplicative mirror descent, improved LP-based fixed-point iteration,
and spectral flattening. Explores multiple sequence lengths and diverse starting points.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def softmax_weights(c, tau):
    """Compute softmax weights over convolution values."""
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    return w


def compute_grad_soft(a, w):
    """Compute gradient: sum_m w[m] * d(a*a)[m]/da[j]."""
    n = len(a)
    grad = np.zeros(n)
    # (a*a)[m] = sum_j a[j]*a[m-j]
    # d(a*a)[m]/da[j] = 2*a[m-j] for 0<=m-j<n, else 0 (when j != m-j)
    # Actually: = a[m-j] + a[j-m] for valid indices
    # But since a[j-m] = a[m-j] only if... no, they're different indices.
    # Let me be precise:
    # (a*a)[m] = sum_{j: valid} a[j] * a[m-j]
    # d/da[j'] = sum_{j: valid} [δ_{j,j'}*a[m-j] + a[j]*δ_{m-j,j'}]
    # = a[m-j'] (if 0<=m-j'<n) + a[j'-m] (if 0<=j'-m<n, i.e., j'>=m)
    # Wait, the second term: a[j] * δ_{m-j, j'} means m-j = j', so j = m-j'.
    # This contributes a[m-j'] if 0<=m-j'<n.
    # Hmm, both terms give the same: a[m-j'] if 0<=m-j'<n.

    # d/da[j'] of a[j']^2 = 2*a[j']
    # And the formula gives 2*a[m-j'] = 2*a[j'] (since m-j' = j'). Same!
    # So the formula d(a*a)[m]/da[j'] = 2*a[m-j'] * 1_{0<=m-j'<n} is always correct.

        m_lo = max(0, j)
        m_hi = min(2*n - 2, j + n - 1)
        if m_hi >= m_lo:
            k_lo = m_lo - j  # = 0
            k_hi = m_hi - j  # = n-1 or less
            grad[j] = 2.0 * np.dot(a[0:k_hi+1], w[m_lo:m_lo+k_hi+1])

def compute_grad_hard(a):
    """Compute subgradient at the argmax lag."""
    n = len(a)
    c = np.convolve(a, a)
    m_star = int(np.argmax(c))
    grad = np.zeros(n)
    # d(a*a)[m_star]/da[j] = 2*a[m_star-j] for max(0,m_star-n+1) <= j <= min(n-1, m_star)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m_star - j]
    return grad


def mirror_step(a, grad, lr):
    """Multiplicative (mirror) descent step with fixed sum."""
    S = a.sum()
    # Center the gradient to preserve sum (approximately)
    grad_centered = grad - grad.mean()
    a_new = a * np.exp(-lr * grad_centered)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new * (S / a_new.sum())
    return a_new

    n = len(a0)
    a = np.array(a0, dtype=np.float64).copy()
    a = np.maximum(a, 1e-10)
    S = a.sum()
    if S < 1e-10:

        S = 1.0
    a = a / S  # Normalize to sum = 1

    best_score = compute_obj(a)

        if time.monotonic() > deadline - 5:

        tau = tau_init + (tau_end - tau_init) * frac

def solve_lp_min_crosspeak(f, S_fixed, n_max=2048):
    """
    Solve LP: min M' s.t. conv(f,g)[m] <= M' for all m, sum(g) = S_fixed, g >= 0.
    Returns (g, M') or (None, None).
    """
    n = len(f)
    if n > n_max:
        return None, None
    
    # Variables: g[0..n-1], M'
    # Total: n+1 variables
    # Objective: minimize M' -> c_obj = [0]*n + [1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M'
    
    # Inequality constraints: conv(f,g)[m] - M' <= 0 for all m
    # A_ub has shape (2n-1, n+1)
    A_ub = np.zeros((2 * n - 1, n + 1))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A_ub[m, j] = f[m - j]
        A_ub[m, n] = -1.0  # coefficient of M'
    
    b_ub = np.zeros(2 * n - 1)
    
    # Equality constraint: sum(g) = S_fixed
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0  # sum of g = S_fixed
    b_eq = np.array([S_fixed])
    
    # Bounds: g[j] >= 0, M' >= 0
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = res.x[:n]
            M_prime = res.x[n]
            return g, M_prime
    except Exception as e:
        print(f"  LP error: {e}")
    
    return None, None


def lp_fixed_point(a0, deadline, max_iters=100, blend_rates=None):
    """
    LP-based fixed-point iteration:
    g = argmin { max conv(f,g) : sum(g)=S, g>=0 }
    f = (1-t)*f + t*g
    """
    if blend_rates is None:
        blend_rates = [1.0, 0.8, 0.5, 0.3, 0.1]
    
    n = len(a0)
    a = np.array(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, compute_obj(a)
    
    best = a.copy()
    best_score = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        
        # Solve LP
        g, M_prime = solve_lp_min_crosspeak(a, S)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        if g.sum() < 1e-10:
            break
        g = g * S / g.sum()
        
        # Try blending
        improved = False
        for t in blend_rates:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            if a_new.sum() < 1e-10:
                continue
            a_new = a_new * S / a_new.sum()
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-10:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break
        
        if not improved:
            # Stuck - try a perturbation
            rng = np.random.default_rng(hash((n, it)) % 2**32)
            a_pert = a + 0.05 * rng.random(n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert * S / a_pert.sum()
            s_pert = compute_obj(a_pert)
            if s_pert < best_score:
                a = a_pert
                best = a.copy()
                best_score = s_pert
            else:
                break  # Truly stuck
    
    return best, best_score


def spectral_flatten(a0, deadline, max_iters=500):
    """Flatten the power spectrum to reduce autocorrelation peak."""
    n = len(a0)
    a = np.array(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, compute_obj(a)
    
    best = a.copy()
    best_score = compute_obj(a)
    N = 2 * n - 1
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        
        A = np.fft.rfft(a, n=N)
        P = np.abs(A)**2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Alpha: strength of flattening
        alpha = 0.05 + 0.3 * (it / max_iters)
        
        # Reduce peaks in power spectrum
        ratio = P / P_mean
        P_new = P * np.exp(-alpha * (ratio - 1.0))
        P_new = np.maximum(P_new, 1e-30)
        P_new = P_new * (P.sum() / P_new.sum())
        
        # Reconstruct with same phase
        A_new = np.sqrt(P_new) * (A / (np.abs(A) + 1e-30))
        a_new = np.fft.irfft(A_new, n=N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-10:
            break
        a_new = a_new * S / a_new.sum()
        
        s_new = compute_obj(a_new)
        if s_new < best_score:
            best = a_new.copy()
            best_score = s_new
            a = a_new
    
    return best, best_score


def run_gradient_descent(a0, deadline, max_iters=3000, lr_init=0.05):
    """Standard projected gradient descent (additive)."""
    n = len(a0)
    a = np.array(a0, dtype=np.float64).copy()
    a = np.maximum(a, 1e-10)
    S = a.sum()
    a = a / S
    
    best = a.copy()
    best_score = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        
        frac = float(it) / float(max_iters)
        lr = 0.001 + (lr_init - 0.001) * 0.5 * (1.0 + np.cos(np.pi * frac))
        
        c = np.convolve(a, a)
        m_star = int(np.argmax(c))
        grad = np.zeros(n)
        lo = max(0, m_star - n + 1)
        hi = min(n - 1, m_star)
        for j in range(lo, hi + 1):
            grad[j] = 2.0 * a[m_star - j]
        
        # Project out mean component (fixed sum constraint)
        grad -= grad.mean()
        
        a = a - lr * grad
        a = np.maximum(a, 1e-10)
        a = a * S / a.sum()
        
        if (it + 1) % 300 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    
    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Combines mirror descent, LP fixed-point, gradient descent,
    and spectral flattening. Explores multiple sequence lengths and starting points.
    
    Strategy:
    1. Generate diverse starting points for various lengths.
    2. Apply quick mirror descent to all starts.
    3. Apply LP fixed-point to top candidates.
    4. Apply spectral flattening to the best.
    5. Final intensive optimization on the best.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    
    def update_best(seq, score):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = np.array(seq, dtype=np.float64).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:.0f}s] New best: score={best_score:.6f}, n={len(seq)}")
    
    # --- Try to use previous best ---
    prev_seq = None
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)) and np.all(gs >= -1e-10):
            gs = np.maximum(gs, 0.0)
            prev_seq = gs
            s_prev = compute_obj(gs)
            print(f"Previous sequence: n={len(gs)}, score={s_prev:.6f}")
            update_best(gs, s_prev)
    except (NameError, Exception) as e:
        print(f"No previous sequence available: {e}")
    
    # --- Generate starting points ---
    starts = []
    
    if prev_seq is not None:
        # Use prefixes of previous best
        for L in [128, 256, 384, 512, min(1024, len(prev_seq))]:
            if L <= len(prev_seq):
                starts.append(prev_seq[:L].copy())
        # Full previous
        starts.append(prev_seq.copy())
    
    # Random starts
    for n in [128, 256, 512, 1024]:
        starts.append(rng.random(n) * 0.8 + 0.2)
    
    # Structured starts
    for n in [256, 512, 1024]:
        i = np.arange(n, dtype=np.float64)
        # Various frequency patterns
        starts.append(0.3 + 0.7 * np.abs(np.cos(2 * np.pi * i * 3.0 / n)))
        starts.append(0.3 + 0.7 * np.abs(np.cos(2 * np.pi * i * 5.0 / n)))
        starts.append(0.2 + 0.8 * np.exp(-0.5 * ((i - n/2) / (n*0.2))**2))
        # Periodic
        starts.append(0.5 + 0.3 * np.cos(2*np.pi*i/4.0) + 0.2 * np.cos(2*np.pi*i/7.0))
        starts = [s for s in starts if s[-1] is not None]
        # Make sure all are positive
        starts[-1] = np.maximum(starts[-1], 0.05)
    
    # Sparse starts (mostly zero with a few peaks)
    for n in [256, 512]:
        a_sparse = np.zeros(n)
        n_peaks = n // 8
        peak_positions = rng.choice(n, size=n_peaks, replace=False)
        a_sparse[peak_positions] = 1.0
        starts.append(a_sparse + 0.01)
    
    # Sort by length
    starts.sort(key=len)
    
    print(f"\n=== Phase 1: Quick optimization ({len(starts)} starts) ===")
    
    # --- Phase 1: Quick mirror descent on all starts ---
    # Time budget per start: depend on how many starts
    n_starts = len(starts)
    time_per_start = min(80.0, (deadline - t_start - 300.0) / max(n_starts, 1))
    
    for idx, a0 in enumerate(starts):
        if time.monotonic() > deadline - 200.0:
            break
        remaining = deadline - time.monotonic() - 200.0
        t_sub = time.monotonic() + min(time_per_start, max(10.0, remaining / (n_starts - idx)))
        
        seq, score = run_mirror_descent(a0, t_sub, max_iters=3000,
                                         tau_init=5.0, tau_end=60.0, lr_init=0.08)
        update_best(seq, score)
    
    # --- Phase 2: LP fixed-point on best and diverse candidates ---
    print(f"\n=== Phase 2: LP fixed-point refinement ===")
    
    # Build candidate list: best + a few diverse ones
    lp_candidates = []
    if best_seq is not None:
        lp_candidates.append(best_seq)
    
    # Add some from different lengths
    seen_lens = {len(best_seq)} if best_seq is not None else set()
    for s in starts:
        if len(s) not in seen_lens and len(lp_candidates) < 6:
            # Quick optimize first
            t_sub = time.monotonic() + 20.0
            seq, _ = run_mirror_descent(s, t_sub, max_iters=1000)
            lp_candidates.append(seq)
            seen_lens.add(len(seq))
    
    for i, cand in enumerate(lp_candidates):
        if time.monotonic() > deadline - 100.0:
            break
        t_sub = deadline - 60.0
        seq, score = lp_fixed_point(cand, t_sub, max_iters=80)
        update_best(seq, score)
        
        # Follow up with mirror descent from the LP result
        if time.monotonic() < deadline - 60.0:
            t_sub2 = deadline - 30.0
            seq2, score2 = run_mirror_descent(seq, t_sub2, max_iters=2000,
                                               tau_init=20.0, tau_end=100.0, lr_init=0.05)
            update_best(seq2, score2)
    
    # --- Phase 3: Spectral flattening ---
    print(f"\n=== Phase 3: Spectral refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 60.0:
        t_sub = deadline - 30.0
        seq, score = spectral_flatten(best_seq, t_sub, max_iters=300)
        update_best(seq, score)
    
    # --- Phase 4: Intensive final optimization ---
    print(f"\n=== Phase 4: Intensive final optimization ===")
    if best_seq is not None:
        # Long mirror descent
        if time.monotonic() < deadline - 60.0:
            t_sub = deadline - 30.0
            seq, score = run_mirror_descent(best_seq, t_sub, max_iters=10000,
                                             tau_init=30.0, tau_end=200.0, lr_init=0.03)
            update_best(seq, score)
        
        # LP pass
        if time.monotonic() < deadline - 20.0:
            t_sub = deadline - 10.0
            seq, score = lp_fixed_point(seq, t_sub, max_iters=30)
            update_best(seq, score)
        
        # Another mirror pass
        if time.monotonic() < deadline - 10.0:
            t_sub = deadline - 5.0
            seq, score = run_mirror_descent(seq, t_sub, max_iters=3000,
                                             tau_init=50.0, tau_end=300.0, lr_init=0.02)
            update_best(seq, score)
        
        # Spectral
        if time.monotonic() < deadline - 5.0:
            t_sub = deadline - 3.0
            seq, score = spectral_flatten(seq, t_sub, max_iters=100)
            update_best(seq, score)
    
    # --- Final validation ---
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = np.minimum(best_seq, 1000.0)
        final_score = compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
        
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        
        return [float(x) for x in best_seq]
    
    # Fallback
    print("Falling back to uniform sequence of length 512.")
    return [1.0] * 512
```
