"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses: (1) FFT-accelerated gradient descent on log-sum-exp of self-convolution,
(2) LP-based direction finding for large jumps, (3) local search polishing.
Explores multiple sequence lengths and diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize(a):
    """Normalize to non-negative with sum=1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    conv = np.convolve(a, a)
    mx = conv.max()
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    return 2.0 * n * mx / (s * s)


def grad_logsumexp(a, T):
    """
    Compute gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    conv = np.convolve(a, a)  # length 2n-1
    # Stable softmax
    shifted = T * conv - (T * conv).max()
    w = np.exp(shifted)
    w_sum = w.sum()
    w /= w_sum

    # B[j] = sum_{m=0}^{n-1} a[m] * w[j+m]
    # Using: v = w[::-1], c = convolve(a, v), B[j] = c[2n-2-j]
    v = w[::-1]
    # Full convolution length: n + (2n-1) - 1 = 3n-2
    N = 1
    while N < 3 * n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    fv = np.fft.rfft(v, N)
    c = np.fft.irfft(fa * fv, N)[:3 * n - 2]
    # B[j] = c[2n-2-j] for j=0,...,n-1
    B = c[2 * n - 2 : n - 2 : -1] if n > 1 else np.array([c[0]])
    # Ensure B has length n
    B = B[:n]

    # grad[j] = 2*B[j] - w[2j]*a[j]
    grad = 2.0 * B - w[2 * np.arange(n)] * a
    return grad


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def gd_optimize(a, time_limit, T_start=50.0, T_end=5000.0, lr_scale=0.005,
                n_iters=20000, verbose=False):
    """
    Gradient descent on log-sum-exp with adaptive temperature and learning rate.
    """
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    if verbose:
        print(f"    GD start: score={best_score:.6f}, n={n}")

    for it in range(n_iters):
        if time.monotonic() - start_time > time_limit:
            break
        frac = it / max(1, n_iters - 1)
        # Temperature schedule: start moderate, increase, then oscillate
        if frac < 0.3:
            T = T_start * (T_end / T_start) ** (frac / 0.3)
        else:
            # Oscillate between moderate and high T
            osc = 0.5 + 0.5 * np.sin(2 * np.pi * (frac - 0.3) / 0.7 * 3)
            T = T_start * 5 * (T_end / (T_start * 5)) ** osc

        # Adaptive learning rate
        lr = lr_scale * (1.0 / (1.0 + 0.1 * it / max(1, n_iters / 10)))

        grad = grad_logsumexp(a, T)
        # Normalize gradient for stable steps
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-30:
            step = lr * grad / np.sqrt(n)
            a_new = a - step
            a_new = project_simplex(a_new)
        else:
            a_new = a.copy()

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            if verbose and (it % 2000 == 0):
                print(f"    GD iter {it:6d}: score={score:.6f}, T={T:.1f}")
        a = a_new

    return best_a, best_score


def build_conv_matrix(a):
    """Build sparse convolution matrix for LP."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def lp_direction(a, M, timeout=20.0):
    """
    Find direction g >= 0 maximizing sum(g) subject to (a*g)[k] <= M for all k.
    Returns normalized g or None.
    """
    n = len(a)
    if n > 5000:
        return None  # Too large for LP
    try:
        A = build_conv_matrix(a)
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            g = np.maximum(0.0, res.x)
            s = g.sum()
            if s > 1.0 + 1e-8:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(a, time_limit, verbose=False):
    """LP-based iterative refinement."""
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")

    stagnant = 0
    it = 0
    max_iters = 200
    while time.monotonic() - start_time < time_limit and it < max_iters:
        it += 1
        conv = np.convolve(a, a)
        score = compute_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 30:
            break

        M = conv.max()
        remaining = time_limit - (time.monotonic() - start_time)
        if remaining < 5:
            break
        timeout = max(2.0, min(15.0, remaining * 0.2))
        g = lp_direction(a, M, timeout=timeout)
        if g is None:
            break
        alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = project_simplex(a_new)
        new_score = compute_score(a_new)
        if new_score <= score + 1e-10:
            a = a_new
        else:
            alpha2 = alpha * 0.3
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = project_simplex(a_new)
            a = a_new

    if verbose:
        print(f"    LP done: best={best_score:.6f}, {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search(a, time_limit, rng, verbose=False):
    """Local search with random mutations."""
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    if verbose:
        print(f"    LS start: score={best_score:.6f}")
    accept_count = 0

    while time.monotonic() - start_time < time_limit:
        # Random smooth mutation
        width = max(2, int(rng.integers(n // 300, n // 5)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()

        # Gaussian perturbation on a window
        noise = rng.normal(0.0, 0.15, size=stop - start)
        candidate[start:stop] *= np.exp(noise)

        # Occasionally add a bump
        if rng.random() < 0.2:
            bump_center = int(rng.integers(0, n))
            bump_width = max(1, n // 20)
            idx = np.arange(n)
            candidate += 0.1 * np.exp(-0.5 * ((idx - bump_center) / bump_width) ** 2)

        # Occasionally enforce symmetry
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])

        candidate = project_simplex(candidate)
        score = compute_score(candidate)
        if score < best_score - 1e-12:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 100 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {accept_count} accepts, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def init_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def init_blackman(n):
    if n == 1:
        return np.ones(1)
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def init_alternating(n, ratio=2.6):
    """Alternating pattern [1, ratio, 1, ratio, ...] - exploits misalignment."""
    a = np.zeros(n)
    a[0::2] = 1.0
    a[1::2] = ratio
    return a


def init_random_smooth(n, rng, n_bumps=4):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 15), max(2, n // 4)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    return a


def init_triangle(n):
    """Triangle that peaks at one end - maximizes asymmetry."""
    t = np.arange(n, dtype=np.float64)
    return np.maximum(0.0, 1.0 - t / (n - 1)) if n > 1 else np.ones(1)


def init_two_peaks(n, rng):
    """Two Gaussian peaks at different positions."""
    a = np.zeros(n)
    t = np.arange(n, dtype=np.float64)
    c1 = float(rng.integers(0, n // 3))
    c2 = float(rng.integers(2 * n // 3, n))
    w1 = max(1.0, n * 0.08)
    w2 = max(1.0, n * 0.12)
    a += np.exp(-0.5 * ((t - c1) / w1) ** 2)
    a += 1.5 * np.exp(-0.5 * ((t - c2) / w2) ** 2)
    return a


def full_optimize(a, time_limit, rng, verbose=True, use_lp=True):
    """
    Multi-stage optimization: GD -> LP -> GD -> LS.
    """
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)

    # Stage 1: Gradient descent (fast, many iterations)
    t1 = time_limit * 0.35
    if verbose:
        print(f"    Stage 1 (GD, {t1:.0f}s)")
    a1, s1 = gd_optimize(a, time_limit=t1, verbose=verbose)
    if s1 < best_score:
        best_score = s1
        best_a = a1.copy()
    a = a1

    # Stage 2: LP refinement (if n is small enough)
    if use_lp and n <= 3000:
        remaining = time_limit - (time.monotonic() - start_time)
        t2 = remaining * 0.35
        if t2 > 10:
            if verbose:
                print(f"    Stage 2 (LP, {t2:.0f}s)")
            a2, s2 = lp_refine(a, time_limit=t2, verbose=verbose)
            if s2 < best_score:
                best_score = s2
                best_a = a2.copy()
            a = a2

    # Stage 3: More GD with different parameters
    remaining = time_limit - (time.monotonic() - start_time)
    t3 = remaining * 0.35
    if t3 > 5:
        if verbose:
            print(f"    Stage 3 (GD fine, {t3:.0f}s)")
        a3, s3 = gd_optimize(a, time_limit=t3, T_start=200.0, T_end=10000.0,
                             lr_scale=0.002, verbose=verbose)
        if s3 < best_score:
            best_score = s3
            best_a = a3.copy()
        a = a3

    # Stage 4: Local search polishing
    remaining = time_limit - (time.monotonic() - start_time)
    if remaining > 5:
        if verbose:
            print(f"    Stage 4 (LS, {remaining:.0f}s)")
        a4, s4 = local_search(a, time_limit=remaining, rng=rng, verbose=verbose)
        if s4 < best_score:
            best_score = s4
            best_a = a4.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function. Tries multiple initializations and lengths,
    then intensively optimizes the best candidate.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20.0  # Leave 20s buffer

    best_score = float('inf')
    best_seq = None

    # Get previous best if available
    try:
        prev_seq = np.array(height_sequence_1, dtype=np.float64)
        prev_score = compute_score(prev_seq)
        print(f"Previous best: score={prev_score:.6f}, n={len(prev_seq)}")
    except (NameError, UnboundLocalError):
        prev_seq = None

    # Define configurations: (n, init_type, time_fraction)
    configs = []

    # Use previous best as a strong starting point
    if prev_seq is not None:
        n_prev = len(prev_seq)
        configs.append((n_prev, 'previous', 0.18))
        # Try slightly different lengths
        configs.append((int(n_prev * 1.2), 'interpolated_prev', 0.08))
        configs.append((int(n_prev * 0.8), 'interpolated_prev', 0.08))

    # Various initializations at different lengths
    configs.extend([
        (500, 'raised_cosine', 0.07),
        (700, 'raised_cosine', 0.06),
        (1000, 'raised_cosine', 0.07),
        (1500, 'raised_cosine', 0.06),
        (500, 'gaussian', 0.05),
        (1000, 'gaussian', 0.05),
        (2000, 'gaussian', 0.04),
        (500, 'blackman', 0.04),
        (1000, 'blackman', 0.04),
        (500, 'alternating', 0.06),
        (1000, 'alternating', 0.05),
        (2000, 'alternating', 0.04),
        (500, 'triangle', 0.04),
        (1000, 'triangle', 0.04),
        (500, 'random1', 0.04),
        (1000, 'random1', 0.04),
        (2000, 'random1', 0.03),
        (500, 'two_peaks', 0.03),
        (1000, 'two_peaks', 0.03),
    ])

    # Track per-config bests for potential re-optimization
    per_config_best = []

    for n, init_type, time_frac in configs:
        remaining = deadline - time.monotonic()
        if remaining < 15.0:
            break
        time_limit = min(time_frac * budget_s, remaining * 0.7)
        if time_limit < 10.0:
            continue

        print(f"\n=== n={n}, init={init_type}, time={time_limit:.0f}s ===")

        # Generate initialization
        if init_type == 'previous':
            a = prev_seq.copy()
        elif init_type == 'interpolated_prev':
            # Interpolate previous sequence to new length
            from numpy import interp
            x_old = np.linspace(0, 1, len(prev_seq))
            x_new = np.linspace(0, 1, n)
            a = interp(x_new, x_old, prev_seq)
            a = np.maximum(0, a)
        elif init_type == 'raised_cosine':
            a = init_raised_cosine(n)
        elif init_type == 'gaussian':
            a = init_gaussian(n, sigma_frac=0.2 + 0.1 * rng.random())
        elif init_type == 'blackman':
            a = init_blackman(n)
        elif init_type == 'alternating':
            ratio = 2.0 + 1.5 * rng.random()
            a = init_alternating(n, ratio=ratio)
        elif init_type == 'triangle':
            a = init_triangle(n)
            if rng.random() < 0.5:
                a = a[::-1]
        elif init_type == 'random1':
            a = init_random_smooth(n, rng, n_bumps=3)
        elif init_type == 'two_peaks':
            a = init_two_peaks(n, rng)
        else:
            a = init_raised_cosine(n)

        a = normalize(a)
        init_score = compute_score(a)
        print(f"    Init score: {init_score:.6f}")

        # Optimize
        use_lp = (n <= 2000 and rng.random() < 0.6)
        a_opt, score = full_optimize(a, time_limit, rng, verbose=(time_limit > 30),
                                     use_lp=use_lp)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  *** NEW BEST: score={score:.6f}, n={n}, elapsed={time.monotonic()-total_start:.1f}s")

        per_config_best.append((a_opt, score, n, init_type))

    # Sort per-config results and re-optimize top few
    per_config_best.sort(key=lambda x: x[1])
    print(f"\n--- Top candidates ---")
    for i, (a_o, s_o, n_o, t_o) in enumerate(per_config_best[:5]):
        print(f"  #{i+1}: score={s_o:.6f}, n={n_o}, init={t_o}")

    # Intensive optimization on the best 2-3 candidates
    remaining = deadline - time.monotonic()
    n_top = min(3, len(per_config_best))
    for i in range(n_top):
        if remaining < 30:
            break
        a_o, s_o, n_o, t_o = per_config_best[i]
        t_alloc = remaining * (0.5 if i == 0 else 0.25)
        print(f"\n=== Intensive opt on candidate #{i+1} (n={n_o}, score={s_o:.6f}, t={t_alloc:.0f}s) ===")
        a_int, s_int = full_optimize(a_o, t_alloc, rng, verbose=True, use_lp=(n_o <= 2500))
        if s_int < best_score:
            best_score = s_int
            best_seq = a_int.copy()
            print(f"  *** NEW BEST: score={s_int:.6f}")
        remaining = deadline - time.monotonic()

    # Final local search on the overall best
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print(f"\n=== Final polishing (t={remaining:.0f}s) ===")
            a_pol, s_pol = local_search(best_seq, time_limit=remaining * 0.5, rng=rng, verbose=True)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = a_pol.copy()

            # One more GD pass
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_gd, s_gd = gd_optimize(best_seq, time_limit=remaining2,
                                         T_start=500, T_end=20000, lr_scale=0.001,
                                         verbose=True)
                if s_gd < best_score:
                    best_score = s_gd
                    best_seq = a_gd.copy()

    if best_seq is None:
        best_seq = init_raised_cosine(500)
        best_score = compute_score(best_seq)

    # Verify with evaluate_sequence
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"\nFinal: score={best_score:.6f}, verified={verify_score:.6f}, n={len(best_seq)}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")

    return [float(x) for x in best_seq]
