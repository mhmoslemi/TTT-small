```python
def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    return a * (S / s)
```

Actually, the constraint (w ⋆ a)[m] = C for all m is n equations, and they're not all independent. The sum of all (w ⋆ a)[m] for m = 0 to n-1 is:
sum_m sum_i a[i]*w[i+m] = sum_i a[i] * sum_m w[i+m] = sum_i a[i] * (sum of w over a window of length n starting at i)

This doesn't simplify to a constant in general. So the n equations plus sum a = 1 might be consistent or overdetermined.

Let me think differently. The condition is that the cross-correlation is constant. In the frequency domain:
DFT(w ⋆ a) = DFT(a) * conj(DFT(w)) (from earlier derivation, with the right convention)

Wait, I derived earlier that DFT(a ⋆ w) = conj(A) * W. So DFT(w ⋆ a) = W * conj(A) (same thing since multiplication is commutative).

For (w ⋆ a)[m] = C for all m (extending a and w by zero), DFT(w ⋆ a) = C * N * δ (where N is the DFT length and δ is 1 at frequency 0, 0 elsewhere).

So: W[f] * conj(A[f]) = C * N * δ[f]

At f = 0: W[0] * conj(A[0]) = C * N
At f != 0: W[f] * conj(A[f]) = 0

So for f != 0: either W[f] = 0 or A[f] = 0.

If W[f] != 0 for all f (which is the case if w is "spread out"), then A[f] = 0 for all f != 0, meaning a is constant (uniform). And indeed, uniform is a solution when w is uniform!

But when w is concentrated (near the peak of c), W[f] is not zero for most f, and we need A[f] = 0 for those f, which constrains a significantly.

This is getting complex. Let me just go with the gradient-based approach but with the FFT-accelerated subgradient.

Let me now write the final code:

```python
"""
Minimize 2*n*max_conv/sum^2 using Adam with FFT-accelerated softmax subgradient.

Key improvements over previous approach:
1. FFT-based subgradient computation (O(n log n) instead of O(n * top_k))
2. Softmax weighting over all lags (smooth, informative gradient)
3. Multiple initializations and lengths
4. Adaptive temperature schedule
5. Periodic noise injection to escape local minima
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient_fft(a, weights, fsize):
    """Compute grad[m] = 2 * sum_i a[i] * w[m+i] using FFT cross-correlation."""
    # Cross-correlation: x[m] = sum_i a[i] * w[i+m]
    # DFT(x) = conj(DFT(a)) * DFT(w) (derived earlier)
    A = fft(a, fsize)
    W = fft(weights, fsize)
    x = np.real(ifft(np.conj(A) * W))
    return 2.0 * x[:len(a)]


def softmax_weights(c, tau):
    """Compute softmax weights over convolution values."""
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / tau)
    w = exp_c / exp_c.sum()
    return w


def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    return a * (S / s)


def run_adam_subgradient(a_init, n, S, max_time, rng, 
                         lr_base=None, tau_frac=0.05, noise_interval=300):
    """Adam with FFT softmax subgradient."""
    a = safe_normalize(a_init, n)
    if lr_base is None:
        lr_base = 0.5 * S / n
    
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    # Precompute FFT size for cross-correlation
    fsize = 1 << (2 * n).bit_length()  # >= 2n for cross-corr
    # Actually need >= n + (2n-1) - 1 = 3n-2 for the cross-corr
    fsize = 1 << (3 * n - 2 - 1).bit_length()
    # That might be too large. Let me use a different approach.
    # For the cross-correlation x[m] = sum_i a[i]*w[i+m], m in [0,n-1], i in [0,n-1]
    # Max index accessed: i+m <= 2n-2, so we need arrays of length >= 2n-1
    # But to avoid circular wrap, we need M >= (n-1) + (2n-2) = 3n-3
    fsize = 1 << (3 * n - 1).bit_length()
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
            
        # Softmax weights
        tau = tau_frac * best_score
        tau = max(tau, 1e-15)
        w = softmax_weights(c, tau)
        
        # FFT subgradient
        grad = compute_subgradient_fft(a, w, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-15:
            break
        grad = grad / gnorm
        
        # Learning rate: warmup then decay
        if t <= 100:
            lr = lr_base * (t / 100.0)
        else:
            lr = lr_base * (100.0 / t) ** 0.7
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        
        # Periodic noise
        if t % noise_interval == 0 and rng.random() < 0.2:
            scale = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, scale, n), S)
    
    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, eta_init=None, tau_frac=0.05):
    """Mirror descent (multiplicative weights) with FFT softmax subgradient."""
    a = safe_normalize(a_init, n)
    if eta_init is None:
        eta_init = 0.3
    
    fsize = 1 << (3 * n - 1).bit_length()
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        
        tau = tau_frac * best_score
        tau = max(tau, 1e-15)
        w = softmax_weights(c, tau)
        grad = compute_subgradient_fft(a, w, fsize)
        
        # Mirror descent: a_new = a * exp(-eta * grad / max|grad|)
        gmax = np.abs(grad).max()
        if gmax <= 1e-15:
            break
        
        eta = eta_init / (1.0 + 0.01 * t)
        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 1e-15:
            a = np.full(n, S / n)
        else:
            a = a_new * (S / s)
    
    return best_local, best_score


def collect_initializations(rng, base_seq=None):
    """Generate diverse initializations."""
    inits = []
    
    # From given sequence
    if base_seq is not None:
        try:
            given = np.asarray(base_seq, dtype=np.float64).reshape(-1)
            given = np.clip(given, 0, 1000)
            if given.size >= 256 and given.sum() > 0:
                inits.append(("given", given))
                # Noisy version
                noise = rng.normal(0, 0.1 * (given.sum() / len(given)), len(given))
                inits.append(("given_noisy", np.maximum(given + noise, 0)))
                # Another noise level
                noise2 = rng.normal(0, 0.3 * (given.sum() / len(given)), len(given))
                inits.append(("given_noisy2", np.maximum(given + noise2, 0)))
        except Exception:
            pass
    
    # Random at various lengths
    for n_try in (4096, 8192, 16384):
        inits.append(("rand_%d" % n_try, rng.random(n_try) + 0.1))
    
    # Smooth/sinusoidal
    for n_try in (8192, 16384):
        tt = np.linspace(0, 1, n_try)
        arr = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt)
        inits.append(("smooth_%d" % n_try, np.maximum(arr, 0.01)))
    
    return inits
```
