def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def gradient_step_fft(a, weights, n):
    """Compute grad = 2 * convolve(a, weights[::-1], 'full')[n-1:2n-1][::-1] using FFT"""
    # We need conv(a, weights[::-1], 'full') which has length n + (2n-1) - 1 = 3n-2
    rev_w = weights[::-1]
    result = fft_convolve(a, rev_w)
    # Extract [n-1 : 2n-1] and reverse
    return 2.0 * result[n-1 : 2*n-1][::-1]

"""
Combined FISTA + LP + Local Search to minimize peak auto-convolution.
Uses log-sum-exp smoothing for FISTA, LP for directional improvement,
and random perturbations for local search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_sum1(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def fista_smooth(a, time_limit=60.0, tau=0.003, verbose=True):
    """FISTA on smoothed max of convolution."""
    start_time = time.monotonic()
    n = len(a)
    a = normalize_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_conv_max = np.convolve(a, a).max()
    
    y = a.copy()
    t_k = 1.0
    base_lr = 1.0 / (2.0 * np.max(a) + 1e-8)
    lr = min(base_lr, 0.5)
    
    for it in range(50000):
        if time.monotonic() - start_time > time_limit:
            break
        
        conv_y = np.convolve(y, y)
        max_c = conv_y.max()
        exp_vals = np.exp((conv_y - max_c) / tau)
        w = exp_vals / exp_vals.sum()
        
        # Gradient: 2 * correlation(y, w)
        grad = 2.0 * np.convolve(y, w[::-1], 'full')[n-1:2*n-1][::-1]
        
        # Gradient step
        z = y - lr * grad
        
        # Project onto {x >= 0, sum(x) = 1}
        a_new = np.maximum(0, z)
        s = a_new.sum()
        if s < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = np.maximum(0.0, y)
        ys = y.sum()
        if ys < 1e-10:
            y = a_new.copy()
        else:
            y = y / ys
        t_k = t_new
        a = a_new
        
        # Adaptive lr
        if it > 0 and it % 100 == 0:
            lr = lr * 0.95
        
        current_max = np.convolve(a, a).max()
        if current_max < best_conv_max - 1e-12:
            best_conv_max = current_max
            best_a = a.copy()
            if verbose and (it % 200 == 0):
                print(f"    FISTA iter {it}: score={2*n*best_conv_max:.6f}")
    
    # Final check
    final_max = np.convolve(best_a, best_a).max()
    if final_max < best_conv_max:
        best_conv_max = final_max
    return best_a, 2.0 * n * best_conv_max


def lse_optimize(a, time_limit=60.0, tau=0.002, verbose=True):
    """Iteratively minimize LSE of convolution (smoother than FISTA)."""
    start_time = time.monotonic()
    n = len(a)

    lr = 0.1
    for it in range(50000):

        w = exp_vals / exp_vals.sum()

            a = a_new

def build_conv_matrix_sparse(a):
    n = len(a)
    n_rows = 2 * n - 1
    # For row i (0-indexed), the constraint is sum_j a[j] * g[i-j] <= M
    # This is: sum_j (a[j] if 0<=i-j<n else 0) * g[j] <= M
    # So the matrix entry (i,j) = a[j] if 0 <= i-j < n, else 0
    # i-j >= 0 means j <= i, and i-j < n means j > i-n
    # So for row i, valid j range is max(0, i-n+1) to min(i, n-1)
    
    rows = []
    cols = []
    data = []
    for i in range(n_rows):
        j_start = max(0, i - n + 1)
        j_end = min(i, n - 1) + 1
        if j_start < j_end:
            js = np.arange(j_start, j_end, dtype=np.int32)
            rows.extend(np.full(len(js), i, dtype=np.int32))
            cols.extend(js)
            data.extend(a[js].tolist())
    
    return csr_matrix((data, (rows, cols)), shape=(n_rows, n))


def lp_refine(a, time_limit=300.0, verbose=True):
    """LP-based iterative refinement."""
    start_time = time.monotonic()
    n = len(a)
    a = normalize_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = score_sequence(a)
    
    conv = np.convolve(a, a)
    M = conv.max()
    
    stagnant = 0
    for it in range(1000):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit:
            break
        
        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        A = build_conv_matrix_sparse(a)
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n
        
        remaining = time_limit - elapsed
        if remaining < 3.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.2))
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, 
                         method='highs', options={'presolve': True, 'time_limit': timeout})
            if not res.success or res.x is None:
                break
            g = np.maximum(0.0, res.x)
        except Exception:
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-8:
            stagnant += 1
            if stagnant > 5:
                break
            # Try with slightly relaxed M
            M *= 1.001
            continue
        
        g = g / sum_g
        alpha = 0.3
        a_new = (1 - alpha) * a + alpha * g
        a_new = normalize_sum1(a_new)
        
        new_score = score_sequence(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            conv = np.convolve(a, a)
            M = conv.max()
            if verbose and it % 10 == 0:
                print(f"    LP iter {it}: score={best_score:.6f}")
        else:
            # Try smaller step
            alpha = 0.1
            a_new = (1 - alpha) * a + alpha * g
            a_new = normalize_sum1(a_new)
            new_score = score_sequence(a_new)
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                stagnant = 0
            else:
                stagnant += 1
        
        if stagnant > 20:
            break
    
    return best_a, best_score


def local_search(a, time_limit=60.0, rng=None, verbose=True):
    """Random perturbation local search."""
    if rng is None:
        rng = np.random.default_rng(42)
    start_time = time.monotonic()
    n = len(a)
    a = normalize_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = score_sequence(a)
    
    for it in range(100000):
        if time.monotonic() - start_time > time_limit:
            break
        
        # Random perturbation
        candidate = a.copy()
        # Perturb a random window
        width = max(2, n // 50)
        start = rng.integers(0, n - width)
        candidate[start:start+width] *= np.exp(rng.normal(0, 0.05, width))
        candidate = normalize_sum1(candidate)
        
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate
            if verbose and it % 100 == 0:
                print(f"    LS iter {it}: score={best_score:.6f}")
    
    return best_a, best_score


def init_raised_cosine(n):

    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def init_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n / 2.0
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def init_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2*np.pi*x) + 0.08 * np.cos(4*np.pi*x))


def init_flat_top(n, width_frac=0.5):
    t = np.arange(n, dtype=np.float64)
    center = (n-1) / 2.0
    half_width = width_frac * n / 2.0
    return np.exp(-((t - center) / half_width) ** 4)
