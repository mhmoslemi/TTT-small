Optimizes non-negative step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated gradient descent, sparse LP refinement with column generation,
and simulated annealing restarts.
"""

import numpy as np

def _compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def _normalize(a, target=None):
    """Normalize to target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-15:
        return a
    return a * (target / S)

def _fft_convolve(a):
    """Compute full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for efficiency
    fsize = 1
    while fsize < size:
        fsize *= 2
    fa = np.fft.rfft(a, fsize)
    fc = np.fft.irfft(fa * fa, fsize)
    return fc[:size]

def _gradient_step(a, lr, mode='softmax', tau=30.0):
    """One gradient step to minimize peak autocorrelation (with fixed sum)."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)

    if mode == 'softmax':

        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[c.argmax()] = 1.0
        else:
            w /= w_sum
    else:
        idx = int(c.argmax())
        w = np.zeros(len(c))
        w[idx] = 1.0

def _lp_column_generation(f, M, max_iters=50, initial_constraints=None):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0 using column generation (constraint generation)."""
    n = len(f)
    total_constraints = 2 * n - 1
    
    if initial_constraints is None:
        # Start with the peak constraint and a few neighbors
        peak_idx = int(np.convolve(f, f).argmax())
        idxs = set()
        idxs.add(peak_idx)
        for offset in range(-5, 6):
            idx = peak_idx + offset
            if 0 <= idx < total_constraints:
                idxs.add(idx)
        # Also add a random subset
        idxs.update(set(np.random.choice(total_constraints, size=min(50, total_constraints), replace=False)))
        active = sorted(idxs)
    else:
        active = list(initial_constraints)
    
    for it in range(max_iters):
        # Build constraint matrix for active constraints
        m_active = len(active)
        A_rows = []
        for m in active:
            row = np.zeros(n)
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            for j in range(lo, hi + 1):
                row[j] = f[m - j]
            A_rows.append(row)
        A = np.array(A_rows)
        
        c_obj = -np.ones(n)
        b = np.full(m_active, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
            if res.status != 0:
                return None
            g = res.x
        except Exception:
            return None
        
        # Check all constraints
       .conv_f_g = np.convolve(f, g)
        violations = .conv_f_g - M
        if violations.max() <= 1e-6:
            return g  # Feasible!

        active_set = set(active)
        for idx in worst:
            active_set.add(int(idx))

"""
Optimizes non-negative step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated gradient descent, sparse LP refinement with constraint generation,
block-coordinate LP alternation, and simulated annealing restarts.
Key insight: taper edges to suppress corner peaks, flatten middle autocorrelation via LP.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize to target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-15:
        return a
    return a * (target / S)


def _gradient_step(a, lr, tau=30.0):
    """One gradient step minimizing peak autocorrelation with fixed sum constraint."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    a = _normalize(a, S_target)

    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        idx = int(c.argmax())
        w = np.zeros(len(c))
        w[idx] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for j in range(n):
        k_max = min(n, 2 * n - 1 - j)
        if k_max > 0:
            grad[j] = 2.0 * np.dot(a[:k_max], w[j:j + k_max])

    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = _normalize(a, S_target)
    return a


def _gradient_step_hard(a, lr):
    """Gradient step using only the hard max (peak) location."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    a = _normalize(a, S_target)

    c = np.convolve(a, a)
    m_star = int(c.argmax())

    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m_star - j]

    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = _normalize(a, S_target)
    return a


def _build_constraint_matrix(f, constraint_indices=None):
    """Build constraint matrix A where A[m,j] = f[m-j] if valid."""
    n = len(f)
    total = 2 * n - 1
    if constraint_indices is None:
        constraint_indices = range(total)
    m_count = len(constraint_indices) if hasattr(constraint_indices, '__len__') else total
    A = np.zeros((m_count, n))
    for idx, m in enumerate(constraint_indices):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[idx, j] = f[m - j]
    return A


def _lp_solve(f, M, constraint_indices=None, time_limit=45):
    """Solve LP: max sum(g) s.t. conv(f,g)[m] <= M for m in indices, g >= 0."""
    n = len(f)
    if constraint_indices is None:
        constraint_indices = range(2 * n - 1)
    if isinstance(constraint_indices, range):
        constraint_indices = list(constraint_indices)
    m_count = len(constraint_indices)
    if m_count == 0:
        return None

    A = _build_constraint_matrix(f, constraint_indices)
    c_obj = -np.ones(n)
    b = np.full(m_count, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _lp_constraint_generation(f, M, max_rounds=30, time_per_lp=20):
    """LP with constraint generation: start with subset, add violated constraints."""
    n = len(f)
    total = 2 * n - 1

    # Initial active set: peak and neighbors
    c_ff = np.convolve(f, f)
    peak = int(c_ff.argmax())
    active = set()
    for off in range(-10, 11):
        idx = peak + off
        if 0 <= idx < total:
            active.add(idx)
    # Add corners
    active.add(0)
    active.add(total - 1)
    # Add random
    active.update(set(np.random.choice(total, size=min(30, total), replace=False)))
    active = sorted(active)

    g = None
    for rnd in range(max_rounds):
        g = _lp_solve(f, M, constraint_indices=active, time_limit=time_per_lp)
        if g is None:
            return None

        # Check all constraints
        conv_fg = np.convolve(f, g)
        violations = conv_fg - M
        max_viol = float(violations.max())

        if max_viol <= 1e-5 * max(M, 1e-10):
            return g  # Feasible

        # Add most violated constraints
        n_add = min(30, total)
        worst = np.argsort(-violations)[:n_add]
        for idx in worst:
            active.add(int(idx))
        active = sorted(active)

        # Cap active set size

            keep = np.argsort(-violations)[:150]
            active_set = set(keep)
            active_set.add(0)
            active_set.add(total - 1)
            active_set.add(peak)
            active = sorted(active_set)

    return g

        a = np.ones(n) * 0.5
    a = _normalize(a)
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    S_t = np.sqrt(2.0 * n)

    for it in range(max_iters):
        if time.monotonic() > deadline - 20.0:
            break

        progress = float(it) / max_iters
        tau = 15.0 + 100.0 * progress
        lr = 0.2 * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, 0.005)

        if use_hard_max or tau > 80:

                best = a.copy()

    # Final check
    s = _compute_obj(a)
    if s < best_score:

        best = a.copy()

    return best, best_score

            # Try gradient escape
            for lr in [0.05, 0.02, 0.01]:
                a_cand = _gradient_step(a, lr, tau=50.0)
                s_cand = _compute_obj(a_cand)
                if s_cand < best_score:
                    a = a_cand
                    best = a.copy()
                    best_score = s_cand
                    break
            else:
                # Random perturbation
                a_pert = a + 0.15 * rng.standard_normal(n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = _normalize(a_pert, S_t)
                s_pert = _compute_obj(a_pert)
                if s_pert < best_score:
                    a = a_pert
                elif s_pert < best_score * 1.05:
                    a = a_pert  # Accept slightly worse for exploration
            continue

        g = _normalize(g, S_t)

        # Now solve LP with g as reference (block coordinate step)
        c_g = np.convolve(g, g)
        M_g = float(c_g.max())
        h = _lp_solve(g, M_g, time_limit=40)

        # Determine best update
        candidates = []
        candidates.append(('g', g, _compute_obj(g)))
        if h is not None:
            h_norm = _normalize(h, S_t)
            candidates.append(('h', h_norm, _compute_obj(h_norm)))

        # Blending candidates
        for t in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a_blend = (1.0 - t) * a + t * g
            a_blend = np.maximum(a_blend, 0.0)
            a_blend = _normalize(a_blend, S_t)
            candidates.append((f'blend_{t}', a_blend, _compute_obj(a_blend)))

        if h is not None:
            for t in [0.5, 0.3, 0.1]:
                a_blend2 = (1.0 - t) * a + t * h
                a_blend2 = np.maximum(a_blend2, 0.0)
                a_blend2 = _normalize(a_blend2, S_t)
                candidates.append((f'blend_h_{t}', a_blend2, _compute_obj(a_blend2)))

        # Pick best candidate
        best_cand = min(candidates, key=lambda x: x[2])
        if best_cand[2] < best_score:
            a = best_cand[1]
            best = a.copy()
            best_score = best_cand[2]
            if rnd % 5 == 0:
                print(f"    LP-BC round {rnd}: score={best_score:.6f} via {best_cand[0]}")
        else:
            # No improvement - try perturbation
            a_pert = a + 0.1 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = _normalize(a_pert, S_t)
            s_pert = _compute_obj(a_pert)
            if s_pert < best_score:
                a = a_pert
                best = a.copy()
                best_score = s_pert
            elif rnd > 10:
                # Gradient escape
                a_cand = _gradient_step(a, 0.03, tau=40.0)
                s_cand = _compute_obj(a_cand)
                if s_cand < best_score:
                    a = a_cand
                    best = a.copy()
                    best_score = s_cand

    return best, best_score


def _tapered_init(n, rng):
    """Create a tapered initial sequence (small at edges)."""
    i = np.arange(n, dtype=np.float64)
    # Raised cosine window
    window = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1)))
    # Add some randomness
    a = window * (0.5 + 0.5 * rng.random(n))
    # Ensure non-negative
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = window.copy()
    return a


def _multi_freq_init(n, rng):
    """Initialize with multi-frequency structure."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n) * 0.3
    n_freqs = rng.integers(2, 6)
    for _ in range(n_freqs):
        freq = rng.uniform(1, 15)
        phase = rng.uniform(0, 2 * np.pi)
        a += 0.2 * np.sin(2 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.0)
    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function. Combines gradient descent, block-coordinate LP,
    constraint generation, and annealed restarts to minimize the peak autocorrelation
    objective for non-negative sequences.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = budget_s
    hard_deadline = t_start + total_budget - 15.0

    best_seq = None
    best_score = float('inf')

    def _update(seq, score):
        nonlocal best_seq, best_score
        if score < best_score:
            old = best_score
            best_score = score
            best_seq = seq.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: {best_score:.6f} (was {old:.6f}), n={len(seq)}, t={elapsed:.0f}s")

    print(f"=== Starting optimization, budget={total_budget}s ===")

    # --- Generate diverse starting points ---
    starts = []

    # Use provided sequence if available
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)) and gs.sum() > 0:
            starts.append(('provided', gs.copy()))
            for L in [256, 512, 1024]:
                if L <= len(gs):
                    starts.append((f'provided_{L}', gs[:L].copy()))
    except NameError:
        pass

    # Tapered inits (key insight: small edges reduce corner peaks)
    for n in [256, 512, 1024]:
        starts.append((f'taper_{n}', _tapered_init(n, rng)))

    # Multi-frequency inits
    for n in [256, 512]:
        starts.append((f'mf_{n}', _multi_freq_init(n, rng)))

    # Random init
    for n in [512]:
        starts.append((f'rand_{n}', 0.3 + 0.4 * rng.random(n)))

    # Sort: provided first, then by length
    def _start_key(item):
        name, seq = item
        if 'provided' in name:
            return (0, len(seq))
        return (1, len(seq))

    starts.sort(key=_start_key)
    print(f"Generated {len(starts)} starting sequences")

    # --- Phase 1: Quick gradient optimization (first 4 min) ---
    phase1_end = t_start + 240
    print(f"\n--- Phase 1: Gradient optimization (deadline: {phase1_end - t_start:.0f}s) ---")

    for name, a0 in starts:
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        seq, score = _optimize_gradient(a0, phase1_end, max_iters=3000)
        _update(seq, score)
        elapsed = time.monotonic() - t0
        print(f"  [{name}] n={len(a0)}, score={score:.6f}, t={elapsed:.1f}s")

    # --- Phase 2: LP-based refinement on top candidates (next ~28 min) ---
    phase2_end = t_start + 1920
    print(f"\n--- Phase 2: LP block-coordinate (deadline: {phase2_end - t_start:.0f}s) ---")

    if best_seq is not None:
        # Refine the best with LP
        t0 = time.monotonic()
        seq, score = _optimize_lp_block(best_seq, phase2_end, max_rounds=100)
        _update(seq, score)
        print(f"  LP on best: score={score:.6f}, t={time.monotonic()-t0:.1f}s")

        # Try other promising starts with same length
        best_n = len(best_seq)
        other_starts = [(nm, s) for nm, s in starts if len(s) == best_n and nm != 'provided']
        for name, a0 in other_starts[:2]:
            if time.monotonic() > phase2_end - 120:
                break
            t0 = time.monotonic()
            seq2, score2 = _optimize_lp_block(a0, phase2_end - 60, max_rounds=50)
            _update(seq2, score2)
            print(f"  LP on [{name}]: score={score2:.6f}, t={time.monotonic()-t0:.1f}s")

        # Try a different length
        for alt_n in [1024, 256]:
            if alt_n == best_n:
                continue
            if time.monotonic() > phase2_end - 180:
                break
            a_alt = _tapered_init(alt_n, rng)
            seq3, score3 = _optimize_gradient(a_alt, phase2_end - 120, max_iters=1500)
            if score3 < best_score * 1.05:  # Only pursue if competitive
                t0 = time.monotonic()
                seq3, score3 = _optimize_lp_block(a_alt, phase2_end - 30, max_rounds=30)
                _update(seq3, score3)
                print(f"  Alt n={alt_n}: score={score3:.6f}")

    # --- Phase 3: Final polish (remaining time) ---
    phase3_end = hard_deadline
    print(f"\n--- Phase 3: Final polish (deadline: {phase3_end - t_start:.0f}s) ---")

    if best_seq is not None:
        # Gradient polish
        t0 = time.monotonic()
        seq, score = _optimize_gradient(best_seq, phase3_end - 60, max_iters=5000)
        _update(seq, score)
        print(f"  Gradient polish: score={score:.6f}, t={time.monotonic()-t0:.1f}s")

        # One more LP pass
        if time.monotonic() < phase3_end - 90:
            t0 = time.monotonic()
            seq, score = _optimize_lp_block(best_seq, phase3_end, max_rounds=30)
            _update(seq, score)
            print(f"  Final LP: score={score:.6f}, t={time.monotonic()-t0:.1f}s")

    # --- Output result ---
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"\n=== FINAL: n={len(best_seq)}, computed_score={final_score:.6f} ===")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("FALLBACK: returning tapered sequence of length 512")
    i = np.arange(512, dtype=np.float64)
    fallback = 0.5 * (1 + np.cos(2 * np.pi * i / 511))
    return [float(x) for x in fallback]
