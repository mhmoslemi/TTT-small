```python
import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def convolve_with_fixed(a, b):
    """Compute a * b (linear convolution) where a is fixed."""
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]

def solve_lp(a, M):
    """
    Solve: max sum(g_j) s.t. g_j >= 0, (a*g)_k <= M for all k.
    """
    n = len(a)
    # (a*g)_k = sum_j a[k-j] * g_j (with valid indices)
    # This is a linear function of g.
    # The coefficient of g_j in (a*g)_k is a[k-j] if 0 <= k-j < n, else 0.
    
    # A_matrix[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # A_matrix has shape (2n-1, n)
    
    A_mat = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_mat[k, j] = a[idx]
    
    # max sum(g) s.t. A_mat @ g <= M, g >= 0
    # linprog minimizes, so we negate the objective
    c = -np.ones(n)
    res = linprog(c, A_ub=A_mat, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        g = res.x
        if g.sum() > 1e-10:
            g /= g.sum()
        return g
    return None

def lp_based_optimize(a0, time_limit, t_param=0.1):
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a_sum
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t = t_param
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        g = solve_lp(a, M)
        if g is None:
            break
        
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        
        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()
        obj_new = 2.0 * n * M_new
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Decrease step size occasionally
        if np.random.random() < 0.1:
            t = max(0.01, t * 0.8)
    
    return best_a, best_obj
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
combined with gradient polish. The LP finds a direction g that, when blended
with the current sequence, is guaranteed to reduce the peak autocorrelation.
Multiple sequence lengths are explored with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def build_conv_matrix(a):
    """Build matrix A where A[k,j] = a[k-j] if valid, else 0.
    (a * g)[k] = (A @ g)[k]"""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A

def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. g >= 0, A@g <= M. Returns normalized g."""
    n = len(a)
    A_mat = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        g = res.x
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None

def lp_optimize(a0, time_limit, alpha=0.1, seed=42):
    """LP-based optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    alpha_curr = alpha
    no_improve = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        g = lp_direction(a, M)
        if g is None:
            break
        
        # Blend
        a_new = (1 - alpha_curr) * a + alpha_curr * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive step size
        if no_improve > 50:
            alpha_curr = max(0.01, alpha_curr * 0.8)
            no_improve = 0
            # Occasionally restart from best with perturbation
            if alpha_curr < 0.02:
                a = best_a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                alpha_curr = 0.1
                no_improve = 0
    
    return best_a, best_obj

def softmax_gradient_optimize(a0, time_limit, t_param=50.0, eta0=0.5, seed=42):
    """Gradient-based optimization with softmax peak weighting."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    eta = eta0
    t = t_param
    no_improve = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Softmax weights
        scaled = t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-12:
            w /= w_sum
        else:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        
        # Compute gradient: grad[j] = sum_k w[k] * (M_k a)[j]
        # (M_k a)[j] = a[k-j] if valid
        grad = np.zeros(n)
        thresh = w.max() * 0.001
        active = np.where(w > thresh)[0]
        
        for k_int in active:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                # grad[j] += w[k] * a[k-j]
                # For j in [j_start, j_end]: k-j ranges from k-j_start to k-j_end
                # = a[k-j_end : k-j_start+1][::-1]
                grad[j_start:j_end+1] += w[k] * a[k-j_end:k-j_start+1][::-1]
        
        # Gradient step with projection onto simplex
        # Use multiplicative update (mirror descent) for non-negativity
        sg_centered = grad - grad.mean()
        log_factor = -eta * sg_centered
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of LP-based
direction finding (guaranteed descent) and multiplicative gradient polish.
The LP finds a direction g maximizing sum(g) subject to a*g <= M, giving
a theoretically guaranteed reduction in peak autocorrelation. Multiple
sequence lengths and diverse initializations are explored.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_conv_matrix(a):
    """Build matrix where A[k,j] = a[k-j] if 0<=k-j<n else 0.
    Then (a * g)[k] = (A @ g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    if np.any(valid):
        A[valid] = a[diff[valid]]
    return A


def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. g >= 0, A@g <= M. Returns normalized g or None."""
    n = len(a)
    a_safe = a + 1e-12
    A_mat = build_conv_matrix(a_safe)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if res.success:
            g = res.x
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_optimize(a0, time_limit, alpha=0.15, seed=42):
    """LP-based optimization with adaptive step size."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:

        a /= s

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    alpha_curr = alpha
    no_improve = 0
    t0 = time.monotonic()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        g = lp_direction(a, M)
        if g is None:
            break

        a_new = (1.0 - alpha_curr) * a + alpha_curr * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        iter_count += 1

            top_configs.append((n, name, a_opt))
            seen_n.add(n)
        if len(top_configs) >= 12:
            break

    # Always include current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    # Also add pure LP starts from diverse inits for n=128
    for n in [128, 256]:
        inits = get_initializations(n, rng, parent)
        for name in ['uniform', 'bimodal', 'u_shape']:
            if name in inits and (n, name, inits[name]) not in top_configs:
                top_configs.append((n, name, inits[name]))

    t_per_deep = max(15.0, (t2_budget) / max(1, len(top_configs)))

    for n, name, a0 in top_configs:
        if time.monotonic() > t2_start + t2_budget - 20:
            break
        t_alloc = min(t_per_deep, t2_start + t2_budget - time.monotonic() - 20)
        if t_alloc < 5:
            break

        for alpha in [0.1, 0.2]:
            if time.monotonic() > t2_start + t2_budget - 20:
                break
            t_single = t_alloc / 2.0
            a_opt, _ = lp_optimize(a0, time_limit=t_single, alpha=alpha,
                                   seed=seed + n * 7 + int(alpha * 100))
            score = try_score(a_opt)
            if score < best_score:
                print(f"  LP: n={n}, {name}, alpha={alpha}: {score:.6f}")

    print(f"  Phase 2 done. Best: {best_score:.6f}")

    # Phase 3: Alternate LP and gradient on best
    print("Phase 3: Alternating LP + Gradient polish")
    remaining = deadline - time.monotonic()
    if best_sequence is not None and remaining > 30:
        n_best = len(best_sequence)
        a_cur = best_sequence.copy()

        cycle_time = min(60.0, remaining / 8.0)
        n_cycles = max(1, int(remaining / cycle_time) - 1)

        for cyc in range(n_cycles):
            if time.monotonic() > deadline - 10:
                break

            # LP step
            t_lp = cycle_time * 0.6
            a_opt, _ = lp_optimize(a_cur, time_limit=t_lp, alpha=0.08,
                                   seed=seed + 500 + cyc)
            score = try_score(a_opt)
            a_cur = a_opt if score < best_score else a_cur

            # Gradient step
            t_grad = cycle_time * 0.4
            if time.monotonic() > deadline - 10:
                break
            a_opt2, _ = gradient_optimize(a_cur, time_limit=t_grad, t_param=80.0,
                                          eta0=0.3, seed=seed + 600 + cyc)
            score2 = try_score(a_opt2)
            if score2 < best_score:
                a_cur = a_opt2
                print(f"  Cycle {cyc}: {score2:.6f}")

    print(f"  Phase 3 done. Best: {best_score:.6f}")

    # Phase 4: Final deep LP on the best
    if best_sequence is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 15
        print(f"Phase 4: Final deep LP ({remaining:.0f}s)")
        a_opt, _ = lp_optimize(best_sequence, time_limit=remaining, alpha=0.05,
                               seed=seed + 900)
        score = try_score(a_opt)
        print(f"  Final LP: {score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f}")
    return [float(x) for x in best_sequence]
```
