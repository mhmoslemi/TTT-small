- Or: let `a_i = (w_i + 1)^2 * target_sum / sum((w_j + 1)^2)` for w_i > -1

I think the softmax parameterization is cleanest. Let me use that.

With `a_i = softmax(w)_i * S` where `S = sqrt(2n)`:
- `sum(a) = S` always
- `a_i > 0` always (never exactly 0, but can be very small)

The gradient w.r.t. `w` is:
`da_i/dw_j = S * (delta_ij * softmax(w)_i - softmax(w)_i * softmax(w)_j)`
`= S * softmax(w)_i * (delta_ij - softmax(w)_j)`

So `d(obj)/dw_j = sum_i d(obj)/da_i * S * softmax(w)_i * (delta_ij - softmax(w)_j)`

This is manageable. But it might be numerically unstable for very peaked softmax distributions.

Alternatively, I can just work directly with `a` and project onto the constraint set after each step. This is simpler and works well in practice.

Let me go with the direct approach:
1. Work with `a` directly (non-negative, sum = S)
2. Compute the smoothed gradient
3. Take a step and project: `a_new = max(0, a - lr * grad)`, then normalize to sum = S
4. Use backtracking line search for the step size

For the line search, I want to ensure the objective decreases. With the smoothed objective (power sum), this should work well.

    """Normalize to target sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (target / S)

def _ps_gradient(a, p):
    """
    Compute gradient of the power-sum smoothed objective.
    Objective: 2n * PS_p(conv(a,a)) / (sum(a))^2
    where PS_p(c) = (sum c^p)^(1/p)
    """
    n = len(a)
    c = np.convolve(a, a)
    S = a.sum()
    
    if p < 1:
        # Hard max subgradient
        m_star = int(c.argmax())
        smax = c[m_star]
        # d(smax)/da[k] = 2*a[m_star-k] if 0 <= m_star-k < n
        grad_smax = np.zeros(n)
        for k in range(n):
            idx = m_star - k
            if 0 <= idx < n:
                grad_smax[k] = 2.0 * a[idx]
    else:
        c_safe = np.maximum(c, 1e-30)
        c_p = c_safe ** p
        sum_cp = c_p.sum()
        smax = sum_cp ** (1.0 / p)
        # grad_c[m] = c_m^(p-1) * (sum_cp)^(1/p - 1)
        grad_c = (c_p / c_safe) * (sum_cp ** (1.0 / p - 1.0))
        
        # d(smax)/da[k] = 2 * sum_j a[j] * grad_c[j+k]
        # Use FFT for efficiency

    grad = (2.0 * n / (S * S)) * grad_smax - (2.0 * n * smax / (S * S * S))
    
    return grad, smax

    S = a.sum()
    if S < 1e-15:

    else:
        a = a * (target / S)
    return a

    """

"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses adaptive power-sum smoothing with projected gradient and backtracking line search,
complemented by L-BFGS-B on softmax reparameterization and simulated annealing.
Multi-start across different initializations and sequence lengths.

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return float('inf')
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize to target sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (target / S)


def _project(a, target):
    """Project onto {a >= 0, sum(a) = target}."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) * target / len(a)
    else:
        a = a * (target / S)
    return a


def _ps_grad_smax(a, p):
    """Compute d(PS_p(conv(a,a)))/da where PS_p(c)=(sum c^p)^(1/p)."""
    n = len(a)
    c = np.convolve(a, a)

    if p < 1.0:
        m_star = int(c.argmax())
        grad = np.zeros(n)
        for k in range(n):
            idx = m_star - k
            if 0 <= idx < n:
                grad[k] = 2.0 * a[idx]
        return grad, c[m_star]

    c_safe = np.maximum(c, 1e-30)
    c_p = c_safe ** p
    sum_cp = c_p.sum()
    if sum_cp < 1e-300:
        return np.zeros(n), 0.0
    smax = sum_cp ** (1.0 / p)
    # grad_c[m] = c_m^(p-1) * (sum_cp)^(1/p - 1)
    grad_c = (c_p / c_safe) * (sum_cp ** (1.0 / p - 1.0))

    # d(smax)/da[k] = 2 * sum_j a[j] * grad_c[j+k] for j+k < 2n-1
    # Use FFT cross-correlation
    N = 1
    while N < 4 * n:
        N *= 2
    A = np.fft.rfft(a, N)
    G = np.fft.rfft(grad_c, N)
    cross = np.fft.irfft(A * np.conj(G), N)
    grad = 2.0 * cross[:n]
    return grad, smax


def _full_gradient(a, p):
    """Full gradient of 2n*PS_p(conv)/(sum)^2."""
    n = len(a)
    S = a.sum()
    grad_smax, smax = _ps_grad_smax(a, p)

Uses adaptive power-sum smoothing with projected gradient and backtracking line search,

def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return float('inf')
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize to target sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (target / S)


def _project(a, target):
    """Project onto {a >= 0, sum(a) = target}."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) * target / len(a)
    else:
        a = a * (target / S)
    return a


def _ps_grad_smax(a, p):
    """Compute d(PS_p(conv(a,a)))/da and the PS value."""
    n = len(a)

        grad = np.zeros(n)
