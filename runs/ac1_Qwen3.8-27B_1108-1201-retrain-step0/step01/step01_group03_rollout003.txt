def mirror_descent_optimize(a, deadline, max_iters=5000, top_k_range=[1,3,5,10],
                             tau_range=[2.0, 5.0, 10.0, 20.0]):
    """Optimize using mirror descent with adaptive parameters."""
    n = len(a)
    a = np.maximum(a, 0)
    a = normalize_to_sum1(a)
    best = a.copy()
    best_score = compute_obj(best)
    
    # Adaptive parameters
    eta = 0.05  # Initial step size
    eta_min = 0.001
    eta_decay = 0.9995
    
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        # Vary top_k and tau periodically
        top_k = top_k_range[it % len(top_k_range)]
        tau = tau_range[(it // len(top_k_range)) % len(tau_range)]
        
        # Compute weighted gradient
        grad, c = weighted_topk_gradient(a, top_k=top_k, tau=tau)

        # a_new[m] ∝ a[m] * exp(-eta * grad[m])
        log_a = np.log(a + 1e-30) - eta * grad
        log_a = np.clip(log_a, -50, 50)
        a = np.exp(log_a - log_a.max())
        a = a / a.sum()
        
        # Decay step size
        eta *= eta_decay
        eta = max(eta, eta_min)
        
        # Periodically check score
        if (it + 1) % 50 == 0:
            score = compute_obj(a)
            if score < best_score:
                best_score = score
                best = a.copy()
        
        # Random restart (small probability)
        if (it + 1) % 500 == 0 and eta > eta_min * 2:
            # Add small random perturbation
            perturbation = rng.random(n) * 0.01
            a = a + perturbation
            a = np.maximum(a, 0)
            a = a / a.sum()
            eta = max(eta, 0.02)  # Reset step size
    
    return best, best_score

def lp_refine(a, deadline, max_iters=50):
    """LP-based refinement."""
    n = len(a)
    if n > 1024:
        return a, compute_obj(a)
    
    a = normalize_to_sum1(a)
    best = a.copy()
    best_score = compute_obj(best)
    
    c = np.convolve(a, a)
    M = float(c.max())

        if time.monotonic() > deadline - 5.0:
            break
        
        # Solve LP: max sum(g) s.t. convolve(a, g) <= M, g >= 0
        c_obj = -np.ones(n)
        A = np.zeros((2 * n - 1, n))
        for m in range(2 * n - 1):
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            for j in range(lo, hi + 1):
                A[m, j] = a[m - j]
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break
        
        if g is None or g.sum() < 1e-10:
            break
        
        # Try blending
        improved = False
        for t in [0.1, 0.3, 0.5, 0.7, 0.9]:
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            s_new = compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break
        
        if not improved:
            # Small perturbation to escape
            break
        
        # Update M
        c = np.convolve(a, a)
        M = float(c.max())
    
    return best, best_score

def adam_optimize(a, deadline, max_iters=5000, top_k=5, tau=5.0):
    """Adam-based optimization."""
    n = len(a)
    a = np.maximum(a, 0)
    a = normalize_to_sum1(a)
    best = a.copy()
    best_score = compute_obj(best)
    
    # Adam parameters
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.01
    eps = 1e-8
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        # Anneal tau
        current_tau = tau + 10.0 * (it / max_iters)
        current_top_k = max(1, top_k - it // (max_iters // 3))
        
        grad, c = weighted_topk_gradient(a, top_k=current_top_k, tau=current_tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        # Project out mean (for sum constraint)
        m_hat = m_hat - m_hat.mean()
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        # Decay LR
        lr *= 0.999
        
        if (it + 1) % 50 == 0:
            score = compute_obj(a)
            if score < best_score:
                best_score = score
                best = a.copy()
    
    return best, best_score
