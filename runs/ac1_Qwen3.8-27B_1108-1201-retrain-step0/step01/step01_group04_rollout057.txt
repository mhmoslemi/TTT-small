    h = h[:n]
    grad = 2.0 * h
    return grad, c, cmax


def _make_arcsine(n):
    """p[i] proportional to 1/sqrt((i+1)*(n-i))."""
    i = np.arange(0, n, dtype=np.float64)
    p = 1.0 / np.sqrt((i + 1.0) * (n - i))
    return _norm1(p)


def _make_beta(n, a, b):
    """Discrete approximation of Beta(a,b) on {0,...,n-1}."""
    x = (np.arange(n) + 0.5) / n
    p = x ** a * (1.0 - x) ** b
    return _norm1(np.maximum(p, 1e-30))


def _make_uniform(n):
    return np.full(n, 1.0 / n)


def _make_geometric(n, r):
    p = r ** np.arange(n, dtype=np.float64)
    return _norm1(p)


def _make_m_shape(n, eps):
    """M-shaped: high at edges, dip in middle. p[i] = 1 + eps*(2i/(n-1) - 1)^2 * 2"""
    x = np.linspace(0, 1, n)
    p = 1.0 + eps * (2 * x - 1) ** 2 * 4.0
    return _norm1(np.maximum(p, 0.01))


def _make_w_shape(n, a, b):
    """W-shaped: p[i] = 1 + a*cos(2*pi*i/(n-1)) + b*cos(4*pi*i/(n-1))"""
    i = np.arange(n, dtype=np.float64)
    p = 1.0 + a * np.cos(2 * np.pi * i / (n - 1)) + b * np.cos(4 * np.pi * i / (n - 1))
    return _norm1(np.maximum(p, 0.01))


def _spectral_init(n, n_iters=30):
    """
    Fienup-like: target |DFT(p)|^2 = DFT(rectangle of length 2n-1, normalized).
    """
    L = _next_fast_len(3 * n)
    target_c = np.zeros(L)
    target_c[:2 * n - 1] = 1.0 / (2 * n - 1)
    target_spec = np.fft.fft(target_c)
    target_mag = np.sqrt(np.abs(target_spec))
    # Initialize
    phase = target_spec / (np.abs(target_spec) + 1e-30)
    target_X = target_mag * np.sqrt(phase + 1e-30)
    p = np.real(np.fft.ifft(target_X, L))[:n]
    p = _norm1(p)
    for _ in range(n_iters):
        X = np.fft.fft(p, L)
        X_new = target_mag * (X / (np.abs(X) + 1e-30))
        p_new = np.real(np.fft.ifft(X_new, L))[:n]
        p = _norm1(p_new)
    return p


def _generate_candidates(n, rng):
    """Generate diverse initial candidates for a given length n."""
    candidates = []
    candidates.append(_make_uniform(n))
    candidates.append(_make_arcsine(n))
    candidates.append(_make_beta(n, 0.5, 0.5))
    candidates.append(_make_beta(n, 0.3, 0.3))
    candidates.append(_make_beta(n, 0.2, 0.8))
    candidates.append(_make_beta(n, 0.8, 0.2))
    candidates.append(_make_beta(n, 0.1, 1.5))
    candidates.append(_make_beta(n, 1.5, 0.1))
    candidates.append(_make_geometric(n, 0.3))
    candidates.append(_make_geometric(n, 0.5))
    candidates.append(_make_geometric(n, 0.7))
    candidates.append(_make_geometric(n, 0.15))
    candidates.append(_make_m_shape(n, 0.3))
    candidates.append(_make_m_shape(n, 0.6))
    candidates.append(_make_m_shape(n, 1.0))
    candidates.append(_make_w_shape(n, 0.4, 0.2))
    candidates.append(_make_w_shape(n, 0.6, -0.2))
    candidates.append(_make_w_shape(n, 0.3, 0.4))
    candidates.append(_make_w_shape(n, -0.3, 0.3))
    candidates.append(_spectral_init(n, 20))
    # Random smooth
    for _ in range(4):
        f1 = rng.integers(2, max(3, n // 64))
        f2 = rng.integers(1, max(2, n // 128))
        p = 1.0
        p += rng.uniform(0.2, 1.5) * np.sin(2 * np.pi * f1 * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        p += rng.uniform(0.1, 0.8) * np.sin(2 * np.pi * f2 * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        candidates.append(_norm1(np.maximum(p, 0.01)))
    return candidates


def _optimize_local(p0, time_limit, rng):
    """
    Projected gradient with ISE surrogate, adaptive T, momentum, restarts.
    """
    p = _norm1(p0.copy())
    n = p.size
    best_p = p.copy()
    best_score = _score(p)
    current_score = best_score

    end = time.monotonic() + time_limit
    T_factor = 0.4
    lr = 0.5 / n
    v = np.zeros(n)
    mom = 0.0
    since_improve = 0
    iter_no = 0
    best_T_factor = T_factor

    while time.monotonic() < end:
        iter_no += 1

        # Adaptive momentum
        if iter_no < 100:
            mom = 0.0
        elif iter_no < 500:
            mom = 0.3
        else:
            mom = 0.5

        # Compute gradient
        c = _self_conv(p)
        cmax = float(c.max())
        T = max(T_factor * cmax, 1e-10)
        diff = (c - cmax) / T
        diff = np.clip(diff, -60.0, 60.0)
        e = np.exp(diff)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(c.argmax())] = 1.0
        else:
            w = e / s

        # grad_j = 2 * sum_m w[m+j] * p[m]
        L = _next_fast_len(3 * n)
        W = np.fft.fft(w, L)
        P = np.fft.fft(p[::-1], L)
        H = np.real(np.fft.ifft(W * P))
        h = H[n - 1: 3 * n - 2]
        if h.size < n:
            h = np.concatenate([h, np.zeros(n - h.size)])
        h = h[:n]
        grad = 2.0 * h

        # Center gradient (tangent to simplex)
        grad_centered = grad - grad.mean()

        # Momentum
        v = mom * v + (1.0 - mom) * grad_centered

        # Line search with backtracking
        found = False
        for lr_try in (lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05, lr * 0.01):
            p_new = _project(p - lr_try * v)
            sc = _score(p_new)
            if np.isfinite(sc) and sc <= current_score + 1e-12:
                found = True
                break
        if not found:
            # Try direct gradient (no momentum)
            for lr_try in (lr * 0.5, lr * 0.1, lr * 0.05, lr * 0.01, lr * 0.005):
                p_new = _project(p - lr_try * grad_centered)
                sc = _score(p_new)
                if np.isfinite(sc) and sc <= current_score + 1e-12:
                    found = True
                    v = grad_centered
                    break

        if found:
            if sc < best_score - 1e-14:
                best_score = sc
                best_p = p_new.copy()
                since_improve = 0
                best_T_factor = T_factor
            else:
                since_improve += 1
            current_score = sc
            p = p_new
            # Gradually decrease T
            T_factor = max(0.03, T_factor * 0.997)
            # Gradually increase lr
            lr = min(lr * 1.01, 3.0 / n)
        else:
            since_improve += 1
            # Reduce T to sharpen gradient
            T_factor = min(1.0, T_factor * 2.0)
            lr = max(lr * 0.5, 0.01 / n)
            v = np.zeros(n)

            if since_improve > 500:
                # Reheat + random kick
                T_factor = 0.5
                lr = 0.5 / n
                v = np.zeros(n)
                kick_scale = np.sqrt(2.0 / n) * min(1.0, 50.0 / since_improve)
                p = _project(p + rng.normal(0.0, kick_scale, n))
                current_score = _score(p)
                since_improve = 0

            if since_improve > 2000:
                # Major restart: blend with best and kick
                p = _project(0.7 * best_p + 0.3 * p + rng.normal(0.0, 2.0 / np.sqrt(n), n))
                current_score = _score(p)
                T_factor = 0.4
                lr = 0.5 / n
                v = np.zeros(n)
                since_improve = 0

        if best_score < 1.5030 and since_improve > 8000:
            break

    return best_p, best_score


def _optimize_refine(p0, time_limit, rng):
    """
    Finer-grained refinement: smaller steps, lower T, focused on the peak.
    Uses coordinate-pair moves as well as gradient steps.
    """
    p = _norm1(p0.copy())
    n = p.size
    best_p = p.copy()
    best_score = _score(p)
    current_score = best_score

    end = time.monotonic() + time_limit
    T_factor = 0.05  # Start with sharp gradient
    lr = 0.1 / n
    v = np.zeros(n)
    mom = 0.3
    since_improve = 0
    iter_no = 0

    while time.monotonic() < end:
        iter_no += 1

        c = _self_conv(p)
        cmax = float(c.max())
        T = max(T_factor * cmax, 1e-10)

        # Softmax weights
        diff = (c - cmax) / T
        diff = np.clip(diff, -60.0, 60.0)
        e = np.exp(diff)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(c.argmax())] = 1.0
        else:
            w = e / s

        # Gradient
        L = _next_fast_len(3 * n)
        W = np.fft.fft(w, L)
        P = np.fft.fft(p[::-1], L)
        H = np.real(np.fft.ifft(W * P))
        h = H[n - 1: 3 * n - 2]
        if h.size < n:
            h = np.concatenate([h, np.zeros(n - h.size)])
        h = h[:n]
        grad = 2.0 * (h - h.mean())

        # Momentum
        v = mom * v + (1.0 - mom) * grad

        # Try gradient step
        found = False
        for lr_try in (lr, lr * 0.5, lr * 0.2, lr * 0.1, lr * 0.05):
            p_new = _project(p - lr_try * v)
            sc = _score(p_new)
            if sc < current_score:
                found = True
                break

        if not found:
            # Try pair moves: move mass from high-gradient to low-gradient positions
            gi = np.argmax(grad)
            lj = np.argmin(grad)
            if grad[gi] > grad[lj] + 1e-15:
                for frac in (0.1, 0.05, 0.02, 0.01, 0.005):
                    delta = frac * min(p[gi], p[lj] + 1e-10)
                    if delta < 1e-15:
                        continue
                    p_new = p.copy()
                    p_new[gi] -= delta
                    p_new[lj] += delta
                    p_new = _norm1(p_new)
                    sc = _score(p_new)
                    if sc < current_score:
                        found = True
                        break

        if not found:
            # Small random perturbation
            for sigma in (0.02 / np.sqrt(n), 0.01 / np.sqrt(n), 0.005 / np.sqrt(n)):
                p_new = _project(p + rng.normal(0, sigma, n))
                sc = _score(p_new)
                if sc < current_score:
                    found = True
                    break

        if found:
            if sc < best_score - 1e-14:
                best_score = sc
                best_p = p_new.copy()
                since_improve = 0
            else:
                since_improve += 1
            current_score = sc
            p = p_new
            T_factor = max(0.01, T_factor * 0.99)
        else:
            since_improve += 1
            T_factor = min(0.3, T_factor * 2.0)
            lr = max(lr * 0.7, 0.005 / n)

            if since_improve > 300:
                p = _project(0.8 * best_p + 0.2 * p + rng.normal(0, 0.5 / np.sqrt(n), n))
                current_score = _score(p)
                since_improve = 0
                T_factor = 0.1

            if since_improve > 3000:
                p = _project(best_p + rng.normal(0, 1.0 / np.sqrt(n), n))
                current_score = _score(p)
                since_improve = 0
                T_factor = 0.15

        if best_score < 1.5028 and since_improve > 5000:
            break

    return best_p, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a nonnegative step function minimizing the self-convolution peak score."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    # Try to load parent
    parent = None
    try:
        pa = np.asarray(height_sequence_1, dtype=np.float64)
        pa = np.where(np.isfinite(pa), pa, 0.0)
        pa = np.maximum(pa, 0.0)
        if pa.ndim == 1 and pa.size >= 64 and pa.sum() > 0.01:
            parent = _norm1(pa)
    except Exception:
        parent = None

    if parent is not None:
        parent_score = _score(parent)
        print(f"Parent: n={parent.size}, score={parent_score:.10f}", flush=True)

    # Determine sequence lengths to try
    lengths = [512, 1024, 2048, 4096, 8192]
    if parent is not None:
        pn = parent.size
        # Include parent length and nearby powers of 2
        if pn not in lengths:
            lengths.append(pn)
        lengths = sorted(set(lengths))
        # Filter: skip very large if time is tight
        if len(lengths) > 5:
            lengths = lengths[-5:]

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None or p.size == 0:
            return
        p = _norm1(p)
        if not np.isfinite(sc):
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print(f"  NEW BEST: n={p.size}, score={sc:.10f}", flush=True)

    # Phase 1: Generate and score candidates for each length
    candidates_by_n = {}
    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 300:
            break
        cands = _generate_candidates(n, rng)
        if parent is not None and parent.size == n:
            cands.insert(0, parent)
        # Score all
        scored = []
        for p in cands:
            sc = _score(p)
            if np.isfinite(sc):
                scored.append((sc, p))
        scored.sort(key=lambda x: x[0])
        candidates_by_n[n] = scored
        if scored:
            print(f"n={n}: best candidate score={scored[0][0]:.10f} (from {len(scored)} candidates)", flush=True)
            try_record(scored[0][1], scored[0][0])

    # Phase 2: Local optimization on top candidates
    # Build a job list: (score, n, p, label)
    jobs = []
    for n, scored in candidates_by_n.items():
        for i, (sc, p) in enumerate(scored[:3]):
            jobs.append((sc, n, p, f"n{n}_c{i}"))
    # Also add parent at its own length if available
    if parent is not None:
        jobs.append((_score(parent), parent.size, parent, "parent"))

    jobs.sort(key=lambda x: x[0])

    # Allocate time to optimization
    final_reserve = 200.0
    n_jobs = min(len(jobs), 8)
    time_per = max(60.0, (deadline - final_reserve - time.monotonic()) / max(1, n_jobs))

    opt_count = 0
    for sc_init, n, p0, label in jobs[:n_jobs]:
        remaining = deadline - final_reserve - time.monotonic()
        if remaining < 30:
            break
        t_alloc = min(time_per, remaining)
        print(f"Optimizing {label} (n={n}, init_score={sc_init:.10f}, time={t_alloc:.0f}s)", flush=True)

        if t_alloc > 200:
            # Two-stage: broad then refine
            t1 = t_alloc * 0.6
            t2 = t_alloc * 0.4
            p_opt, sc_opt = _optimize_local(p0, t1, rng)
            try_record(p_opt, sc_opt)
            if t2 > 30:
                p_opt2, sc_opt2 = _optimize_refine(p_opt, t2, rng)
                try_record(p_opt2, sc_opt2)
        else:
            p_opt, sc_opt = _optimize_local(p0, t_alloc, rng)
            try_record(p_opt, sc_opt)

        opt_count += 1

    # Phase 3: Final polish on the global best
    if best_p is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"Final polish (remaining={remaining:.0f}s, current best={best_score:.10f})", flush=True)
            p_final, sc_final = _optimize_refine(best_p, remaining * 0.8, rng)
            try_record(p_final, sc_final)
            # One more round
            remaining = deadline - time.monotonic()
            if remaining > 20:
                p_fin2, sc_fin2 = _optimize_refine(best_p, remaining * 0.9, rng)
                try_record(p_fin2, sc_fin2)

    if best_p is None:
        # Fallback
        best_p = _make_arcsine(1024)
        best_score = _score(best_p)

    print(f"Final: n={best_p.size}, score={best_score:.10f}", flush=True)

    # Ensure valid output
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = best_p.sum()
    if s < 0.01:
        best_p = _make_uniform(256)
    return [float(x) for x in best_p]
