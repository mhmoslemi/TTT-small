```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
def smoothed_objective_grad(a, tau=10.0):
    """Compute gradient of smoothed max of auto-convolution."""
    n = len(a)
    conv = np.convolve(a, a)
    # Shift for numerical stability
    conv_shifted = conv - conv.max()
    weights = np.exp(tau * conv_shifted)
    weights /= weights.sum()
    
    # Gradient = sum_k weights[k] * grad_k
    # grad_k[m] = a[k-m] + a[m-k] (with boundary handling)
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-10:
            continue
        # Add contribution from lag k
        ms = max(0, k - n + 1)
        me = min(n - 1, k)
        if ms <= me:
            grad[ms:me+1] += weights[k] * a[k-me:k-ms+1][::-1]
        ms2 = max(0, k)
        me2 = min(n - 1, k + n - 1)
        if ms2 <= me2:
            grad[ms2:me2+1] += weights[k] * a[ms2-k:me2-k+1]
    
    return grad, conv.max()
```

"""
Hybrid optimization: smoothed gradient descent + LP refinement + local search.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

# === Core functions ===

def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def auto_convolve(a):
    """Compute auto-convolution efficiently."""
    return np.convolve(a, a)

def subgradient_of_max_conv(a, tol=1e-6):
    """
    Compute subgradient of max_k conv(a,a)[k] with respect to a.
    Uses the active set (lags where conv is within tol of max).
    Returns (subgradient, max_value, num_active).
    """
    n = len(a)
    conv = np.convolve(a, a)
    max_val = conv.max()
    threshold = max_val * (1.0 - tol)
    active = np.where(conv >= threshold)[0]
    
    subgrad = np.zeros(n)
    for k in active:
        # grad of conv[k] w.r.t. a[m] = a[k-m] (if valid) + a[m-k] (if valid)
        # First term: a[k-m], valid for m in [max(0, k-n+1), min(n-1, k)]
        ms = max(0, k - n + 1)
        me = min(n - 1, k)
        if ms <= me:
            subgrad[ms:me+1] += a[k-me:k-ms+1][::-1]
        # Second term: a[m-k], valid for m in [max(0, k), min(n-1, k+n-1)]
        ms2 = max(0, k)
        me2 = min(n - 1, k + n - 1)
        if ms2 <= me2:
            subgrad[ms2:me2+1] += a[ms2-k:me2-k+1]
    
    return subgrad, max_val, len(active)

def smoothed_subgradient(a, tau=10.0):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*conv[k])).
    This is a smooth approximation to max_k conv[k].
    """
    n = len(a)
    conv = np.convolve(a, a)
    conv_norm = conv - conv.max()
    weights = np.exp(tau * conv_norm)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n), conv.max()
    weights /= wsum
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-10:
            continue
        ms = max(0, k - n + 1)
        me = min(n - 1, k)
        if ms <= me:
            grad[ms:me+1] += w * a[k-me:k-ms+1][::-1]
        ms2 = max(0, k)
        me2 = min(n - 1, k + n - 1)
        if ms2 <= me2:
            grad[ms2:me2+1] += w * a[ms2-k:me2-k+1]
    
    return grad, conv.max()

# === Optimization methods ===

def multiplicative_grad_descent(a, time_limit=60.0, verbose=True):
    """
    Multiplicative weight gradient descent with adaptive smoothing.
    """
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    tau = 5.0  # smoothing parameter
    eta = 0.1  # learning rate
    stagnant = 0
    
    it = 0
    while time.monotonic() - start < time_limit:
        grad, max_val = smoothed_subgradient(a, tau)
        score = 2.0 * n * max_val / (a.sum() ** 2)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 50 == 0 or it < 5):
                print(f"    MG iter {it:4d}: score={score:.6f}, tau={tau:.1f}")
        else:
            stagnant += 1
        
        if stagnant > 30:
            # Adapt parameters
            if tau < 100.0:
                tau *= 1.5
            eta *= 0.7
            stagnant = 0
            if eta < 1e-4:
                break
        
        # Multiplicative update: a_new = a * exp(-eta * grad)
        # Center gradient to maintain scale
        grad_centered = grad - grad.mean()
        # Scale gradient for stability
        grad_scale = np.abs(grad_centered).max()
        if grad_scale > 1e-15:
            grad_centered /= grad_scale
        
        log_ratio = -eta * grad_centered
        # Clip for numerical stability
        log_ratio = np.clip(log_ratio, -5, 5)
        a_new = a * np.exp(log_ratio)
        a = normalize_nonneg(a_new)
        
        it += 1
    
    if verbose:
        print(f"    MG done: best={best_score:.6f}, {it} iters, {time.monotonic()-start:.1f}s")
    return best_a, best_score

def lp_refine(a, time_limit_s=60.0, verbose=True):
    """LP-based refinement (same as before but cleaner)."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    stagnant = 0
    it = 0
    for it in range(500):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        max_b = conv.max()
        score = 2.0 * n * max_b / (a.sum() ** 2)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 30:
            break
        
        remaining = time_limit_s - elapsed
        if remaining < 10.0:
            break
        
        # Build sparse convolution matrix
        n_rows = 2 * n - 1
        col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
        row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
        data = np.tile(a, n).astype(np.float64)
        A = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))
        
        c = -np.ones(n)
        b_ub = np.full(n_rows, max_b)
        bounds = [(0.0, None)] * n
        
        timeout = max(5.0, min(60.0, remaining * 0.4))
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': timeout})
            if res.success and res.x is not None:
                g = np.maximum(0.0, res.x)
            else:
                break
        except Exception:
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement at iter {it}")
            break
        
        g = g / sum_g
        alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a = normalize_nonneg(a_new)
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.6f}, {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

def local_search(a, time_limit_s=60.0, rng=None, verbose=True):
    """Random local search with multiplicative perturbations."""
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        # Perturb a random block
        width = max(2, int(rng.integers(2, max(3, n // 50))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.15, size=stop - start))
        
        # Occasionally try a global perturbation
        if rng.random() < 0.1:
            candidate *= np.exp(rng.normal(0.0, 0.05, size=n))
        
        # Occasionally enforce symmetry
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 100 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")
    
    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))

def initial_two_hump(n, k=1):
    """Two-hump initialization: high at edges, low in middle."""
    t = np.arange(n, dtype=np.float64)
    return np.maximum(0.0, 1.0 - np.cos(2.0 * np.pi * k * t / n))

def initial_gaussian(n, sigma_frac=0.2):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)

def initial_random_smooth(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(2, n // 20), max(3, n // 5)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s
