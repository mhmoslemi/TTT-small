"""
Iterative optimization of non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based linearization, multiplicative weights, and local search.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def build_conv_matrix_col(a):
    """Build the matrix A such that (A @ b)[k] = conv(a, b)[k].
    A has shape (2n-1, n), A[k,j] = a[j] if 0<=k-j<n, else 0."""
    n = len(a)
    n_rows = 2 * n - 1
    # For each column j, the non-zero rows are k = j, j+1, ..., j+n-1 (clipped to [0, 2n-2])
    # A[k,j] = a[j] for k in [j, j+n-1]
    row_idx = []
    col_idx = []
    data = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2*n - 2)
        for k in range(k_start, k_end + 1):
            row_idx.append(k)
            col_idx.append(j)
            data.append(a[j])
    return csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))

# Actually, let me vectorize this:
def build_conv_matrix_vec(a):
    n = len(a)
    n_rows = 2 * n - 1
    # Column j has non-zeros at rows j, j+1, ..., j+n-1 (clipped)
    col_idx = np.repeat(np.arange(n), n)  # each column j appears n times
    # For column j, rows are j, j+1, ..., j+n-1
    # But we need to clip to [0, 2n-2]
    # Actually for k = j to j+n-1, k ranges from j to j+n-1
    # k is valid if 0 <= k <= 2n-2, which is always true for j in [0,n-1] and k in [j, j+n-1]
    # since j+n-1 <= 2n-2 (as j <= n-1)
    row_idx = np.concatenate([np.arange(j, j+n) for j in range(n)])
    data = np.tile(a, n)
    return csr_matrix((data.astype(np.float64), (row_idx, col_idx)), shape=(n_rows, n))

def lp_step(a, M, time_limit=30.0):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns b or None."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix_vec(a)
    c = -np.ones(n)
    b_vec = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit, 'dual_feasibility_tolerance': 1e-8})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception as e:
        print(f"    LP error: {e}")
    return None

def lp_refine(a, time_limit_s=600.0, max_iters=1000, verbose=False):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 50:
            break
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(60.0, remaining * 0.2))
        g = lp_step(a, M, time_limit=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-8:
            break
        g = g / sum_g
        # Adaptive step size
        delta = sum_g - 1.0
        alpha = min(0.5, max(0.02, 0.5 * delta))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-12:
            a = a_new
        else:
            # Try smaller step
            alpha2 = alpha * 0.3
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

def multiplicative_weights_refine(a, time_limit_s=300.0, max_iters=5000, verbose=False):
    """Mirror descent / multiplicative weights to flatten the autocorrelation."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        max_b = conv.max()
        score = 2.0 * n * max_b / (a.sum() ** 2)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 200:
                break
        # Find active lags (near the max)
        threshold = max_b * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            break
        # Compute "gradient" for active lags
        # For lag k, d conv_k / da_i = a_{k-i} (with bounds)
        # We want to decrease a_i where the gradient is large (i.e., where changing a_i most affects the peak)
        grad = np.zeros(n)
        for k in active:
            # conv_k = sum_j a_j * a_{k-j}
            # d conv_k / da_i = a_{k-i} (if 0 <= k-i < n)
            # So grad[i] += a_{k-i}
            valid = (k - np.arange(n) >= 0) & (k - np.arange(n) < n)
            grad[valid] += a[k - np.arange(n)[valid]]
        # Multiplicative update: a_i <- a_i * exp(-eta * grad_i)
        # Choose eta adaptively
        eta = 0.1 / (1.0 + it * 0.01)
        a_new = a * np.exp(-eta * grad / (grad.max() + 1e-15))
        a_new = normalize_nonneg(a_new)
        a = a_new
    return best_a, best_score

def local_search(a, time_limit_s=120.0, rng=None, verbose=False):
    if rng is None:
        rng = np.random.default_rng(42)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        # Random block perturbation
        width = max(2, int(rng.integers(n // 100, n // 5)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.05, size=stop - start))
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
    return best_a, best_score

def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))

def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)

def initial_blackman(n):
    if n == 1:
        return np.ones(1)
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2*np.pi*x) + 0.08 * np.cos(4*np.pi*x))

def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s

For the uniform sequence, max_b = n and S = n (if a_i = 1), so score = 2n*n/n^2 = 2. The average conv value is n^2/(2n-1) ≈ n/2, and the peak is n, so peak/average ≈ 2.

To beat score 2, we need peak/average < 2, i.e., the autocorrelation needs to be flatter than the triangular shape.

Now, can we make the autocorrelation of a non-negative sequence flatter than triangular? 

The triangular shape is the autocorrelation of a "box" (uniform) sequence. To get a flatter autocorrelation, we need a "more spread out" or "less concentrated" sequence.

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines LP-based successive linearization, multiplicative weights (mirror descent),
and local search, with multi-start over different lengths and initializations.
Uses adaptive time allocation to focus on the most promising candidates.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-15:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def lp_solve(a, M, time_limit=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_vec = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=600.0, max_iters=2000, verbose=False):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 100 == 0 or it < 5):
                print(f"    LP iter {it}: score={score:.8f}")
        else:
            stagnant += 1
        if stagnant > 80:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.8f}")
            break
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 4.0:
            break
        timeout = max(2.0, min(90.0, remaining * 0.15))
        g = lp_solve(a, M, time_limit=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 5e-6:
            if verbose:
                print(f"    LP no improvement (sum_g={sum_g:.8f}) at iter {it}")
            break
        g = g / sum_g
        delta = sum_g - 1.0
        alpha = min(0.5, max(0.01, 0.4 * delta))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-13:
            a = a_new
        else:
            alpha2 = alpha * 0.25
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


def multiplicative_weights(a, time_limit_s=200.0, max_iters=20000, verbose=False):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        max_b = conv.max()
        score = 2.0 * n * max_b / (a.sum() ** 2)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 500:
                break
        threshold = max_b * 0.9
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            break
        grad = np.zeros(n)
        idx = np.arange(n)
        for k in active:
            valid = (k - idx >= 0) & (k - idx < n)
            grad[valid] += a[k - idx[valid]]
        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        eta = 0.3 / (1.0 + 0.005 * it)
        a_new = a * np.exp(-eta * grad / grad_max)
        a_new = normalize_nonneg(a_new)
        a = a_new
    return best_a, best_score


def local_search(a, time_limit_s=120.0, rng=None, verbose=False):
    if rng is None:
        rng = np.random.default_rng(123)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(rng.integers(max(2, n // 200), max(3, n // 4))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        noise = rng.normal(0.0, 0.08, size=stop - start)
        candidate[start:stop] *= np.exp(noise)
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score - 1e-15:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 100 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.8f}")
    return best_a, best_score


def coordinate_descent(a, time_limit_s=120.0, verbose=False):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    it = 0
    while time.monotonic() - start_time < time_limit_s:
        improved = False
        for i in range(n):
            if time.monotonic() - start_time > time_limit_s:
                break
            orig = a[i]
            for frac in [0.7, 0.85, 1.15, 1.3]:
                a[i] = orig * frac
                s = a.sum()
                if s < 1e-15:
                    a[i] = orig
                    continue
                a /= s
                sc = score_sequence(a)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            a[i] = orig
            a /= a.sum()
        it += 1
        if not improved and it > 2:
            break
    return best_a, best_score


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    if n <= 1:
        return np.ones(max(1, n))
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_kaiser_like(n, beta=2.0):
    if n <= 1:
        return np.ones(max(1, n))
    t = np.arange(n, dtype=np.float64)
    x = 2.0 * t / (n - 1) - 1.0
    from scipy.special import i0
    return i0(beta * np.sqrt(np.maximum(0, 1 - x**2))) / i0(beta)


def random_smooth_init(n, rng, n_bumps=4):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(2, n // 15), max(3, n // 4)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def get_initial(n, init_type, rng):
    if init_type == 'raised_cosine':
        return initial_raised_cosine(n)
    elif init_type == 'gaussian':
        return initial_gaussian(n, sigma_frac=rng.choice([0.2, 0.3, 0.4, 0.5]))
    elif init_type == 'blackman':
        return initial_blackman(n)
    elif init_type == 'kaiser':
        return initial_kaiser_like(n, beta=float(rng.choice([1.0, 2.0, 3.0, 5.0])))
    elif init_type == 'random':
        return random_smooth_init(n, rng, n_bumps=int(rng.integers(2, 8)))
    return initial_raised_cosine(n)
