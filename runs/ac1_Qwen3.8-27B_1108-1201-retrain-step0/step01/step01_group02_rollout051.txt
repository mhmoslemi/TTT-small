import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a**2)

def normalize(a):
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def compute_subgradient(a, top_k=5):
    """Compute subgradient of max(conv(a,a)) using top-k peaks."""
    n = len(a)
    conv = np.convolve(a, a)
    # Find top-k peaks
    k = min(top_k, len(conv))
    peak_indices = np.argsort(conv)[-k:][::-1]
    # Weight peaks by how close they are to the max
    peak_vals = conv[peak_indices]
    max_val = peak_vals[0]
    # Exponential weighting
    weights = np.exp(-5.0 * (max_val - peak_vals) / max_val)
    weights /= weights.sum()
    
    # Compute gradient for each peak lag
    grad = np.zeros(n)
    for w, lag in zip(weights, peak_indices):
        # dc[lag]/da[i] = 2*a[lag-i] if 0 <= lag-i < n
        # So grad contribution = w * 2 * a[lag-i] for valid i
        # lag-i ranges from lag-(n-1) to lag
        # Valid when 0 <= lag-i < n, i.e., lag-n+1 <= i <= lag
        i_start = max(0, lag - n + 1)
        i_end = min(n - 1, lag)
        if i_start <= i_end:
            # a[lag-i] for i in [i_start, i_end]
            # When i = i_start: a[lag - i_start]
            # When i = i_end: a[lag - i_end]
            indices = np.arange(i_start, i_end + 1)
            a_indices = lag - indices  # These are the indices into a
            grad[i_start:i_end+1] += w * 2.0 * a[a_indices]
    
    return grad, conv, peak_indices

def subgradient_descent(a, time_limit=60, lr_init=0.01, momentum=0.9, top_k=5):
    """Subgradient descent with momentum and adaptive step size."""
    start = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    lr = lr_init

        # Update with momentum

        a_new = a - lr * v

        if s < 1e-15:

            best_a = a_new.copy()
            patience = 0
            # Adaptive lr: increase slightly

        else:

            lr = lr * 0.95
        
        a = a_new

    return best_a, best_score

Let me also think about the **spectral approach**. The autocorrelation c = conv(a, a) has DFT:
`DFT(c)[m] = |DFT(a)[m]|^2`

So `max c[k]` is the maximum time-domain value of the inverse DFT of `|A[m]|^2` where A = DFT(a).

We want to minimize `max_k c[k]` subject to `sum(a) = 1, a >= 0`.

`sum(a) = A[0]` (the DC component of the DFT). So we want `A[0] = 1`.

The constraint `a >= 0` translates to: the inverse DFT of A must be non-negative.

This is a complex constraint. But the insight is: we want `|A[m]|^2` to be as flat as possible (so that its inverse DFT has a low peak). The flatter the spectrum, the lower the peak of the autocorrelation.

For a raised cosine window, the spectrum is also somewhat smooth but not flat. A "flat-top" spectrum would be ideal, but it's hard to achieve with non-negative time-domain sequences.

This suggests that we should try to make the spectrum of a as flat as possible (while keeping a >= 0 and sum(a) = 1).

One approach: **Iterative spectral flattening**.
1. Start with some a.
2. Compute A = DFT(a).
3. Modify A to be flatter (e.g., replace |A| with a smoothed version of |A|).
4. Take inverse DFT to get new a.
5. Project to a >= 0.
6. Iterate.

But this might not directly minimize the peak... it minimizes a related quantity (the L2 norm of the spectrum or some average measure).

Actually, a better spectral approach: since we want to minimize `max_k c[k]`, and `c = IDFT(|A|^2)`, we want to minimize the peak of `IDFT(|A|^2)`. This is like a "peak reduction" problem.

**Peak reduction in the time domain** = **flattening in the frequency domain** (roughly).

So the algorithm would be:
1. Compute c = conv(a, a).
2. Find the peak of c.
3. In the frequency domain, this peak corresponds to energy concentrated at specific frequencies.
4. Reduce the energy at those frequencies.
5. Transform back.

This is essentially what the subgradient method does, but in the frequency domain it might be more efficient.

                a_indices = lag - indices
                grad[i_start:i_end+1] += w * 2.0 * a[a_indices]

        a_new = a * np.exp(-lr_normalized * grad / max(grad) if max(grad) > 0 else 0)

        return np.zeros(n), conv
    weights = np.exp(-4.0 * (peak_vals[0] - peak_vals) / peak_vals[0])
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, lag in zip(weights, peak_indices):
        i_start = max(0, lag - n + 1)
        i_end = min(n - 1, lag)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            a_idx = lag - idx
            grad[i_start:i_end + 1] += w * 2.0 * a[a_idx]
    return grad, conv


def multiplicative_descent(a, time_limit=30.0, beta_init=5.0, beta_end=50.0,
                           lr_init=0.15, use_soft=True):
    """Multiplicative weight descent with adaptive beta and learning rate."""
    start = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    lr = lr_init
    stagnant = 0
    it = 0

    while time.monotonic() - start < time_limit:
        progress = min(1.0, (time.monotonic() - start) / time_limit)
        beta = beta_init + (beta_end - beta_init) * progress

        if use_soft and beta < 100.0:
            grad, conv = compute_soft_gradient(a, beta=beta)
        else:
            grad, conv = compute_topk_gradient(a, top_k=8)

        max_grad = grad.max()
        if max_grad < 1e-15:
            break

        # Multiplicative update with clipping to avoid extreme changes
        effective_lr = lr * max_grad
        a_new = a * np.exp(-effective_lr * grad)
        a_new = np.maximum(0.0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / s

        new_score = score_sequence(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            stagnant = 0
            lr = min(lr * 1.03, 2.0)
        else:
            stagnant += 1
            lr *= 0.97
            if stagnant > 80:
                # Try a kick: random smooth perturbation
                rng_kick = np.random.default_rng(it)
                a_new = a * np.exp(rng_kick.normal(0, 0.05, size=n))
                a_new = normalize(a_new)
                new_score = score_sequence(a_new)
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    stagnant = 0
                a = a_new
        a = a_new
        it += 1

    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_parts = []
    for j in range(n):
        row_parts.append(np.arange(j, j + n, dtype=np.int32))
    row_idx = np.concatenate(row_parts)
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def lp_polish(a, time_limit=60.0, max_iters=200):
    """LP-based polishing: find g maximizing sum(g) s.t. conv(a,g) <= peak."""
    start = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_limit - 5:
            break

        conv = np.convolve(a, a)
        M = conv.max()
        n_rows = 2 * n - 1

        A = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(n_rows, M)
        bounds = [(0.0, None)] * n

        remaining = time_limit - elapsed
        timeout = max(2.0, min(30.0, remaining * 0.4))

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': timeout})
            if not res.success or res.x is None:
                break
            g = np.maximum(0.0, res.x)
            sum_g = g.sum()
            if sum_g <= 1.0 + 1e-8:
                stagnant += 1
                if stagnant > 3:
                    break
                # Try with slightly larger M
                M *= 1.001
                A = build_conv_matrix(a)
                b_ub = np.full(n_rows, M)
                res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs', options={'presolve': True, 'time_limit': timeout})
                if not res.success or res.x is None:
                    break
                g = np.maximum(0.0, res.x)
                sum_g = g.sum()
                if sum_g <= 1.0 + 1e-8:
                    break

            g = g / sum_g
            alpha = min(0.5, max(0.05, 1.5 / (it + 3)))
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = normalize(a_new)
            new_score = score_sequence(a_new)

            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                stagnant = 0
            else:
                # Try smaller alpha
                alpha2 = alpha * 0.3
                a_new = (1.0 - alpha2) * a + alpha2 * g
                a_new = normalize(a_new)
                new_score = score_sequence(a_new)
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_a = a_new.copy()
                    a = a_new
                    stagnant = 0
                else:
                    stagnant += 1
                    a = a_new
                    if stagnant > 5:
                        break
        except Exception:
            break

    return best_a, best_score


def local_search(a, time_limit=30.0, rng=None):
    """Random local search with smooth mutations."""
    if rng is None:
        rng = np.random.default_rng(42)
    start = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    while time.monotonic() - start < time_limit:
        # Smooth random perturbation
        width = int(rng.integers(max(3, n // 50), n // 5))
        pos = int(rng.integers(0, max(1, n - width)))
        candidate = a.copy()
        noise = np.exp(rng.normal(0.0, 0.08, size=width))
        candidate[pos:pos + width] *= noise
        candidate = normalize(candidate)
        sc = score_sequence(candidate)
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            a = candidate.copy()
        # Symmetry exploitation
        if rng.random() < 0.2:
            candidate = 0.5 * (a + a[::-1])
            candidate = normalize(candidate)
            sc = score_sequence(candidate)
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
                a = candidate.copy()

    return best_a, best_score


def propose_candidate(budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses soft-max multiplicative descent + LP polishing + local search.
    Explores multiple lengths and initializations within the time budget.
    """
    rng = np.random.default_rng(777)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 25.0

    best_score = float('inf')
    best_seq = None

    # Try to use previous best
    prev_seq = None
    try:
        prev_seq = np.array(height_sequence_1, dtype=np.float64)
        if len(prev_seq) > 0 and np.all(np.isfinite(prev_seq)):
            prev_seq = np.maximum(0, prev_seq)
            print(f"Previous best: n={len(prev_seq)}, score={score_sequence(prev_seq):.6f}")
    except (NameError, TypeError, ValueError):
        prev_seq = None

    # Configuration list: (n, init_type, md_time, lp_time, ls_time)
    # Total time per config should not exceed what's available
    configs = []

    # Start with previous best (most likely to improve)
    if prev_seq is not None:
        configs.append((len(prev_seq), 'previous', 45, 120, 20))

    # Various lengths
    configs.extend([
        (500, 'raised_cosine', 40, 100, 15),
        (750, 'raised_cosine', 50, 120, 15),
        (1000, 'raised_cosine', 60, 140, 20),
        (1500, 'raised_cosine', 70, 160, 20),
        (2000, 'raised_cosine', 80, 180, 20),
        (1000, 'gaussian', 50, 120, 15),
        (2000, 'gaussian', 70, 140, 15),
        (1000, 'blackman', 40, 100, 15),
        (1500, 'fourier', 60, 120, 15),
        (2000, 'fourier', 70, 140, 15),
        (1000, 'random', 40, 80, 10),
        (2000, 'random', 60, 100, 10),
        (3000, 'raised_cosine', 80, 180, 20),
    ])

    for n, init_type, md_time, lp_time, ls_time in configs:
        remaining = deadline - time.monotonic()
        needed = md_time + lp_time + ls_time + 10
        if remaining < needed:
            continue
        if remaining < 60:
            break

        # Generate initial sequence
        if init_type == 'previous' and prev_seq is not None:
            a = prev_seq.copy()
            if len(a) != n:
                # Resample
                old_t = np.linspace(0.0, 1.0, len(a))
                new_t = np.linspace(0.0, 1.0, n)
                a = np.interp(new_t, old_t, a)
                a = np.maximum(0, a)
        elif init_type == 'raised_cosine':
            a = init_raised_cosine(n)
        elif init_type == 'gaussian':
            a = init_gaussian(n, sigma_frac=rng.uniform(0.2, 0.4))
        elif init_type == 'blackman':
            a = init_blackman(n)
        elif init_type == 'fourier':
            a = init_fourier_smooth(n, rng, n_harmonics=int(rng.integers(6, 20)))
        elif init_type == 'random':
            a = init_random_bumps(n, rng, n_bumps=int(rng.integers(3, 10)))
        else:
            a = init_raised_cosine(n)

        a = normalize(a)

        # Phase 1: Multiplicative descent
        a_md, score_md = multiplicative_descent(a, time_limit=md_time)
        if score_md < best_score:
            best_score = score_md
            best_seq = a_md.copy()
            print(f"n={n}, init={init_type}: MD score={score_md:.6f} (best={best_score:.6f})")

        # Phase 2: LP polishing
        remaining = deadline - time.monotonic()
        if remaining > lp_time * 0.6:
            a_lp, score_lp = lp_polish(a_md, time_limit=lp_time)
            if score_lp < best_score:
                best_score = score_lp
                best_seq = a_lp.copy()
                print(f"n={n}, init={init_type}: LP score={score_lp:.6f} (best={best_score:.6f})")

        # Phase 3: Local search
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_ls, score_ls = local_search(a_md if score_md < (score_lp if 'score_lp' in dir() else score_md)
                                           if 'score_lp' in dir() else a_md,
                                          time_limit=min(ls_time, remaining - 5), rng=rng)
            if score_ls < best_score:
                best_score = score_ls
                best_seq = a_ls.copy()
                print(f"n={n}, init={init_type}: LS score={score_ls:.6f} (best={best_score:.6f})")

    # Final intensive polishing on the best sequence
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        print(f"\n=== Final polish: score={best_score:.6f}, n={len(best_seq)}, remaining={remaining:.0f}s ===")

        if remaining > 30:
            # More LP on best
            t_lp = min(remaining * 0.5, 300)
            a_f1, s_f1 = lp_polish(best_seq, time_limit=t_lp)
            if s_f1 < best_score:
                best_score = s_f1
                best_seq = a_f1.copy()
                print(f"  Final LP1: {best_score:.6f}")

            remaining = deadline - time.monotonic()
            if remaining > 20:
                # More MD with higher beta
                t_md = min(remaining * 0.4, 120)
                a_f2, s_f2 = multiplicative_descent(best_seq, time_limit=t_md,
                                                    beta_init=30.0, beta_end=100.0, lr_init=0.05)
                if s_f2 < best_score:
                    best_score = s_f2
                    best_seq = a_f2.copy()
                    print(f"  Final MD: {best_score:.6f}")

            remaining = deadline - time.monotonic()
            if remaining > 15:
                # Final LP
                t_lp2 = min(remaining * 0.6, 200)
                a_f3, s_f3 = lp_polish(best_seq, time_limit=t_lp2)
                if s_f3 < best_score:
                    best_score = s_f3
                    best_seq = a_f3.copy()
                    print(f"  Final LP2: {best_score:.6f}")

            remaining = deadline - time.monotonic()
            if remaining > 10:
                # Final local search
                a_f4, s_f4 = local_search(best_seq, time_limit=min(remaining * 0.5, 60), rng=rng)
                if s_f4 < best_score:
                    best_score = s_f4
                    best_seq = a_f4.copy()
                    print(f"  Final LS: {best_score:.6f}")

    if best_seq is None:
        best_seq = init_raised_cosine(500)
        best_score = score_sequence(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    # Verify with evaluate_sequence
    verify = evaluate_sequence([float(x) for x in best_seq])
    print(f"\nFinal result: score={best_score:.6f}, n={len(best_seq)}")
    print(f"Verified via evaluate_sequence: {verify:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")

    return [float(x) for x in best_seq]
