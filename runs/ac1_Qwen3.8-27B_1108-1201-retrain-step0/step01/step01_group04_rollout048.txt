"""
Step-function search minimizing 2n*max(conv(p,p))/sum(p)^2.
Combines diverse shape initialization, projected gradient descent
with smoothed-max surrogate, Fourier-domain moves, and block swaps.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len

def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    n = int(p.size)
    if not np.isfinite(s) or n == 0 or s <= 1e-12:
        return np.full(max(1, n), 1.0/max(1, n))
    return p / s

def conv_sq(p):
    n = int(p.size)
    if n == 0:
        return np.empty(0)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]

def internal_score(p):
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(p.sum())
    if s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    return float(2.0 * n * cmax / (s * s))

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / n)
    return out / s

def gradient_smoothed_max(p, T_factor):
    """Returns (gradient, smoothed_value, c) for smoothed max of conv(p,p)."""
    n = int(p.size)
    L = next_fast_len(2 * n - 1)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]
    cmax = float(np.max(c))
    if cmax <= 0:
        return np.zeros(n), 0.0, c
    T = max(T_factor * cmax, 1e-300)
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = float(e.sum())
    w = e / s
    surr = cmax + T * np.log(s)
    
    # grad[j] = 2 * sum_m w[m+j] * p[m]
    # = 2 * (p * w_reversed)[2n-2-j]
    w_rev = w[::-1]
    L2 = next_fast_len(n + (2 * n - 1) - 1)
    Fp2 = fft(p, L2)
    Fwr = fft(w_rev, L2)
    conv_pw = np.real(ifft(Fp2 * Fwr))
    # grad[j] = 2 * conv_pw[2n-2-j] for j=0,...,n-1
    grad = 2.0 * conv_pw[2 * n - 2 : n - 2 : -1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad, surr, c

def make_hann(n):
    n = max(2, int(n))
    x = np.arange(n)
    p = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    p = np.maximum(p, 1e-15)
    return safe_normalize(p)

def make_hamming(n):
    n = max(2, int(n))
    x = np.arange(n)
    p = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / (n - 1))
    return safe_normalize(p)

def make_blackman(n):
    n = max(2, int(n))
    x = np.arange(n)
    p = 0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * x / (n - 1))
    return safe_normalize(np.maximum(p, 1e-15))

def make_raised_cosine(n, a=1.0):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    # p = 1 + a * cos(2*pi*x) -- one full period
    p = 1.0 + float(a) * np.cos(2.0 * np.pi * x)
    return safe_normalize(np.maximum(p, 1e-15))

def make_fejer(n):
    """Fejér kernel shape: triangular in frequency."""
    n = max(2, int(n))
    x = np.arange(n)
    p = 1.0 - np.abs(x - (n - 1) / 2.0) / ((n - 1) / 2.0 + 1e-10)
    return safe_normalize(np.maximum(p, 1e-15))

def make_two_bump(n, sigma=0.15, center1=0.2, center2=0.8):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-((x - center1) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - center2) ** 2) / (2 * sigma ** 2))
    return safe_normalize(p)

def make_dome(n, width=0.3):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-((x - 0.5) ** 2) / (2 * width ** 2))
    return safe_normalize(p)

def make_paren(n, a=2.0):
    """U-shape: high at edges, low in middle."""
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + float(a) * (2.0 * x - 1.0) ** 2
    return safe_normalize(p)

def make_super_gauss(n, alpha=2.0):
    n = max(2, int(n))
    x = np.linspace(-1.0, 1.0, n)
    p = np.exp(-np.abs(x) ** (2.0 * float(alpha)))
    return safe_normalize(p)

def make_random_smooth(n, rng, nharmonics=5):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    p = np.ones(n)
    for k in range(1, min(nharmonics, n // 4) + 1):
        a = float(rng.uniform(0.0, 1.0 / k))
        b = float(rng.uniform(0.0, 1.0 / k))
        ph = float(rng.uniform(0, 2 * np.pi))
        p += a * np.cos(2 * np.pi * k * x + ph)
    p = np.maximum(p, 0.01)
    return safe_normalize(p)

def fourier_perturb(p, rng, amplitude=0.1):
    """Add a smooth perturbation in the frequency domain."""
    n = int(p.size)
    L = next_fast_len(n)
    Fp = fft(p, L)
    # Random smooth filter (low-pass weighted random complex)
    freqs = np.fft.fftfreq(L)
    # Weight by some smooth function to keep perturbation smooth
    smooth = np.exp(-np.abs(freqs) * L * 0.02)
    delta = (rng.normal(0, 1, L) + 1j * rng.normal(0, 1, L)) * smooth * float(amplitude)
    Fp2 = Fp + delta
    p2 = np.real(ifft(Fp2, n))
    return project_simplex(p2)

def block_swap(p, rng):
    """Swap two random blocks of the sequence."""
    n = int(p.size)
    if n < 8:
        return p
    # Choose block size
    max_block = max(2, n // 8)
    b1 = int(rng.integers(1, max_block))
    b2 = int(rng.integers(1, max_block))
    s1 = int(rng.integers(0, n - b1))
    s2 = int(rng.integers(0, n - b2))
    if not (s1 + b1 <= s2 or s2 + b2 <= s1):
        return p
    p2 = p.copy()
    temp = p2[s1:s1+b1].copy()
    p2[s1:s1+b1] = p2[s2:s2+b2]
    p2[s2:s2+b2] = temp
    return project_simplex(p2)
