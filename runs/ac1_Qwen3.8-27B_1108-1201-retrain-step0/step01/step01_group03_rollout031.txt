"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-accelerated LSE-smoothed gradient descent with Nesterov momentum,
adaptive temperature annealing, LP-based refinement, and coordinate descent polish.
Explores multiple starting points and sequence lengths within the time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    mc = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return 2.0 * n * mc / (S * S)


def _norm(a, S_target):
    """Normalize sequence to have a specific sum."""
    S = a.sum()
    if S < 1e-15:
        return a
    return a * (S_target / S)


def _compute_weights(c, tau, n_top=None):
    """Compute softmax weights over convolution values."""
    if n_top is not None and n_top < len(c):
        idx = np.argpartition(c, -n_top)[-n_top:]
        c_work = np.full_like(c, -1e30)
        c_work[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_work = c
        c_max = c.max()
    w = np.exp(tau * (c_work - c_max))
    wsum = w.sum()
    if wsum < 1e-30:
        return None
    w /= wsum
    return w


def _grad_lse(a, tau, n_top=None):
    """Compute LSE-smoothed gradient using FFT for efficiency."""
    n = len(a)
    c = np.convolve(a, a)
    w = _compute_weights(c, tau, n_top)
    if w is None:
        # Fall back to hard max gradient
        k = int(c.argmax())
        grad = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] = 2.0 * a[k - i]
        return grad
    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # Using FFT: grad[i] = 2 * (a * w_rev)[2n-2-i]
    N = 1
    while N < 3 * n:
        N <<= 1
    A = np.fft.rfft(a, N)
    w_rev = w[::-1]
    WR = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(A * WR, N)
    # Extract: grad[i] = 2 * conv[2n-2-i] for i=0..n-1
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad


def _grad_hard(a):
    """Compute hard-max gradient."""
    n = len(a)
    c = np.convolve(a, a)
    k = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    for i in range(lo, hi + 1):
        grad[i] = 2.0 * a[k - i]
    return grad


def _lp_solve(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    n = len(f)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, n_iter=200):
    """LP-based refinement: iteratively solve LP and blend."""
    n = len(a)
    S_t = float(a.sum())
    best = a.copy()
    best_obj = _obj(a)
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        c = np.convolve(a, a)
        M = float(c.max())
        g = _lp_solve(a, M)
        if g is None:
            break
        improved = False
        for t in [0.7, 0.5, 0.3, 0.15, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _norm(a_new, S_t)
            s = _obj(a_new)
            if s < best_obj:
                a = a_new
                best = a_new.copy()
                best_obj = s
                improved = True
                break
        if not improved:
            break
    return best, best_obj


def _gd_optimize(a0, deadline, max_iters=10000, tau_start=2.0, tau_end=150.0,
                 lr_base=0.08, momentum=0.9, n_top=None, check_interval=100):
    """Smoothed gradient descent with Nesterov momentum and adaptive temperature."""
    n = len(a0)
    S_t = float(a0.sum())
    a = np.maximum(a0, 0.0).copy()
    a = _norm(a, S_t)
    v = np.zeros(n)
    best = a.copy()
    best_obj = _obj(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(max_iters - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_base * (1.0 + np.cos(np.pi * frac)) * 0.5
        lr = max(lr, 0.001)

        grad = _grad_lse(a, tau, n_top)
        grad -= grad.mean()

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        a_new = _norm(a_new, S_t)

        if (it + 1) % check_interval == 0:
            s = _obj(a_new)
            if s < best_obj:
                best_obj = s
                best = a_new.copy()

        a = a_new

    s = _obj(a)
    if s < best_obj:
        best_obj = s
        best = a.copy()
    return best, best_obj


def _coord_descent(a0, deadline, n_sweeps=5):
    """Coordinate descent: try adjusting each element."""
    n = len(a0)
    S_t = float(a0.sum())
    a = a0.copy()
    best_obj = _obj(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            old = a[i]
            if old > 1e-12:
                # Try reducing
                for frac in [0.6, 0.8, 1.1, 1.3, 1.5]:
                    a[i] = old * frac
                    a = _norm(a, S_t)
                    s = _obj(a)
                    if s < best_obj:
                        best_obj = s
                        old = a[i]
                        improved = True
                    else:
                        a[i] = old
                # Reset to best found for this coordinate
                a[i] = old
            else:
                for val in [S_t * 0.001, S_t * 0.005]:
                    a[i] = val
                    a = _norm(a, S_t)
                    s = _obj(a)
                    if s < best_obj:
                        best_obj = s
                        improved = True
                        break
                    else:
                        a[i] = 0.0
        a = _norm(a, S_t)
        if not improved:
            break
    return a, best_obj


def _perturb_restart(a0, rng, deadline, max_iters=3000):
    """Perturb a sequence and re-optimize."""
    n = len(a0)
    S_t = float(a0.sum())
    noise = rng.random(n) * 0.1 * S_t / n
    a_pert = a0 + noise
    a_pert = np.maximum(a_pert, 0.0)
    a_pert = _norm(a_pert, S_t)
    return _gd_optimize(a_pert, deadline, max_iters=max_iters,
                        tau_start=1.0, tau_end=80.0, lr_base=0.05)
