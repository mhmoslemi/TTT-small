"""
Optimization of non-negative sequences to minimize peak autocorrelation.
Uses scipy L-BFGS-B with LogSumExp smoothing, multiple initializations,
and targeted local search. Explores different sequence lengths.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_from_conv(c, n):
    """Compute score from convolution and n. Assumes sum(b)=1."""
    return 2.0 * n * np.max(c)

def objective_and_grad(b, T):
    """
    Compute LogSumExp objective and gradient.
    b is assumed to have sum = 1 (we'll handle normalization).
    """
    n = len(b)
    c = fft_convolve(b, b)
    # LogSumExp: T * log(sum(exp(c/T)))
    c_max = np.max(c)
    lse = T * (c_max / T + np.log(np.sum(np.exp((c - c_max) / T))))
    # Gradient: w = exp(c/T) / sum(exp(c/T))
    w = np.exp((c - c_max) / T)
    w /= np.sum(w)
    
    # grad[k] = 2 * (w cross-corr b_ext)[k]
    # where b_ext = [b, 0, ..., 0] (length 2n-1)
    N = len(c)  # 2n-1
    size = 1
    while size < 2 * N:
        size <<= 1
    W = np.fft.rfft(w, size)
    b_ext = np.zeros(N)
    b_ext[:n] = b
    B = np.fft.rfft(b_ext, size)
    # cross-correlation: ifft(conj(W) * B)
    cross = np.fft.irfft(np.conj(W) * B, size)[:n]
    grad = 2.0 * cross
    
    return lse, grad

def normalize_to_sum1(b):
    s = np.sum(b)
    if s < 1e-10:
        return b.copy()
    return b / s

def actual_score(b):
    """Actual score (same formula as evaluate_sequence, assuming b >= 0)."""
    n = len(b)
    sum_a = np.sum(b)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(b, b)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def make_ramp(n, rng, exponent=1.0):
    """Create a ramp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    # Power law ramp
    b = (t + 1.0) ** exponent
    b += 0.05 * rng.standard_normal(n)
    b = np.abs(b) + 0.01
    return b / np.sum(b)

def make_exponential(n, rng, decay_range=(0.5, 0.99)):
    """Create an exponential decay initial sequence."""
    r = rng.uniform(*decay_range)
    t = np.arange(n, dtype=np.float64)
    b = r ** t
    b += 0.05 * rng.standard_normal(n)
    b = np.abs(b) + 0.01
    return b / np.sum(b)

def make_bimodal(n, rng):
    """Create a bimodal initial sequence (high at both ends)."""
    t = np.arange(n, dtype=np.float64)
    center = n / 2.0
    b = np.abs(t - center) / center
    b += 0.05 * rng.standard_normal(n)
    b = np.abs(b) + 0.01
    return b / np.sum(b)

def make_symmetric_ramp(n, rng):
    """Symmetric ramp (high in middle, low at ends)."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    b = 1.0 - np.abs(t - center) / center
    b = np.abs(b) + 0.01
    b += 0.05 * rng.standard_normal(n)
    b = np.abs(b) + 0.01
    return b / np.sum(b)

def make_chirp(n, rng, freq_range=(1, 5)):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    f = rng.uniform(*freq_range)
    b = 0.5 + 0.3 * np.cos(2 * np.pi * f * t / n)
    # Add a ramp component
    b += 0.2 * t / n
    b += 0.05 * rng.standard_normal(n)
    b = np.abs(b) + 0.01
    return b / np.sum(b)

def make_random(n, rng):
    """Random positive initialization."""
    b = np.abs(rng.standard_normal(n)) + 0.1
    return b / np.sum(b)

def optimize_with_lbfgsb(b_init, T, maxiter=1000):
    """
    Optimize using L-BFGS-B to minimize the LogSumExp score.
    b is non-negative; we optimize in the space where sum(b) is free.
    Actually, since the score is scale-invariant, we can fix sum(b) = 1.
    But L-BFGS-B doesn't handle equality constraints well.
    
    Alternative: optimize in log-space. Let b = exp(x) / sum(exp(x)).
    Then the non-negativity is automatic.
    """
    # Work in log-space
    b_safe = np.maximum(b_init, 1e-10)
    x = np.log(b_safe) - np.log(np.sum(b_safe))
    
    def obj(x):
        b = np.exp(x)
        b = b / np.sum(b)
        val, _ = objective_and_grad(b, T)
        return val
    
    def obj_grad(x):
        b = np.exp(x)
        b = b / np.sum(b)
        val, grad_b = objective_and_grad(b, T)
        # Chain rule: db/dx_k = b_k - b_k * sum(b) / sum(b) ... 
        # Actually if b_k = exp(x_k) / sum(exp(x)), then:
        # db_k/dx_j = b_k * delta(kj) - b_k * b_j = b_k (delta(kj) - b_j)
        # So grad_x_j = sum_k grad_b_k * b_k * (delta(kj) - b_j)
        # = grad_b_j * b_j - b_j * sum_k grad_b_k * b_k
        inner = np.dot(grad_b, b)
        grad_x = b * (grad_b - inner)
        return val, grad_x
    
    result = minimize(obj_grad, x, method='L-BFGS-B', jac=True,
                      options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
    b_opt = np.exp(result.x)
    b_opt = b_opt / np.sum(b_opt)
    return b_opt, result.fun

def polish_local_search(b_init, time_limit, rng):
    """
    Targeted local search to reduce the peak of the convolution.
    Works with b normalized to sum = 1.
    """
    current = b_init.copy()
    current_score = actual_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    # Temperature for simulated annealing acceptance
    temp = 0.001
    cooling = 0.999
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        peak_val = np.max(c)
        peak_idx = int(np.argmax(c))
        
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.4:
            # Targeted: reduce peak contributors
            # The peak is at peak_idx: c[peak_idx] = sum_i b[i]*b[peak_idx-i]
            # Find which indices contribute most
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            
            # Reduce top contributors
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            
            # Increase some others to maintain sum
            other = np.setdiff1d(np.arange(n), top_idx)
            n_inc = min(5, len(other))
            if n_inc > 0:
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
                
        elif r < 0.6:
            # Reduce the peak by spreading mass
            # Decrease values near the "diagonal" that creates the peak
            half = peak_idx // 2
            width = max(2, n // 20)
            center_idx = min(max(half, width), n - width)
            sl = slice(center_idx - width, center_idx + width)
            candidate[sl] *= np.exp(rng.normal(-0.03, 0.01, size=2*width))
            # Compensate elsewhere
            other_sl = slice(0, n // 4) if center_idx > n // 2 else slice(3*n//4, n)
            candidate[other_sl] *= np.exp(rng.normal(0.03, 0.01, size=len(other_sl)))
            
        elif r < 0.75:
            # Small random perturbation
            candidate *= np.exp(rng.normal(0, 0.02, size=n))
            
        elif r < 0.85:
            # Block perturbation
            width = max(1, n // 30)
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.03, size=width))
            # Compensate in a different block
            start2 = int(rng.integers(0, n - width))
            candidate[start2:start2+width] *= np.exp(rng.normal(0, 0.03, size=width))
            
        else:
            # Symmetry-based perturbation
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0, 0.01, size=n))
        
        candidate = np.maximum(candidate, 1e-10)
        candidate = candidate / np.sum(candidate)
        
        new_score = actual_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-10)):
            current = candidate
            current_score = new_score
        
        temp *= cooling
    
    return best, best_score

Minimize peak autocorrelation of non-negative step functions using:
- LogSumExp smoothed objective with L-BFGS-B in log-space (softmax parameterization)
- Temperature annealing schedule for progressive sharpening
- Multiple diverse initializations (asymmetric ramps, exponentials, chirps, perturbations)
- Targeted local search with peak-contributor analysis
- Multi-scale exploration across sequence lengths

import numpy as np
from scipy.optimize import minimize
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def actual_score(b):
    """Compute the actual score: 2*n*max(conv(b,b))/sum(b)^2."""
    n = len(b)
    sum_b = np.sum(b)
    if sum_b < 0.01:
        return np.inf
    c = fft_convolve(b, b)
    return 2.0 * n * np.max(c) / (sum_b ** 2)


def lse_objective_and_grad(b, T):
    """
    Compute LogSumExp(T * max(conv(b,b))) and its gradient w.r.t. b.
    Returns (lse_value, gradient).
    """
    n = len(b)
    c = fft_convolve(b, b)  # length 2n-1
    c_max = np.max(c)
    
    # Softmax weights
    exp_shifted = np.exp((c - c_max) / T)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    
    # LSE value
    lse_val = T * (c_max / T + np.log(sum_exp))
    
    # Gradient: df/db_k = 2 * sum_j w[j+k] * b[j] for j=0..n-1
    # This is 2 * (w cross-corr b)[k]
    N = 1
    while N < (2 * n - 1) + n:
        N <<= 1
    W = np.fft.rfft(w, N)
    B = np.fft.rfft(b, N)
    cross = np.fft.irfft(np.conj(W) * B, N)[:n]
    
    grad = 2.0 * cross
    return lse_val, grad


def softmax_from_log(x):
    """Stable softmax from log-values."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / np.sum(e)


def make_asymmetric_ramp(n, rng, exponent=1.0, side='left'):
    """Asymmetric power-law ramp."""
    t = np.arange(n, dtype=np.float64)
    if side == 'left':
        base = (t + 1.0) ** exponent
    else:
        base = (n - t) ** exponent
    noise = 0.05 * rng.standard_normal(n)
    b = base + noise
    b = np.abs(b) + 0.01
    return b / np.sum(b)

    """Skewed Gaussian (asymmetric bell)."""
    t = np.arange(n, dtype=np.float64) / n
    center = rng.uniform(0.3, 0.7)
    width = rng.uniform(0.1, 0.3)
    skew = rng.uniform(0.5, 3.0)
    x = (t - center) / width
    b = np.exp(-0.5 * x * x * np.exp(skew * np.sign(x) * np.abs(x)))
    b += 0.03 * rng.standard_normal(n)
    b = np.abs(b) + 0.01
    return b / np.sum(b)


def make_bimodal_skew(n, rng):
    """Bimodal skewed distribution."""
    t = np.arange(n, dtype=np.float64) / n
    c1 = rng.uniform(0.1, 0.3)
    c2 = rng.uniform(0.6, 0.9)
    w1 = rng.uniform(0.05, 0.15)
    w2 = rng.uniform(0.05, 0.15)
    a1 = rng.uniform(0.5, 1.0)
    a2 = rng.uniform(0.5, 1.0)
    b = a1 * np.exp(-((t - c1) / w1) ** 2) + a2 * np.exp(-((t - c2) / w2) ** 2)
    b += 0.02 * rng.standard_normal(n)
    b = np.abs(b) + 0.01
    return b / np.sum(b)


def make_chirp_asymmetric(n, rng):
    """Asymmetric chirp."""
    t = np.arange(n, dtype=np.float64)
    f = rng.uniform(2.0, 8.0)
    chirp = 0.5 + 0.3 * np.cos(2 * np.pi * f * t / n + 0.5 * np.pi * t / n)
    ramp = 0.3 * (t / n) ** rng.uniform(0.5, 2.0)
    b = chirp + ramp
    b += 0.03 * rng.standard_normal(n)
    b = np.abs(b) + 0.01
    return b / np.sum(b)


def make_random_smooth(n, rng):
    """Random smooth positive sequence."""
    # Generate random Fourier coefficients for smoothness
    n_freq = max(3, n // 10)
    freqs = rng.integers(1, n_freq + 1, size=n_freq)
    amps = rng.uniform(0, 1, size=n_freq)
    phases = rng.uniform(0, 2 * np.pi, size=n_freq)
    t = np.arange(n, dtype=np.float64)
    b = np.ones(n)
    for k in range(n_freq):
        b += amps[k] * np.cos(2 * np.pi * freqs[k] * t / n + phases[k])
    b = np.abs(b) + 0.1
    return b / np.sum(b)


def optimize_logspace(b_init, T, maxiter=2000, time_limit=60.0):
    """
    Optimize in log-space using L-BFGS-B.
    b = softmax(x), minimize LSE(conv(b,b), T).
    """
    b_safe = np.maximum(b_init, 1e-15)
    x0 = np.log(b_safe / np.sum(b_safe))
    x0 -= np.max(x0)  # for numerical stability
    
    t_start = time.monotonic()
    result_holder = {'converged': False}
    
    def fun_grad(x):
        if time.monotonic() - t_start > time_limit:
            result_holder['converged'] = True
        b = softmax_from_log(x)
        lse_val, grad_b = lse_objective_and_grad(b, T)
        # Chain rule for softmax: grad_x_j = b_j * (grad_b_j - sum_k grad_b_k * b_k)
        inner = np.dot(grad_b, b)
        grad_x = b * (grad_b - inner)
        return lse_val, grad_x
    
    result = minimize(fun_grad, x0, method='L-BFGS-B', jac=True,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                               'maxls': 50})
    
    b_opt = softmax_from_log(result.x)
    return b_opt, result


def polish_local_search(b_init, time_limit, rng, verbose=False):
    """
    Targeted local search for peak reduction.
    Normalizes b to sum=1 and tries to reduce max(conv(b,b)).
    """
    current = b_init.copy()
    current = current / np.sum(current)
    current_score = 2.0 * len(current) * np.max(fft_convolve(current, current))
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    temp = 0.002
    cooling = 0.9995
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))
        
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.35:
            # Targeted peak reduction: identify top contributors to peak
            # c[peak_idx] = sum_i b[i] * b[peak_idx - i] (valid indices)
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            # Reduce top contributors
            reductions = rng.uniform(0.85, 0.98, size=len(top_idx))
            candidate[top_idx] *= reductions
            
            # Distribute the removed mass to less-contributing indices
            total_removed = 1.0  # in normalized terms, approximately
            other_mask = np.ones(n, dtype=bool)
            other_mask[top_idx] = False
            other_idx = np.where(other_mask)[0]
            if len(other_idx) > 0:
                n_inc = min(max(3, n // 20), len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                increments = rng.uniform(1.02, 1.10, size=n_inc)
                candidate[inc_idx] *= increments
                
        elif r < 0.55:
            # Asymmetric shift: push mass from one end to the other
            shift_amount = rng.uniform(0.02, 0.08)
            # Reduce left side, increase right side (or vice versa)
            if rng.random() < 0.5:
                half = n // 4
                candidate[:half] *= (1 - shift_amount)
                candidate[-half:] *= (1 + shift_amount * 0.5)
            else:
                half = n // 4
                candidate[-half:] *= (1 - shift_amount)
                candidate[:half] *= (1 + shift_amount * 0.5)
                
        elif r < 0.7:
            # Localized block perturbation
            width = max(2, n // 25)
            start = int(rng.integers(0, n - width))
            # Reduce one block
            candidate[start:start + width] *= np.exp(rng.normal(-0.05, 0.02, size=width))
            # Increase another block
            start2 = int(rng.integers(0, n - width))
            candidate[start2:start2 + width] *= np.exp(rng.normal(0.05, 0.02, size=width))
            
        elif r < 0.8:
            # Multi-scale perturbation
            for scale in [n // 4, n // 10, n // 30]:
                width = max(1, scale)
                start = int(rng.integers(0, max(1, n - width)))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))
                
        elif r < 0.9:
            # Global smooth perturbation
            freq = rng.integers(1, 5)
            t = np.arange(n, dtype=np.float64)
            perturbation = 0.03 * np.cos(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
            candidate *= np.exp(perturbation)
            
        else:
            # Mirrored perturbation (breaks symmetry)
            mirror = current[::-1].copy()
            mix = rng.uniform(0.1, 0.3)
            candidate = (1 - mix) * current + mix * mirror
            candidate *= np.exp(rng.normal(0, 0.01, size=n))
        
        candidate = np.maximum(candidate, 1e-10)
        candidate = candidate / np.sum(candidate)
        
        new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            stagnant = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if verbose and time.monotonic() - t0 > 5:
                    print(f"    polish: score={best_score:.6f}, t={time.monotonic()-t0:.1f}s")
        else:
            stagnant += 1
            if rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-10)):
                current = candidate
                current_score = new_score
        
        temp *= cooling
        if stagnant > 2000:
            # Restart with perturbation
            current = best * np.exp(rng.normal(0, 0.05, size=n))
            current = np.maximum(current, 1e-10)
            current = current / np.sum(current)
            current_score = 2.0 * n * np.max(fft_convolve(current, current))
            stagnant = 0
    
    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimum peak autocorrelation sequences.
    Phase 1: L-BFGS-B with LogSumExp annealing from diverse initializations
    Phase 2: Targeted local search polish
    Phase 3: Iterated refinement (L-BFGS-B + local search cycling)
    """
    t_global = time.monotonic()
    time_budget = max(120.0, float(budget_s) - 40.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Check for provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            start_seq = np.asarray(height_sequence_1, dtype=np.float64)
            start_seq = np.clip(start_seq, 0.0, 1000.0)
            if np.sum(start_seq) > 0.01:
                s = actual_score(start_seq)
                print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
                best_score = s
                best_seq = start_seq / np.sum(start_seq)
        except Exception as e:
            print(f"Error reading provided sequence: {e}")
    
    # Define initialization strategies
    def get_init(n, init_type, rng):
        if init_type == 'provided':
            if start_seq is not None:
                if len(start_seq) == n:
                    return start_seq / np.sum(start_seq)
                else:
                    x_old = np.linspace(0, 1, len(start_seq))
                    x_new = np.linspace(0, 1, n)
                    return np.interp(x_new, x_old, start_seq) / (np.interp(x_new, x_old, start_seq) / np.sum(np.interp(x_new, x_old, start_seq)) + 1e-15)
        elif init_type == 'ramp_left':
            return make_asymmetric_ramp(n, rng, exponent=rng.uniform(0.5, 2.5), side='left')
        elif init_type == 'ramp_right':
            return make_asymmetric_ramp(n, rng, exponent=rng.uniform(0.5, 2.5), side='right')
        elif init_type == 'skew_gauss':
            return make_skew_gaussian(n, rng)
        elif init_type == 'bimodal':
            return make_bimodal_skew(n, rng)
        elif init_type == 'chirp':
            return make_chirp_asymmetric(n, rng)
        elif init_type == 'smooth_random':
            return make_random_smooth(n, rng)
        elif init_type == 'uniform_perturb':
            b = np.ones(n) + 0.3 * rng.standard_normal(n)
            return np.abs(b) + 0.01 * rng.standard_normal(n) + 0.01
        else:
            b = np.abs(rng.standard_normal(n)) + 0.1
            return b / np.sum(b)
    
    # Phase 1: L-BFGS-B optimization
    print("\n=== Phase 1: L-BFGS-B with LogSumExp annealing ===")
    
    # (length, list of init types, time per init-temperature combo)
    search_configs = [
        (256, ['provided', 'ramp_left', 'ramp_right', 'skew_gauss', 'bimodal', 'chirp', 'smooth_random', 'uniform_perturb']),
        (384, ['ramp_left', 'ramp_right', 'skew_gauss', 'chirp', 'smooth_random']),
        (512, ['ramp_left', 'ramp_right', 'skew_gauss', 'chirp']),
        (768, ['ramp_left', 'skew_gauss']),
        (1024, ['ramp_left']),
    ]
    
    # Temperature schedule: start smooth, end sharp
    T_schedule = [100.0, 20.0, 5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1]
    
    total_inits = sum(len(inits) for _, inits in search_configs)
    avg_time_per = (deadline - t_global) / (total_inits * len(T_schedule) * 0.6)
    
    for n, init_types in search_configs:
        if time.monotonic() > deadline - 60:
            break
        
        for init_type in init_types:
            if time.monotonic() > deadline - 30:
                break
            
            b0 = get_init(n, init_type, rng)
            init_score = 2.0 * n * np.max(fft_convolve(b0, b0))
            
            # If this is a great start, skip to finer temperatures
            if init_score < 1.55:
                Ts_to_try = [10.0, 3.0, 1.0, 0.5, 0.3, 0.2, 0.1]
            elif init_score < 1.8:
                Ts_to_try = [50.0, 10.0, 3.0, 1.0, 0.5, 0.3, 0.1]
            else:
                Ts_to_try = T_schedule
            
            b_current = b0.copy()
            
            for T in Ts_to_try:
                if time.monotonic() > deadline - 20:
                    break
                
                t_remaining = deadline - time.monotonic()
                t_alloc = min(30.0, max(5.0, t_remaining / (len(Ts_to_try) + 1)))
                
                try:
                    b_opt, result = optimize_logspace(b_current, T, maxiter=500, time_limit=t_alloc)
                    opt_score = 2.0 * n * np.max(fft_convolve(b_opt, b_opt))
                    
                    if opt_score < best_score:
                        best_score = opt_score
                        best_seq = b_opt / np.sum(b_opt)
                        print(f"  n={n}, init='{init_type}', T={T:.1f}: score={opt_score:.6f} *** NEW BEST ***")
                    elif verbose if 'verbose' in dir() else False:
                        print(f"  n={n}, init='{init_type}', T={T:.1f}: score={opt_score:.6f}")
                    
                    b_current = b_opt
                except Exception as e:
                    print(f"  Error n={n}, init='{init_type}', T={T}: {e}")
                    break
        
        print(f"  [n={n} done] Best so far: {best_score:.6f}, time_left={deadline-time.monotonic():.0f}s")
    
    # Phase 2: Local search polish on best candidate
    print(f"\n=== Phase 2: Local search polish ===")
    if best_seq is not None:
        ptime = min(300.0, max(30.0, (deadline - time.monotonic()) * 0.4))
        print(f"  Starting polish: score={best_score:.6f}, time={ptime:.0f}s")
        
        polished, p_score = polish_local_search(best_seq, ptime, rng, verbose=True)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished
            print(f"  After polish: {best_score:.6f}")
        
        # Phase 3: Iterate L-BFGS-B + polish
        print(f"\n=== Phase 3: Iterated refinement ===")
        for iteration in range(3):
            if time.monotonic() > deadline - 60:
                break
            
            # L-BFGS-B from polished result
            t_lbg = min(60.0, (deadline - time.monotonic()) * 0.3)
            for T in [1.0, 0.5, 0.3, 0.1]:
                if time.monotonic() > deadline - 40:
                    break
                try:
                    b_opt, _ = optimize_logspace(best_seq, T, maxiter=300, time_limit=t_lbg / len([1.0,0.5,0.3,0.1]))
                    s = 2.0 * len(b_opt) * np.max(fft_convolve(b_opt, b_opt))
                    if s < best_score:
                        best_score = s
                        best_seq = b_opt / np.sum(b_opt)
                        print(f"  Iter {iteration}, T={T}: {best_score:.6f} *** NEW BEST ***")
                    best_seq = b_opt / np.sum(b_opt)
                except:
                    pass
            
            # Polish again
            if time.monotonic() < deadline - 30:
                ptime2 = min(120.0, deadline - time.monotonic() - 20)
                polished, p_score = polish_local_search(best_seq, ptime2, rng, verbose=False)
                if p_score < best_score:
                    best_score = p_score
                    best_seq = polished
                    print(f"  Iter {iteration}, after polish: {best_score:.6f}")
    
    # Phase 4: Final exploration with new random seeds
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 4: Final exploration ===")
        for trial in range(3):
            if time.monotonic() > deadline - 30:
                break
            n = int(rng.integers(200, 600))
            b0 = get_init(n, rng.choice(['ramp_left', 'skew_gauss', 'bimodal', 'chirp']), rng)
            try:
                b_opt, _ = optimize_logspace(b0, 10.0, maxiter=300, time_limit=20)
                b_opt, _ = optimize_logspace(b_opt, 1.0, maxiter=300, time_limit=15)
                b_opt, _ = optimize_logspace(b_opt, 0.3, maxiter=300, time_limit=10)
                s = 2.0 * n * np.max(fft_convolve(b_opt, b_opt))
                if s < best_score:
                    best_score = s
                    best_seq = b_opt / np.sum(b_opt)
                    print(f"  Phase4 trial {trial}, n={n}: {best_score:.6f} *** NEW BEST ***")
            except:
                pass
    
    # Phase 5: Final polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        ptime_final = max(10.0, deadline - time.monotonic() - 10)
        polished, p_score = polish_local_search(best_seq, ptime_final, rng, verbose=False)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished
            print(f"\n  Final polish: {best_score:.6f}")
    
    # Convert to output format
    if best_seq is not None:
        n = len(best_seq)
        # Scale for numerical stability (score is scale-invariant)
        target_sum = np.sqrt(2.0 * n)
        scale = target_sum / np.sum(best_seq)
        final = np.clip(best_seq * scale, 0.0, 1000.0)
        final_list = [float(x) for x in final]
        
        # Verify
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}, time_used={time.monotonic()-t_global:.1f}s")
        print(f"{'='*50}")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    
    return [1.0] * 256
