import numpy as np
import time

def _compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def _normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)

def _pseudogradient(a, active_set):
    """Compute pseudogradient for active set of lag indices."""
    n = len(a)
    g = np.zeros(n)
    for m in active_set:
        # g[j] += 2*a[m-j] for j where 0 <= m-j < n
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            g[j] += 2.0 * a[m - j]
    return g / len(active_set)

def _optimize_ppa(a0, deadline, max_iters=50000):
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a)
    
    S_t = np.sqrt(2.0 * n)
    best = a.copy()
    best_score = _compute_obj(a)
    
    # PPA parameters
    lr = 0.05
    epsilon = 0.1  # relative tolerance for active set
    min_epsilon = 0.001
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        
        # Compute autocorrelation
        c = np.convolve(a, a)
        max_c = c.max()
        
        # Active set: lags where c[k] >= max_c - epsilon * max_c
        threshold = max_c - epsilon * max_c
        active = np.where(c >= threshold)[0]
        
        # Compute pseudogradient
        g = _compute_pseudogradient(a, active)
        
        # Project out mean (for sum constraint)
        g = g - g.mean()
        
        # Line search (backtracking)
        alpha = lr
        a_new = a - alpha * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S_t)
        
        # Check improvement
        c_new = np.convolve(a_new, a_new)
        max_new = c_new.max()
        
        if max_new >= max_c:
            # Backtrack
            alpha *= 0.5
            a_new = a - alpha * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, S_t)
            c_new = np.convolve(a_new, a_new)
            max_new = c_new.max()
            if max_new >= max_c:
                # Reduce learning rate and epsilon
                lr *= 0.9
                epsilon = max(min_epsilon, epsilon * 0.95)
                continue
        
        a = a_new
        
        # Update best
        score = _compute_obj(a)
        if score < best_score:
            best_score = score
            best = a.copy()

        if it % 100 == 0:

        # Periodic learning rate adjustment
        if it % 500 == 0:
            lr = max(0.001, lr * 1.1)

PPA (Projected PseudogrAdient) based optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines adaptive active set PPA with perturbation-based restarts
and multi-length exploration.
"""

import numpy as np
import time

def _compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def _normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a

"""
PPA (Projected PseudogrAdient) optimization with adaptive active sets,
FFT-based autocorrelation, multi-phase annealing, and perturbation restarts.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses multi-scale strategy: wide active set → narrow → very narrow,
combined with multiple starting points of varying lengths.
"""

import numpy as np
import time


def _compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _autocorr(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    ac = np.fft.irfft(fa * fa.conj(), size)
    return ac[:2 * n - 1]


def _normalize(a, target=None):
    """Normalize sequence to target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def _ppa_step(a, active, lr):
    """One PPA step: compute pseudogradient, step, project."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    g = np.zeros(n)
    for m in active:
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[m - hi:m - lo]
    g /= max(1, len(active))
    g -= g.mean()
    a_new = a - lr * g
    a_new = np.maximum(a_new, 0.0)
    a_new = _normalize(a_new, S_t)
    return a_new
