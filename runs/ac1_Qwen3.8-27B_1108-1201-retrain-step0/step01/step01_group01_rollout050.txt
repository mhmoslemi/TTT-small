def project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Multi-stage softmax gradient descent for minimizing normalized autocorrelation peak.
Uses adaptive temperature continuation, FFT-based gradient, and peak chasing.

def project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1)
    return np.maximum(v - theta, 0.0)

        a += amp * np.cos(2*np.pi*f*x/n + phase)
    a = np.maximum(a, 0.01)
    inits.append(a / np.sum(a))
    
    return inits


def softmax_descent(a, deadline, t_init, t_max, eta_init, eta_min, rng, 
                    patience=300, verbose=False):
    """
    Softmax gradient descent with adaptive temperature.
    Minimize max(convolve(a,a)) subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    
    step = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Temperature schedule: exponential ramp
        t = min(t_init * np.exp(0.005 * step), t_max)
        eta = max(eta_init * np.exp(-0.002 * step), eta_min)
        
        # Compute convolution
        c = np.convolve(a, a)
        
        # Softmax weights
        c_max = np.max(c)
        exp_c = np.exp(t * (c - c_max))
        w = exp_c / np.sum(exp_c)
        
        # Gradient: 2 * correlate(w, a, 'valid')
        grad = 2.0 * np.correlate(w, a, mode='valid')
        
        # Center for sum constraint
        grad -= np.mean(grad)
        
        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Update
        a = a - eta * grad / gnorm
        
        # Project
        a = project_simplex(a, 1.0)
        
        # Evaluate
        c = np.convolve(a, a)
        score = 2.0 * n * np.max(c)
        
        if score < best_score - 1e-10:
            best_score = score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        step += 1
        
        if verbose and step % 500 == 0:
            print(f"    step={step}, score={best_score:.6f}, t={t:.1f}, eta={eta:.6f}")
        
        if no_improve >= patience:
            # Reheat: reset temperature and add perturbation
            step = 0
            no_improve = 0
            if rng is not None:
                noise = rng.normal(0, 0.02, n)
                a = a + noise
                a = project_simplex(a, 1.0)
    
    return best, best_score
