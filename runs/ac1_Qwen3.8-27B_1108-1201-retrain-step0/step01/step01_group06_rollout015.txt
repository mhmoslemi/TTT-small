"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a Gerchberg-Saxton inspired
iterative algorithm that flattens the power spectrum of the sequence.
Combines spectral domain magnitude shaping with non-negativity projection,
cross-validated with multiplicative mirror descent for final polish.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_score_normalized(a):
    """Compute 2n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def gs_flatten(a0, time_limit, gamma_start=0.3, gamma_end=0.05, 
               n_power_of_2=None, max_iter=None):
    """
    Gerchberg-Saxton style: iteratively flatten the power spectrum.
    a0: initial non-negative sequence (will be normalized to sum=1)
    Returns (best_a, best_obj_normalized)
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
        a_sum = 1.0
    else:
        a /= a_sum
    
    # FFT size
    N = 1 << (2 * n - 2).bit_length()
    
    best_obj = float('inf')
    best_a = a.copy()
    
    t0 = time.monotonic()
    iter_count = 0
    patience = 0
    max_patience = 200
    
    gamma = gamma_start
    
    while time.monotonic() - t0 < time_limit and (max_iter is None or iter_count < max_iter):
        iter_count += 1
        
        # Forward transform
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compute target magnitude for non-DC components
        # Total power = N * sum(a^2)
        # DC power = |A[0]|^2 = (sum a)^2 = 1 (since sum=1)
        # Non-DC power = N * sum(a^2) - 1
        # For flat non-DC: each non-DC bin has power (N*sum(a^2)-1)/(N-1)
        sum_a2 = np.sum(a * a)
        non_dc_power = N * sum_a2 - a_sum ** 2
        if non_dc_power > 0:
            target_mag_sq = non_dc_power / (N - 1)
            target_mag = np.sqrt(target_mag_sq)
        else:
            target_mag = 0.0
        
        # Blend magnitude toward target (keep DC fixed)
        new_mag = mag.copy()
        new_mag[1:] = (1.0 - gamma) * mag[1:] + gamma * target_mag
        
        # Reconstruct
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A, N))[:n]
        
        # Project onto non-negative
        a_new = np.maximum(a_new, 0.0)
        
        # Normalize
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-12:
            break
        a_new /= a_new_sum
        
        a = a_new
        
        # Evaluate
        obj = compute_score_normalized(a)
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        # Adapt gamma
        if patience >= max_patience:
            break
        
        # Decay gamma
        progress = min(1.0, iter_count / max(1, int(time_limit * 10)))
        gamma = gamma_start + (gamma_end - gamma_start) * progress
    
    return best_a, best_obj


def mirror_descent_polish(a0, time_limit, t_param=40.0, eta0=0.3):
    """Multiplicative mirror descent for fine-tuning."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a_sum
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    eta = eta0
    patience = 0
    max_patience = 100
    current_t = t_param
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Soft-max weights
        scaled = current_t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        thresh = w.max() * 0.005
        active_idx = np.where(w > thresh)[0]
        
        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]
        
        # Multiplicative update
        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.ones(n) / n
        else:
            a /= a_sum
        
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            patience = 0
            eta *= 0.7
            current_t *= 1.5
            if eta < 0.01:
                break
    
    return best_a, best_obj

This is a QCQP (quadratic constraint quadratic program). scipy doesn't directly support this, but we can use successive convex approximation:
- Fix a, linearize the quadratic constraints.
- Solve the resulting LP/QP.
- Update a.
- Repeat.

This is the "cutting plane" or "successive approximation" approach. At each iteration:
- Linearize: a[j]*a[k-j] ≈ a[j]*a_fixed[k-j] + a_fixed[j]*a[k-j] - a_fixed[j]*a_fixed[k-j]
- This gives a linear constraint: a[j]*a_fixed[k-j] + a_fixed[j]*a[k-j] - a_fixed[j]*a_fixed[k-j] <= t
- Plus the original linear constraints: sum(a) = 1, a >= 0.
- Solve the LP: minimize t subject to these constraints.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. Gerchberg-Saxton (GS) spectral flattening with proper target computation
2. Multiplicative mirror descent for fine-tuning
3. Multi-length, multi-initialization sweep with adaptive time allocation
The GS algorithm iteratively projects between the set of sequences with
target power spectra and the non-negative simplex, using relaxed projections
with adaptive blending.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def gs_flatten(a0, time_limit, gamma_start=0.4, gamma_end=0.05, use_proper_target=True):
    """
    Gerchberg-Saxton style spectral flattening.
    a0: initial non-negative sequence (normalized to sum=1).
    Returns (best_a, best_obj_normalized).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
        a_sum = 1.0
    else:
        a = a / a_sum

    N = 1 << (2 * n - 2).bit_length()

    best_obj = compute_obj_normalized(a)
    best_a = a.copy()

    t0 = time.monotonic()
    iter_count = 0
    patience = 0
    max_patience = 500
    gamma = gamma_start

    # Precompute target magnitude if using proper target
    if use_proper_target:
        M = 2 * n - 1
        c_flat = np.zeros(N)
        c_flat[:M] = 1.0 / M
        C_flat = np.fft.fft(c_flat)
        target_mag = np.sqrt(np.maximum(C_flat.real, 0.0))
        target_mag[0] = np.sqrt(M) * (1.0 / np.sqrt(M))  # = 1, will rescale
        # Actually target_mag[0] should be sum(a) = 1 for normalized a
        # The DC component of C_flat is sum(c_flat) = 1
        # So |A[0]|^2 = 1, meaning |A[0]| = 1 = sum(a). Good.
    else:
        target_mag = None

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        # Forward transform
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compute target magnitude
        if use_proper_target:
            # Scale target by current sum (which should be 1)
            tgt = target_mag * a_sum
        else:
            # Simple flat non-DC target
            sum_a2 = np.sum(a * a)
            non_dc_power = N * sum_a2 - a_sum ** 2
            if non_dc_power > 0:
                flat_tgt = np.sqrt(non_dc_power / (N - 1))
            else:
                flat_tgt = 0.0
            tgt = np.full(N, flat_tgt)
            tgt[0] = a_sum

        # Blend magnitude toward target
        new_mag = mag.copy()
        new_mag[1:] = (1.0 - gamma) * mag[1:] + gamma * tgt[1:]
        new_mag[0] = a_sum  # Keep DC fixed

        # Inverse transform
        new_A = new_mag * np.exp(1j * phase)
        x = np.real(np.fft.ifft(new_A, N))

        # Project: keep first n samples, clip to non-negative
        a_new = np.maximum(x[:n], 0.0)

        # Normalize
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-12:
            break
        a = a_new / a_new_sum

        # Evaluate
        obj = compute_obj_normalized(a)

        if obj < best_obj - 1e-11:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Adapt gamma (cooling schedule)
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(1.0, time_limit))
        gamma = gamma_start + (gamma_end - gamma_start) * (progress ** 0.7)

        if patience >= max_patience:
            break

    return best_a, best_obj


def mirror_descent(a0, time_limit, t_param=40.0, eta0=0.4):
    """
    Multiplicative mirror descent on the probability simplex.
    a0 should be non-negative (will be normalized to sum=1).
    Returns (best_a, best_obj_normalized).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / a_sum

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    patience = 0
    max_patience = 120
    current_t = t_param
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Soft-max weights over convolution peaks
        scaled = current_t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum

        # Compute weighted subgradient
        sg = np.zeros(n)
        thresh = w.max() * 0.003
        active_idx = np.where(w > thresh)[0]

        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]

        # Multiplicative update with centering
        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.ones(n) / n
        else:
            a = a / a_sum

        # Evaluate
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()

        if current_obj < best_obj - 1e-11:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            patience = 0
            eta *= 0.65
            current_t *= 1.4
            if eta < 0.005:
                current_t = min(current_t * 2.0, 500.0)
                eta = 0.2
                max_patience = 80
            if eta < 0.002:
                break

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    candidates = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    candidates.append(np.ones(n) / n)

    # Gaussians with various widths
    for sf in [0.08, 0.12, 0.18, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - n / 2.0) / (sf * n)) ** 2)
        g = np.maximum(g, 1e-10)
        g /= g.sum()
        candidates.append(g)

    # Inverted triangle (valley in middle)
    tri = 1.0 - 2.0 * np.abs(x - (n - 1) / 2.0) / max(1, (n - 1))
    tri = np.maximum(tri, 0.01)
    tri /= tri.sum()
    candidates.append(tri)

    # Bimodal (peaks at edges)
    bi = np.exp(-0.5 * (x / (0.08 * n)) ** 2) + np.exp(-0.5 * ((x - (n - 1)) / (0.08 * n)) ** 2)
    bi = np.maximum(bi, 1e-10)
    bi /= bi.sum()
    candidates.append(bi)

    # Cosine bump
    cos = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    cos = np.maximum(cos, 0.01)
    cos /= cos.sum()
    candidates.append(cos)

    # Multi-peak
    for num_peaks in [3, 4, 6]:
        centers = np.linspace(n * 0.1, n * 0.9, num_peaks)
        w = np.zeros(n)
        for c in centers:
            w += np.exp(-0.5 * ((x - c) / (n / (2.0 * num_peaks))) ** 2)
        w = np.maximum(w, 0.001)
        w /= w.sum()
        candidates.append(w)

    # Sine modulation
    for freq in [2, 3, 5, 8]:
        s = 1.0 + 0.4 * np.sin(2.0 * np.pi * freq * x / n)
        s = np.maximum(s, 0.01)
        s /= s.sum()
        candidates.append(s)

    # Power law (various)
    for p in [0.3, 0.5, 2.0, 5.0]:
        pw = (x + 1) ** p
        pw /= pw.sum()
        candidates.append(pw)
        pw_r = pw[::-1].copy()
        candidates.append(pw_r)

    # Random Dirichlet
    for alpha in [0.3, 1.0, 3.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        candidates.append(r)

    # Random with structure: sum of random Gaussians
    for _ in range(2):
        w = np.ones(n) * 0.1
        num_bumps = rng.integers(2, 6)
        for _ in range(num_bumps):
            center = rng.uniform(0, n)
            width = rng.uniform(0.05 * n, 0.2 * n)
            amp = rng.uniform(0.5, 2.0)
            w += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
        w = np.maximum(w, 0.0)
        if w.sum() > 0:
            w /= w.sum()
            candidates.append(w)

    # From parent
    if parent is not None and parent.size > 0:
        if parent.size == n:
            pp = np.maximum(parent, 0.0)
            if pp.sum() > 1e-10:
                pp = pp / pp.sum()
                candidates.append(pp)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                pp = pp / pp.sum()
                candidates.append(pp)

    return candidates
