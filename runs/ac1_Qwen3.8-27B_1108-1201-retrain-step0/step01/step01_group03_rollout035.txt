```python
import cvxpy as cp

def build_hankel(w, n):
    """Build the Hankel matrix Q from weights w (length 2n-1)."""
    # Q[i][j] = w[i+j] * (1 if i==j else 1/2)
    # Actually: Q[i][j] = w[i+j] for the "Hankel" part, with diagonal correction
    # Let's build it as: F[i][j] = w[i+j], then Q = (F + diag_correction) / 2
    # where diag_correction[i][i] = w[2i]
    
    # F is a Hankel matrix with first column w[0:n] and last row w[n-1:2n-1]
    F = np.zeros((n, n))
    for i in range(n):
        F[i, :] = w[i:i+n]
    # Diagonal correction: add w[2i] to Q[i][i], which means adding w[2i] to F[i][i] (since Q = (F + D)/2 and we want Q[i][i] = w[2i] = (F[i][i] + D[i][i])/2 = (w[2i] + D[i][i])/2, so D[i][i] = w[2i])
    # So Q = (F + diag(w[0], w[2], ..., w[2n-2])) / 2
    D = np.diag(w[0:2*n-1:2])  # w[0], w[2], w[4], ..., w[2n-2]
    Q = (F + D) / 2.0
    return Q

def solve_qp(Q, n, S):
    """Solve: min a^T Q a s.t. sum(a) = S, a >= 0."""
    a_var = cp.Variable(n)
    constraints = [a_var >= 0, cp.sum(a_var) == S]
    obj = cp.quad_form(a_var, Q, assume_PSD=True)
    prob = cp.Problem(cp.Minimize(obj), constraints)
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
    except:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            prob.solve(solver=cp.SCS, verbose=False)
    if a_var.value is not None:
        return np.maximum(a_var.value, 0)
    return None
```

```python
def build_hankel(w, n):
    """Build Hankel matrix Q with Q[i][j] = w[i+j]."""
    # w has length 2n-1
    Q = np.zeros((n, n))
    for i in range(n):
        Q[i, :] = w[i:i+n]
    return Q


def solve_qp_cvxpy(Q, n, S=1.0):
    """Solve: min a^T Q a s.t. sum(a) = S, a >= 0."""
    a_var = cp.Variable(n)
    constraints = [a_var >= 0, cp.sum(a_var) == S]
    obj = cp.quad_form(a_var, cp.psd_wrap(Q))
    prob = cp.Problem(cp.Minimize(obj), constraints)
    
    # Try solvers in order of preference
    for solver in [cp.MOSEK, cp.ECOS2, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if a_var.value is not None:
                return np.maximum(np.asarray(a_var.value).flatten(), 0)
        except Exception:
            continue
    return None
```

Actually, for a QP with a single equality constraint and box constraints, there's a very efficient approach:

1. If we ignore the non-negativity, the solution is:
   a* = (Q^{-1} 1) / (1^T Q^{-1} 1) * S
   
   (This is the standard solution to min a^T Q a s.t. 1^T a = S.)

2. If some components of a* are negative, we set them to 0 and solve the reduced problem.

This is the active-set method for QPs. For n=512, each iteration involves solving a linear system of size (n - |active|) × (n - |active|). With a good initial active set guess, this converges in few iterations.

Let me implement this:

```python
def solve_qp_active_set(Q, n, S=1.0, max_iters=100):
    """
    Solve: min a^T Q a s.t. sum(a) = S, a >= 0 using active set.
    """
    a = np.ones(n) * S / n  # initial guess (uniform)
    active = np.zeros(n, dtype=bool)  # which variables are at 0
    
    for it in range(max_iters):
        # Solve equality-constrained QP on free variables
        free = ~active
        n_free = free.sum()
        if n_free == 0:
            break
        if n_free == 1:
            idx = np.where(free)[0][0]
            a_free = np.array([S])
        else:
            Q_ff = Q[np.ix_(free, free)]
            # Solve: 2*Q_ff*a_free + lambda*1 = 0, sum(a_free) = S
            # => a_free = Q_ff^{-1} * 1 * (S / (1^T Q_ff^{-1} 1))
            try:
                Q_ff_inv_1 = np.linalg.solve(Q_ff, np.ones(n_free))
                denom = Q_ff_inv_1.sum()
                if abs(denom) < 1e-15:
                    break
                a_free = Q_ff_inv_1 * (S / denom)
            except np.linalg.LinAlgError:
                break
        
        # Assemble full solution
        a_new = np.zeros(n)
        a_new[free] = a_free
        
        # Check KKT: for active variables, the multiplier should be >= 0
        # Multiplier for variable i: mu_i = (2*Q@a)[i] + lambda
        # where lambda is the Lagrange multiplier for the equality constraint
        # From KKT: 2*Q*a + lambda*1 - mu = 0
        # For free variables: 2*(Q*a)[i] + lambda = 0 => lambda = -2*(Q*a)[i] (for any free i)
        # For active variables: mu_i = 2*(Q*a)[i] + lambda = 2*(Q*a)[i] - 2*(Q*a)[free_i]
        
        Qa = Q @ a_new
        if n_free >= 2:
            # lambda = -2 * Qa[free][0] (approximately, should be same for all free)
            lambda_val = -2 * Qa[free][0]
            # Actually, for the equality-constrained QP, the KKT gives:
            # 2*Q_ff*a_free = -lambda*1_free => lambda = -2*(Q_ff@a_free)[0]
            # But Q_ff@a_free = Q@a_new (restricted to free) ... let me just compute it
            # For the active set to be valid: mu_i >= 0 for all active i
            # mu_i = 2*(Q*a_new)[i] + lambda for active i
            # We need 2*(Q*a_new)[i] + lambda >= 0 for all active i
            # lambda = -2*(Q*a_new)[free][0] (from the free variable condition)
            
            for i in np.where(active)[0]:
                mu_i = 2 * Qa[i] + lambda_val
                if mu_i < -1e-10:
                    # This active variable should be freed
                    active[i] = False
                    break
            else:
                # Also check: are any free variables < 0?
                if a_free.min() < -1e-10:
                    # The most negative free variable should become active
                    idx = np.where(free)[0][int(a_free.argmin())]
                    active[idx] = True
                    a_new[idx] = 0
                    # Recompute
                    continue
                # KKT satisfied
                return np.maximum(a_new, 0)
        else:
            # Only one free variable, check if it's >= 0
            if a_free[0] >= 0:
                return np.maximum(a_new, 0)
            else:
                break
    
    # Fallback: project onto the constraint set
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a * S / s
    return a
```

```python
"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses iterative reweighted quadratic programming: at each step, builds a Hankel
matrix from softmax-weighted convolution peaks and solves the resulting convex QP.
Combined with local search polish and multiple starting points.
"""

import numpy as np
import time
import cvxpy as cp


def build_hankel(w, n):
    """Build Hankel matrix Q with Q[i][j] = w[i+j]. w must have length 2n-1."""
    Q = np.zeros((n, n))
    for i in range(n):
        Q[i, :] = w[i:i + n]
    return Q


def solve_qp_cvxpy(Q, n):
    """Solve: min a^T Q a s.t. sum(a)=1, a>=0. Returns normalized a or None."""
    a_var = cp.Variable(n)
    constraints = [a_var >= 0, cp.sum(a_var) == 1.0]
    obj = cp.quad_form(a_var, cp.psd_wrap(Q))
    prob = cp.Problem(cp.Minimize(obj), constraints)
    solvers = [cp.MOSEK, cp.ECOS2, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                val = a_var.value
                if val is not None:
                    val = np.maximum(np.asarray(val).flatten(), 0.0)
                    S = val.sum()
                    if S > 1e-12:
                        return val / S
        except Exception:
            continue
    return None


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = np.convolve(a, a)
    S = float(a.sum())
    if S < 1e-12:
        return float('inf')
    return float(2.0 * n * float(c.max()) / (S * S))


def qp_optimize(a0, deadline, max_outer, max_inner, beta_lo, beta_hi):
    """
    Iterative reweighted QP optimization.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_score = compute_score(a)

    if max_outer == 1:
        betas = [beta_lo]
    else:
        betas = np.geomspace(beta_lo, beta_hi, max_outer)

    for oi in range(max_outer):
        if time.monotonic() > deadline - 30.0:
            break
        beta = betas[oi]

        for ii in range(max_inner):
            if time.monotonic() > deadline - 30.0:
                break

            conv = np.convolve(a, a)
            conv_shift = conv - conv.max()
            w = np.exp(beta * conv_shift)
            ws = w.sum()
            if ws < 1e-300:
                w = np.zeros(len(conv))
                w[int(np.argmax(conv))] = 1.0
            else:
                w = w / ws

            Q = build_hankel(w, n)
            Q += 1e-14 * np.eye(n)

            a_new = solve_qp_cvxpy(Q, n)
            if a_new is not None:
                a = a_new
                s = compute_score(a)
                if s < best_score:
                    best_score = s
                    best_a = a.copy()

        if (oi + 1) % 5 == 0:
            print(f"      outer={oi+1}/{max_outer}, beta={beta:.1f}, best={best_score:.6f}")

    return best_a, best_score


def local_polish(a, deadline, max_iters, seed_val):
    """Random local search to polish a sequence."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(seed_val)
    step = 0.05
    n_accept = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 15.0:
            break

        a_t = a.copy()
        k = int(rng.integers(1, min(6, n)))
        idx = rng.integers(0, n, size=k)
        a_t[idx] += rng.normal(0.0, step, size=k)
        a_t = np.maximum(a_t, 0.0)
        st = a_t.sum()
        if st < 1e-12:
            continue
        a_t = a_t / st

        s = compute_score(a_t)
        if s < best_score:
            best_score = s
            best_a = a_t.copy()
            a = a_t
            n_accept += 1
            step = max(0.0005, step * 0.998)
        elif it % 3000 == 0 and n_accept < 3000:
            step = max(0.0005, step * 0.9)

    return best_a, best_score


def coordinate_descent(a, deadline, max_sweeps=200):
    """
    Coordinate descent: optimize one element at a time.
    For fixing all but a[i], the objective max_t (a*a)[t] is piecewise quadratic
    in a[i]. We find the optimal a[i] via ternary search on [0, upper].
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return a, compute_score(a)
    a = a / S

    best_a = a.copy()
    best_score = compute_score(a)

    def eval_single(a_arr, i, val):
        """Compute score with a[i] replaced by val, renormalizing."""
        a_try = a_arr.copy()
        a_try[i] = val
        a_try = np.maximum(a_try, 0.0)
        st = a_try.sum()
        if st < 1e-12:
            return float('inf')
        a_try = a_try / st
        c = np.convolve(a_try, a_try)
        return float(2.0 * n * c.max() / st**2) * (st / a_arr.sum())**2

    # Simpler: just do the full evaluation
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline - 20.0:
            break
        improved = False
        for i in range(n):
            # Ternary search for optimal a[i]
            # Range: [0, 3 * a[i]] (generous upper bound)
            lo = 0.0
            hi = max(3.0 * a[i], 0.01)

            # Sample the objective on a grid first
            grid = np.linspace(lo, hi, 20)
            scores = []
            for gv in grid:
                a_try = a.copy()
                a_try[i] = gv
                s_try = compute_score(a_try)
                scores.append(s_try)
            best_idx = int(np.argmin(scores))
            
            # Refine around best grid point
            gl = grid[max(0, best_idx - 1)]
            gr = grid[min(len(grid) - 1, best_idx + 1)]
            for _ in range(30):
                m1 = gl + (gr - gl) / 3.0
                m2 = gr - (gr - gl) / 3.0
                a_try1 = a.copy(); a_try1[i] = m1
                a_try2 = a.copy(); a_try2[i] = m2
                s1 = compute_score(a_try1)
                s2 = compute_score(a_try2)
                if s1 < s2:
                    gr = m2
                else:
                    gl = m1

            optimal_val = (gl + gr) / 2.0
            a_try = a.copy()
            a_try[i] = optimal_val
            s_try = compute_score(a_try)
            if s_try < best_score - 1e-12:
                a = a_try
                best_score = s_try
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_score
```
