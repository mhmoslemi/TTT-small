1. Compute `conv(a,a)` using FFT: O(n log n).
2. Find the active set S (indices where conv is close to max): O(n).
3. Compute the average gradient: For each k in S, the gradient is `2 * (M_k a)`. The sum over the active set can be computed efficiently.
4. Take a gradient step and project: O(n log n) for the projection (sorting).

Total per iteration: O(n log n) (dominated by FFT and sorting).

Minimize peak auto-convolution of non-negative sequences using projected gradient descent
with FFT-based convolution, multiple initializations at varying lengths, and LP-based polishing.
The key insight is that asymmetric sequences can achieve lower peak convolution than uniform ones
because the convolution peak shifts off-center where fewer terms contribute.

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def convolve_full(a):
    """Compute full convolution using FFT for O(n log n) speed."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:size]
    return np.maximum(conv, 0)  # Remove tiny negative values from FFT roundoff


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def score_normalized(a):
    """Score assuming sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    conv = convolve_full(a)
    return 2.0 * n * conv.max()


def score_sequence(a):

    configs = [
        # If we have a previous solution, try to improve it
        (500, 'prev', {}),
        # Larger sequences with various inits
        (500, 'ramp', {'p': 2.0}),
        (500, 'ramp', {'p': 3.0}),
        (500, 'exp_ramp', {'rate': 3.0}),
        (800, 'ramp', {'p': 2.0}),
        (800, 'exp_ramp', {'rate': 2.5}),
        (800, 'bump', {'center_frac': 0.3, 'width_frac': 0.25}),
        (1200, 'ramp', {'p': 2.0}),
        (1200, 'exp_ramp', {'rate': 2.0}),
        (1200, 'two_bumps', {}),
        (2000, 'ramp', {'p': 1.5}),
        (2000, 'exp_ramp', {'rate': 1.5}),
        (2000, 'bump', {'center_frac': 0.25, 'width_frac': 0.3}),
        (3000, 'ramp', {'p': 1.5}),
        (3000, 'exp_ramp', {'rate': 1.5}),
        (5000, 'ramp', {'p': 1.0}),
        (5000, 'exp_ramp', {'rate': 1.0}),
    ]
    
    # Assign time fractions (more time for larger n and later configs)
    n_configs = len(configs)
    # Weight: more time for configs that are likely to be productive
    weights = []
    for i, (n, init_type, _) in enumerate(configs):
        if init_type == 'prev' and have_prev:
            weights.append(3.0)
        elif n <= 800:
            weights.append(1.0)
        elif n <= 2000:
            weights.append(1.5)
        else:
            weights.append(2.0)
    total_weight = sum(weights)
    
    cfg_idx = 0
    for n, init_type, params in configs:
        remaining = deadline - time.monotonic()
        if remaining < 15.0:
            break
        
        time_frac = weights[cfg_idx] / total_weight
        time_for_config = min(time_frac * (budget_s - 300.0), remaining * 0.9)
        time_for_config = max(10.0, time_for_config)
        
        # Skip 'prev' if no previous solution
        if init_type == 'prev' and not have_prev:
            cfg_idx += 1
            continue
        
        print(f"\n=== n={n}, init={init_type}, time={time_for_config:.0f}s, "
              f"remaining_total={remaining:.0f}s ===")
        
        # Generate initialization
        if init_type == 'prev':
            a = prev.copy()
            if len(a) != n:
                # Resample to new length
                idx = np.linspace(0, len(a)-1, n)
                a = np.interp(np.arange(n), idx, a)
        elif init_type == 'ramp':
            a = make_init_ramp(n, p=params.get('p', 2.0))
        elif init_type == 'exp_ramp':
            a = make_init_exp_ramp(n, rate=params.get('rate', 2.0))
        elif init_type == 'bump':
            a = make_init_bump(n,
                              center_frac=params.get('center_frac', 0.3),
                              width_frac=params.get('width_frac', 0.25))
        elif init_type == 'two_bumps':
            a = make_init_two_bumps(n)
        elif init_type == 'raised_cosine':
            a = make_init_raised_cosine(n)
        else:
            a = make_init_raised_cosine(n)
        
        a = project_simplex(a)
        init_score = score_normalized(a)
        print(f"    Init score: {init_score:.6f}")
        
        # Phase 1: Gradient descent (fast)
        gd_time = time_for_config * 0.7
        a_gd, score_gd = gradient_descent_smooth(
            a, time_limit=gd_time, beta=80.0, verbose=False, seed=seed+cfg_idx)
        
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"  *** NEW BEST (GD-soft): score={best_score:.6f}, n={n}")
        
        # Also try hard max GD briefly
        remaining_here = time.monotonic() + time_for_config - time.monotonic()
        hard_time = min(time_for_config * 0.15, max(5.0, time_for_config - gd_time - 10))
        if hard_time > 5:
            a_h, score_h = gradient_descent_hard(
                a_gd, time_limit=hard_time, verbose=False, seed=seed+100+cfg_idx)
            if score_h < best_score:
                best_score = score_h
                best_seq = a_h.copy()
                print(f"  *** NEW BEST (GD-hard): score={best_score:.6f}, n={len(a_h)}")
        
        cfg_idx += 1
    
    # Print current best
    print(f"\n=== After GD phase: best={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0} ===")
    
    # Phase 2: LP polish on the best candidate
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            lp_time = min(remaining * 0.5, 300.0)
            print(f"\n=== LP polish (time={lp_time:.0f}s) ===")
            a_lp, score_lp = lp_refine(best_seq, time_limit_s=lp_time, verbose=True)
            if score_lp < best_score:
                best_score = score_lp
                best_seq = a_lp.copy()
                print(f"  *** NEW BEST (LP): score={best_score:.6f}")
    
    # Phase 3: Local search polish
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            ls_time = min(remaining * 0.4, 60.0)
            print(f"\n=== Local search polish (time={ls_time:.0f}s) ===")
            a_ls, score_ls = local_search_polish(best_seq, time_limit_s=ls_time, verbose=True)
            if score_ls < best_score:
                best_score = score_ls
                best_seq = a_ls.copy()
                print(f"  *** NEW BEST (LS): score={best_score:.6f}")
    
    # Phase 4: Another GD round on the polished sequence with different params
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            gd_time2 = min(remaining * 0.5, 120.0)
            print(f"\n=== Final GD round (time={gd_time2:.0f}s) ===")
            a_f, score_f = gradient_descent_smooth(
                best_seq, time_limit=gd_time2, beta=150.0, verbose=False, seed=999)
            if score_f < best_score:
                best_score = score_f
                best_seq = a_f.copy()
                print(f"  *** NEW BEST (Final GD): score={best_score:.6f}")
    
    # Fallback
    if best_seq is None:
        best_seq = make_init_raised_cosine(500)
        best_seq = project_simplex(best_seq)
        best_score = score_normalized(best_seq)
        print(f"  Fallback: score={best_score:.6f}")
    
    # Final verification
    print(f"\nFinal: score={best_score:.6f}, n={len(best_seq)}")
    verify = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified via evaluate_sequence: {verify:.6f}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")
    
    return [float(x) for x in best_seq]
