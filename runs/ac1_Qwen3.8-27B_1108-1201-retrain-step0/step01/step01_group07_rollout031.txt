def fft_convolve(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    L = 1
    while L < 2*n - 1:
        L *= 2
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)
    return c[:2*n-1]

def fft_correlate_valid(q, a):
    """Compute correlate(q, a, 'valid') using FFT.
    q has length 2n-1, a has length n. Result has length n.

import numpy as np
import time

def get_fft_size(n):
    """Get next power of 2 >= 2n-1."""
    L = 1
    need = 2*n - 1
    while L < need:
        L *= 2
    return L

def fft_convolve_square(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    L = get_fft_size(n)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)
    return c[:2*n-1]

def fft_gradient(a, q):
    """Compute gradient of sum_k q[k]*conv(a,a)[k] w.r.t. a.
    grad[j] = 2 * sum_i a[i]*q[i+j] - q[2j]*a[j]
    """
    n = len(a)
    # First term: 2 * correlate(q, a, 'valid')
    # q has length 2n-1, a has length n
    L = 1
    while L < 3*n - 2:
        L *= 2
    Q = np.fft.rfft(q, L)
    A = np.fft.rfft(a, L)
    corr = np.fft.irfft(Q * np.conj(A), L)
    # Valid part: starts at index n-1, length n
    corr_valid = corr[n-1:2*n-1]
    term1 = 2.0 * corr_valid
    term2 = q[0::2] * a  # q[2j] * a[j] for j=0,...,n-1
    return term1 - term2

def project_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = a.size
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, a.max()
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    res[res.argmax()] += S - res.sum()
    return res

def compute_score(a):
    """Compute the objective function value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_square(a)
    return 2.0 * n * c.max() / (S * S)

def optimize_adam(a0, S, tau0, tau_min, lr, t_budget, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimization with tau annealing and simplex projection."""
    n = a0.size
    a = (a0.astype(np.float64) / max(a0.sum(), 1e-12) * S).copy()
    a = project_simplex(a, S)
    
    c = fft_convolve_square(a)
    best_mc = c.max()
    best_a = a.copy()
    
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < t_budget:
        t += 1
        # Anneal tau
        f = min(1.0, (time.monotonic() - t0) / t_budget)
        tau = tau0 * (tau_min / tau0) ** f
        
        # Compute soft-max weights
        c = fft_convolve_square(a)
        c_shift = c - c.max()
        exp_c = np.exp(c_shift / tau)
        q = exp_c / exp_c.sum()
        
        # Compute gradient
        g = fft_gradient(a, q)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new, S)
        
        # Check improvement every 5 steps
        if t % 5 == 0:
            mc = fft_convolve_square(a_new).max()
            if mc < best_mc:
                best_mc = mc
                best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_mc

def fft_conv_square(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = a.size

    if prov is not None:
        sc_prov = evaluate_sequence(prov.tolist())
        print(f"Provided sequence: n={prov.size}, score={sc_prov:.6f}")
        bs = prov.copy()
        bsc = sc_prov

    # Phase 1: Exploration with various n and initializations
    # Time allocation: 20% of total
    p1_frac = 0.20
    p1_time = (deadline - time.monotonic()) * p1_frac
    p1_end = time.monotonic() + p1_time

    # Candidate configurations: (n, init_type, seed)
    configs = []
    if prov is not None:
        configs.append((len(prov), 'provided', 0))

        configs.append((n, 'chirp', 0))

        configs.append((n, 'multisine', 7))
        configs.append((n, 'gaussian', 3))

    # Deduplicate by (n, init_type)
    seen = set()
    unique_configs = []
    for cfg in configs:
        key = (cfg[0], cfg[1])
        if key not in seen:
            seen.add(key)
            unique_configs.append(cfg)
    configs = unique_configs

    t_per = max(10.0, (p1_end - time.monotonic()) / max(1, len(configs)))
    print(f"\nPhase 1: {len(configs)} configs, ~{t_per:.0f}s each")

    for i, (n, itype, sd) in enumerate(configs):
        tb = p1_end - time.monotonic()
        if tb < 8:
            break
        tb = min(t_per, tb)

        if itype == 'provided' and prov is not None:
            if len(prov) == n:
                a0 = prov.copy()
            else:
                x_old = np.linspace(0.0, 1.0, len(prov))
                x_new = np.linspace(0.0, 1.0, n)
                a0 = np.maximum(np.interp(x_new, x_old, prov), 0.0)
        elif itype == 'chirp':
            a0 = init_chirp(n, sd)
        elif itype == 'windowed_chirp':
            a0 = init_windowed_chirp(n, sd)
        elif itype == 'multisine':
            a0 = init_multisine(n, sd)
        elif itype == 'gaussian':
            a0 = init_gaussian(n, sd)
        else:
            a0 = init_chirp(n, sd)

        S = np.sqrt(2.0 * n)
        print(f"  P1[{i+1}/{len(configs)}] n={n} {itype}_{sd} t={tb:.0f}s", end=" ", flush=True)
        a_r, mc = optimize_adam(a0, S, tb, tau_start=1.0, tau_end=0.05,
                                 lr_start=0.01, lr_end=0.001, verbose=False)
        sc = evaluate_sequence(a_r.tolist())
        print(f"-> mc={mc:.6f}, score={sc:.6f}", flush=True)

        if sc < bsc:
            bsc = sc
            bs = a_r.copy()
            print(f"  *** NEW BEST: {bsc:.6f} (n={n})", flush=True)

    # Phase 2: Focused optimization of best candidate
    rem = deadline - time.monotonic()
    if rem > 60 and bs is not None:
        n_best = bs.size
        S_best = np.sqrt(2.0 * n_best)
        print(f"\nPhase 2: Focused optimization n={n_best}, t={rem:.0f}s")

        # Multiple rounds with decreasing tau
        rounds = [
            # (tau_start, tau_end, lr_start, lr_end, time_frac)
            (0.5, 0.02, 0.005, 0.0005, 0.35),
            (0.2, 0.01, 0.003, 0.0003, 0.30),
            (0.1, 0.005, 0.002, 0.0002, 0.20),
            (0.05, 0.002, 0.001, 0.0001, 0.15),
        ]

        for idx, (ts, te, lrs, lre, frac) in enumerate(rounds):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            tb = rem * frac
            print(f"  Round {idx+1}: tau=[{ts},{te}] lr=[{lrs},{lre}] t={tb:.0f}s",
                  end=" ", flush=True)
            a_r, mc = optimize_adam(bs, S_best, tb, tau_start=ts, tau_end=te,
                                     lr_start=lrs, lr_end=lre, verbose=True)
            sc = evaluate_sequence(a_r.tolist())
            print(f"-> mc={mc:.6f}, score={sc:.6f}", flush=True)
            if sc < bsc:
                bsc = sc
                bs = a_r.copy()
                print(f"  *** improved to {bsc:.6f}", flush=True)

    # Phase 3: Try a different length for the best solution (interpolation)
    rem = deadline - time.monotonic()
    if rem > 120 and bs is not None:
        print(f"\nPhase 3: Trying different lengths, t={rem:.0f}s")
        n_best = bs.size
        for n_try in [n_best * 2, n_best + 1000]:
            if n_try > 20000:
                continue
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            tb = min(rem * 0.4, 300.0)
            S_try = np.sqrt(2.0 * n_try)
            # Interpolate to new length
            x_old = np.linspace(0.0, 1.0, n_best)
            x_new = np.linspace(0.0, 1.0, n_try)
            a0 = np.maximum(np.interp(x_new, x_old, bs), 0.0)
            print(f"  n={n_try}: optimizing for {tb:.0f}s", end=" ", flush=True)
            a_r, mc = optimize_adam(a0, S_try, tb, tau_start=0.3, tau_end=0.01,
                                     lr_start=0.005, lr_end=0.0005, verbose=False)
            sc = evaluate_sequence(a_r.tolist())
            print(f"-> score={sc:.6f}", flush=True)
            if sc < bsc:
                bsc = sc
                bs = a_r.copy()
                print(f"  *** improved to {bsc:.6f} (n={n_try})", flush=True)
