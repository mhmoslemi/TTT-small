def grad_bk(a, k):
    """Gradient of b_k w.r.t. a."""

    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)

def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""

    n = len(v)
    if n == 0:
        return v

    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]

        return np.full(n, target / n)

"""
Hybrid multiplicative descent with adaptive peak selection and local refinement
to minimize the normalized autocorrelation peak of a non-negative sequence.
"""

import numpy as np
import time


def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) for a sequence with sum=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _grad_bk(a, k):
    """Gradient of b_k = sum_i a_i*a_{k-i} w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _make_two_level(n, eps):
    """Create an anti-symmetric two-level sequence."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _make_smooth_profile(n, freq, eps, phase=0.0):
    """Create a smooth profile: 1 + eps*cos(2*pi*freq*i/n + phase)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_multi_freq(n, freqs, amps, phases):
    """Create a multi-frequency profile."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for f, amp, ph in zip(freqs, amps, phases):
        a += amp * np.cos(2.0 * np.pi * f * i / n + ph)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def _perturb(a, rng, scale=0.05):
    """Add smooth random perturbation."""
    n = len(a)
    # Low-frequency perturbation
    freqs = rng.integers(1, max(2, n // 8), size=min(3, n // 2))

"""
FFT-accelerated hybrid optimization (multiplicative + additive descent) with
adaptive peak selection, line search, and local refinement to minimize the
normalized autocorrelation peak of a non-negative sequence. Uses FFT-based
convolution for ~100x speedup over direct convolution, enabling far more
iterations with smaller step sizes for finer convergence.

import numpy as np
import time


def _fft_convolve(a):
    """Compute convolve(a, a) using FFT for speed."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_from_c(c, n):
    """Compute score from convolution."""
    return 2.0 * n * np.max(c)


def _grad_weighted(a, k_idx, wts):
    """Compute weighted sum of gradients for given peak locations."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(k_idx, wts):
        if w < 1e-14:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[k - idx]
    return grad


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _make_two_level(n, eps):
    """Create an anti-symmetric two-level sequence."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _make_smooth(n, freq, eps, phase=0.0):
    """Create a smooth cosine profile."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_plateau(n, frac, h):
    """Create a plateau sequence: high in the middle, low at edges."""
    a = np.ones(n)
    m = int(n * frac)
    center = n // 2
    lo = max(0, center - m // 2)
    hi = min(n, center + m // 2 + 1)
    a[lo:hi] = h
    S = np.sum(a)
    return a / S


def _perturb(a, rng, scale=0.03):
    """Add smooth random perturbation."""
    n = len(a)
    n_freqs = min(4, n // 2)
    freqs = rng.integers(1, max(2, n // 6), size=n_freqs)
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2.0 * np.pi * f * np.arange(n) / n + phase)
    max_abs = np.max(np.abs(noise))
    if max_abs > 1e-15:
        noise /= max_abs
    a_new = a + scale * noise
    a_new = _project_simplex(a_new, 1.0)
    return a_new


def _optimize(a, deadline, rng, method="mult", M_start=1, M_end=50,
              eta_init=None, cool_rate=0.99, min_eta=1e-12,
              reheat_every=200, temp=5.0, use_line_search=True,
              ls_factor=0.7):
    """
    Main optimization loop using FFT-accelerated convolution.
    method: 'mult' for multiplicative, 'add' for additive with projection.
    """
    n = len(a)
    if n == 0:
        return a, np.inf

    if eta_init is None:
        eta_init = 1.0 / np.sqrt(n)

    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n

    best = a.copy()
    best_score = _score_from_c(_fft_convolve(best), n)
    eta = eta_init
    no_improve = 0
    iter_count = 0
    n_improvements = 0

    while time.monotonic() < deadline:
        # Adaptive M
        progress = min(1.0, iter_count / 2000.0)
        M = int(M_start + (M_end - M_start) * progress)
        M = min(M, 2 * n - 1)

        c = _fft_convolve(a)
        c_max = np.max(c)
        cur_score = 2.0 * n * c_max

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
            n_improvements += 1
        else:
            no_improve += 1

        if no_improve >= reheat_every:
            eta *= cool_rate
            no_improve = 0
            if eta < min_eta:
                break

        # Get top M peaks with softmax weighting
        if M >= len(c):
            k_idx = np.arange(len(c))
            wts = np.ones(len(c)) / len(c)
        else:
            k_idx = np.argpartition(c, -M)[-M:]
            c_sel = c[k_idx]
            wts = np.exp(-temp * (c_max - c_sel) / max(c_max, 1e-15))
            w_sum = np.sum(wts)
            if w_sum < 1e-30:
                break
            wts /= w_sum

        # Compute weighted gradient
        grad = _grad_weighted(a, k_idx, wts)

        # Center gradient for sum constraint
        grad_mean = np.mean(grad)
        grad -= grad_mean

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        grad_normed = grad / grad_norm

        if method == "mult":
            # Multiplicative update
            log_update = -eta * grad_normed
            log_update = np.clip(log_update, -50.0, 50.0)
            a_new = a * np.exp(log_update)
            a_new[a_new < 1e-18] = 0.0
            S_new = np.sum(a_new)
            if S_new < 1e-15:
                break
            a_new /= S_new
        else:
            # Additive update with projection

            a_new = _project_simplex(a_new, 1.0)

            c_new = _fft_convolve(a_new)
            score_new = 2.0 * n * np.max(c_new)
            if score_new < cur_score - 1e-14:
                a = a_new
            else:
                eta *= ls_factor
                continue
        else:
            a = a_new

        iter_count += 1

    return best, best_score


def _local_refine(a, deadline, rng):
    """Local refinement with very small steps focused on the single peak."""
    n = len(a)
    best = a.copy()
    c = _fft_convolve(best)
    best_score = _score_from_c(c, n)
    eta = 0.01 / np.sqrt(n)

    while time.monotonic() < deadline and eta > 1e-14:
        c = _fft_convolve(best)
        c_max = np.max(c)
        k_star = int(np.argmax(c))

        # Gradient of the peak
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] = 2.0 * best[k_star - idx]

        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Try small additive step
        candidate = best - eta * grad / gnorm
        candidate = _project_simplex(candidate, 1.0)
        c_cand = _fft_convolve(candidate)
        score_cand = _score_from_c(c_cand, n)

        if score_cand < best_score - 1e-14:
            best = candidate
            best_score = score_cand
        else:
            # Try multiplicative
            log_upd = np.clip(-eta * grad / gnorm, -5, 5)
            candidate2 = best * np.exp(log_upd)
            candidate2[candidate2 < 1e-18] = 0.0
            S2 = np.sum(candidate2)
            if S2 > 1e-15:
                candidate2 /= S2
                c_cand2 = _fft_convolve(candidate2)
                score_cand2 = _score_from_c(c_cand2, n)
                if score_cand2 < best_score - 1e-14:
                    best = candidate2
                    best_score = score_cand2
                else:
                    eta *= 0.5
                    continue
            else:
                eta *= 0.5
                continue

        for d in range(n_div):
            if time.monotonic() >= deadline:
                break
            scale = 0.02 + 0.03 * (d % 3)
            a_pert = _perturb(global_best, rng, scale=scale)
            d_deadline = min(time.monotonic() + per_div, deadline)
            method = "mult" if d % 2 == 0 else "add"
            best_a_d, best_s_d = _optimize(
                a_pert, d_deadline, rng, method=method,
                M_start=5, M_end=40,
                eta_init=0.5 / np.sqrt(n),
                cool_rate=0.99, min_eta=1e-12,
                reheat_every=150, temp=5.0,
                use_line_search=True, ls_factor=0.75
            )
            _update_best(best_a_d, best_s_d, f"diverge {d+1}")

    # Final
    if global_best is None:
        global_best = a0

    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"\nFINAL SCORE: {final_score:.6f}")
    return result
