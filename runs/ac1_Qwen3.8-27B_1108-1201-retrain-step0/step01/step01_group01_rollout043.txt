import time

def _fft_convolve(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()  # Next power of 2 >= 2n-1
    fa = np.fft.rfft(a, n=size)
    return np.fft.irfft(fa * fa, n=size)[:2*n-1]

def _fft_cross_correlate(a, w):
    """
    Compute S[i] = sum_{j=0}^{n-1} a[j] * w[j+i] for i = 0, ..., n-1.
    a has length n, w has length 2n-1.
    """
    n = len(a)
    w_rev = w[::-1]
    size = 1 << (n + 2*n - 2).bit_length()  # Next power of 2 >= 3n-2
    conv = np.fft.irfft(np.fft.rfft(a, n=size) * np.fft.rfft(w_rev, n=size), n=size)
    return conv[n-1 : 2*n-1 : -1]

def _score(a):
    """Compute the score 2n*max_b/sum_a^2."""
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    b = _fft_convolve(a)
    return 2.0 * len(a) * np.max(b) / (S * S)

def _subgradient_polish(a, deadline, M=10, verbose=True):
    """
    Subgradient polishing with top-M peak weighting and adaptive step size.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    a = a / S
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(_fft_convolve(a))
    
    eta = 0.5 / np.sqrt(n)
    n_iter = 0
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() < deadline:
        b = _fft_convolve(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                eta *= 0.7
                no_improve = 0
                if eta < 1e-10:
                    break
        
        # Top-M peak indices
        if M >= len(b):
            k_idx = np.arange(len(b))
            wts = np.ones(len(b)) / len(b)
        else:
            k_idx = np.argpartition(b, -M)[-M:]
            b_sel = b[k_idx]
            # Weight by proximity to max
            wts = np.exp(-10.0 * (b_max - b_sel) / max(b_max, 1e-15))
            wts /= np.sum(wts)
        
        # Compute weighted gradient
        # g_i = sum_k wts[k] * 2 * a[k-i] (with valid range)
        # This is 2 * cross_correlate(a, weighted_w)
        w_full = np.zeros(2*n-1)
        w_full[k_idx] = wts
        S_vals = _fft_cross_correlate(a, w_full)
        g = 2.0 * S_vals
        
        # Center gradient (for sum constraint)
        g -= np.mean(g)
        
        # Euclidean projection step
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        a = a - eta * g / g_norm
        a = np.maximum(a, 0)
        a_sum = np.sum(a)
        if a_sum < 1e-15:
            a = best_a.copy()
            break
        a = a / a_sum
        
        n_iter += 1
    
    return best_a, best_score, n_iter

def _fft_convolve_self(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, n=size)
    return np.fft.irfft(fa * fa, n=size)[:2 * n - 1]


def _fft_grad_autocorr(a, w):
    """
    Compute g_i = 2 * sum_{j=0}^{n-1} a[j] * w[j+i] for i = 0..n-1.
    a has length n, w has length 2n-1 (weights over convolution indices).
    """
    n = len(a)
    w_rev = w[::-1]
    size = 1 << (3 * n - 2).bit_length()
    fa = np.fft.rfft(a, n=size)
    fw = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(fa * fw, n=size)
    return 2.0 * conv[n - 1:2 * n - 1:-1]


def _true_score(a):
    """Compute 2n * max(b) for normalized a (sum=1)."""
    n = len(a)
    b = _fft_convolve_self(a)
    return 2.0 * n * np.max(b)


def _obj_and_grad_x(x, T):
    """
    Smoothed objective and gradient in x-space (a = softmax(x)).
    Returns (f, grad_x).
    """
    # Softmax
    x_s = x - np.max(x)
    a = np.exp(x_s)
    S = np.sum(a)
    a = a / S
    n = len(a)
    
    # Convolution
    b = _fft_convolve_self(a)
    
    # Log-sum-exp
    Tb = T * b
    m = np.max(Tb)
    lse = m + np.log(np.sum(np.exp(Tb - m)))
    f = lse / T
    
    # Weights
    w = np.exp(Tb - m)
    w /= np.sum(w)
    
    # Gradient wrt a
    g_a = _fft_grad_autocorr(a, w)
    
    # Gradient wrt x via chain rule
    # d(softmax(x))_j / d(x_i) = a_j * (delta_ij - a_i)
    # df/dx_i = sum_j (df/da_j) * a_j * (delta_ij - a_i)
    #         = a_i * (g_a_i - sum_j a_j * g_a_j)
    weighted_mean = np.dot(a, g_a)
    grad_x = a * (g_a - weighted_mean)
    
    return f, grad_x

    Run L-BFGS-B optimization starting from a_init with smoothing parameter T.

    x0 = x0 - np.max(x0)  # Center for numerical stability
    
    t_start = time.monotonic()
    
    def callback(xk):
        if time.monotonic() - t_start > time_limit:
            raise TimeoutError("Time limit exceeded")
        return False
    
    try:
        result = scipy_minimize(
            _obj_and_grad_x,
            x0,
            args=(T,),
            jac=True,
            method='L-BFGS-B',

        a_out /= np.sum(a_out)
        return a_out, False
    
    x_opt = result.x
    a_out = np.exp(x_opt - np.max(x_opt))
    a_out /= np.sum(a_out)
    return a_out, result.success


def _mirror_descent_step(a, T, eta):
    """Single mirror descent step."""
    n = len(a)
    b = _fft_convolve_self(a)
    Tb = T * b
    m = np.max(Tb)
    w = np.exp(Tb - m)
    w /= np.sum(w)
    g = _fft_grad_autocorr(a, w)

    log_a_new -= np.max(log_a_new)
    a_new = np.exp(log_a_new)
    a_new /= np.sum(a_new)
    return a_new, b

    Returns (a_optimized, converged_flag).
    """
    a_safe = np.maximum(a_init, 1e-30)
    x0 = np.log(a_safe)
    x0 = x0 - np.max(x0)

    t_start = time.monotonic()
    aborted = [False]

    def _callback(xk):
        if time.monotonic() - t_start > time_limit:
            aborted[0] = True
            return True
        return False

    try:
        result = scipy_minimize(
            _obj_and_grad_x,
            x0,
            args=(T,),
            jac=True,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'maxfun': maxiter,
                     'ftol': 1e-14, 'gtol': 1e-9, 'callback': _callback}
        )
        x_opt = result.x
    except Exception:
        x_opt = x0

    a_out = np.exp(x_opt - np.max(x_opt))
    a_out /= np.sum(a_out)
    return a_out, (not aborted[0])


def _mirror_descent_run(a_init, deadline, T_start=20.0, T_end=500.0, eta_init=0.4):
    """Mirror descent with temperature annealing. Returns (best_a, best_score, n_iter)."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _true_score(best_a)

    t0 = time.monotonic()
    total_time = max(1.0, deadline - t0)
    eta = eta_init
    n_iter = 0

    while time.monotonic() < deadline:
        frac = min(1.0, (time.monotonic() - t0) / total_time)
        T = T_start + (T_end - T_start) * frac

        # Compute b and gradient
        b = _fft_convolve_self(a)
        Tb = T * b
        m = np.max(Tb)
        w = np.exp(Tb - m)
        w /= np.sum(w)
        g = _fft_grad_autocorr(a, w)

        # Mirror descent update in log space
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_sum = np.sum(a_new)
        if a_sum < 1e-15:
            break
        a = a_new / a_sum

        cur_score = 2.0 * n * np.max(b)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        eta *= 0.9998
        n_iter += 1

    return best_a, best_score, n_iter


def make_two_level(n, eps):
    """Anti-symmetric two-level sequence, normalized to sum=1."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    return a / np.sum(a)


def make_cosine(n, eps, freq=1):
    """Smooth cosine-modulated sequence."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * i / n)
    a = np.maximum(a, 0.0)
    return a / np.sum(a)


def make_ramp(n, ratio=0.6):
    """Linear ramp from (1+ratio) to (1-ratio)."""
    a = np.linspace(1.0 + ratio, 1.0 - ratio, n)
    a = np.maximum(a, 0.0)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer: L-BFGS-B on smoothed objective + mirror descent polish.
    Minimizes 2n*max(self-conv) / sum^2 over non-negative sequences.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    # Get initial sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, None)
    except NameError:
        parent = np.ones(6121)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size
    a_best = parent / np.sum(parent)

    best_score = _true_score(a_best)
    print(f"n={n}, Initial score: {best_score:.6f}")
    print(f"Effective time budget: {budget_s - 15:.0f}s")

    # === Phase 1: L-BFGS-B with annealed T (70% of time) ===
    print("\n=== Phase 1: L-BFGS-B with T-annealing ===")
    phase1_end = time.monotonic() + (deadline - time.monotonic()) * 0.70

    # Build diverse starting points
    starts = [a_best.copy()]
    for eps in [0.3, 0.5, 0.7, 0.9]:
        starts.append(make_two_level(n, eps))
    for eps, freq in [(0.5, 1), (0.7, 1), (0.5, 2), (0.8, 1)]:
        starts.append(make_cosine(n, eps, freq))
    starts.append(make_ramp(n, 0.5))
    starts.append(make_ramp(n, 0.8))

    # T schedule: multiple passes with increasing sharpness
    T_schedule = []
    for eps_list in [0.3, 0.5, 0.7]:
        for T in [10, 30, 100, 300]:
            T_schedule.append((T, eps_list))
    # Additional passes on best
    for T in [100, 300, 1000, 3000]:
        T_schedule.append((T, 0.0))  # No perturbation

    run = 0
    best_start_idx = 0

    for T, noise_scale in T_schedule:
        if time.monotonic() >= phase1_end:
            break

        # Choose starting point
        if run == 0:
            a_start = a_best.copy()
        elif noise_scale > 0 and rng.random() < 0.25:
            a_pert = a_best + noise_scale * 0.01 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert /= np.sum(a_pert)
            a_start = a_pert
        else:
            # Cycle through diverse starts for early runs
            if run < len(starts):
                a_start = starts[run].copy()
            else:
                a_start = a_best.copy()

        # Time limit for this run
        remaining = phase1_end - time.monotonic()
        n_remaining_runs = len(T_schedule) - run
        per_run = max(8.0, min(50.0, remaining / max(1, n_remaining_runs)))

        a_new, converged = _run_lbfgs(a_start, T, maxiter=250, time_limit=per_run)
        new_score = _true_score(a_new)

        if new_score < best_score:
            a_best = a_new.copy()
            best_score = new_score
            print(f"  [{run:3d}] T={T:4d} start={run if run < len(starts) else 'best'}: "
                  f"{best_score:.6f} *** IMPROVED ***")
        elif run % 8 == 0:
            print(f"  [{run:3d}] T={T:4d}: {new_score:.6f} (best={best_score:.6f})")

        run += 1

    # === Phase 2: Mirror descent polishing (15% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 2: Mirror Descent Polish (score={best_score:.6f}) ===")
        phase2_end = time.monotonic() + remaining * 0.50

        a_md, score_md, iters = _mirror_descent_run(
            a_best, phase2_end, T_start=50.0, T_end=800.0, eta_init=0.3
        )
        if score_md < best_score:
            a_best = a_md.copy()
            best_score = score_md
            print(f"  Mirror descent: {best_score:.6f} ({iters} iters) *** IMPROVED ***")
        else:
            print(f"  Mirror descent: {score_md:.6f} (no improvement, {iters} iters)")

    # === Phase 3: Final high-T L-BFGS-B runs (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 3: Final L-BFGS-B Refinement (score={best_score:.6f}) ===")
        for T in [500, 1000, 2000, 5000]:
            if time.monotonic() >= deadline - 5:
                break
            a_new, _ = _run_lbfgs(a_best, T, maxiter=150, time_limit=remaining * 0.2)
            new_score = _true_score(a_new)
            if new_score < best_score:
                a_best = a_new.copy()
                best_score = new_score
                print(f"  T={T:5d}: {best_score:.6f} *** IMPROVED ***")
            else:
                print(f"  T={T:5d}: {new_score:.6f}")

    # === Final: One more mirror descent pass if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Final Mirror Pass (score={best_score:.6f}) ===")
        a_md, score_md, iters = _mirror_descent_run(
            a_best, deadline - 5, T_start=200.0, T_end=2000.0, eta_init=0.1
        )
        if score_md < best_score:
            a_best = a_md.copy()
            best_score = score_md
            print(f"  Final mirror: {best_score:.6f} ({iters} iters) *** IMPROVED ***")

    # Final evaluation
    result = [float(max(0.0, x)) for x in a_best]
    final_score = float(evaluate_sequence(result))
    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {final_score:.6f}")
    print(f"Target:      1.5027")
    print(f"Previous:    1.514844")
    print(f"{'=' * 50}")

    return result
