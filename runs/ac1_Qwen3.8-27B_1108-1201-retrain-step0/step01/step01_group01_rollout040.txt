Let me think about the "soft gradient" idea. Instead of using just the top peak, use:
g = sum_k softmax(b_k / T) * grad_b_k

where T is a temperature parameter. As T → 0, this concentrates on the max. As T is large, it spreads out.

Using a moderate T gives a smoother gradient that can help escape local minima.

Another idea: use the "smoothing" approach:
f_T(a) = T * log(sum_k exp(b_k/T))

This is a smooth approximation to max_k b_k. Its gradient is:
∇f_T = sum_k (exp(b_k/T) / sum_j exp(b_j/T)) * ∇b_k

This is the softmax-weighted average of gradients. As T → 0, it approaches the subgradient of the max.

The advantage of using f_T instead of max is that it's differentiable everywhere, so we can use proper gradient descent methods (like L-BFGS) instead of subgradient methods.

But for n=6121, b has 2n-1 = 12241 elements. Computing the softmax over all of them is fine (it's just a vector operation). The expensive part is computing ∇b_k for the relevant k's.

Actually, for the gradient of f_T, we need:
∇f_T = sum_k p_k * ∇b_k where p_k = softmax(b/T)

And ∇b_k is a vector of length n. So the full gradient is a weighted sum of (2n-1) vectors of length n, which is O(n^2) work. That's the same as computing the convolution.

But we can be smarter: only the top-K peaks have significant softmax weights (for small T). So we can approximate:
∇f_T ≈ sum_{k ∈ top-K} p_k * ∇b_k

This is O(K*n) work for the gradient, which is fast if K is small (like 10-100).

"""
FFT-accelerated subgradient optimization with Adam, soft gradients,
and simulated annealing to minimize 2n*max(conv(a,a))/(sum(a))^2
for non-negative sequences.
"""
import numpy as np
import time
from scipy.fft import rfft, irfft

def _fft_self_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def _grad_bk(a, k):
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return grad
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    if k % 2 == 0:
        i_diag = k // 2
        if lo <= i_diag <= hi:
            grad[i_diag] = a[i_diag]
    return grad

def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_score(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    b = _fft_self_conv(a)
    return 2.0 * n * np.max(b) / (S * S)

def _adam_optimize(a, deadline, rng, T_init=0.01, T_final=0.001, lr=0.005, top_k=50):
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    a /= S
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_conv(a))
    
    total_time = deadline - time.monotonic()
    
    while time.monotonic() < deadline:
        t += 1
        b = _fft_self_conv(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max
        
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        
        # Anneal temperature
        frac = min(1.0, t / max(1, int(0.8 * total_time / 0.001)))  # This is rough
        # Better: use exponential annealing
        elapsed = time.monotonic() - (deadline - total_time)
        progress = elapsed / total_time
        T = T_init * (T_final / T_init) ** progress
        
        # Compute softmax weights
        # Use top-k for efficiency
        if top_k >= len(b):
            k_idx = np.arange(len(b))
            b_sel = b
        else:
            k_idx = np.argpartition(b, -top_k)[-top_k:]
            b_sel = b[k_idx]
        
        # Softmax weights
        b_shifted = b_sel - np.max(b_sel)
        w = np.exp(b_shifted / T)
        w /= np.sum(w)
        
        # Compute gradient
        grad = np.zeros(n)
        for i in range(len(k_idx)):
            if w[i] < 1e-10:
                continue
            grad += w[i] * _grad_bk(a, k_idx[i])
        
        # Center gradient (for sum constraint)
        grad -= np.mean(grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex
        a = _project_simplex(a, 1.0)
        
        if t % 10000 == 0:
            print(f"  Iter {t}: score={cur_score:.6f}, best={best_score:.6f}, T={T:.6f}")
    
    return best_a, best_score

FFT-accelerated optimization with soft gradients and adaptive step size
to minimize 2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""
import numpy as np
import time
from scipy.fft import rfft, irfft


def _fft_self_conv(a):
    """Compute full self-convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)[:size]
    # Clip tiny negative values from FFT numerical errors
    return np.maximum(result, 0.0)


def _grad_bk(a, k):
    """Gradient of b_k = (a*a)_k with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return grad
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    if k % 2 == 0:
        i_diag = k // 2
        if lo <= i_diag <= hi:
            grad[i_diag] = a[i_diag]
    return grad


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if
