```python
def solve_qp_active_set(Q, n, max_active_iters=50):
    """Solve min a^T Q a s.t. a >= 0, sum(a) = 1 using active set method."""
    free = np.ones(n, dtype=bool)  # indices not fixed at 0
    for _ in range(max_active_iters):
        Q_ff = Q[np.ix_(free, free)]
        try:
            Q_inv = np.linalg.inv(Q_ff)
        except:
            Q_inv = np.linalg.pinv(Q_ff)
        v = Q_inv @ np.ones(free.sum())
        denom = v.sum()
        if denom <= 0:
            break
        a_free = v / denom
        # Check if all free variables are positive
        if np.all(a_free > -1e-10):
            a = np.zeros(n)
            a[free] = np.maximum(0, a_free)
            return a / a.sum()
        # Find the most negative variable and fix it at 0
        min_idx = np.argmin(a_free)
        free[np.where(free)[0][min_idx]] = False
    # Fallback
    a = np.zeros(n)
    a[free] = 1.0 / free.sum()
    return a
```

```python
def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def build_hankel(alpha, n):
    """Build Hankel matrix Q with Q[i,j] = alpha[i+j]."""
    # alpha has length 2n-1, indices 0..2n-2
    i = np.arange(n)[:, None]
    j = np.arange(n)[None, :]
    return alpha[i + j]


def solve_qp_hankel(alpha, n, max_active=100):
    """
    Solve min a^T Q a s.t. a >= 0, sum(a) = 1
    where Q[i,j] = alpha[i+j].
    Uses active set method.
    """
    free = np.ones(n, dtype=bool)
    a = np.ones(n) / n  # fallback
    
    for _ in range(max_active):
        n_free = int(free.sum())
        if n_free == 0:
            break
        free_idx = np.where(free)[0]
        # Build sub-Hankel: Q[i,j] = alpha[free_idx[i] + free_idx[j]]
        Q_sub = alpha[free_idx[:, None] + free_idx[None, :]]
        
        try:
            Q_inv = np.linalg.inv(Q_sub)
        except np.linalg.LinAlgError:
            Q_inv = np.linalg.pinv(Q_sub)
        
        v = Q_inv @ np.ones(n_free)
        denom = v.sum()
        if denom <= 1e-15:
            break
        a_free = v / denom
        
        if np.all(a_free > -1e-12):
            a = np.zeros(n)
            a[free_idx] = np.maximum(0, a_free)
            s = a.sum()
            if s > 1e-15:
                a /= s
            return a
        
        # Fix the most negative variable
        min_local = np.argmin(a_free)
        free[free_idx[min_local]] = False
    
    # Fallback: uniform on free set
    a = np.zeros(n)
    if any(free):
        a[free] = 1.0 / free.sum()
    else:
        a = np.ones(n) / n
    return a


def proj_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def hankel_matvec(alpha, v, n):
    """Compute Q @ v where Q[i,j] = alpha[i+j], using FFT."""
    # Q @ v has entries (Qv)_i = sum_j alpha[i+j] v_j
    # Use convolution: (Qv)_i = (alpha * v_rev)_{i+n-1} where v_rev = v[::-1]
    v_rev = v[::-1]
    # alpha has length 2n-1, v_rev has length n
    # convolution has length (2n-1) + n - 1 = 3n-2
    conv = np.convolve(alpha, v_rev, mode='full')
    # (Qv)_i = conv[i + n - 1] for i = 0..n-1
    return conv[n-1:n-1+n]


def minimax_solve(a, time_limit_s=100, n_inner_steps=1, verbose=True):

    """
    start_time = time.monotonic()

    a = normalize_nonneg(a.copy().astype(np.float64))
    
    # Initialize dual variable
    b = np.convolve(a, a)
    alpha = b.copy()
    alpha = np.maximum(alpha, 1e-15)
    alpha /= alpha.sum()
    
    best_a = a.copy()
    best_val = b.max()
    
    T = 0
    for T in range(100000):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        # Solve QP: min a'^T Q_alpha a' s.t. a' >= 0, sum(a') = 1
        a_new = solve_qp_hankel(alpha, n)
        
        # Update dual: multiplicative weights
        b_new = np.convolve(a_new, a_new)
        # Smoothed update
        temp = max(1.0, 2.0 / (1.0 + 0.01 * T))
        alpha_new = alpha * np.exp(temp * b_new / max(b_new.max(), 1e-15))
        alpha_new = np.maximum(alpha_new, 1e-15)
        alpha_new /= alpha_new.sum()
        alpha = alpha_new
        
        # Primal update: move towards a_new

        t_step = 2.0 / (2.0 + T)
        a = (1 - t_step) * a + t_step * a_new
        a = normalize_nonneg(a)
        
        # Track best
```

In the minimax game:
- Player 1 (min) plays $a_t$
- Player 2 (max) plays $\alpha_t$
- Payoff to Player 2: $\sum_k \alpha_t(k) (a_t * a_t)_k$

For Player 2 (the maximizer), MWU gives:
$\alpha_{t+1}(k) \propto \alpha_t(k) \cdot \exp(\eta \cdot (a_t * a_t)_k)$

For Player 1 (the minimizer), the best response to $\alpha_t$ is:
$a_t = \arg\min_{a \in \Delta_n} \sum_k \alpha_t(k) (a*a)_k = \arg\min_{a \in \Delta_n} a^T Q_{\alpha_t} a$

The convergence rate of MWU is $O(1/\sqrt{T})$ in the regret, which translates to an $O(1/\sqrt{T})$ gap in the minimax value.

After $T$ iterations:
- $\bar{\alpha} = \frac{1}{T}\sum_{t=1}^T \alpha_t$ satisfies: $\max_{a} \sum_k \bar{\alpha}_k (a*a)_k \leq V^* + O(1/\sqrt{T})$ where $V^*$ is the game value.
- $\bar{a} = \frac{1}{T}\sum_{t=1}^T a_t$ satisfies: $\max_k (\bar{a}*\bar{a})_k \leq V^* + O(1/\sqrt{T})$.

But in practice, the convergence is often much faster, especially with the "best response" inner solve.

Actually, let me just use a simpler approach: at each step, set $\alpha$ proportional to $(a * a)$ (the current convolution values). This is a "greedy" dual update that should work well in practice.

$\alpha_t(k) = \frac{(a_t * a_t)_k}{\sum_j (a_t * a_t)_j}$

This is equivalent to setting the dual to the "maximizing distribution" that's spread according to the current primal solution. It's not exactly MWU but it's simpler and often works well.

Minimax QP optimization for non-negative sequence peak auto-convolution minimization.
Uses alternating best responses in the zero-sum game:
  min_{a in Delta} max_k (a*a)_k = max_{alpha in Delta} min_{a in Delta} a^T Q_alpha a
The inner QP is solved via active-set method on Hankel-structured matrices.
