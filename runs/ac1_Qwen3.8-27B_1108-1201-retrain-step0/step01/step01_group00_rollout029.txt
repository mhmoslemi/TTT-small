```python
def subgradient_step(p, lr):
    """One subgradient step for min max(conv(p,p)) s.t. p in simplex."""
    c = convolve(p, p)
    m = argmax(c)
    # Gradient of c[m] w.r.t. p
    g = np.zeros(len(p))
    for k in range(len(p)):
        g[k] = 0
        if 0 <= m - k < len(p):
            g[k] += p[m - k]
        if 0 <= k - m < len(p):
            g[k] += p[k - m]
    # Gradient step
    p_new = p - lr * g
    # Project to simplex
    p_new = project_to_simplex(p_new)
    return p_new
```

```python
def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
```

```python
"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions. Uses gradient descent with top-K peak tracking, simulated
annealing local search, and fine-tuning with exact subgradient.
"""

import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def compute_gradient_topk(p, K, tau):
    """
    Compute gradient of smoothed max of conv(p,p) using top-K peaks.
    Returns (value, gradient, convolution).
    """
    n = len(p)
    c = fft_convolve(p, p)
    
    # Get top-K values
    if K >= len(c):
        # Use all with softmax
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        exp_s = np.exp(shifted)
        w = exp_s / np.sum(exp_s)
    else:
        # Use top-K with softmax
        top_k_vals = np.sort(c)[-K:]
        threshold = top_k_vals[0]
        # Weight: softmax over the top-K values (only those above threshold get weight)
        # Actually, let's use: for values above threshold, weight = exp((val-threshold)/tau)
        # For values below, weight = 0
        mask = c >= threshold
        if not np.any(mask):
            mask = c >= np.max(c)
        shifted = (c[mask] - np.max(c)) / tau
        exp_s = np.exp(shifted)
        w = np.zeros(len(c))
        w[mask] = exp_s / np.sum(exp_s)
    
    value = np.dot(w, c)  # weighted average as smooth approximation of max
    
    # Gradient: g[j] = sum_m w[m] * (p[m-j] + p[j-m]) with bounds
    # Efficient computation via FFT
    p_rev = p[::-1]
    corr = fft_convolve(p_rev, w)
    g = 2.0 * corr[n-1:2*n-1]
    
    return value, g, c, w

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def optimize_gradient_v2(n, p_init, time_limit, rng, K=10):
    """
    Gradient descent on probability vector p to minimize max(conv(p,p)).
    Uses top-K peak tracking with annealed temperature.
    """
    p = np.maximum(p_init, 0).astype(np.float64)
    p = p / np.sum(p)
    
    S = np.sqrt(2.0 * n)
    best_p = p.copy()
    best_val = np.max(fft_convolve(p, p))
    
    # Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0
    
    # Temperature schedule
    tau_start = 2.0 / np.max(fft_convolve(p, p))
    tau_end = 0.05 / np.max(fft_convolve(p, p))
    
    t0 = time.monotonic()
    no_improve = 0
    plateau_best = best_val
    best_lr = 0.5
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.8)
        
        val, g, c, w = compute_gradient_topk(p, K, tau)
        
        step += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        
        # Adaptive learning rate
        lr = best_lr / (1.0 + 0.1 * step ** 0.5)
        p_new = p - lr * m_hat / (np.sqrt(v_hat) + eps)
        p_new = project_simplex(p_new)
        
        # Evaluate
        new_val = np.max(fft_convolve(p_new, p_new))
        
        if new_val < best_val:
            best_val = new_val
            best_p = p_new.copy()
            no_improve = 0
            # Slightly increase lr if we're making progress
            best_lr = min(1.0, best_lr * 1.01)
        else:
            no_improve += 1
            if no_improve > 200:
                # Restart with smaller lr and some noise
                best_lr *= 0.7
                p = best_p + 0.01 * rng.standard_normal(n)
                p = project_simplex(p)
                no_improve = 0
        
        if no_improve % 500 == 0 and no_improve > 0:
            if new_val > plateau_best:
                plateau_best = new_val
        
        p = p_new
        
        if step % 5000 == 0:

    return best_p, 2.0 * n * best_val

def polish_local_search(a, time_limit, rng):
    """
    Simulated annealing with targeted peak reduction.
    Works on the actual sequence (not probability vector).
    """
    current = a.astype(np.float64).copy()
    n = len(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    temperature = 0.001
    cooling = 0.99985
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.4:
            # Targeted: reduce peak contributors
            conv = fft_convolve(current, current)
            # Find top peaks
            top_k = min(5, len(conv))
            peak_indices = np.argsort(conv)[-top_k:]
            # For each peak, find top contributors
            all_contribs = np.zeros(n)
            for pi in peak_indices:
                for i in range(min(n, pi + 1)):
                    j = pi - i
                    if 0 <= j < n:
                        all_contribs[i] += current[i] * current[j]
            top_idx = np.argsort(all_contribs)[-max(3, n//12):]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            # Increase others
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
        elif r < 0.6:
            # Pair swap: move mass from high-contributing to low
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top = np.argsort(contribs)[-3:]
            bot = np.argsort(contribs)[:min(5, n//4)]
            amt = current[top[0]] * 0.05
            candidate[top[0]] -= amt
            candidate[bot[0]] += amt
        elif r < 0.8:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 15))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.9:
            # Partial symmetrization
            t = 0.1 + 0.1 * rng.random()
            candidate = (1-t) * current + t * current[::-1]
            candidate *= np.exp(rng.normal(0.0, 0.005, size=n))
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.005, size=n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best, best_score
```

```python
"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions. Combines gradient descent with annealed softmax peak tracking,
simulated annealing local search with targeted peak reduction, and exact
subgradient fine-tuning. Uses FFT-based convolution for efficiency.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Fast local computation of the score (same formula as evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_softmax(p, tau):
    """
    Compute gradient of softmax_max(conv(p,p), tau) w.r.t. p.
    Returns (smooth_value, gradient, convolution, weights).
    """
    n = len(p)
    c = fft_convolve(p, p)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    value = c_max + tau * np.log(np.sum(exp_s))
    # Gradient: g[k] = 2 * sum_i p[i] * w[k+i]
    p_rev = p[::-1]
    corr = fft_convolve(p_rev, w)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return value, g, c, w


def compute_grad_topk(p, K, tau):
    """
    Compute gradient using top-K peaks with softmax weighting.
    More focused than full softmax when peaks are separated.
    """
    n = len(p)
    c = fft_convolve(p, p)
    
    if K >= len(c):
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        exp_s = np.exp(shifted)
        w = exp_s / np.sum(exp_s)
    else:
        sorted_c = np.sort(c)[::-1]
        threshold = sorted_c[K - 1]
        mask = c >= threshold
        if not np.any(mask):
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            c_max = np.max(c)
            shifted = np.zeros(len(c))
            shifted[mask] = (c[mask] - c_max) / tau
            exp_s = np.exp(shifted)
            total = np.sum(exp_s)
            w = exp_s / total
    
    value = np.dot(w, c)
    p_rev = p[::-1]
    corr = fft_convolve(p_rev, w)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return value, g, c, w


def optimize_gradient(n, p_init, time_limit, rng, K=10):
    """
    Adam gradient descent on probability vector p to minimize max(conv(p,p)).
    Uses annealed temperature and top-K peak tracking.
    Returns (best_p, best_score) where best_score = 2n * max(conv(best_p, best_p)).
    """
    p = np.maximum(p_init, 0.0).astype(np.float64)
    p_sum = np.sum(p)
    if p_sum < 1e-10:
        p = np.ones(n) / n
    else:
        p = p / p_sum

    c_init = fft_convolve(p, p)
    max_init = np.max(c_init)

    best_p = p.copy()
    best_val = max_init

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0

    # Temperature: start relative to the scale of convolution values
    tau_base = max_init
    tau_start = 3.0 * tau_base
    tau_end = 0.05 * tau_base

    # Learning rate schedule
    lr_base = 0.8
    best_lr = lr_base

    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        frac = min(1.0, elapsed * 1.1)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)

        if K > 0:
            val, g, c, w = compute_grad_topk(p, K, tau)
        else:
            val, g, c, w = compute_grad_softmax(p, tau)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # Cosine-annealed learning rate
        lr = best_lr * (0.5 + 0.5 * np.cos(np.pi * min(1.0, step / max(1, int(time_limit * 500 / 0.001)))))
        lr = max(lr, lr_base * 0.05)

        p_new = p - lr * m_hat / (np.sqrt(v_hat) + eps)
        p_new = project_simplex(p_new)

        new_val = np.max(fft_convolve(p_new, p_new))

            best_val = new_val
            best_p = p_new.copy()
            no_improve = 0

        else:
            no_improve += 1

                noise = 0.02 * rng.standard_normal(n)
                p = best_p + noise
                p = project_simplex(p)
                no_improve = 0
                # Reset Adam state
                m = np.zeros(n)
                v = np.zeros(n)
                step = 0

                  f"tau={tau:.2e}, lr={lr:.4f}, t={elapsed_now:.1f}s")

    return best_p, 2.0 * n * best_val

    n = len(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score

    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9999
    min_temp = 1e-6

    iter_count = 0

        candidate = current.copy()
        r = rng.random()
        iter_count += 1

            conv = fft_convolve(current, current)
            top_k_peaks = min(3, len(conv))
            peak_indices = np.argsort(conv)[-top_k_peaks:]
            all_contribs = np.zeros(n)
            for pi in peak_indices:
                pi = int(pi)
                for i in range(min(n, pi + 1)):
                    j = pi - i
                    if 0 <= j < n:
                        all_contribs[i] += current[i] * current[j]
            top_frac = max(0.1, 0.3 / (1.0 + iter_count / 5000.0))
            n_top = max(3, int(n * top_frac))
            top_idx = np.argsort(all_contribs)[-n_top:]
            candidate[top_idx] *= np.exp(rng.normal(-0.025, 0.008, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(12, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.025, 0.008, size=n_inc))
        elif r < 0.55:
            # Pair: move mass from high-contributing to low
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_i = np.argmax(contribs)
            bot_candidates = np.where(contribs < np.median(contribs))[0]
            if len(bot_candidates) > 0:
                bot_i = bot_candidates[rng.integers(0, len(bot_candidates))]
                amt = current[top_i] * (0.02 + 0.03 * rng.random())
                candidate[top_i] -= amt
                candidate[bot_i] += amt
        elif r < 0.75:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 12))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.85:
            # Multi-small-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 20))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.01, size=width))
        elif r < 0.93:
            # Partial symmetrization
            t_sym = 0.05 + 0.1 * rng.random()
            candidate = (1.0 - t_sym) * current + t_sym * current[::-1]
            candidate *= np.exp(rng.normal(0.0, 0.003, size=n))
        else:
            # Global micro-perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.004, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif np.log(rng.random()) < -(new_score - current_score) / max(temperature, 1e-12):
            current = candidate
            current_score = new_score

        temperature *= cooling

    return best, best_score


def fine_tune_subgradient(a, time_limit, rng):
    """
    Fine-tuning using exact subgradient of the max convolution value.
    Very small steps focused on the single worst peak.
    """
    current = a.astype(np.float64).copy()
    n = len(current)
    p = current / np.sum(current)
    p = np.maximum(p, 0.0)
    p = p / np.sum(p)

    best_p = p.copy()
    best_val = np.max(fft_convolve(p, p))

    t0 = time.monotonic()
    lr = 0.001
    momentum = np.zeros(n)
    beta = 0.9

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(p, p)
        m = int(np.argmax(c))

        # Exact subgradient: g[k] = 2 * p[m-k] if 0 <= m-k < n, else 0
        g = np.zeros(n)
        lo = max(0, m - (n - 1))
        hi = min(n - 1, m)
        for k in range(lo, hi + 1):
            g[k] = 2.0 * p[m - k]

        momentum = beta * momentum + (1.0 - beta) * g
        p_new = p - lr * momentum
        p_new = project_simplex(p_new)

        new_val = np.max(fft_convolve(p_new, p_new))
        if new_val < best_val:
            best_val = new_val
            best_p = p_new.copy()
            lr = min(0.005, lr * 1.01)
        else:
            lr *= 0.95
            if lr < 1e-7:
                break

        p = p_new

    a_best = best_p * np.sum(current)
    return a_best, local_score(a_best)


def make_chirp_init(n, rng):
    """Chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    alpha = 0.2 + 0.4 * rng.random()
    f0 = rng.uniform(0.5, 2.0)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * (f0 * t / n + 0.5 * t * t / (n * n)))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    n_tones = int(rng.integers(3, 8))
    for _ in range(n_tones):
        k = int(rng.integers(1, max(2, n // 3)))
        amp = 0.05 + 0.2 * rng.random()
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.5):
    """Random Gaussian initial sequence."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_inverse_barker_like(n, rng):
    """Initialize with a pattern inspired by low-autocorrelation sequences."""
    # Use a weighted sum of shifted rectangles to create structure
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n) * 0.5
    n_rects = int(rng.integers(5, 15))
    for _ in range(n_rects):
        center = int(rng.integers(0, n))
        width = int(rng.integers(1, n // 4))
        amp = rng.uniform(-0.3, 0.3)
        for j in range(n):
            d = abs(j - center)
            if d < width:
                a[j] += amp * (1.0 - d / width)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for optimal non-negative step function coefficients.
    Phase 1: Multi-start gradient optimization with top-K peak tracking.
    Phase 2: Simulated annealing local search with targeted perturbations.
    Phase 3: Exact subgradient fine-tuning.
    Phase 4: Final local search polish.
    """
    t_global = time.monotonic()
    time_budget = max(10.0, float(budget_s) - 60.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    start_seq = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            s = local_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()
    except Exception:
        pass

    # Phase 1: Multi-start gradient optimization
    print("\n=== Phase 1: Gradient optimization ===")

    configs = [
        # (n, num_inits, time_per_run, K)
        (256, 5, 50.0, 10),
        (384, 3, 45.0, 10),
        (512, 3, 45.0, 15),
        (768, 2, 40.0, 15),
    ]

    for n, n_inits, t_per_run, K in configs:
        if time.monotonic() > deadline - 240:
            break

        for init_i in range(n_inits):
            if time.monotonic() > deadline - 240:
                break

            # Choose initialization
            if init_i == 0 and start_seq is not None:
                if n == len(start_seq):
                    p0 = start_seq / np.sum(start_seq)
                    init_name = "provided"
                else:
                    x_old = np.linspace(0, 1, len(start_seq))
                    x_new = np.linspace(0, 1, n)
                    a_res = np.interp(x_new, x_old, start_seq)
                    p0 = a_res / np.sum(a_res)
                    init_name = "provided_res"
            elif init_i == 1:
                p0 = make_chirp_init(n, rng)
                p0 = p0 / np.sum(p0)
                init_name = "chirp"
            elif init_i == 2:
                p0 = make_multi_tone_init(n, rng)
                p0 = p0 / np.sum(p0)
                init_name = "multi_tone"
            elif init_i == 3:
                p0 = make_random_init(n, rng, spread=0.3 + 0.3 * rng.random())
                p0 = p0 / np.sum(p0)
                init_name = "random"
            else:
                p0 = make_inverse_barker_like(n, rng)
                p0 = p0 / np.sum(p0)
                init_name = "barker_like"

            t0 = time.monotonic()
            p_best, score = optimize_gradient(n, p0, t_per_run, rng, K=K)
            elapsed = time.monotonic() - t0

            a_best = p_best * np.sqrt(2.0 * n)
            actual_score = local_score(a_best)

            print(f"  n={n}, init='{init_name}', score={actual_score:.6f}, time={elapsed:.1f}s")

            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Phase 2: Local search polish on best candidate
    print(f"\n=== Phase 2: Local search polish ===")
    if best_seq is not None:
        polish_time = min(400.0, max(60.0, (deadline - time.monotonic()) - 120.0))
        if polish_time > 30:
            print(f"  Starting polish, score={best_score:.6f}, time={polish_time:.0f}s")
            polished, polish_score = polish_local_search(best_seq, polish_time, rng)
            if polish_score < best_score:
                best_score = polish_score
                best_seq = polished.copy()
                print(f"  After polish: {best_score:.6f}")
            else:
                print(f"  Polish: no improvement (still {best_score:.6f})")

    # Phase 3: Fine-tune with subgradient
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\n=== Phase 3: Subgradient fine-tuning ===")
        ft_time = min(60.0, (deadline - time.monotonic()) - 60.0)
        if ft_time > 10:
            a_ft, ft_score = fine_tune_subgradient(best_seq, ft_time, rng)
            print(f"  Subgradient fine-tune: score={ft_score:.6f}")
            if ft_score < best_score:
                best_score = ft_score
                best_seq = a_ft.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Phase 4: Final short local search
    if time.monotonic() < deadline - 20 and best_seq is not None:
        remaining = (deadline - time.monotonic()) - 20.0
        if remaining > 20:
            print(f"\n=== Phase 4: Final polish ({remaining:.0f}s) ===")
            polished2, ps2 = polish_local_search(best_seq, remaining, rng)
            if ps2 < best_score:
                best_score = ps2
                best_seq = polished2.copy()
                print(f"  Final polish: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
