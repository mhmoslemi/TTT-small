def normalized_fast_score(a, n=None):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if n is None:
        n = a.size
    else:
        if a.size != n:
            a = linear_upsample(a, n)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    S = float(np.sqrt(2.0 * n))
    if s <= 1e-12 or not np.isfinite(s):
        a = np.full(n, S / n, dtype=np.float64)

    c = self_conv_fast(a, n)
    sc = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(sc):
        sc = float('inf')
    return a, sc

```python
"""
Search for a non-negative sequence minimizing

    2 * n * max(self-convolution) / (sum sequence)^2

using a combination of:

  1. Length-8192 fixed search (the supplied construction length).
  2. Initial candidates from the previous best, its reverse, endpoint-zero
     variants, smoothed variants, low-frequency perturbations and several
     analytic profiles.
  3. FFT-based convolution and auxiliary log-sum-exp gradients to approximate
     the max-convolution objective smoothly.
  4. Adam-style updates in two parameterizations:
        - projection onto the non-negative simplex,
        - log/softmax (entropic) parameterization.
  5. Exact active-lag subgradient polishing with random low-frequency kicks.
  6. A long final polishing phase on the globally best sequence.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def _pow2_at_least(m):
    """Smallest power of two >= m."""
    m = int(m)
    if m <= 0:
        return 1
    return 1 << (m - 1).bit_length()


def linear_upsample(arr, n):
    """Interpolate a vector to length n on [0, 1]."""
    arr = np.asarray(arr, dtype=np.float64).reshape(-1)
    if arr.size == n:
        return arr.copy()
    if arr.size == 0:
        return np.ones(n, dtype=np.float64)
    if arr.size == 1:
        return np.full(n, float(arr[0]), dtype=np.float64)
    arr = np.clip(arr, 0.0, 1000.0)
    old = np.linspace(0.0, 1.0, arr.size)
    new = np.linspace(0.0, 1.0, n)
    return np.interp(new, old, arr)


def normalized_fast_score(a, n=None):
    """Normalize to sum sqrt(2n) and return (sequence, max_conv)."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if n is None:
        n = a.size
    else:
        if a.size != n:
            a = linear_upsample(a, n)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    S = float(np.sqrt(2.0 * n))
    if not np.isfinite(s) or s <= 1e-12:
        a = np.full(n, S / n, dtype=np.float64)
    else:
        a = a * (S / s)
    c = self_conv_fast(a, n)
    sc = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(sc):
        sc = float("inf")
    return a, sc


def self_conv_fast(a, n):
    """Full linear self-convolution via FFT."""
    L = 2 * n - 1
    size = _pow2_at_least(L)
    A = fft(a, size)
    cfull = ifft(A * A)
    c = cfull[:L].real
    np.maximum(c, 0.0, out=c)
    return c


def full_conv_and_logsumexp_grad(a, beta, n):
    """
    Return c, max_c, and gradient of (1/beta)*logsumexp(beta*c) with respect to a.
    The gradient is not scaled by 1/beta; only the direction matters for our
    optimizers, and the call already computes c for bookkeeping.
    """
    L = 2 * n - 1
    size = _pow2_at_least(3 * n - 2)  # enough for a * reversed(w)
    A = fft(a, size)
    cfull = ifft(A * A)
    c = cfull[:L].real
    np.maximum(c, 0.0, out=c)
    maxc = float(c.max()) if c.size else 0.0

    if not np.isfinite(maxc):
        maxc = 0.0

    # Stable weights proportional to exp(beta * c).
    w = np.exp(beta * (c - maxc))
    Z = float(w.sum())
    if (not np.isfinite(Z)) or Z <= 1e-300:
        w = np.zeros(L, dtype=np.float64)
        if L > 0:
            w[int(np.argmax(c))] = 1.0
        Z = 1.0

    # h_m = sum_j a_j w_{m+j}; grad_a = 2*h / Z.
    B = fft(w[::-1], size)
    H = ifft(A * B)
    h = H[2 * n - 2 : n - 2 : -1].real
    grad = 2.0 * h / Z
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return c, maxc, grad


def top_gradient_uniform(a, top_idx):
    """Average (uniform) subgradient over the selected lags."""
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    L = 2 * n - 1
    for k in top_idx:
        k = int(k)
        if k < 0 or k >= L:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[lo:hi + 1][::-1]
    return grad


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    v = np.nan_to_num(np.asarray(v, dtype=np.float64), nan=0.0, posinf=0.0, neginf=0.0)
    v = np.maximum(v, 0.0)
    n = v.size
    if n == 0:
        return v
    if n == 1:
        return np.array([S], dtype=np.float64)
    s = float(v.sum())
    if s <= 1e-300:
        return np.full(n, S / n, dtype=np.float64)

    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cand = (cumsum - S) / idx
    valid = theta_cand <= w

    if np.any(valid):
        rho = int(np.max(np.where(valid)[0])) + 1
    else:
        rho = n

    theta = (cumsum[rho - 1] - S) / float(rho)
    out = np.maximum(v - theta, 0.0)
    out_sum = float(out.sum())
    if out_sum <= 1e-300:
        return np.full(n, S / n, dtype=np.float64)
    out *= S / out_sum
    return out


def softmax_scaled(y, S):
    """Return S*softmax(y), with stable exponentiation."""
    y = np.nan_to_num(np.asarray(y, dtype=np.float64), nan=0.0, posinf=0.0, neginf=-700.0)
    y = y - float(np.max(y))
    y = np.maximum(y, -700.0)
    p = np.exp(y)
    s = float(p.sum())
    if (not np.isfinite(s)) or s <= 1e-300:
        return np.full(y.shape, S / y.size, dtype=np.float64)
    return S * (p / s)


def smooth_noise_unit(n, rng):
    """Smooth zero-mean unit-variance low-frequency noise."""
    x = np.linspace(0.0, 1.0, n)
    noise = np.zeros(n, dtype=np.float64)
    for _ in range(5):
        f = float(rng.integers(1, 33))
        phi = float(rng.uniform(0.0, 2.0 * np.pi))
        noise += np.sin(2.0 * np.pi * f * x + phi)
    noise -= float(np.mean(noise))
    std = float(np.std(noise))
    if std > 1e-15:
        noise /= std
    else:
        noise[:] = 0.0
    return noise


def make_profiles(n, rng):
    """Several non-negative analytic profiles, not normalized."""
    x = np.linspace(0.0, 1.0, n)
    profiles = []

    # Linear, mild tilt.
    profiles.append(1.0 + 0.8 * (1.0 - 2.0 * x))
    # Linear decreasing toward zero.
    profiles.append(1.0 - 0.99 * x)
    # Quadratic decreasing small endpoint.
    profiles.append((1.0 - 0.98 * x) ** 2)
    # Square-root decreasing small endpoint.
    profiles.append(np.sqrt(np.maximum(1.0 - 0.98 * x, 1e-12)))
    # Exponential decreasing.
    profiles.append(np.exp(-2.0 * x))
    # Cosine tapered to zero.
    profiles.append(0.5 * (1.0 + np.cos(np.pi * x)))

    # One random smooth positive profile.
    a = 0.8
    for _ in range(6):
        f = float(rng.integers(1, 9))
        phi = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.05, 0.35))
        a += amp * np.sin(np.pi * f * x + phi)
    profiles.append(np.maximum(a, 0.0))

    out = []
    for p in profiles:
        p = np.asarray(p, dtype=np.float64).reshape(-1)
        p = np.maximum(p, 0.0)
        if float(np.sum(p)) <= 1e-15:
            p = np.ones(n, dtype=np.float64)
        out.append(p)
    return out


def build_candidates(parent, n, rng):
    """Build a modest set of initial candidate sequences."""
    raw = []

    if parent is not None:
        p, _ = normalized_fast_score(parent, n)
        raw.append(("parent", p))
        raw.append(("parent_rev", p[::-1].copy()))

        block = max(4, n // 4096)
        end0 = p.copy()
        end0[-block:] = 0.0
        raw.append(("parent_end0", end0))

        start0 = p.copy()
        start0[:block] = 0.0
        raw.append(("parent_start0", start0))

        S = float(np.sqrt(2.0 * n))
        amp = float(rng.uniform(0.05, 0.15))
        raw.append(("parent_noise", p + amp * (S / n) * smooth_noise_unit(n, rng)))

        kernel = np.ones(16, dtype=np.float64) / 16.0
        p_smooth = np.convolve(p, kernel, mode="same")
        raw.append(("parent_smooth", p_smooth))

    for i, prof in enumerate(make_profiles(n, rng)):
        raw.append(("profile%d" % i, prof))

    cands = []
    for name, arr in raw:
        a, sc = normalized_fast_score(arr, n)
        if np.isfinite(sc):
            cands.append((name, a, sc))

    if not cands:
        a = np.full(n, 1.0, dtype=np.float64)
        a, sc = normalized_fast_score(a, n)
        cands.append(("fallback", a, sc))

    return cands


def run_log_soft_full(a, n, S, dur, rng,
                      beta_start=15.0, beta_end=1500.0,
                      lr_start=0.20, lr_end=0.025):
    """Adam on log-softmax parameter with log-sum-exp smooth max."""
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size != n:
        a = linear_upsample(a, n)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12 or not np.isfinite(s):
        a = np.full(n, S / n, dtype=np.float64)
    else:
        a = a * (S / s)

    y = np.log(np.maximum(a, 1e-300))
    y = np.nan_to_num(y, nan=0.0, posinf=0.0, neginf=-700.0)
    y -= float(np.max(y))
    y = np.maximum(y, -700.0)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1, b2 = 0.85, 0.999
    eps = 1e-8

    c0 = self_conv_fast(a, n)
    best_s = float(c0.max()) if c0.size else float("inf")
    best_a = a.copy()

    dur = float(max(1e-6, dur))
    start = time.monotonic()
    end = start + dur
    t = 0
    no_impr = 0

    while time.monotonic() < end:
        elapsed = time.monotonic() - start
        prog = min(1.0, elapsed / dur)
        beta = beta_start * ((beta_end / max(beta_start, 1e-15)) ** prog)

        c, maxc, grad = full_conv_and_logsumexp_grad(a, beta, n)
        if maxc < best_s - 1e-12:
            best_s = maxc
            best_a = a.copy()
            no_impr = 0
        else:
            no_impr += 1

        gscale = float(np.max(np.abs(grad)))
        if (not np.isfinite(gscale)) or gscale <= 1e-15:
            break
        grad /= gscale

        # Gradient with respect to log-probabilities:
        # a = S*softmax(y), da_i/dy_j = a_j*(delta_ij - a_i/S).
        dot = float(np.dot(grad, a)) / S
        gy = a * (grad - dot)
        gy_scale = float(np.max(np.abs(gy)))
        if (not np.isfinite(gy_scale)) or gy_scale <= 1e-15:
            break
        gy /= gy_scale

        t += 1
        lr = lr_start * ((lr_end / max(lr_start, 1e-15)) ** min(1.0, t / 15000.0))

        m = b1 * m + (1.0 - b1) * gy
        v = b2 * v + (1.0 - b2) * (gy * gy)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if no_impr > 8000 and rng.random() < 0.02:
            kick = float(rng.uniform(0.05, 0.25))
            y += kick * smooth_noise_unit(n, rng)

        y = np.nan_to_num(y, nan=0.0, posinf=0.0, neginf=-700.0)
        y -= float(np.mean(y))
        y -= float(np.max(y))
        y = np.maximum(y, -700.0)
        a = softmax_scaled(y, S)

    if not np.isfinite(best_s):
        c = self_conv_fast(best_a, n)
        best_s = float(c.max()) if c.size else float("inf")
    return best_a, best_s


def run_projected_full(a, n, S, dur, rng,
                       beta_start=30.0, beta_end=3000.0,
                       lr_start=0.004, lr_end=0.0003):
    """Adam with Euclidean projection onto the non-negative simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size != n:
        a = linear_upsample(a, n)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12 or not np.isfinite(s):
        a = np.full(n, S / n, dtype=np.float64)
    else:
        a = a * (S / s)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1, b2 = 0.85, 0.999
    eps = 1e-8

    c0 = self_conv_fast(a, n)
    best_s = float(c0.max()) if c0.size else float("inf")
    best_a = a.copy()

    dur = float(max(1e-6, dur))
    start = time.monotonic()
    end = start + dur
    t = 0
    no_impr = 0

    while time.monotonic() < end:
        elapsed = time.monotonic() - start
        prog = min(1.0, elapsed / dur)
        beta = beta_start * ((beta_end / max(beta_start, 1e-15)) ** prog)

        c, maxc, grad = full_conv_and_logsumexp_grad(a, beta, n)
        if maxc < best_s - 1e-12:
            best_s = maxc
            best_a = a.copy()
            no_impr = 0
        else:
            no_impr += 1

        gscale = float(np.max(np.abs(grad)))
        if (not np.isfinite(gscale)) or gscale <= 1e-15:
            break
        grad /= gscale

        t += 1
        lr = lr_start * ((lr_end / max(lr_start, 1e-15)) ** min(1.0, t / 15000.0))

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a_new = a - lr * (m_hat / (np.sqrt(v_hat) + eps))

        if no_impr > 8000 and rng.random() < 0.05:
            kick = float(rng.uniform(0.05, 0.3))
            a_new += (kick * S / n) * smooth_noise_unit(n, rng)

        a_new = np.nan_to_num(a_new, nan=0.0, posinf=0.0, neginf=0.0)
        a = project_simplex(a_new, S)

    if not np.isfinite(best_s):
        c = self_conv_fast(best_a, n)
        best_s = float(c.max()) if c.size else float("inf")
    return best_a, best_s


def run_projected_top(a, n, S, dur, rng,
                      top_k_start=8, top_k_end=40,
                      window_frac_start=0.08, window_frac_end=1e-4,
                      lr_start=0.004, lr_end=0.0003,
                      initial_score=None):
    """Polish with exact top-lag subgradients and simplex projection."""
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size != n:
        a = linear_upsample(a, n)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12 or not np.isfinite(s):
        a = np.full(n, S / n, dtype=np.float64)
    else:
        a = a * (S / s)

    best_a = a.copy()
    if initial_score is not None and np.isfinite(initial_score):
        best_s = float(initial_score)
    else:
        c0 = self_conv_fast(a, n)
        best_s = float(c0.max()) if c0.size else float("inf")

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1, b2 = 0.85, 0.999
    eps = 1e-8

    dur = float(max(1e-6, dur))
    start = time.monotonic()
    end = start + dur
    t = 0
    no_impr = 0
    L = 2 * n - 1

    while time.monotonic() < end:
        c = self_conv_fast(a, n)
        maxc = float(c.max()) if c.size else 0.0
        if not np.isfinite(maxc):
            break

        if maxc < best_s - 1e-12:
            best_s = maxc
            best_a = a.copy()
            no_impr = 0
        else:
            no_impr += 1

        elapsed = time.monotonic() - start
        prog = min(1.0, elapsed / dur)
        top_k = int(top_k_start + (top_k_end - top_k_start) * prog)
        top_k = max(1, min(top_k, L))

        window = max(maxc * window_frac_end, maxc * window_frac_start * (1.0 - prog))
        active = np.flatnonzero(c >= maxc - window)
        if active.size == 0:
            active = np.array([int(np.argmax(c))], dtype=np.int64)

        if active.size > top_k:
            top_idx = np.argpartition(c, -top_k)[-top_k:]
        else:
            top_idx = active

        grad = top_gradient_uniform(a, top_idx)
        gscale = float(np.max(np.abs(grad)))
        if (not np.isfinite(gscale)) or gscale <= 1e-15:
            break
        grad /= gscale

        t += 1
        lr = lr_start * ((lr_end / max(lr_start, 1e-15)) ** min(1.0, t / 10000.0))

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a_new = a - lr * (m_hat / (np.sqrt(v_hat) + eps))

        if no_impr > 5000 and rng.random() < 0.08:
            kick = float(rng.uniform(0.05, 0.3))
            a_new += (kick * S / n) * smooth_noise_unit(n, rng)

        a_new = np.nan_to_num(a_new, nan=0.0, posinf=0.0, neginf=0.0)
        a = project_simplex(a_new, S)

    if not np.isfinite(best_s):
        c = self_conv_fast(best_a, n)
        best_s = float(c.max()) if c.size else float("inf")
    return best_a, best_s


def run_candidate(name, a, n, S, dur, rng, initial_score):
    """Run several methods on one candidate for `dur` seconds."""
    a_best = np.asarray(a, dtype=np.float64).copy()
    if a_best.size != n:
        a_best = linear_upsample(a_best, n)
    s_best = float(initial_score)

    dur = float(max(0.0, dur))
    if dur > 200.0:
        seg_log = 0.36 * dur
        seg_pf = 0.18 * dur
        seg_top = dur - seg_log - seg_pf
        if seg_top < 5.0:
            seg_top = 0.45 * dur
            seg_log = dur - seg_top
            seg_pf = 0.0
    elif dur > 60.0:
        seg_log = 0.55 * dur
        seg_top = dur - seg_log
        seg_pf = 0.0
    else:
        seg_log = dur
        seg_pf = 0.0
        seg_top = 0.0

    if seg_log > 5.0:
        a1, s1 = run_log_soft_full(
            a_best, n, S, seg_log, rng,
            beta_start=10.0, beta_end=1200.0,
            lr_start=0.20, lr_end=0.025
        )
        if np.isfinite(s1) and s1 < s_best:
            a_best, s_best = a1, s1

    if seg_pf > 5.0:
        a2, s2 = run_projected_full(
            a_best, n, S, seg_pf, rng,
            beta_start=40.0, beta_end=3000.0,
            lr_start=0.004, lr_end=0.0003
        )
        if np.isfinite(s2) and s2 < s_best:
            a_best, s_best = a2, s2

    if seg_top > 5.0:
        a3, s3 = run_projected_top(
            a_best, n, S, seg_top, rng,
            top_k_start=8, top_k_end=40,
            window_frac_start=0.08, window_frac_end=1e-3,
            lr_start=0.004, lr_end=0.0003,
            initial_score=s_best
        )
        if np.isfinite(s3) and s3 < s_best:
            a_best, s_best = a3, s3

    return a_best, s_best


def polish_sequence(a, n, S, max_time, rng, initial_score):
    """Long final polishing phase for the global best sequence."""
    a_best = np.asarray(a, dtype=np.float64).copy()
    if a_best.size != n:
        a_best = linear_upsample(a_best, n)
    s_best = float(initial_score)
    if (not np.isfinite(s_best)) or s_best <= 0.0:
        _, s_best = normalized_fast_score(a_best, n)

    max_time = float(max(0.0, max_time))
    start = time.monotonic()
    end = start + max_time

    # Smooth log-sum-exp polish.
    seg = min(0.22 * max_time, max(0.0, end - time.monotonic()))
    if seg > 5.0:
        a1, s1 = run_log_soft_full(
            a_best, n, S, seg, rng,
            beta_start=80.0, beta_end=6000.0,
            lr_start=0.12, lr_end=0.015
        )
        if np.isfinite(s1) and s1 < s_best:
            a_best, s_best = a1, s1

    # Full-gradient projected polish.
    seg = min(0.15 * max_time, max(0.0, end - time.monotonic()))
    if seg > 5.0:
        a2, s2 = run_projected_full(
            a_best, n, S, seg, rng,
            beta_start=150.0, beta_end=8000.0,
            lr_start=0.002, lr_end=0.00015
        )
        if np.isfinite(s2) and s2 < s_best:
            a_best, s_best = a2, s2

    chunk = min(120.0, max(20.0, 0.12 * max_time))

    # Repeated top-subgradient polish plus occasional random jumps.
    while time.monotonic() < end - 20.0:
        seg = min(chunk, max(0.0, end - time.monotonic()))
        if seg > 5.0:
            a3, s3 = run_projected_top(
                a_best, n, S, seg, rng,
                top_k_start=12, top_k_end=60,
                window_frac_start=0.02, window_frac_end=3e-5,
                lr_start=0.0015, lr_end=0.0001,
                initial_score=s_best
            )
            if np.isfinite(s3) and s3 < s_best:
                a_best, s_best = a3, s3
                chunk = max(20.0, chunk * 0.9)
            else:
                chunk = min(180.0, chunk * 1.05 + 5.0)

        # Try a few low-frequency perturbations; keep and polish if improved.
        if end - time.monotonic() > 90.0:
            for _ in range(5):
                amp = float(rng.uniform(0.08, 0.8))
                cand = a_best + (amp * S / n) * smooth_noise_unit(n, rng)
                cand = project_simplex(cand, S)
                _, jc = normalized_fast_score(cand, n)
                if np.isfinite(jc) and jc < s_best - 1e-10:
                    a_best, s_best = cand, jc
                    if end - time.monotonic() > 60.0:
                        a4, s4 = run_projected_top(
                            a_best, n, S, min(45.0, end - time.monotonic()), rng,
                            top_k_start=8, top_k_end=40,
                            window_frac_start=0.02, window_frac_end=3e-5,
                            lr_start=0.0015, lr_end=0.0001,
                            initial_score=s_best
                        )
                        if np.isfinite(s4) and s4 < s_best:
                            a_best, s_best = a4, s4

    # Tiny final exact-lag cleanup.
    left = max(0.0, end - time.monotonic())
    if left > 20.0:
        a5, s5 = run_projected_top(
            a_best, n, S, min(60.0, left), rng,
            top_k_start=1, top_k_end=4,
            window_frac_start=0.005, window_frac_end=1e-5,
            lr_start=0.0008, lr_end=0.00006,
            initial_score=s_best
        )
        if np.isfinite(s5) and s5 < s_best:
            a_best, s_best = a5, s5

    return a_best, s_best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Return the best sequence found before the time budget expires."""
    budget_s = float(max(1.0, budget_s))
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 12.0)
    rng = np.random.default_rng(seed)

    # Fixed search length 8192 (the supplied construction length).
    n = 8192
    S = float(np.sqrt(2.0 * n))

    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
    except Exception:
        parent = None

    # If all else fails, uniform is valid and safe.
    def _fallback():
        if parent is not None and parent.size > 0:
            a, _ = normalized_fast_score(parent, n)
        else:
            a = np.full(n, 1.0, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s > 0.0 and np.isfinite(s):
            a = a / s
        return [float(x) for x in a]

    if deadline - start <= 15.0:
        return _fallback()

    print("Propose candidate: n=%d, budget=%.1fs" % (n, budget_s))

    cands = build_candidates(parent, n, rng)

    # Initial best.
    init_a = cands[0][1]
    init_s = cands[0][2]
    for _, a, sc in cands:
        if np.isfinite(sc) and sc < init_s:
            init_a, init_s = a, sc

    # Print a compact score table.
    compact = sorted([(sc, 0, name, a) for name, a, sc in cands])
    for sc, _, name, _ in compact[:8]:
        print("  candidate %-15s score=%.6f" % (name, sc))

    remaining = max(1.0, deadline - time.monotonic())
    polish_time = min(0.45 * remaining + 60.0, 0.60 * remaining)
    explore_time = remaining - polish_time - 30.0
    if explore_time < 30.0:
        polish_time = remaining
        explore_time = 0.0

    # Schedule candidates.  Keep the old best and reverse first, then sort
    # the rest by fast FFT score.
    primary = []
    others_raw = []
    for name, a, sc in cands:
        if name in ("parent", "parent_rev"):
            primary.append((name, a, sc))
        else:
            others_raw.append((name, a, sc))

    other_sorted = sorted([(sc, i, name, a) for i, (name, a, sc) in enumerate(others_raw)])
    others = [(name, a, sc) for sc, i, name, a in other_sorted]

    schedule = []
    if primary:
        if len(primary) >= 1:
            schedule.append((primary[0], max(0.0, 0.25 * explore_time)))
        if len(primary) >= 2:
            schedule.append((primary[1], max(0.0, 0.12 * explore_time)))
        rest_explore = explore_time - 0.25 * explore_time - (0.12 * explore_time if len(primary) >= 2 else 0.0)
    else:
        rest_explore = explore_time

    if others:
        per_other = max(8.0, rest_explore / float(len(others)))
        for item in others:
            schedule.append((item, per_other))

    total_dur = sum(d for _, d in schedule)
    if total_dur > explore_time and total_dur > 0.0:
        scale = explore_time / total_dur
        schedule = [(item, d * scale) for item, d in schedule]

    best_a = init_a.copy()
    best_s = float(init_s)

    for (name, a, sc), dur in schedule:
        now = time.monotonic()
        reserve = max(30.0, min(120.0, polish_time * 0.1))
        if now + dur > deadline - reserve:
            dur = deadline - reserve - now
        if dur < 8.0:
            break

        a2, s2 = run_candidate(name, a, n, S, dur, rng, sc)
        if np.isfinite(s2) and s2 < best_s:
            best_a, best_s = a2, s2
            print("  explore %-15s -> %.6f" % (name, best_s))
        else:
            print("  explore %-15s -> %.6f (best %.6f)" % (name, s2 if np.isfinite(s2) else float("nan"), best_s))

    # Polish using all remaining time.
    left = max(10.0, deadline - time.monotonic())
    if left > 20.0:
        a_final, s_final = polish_sequence(best_a, n, S, left, rng, best_s)
        if np.isfinite(s_final) and s_final < best_s:
            best_a, best_s = a_final, s_final
        else:
            best_s = best_s if np.isfinite(best_s) else s_final
        print("  after polish, internal max_conv=%.6f" % best_s)

    # Final sanitation.
    best_a = np.asarray(best_a, dtype=np.float64).reshape(-1)
    if best_a.size != n:
        best_a = linear_upsample(best_a, n)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(best_a))
    if (not np.isfinite(s)) or s <= 0.0:
        best_a = np.full(n, 1.0, dtype=np.float64)
        s = float(n)

    # Scale to sum 1.0; objective is scale invariant and this avoids the
    # sum < 0.01 guard with a comfortable margin.
    if s > 1e-300:
        best_a = best_a / s
    else:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    final_list = [float(x) for x in best_a]

    # Authoritative check (and silent fallback if somehow invalid).
    try:
        val = evaluate_sequence(final_list)
        if np.isfinite(val):
            print("  evaluated final score=%.6f" % val)
        else:
            print("  final evaluation non-finite, using fallback")
            final_list = _fallback()
    except Exception:
        # If evaluation is unavailable or fails unexpectedly, still return a
        # valid normalized sequence.
        final_list = _fallback()

    return final_list
```
