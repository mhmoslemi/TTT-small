```python
"""
Exponential subgradient descent with multiplicative weights to minimize
the normalized autocorrelation peak (2*n*max(conv(a,a))/sum(a)^2) of a
non-negative sequence. Uses adaptive step size, multiple starts, and
periodic restarts to escape local minima.
"""

import numpy as np
import time


def _compute_score_fast(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    C = np.convolve(a, a)
    return 2.0 * n * np.max(C)


def _gradient_at_peak(a, k_star):
    """Gradient of C[k_star] w.r.t. a, where C = convolve(a, a)."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        indices = np.arange(lo, hi + 1)
        g[indices] = 2.0 * a[k_star - indices]
    return g


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _normalize(a):
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _make_cosine(n, freq, alpha, phase=0.0):
    """a[i] = 1 + alpha * cos(2*pi*freq*i/n + phase), clipped and normalized."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _make_ramp(n, power=2.0):
    """a[i] = (i/(n-1))^power, normalized."""
    i = np.arange(n, dtype=np.float64)
    a = (i / (n - 1)) ** power
    a = np.maximum(a, 1e-10)
    return _normalize(a)


def _make_symmetric_bump(n, alpha):
    """a[i] = 1 + alpha * (2*i/(n-1) - 1)^2, normalized. High at ends, low in middle."""
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / (n - 1) - 1.0
    a = 1.0 + alpha * t * t
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _exp_descent(a, deadline, eta_init=0.05, eta_floor=1e-8, patience=20000, verbose=False):
    """
    Multiplicative subgradient descent to minimize max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_score = _compute_score_fast(best_a)
    eta = eta_init
    no_improve = 0
    iteration = 0
    
    while time.monotonic() < deadline:
        C = np.convolve(a, a)
        k_star = int(np.argmax(C))
        cur_score = 2.0 * n * C[k_star]
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            if verbose and iteration % 100000 == 0:
                print(f"  iter {iteration}: score={best_score:.6f}, eta={eta:.2e}, k*={k_star}")
        else:
            no_improve += 1
        
        if no_improve >= patience:
            break
        
        # Gradient
        g = _gradient_at_peak(a, k_star)
        
        # Multiplicative update
        a = a * np.exp(-eta * g)
        a = _normalize(a)
        
        # Adaptive step size: decrease if no improvement
        if no_improve > 5000:
            eta = max(eta * 0.99, eta_floor)
        
        iteration += 1
    
    return best_a, best_score


def _subgrad_descent(a, deadline, eta_init=0.01, eta_floor=1e-10, patience=10000, verbose=False):
    """
    Additive subgradient descent with simplex projection.
    """
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_score = _compute_score_fast(best_a)
    eta = eta_init
    no_improve = 0
    iteration = 0
    
    while time.monotonic() < deadline:
        C = np.convolve(a, a)
        k_star = int(np.argmax(C))
        cur_score = 2.0 * n * C[k_star]
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            if verbose and iteration % 100000 == 0:
                print(f"  iter {iteration}: score={best_score:.6f}, eta={eta:.2e}, k*={k_star}")
        else:
            no_improve += 1
        
        if no_improve >= patience:
            break
        
        # Gradient
        g = _gradient_at_peak(a, k_star)
        
        # Center the gradient (for sum constraint)
        g = g - np.mean(g)
        
        # Step
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        a = a - eta * g / g_norm
        a = _project_simplex(a, 1.0)
        
        # Adaptive step size
        if no_improve > 3000:
            eta = max(eta * 0.99, eta_floor)
        
        iteration += 1
    
    return best_a, best_score


def _topM_descent(a, deadline, M=5, eta_init=0.05, eta_floor=1e-8, patience=20000, verbose=False):
    """
    Subgradient descent using top-M peaks with softmax weighting.
    """
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_score = _compute_score_fast(best_a)
    eta = eta_init
    no_improve = 0
    iteration = 0
    
    while time.monotonic() < deadline:
        C = np.convolve(a, a)
        cur_score = 2.0 * n * np.max(C)
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            if verbose and iteration % 100000 == 0:
                print(f"  iter {iteration}: score={best_score:.6f}, eta={eta:.2e}")
        else:
            no_improve += 1
        
        if no_improve >= patience:
            break
        
        # Select top M peaks
        if M >= len(C):
            k_indices = np.arange(len(C))
            wts = np.ones(len(C)) / len(C)
        else:
            k_indices = np.argpartition(C, -M)[-M:]
            C_sel = C[k_indices]
            # Softmax weighting
            wts = np.exp(C_sel / max(np.max(C_sel), 1e-15) - 1.0)
            wts /= np.sum(wts)
        
        # Weighted gradient
        g = np.zeros(n)
        for ki in range(len(k_indices)):
            if wts[ki] < 1e-8:
                continue
            g += wts[ki] * _gradient_at_peak(a, int(k_indices[ki]))
        
        # Multiplicative update
        a = a * np.exp(-eta * g)
        a = _normalize(a)
        
        if no_improve > 5000:
            eta = max(eta * 0.99, eta_floor)
        
        iteration += 1
    
    return best_a, best_score
```

```python
    Balance descent: equalize the combined gradient from top peaks.

        cur_max = np.max(C)
        cur_score = 2.0 * n * cur_max

            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50000:
            break
        
        # Select top peaks (those within 5% of max)
        threshold = cur_max * (1 - top_fraction)
        active = C >= threshold
        
        # Combined gradient
        G = np.zeros(n)
        active_indices = np.where(active)[0]
        for k in active_indices:
            G += _gradient_at_peak(a, int(k))
        
        # Normalize G
        G = G / (np.max(G) + 1e-15)
        
        # Update: reduce a where G is high, increase where G is low
        # Use multiplicative update
        a = a * np.exp(-eta * (G - np.mean(G)))
        a = _normalize(a)
        
        if no_improve > 10000:

    return best_a, best_score
```
