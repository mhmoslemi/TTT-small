    n = len(v)
    if n == 0:

    cumsum = np.cumsum(w)

def solve_direction_lp(a, M, n, constraint_idx):
    """Solve LP: max 1^T b s.t. A b <= M, b >= 0"""
    k = len(constraint_idx)
    if k == 0:
        return None
    
    # Build sparse constraint matrix
    rows_list = []
    cols_list = []
    vals_list = []
    
    for i, k_idx in enumerate(constraint_idx):
        k_idx = int(k_idx)
        j_lo = max(0, k_idx - n + 1)
        j_hi = min(n - 1, k_idx)
        if j_lo > j_hi:
            continue
        jj = np.arange(j_lo, j_hi + 1)
        kk = k_idx - jj  # = a[k_idx - j]
        rows_list.extend([i] * len(jj))
        cols_list.extend(jj.tolist())
        vals_list.extend(a[kk].tolist())
    
    if not vals_list:
        return None
    
    A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(k, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(k, M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub,
                        bounds=[(0, None)] * n, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

to find the optimal descent direction that simultaneously respects all active
convolution constraints. This gives a globally-informed direction. We use a
reduced LP (only top-k active constraints) with a cutting plane loop to
iteratively add violated constraints, keeping the LP small and fast.
Combined with exact line search and adaptive restarts, this should significantly
outperform pure subgradient methods.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_conv_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_conv_ab(a, b):
    """Full cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cands = (cumsum - S) / idx
    valid = theta_cands <= w
    if not np.any(valid):
        return np.zeros(n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        n = len(a)
        return np.full(n, S / n, dtype=np.float64)
    return a * (S / s)


def build_sparse_A(a, n, constraint_idx):
    """Build sparse matrix where A[i, j] = a[k_i - j] if valid."""
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(constraint_idx):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        jj = np.arange(j_lo, j_hi + 1)
        kk = k - jj
        rows.extend([i] * len(jj))
        cols.extend(jj.tolist())
        vals.extend(a[kk].tolist())
    if not vals:
        return None
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_idx), n))


def solve_lp_direction(a, M, n, constraint_idx):
    """Solve: maximize sum(b) s.t. (a*b)(k) <= M for k in constraint_idx, b >= 0."""
    if len(constraint_idx) == 0:
        return None
    A = build_sparse_A(a, n, constraint_idx)
    if A is None:
        return None
    c_obj = -np.ones(n)
    b_ub = np.full(len(constraint_idx), M)
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def line_search_mixture(a, b_dir, S, n):
    """Find best t in [0,1] minimizing max of ((1-t)*a + t*b_dir) self-conv."""
    aa = fft_conv_self(a)
    ab = fft_conv_ab(a, b_dir)
    bb = fft_conv_self(b_dir)

    best_t = 0.0
    best_val = aa.max()

    # Coarse search
    for t in np.linspace(0.02, 1.0, 20):
        combo = (1 - t) ** 2 * aa + 2 * t * (1 - t) * ab + t ** 2 * bb
        val = combo.max()
        if val < best_val:
            best_val = val
            best_t = t

    # Fine search around best
    if best_t > 0:
        lo = max(0.0, best_t - 0.05)
        hi = min(1.0, best_t + 0.05)
        for t in np.linspace(lo, hi, 20):
            combo = (1 - t) ** 2 * aa + 2 * t * (1 - t) * ab + t ** 2 * bb
            val = combo.max()
            if val < best_val:
                best_val = val
                best_t = t

    return best_t, best_val


def lp_optimize(a_init, n, S, max_time, rng, top_k_base=50, verbose=True):
    """LP-based optimizer with cutting planes and line search."""
    a = normalize_to_sum(a_init, S)
    c = fft_conv_self(a)
    M = c.max()
    best_a = a.copy()
    best_M = M
    start = time.monotonic()
    iter_count = 0
    no_improve = 0
    stagnant_threshold = 30

    while time.monotonic() - start < max_time:
        iter_count += 1
        c = fft_conv_self(a)
        M = c.max()

        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and iter_count % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP iter {iter_count}: M={M:.8f}, best={best_M:.8f}, "
                  f"t={elapsed:.1f}s")

        if no_improve > stagnant_threshold:
            if verbose:
                print(f"    Stagnant after {no_improve} iters, restarting")
            a = project_simplex(best_a + rng.normal(0, 0.02 * S / n, n), S)
            no_improve = 0
            continue

        # Get active constraints
        top_k = min(top_k_base + iter_count, 2 * n - 1)
        active_idx = np.argsort(c)[::-1][:top_k]

        # Solve LP with cutting plane
        b_dir = solve_lp_direction(a, M, n, active_idx)

        # Cutting plane: check for violations
        if b_dir is not None:
            for cp_iter in range(3):
                conv_check = fft_conv_ab(a, b_dir)
                violations = np.where(conv_check > M * (1 + 1e-6) + 1e-10)[0]
                if len(violations) == 0:
                    break
                # Add most violated
                viol_sorted = violations[np.argsort(conv_check[violarations])
                                         ][::-1][:10] if len(violations) > 10 \
                    else violations
                new_idx = np.unique(np.concatenate([active_idx, viol_sorted]))
                b_dir = solve_lp_direction(a, M, n, new_idx)
                if b_dir is None:
                    break

        if b_dir is None:
            # Fallback: use subgradient direction
            k_star = int(np.argmax(c))
            b_dir = np.zeros(n)
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            if lo <= hi:
                b_dir[lo:hi + 1] = a[k_star - hi:k_star - lo + 1][::-1]
            if b_dir.max() <= 0:
                break

        # Rescale direction
        s_dir = b_dir.sum()
        if s_dir < 1e-15:
            break
        b_dir = b_dir * (S / s_dir)
        b_dir = np.maximum(b_dir, 0)

        # Line search
        t, new_M = line_search_mixture(a, b_dir, S, n)

        if t < 0.005:
            t = 0.05

        # Update
        a_new = (1 - t) * a + t * b_dir
        a = normalize_to_sum(a_new, S)

    return best_a, best_M


def adam_optimize(a_init, n, S, max_time, rng, top_k=10):
    """Adam with simplex projection as fallback/complement."""
    a = normalize_to_sum(a_init, S)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr_init = 0.3 * S / n
    t = 0
    start = time.monotonic()
    best_a = a.copy()
    best_M = fft_conv_self(a).max()
    no_improve = 0

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_conv_self(a)
        cur_M = c.max()
        if cur_M < best_M - 1e-12:
            best_M = cur_M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            a = project_simplex(best_a + rng.normal(0, 0.01 * S / n, n), S)
            no_improve = 0

        top_idx = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        top_vals = c[top_idx]
        total_w = top_vals.sum()
        if total_w > 1e-15:
            weights = top_vals / total_w
            for w_val, k in zip(weights, top_idx):
                k = int(k)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                if lo <= hi:
                    grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = lr_init * (0.5 ** (float(t) / 400.0))
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

        if t % 2000 == 0 and rng.random() < 0.15:
            a = project_simplex(a + rng.normal(0, 0.01 * S / n, n), S)

    return best_a, best_M

def make_initializations(rng, lengths):
    """Generate diverse initializations."""
    inits = []
    for n in lengths:
        # Random
        inits.append(("rand", rng.random(n), n))
        # Smooth sinusoidal
        tt = np.linspace(0, 1, n)
        arr = 1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + \
              0.2 * np.sin(2 * np.pi * 5 * tt + 0.5) + \
              0.1 * np.cos(2 * np.pi * 11 * tt)
        inits.append(("smooth", np.maximum(arr, 0), n))
        # Gaussian bump
        arr = np.exp(-0.5 * ((np.arange(n) - n * 0.5) / (n * 0.25)) ** 2)
        inits.append(("gauss", arr, n))
        # Two bumps
        arr = np.zeros(n)
        arr[int(n * 0.25):int(n * 0.45)] = 1.0
        arr[int(n * 0.55):int(n * 0.75)] = 0.7
        inits.append(("2bump", arr, n))
        # Alternating with decay
        arr = np.ones(n) * (1 + 0.3 * np.cos(2 * np.pi * np.arange(n) / (n / 8.0)))
        inits.append(("alt", np.maximum(arr, 0), n))
    return inits

    if time.monotonic() < deadline - 60:
        time_left = deadline - time.monotonic() - 60
        # Choose lengths to try
        lengths_to_try = [4096, 8192]

            S_n = np.sqrt(2.0 * float(n))

    if global_best_a is not None and time.monotonic() < deadline - 5:
        time_left = deadline - time.monotonic() - 5

        S_best = np.sqrt(2.0 * float(n_best))

        # Very small perturbations + LP
        for trial in range(5):

                break
            seg_time = min(time_left / (5 - trial), 30.0)

            update_best(a_t, f"final_trial_{trial}")
            time_left = deadline - time.monotonic() - 5

    # Final output
    if global_best_a is not None:
        final_a = np.clip(global_best_a, 0, 1000)
        if final_a.sum() < 0.01:
            final_a = np.ones(256)
        final_val = verify_sequence(final_a)
        print(f"\n=== FINAL: {final_val:.6f} (n={len(final_a)}) ===")
    else:
        final_a = np.ones(256)
        print("\n=== FINAL: fallback uniform ===")

    return [float(x) for x in final_a]
