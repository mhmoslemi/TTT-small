    """Compute t*log(sum exp(c/t)) and softmax weights w[k] = exp(c[k]/t)/Z."""
    c_shifted = (c - c.max()) / t
    exp_c = np.exp(c_shifted)
    Z = exp_c.sum()
    val = t * (np.log(Z) + c.max() / t)

    return val, w

    """Gradient of t*log(sum exp(c[k]/t)) w.r.t. a, given weights w.
    
    grad[m] = 2 * sum_{i=0}^{n-1} w[m+i] * a[i]
    Computed via FFT cross-correlation.

    """Convert log-parameters b to sequence a with sum S."""
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted)
    a = exp_b / exp_b.sum() * S
    return a


def grad_wrt_b(a, g_a, S):
    """Chain rule: dF/db[m] = a[m] * (g_a[m] - sum(g_a*a)/S)."""
    correction = np.dot(g_a, a) / S
    return a * (g_a - correction)

    b = b_init.copy().astype(np.float64)
    a = softmax_params_to_a(b, S)
    c = fft_convolve_self(a)
    _, w = soft_max_and_weights(c, t_start)
    best_score = c.max()
    best_b = b.copy()

    v = np.zeros(n, dtype=np.float64)
    t_step = 0
    start = time.monotonic()
    
    # Annealing: linear in log space from t_start to t_end
    log_t_start = np.log(t_start)
    log_t_end = np.log(t_end)
    
    while time.monotonic() - start < max_time:
        t_step += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max_time)
        t = np.exp(log_t_start + frac * (log_t_end - log_t_start))

        g_b = grad_wrt_b(a, g_a, S)

        v = momentum2 * v + (1.0 - momentum2) * (g_b ** 2)

    """Compute t*log(sum exp(c/t)) and normalized weights w[k]."""
    c_max = c.max()
    exp_c = np.exp((c - c_max) / t)
    Z = exp_c.sum()
    val = t * (np.log(Z) + c_max / t)
    w = exp_c / Z
    return val, w


def grad_smooth_max_wrt_a(a, w, t):
    """Gradient of F(a,t) = t*log(sum exp(c[k]/t)) w.r.t. a.
    
    grad[m] = (2/t) * sum_{i=0}^{n-1} w[m+i] * a[i]
    Uses FFT cross-correlation for O(n log n) computation.
    """
    n = len(a)
    w_flip = w[::-1]
    conv = fft_convolve_ab(w_flip, a)
    result = conv[n - 1:2 * n - 1][::-1]
    return (2.0 / t) * result


def softmax_params_to_a(b, S):
    """Convert log-parameters b to sequence a = S * softmax(b)."""
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted)
    a = exp_b / exp_b.sum() * S
    return a


def grad_wrt_b(a, g_a, S):
    """Chain rule for softmax parameterization.
    dF/db[m] = a[m] * (g_a[m] - dot(g_a, a)/S)
    """
    correction = np.dot(g_a, a) / S
    return a * (g_a - correction)


def run_smooth_adam(b_init, n, S, t_start, t_end, max_time, lr, rng,
                    beta1=0.8, beta2=0.999, eps=1e-8, perturb_every=0, perturb_std=0.0):
    """Adam optimizer on b with temperature annealing. Returns (best_b, best_max)."""
    b = b_init.copy().astype(np.float64)
    a = softmax_params_to_a(b, S)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_b = b.copy()
    best_a = a.copy()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    step = 0
    start = time.monotonic()

    log_t_start = np.log(t_start)
    log_t_end = np.log(t_end)

    it_count = 0
    while time.monotonic() - start < max_time:
        step += 1
        it_count += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max_time)
        t = np.exp(log_t_start + frac * (log_t_end - log_t_start))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_b = b.copy()
            best_a = a.copy()

        _, w = soft_max_and_weights(c, t)
        g_a = grad_smooth_max_wrt_a(a, w, t)
        g_b = grad_wrt_b(a, g_a, S)

        m = beta1 * m + (1.0 - beta1) * g_b
        v = beta2 * v + (1.0 - beta2) * (g_b ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        lr_t = lr * (1.0 + 2.0 * frac) ** (-1.0)
        b = b - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = softmax_params_to_a(b, S)

        if perturb_every > 0 and step % perturb_every == 0 and rng.random() < 0.03:
            b += rng.normal(0, perturb_std, n)

    return best_b, best_score, best_a


def try_lbfgs_polish(b_init, n, S, t, max_time, rng):
    """Use scipy L-BFGS for polishing with a fixed temperature."""
    from scipy.optimize import minimize

    state = {'count': 0}

    def objective(b):
        state['count'] += 1
        a = softmax_params_to_a(b, S)
        c = fft_convolve_self(a)
        wsum = c.sum()
        val, _ = soft_max_and_weights(c, t)
        return val

    def gradient(b):
        a = softmax_params_to_a(b, S)
        c = fft_convolve_self(a)
        _, w = soft_max_and_weights(c, t)
        g_a = grad_smooth_max_wrt_a(a, w, t)
        return grad_wrt_b(a, g_a, S)

    best_b = b_init.copy()
    try:
        result = minimize(objective, b_init, jac=gradient, method='L-BFGS-B',
                          options={'maxiter': 500, 'maxfun': 2000,
                                   'ftol': 1e-12, 'gtol': 1e-10,
                                   'iprint': -1, 'maxls': 50})
        if result.fun < objective(b_init):
            best_b = result.x.copy()
    except Exception:
        pass
    return best_b


def propose_candidate(seed=42, budget_s=2400):
    """Search for optimal non-negative sequence minimizing 2n*max_conv/sum^2."""
    rng = np.random.default_rng(seed)
    total_budget = budget_s
    deadline = time.monotonic() + total_budget - 15

    # Get given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).flatten()
        given = np.clip(given, 0, 1000)
    except Exception:
        given = None

    # Try multiple lengths
    lengths_to_try = [8192]
    if given is not None and len(given) != 8192 and len(given) >= 256:
        lengths_to_try.insert(0, len(given))
    if total_budget > 1500:
        lengths_to_try.append(4096)

    global_best = np.inf
    global_a = None

    for n in lengths_to_try:
        if time.monotonic() > deadline - 30:
            break

        S = np.sqrt(2.0 * n)
        n_alloc = min(600.0, 0.65 * (deadline - time.monotonic()))

        # Build initializations for this length
        inits = []

        # From given
        if given is not None and len(given) == n and given.sum() > 1e-6:
            b_given = np.log(given + 1e-10)
            inits.append(("given", b_given))

        # Random
        for i in range(2):
            a_r = rng.random(n) + 0.05
            a_r = a_r * (S / a_r.sum())
            inits.append((f"rand{i}", np.log(a_r)))

        # Asymmetric ramp (concentrated left)
        tt = np.linspace(0, 1, n)
        a_ramp = np.exp(-4.0 * tt) + 0.01
        a_ramp = a_ramp * (S / a_ramp.sum())
        inits.append(("ramp", np.log(a_ramp)))

        # Cosine modulated
        a_cos = 1.0 + 0.5 * np.cos(2 * np.pi * 3 * tt) + 0.3 * np.cos(2 * np.pi * 7 * tt)
        a_cos = np.maximum(a_cos, 0.01)
        a_cos = a_cos * (S / a_cos.sum())
        inits.append(("cos", np.log(a_cos)))

        # Two-hump (asymmetric)
        a_hump = np.exp(-20 * (tt - 0.3) ** 2) + 0.5 * np.exp(-20 * (tt - 0.7) ** 2) + 0.01
        a_hump = a_hump * (S / a_hump.sum())
        inits.append(("hump", np.log(a_hump)))

        per_init = max(3.0, n_alloc / len(inits))

        for name, b_init in inits:
            if time.monotonic() > deadline - 30:
                break
            alloc = min(per_init, deadline - time.monotonic() - 30)
            if alloc < 3.0:
                break

            b_best, score, a_best = run_smooth_adam(
                b_init, n, S,
                t_start=0.5, t_end=0.05,
                max_time=alloc,
                lr=0.03,
                rng=rng,
                perturb_every=3000,
                perturb_std=0.05
            )

            a_check = np.clip(a_best, 0, 1000)
            ver = evaluate_sequence(a_check.tolist())

            print(f"[n={n} {name}] smooth={score:.6f} verified={ver:.6f} t={time.monotonic()-(deadline-n_alloc):.0f}s")

            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()

        # Polish the best for this length
        if len(inits) > 0 and time.monotonic() < deadline - 20:
            # Find the best b for this n (re-run quickly on best)
            # Use the last best we found
            if global_a is not None and len(global_a) == n:
                b_cur = np.log(np.maximum(global_a, 1e-10))
                # L-BFGS polish
                if time.monotonic() < deadline - 10:
                    b_pol = try_lbfgs_polish(b_cur, n, S, t=0.03, max_time=30, rng=rng)
                    a_pol = softmax_params_to_a(b_pol, S)
                    ver_pol = evaluate_sequence(np.clip(a_pol, 0, 1000).tolist())
                    print(f"[n={n} lbfgs_polish] verified={ver_pol:.6f}")
                    if np.isfinite(ver_pol) and ver_pol < global_best:
                        global_best = ver_pol
                        global_a = a_pol.copy()

    # Final polish phase on the global best
    if global_a is not None and time.monotonic() < deadline - 5:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        b_cur = np.log(np.maximum(global_a, 1e-10))

        # Low-temperature Adam polish
        alloc = min(120.0, deadline - time.monotonic() - 5)
        if alloc > 10:
            b_p, sp, a_p = run_smooth_adam(
                b_cur, n, S,
                t_start=0.05, t_end=0.01,
                max_time=alloc,
                lr=0.01,
                rng=rng
            )
            ver_p = evaluate_sequence(np.clip(a_p, 0, 1000).tolist())
            print(f"[final_adam_polish] verified={ver_p:.6f}")
            if np.isfinite(ver_p) and ver_p < global_best:
                global_best = ver_p
                global_a = a_p.copy()

        # L-BFGS polish
        if time.monotonic() < deadline - 5 and global_a is not None:
            b_cur2 = np.log(np.maximum(global_a, 1e-10))
            b_p2 = try_lbfgs_polish(b_cur2, n, S, t=0.02, max_time=60, rng=rng)
            a_p2 = softmax_params_to_a(b_p2, S)
            ver_p2 = evaluate_sequence(np.clip(a_p2, 0, 1000).tolist())
            print(f"[final_lbfgs] verified={ver_p2:.6f}")
            if np.isfinite(ver_p2) and ver_p2 < global_best:
                global_best = ver_p2
                global_a = a_p2.copy()

    # Return
    if global_a is None:
        n = 8192
        S = np.sqrt(2.0 * n)
        global_a = np.full(n, S / n)

    global_a = np.clip(global_a, 0, 1000)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final = evaluate_sequence([float(x) for x in global_a])
    print(f"=== FINAL ANSWER: {final:.6f} (n={len(global_a)}) ===")
    return [float(x) for x in global_a]
