```
    n=v.size
    if n==0: return v.copy()

    if not np.all(np.isfinite(v)):
       v[~np.isfinite(v)] = 0.0
    w = np.sort(v)[::-1]

    lo=1; hi=n
    # Need handle n=1: theta = v[0]-S, out max? Actually projection of single variable to sum S nonneg is S if S>=0.

       if cum[mid-1] - float(mid)*w[mid-1] < S:
```

```python
"""
Search for non-negative step-function heights minimizing
    2*n*max(conv(a,a)) / sum(a)^2.

The normalizing sum is fixed to 1 during search, so the objective becomes
    2*n*max(conv(a,a)).

The search uses:
  - FFT-based convolution and exact gradients of a log-sum-exp surrogate.
  - Projected gradient descent onto the non-negative simplex.
  - Gamma (temperature) annealing from a smooth surrogate to the max norm.
  - A final top-k peak averaging mode to polish the exact maximum.
  - Multiple initializations, including resampling the provided sequence
    to several longer lengths.
"""

import numpy as np
import time


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _fft_len(n):
    return _next_pow2(max(1, 2 * int(n) - 1))


def _normalize(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    out = arr / s
    out = np.maximum(out, 1e-18)
    out /= float(out.sum())
    return out


def _init_uniform(n):
    return np.full(max(1, int(n)), 1.0 / max(1, int(n)), dtype=np.float64)


def _init_chirp(n, seed):
    n = max(2, int(n))
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64) - 0.5 * (n - 1)
    a = 1.0 + 0.85 * np.cos(np.pi * t * t / float(n))
    a += 0.08 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0.0:
        return _init_uniform(n)
    a /= s
    return a


def _init_random(n, seed):
    n = max(2, int(n))
    rng = np.random.default_rng(seed)
    a = rng.exponential(1.0, size=n)
    a = np.maximum(a, 1e-18)
    a /= float(a.sum())
    return a


def _init_cosine(n, seed):
    n = max(2, int(n))
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    a = np.ones(n, dtype=np.float64)
    for m in (1, 2, 3, 5):
        amp = rng.uniform(0.05, 0.35) / float(m)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * x + ph)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0.0:
        return _init_uniform(n)
    a /= s
    return a


def _perturb_sequence(a, seed, rel_sigma=0.1):
    a = _normalize(a)
    n = a.size
    rng = np.random.default_rng(seed)
    scale = float(rel_sigma) * float(a.mean())
    if scale <= 0.0:
        return a
    b = a + rng.normal(0.0, scale, size=n)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s <= 0.0:
        return a
    b /= s
    return b


def _resample_nonneg(a, n):
    a = _normalize(a)
    n = max(2, int(n))
    if a.size == 0:
        return _init_uniform(n)
    if a.size == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if a.size == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 0.0:
        return _init_uniform(n)
    out /= s
    return out


def _project_simplex(v, S=1.0):
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    if n == 1:
        return np.array([max(0.0, float(S))], dtype=np.float64)

    s = float(v.sum())
    minv = float(v.min())

    # Fast path: already nonnegative and very close to the simplex.
    if minv >= 0.0:
        theta = (s - S) / float(n)
        out = v - theta
        if float(out.min()) >= -1e-14:
            out = np.maximum(out, 0.0)
            s2 = float(out.sum())
            if s2 > 0.0:
                out *= S / s2
            else:
                out = np.full(n, S / float(n), dtype=np.float64)
            return out

    # Exact projection by sorting.
    w = np.sort(v)[::-1]
    cum = np.cumsum(w)

    lo = 1
    hi = n
    while lo < hi:
        mid = (lo + hi + 1) // 2
        # Want largest k such that theta_k < w[k-1]:
        # (cum[k-1]-S)/k < w[k-1]  <=>  cum[k-1] - k*w[k-1] < S
        if cum[mid - 1] - float(mid) * w[mid - 1] < S:
            lo = mid
        else:
            hi = mid - 1

    k = lo
    theta = (cum[k - 1] - S) / float(k)
    out = np.maximum(v - theta, 0.0)
    s2 = float(out.sum())
    if s2 > 0.0:
        out *= S / s2
    else:
        out = np.full(n, S / float(n), dtype=np.float64)
    return out


def _conv_square_fft(a, M):
    A = np.fft.rfft(a, M)
    c = np.fft.irfft(A * A, M)
    L = max(1, 2 * int(a.size) - 1)
    c = c[:L]
    c = np.maximum(c, 0.0)
    return c, A


def _corr_q_a_fft(q, a, A, M):
    # Returns r[j] = sum_i a[i] * q[i+j], for j=0..n-1.
    Q = np.fft.rfft(q, M)
    r = np.fft.irfft(Q * np.conj(A), M)
    r = r[:a.size]
    r = np.where(np.isfinite(r), r, 0.0)
    return r


def _lse_q(c, gamma, n, topk=None):
    L = c.size
    if L <= 0:
        return np.zeros(0, dtype=np.float64)

    # Top-k / exact-max mode.
    if topk is not None or (gamma is not None and np.isinf(gamma)):
        if topk is None:
            topk = max(3, min(15, int(n) // 100))
        topk = int(topk)
        if topk <= 1:
            q = np.zeros(L, dtype=np.float64)
            q[int(np.argmax(c))] = 1.0
            return q
        if topk >= L:
            return np.full(L, 1.0 / float(L), dtype=np.float64)
        idx = np.argpartition(c, -topk)[-topk:]
        vals = c[idx]
        z = vals - float(vals.max())
        z = np.minimum(z, 0.0)
        w = np.exp(z)
        s = float(w.sum())
        if not np.isfinite(s) or s <= 0.0:
            w = np.ones(topk, dtype=np.float64)
            s = float(w.sum())
        q = np.zeros(L, dtype=np.float64)
        q[idx] = w / s
        return q

    # Smooth log-sum-exp probabilities.
    if gamma is None or gamma <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)

    cmax = float(c.max())
    z = (c - cmax) * float(gamma) * float(n)
    z = np.minimum(z, 0.0)
    q = np.exp(z)
    s = float(q.sum())
    if not np.isfinite(s) or s <= 0.0:
        q = np.zeros(L, dtype=np.float64)
        q[int(np.argmax(c))] = 1.0
    else:
        q /= s
    return q


def _lse_value(c, gamma, n):
    cmax = float(c.max())
    if gamma is None or gamma <= 0.0 or np.isinf(gamma):
        return cmax
    z = (c - cmax) * float(gamma) * float(n)
    z = np.minimum(z, 0.0)
    s = float(np.sum(np.exp(z)))
    if not np.isfinite(s) or s <= 0.0:
        return cmax
    return cmax + (1.0 / (float(gamma) * float(n))) * float(np.log(s))


def _run_fixed_gamma(a, gamma, budget_s, M, initial_lr=0.6, topk=None, verbose=False):
    n = int(a.size)
    a = _normalize(a)

    c, A = _conv_square_fft(a, M)
    cmax = float(c.max())
    F = _lse_value(c, gamma, n)

    best_a = a.copy()
    best_mc = cmax
    best_F = F

    lr = float(initial_lr)
    min_lr = 1e-5
    max_lr = 30.0
    fail = 0
    step = 0
    top_mode = (topk is not None) or (gamma is not None and np.isinf(gamma))

    t0 = time.monotonic()
    limit_t = t0 + max(0.0, float(budget_s))

    while time.monotonic() < limit_t:
        q = _lse_q(c, gamma, n, topk)
        if q is None or float(q.max()) <= 0.0:
            break

        r = _corr_q_a_fft(q, a, A, M)
        g = 2.0 * r
        g = g - float(g.mean())

        gnorm = float(np.linalg.norm(g))
        if not np.isfinite(gnorm) or gnorm <= 1e-30:
            break

        max_abs_g = float(np.max(np.abs(g)))
        if max_abs_g <= 1e-30:
            break

        # Prevent a single coordinate from taking a destructive absolute step.
        max_step = 10.0 * float(a.mean())
        lr_eff = min(lr, max_step / max_abs_g)
        if lr_eff <= 0.0:
            break

        v = a - lr_eff * g
        a_new = _project_simplex(v, 1.0)
        a_new = np.maximum(a_new, 1e-15)
        s2 = float(a_new.sum())
        if s2 <= 0.0:
            break
        a_new /= s2

        c_new, A_new = _conv_square_fft(a_new, M)
        cmax_new = float(c_new.max())
        F_new = _lse_value(c_new, gamma, n)

        if not np.isfinite(cmax_new) or not np.isfinite(F_new):
            lr = max(min_lr, 0.5 * lr)
            fail += 1
            if fail > 150:
                break
            continue

        tolc = 1e-12
        if top_mode:
            improve = cmax_new < cmax - tolc
        else:
            tolf = 1e-12 * max(1.0, abs(F))
            improve = (cmax_new < cmax - tolc) or (F_new < F - tolf)

        prev_cmax = cmax
        if improve:
            a = a_new
            c = c_new
            A = A_new
            cmax = cmax_new
            F = F_new

            if cmax < best_mc:
                best_mc = cmax
                best_a = a.copy()
            if F < best_F:
                best_F = F

            if cmax < prev_cmax:
                lr = min(max_lr, lr * 1.08)
            elif (not top_mode) and F_new < F:
                lr = min(max_lr, lr * 1.01)
            else:
                lr = max(min_lr, lr * 0.995)
            fail = 0
        else:
            lr = max(min_lr, 0.5 * lr)
            fail += 1
            if fail > 180 or lr <= min_lr * 1.2:
                break

        step += 1
        if verbose and (step % 200 == 0):
            gam_str = "top" if top_mode else str(gamma)
            print(f"        gamma={gam_str} step={step} mc={cmax:.9f} score={2*n*cmax:.6f} lr={lr:.3g}")

    return a, best_a, best_mc, best_F, cmax


def _optimize_sequence(a0, budget_s, schedule=None, verbose=True):
    a0 = _normalize(a0)
    n = int(a0.size)
    M = _fft_len(n)

    if schedule is None:
        schedule = [
            (1.0, 0.08),
            (3.0, 0.14),
            (8.0, 0.20),
            (20.0, 0.20),
            (50.0, 0.20),
            (120.0, 0.10),
            (np.inf, 0.08),
        ]

    c, _ = _conv_square_fft(a0, M)
    init_mc = float(c.max())
    best_a = a0.copy()
    best_mc = init_mc

    if budget_s <= 0.0:
        return best_a, 2.0 * n * best_mc

    total_frac = 0.0
    for _, f in schedule:
        total_frac += float(f)
    if total_frac <= 0.0:
        return best_a, 2.0 * n * best_mc

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s))

    for gamma, frac in schedule:
        tb = float(frac) / total_frac * float(budget_s)
        tb = min(tb, max(0.0, deadline - time.monotonic() - 1.0))
        if tb < 1.0:
            continue

        top_direct = np.isinf(gamma)
        topk = None

        a_end, a_best_blk, mc_best_blk, F_best_blk, mc_end = _run_fixed_gamma(
            best_a, gamma, tb, M,
            initial_lr=0.6,
            topk=topk,
            verbose=verbose
        )

        if mc_best_blk < best_mc:
            best_mc = mc_best_blk
            best_a = a_best_blk.copy()

        # Carry forward a point that is either the best maximum or nearly so.
        if abs(mc_end - mc_best_blk) <= 1e-8 * max(1.0, mc_best_blk):
            best_a_for_next = a_end
        else:
            best_a_for_next = a_best_blk

        if mc_end < best_mc:
            best_mc = mc_end
            best_a = best_a_for_next.copy()

        if verbose:
            gname = "top" if top_direct else str(gamma)
            print(f"      block gamma={gname} done: current_mc={mc_end:.9f}, best_mc={best_mc:.9f}")

        best_a = best_a_for_next

    return best_a, 2.0 * n * best_mc


def _create_candidates(prov, seed):
    cands = []

    if prov is not None:
        p = _normalize(prov)
        if p.size > 12000:
            p = _resample_nonneg(p, 12000)

        cands.append(("provided", p.copy()))
        for n in (6000, 8000, 12000):
            if n != p.size:
                cands.append((f"resample_{n}", _resample_nonneg(p, n)))
            else:
                cands.append((f"resample_{n}", p.copy()))

    cands.append(("chirp_8000", _init_chirp(8000, seed)))
    cands.append(("cosine_8000", _init_cosine(8000, seed + 7)))
    cands.append(("random_8000", _init_random(8000, seed + 13)))

    # Keep the list modest but diverse.
    if len(cands) > 6:
        cands = cands[:6]

    # Remove exact duplicate names/lengths if any, but keep first occurrence.
    seen = {}
    out = []
    for name, arr in cands:
        key = (name, int(arr.size))
        if key in seen:
            continue
        seen[key] = True
        out.append((name, arr))
    return out


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search entry point.
    Returns a list of non-negative floats achieving the best known score
    found within the allotted time.
    """
    total = float(budget_s if budget_s is not None else 2400.0)
    if total <= 0.0:
        return [1.0]

    t_start = time.monotonic()
    deadline = t_start + max(30.0, total - 30.0)

    # Load provided sequence if available.
    prov = None
    try:
        raw = height_sequence_1
        arr = np.asarray(raw, dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        if arr.size > 0 and float(arr.sum()) > 0.01:
            prov = _normalize(arr)
    except Exception:
        prov = None

    if prov is None:
        prov = _init_chirp(3000, seed)

    quick_schedule = [
        (2.0, 0.25),
        (10.0, 0.45),
        (np.inf, 0.30),
    ]

    candidates = _create_candidates(prov, seed)

    best_seq = None
    best_score = np.inf

    # Quick scouting phase.
    print(f"[search] starting; deadline in {deadline - time.monotonic():.1f}s")
    print(f"[search] candidates: {len(candidates)}")

    results = []
    for idx, (name, arr) in enumerate(candidates):
        now = time.monotonic()
        if now > deadline - 60.0:
            break
        remaining_cands = max(1, len(candidates) - idx)
        qt = 25.0
        qt = min(qt, max(5.0, (deadline - now) / float(remaining_cands) * 0.75))
        if qt < 3.0:
            break

        print(f"[quick] {name} n={arr.size} budget={qt:.1f}s")
        a_opt, sc_opt = _optimize_sequence(arr, qt, schedule=quick_schedule, verbose=False)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
        results.append((float(sc_opt), idx, name, a_opt))
        print(f"[quick] {name} -> {sc_opt:.6f}")

    if not results:
        a_opt, sc_opt = _optimize_sequence(prov, max(1.0, deadline - time.monotonic() - 20.0), verbose=True)
        best_seq = a_opt
        best_score = sc_opt

    # Sort scouting results by score.
    results.sort()

    # Prefer one larger candidate for the first main run, if available.
    large = [r for r in results if r[3].size >= 5000]
    if large:
        large.sort()
        main1 = large[0]
    else:
        main1 = results[0]

    others = [r for r in results if r[1] != main1[1]]
    others.sort()
    main2 = others[0] if others else None

    rem = max(0.0, deadline - time.monotonic())
    main_schedule = [
        (1.0, 0.08),
        (3.0, 0.14),
        (8.0, 0.20),
        (20.0, 0.20),
        (50.0, 0.20),
        (120.0, 0.10),
        (np.inf, 0.08),
    ]

    if rem <= 30.0:
        pass
    else:
        if main2 is not None:
            alloc1 = 0.62 * rem
            alloc2 = 0.27 * rem
            final_alloc = rem - alloc1 - alloc2
            if final_alloc < 0.0:
                final_alloc = 0.0
                alloc1 = 0.70 * rem
                alloc2 = 0.30 * rem
        else:
            final_alloc = 0.20 * rem
            alloc1 = rem - final_alloc
            alloc2 = 0.0

        # Main run 1.
        now = time.monotonic()
        tb1 = min(alloc1, max(1.0, deadline - now - 30.0))
        if tb1 > 5.0:
            print(f"[main1] {main1[2]} n={main1[3].size} budget={tb1:.1f}s")
            a_opt, sc_opt = _optimize_sequence(main1[3], tb1, schedule=main_schedule, verbose=True)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
            print(f"[main1] best score now {best_score:.6f}")

        # Main run 2.
        now = time.monotonic()
        tb2 = min(alloc2, max(1.0, deadline - now - 30.0))
        if main2 is not None and tb2 > 5.0:
            print(f"[main2] {main2[2]} n={main2[3].size} budget={tb2:.1f}s")
            a_opt, sc_opt = _optimize_sequence(main2[3], tb2, schedule=main_schedule, verbose=True)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
            print(f"[main2] best score now {best_score:.6f}")

    # Optional perturbation restarts near the end.
    now = time.monotonic()
    if best_seq is not None and (deadline - now) > 220.0:
        polish_schedule = [
            (5.0, 0.15),
            (20.0, 0.20),
            (60.0, 0.25),
            (np.inf, 0.40),
        ]
        for relsig in (0.05, 0.12):
            now = time.monotonic()
            if (deadline - now) < 100.0:
                break
            tb = min(70.0, max(20.0, 0.18 * (deadline - now)))
            a_var = _perturb_sequence(best_seq, seed + 100 + int(relsig * 1000), relsig)
            print(f"[perturb] rel_sigma={relsig} n={a_var.size} budget={tb:.1f}s")
            a_opt, sc_opt = _optimize_sequence(a_var, tb, schedule=polish_schedule, verbose=False)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[perturb] improved to {best_score:.6f}")
            else:
                print(f"[perturb] no improvement: {sc_opt:.6f}")

    # Final polish of the best sequence.
    now = time.monotonic()
    if best_seq is not None and (deadline - now) > 60.0:
        tb = max(1.0, deadline - now - 10.0)
        polish_schedule = [
            (2.0, 0.10),
            (10.0, 0.20),
            (30.0, 0.25),
            (80.0, 0.25),
            (np.inf, 0.20),
        ]
        print(f"[final polish] n={best_seq.size} budget={tb:.1f}s")
        a_opt, sc_opt = _optimize_sequence(best_seq, tb, schedule=polish_schedule, verbose=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
        print(f"[final polish] best score now {best_score:.6f}")

    # Fallbacks.
    if best_seq is None:
        best_seq = prov.copy()
        try:
            c, _ = _conv_square_fft(best_seq, _fft_len(best_seq.size))
            best_score = 2.0 * best_seq.size * float(c.max())
        except Exception:
            best_score = np.inf

    # Optional exact evaluation for logging. The judged value may differ by
    # tiny FFT rounding if not computed here, but this only affects prints.
    try:
        final_list = [float(x) for x in best_seq]
        if time.monotonic() < (t_start + max(30.0, total - 5.0)) and len(final_list) <= 12000:
            exact = evaluate_sequence(final_list)
            print(f"[exact evaluate_sequence] {exact:.9f}")
        else:
            print(f"[approx score] {best_score:.9f}")
    except Exception as e:
        print(f"[exact eval skipped] {repr(e)}")

    # Ensure a valid Python list is returned.
    out = [float(max(0.0, x)) for x in np.asarray(best_seq, dtype=np.float64).ravel()]
    if not out:
        out = [1.0]
    s = float(np.sum(out))
    if s <= 0.0:
        out = [1.0]
    else:
        out = [x / s for x in out]

    return out
```
