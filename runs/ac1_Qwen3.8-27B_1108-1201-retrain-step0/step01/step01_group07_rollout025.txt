    # a is the current sequence of length n
    N = 2 * n - 1  # or 2*n for FFT efficiency
    # Pad to length N
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    # We want |A|^2 to be flat, i.e., |A| to be flat
    # Set |A| to constant = sqrt(sum(|A|^2) / N) = sqrt(sum(a^2) / N) * N... 
    # Actually, sum(|A|^2) = N * sum(a_pad^2) by Parseval
    # We want |A_k| = C for all k, where C = sqrt(sum(|A|^2) / N)
    C = np.sqrt(np.sum(np.abs(A)**2) / N)
    A_new = C * A / np.maximum(np.abs(A), 1e-12)  # maintain phase
    a_new = np.fft.irfft(A_new, N)
    # Project to non-negative, supported on [0, n-1]
    a_new = np.maximum(a_new[:n], 0)
    # Rescale to maintain sum
    S = np.sum(a)  # original sum
    if np.sum(a_new) > 1e-12:
        a_new = a_new * S / np.sum(a_new)
    return a_new

where we use the convention that a[m] = 0 for m outside [0,n).

So the gradient is:
`grad[i] = sum_k q[k] * (a[k-i] * I(0<=k-i<n) + a[i-k] * I(0<=i-k<n))`

= `sum_k q[k] * a[k-i] * I(0<=k-i<n)` + `sum_k q[k] * a[i-k] * I(0<=i-k<n)`

The first sum: let m = k-i, so k = m+i. Sum over m from 0 to n-1: q[m+i] * a[m]. This is a cross-correlation of q and a.
The second sum: let m = i-k, so k = i-m. Sum over m from 0 to n-1: q[i-m] * a[m]. This is also a cross-correlation (with the other direction).

Both of these can be computed efficiently with FFT!

Actually, I realize the first sum is `sum_m q[m+i] * a[m]` for m=0..min(n-1, 2n-2-i), and the second is `sum_m q[i-m] * a[m]` for m=max(0,i-(2n-2))..min(n-1,i).

These are correlations that can be computed with FFT in O(n log n).

Great, so the full gradient computation is O(n log n) using FFT. This makes the gradient descent much faster!

def project_simplex(a, S):
    """Project a onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(a.shape, S / a.size)
    if np.isclose(s, S, atol=1e-12):
        return a
    # Bisection for the threshold

    for _ in range(60):
        mid = (lo + hi) / 2
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    res = np.maximum(a - t, 0.0)
    res[res.argmax()] += S - res.sum()
    return res

def fft_conv(a, n):
    """Compute linear convolution conv(a, a) using FFT."""
    N = 1
    while N < 2*n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[:2*n-1]

def gs_step(a, n, N, alpha=1.0):
    """One GS/Fienup iteration."""
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    # Target: flat magnitude
    target_mag = np.sqrt(np.sum(mag**2) / N)  # preserve energy
    new_mag = (1 - alpha) * mag + alpha * target_mag
    new_mag = np.maximum(new_mag, 1e-15)
    phase = A / np.maximum(mag, 1e-15)
    A_new = new_mag * phase
    a_new = np.fft.irfft(A_new, N)
    # Project to non-negative, supported on [0, n)
    a_new = np.maximum(a_new[:n], 0.0)
    return a_new

def run_pgd(a0, n, tau_hi, tau_lo, lr, t_budget, momentum=0.9):
    """Projected gradient descent with momentum."""
    S = np.sqrt(2.0 * n)
    a = project_simplex(a0 / max(a0.sum(), 1e-12) * S, S)
    v = np.zeros_like(a)  # momentum velocity
    best = a.copy()
    best_mc = fft_conv(a, n).max()
    best_score = evaluate_sequence(best.tolist())
    
    t0 = time.monotonic()
    lr_current = lr
    step = 0
    stale = 0
    
    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / t_budget)
        tau = tau_hi * (tau_lo / tau_hi) ** frac
        
        obj, grad, c = softmax_objective_grad(a, n, tau)
        
        # Nesterov momentum
        v = momentum * v + grad
        a_new = a - lr_current * v
