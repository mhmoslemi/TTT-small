    # A[k, j] = f[k-j] if 0<=k-j<n, else 0

    b = np.full(2 * n - 1, M)

            best_score = best_obj_local
            best = a.copy()

    return best, best_score


def subgradient_optimize_fn(a0, deadline, max_iters=3000):
    """Subgradient descent with cosine annealing."""
    n = len(a0)
    S_t = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    a = normalize_seq(a, S_t)
    best = a.copy()
    best_score = compute_obj(a)

    lr_init = 0.15
    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * it / max_iters))
        lr = max(lr, 0.003)

        a = subgradient_step_fn(a, lr)

        if (it + 1) % 100 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    return best, best_score


def combined_optimize_fn(a0, deadline):
    """Full pipeline: subgradient then LP refinement."""
    n = len(a0)
    # Phase 1: Subgradient
    leftover = deadline - time.monotonic()
    grad_time = min(leftover * 0.4, 120.0)
    a, score = subgradient_optimize_fn(a0, time.monotonic() + grad_time,
                                       max_iters=2000)
    best = a.copy()
    best_score = score

    # Phase 2: LP refinement
    if n <= 2000:
        seq, lp_score = lp_refine_fn(best, deadline, max_iters=80)
        if lp_score < best_score:
            best_score = lp_score
            best = seq

    return best, best_score


def coordinate_polish_fn(a, deadline, max_rounds=50):
    """Coordinate descent polishing."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline - 5.0:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline - 5.0:
                break
            old_val = a[j]
            for scale in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * scale
                if new_val < 0:
                    new_val = 0.0
                if new_val > 1000.0:
                    new_val = 1000.0
                a[j] = new_val
                s = compute_obj(a)
                if s < best_score:
                    best_score = s
                    best = a.copy()
                    improved = True
                else:
                    a[j] = old_val
            a[j] = old_val
        if not improved:
            break

    return best, best_score


def generate_starts(seed):
    """Generate diverse starting sequences."""
    rng = np.random.default_rng(seed)
    starts = []

    # Thue-Morse based
    for n in [256, 384, 512, 640, 768, 1024]:
        tm = thue_morse_seq(n)
        starts.append((n, tm + 0.3, 'thue_morse'))
        starts.append((n, 1.0 - tm + 0.3, 'thue_morse_inv'))

    # Structured: sinusoidal
    for n in [256, 384, 512, 640]:
        i = np.arange(n, dtype=np.float64)
        for freq in [1, 3, 5, 7]:
            sig = 0.5 + 0.4 * np.cos(2.0 * np.pi * freq * i / n)
            starts.append((n, sig, f'sine_f{freq}'))

    # Bimodal Gaussian
    for n in [384, 512, 640]:
        i = np.arange(n, dtype=np.float64)
        c1, c2 = n * 0.3, n * 0.7
        sig = np.exp(-0.5 * ((i - c1) / (n * 0.12)) ** 2) + \
              np.exp(-0.5 * ((i - c2) / (n * 0.12)) ** 2) + 0.05
        starts.append((n, sig, 'bimodal'))

    # Random smooth
    for n in [256, 512, 768]:
        raw = rng.random(n)
        # Smooth via cumulative sum difference
        smoothed = np.convolve(raw, np.ones(5) / 5.0, mode='same')
        starts.append((n, smoothed + 0.2, 'smooth_rand'))

    # Pure random
    for n in [256, 512]:
        starts.append((n, rng.random(n) * 0.5 + 0.25, 'random'))

    # Asymmetric ramp (exploits asymmetry to reduce center conv)
    for n in [384, 512, 640]:
        i = np.arange(n, dtype=np.float64)
        ramp = np.exp(-i / (n * 0.5)) + 0.1
        starts.append((n, ramp, 'ramp'))
        ramp2 = np.exp(-(n - 1 - i) / (n * 0.5)) + 0.1
        starts.append((n, ramp2, 'ramp_rev'))

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Explores diverse initializations with combined
    subgradient + LP optimization, followed by coordinate polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Try to load previous best
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size >= 16 and np.all(np.isfinite(prev)) and np.all(prev >= 0):
            # Use previous best as primary starting point
            n_prev = len(prev)
            a_start = prev.copy()
            print(f"Loading previous best: n={n_prev}")

            # Optimize from previous best first
            t0 = time.monotonic()
            seq, score = combined_optimize_fn(a_start, deadline)
            elapsed = time.monotonic() - t0
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
            print(f"  From previous: score={score:.6f}, elapsed={elapsed:.1f}s")

            # Also try truncated and extended versions
            for trim_frac in [0.75, 0.6]:
                L = int(n_prev * trim_frac)
                if L >= 64 and time.monotonic() < deadline - 120:
                    a_trim = prev[:L].copy()
                    seq2, score2 = combined_optimize_fn(a_trim, deadline)
                    if score2 < best_score:
                        best_score = score2
                        best_seq = seq2.copy()
                        print(f"  From trimmed (L={L}): score={score2:.6f}")
    except NameError:
        pass

    # Generate diverse starts
    starts = generate_starts(seed)

    # Sort by length (shorter first for quick wins)
    starts.sort(key=lambda x: x[0])

    # Remove duplicates (same length and type)
    seen = set()
    filtered_starts = []
    for s in starts:
        key = (s[0], s[2])
        if key not in seen:
            seen.add(key)
            filtered_starts.append(s)
    starts = filtered_starts

    print(f"\nExploring {len(starts)} diverse starts...")

    for idx, (n, a0, name) in enumerate(starts):
        if time.monotonic() > deadline - 90.0:
            break
        # Skip if same length as previous best (already tried)
        if best_seq is not None and n == len(best_seq) and 'thue' not in name:
            # Still try different types for same length
            pass

        t0 = time.monotonic()
        seq, score = combined_optimize_fn(a0, deadline)
        elapsed = time.monotonic() - t0
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  [{idx}] n={n}, {name}: NEW BEST score={score:.6f} "
                  f"(elapsed={elapsed:.1f}s)")
        else:
            print(f"  [{idx}] n={n}, {name}: score={score:.6f} "
                  f"(elapsed={elapsed:.1f}s, best={best_score:.6f})")

        # If we're getting close to deadline, only try short sequences
        if time.monotonic() > deadline - 300.0 and n > 384:
            # Skip remaining long ones
            continue

    # Final polish phase
    if best_seq is not None and time.monotonic() < deadline - 30.0:
        print(f"\nFinal polish phase (current best={best_score:.6f})...")

        # Coordinate polish
        remaining = deadline - time.monotonic()
        polish_time = min(remaining * 0.4, 120.0)
        seq, score = coordinate_polish_fn(best_seq, time.monotonic() + polish_time)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  After coordinate polish: {best_score:.6f}")

        # One more LP pass
        if time.monotonic() < deadline - 20.0 and len(best_seq) <= 2000:
            seq, score = lp_refine_fn(best_seq, deadline - 5.0, max_iters=30)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  After final LP: {best_score:.6f}")

        # Final coordinate polish
        if time.monotonic() < deadline - 10.0:
            seq, score = coordinate_polish_fn(best_seq, deadline - 3.0, max_rounds=20)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  After final polish: {best_score:.6f}")

    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.6f} ===")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    print("Fallback: returning uniform sequence")
    return [1.0] * 512
