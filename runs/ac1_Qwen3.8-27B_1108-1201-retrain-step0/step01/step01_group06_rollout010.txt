    # c = ifft(fft(dx[::-1], N) * fft(a, N)) ... no, that's not right either.
    
    # Let me think again. The linear convolution (c * a)[j] = sum_k c[k] * a[j-k]
    # I want: sum_k dx[k] * a[k-j] = sum_k dx[k] * a[-(j-k)] = (dx * a_rev)(j) where a_rev[m] = a[-m]
    # = sum_k dx[k] * a_rev[j-k] where a_rev[m] = a[-m] = a[|m|]... no.
    
    # OK let me define it differently. Let's compute:
    # h[j] = sum_k dx[k] * a[k-j] for j in [0, n-1]
    # 
    # This is equivalent to: h = cross_correlation(dx, a) evaluated at lags 0 to n-1
    # where cross_correlation(dx, a)[j] = sum_k dx[k+j] * a[k] (standard definition)
    # 
    # Wait: sum_k dx[k] * a[k-j] = sum_m dx[m+j] * a[m] (letting m=k-j)
    # = cross_correlation(dx, a)[j] in the standard definition
    # where cross_correlation(f, g)[j] = sum_m f[m+j] * g[m]
    
    # Using FFT: cross_correlation(dx, a) = ifft(conj(fft(dx, N)) * fft(a, N))[n-1 : n-1+n]
    # Actually, the standard formula is:
    # R(j) = sum_m f[m] * g[m-j] = ifft(fft(f) * conj(fft(g))) ... 
    
    # I'll just use np.correlate for correctness (it's fast enough for n < 8192):
    # np.correlate(dx, a, 'full') gives R[k] = sum_m dx[m] * a[m+k-(n-1)] ... 
    
    # Let me just test: np.correlate([1,2,3], [4,5], 'full')
    # = [1*4, 1*5+2*4, 2*5+3*4, 3*5] = [4, 13, 22, 15]
    # Standard: R[k] = sum_m f[m] * g[m-k] ... 
    
    # I'll just use the direct method since it's clearest:
    # h[j] = sum_{m=0}^{n-1} dx[m+j] * a[m] for j in [0, n-1]
    # where dx has length 2n-1 and dx[k]=0 for k<0 or k>=2n-1
    
    # This can be done with a loop or with sliding dot product.
    # For n up to ~4096, even a simple loop takes ~4096*4096 = 16M operations which is fine.
    # But let's use scipy or numpy for efficiency.
    
    # Using sliding_window_view:
    if n <= 2048:
        # Direct computation using matrix operations
        # h[j] = dot(dx[j:j+n], a)
        dx_ext = np.zeros(n + 2*n - 1)  # extra space
        dx_ext[:2*n-1] = dx
        # h[j] = sum_{m=0}^{n-1} dx_ext[j+m] * a[m]
        # This is a "moving dot product"
        # Can be done with convolution: h = convolve(dx_ext[0:3n-2], a)[n-1 : 3n-2-n+1]... 
        # Actually: h[j] = sum_m dx_ext[j+m]*a[m] = (a_reversed * dx_ext)[j+n-1] where * is linear conv
        a_rev = a[::-1]
        h_full = np.convolve(dx_ext, a_rev, 'full')
        h = h_full[n-1 : n-1+n]
    else:
        # Use FFT for larger n
        N_fft = 1 << (3 * n).bit_length()
        F_dx = np.fft.rfft(dx, N_fft)
        F_a = np.fft.rfft(a, N_fft)
        # h[j] = sum_m dx[m+j]*a[m] = cross-correlation at lag j
        # = ifft(conj(F_dx) * F_a) ... let me verify
        # Ifft(conj(F_dx)*F_a)[k] = sum_m dx[m]*a[m-k] (circular cross-correlation)
        # For linear: we need N_fft >= (2n-1) + n - 1 = 3n-2
        c = np.fft.irfft(np.conj(F_dx) * F_a, N_fft)
        h = c[:n]  # lags 0 to n-1
        # Hmm, I need to be careful about the lag indexing.
        # Actually for the circular case with sufficient zero-padding:
        # c[j] = sum_m dx[m] * a[(m-j) mod N_fft]
        # For 0 <= j < n and with enough padding, this equals sum_m dx[m] * a[m-j] (linear)
        # But I want h[j] = sum_m dx[m+j]*a[m] = sum_k dx[k]*a[k-j] (letting k=m+j)
        # = c[-j] in the circular sense = c[N_fft-j] for j>0
        # Hmm, this is getting confusing. Let me just use the convolve approach for all sizes.
        h_full = np.convolve(dx_ext, a_rev, 'full')
        h = h_full[n-1 : n-1+n]
    
    grad_a = 2.0 * h
    # Project onto tangent space (sum(a)=1): subtract mean
    grad_a = grad_a - grad_a.mean()
    
    obj_val = 2.0 * n * S
    return obj_val, grad_a


def mirror_descent_step(a, t, eta):
    """One step of mirror descent on the simplex using soft-max smoothed objective."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Soft-max weights
    m = conv.max()
    if m < 1e-15:
        return a
    
    scaled = t * conv / m
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w /= w_sum
    
    # Gradient computation (same as above but inline for speed)
    # h[j] = sum_{m=0}^{n-1} w[m+j] * a[m]
    # Using the convolution trick
    a_rev = a[::-1]
    h_full = np.convolve(w, a_rev, 'full')
    h = h_full[n-1 : n-1+n]
    
    grad = 2.0 * h
    grad = grad - grad.mean()  # Project to tangent of simplex
    
    # Multiplicative update
    log_factor = -eta * grad
    log_factor -= log_factor.max()
    a_new = a * np.exp(log_factor)
    a_new = np.maximum(a_new, 0.0)
    a_sum = a_new.sum()
    if a_sum < 1e-12:
        a_new = np.ones(n) / n
    else:
        a_new /= a_sum
    
    return a_new


def optimize_sequence(a0, time_limit, t_start=50.0, t_end=500.0, eta_start=0.5, eta_end=0.05):
    """
    Optimize using mirror descent with adaptive temperature.
    t increases over time (sharpening the max approximation).
    eta decreases over time (smaller steps for fine-tuning).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a_sum
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    patience = 0
    max_patience = 100
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Adaptive temperature: increase over time
        current_t = t_start * (t_end / t_start) ** progress
        # Adaptive eta: decrease over time
        current_eta = eta_start * (eta_end / eta_start) ** progress
        
        a = mirror_descent_step(a, current_t, current_eta)
        
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            patience = 0
            # Perturb to escape local minimum
            if best_obj < 2.0:  # Only perturb if we've made progress
                noise = np.random.randn(n) * 0.05
                a_perturbed = best_a + noise
                a_perturbed = np.maximum(a_perturbed, 0)
                a_perturbed /= a_perturbed.sum()
                a = a_perturbed
            else:
                break
    
    return best_a, best_obj


def lbfgs_optimize(a0, time_limit, t=100.0):
    """
    Use L-BFGS-B to optimize the soft-max smoothed objective.
    Parameterize a = softmax(x) to handle non-negativity and sum constraint.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a.sum()
    
    # Initialize x such that softmax(x) = a
    # x = log(a) + constant (doesn't matter due to softmax invariance)
    a_safe = np.maximum(a, 1e-30)
    x0 = np.log(a_safe)
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    def objective_func(x):
        if time.monotonic() - t0 > time_limit:
            return np.inf
        x_shifted = x - x.max()
        a_curr = np.exp(x_shifted)
        a_curr /= a_curr.sum()
        conv = fast_convolve(a_curr)
        # Soft-max smoothed max
        m = conv.max()
        if m < 1e-15:
            return 0.0
        tx = t * conv
        tx_max = tx.max()
        exp_tx = np.exp(tx - tx_max)
        S = (tx_max + np.log(exp_tx.sum())) / t
        return 2.0 * n * S
    
    def gradient_func(x):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        x_shifted = x - x.max()
        a_curr = np.exp(x_shifted)
        a_curr /= a_curr.sum()
        conv = fast_convolve(a_curr)
        
        m = conv.max()
        if m < 1e-15:
            return np.zeros(n)
        
        tx = t * conv
        tx_max = tx.max()
        exp_tx = np.exp(tx - tx_max)
        sum_exp = exp_tx.sum()
        dx = exp_tx / sum_exp  # softmax of t*conv
        
        # Grad of smooth max w.r.t. conv: dx
        # Grad of conv w.r.t. a: 2 * cross_correlation(dx, a)
        a_rev = a_curr[::-1]
        h_full = np.convolve(dx, a_rev, 'full')
        h = h_full[n-1 : n-1+n]
        grad_a = 2.0 * h
        
        # Grad of smooth_obj w.r.t. a: 2n * grad_a
        # Grad w.r.t. x (where a = softmax(x)):
        # a_i = exp(x_i)/sum(exp(x))
        # d(a_i)/d(x_j) = a_i*(delta_ij - a_j)
        # d(obj)/d(x_j) = sum_i d(obj)/d(a_i) * d(a_i)/d(x_j)
        #               = sum_i (2n*grad_a[i]) * a_i*(delta_ij - a_j)
        #               = 2n * (grad_a[j]*a_j - a_j*sum_i grad_a[i]*a_i)
        #               = 2n * a_j * (grad_a[j] - sum_i grad_a[i]*a_i)
        
        inner = np.dot(grad_a, a_curr)
        grad_x = 2.0 * n * a_curr * (grad_a - inner)
        return grad_x
    
    # Run L-BFGS-B
    result = minimize(
        objective_func,
        x0,
        jac=gradient_func,
        method='L-BFGS-B',
        options={'maxiter': 5000, 'maxls': 50, 'gtol': 1e-6}
    )
    
    # Extract result
    x_opt = result.x
    x_shifted = x_opt - x_opt.max()
    a_opt = np.exp(x_shifted)
    a_opt /= a_opt.sum()
    
    conv = fast_convolve(a_opt)
    opt_obj = 2.0 * n * conv.max()
    
    if opt_obj < best_obj:
        best_obj = opt_obj
        best_a = a_opt.copy()
    
    return best_a, best_obj
