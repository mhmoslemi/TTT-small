```python
"""
Optimize non-negative step heights to minimize
    2*n*max(convolve(a,a)) / (sum(a))**2.

Strategy:
- Work on normalized sequences (sum = 1).
- Use the supplied global sequence if available as the main warm start.
- Stage 1: annealed smooth projected optimization with L-BFGS-B in softmax
  variables using a log-sum-exp surrogate for the peak coefficient.
- Stage 2: fast projected gradient iterations using the same surrogate,
  but in linear coordinates with an adaptive step size.
- Stage 3: linearized Frank-Wolfe style LP steps, where for fixed q we solve
    min t  s.t.  (q*p)_k <= t  for selected active k,  sum(p)=1, p>=0.
- Longer sequences are explored by interpolating the best short sequence and
  running fast manual optimization.
- All final comparisons use the direct convolution score, and the final answer
  is validated with evaluate_sequence when available.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog, minimize
except Exception:
    linprog = None
    minimize = None

try:
    from scipy.signal import fftconvolve
except Exception:
    fftconvolve = None


def normalize_sequence(a, target=1.0):
    """Normalize a non-negative sequence to a prescribed sum."""
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return arr
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 1e-15:
        return np.full(arr.shape, max(0.0, float(target)) / max(1, arr.size), dtype=np.float64)
    return arr * (max(0.0, float(target)) / s)


def compute_conv(a):
    """Full linear convolution of a with itself."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if fftconvolve is not None and n > 512:
        c = fftconvolve(a, a)
        c = np.real(c).astype(np.float64, copy=False)
        if c.size and c.min() < 0.0:
            c = np.maximum(c, 0.0)
        return c
    return np.convolve(a, a)


def score_from_array(a, c):
    """Objective from an array and its self-convolution."""
    a = np.asarray(a, dtype=np.float64).ravel()
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    if c.size == 0:
        return 0.0
    m = float(np.max(c))
    return float(2.0 * a.size * m / (s * s))


def compute_score_fast(a):
    """Fast objective using direct convolution for small n and FFT for larger n."""
    a = np.asarray(a, dtype=np.float64).ravel()
    c = compute_conv(a)
    return score_from_array(a, c)


def compute_score_direct(a):
    """Objective using direct np.convolve, matching the evaluator style."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    c = np.convolve(a, a)
    if c.size == 0:
        return 0.0
    m = float(np.max(c))
    return float(2.0 * a.size * m / (s * s))


def softmax_weights(c, beta):
    """Return normalized softmax weights on convolution coefficients."""
    c = np.asarray(c, dtype=np.float64).ravel()
    beta = float(beta)
    if c.size == 0:
        return np.array([], dtype=np.float64)
    if beta <= 0.0:
        w = np.ones(c.size, dtype=np.float64)
        s = float(w.sum())
        if s > 0.0:
            w /= s
        return w
    m = float(np.max(c))
    z = beta * (c - m)
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(c.size, dtype=np.float64)
        idx = int(np.argmax(c))
        w[idx] = 1.0
        return w
    w = e / s
    return w.astype(np.float64, copy=False)


def pair_grad_from_weights(p, w):
    """
    Compute derivative of sum_k w_k * (p*p)_k with respect to p.

    For signal p, weights w over full convolution indices:
        grad_j = 2 * sum_m p_m * w_{j+m}.
    This is obtained by convolution(w, reverse(p)) sliced appropriately.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.array([], dtype=np.float64)

    expected = 2 * n - 1
    if w.size != expected:
        tmp = np.zeros(expected, dtype=np.float64)
        m = min(w.size, expected)
        tmp[:m] = w[:m]
        w = tmp
        if w.size and w.max() > 0.0:
            w /= float(w.sum())

    if fftconvolve is not None and n > 512:
        conv = fftconvolve(w, p[::-1])
        conv = np.real(conv).astype(np.float64, copy=False)
        if conv.size and conv.min() < 0.0:
            conv = np.maximum(conv, 0.0)
    else:
        conv = np.convolve(w, p[::-1])

    g = 2.0 * conv[n - 1:2 * n - 1]
    if g.size < n:
        tmp = np.zeros(n, dtype=np.float64)
        tmp[:g.size] = g
        g = tmp
    elif g.size > n:
        g = g[:n]
    return g.astype(np.float64, copy=False)


def add_random_noise(a, rng, eps=0.01):
    """Multiplicative random perturbation, projected back to simplex."""
    a = normalize_sequence(a, 1.0)
    n = a.size
    noise = rng.normal(0.0, float(eps), size=n)
    b = a * (1.0 + noise)
    b = np.maximum(b, 0.0)
    if b.sum() <= 1e-15:
        b = np.full(n, 1.0 / n, dtype=np.float64)
    return normalize_sequence(b, 1.0)


def interpolate_sequence(a, new_n):
    """Linearly interpolate a sequence to a new length and renormalize."""
    a = normalize_sequence(a, 1.0)
    old_n = a.size
    if old_n == 0:
        return np.zeros(new_n, dtype=np.float64)
    if old_n == int(new_n):
        return a.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0.0, old_n - 1, int(new_n), dtype=np.float64)
    b = np.interp(x_new, x_old, a)
    b = np.maximum(b, 0.0)
    return normalize_sequence(b, 1.0)


def make_random_sequence(n, rng):
    """Random smooth or rough starting sequence of length n."""
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    typ = int(rng.integers(0, 6))
    i = np.arange(n, dtype=np.float64)
    if typ == 0:
        a = 0.5 + 0.5 * rng.random(n)
    elif typ == 1:
        a = np.exp(rng.normal(0.0, 0.45, size=n))
    elif typ == 2:
        f = int(rng.integers(1, 6))
        a = 0.6 + 0.4 * np.cos(2.0 * np.pi * f * i / max(1, n - 1))
    elif typ == 3:
        f = int(rng.integers(1, 6))
        a = 0.15 + np.abs(np.sin(2.0 * np.pi * f * i / max(1, n - 1)))
    elif typ == 4:
        center = 0.5 * n
        a = np.exp(-0.5 * ((i - center) / (0.22 * n)) ** 2)
    else:
        a = 1.0 / (1.0 + np.exp(-0.8 * i / max(1.0, n / 10.0)))
    return normalize_sequence(a, 1.0)


def postprocess_sequence(a):
    """Try tiny-value muting, keeping the direct score if it improves."""
    a = normalize_sequence(a, 1.0)
    best = a.copy()
    best_score = compute_score_direct(best)
    thresholds = [1e-13, 1e-11, 1e-9, 1e-7]
    for th in thresholds:
        b = best.copy()
        mask = b < float(th)
        if np.any(mask):
            b[mask] = 0.0
            if b.sum() > 1e-12:
                b = normalize_sequence(b, 1.0)
                s = compute_score_direct(b)
                if np.isfinite(s) and s < best_score:
                    best = b
                    best_score = s
    return best


def z_from_p(p, base, n, free_idx, lower=-30.0, upper=30.0):
    """
    Parametrize positive simplex by logits relative to a fixed base coordinate.
    p_base = 1 / (1 + sum_i exp(z_i)), p_i = exp(z_i) / (...) .
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    stable = np.where(p > 1e-300, p, 1e-300)
    denom = float(stable[base])
    z = np.zeros(n - 1, dtype=np.float64)
    for idx, j in enumerate(free_idx):
        val = float(stable[j]) / denom
        if val <= 0.0:
            z[idx] = lower
        else:
            zv = float(np.log(val))
            if zv < lower:
                z[idx] = lower
            elif zv > upper:
                z[idx] = upper
            else:
                z[idx] = zv
    return z


def p_from_z(z, base, n, free_idx):
    """Inverse of z_from_p for simplex parametrization."""
    z = np.asarray(z, dtype=np.float64).ravel()
    p = np.zeros(n, dtype=np.float64)
    if n <= 0:
        return p
    if z.size == 0:
        p[base] = 1.0
        return p
    m = float(np.max(np.maximum(z, 0.0)))
    y = np.exp(z - m)
    denom = 1.0 + float(np.sum(y))
    if not np.isfinite(denom) or denom <= 0.0:
        p[base] = 1.0
        return p
    p[base] = 1.0 / denom
    for idx, j in enumerate(free_idx):
        p[j] = y[idx] / denom
    return normalize_sequence(p, 1.0)


class SmoothObjective:
    """
    Differentiable surrogate:
        2n * LSE(beta*c)/beta
    where c = a*a, parametrized in the positive simplex by logits.
    """

    def __init__(self, beta, base, n, free_idx):
        self.beta = float(beta)
        self.base = int(base)
        self.n = int(n)
        self.free_idx = np.asarray(free_idx, dtype=int)

    def __call__(self, z):
        p = p_from_z(z, self.base, self.n, self.free_idx)
        c = compute_conv(p)
        if c.size == 0:
            return 0.0, np.zeros(max(0, self.n - 1), dtype=np.float64)

        beta = self.beta
        maxc = float(np.max(c))
        if beta <= 0.0:
            w = np.ones(c.size, dtype=np.float64)
            s = float(w.sum())
            if s > 0.0:
                w /= s
            lse_value = maxc
        else:
            shift = beta * (c - maxc)
            e = np.exp(shift)
            s = float(np.sum(e))
            if not np.isfinite(s) or s <= 0.0:
                w = np.zeros(c.size, dtype=np.float64)
                w[int(np.argmax(c))] = 1.0
                lse_value = maxc
            else:
                w = e / s
                lse_value = (np.log(s) + beta * maxc) / beta

        value = 2.0 * self.n * lse_value

        pair_grad = pair_grad_from_weights(p, w)
        grad_p = 2.0 * self.n * pair_grad
        dot = float(np.dot(p, grad_p))

        grad = np.zeros(self.free_idx.size, dtype=np.float64)
        for idx, j in enumerate(self.free_idx):
            grad[idx] = p[j] * (grad_p[j] - dot)

        if not np.all(np.isfinite(grad)):
            grad = np.zeros_like(grad)
        return float(value), grad.astype(np.float64, copy=False)


class TimeCallback:
    """Callback for scipy minimizers that stops after a deadline."""

    def __init__(self, deadline):
        self.deadline = float(deadline)

    def __call__(self, *args):
        return time.monotonic() > self.deadline


def lobfgs_anneal(a, deadline, best, best_score, rng=None, high_only=False):
    """
    Run scaled log-sum-exp L-BFGS-B optimization in simplex-log coordinates.
    Returns current a, best sequence, and best direct score.
    """
    a = normalize_sequence(a, 1.0)
    if best is None:
        best = a.copy()
    best = normalize_sequence(best, 1.0)
    if best_score is None or not np.isfinite(best_score):
        best_score = compute_score_direct(best)

    n = a.size
    if n < 4 or minimize is None:
        return a, best, best_score
    if rng is None:
        rng = np.random.default_rng(20240517)

    if high_only:
        betas = [
            max(1000.0, 300.0 * n),
            max(3000.0, 900.0 * n),
            max(8000.0, 2000.0 * n),
        ]
    else:
        betas = [
            max(100.0, 20.0 * n),
            max(500.0, 80.0 * n),
            max(2000.0, 300.0 * n),
            max(5000.0, 800.0 * n),
        ]

    for beta in betas:
        if time.monotonic() > deadline - 10.0:
            break

        p_safe = np.maximum(a, 1e-12)
        p_safe = normalize_sequence(p_safe, 1.0)
        base = int(np.argmax(p_safe))

        free_idx = np.zeros(n - 1, dtype=int)
        j = 0
        for x in range(n):
            if x != base:
                free_idx[j] = x
                j += 1

        z0 = z_from_p(p_safe, base, n, free_idx)
        obj = SmoothObjective(beta, base, n, free_idx)

        bounds = []
        for _ in range(n - 1):
            bounds.append((-30.0, 30.0))

        if n <= 512:
            maxfun = 5000
        elif n <= 1024:
            maxfun = 2000
        else:
            maxfun = 800

        options = {
            "maxiter": 1000,
            "maxfun": maxfun,
            "ftol": 1e-15,
            "gtol": 1e-8,
            "maxls": 30,
        }

        cb = TimeCallback(deadline - 8.0)
        try:
            res = minimize(
                obj,
                z0,
                method="L-BFGS-B",
                jac=True,
                bounds=bounds,
                options=options,
                callback=cb,
            )
            z1 = res.x if hasattr(res, "x") and res.x.size == n - 1 else z0
        except Exception:
            z1 = z0

        p1 = p_from_z(z1, base, n, free_idx)
        s1 = compute_score_direct(p1)
        if np.isfinite(s1):
            if s1 < best_score - 1e-12:
                best_score = s1
                best = p1.copy()
                a = p1.copy()
            else:
                # Keep the smoothed minimizer if it is not far away; it can
                # help the next, sharper beta escape previous approximations.
                if s1 < best_score * 1.02:
                    a = p1.copy()
                else:
                    a = add_random_noise(best, rng, 0.02)

        if time.monotonic() > deadline - 10.0:
            break

    return a, best, best_score


def smooth_step(a, step, beta):
    """One projected gradient step using the smooth surrogate."""
    a = normalize_sequence(a, 1.0)
    n = a.size
    if n == 0:
        return a
    c = compute_conv(a)
    w = softmax_weights(c, beta)
    g = 2.0 * n * pair_grad_from_weights(a, w)
    if not np.all(np.isfinite(g)):
        return a
    g -= g.mean()
    gn = float(np.linalg.norm(g))
    if gn <= 1e-14:
        return a
    b = a - float(step) * (g / gn)
    b = np.maximum(b, 0.0)
    if b.sum() <= 1e-15:
        return a
    return normalize_sequence(b, 1.0)


def manual_smooth_optimize(a, deadline, best, best_score, rng=None,
                           beta_factor0=20.0, beta_factor1=800.0,
                           step0=0.03, step1=0.002):
    """
    Fast projected-gradient loop for the smooth surrogate.
    Uses candidate evaluation and adaptive step size.
    """
    a = normalize_sequence(a, 1.0)
    if best is None:
        best = a.copy()
    best = normalize_sequence(best, 1.0)

    fast_best_score = compute_score_fast(best)
    if best_score is None or not np.isfinite(best_score):
        best_score = compute_score_direct(best)

    n = a.size
    if n < 4:
        return a, best, best_score
    if rng is None:
        rng = np.random.default_rng(987654321)

    beta0 = max(100.0, float(beta_factor0) * n)
    beta1 = max(5000.0, float(beta_factor1) * n)

    c = compute_conv(a)
    score = score_from_array(a, c)
    if score < fast_best_score:
        best = a.copy()
        fast_best_score = score

    step = float(step0)
    min_step = 0.0005
    last_best_iter = 0
    stagnant = 0
    it = 0
    t_start = time.monotonic()
    total_wall = max(1e-9, deadline - t_start)

    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - t_start
        progress = min(1.0, elapsed / total_wall)
        beta = beta0 + (beta1 - beta0) * progress

        w = softmax_weights(c, beta)
        g = 2.0 * n * pair_grad_from_weights(a, w)
        if not np.all(np.isfinite(g)):
            break
        g -= g.mean()
        gn = float(np.linalg.norm(g))

        if gn <= 1e-13:
            stagnant += 1
            if stagnant > 8:
                break
            a = add_random_noise(a, rng, 0.01)
            c = compute_conv(a)
            score = score_from_array(a, c)
            continue

        cand = a - step * (g / gn)
        cand = np.maximum(cand, 0.0)
        if cand.sum() <= 1e-15:
            break
        cand = normalize_sequence(cand, 1.0)

        c2 = compute_conv(cand)
        score2 = score_from_array(cand, c2)

        if score2 < score:
            a = cand
            c = c2
            score = score2
            step = min(float(step0), step * 1.03)
            stagnant = 0
        elif score2 < score * 1.002:
            # Accept a very small deterioration to avoid stalling.
            a = cand
            c = c2
            score = score2
            step = min(float(step0), step * 1.01)
            stagnant = 0
        else:
            step = max(min_step, step * 0.5)
            stagnant += 1
            if stagnant > 25:
                a = add_random_noise(best, rng, 0.02)
                c = compute_conv(a)
                score = score_from_array(a, c)
                step = float(step0)
                stagnant = 0

        if score < fast_best_score:
            fast_best_score = score
            best = a.copy()
            last_best_iter = it

        if it % 20000 == 0:
            print("  manual: n={} it={} current={} best_fast={}".format(
                n, it, repr(float(score)), repr(float(fast_best_score))))

        if it - last_best_iter > 50000 and time.monotonic() > deadline - 60.0:
            break

    best2 = postprocess_sequence(best)
    s2 = compute_score_direct(best2)
    if np.isfinite(s2):
        return a, best2, s2
    return a, best, float(fast_best_score)


def select_lp_constraints(a, max_count=None):
    """Select active convolution indices for the LP linearized step."""
    a = normalize_sequence(a, 1.0)
    n = a.size
    c = compute_conv(a)
    L = 2 * n - 1
    if max_count is None:
        max_count = max(64, min(L, n // 2 if n > 64 else n))
    max_count = int(max(8, min(L, max_count)))

    order = np.argsort(c)[::-1]
    top = order[:max_count].astype(int)
    even_count = min(L, max(16, L // 64))
    if even_count > 1:
        even = np.linspace(0.0, float(L - 1), even_count, dtype=np.float64).astype(int)
    else:
        even = np.array([L - 1], dtype=int)
    combined = np.concatenate((top, even))
    return np.unique(combined)


def lp_solve_response(q, idxs, pmax=0.2):
    """
    Solve LP:
        min t
        s.t. (q*p)_k <= t  for selected k,
             sum p = 1, 0 <= p_i <= pmax.
    """
    if linprog is None:
        return None
    q = normalize_sequence(q, 1.0)
    n = q.size
    idxs = np.asarray(idxs, dtype=int).ravel()
    if n <= 0 or idxs.size == 0:
        return None

    m = idxs.size
    A = np.zeros((m, n + 1), dtype=np.float64)
    for r, k in enumerate(idxs):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        i_max = ki - lo
        i_min = ki - hi
        A[r, lo:hi + 1] = q[i_min:i_max + 1][::-1]
        A[r, n] = -1.0

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[-1] = 1.0

    bounds = []
    for _ in range(n):
        bounds.append((0.0, float(pmax)))
    bounds.append((0.0, np.inf))

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    A_eq[0, n] = 0.0
    b_eq = np.array([1.0], dtype=np.float64)
    b_ub = np.zeros(m, dtype=np.float64)

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={
                "presolve": True,
                "dual_feasibility_tolerance": 1e-9,
            },
        )
        if res.status == 0:
            p = np.asarray(res.x[:-1], dtype=np.float64)
            p = np.maximum(p, 0.0)
            s = float(p.sum())
            if s > 1e-12:
                p = p / s
                if np.max(np.abs(p - q)) > 1e-9:
                    return p
    except Exception:
        return None
    return None


def line_search_blend(a, b, fa):
    """Line search for (1-lam)*a + lam*b, returning best lambda, point, score."""
    a = normalize_sequence(a, 1.0)
    b = normalize_sequence(b, 1.0)
    best_lam = 0.0
    best_a = a.copy()
    best_f = float(fa)

    k = 25
    lams = np.linspace(0.0, 1.0, k + 1, dtype=np.float64)
    for lam in lams:
        if float(lam) == 0.0:
            x = best_a
            f = best_f
        else:
            x = (1.0 - float(lam)) * a + float(lam) * b
            x = np.maximum(x, 0.0)
            if x.sum() <= 1e-15:
                continue
            x = normalize_sequence(x, 1.0)
            f = compute_score_fast(x)
        if np.isfinite(f) and f < best_f:
            best_f = f
            best_a = x.copy()
            best_lam = float(lam)

    if 0.0 < best_lam < 1.0:
        offsets = [-0.10, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.10]
        for off in offsets:
            lam = best_lam + float(off)
            if lam <= 0.0 or lam >= 1.0:
                continue
            x = (1.0 - lam) * a + lam * b
            x = np.maximum(x, 0.0)
            if x.sum() <= 1e-15:
                continue
            x = normalize_sequence(x, 1.0)
            f = compute_score_fast(x)
            if np.isfinite(f) and f < best_f:
                best_f = f
                best_a = x.copy()
                best_lam = lam

    return best_lam, best_a, best_f


def lp_refine(a, deadline, best, best_score, max_iters=70, rng=None):
    """
    Linearized Frank-Wolfe style refinement.
    For fixed q, find p minimizing max(q*p) over selected active diagonals,
    then blend and line search the true objective.
    """
    a = normalize_sequence(a, 1.0)
    if best is None:
        best = a.copy()
    best = normalize_sequence(best, 1.0)
    if best_score is None or not np.isfinite(best_score):
        best_score = compute_score_direct(best)
    if linprog is None:
        return a, best, best_score

    n = a.size
    if n > 2048:
        return a, best, best_score

    fast_best_score = compute_score_fast(best)
    no_imp = 0
    pmax = max(0.2, 2.0 / n)

    for it in range(int(max_iters)):
        if time.monotonic() > deadline:
            break

        max_count = min(2 * n - 1, max(64, n // 2))
        idxs = select_lp_constraints(a, max_count=max_count)
        p = lp_solve_response(a, idxs, pmax=pmax)
        if p is None:
            max_count = min(2 * n - 1, max(32, n // 4))
            idxs = select_lp_constraints(a, max_count=max_count)
            p = lp_solve_response(a, idxs, pmax=pmax)
            if p is None:
                break

        fa = compute_score_fast(a)
        _, a_new, f_new = line_search_blend(a, p, fa)

        if np.isfinite(f_new) and f_new < fast_best_score - 1e-12:
            fast_best_score = f_new
            a = a_new.copy()
            best = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            # A small smooth step can help move when the LP line search is flat.
            try:
                a_try = smooth_step(a, 0.02, beta=max(200.0, 80.0 * n))
            except Exception:
                a_try = a.copy()
            f_try = compute_score_fast(a_try)
            if np.isfinite(f_try) and f_try < fast_best_score - 1e-12:
                fast_best_score = f_try
                a = a_try.copy()
                best = a_try.copy()
                no_imp = 0
            else:
                if no_imp > 12:
                    break

        if (it + 1) % 20 == 0:
            print("  lp: n={} it={} fast_best={}".format(n, it + 1, repr(float(fast_best_score))))

    best2 = postprocess_sequence(best)
    s2 = compute_score_direct(best2)
    if np.isfinite(s2):
        return a, best2, s2
    return a, best, float(fast_best_score)


def refine_sequence(a0, deadline, use_lp=True, rng=None):
    """
    Multi-stage refinement of one starting sequence.
    Returns best sequence and its direct score.
    """
    a = normalize_sequence(a0, 1.0)
    if a.size == 0:
        a = np.ones(512, dtype=np.float64)
    best = a.copy()
    best_score = compute_score_direct(best)
    n = a.size

    if rng is None:
        rng = np.random.default_rng(123456789)

    if n < 4:
        best2 = postprocess_sequence(best)
        s2 = compute_score_direct(best2)
        if np.isfinite(s2):
            return best2, s2
        return best, best_score

    # Too long for the heavy L-BFGS/LP machinery: use fast projected gradient.
    if n > 1024:
        a, best, best_score = manual_smooth_optimize(
            a, deadline, a, best_score, rng=rng,
            beta_factor0=20.0, beta_factor1=1000.0,
            step0=0.02, step1=0.001,
        )
        best2 = postprocess_sequence(best)
        s2 = compute_score_direct(best2)
        if np.isfinite(s2) and s2 < best_score:
            best = best2
            best_score = s2
        return best, best_score

    t0 = time.monotonic()
    total = max(1e-9, deadline - t0)

    # Stage 1: L-BFGS-B smooth optimization.
    d1 = t0 + 0.35 * total
    try:
        a1, b1, s1 = lobfgs_anneal(a, d1, a, best_score, rng=rng, high_only=False)
    except Exception:
        a1, b1, s1 = a, best, best_score
    if np.isfinite(s1) and s1 < best_score:
        best = b1
        best_score = s1
        a = a1
    else:
        a = a1

    # Stage 2: fast projected gradient.
    remaining = max(0.0, deadline - time.monotonic())
    d2 = time.monotonic() + 0.32 * remaining
    try:
        a2, b2, s2 = manual_smooth_optimize(
            a, d2, best, best_score, rng=rng,
            beta_factor0=20.0, beta_factor1=900.0,
            step0=0.03, step1=0.002,
        )
    except Exception:
        a2, b2, s2 = a, best, best_score
    if np.isfinite(s2) and s2 < best_score:
        best = b2
        best_score = s2
        a = a2
    else:
        a = a2

    # Stage 3: LP linearized steps.
    if use_lp:
        remaining = max(0.0, deadline - time.monotonic())
        d3 = time.monotonic() + 0.22 * remaining
        max_iters = 80 if n <= 512 else 40
        try:
            a3, b3, s3 = lp_refine(a, d3, best, best_score, max_iters=max_iters, rng=rng)
        except Exception:
            a3, b3, s3 = a, best, best_score
        if np.isfinite(s3) and s3 < best_score:
            best = b3
            best_score = s3
            a = a3
        else:
            a = a3

    # Stage 4: final sharper L-BFGS-B pass.
    if time.monotonic() < deadline - 25.0:
        try:
            a4, b4, s4 = lobfgs_anneal(a, deadline, best, best_score, rng=rng, high_only=True)
        except Exception:
            a4, b4, s4 = a, best, best_score
        if np.isfinite(s4) and s4 < best_score:
            best = b4
            best_score = s4
            a = a4
        else:
            a = a4

    best2 = postprocess_sequence(best)
    s2 = compute_score_direct(best2)
    if np.isfinite(s2) and s2 < best_score:
        best = best2
        best_score = s2

    return best, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Main entry point. Searches for the best non-negative height sequence
    within the given time budget and returns it as a Python list of floats.
    """
    rng = np.random.default_rng(int(seed))
    t_start = time.monotonic()
    budget = max(120.0, float(budget_s))
    deadline = t_start + max(60.0, budget - 20.0)

    initial_candidates = []

    # Use the provided state of the art sequence if present.
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if gs.size >= 16 and np.all(np.isfinite(gs)) and np.any(gs > 0.0):
            initial_candidates.append(normalize_sequence(gs, 1.0))
    except Exception:
        pass

    # Deterministic and random 512 starts as diverse warm starts.
    n512 = 512
    i512 = np.arange(n512, dtype=np.float64)

    initial_candidates.append(normalize_sequence(np.ones(n512, dtype=np.float64), 1.0))
    initial_candidates.append(normalize_sequence(np.abs(np.sin(2.0 * np.pi * 3.0 * i512 / n512)) + 0.2, 1.0))
    initial_candidates.append(normalize_sequence(np.abs(np.cos(np.pi * i512 / (n512 - 1.0))) + 0.1, 1.0))
    initial_candidates.append(make_random_sequence(512, rng))
    initial_candidates.append(make_random_sequence(512, rng))

    # Pick the best initial candidate by direct score.
    primary = initial_candidates[0]
    primary_score = compute_score_direct(primary)
    for a in initial_candidates:
        s = compute_score_direct(a)
        if np.isfinite(s) and s < primary_score:
            primary = a
            primary_score = s

    # Optionally interpolate the best start to 512 if its length is unusual.
    if primary.size < 256:
        primary = interpolate_sequence(primary, 512)
    elif primary.size > 1024:
        primary = interpolate_sequence(primary, 512)

    allocate = max(60.0, deadline - t_start)

    best_seq = None
    best_score = float("inf")

    def considered(a, s):
        nonlocal best_seq, best_score
        if a is None or len(a) == 0:
            return
        if not np.isfinite(s):
            return
        if s < best_score:
            best_score = float(s)
            best_seq = np.asarray(a, dtype=np.float64).ravel().copy()

    # Main refinement on the best 512-like candidate.
    d1 = t_start + 0.55 * allocate
    try:
        b1, s1 = refine_sequence(primary, d1, use_lp=True, rng=rng)
    except Exception:
        b1, s1 = primary, primary_score
    considered(b1, s1)
    print("primary refined: score={}".format(repr(float(best_score))))

    # Upsample the current best to 1024 and refine.
    if best_seq is not None and time.monotonic() < deadline - 180.0:
        d2 = min(deadline - 30.0, t_start + 0.78 * allocate)
        up1024 = interpolate_sequence(best_seq, 1024)
        try:
            b2, s2 = refine_sequence(up1024, d2, use_lp=True, rng=rng)
        except Exception:
            b2, s2 = up1024, compute_score_direct(up1024)
        considered(b2, s2)
        print("1024 refined: best_score={}".format(repr(float(best_score))))

    # Try a fresh random 1024 candidate if time remains.
    if time.monotonic() < deadline - 240.0:
        d3 = min(deadline - 45.0, t_start + 0.90 * allocate)
        rnd1024 = make_random_sequence(1024, rng)
        try:
            b3, s3 = refine_sequence(rnd1024, d3, use_lp=True, rng=rng)
        except Exception:
            b3, s3 = rnd1024, compute_score_direct(rnd1024)
        considered(b3, s3)
        print("random 1024 tried: best_score={}".format(repr(float(best_score))))

    # If time remains, try a length-2048 continuation using fast manual optimization.
    if best_seq is not None and time.monotonic() < deadline - 120.0:
        up2048 = interpolate_sequence(best_seq, 2048)
        d4 = min(deadline - 15.0, time.monotonic() + max(60.0, 0.12 * allocate))
        try:
            b4, s4 = refine_sequence(up2048, d4, use_lp=False, rng=rng)
        except Exception:
            b4, s4 = up2048, compute_score_direct(up2048)
        considered(b4, s4)
        print("2048 manual tried: best_score={}".format(repr(float(best_score))))

    # Final polished refinement on the global best.
    if best_seq is not None and time.monotonic() < deadline - 20.0:
        dl = min(deadline - 8.0, time.monotonic() + max(20.0, 0.05 * allocate))
        try:
            b5, s5 = refine_sequence(best_seq, dl, use_lp=(len(best_seq) <= 1024), rng=rng)
        except Exception:
            b5, s5 = best_seq, best_score
        considered(b5, s5)

    if best_seq is None:
        # Safe fallback.
        fb = np.ones(256, dtype=np.float64)
        fb = normalize_sequence(fb, 1.0)
        considered(fb, compute_score_direct(fb))

    final = np.maximum(np.asarray(best_seq, dtype=np.float64).ravel(), 0.0)
    final = normalize_sequence(final, 1.0)
    final = postprocess_sequence(final)

    # If possible, verify with the official evaluator.
    official_score = None
    try:
        official_score = float(evaluate_sequence([float(x) for x in final]))
        print("official evaluate_sequence score={}".format(repr(official_score)))
    except Exception as exc:
        print("official evaluator unavailable/failed: {}".format(exc))

    out = []
    for x in final:
        out.append(float(x))
    return out
```
