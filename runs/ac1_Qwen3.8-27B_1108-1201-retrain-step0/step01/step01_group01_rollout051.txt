Coordinate descent with exact 1D line search (golden section) to minimize
the peak autocorrelation of a non-negative sequence of fixed length.
Uses proportional redistribution to maintain the simplex constraint.
Multiple phases: coordinate descent, pair swaps, random perturbations.

import numpy as np
import time

def _autocorr(a):
    """Compute linear autocorrelation of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()  # next power of 2
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * np.conjugate(fa), fft_size)[:size]
    return c

def _score(c, n):
    """Score from autocorrelation (assuming sum(a)=1)."""
    return 2.0 * n * np.max(c)

def _project_simplex(v, target=1.0):
    """Project onto simplex {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
