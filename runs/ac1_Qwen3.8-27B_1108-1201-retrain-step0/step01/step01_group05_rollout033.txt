import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.maximum(v, 0)  # quick clip first
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = (u * np.arange(1, n+1) - cssv) > 0
    if not np.any(rho):
        return np.full(n, S/n)
    rho = int(np.max(np.where(rho)[0])) + 1
    theta = (cssv[rho-1] - S) / rho
    return np.maximum(v - theta, 0)

def compute_soft_gradient(a, tau):
    """Compute exact gradient of the soft-max smoothed objective."""
    n = len(a)
    c = fft_convolve_self(a)
    # Numerically stable softmax
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / tau)
    p = exp_c / exp_c.sum()
    
    # grad[m] = 2 * sum_{k=0}^{n-1} p[m+k] * a[k]
    # = 2 * (p_rev * a)[2n-2-m] for m=0..n-1
    p_rev = p[::-1]
    conv_len = (2*n-1) + n - 1  # = 3n-2
    L = 1 << (conv_len - 1).bit_length()
    P = fft(p_rev, L)
    A = fft(a, L)
    conv = np.real(ifft(P * A))[:conv_len]
    # Need conv[2n-2-m] for m=0..n-1 → indices 2n-2, 2n-3, ..., n-1
    grad = 2.0 * conv[2*n-2 : n-2 : -1]
    return grad, p, c

def soft_objective(a, tau):
    """Soft-max approximation of max(c)."""
    c = fft_convolve_self(a)
    c_max = c.max()
    return tau * np.log(np.sum(np.exp((c - c_max) / tau))) + c_max

def run_soft_adam(a_init, n, S, max_time, rng, tau_start, tau_end, top_k_anneal):
    """Adam with soft-max smoothed objective and temperature annealing."""
    a = project_simplex(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    
    start = time.monotonic()
    n_steps = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        # Anneal temperature
        progress = min(1.0, t / top_k_anneal)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Compute smooth gradient
        grad, p, _ = compute_soft_gradient(a, tau)
        
        # Adam update
        lr = 0.1 * S / n  # base learning rate
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        
        # Periodic restart with noise

    return best_local, best_score, n_steps

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key approach: Use soft-max (log-sum-exp) smoothing to convert the non-smooth
max objective into a smooth, differentiable one. The gradient is computed
efficiently via FFT-based cross-correlation. Temperature annealing starts
from a smooth (large tau) approximation and refines to the sharp max.

Optimizers used:
  - Adam with Euclidean projection onto the non-negative simplex
  - Mirror descent (multiplicative weights) respecting non-negativity naturally

Multiple initializations and lengths are tried, with a final polish phase.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    # Standard projected gradient algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) - cssv > 0
    if not np.any(cond):
        return np.full(n, S / n, dtype=np.float64)
    rho = int(np.max(np.where(cond)[0])) + 1
    theta = (cssv[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix any numerical drift
    dr = S - result.sum()
    if abs(dr) > 1e-10:
        # Distribute the drift to non-zero elements
        mask = result > 0
        if mask.sum() > 0:
            result[mask] += dr / mask.sum()
            result = np.maximum(result, 0.0)
    return result


def compute_soft_gradient(a, tau):
    """
    Compute exact gradient of F_tau(a) = tau * log(sum_j exp(c_j/tau))
    where c = a*a (full self-convolution).
    
    Returns (grad, c) where grad has length n.
    """
    n = len(a)
    c = fft_convolve_self(a)  # length 2n-1
    
    # Numerically stable softmax
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    p = exp_c / exp_c.sum()  # length 2n-1, sums to 1
    
    # grad[m] = 2 * sum_{k=0}^{n-1} p[m+k] * a[k] for m=0..n-1
    # This equals 2 * (p_rev * a)[2n-2-m] where * is linear convolution
    p_rev = p[::-1]  # length 2n-1
    conv_len = (2 * n - 1) + n - 1  # = 3n-2
    L = 1 << (conv_len - 1).bit_length()
    P = fft(p_rev, L)
    A = fft(a, L)
    conv = np.real(ifft(P * A))[:conv_len]
    # Need conv[2n-2-m] for m=0..n-1: indices 2n-2, 2n-3, ..., n-1
    grad = 2.0 * conv[2 * n - 2 : n - 2 : -1]
    return grad, c


def run_adam_soft(a_init, n, S, max_time, rng, tau_start=1.0, tau_end=0.01,
                  lr=0.1, anneal_steps=5000):
    """Adam optimizer with soft-max smoothed objective and temperature annealing."""
    a = project_simplex(a_init.copy().astype(np.float64), S)
    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        # Anneal temperature
        progress = min(1.0, float(t) / anneal_steps)
        # Exponential annealing
        tau = tau_start * (tau_end / tau_start) ** progress
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Compute smooth gradient
        grad, _ = compute_soft_gradient(a, tau)
        
        # Adam update
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** t)
        v_hat = v_buf / (1.0 - beta2 ** t)
        
        # Adaptive learning rate: scale with problem size
        step_scale = S / n
        a = project_simplex(a - lr * step_scale * m_hat / (np.sqrt(v_hat) + eps), S)
        
        # Periodic noise injection to escape local minima
        if t % 3000 == 0 and rng.random() < 0.15:
            noise = rng.normal(0.0, 0.002 * S / n, n)
            a = project_simplex(a + noise, S)
        
        if t % 5000 == 0:
            left = max_time - (time.monotonic() - start)
            print("    adam step=%d tau=%.5f maxc=%.8f best=%.8f time_left=%.1f"
                  % (t, tau, cur, best_score, left))
    
    return best_local, best_score


def run_mirror_soft(a_init, n, S, max_time, rng, tau_start=0.5, tau_end=0.01,
                    eta=0.5, anneal_steps=3000):
    """Mirror descent (multiplicative weights) with soft-max smoothed objective."""
    a = project_simplex(a_init.copy().astype(np.float64), S)
    # Ensure no zeros for multiplicative updates
    a = np.maximum(a, 1e-15)
    a = project_simplex(a, S)
    
    t = 0
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, float(t) / anneal_steps)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        grad, _ = compute_soft_gradient(a, tau)
        
        # Normalize gradient for stable updates
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad_norm = grad / gmax
        
        # Multiplicative update
        step = eta / (1.0 + 0.01 * t)
        a_new = a * np.exp(-step * grad_norm)
        a_new = np.maximum(a_new, 1e-15)
        a = project_simplex(a_new, S)
        a = np.maximum(a, 1e-15)  # Keep strictly positive
        
        if t % 5000 == 0:
            left = max_time - (time.monotonic() - start)
            print("    mirror step=%d tau=%.5f maxc=%.8f best=%.8f time_left=%.1f"
                  % (t, tau, cur, best_score, left))
    
    return best_local, best_score


def run_subgradient(a_init, n, S, max_time, rng, top_k=10, lr=0.5):
    """Subgradient method using top-k active lags (for comparison/backup)."""
    a = project_simplex(a_init.copy().astype(np.float64), S)
    t = 0
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Top-k subgradient
        top_indices = np.argpartition(c, -top_k)[-top_k:]
        top_vals = c[top_indices]
        weights = top_vals / top_vals.sum()
        
        grad = np.zeros(n, dtype=np.float64)
        for w_val, k in zip(weights, top_indices):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        
        step = lr * (S / n) / (1.0 + 0.005 * t)
        a = project_simplex(a - step * grad, S)
        
        if t % 5000 == 0:
            left = max_time - (time.monotonic() - start)
            print("    subgrad step=%d maxc=%.8f best=%.8f time_left=%.1f"
                  % (t, cur, best_score, left))
    
    return best_local, best_score
