Step-function search for low max autocorrelation score.
Uses adaptive-temperature smoothed-max gradient descent with LP best-response,

except Exception:
    from numpy.fft import fft, ifft
    def next_fast_len(n):
        n = int(max(1, n))
        m = 1
        while m < n:
            m <<= 1
        return m

try:

    from scipy.sparse import csr_matrix

except Exception:

def _fft_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))

def conv_sq(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)

def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax):
        return np.inf
    return float(2.0 * n * cmax / (s * s))

def surrogate(c, T, cmax):
    if c.size == 0:
        return 0.0
    if not np.isfinite(cmax) or cmax <= 0:
        return 0.0
    if not np.isfinite(T) or T <= 0:
        return float(cmax)
    diff = (c - cmax) / T
    e = np.exp(np.maximum(diff, -50.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return float(cmax)
    return float(cmax + T * np.log(s))

def gradient_smoothed(p, T_factor):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64), 0.0, 0.0
    
    L = int(next_fast_len(max(1, 3 * n - 2)))
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]
    
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax <= 0:
        return np.zeros(n), 0.0, 0.0
    
    T = max(float(T_factor) * cmax, 1e-300)
    diff = (c - cmax) / T
    e = np.exp(np.maximum(diff, -50.0))
    raw = float(e.sum())
    if not np.isfinite(raw) or raw <= 0:
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        raw = 1.0
    w = e / raw
    
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    
    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2:n - 2:-1]
    
    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    
    g = 2.0 * np.maximum(G, 0.0)
    surr_val = float(cmax + T * np.log(raw))
    return g, T, surr_val

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s

def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s

def lp_best_response(p, budget_s):
    if not _HAVE_LP:
        return None, np.inf
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 2 or n > 1200:
        return None, np.inf
    if budget_s <= 0.5:
        return None, np.inf
    
    c = conv_sq(p)
    M = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(M) or M <= 0:
        return None, np.inf
    
    p = safe_normalize(p)
    end = time.monotonic() + max(0.2, budget_s)
    
    try:
        mid = np.arange(n, dtype=np.int32)
        rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel().astype(np.int32)
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(p, n).astype(np.float64, copy=False)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs",
                      options={"disp": False, "time_limit": max(1.0, budget_s - 0.5)})
    except Exception:
        return None, np.inf
    
    if res is None or not hasattr(res, 'x') or res.x is None:
        return None, np.inf
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None, np.inf
    g = g / s
    return g, np.inf

def make_initials(n, rng):
    out = []
    out.append(np.full(n, 1.0/n))
    
    # Geometric
    for s in (0.3, 0.8, 1.5, 3.0, 6.0, 12.0):
        x = np.linspace(0, 1, n)
        p = np.exp(-s * x)
        out.append(safe_normalize(p))
        out.append(safe_normalize(p[::-1]))
    
    # Sinusoidal bumps
    for f in (1, 2, 3, 5, 8):
        x = np.linspace(0, 1, n)
        p = 0.5 + 0.5 * np.cos(2 * np.pi * f * x)
        out.append(safe_normalize(np.maximum(p, 0.01)))
    
    # Two peaks
    for mu1, mu2 in ((0.2, 0.8), (0.1, 0.5), (0.3, 0.7), (0.15, 0.85)):
        x = np.linspace(0, 1, n)
        sig = max(0.02, 0.05 * mu1 * (1-mu1))
        p = np.exp(-((x-mu1)**2)/(2*sig**2)) + np.exp(-((x-mu2)**2)/(2*sig**2))
        out.append(safe_normalize(p))
    
    # Cosine window variants
    x = np.linspace(0, 1, n)
    for alpha in (0.5, 1.0, 2.0, 4.0, 8.0):
        p = np.cos(np.pi * alpha * (x - 0.5))
        out.append(safe_normalize(np.maximum(p, 0.001)))
    
    # Random smooth
    for _ in range(8):
        f1 = int(rng.integers(1, max(2, n//50)))
        f2 = int(rng.integers(1, max(2, n//25)))
        p = 1.0
        p += 0.5 * rng.uniform(0.5, 1.0) * np.sin(2*np.pi*f1*np.linspace(0,1,n) + rng.uniform(0, 2*np.pi))
        p += 0.3 * rng.uniform(0.3, 0.7) * np.sin(2*np.pi*f2*np.linspace(0,1,n) + rng.uniform(0, 2*np.pi))
        out.append(safe_normalize(np.maximum(p, 0.01)))
    
    # Sparse (block) sequences
    for block_frac in (0.1, 0.2, 0.3, 0.5):
        m = max(2, int(n * block_frac))
        p = np.zeros(n)
        start = int(rng.integers(0, n - m + 1))
        p[start:start+m] = 1.0
        out.append(safe_normalize(p))
    
    # Beta-distribution shaped
    for a, b in ((2, 5), (5, 2), (3, 3), (1, 8), (8, 1), (0.5, 0.5)):
        x = np.linspace(0.001, 0.999, n)
        p = x**a * (1-x)**b
        out.append(safe_normalize(np.maximum(p, 0)))
    
    return out

def optimize_sequence(p0, rng, time_limit, label=""):
    p = project_simplex(safe_normalize(p0))
    n = int(p.size)
    if n == 0:
        return p, np.inf
    
    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score
    prev_p = p.copy()
    
    end = time.monotonic() + max(0.1, float(time_limit))
    T_factor = 0.5
    iter_no = 0
    since_improve = 0
    momentum = 0.0
    
    while time.monotonic() < end:
        iter_no += 1
        
        # Adaptive temperature
        T_factor = max(0.005, 0.5 * (0.995 ** iter_no))
        if since_improve > 200:
            T_factor = max(0.003, T_factor * 0.5)
        
        g, T, surr = gradient_smoothed(p, T_factor)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        
        gabs = float(np.max(np.abs(g))) if g.size else 0.0
        pmax = float(np.max(p)) if n > 0 else 0.0
        p_scale = max(1.0/float(n), 0.1 * pmax, 1e-12)
        
        if gabs > 1e-300:
            base = p_scale / gabs
            lr_candidates = (0.5*base, 0.15*base, 0.05*base, 0.01*base, 0.003*base)
        else:
            lr_candidates = (0.1, 0.05, 0.02, 0.01, 0.005)
        
        accepted = False
        for lr in lr_candidates:
            v = p - float(lr) * g
            if momentum > 0:
                v = v + momentum * (p - prev_p)
            pt = project_simplex(v)
            
            c_t = conv_sq(pt)
            cmax_t = float(np.max(c_t)) if c_t.size else 0.0
            s_t = float(pt.sum())
            score_t = float(2.0 * n * cmax_t / (s_t * s_t)) if s_t > 1e-12 else np.inf
            surr_t = surrogate(c_t, T, cmax_t)
            
            if np.isfinite(score_t) and np.isfinite(surr_t):
                if surr_t < surr - 1e-10 * max(1.0, abs(surr)):
                    accepted = True
                    break
                elif score_t < best_score - 1e-12 and surr_t < surr + 0.5*max(1.0, surr):
                    accepted = True
                    break
                elif score_t < current_score - 1e-10 and surr_t < surr + 0.3*max(1.0, surr):
                    accepted = True
                    break
        
        if accepted:
            if score_t < best_score:
                best_score = score_t
                best_p = pt.copy()
                since_improve = 0
            else:
                since_improve += 1
