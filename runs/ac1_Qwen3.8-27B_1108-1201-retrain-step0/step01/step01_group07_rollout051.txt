def compute_objective_grad(a, tau, top_k=None):
    """
    Compute gradient of tau*log(sum(exp(c/tau))) for c = convolve(a,a).
    Optionally restrict to top-k values.
    Returns (value, gradient).
    """
    n = len(a)
    c = np.convolve(a, a)
    
    if top_k is not None and top_k < len(c):
        # Zero out all but top-k values for the softmax
        # (This makes the objective focus on the largest values)
        threshold = np.partition(c, -top_k)[-top_k]

    else:
        c_masked = c

    c_max = c_masked.max()

    q = exp_c / exp_c.sum()

    # Actually, the value of the objective is just c.max() (or the top-k weighted sum)
    # For monitoring, let's return c.max()
    
    return c.max(), grad


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = a.size
    a = np.maximum(a, 0.0)
    s = a.sum()
    if abs(s - S) < 1e-12:
        return a
    if s == 0:
        return np.full(n, S / n)
    
    # Binary search for the threshold
    lo = -a.max()
    hi = 0.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a + mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a + t, 0.0)
    # Fix rounding
    diff = S - res.sum()
    if abs(diff) > 1e-10:
        idx = res.argmax()
        res[idx] += diff
    return res


def adam_step(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """One step of Adam optimizer. Returns updated m, v."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v


def init_chirp(n, seed=0):
    """Chirp-based initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.5 * np.cos(np.pi * t * t / (2.0 * n))
    rng = np.random.default_rng(seed)
    a += 0.1 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_random(n, seed=0):
    """Random initialization."""
    return np.random.default_rng(seed).exponential(1.0, size=n)


def init_from_provided(prov, n):
    """Interpolate provided sequence to length n."""
    if len(prov) == n:
        return prov.copy()
    x_old = np.linspace(0.0, 1.0, len(prov))
    x_new = np.linspace(0.0, 1.0, n)
    return np.maximum(np.interp(x_new, x_old, prov), 0.0)

So we want `c[m] ≈ 1` for all m. The deviation from 1 is what we're minimizing (in a max sense).

For a random sequence with `a[i] ~ Exp(1)` normalized to have sum `sqrt(2n)`, the variance of `c[m]` is approximately `sum(a^4) / n ≈ (9 * (sum(a)/n)^4 * n) / n = 9 * (sqrt(2n)/n)^4 ... ` this is getting complicated. The point is that random sequences have large variance in c, so the max is well above the mean.

An optimized sequence should have much lower variance, with `c[m]` close to 1 for all m.

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using Adam with soft-max smoothing,
tau annealing, top-k focusing, and multi-phase strategy.
"""

import numpy as np
import time

def softmax_grad(a, tau, top_k=None):
    """Compute gradient of soft-max smoothed max convolution."""
    n = a.size
    c = np.convolve(a, a)
    if top_k is not None and top_k < c.size:
        # Create masked version for top-k
        thr = np.partition(c, c.size - top_k)[c.size - top_k]
        c_work = c.copy()
        c_work[c_work < thr] = -1e30  # Effectively -inf
    else:
        c_work = c
    
    c_max = c_work.max()
    exp_c = np.exp((c_work - c_max) / tau)
    q = exp_c / exp_c.sum()
    
    # Gradient = 2 * correlate(q, a, 'full')[:n]
    grad = 2.0 * np.correlate(q, a, 'full')[:n]
    return c.max(), grad

def project_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = a.size
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= S:
        return a + (S - s) / n
    # Binary search for threshold
    lo, hi = -a.max(), 0.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a + mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a + t, 0.0)
    res[res.argmax()] += S - res.sum()
    return res

def adam_update(a, grad, state, step, lr):
    """Adam optimizer step."""
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = state['m']
    v = state['v']
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1 - beta1 ** step)
    v_hat = v / (1 - beta2 ** step)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    state['m'] = m
    state['v'] = v
    return a_new

def run_opt(a0, t_budget, tau_start, tau_end, top_k=None, lr=1.0, name=""):
    """Main optimization loop."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = a0.astype(np.float64).copy()
    a = a / max(a.sum(), 1e-12) * S
    a = project_simplex(a, S)
    
    best = a.copy()
    best_mc = np.convolve(best, best).max()
    state = {'m': np.zeros(n), 'v': np.zeros(n)}
    
    t0 = time.monotonic()
    step = 0
    no_improve = 0
    
    while time.monotonic() - t0 < t_budget:
        el = time.monotonic() - t0
        f = min(1.0, el / max(t_budget, 1e-6))
        tau = tau_start * (tau_end / tau_start) ** f
        
        # Adaptive top-k
        k = None
        if top_k is not None:
            k = max(5, int(top_k * (1.0 - 0.7 * f)))
        
        mc, grad = softmax_grad(a, tau, k)
        
        # Learning rate schedule
        lr_now = lr * (1.0 + 0.5 * (1.0 - f))
        
        step += 1
        a = adam_update(a, grad, state, step, lr_now)
        a = project_simplex(a, S)
        
        # Monitor (every 50 steps)
        if step % 50 == 0:
            mc_now = np.convolve(a, a).max()
            if mc_now < best_mc - 1e-10:

                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1

                a = project_simplex(a + noise, S)
                no_improve = 0
    
    final_mc = np.convolve(best, best).max()
    score = 2.0 * n * final_mc / (S * S)
    print(f"  {name}: score={score:.6f}, max_c={final_mc:.6f}, steps={step}")
    return best, score
