import numpy as np

def _fft_size(n):
    """Next power of 2 >= 2n-1."""
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    N = _fft_size(n)
    A = np.fft.rfft(a, N)
    result = np.fft.irfft(A * A, N)
    return result[:2 * n - 1]


def compute_norm_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1, a>=0."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def normalize_sequence(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def gs_flat_spectrum(a, n_iters=50, beta=3.0):

    n = len(a)
    N = _fft_size(n)
    for _ in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        mag_new = mag ** (1.0 / beta)
        phase = A / mag.replace(0.0, 1.0)
        A_new = mag_new * phase
        a_new = np.real(np.fft.ifft(A_new, N))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    return a


def compute_flat_gradient(a, N):
    """Gradient of sum|FFT(a)|^4 (proxy for convolution variance)."""
    A = np.fft.fft(a, N)
    P = A * np.conj(A)
    grad = 4.0 * np.real(np.fft.ifft(P * A, N))[:len(a)]
    return grad

def generate_initializations(n, seed, parent=None):
    """Generate diverse initializations of length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussians
    for sf in [0.1, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Bimodal (edges)
    bi = np.zeros(n)
    bi[:max(1, n // 4)] = 1.0
    bi[-max(1, n // 4):] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Cosine bump
    cos_bump = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_bump /= cos_bump.sum()
    inits['cosine'] = cos_bump

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Random Dirichlet (concentrated)
    inits['random_conc'] = rng.dirichlet(np.ones(n) * 0.1)

    # Periodic-ish
    period = max(2, n // 8)
    per = np.zeros(n)
    per[::period] = 1.0
    per /= per.sum()
    inits['periodic'] = per

    # Triangular (peak in middle)
    tri = np.ones(n)
    half = n // 2
    tri[:half] = np.linspace(1, 0.2, half)
    tri[half:] = np.linspace(0.2, 1, n - half)
    tri /= tri.sum()
    inits['triangular'] = tri

    # From parent if available
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp

    return inits


def upsample(a, new_n):
    """Upsample sequence to new length via interpolation."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize_sequence(a_new)


def blend(a1, a2, alpha):
    """Blend two sequences of the same length."""
    return normalize_sequence(alpha * a1 + (1 - alpha) * a2)

        if time.monotonic() > phase1_deadline - 10:
            break
        t_alloc = min(time_per_scan, phase1_deadline - time.monotonic() - 5)
        if t_alloc < 1.0:
            break

        # GS pre-processing
        a_gs = gs_flat_spectrum(a0, n_iters=20, beta=3.0)
        # Short mirror descent
        t_md = max(0.5, t_alloc * 0.7)
        a_opt, obj_opt = mirror_descent(a_gs, time_limit=t_md, t_param=30.0, eta0=0.5)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan: n={n}, {name}, score={score:.6f}")

    print(f"  Phase 1 done. Best: {best_score:.6f}")

    # ============ PHASE 2: Deep optimization (1500s) ============
    print("Phase 2: Deep optimization...")
    phase2_deadline = time.monotonic() + min(1500.0, (deadline - time.monotonic()) * 0.65)

    scan_results.sort(key=lambda x: x[0])

    # Select top configs for deep optimization
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n,)
        if key not in seen:
            top_configs.append((n, name, a_opt))
            seen.add(key)
        if len(top_configs) >= 10:
            break

    # Also add best if not already there
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'current_best', best_sequence.copy()))

    # Ensure we have configs for a few key lengths
    for n in [128, 256, 512]:
        if n not in [c[0] for c in top_configs]:
            inits = generate_initializations(n, seed + n, parent)
            best_init_name = 'uniform'
            top_configs.append((n, best_init_name, inits[best_init_name]))

    remaining = phase2_deadline - time.monotonic() - 10
    n_configs = len(top_configs)
    time_per_config = max(5.0, remaining / max(1, n_configs))

    pool = []  # Pool of good solutions for genetic operations
    for score, n, name, a_opt in scan_results[:10]:
        pool.append(a_opt.copy())

    for ci, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > phase2_deadline - 10:
            break
        t_alloc = min(time_per_config, phase2_deadline - time.monotonic() - 10)
        if t_alloc < 3.0:
            break

        print(f"  Deep: n={n}, {name}, alloc={t_alloc:.1f}s")

        # Multiple strategies
        strategies = [
            ('gs_md', 0.4),      # GS then mirror descent
            ('md_flat', 0.3),    # MD with flat gradient
            ('lp', 0.3 if n <= 512 else 0.0),  # LP refinement
        ]

        for strat_name, frac in strategies:
            if t_alloc < 2.0:
                break
            t_strat = t_alloc * frac

            if strat_name == 'gs_md':
                a_gs = gs_flat_spectrum(a0, n_iters=min(50, int(t_strat * 5)), beta=2.5)
                a_opt, _ = mirror_descent(a_gs, time_limit=t_strat, t_param=40.0, eta0=0.5)
            elif strat_name == 'md_flat':
                a_opt, _ = mirror_descent(a0, time_limit=t_strat, t_param=60.0,
                                          eta0=0.4, use_flat_grad=True, flat_weight=0.4)
            elif strat_name == 'lp':
                a_opt, _ = lp_refine(a0, time_limit=t_strat)
            else:
                continue

            update_best(a_opt)
            if np.isfinite(compute_norm_obj(a_opt)):
                pool.append(a_opt.copy())
                if len(pool) > 20:
                    pool = pool[:20]

            t_alloc = t_alloc - frac * (t_alloc / frac) if t_alloc > 0 else 0
            t_alloc = phase2_deadline - time.monotonic() - 10

        # Occasionally do genetic blending
        if len(pool) >= 2 and time.monotonic() < phase2_deadline - 10:
            a1 = pool[rng.integers(len(pool))]
            a2 = pool[rng.integers(len(pool))]
            if len(a1) == len(a2):
                for alpha in [0.3, 0.5, 0.7]:
                    a_blend = blend(a1, a2, alpha)
                    a_opt, _ = mirror_descent(a_blend, time_limit=min(5.0, phase2_deadline - time.monotonic() - 5),
                                              t_param=50.0, eta0=0.4)
                    update_best(a_opt)
                    if time.monotonic() > phase2_deadline - 10:
                        break

    print(f"  Phase 2 done. Best: {best_score:.6f}")

    # ============ PHASE 3: Final polish (remaining time) ============
    print("Phase 3: Final polish...")
    if best_sequence is not None and time.monotonic() < deadline - 10:
        remaining = deadline - 15 - time.monotonic()
        n = len(best_sequence)

        # Try multiple polish strategies
        polish_configs = [
            (40.0, 0.3, 0.3),
            (80.0, 0.3, 0.4),
            (120.0, 0.4, 0.0),
            (60.0, 0.5, 0.0),
            (100.0, 0.3, 0.3),
        ]

        # Also try upsampled versions for larger n
        for target_n in [256, 512, 1024]:
            if n < target_n and remaining > 30:
                a_up = upsample(best_sequence, target_n)
                t_up = min(20.0, remaining * 0.1)
                a_opt, _ = mirror_descent(a_up, time_limit=t_up, t_param=50.0, eta0=0.4)
                update_best(a_opt)
                remaining = deadline - 15 - time.monotonic()

                a_up2 = gs_flat_spectrum(a_up, n_iters=30, beta=3.0)
                a_opt2, _ = mirror_descent(a_up2, time_limit=min(10.0, remaining * 0.1), t_param=60.0, eta0=0.3)
                update_best(a_opt2)
                remaining = deadline - 15 - time.monotonic()

        for t_p, eta_p, fw in polish_configs:
            if remaining < 3.0:
                break
            t_single = min(remaining * 0.4, 60.0)
            use_fg = fw > 0.1
            a_opt, _ = mirror_descent(best_sequence, time_limit=t_single,
                                      t_param=t_p, eta0=eta_p,
                                      use_flat_grad=use_fg, flat_weight=fw)
            update_best(a_opt)

            # LP refinement if n is small enough
            if n <= 512 and remaining > 10:
                a_lp, _ = lp_refine(a_opt, time_limit=min(10.0, remaining * 0.2))
                update_best(a_lp)

            remaining = deadline - 15 - time.monotonic()

        # GS polish on the best
        if remaining > 5.0 and best_sequence is not None:
            a_gs = gs_flat_spectrum(best_sequence, n_iters=min(50, int(remaining)), beta=2.0)
            update_best(a_gs)
            a_opt, _ = mirror_descent(a_gs, time_limit=min(remaining * 0.5, 30.0), t_param=100.0, eta0=0.2)
            update_best(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f}, length={len(best_sequence)}")
    return best_sequence


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Uses hybrid optimization (GS spectrum flattening +
    multiplicative mirror descent + LP refinement) with multi-start search
    to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    result = run_full_search(seed=seed, budget_s=budget_s)
    return [float(x) for x in result]
