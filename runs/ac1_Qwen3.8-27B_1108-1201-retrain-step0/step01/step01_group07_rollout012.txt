One important detail: the Nesterov momentum with projection. The standard Nesterov is:
y_{k+1} = x_k + beta*(x_k - x_{k-1})
x_{k+1} = Proj(y_{k+1} - alpha * grad(f(y_{k+1})))

The projection is applied AFTER the gradient step. The momentum term (x_k - x_{k-1}) is computed in the unconstrained space, which might cause issues if the projection is "active" (i.e., many coordinates are at 0). In that case, the momentum might point in a direction that immediately gets projected away.

A simpler approach that works well in practice:
y_{k+1} = Proj(x_k + beta*(x_k - x_{k-1}))  [project the extrapolation]
x_{k+1} = Proj(y_{k+1} - alpha * grad(f(y_{k+1})))  [project the update]

Or even simpler (and what many implementations use):
v_{k+1} = beta * v_k + (1-beta) * grad(f(x_k))  [exponential average of gradients]
x_{k+1} = Proj(x_k - alpha * v_{k+1})

This is "momentum SGD" (Polyak momentum), not exactly Nesterov, but often works as well or better in practice.

Let me use the Polyak momentum version for simplicity:
v = beta * v + (1-beta) * grad
x_new = Proj(x - alpha * v)

With beta starting at 0.9 and increasing to 0.99.

Actually, for safety, let me use a combination: start with plain gradient descent (no momentum) for the first few hundred steps (to settle into a good direction), then add momentum.

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses Nesterov-accelerated projected gradient descent with log-sum-exp smoothing,
FFT-based convolution, adaptive learning rate, and multi-phase strategy.
"""

import numpy as np
import time
import math

def project_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    s = a.sum()
    if s <= S + 1e-12:
        return a + (S - s) / n
    # Binary search for threshold
    lo, hi = 0.0, a.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    # Fix rounding: distribute remainder to largest element
    diff = S - res.sum()
    if abs(diff) > 1e-15:
        idx = res.argmax()
        res[idx] += diff
    return res

def fast_conv(a):
    """Compute conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, n=fft_size)
    c = np.fft.irfft(fa * fa, n=fft_size)[:size]
    return c

def compute_smooth_grad(a, tau):
    """Compute gradient of tau*log-sum-exp(conv(a,a)/tau) using FFT."""
    n = len(a)
    c = fast_conv(a)
    # Softmax
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    
    # Gradient: (2/tau) * (a cross-correlated with q), first n components
    # (a ⋆ q)[j] = sum_i a[i] * q[i+j]
    fft_size = 1
    while fft_size < 3 * n - 2:
        fft_size *= 2
    fa = np.fft.rfft(a, n=fft_size)
    fq = np.fft.rfft(q, n=fft_size)
    corr = np.fft.irfft(fa * np.conj(fq), n=fft_size)
    grad = (2.0 / tau) * corr[:n]
    
    return grad, c

def init_chirp(n, seed=0):
    """Chirp-based initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.9 * np.cos(np.pi * t * t / n)
    rng = np.random.default_rng(seed)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

def init_random(n, seed=0):
    """Random initialization."""
    return np.random.default_rng(seed).exponential(1.0, size=n)

def normalize(a, S):
    """Normalize to sum = S."""
    s = a.sum()
    if s < 1e-12:
        return np.full_like(a, S / len(a))
    return a * (S / s)

def nesterov_opt(a0, S, t_budget, tau_start=0.5, tau_end=0.01, 
                  lr0=1.0, beta_start=0.0, beta_end=0.9, verbose=False):
    """Nesterov/Polyak momentum projected gradient descent with tau annealing."""
    n = a0.size
    a = normalize(a0.astype(np.float64), S).copy()
    a = project_simplex(a, S)
    v = np.zeros(n)
    
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    best_mc = fast_conv(best).max()
    
    t0 = time.monotonic()
    step = 0
    lr = lr0
    stagnation = 0
    last_best_time = t0
    
    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(t_budget, 1e-6))
        
        # Anneal tau
        tau = tau_start * (tau_end / tau_start) ** frac
        # Anneal momentum
        beta = beta_start + (beta_end - beta_start) * frac
        
        # Compute gradient at current point (or extrapolated point)
        if beta > 0 and step > 0:
            y = a + beta * (a - a_prev)
        else:
            y = a.copy()
        
        grad, c = compute_smooth_grad(y, tau)
        
        # Extrapolation (Nesterov)
        if step > 0:
            y = a + beta * (a - a_prev)
        else:
            y = a.copy()
        
        a_prev = a.copy()
        
        # Gradient step
        a_new = y - lr * grad
        a_new = project_simplex(a_new, S)
        
        # Evaluate (every 50 steps to save time)
        step += 1
        if step % 50 == 0 or step < 10:
            mc = fast_conv(a_new).max()
            if mc < best_mc - 1e-10:
                best_mc = mc
                best = a_new.copy()
                stagnation = 0
                if lr < lr0 * 5.0:
                    lr = min(lr * 1.1, lr0 * 5.0)
                last_best_time = time.monotonic()
            else:
                if mc < best_mc:
                    # Small improvement, accept
                    best_mc = mc  # Don't update best sequence, just track
                else:
                    stagnation += 50
                    if mc > best_mc:
                        lr = max(lr * 0.7, 0.001)
        
        a = a_new
        
        # Safety: if stuck for too long, reduce step size
        if time.monotonic() - last_best_time > 60 and stagnation > 200:
            lr = max(lr * 0.5, 0.001)
