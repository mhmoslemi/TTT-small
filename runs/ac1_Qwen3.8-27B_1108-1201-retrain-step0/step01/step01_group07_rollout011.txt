import time

def compute_conv(a, b):
    """Compute full convolution using FFT for speed."""
    n = len(a) + len(b) - 1
    N = 1
    while N < n:
        N *= 2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]

from scipy.optimize import linprog
import time

def fftconvolve_full(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    N = 1
    while N < n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]

    c = -np.ones(n)
    b_ub = np.full(num_rows, M)

    if res.status != 0:
        return None
    g = res.x
    # Clean up tiny negatives
    g = np.maximum(g, 0.0)
    return g

def run_lp_method(p0, n_iter_max, t_budget, verbose=True):
    """Run LP-based best response iterations."""
    t0 = time.monotonic()
    p = p0.astype(np.float64).copy()
    p = p / p.sum()
    n = len(p)
    
    c = fftconvolve_full(p, p)
    M = float(c.max())
    best_p = p.copy()
    best_score = compute_score(p)
    
    if verbose:
        print(f"  LP start: n={n}, M={M:.8f}, score={best_score:.8f}")
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > t_budget:
            break
        
        # Solve LP
        t_lp_start = time.monotonic()
        g = lp_best_response(p, M)
        t_lp = time.monotonic() - t_lp_start

            break

        if g_sum < 1e-12:
            break
        
        # Rescale g to match sum(p) = 1
        g = g / g_sum
        
        # Try different blend factors
        best_t = 0.0
        best_new_M = M
        for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
            p_new = (1 - t) * p + t * g
            p_new = np.maximum(p_new, 0)
            p_new = p_new / p_new.sum()
            c_new = fftconvolve_full(p_new, p_new)
            new_M = float(c_new.max())
            if new_M < best_new_M:
                best_new_M = new_M
                best_t = t
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.01, 0.02, 0.03]:
                p_new = (1 - t) * p + t * g
                p_new = np.maximum(p_new, 0)
                p_new = p_new / p_new.sum()
                c_new = fftconvolve_full(p_new, p_new)
                new_M = float(c_new.max())
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_t = t

            if verbose and it % 10 == 0:
                print(f"  No improvement at iter {it}, M={M:.8f}")

                M = new_M

Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using iterative LP best-response.
The LP finds the non-negative sequence g maximizing sum(g) subject to
cross-convolution conv(p,g) being bounded by the current max of conv(p,p).
This provides a principled descent direction. Combined with multi-scale
strategy and subgradient polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fftconvolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    N = 1
    while N < 2 * n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    result = np.fft.irfft(fa * fb, N)
    return result[:n]


def build_conv_matrix(p):
    """Build matrix A where A[k,j] = p[k-j] if 0<=k-j<len(p), else 0.
    Shape: (2n-1, n) where n = len(p)."""
    n = len(p)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    # For each column j, A[k, j] = p[k-j] for k = j, j+1, ..., j+n-1 (clipped)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = p[:end - j]
    return A


def solve_lp(p, M):
    """Solve: max 1^T g subject to A g <= M*1, g >= 0.
    Returns g (optimal) or None if infeasible."""
    n = len(p)
    A = build_conv_matrix(p)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0.0)
    return None


def score(p):
    """Compute 2n * max(conv(p,p)) for probability vector p."""
    n = len(p)
    c = fftconvolve(p, p)
    return 2.0 * n * float(np.max(c))


def run_lp_iterations(p, max_iters, time_budget, verbose=True):
    """
    Run LP best-response iterations on probability vector p.
    Returns (best_p, best_M) where best_M = max(conv(best_p, best_p)).
    """
    t0 = time.monotonic()
    n = len(p)
    p = p.copy()
    p = np.maximum(p, 0)
    p = p / p.sum()
    
    c = fftconvolve(p, p)
    M = float(c.max())
    best_p = p.copy()
    best_M = M
    
    stall_count = 0
    stall_threshold = 0.001  # relative improvement threshold
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Solve LP
        g = solve_lp(p, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum  # Normalize to probability vector
        
        # Check if g is close to p (convergence)
        if np.max(np.abs(g - p)) < 1e-10:
            break
        
        # Line search: find best blend factor
        best_t = 0.0
        best_new_M = M
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            p_try = (1 - t) * p + t * g
            p_try = np.maximum(p_try, 0)
            s = p_try.sum()
            if s < 1e-15:
                continue
            p_try = p_try / s
            c_try = fftconvolve(p_try, p_try)
            new_M = float(c_try.max())
            if new_M < best_new_M:
                best_new_M = new_M
                best_t = t
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.02, 0.05, 0.08]:
                p_try = (1 - t) * p + t * g
                p_try = np.maximum(p_try, 0)
                s = p_try.sum()
                if s < 1e-15:
                    continue
                p_try = p_try / s
                c_try = fftconvolve(p_try, p_try)
                new_M = float(c_try.max())
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_t = t
        
        if best_t > 0:
            p = (1 - best_t) * p + best_t * g
            p = np.maximum(p, 0)
            p = p / p.sum()
            M = best_new_M
            
            if M < best_M * (1 - stall_threshold):
                best_M = M
                best_p = p.copy()
                stall_count = 0
            else:
                stall_count += 1
        else:
            stall_count += 1
        
        if verbose and (it < 5 or it % 50 == 0):
            elapsed = time.monotonic() - t0
            print(f"    iter {it}: M={M:.10f}, best_M={best_M:.10f}, t={best_t:.2f}, "
                  f"elapsed={elapsed:.1f}s, score={2*n*best_M:.8f}")
        
        if stall_count > 30:
            if verbose:
                print(f"    Stalled at iter {it}, M={M:.10f}")
            break
    
    return best_p, best_M
