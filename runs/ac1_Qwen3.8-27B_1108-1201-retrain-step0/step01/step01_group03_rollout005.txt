    for n in [256, 512, 1024]:
        i = np.arange(n, dtype=np.float64)
        starts.append(('chirp3', 0.3 + 0.7 * np.abs(np.cos(2 * np.pi * i * 3.0 / n))))
        starts.append(('chirp7', 0.3 + 0.7 * np.abs(np.cos(2 * np.pi * i * 7.0 / n))))
        starts.append(('sine5', np.abs(np.sin(2 * np.pi * i * 5.0 / n)) + 0.1))
        starts.append(('gauss', np.exp(-0.5 * ((i - n / 2.0) / (n * 0.22)) ** 2)))
        starts.append(('sigmoid', 1.0 / (1.0 + np.exp(-0.4 * (i - n / 2.0)))))
        # Two Gaussian bumps
        starts.append(('twin', np.exp(-0.5 * ((i - n * 0.25) / (n * 0.1)) ** 2) +
                               np.exp(-0.5 * ((i - n * 0.75) / (n * 0.1)) ** 2)))
        # Fractional Fourier-like
        starts.append(('frac', np.abs(np.cos(np.pi * i * i / (4.0 * n))) + 0.1))

    # Sort: global variable first, then by length preference
    def _priority(x):
        name, arr = x
        n = len(arr)
        if name.startswith('global'):
            cat = 0
        elif 256 <= n <= 1024:
            cat = 1
        elif n <= 256:
            cat = 2
        else:
            cat = 3
        return (cat, n)

    starts.sort(key=_priority)

    print(f"Total starts: {len(starts)}, budget: {budget_s}s")
    print(f"Starts by type: {[s[0] for s in starts[:10]]}")

    # Time allocation: give generous time to top starts
    n_starts = len(starts)
    total_time = deadline - t_start

    for idx, (name, a0) in enumerate(starts):
        remaining = deadline - time.monotonic()
        if remaining < 25.0:
            break

        # Time allocation: first 3 get 15%, next 5 get 10%, rest get 5%
        if idx < 3:
            alloc = min(0.15 * total_time, 300.0)
        elif idx < 8:
            alloc = min(0.10 * total_time, 200.0)
        else:
            alloc = min(0.04 * total_time, 120.0)
        alloc = min(alloc, remaining - 15.0)
        if alloc < 20.0:
            continue

        t0 = time.monotonic()
        print(f"[{idx+1}/{n_starts}] {name} n={len(a0)} alloc={alloc:.0f}s")

        seq, score = _warmup_lp(a0, t0 + alloc, warmup_time=min(15.0, alloc * 0.15))
        _update_best(seq, score)

        elapsed = time.monotonic() - t0
        print(f"  -> score={score:.6f}, took {elapsed:.1f}s, best={best_score:.6f}")

    # Intensive final refinement on best
    if best_seq is not None and time.monotonic() < deadline - 10.0:
        print(f"\n=== Final Refinement (current best: {best_score:.6f}) ===")
        refine_deadline = deadline - 5.0
        for pass_i in range(10):
            if time.monotonic() > refine_deadline:
                break
            remaining = refine_deadline - time.monotonic()
            if remaining < 15.0:
                break
            # Use more LP iters for refinement
            seq, score = _lp_optimize(best_seq, refine_deadline, max_iters=500, use_bisect=True)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Refine {pass_i}: {best_score:.6f}")
            else:
                # Try subgradient nudge then LP again
                n = len(best_seq)
                S_t = np.sqrt(2.0 * n)
                perturbed = _subgrad_step(best_seq, 0.02, k=5)
                perturbed = _normalize(perturbed, S_t)
                seq2, score2 = _lp_optimize(perturbed, refine_deadline, max_iters=200)
                if score2 < best_score:
                    best_score = score2
                    best_seq = seq2.copy()
                    print(f"  Perturb+LP {pass_i}: {best_score:.6f}")
                else:
                    print(f"  Stuck at pass {pass_i}")
                    break

    # Final output
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.6f} ===")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Fallback to uniform sequence.")
    return [1.0] * 256
