def normalize_sum(a, target=None):
    if a is None: return None

    a = np.clip(a, 0.0, None)

    S2 = float(a.sum())
    if S2 <= 1e-15:

def active_gradient_with_c(a, c, top_k=4):
    n = a.size
    if c is None: c = np.convolve(a,a,'full')
    peak = float(c.max())
    if top_k <= 0: top_k = 1
    if top_k >= c.size:
       idx = np.arange(c.size)
    else:
       idx = np.argpartition(c, -top_k)[-top_k:]
    ci = c[idx]
    # If all ci zero? weights uniform
    if ci.sum() > EPS:
       w = ci / ci.sum()
    else:
       w = np.ones_like(ci)/ci.size
    grad = np.zeros(n, dtype=np.float64)
    for m, wt in zip(idx, w):
       m=int(m)
       lo = 0 if m < n-1 else m-n+1
       hi = n-1 if m >= n-1 else m
       # simpler
       lo = max(0, m - n + 1)
       hi = min(n - 1, m)
       if hi >= lo:
           grad[lo:hi+1] += (2.0 * float(wt)) * a[m - hi : m - lo + 1][::-1]
    grad -= grad.mean()
    return grad, peak

def lp_descent_from_matrix(a, mat, c, M, tl):
    n = mat.shape[1]
    L = mat.shape[0]
    A = mat * 2.0
    A_ub = np.hstack((A, np.ones((L,1), dtype=np.float64)))
    b = M - c
    b = np.maximum(b, 0.0)
    c_obj = np.zeros(n+1, dtype=np.float64); c_obj[n] = -1.0
    fmax = float(a.max())
    upper = max(2.0*fmax, 1e-9)
    bounds = []
    for val in a:
        lower = -float(val) - 1e-15
        bounds.append((lower, upper))
    bounds.append((0.0, float(M) + 1e-9))
    A_eq = np.zeros((1, n+1), dtype=np.float64); A_eq[0, :n] = 1.0
    b_eq = np.zeros(1, dtype=np.float64)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options={'time_limit': float(tl)})
    except Exception:
        return None, 0.0
    if res.status == 0 and res.x is not None:
        x = np.asarray(res.x, dtype=np.float64)
        d = x[:n].copy()
        eps = float(x[n])
        if abs(float(d.sum())) > 1e-8:
            d -= d.mean()
        if eps > 1e-9:
            return d, eps
    return None, 0.0

def make_initial_pattern(n, kind, rng):
    n = int(n)
    if n <= 0: n=64
    i = np.arange(n, dtype=np.float64)
    c = (n-1)/2.0
    if kind == 'random':
       return rng.random(n)*0.5+0.25

import numpy as np
import time
from scipy.optimize import linprog

LP_N_MAX = 1024
_TINY = 1e-12


def target_sum(n):
    return float(np.sqrt(2.0 * float(n)))


def normalize_sum(a, target=None):
    """Clip to non-negative/reasonable values and normalize to target sum."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    n = a.size
    if target is None:
        target = target_sum(n)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)

    if float(a.sum()) <= 1e-15:
        return np.full(n, target / float(n), dtype=np.float64)

    # Scale to target, clipping only if necessary to respect evaluator's 1000 cap.
    for _ in range(4):
        s = float(a.sum())
        if s <= 1e-15:
            return np.full(n, target / float(n), dtype=np.float64)
        a = a * (target / s)
        if float(np.max(a)) <= 1000.0:
            break
        a = np.minimum(a, 1000.0)

    s = float(a.sum())
    if s <= 1e-15:
        return np.full(n, target / float(n), dtype=np.float64)

    a = a * (target / s)
    if float(np.max(a)) > 1000.0:
        a = np.minimum(a, 1000.0)

    return a


def score_with_conv(a):
    """Return (objective, full self-convolution), mimicking evaluator clipping."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf"), None

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size

    a = np.minimum(a, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf"), None

    c = np.convolve(a, a, "full")
    mc = float(c.max())
    return float(2.0 * n * mc / (s * s)), c


def objective(a):
    s, _ = score_with_conv(a)
    return s


def active_gradient_with_c(a, c, top_k=4):
    """Subgradient / softmax-active-set gradient for peak of a*a."""
    n = a.size
    if c is None:
        c = np.convolve(a, a, "full")

    top_k = int(top_k)
    if top_k <= 0:
        top_k = 1

    L = int(c.size)
    if top_k >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -top_k)[-top_k:]

    ci = np.asarray(c[idx], dtype=np.float64)
    s = float(ci.sum())
    if s > _TINY:
        w = ci / s
    else:
        w = np.ones_like(ci) / float(ci.size)

    grad = np.zeros(n, dtype=np.float64)
    for k in range(idx.size):
        m = int(idx[k])
        wt = float(w[k])
        if wt <= 0.0:
            continue

        lo = m - n + 1
        if lo < 0:
            lo = 0
        hi = m
        if hi > n - 1:
            hi = n - 1

        if hi >= lo:
            grad[lo:hi + 1] += (2.0 * wt) * a[m - hi:m - lo + 1][::-1]

    grad -= grad.mean()
    return grad, float(c.max())


def apply_gradient(a, grad, lr, target):
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    return normalize_sum(a, target)


def run_gradient(a, best, best_s, end_time,
                 max_iters=2000, lr0=0.25, lr1=0.02,
                 tk0=4, tk1=2, check_every=20, target=None):
    """Projected subgradient descent until end_time."""
    n = a.size
    if target is None:
        target = target_sum(n)

    for it in range(max_iters):
        if time.monotonic() >= end_time:
            break

        frac = min(1.0, float(it) / float(max(1, max_iters - 1)))
        lr = lr0 + (lr1 - lr0) * frac
        tk = int(round(float(tk0) + (float(tk1) - float(tk0)) * frac))
        if tk < 1:
            tk = 1

        c = np.convolve(a, a, "full")
        grad, _ = active_gradient_with_c(a, c, tk)
        a = apply_gradient(a, grad, lr, target)

        if (it + 1) % check_every == 0:
            s = objective(a)
            if s < best_s:
                best = a.copy()
                best_s = s

    s = objective(a)
    if s < best_s:
        best = a.copy()
        best_s = s

    return a, best, best_s


def build_matrix(f):
    """Dense matrix V with (Vg)[k] = sum_j f[k-j] g[j]."""
    n = int(f.size)
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)

    for m in range(n):
        A[m, :m + 1] = f[m::-1]

    for m in range(n, L):
        lo = m - n + 1
        A[m, lo:n] = f[n - 1:m - n:-1]

    return A


def lp_parent_from_matrix(mat, M, time_limit):
    """LP from the previous state-of-the-art direction:
       max sum(g) s.t. V_f g <= M, g >= 0.
    """
    n = int(mat.shape[1])
    if n < 3:
        return None

    c_obj = -np.ones(n, dtype=np.float64)
    b = np.full(mat.shape[0], float(M), dtype=np.float64)
    bounds = []
    for _ in range(n):
        bounds.append((0.0, None))

    try:
        res = linprog(
            c_obj,
            A_ub=mat,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"time_limit": float(time_limit)}
        )
    except Exception:
        return None

    if res is not None and res.status == 0 and res.x is not None:
        g = np.asarray(res.x, dtype=np.float64)
        g = np.maximum(g, 0.0)
        if float(g.sum()) > 1e-12 and np.all(np.isfinite(g)):
            return g

    return None


def lp_descent_from_matrix(a, mat, c, M, time_limit):
    """Linearized peak-reduction LP:
       max eps
       s.t. 2 V_a d + eps <= M - c,
            sum(d)=0,
            -a <= d <= 2*max(a), eps >= 0.
    """
    n = int(mat.shape[1])
    L = int(mat.shape[0])
    if n < 3:
        return None, 0.0

    A_ub = np.hstack((mat * 2.0, np.ones((L, 1), dtype=np.float64)))
    b_ub = np.maximum(float(M) - c, 0.0)

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = -1.0

    fmax = float(np.max(a))
    upper = max(2.0 * fmax, 1e-9)

    bounds = []
    for val in a:
        val = float(val)
        bounds.append((-val - 1e-15, upper))
    bounds.append((0.0, float(M) + 1e-9))

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1, dtype=np.float64)

    try:
        res = linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"time_limit": float(time_limit)}
        )
    except Exception:
        return None, 0.0

    if res is not None and res.status == 0 and res.x is not None:
        x = np.asarray(res.x, dtype=np.float64)
        d = x[:n].copy()
        eps = float(x[n])

        if abs(float(d.sum())) > 1e-8:
            d -= d.mean()

        if eps > 1e-8 and np.all(np.isfinite(d)):
            return d, eps

    return None, 0.0


def try_blend_candidates(a, g, target, t_list):
    """Try blends (1-t)a + t*normalize(g)."""
    best_cand = None
    best_score = float("inf")

    gn = normalize_sum(g, target)
    if gn is None:
        return None, best_score

    for t in t_list:
        t = float(t)
        cand = (1.0 - t) * a + t * gn
        cand = normalize_sum(cand, target)
        if cand is None:
            continue
        s = objective(cand)
        if s < best_score:
            best_score = s
            best_cand = cand.copy()

    return best_cand, best_score


def try_descent_candidates(a, d, target, t_list):
    """Try steps a + t*d along a linearized descent direction."""
    best_cand = None
    best_score = float("inf")

    for t in t_list:
        t = float(t)
        cand = a + t * d
        cand = np.maximum(cand, 0.0)
        cand = normalize_sum(cand, target)
        if cand is None:
            continue
        s = objective(cand)
        if s < best_score:
            best_score = s
            best_cand = cand.copy()

    return best_cand, best_score


def random_smooth_perturb(a, rng, sigma, target=None):
    """Add small smoothed noise and renormalize."""
    n = a.size
    if target is None:
        target = target_sum(n)

    if n <= 0:
        return None

    noise = rng.normal(0.0, float(sigma), size=n)
    if n >= 5:
        kernel = np.array([0.2, 0.6, 0.2], dtype=np.float64)
        noise = np.convolve(noise, kernel, mode="same")

    cand = a + noise
    return normalize_sum(cand, target)


def make_interpolated(a, n2):
    """Linearly interpolate / resample a to length n2."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    n2 = int(n2)
    if n == 0 or n2 <= 1:
        return None

    x = np.linspace(0.0, 1.0, n)
    y = np.linspace(0.0, 1.0, n2)
    return np.interp(y, x, a)


def make_initial_pattern(n, kind, rng):
    """Useful deterministic/random starts."""
    n = int(n)
    if n <= 0:
        n = 64

    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0

    if kind == "random":
        return rng.random(n) * 0.5 + 0.25

    if kind == "triangle":
        denom = max(1e-12, c)
        return np.maximum(1.0 - np.abs(i - c) / denom, 1e-6)

    if kind == "gaussian":
        return np.exp(-0.5 * ((i - c) / max(1.0, 0.2 * n)) ** 2) + 1e-3

    if kind == "ramp":
        if n == 1:
            return np.array([1.0], dtype=np.float64)
        return np.maximum(1.0 + 0.8 * (i / (n - 1.0) - 0.5), 0.05)

    if kind == "sine":
        ph1 = float(rng.uniform(0.0, 2.0 * np.pi))
        ph2 = float(rng.uniform(0.0, 2.0 * np.pi))
        ph3 = float(rng.uniform(0.0, 2.0 * np.pi))
        val = (
            1.0
            + 0.55 * np.cos(2.0 * np.pi * i / float(n) + ph1)
            + 0.30 * np.sin(4.0 * np.pi * i / float(n) + ph2)
            + 0.12 * np.sin(6.0 * np.pi * i / float(n) + ph3)
        )
        return np.maximum(val, 0.05)

    return rng.random(n) * 0.5 + 0.25

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative step-function sequence minimizing
    2*n*max(convolve(seq, seq)) / (sum(seq))^2.

    Strategy:
    - If height_sequence_1 exists, refine it heavily using two LP directions
      and subgradient polishing.
    - Also try small perturbations and a linearly upsampled version when time
      permits.
    - Without height_sequence_1, run several structured random starts.
    """
    t0 = time.monotonic()
    budget_s = float(budget_s)

    # Accept a few common alternative keyword spellings.
    for key in ("max_time", "time_budget", "time", "budget"):
        if key in kwargs:
            try:
                budget_s = max(budget_s, float(kwargs[key]))
            except Exception:
                pass

    deadline = t0 + max(1.0, budget_s - 5.0)
    rng = np.random.default_rng(int(seed))

    best_seq = None
    best_score = float("inf")

    work_items = []

    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        h = None

    if h is not None and h.size >= 8 and np.all(np.isfinite(h)):
        h = np.where(np.isfinite(h), h, 0.0)
        h = np.maximum(h, 0.0)
        h = np.minimum(h, 1000.0)

        # If unexpectedly huge, downsample to a tractable size.
        if h.size > 2 * LP_N_MAX:
            h_small = make_interpolated(h, 2 * LP_N_MAX)
            if h_small is not None:
                h = h_small

        work_items.append(("global", h, 0.70, 0.03))

        # Small perturbation of the known SOTA point.
        sigma_seed = max(1e-4, 0.05 * float(np.mean(h)) if h.size else 1e-3)
        h_noisy = h + rng.normal(0.0, sigma_seed, size=h.size)
        work_items.append(("global_noise", h_noisy, 0.15, 0.08))

        # Try a linearly upsampled version only if it stays within LP limits.
        if 8 <= h.size <= 512:
            n2 = min(LP_N_MAX, 2 * int(h.size))
            h_interp = make_interpolated(h, n2)
            if h_interp is not None:
                work_items.append(("global_interp", h_interp, 0.15, 0.12))

    else:
        # Fallback: explore several low-dimensional heuristic starts at n=512.
        n = 512
        patterns = (
            ("sine", 0.35),
            ("ramp", 0.25),
            ("gaussian", 0.20),
            ("random", 0.20),
        )
        for name, w in patterns:
            seq = make_initial_pattern(n, name, rng)
            work_items.append((name, seq, w, 0.20))

    total_w = 0.0
    for _, _, w, _ in work_items:
        total_w += float(w)

    if total_w <= 0.0:
        seq = make_initial_pattern(512, "sine", rng)
        work_items = [("random_fallback", seq, 1.0, 0.20)]
        total_w = 1.0

    time_available = max(1.0, float(deadline) - t0)
    min_alloc = 60.0 if time_available > 300.0 else max(5.0, time_available / max(1, len(work_items)))

    current = t0
    for label, seq, w, grad_frac in work_items:
        if current >= float(deadline) - 5.0:
            break

        frac = float(w) / total_w
        alloc = max(min_alloc, frac * time_available)
        end = min(float(deadline) - 2.0, current + alloc)

        if end <= current:
            end = current + max(1.0, frac * time_available)

        allow_lp = (seq is not None) and (len(np.asarray(seq)) <= LP_N_MAX)

        try:
            seq_best, score = optimize_start(
                seq,
                deadline=end,
                rng=rng,
                allow_lp=allow_lp,
                gradient_fraction=grad_frac
            )
            if seq_best is not None and np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = np.asarray(seq_best, dtype=np.float64).copy()
                print(f"[{label}] score={best_score:.6f} n={len(best_seq)}")
        except Exception as e:
            print(f"[{label}] optimizer exception: {e}")

        current = end

    if best_seq is None or not np.isfinite(best_score):
        # Safe fallback.
        a = make_initial_pattern(512, "sine", rng)
        a = normalize_sum(a)
        best_seq = a
        best_score = objective(best_seq)

    # Final validation with the official evaluator if available.
    try:
        ev = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(ev):
            if ev < best_score:
                best_score = ev
            print(f"evaluate_sequence final score={ev:.6f} n={len(best_seq)}")
        else:
            print(f"evaluate_sequence returned non-finite value; returning fallback.")
            best_seq = np.ones(512, dtype=np.float64)
            best_score = objective(best_seq)
            print(f"fallback score={best_score:.6f}")
    except Exception as e:
        print(f"evaluate_sequence unavailable or failed: {e}")

    return [float(x) for x in best_seq]
