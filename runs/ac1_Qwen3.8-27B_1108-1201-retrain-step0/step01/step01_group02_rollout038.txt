"""
Combined mirror descent + LP refinement to minimize peak auto-convolution.
Uses FFT-based convolution for speed, multiplicative weight updates for the
main optimization loop, and LP polishing for final improvement.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    if n < 64:
        return np.convolve(a, a)
    size = 1
    target = 2 * n - 1
    while size < target:
        size *= 2
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:target]
    return conv

def score_sequence(a):
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def get_subgradient(a, conv, k_star):
    """Compute subgradient of max_k (a*a)(k) at k_star."""
    n = len(a)
    g = np.zeros(n)
    # g_i = a_{k_star - i} if 0 <= k_star - i < n
    valid = (k_star >= 0) & (k_star < 2*n - 1)
    if not valid:
        return g
    # Indices: for each i in [0, n), check if k_star - i is in [0, n)
    # k_star - i >= 0 => i <= k_star
    # k_star - i < n => i > k_star - n
    i_start = max(0, k_star - n + 1)
    i_end = min(n, k_star + 1)
    for i in range(i_start, i_end):
        g[i] = a[k_star - i]
    return g

def mirror_descent(a, time_limit_s=600.0, verbose=True, seed=42):
    """Run mirror descent (multiplicative weights) on the simplex."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    a = normalize_simplex(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"  MD start: score={best_score:.6f}, n={n}")
    
    # Adaptive step size
    eta = 1.0
    patience = 0
    max_patience = 200
    
    for it in range(100000):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        score = 2.0 * n * conv[k_star]  # since sum(a) = 1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            patience = 0
            if verbose and (it % 100 == 0 or it < 10):
                print(f"  MD iter {it:5d}: score={score:.6f}")
        else:
            patience += 1
            if patience > max_patience:
                # Reduce step size and reset patience
                eta *= 0.7
                patience = 0
                if verbose and patience == 0:
                    print(f"  MD iter {it:5d}: reducing eta to {eta:.4f}")
                if eta < 1e-6:
                    break
        
        g = get_subgradient(a, conv, k_star)
        
        # Multiplicative update with normalization
        log_a = np.log(a + 1e-300) - eta * g

        a_new = np.exp(log_a - log_a.max())
        s = a_new.sum()
        if s < 1e-15:

        a_new = a_new / s

        a = a_new

        best_score = final_score
        best_a = a.copy()

    return best_a, best_score

def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))

def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    if n > 5000:
        return None  # Too large for LP
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

def lp_refine(a, time_limit_s=300.0, max_iters=200, verbose=True):
    start_time = time.monotonic()
    a = normalize_simplex(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"  LP start: score={best_score:.6f}, n={n}")
    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = fast_convolve(a)
        score = 2.0 * n * conv.max()  # sum(a) = 1
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 10 == 0 or it < 5):
                print(f"  LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 30:
            break
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.25))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g
        alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_simplex(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-14:
            a = a_new
        else:
            alpha2 = alpha * 0.3
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_simplex(a_new)
            a = a_new
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"  LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))

def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)

def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))

def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s

Combined FFT-accelerated mirror descent and LP refinement for minimizing
peak auto-convolution of non-negative sequences.

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Compute full linear convolution using FFT for efficiency."""
    n = len(a)
    if n < 128:
        return np.convolve(a, a)
    target = 2 * n - 1
    size = 1
    while size < target:
        size *= 2
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:target]
    return conv


def fast_cross_correlate(p, a):
    """Compute cross-correlation: result[i] = sum_k p[k] * a[k-i] for valid indices."""
    n = len(a)
    size = 1
    target = len(p) + n - 1
    while size < target:
        size *= 2
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a[::-1], size)
    conv = np.fft.irfft(P * A, size)
    # result[i] = conv[i + n - 1] for i = 0, ..., n-1
    return conv[n-1 : 2*n-1]


def score_sequence(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))

    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))

    return np.exp(-0.5 * ((t - center) / sigma) ** 2)

            k_star = int(np.argmax(conv))
            g = np.zeros(n)
            i_start = max(0, k_star - n + 1)
            i_end = min(n, k_star + 1)
            for i in range(i_start, i_end):
                g[i] = a[k_star - i]
        else:
            # Soft gradient using log-sum-exp
            eps = max(eps_factor * peak, 1e-10)
            shifted = conv / eps
            shifted -= shifted.max()
            p = np.exp(shifted)
            p_sum = p.sum()
            if p_sum < 1e-300:
                break
            p /= p_sum
            # g_i = sum_k p_k * a_{k-i}
            g = fast_cross_correlate(p, a)
        
        # Multiplicative update
        # Scale step to avoid underflow
        g_max = g.max()
        if g_max < 1e-15:
            break
        eta_eff = eta / g_max  # normalize so max exponent is -eta
        log_weights = -eta_eff * g
        log_weights -= log_weights.min()  # shift for stability
        weights = np.exp(log_weights)
        a_new = a * weights
        a_new = normalize_simplex(a_new)
        a = a_new
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"  MD done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def build_conv_matrix(a):
    """Build sparse matrix representing convolution with a."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    """Solve the LP: max sum(g) s.t. conv(a, g) <= M, g >= 0."""
    n = len(a)
    if n > 5000:
        return None
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=300.0, max_iters=300, verbose=True):

    if verbose:
        print(f"  LP start: score={best_score:.6f}, n={n}")
    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break

        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"  LP no improvement at iter {it}")
            break

                best_a_new = a_try

"""
Combined FFT-accelerated mirror descent and LP refinement for minimizing
peak auto-convolution of non-negative sequences. Uses multiplicative weight
updates with soft (log-sum-exp) gradients for fast iterations, followed by
LP-based directed refinement for final polish.

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Compute full linear convolution using FFT for efficiency."""
    n = len(a)
    if n < 64:
        return np.convolve(a, a)
    target = 2 * n - 1
    size = 1
    while size < target:
        size *= 2
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:target]
    return conv


def fast_cross_correlate(p, a):
    """Compute g_i = sum_k p[k]*a[k-i] for i=0..n-1 using FFT."""
    n = len(a)
    target = len(p) + n - 1
    size = 1
    while size < target:
        size *= 2
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a[::-1], size)
    conv = np.fft.irfft(P * A, size)[:target]
    return conv[n-1:2*n-1]


def score_sequence(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_gaussian(n, sigma_frac=0.25):

    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def mirror_descent(a, time_limit_s=300.0, verbose=True, mode='soft', eps_factor=0.05, seed=0):

    n = len(a)
    best_a = a.copy()

    if verbose:

                eta *= 0.6

            i_start = max(0, k_star - n + 1)
            i_end = min(n, k_star + 1)
            if i_end > i_start:

def build_conv_matrix(a):
    """Build sparse matrix for convolution constraints."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 4000:
        return None
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:

                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=300.0, max_iters=500, verbose=True):
    """LP-based iterative refinement with backtracking step sizes."""
    start_time = time.monotonic()
    a = normalize_simplex(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"  LP start: score={best_score:.6f}, n={n}")
    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = fast_convolve(a)
        peak = conv.max()
        score = 2.0 * n * peak
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 5 == 0 or it < 5):
                print(f"  LP iter {it:4d}: score={score:.6f}, t={elapsed:.0f}s")
        else:
            stagnant += 1
        if stagnant > 25:
            break
        remaining = time_limit_s - elapsed
        if remaining < 10.0:
            break
        timeout = max(3.0, min(45.0, remaining * 0.25))
        g = solve_lp_step(a, peak, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"  LP no improvement at iter {it}")
            break
        g = g / sum_g
        # Backtracking: try multiple step sizes
        best_a_new = a
        best_score_new = score
        for alpha in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05]:
            a_try = (1.0 - alpha) * a + alpha * g
            a_try = normalize_simplex(a_try)
            s_try = score_sequence(a_try)
            if s_try < best_score_new:
                best_score_new = s_try
                best_a_new = a_try
        a = best_a_new

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"  LP done: best={best_score:.6f}, {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search(a, time_limit_s=60.0, rng=None, verbose=True):
    """Random local search with multiplicative perturbations."""
    if rng is None:
        rng = np.random.default_rng(123)
    start_time = time.monotonic()
    a = normalize_simplex(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"  LS start: score={best_score:.6f}")
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        width = max(3, min(n, int(rng.integers(n // 200, n // 4))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.02, size=stop - start))
        if rng.random() < 0.15:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = normalize_simplex(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 20 == 0:
                print(f"  LS accept {accept_count}: score={best_score:.6f}")
    if verbose:
        print(f"  LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.

    Strategy:
    1. Use provided starting point (height_sequence_1) if available
    2. FFT-accelerated mirror descent with soft/hard gradients (fast bulk optimization)
    3. LP-based directed refinement (precise polish)
    4. Local search for final tweaks
    5. Multiple phases to use time effectively
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 30.0)

    best_score = float('inf')
    best_seq = None

    # Get parent sequence if available
    parent = None
    try:
        if 'height_sequence_1' in globals():
            parent = np.array(height_sequence_1, dtype=np.float64)
            parent = np.maximum(0, parent)
            s = parent.sum()
            if s > 0.01:
                parent = parent / s
                print(f"Parent: n={len(parent)}, score={score_sequence(parent):.6f}")
            else:
                parent = None
    except Exception:
        parent = None

    # === Phase 1: Mirror descent on parent (soft gradient) ===
    if parent is not None:
        remaining = deadline - time.monotonic()
        t_limit = min(remaining * 0.30, 500.0)
        if t_limit > 30:
            print(f"\n=== Phase 1a: MD soft on parent (n={len(parent)}, t={t_limit:.0f}s) ===")
            try:
                a_r, sc = mirror_descent(parent, time_limit_s=t_limit, verbose=True, mode='soft', eps_factor=0.05)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_r.copy()
                    print(f"  *** best={sc:.6f}")
            except Exception as e:
                print(f"  Phase 1a error: {e}")

    # === Phase 1b: Hard gradient MD ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_limit = min(remaining * 0.15, 200.0)
        if t_limit > 20:
            print(f"\n=== Phase 1b: MD hard (t={t_limit:.0f}s) ===")
            try:
                a_r, sc = mirror_descent(best_seq, time_limit_s=t_limit, verbose=True, mode='hard')
                if sc < best_score:
                    best_score = sc
                    best_seq = a_r.copy()
                    print(f"  *** best={sc:.6f}")
            except Exception as e:
                print(f"  Phase 1b error: {e}")

    # === Phase 2: LP refinement ===
    if best_seq is not None and len(best_seq) <= 3000:
        remaining = deadline - time.monotonic()
        t_limit = min(remaining * 0.25, 350.0)
        if t_limit > 30:
            print(f"\n=== Phase 2: LP refine (n={len(best_seq)}, t={t_limit:.0f}s) ===")
            try:
                a_r, sc = lp_refine(best_seq, time_limit_s=t_limit, verbose=True)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_r.copy()
                    print(f"  *** best={sc:.6f}")
            except Exception as e:
                print(f"  Phase 2 error: {e}")

    # === Phase 3: Try alternative initializations ===
    alts = [
        (500, 'rc'), (800, 'bm'), (500, 'bm'), (1000, 'rc'),
    ]
    for n_alt, typ in alts:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        t_limit = min(remaining * 0.12, 150.0)
        if t_limit < 25:
            continue
        print(f"\n=== Phase 3: alt n={n_alt}, type={typ} (t={t_limit:.0f}s) ===")
        try:
            if typ == 'rc':
                a_init = normalize_simplex(initial_raised_cosine(n_alt))
            else:
                a_init = normalize_simplex(initial_blackman(n_alt))
            a_r, sc = mirror_descent(a_init, time_limit_s=t_limit * 0.7, verbose=False, mode='soft', eps_factor=0.05, seed=seed + n_alt)
            if sc < best_score and n_alt <= 2000:
                remaining2 = deadline - time.monotonic()
                if remaining2 > 30:
                    a_p, sc2 = lp_refine(a_r, time_limit_s=min(remaining2 * 0.2, 100.0), verbose=False)
                    if sc2 < best_score:
                        best_score = sc2
                        best_seq = a_p.copy()
                        print(f"  *** best={sc2:.6f} from alt n={n_alt}")
        except Exception as e:
            print(f"  Phase 3 error: {e}")

    # === Phase 4: Soft MD with smaller eps (more focused) ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_limit = min(remaining * 0.20, 250.0)
        if t_limit > 20:
            print(f"\n=== Phase 4: MD soft eps=0.02 (t={t_limit:.0f}s) ===")
            try:
                a_r, sc = mirror_descent(best_seq, time_limit_s=t_limit, verbose=True, mode='soft', eps_factor=0.02)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_r.copy()
                    print(f"  *** best={sc:.6f}")
            except Exception as e:
                print(f"  Phase 4 error: {e}")

    # === Phase 5: Final LP polish ===
    if best_seq is not None and len(best_seq) <= 3000:
        remaining = deadline - time.monotonic()
        t_limit = min(remaining * 0.20, 200.0)
        if t_limit > 20:
            print(f"\n=== Phase 5: Final LP (t={t_limit:.0f}s) ===")
            try:
                a_r, sc = lp_refine(best_seq, time_limit_s=t_limit, verbose=True)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_r.copy()
                    print(f"  *** best={sc:.6f}")
            except Exception as e:
                print(f"  Phase 5 error: {e}")

    # === Phase 6: Local search ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_limit = min(remaining * 0.15, 100.0)
        if t_limit > 15:
            print(f"\n=== Phase 6: Local search (t={t_limit:.0f}s) ===")
            try:
                a_r, sc = local_search(best_seq, time_limit_s=t_limit, rng=rng, verbose=True)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_r.copy()
                    print(f"  *** best={sc:.6f}")
            except Exception as e:
                print(f"  Phase 6 error: {e}")

    # === Phase 7: Final MD squeeze ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print(f"\n=== Phase 7: Final MD (t={remaining:.0f}s) ===")
            try:
                a_r, sc = mirror_descent(best_seq, time_limit_s=remaining * 0.9, verbose=True, mode='soft', eps_factor=0.01)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_r.copy()
                    print(f"  *** best={sc:.6f}")
            except Exception as e:
                print(f"  Phase 7 error: {e}")

    # Fallback
    if best_seq is None:
        best_seq = normalize_simplex(initial_raised_cosine(500))
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Ensure valid output
    best_seq = np.maximum(0.0, best_seq)
    best_seq = np.minimum(1000.0, best_seq)
    s = best_seq.sum()
    if s < 0.01:
        best_seq = normalize_simplex(initial_raised_cosine(500))
        best_score = score_sequence(best_seq)

    print(f"\nFinal result: score={best_score:.6f}, n={len(best_seq)}")
    try:
        verify_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"Verified score via evaluate_sequence: {verify_score:.6f}")
    except Exception:
        print("Verification failed")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")

    return [float(x) for x in best_seq]
