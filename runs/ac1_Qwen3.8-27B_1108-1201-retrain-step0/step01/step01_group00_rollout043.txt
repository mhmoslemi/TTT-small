This is a system of quadratic inequalities, which is non-convex. But we can use SDP relaxation: replace a^T B_k a with tr(B_k * aa^T) and relax aa^T to X >= 0 (PSD). Then we have:
minimize 0 (or minimize trace(X) for compactness)
subject to: tr(B_k * X) <= T for all k
diag(X) >= 0 (implied by X >= 0)
trace of the "sum constraint": (1^T a)^2 = S^2... this is the tricky part.

The sum constraint (sum a_i)^2 = S^2 is equivalent to 1^T X 1 = S^2, which is linear in X! So the SDP relaxation is:
Find X >= 0 such that:
- tr(B_k * X) <= T for all k = 0, ..., 2n-2
- 1^T X 1 = S^2
- X is PSD

If this has a feasible solution with rank(X) = 1, then we've found an optimal a. Otherwise, we need to use rank-reduction techniques.

But this SDP would be huge (X is n x n, with O(n^2) variables and O(n) constraints each involving a structured matrix B_k). For n = 256, X has 256*257/2 ≈ 32896 variables. That's too many for most SDP solvers to handle efficiently within the time budget.

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """FFT-based linear convolution."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def lp_peak_reduction(a, top_k=5, epsilon=0.01, D=0.1):
    """
    Solve LP to find direction d that reduces the top-k peaks.
    Returns (d, t_opt) where d is the direction and t_opt is the LP objective.
    """
    n = len(a)
    c = fft_convolve(a, a)
    M = np.max(c)
    
    # Get top-k lag indices
    top_idx = np.argsort(c)[-top_k:]
    
    # Build LP: minimize t
    # subject to: conv(a,d)[k] <= t for k in top_idx
    #             sum(d) = 0
    #             -D <= d[i] <= D
    #
    # Variables: d[0],...,d[n-1], t (total n+1 variables)
    # 
    # conv(a,d)[k] = sum_j a[j] * d[k-j] (valid j)
    # In terms of d variables: for a given k,
    #   conv(a,d)[k] = sum_{j=0}^{n-1} a[j] * d[k-j] where 0 <= k-j <= n-1
    #   = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * d[k-j]
    #   = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[k-i] * d[i]  (substituting i = k-j)
    
    # A matrix for constraints: (len(top_idx), n+1)
    A_constraints = np.zeros((len(top_idx), n + 1))
    b_constraints = np.zeros(len(top_idx))
    
    for row, k in enumerate(top_idx):
        # conv(a,d)[k] <= t
        # sum_i a[k-i] * d[i] - t <= 0
        for i in range(max(0, k - n + 1), min(k + 1, n)):
            A_constraints[row, i] = a[k - i]
        A_constraints[row, n] = -1  # coefficient of t
    
    # Objective: minimize t => c = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Equality constraint: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -D <= d[i] <= D, t is free (but bounded for practical purposes)
    bounds = [(-D, D)] * n + [(-1.0, 1.0)]  # t bounded to [-1, 1]
    
    try:
        result = linprog(c_obj, A_ub=A_constraints, b_ub=b_constraints,
                        A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if result.success:
            d = result.x[:n]
            t_opt = result.x[n]
            return d, t_opt
    except:
        pass
    
    return None, 0.0

    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    S = np.sum(a)
    if S > 0:
        a = a / S  # normalize
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = 0
    stuck_count = 0
    
    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a, a)
        M = np.max(c)
        S = np.sum(a)
        score = 2.0 * n * M / (S * S)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Determine top-k based on how close lags are to the max
        threshold = M * 0.95
        active = np.where(c >= threshold)[0]
        top_k = min(len(active), 10)
        if top_k < 3:
            # Use top 5 by value
            top_k = 5
        
        # LP-based direction
        d, t_opt = lp_peak_reduction(a, top_k=top_k)
        
        if d is not None and t_opt < -0.001:
            # Take a step
            # Find max step size to maintain non-negativity
            neg_mask = d < 0
            if np.any(neg_mask):
                max_alpha = np.min(a[neg_mask] / (-d[neg_mask])) * 0.9
            else:
                max_alpha = 0.1
            
            alpha = min(max_alpha, 0.05)
            a_new = a + alpha * d
            a_new = np.clip(a_new, 0, None)
            a_new_sum = np.sum(a_new)
            if a_new_sum > 0.01:
                a_new = a_new / a_new_sum
                a = a_new
                stuck_count = 0
            else:
                stuck_count += 1
        else:
            stuck_count += 1
        
        # If stuck, apply perturbation
        if stuck_count > 50:
            # Random perturbation
            noise = rng.normal(0, 0.05, size=n)
            a = a + noise
            a = np.clip(a, 0, None)
            S = np.sum(a)
            if S > 0.01:
                a = a / S
            stuck_count = 0
        
        step += 1
        if step % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    step={step}, score={score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_score

    """
    Compute grad[i] = 2 * sum_m w[i+m] * a[m] for i = 0, ..., n-1
    where w has length 2n-1 and a has length n.
    w[k] = 0 for k outside [0, 2n-2].
    """

    n_w = len(w)  # 2n - 1
    # grad[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]
    # i+m ranges from i to i+n-1, which is within [0, 2n-2] for i in [0, n-1]

    # Let's reverse a: a_rev = a[::-1]
    # sum_m w[i+m]*a[m] = sum_m w[i+m]*a_rev[n-1-m]
    # Let k = n-1-m: m = n-1-k

    # conv(w, a_rev)[i+n-1] = sum_j w[j]*a_rev[i+n-1-j]
    # Let k = i+n-1-j: j = i+n-1-k
    # = sum_k w[i+n-1-k]*a_rev[k]

    conv_w_arev = fft_convolve(w, a_rev)  # length n_w + n - 1 = 3n-2
    # grad[i] = 2 * conv_w_arev[i + n - 1]
    grad = 2.0 * conv_w_arev[n-1 : 2*n - 1]
    
    return grad

Actually, wait. If sum(d) = 0 and alpha is small, then sum(a + alpha*d) = sum(a) + alpha*sum(d) = sum(a). So the sum doesn't change! Great, so we don't need to renormalize (unless the projection changes the sum).

1. Find d with sum(d) = 0, d reducing the peaks.
2. alpha = min(0.1, min_{i: d[i]<0} a[i]/(-d[i])) * 0.9
3. a = a + alpha*d (no projection needed if alpha is chosen correctly)
4. No renormalization needed since sum(d) = 0.

"""
Hybrid optimizer combining gradient descent with LP-based direction finding
for minimizing the peak autocorrelation of non-negative step functions.
Uses FFT convolution, smoothed max gradient, LP for stuck-point escape,
and multi-phase search across sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """FFT-based linear convolution of two 1-D arrays."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def smoothmax_and_weights(c, tau):
    """Compute smooth approximation to max(c) and softmax weights."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    value = c_max + tau * np.log(sum_exp)
    weights = exp_s / sum_exp
    return value, weights


def grad_smoothmax_conv(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Uses the formula: grad[i] = 2 * sum_m w[i+m] * a[m]
    = 2 * conv(w, a_rev)[i + n - 1]
    """
    n = len(a)
    c = fft_convolve(a, a)
    value, w = smoothmax_and_weights(c, tau)
    a_rev = a[::-1]
    conv_w_arev = fft_convolve(w, a_rev)
    # conv_w_arev has length (2n-1) + n - 1 = 3n-2
    grad = 2.0 * conv_w_arev[n - 1:2 * n - 1]
    return value, grad, c


def lp_direction(a, top_k=5, D=0.1):
    """
    Solve LP to find direction d that reduces top-k autocorrelation peaks.
    Minimizes t subject to conv(a,d)[k] <= t for top-k lags, sum(d)=0, |d[i]|<=D.
    Returns (d, t_opt) or (None, 0) on failure.
    """
    n = len(a)
    c = fft_convolve(a, a)
    # Get top-k lag indices (highest autocorrelation values)
    top_idx = np.argsort(c)[-top_k:]
    n_constraints = len(top_idx)
    
    # Variables: d[0],...,d[n-1], t (total n+1)
    # Objective: minimize t => c_obj = [0,...,0,1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: conv(a,d)[k] - t <= 0 for each top lag k
    # conv(a,d)[k] = sum_i a[k-i] * d[i] for valid i (0 <= k-i < n, 0 <= i < n)
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    
    for row, k in enumerate(top_idx):
        for i in range(max(0, k - n + 1), min(k + 1, n)):
            A_ub[row, i] = a[k - i]
        A_ub[row, n] = -1.0  # coefficient of t
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-D, D)] * n + [(-1.0, 1.0)]
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                         A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                         method='highs')
        if result.success:
            d = result.x[:n]
            t_opt = result.x[n]
            return d, t_opt
    except Exception:
        pass
    return None, 0.0


def optimize_gradient(a_init, time_budget, rng, tau_start=None, tau_end=None):
    """
    Gradient descent with annealed softmax on smooth max of autocorrelation.
    a is maintained as a probability vector (sum=1, non-negative).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    if S <= 0:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    if tau_start is None:
        tau_start = 5.0 / n
    if tau_end is None:
        tau_end = 0.1 / n
    
    t0 = time.monotonic()
    step = 0
    lr = 0.5
    stuck = 0
    stuck_best = best_score
    
    while time.monotonic() - t0 < time_budget:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_budget)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)
        
        value, grad, c = grad_smoothmax_conv(a, tau)
        
        # Project gradient to maintain sum(a) = 1
        grad_mean = np.mean(grad)
        grad_proj = grad - grad_mean
        
        # Adaptive step size
        alpha = lr / (np.linalg.norm(grad_proj) + 1e-10)
        alpha = min(alpha, 0.05)
        
        a_new = a - alpha * grad_proj
        a_new = np.clip(a_new, 0, None)
        S_new = np.sum(a_new)
        if S_new > 0.01:
            a_new = a_new / S_new
            new_score = compute_score(a_new)
        else:
            new_score = np.inf
        
        if np.isfinite(new_score) and new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stuck = 0
        else:
            a = a_new  # still move, but track best separately
            stuck += 1
            if stuck > 200:
                # Perturb to escape
                noise = rng.normal(0, 0.02, size=n)
                a = a + noise
                a = np.clip(a, 0, None)
                S = np.sum(a)
                if S > 0.01:
                    a = a / S
                stuck = 0
                lr *= 0.8
        
        step += 1
        if step % 500 == 0:
            lr *= 0.95
            curr_score = compute_score(a)
            print(f"      grad step={step}, curr={curr_score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")
    
    return best_a, best_score


def optimize_lp_primary(a_init, time_budget, rng):

    """
    n = len(a_init)

    lp_stuck = 0
    
    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a, a)
        M = np.max(c)
        score = 2.0 * n * M  # since sum(a)=1
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Try LP direction
        # Use top-k where k adapts based on peak structure
        threshold = M * 0.9
        n_active = np.sum(c >= threshold)
        top_k = min(max(3, n_active), 15)
        
        d, t_opt = lp_direction(a, top_k=top_k, D=0.05)
        
        took_step = False
        if d is not None and t_opt < -1e-4:
            # Find safe step size
            neg_mask = d < -1e-10
            if np.any(neg_mask):
                max_alpha = np.min(a[neg_mask] / (-d[neg_mask])) * 0.8
            else:
                max_alpha = 0.1
            alpha = min(max_alpha, 0.03)
            if alpha > 1e-6:
                a = a + alpha * d
                a = np.clip(a, 1e-10, None)
                S = np.sum(a)
                a = a / S
                took_step = True
                lp_stuck = 0
        
        if not took_step:
            lp_stuck += 1
        
        if lp_stuck > 30:
            # Use a gradient step to break free
            tau = 2.0 / n
            value, grad, c2 = grad_smoothmax_conv(a, tau)
            grad_proj = grad - np.mean(grad)
            alpha = 0.02 / (np.linalg.norm(grad_proj) + 1e-10)
            a = a - alpha * grad_proj
            a = np.clip(a, 0, None)
            S = np.sum(a)
            if S > 0.01:
                a = a / S
            lp_stuck = 0
        
        if lp_stuck > 100:
            # Random perturbation
            noise = rng.normal(0, 0.03, size=n)
            a = a + noise
            a = np.clip(a, 0.001, None)
            a = a / np.sum(a)
            lp_stuck = 0
        
        iteration += 1
        if iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={iteration}, score={score:.6f}, best={best_score:.6f}, t_opt={t_opt:.4f}, t={elapsed:.1f}s")
    
    return best_a, best_score


def polish_local(a_init, time_budget, rng):
    """
    Local polish using targeted peak reduction perturbations.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    if S > 0:
        a = a / S
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    temp = 0.001
    cooling = 0.9995
    
    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a, a)
        peak_idx = np.argmax(c)
        M = c[peak_idx]
        
        r = rng.random()
        candidate = a.copy()
        
        if r < 0.4:
            # Targeted: reduce top contributors to the peak
            # Compute contribution of each a[i] to c[peak_idx]
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
                    if i != j:
                        contribs[j] += a[i] * a[j]
            top_k = max(2, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            scale_down = np.exp(rng.normal(-0.06, 0.02, size=len(top_idx)))
            candidate[top_idx] *= scale_down
            # Compensate by increasing others
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(6, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.06, 0.02, size=n_inc))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.75:
            # Symmetrize partially
            sym = 0.5 * (a + a[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.015, size=n))
        elif r < 0.9:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, max(1, n - width)))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.025, size=width))
        else:
            # Small global
            candidate = a * np.exp(rng.normal(0, 0.01, size=n))
        
        candidate = np.clip(candidate, 0, None)
        S_c = np.sum(candidate)
        if S_c < 0.01:
            continue
        candidate = candidate / S_c
        new_score = compute_score(candidate)
        
        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
        
        curr_score = compute_score(a)
        if new_score < curr_score:
            a = candidate
        elif rng.random() < np.exp(-(new_score - curr_score) / max(temp, 1e-12)):
            a = candidate
        
        temp *= cooling
    
    return best_a, best_score


def make_init(n, init_type, rng):
    """Create initial sequence of given type."""
    if init_type == 'uniform':
        return np.ones(n) / n
    elif init_type == 'random':
        a = np.abs(rng.standard_normal(n)) + 0.1
        return a / np.sum(a)
    elif init_type == 'chirp':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.5 * np.cos(np.pi * t * t / n)
        a = np.clip(a, 0.01, None)
        return a / np.sum(a)
    elif init_type == 'multitone':
        t = np.arange(n, dtype=np.float64)
        a = 1.0
        for k, amp in [(1, 0.3), (2, 0.2), (3, 0.15), (5, 0.1), (8, 0.05)]:
            a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
        a = np.clip(a, 0.01, None)
        return a / np.sum(a)
    elif init_type == 'bell':
        t = np.arange(n, dtype=np.float64)
        center = n / 2.0
        a = np.exp(-0.5 * ((t - center) / (n / 4.0)) ** 2)
        a += 0.05 * np.abs(rng.standard_normal(n))
        return a / np.sum(a)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
        return a / np.sum(a)

                best_seq = a_res.copy()
                print(f"    *** New best: {best_score:.6f} (n={n}) ***")
    
    # Phase 2: Gradient descent refinement on best candidate
    print(f"\n=== Phase 2: Gradient refinement (best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        grad_time = min(300.0, (deadline - time.monotonic()) * 0.4)
        print(f"  Refining n={n_best} with gradient, time={grad_time:.0f}s")
        
        # Multiple gradient runs with different tau schedules
        for run_i in range(3):
            if time.monotonic() > deadline - 60:
                break
            a0 = best_seq.copy()
            if run_i > 0:
                a0 = a0 * np.exp(rng.normal(0, 0.01, size=n_best))
                a0 = a0 / np.sum(a0)
            
            tau_s = [5.0 / n_best, 10.0 / n_best, 2.0 / n_best][run_i]
            tau_e = [0.1 / n_best, 0.05 / n_best, 0.2 / n_best][run_i]
            
            t_per = grad_time / (3 - run_i)
            a_res, score_res = optimize_gradient(a0, t_per, rng, tau_start=tau_s, tau_end=tau_e)
            score_res = compute_score(a_res)
            print(f"  Grad run {run_i}: score={score_res:.6f}")
            
            if score_res < best_score:
                best_score = score_res
                best_seq = a_res.copy()
                print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 3: Local search polish
    print(f"\n=== Phase 3: Local polish (best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        polish_time = min(300.0, (deadline - time.monotonic()) * 0.5)
        print(f"  Polishing with {polish_time:.0f}s")
        a_res, score_res = polish_local(best_seq, polish_time, rng)
        score_res = compute_score(a_res)
        print(f"  After polish: score={score_res:.6f}")
        
        if score_res < best_score:
            best_score = score_res
            best_seq = a_res.copy()
            print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 4: Additional exploration
    print(f"\n=== Phase 4: Additional exploration (best={best_score:.6f}) ===")
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        extra_configs = [
            (256, 'lp'),
            (320, 'lp'),
            (448, 'grad'),
            (512, 'lp'),
            (256, 'grad'),
        ]
        
        for n, method in extra_configs:
            if time.monotonic() > deadline - 20:
                break
            t_per = min(remaining / 2.0, 120.0)
            a0 = make_init(n, 'chirp', rng)
            
            if method == 'lp':
                a_res, score_res = optimize_lp_primary(a0, t_per, rng)
            else:
                a_res, score_res = optimize_gradient(a0, t_per, rng)
            
            score_res = compute_score(a_res)
            print(f"  Extra n={n}, method={method}: score={score_res:.6f}")
            
            if score_res < best_score:
                best_score = score_res
                best_seq = a_res.copy()
                print(f"    *** New best: {best_score:.6f} ***")
    
    # Final: one more polish on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 10:
        final_time = deadline - time.monotonic() - 5
        if final_time > 10:
            a_res, score_res = polish_local(best_seq, final_time, rng)
            score_res = compute_score(a_res)
            if score_res < best_score:
                best_score = score_res
                best_seq = a_res.copy()
                print(f"\nFinal polish improved to: {best_score:.6f}")
    
    # Convert to final output
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
