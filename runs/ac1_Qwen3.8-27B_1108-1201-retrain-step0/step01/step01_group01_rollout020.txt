def smooth_objective(x, t, n):
    """Compute the smooth approximation of 2n*max(b_k) with softmax parameterization."""
    # x is the parameter vector, a = softmax(x)
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    a = exp_x / np.sum(exp_x)

    b = np.convolve(a, a)

import numpy as np
from scipy.optimize import minimize
import time

def _softmax(x):
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    return exp_x / np.sum(exp_x)

def _objective_and_grad(x, t, n):
    """Compute objective and gradient of 2n*smooth_max(b)/(sum a)^2 w.r.t. x.
    Since a = softmax(x) has sum 1, this simplifies to 2n*smooth_max(b)."""
    a = _softmax(x)
    b = np.convolve(a, a)
    
    # Log-sum-exp
    tb = t * b
    tb_max = np.max(tb)
    log_sum_exp = tb_max + np.log(np.sum(np.exp(tb - tb_max)))
    obj = 2.0 * n * log_sum_exp / t
    
    # Weights for the smooth max
    weights = np.exp(tb - tb_max)
    weights /= np.sum(weights)
    
    # Gradient w.r.t. a
    conv_wa = np.convolve(weights, a[::-1])
    grad_a = 2.0 * conv_wa[n-1:2*n-1]
    
    # Convert to gradient w.r.t. x
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x

def _make_initial(x_type, n, rng, eps=0.3):
    """Create initial x vectors for softmax parameterization."""
    if x_type == 'uniform':
        return np.zeros(n)
    elif x_type == 'anti_sym':
        x = np.zeros(n)
        x[:n//2] = eps
        x[n//2:] = -eps
        return x
    elif x_type == 'smooth_anti':
        i = np.arange(n, dtype=np.float64)
        x = eps * np.cos(np.pi * i / (n-1))
        return x
    elif x_type == 'four_level':
        x = np.zeros(n)
        q = n // 4
        x[:q] = eps
        x[q:2*q] = eps/3
        x[2*q:3*q] = -eps/3
        x[3*q:] = -eps
        return x
    elif x_type == 'random':
        return rng.normal(0, 0.1, n)
    elif x_type == 'from_sequence':
        # Convert a sequence to softmax parameters
        # a = softmax(x) => x = log(a) + const
        # We need to pass the sequence
        raise ValueError("Pass sequence separately")
    return np.zeros(n)

def _optimize_smooth(x0, n, t_schedule, rng, max_iter_per_t=100, tol=1e-8):
    """Run L-BFGS with increasing temperature schedule."""
    x = x0.copy()
    best_obj = np.inf
    best_a = None
    
    for t in t_schedule:
        # Run L-BFGS for this temperature
        result = minimize(
            _objective_and_grad,
            x,
            args=(t, n),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter_per_t, 'gtol': tol, 'disp': False}
        )
        x = result.x
        a = _softmax(x)
        obj = result.fun
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj, x

def _project_simplex(v, target=1.0):
    """Project vector onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target))[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _subgradient_polish(a, deadline, rng, M=10, eta_init=None):
    """Subgradient descent polish with exact max."""
    n = len(a)
    if eta_init is None:
        eta_init = 0.05 / np.sqrt(n)
    
    a = _project_simplex(a.copy(), 1.0)
    best = a.copy()
    best_score = _eval_exact(best)
    eta = eta_init
    no_improve = 0
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max  # sum(a) = 1
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            eta *= 0.9
            no_improve = 0
            if eta < 1e-10:
                break
        
        # Get top M peaks
        if M >= len(b):
            k_idx = np.arange(len(b))
            wts = np.ones(len(b)) / len(b)
        else:
            k_idx = np.argpartition(b, -M)[-M:]
            b_sel = b[k_idx]
            wts = np.exp(-10.0 * (b_max - b_sel) / max(b_max, 1e-15))
            wts /= np.sum(wts)
        
        # Gradient
        conv_wa = np.convolve(wts, a[::-1])
        grad_a = 2.0 * conv_wa[n-1:2*n-1]
        
        # Center for sum constraint
        grad_a -= np.mean(grad_a)
        
        # Update
        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-15:
            break
        a = a - eta * grad_a / gnorm
        a = _project_simplex(a, 1.0)
    
    return best, best_score
