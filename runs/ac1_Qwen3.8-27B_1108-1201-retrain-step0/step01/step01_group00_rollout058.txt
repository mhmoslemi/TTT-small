Hybrid gradient + local search optimizer for minimizing the autocorrelation peak
of non-negative step functions. Uses FFT-based log-sum-exp gradient, Adam optimizer
with adaptive temperature annealing, multi-start over initializations and lengths,
and targeted local search polish.

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))


def compute_soft_max_and_grad(a, T):
    """
    Compute log-sum-exp(T * (a*a)) and its gradient w.r.t. a.
    L = (1/T) * log(sum_k exp(T * c_k)) where c = a * a
    grad[j] = (2/T) * sum_k w[k] * (a[k-j] + a[j-k])
    """
    n = len(a)
    c = fft_convolve(a, a)
    
    # Compute softmax weights
    Tc = T * c
    max_Tc = np.max(Tc)
    exp_Tc = np.exp(Tc - max_Tc)
    sum_exp = np.sum(exp_Tc)
    w = exp_Tc / sum_exp
    
    # L value
    L = (max_Tc + np.log(sum_exp)) / T
    
    # Compute gradient using FFT
    # T1[j] = sum_m a[m] * w[m+j] = cross-correlation
    # T2[j] = sum_m a[m] * w[j-m] = convolution
    L_size = 1
    while L_size < 3 * n - 1:
        L_size <<= 1
    
    A = np.fft.rfft(a, L_size)
    W = np.fft.rfft(w, L_size)
    
    T1 = np.fft.irfft(np.conj(A) * W, L_size)[:n]
    T2 = np.fft.irfft(A * W, L_size)[:n]
    
    grad = (2.0 / T) * (T1 + T2)
    
    return L, grad, c


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def optimize_adam(n, theta_init, time_limit, rng):
    """
    Adam optimization with annealed soft-max temperature.
    """
    S = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf
    
    # Temperature schedule: start low, increase, then decrease
    # We want T to go from about 1/n to n and back
    tau_min = 0.5 / n
    tau_max = 50.0 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = elapsed / time_limit
        
        # Temperature: use a "U-shaped" schedule
        # Start at medium, go up (explore), then come down (exploit)
        if frac < 0.4:
            # Phase 1: increase temperature (explore)
            T = tau_min + (tau_max - tau_min) * (frac / 0.4)
        elif frac < 0.7:
            # Phase 2: high temperature (explore)
            T = tau_max
        else:
            # Phase 3: decrease temperature (exploit)
            T = tau_max * (tau_min / tau_max) ** ((frac - 0.7) / 0.3)
        
        a = softmax_normalize(theta) * S
        
        # Compute gradient
        L, grad_a, c = compute_soft_max_and_grad(a, T)
        
        # Project gradient to theta space
        # a_i = softmax(theta_i) * S
        # da_i/dtheta_j = a_i * (delta_ij - a_j / S)  (since sum(a) = S is fixed)
        # Actually: a_i = S * exp(theta_i) / sum(exp(theta))
        # da_i/dtheta_j = S * exp(theta_i) * (delta_ij * sum(exp) - exp(theta_j) * sum(exp)) / sum(exp)^2
        # = a_i * (delta_ij - a_j / S)
        
        # g_theta_j = sum_i (dL/da_i) * (da_i/dtheta_j)
        # = sum_i grad_a[i] * a_i * (delta_ij - a_j/S)

        inner = np.dot(grad_a, a)
        g_theta = a * (grad_a - inner / S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()

3. For each pair (i, j) with i + j = k*, the product a[i]*a[j] contributes to c[k*]
4. Move mass from high-product pairs to low-product pairs
5. Specifically: decrease a[i] and a[j] slightly, increase a[i'] and a[j'] where i'+j' ≠ k*

But this needs to be done carefully to not increase c at other lags.

A simpler approach that often works:
1. Compute c
2. Find the top few lags with highest c values
3. For each such lag k, find the pairs (i, j) with i+j=k that have the largest a[i]*a[j]
4. Reduce those a[i] values
5. Compensate by increasing a at positions that contribute less to all high lags

This is essentially what the local search in the previous code did. Let me refine it.

"""
Hybrid optimizer for minimizing autocorrelation peak of non-negative sequences.
Uses FFT-based gradient descent with annealed soft-max, Adam optimizer,
multi-start strategy, and targeted local search for final polish.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))


def compute_soft_max_grad(a, T):
    """
    Compute L = (1/T)*log(sum exp(T*c_k)) and dL/da where c = a*a.
    """
    n = len(a)
    c = fft_convolve(a, a)
    
    Tc = T * c
    max_Tc = np.max(Tc)
    exp_Tc = np.exp(Tc - max_Tc)
    sum_exp = np.sum(exp_Tc)
    w = exp_Tc / sum_exp
    L = (max_Tc + np.log(sum_exp)) / T
    
    # Gradient via FFT
    L_fft = 1
    while L_fft < 3 * n - 1:
        L_fft <<= 1
    
    A = np.fft.rfft(a, L_fft)
    W = np.fft.rfft(w, L_fft)
    
    T1 = np.fft.irfft(np.conj(A) * W, L_fft)[:n]
    T2 = np.fft.irfft(A * W, L_fft)[:n]
    
    grad = (2.0 / T) * (T1 + T2)
    return L, grad


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def optimize_gradient(n, theta_init, time_budget, rng):

    Adam optimization with annealed soft-max temperature.
    Returns (best_theta, best_score).
    """
    S = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    
    # Adam state
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf
    best_a = None
    
    # Temperature range
    T_min = 2.0 / n
    T_max = 80.0 / n
    
    # Learning rate schedule parameters
    lr_init = 0.5
    lr_end = 0.01
    
    no_improve_steps = 0
    patience = 3000
    
    while time.monotonic() - t0 < time_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_budget)
        
        # Temperature: start low, increase to max, then decrease
        if frac < 0.3:
            # Ramp up
            T = T_min + (T_max - T_min) * (frac / 0.3) ** 1.5
        elif frac < 0.6:
            # Hold high
            T = T_max
        else:
            # Ramp down
            T = T_max * (T_min / T_max) ** ((frac - 0.6) / 0.4)
        
        # Learning rate: decrease over time
        lr = lr_init * (lr_end / lr_init) ** frac
        
        a = softmax_normalize(theta) * S
        L, grad_a = compute_soft_max_grad(a, T)
        
        # Project gradient to theta space
        inner = np.dot(grad_a, a)
        g_theta = a * (grad_a - inner / S)
        
        # Adam update
        step += 1
        m1 = beta1 * m1 + (1 - beta1) * g_theta
        m2 = beta2 * m2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = m1 / (1.0 - beta1 ** step)
        m2_hat = m2 / (1.0 - beta2 ** step)
        
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

            # Wait: original score = 2n * max_c / (sum_a)^2 = 2n * max_c / S^2 = 2n * max_c / (2n) = max_c
            # So the score IS just max_c when S = sqrt(2n)!

            candidate += rng.normal(0, 0.001 * np.max(current), n)
            
        elif r < 0.7:
