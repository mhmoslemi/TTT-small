Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses Successive Convex Approximation (SCA) with LP solves as the primary method,
combined with momentum-based gradient descent for exploration and escape from
local minima. Multiple starting points of varying lengths are explored.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def _sca_lp(a0, S):
    """
    Solve the SCA LP: minimize M subject to
      2*(a0 * a)[k] - (a0*a0)[k] <= M  for all k
      sum(a) = S
      a >= 0
    Returns new a or None.
    """
    n = len(a0)
    a0_conv = np.convolve(a0, a0)
    
    # Build constraint matrix
    # Variables: [a[0], ..., a[n-1], M]
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = a0_conv.copy()
    
    for k in range(2 * n - 1):
        # Row k: 2*sum_j a0[k-j]*a[j] - M <= a0_conv[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[k, j] = 2.0 * a0[k - j]
        A_ub[k, n] = -1.0
    
    # Equality: sum(a) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0.0, None)] * (n + 1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    Uses soft-max weighted gradient over top peaks.
    Returns (new_a, new_momentum).

    c = np.convolve(a, a)

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses Successive Convex Approximation (SCA) with LP solves as the primary method,
combined with momentum-based gradient descent for exploration. Multiple starting
points of varying lengths are explored within the time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def _sca_lp(a0, S):
    """
    Solve the SCA LP: minimize M subject to
      2*(a0 * a)[k] - (a0*a0)[k] <= M  for all k
      sum(a) = S
      a >= 0
    Returns new a or None.
    """
    n = len(a0)
    a0_conv = np.convolve(a0, a0)

    # Build constraint matrix
    # Variables: [a[0], ..., a[n-1], M]
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = a0_conv.copy()

    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[k, j] = 2.0 * a0[k - j]
        A_ub[k, n] = -1.0

    # Equality: sum(a) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    bounds = [(0.0, None)] * (n + 1)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None


def _grad_step(a, momentum, lr, S):
    """
    One momentum-based gradient step minimizing peak convolution.
    Uses soft-max weighted gradient over top peaks.
    Returns (new_a, new_momentum).
    """
    n = len(a)
    c = np.convolve(a, a)

    # Soft-max over top peaks
    top_k = min(10, len(c))
    top_idx = np.argsort(c)[-top_k:]
    c_top = c[top_idx]
    spread = c_top[-1] - c_top[0]
    temp = max(0.01, spread / max(top_k, 1)) if spread > 0 else 1.0
    w = np.exp((c_top - c_top[0]) / temp)
    w /= w.sum()

    grad = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * wi * a[ki - j]

    # Project out mean (maintain fixed sum)
    grad -= grad.mean()

    # Momentum update
    momentum = 0.9 * momentum + lr * grad
    a = a - momentum
    a = np.maximum(a, 0.0)
    a = _normalize(a, S)
    return a, momentum


def _optimize(a0, deadline, max_sca_iters=5000, max_grad_between=3, seed=0):
    """
    Main optimization: alternate between SCA-LP steps and gradient steps.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a)
    S = np.sqrt(2.0 * n)

    score = _compute_obj(a)
    best = a.copy()
    best_score = score

    momentum = np.zeros(n)
    lr = 0.1
    stagnation_count = 0
    best_stagnation = 80

    for it in range(max_sca_iters):
        if time.monotonic() > deadline - 5.0:
            break

        # SCA LP step
        a_lp = _sca_lp(a, S)

        if a_lp is not None and a_lp.sum() > 1e-6:
            a_lp = np.maximum(a_lp, 0.0)
            a_lp = _normalize(a_lp, S)

            # Line search
            improved = False
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
                a_trial = (1.0 - alpha) * a + alpha * a_lp
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = _normalize(a_trial, S)
                s_trial = _compute_obj(a_trial)
                if s_trial < score - 1e-12:
                    a = a_trial
                    score = s_trial
                    improved = True
                    break

        if not improved:
            # Gradient step to escape
            for _ in range(max_grad_between):
                a, momentum = _grad_step(a, momentum, lr, S)
            score = _compute_obj(a)

        if score < best_score:
            best_score = score
            best = a.copy()
            stagnation_count = 0
            if best_score < 1.515:
                print(f"        iter {it}: best={best_score:.6f}")

            stagnation_count += 1

        lr = max(0.005, 0.1 * (0.9995 ** it))

        # Stagnation: random perturbation to restart

            pert = rng.random(n) * 0.15
            a = a + pert
            a = np.maximum(a, 0.0)
            a = _normalize(a, S)
            momentum = np.zeros(n)
            lr = 0.08
            score = _compute_obj(a)
            stagnation_count = 0
            best_stagnation = max(30, best_stagnation // 2)

    return best, best_score

    # Magnitude: near-flat with gentle modulation
    mod_freq = rng.uniform(1.0, 5.0)
    mag = 1.0 + 0.4 * np.cos(2.0 * np.pi * freqs * mod_freq + rng.uniform(0, 2 * np.pi))
    mag = np.maximum(mag, 0.1)
    # Boost DC
    mag[0] *= 2.0

    # Random phases
    phases = rng.random(n_freqs) * 2.0 * np.pi
    phases[0] = 0.0

    spec = mag * np.exp(1j * phases)
    a = np.fft.irfft(spec * np.sqrt(n), n=n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a)
    return a
