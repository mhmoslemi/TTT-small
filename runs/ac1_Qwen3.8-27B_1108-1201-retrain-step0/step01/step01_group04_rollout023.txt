Uses softmax parameterization, smoothed max surrogate, and Adam optimizer.

import numpy as np
from scipy.fft import fft, ifft, next_fast_len

def _conv(p):
    """Self-convolution via FFT."""
    n = p.size
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1]

def _score(p):
    """Compute the objective score."""
    n = p.size
    s = p.sum()
    if s < 1e-12:
        return np.inf
    c = _conv(p)
    return 2.0 * n * c.max() / (s * s)

def _softmax(theta):
    theta = theta - theta.max()
    e = np.exp(theta)
    return e / e.sum()

    m[:] = b1 * m + (1 - b1) * grad
    v[:] = b2 * v + (1 - b2) * grad * grad
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    theta -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return theta

        p = np.maximum(p, 0.0)
        s = p.sum()
        if s < 1e-12:
            return
        if not np.isfinite(sc):
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print(f"  NEW BEST: score={sc:.10f} n={p.size}", flush=True)

    # Baseline: uniform sequences of various lengths
    for n in (64, 128, 256, 512, 700, 1024, 2048, 4096):
        if time.monotonic() > deadline - 60:
            break
        p = np.ones(n) / n
        sc = _score(p)
        try_record(p, sc)

    print(f"Uniform baseline best: {best_score:.10f}", flush=True)

    # If parent is available, try it
    if parent is not None:
        n_par = parent.size
        sc_par = _score(parent)
        try_record(parent, sc_par)
        print(f"Parent: n={n_par} score={sc_par:.10f}", flush=True)

    # Determine which n values to optimize
    candidate_ns = [700, 1024, 2048]
    if parent is not None and parent.size in candidate_ns:
        # Already included
        pass
    elif parent is not None:
        candidate_ns.insert(0, parent.size)

    # Filter to reasonable range
    candidate_ns = [n for n in candidate_ns if 32 <= n <= 8192]
    candidate_ns = list(dict.fromkeys(candidate_ns))  # deduplicate preserving order

    # Time allocation
    elapsed = time.monotonic() - start_time
    remaining = deadline - time.monotonic()
    n_strategies = len(candidate_ns) * 2 + 1  # adam + subgrad per n, plus final

    time_per = max(10.0, remaining / max(1, n_strategies))

    # Optimization phase 1: Adam with smoothed max
    for n in candidate_ns:
        if time.monotonic() > deadline - 30:
            break
        alloc = min(time_per, deadline - time.monotonic() - 20)
        if alloc < 10:
            break

        print(f"  Adam optimizing n={n} budget={alloc:.1f}s", flush=True)

        # Multiple restarts
        n_restarts = max(1, int(alloc / 30))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            rb = min(30.0, (deadline - time.monotonic() - 20) / max(1, n_restarts - r))
            if rb < 5:
                break

            if r == 0 and parent is not None and parent.size == n:
                p_init = parent / parent.sum()
                theta0 = np.log(np.maximum(p_init, 1e-30))
            elif r == 0:
                theta0 = None
            else:
                theta0 = None

            p_best, sc_best = _optimize_adam(n, rng, rb, theta0=theta0)
            try_record(p_best, sc_best)

    # Optimization phase 2: Subgradient method
    for n in candidate_ns[:3]:
        if time.monotonic() > deadline - 30:
            break
        alloc = min(120.0, (deadline - time.monotonic() - 20) / max(1, len(candidate_ns[:3])))
        if alloc < 10:
            break

        print(f"  Subgradient n={n} budget={alloc:.1f}s", flush=True)
        p_best, sc_best = _optimize_subgrad(n, rng, alloc)
        try_record(p_best, sc_best)

    # Optimization phase 3: Mirror descent
    for n in candidate_ns[:2]:
        if time.monotonic() > deadline - 30:
            break
        alloc = min(90.0, (deadline - time.monotonic() - 20) / max(1, len(candidate_ns[:2])))
        if alloc < 10:
            break

        print(f"  Mirror descent n={n} budget={alloc:.1f}s", flush=True)
        p_best, sc_best = _optimize_mirror(n, rng, alloc)
        try_record(p_best, sc_best)

    # Final polish: short Adam refinement on best
    if best_p is not None and time.monotonic() < deadline - 15:
        n_best = best_p.size
        alloc = min(60.0, deadline - time.monotonic() - 10)
        if alloc > 10:
            print(f"  Final polish n={n_best} budget={alloc:.1f}s", flush=True)
            p_init = best_p / best_p.sum()
            theta0 = np.log(np.maximum(p_init, 1e-30))
            p_best, sc_best = _optimize_adam(n_best, rng, alloc, theta0=theta0)
            try_record(p_best, sc_best)

    # Ensure we have a valid result
    if best_p is None:
        best_p = np.ones(700) / 700
        best_score = _score(best_p)

    # Convert to list of floats, ensuring validity
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)

    s = best_p.sum()
    if s < 0.01:
        best_p = np.ones(700) / 700

    final_score = _score(best_p)
    print(f"  FINAL: n={best_p.size} score={final_score:.10f}", flush=True)

    return [float(x) for x in best_p]
