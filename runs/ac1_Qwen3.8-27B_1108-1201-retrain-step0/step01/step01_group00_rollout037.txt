   - Reduce |A[m]| for m values that contribute most to the peak
4. Inverse DFT to get new a
5. Threshold: a[i] = max(0, a[i])
6. Rescale to maintain sum = S
7. Repeat

        corr = np.fft.irfft(np.conj(fa) * fw, size)[:n]
        grad = 2.0 * corr
    else:
        # Exact subgradient
        k_star = int(np.argmax(conv))
        value = conv[k_star]
        grad = np.zeros(n)
        for j in range(n):
            k_minus_j = k_star - j
            if 0 <= k_minus_j < n:
                grad[j] = 2.0 * a[k_minus_j]
    
    return value, grad, conv

def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    # Binary search for the threshold
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def optimize_nesterov(n, a_init, time_limit, rng):
    """
    Nesterov accelerated projected gradient descent on the simplex.
    Minimizes max_conv(a) subject to sum(a) = 1, a >= 0.
    """
    a = a_init.copy()
    a = project_simplex(a)  # Ensure on simplex
    a_prev = a.copy()
    t = 1.0
    best_a = a.copy()

    tau_start = 0.1
    tau_end = 0.001

    lr = 0.01

        elapsed = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** min(1.0, elapsed * 1.5)

        t_prev = t

        y = project_simplex(y)

        value, grad, conv = max_conv_and_grad(y, tau)

            val_new, _, _ = max_conv_and_grad(candidate, 0)
            if val_new <= value + 0.01 * step * np.sum(grad * (y - candidate)):
                break
            step *= 0.5

"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions. Uses spectral initialization, Adam gradient descent on softmax
parameterization with annealing, and aggressive coordinate descent polish.
Focuses on larger sequence lengths for better asymptotic scores.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score computation: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def max_conv_smooth(a, tau):
    """Compute smoothed max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    value = max_c + tau * np.log(np.sum(exp_vals))
    # grad_j = 2 * sum_m a[m] * w[m+j] (valid range)
    # = cross_correlation(a, w)[j]
    size = 2 * n
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(fa) * fw, size)[:n]
    grad = 2.0 * corr
    return value, grad, conv


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1} via sorting method."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_init(n, rng, n_iter=50):
    """
    Spectral iterative method to find a good initial sequence.
    Alternates between time-domain (non-negativity) and frequency-domain
    (peak reduction) constraints.
    """
    a = 1.0 + 0.5 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)

    for it in range(n_iter):
        # Compute circular autocorrelation via FFT
        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        corr = np.fft.irfft(power, 2 * n)[:n]
        # Reduce peak: subtract a fraction of the correlation
        alpha = 1.0 / (n + 1.0)
        a = a - alpha * corr
        # Enforce non-negativity and normalize
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s

    return a


def ramp_init(n, rng, style='linear'):
    """Create various ramp-like initializations."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if style == 'linear':
        a = 1.0 + 0.3 * t
    elif style == 'quadratic':
        a = 1.0 + 0.4 * t ** 2
    elif style == 'sinusoidal':
        a = 1.0 + 0.3 * np.sin(np.pi * t)
    elif style == 'cosine_ramp':
        a = 0.5 + 0.5 * np.cos(np.pi * (t - 0.5))
        a = 1.0 + 0.3 * (1.0 - a)  # inverted
    elif style == 'sqrt':
        a = 1.0 + 0.5 * np.sqrt(t)
    else:
        a = 1.0 + 0.3 * t
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def optimize_adam(n, theta_init, time_limit, rng):
    """
    Adam optimization on softmax parameterization.
    Minimizes max_conv(a) where a = softmax(theta) * sqrt(2n).
    """
    S = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = 1.0
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf

    tau_start = 20.0 / n
    tau_end = 0.1 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)

        a = softmax_normalize(theta) * S
        value, g_a, conv = max_conv_smooth(a, tau)

        # Gradient w.r.t. theta: S * p * (g - dot(p, g)) where p = a/S
        p = a / S
        inner = np.dot(p, g_a)
        g_theta = S * p * (g_a - inner)

        step += 1
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * (g_theta * g_theta)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr_eff = lr / np.sqrt(step + 1)
        theta = theta - lr_eff * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate
        if step % 50 == 0:
            a_check = softmax_normalize(theta) * S
            sc = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (S * S)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()

        if step % 2000 == 0:
            a_check = softmax_normalize(theta) * S
            sc = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (S * S)
            print(f"      step={step}, score={sc:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")

    return best_theta, best_score


def coordinate_descent(a_init, time_limit, rng):
    """
    Aggressive coordinate descent / local search.
    Tries: single-element changes, pair transfers, block modifications.
    Works on the simplex (sum a = 1).
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    n = len(a)

    current_score = 2.0 * n * np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    patience = 0
    max_patience = 5000

    while time.monotonic() - t0 < time_limit and patience < max_patience:
        r = rng.random()

        if r < 0.4:
            # Pair transfer: move mass from i to j
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.15)
            amt = a[i] * frac
            candidate = a.copy()
            candidate[i] = a[i] - amt
            candidate[j] = a[j] + amt
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs * (np.sum(a) / cs) * cs  # keep same sum
            # Actually just re-normalize
            candidate = candidate / np.sum(candidate)

            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        elif r < 0.6:
            # Single element scale
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.05))
            candidate = a.copy()
            candidate[j] *= factor
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs

            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        elif r < 0.8:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 20))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, size=width))
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs

            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        else:
            # Peak-targeted: reduce elements contributing most to the peak
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Find top contributors to conv[peak_idx]
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
                    if i != j:
                        contribs[j] = a[i] * a[i]  # approximate
            top_k = max(3, n // 32)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate = a.copy()
            candidate[top_idx] *= (1.0 - rng.uniform(0.01, 0.05, size=len(top_idx)))
            # Redistribute to random other elements
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_add = min(5, len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                total_removed = np.sum(a[top_idx] - candidate[top_idx])
                candidate[add_idx] += total_removed / n_add
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs

            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

    return best_a, best_score

        t_per_init = max(5.0, t_left / (n_init + 0.5))

            if time.monotonic() > deadline - 120:
                break
            init_type = init_types[(init_i + int(np.log2(n))) % len(init_types)]

            a0 = make_init(n, init_type, rng)
            a0 = np.clip(a0, 0.001, 1000.0)

            if method == 'adam':
                theta_init = np.log(a0 / np.sum(a0) + 1e-10)
                theta_best, grad_score = optimize_adam(n, theta_init, t_per_init, rng)
                a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n)
            else:
                a_norm = a0 / np.sum(a0)
                a_best, grad_score = optimize_projected_grad(n, a_norm, t_per_init, rng)
                a_best = a_best * np.sqrt(2.0 * n)

            actual_score = score_fast(a_best)
            print(f"  n={n}, method={method}, init='{init_type}', "
                  f"score={actual_score:.6f}, time={time.monotonic()-t_global:.1f}s")

            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Phase 2: Coordinate descent polish on best sequence
    print(f"\n=== Phase 2: Coordinate descent polish ===")
    print(f"  Best before polish: {best_score:.6f}")
    if best_seq is not None:
        polish_time = min(300.0, max(30.0, deadline - time.monotonic() - 60))
        print(f"  Polish time budget: {polish_time:.0f}s")
        a_norm = best_seq / np.sum(best_seq)
        polished, polish_score_norm = coordinate_descent(a_norm, polish_time, rng)
        # Convert back to original scale
        polished_scaled = polished * np.sqrt(2.0 * len(polished))
        final_polish_score = score_fast(polished_scaled)
        print(f"  After polish: {final_polish_score:.6f}")

        if final_polish_score < best_score:
            best_score = final_polish_score
            best_seq = polished_scaled.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 3: Quick exploration with remaining time
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print(f"\n=== Phase 3: Additional exploration ===")
        remaining = deadline - time.monotonic() - 30
        # Try a few more spectral inits with different parameters
        for trial in range(3):
            if time.monotonic() > deadline - 30:
                break
            n_try = [512, 768, 1024][trial % 3]
            a0 = spectral_init(n_try, rng, n_iter=80)
            theta_init = np.log(a0 / np.sum(a0) + 1e-10)
            t_alloc = min(60.0, remaining / (4 - trial))
            theta_best, _ = optimize_adam(n_try, theta_init, t_alloc, rng)
            a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n_try)
            sc = score_fast(a_best)
            print(f"  trial {trial}: n={n_try}, score={sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
