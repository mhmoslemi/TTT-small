subject to: (f * g)[k] <= M for all k = 0, ..., 2n-2
           g_j >= 0 for all j

The constraint (f*g)[k] = sum_j f[k-j] * g_j <= M.

For n = 8192, this has 8192 variables and ~16383 constraints. The constraint matrix is a (2n-1) x n Toeplitz matrix. This is a large LP.

With scipy.optimize.linprog, this might be too slow. But with a structured solver or by using the dual...

The dual is:
min M * sum_k y_k
subject to: sum_k y_k * f[k-j] >= 1 for all j = 0, ..., n-1
           y_k >= 0

This has 2n-1 variables and n constraints. The constraint matrix is n x (2n-1), which is also Toeplitz-like.

For n=8192, that's 16383 variables and 8192 constraints. Still large.

I think for practical purposes, I'll skip the LP approach and focus on gradient-based methods with improvements.

Let me now write the code:

"""
Minimize 2*n*max(conv) / sum^2 over non-negative sequences.

Uses soft-max smoothing of the max-convolution objective combined with
Adam optimization, multiplicative updates, and randomized restarts.
Explores multiple sequence lengths and initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_weights(c, tau):
    """Compute softmax weights over the convolution values."""
    c_max = c.max()
    exp_vals = np.exp(tau * (c - c_max))
    total = exp_vals.sum()
    if total < 1e-300:
        # Fallback: uniform over top-1
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return exp_vals / total


def compute_gradient(a, c, weights):
    """Compute the gradient of the weighted sum of convolution values.
    
    d(sum_k w_k * c_k)/d(a_j) = sum_k w_k * (a[k-j] + a[j-k])
    (where terms are included only when indices are valid)
    """
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    # Only iterate over indices where weight is significant
    threshold = weights.max() * 1e-10
    active = np.where(weights > threshold)[0]
    for k in active:
        w_k = weights[k]
        k = int(k)
        # d(c_k)/d(a_j) = a[k-j] if 0 <= k-j < n, plus a[j-k] if 0 <= j-k < n
        # For a given k, as j varies:
        #   a[k-j] is valid when 0 <= k-j < n, i.e., k-n < j < k (i.e., max(0,k-n+1) <= j <= min(n-1,k))
        #   a[j-k] is valid when 0 <= j-k < n, i.e., k <= j < k+n (i.e., max(0,k) <= j <= min(n-1,k+n-1))
        # Combined: for j in [max(0,k-n+1), min(n-1,k+n-1)]:
        #   d(c_k)/d(a_j) = [a[k-j] if k-j in [0,n)) + [a[j-k] if j-k in [0,n))
        
        lo = max(0, k - n + 1)
        hi = min(n - 1, k + n - 1)
        if lo > hi:
            continue
        # For j in [lo, hi]:
        # Term 1: a[k-j] valid when 0 <= k-j < n => max(0, k-n+1) <= j <= k
        # Term 2: a[j-k] valid when 0 <= j-k < n => k <= j <= min(n-1, k+n-1)
        
        # Term 1 contribution: for j in [max(0,k-n+1), min(n-1,k)], add w_k * a[k-j]
        t1_lo = max(0, k - n + 1)
        t1_hi = min(n - 1, k)
        if t1_lo <= t1_hi:
            # a[k-j] for j = t1_lo, ..., t1_hi gives a[k-t1_hi], ..., a[k-t1_lo]
            # i.e., a[0] to a[k-t1_lo] reversed... let me think
            # j = t1_lo: a[k - t1_lo]
            # j = t1_hi: a[k - t1_hi]
            # So the values are a[k-t1_lo], a[k-t1_lo-1], ..., a[k-t1_hi] = a[0] (if t1_hi = k) or a[n-1-t1_lo+k]... 
            # Actually: k - t1_lo = k - max(0, k-n+1) and k - t1_hi = k - min(n-1, k)
            # If k <= n-1: t1_hi = k, so k - t1_hi = 0. t1_lo = max(0, k-n+1) = 0 (since k < n). So a[0] to a[k].
            # If k >= n: t1_hi = n-1, so k - t1_hi = k-n+1. t1_lo = k-n+1. So a[k-n+1] to a[k-(k-n+1)] = a[n-1].
            
            # The vector a[k-t1_hi : k-t1_lo+1] reversed gives us the values for j = t1_lo to t1_hi
            # Wait: for j = t1_hi, we need a[k - t1_hi]. For j = t1_lo, we need a[k - t1_lo].
            # So the values are a[k-t1_lo], a[k-t1_lo-1], ..., a[k-t1_hi]
            # Which is a[k-t1_hi : k-t1_lo+1][::-1]
            grad[t1_lo:t1_hi+1] += w_k * a[k-t1_hi:k-t1_lo+1][::-1]
        
        # Term 2 contribution: for j in [max(0,k), min(n-1, k+n-1)], add w_k * a[j-k]
        t2_lo = max(0, k)
        t2_hi = min(n - 1, k + n - 1)
        if t2_lo <= t2_hi:
            # For j = t2_lo: a[t2_lo - k]. For j = t2_hi: a[t2_hi - k].
            # Values are a[t2_lo-k], a[t2_lo-k+1], ..., a[t2_hi-k]
            grad[t2_lo:t2_hi+1] += w_k * a[t2_lo-k:t2_hi-k+1]
    
    return grad

    Alternative: use the fact that if we define the "reversed" convolution,
    we can compute the gradient via another convolution.

    = sum_k w_k * a_{k-j} + sum_k w_k * a_{j-k}

    sum_{k=j}^{j+n-1} w_k * a_{k-j} = sum_{m=0}^{n-1} w_{j+m} * a_m

    For fixed j, j-k ranges over [0, n-1], so k ranges over [j-n+1, j].

    # corr1[j] = sum_{u=0}^{n-1} w[u+j] * a[u]

    #   term1[j] = dot(weights[j:j+n], a)  (but weights has length 2n-1, so j+n <= 2n-1 means j <= n-1, OK)
    #   term2[j] = dot(weights[max(0,j-n+1):j+1], a[min(n-1,j):-1 if j>=n-1 else None][::-1])

    # cross_corr(w, a_flipped)[j] = sum_m w[m+j] * a_flipped[m] = sum_m w[m+j] * a[n-1-m]

    # Let me try: convolve(a, w)[n-1+j] = sum_m a[m] * w[n-1+j-m]

    # = sum_m a[m] * w[n-1+j-m]

    # = sum_k a[n-1+j-k] * w[k] (for valid k)
    # Hmm, getting complicated.
    
    # Let me just use a different approach: compute the gradient using the FFT of the weighted convolution.

    grad = 2.0 * corr[:n]
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def run_softmax_adam(a_init, n, S, max_time, rng, tau_init, tau_final, top_k_fallback=10):
    """Adam optimizer with soft-max smoothed objective.
    
    The objective is (1/tau) * log(sum exp(tau * c_k)), which smoothly approximates max(c_k).
    As tau increases, it approaches the hard max.
    """
    a = safe_normalize(a_init, n)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    # Learning rate schedule
    lr_init = 0.3 * S / n
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    
    total_iters_est = max_time / 0.05  # rough estimate of iterations per 50ms
    
    while time.monotonic() - start < max_time:
        t += 1
        elapsed = time.monotonic() - start
        
        # Anneal temperature from tau_init to tau_final
        progress = min(1.0, t / max(1, total_iters_est * 0.8))
        tau = tau_init * (tau_final / tau_init) ** progress
        tau = max(tau, 1.0)
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Compute weights
        if tau > 50:
            # For high tau, use top-k for efficiency
            top_idx = np.argsort(c)[::-1][:top_k_fallback]
            top_vals = c[top_idx]
            w_top = np.exp(tau * (top_vals - top_vals[0]))
            w_top /= w_top.sum()
            weights = np.zeros(2*n-1)
            weights[top_idx] = w_top
        else:
            weights = softmax_weights(c, tau)
        
        # Compute gradient using FFT-based method
        grad = compute_gradient_v2(a, weights)
        
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-12:
            break
        grad = grad / gnorm
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr = lr_init * (0.5 ** (float(t) / 200.0))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)
        
        # Periodic random perturbation to escape local minima
        if t % 1000 == 0 and rng.random() < 0.3:
            noise = rng.normal(0.0, 0.02 * S / n, n)
            a = project_simplex(a + noise, S)
    
    return best_local, best_score


def run_multiplicative(a_init, n, S, max_time, rng, tau_init, tau_final):
    """Multiplicative weights / mirror descent with soft-max objective."""
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    eta_init = 0.2
    
    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, t / 20000.0)
        tau = tau_init * (tau_final / tau_init) ** progress
        tau = max(tau, 1.0)
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        if tau > 50:
            top_idx = np.argsort(c)[::-1][:10]
            top_vals = c[top_idx]
            w_top = np.exp(tau * (top_vals - top_vals[0]))
            w_top /= w_top.sum()
            weights = np.zeros(2*n-1)
            weights[top_idx] = w_top
        else:
            weights = softmax_weights(c, tau)
        
        grad = compute_gradient_v2(a, weights)
        gmax = np.abs(grad).max()
        if gmax <= 1e-12:
            break
        grad = grad / gmax
        
        eta = eta_init * max(0.01, 1.0 / (1.0 + 0.002 * t))
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / s)
    
    return best_local, best_score


def run_topk_subgradient(a_init, n, S, max_time, rng, top_k, lr_init=None):
    """Adam with hard top-k subgradient (like the previous approach but with tuning)."""
    if lr_init is None:
        lr_init = 0.4 * S / n
    
    a = safe_normalize(a_init, n)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total <= 1e-15:
            break
        wgt = top_vals / total
        
        grad = np.zeros(n, dtype=np.float64)
        for w_val, k in zip(wgt, top_indices):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-12:
            break
        grad = grad / gnorm
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr = lr_init * (0.5 ** (float(t) / 300.0))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)
        
        if t % 2000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0.0, 0.01 * S / n, n)
            a = project_simplex(a + noise, S)
    
    return best_local, best_score


def make_smooth_init(n, rng, freq1=3, freq2=7, amp1=0.3, amp2=0.15):
    """Create a smooth sinusoidal initialization."""
    t = np.linspace(0.0, 1.0, n)
    base = 1.0 + amp1 * np.sin(2.0 * np.pi * freq1 * t) + amp2 * np.cos(2.0 * np.pi * freq2 * t)
    base = np.maximum(base, 0.01)
    return base * (rng.random() * 0.1 + 0.95)  # small random perturbation


def make_ramp_init(n, rng, shape='cosine'):
    """Create a ramp-shaped initialization (monotonically varying)."""
    t = np.linspace(0.0, 1.0, n)
    if shape == 'cosine':
        base = 0.5 + 0.5 * np.cos(np.pi * t)  # peaks in middle
    else:
        base = t  # linear ramp
    base = np.maximum(base, 0.01)
    return base * (rng.random() * 0.05 + 0.975)
