"""
LP-guided iterative optimization for minimizing the normalized autocorrelation peak
of a non-negative sequence. Combines LP-based direction finding with subgradient
polishing and multi-start diversification.
"""

import numpy as np
from scipy import sparse
from scipy.optimize import linprog
import time


def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) / (sum(a))^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * n * np.max(b) / (S * S))


def _normalize(a, target=1.0):
    S = np.sum(a)
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S * target


def _build_constraint_matrix(a, lags):
    """
    Build the (len(lags) x n) constraint matrix for the LP.
    Row j corresponds to lag lags[j].
    A[j, i] = a[lags[j] - i] if 0 <= lags[j] - i < n, else 0.
    This encodes (a*x)_{lags[j]} = sum_i A[j,i] * x_i.
    """
    n = len(a)
    K = len(lags)
    A = np.zeros((K, n))
    for j, k in enumerate(lags):
        # j ranges from max(0, k-n+1) to min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[j, lo:hi+1] = a[k-hi:k-lo+1][::-1]
    return A


def _lp_direction(a, M, K=200, verbose=False):
    """
    Solve LP to find direction x maximizing sum(x) subject to:
      (a*x)_k <= M for top K lags
      x >= 0
    Returns normalized x (sum=1) or None if LP fails.
    """
    n = len(a)
    b = np.convolve(a, a)
    
    # Find top K lags
    if K >= len(b):
        lags = np.arange(len(b))
    else:
        lags = np.argpartition(b, -K)[-K:]
    
    # Build constraint matrix
    A = _build_constraint_matrix(a, lags)
    
    # LP: maximize sum(x) = minimize -sum(x)
    # subject to A @ x <= M, x >= 0
    c = -np.ones(n)
    
    # Use HiGHS solver
    result = linprog(c, A_ub=A, b_ub=np.full(len(lags), M),
                     bounds=[(0, None)] * n, method='highs')
    
    if not result.success:
        return None, lags
    
    x = result.x
    S = np.sum(x)
    if S < 1e-15:
        return None, lags
    
    x = x / S  # normalize to sum=1
    return x, lags


def _lp_optimize(a, deadline, K=200, t_init=0.3, verbose=True):
    """
    LP-guided iterative optimization.
    """
    n = len(a)
    a = _normalize(a.copy(), 1.0)
    best = a.copy()
    best_score = _compute_score(best)
    
    if verbose:
        print(f"  LP start: score={best_score:.6f}")
    
    t = t_init
    no_improve = 0
    max_no_improve = 50
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        M = np.max(b)
        
        x, lags = _lp_direction(a, M, K=K)
        if x is None:
            break
        
        # Try different step sizes (backtracking)
        improved = False
        for t_try in [t, t*0.5, t*0.25, t*0.1]:
            a_new = (1 - t_try) * a + t_try * x
            a_new = np.maximum(a_new, 0)
            a_new = _normalize(a_new, 1.0)
            
            score_new = _compute_score(a_new)
            if score_new < _compute_score(a) - 1e-10:
                a = a_new
                t = max(t_try, 0.01)
                improved = True
                break
        
        if _compute_score(a) < best_score - 1e-10:
            best_score = _compute_score(a)
            best = a.copy()
            no_improve = 0
            if verbose and best_score < 1.51:
                print(f"  LP: new best! score={best_score:.6f}")
        else:
            no_improve += 1
        
        if not improved:
            t *= 0.5
            if t < 0.001:
                break
            no_improve += 1
        
        if no_improve >= max_no_improve:
            # Reheat: increase K and reset t
            K = min(K * 2, 500)
            t = t_init
            no_improve = 0
            if K >= 500:
                break
    
    if verbose:
        print(f"  LP end: best score={best_score:.6f}")
    
    return best, best_score


def _subgradient_optimize(a, deadline, rng, M_peaks=20, eta_init=None):
    """
    Subgradient descent to minimize max_k b_k subject to sum(a)=1, a>=0.
    """
    n = len(a)
    if eta_init is None:
        eta_init = 0.05 / np.sqrt(n)
    
    a = _normalize(np.asarray(a, dtype=np.float64).copy(), 1.0)
    best = a.copy()
    best_score = _compute_score(best)
    eta = eta_init
    no_improve = 0
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max  # sum=1
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 200:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-10:
                break
        
        # Get top M_peaks locations
        k_idx = np.argpartition(b, -M_peaks)[-M_peaks:]
        b_sel = b[k_idx]
        wts = np.exp(-3.0 * (b_max - b_sel) / max(b_max, 1e-15))
        wts /= np.sum(wts)
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for ki in range(M_peaks):
            if wts[ki] < 1e-8:
                continue
            k = k_idx[ki]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += wts[ki] * 2.0 * a[k - idx]
        
        # Center for sum constraint
        grad -= np.mean(grad)
        
        # Update
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        a = a - eta * grad / grad_norm
        a = np.maximum(a, 0)
        a = _normalize(a, 1.0)
    
    return best, best_score


def _make_two_level(n, eps):
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    return _normalize(a, 1.0)


def _make_smooth_anti(n, eps):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    return _normalize(a, 1.0)


def _make_four_level(n, eps):
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2*q] = 1.0 + eps/3
    a[2*q:3*q] = 1.0 - eps/3
    a[3*q:] = 1.0 - eps
    return _normalize(a, 1.0)


def _small_n_optimize(m, deadline, rng):
    """
    Optimize on a small grid of size m, then interpolate to full size.
    Returns the interpolated sequence.
    """
    # Generate diverse starts on small grid
    starts = [
        _make_two_level(m, 0.5),

        _make_smooth_anti(m, 0.6),

        np.maximum(rng.random(m), 0.01),
    ]
    starts = [_normalize(s, 1.0) for s in starts]
    
    best_small = None
    best_small_score = np.inf
    
    n_starts = len(starts)
    per_start = (deadline - time.monotonic()) / n_starts
    
    for i, s in enumerate(starts):

        d = min(time.monotonic() + per_start, deadline)
        b, sc = _lp_optimize(s, d, K=min(30, 2*m), t_init=0.3, verbose=False)
        if sc < best_small_score:
            best_small_score = sc
            best_small = b.copy()
    
    if best_small is None:
        best_small = starts[0]
    
    return best_small, best_small_score

"""
LP-guided iterative optimization with subgradient polishing for minimizing
the normalized autocorrelation peak of a non-negative sequence.
Combines linear programming for direction finding, adaptive step sizes,
small-n full-constraint solves, and multi-start diversification.
"""

import numpy as np
from scipy.optimize import linprog
import time


def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) / (sum(a))^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * n * np.max(b) / (S * S))


def _normalize(a, target=1.0):
    """Normalize a to have the given sum, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-15:
        return np.full(len(a), target / len(a))
    return a / S * target


def _build_lp_matrix(a, lags):
    """
    Build constraint matrix for LP.
    A[j, i] = a[lags[j] - i] if 0 <= lags[j] - i < n, else 0.
    So A @ x = [(a*x)_{lags[0]}, ..., (a*x)_{lags[K-1]}].
    """
    n = len(a)
    K = len(lags)
    A = np.zeros((K, n))
    for j in range(K):
        k = lags[j]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[j, lo:hi+1] = a[k-hi:k-lo+1][::-1]
    return A


def _lp_direction(a, M, K=200):
    """
    Solve LP: max sum(x) s.t. (a*x)_k <= M for top K lags, x >= 0.
    Returns normalized direction (sum=1) or None.
    """
    n = len(a)
    b = np.convolve(a, a)
    
    if K >= len(b):
        lags = np.arange(len(b))
    else:
        lags = np.argpartition(b, -K)[-K:]
    
    A = _build_lp_matrix(a, lags)
    c = -np.ones(n)
    
    try:
        result = linprog(c, A_ub=A, b_ub=np.full(K, M),
                         bounds=[(0, None)] * n, method='highs',
                         options={'presolve': True})
    except Exception:
        return None
    
    if not result.success or result.x is None:
        return None
    
    x = result.x
    S = np.sum(x)
    if S < 1e-15:
        return None
    return x / S


def _lp_optimize(a, deadline, K_init=150, t_init=0.3, verbose=True, label=""):
    """
    LP-guided iterative optimization with adaptive K and backtracking.
    """
    n = len(a)
    a = _normalize(a.copy(), 1.0)
    best = a.copy()
    best_score = _compute_score(best)
    
    if verbose:
        print(f"  [{label}] Start: score={best_score:.8f}")
    
    t = t_init
    K = K_init
    no_improve = 0
    total_iters = 0
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        M = np.max(b)
        
        x = _lp_direction(a, M, K=K)
        if x is None:
            if verbose:
                print(f"  [{label}] LP failed, stopping.")
            break
        
        total_iters += 1
        
        # Backtracking line search
        cur_score = _compute_score(a)
        improved = False
        for t_try in [t, t * 0.5, t * 0.25, t * 0.1, t * 0.05]:
            a_new = (1 - t_try) * a + t_try * x
            a_new = np.maximum(a_new, 0)
            a_new = _normalize(a_new, 1.0)
            score_new = _compute_score(a_new)
            if score_new < cur_score - 1e-12:
                a = a_new
                cur_score = score_new
                t = max(t_try, 0.005)
                improved = True
                break
        
        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
            if verbose and total_iters % 50 == 0:
                print(f"  [{label}] iter={total_iters}, score={best_score:.8f}")
        else:
            no_improve += 1
        
        if not improved:
            no_improve += 5
        
        # Adaptive strategy
        if no_improve >= 30:
            K = min(K + 50, min(500, 2 * n - 1))
            t = t_init
            no_improve = 0
            if K >= 500:
                # Try subgradient as fallback
                if verbose:
                    print(f"  [{label}] K maxed out, stopping LP phase.")
                break
    
    if verbose:
        print(f"  [{label}] End: best_score={best_score:.8f}, iters={total_iters}, K={K}")
    
    return best, best_score


def _subgradient_optimize(a, deadline, rng, M_peaks=30, eta_init=None, verbose=False, label=""):
    """
    Subgradient descent with top-M weighting.
    """
    n = len(a)
    if eta_init is None:
        eta_init = 0.03 / np.sqrt(n)
    
    a = _normalize(np.asarray(a, dtype=np.float64).copy(), 1.0)
    best = a.copy()
    best_score = _compute_score(best)
    eta = eta_init
    no_improve = 0
    
    if verbose:
        print(f"  [SG:{label}] Start: score={best_score:.8f}")
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * b_max  # sum=1
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 300:
            eta *= 0.6
            no_improve = 0
            if eta < 1e-10:
                break
        
        # Top M peaks with softmax weighting
        k_idx = np.argpartition(b, -M_peaks)[-M_peaks:]
        b_sel = b[k_idx]
        wts = np.exp(-2.0 * (b_max - b_sel) / max(b_max, 1e-15))
        wts /= np.sum(wts)
        
        # Weighted subgradient
        grad = np.zeros(n)
        for ki in range(M_peaks):
            w = wts[ki]
            if w < 1e-8:
                continue
            k = k_idx[ki]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[k - idx]
        
        # Center for sum constraint
        grad -= np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        a = a - eta * grad / gnorm
        a = _normalize(a, 1.0)
    
    if verbose:
        print(f"  [SG:{label}] End: best_score={best_score:.8f}")
    
    return best, best_score


def _make_two_level(n, eps):
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    return _normalize(a, 1.0)


def _make_smooth_anti(n, eps):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / max(1, n - 1))
    a = np.maximum(a, 0.0)
    return _normalize(a, 1.0)


def _make_four_level(n, eps):
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2*q] = 1.0 + eps / 3
    a[2*q:3*q] = 1.0 - eps / 3
    a[3*q:] = 1.0 - eps
    return _normalize(a, 1.0)


def _make_cosine_blend(n, eps1, eps2):
    """Blend of two cosine profiles."""
    i = np.arange(n, dtype=np.float64)
    x = i / max(1, n - 1)
    a = 1.0 + eps1 * np.cos(np.pi * x) + eps2 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.0)
    return _normalize(a, 1.0)

def _perturb(a, rng, scale=0.1):
    """Add smooth random perturbation."""
    n = len(a)
    noise = np.zeros(n)
    n_freqs = min(5, n // 4)
    for _ in range(n_freqs):
        f = rng.integers(1, max(2, n // 5))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    if np.max(np.abs(noise)) > 1e-10:
        noise /= np.max(np.abs(noise))
    a_new = a + scale * noise
    return _normalize(a_new, 1.0)

"""
LP-guided iterative optimization with subgradient polishing, small-n full-constraint
LP + interpolation, and multi-start diversification to minimize the normalized
autocorrelation peak of a non-negative sequence. The key innovation is using
linear programming to find the optimal descent direction at each iteration,
which converges much faster than pure subgradient methods.
"""

import numpy as np
from scipy.optimize import linprog
import time


def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) / (sum(a))^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * n * np.max(b) / (S * S))


def _normalize(a, target=1.0):
    """Normalize a to have the given sum, ensuring non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    S = np.sum(a)
    if S < 1e-15:
        return np.full(len(a), target / len(a))
    return a / S * target


def _build_lp_matrix(a, lags):
    """
    Build constraint matrix for LP.
    A[j, i] = a[lags[j] - i] if 0 <= lags[j] - i < n, else 0.
    So A @ x gives [(a*x)_{lags[0]}, ..., (a*x)_{lags[K-1]}].
    """
    n = len(a)
    K = len(lags)
    A = np.zeros((K, n))
    for j in range(K):
        k = lags[j]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[j, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return A


def _lp_direction(a, M, K=200):
    """
    Solve LP: max sum(x) s.t. (a*x)_k <= M for top K lags, x >= 0.
    Returns normalized direction (sum=1) or None.
    """
    n = len(a)
    b = np.convolve(a, a)

    if K >= len(b):
        lags = np.arange(len(b))
    else:
        lags = np.argpartition(b, -K)[-K:]

    K_actual = len(lags)
    A = _build_lp_matrix(a, lags)
    c = -np.ones(n)

    try:
        result = linprog(c, A_ub=A, b_ub=np.full(K_actual, M),
                         bounds=[(0, None)] * n, method='highs',
                         options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    x = result.x
    S = np.sum(x)
    if S < 1e-15:
        return None
    return x / S


def _lp_optimize(a, deadline, K_init=150, t_init=0.3, verbose=True, label=""):
    """
    LP-guided iterative optimization with adaptive K and backtracking line search.
    """
    n = len(a)
    a = _normalize(a.copy(), 1.0)
    best = a.copy()
    best_score = _compute_score(best)

    if verbose:
        print(f"  [{label}] Start: score={best_score:.8f}")

    t = t_init
    K = K_init
    no_improve = 0
    total_iters = 0
    momentum = 0.0
    vel = np.zeros(n)

    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        M = np.max(b)

        x = _lp_direction(a, M, K=K)
        if x is None:
            if verbose:
                print(f"  [{label}] LP failed at iter {total_iters}, stopping.")
            break

        total_iters += 1

        # Backtracking line search
        cur_score = _compute_score(a)
        improved = False
        for t_try in [t, t * 0.5, t * 0.25, t * 0.1, t * 0.05]:
            a_new = (1 - t_try) * a + t_try * x
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, 1.0)
            score_new = _compute_score(a_new)
            if score_new < cur_score - 1e-13:
                # Momentum update
                vel = momentum * vel + (1 - momentum) * (a_new - a)
                a = a_new
                cur_score = score_new
                t = max(t_try, 0.005)
                improved = True
                break

        if cur_score < best_score - 1e-11:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
            if verbose and total_iters % 100 == 0:
                print(f"  [{label}] iter={total_iters}, score={best_score:.8f}, K={K}, t={t:.4f}")
        else:
            no_improve += 1

        if not improved:
            no_improve += 10

        # Adaptive strategy: escalate K, reset step size
        if no_improve >= 40:
            K = min(K + 75, min(600, 2 * n - 1))
            t = t_init
            no_improve = 0
            momentum = min(momentum + 0.05, 0.5)
            if K >= 600:
                if verbose:
                    print(f"  [{label}] K maxed at {K}, stopping after {total_iters} iters.")
                break

    if verbose:
        print(f"  [{label}] End: best={best_score:.8f}, iters={total_iters}, K={K}, t={t:.4f}")

    return best, best_score


def _subgradient_optimize(a, deadline, rng, M_peaks=30, eta_init=None,
                          verbose=False, label=""):
    """
    Subgradient descent with top-M softmax weighting and momentum.
    """
    n = len(a)
    if eta_init is None:
        eta_init = 0.03 / np.sqrt(n)

    a = _normalize(np.asarray(a, dtype=np.float64).copy(), 1.0)
    best = a.copy()
    best_score = _compute_score(best)
    eta = eta_init
    no_improve = 0
    vel = np.zeros(n)
    mu = 0.85  # momentum

    if verbose:
        print(f"  [SG:{label}] Start: score={best_score:.8f}")

    iters = 0
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * b_max  # sum=1

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 400:
            eta *= 0.5
            no_improve = 0
            mu = max(mu - 0.1, 0.3)
            if eta < 1e-11:
                break

        # Top M peaks with softmax weighting
        k_idx = np.argpartition(b, -M_peaks)[-M_peaks:]
        b_sel = b[k_idx]
        wts = np.exp(-2.5 * (b_max - b_sel) / max(b_max, 1e-15))
        wts /= np.sum(wts)

        # Weighted subgradient
        grad = np.zeros(n)
        for ki in range(M_peaks):
            w = wts[ki]
            if w < 1e-8:
                continue
            k = k_idx[ki]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[k - idx]

        # Center for sum constraint
        grad -= np.mean(grad)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Momentum update
        vel = mu * vel + (1 - mu) * grad / gnorm
        vnorm = np.linalg.norm(vel)
        if vnorm < 1e-15:
            break
        a = a - eta * vel / vnorm
        a = _normalize(a, 1.0)
        iters += 1

    if verbose:
        print(f"  [SG:{label}] End: best={best_score:.8f}, iters={iters}")

    return best, best_score


def _make_two_level(n, eps):
    """Anti-symmetric two-level: high first half, low second half."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    return _normalize(a, 1.0)


def _make_smooth_anti(n, eps):
    """Smooth anti-symmetric: 1 + eps*cos(pi*i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / max(1, n - 1))
    a = np.maximum(a, 0.0)
    return _normalize(a, 1.0)


def _make_four_level(n, eps):
    """Four-level anti-symmetric profile."""
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2 * q] = 1.0 + eps / 3
    a[2 * q:3 * q] = 1.0 - eps / 3
    a[3 * q:] = 1.0 - eps
    return _normalize(a, 1.0)


def _make_cosine_blend(n, eps1, eps2):
    """Blend of fundamental and second harmonic."""
    i = np.arange(n, dtype=np.float64)
    x = i / max(1, n - 1)
    a = 1.0 + eps1 * np.cos(np.pi * x) + eps2 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.0)
    return _normalize(a, 1.0)


def _make_optimal_two_level(n):
    """Two-level with optimal H:L = 2:1 ratio (minimizes max of center/edge peaks)."""
    a = np.ones(n)
    a[:n // 2] = 2.0
    a[n // 2:] = 1.0
    return _normalize(a, 1.0)

def _perturb(a, rng, scale=0.1):
    """Add smooth random perturbation and re-normalize."""
    n = len(a)
    noise = np.zeros(n)
    n_freqs = min(5, max(1, n // 4))
    for _ in range(n_freqs):
        f = rng.integers(1, max(2, n // 5))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    max_abs = np.max(np.abs(noise))
    if max_abs > 1e-10:
        noise /= max_abs
    return _normalize(a + scale * noise, 1.0)

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {final_score:.8f}")
    print(f"{'=' * 60}")

    return result
