def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()  # next power of 2
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)[:size]
    return result

    result = np.fft.irfft(fa * fa, fft_len)[:size]

    result[result < 0] = 0

    n = len(a)

"""
Optimizes non-negative step function heights to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based softmax gradient descent with annealing, LP refinement,
and random perturbation to escape local minima. Multiple sequence lengths explored.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_len = 1
    while fft_len < size:
        fft_len *= 2
    fa = np.fft.rfft(a, fft_len)
    result = np.fft.irfft(fa * fa, fft_len)[:size]
    result[result < 0] = 0
    return result


def _compute_score(a):
    n = len(a)
    c = _fast_convolve(a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, S_target):
    S = a.sum()
    if S < 1e-15:
        return np.ones_like(a)
    return a * (S_target / S)


def _softmax_grad(a, tau):
    """Compute softmax-weighted gradient."""
    n = len(a)
    c = _fast_convolve(a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2*n-1)
        w[c.argmax()] = 1.0
    else:
        w /= w_sum
    # grad_j = 2 * dot(w[j:j+n], a)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(w[j:j+n], a)
    return grad, c


def _hard_grad(a):
    """Compute subgradient at the peak."""
    n = len(a)
    c = _fast_convolve(a)
    k_star = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[k_star - j]
    return grad, c


def _lp_min_peak(f, S_target, timeout=30):
    """
    LP: minimize t subject to conv(f, g) <= t, sum(g) = S_target, g >= 0.
    Returns g or None.
    """
    n = len(f)
    if n > 1500:
        return None
    
    t_start = time.monotonic()
    
    # Variables: g_0, ..., g_{n-1}, t (total n+1 variables)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Constraints:
    # (2n-1) convolution constraints: (conv(f,g))_k - t <= 0
    # 1 equality: sum(g) = S_target (as two inequalities)
    n_constraints = 2 * n - 1 + 2
    A = np.zeros((n_constraints, n + 1))
    b = np.zeros(n_constraints)
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A[k, i] = f[k - i]
        A[k, n] = -1.0  # -t
        b[k] = 0
    
    # sum(g) <= S_target
    A[2*n-1, :n] = 1.0
    b[2*n-1] = S_target
    
    # -sum(g) <= -S_target
    A[2*n, :n] = -1.0
    b[2*n] = -S_target
    
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={"time_limit": timeout, "presolve": True})
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_max_sum(f, M, timeout=30):
    """
    LP: maximize sum(g) subject to conv(f, g) <= M, g >= 0.
    This is the approach from the SOTA description.
    """
    n = len(f)
    if n > 1500:
        return None
    
    t_start = time.monotonic()
    
    c_obj = -np.ones(n)
    
    n_constraints = 2 * n - 1
    A = np.zeros((n_constraints, n))
    b = np.full(n_constraints, M)
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A[k, i] = f[k - i]
    
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={"time_limit": timeout, "presolve": True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _optimize_single(a0, deadline, max_grad_iters=5000, max_lp_iters=50,
                     use_soft=True, tau_start=10.0, tau_end=200.0):
    """Optimize a single sequence. Returns (best_seq, best_score)."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    
    S_target = np.sqrt(2.0 * n)
    a = _normalize(a, S_target)
    
    score = _compute_score(a)
    best = a.copy()
    best_score = score
    
    # Phase 1: Gradient descent
    lr_init = 0.3
    for it in range(max_grad_iters):
        if time.monotonic() > deadline - 60.0:
            break
        
        # Anneal tau and learning rate
        progress = it / max_grad_iters
        tau = tau_start + (tau_end - tau_start) * progress
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, 0.005)
        
        if use_soft:
            grad, c = _softmax_grad(a, tau)
        else:
            grad, c = _hard_grad(a)
        
        # Project out mean (fixed sum constraint)
        grad -= grad.mean()
        
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        a = _normalize(a, S_target)
        
        if (it + 1) % 200 == 0:
            s = _compute_score(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    # Phase 2: LP refinement
    if n <= 1024 and time.monotonic() < deadline - 30.0:
        max_lp = min(max_lp_iters, 30)
        for lp_it in range(max_lp):
            if time.monotonic() > deadline - 10.0:
                break
            
            # Try both LP formulations
            g = _lp_min_peak(a, S_target, timeout=20)
            if g is None:
                g = _lp_max_sum(a, float(_fast_convolve(a).max()), timeout=20)
                if g is not None:
                    g = _normalize(g, S_target)
            
            if g is None:
                break
            
            # Try blending
            improved = False
            for blend in [1.0, 0.8, 0.6, 0.4, 0.2]:
                a_new = (1.0 - blend) * a + blend * np.maximum(g, 0)
                a_new = np.maximum(a_new, 0.0)
                if a_new.sum() < 1e-10:
                    continue
                a_new = _normalize(a_new, S_target)
                s_new = _compute_score(a_new)
                if s_new < best_score:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    improved = True
                    break
            
            if not improved:
                # Small gradient step to escape
                grad, _ = _softmax_grad(a, 50.0)
                grad -= grad.mean()
                a = a - 0.05 * grad
                a = np.maximum(a, 0.0)
                a = _normalize(a, S_target)
                s = _compute_score(a)
                if s < best_score:
                    best = a.copy()
                    best_score = s
    
    return best, best_score


def _perturb_and_refine(a, rng, deadline, S_target):
    """Perturb a sequence and run a short optimization."""
    n = len(a)
    # Random perturbation
    noise_scale = 0.1
    a_pert = a + noise_scale * rng.standard_normal(n)
    a_pert = np.maximum(a_pert, 0.0)
    a_pert = _normalize(a_pert, S_target)
    
    # Short optimization

                                     max_grad_iters=1000, max_lp_iters=10,
                                     use_soft=True, tau_start=20, tau_end=100)
    return a_new, score

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_len = 1
    while fft_len < size:
        fft_len *= 2
    fa = np.fft.rfft(a, fft_len)
    result = np.fft.irfft(fa * fa, fft_len)[:size]
    np.maximum(result, 0, out=result)
    return result


def compute_score(a):
    n = len(a)
    c = fast_convolve(a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, S_target):
    S = a.sum()
    if S < 1e-15:
        out = np.ones_like(a)
        return out * (S_target / n) if False else a * 0 + S_target / len(a)
    return a * (S_target / S)


def softmax_grad_sliding(a, tau, center=True):
    """Compute softmax-weighted gradient using sliding window (Toeplitz) approach."""
    n = len(a)
    c = fast_convolve(a)
    if center:
        c = c - c.mean()
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(c.argmax())] = 1.0
    else:
        w /= w_sum
    # grad[j] = 2 * dot(w[j:j+n], a) using sliding window
    W = np.lib.stride_tricks.sliding_window_view(w, n)
    grad = 2.0 * (W @ a)
    return grad, c
