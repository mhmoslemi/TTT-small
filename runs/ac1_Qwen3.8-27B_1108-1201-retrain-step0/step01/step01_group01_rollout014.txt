def _adaptive_subgradient(a, deadline, rng):
    n = len(a)
    a = a / sum(a)
    best_a = a.copy()
    best_score = compute_score(a)
    
    eta = 0.1 / sqrt(n)
    momentum = zeros(n)
    beta = 0.9  # momentum coefficient
    
    while time < deadline:
        b = convolve(a, a)
        b_max = max(b)
        
        # Find all "active" peaks (within 1% of max)
        threshold = b_max * 0.99
        active = [k for k in range(len(b)) if b[k] >= threshold]
        
        # Compute weighted subgradient
        grad = zeros(n)
        for k in active:
            w = (b[k] - threshold) / (b_max - threshold + 1e-15) + 0.1
            grad += w * grad_bk(a, k)

        a = a - eta * momentum
        a = project_simplex(a, 1.0)

        if score(a) < best_score:
            best_score = score(a)
            best_a = a.copy()
            # Can increase eta slightly
        else:
            eta *= 0.95

"""
Accelerated subgradient descent with Nesterov momentum, FFT-based convolution,
and directed local search polish to minimize the peak normalized autocorrelation
of a non-negative sequence. Multi-start with diverse anti-symmetric profiles.
"""

import numpy as np
import time


def _next_pow2(n):
    return 1 << (n - 1).bit_length()


def _convolve_full(a):
    n = len(a)
    size = 2 * n - 1
    fs = _next_pow2(size)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]


def _score(b, n):
    return 2.0 * n * float(np.max(b))


def _grad_bk(a, k):
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad


def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _make_start(n, eps, profile='linear', rng=None):
    i = np.arange(n, dtype=np.float64)
    t = i / max(n - 1, 1)
    if profile == 'linear':
        h = 1.0 - 2.0 * t
    elif profile == 'cosine':
        h = np.cos(np.pi * t)
    elif profile == 'tanh':
        h = np.tanh(3.0 * (0.5 - t))
    elif profile == 'random':
        if rng is None:
            rng = np.random.default_rng(0)
        h = rng.standard_normal(n)
        # Make it smooth
        from scipy.ndimage import gaussian_filter1d
        h = gaussian_filter1d(h, sigma=max(1, n // 50))
        h = h - np.mean(h)
    else:
        h = 1.0 - 2.0 * t
    a = 1.0 + eps * h
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _subgradient_optimize(a_init, deadline, M=10, eta_scale=1.0, seed=0):
    n = len(a_init)
    rng = np.random.default_rng(seed)
    
    a = _project_simplex(np.asarray(a_init, dtype=np.float64), 1.0)
    x_prev = a.copy()
    
    b = _convolve_full(a)
    best_f = _score(b, n)
    best_a = a.copy()
    
    k = 0
    stagnation = 0
    
    while time.monotonic() < deadline:
        k += 1
        
        # Nesterov parameters
        if k == 1:
            momentum = 0.0
        else:
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
            t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (k - 1) ** 2)) / 2.0
            momentum = (t_k - 1.0) / t_prev
        
        # Lookahead
        y = a + momentum * (a - x_prev)
        y = _project_simplex(y, 1.0)
        
        b_y = _convolve_full(y)
        b_max = float(np.max(b_y))
        
        # Top M peaks
        if M >= len(b_y):
            k_idx = np.arange(len(b_y))
            wts = np.ones(len(b_y)) / len(b_y)
        else:
            k_idx = np.argpartition(b_y, -M)[-M:]
            b_sel = b_y[k_idx]
            wts = np.exp(-10.0 * (b_max - b_sel) / max(b_max, 1e-15))
            wts /= np.sum(wts)
        
        # Subgradient
        grad = np.zeros(n)
        for ki in range(len(k_idx)):
            w = wts[ki]
            if w > 1e-12:
                grad += w * _grad_bk(y, k_idx[ki])
        grad -= np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        eta = eta_scale / np.sqrt(n) / np.sqrt(k + 1)
        x_prev = a.copy()
        a = a - eta * grad / gnorm
        a = _project_simplex(a, 1.0)
        
        if k % 50 == 0:

            f_a = _score(b_a, n)

                stagnation = 0
            else:
                stagnation += 1

                a = _project_simplex(a, 1.0)
                x_prev = a.copy()
                stagnation = 0

    b_a = _convolve_full(a)
    f_a = _score(b_a, n)
    if f_a < best_f:
        best_f = f_a
        best_a = a.copy()
    
    return best_a, best_f


def _directed_local_search(a, deadline, initial_step=0.002, seed=42):

    Directed local search: transfer small amounts from positions
    that contribute to the max autocorrelation lag to positions
    that don't.
    """

    total_len = 2 * n - 1
    a = _project_simplex(np.asarray(a, dtype=np.float64), 1.0)
    b = _convolve_full(a)
    f = _score(b, n)
    rng = np.random.default_rng(seed)
    
    step = initial_step
    
    for sweep in range(1000):
        if time.monotonic() >= deadline:
            break
        
        # Find the active lag
        k_star = int(np.argmax(b))
        
        # Find top contributors to b_{k_star}
        # b_{k_star} = sum_i a_i * a_{k_star - i}
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo > hi:
            break
        contrib = a[lo:hi+1] * a[k_star - hi : k_star - lo : -1]
        top_contrib_idx = lo + np.argsort(contrib)[-5:]  # top 5 contributors
        
        # Find positions with low a values (good targets for receiving)
        low_positions = np.argpartition(a, 10)[:10]  # bottom 10
        
        improved_this_sweep = False
        
        for _ in range(200):
            if time.monotonic() >= deadline:
                break
            
            # Choose source from top contributors, target from low positions
            src = rng.choice(top_contrib_idx)
            tgt = rng.choice(low_positions)
            
            if src == tgt or a[src] < step:
                continue
            
            # Compute the change in b
            # b_k changes by 2*step*(a_{k-tgt} - a_{k-src})
            # We need to check the new max
            v = np.zeros(total_len)
            # a_{k-tgt} for k=0..2n-2: nonzero for k in [tgt, tgt+n-1]
            lo_t, hi_t = tgt, tgt + n
            v[lo_t:min(hi_t, total_len)] = a[:min(n, total_len - lo_t)]
            # a_{k-src} for k=0..2n-2: nonzero for k in [src, src+n-1]
            lo_s, hi_s = src, src + n
            v[lo_s:min(hi_s, total_len)] -= a[:min(n, total_len - lo_s)]

                b = b_new
                f = f_new

                low_positions = np.argpartition(a, 10)[:10]
