def _fft_len(n):

def conv_sq(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)

        return np.empty(0, dtype=np.float64)

def internal_score(p):

def surrogate_and_grad(p, T_factor):

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s

        if not np.isfinite(sc):
            sc = internal_score(p)
        # Also verify with actual evaluator
        try:
            eval_sc = evaluate_sequence([float(x) for x in p])
            if np.isfinite(eval_sc) and eval_sc < sc:
                sc = eval_sc
        except Exception:
            pass
        if np.isfinite(sc) and sc < best_score_overall:
            best_score_overall = float(sc)
            best_overall = p.copy()
            print("NEW BEST: score=%.8f n=%d" % (best_score_overall, p.size), flush=True)

    # Phase 1: Quick scoring of initial candidates for each length
    initials_by_len = {}
    for n in candidate_lengths:
        inits = generate_initials(n, rng)
        if parent is not None:
            # Resize parent to this length via interpolation
            from numpy import interp
            parent_resized = interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            inits.append(safe_normalize(parent_resized))
        scored = []
        for p in inits:
            sc = internal_score(p)
            if np.isfinite(sc):
                scored.append((sc, p))
        scored.sort(key=lambda x: x[0])
        initials_by_len[n] = scored
        for sc, p in scored[:3]:
            try_record(p, sc)

    # Phase 2: Allocate time to optimization jobs
    jobs = []
    for n in candidate_lengths:
        scored = initials_by_len.get(n, [])
        for sc, p in scored[:4]:
            jobs.append((sc, p, n))
    jobs.sort(key=lambda x: x[0])

    # Prioritize: give most time to the best scores, especially parent length
    final_reserve = max(120.0, (deadline - start_time) * 0.10)
    available = max(0.0, (deadline - final_reserve) - start_time)

    # Allocate time: top job gets 40%, next 25%, rest split
    n_jobs = max(1, len(jobs))
    if n_jobs == 1:
        times = [available]
    elif n_jobs == 2:
        times = [available * 0.55, available * 0.45]
    elif n_jobs == 3:
        times = [available * 0.45, available * 0.30, available * 0.25]
    else:
        times = [available * 0.35, available * 0.25, available * 0.20, available * 0.20]
        for i in range(len(jobs) - 4):
            times.append(0.0)

    print("Phase 2: Optimizing %d jobs over %.0f seconds" % (n_jobs, available), flush=True)

    for j in range(len(jobs)):
        if time.monotonic() > deadline - final_reserve:
            break
        sc, p, n = jobs[j]
        t_alloc = max(30.0, times[j]) if j < len(times) else 30.0
        remaining = max(0.0, deadline - final_reserve - time.monotonic())
        if remaining < 30.0:
            break
        t_alloc = min(t_alloc, remaining)
        label = "J%d(n=%d)" % (j, n)
        print(label + " starting with score=%.8f, budget=%.0fs" % (sc, t_alloc), flush=True)
        p_best, sc_best = optimize_local(p, rng, t_alloc, label + ": ")
        try_record(p_best, sc_best)
        jobs[j] = (sc_best, p_best, n)

    # Phase 3: Final polish on the overall best
    if best_overall is not None and time.monotonic() < deadline - 30.0:
        remaining = max(10.0, deadline - start_time - 10.0)
        n_best = int(best_overall.size)
        print("Final polish: n=%d, budget=%.0fs" % (n_best, remaining), flush=True)
        p_best, sc_best = optimize_local(best_overall, rng, remaining, "FINAL: ")
        try_record(p_best, sc_best)

    # If we still haven't beaten the parent, try one more thing
    if parent is not None and (best_overall is None or best_score_overall > internal_score(parent)):
        if time.monotonic() < deadline - 60.0:
            remaining = max(30.0, deadline - start_time - 30.0)
            print("Re-optimizing parent: budget=%.0fs" % remaining, flush=True)
            p_best, sc_best = optimize_local(parent, rng, remaining, "PARENT: ")
            try_record(p_best, sc_best)

    # Final output
    if best_overall is None:
        best_overall = make_uniform(base_n)
        best_score_overall = internal_score(best_overall)

    best_overall = np.asarray(best_overall, dtype=np.float64)
    best_overall = np.where(np.isfinite(best_overall), best_overall, 0.0)
    best_overall = np.maximum(best_overall, 0.0)
    best_overall = np.minimum(best_overall, 1000.0)
    s = float(best_overall.sum())
    if s > 1e-12:
        best_overall = best_overall / s  # Normalize to sum 1 for the evaluator

    print("Final: score=%.8f n=%d" % (best_score_overall, int(best_overall.size)), flush=True)
    return [float(x) for x in best_overall]
