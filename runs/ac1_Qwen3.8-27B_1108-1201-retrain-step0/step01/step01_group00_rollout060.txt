Optimization of non-negative step functions to minimize 2n*max(autoconv)/sum^2.
Uses softmax parameterization, log-sum-exp smoothing, Adam optimizer,
and multi-start with annealing. Includes iterative refinement by
re-initializing near the best solution with perturbations targeting peak locations.

import numpy as np

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def softmax_stable(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def compute_score(a):
    """Compute the actual (unsmoothed) score."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def compute_smoothed_objective(a, tau):
    """Compute smoothed objective and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    sum_a = np.sum(a)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    log_sum_exp = c_max + tau * np.log(np.sum(exp_shifted))
    w = exp_shifted / np.sum(exp_shifted)  # softmax weights
    
    # h[j] = sum_m a_m * w_{j+m} = cross-correlation of a with w
    a_rev = a[::-1]
    h_full = fft_convolve(a_rev, w)
    h = h_full[n-1:2*n-1]  # length n

    grad_score = 2.0 * n * (grad_L * sum_a - 2.0 * log_sum_exp) / (sum_a ** 3)
    
    return score, grad_score, c, w

def objective_and_grad(theta, tau, n):
    """Compute smooth objective and gradient w.r.t. theta (unconstrained)."""
    a = softmax_stable(theta)  # sum(a) = 1
    c = fft_convolve(a, a)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    L = c_max + tau * np.log(np.sum(exp_shifted))
    w = exp_shifted / np.sum(exp_shifted)
    
    # h[j] = sum_m a_m * w_{j+m}
    a_rev = a[::-1]
    h_full = fft_convolve(a_rev, w)
    h = h_full[n-1:2*n-1]
    
    # dL/da_j = 2*h[j]/tau
    # da_j/dtheta_k = a_j * (delta_jk - a_k)
    # dL/dtheta_k = sum_j (2h_j/tau) * a_j * (delta_jk - a_k)
    #             = (2/tau) * [h_k * a_k - a_k * sum_j h_j * a_j]
    #             = (2*a_k/tau) * [h_k - dot(h,a)]
    
    inner = np.dot(h, a)
    dL_dtheta = (2.0 * a / tau) * (h - inner)
    
    # score = 2n * L (since sum(a) = 1)
    score = 2.0 * n * L
    grad = 2.0 * n * dL_dtheta
    
    return score, grad

def run_adam(n, theta_init, tau_start, tau_end, time_limit, seed=42):
    """Run Adam optimization with annealed temperature."""
    rng = np.random.default_rng(seed)
    theta = theta_init.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = np.inf
    plateau = 0
    plateau_best = best_score
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Exponential annealing
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)
        
        score, grad = objective_and_grad(theta, tau, n)
        
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        v_hat = v / (1 - beta2**step)

    n = len(a)
    current = a.copy()
    current_score = compute_score(current)
    best = current.copy()

But we can't make it exactly constant with a >= 0 on [0,1] (that would require a to be a constant function, giving a triangular convolution).

For a(t) = 1 on [0,1]: (a*a)(t) = t for t in [0,1], 2-t for t in [1,2]. Max = 1.

"""
Optimization of non-negative step functions to minimize 2n*max(autoconv)/sum^2.
Uses softmax parameterization, log-sum-exp continuation method, Adam optimizer,
multi-start strategy, and targeted peak-reduction local search with gradient checks.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax_stable(theta):
    """Numerically stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def compute_score(a):
    """Compute the actual (unsmoothed) score: 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def objective_and_grad(theta, tau, n):
    """
    Compute smooth objective and gradient w.r.t. theta using softmax parameterization.
    a = softmax(theta), score = 2n * logsumexp(conv(a,a), tau)
    """
    a = softmax_stable(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    L = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    # h[j] = sum_m a[m] * w[j+m] via cross-correlation
    a_rev = a[::-1]
    h_full = fft_convolve(a_rev, w)
    h = h_full[n - 1:2 * n - 1]

    # dL/dtheta_k = (2*a_k/tau) * (h_k - dot(h,a))
    inner = np.dot(h, a)
    dL_dtheta = (2.0 * a / tau) * (h - inner)

    score = 2.0 * n * L
    grad = 2.0 * n * dL_dtheta

    return score, grad


def verify_gradient(theta, tau, n, eps=1e-5):
    """Numerical gradient check."""
    score, grad_analytic = objective_and_grad(theta, tau, n)
    grad_numeric = np.zeros(n)
    for i in range(min(n, 5)):
        theta_plus = theta.copy()
        theta_plus[i] += eps
        s_plus, _ = objective_and_grad(theta_plus, tau, n)
        theta_minus = theta.copy()
        theta_minus[i] -= eps
        s_minus, _ = objective_and_grad(theta_minus, tau, n)
        grad_numeric[i] = (s_plus - s_minus) / (2 * eps)
    diff = np.max(np.abs(grad_analytic[:5] - grad_numeric))
    return diff


def run_adam(n, theta_init, tau_start, tau_end, time_limit, seed=42, lr_scale=1.0):
    """Run Adam optimization with annealed log-sum-exp temperature."""
    rng = np.random.default_rng(seed)
    theta = theta_init.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps_adam = 0.8, 0.999, 1e-8
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = compute_score(softmax_stable(theta))
    plateau = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Gamma-function shaped annealing: fast initial drop, slow final convergence
        shape = elapsed_frac ** 0.6
        tau = tau_start * (tau_end / tau_start) ** shape

        score, grad = objective_and_grad(theta, tau, n)

        if not np.all(np.isfinite(grad)):
            break

        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # Learning rate: base decay + tau-dependent scaling
        lr = 0.3 * lr_scale / (1.0 + 0.002 * step)
        lr *= max(0.1, (tau / tau_start) ** 0.4)

        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = softmax_stable(theta)
        actual = compute_score(a)

        if np.isfinite(actual) and actual < best_score:
            best_score = actual
            best_theta = theta.copy()
            plateau = 0
        else:
            plateau += 1

        if step % 1000 == 0:
            print(f"        step={step}, score={actual:.6f}, best={best_score:.6f}, tau={tau:.5f}")

    return best_theta, best_score


def perturb_around_best(n, theta_best, time_budget, rng, n_restarts=3):
    """
    Try multiple perturbations around the best solution and optimize each briefly.
    Returns the best (theta, score) found.
    """
    best_theta = theta_best.copy()
    best_score = compute_score(softmax_stable(theta_best))
    time_per = time_budget / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > time_budget + time.monotonic() - time_budget:
            break
        # Different perturbation scales
        scales = [0.05, 0.1, 0.2, 0.03, 0.15]
        scale = scales[r % len(scales)]
        theta_pert = theta_best + rng.normal(0, scale, size=n)
        tau = 2.0 / n
        tau_end_pert = 0.1 / n

        t0 = time.monotonic()
        theta_res, score_res = run_adam(n, theta_pert, tau, tau_end_pert, time_per,
                                        seed=rng.integers(0, 10000), lr_scale=0.5)
        if score_res < best_score:
            best_score = score_res
            best_theta = theta_res.copy()
            print(f"    Perturb {r}: new best = {best_score:.6f}")

    return best_theta, best_score


def peak_reduction_search(a, time_limit, rng):
    """
    Targeted local search: identify peak lags in the autoconvolution and
    reduce contributing elements. Uses a greedy approach with rollback.
    """
    n = len(a)
    current = a.astype(np.float64).copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()

    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001]
    ss_idx = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit and ss_idx < len(step_sizes):
        c = fft_convolve(current, current)
        peak_val = np.max(c)
        peak_indices = np.where(c >= 0.98 * peak_val)[0]

        improved = False
        for pk in peak_indices[:20]:
            if time.monotonic() - t0 > time_limit:
                break

            # Find top contributors to c[pk]
            contribs = np.zeros(n)
            for i in range(min(n, pk + 1)):
                j = pk - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]

            top_k = min(8, max(2, n // 32))
            top_idx = np.argsort(contribs)[-top_k:]

            candidate = current.copy()
            ss = step_sizes[ss_idx]
            for idx in top_idx:
                candidate[idx] *= (1.0 - ss * (0.5 + rng.random()))
            # Slightly increase some non-contributing elements
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 4:
                inc_idx = rng.choice(other, size=min(4, len(other)), replace=False)
                candidate[inc_idx] *= (1.0 + ss * 0.5 * rng.random(size=len(inc_idx)))

            candidate = np.clip(candidate, 1e-10, 1000.0)
            new_score = compute_score(candidate)

            if new_score < current_score - 1e-8:
                current = candidate
                current_score = new_score
                improved = True
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
                stuck = 0
            else:
                stuck += 1

        if not improved:
            stuck += 1
            if stuck > 50:
                ss_idx += 1
                stuck = 0

    return best, best_score


def make_initialization(n, rng, init_type='random'):
    """Create various initializations."""
    t = np.arange(n, dtype=np.float64)

    if init_type == 'chirp':
        alpha = 0.3 + 0.4 * rng.random()
        freq = 1.0 + 3.0 * rng.random()
        a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * t * t / (2.0 * n) + rng.uniform(0, 2 * np.pi))
    elif init_type == 'multi_tone':
        a = 1.0
        n_tones = 3 + int(rng.integers(0, 4))
        for _ in range(n_tones):
            k = int(rng.integers(1, n // 4))
            amp = 0.05 + 0.2 * rng.random()
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    elif init_type == 'cosine_window':
        # Cosine window modulated by a low-frequency oscillation
        window = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / (n - 1)))
        mod_freq = int(rng.integers(2, max(3, n // 10)))
        a = window * (1.0 + 0.3 * np.cos(2.0 * np.pi * mod_freq * t / n))
    elif init_type == 'bipolar_smoothed':
        # Start with a bipolar-like pattern and smooth it
        a = 1.0 + 0.5 * np.sign(np.sin(2.0 * np.pi * rng.integers(2, n // 5) * t / n))
        # Smooth with a moving average
        kernel_size = max(3, n // 16)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    else:  # random
        spread = 0.3 + 0.5 * rng.random()
        a = 1.0 + spread * rng.standard_normal(n)

    a = np.clip(a, 0.01, None)
    return a
