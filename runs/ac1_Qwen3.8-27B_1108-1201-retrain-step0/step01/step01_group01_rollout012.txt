def _fft_convolve(a, a2=None):
    """Compute full convolution of a with a2 (or a with itself) using FFT."""

    n1 = len(a)
    n2 = len(a2)

    fsize = 1
    while fsize < size:
        fsize *= 2
    A = np.fft.rfft(a, fsize)
    B = np.fft.rfft(a2, fsize)
    return np.fft.irfft(A * B, fsize)[:size]

    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = _fft_convolve(a)

def _smooth_grad(a, T):
    """
    Compute gradient of T*logsumexp(conv(a,a)/T) w.r.t. a.
    Also returns the current max of conv(a,a).
    """
    n = len(a)
    size = 2 * n - 1

    while fsize < size:
        fsize *= 2

    B = np.fft.irfft(A * A.conj(), fsize)[:size]

    Bm = B - np.max(B)
    exp_B = np.exp(Bm / T)
    W = exp_B / np.sum(exp_B)

def _next_pow2(m):
    """Next power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1

def _fft_convolve(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a

    A = np.fft.rfft(a, fsize)

    a = 1.0 + eps * (1.0 - t * t)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_multi_freq(n, eps, rng):
    """Sum of a few random cosines."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(3):
        f = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = eps * rng.uniform(0.3, 1.0)
        a += amp * np.cos(2 * np.pi * f * i / n + phase)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _optimize(a0, deadline, T0=0.05, T_min=0.001, lr0=0.3,
              patience=40, dim_rate=0.85, use_mw=False):
    """
    Smooth optimization loop with temperature annealing.
    Uses either projected gradient or multiplicative weight (mirror descent).
    """
    n = len(a0)
    a = _project_simplex(a0.copy(), 1.0)

    best = a.copy()
    best_score = _compute_score(best)
    T = T0
    lr = lr0
    no_improve = 0
    stall_count = 0

    while time.monotonic() < deadline:
        grad, max_b = _smooth_grad_and_peak(a, T)
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
            stall_count = 0
        else:
            no_improve += 1

        # Adapt step size
        if no_improve >= patience:
            lr *= dim_rate
            no_improve = 0
            stall_count += 1
            if lr < 1e-10:
                break
            # Periodic reheat with perturbation
            if stall_count % 4 == 0:
                scale = 0.02 * lr / max(lr0, 1e-15)
                a = _project_simplex(a + scale * np.random.randn(n), 1.0)

        if use_mw:
            # Multiplicative weight (mirror descent)
            a_new = a * np.exp(-lr * grad / n)
            S = np.sum(a_new)
            if S < 1e-15:
                a = best.copy()
                break
            a = a_new / S
        else:
            # Projected gradient
            gc = grad - np.mean(grad)
            gn = np.linalg.norm(gc)
            if gn < 1e-15:
                break
            a = a - lr * gc / gn
            a = _project_simplex(a, 1.0)

        # Anneal temperature
        T = max(T_min, T * 0.9995)

    return best, best_score


def _optimize_subgrad_direct(a0, deadline, n_top=5, lr0=0.05, patience=60, dim_rate=0.85):
    """
    Direct subgradient: use gradient at the actual peak locations.
    More aggressive than smooth method, good for final polishing.
    """
    n = len(a0)
    a = _project_simplex(a0.copy(), 1.0)
    best = a.copy()
    best_score = _compute_score(best)
    lr = lr0
    no_improve = 0

    while time.monotonic() < deadline:
        b = _fft_convolve(a)
        max_b = np.max(b)
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            lr *= dim_rate
            no_improve = 0
            if lr < 1e-12:
                break

        # Find top peaks
        k_idx = np.argpartition(b, -n_top)[-n_top:]
        k_idx = k_idx[np.argsort(b[k_idx])[::-1]]
        b_sel = b[k_idx]
        # Weight by exponential of gap
        wts = np.exp(-10.0 * (max_b - b_sel) / max(max_b, 1e-15))
        wts /= np.sum(wts)

        # Weighted gradient
        fsize = _next_pow2(2 * n - 1)
        grad = np.zeros(n)
        for ki in range(len(k_idx)):
            if wts[ki] < 1e-10:
                continue
            k = int(k_idx[ki])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += wts[ki] * 2.0 * a[k - hi:k - lo + 1][::-1]

        gc = grad - np.mean(grad)
        gn = np.linalg.norm(gc)
        if gn < 1e-15:
            break

        a = a - lr * gc / gn
        a = _project_simplex(a, 1.0)

    return best, best_score


def _perturb(a, rng, scale=0.03):
    """Smooth random perturbation."""
    n = len(a)
    i = np.arange(n, dtype=np.float64)
    noise = np.zeros(n)
    for _ in range(rng.integers(2, 6)):
        f = rng.integers(1, max(2, n // 10))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.5, 1.0)
        noise += amp * np.cos(2 * np.pi * f * i / n + phase)
    noise /= (np.max(np.abs(noise)) + 1e-15)
    return _project_simplex(a + scale * noise, 1.0)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-start FFT-based smooth optimization minimizing normalized
    autocorrelation peak. Combines log-sum-exp smoothing, multiplicative
    weight updates, and direct subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    n = 6121

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(n)
    parent = np.clip(parent, 0.0, None)
    if parent.size != n:
        parent = np.ones(n)
    S = np.sum(parent)
    if S > 1e-10:
        parent = parent / S
    else:
        parent = np.ones(n) / n

    global_best = parent.copy()
    global_best_score = _compute_score(parent)
    print(f"Initial score: {global_best_score:.6f}")

    # Diverse starting points
    starts = [
        parent.copy(),
        _make_two_level(n, 0.5),
        _make_two_level(n, 0.8),
        _make_cos_profile(n, 0.7, 1),
        _make_cos_profile(n, 0.5, 2),
        _make_cos_profile(n, 0.9, 1),
        _make_sigmoid(n, 6.0),
        _make_quad_profile(n, 0.6),
        _make_multi_freq(n, 0.4, rng),
    ]

    n_starts = len(starts)
    total_time = max(1.0, deadline - time.monotonic())

    # Phase 1: Coarse multi-start (50% of time)
    phase1_budget = 0.50 * total_time
    time_first = 0.25 * phase1_budget
    time_each = (phase1_budget - time_first) / (n_starts - 1)

    print(f"Phase 1: Multi-start ({phase1_budget:.0f}s total)")
    for i, a_start in enumerate(starts):
        if time.monotonic() >= deadline:
            break
        t_alloc = time_first if i == 0 else time_each
        dl = min(time.monotonic() + t_alloc, deadline)

        if i == 0:
            best_a, best_s = _optimize(a_start, dl, T0=0.08, T_min=0.01,
                                        lr0=0.5, patience=30, dim_rate=0.88)
        elif i % 2 == 0:
            best_a, best_s = _optimize(a_start, dl, T0=0.05, T_min=0.01,
                                        lr0=0.4, patience=30, dim_rate=0.88,
                                        use_mw=True)
        else:
            best_a, best_s = _optimize(a_start, dl, T0=0.03, T_min=0.005,
                                        lr0=0.3, patience=30, dim_rate=0.85)

        if best_s < global_best_score:
            global_best = best_a.copy()
            global_best_score = best_s
            print(f"  Start {i+1}/{n_starts}: NEW BEST {global_best_score:.6f}")
        else:
            print(f"  Start {i+1}/{n_starts}: {best_s:.6f} (best={global_best_score:.6f})")

    # Phase 2: Medium refinement (25% of time)
    phase2_budget = 0.25 * total_time
    dl2 = min(time.monotonic() + phase2_budget, deadline)
    if global_best_score < np.inf:
        print(f"\nPhase 2: Refinement ({phase2_budget:.0f}s), current={global_best_score:.6f}")
        # Alternate between smooth and direct subgradient
        n_iter2 = 3
        per2 = (dl2 - time.monotonic()) / n_iter2
        for it in range(n_iter2):
            if time.monotonic() >= dl2:
                break
            sub_dl = min(time.monotonic() + per2, dl2)
            if it % 2 == 0:
                best_a, best_s = _optimize(
                    global_best, sub_dl, T0=0.01, T_min=0.002,
                    lr0=0.2, patience=50, dim_rate=0.9)
            else:
                best_a, best_s = _optimize_subgrad_direct(
                    global_best, sub_dl, n_top=3, lr0=0.03,
                    patience=50, dim_rate=0.9)
            if best_s < global_best_score:
                global_best = best_a.copy()
                global_best_score = best_s
                print(f"  Iter {it+1}/{n_iter2}: NEW BEST {global_best_score:.6f}")
            else:
                print(f"  Iter {it+1}/{n_iter2}: {best_s:.6f}")

    # Phase 3: Fine polishing (15% of time)
    phase3_budget = 0.15 * total_time
    dl3 = min(time.monotonic() + phase3_budget, deadline)
    if global_best_score < np.inf:
        print(f"\nPhase 3: Polishing ({phase3_budget:.0f}s), current={global_best_score:.6f}")
        best_a, best_s = _optimize(
            global_best, dl3, T0=0.003, T_min=0.0005,
            lr0=0.08, patience=100, dim_rate=0.92)
        if best_s < global_best_score:
            global_best = best_a.copy()
            global_best_score = best_s
            print(f"  Polish: NEW BEST {global_best_score:.6f}")
        else:
            print(f"  Polish: {best_s:.6f}")

        # Direct subgradient polish
        remaining3 = dl3 - time.monotonic()
        if remaining3 > 10:
            best_a2, best_s2 = _optimize_subgrad_direct(
                global_best, dl3, n_top=2, lr0=0.01,
                patience=80, dim_rate=0.92)
            if best_s2 < global_best_score:
                global_best = best_a2.copy()
                global_best_score = best_s2
                print(f"  Direct polish: NEW BEST {global_best_score:.6f}")

    # Phase 4: Diversification (10% of time)
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_best_score < np.inf:
        print(f"\nPhase 4: Diversification ({remaining:.0f}s)")
        n_div = max(1, int(remaining / 30))
        per_div = remaining / n_div

        for d in range(n_div):
            if time.monotonic() >= deadline:
                break
            a_pert = _perturb(global_best, rng, scale=0.02)
            sub_dl = min(time.monotonic() + per_div, deadline)
            best_a_d, best_s_d = _optimize(
                a_pert, sub_dl, T0=0.005, T_min=0.001,
                lr0=0.1, patience=40, dim_rate=0.88,
                use_mw=(d % 2 == 1))
            if best_s_d < global_best_score:
                global_best = best_a_d.copy()
                global_best_score = best_s_d
                print(f"  Div {d+1}/{n_div}: NEW BEST {global_best_score:.6f}")

    # Final
    result = [float(max(0.0, x)) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"\nFINAL SCORE: {final_score:.6f}")
    return result
