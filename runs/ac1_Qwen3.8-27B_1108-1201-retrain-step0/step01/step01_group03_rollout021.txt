```python
"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses subgradient descent, LP-based refinement, and multi-start strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)

def subgradient_step(a, lr, soft_tau=None):
    n = len(a)
    c = np.convolve(a, a)
    
    if soft_tau is not None:
        # Soft-max weighted subgradient
        c_shifted = c - c.max()
        w = np.exp(soft_tau * c_shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * w[k] * a[k - j]
    else:
        # Hard max subgradient
        c_max = c.max()
        active = np.where(c >= c_max - 1e-10)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * a[k - j]
        grad /= len(active)
    
    # Project to sum-zero
    grad -= grad.mean()
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-12:
        return a
    a_new /= S
    return a_new
```

Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_sum(a, target=1.0):
    """Normalize array to have given sum."""
    S = np.sum(a)
    if S < 1e-14:
        return a.copy()
    return a * (target / S)


def compute_subgrad_weights(a, soft_tau=None):
    """Compute weights w over lags for subgradient computation."""
    n = len(a)
    c = np.convolve(a, a)
    if soft_tau is not None:
        c_shifted = c - c.max()
        w = np.exp(soft_tau * c_shifted)
        w /= w.sum()
    else:
        c_max = float(c.max())
        tol = max(c_max * 1e-4, 1e-12)
        w = (c >= c_max - tol).astype(np.float64)
        sw = w.sum()
        if sw < 1e-14:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= sw
    return w


def compute_subgradient(a, w):
    """Compute subgradient using precomputed lag weights."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        d_max = min(n, len(w) - j)
        if d_max > 0:
            grad[j] = 2.0 * np.dot(a[:d_max], w[j:j + d_max])
    return grad


def project_simplex(a):
    """Project onto non-negative simplex (sum=1, a>=0)."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-14:
        return np.full_like(a, 1.0 / len(a))
    return a / S
