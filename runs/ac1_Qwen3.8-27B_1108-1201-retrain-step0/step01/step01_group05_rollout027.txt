def compute_subgradient(a, c, top_indices):
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total <= 1e-15:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]

def compute_smooth_subgradient(a, c, T, max_active=200):
    """Gradient of T*log(sum(exp(c/T))) approximated with active set."""
    n = len(a)
    c_max = c.max()

    S = np.sqrt(2.0 * float(n))

    return a * (S / s)

    v = np.zeros(n, dtype=np.float64)

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
  1. Multiple structured initializations: given sequence, chirp-like, sinusoidal.
  2. Phase 1: Adam with log-sum-exp smoothed objective, adaptive temperature annealing.
  3. Phase 2: Focused top-k subgradient Adam for fine polishing.
  4. Phase 3: Coordinate descent on worst peaks.
  5. Restart mechanism to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, c, top_indices):
    """Weighted subgradient of max(c) using top-k active lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total <= 1e-15:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def compute_smooth_subgradient(a, c, T, max_active=150):
    """Approximate gradient of T*logsumexp(c/T) using active set."""
    n = len(a)
    c_max = c.max()
    scaled = (c - c_max) / T
    scaled = np.clip(scaled, -40.0, 40.0)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum <= 1e-15:
        return np.zeros(n)
    w = w / w_sum

    threshold = 1e-5
    active_mask = w > threshold
    top_idx = np.argsort(c)[::-1][:30]
    active_mask[top_idx] = True
    active = np.where(active_mask)[0]

    if len(active) > max_active:
        active_weights = w[active]
        keep = np.argsort(active_weights)[::-1][:max_active]
        active = active[keep]

    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        w_k = w[k]
        if w_k <= 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_k * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def run_phase1(a_init, n, S, max_time, rng, T_init, T_min):
    """Phase 1: Adam with log-sum-exp smoothed objective."""
    a = safe_normalize(a_init, n)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.4 * S / n

    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    stuck_count = 0
    T = T_init

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        # Temperature annealing
        T = max(T_min, T_init * (0.9997 ** t))

        # If stuck, add perturbation
        if stuck_count > 3000:
            noise = rng.normal(0.0, 0.02 * S / n, n)
            a = project_simplex(a + noise, S)
            stuck_count = 0
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0

        # Compute gradient (switch to top-k as T gets small)
        if T < T_min * 1.5:
            top_indices = np.argsort(c)[::-1][:5]
            grad = compute_subgradient(a, c, top_indices)
        else:
            grad = compute_smooth_subgradient(a, c, T, max_active=100)

        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-12:
            break
        grad = grad / gnorm

        # Adam update with cycling LR
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** max(t, 1))
        v_hat = v / (1.0 - beta2 ** max(t, 1))

        lr = base_lr * min(1.0, 20.0 / max(t, 1)) * (0.98 ** (t % 200))

        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

    return best_local, best_score


def run_phase2(a_init, n, S, max_time, rng, top_k=4, lr_scale=0.25):
    """Phase 2: Adam with focused top-k subgradient."""
    a = safe_normalize(a_init, n)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    base_lr = lr_scale * S / n

    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    stuck_count = 0

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        if stuck_count > 5000:
            noise = rng.normal(0.0, 0.005 * S / n, n)
            a = project_simplex(a + noise, S)
            stuck_count = 0

        top_indices = np.argsort(c)[::-1][:top_k]
        grad = compute_subgradient(a, c, top_indices)
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-12:
            break
        grad = grad / gnorm

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** max(t, 1))
        v_hat = v / (1.0 - beta2 ** max(t, 1))

        lr = base_lr * min(1.0, 30.0 / max(t, 1))
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

    return best_local, best_score


def coordinate_polish(a_init, n, S, max_time, rng, top_k=3):
    """Coordinate descent: try small moves on elements contributing to worst peaks."""
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()

    # Build a list of "candidate moves" near the worst peaks
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        top_idx = np.argsort(c)[::-1][:top_k]

        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi < lo:
                continue

            # Find the largest contributor in this range
            seg = a[lo:hi + 1]
            max_local = int(np.argmax(seg))
            src = lo + max_local

            # Try several delta values
            for delta_frac in (0.002, 0.001, 0.0005):
                delta = delta_frac * S / n
                # Try moving mass from src to several targets
                targets = [int(rng.integers(0, n)) for _ in range(5)]
                # Also try symmetric target
                targets.append(n - 1 - src)

                for tgt in targets:
                    if tgt == src:
                        continue
                    a_new = a.copy()
                    a_new[src] -= delta
                    a_new[tgt] += delta
                    if a_new[src] < 0:
                        continue
                    s = a_new.sum()
                    a_new = a_new * (S / s)
                    c_new = fft_convolve_self(a_new)
                    if c_new.max() < cur:
                        a = a_new
                        cur = c_new.max()
                        break

    return best_local, best_score
