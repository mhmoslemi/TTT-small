"""
Hybrid optimization for minimizing peak auto-convolution of non-negative sequences.
Uses subgradient descent (key insight: ∂max(conv(a,a))/∂a[m] = 2*a[k*-m] at argmax k*),
LP-based refinement with multi-peak constraints, spectral flattening, and local search.
Explores multiple sequence lengths and initializations.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
from scipy.ndimage import gaussian_filter1d
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, 0.01 / len(a))
    return a / s


def init_constant(n):
    return np.ones(n, dtype=np.float64) / n


def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))
    return normalize(a)


def init_asymmetric_step(n, delta=0.15):
    a = np.ones(n, dtype=np.float64)
    mid = n // 2
    a[:mid] += delta
    a[mid:] -= delta
    a = np.maximum(0, a)
    return normalize(a)


def init_smooth_step(n, delta=0.15, width_frac=0.1):
    t = np.arange(n, dtype=np.float64)
    mid = n / 2.0
    width = width_frac * n
    a = 1.0 + delta * np.tanh((t - mid) / width)
    a = np.maximum(0, a)
    return normalize(a)


def init_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    return normalize(a)


def init_random_smooth(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(2, n // 10), max(3, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return init_constant(n)
    return a / s


def subgradient_descent(a, time_limit=60.0, initial_eta=0.03, momentum=0.9, verbose=False):
    """Subgradient descent on max(conv(a,a)) with momentum and adaptive step size."""
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    v = np.zeros(n)
    stagnant = 0

    it = 0
    while time.monotonic() - start_time < time_limit:
        b = np.convolve(a, a)
        k_star = int(np.argmax(b))

        m_lo = max(0, k_star - n + 1)
        m_hi = min(n - 1, k_star)
        valid_m = np.arange(m_lo, m_hi + 1)
        grad = np.zeros(n)
        grad[valid_m] = 2.0 * a[k_star - valid_m]

        # Multi-peak: also consider second and third peaks if close to max
        sorted_idx = np.argsort(b)[::-1]
        b_max = b[sorted_idx[0]]
        for idx in sorted_idx[1:4]:
            if b[idx] > 0.9 * b_max:
                k2 = int(idx)
                m_lo2 = max(0, k2 - n + 1)
                m_hi2 = min(n - 1, k2)
                valid_m2 = np.arange(m_lo2, m_hi2 + 1)
                grad[valid_m2] += 2.0 * a[k2 - valid_m2]

        v = momentum * v + (1.0 - momentum) * grad
        eta = initial_eta / (1.0 + 0.01 * it)

        a_new = a - eta * v
        a_new = np.maximum(0.0, a_new)
        a_new = normalize(a_new)

        new_score = score_sequence(a_new)
        if new_score < best_score + 1e-12:
            a = a_new
            if new_score < best_score - 1e-10:
                best_score = new_score
                best_a = a.copy()
                stagnant = 0
                if verbose and it % 100 == 0:
                    print(f"      SG iter {it}: score={best_score:.6f}")
            else:
                stagnant += 1
        else:
            # Try smaller step
            a_new2 = a - 0.1 * eta * v
            a_new2 = np.maximum(0.0, a_new2)
            a_new2 = normalize(a_new2)
            if score_sequence(a_new2) < best_score:
                a = a_new2
            stagnant += 1

        if stagnant > 200:
            break
        it += 1

    return best_a, best_score


def build_conv_matrix(a):
    """Build sparse matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    n_rows = 2 * n - 1
    row_idx = []
    col_idx = []
    data = []
    for k in range(n_rows):
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            cols = np.arange(m_lo, m_hi + 1)
            vals = a[k - cols]
            mask = vals > 0
            row_idx.extend([k] * int(mask.sum()))
            col_idx.extend(cols[mask].tolist())
            data.extend(vals[mask].tolist())
    A = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))
    return A


def lp_refine(a, time_limit=120.0, top_k=3, verbose=False):
    """LP-based refinement: find g maximizing sum(g) s.t. conv(a,g) <= M, g >= 0."""
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")

    stagnant = 0
    it = 0
    while time.monotonic() - start_time < time_limit:
        remaining = time_limit - (time.monotonic() - start_time)
        if remaining < 5.0:
            break

        b = np.convolve(a, a)
        score = 2.0 * n * b.max() / (a.sum() ** 2)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 10 == 0 or it < 3):
                print(f"    LP iter {it}: score={score:.6f}")
        else:
            stagnant += 1

        if stagnant > 30:
            if verbose:
                print(f"    LP stagnant at iter {it}")
            break

        # Find top-K peaks for constraints
        sorted_idx = np.argsort(b)[::-1]
        top_positions = sorted_idx[:top_k]
        M = b[top_positions].max()

        # Build constraint matrix A where A @ g = conv(a, g)
        A = build_conv_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        timeout = max(3.0, min(30.0, remaining * 0.4))
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': timeout})
            if res.success and res.x is not None:
                g = np.maximum(0.0, res.x)
                sum_g = g.sum()
                if sum_g <= a.sum() * (1.0 + 1e-10):
                    break
                g = g / sum_g
                # Blend
                alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.2 * it)))
                a_new = (1.0 - alpha) * a + alpha * g
                a_new = normalize(a_new)
                new_score = score_sequence(a_new)
                if new_score <= score + 1e-12:
                    a = a_new
                else:
                    alpha2 = alpha * 0.2
                    a_new = (1.0 - alpha2) * a + alpha2 * g
                    a_new = normalize(a_new)
                    a = a_new
        except Exception:
            break
        it += 1

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it} iters")
    return best_a, best_score


def spectral_flatten(a, time_limit=30.0, verbose=False):
    """Try to flatten the power spectrum to reduce convolution peaks."""
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    Spec start: score={best_score:.6f}")

    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        if time.monotonic() - start_time > time_limit:
            break
        A = np.fft.rfft(a, n=4 * n)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        P_new = P * (P_mean / (P + 1e-10)) ** alpha
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n=4 * n)[:n]
        a_new = np.maximum(0.0, a_new)
        a_new = normalize(a_new)
        s = score_sequence(a_new)
        if s < best_score:
            best_score = s
            best_a = a_new.copy()
            if verbose:
                print(f"    Spec alpha={alpha}: score={best_score:.6f}")
            a = a_new.copy()

    return best_a, best_score


def local_search(a, time_limit=60.0, rng=None, verbose=False):
    """Random perturbation local search with adaptive width."""
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LS start: score={best_score:.6f}")

    base_width = max(2, n // 100)
    max_width = max(3, n // 5)
    accept_count = 0
    base_scale = 0.1

    while time.monotonic() - start_time < time_limit:
        width = int(rng.integers(base_width, max_width + 1))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        scale = base_scale / (1.0 + accept_count * 0.01)
        candidate[start:stop] *= np.exp(rng.normal(0.0, scale, size=stop - start))
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = normalize(candidate)
        score = score_sequence(candidate)
        if score < best_score - 1e-12:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 100 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def smooth_sequence(a, sigma_frac=0.01):
    """Apply mild Gaussian smoothing."""
    n = len(a)
    sigma = max(1.0, sigma_frac * n)
    a_smooth = gaussian_filter1d(a, sigma=sigma, mode='nearest')
    return normalize(a_smooth)


def refine_sequence(a, time_budget=180.0, verbose=False):
    """Full refinement pipeline: subgradient -> LP -> spectral -> local search."""
    start_time = time.monotonic()
    a = normalize(a.astype(np.float64))
    best_a = a.copy()
    best_score = score_sequence(a)

    # Phase 1: Subgradient descent
    sg_time = min(40.0, time_budget * 0.2)
    if time_budget > 20:
        a, s = subgradient_descent(a, time_limit=sg_time, verbose=verbose)
        if s < best_score:
            best_score = s
            best_a = a.copy()

    # Phase 2: LP refinement
    lp_time = min(80.0, time_budget * 0.35)
    remaining = time_budget - (time.monotonic() - start_time)
    if remaining > 15:
        lp_t = min(lp_time, remaining * 0.6)
        a, s = lp_refine(a, time_limit=lp_t, verbose=verbose)
        if s < best_score:
            best_score = s
            best_a = a.copy()

    # Phase 3: Spectral flattening
    remaining = time_budget - (time.monotonic() - start_time)
    if remaining > 10:
        spec_t = min(20.0, remaining * 0.3)
        a, s = spectral_flatten(a, time_limit=spec_t, verbose=verbose)
        if s < best_score:
            best_score = s
            best_a = a.copy()

    # Phase 4: More subgradient
    remaining = time_budget - (time.monotonic() - start_time)
    if remaining > 15:
        sg_t = min(40.0, remaining * 0.4)
        a, s = subgradient_descent(a, time_limit=sg_t, verbose=verbose)
        if s < best_score:
            best_score = s
            best_a = a.copy()

    # Phase 5: Local search polish
    remaining = time_budget - (time.monotonic() - start_time)
    if remaining > 10:
        ls_t = min(30.0, remaining * 0.8)
        a, s = local_search(best_a, time_limit=ls_t, verbose=verbose)
        if s < best_score:
            best_score = s
            best_a = a.copy()

    return best_a, best_score
