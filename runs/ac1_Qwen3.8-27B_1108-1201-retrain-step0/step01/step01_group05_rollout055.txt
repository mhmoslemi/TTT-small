from scipy.fft import fft, ifft
from scipy.optimize import linprog

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def solve_lp_direction(f, M, active_lags, n):
    """Solve the LP: max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0"""
    K = len(active_lags)
    # Build constraint matrix C (K x n)
    # C[k_idx, j] = f[k - j] if 0 <= k - j < n else 0
    C = np.zeros((K, n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            C[idx, lo:hi+1] = f[k - hi:k - lo + 1][::-1]

    row = np.zeros(n)

        row[lo:hi+1] = f[k-hi:k-lo+1][::-1]

def solve_direction_lp(f, M, active_lags, n):
    """Solve LP: max sum(g) s.t. (f*g)[active[k]] <= M, g >= 0"""
    K = len(active_lags)
    # Build constraint matrix
    C = np.zeros((K, n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)

    if res.success:
        return res.x
    return None

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using LP-based direction
finding combined with FISTA polishing.

Key innovations over previous approach:
1. LP-based direction finding (solves max sum(g) s.t. ||f*g||_inf <= M)
   to find the optimal feasible direction at each step
2. Adaptive active set management (grow/shrink the set of binding constraints)
3. Line search for optimal step size
4. FISTA polishing with adaptive step size for fine-tuning
5. Multiple lengths and initializations for robustness
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, n):
    """Clamp to [0, 1000] and normalize to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    return a * (S / s)


def build_lp_matrix(f, active_lags, n):
    """Build the LP constraint matrix for the given active lags."""
    K = len(active_lags)
    C = np.zeros((K, n), dtype=np.float64)
    for idx, k in enumerate(active_lags):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            C[idx, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return C


def solve_direction_lp(f, M, active_lags, n):
    """
    Solve LP: max sum(g) subject to (f*g)[k] <= M for k in active_lags, g >= 0.
    Returns the optimal g vector, or None if the LP fails.
    """
    K = len(active_lags)
    if K == 0:
        return None
    C = build_lp_matrix(f, active_lags, n)
    # Check if any row is all zeros (degenerate constraint)
    nonzero_rows = np.any(C != 0, axis=1)
    if not np.all(nonzero_rows):
        C = C[nonzero_rows]
        K = len(C)
        if K == 0:
            return None
    
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def line_search_convex(f, direction, n, cur_max):
    """
    Find step size t that minimizes max_conv(f + t*direction).
    Returns (best_t, best_val).
    """
    best_t = 0.0
    best_val = cur_max
    
    # Coarse search
    candidates = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
    for t in candidates:
        a = f + t * direction
        a = np.maximum(a, 0.0)
        c = fft_convolve_self(a)
        val = c.max()
        if val < best_val:
            best_val = val
            best_t = t
    
    # Fine search around best
    if best_t > 0:
        lo = max(0.0, best_t * 0.5)
        hi = best_t * 1.5 + 0.05
        for t in np.linspace(lo, hi, 15):
            a = f + t * direction
            a = np.maximum(a, 0.0)
            c = fft_convolve_self(a)
            val = c.max()
            if val < best_val:
                best_val = val
                best_t = t
    
    return best_t, best_val


def compute_subgradient(a, c, top_indices):
    """Compute weighted subgradient of max(c) using top-k active lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-15:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad

    S = np.sqrt(2.0 * n)
    a = normalize_seq(f_init, n)
    c = fft_convolve_self(a)
    best_a = a.copy()
    best_val = c.max()
    
    # Initialize FISTA parameters
    y = a.copy()
    t_k = 1.0

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using LP-based direction
finding with adaptive active sets, combined with FISTA and mirror descent polishing.

Key ideas:
1. LP-based direction: solve max sum(g) s.t. ||f*g||_inf <= M, g>=0 to find optimal
   feasible direction; use line search for step size.
2. Adaptive active set: start with few top lags, grow over iterations.
3. Dual multiplicative weights as fast fallback when LP is too slow.
4. FISTA and mirror descent for fine-tuning after LP phase.
5. Multiple lengths and initializations for robustness.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, n):
    """Clamp to [0, 1000] and normalize to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s < 1e-15:
        return np.full(n, S / n, dtype=np.float64)
    return a * (S / s)


def build_lp_row(f, k, n):
    """Build one row of LP constraint matrix for lag k."""
    row = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        row[lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return row


def solve_direction_lp(f, M, active_lags, n, timeout=15.0):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0.
    Returns optimal g or None on failure.
    """
    active_lags = list(active_lags)
    K = len(active_lags)
    if K == 0:
        return None

    # Build constraint matrix (only non-zero rows)
    C_rows = []
    for k in active_lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            row = np.zeros(n, dtype=np.float64)
            row[lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
            C_rows.append(row)

    if not C_rows:
        return None

    C = np.vstack(C_rows)
    K = len(C)

    c_obj = -np.ones(n)
    b_ub = np.full(K, M, dtype=np.float64)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def dual_mp_direction(f, M, active_lags, n, iterations=200):
    """
    Dual multiplicative weights to approximate LP solution.
    Dual: min M*sum(y) s.t. sum_k y_k * f[k-j] >= 1 for all j, y >= 0.
    Returns a primal-like direction g derived from the dual.
    """
    K = len(active_lags)
    if K == 0:
        return None
    active_lags = [int(k) for k in active_lags]

    # Build the "convolution matrix" rows for active lags
    # Row k: (C_k)[j] = f[k-j] if valid
    # We need: for each j, sum_k y_k * f[k-j] >= 1
    # Equivalently: A @ y >= 1 where A[j,k] = f[k-j] if valid else 0

    # To save memory, compute A @ y on the fly
    y = np.ones(K, dtype=np.float64) / K
    eta = 1.0

    for it in range(iterations):
        # Compute violation at each j: v[j] = 1 - sum_k y_k * f[k-j]
        # Maximize violation to find the tightest constraint
        # v[j] = 1 - sum_k y_k * f[k-j]
        # We want j that maximizes v[j] (most violated)
        # sum_k y_k * f[k-j] = sum over active lag k of y_k * f[k-j]

        # Efficient computation: for each active lag k, f[k-j] is a shifted version of f
        # conv[j] = sum_k y_k * f[k-j]
        conv_arr = np.zeros(n, dtype=np.float64)
        for ki, k in enumerate(active_lags):
            if y[ki] < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                # f[k-j] for j in [lo, hi] = f[k-hi : k-lo+1][::-1]
                conv_arr[lo:hi + 1] += y[ki] * f[k - hi:k - lo + 1][::-1]

        violation = 1.0 - conv_arr
        j_star = int(np.argmax(violation))

        # Update y: y_k *= exp(eta * f[k - j_star])
        update = np.zeros(K, dtype=np.float64)
        for ki, k in enumerate(active_lags):
            idx = k - j_star
            if 0 <= idx < n:
                update[ki] = f[idx]
        y = y * np.exp(eta * update)
        y_sum = y.sum()
        if y_sum < 1e-30:
            y = np.ones(K) / K
        else:
            y = y / y_sum
        eta *= 0.999

    # From dual solution, derive primal direction via complementary slackness
    # g[j] > 0 only where conv_arr[j] = 1 (tight constraint)
    # Approximate: g[j] proportional to how "tight" constraint j is
    tightness = np.maximum(0.0, 1.0 - conv_arr + 0.01)
    g = tightness ** 2
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs * S_global  # will be rescaled by caller

S_global = 1.0


def line_search_convex(f, direction, n, cur_max):
    """Find step t minimizing max_conv(f + t*direction). Returns (best_t, best_val)."""
    best_t = 0.0
    best_val = cur_max

    candidates = [0.01, 0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0]
    for t in candidates:
        a = f + t * direction
        if np.any(a < -1e-10):
            a = np.maximum(a, 0.0)
        else:
            a = np.maximum(a, 0.0)
        c = fft_convolve_self(a)
        val = c.max()
        if val < best_val:
            best_val = val
            best_t = t

    # Fine search
    if best_t > 0:
        lo = max(0.0, best_t * 0.7)
        hi = best_t * 1.3 + 0.02
        for t in np.linspace(lo, hi, 12):
            a = np.maximum(f + t * direction, 0.0)
            c = fft_convolve_self(a)
            val = c.max()
            if val < best_val:
                best_val = val
                best_t = t

    return best_t, best_val


def compute_subgradient(a, c, top_indices):
    """Weighted subgradient of max(c) using top-k active lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-15:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad

    Each iteration: select active lags, solve LP or dual MP, line search, update.
    """
    global S_global
    S = np.sqrt(2.0 * float(n))
    S_global = S
    a = normalize_seq(f_init, n)
    c = fft_convolve_self(a)
    best_a = a.copy()
    best_val = c.max()
    start = time.monotonic()
    iteration = 0
    stale_rounds = 0

    if verbose:
        print(f"    Start: max_c = {best_val:.8f}")

    while time.monotonic() - start < budget:

        rem = budget - (time.monotonic() - start)
        if rem < 2.0:
            break

            best_val = cur_max

        # Select active lags

        first_half = c[:n]
        active = np.argsort(first_half)[::-1][:n_active]

        else:

                direction = -compute_subgradient(a, c, top10)

def run_fista(f_init, n, budget, rng, top_k=20, verbose=True):
    """FISTA for fine-tuning the sequence."""
    S = np.sqrt(2.0 * float(n))
    a = normalize_seq(f_init, n)
    y = a.copy()
    t_k = 1.0
    # Lipschitz estimate: the gradient of max_k (a*a)[k] has norm bounded by
    # 2 * max_k sum_i |a[i]| * |a[k-i]| <= 2 * (sum |a|)^2 = 2*S^2 = 4n
    # But in practice the subgradient is much smaller. Use adaptive step.
    step = 1.0 / (4.0 * n)  # initial conservative step
    c = fft_convolve_self(a)
    best_a = a.copy()
    best_val = c.max()
    start = time.monotonic()
    iteration = 0

    if verbose:
        print(f"    FISTA start: max_c = {best_val:.8f}")

    while time.monotonic() - start < budget:
        iteration += 1

        c_y = fft_convolve_self(y)
        top_idx = np.argsort(c_y[:n])[::-1][:top_k]
        grad = compute_subgradient(y, c_y, top_idx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        a_new = project_simplex(y - step * grad, S)

        # Simple backtracking: check if objective decreased
        if iteration % 100 == 0:
            c_a = fft_convolve_self(a_new)
            if c_a.max() < best_val:
                best_val = c_a.max()
                best_a = a_new.copy()
                if verbose:
                    print(f"    FISTA iter {iteration}: max_c = {best_val:.8f}, t={time.monotonic()-start:.1f}s")

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        t_k = t_new
        a = a_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best_val:
        best_val = c_final.max()
        best_a = a.copy()

    return best_a, best_val


def run_mirror(f_init, n, budget, rng, eta_init=0.2, top_k=15, verbose=True):
    """Mirror descent for fine-tuning."""
    S = np.sqrt(2.0 * float(n))
    a = normalize_seq(f_init, n)
    c = fft_convolve_self(a)
    best_a = a.copy()
    best_val = c.max()
    eta = eta_init
    start = time.monotonic()
    iteration = 0

    if verbose:
        print(f"    Mirror start: max_c = {best_val:.8f}")

    while time.monotonic() - start < budget:
        iteration += 1
        cur = c.max()
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            if verbose and iteration % 500 == 0:
                print(f"    Mirror iter {iteration}: max_c = {best_val:.8f}, t={time.monotonic()-start:.1f}s")

        top_idx = np.argsort(c[:n])[::-1][:top_k]
        grad = compute_subgradient(a, c, top_idx)
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad_n = grad / gmax

        a_new = a * np.exp(-eta * grad_n)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.full(n, S / n)
        else:
            a = a_new * S / s
        c = fft_convolve_self(a)
        eta = eta_init / (1.0 + 0.005 * iteration)

    return best_a, best_val


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function. Returns a list of non-negative floats minimizing
    2*n*max_conv/sum^2.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 15.0  # safety margin

    best_overall = np.inf
    best_seq = None

    # Collect initializations
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).flatten()
        if given.size >= 1024 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append((given.size, np.clip(given, 0, 1000), "given"))
    except Exception:
        pass

    # Add diverse initializations
    for n_try in [4096, 8192]:
        inits.append((n_try, rng.random(n_try) + 0.1, f"rand_{n_try}"))

    # Half-sine
    n_h = 8192
    t_h = np.linspace(0, np.pi, n_h)
    inits.append((n_h, np.sin(t_h) ** 0.5, "hsin"))

    # Cosine bell
    t_c = np.linspace(-1, 1, 8192)
    inits.append((8192, np.cos(np.pi * t_c / 2) ** 2, "cbell"))

    # Smoothed random (Gaussian kernel)
    n_s = 8192
    base = rng.random(n_s)
    # Simple boxcar smoothing
    k = 16
    kernel = np.ones(k) / k
    smoothed = np.convolve(base, kernel, mode='same')
    inits.append((n_s, smoothed, "smooth_rand"))

    print(f"=== Phase 1: Exploration ({len(inits)} inits) ===")

    # Budget allocation: 25% explore, 75% polish
    # Give more time to the "given" init since it's already good
    explore_total = 0.25 * total_budget
    # Weight: given gets 40% of explore, others split the rest
    given_idx = 0
    for i, (n_x, arr_x, name_x) in enumerate(inits):
        if name_x == "given":
            given_idx = i
            break

    per_init_base = explore_total / len(inits)
    per_init = np.full(len(inits), per_init_base)
    total_w = per_init.sum()
    # Give the given one more time
    per_init[given_idx] = 0.4 * explore_total
    others = (explore_total - per_init[given_idx]) / (len(inits) - 1)
    for i in range(len(inits)):
        if i != given_idx:
            per_init[i] = others

    for i, (n_x, arr_x, name_x) in enumerate(inits):
        if time.monotonic() > deadline - 0.75 * total_budget - 5:
            break
        budget_i = per_init[i]
        print(f"  [{name_x}] n={n_x}, budget={budget_i:.1f}s")
        t0 = time.monotonic()

        # LP-based optimization
        a_best, val = run_lp_optimization(
            arr_x, n_x, budget_i * 0.75, rng,
            top_k_init=50, top_k_max=200, verbose=(name_x == "given")
        )

        # Quick FISTA polish on the result
        rem_i = budget_i - (time.monotonic() - t0)
        if rem_i > 5:
            a_f, val_f = run_fista(a_best, n_x, rem_i, rng, top_k=15, verbose=False)
            if val_f < val:
                a_best, val = a_f, val_f

        # Verify
        verify = evaluate_sequence(a_best.tolist())
        print(f"  [{name_x}] local={val:.6f}, verify={verify:.6f}")

        if verify < best_overall:
            best_overall = verify
            best_seq = a_best.copy()
            print(f"  *** New best: {best_overall:.6f} ***")

    # --- Phase 2: Polish the best sequence ---
    print(f"\n=== Phase 2: Polish (budget={total_budget * 0.75:.0f}s) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        polish_budget = deadline - time.monotonic() - 5.0
        if polish_budget < 10:
            polish_budget = 10.0

        # Split: 55% LP, 25% FISTA, 20% mirror
        t_lp = 0.55 * polish_budget
        t_fista = 0.25 * polish_budget
        t_mirror = 0.20 * polish_budget

        print(f"  LP polish ({t_lp:.0f}s), n={n_best}...")
        a1, v1 = run_lp_optimization(
            best_seq, n_best, t_lp, rng,
            top_k_init=100, top_k_max=500, verbose=True
        )
        print(f"  LP polish result: {v1:.8f}")
        if v1 < best_overall:
            best_overall = v1
            best_seq = a1

        remain = polish_budget - t_lp - 3
        if remain > 5:
            t_f = min(t_fista, remain)
            print(f"  FISTA polish ({t_f:.0f}s)...")
            a2, v2 = run_fista(best_seq, n_best, t_f, rng, top_k=25, verbose=True)
            print(f"  FISTA result: {v2:.8f}")
            if v2 < best_overall:
                best_overall = v2
                best_seq = a2

        remain = polish_budget - t_lp - t_fista - 3
        if remain > 5:
            t_m = min(t_mirror, remain)
            print(f"  Mirror polish ({t_m:.0f}s)...")
            a3, v3 = run_mirror(best_seq, n_best, t_m, rng, eta_init=0.05, top_k=10, verbose=True)
            print(f"  Mirror result: {v3:.8f}")
            if v3 < best_overall:
                best_overall = v3
                best_seq = a3
    else:
        best_seq = np.ones(256, dtype=np.float64)

    # Final assembly
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones(256, dtype=np.float64)

    final_val = evaluate_sequence(best_seq.tolist())
    print(f"\n=== FINAL: {final_val:.6f} (target: 1.5027) ===")

    # If final verify is somehow worse (shouldn't happen), use best_overall seq
    if not np.isfinite(final_val):
        best_seq = np.ones(256, dtype=np.float64)
        final_val = evaluate_sequence(best_seq.tolist())

    return [float(x) for x in best_seq]
