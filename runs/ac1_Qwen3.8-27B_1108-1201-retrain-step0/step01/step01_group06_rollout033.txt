import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)

def convolve_ab(a, b):
    """Compute conv(a, b) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]

def build_constraint_matrix(a):
    """Build A matrix where A[k, m] = a[k-m] if valid, else 0."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):

            j = k - m
            if 0 <= j < n:
                A[k, m] = a[j]
    return A

def lp_step(a, M):

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
combined with soft-max mirror descent polishing. The LP step solves:
  max sum(b) s.t. conv(a,b) <= M, b >= 0
which is guaranteed to produce a non-increasing objective. Multiple
sequence lengths and initializations are explored.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A[k, m] = a[j]
    return A

    A = build_conv_matrix(a)

        return np.maximum(res.x, 0)
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
with restarts and multiple sequence lengths. The core iteration solves:
  max sum(b) s.t. conv(a,b) <= M, b >= 0
which guarantees non-increasing objective. Combined with soft-max mirror
descent polishing and diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve_auto(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a,b)[k] for linear conv."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = a[k - j]
    return A

def lp_step(a, M, max_iter=None):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    opts = {'maxiter': max_iter or 10000}
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=opts)
    
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None

def lp_optimize(a0, time_limit, mix_rate=1.0, restart_every=200, verbose=False):
    """Iterative LP refinement with periodic restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve_auto(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        b = lp_step(a, M)
        if b is None:
            break
        
        b_sum = b.sum()
        if b_sum < 1e-12:
            break
        
        b = b / b_sum
        
        # Compute new objective
        conv_b = fast_convolve_auto(b)
        obj_b = 2.0 * n * conv_b.max()
        
        if obj_b < best_obj - 1e-10:
            best_obj = obj_b
            best_a = b.copy()
            stall = 0
        else:
            stall += 1
        
        # Update a
        if mix_rate >= 1.0:
            a = b
        else:
            a = (1 - mix_rate) * a + mix_rate * b
            a = np.maximum(a, 0)
            a_sum = a.sum()
            if a_sum < 1e-12:
                a = best_a.copy()
            else:
                a = a / a_sum
        
        conv = fast_convolve_auto(a)
        
        if verbose and (stall % 100 == 0):
            print(f"    LP iter: obj={2*n*conv.max():.6f}, best={best_obj:.6f}, stall={stall}")
        
        # Restart if stalled
        if stall >= max_stall:
            # Perturb and restart
            a = best_a.copy()
            noise = np.random.default_rng(int(time.monotonic()*1000)).uniform(0, 0.1, n)
            a = a + noise
            a = np.maximum(a, 0)

            stall = 0

def mirror_descent_polish(a0, time_limit, t_param=50.0, eta0=0.3):
    """Soft-max mirror descent polishing step."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve_auto(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    eta = eta0
    patience = 0
    max_patience = 60
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Soft-max weights
        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum
        
        # Subgradient
        sg = np.zeros(n)
        thresh = w.max() * 0.001
        active = np.where(w > thresh)[0]
        
        for k_int in active:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end+1] += w[k] * a[k-j_end:k-j_start+1][::-1]
        
        # Multiplicative update
        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = best_a.copy()
        else:
            a = a / a_sum
        
        conv = fast_convolve_auto(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            patience = 0
            eta *= 0.5
            t_param *= 1.5
            if eta < 0.003:

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian (various widths)
    for sf in [0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - n*0.5) / (sf * n))**2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g
    
    # Exponential (geometric-like)
    for r in [0.95, 0.98, 0.99]:
        e = np.array([r**j for j in range(n)])
        e /= e.sum()
        inits[f'exp_{r}'] = e
    
    # Bimodal
    w = max(1, n // 6)
    bi = np.zeros(n)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Ramp
    ramp = np.arange(1, n+1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Parent-derived
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0).astype(np.float64)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
            if pp.sum() > 1e-10:
                inits['parent_interp'] = pp / pp.sum()
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
combined with soft-max mirror descent polishing. The core LP iteration solves:
  max sum(b) s.t. conv(a,b) <= M, b >= 0
which guarantees non-increasing objective. Multiple sequence lengths (64-1024)
and diverse initializations are explored with restart mechanisms to escape
local minima.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve_auto(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a,b)[k] for linear convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def lp_step(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'maxiter': 5000})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(a0, time_limit, mix_rate=1.0, verbose=False):
    """Iterative LP refinement with periodic restarts on stall."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve_auto(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    stall = 0
    max_stall = 300
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        b = lp_step(a, M)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-12:
            break

        b = b / b_sum

        conv_b = fast_convolve_auto(b)
        obj_b = 2.0 * n * conv_b.max()

        if obj_b < best_obj - 1e-10:
            best_obj = obj_b
            best_a = b.copy()
            stall = 0
        else:
            stall += 1

        if mix_rate >= 1.0:
            a = b
        else:
            a = (1.0 - mix_rate) * a + mix_rate * b
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum < 1e-12:
                a = best_a.copy()
            else:
                a = a / a_sum

        conv = fast_convolve_auto(a)

            else:

            stall = 0

def mirror_descent_polish(a0, time_limit, t_param=50.0, eta0=0.3):
    """Soft-max mirror descent polishing step on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve_auto(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    eta = eta0
    patience = 0
    max_patience = 50
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum

        sg = np.zeros(n)
        thresh = w.max() * 0.001
        active = np.where(w > thresh)[0]

        for k_int in active:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]

        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = best_a.copy()
        else:
            a = a / a_sum

        conv = fast_convolve_auto(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            patience = 0
            eta *= 0.5
            t_param *= 1.5
            if eta < 0.003:
                break

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    for r in [0.90, 0.95, 0.98, 0.995]:
        e = np.array([r ** j for j in range(n)])
        e /= e.sum()
        inits[f'geom_{r}'] = e

    w = max(1, n // 5)
    bi = np.zeros(n)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    cos_b = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    cos_b /= cos_b.sum()
    inits['cosine'] = cos_b

    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0).astype(np.float64)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size),
                           np.maximum(parent, 0))
            if pp.sum() > 1e-10:
                inits['parent_interp'] = pp / pp.sum()

    return inits

            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Final LP: score={score:.6f}")

        # Mirror descent with various parameters
        if time.monotonic() < deadline - 5:
            t_md = (deadline - 5 - time.monotonic()) * 0.5
            for t_p in [80.0, 150.0]:
                if t_md < 5:
                    break
                a_md, _ = mirror_descent_polish(best_sequence, time_limit=t_md,
                                                t_param=t_p, eta0=0.2)
                score_md = float(evaluate_sequence(a_md.tolist()))
                if np.isfinite(score_md) and score_md < best_score:
                    best_score = score_md
                    best_sequence = a_md.copy()
                    print(f"  Final MD (t={t_p}): score={score_md:.6f}")
                t_md = (deadline - 5 - time.monotonic()) * 0.5

        # One more LP pass on the polished result
        if time.monotonic() < deadline - 3:
            t_final = deadline - 3 - time.monotonic()
            if t_final > 5:
                a_f, _ = lp_optimize(best_sequence, time_limit=t_final, mix_rate=1.0)
                score_f = float(evaluate_sequence(a_f.tolist()))
                if np.isfinite(score_f) and score_f < best_score:
                    best_score = score_f
                    best_sequence = a_f.copy()
                    print(f"  Final LP2: score={score_f:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Ensure all values are valid (non-negative, finite, within bounds)
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)

    elapsed = time.monotonic() - t_start
    print(f"FINAL: score={best_score:.6f}, n={len(best_sequence)}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
