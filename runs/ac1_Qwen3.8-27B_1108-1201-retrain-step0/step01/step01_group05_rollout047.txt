        weights = top_vals / total
        # Use FFT: grad[i] = 2 * sum_k weights[k] * a[k-i]
        # = 2 * cross_correlation(a, weights)[i]
        N = 1 << (n + 2 * n - 2).bit_length()  # >= 3n-2
        A_f = fft(a, N)
        W_f = fft(weights, N)
        corr = np.real(ifft(A_f * np.conj(W_f)))
        return 2.0 * corr[:n]
    
    # p-norm gradient
    c_clipped = np.maximum(c, 1e-30)
    cp = c_clipped ** (p - 1)
    norm_val = (np.sum(c_clipped ** p)) ** (1.0 / p)
    if norm_val <= 1e-30:
        return np.zeros(n)
    
    weights = cp / norm_val  # these sum to > 1 but that's OK for gradient direction
    
    # FFT-based cross-correlation
    N = 1 << (n + 2 * n - 2).bit_length()
    A_f = fft(a, N)
    W_f = fft(weights, N)
    corr = np.real(ifft(A_f * np.conj(W_f)))
    return 2.0 * corr[:n]


def run_optimize(a_init, n, S, max_time, rng, p_schedule, lr_init, patience=200):
    """
    Adam optimizer with p-norm objective and adaptive p.
    p_schedule: dict with 'start', 'end', 'max_steps' for p schedule.
    """
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    best_ever = best_score
    
    p_start = p_schedule.get('start', 4)
    p_end = p_schedule.get('end', 100)
    p_steps = p_schedule.get('max_steps', 500)
    
    stall_counter = 0
    stall_threshold = 1e-8
    start_time = time.monotonic()
    
    while time.monotonic() - start_time < max_time:
        t += 1
        
        # Adaptive p
        if t <= p_steps:
            p = p_start + (p_end - p_start) * (t / p_steps)
        else:
            p = p_end
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
            stall_counter = 0
        else:
            stall_counter += 1
            if stall_counter > 5 * patience:
                # Reset with perturbation
                a = a + rng.normal(0, 0.01 * S / n, n)
                a = project_simplex(a, S)
                c = fft_convolve_self(a)
                best_score = max(best_score, c.max())
                stall_counter = 0
        
        if stall_counter > patience:
            # Reduce learning rate
            lr_init *= 0.8
            stall_counter = 0
        
        lr = lr_init / (1.0 + 0.001 * t)
        
        # Compute gradient
        grad = compute_pnorm_gradient(a, c, p, n)
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-15:
            break
        grad = grad / gnorm
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)
    
    return best_local, best_score


def run_multiplicative(a_init, n, S, max_time, rng, p_schedule, eta_init):
    """Multiplicative weights / mirror descent with p-norm objective."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    
    p_start = p_schedule.get('start', 4)
    p_end = p_schedule.get('end', 100)
    p_steps = p_schedule.get('max_steps', 300)
    
    stall_counter = 0
    start_time = time.monotonic()
    t = 0
    
    while time.monotonic() - start_time < max_time:
        t += 1
        if t <= p_steps:
            p = p_start + (p_end - p_start) * (t / p_steps)
        else:
            p = p_end
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
            stall_counter = 0
        else:
            stall_counter += 1
        
        if stall_counter > 300:
            eta_init *= 0.7
            stall_counter = 0
        
        lr = eta_init / (1.0 + 0.002 * t)
        
        grad = compute_pnorm_gradient(a, c, p, n)
        gmax = np.abs(grad).max()
        if gmax <= 1e-15:
            break
        grad = grad / gmax
        
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new <= 1e-15:
            a = np.full(n, S / n)
        else:
            a = a_new * (S / s_new)
    
    return best_local, best_score

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Key improvements over previous approach:
- FFT-accelerated p-norm gradient (O(N log N) per step instead of O(n^2))
- Adaptive p-norm schedule (smooth -> sharp) for better convergence
- Log-space computation for numerical stability with large p
- Focused polishing of the best sequence with longer runtime
- Multiple optimization methods (Adam, multiplicative weights) run sequentially
- Periodic perturbation to escape local minima
- Try both n=8192 and n=16384
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    # Clip tiny negative values from FFT numerical error
    c = np.maximum(c, 0.0)
    return c[:size]


def compute_gradient_fft(a, c, p, n):
    """
    Compute gradient direction for p-norm of self-convolution.
    Uses FFT for O(N log N) computation.
    
    grad[i] = 2 * sum_k w[k] * a[k-i] where w[k] ∝ c[k]^(p-1)
    Computed as: 2 * (w * a_reversed)[i + n - 1]
    """
    c_safe = np.maximum(c, 1e-30)
    
    if p > 50:
        # Use log-space for numerical stability, effectively top-k
        log_c = np.log(c_safe)
        log_w = (p - 1) * log_c
        # Subtract max for stability
        log_w -= log_w.max()
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum <= 1e-30:
            # Fall back to argmax
            k_max = int(np.argmax(c))
            grad = np.zeros(n)
            lo = max(0, k_max - n + 1)
            hi = min(n - 1, k_max)
            grad[lo:hi + 1] = 2.0 * a[k_max - hi:k_max - lo + 1][::-1]
            return grad
        w = w / w_sum
    else:
        cp1 = c_safe ** (p - 1)
        w_sum = cp1.sum()
        if w_sum <= 1e-30:
            return np.zeros(n)
        w = cp1 / w_sum
    
    # Compute grad[i] = 2 * sum_k w[k] * a[k-i] via FFT
    # This equals 2 * (w * a_rev)[i + n - 1] where a_rev = a[::-1]
    a_rev = a[::-1]
    N = 1 << (3 * n - 2).bit_length()
    W = fft(w, N)
    B = fft(a_rev, N)
    conv_result = np.real(ifft(W * B))
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def run_adam_pnorm(a_init, n, S, max_time, rng, p_start, p_end, p_ramp_steps,
                   lr_init, patience=200, perturbation_rate=0.02):
    """Adam optimizer with adaptive p-norm objective."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    c = fft_convolve_self(a)
    best_score = float(c.max())
    best_local = a.copy()
    
    stall_counter = 0
    best_ever = best_score
    start_time = time.monotonic()
    
    while time.monotonic() - start_time < max_time:
        t += 1
        
        # Adaptive p
        if t <= p_ramp_steps:
            p = p_start + (p_end - p_start) * (t / float(p_ramp_steps))
        else:
            p = p_end
        
        c = fft_convolve_self(a)
        cur_max = float(c.max())
        
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
            stall_counter = 0
        else:
            stall_counter += 1
        
        if stall_counter > patience * 3:
            # Larger perturbation
            noise_scale = perturbation_rate * S / n
            a = a + rng.normal(0, noise_scale, n)
            a = project_simplex(a, S)
            stall_counter = 0
        elif stall_counter > patience:
            # Reduce learning rate
            lr_init = max(lr_init * 0.85, 1e-6 * S / n)
            stall_counter = 0
        
        lr = lr_init / (1.0 + 0.0005 * t)
        
        grad = compute_gradient_fft(a, c, p, n)
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-14:
            break
        grad = grad / gnorm
        
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** t)
        v_hat = v_buf / (1.0 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)
        
        if t % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print("  Adam t=%d p=%.1f maxc=%.8f best=%.8f lr=%.2e elapsed=%.1fs"
                  % (t, p, cur_max, best_score, lr, elapsed))
    
    return best_local, best_score


def run_multiplicative_pnorm(a_init, n, S, max_time, rng, p_start, p_end,
                             p_ramp_steps, eta_init, patience=200):
    """Multiplicative weights with adaptive p-norm objective."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    
    c = fft_convolve_self(a)
    best_score = float(c.max())
    best_local = a.copy()
    
    stall_counter = 0
    start_time = time.monotonic()
    t = 0
    effective_eta = eta_init
    
    while time.monotonic() - start_time < max_time:
        t += 1
        
        if t <= p_ramp_steps:
            p = p_start + (p_end - p_start) * (t / float(p_ramp_steps))
        else:
            p = p_end
        
        c = fft_convolve_self(a)
        cur_max = float(c.max())
        
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
            stall_counter = 0
        else:
            stall_counter += 1
        
        if stall_counter > patience * 2:
            effective_eta *= 0.8
            stall_counter = 0
        
        lr = effective_eta / (1.0 + 0.001 * t)
        
        grad = compute_gradient_fft(a, c, p, n)
        gmax = np.abs(grad).max()
        if gmax <= 1e-14:
            break
        grad = grad / gmax
        
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new <= 1e-15:
            a = np.full(n, S / n)
        else:
            a = a_new * (S / s_new)
        
        if t % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print("  Multi t=%d p=%.1f maxc=%.8f best=%.8f eta=%.2e elapsed=%.1fs"
                  % (t, p, cur_max, best_score, lr, elapsed))
    
    return best_local, best_score


def local_search(a_init, n, S, max_time, rng):
    """Local perturbation search: try small random changes, keep if better."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    
    c = fft_convolve_self(a)
    best_score = float(c.max())
    best_local = a.copy()
    
    start_time = time.monotonic()
    scale = 0.05 * S / n
    
    while time.monotonic() - start_time < max_time:
        # Random perturbation
        n_perturb = max(1, n // 100)
        indices = rng.choice(n, size=n_perturb, replace=False)
        a_trial = a.copy()
        a_trial[indices] += rng.normal(0, scale, n_perturb)
        a_trial = project_simplex(a_trial, S)
        
        c_trial = fft_convolve_self(a_trial)
        trial_score = float(c_trial.max())
        
        if trial_score < best_score:
            a = a_trial
            best_score = trial_score
            best_local = a.copy()
            # Sometimes increase scale on success
            scale *= 1.1
        else:
            # Decrease scale on failure
            scale = max(scale * 0.95, 0.001 * S / n)
        
        if time.monotonic() - start_time > max_time * 0.5 and \
           time.monotonic() - start_time < max_time * 0.5 + 0.1:
            print("  LocalSearch mid: best=%.8f scale=%.2e" % (best_score, scale))
    
    return best_local, best_score
