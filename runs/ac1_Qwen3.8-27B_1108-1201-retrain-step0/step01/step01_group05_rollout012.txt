Wait, I want to also think about whether there's a smarter mathematical approach. 

The problem is: minimize max_k c[k] where c = conv(a, a), subject to a >= 0, sum(a) = S.

This is equivalent to: minimize max_k c[k] where c_k = sum_j a[j]*a[k-j], a >= 0, sum(a) = S.

If we define the generating function A(x) = sum_j a[j] x^j, then c_k = [x^k] A(x)^2.

We want to minimize the maximum coefficient of A(x)^2.

This is a polynomial optimization problem. It's NP-hard in general (I think), so we need heuristics.

One approach from the literature: the "iterative amplitude adjustment" (IAA) algorithm, which is similar to the LP-based approach.

Another approach: use the fact that c = conv(a,a) is the "energy" sequence, and we want to make it as flat as possible. This is like a "spectral flattening" problem.

In the frequency domain: let F(omega) = FFT(a). Then the circular convolution (which is close to linear convolution for a well-behaved sequence) has FFT = |F(omega)|^2. So we want to make |F(omega)|^2 as flat as possible, which means making |F(omega)| as flat as possible.

But this is the circular convolution, not the linear one. For a sequence that's well-concentrated (no wrap-around), the linear and circular convolutions are the same. So if our sequence is Gaussian-like (concentrated in the middle), this approximation is good.

This suggests a "spectral" approach:
1. Compute F = FFT(a)
2. Update F to make |F| more uniform: F_new = F * (|F|_target / |F|)
3. a_new = IFFT(F_new)
4. Project a_new onto {a >= 0, sum(a) = S}

This is similar to the "Graham-Slative" algorithm or the "Gerchberg-Saxton" algorithm used in phase retrieval.

    """Iterative spectral flattening."""
    a = a.copy()
    size = 2*n - 1
    fsize = next_power_of_2(size * 2)  # avoid circular convolution issues

        # But we need to be careful: we want the LINEAR convolution to be flat

        # So we want |F|^2 to be flat, i.e., |F| to be constant

        # where target is chosen so that sum(a_new) = S

        ratio = target_mean / (absF + 1e-12)
        
        F_new = F * ratio
        a_new = np.real(ifft(F_new, fsize))[:n]
        
        # Project onto {a >= 0, sum(a) = S}
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-12:
            a_new = a_new * (S / s)
        
        a = a_new
    
    return a

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences a.

Algorithm:
1. LP-based optimization: At each step, solve a restricted linear program
   to find the best update direction b that maximizes sum(b) subject to
   (f*b)[k] <= M for the top-K lags. Update f <- (1-t)*f + t*b.
2. Spectral flattening: Iteratively adjust the spectrum to make |FFT(a)|^2
   more uniform, which flattens the circular convolution.
3. Subgradient polishing: Use weighted top-k subgradient with adaptive step.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def next_power_of_2(n):
    return 1 << (n - 1).bit_length()


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = v.copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = next_power_of_2(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def build_lp_matrix(f, indices):
    """Build the constraint matrix for the restricted LP.
    
    For each index k in indices, the constraint is:
      sum_j f[k-j] * b[j] <= M
    where the sum is over j such that 0 <= k-j < n.
    """
    n = len(f)
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1) + 1
        if j_start < j_end:
            # A[i, j] = f[k-j] for j in [j_start, j_end)
            # When j = j_start: k-j = k - j_start
            # When j = j_end-1: k-j = k - j_end + 1
            # So the values are f[k-j_start], f[k-j_start-1], ..., f[k-j_end+1]
            # = f[k-j_end+1 : k-j_start+1][::-1]
            seg_start = k - j_end + 1
            seg_end = k - j_start + 1
            A[i, j_start:j_end] = f[seg_start:seg_end][::-1]
    return A


def solve_restricted_lp(f, M, top_indices):
    """Solve the restricted LP: max sum(b) s.t. (f*b)[k] <= M for k in top_indices, b >= 0."""
    n = len(f)
    A = build_lp_matrix(f, top_indices)
    b_ub = np.full(len(top_indices), M)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    
    if res.success:
        return res.x
    return None


def compute_topk_subgradient(a, c, top_indices):
    """Weighted subgradient using top-k active lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total <= 1e-15:
        # Uniform subgradient
        return np.ones(n)
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1) + 1
        if j_start < j_end:
            seg = a[k - (j_end - 1):k - j_start + 1][::-1]
            # grad[j] += w_val * 2 * a[k-j] for j in [j_start, j_end)
            grad[j_start:j_end] += w_val * 2.0 * seg
    return grad

    f = safe_normalize(f_init, n)
    
    c = fft_convolve_self(f)
    best_score = c.max()
    best_f = f.copy()
    
    step = 0.1
    min_step = 0.01
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(f)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_f = f.copy()
        
        top_indices = np.argsort(c)[::-1][:top_k]
        b = solve_restricted_lp(f, cur, top_indices)
        
        if b is not None and b.sum() > 1e-12:
            b = b * (S / b.sum())
            f = (1.0 - step) * f + step * b
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s > 1e-12:
                f = f * (S / s)
            step = max(min_step, step * 0.999)

                lr = 0.1 * S / n

            print(f"  LP opt it={it} score={best_score:.6f} step={step:.4f} time={elapsed:.1f}s")

def run_spectral_optimization(f_init, max_time, verbose=True):

    f = safe_normalize(f_init, n)
    
    fsize = next_power_of_2(2 * n * 2)  # extra padding to avoid circular issues

    best_score = c.max()
    best_f = f.copy()

    while time.monotonic() - start < max_time:
        it += 1
        F = fft(f, fsize)

        alpha = 0.5
        ratio = target / (absF + 1e-12)
        # Blend with original
        F_new = F * (1.0 + alpha * (ratio - 1.0))

            f_new = f_new * (S / s)
        else:
            f_new = f.copy()

        if it % 100 == 0:

            cur = c.max()

def run_subgradient_polish(f_init, max_time, rng, top_k=500):

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences a.

Algorithm:
1. LP-based optimization: At each step, solve a restricted linear program
   to find the best update direction b that maximizes sum(b) subject to
   (f*b)[k] <= M for the top-K lags. Update f <- (1-t)*f + t*b.
2. Spectral flattening: Iteratively adjust the spectrum to make |FFT(a)|^2
   more uniform, which flattens the convolution.
3. Subgradient polishing: Use weighted top-k subgradient with adaptive step.
4. Multi-start with restarts and noise injection to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def next_power_of_2(n):
    """Return the smallest power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = next_power_of_2(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_two(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = next_power_of_2(size)
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def build_lp_matrix(f, indices):
    """Build the constraint matrix for the restricted LP.
    
    Row i corresponds to constraint k=indices[i]:
      sum_j f[k-j] * b[j] <= M
    """
    n = len(f)
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = int(indices[i])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1) + 1
        if j_start < j_end:
            seg_start = k - j_end + 1
            seg_end = k - j_start + 1
            A[i, j_start:j_end] = f[seg_start:seg_end][::-1]
    return A


def solve_restricted_lp(f, M, top_indices, S):
    """Solve restricted LP: max sum(b) s.t. (f*b)[k] <= M, 0 <= b <= S."""
    n = len(f)
    A = build_lp_matrix(f, top_indices)
    b_ub = np.full(len(top_indices), M)
    c_obj = -np.ones(n)
    bounds = [(0.0, S)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    
    if res.success:
        return res.x
    return None


def compute_topk_subgradient(a, c, top_indices):
    """Weighted subgradient of max(c) using top-k active lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total <= 1e-15:
        return np.ones(n)
    weights = top_vals / total
    for i in range(len(top_indices)):
        w_val = weights[i]
        k = int(top_indices[i])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1) + 1
        if j_start < j_end:
            seg_start = k - j_end + 1
            seg_end = k - j_start + 1
            grad[j_start:j_end] += w_val * 2.0 * a[seg_start:seg_end][::-1]
    return grad


def run_lp_optimization(f_init, max_time, rng, top_k=300):
    """LP-based optimization with periodic noise and stuck detection."""
    n = len(f_init)
    S = np.sqrt(2.0 * float(n))
    f = safe_normalize(f_init, n)
    
    c = fft_convolve_self(f)
    best_score = c.max()
    best_f = f.copy()
    
    step = 0.15
    min_step = 0.02
    stuck_count = 0
    last_improve = best_score
    start = time.monotonic()
    it = 0
    lp_fail_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(f)
        cur = c.max()
        if cur < best_score - 1e-9:
            best_score = cur
            best_f = f.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        top_indices = np.argsort(c)[::-1][:top_k]
        b = solve_restricted_lp(f, cur, top_indices, S)
        
        if b is not None and b.sum() > 1e-10:
            lp_fail_count = 0
            # Check full cross-convolution constraint
            cross = fft_convolve_two(f, b)
            max_cross = cross.max()
            if max_cross > cur and max_cross > 1e-15:
                b = b * (cur / max_cross)
            if b.sum() > 1e-10:
                b = b * (S / b.sum())
                f = (1.0 - step) * f + step * b
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s > 1e-12:
                    f = f * (S / s)
                step = max(min_step, step * 0.9995)
            else:
                lp_fail_count += 1
        else:
            lp_fail_count += 1
        
        # If LP keeps failing, use subgradient fallback
        if lp_fail_count > 20:
            grad = compute_topk_subgradient(f, c, top_indices)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-12:
                lr = 0.05 * S / n
                f = project_simplex(f - lr * grad, S)
            lp_fail_count = 0
        
        # Periodic noise with increasing amplitude when stuck
        if stuck_count > 200:
            noise_scale = min(0.05, 0.005 + 0.001 * (stuck_count // 200)) * S / np.sqrt(n)
            f = project_simplex(f + rng.normal(0, noise_scale, n), S)
            step = max(0.05, step * 1.1)
        
        if it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it={it} best={best_score:.6f} step={step:.4f} stuck={stuck_count} t={elapsed:.1f}s")
    
    return best_f, best_score


def run_spectral_optimization(f_init, max_time, rng):
    """Spectral flattening: make |FFT(a)|^2 more uniform."""
    n = len(f_init)
    S = np.sqrt(2.0 * float(n))
    f = safe_normalize(f_init, n)
    
    fsize = next_power_of_2(4 * n)
    
    c = fft_convolve_self(f)
    best_score = c.max()
    best_f = f.copy()
    
    start = time.monotonic()
    it = 0
    alpha = 0.4
    
    while time.monotonic() - start < max_time:
        it += 1
        F = fft(f, fsize)
        absF = np.abs(F)
        
        # Target: make |F| more uniform (geometric mean as target)
        log_abs = np.log(absF + 1e-30)
        target = np.exp(np.mean(log_abs))
        
        ratio = target / (absF + 1e-30)
        # Soft blend
        F_new = F * (1.0 + alpha * (ratio - 1.0))
        
        f_new = np.real(ifft(F_new, fsize))[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-12:
            f_new = f_new * (S / s)
        else:
            f_new = f.copy()
        
        f = f_new
        
        if it % 200 == 0:
            c = fft_convolve_self(f)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_f = f.copy()
            # Occasional noise
            if rng.random() < 0.1:
                f = project_simplex(f + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            if it % 2000 == 0:
                elapsed = time.monotonic() - start
                print(f"    Spectral it={it} best={best_score:.6f} t={elapsed:.1f}s")
            alpha = max(0.05, alpha * 0.998)
    
    c = fft_convolve_self(f)
    if c.max() < best_score:
        best_score = c.max()
        best_f = f.copy()
    
    return best_f, best_score


def run_subgradient_polish(f_init, max_time, rng, top_k=800):
    """Subgradient polishing with adaptive step and noise."""
    n = len(f_init)
    S = np.sqrt(2.0 * float(n))
    f = safe_normalize(f_init, n)
    
    c = fft_convolve_self(f)
    best_score = c.max()
    best_f = f.copy()
    
    lr = 0.03 * S / n
    start = time.monotonic()
    it = 0
    best_lr = lr
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(f)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_f = f.copy()
            no_improve = 0
            best_lr = lr
        else:
            no_improve += 1
        
        top_indices = np.argsort(c)[::-1][:top_k]
        grad = compute_topk_subgradient(f, c, top_indices)
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-12:
            break
        grad = grad / gnorm
        
        f = project_simplex(f - lr * grad, S)
        lr = max(1e-6 * S / n, lr * 0.9999)
        
        # When stuck, try larger step with noise
        if no_improve > 300:
            f = project_simplex(f + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            lr = best_lr
            no_improve = 0
        
        if it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SG it={it} best={best_score:.6f} lr={lr:.2e} t={elapsed:.1f}s")
    
    return best_f, best_score


def gaussian_init(n, rng, sigma_frac=1.0/6.0):
    """Gaussian-shaped initialization with small random perturbation."""
    center = (n - 1) / 2.0
    sigma = n * sigma_frac
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = a + 0.02 * rng.random(n)
    return a
