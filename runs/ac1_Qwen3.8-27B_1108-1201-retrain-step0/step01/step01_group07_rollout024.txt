import time

def normalize_simplex(a, S):
    """Normalize to simplex: a >= 0, sum(a) = S."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-12:
        return np.full_like(a, S / len(a))
    return a * S / s

def conv_max_and_grad(a, tau):
    """Compute soft max of conv(a,a) and its gradient w.r.t. a."""
    c = np.convolve(a, a)
    c_max = c.max()
    c_shift = c - c_max
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    # Gradient of sum_k q_k * c_k w.r.t. a_j = 2 * sum_k q_k * a_{k-j} (valid)
    g_a = 2.0 * np.correlate(q, a, 'valid')
    return c_max, q, g_a, c

def run_adam(a0, S, t_budget, tau0=5.0, tau_end=0.1, verbose=False):
    """Adam optimizer with simplex constraint and tau annealing."""
    n = a0.size
    a = normalize_simplex(a0.copy(), S)
    
    best_a = a.copy()
    best_mc = np.convolve(a, a).max()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    lr0 = 1.0
    
    t_start = time.monotonic()
    
    while time.monotonic() - t_start < t_budget:
        t += 1
        f = min(1.0, (time.monotonic() - t_start) / t_budget)
        tau = tau0 * (tau_end / tau0) ** f
        lr = lr0 / (1 + t * 0.01)
        
        c_max, q, g_a, c = conv_max_and_grad(a, tau)
        
        # Project gradient to tangent space of simplex (remove mean component)
        # The tangent space of {a >= 0, sum(a) = S} at a point in the interior
        # is {d : sum(d) = 0}. For boundary points, it's more complex.
        # Simple approach: subtract the mean to make sum(grad) = 0
        g_a = g_a - g_a.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g_a
        v = beta2 * v + (1 - beta2) * g_a**2

        v_hat = v / (1 - beta2**t)

        a = a_new

            print(f"  iter {t}, tau={tau:.3f}, max_c={best_mc:.6f}, score={score:.6f}")

    exp_c = np.exp(c_max / tau)

    return q, c

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences using:
- Adam optimizer with log-sum-exp smoothing and simplex constraint
- L-BFGS-B polishing with second-order information
- Periodic random perturbations to escape local minima
- Multi-phase search: exploration, polishing, and fine-tuning
- Multiple initializations (provided, Hann, Gaussian, chirp) and lengths
"""

import numpy as np
import time
from scipy.optimize import minimize


def normalize_to_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S} via positive scaling."""
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.full(a.shape, S / a.size, dtype=np.float64)
    return a * (S / s)


def soft_max_conv(a, tau):
    """Compute softmax weights over convolution values."""
    c = np.convolve(a, a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    return q, c


def grad_soft_max(a, q):
    """Gradient of soft-max convolution w.r.t. a."""
    return 2.0 * np.correlate(q, a, 'valid')


def current_score(a):
    """Compute the actual evaluation score."""
    n = a.size
    S = a.sum()
    if S < 0.01:
        return np.inf
    mc = np.convolve(a, a).max()
    return 2.0 * n * mc / (S * S)


def adam_optimize(a0, S, t_budget, tau_start=10.0, tau_end=0.5, lr0=0.5,
                  perturb_every=0, perturb_scale=0.02, verbose=False):
    """Adam optimizer with simplex constraint, tau annealing, and optional perturbation."""
    n = a0.size
    a = normalize_to_simplex(a0.copy(), S)
    best_a = a.copy()
    best_mc = np.convolve(a, a).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.75, 0.9, 1e-8
    t = 0
    t_start = time.monotonic()
    prev_mc = best_mc

    while time.monotonic() - t_start < t_budget:
        t += 1
        f = min(1.0, (time.monotonic() - t_start) / max(t_budget, 1e-6))
        tau = tau_start * (tau_end / tau_start) ** f
        step_lr = lr0 / (1.0 + 0.01 * t)

        q, c = soft_max_conv(a, tau)
        g = grad_soft_max(a, q)

        # Project gradient to simplex tangent space (zero-mean)
        g = g - g.mean()

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * g * g
        m_c = m / (1.0 - beta1 ** t)
        v_c = v / (1.0 - beta2 ** t)

        a_new = a - step_lr * m_c / (np.sqrt(v_c) + eps)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to_simplex(a_new, S)

        mc = c.max()
        if mc < best_mc:
            best_mc = mc
            best_a = a_new.copy()

        a = a_new

        # Periodic perturbation to escape local minima
        if perturb_every > 0 and t % perturb_every == 0:
            rng = np.random.default_rng(t)
            noise = rng.standard_normal(n) * perturb_scale
            a = normalize_to_simplex(a + noise, S)
            mc = np.convolve(a, a).max()
            if mc < best_mc:
                best_mc = mc
                best_a = a.copy()

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"      Adam: t={t}, tau={tau:.4f}, mc={best_mc:.6f}, "
                  f"score={2.0*n*best_mc/S**2:.6f}, elapsed={elapsed:.0f}s")

    return best_a, best_mc


def lbfbsb_optimize(a0, S, max_iter=3000, tau=1.0, verbose=False):
    """L-BFGS-B optimization with smooth approximation of the max."""
    n = a0.size
    a_init = normalize_to_simplex(a0.copy(), S)
    t_start = time.monotonic()

    def obj(x):
        a = np.maximum(x, 0.0)
        a = normalize_to_simplex(a, S)
        c = np.convolve(a, a)
        cm = c.max()
        cs = c - cm
        return tau * np.log(np.exp(cs / tau).sum())

    def grad(x):
        a = np.maximum(x, 0.0)
        a = normalize_to_simplex(a, S)
        c = np.convolve(a, a)
        cm = c.max()
        cs = c - cm
        ec = np.exp(cs / tau)
        q = ec / ec.sum()
        g = 2.0 * np.correlate(q, a, 'valid')
        # Subtract mean for tangent space projection (approximate)
        return g - g.mean()

    try:
        result = minimize(
            obj, a_init, method='L-BFGS-B',
            jac=grad,
            bounds=[(0.0, None)] * n,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-12, 'maxls': 40}
        )
        a_opt = normalize_to_simplex(np.maximum(result.x, 0.0), S)
    except Exception as e:
        if verbose:
            print(f"      L-BFGS-B error: {e}")
        a_opt = a_init.copy()

    mc = np.convolve(a_opt, a_opt).max()
    if verbose:
        elapsed = time.monotonic() - t_start
        print(f"      L-BFGS-B: success={result.success}, nit={result.nit}, "
              f"mc={mc:.6f}, score={2.0*n*mc/S**2:.6f}, elapsed={elapsed:.0f}s")
    return a_opt, mc


def refinement_optimize(a0, S, t_budget, tau_start=2.0, tau_end=0.2, lr0=0.2, verbose=False):
    """Fine-tuning Adam with lower tau and smaller learning rate."""
    return adam_optimize(a0, S, t_budget, tau_start=tau_start, tau_end=tau_end,
                         lr0=lr0, perturb_every=2000, perturb_scale=0.01, verbose=verbose)


def init_hann(n):
    """Hann window initialization."""
    t = np.linspace(0.0, np.pi, n)
    return 0.5 * (1.0 + np.cos(t))


def init_gaussian(n, sigma_frac=0.35):
    """Gaussian bump initialization."""
    center = (n - 1) / 2.0
    t = np.arange(n, dtype=np.float64) - center
    sigma = sigma_frac * center
    return np.exp(-t * t / (2.0 * sigma * sigma))


def init_chirp(n, seed=0):
    """Chirp-based initialization targeting flat spectrum."""
    t = np.arange(n, dtype=np.float64) - (n - 1) * 0.5
    a = 1.0 + 0.8 * np.cos(np.pi * t * t / n)
    rng = np.random.default_rng(seed)
    a += 0.03 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_triangle(n):
    """Triangular (Bartlett) window."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return 1.0 - np.abs(t - center) / center
