        return p, np.inf

    c = conv_sq(p)
    M = float(c.max())
    if not np.isfinite(M) or M <= 0:
        return p, np.inf

    # Build sparse constraint matrix: A[i, j] = p[i-j] if 0<=i-j<n else 0
    # Actually A[k, j] = p[k-j] if 0 <= k-j <= n-1, else 0
    # For k in 0..2n-2, j in 0..n-1
    rows = []
    cols = []
    data = []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            data.append(p[k - j])

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'disp': False, 'time_limit': budget_s})
    except Exception:
        return p, np.inf

    if res is None or res.x is None:
        return p, np.inf

    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0)
    s = g.sum()
    if s <= 1e-12:
        return p, np.inf
    g = g / s

    # Try blending
    best_p = p
    best_score = internal_score(p)
    for alpha in (1.0, 0.8, 0.6, 0.4, 0.25, 0.1):
        q = project_simplex((1 - alpha) * p + alpha * g)
        sc = internal_score(q)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = q

    return best_p, best_score


def make_bell(n, width):
    """Gaussian bell shape."""
    x = np.linspace(0, 1, n)
    p = np.exp(-((x - 0.5) ** 2) / (2 * width ** 2))
    return safe_normalize(p)


def make_raised_cosine(n, beta):
    """Raised cosine window."""
    x = np.linspace(0, 1, n)
    p = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5))) ** beta
    return safe_normalize(np.maximum(p, 0))


def make_parabolic(n, a):
    """Parabolic (cap) shape: 1 - a*(2x-1)^2."""
    x = np.linspace(0, 1, n)
    p = 1.0 - a * (2 * x - 1) ** 2
    return safe_normalize(np.maximum(p, 0.01))


def make_power_law(n, alpha):
    """Power law taper: (sin(pi*x))^alpha."""
    x = np.linspace(1e-6, 1 - 1e-6, n)
    p = np.sin(np.pi * x) ** alpha
    return safe_normalize(p)


def make_fourier(n, coeffs):
    """Fourier series shape."""
    x = np.linspace(0, 1, n)
    p = np.ones(n)
    for k, (amp, phase) in enumerate(coeffs):
        p += amp * np.cos(2 * np.pi * (k + 1) * x + phase)
    p = np.maximum(p, 0.01)
    return safe_normalize(p)


def generate_initials(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    # Previous best
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.size == n and parent_arr.sum() > 0:
            inits.append(safe_normalize(parent_arr))
        elif parent_arr.size > n and parent_arr.sum() > 0:
            # Resample to length n
            idx = np.linspace(0, parent_arr.size - 1, n).astype(int)
            inits.append(safe_normalize(parent_arr[idx]))
        elif parent_arr.size > 0 and parent_arr.sum() > 0:
            # Interpolate
            from numpy import interp
            x_old = np.linspace(0, 1, parent_arr.size)
            x_new = np.linspace(0, 1, n)
            inits.append(safe_normalize(interp(x_new, x_old, parent_arr)))
    except Exception:
        pass

    # Parametric families
    for w in (0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5):
        inits.append(make_bell(n, w))
    for b in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0):
        inits.append(make_raised_cosine(n, b))
    for a in (0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        inits.append(make_parabolic(n, a))
    for alpha in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 10.0):
        inits.append(make_power_law(n, alpha))

    # Fourier shapes
    for _ in range(10):
        coeffs = []
        for k in range(5):
            amp = rng.uniform(-0.3, 0.3) / (k + 1)
            phase = rng.uniform(0, 2 * np.pi)
            coeffs.append((amp, phase))
        inits.append(make_fourier(n, coeffs))

    # Random smooth
    for _ in range(5):
        f1 = rng.integers(1, max(2, n // 100))
        f2 = rng.integers(1, max(2, n // 50))
        p = 1.0
        p += rng.uniform(0.1, 0.5) * np.sin(2 * np.pi * f1 * np.linspace(0, 1, n) + rng.uniform(0, 6.28))
        p += rng.uniform(0.05, 0.3) * np.sin(2 * np.pi * f2 * np.linspace(0, 1, n) + rng.uniform(0, 6.28))
        inits.append(safe_normalize(np.maximum(p, 0.01)))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for nonnegative step function with low evaluate_sequence score."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 5.0)

    # Choose sequence length
    n = 700
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 20:
            n = int(parent_arr.size)
    except Exception:
        pass
    if n < 50:
        n = 700
    if n > 2000:
        n = 700

    print("Starting search with n=%d, budget=%.0fs" % (n, budget_s), flush=True)

    # Generate and score initializations
    inits = generate_initials(n, rng)
    scored_inits = []
    for p in inits:
        p = safe_normalize(p)
        sc = internal_score(p)
        if np.isfinite(sc):
            scored_inits.append((sc, p))
    scored_inits.sort(key=lambda x: x[0])

    best_p = scored_inits[0][1].copy() if scored_inits else np.full(n, 1.0 / n)
    best_score = scored_inits[0][0] if scored_inits else internal_score(best_p)

    print("Best initial: %.10f (from %d candidates)" % (best_score, len(scored_inits)), flush=True)

    # Phase 1: FISTA on top candidates
    top_k = min(8, len(scored_inits))
    remaining = deadline - time.monotonic()
    if remaining > 60:
        time_per = min(300.0, max(30.0, remaining * 0.6 / top_k))
        print("Phase 1: FISTA on top %d candidates, %.0fs each" % (top_k, time_per), flush=True)

        for i in range(top_k):
            if time.monotonic() >= deadline - 60:
                break
            sc, p = scored_inits[i]
            limit = min(time_per, deadline - time.monotonic() - 30)
            if limit < 10:
                break
            use_lp = (n <= 800)
            lp_int = 500 if n <= 500 else 1000
            p_best, sc_best = lfista_optimize(p, 0.1, 0.001, limit, rng, use_lp, lp_int)
            if np.isfinite(sc_best) and sc_best < best_score:
                best_score = sc_best
                best_p = p_best.copy()
                print("  Candidate %d improved to %.10f" % (i, best_score), flush=True)

    # Phase 2: LP from best
    remaining = deadline - time.monotonic()
    if remaining > 30 and n <= 800 and _HAVE_LP:
        print("Phase 2: LP from best (score=%.10f)" % best_score, flush=True)
        lp_p, lp_score = lp_best_response(best_p, min(60.0, remaining * 0.4))
        if np.isfinite(lp_score) and lp_score < best_score:
            best_score = lp_score
            best_p = lp_p.copy()
            print("  LP improved to %.10f" % best_score, flush=True)

    # Phase 3: More FISTA from best with different parameters
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("Phase 3: Extended FISTA from best (score=%.10f)" % best_score, flush=True)
        limit = min(remaining * 0.5, 300.0)
        if limit > 15:
            # Try different T schedules
            for T_init in (0.2, 0.05, 0.5):
                if time.monotonic() >= deadline - 15:
                    break
                t_limit = max(10.0, (deadline - time.monotonic() - 10) / 3.0)
                p_best2, sc_best2 = lfista_optimize(best_p, T_init, 0.0005, t_limit, rng, (n <= 800), 800)
                if np.isfinite(sc_best2) and sc_best2 < best_score:
                    best_score = sc_best2
                    best_p = p_best2.copy()
                    print("  T=%.2f improved to %.10f" % (T_init, best_score), flush=True)

    # Phase 4: Final LP + polish
    remaining = deadline - time.monotonic()
    if remaining > 20 and n <= 800 and _HAVE_LP:
        lp_p, lp_score = lp_best_response(best_p, min(30.0, remaining * 0.5))
        if np.isfinite(lp_score) and lp_score < best_score:
            best_score = lp_score
            best_p = lp_p.copy()
            print("Final LP: %.10f" % best_score, flush=True)

    # Phase 5: Final short FISTA
    remaining = deadline - time.monotonic()
    if remaining > 10:
        p_best, sc_best = lfista_optimize(best_p, 0.02, 0.0003, remaining - 5, rng, False, 10000)
        if np.isfinite(sc_best) and sc_best < best_score:
            best_score = sc_best
            best_p = p_best.copy()

    print("Final score: %.10f, length=%d" % (best_score, int(best_p.size)), flush=True)

    # Ensure output contract
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = best_p.sum()
    if s > 0:
        pass  # Keep as-is (scale invariant)
    else:
        best_p = np.full(n, 1.0 / n)

    return [float(x) for x in best_p]
