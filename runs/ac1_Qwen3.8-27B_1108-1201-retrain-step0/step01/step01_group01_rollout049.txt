"""
Momentum subgradient descent with log-sum-exp smoothing for minimizing
the normalized autocorrelation peak of non-negative sequences.
Includes diverse initializations, adaptive cooling, and coordinate descent polish.
"""

import numpy as np
import time


def _score(a):
    """Compute 2*n*max(convolve(a,a)) / sum(a)^2."""
    n = len(a)
    b = np.convolve(a, a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    return float(2.0 * n * np.max(b) / (S * S))


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_bk_vectorized(k, a, n):
    """Compute gradient of b[k] w.r.t. a, vectorized."""
    grad = np.zeros(n)
    # Part 1: a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        # j goes from lo1 to hi1, k-j goes from k-lo1 to k-hi1
        # We want grad[j] += a[k-j], so grad[lo1:hi1+1] += a[k-hi1:k-lo1+1][::-1]
        grad[lo1:hi1 + 1] += a[k - hi1:k - lo1 + 1][::-1]
    # Part 2: a[j-k] for j in [max(0,k), min(n-1,k+n-1)]
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        # j goes from lo2 to hi2, j-k goes from lo2-k to hi2-k
        grad[lo2:hi2 + 1] += a[lo2 - k:hi2 - k + 1]
    return grad


def _smoothed_grad(a, b, t, M=20):
    """Compute gradient of log-sum-exp smoothed max of b_k."""
    n = len(a)
    # Select top M lags
    if M < len(b):
        idx = np.argpartition(b, -M)[-M:]
    else:
        idx = np.arange(len(b))
    
    b_sel = b[idx]
    b_max = np.max(b_sel)
    w = np.exp((b_sel - b_max) / t)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(len(idx)) / len(idx)
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for i in range(len(idx)):
        if w[i] < 1e-8:
            continue
        k = idx[i]
        grad += w[i] * _grad_bk_vectorized(k, a, n)
    return grad


def _momentum_optimize(a, deadline, rng, t0=0.1, t_min=0.01, eta0=0.1, 
                        cool_rate=0.999, momentum0=0.9, momentum_max=0.98,
                        M=20, reheat_every=200, reheat_factor=3.0):
    """
    Momentum subgradient descent with log-sum-exp smoothing.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    else:
        a = np.ones(n) / n
    
    best = a.copy()
    best_score = _score(a)
    
    t = t0
    eta = eta0
    momentum = momentum0
    v = np.zeros(n)  # momentum buffer
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        cur_score = 2.0 * n * np.max(b) / (np.sum(a) ** 2)
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Reheat if stuck
        if no_improve >= reheat_every:
            t = min(t * reheat_factor, t0)
            eta = min(eta * 1.5, 0.5)
            no_improve = 0
            momentum = max(momentum * 0.9, 0.5)
        
        # Compute smoothed gradient
        grad = _smoothed_grad(a, b, t, M)
        
        # Center gradient (for sum constraint)
        grad -= np.mean(grad)
        
        # Momentum update
        v = momentum * v - eta * grad
        a = a + v
        a = _project_simplex(a, 1.0)
        
        # Annealing
        t = max(t_min, t * cool_rate)
        eta *= cool_rate
        momentum = min(momentum_max, momentum * 0.9999)
        
        iter_count += 1
    
    return best, best_score, iter_count


def _coordinate_descent(a, deadline, rng, max_sweeps=100):
    """
    Coordinate descent: try moving mass from one coordinate to another.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    a = a / S
    
    best = a.copy()
    best_score = _score(a)
    
    # Precompute which pairs are worth trying
    # Sort indices by value - try moving from high to low
    order = np.argsort(a)[::-1]
    
    sweeps = 0
    while time.monotonic() < deadline and sweeps < max_sweeps:
        improved = False
        # Try a random subset of coordinate moves
        n_trials = min(n, 500)
        trial_indices = rng.integers(0, n, size=n_trials)
        
        for idx in trial_indices:
            if time.monotonic() >= deadline:
                break
            src = idx
            if a[src] < 1e-8:
                continue
            # Try moving a small amount from src to a few random destinations
            for _ in range(3):
                dst = rng.integers(0, n)
                if dst == src:
                    continue
                # Try delta = a[src] * 0.01 (1% of src's value)
                delta = a[src] * 0.01
                if delta < 1e-10:
                    continue
                
                a_test = a.copy()
                a_test[src] -= delta
                a_test[dst] += delta
                
                s_test = _score(a_test)
                if s_test < best_score:
                    a = a_test
                    best = a.copy()
                    best_score = s_test
                    improved = True

                    a_test2[dst] -= delta2
                    a_test2[src] += delta2

                    if s_test2 < best_score:
                        a = a_test2
                        best = a.copy()
                        best_score = s_test2
                        improved = True
        
        sweeps += 1
    
    return best, best_score


def _make_linear_ramp(n, direction=1):
    """Create a linear ramp sequence."""
    a = np.linspace(0.1, 1.0, n)
    if direction == -1:
        a = a[::-1]
    a /= np.sum(a)
    return a


def _make_cosine_ramp(n, direction=1):
    """Create a cosine ramp sequence."""
    i = np.arange(n)
    a = 0.5 + 0.4 * np.cos(np.pi * i / n)
    a = np.maximum(a, 0.01)
    if direction == -1:
        a = a[::-1]
    a /= np.sum(a)
    return a


def _make_gaussian_bump(n, center_frac=0.3, width=0.2):
    """Create a Gaussian bump at a specific location."""
    i = np.arange(n)
    center = center_frac * (n - 1)
    a = 0.5 + 0.5 * np.exp(-0.5 * ((i - center) / (width * n)) ** 2)
    a /= np.sum(a)
    return a


def _make_random_smooth(n, rng, n_freqs=5):

        a += amp * np.cos(2 * np.pi * f * i / n + phase)
    a = np.maximum(a, 0.01)
    a /= np.sum(a)
    return a

"""
Momentum subgradient descent with log-sum-exp smoothing and FFT-based convolution
for minimizing the normalized autocorrelation peak of non-negative sequences.
Includes diverse initializations, adaptive cooling, momentum, and coordinate descent polish.
"""

import numpy as np
import time


def _fft_convolve(a, a2=None):
    """FFT-based linear convolution for speed."""
    if a2 is None:
        a2 = a
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(a2, fft_size)
    result = np.fft.irfft(fa * fb, fft_size)[:size]
    # Clean tiny negative values from FFT numerical error
    result[np.abs(result) < 1e-15] = 0.0
    return result


def _score(a):
    """Compute 2*n*max(convolve(a,a)) / sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    b = _fft_convolve(a)
    return float(2.0 * n * np.max(b) / (S * S))


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_bk_vectorized(k, a, n):
    """Compute gradient of b[k] w.r.t. a, vectorized."""
    grad = np.zeros(n)
    # Part 1: a[k-j] for valid j (i.e., 0 <= k-j < n, so k-n+1 <= j <= k)
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        # For j in [lo1, hi1]: grad[j] += a[k-j]
        # k-j ranges from k-lo1 down to k-hi1
        grad[lo1:hi1 + 1] += a[k - hi1:k - lo1 + 1][::-1]
    # Part 2: a[j-k] for valid j (i.e., 0 <= j-k < n, so k <= j <= k+n-1)
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        # For j in [lo2, hi2]: grad[j] += a[j-k]
        # j-k ranges from lo2-k to hi2-k
        grad[lo2:hi2 + 1] += a[lo2 - k:hi2 - k + 1]
    return grad


def _smoothed_grad(a, b, t, M=20):
    """Compute gradient of log-sum-exp smoothed max of b_k."""
    n = len(a)
    if M < len(b):
        idx = np.argpartition(b, -M)[-M:]
    else:
        idx = np.arange(len(b))

    b_sel = b[idx]
    b_max = np.max(b_sel)
    w = np.exp((b_sel - b_max) / t)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(len(idx)) / len(idx)
    else:
        w /= w_sum

    grad = np.zeros(n)
    for i in range(len(idx)):
        if w[i] < 1e-10:
            continue
        k = idx[i]
        grad += w[i] * _grad_bk_vectorized(k, a, n)
    return grad


def _momentum_optimize(a, deadline, rng, t0=0.05, t_min=0.002, eta0=0.05,
                       cool_rate=0.9995, momentum0=0.8, momentum_max=0.97,
                       M=20, reheat_every=200, reheat_factor=2.5,
                       admissible_increase=1.05):

    Momentum subgradient descent with log-sum-exp smoothing and adaptive step size.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    else:
        a = np.ones(n) / n

    best = a.copy()
    best_score = _score(a)

    t = t0
    eta = eta0
    momentum = momentum0
    v = np.zeros(n)
    no_improve = 0
    iter_count = 0
    iter_start = time.monotonic()

    while time.monotonic() < deadline:
        b = _fft_convolve(a)
        cur_score = 2.0 * n * np.max(b) / (np.sum(a) ** 2)

        if cur_score < best_score * (1.0 - 1e-9):
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Reheat if stuck
        if no_improve >= reheat_every:
            t = min(t * reheat_factor, t0)
            eta = min(eta * 1.3, 0.3)
            momentum = max(0.5, momentum * 0.85)
            no_improve = 0

        # Compute smoothed gradient
        grad = _smoothed_grad(a, b, t, M)
        grad -= np.mean(grad)

        # Normalize gradient for stable updates
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Momentum update
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = _project_simplex(a_new, 1.0)

        # Check if update is acceptable (allow small increases)
        b_new = _fft_convolve(a_new)
        score_new = 2.0 * n * np.max(b_new) / (np.sum(a_new) ** 2)

            eta = min(eta * 1.001, 0.3)  # Slightly increase eta

        iter_count += 1

def _coordinate_descent(a, deadline, rng, n_trials_per_sweep=200):
    """
    Coordinate descent: try moving mass between coordinate pairs.
    Uses FFT convolution for fast evaluation.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    a = a / S

    best = a.copy()
    best_score = _score(a)
    sweeps = 0
    no_improve_sweeps = 0

    # Pre-sort: high-value coordinates are good sources
    while time.monotonic() < deadline:
        improved = False
        trial_indices = rng.integers(0, n, size=n_trials_per_sweep)

        for idx in trial_indices:
            if time.monotonic() >= deadline:
                break
            src = idx
            if a[src] < 1e-10:
                continue
            # Try 2 random destinations
            for attempt in range(2):
                dst = rng.integers(0, n)
                if dst == src:
                    continue
                delta = a[src] * 0.05
                if delta < 1e-12:
                    continue

                a_test = a.copy()
                a_test[src] -= delta
                a_test[dst] += delta
                s_test = _score(a_test)
                if s_test < best_score:
                    a = a_test
                    best = a.copy()
                    best_score = s_test
                    improved = True
                    no_improve_sweeps = 0

        sweeps += 1
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 10:
                break
        else:
            no_improve_sweeps = 0

    return best, best_score

def _make_random_smooth(n, rng, n_freqs=5, amp_range=(0.1, 0.4)):

def _make_staircase(n, steps=4, direction=1):

    a = np.zeros(n)

        end = start + chunk if s < steps - 1 else n
        a[start:end] = val
    if direction == -1:

    a /= np.sum(a)
    return a

            t0=0.01, t_min=0.0005,
            eta0=0.008, cool_rate=0.9999,
            momentum0=0.92, momentum_max=0.985,
            M=50, reheat_every=500, reheat_factor=1.5,
            admissible_increase=1.02
        )
        print(f"  Final result: score={final_s:.6f}, iters={iters}")
        if final_s < results[0][0]:
            results[0] = (final_s, best_a, 0)

    # Return the best
    final_seq = [float(max(0.0, min(1000.0, x))) for x in results[0][1]]
    final_score = float(evaluate_sequence(final_seq))
    print(f"\nFINAL EVALUATED SCORE: {final_score:.6f}")
    return final_seq
