    probs = exp_shifted / W
    
    # Gradient of f_smooth w.r.t. a
    # d(f_smooth)/d(a[m]) = 2 * sum_j probs[m+j] * a[j]
    # = 2 * full_conv[n-1 + m] where full_conv = conv(probs, a_reversed)
    N = 1 << (3 * n - 2).bit_length()
    P = fft(probs, N)
    FA = fft(a_safe[::-1], N)
    full_conv = np.real(ifft(P * FA))
    grad_f = 2.0 * full_conv[n - 1:2 * n - 1]
    
    # Scale-invariant: obj = 2*n*f_smooth / S^2
    S = a_safe.sum()
    obj = 2.0 * n * f_smooth / (S * S)
    
    # Gradient of obj
    grad_obj = (2.0 * n / (S * S)) * (grad_f - 2.0 * f_smooth / S)
    
    return obj, grad_obj


def run_lbfgs(a_init, s_schedule, max_time, max_iter=5000):
    """Run L-BFGS-B with adaptive smoothing."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)

            return np.inf

            return 1e10
        return obj

        phase_start = time.monotonic()

        result = minimize(

                'maxiter': iters_per_phase,
                'maxfun': iters_per_phase * 2,
                'maxls': 20,
                'ftol': 1e-10,
                'gtol': 1e-8,
                'maxfun': iters_per_phase,

def polish_exact(a_init, max_time, top_k=10):
    """Polish using exact max with top-k subgradient (like the previous approach but simpler)."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    S = np.sqrt(2.0 * n)
    
    # Normalize
    s = a.sum()
    if s > 0:
        a = a * (S / s)
    
    best_obj = exact_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    
    lr = 0.1 * S / n
    
    while time.monotonic() - start < max_time:
        conv = fft_convolve_self(a)
        cur = 2.0 * n * conv.max() / (S * S)
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()
        
        # Top-k subgradient
        top_idx = np.argsort(conv)[::-1][:top_k]
        top_vals = conv[top_idx]
        total = top_vals.sum()
        if total < 1e-15:
            break
        weights = top_vals / total
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Project gradient step onto simplex
        a_new = project_simplex(a - lr * grad, S)
        a = a_new
        lr *= 0.999
    
    return best_a, best_obj


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):

def smooth_obj_and_grad(a, s):
