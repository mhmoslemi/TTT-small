def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def make_asym_gauss(n, mu, sigma, beta):
    """Asymmetric Gaussian: exp(-((x-mu)^2)/(2*sigma^2)) * exp(beta*(x-mu))."""
    x = np.linspace(0, 1, n)
    p = np.exp(-((x - mu) ** 2) / (2 * sigma ** 2)) * np.exp(beta * (x - mu))
    p = np.maximum(p, 0)
    s = p.sum()
    if s <= 0:
        return np.full(n, 1.0 / n)
    return p / s


def make_beta_dist(n, alpha, beta_p):
    """Beta distribution shape: x^alpha * (1-x)^beta."""
    x = np.linspace(1e-6, 1 - 1e-6, n)
    p = x ** alpha * (1 - x) ** beta_p
    s = p.sum()
    if s <= 0:
        return np.full(n, 1.0 / n)
    return p / s


def make_exp(n, rate):
    """Exponential decay: exp(-rate * x)."""
    x = np.linspace(0, 1, n)
    p = np.exp(-rate * x)
    s = p.sum()
    return p / s


def make_two_bumps(n, mu1, sig1, w1, mu2, sig2, w2):
    """Sum of two Gaussians."""
    x = np.linspace(0, 1, n)
    p = w1 * np.exp(-((x - mu1) ** 2) / (2 * sig1 ** 2)) + \
        w2 * np.exp(-((x - mu2) ** 2) / (2 * sig2 ** 2))
    p = np.maximum(p, 0)
    s = p.sum()
    if s <= 0:
        return np.full(n, 1.0 / n)
    return p / s


def make_poly(n, coeffs):
    """Polynomial in x on [0,1]."""
    x = np.linspace(0, 1, n)
    p = np.zeros(n)
    for c in coeffs:
        p = p * x + c
    p = np.maximum(p, 0)
    s = p.sum()
    if s <= 0:
        return np.full(n, 1.0 / n)
    return p / s
