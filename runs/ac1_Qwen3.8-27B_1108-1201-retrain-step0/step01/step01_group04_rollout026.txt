```python
LP-based best-response search for minimizing peak autocorrelation of non-negative sequences.
Uses iterative LP to find best responses, with multiple restarts and gradient polish.
"""

from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft, next_fast_len

def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        return np.full(max(1, p.size), 1.0/max(1, p.size), dtype=np.float64)
    return p / s

def conv_full(p):
    n = len(p)
    L = next_fast_len(2*n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))[:2*n-1]
    return c

def score(p):
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    s = float(p.sum())
    if s <= 1e-12:
        return np.inf
    c = conv_full(p)
```

```python
"""
LP-based best-response search for minimizing the peak self-convolution
of a non-negative sequence. Uses iterative LP to find best responses,
combined with gradient polishing for local refinement.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft, next_fast_len


def safe_normalize(p):
    """Normalize p to be non-negative with sum 1."""
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        return np.full(max(1, int(p.size)), 1.0/max(1, int(p.size)), dtype=np.float64)
    return p / s


def conv_full(p):
    """Full self-convolution using FFT."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))[:2 * n - 1]
    return c


def score(p):
    """Compute the evaluation score."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(p.sum())
    if s <= 1e-12:
        return np.inf
    c = conv_full(p)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (s * s))


def build_conv_matrix(p):
    """Build the sparse convolution constraint matrix for the LP.
    A[k, j] = p[j] if 0 <= k-j <= n-1, else 0.
    Shape: (2n-1, n)
    """
    n = int(p.size)
    # Column j has p[j] in rows j to j+n-1 (clipped to 2n-2)
    # Since j < n, j+n-1 < 2n-1, so rows are j to j+n-1
    total_rows = 2 * n - 1
    rows = []
    cols = []
    data = []
    for j in range(n):
        if p[j] > 0:
            r_start = j
            r_end = min(j + n, total_rows)
            length = r_end - r_start
            rows.extend(range(r_start, r_end))
            cols.extend([j] * length)
            data.extend([p[j]] * length)
    return csr_matrix((data, (rows, cols)), shape=(total_rows, n))


def lp_best_response(p, M):
    """Solve: max sum(g) s.t. (p*g)[k] <= M for all k, g >= 0.
    Returns normalized g.
    """
    n = int(p.size)
    A = build_conv_matrix(p)
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                      method='highs', options={'disp': False})
    except Exception:
        return safe_normalize(p.copy())
    
    if res is None or res.status != 0:
        return safe_normalize(p.copy())
    
    g = res.x
    g = np.asarray(g, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-12:
        return safe_normalize(p.copy())
    return g / s


def make_init(n, init_type, rng, parent=None):
    """Create an initial sequence of length n."""
    n = int(n)
    if init_type == 'uniform':
        return np.full(n, 1.0/n, dtype=np.float64)
    elif init_type == 'geo':
        r = float(rng.uniform(0.85, 0.98))
        p = r ** np.arange(n, dtype=np.float64)
        return safe_normalize(p)
    elif init_type == 'u_shape':
        x = np.linspace(0, 1, n)
        a = float(rng.uniform(1.0, 5.0))
        p = 1.0 + a * (2*x - 1)**2
        return safe_normalize(np.maximum(p, 0.01))
    elif init_type == 'endpoint':
        # Concentrate at endpoints
        decay = float(rng.uniform(5.0, 30.0))
        x = np.linspace(0, 1, n)
        p = np.exp(-decay * x) + np.exp(-decay * (1-x))
        return safe_normalize(p)
    elif init_type == 'two_bump':
        x = np.linspace(0, 1, n)
        mu1 = float(rng.uniform(0.1, 0.35))
        mu2 = float(rng.uniform(0.65, 0.9))
        sig = float(rng.uniform(0.02, 0.1))
        p = np.exp(-0.5*((x-mu1)/sig)**2) + np.exp(-0.5*((x-mu2)/sig)**2)
        return safe_normalize(np.maximum(p, 1e-10))
    elif init_type == 'random_smooth':
        x = np.linspace(0, 1, n)
        p = 1.0
        for _ in range(int(rng.integers(2, 6))):
            f = float(rng.uniform(0.1, 3.0))
            phase = float(rng.uniform(0, 2*np.pi))
            amp = float(rng.uniform(0.3, 1.0))
            p += amp * np.sin(2*np.pi*f*x + phase)
        return safe_normalize(np.maximum(p, 0.01))
    elif init_type == 'parent':
        if parent is not None and parent.size > 0:
            # Resample parent to length n
            from numpy import interp
            old_x = np.linspace(0, 1, parent.size)
            new_x = np.linspace(0, 1, n)
            p = interp(new_x, old_x, parent)
            return safe_normalize(p)
        return np.full(n, 1.0/n, dtype=np.float64)
    else:
        return np.full(n, 1.0/n, dtype=np.float64)


def lp_search(p, max_time, rng, verbose=True):
    """Run LP best-response iteration with periodic restarts."""
    p = safe_normalize(p.copy())
    n = int(p.size)
    best_p = p.copy()
    best_sc = score(p)
    start = time.monotonic()
    
    if verbose:
        print(f"  LP search start: n={n}, initial score={best_sc:.6f}")
    
    iter_count = 0
    since_improve = 0
    t = 1.0  # Full step initially
    
    while time.monotonic() - start < max_time:
        iter_count += 1
        c = conv_full(p)
        M = float(np.max(c))
        if M <= 0:
            break
        
        g = lp_best_response(p, M)
        
        # Check convergence
        diff = float(np.max(np.abs(p - g)))
        if diff < 1e-12:
            if verbose:
                print(f"  Converged at iter {iter_count}")
            break

def simplex_project(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)

    s = out.sum()
```

```python
        return np.empty(0)

        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def make_init(n, init_type, rng, parent=None):
    n = int(n)
    if init_type == 'uniform':
        return np.full(n, 1.0 / n, dtype=np.float64)
    elif init_type == 'geo_up':
        r = float(rng.uniform(0.90, 0.985))
        p = r ** np.arange(n, dtype=np.float64)
        return safe_normalize(p)
    elif init_type == 'geo_down':
        r = float(rng.uniform(0.90, 0.985))
        p = r ** np.arange(n, dtype=np.float64)
        return safe_normalize(p[::-1])
    elif init_type == 'u_shape':
        x = np.linspace(0, 1, n)
        a = float(rng.uniform(1.0, 6.0))
        p = 1.0 + a * (2 * x - 1) ** 2
        return safe_normalize(np.maximum(p, 0.01))
    elif init_type == 'endpoint':
        decay = float(rng.uniform(8.0, 40.0))
        x = np.linspace(0, 1, n)
        p = np.exp(-decay * x) + np.exp(-decay * (1 - x))
        return safe_normalize(p)
    elif init_type == 'two_bump':
        x = np.linspace(0, 1, n)
        mu1 = float(rng.uniform(0.05, 0.3))
        mu2 = float(rng.uniform(0.7, 0.95))
        sig = float(rng.uniform(0.02, 0.08))
        p = np.exp(-0.5 * ((x - mu1) / sig) ** 2) + np.exp(-0.5 * ((x - mu2) / sig) ** 2)
        return safe_normalize(np.maximum(p, 1e-10))
    elif init_type == 'random_smooth':
        x = np.linspace(0, 1, n)
        p = 1.0
        n_harmonics = int(rng.integers(2, 7))
        for _ in range(n_harmonics):
            f = float(rng.uniform(0.05, 4.0))
            phase = float(rng.uniform(0, 2 * np.pi))
            amp = float(rng.uniform(0.2, 0.9))
            p += amp * np.sin(2 * np.pi * f * x + phase)
        return safe_normalize(np.maximum(p, 0.01))
    elif init_type == 'sinusoidal':
        x = np.linspace(0, 1, n)
        freq = float(rng.uniform(0.5, 3.0))
        amp = float(rng.uniform(0.3, 0.95))
        p = 1.0 + amp * np.cos(2 * np.pi * freq * x)
        return safe_normalize(np.maximum(p, 0.005))
    elif init_type == 'parent':
        if parent is not None and parent.size > 1:
            old_x = np.linspace(0, 1, parent.size)
            new_x = np.linspace(0, 1, n)
            p = np.interp(new_x, old_x, parent)
            return safe_normalize(p)
        return np.full(n, 1.0 / n, dtype=np.float64)
    elif init_type == 'beta':
        x = np.linspace(1e-6, 1 - 1e-6, n)
        alpha = float(rng.uniform(0.5, 4.0))
        beta = float(rng.uniform(0.5, 4.0))
        p = x ** alpha * (1 - x) ** beta
        return safe_normalize(np.maximum(p, 1e-10))
    else:
        return np.full(n, 1.0 / n, dtype=np.float64)


def lp_search(p, max_time, rng, verbose=True):
    p = safe_normalize(p.copy())
    n = int(p.size)
    best_p = p.copy()
    best_sc = score(p)
    start = time.monotonic()

    if verbose:
        print("  LP start: n=%d, score=%.8f" % (n, best_sc), flush=True)

    iter_count = 0
    since_improve = 0
    t = 1.0

    while time.monotonic() - start < max_time:
        iter_count += 1
        c = conv_full(p)
        M = float(np.max(c))
        if M <= 0:
            break

        g = lp_best_response(p, M)
        diff = float(np.max(np.abs(p - g)))

        if diff < 1e-11:
            if verbose:
                print("  Converged at iter %d, score=%.8f" % (iter_count, best_sc), flush=True)
            break

        p_new = (1 - t) * p + t * g
        p_new = safe_normalize(p_new)
        sc = score(p_new)

        if sc < best_sc:
            best_sc = sc
            best_p = p_new.copy()
            since_improve = 0
        else:
            since_improve += 1

        p = p_new

        if since_improve > 20 and iter_count % 8 == 0:
            eps = float(rng.uniform(0.05, 0.35))
            pert = make_init(n, 'random_smooth', rng, None)
            p = safe_normalize((1 - eps) * best_p + eps * pert)
            t = 1.0
            since_improve = 0

        if since_improve > 60:
            eps = float(rng.uniform(0.2, 0.5))
            pert = make_init(n, 'random_smooth', rng, None)
            p = safe_normalize((1 - eps) * best_p + eps * pert)
            t = 1.0
            since_improve = 0

        if best_sc < 1.45 and since_improve > 100:
            break

        if iter_no_check(iter_count):
            pass

    if verbose:
        print("  LP done: %d iters, best=%.8f" % (iter_count, best_sc), flush=True)

    return best_p, best_sc

    return False


def compute_gradient(p, T_factor):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    c = conv_full(p)
    cmax = float(np.max(c)) if c.size > 0 else 0.0
    if cmax <= 0:
        return np.zeros(n), cmax
    T = max(T_factor * cmax, 1e-300)
    diff = (c - cmax) / T
    e = np.exp(diff)
    raw = float(e.sum())
    if not np.isfinite(raw) or raw <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / raw

    L = next_fast_len(3 * n - 2)
    Fp = fft(p, L)
    w_rev = w[::-1]
    Fw = fft(w_rev, L)
    h = np.real(ifft(Fp * Fw))

    G = h[2 * n - 2:n - 2:-1]
    if G.size != n:
        tmp = np.zeros(n)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp

    grad = 2.0 * np.maximum(G, 0.0)
    return grad, cmax


def gradient_polish(p, max_time, rng, verbose=True):
    p = safe_normalize(p.copy())
    n = int(p.size)
    best_p = p.copy()
    best_sc = score(p)
    cur_sc = best_sc
    prev_p = p.copy()
    start = time.monotonic()
    factor = 0.05
    iter_no = 0
    since_improve = 0
    momentum = 0.0

    if verbose:
        print("  Grad polish start: n=%d, score=%.8f" % (n, best_sc), flush=True)

    while time.monotonic() - start < max_time:
        iter_no += 1

        if iter_no > 100:
            momentum = 0.3

        grad, cmax = compute_gradient(p, factor)
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)

        gabs = float(np.max(np.abs(grad))) if grad.size > 0 else 0.0
        if gabs < 1e-300:
            break

        p_scale = max(1.0 / n, 0.05 * float(np.max(p)), 1e-12)
        base_lr = p_scale / gabs

        accepted = False
        for lr_mult in [0.5, 0.2, 0.08, 0.03, 0.01, 0.003]:
            lr = base_lr * lr_mult
            v = p - lr * grad
            if momentum > 0:
                v = v + momentum * (p - prev_p)
            pt = simplex_project(v)
            sc = score(pt)

            if np.isfinite(sc) and sc < cur_sc - 1e-14:
                p = pt
                cur_sc = sc
                if sc < best_sc:
                    best_sc = sc
                    best_p = pt.copy()
                    since_improve = 0
                else:
                    since_improve += 1
                prev_p = p
                accepted = True
                break

        if not accepted:
            factor = max(0.003, factor * 0.85)
            since_improve += 1

            if since_improve > 80:
                sigma = float(rng.uniform(0.01, 0.1)) / np.sqrt(n)
                pt = simplex_project(p + rng.normal(0, sigma, n))
                sc = score(pt)
                if np.isfinite(sc) and sc < cur_sc - 1e-14:
                    p = pt
                    cur_sc = sc
                    if sc < best_sc:
                        best_sc = sc
                        best_p = pt.copy()
                        since_improve = 0
                    prev_p = p
                    accepted = True

            if since_improve > 250:
                eps = float(rng.uniform(0.02, 0.15))
                pert = make_init(n, 'random_smooth', rng, None)
                p = simplex_project((1 - eps) * best_p + eps * pert)
                cur_sc = score(p)
                prev_p = p
                since_improve = 0
                factor = 0.05

        if iter_no % 1000 == 0 and verbose:
            elapsed = time.monotonic() - start
            print("    iter %d, best=%.8f, T=%.4f, elapsed=%.0fs" % (iter_no, best_sc, factor, elapsed), flush=True)

    if verbose:
        print("  Grad polish done: best=%.8f" % best_sc, flush=True)

    return best_p, best_sc


def propose_candidate(seed=42, budget_s=1800):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 10 and parent_arr.sum() > 0:
            parent = safe_normalize(parent_arr)
    except Exception:
        parent = None

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = safe_normalize(p)
        if not np.isfinite(sc):
            sc = score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()

    init_types = ['geo_up', 'geo_down', 'u_shape', 'endpoint', 'two_bump', 'sinusoidal', 'beta', 'random_smooth']

    lengths_to_try = []
    if parent is not None:
        pn = int(parent.size)
        lengths_to_try.append(pn)
        lengths_to_try.append(max(200, pn - 100))
        lengths_to_try.append(pn + 100)
        lengths_to_try.append(pn + 300)
    lengths_to_try.extend([300, 500, 700, 1000, 1500, 2000, 2500])

    seen = set()
    unique_lengths = []
    for ln in lengths_to_try:
        ln = int(max(100, min(3000, ln)))
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)

    unique_lengths.sort()

    jobs = []
    for n in unique_lengths:
        for it in init_types[:4]:
            jobs.append((n, it))

    rng.shuffle(jobs)

    n_jobs = len(jobs)
    if n_jobs == 0:
        best_p = np.full(700, 1.0/700)
        return [float(x) for x in best_p]

    time_per_job = max(15.0, min(80.0, (deadline - start_time) * 0.7 / n_jobs))

    job_idx = 0
    for n, it in jobs:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        p_init = make_init(n, it, rng, parent)
        sc_init = score(p_init)
        try_record(p_init, sc_init)

        if remaining < 30:
            continue

        alloc = min(time_per_job, remaining * 0.5)
        p_best, sc_best = lp_search(p_init, alloc, rng, verbose=(best_score > 1.55))
        try_record(p_best, sc_best)

        job_idx += 1

    remaining = deadline - time.monotonic()
    if best_p is not None and remaining > 30:
        print("Before polish: score=%.8f, n=%d, time_left=%.0fs" % (best_score, best_p.size, remaining), flush=True)
        polish_time = max(10.0, remaining - 10.0)
        p_pol, sc_pol = gradient_polish(best_p, polish_time, rng, verbose=True)
        try_record(p_pol, sc_pol)

    if best_score < 1.50 and best_p is not None and (deadline - time.monotonic()) > 15:
        p_pol2, sc_pol2 = gradient_polish(best_p, deadline - time.monotonic() - 5, rng, verbose=True)
        try_record(p_pol2, sc_pol2)

    if best_p is None:
        best_p = np.full(700, 1.0 / 700, dtype=np.float64)
        if parent is not None:
            best_p = parent.copy()

    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s <= 1e-12 or best_p.size == 0:
        best_p = np.full(700, 1.0 / 700, dtype=np.float64)

    final_sc = score(best_p)
    print("FINAL: score=%.8f, n=%d" % (final_sc, best_p.size), flush=True)

    return [float(x) for x in best_p]
```
