"""
Nesterov-accelerated projected subgradient descent with progressive peak-focusing,
adaptive step size, and simulated annealing restarts to minimize the normalized
autocorrelation peak of a non-negative sequence. Uses cosine-profile parameterization
for coarse-to-fine optimization.
"""

import numpy as np
import time


def _grad_bk(a, k):
    """Gradient of b_k = sum_j a_j*a_{k-j} w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _objective(a):
    """Compute 2*n*max(convolve(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _softmax_weighted_grad(a, M=10, alpha=50.0):
    """Compute softmax-weighted gradient focusing on top M peaks."""
    n = len(a)
    b = np.convolve(a, a)
    if M < len(b):
        k_idx = np.argpartition(b, -M)[-M:]
    else:
        k_idx = np.arange(len(b))
    b_sel = b[k_idx]
    b_max = np.max(b)
    scaled = alpha * (b_sel - b_max)
    scaled -= np.max(scaled)
    wts = np.exp(scaled)
    wts_sum = np.sum(wts)
    if wts_sum < 1e-30:
        wts = np.ones(len(k_idx)) / len(k_idx)
    else:
        wts /= wts_sum
    grad = np.zeros(n)
    for ki in range(len(k_idx)):
        if wts[ki] > 1e-12:
            grad += wts[ki] * _grad_bk(a, k_idx[ki])
    return grad, b_max


def _nesterov_optimize(a, deadline, rng, alpha_init=5.0, alpha_end=500.0,
                       M_init=30, M_end=5, momentum_init=0.7, momentum_end=0.9,
                       eta_init=None, use_anneal=True, T_init=0.05, T_end=0.0):
    """
    Nesterov-accelerated projected subgradient descent with progressive focusing.
    Returns (best_a, best_score).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    v = np.zeros(n)
    best = a.copy()
    best_score = _objective(best)

    if eta_init is None:
        eta_init = 0.05 / np.sqrt(n)
    eta = eta_init

    start_time = time.monotonic()
    est_total = max(1.0, deadline - start_time)
    iteration = 0
    no_improve_count = 0
    reheat_interval = 2000

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / est_total)

        # Progressive parameter scheduling
        alpha = alpha_init * (alpha_end / alpha_init) ** frac
        M = max(M_end, int(M_init - (M_init - M_end) * frac))
        momentum = momentum_init + (momentum_end - momentum_init) * frac

        # Nesterov lookahead
        a_y = a + momentum * v
        a_y = _project_simplex(a_y, 1.0)

        # Compute smoothed gradient
        grad, b_max = _softmax_weighted_grad(a_y, M=M, alpha=alpha)
        # Center gradient (remove mean for sum constraint)
        grad -= np.mean(grad)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Update momentum and position
        v = momentum * v - eta * grad / grad_norm
        a_new = a + v
        a_new = _project_simplex(a_new, 1.0)

        # Evaluate
        score_new = _objective(a_new)

        # Simulated annealing acceptance
        accept = True
        if use_anneal and score_new > best_score:
            T = T_init * max(0.0, 1.0 - frac) ** 1.5
            if T > 1e-10:
                delta = score_new - best_score
                # Work in normalized units
                delta_norm = delta / best_score
                if delta_norm < T:
                    accept = True
                else:
                    accept = rng.random() < np.exp(-delta_norm / T)
            if not accept:
                a_new = a

        a = a_new
        iteration += 1

        if score_new < best_score - 1e-12:
            best_score = score_new
            best = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Adaptive step size: reduce if too many non-improving steps
        if no_improve_count > reheat_interval:
            eta *= 0.7
            no_improve_count = 0
            # Occasionally perturb to escape local minima
            if rng.random() < 0.3:
                noise = rng.standard_normal(n)
                noise -= np.mean(noise)
                noise /= (np.linalg.norm(noise) + 1e-15)
                a = _project_simplex(a + 0.02 * noise, 1.0)
            v *= 0.5

        # Safety: ensure eta doesn't get too small
        if eta < 1e-10:
            break

    return best, best_score

    """Create sequence from cosine coefficients: a_i = coeffs[0] + sum_m coeffs[m]*cos(m*pi*i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    a = np.full(n, coeffs[0])
    for m in range(1, len(coeffs)):
        if abs(coeffs[m]) > 1e-15:
            a += coeffs[m] * np.cos(m * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _coarse_cosine_optimization(n, deadline_coarse, rng):
    """Optimize over cosine profile coefficients using scipy minimize."""
    from scipy.optimize import minimize as sp_minimize

    # Parameterize: a_i = c0 + c1*cos(pi*i/(n-1)) + c2*cos(2*pi*i/(n-1)) + ...
    n_harm = 5
    x0 = np.zeros(n_harm)
    x0[0] = 1.0
    x0[1] = 0.8  # Strong anti-symmetric component

    def obj(x):
        a = _make_cosine_profile(n, x)
        return _objective(a)

    best_x = x0.copy()
    best_val = obj(x0)

    # Try multiple starting points
    starts = [x0]
    for eps in [0.3, 0.5, 0.6, 0.9, 1.2]:
        s = np.zeros(n_harm)
        s[0] = 1.0
        s[1] = eps
        starts.append(s)
    # Add some random starts
    for _ in range(3):
        s = np.zeros(n_harm)
        s[0] = 1.0
        s[1] = rng.uniform(0.3, 1.2)
        for m in range(2, n_harm):
            s[m] = rng.uniform(-0.1, 0.1)
        starts.append(s)

    for x_start in starts:
        if time.monotonic() > deadline_coarse:
            break
        try:
            res = sp_minimize(obj, x_start, method='Nelder-Mead',
                             options={'maxiter': 500, 'xatol': 1e-6, 'fatol': 1e-8})
            if res.fun < best_val:
                best_val = res.fun
                best_x = res.x
        except Exception:
            pass

    return _make_cosine_profile(n, best_x), best_val


def _local_perturbation_search(a_best, deadline, rng, n_perturb=10):
    """Try local perturbations around the best solution."""
    n = len(a_best)
    best = a_best.copy()
    best_score = _objective(best)
    per_pert = max(5.0, (deadline - time.monotonic()) / max(1, n_perturb))

    for i in range(n_perturb):
        if time.monotonic() >= deadline:
            break
        # Different perturbation strategies
        strategy = i % 4
        if strategy == 0:
            # Smooth random perturbation
            freqs = rng.integers(1, max(2, n // 8), size=min(3, n // 4))
            noise = np.zeros(n)
            for f in freqs:
                phase = rng.uniform(0, 2 * np.pi)
                amp = rng.uniform(-1, 1)
                noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
            noise /= (np.max(np.abs(noise)) + 1e-15)
            a_pert = _project_simplex(best + 0.05 * noise, 1.0)
        elif strategy == 1:
            # Anti-symmetric perturbation
            i_idx = np.arange(n)
            noise = 0.03 * np.cos(np.pi * i_idx / (n - 1) + rng.uniform(0, 2 * np.pi))
            a_pert = _project_simplex(best + noise, 1.0)
        elif strategy == 2:
            # Localized perturbation (random block)
            start = rng.integers(0, n - 500)
            length = rng.integers(200, 500)
            noise = np.zeros(n)
            noise[start:start+length] = rng.uniform(-0.02, 0.02, length)
            a_pert = _project_simplex(best + noise, 1.0)
        else:
            # Mirror perturbation
            a_pert = best.copy()
            flip = rng.integers(0, n // 4)
            a_pert[flip:flip+n//10] = best[n-1-flip-n//10:n-1-flip]
            a_pert = _project_simplex(a_pert, 1.0)

        p_deadline = min(time.monotonic() + per_pert, deadline)
        a_opt, s_opt = _nesterov_optimize(a_pert, p_deadline, rng,
                                           alpha_init=50.0, alpha_end=300.0,
                                           M_init=10, M_end=3,
                                           eta_init=0.02 / np.sqrt(n),
                                           use_anneal=False)
        if s_opt < best_score:
            best_score = s_opt
            best = a_opt.copy()
            print(f"  Perturb {i+1}: New best! Score: {best_score:.6f}")

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization: coarse cosine profile search, Nesterov gradient
    descent with progressive focusing, perturbation search, and final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    # Get starting sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size
    S = np.sum(parent)
    if S < 1e-10:
        parent = np.ones(n)
        S = n
    parent = parent / S

    global_best = parent.copy()
    global_best_score = _objective(global_best)
    print(f"Initial score: {global_best_score:.6f} (n={n})")

    # Phase 1: Coarse cosine profile optimization (30 seconds)
    t1_deadline = time.monotonic() + 30.0
    print("Phase 1: Coarse cosine profile optimization...")
    a_coarse, s_coarse = _coarse_cosine_optimization(n, t1_deadline, rng)
    print(f"  Coarse best: {s_coarse:.6f}")
    if s_coarse < global_best_score:
        global_best = a_coarse.copy()
        global_best_score = s_coarse
        print(f"  New global best from cosine: {global_best_score:.6f}")

    # Phase 2: Main Nesterov optimization from best start (60% of remaining time)
    remaining = deadline - time.monotonic()
    t2_deadline = time.monotonic() + 0.6 * remaining
    print(f"Phase 2: Main Nesterov optimization ({t2_deadline - time.monotonic():.0f}s budget)...")

    # Start from the better of parent and coarse
    a_start = global_best.copy()
    a_opt, s_opt = _nesterov_optimize(a_start, t2_deadline, rng,
                                       alpha_init=5.0, alpha_end=500.0,
                                       M_init=30, M_end=5,
                                       momentum_init=0.7, momentum_end=0.92,
                                       eta_init=0.05 / np.sqrt(n),
                                       use_anneal=True, T_init=0.03, T_end=0.0)
    if s_opt < global_best_score:
        global_best_score = s_opt
        global_best = a_opt.copy()
        print(f"  Main optimization result: {global_best_score:.6f}")
    else:
        print(f"  Main optimization result: {s_opt:.6f} (no improvement, best={global_best_score:.6f})")

    # Phase 3: Try anti-symmetric start with Nesterov (15% of remaining)
    remaining = deadline - time.monotonic()
    t3_deadline = time.monotonic() + 0.15 * remaining
    if remaining > 30:
        print(f"Phase 3: Anti-symmetric start optimization ({t3_deadline - time.monotonic():.0f}s)...")
        i_idx = np.arange(n, dtype=np.float64)
        a_anti = 1.0 + 0.7 * np.cos(np.pi * i_idx / (n - 1))
        a_anti = np.maximum(a_anti, 0.0)
        a_anti = a_anti / np.sum(a_anti)
        a_opt3, s_opt3 = _nesterov_optimize(a_anti, t3_deadline, rng,
                                             alpha_init=10.0, alpha_end=400.0,
                                             M_init=20, M_end=5,
                                             momentum_init=0.75, momentum_end=0.9,
                                             eta_init=0.04 / np.sqrt(n),
                                             use_anneal=True, T_init=0.02)
        if s_opt3 < global_best_score:
            global_best_score = s_opt3
            global_best = a_opt3.copy()
            print(f"  Anti-symmetric result: {global_best_score:.6f} (NEW BEST)")
        else:
            print(f"  Anti-symmetric result: {s_opt3:.6f}")

    # Phase 4: Perturbation search around best (10% of remaining)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t4_deadline = time.monotonic() + 0.15 * remaining
        print(f"Phase 4: Local perturbation search ({t4_deadline - time.monotonic():.0f}s)...")
        a_pert, s_pert = _local_perturbation_search(global_best, t4_deadline, rng, n_perturb=5)
        if s_pert < global_best_score:
            global_best_score = s_pert
            global_best = a_pert.copy()
            print(f"  Perturbation search result: {global_best_score:.6f} (NEW BEST)")

    # Phase 5: Final polish with very small steps (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"Phase 5: Final polish ({remaining:.0f}s)...")
        a_polish, s_polish = _nesterov_optimize(global_best, deadline, rng,
                                                 alpha_init=200.0, alpha_end=1000.0,
                                                 M_init=5, M_end=2,
                                                 momentum_init=0.85, momentum_end=0.9,
                                                 eta_init=0.01 / np.sqrt(n),
                                                 use_anneal=False)
        if s_polish < global_best_score:
            global_best_score = s_polish
            global_best = a_polish.copy()
            print(f"  Polish result: {global_best_score:.6f} (NEW BEST)")
        else:
            print(f"  Polish result: {s_polish:.6f}")

    # Final verification
    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"FINAL SCORE: {final_score:.6f}")
    return result
