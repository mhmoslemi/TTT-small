def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))

def normalize(a):
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def mirror_descent_step(a, lr, eps):
    """One step of soft-max smoothed mirror descent."""
    conv = np.convolve(a, a)
    # Softmax weights
    conv_max = conv.max()
    if conv_max == 0:
        return a
    exp_vals = np.exp(eps * (conv - conv_max))
    w = exp_vals / exp_vals.sum()
    
    # Gradient: grad_j = sum_k w_k * a_{k-j}
    # = (w * a_rev)[n-1+j]
    n = len(a)
    a_rev = a[::-1]
    conv_w = np.convolve(w, a_rev)
    grad = conv_w[n-1:2*n-1]
    
    # Mirror descent update
    a_new = a * np.exp(-lr * grad)
    a_new = np.maximum(0, a_new)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s

    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

        perturbation = a.copy()
        # Pick a random segment to perturb

        if T < T_min:

    return best_a, best_score

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/(sum(a))^2` over non-negative sequences. The key insights are:

1. The problem is scale-invariant, so I normalize `sum(a)=1`
2. The problem is **non-convex** (max of indefinite quadratic forms), so local minima exist
3. Mirror descent (exponentiated gradient) is natural for the simplex constraint
4. Soft-max smoothing makes the objective differentiable for use with L-BFGS
5. The theoretical lower bound is ~1 (flat convolution), we're at 1.516, target 1.5027

My approach combines:
- **Soft-max smoothed mirror descent** (many fast iterations)
- **L-BFGS on the smooth surrogate** (fewer but more precise steps)
- **Simulated annealing** for escaping local minima
- **Multiple lengths and initializations** for diversity

"""
Multi-stage optimization for minimizing peak auto-convolution of non-negative sequences.
Combines soft-max smoothed mirror descent, L-BFGS on smooth surrogates, and simulated
annealing with multiple sequence lengths and diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_conv(a, b=None):
    """FFT-based full convolution."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size)[:size]


def score_sequence(a):
    n = len(a)
    conv = fast_conv(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


def normalize(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def compute_softmax_grad(a, eps):
    """Compute the gradient of the soft-max smoothed peak convolution."""
    n = len(a)
    conv = fast_conv(a)
    conv_max = conv.max()
    exp_vals = np.exp(eps * (conv - conv_max))
    w = exp_vals / exp_vals.sum()
    # grad_j = sum_k w_k * a_{k-j} = (w * a_rev)[n-1+j]
    a_rev = a[::-1]
    conv_w = fast_conv(w, a_rev)
    grad = conv_w[n - 1:2 * n - 1]
    return grad, conv, w


def md_step(a, lr, eps):
    """One mirror descent step."""
    grad, _, _ = compute_softmax_grad(a, eps)
    # Scale gradient to unit mean for stable updates
    g_mean = grad.mean()
    if g_mean > 0:
        grad = grad / g_mean
    a_new = a * np.exp(-lr * grad)
    a_new = np.maximum(0, a_new)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def run_mirror_descent(a_init, time_limit, lr_init=0.3, eps_init=8.0, verbose=False):
    """Run soft-max smoothed mirror descent with adaptive step size and annealing."""
    start = time.monotonic()
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    lr = lr_init
    eps = eps_init
    stagnant = 0
    best_stagnant_score = best_score

    for it in range(500000):
        if time.monotonic() - start > time_limit:
            break

        a = md_step(a, lr, eps)

        if it % 5 == 0:
            sc = score_sequence(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                stagnant = 0
                if verbose and (it < 30 or it % 200 == 0):
                    print(f"        MD it={it}: score={sc:.6f}, lr={lr:.4f}, eps={eps:.2f}")
            else:
                stagnant += 1

            # Adaptive: reduce lr if stuck
            if stagnant >= 30:
                lr *= 0.7
                stagnant = 0

            # Anneal parameters
            if it % 100 == 0:
                lr = max(0.005, lr * 0.997)
                eps = max(0.05, eps * 0.97)

            # If very stuck, try a kick
            if stagnant >= 150:
                a = a + 0.01 * np.random.default_rng(it).random(n)
                a = normalize(a)
                stagnant = 0

    return best_a, best_score


def lbgfsg_obj_and_grad(theta, eps, n):
    """Objective and gradient for L-BFGS on log-parameterization."""
    a_local = np.exp(theta - theta.max())  # numerical stability
    a_local = a_local / a_local.sum()

    conv = fast_conv(a_local)
    conv_max = conv.max()
    exp_vals = np.exp(eps * (conv - conv_max))
    f_val = conv_max + np.log(exp_vals.sum()) / eps

    w = exp_vals / exp_vals.sum()
    a_rev = a_local[::-1]
    conv_w = fast_conv(w, a_rev)
    grad_a = conv_w[n - 1:2 * n - 1]

    # Chain rule for log-parameterization
    sum_ga_a = np.dot(grad_a, a_local)
    grad_theta = grad_a * a_local - a_local * sum_ga_a

    return f_val, grad_theta


def run_lbgfgs(a_init, time_limit, verbose=False):
    """Run L-BFGS on soft-max smoothed objective with annealed epsilon."""
    start = time.monotonic()
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    eps_schedule = [20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.08]
    theta = np.log(a + 1e-15)

    for eps in eps_schedule:
        if time.monotonic() - start > time_limit:
            break
        remaining = time_limit - (time.monotonic() - start)
        if remaining < 2:
            break

        max_iter = min(100, max(20, int(remaining * 5)))

        def fun(th):
            return lbgfsg_obj_and_grad(th, eps, n)

        try:
            res = scipy_minimize(fun, theta, jac=True, method='L-BFGS-B',
                                 options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                          'gtol': 1e-6, 'ftol': 1e-8})
            if res.x is not None:
                a_new = np.exp(res.x - res.x.max())
                a_new = normalize(a_new)
                sc = score_sequence(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    theta = res.x.copy()
                    if verbose:
                        print(f"        L-BFGS eps={eps:.2f}: score={sc:.6f}")
                else:
                    # Keep old theta
                    pass
        except Exception:
            pass

    return best_a, best_score


def simulated_annealing(a_init, time_limit, T_init=0.05, T_min=0.001, rng=None, verbose=False):
    """Simulated annealing with adaptive perturbation scale."""
    if rng is None:
        rng = np.random.default_rng(123)
    start = time.monotonic()
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    T = T_init
    accept_mid = 0
    eval_count = 0

    while time.monotonic() - start < time_limit:
        # Choose perturbation: mix of local and global
        if rng.random() < 0.7:
            # Local: perturb a segment
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            pos = int(rng.integers(0, n - width))
            a_trial = a.copy()
            a_trial[pos:pos + width] *= np.exp(rng.normal(0, 0.2, width))
        else:
            # Global: multiplicative noise on all entries
            a_trial = a * np.exp(rng.normal(0, 0.05, n))

        a_trial = normalize(a_trial)
        sc = score_sequence(a_trial)
        eval_count += 1

        delta = sc - best_score
        if delta < 0:
            a = a_trial
            best_score = sc
            best_a = a_trial.copy()
            accept_mid += 1
            if verbose and eval_count % 500 == 0:
                print(f"        SA eval={eval_count}: score={best_score:.6f}, T={T:.4f}")
        elif delta / max(T, 1e-10) < 30 and rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial

        # Temperature schedule
        if eval_count % 50 == 0:
            T *= 0.97
        if T < T_min:
            T = T_init * 0.3  # Reheat
            # Also try a bigger kick
            a = a + 0.05 * rng.standard_normal(n)
            a = normalize(a)

    return best_a, best_score


def local_refine(a_init, time_limit, rng=None, verbose=False):
    """Fine-grained local search with small deterministic perturbations."""
    if rng is None:
        rng = np.random.default_rng(99)
    start = time.monotonic()
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    sigma = 0.02

    while time.monotonic() - start < time_limit:
        batch = 20
        best_trial = None
        best_trial_score = best_score

        for _ in range(batch):
            a_trial = a.copy()
            a_trial *= np.exp(rng.normal(0, sigma, n))
            a_trial = normalize(a_trial)
            sc = score_sequence(a_trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial = a_trial

        if best_trial is not None:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            if verbose:
                print(f"        LR: score={best_score:.6f}, sigma={sigma:.4f}")
        else:
            sigma *= 0.7
            if sigma < 0.001:
                break

    return best_a, best_score


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_bi_gaussian(n, rng):
    """Two Gaussian bumps at different positions."""
    t = np.arange(n, dtype=np.float64)
    c1 = float(rng.integers(0, n // 3))
    c2 = float(n - rng.integers(0, n // 3))
    w1 = float(rng.integers(n // 10, n // 4))
    w2 = float(rng.integers(n // 10, n // 4))
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    return np.maximum(0, a)


def initial_random_smooth(n, rng, n_bumps=4):
    a = np.zeros(n)
    t = np.arange(n, dtype=np.float64)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 12), max(2, n // 4)))
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def multi_stage_optimize(a_init, time_budget, rng, verbose=True):
    """Run multiple optimization stages on a single sequence."""
    start = time.monotonic()
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    # Stage 1: Mirror descent (bulk of time)
    md_time = time_budget * 0.4
    if time_budget > 5:
        a1, s1 = run_mirror_descent(a, md_time, lr_init=0.3, eps_init=10.0, verbose=verbose)
        if s1 < best_score:
            best_score = s1
            best_a = a1
            if verbose:
                print(f"      After MD: {best_score:.6f}")

    # Stage 2: L-BFGS refinement
    elapsed = time.monotonic() - start
    remaining = time_budget - elapsed
    if remaining > 5:
        a2, s2 = run_lbgfgs(best_a, remaining * 0.35, verbose=verbose)
        if s2 < best_score:
            best_score = s2
            best_a = a2
            if verbose:
                print(f"      After L-BFGS: {best_score:.6f}")

    # Stage 3: Simulated annealing
    elapsed = time.monotonic() - start
    remaining = time_budget - elapsed
    if remaining > 3:
        a3, s3 = simulated_annealing(best_a, remaining * 0.3, T_init=0.03, rng=rng, verbose=verbose)
        if s3 < best_score:
            best_score = s3
            best_a = a3
            if verbose:
                print(f"      After SA: {best_score:.6f}")

    # Stage 4: Local refinement
    elapsed = time.monotonic() - start
    remaining = time_budget - elapsed
    if remaining > 2:
        a4, s4 = local_refine(best_a, remaining * 0.9, rng=rng, verbose=verbose)
        if s4 < best_score:
            best_score = s4
            best_a = a4
            if verbose:
                print(f"      After LR: {best_score:.6f}")

    # Stage 5: Final MD with small lr
    elapsed = time.monotonic() - start
    remaining = time_budget - elapsed
    if remaining > 3:
        a5, s5 = run_mirror_descent(best_a, remaining * 0.9, lr_init=0.05, eps_init=2.0, verbose=verbose)
        if s5 < best_score:
            best_score = s5
            best_a = a5

    return best_a, best_score
