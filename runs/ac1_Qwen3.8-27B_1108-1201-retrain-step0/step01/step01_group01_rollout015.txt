    Compute the gradient of (1/tau)*log(sum_k exp(tau*b_k)) w.r.t. a.
    Uses FFT for speed.
    Returns (grad, lse_value).

    n = len(a)

"""
Hybrid log-sum-exp subgradient and LP polishing to minimize the normalized
autocorrelation peak of a non-negative sequence of length 6121.
Uses FFT-accelerated gradients, adaptive temperature scheduling, multiple
diverse initializations, and periodic LP-based direction finding for refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

N = 6121


def project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_max_b(a):
    """Compute max of full self-convolution."""
    return np.max(np.convolve(a, a))


def lse_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*b_k)) w.r.t. a.
    Uses correlate for speed. Returns (grad, lse_value, b_max).
    """
    n = len(a)
    b = np.convolve(a, a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    w = exp_shifted / sum_exp
    lse_val = np.log(sum_exp) / tau + b_max

    # grad_i = 2 * sum_{j=0}^{n-1} a_j * w_{i+j} = 2 * correlate(w, a, 'valid')[i]
    grad = 2.0 * np.correlate(w, a, mode='valid')

    return grad, lse_val, b_max


def lse_optimize(a_init, deadline, tau_max=500.0, tau_min=5.0,
                 eta_init=None, cool_rate=0.998, reheat_every=200,
                 check_every=10, patience=40):
    """
    Log-sum-exp subgradient optimization with adaptive temperature and step size.
    """
    n = len(a_init)
    a = a_init.copy()
    S = a.sum()
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n

    best = a.copy()
    best_score = 2.0 * n * compute_max_b(best)

    if eta_init is None:
        eta_init = 2.0 / np.sqrt(n)

    eta = eta_init
    tau = tau_max
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        grad, lse_val, b_max = lse_gradient(a, tau)

        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        a = a - eta * grad / gnorm
        a = project_simplex(a, 1.0)

        if it % check_every == 0:
            cur_score = 2.0 * n * compute_max_b(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve >= patience:
                eta *= cool_rate
                tau = max(tau_min, tau * 0.93)
                no_improve = 0

                # Reheat: occasionally jump back to explore
                if it % reheat_every < check_every and tau < tau_max * 0.3:
                    tau = min(tau_max, tau * 3.0)
                    eta = min(eta_init, eta * 2.0)

                if eta < 1e-12:
                    break

    return best, best_score


def lp_polish(f, deadline, n_constraints=150, max_iters=80):
    """
    LP-based polishing: iteratively find direction g via LP and line-search mix.
    """
    n = len(f)
    f = f.copy()
    S = f.sum()
    if S > 1e-10:
        f /= S

    best = f.copy()
    best_score = 2.0 * n * compute_max_b(best)

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        b = np.convolve(f, f)
        M = b.max()

        # Get top K peak locations
        K = min(n_constraints, len(b))
        top_idx = np.argpartition(b, -K)[-K:]
        # Sort by value descending for better LP
        top_idx = top_idx[np.argsort(b[top_idx])[::-1]]

        # Build LP constraint matrix (K x n) using COO
        rows = []
        cols = []
        vals = []
        for r, k in enumerate(top_idx):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            js = np.arange(j_lo, j_hi + 1)
            fidx = k - js
            rows.extend([r] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[fidx].tolist())

        A = coo_matrix((vals, (rows, cols)), shape=(K, n)).tocsr()
        b_ub = M * np.ones(K)
        c = -np.ones(n)

        start_time = time.monotonic()
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')

        if time.monotonic() - start_time > 30:
            print(f"  LP iter {it}: took too long, stopping LP phase")
            break

        if res.status != 0 or res.x is None:
            print(f"  LP iter {it}: infeasible (status={res.status})")
            break

        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            print(f"  LP iter {it}: zero solution")
            break
        g = g / g_sum

        # Line search on step size
        best_t = 0.0
        best_s = best_score
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            f_try = (1.0 - t) * f + t * g
            s_try = 2.0 * n * compute_max_b(f_try)
            if s_try < best_s - 1e-14:
                best_s = s_try
                best_t = t

        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g
            if best_s < best_score - 1e-12:
                best_score = best_s
                best = f.copy()
                print(f"  LP iter {it}: score={best_score:.8f}, t={best_t}")
        else:
            # Try very small steps
            improved = False
            for t in [0.005, 0.01, 0.02]:
                f_try = (1.0 - t) * f + t * g
                s_try = 2.0 * n * compute_max_b(f_try)
                if s_try < best_score - 1e-12:
                    f = f_try
                    best_score = s_try
                    best = f.copy()
                    improved = True
                    print(f"  LP iter {it}: small step score={best_score:.8f}, t={t}")
                    break
            if not improved:
                print(f"  LP iter {it}: no improvement, stopping")
                break

    return best, best_score


def make_init_two_level(n, eps):
    """Two-level anti-symmetric initialization."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = a.sum()
    return a / S


def make_init_smooth(n, freq, amp):
    """Smooth cosine initialization."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def make_init_multi_freq(n, rng):
    """Random multi-frequency initialization."""
    a = np.ones(n)
    n_freqs = int(rng.integers(2, 6))
    i = np.arange(n, dtype=np.float64)
    for _ in range(n_freqs):
        freq = int(rng.integers(1, max(2, n // 4)))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        a += amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.0)
    S = a.sum()
    return a / S


def make_init_triangular(n, width_ratio=0.5):
    """Triangular (tent) shaped initialization."""
    i = np.arange(n, dtype=np.float64)
    center = n / 2.0
    a = 1.0 - width_ratio * np.abs(i - center) / center
    a = np.maximum(a, 0.01)
    S = a.sum()
    return a / S
