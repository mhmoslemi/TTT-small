"""
Multi-strategy optimization for minimizing the peak autocorrelation of 
non-negative step functions. Uses FFT-based initialization, gradient 
descent with annealed soft-max, and targeted local search.
"""

import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s

def soft_max(x, tau):
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    e = np.exp(shifted)
    return x_max + tau * np.log(np.sum(e)), e / np.sum(e)

def grad_soft_max_conv(a, tau):
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max(c, tau)
    # Compute gradient: v[m] = (2/tau) * sum_j a[j] * w[m+j]
    # = (2/tau) * conv(a[::-1], w)[n-1 : 2n-1]
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = (2.0 / tau) * corr[n-1:2*n-1]
    return value, g, c

def optimize_adam(n, a_init, time_limit, rng, tau_start=None, tau_end=None):
    """Adam optimization with annealed soft-max on softmax-parameterized sequence."""
    S = np.sqrt(2.0 * n)

    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr0 = 3.0
    t0 = time.monotonic()
    step = 0
    best_a = a_init.copy()

def fft_phase_init(n, rng, n_iter=50):
    """Initialize using FFT phase randomization (Gerchberg-Saxton style)."""
    N = 2 * n - 1
    # Round to power of 2 for efficiency
    N_fft = 1
    while N_fft < N:
        N_fft <<= 1
    
    a = np.ones(n) / n  # start flat
    for _ in range(n_iter):
        # Zero-pad and FFT
        A = np.fft.fft(a, N_fft)
        # Flatten magnitude, keep phase
        phase = A / np.abs(A + 1e-10)
        # Inverse FFT with flat magnitude
        a_full = np.fft.ifft(phase).real
        # Take first n, clip, normalize
        a = np.clip(a_full[:n], 0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    
    return a

def random_init(n, rng, spread=0.5):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None) / np.sum(np.clip(a, 0.01, None))

def chirp_init(n, rng, alpha=0.5):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)

def multi_tone_init(n, rng):
    t = np.arange(n, dtype=np.float64)

def local_search(a_init, time_limit, rng, n_peak=5):
    """Targeted local search focusing on reducing convolution peaks."""
    current = a_init.copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.005
    cooling = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.4:
            # Peak-directed: reduce elements contributing to peak
            conv = fft_convolve(current, current)
            top_lags = np.argsort(conv)[-min(n_peak, len(conv)):]
            # For each top lag, find contributing elements
            contrib = np.zeros(n)
            for lag in top_lags:
                for i in range(n):
                    j = lag - i
                    if 0 <= j < n:
                        contrib[i] += current[i] * current[j]
            # Reduce top contributors, increase others
            n_top = max(2, n // 8)
            top_idx = np.argsort(contrib)[-n_top:]
            bot_idx = np.argsort(contrib)[:n_top]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.02, size=len(bot_idx)))
        elif r < 0.6:
            # Block perturbation
            n_blocks = rng.integers(1, 4)
            for _ in range(n_blocks):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start+width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.75:
            # Symmetry push
            target_sym = 0.5 * (current + current[::-1])
            candidate = current + 0.1 * (target_sym - current) + 0.01 * rng.standard_normal(n)
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0, 0.02, size=n))
        
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s > 1e-10:
            candidate = candidate / s * np.sum(current)  # preserve scale
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current = candidate
            current_score = new_score
        
        temp *= cooling
    
    return best, best_score

"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions. Combines FFT-phase initialization (Gerchberg-Saxton), gradient
descent with annealed soft-max on softmax-parameterized sequences, coarse-to-fine
upsampling, and targeted local search with convolution-peak-directed mutations.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def soft_max_and_weights(x, tau):
    """Compute soft-max value and normalized weights."""
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    return x_max + tau * np.log(s), e / s


def grad_soft_max_conv(a, tau):
    """Compute soft-max(conv(a,a)) and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_and_weights(c, tau)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = (2.0 / tau) * corr[n - 1:2 * n - 1]
    return value, g, c


def fft_phase_init(n, rng, n_iter=80, N_override=None):
    """Gerchberg-Saxton style initialization: flatten magnitude spectrum."""
    if N_override is None:
        N = 2 * n - 1
        N_fft = 1
        while N_fft < N:
            N_fft <<= 1
    else:
        N_fft = N_override

    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def chirp_init(n, rng, alpha=0.5):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def multi_tone_init(n, rng, n_tones=4):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(n_tones):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.4 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def random_init(n, rng, spread=0.4):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def upsample(a, new_n, rng):
    """Upsample a sequence to a longer length via interpolation + noise."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.02 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)


def optimize_gradient(n, a_init, time_limit, rng, tau_start_mult=40.0, tau_end_mult=0.3):
    """
    Adam gradient descent on softmax-parameterized sequence with annealed
    soft-max temperature. Normalizes to sum = sqrt(2n).
    """
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S
    theta = np.log(a / S + 1e-12)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr0 = 2.5
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)
    tau_start = tau_start_mult / n
    tau_end = tau_end_mult / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.6)

        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)

        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.5 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 300 == 0:
            a_check = softmax_normalize(theta) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
            if step % 1500 == 0:
                print(f"        step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")

    a_final = softmax_normalize(best_theta) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
    return a_final, best_score


def local_search(a_init, time_limit, rng, top_k=8):
    """
    Simulated annealing with convolution-peak-directed perturbations.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.004
    cooling = 0.9996
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        iter_count += 1

        if r < 0.35:
            # Peak-directed reduction
            conv = fft_convolve(current, current)
            n_top = min(top_k, len(conv))
            top_lags = np.argsort(conv)[-n_top:]
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            n_reduce = max(2, n // 10)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.06, 0.02, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.06, 0.02, size=len(bot_idx)))
        elif r < 0.5:
            # Adjacent swap-like: reduce one block, increase adjacent
            width = max(1, int(rng.integers(1, max(2, n // 20))))
            start = int(rng.integers(0, n - 2 * width))
            candidate[start:start + width] *= np.exp(rng.normal(-0.08, 0.02, size=width))
            candidate[start + width:start + 2 * width] *= np.exp(rng.normal(0.08, 0.02, size=width))
        elif r < 0.65:
            # Multi-block perturbation
            n_blocks = int(rng.integers(2, 5))
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 15))))
                bs = int(rng.integers(0, n - bw))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0.0, 0.05, size=bw))
        elif r < 0.8:
            # Partial symmetrization
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.15 * (sym - current)
            candidate += 0.01 * rng.standard_normal(n)
        elif r < 0.9:
            # Sparsification attempt: reduce a random subset significantly
            n_sparse = max(1, n // 5)
            sparse_idx = rng.choice(n, size=n_sparse, replace=False)
            candidate[sparse_idx] *= np.exp(rng.normal(-0.3, 0.1, size=n_sparse))
            dense_idx = np.setdiff1d(np.arange(n), sparse_idx)
            if len(dense_idx) > 0:
                n_inc = min(5, len(dense_idx))
                inc_idx = rng.choice(dense_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.3, 0.1, size=n_inc))
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0.0, 0.015, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current
        else:
            candidate = candidate / s * np.sum(current)

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 10:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score

        temp *= cooling

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for optimal non-negative step function minimizing
    2*n*max(conv)/sum^2. Uses FFT-phase init, gradient descent, and local search.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 25.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Phase 1: Generate diverse initial candidates
    print("\n=== Phase 1: Initialization ===")
    candidates = []

    lengths_to_try = [256, 384, 512, 768, 1024, 1536]

    for n in lengths_to_try:
        if time.monotonic() > deadline - 90:
            break
        inits = []

        # FFT phase init (multiple variants)
        a_fft1 = fft_phase_init(n, rng, n_iter=60)
        inits.append(('fft1', a_fft1))

        a_fft2 = fft_phase_init(n, rng, n_iter=100)
        inits.append(('fft2', a_fft2))

        # Chirp
        a_chirp = chirp_init(n, rng, alpha=0.3 + 0.4 * rng.random())
        inits.append(('chirp', a_chirp))

        # Multi-tone
        a_mt = multi_tone_init(n, rng, n_tones=3 + rng.integers(0, 3))
        inits.append(('multitone', a_mt))

        # Random
        a_rand = random_init(n, rng, spread=0.2 + 0.4 * rng.random())
        inits.append(('random', a_rand))

        for name, a in inits:
            score = local_score(a)
            print(f"  n={n}, {name}: score={score:.6f}")
            candidates.append((score, n, a.copy(), name))

    # Sort and keep top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"\n  Top 3: {[(f'{c[0]:.4f}', c[1], c[3]) for c in candidates[:3]]}")

    # Phase 2: Gradient optimization on best candidates
    print("\n=== Phase 2: Gradient optimization ===")
    n_cand = min(8, len(candidates))
    time_left = deadline - time.monotonic() - 150  # reserve for polish
    time_per = max(8.0, time_left / n_cand)

    # Reduce time per for larger n (slower iterations)
    optimized = []
    for i in range(n_cand):
        if time.monotonic() > deadline - 60:
            break
        score0, n, a, name = candidates[i]
        # Scale time by n (larger n = slower per iteration)
        t_alloc = min(time_per * 1.5, deadline - time.monotonic() - 30)
        t_alloc = t_alloc * min(1.0, 512.0 / n)
        t_alloc = max(5.0, t_alloc)

        print(f"  [{i+1}/{n_cand}] n={n}, init={name}, score0={score0:.6f}, t_alloc={t_alloc:.1f}s")
        a_opt, s_opt = optimize_gradient(n, a, t_alloc, rng)
        print(f"    -> score={s_opt:.6f}")

        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")

        optimized.append((s_opt, n, a_opt))

    # Phase 3: Coarse-to-fine: upsample best short sequences and optimize
    print("\n=== Phase 3: Coarse-to-fine upsample ===")
    if optimized:
        optimized.sort(key=lambda x: x[0])
        best_short = optimized[0]
        for target_n in [512, 1024, 2048]:
            if time.monotonic() > deadline - 60:
                break
            if len(best_short[2]) >= target_n:
                continue
            a_up = upsample(best_short[2], target_n, rng)
            t_alloc = min(60.0, deadline - time.monotonic() - 40)
            if t_alloc < 10:
                break
            print(f"  Upsample to n={target_n}, t_alloc={t_alloc:.1f}s")
            a_opt, s_opt = optimize_gradient(target_n, a_up, t_alloc, rng)
            print(f"    -> score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} (n={target_n}) ***")

    # Phase 4: Local search polish on best candidate
    print(f"\n=== Phase 4: Local search polish ===")
    if best_seq is not None:
        polish_time = min(200.0, max(15.0, deadline - time.monotonic() - 20))
        print(f"  Polishing n={len(best_seq)}, score={best_score:.6f}, time={polish_time:.1f}s")
        polished, p_score = local_search(best_seq, polish_time, rng, top_k=6)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")

    # Phase 5: If time remains, try a few more random restarts on best length
    remaining = deadline - time.monotonic() - 20
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Phase 5: Additional restarts ===")
        n_best = len(best_seq)
        n_restarts = min(3, int(remaining // 40))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break
            a_new = best_seq.copy() * np.exp(rng.normal(0, 0.1, size=n_best))
            a_new = np.clip(a_new, 0.01, None)
            t_alloc = min(30.0, (deadline - time.monotonic() - 20) / (n_restarts - r))
            a_opt, s_opt = optimize_gradient(n_best, a_new, t_alloc, rng)
            print(f"  Restart {r+1}: score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} ***")
                # Quick polish
                if deadline - time.monotonic() > 30:
                    p2, s2 = local_search(best_seq, min(30.0, deadline - time.monotonic() - 20), rng)
                    if s2 < best_score:
                        best_score = s2
                        best_seq = p2.copy()

    # Final: verify with evaluate_sequence
    print(f"\n{'='*50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
