    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return float('inf')
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize_nonneg(a):
    """Normalize a to be non-negative with sum 1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * (1.0 / len(a))
    return a / s


def init_raised_cosine(n):
    """Raised cosine window."""
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def init_gaussian(n, sigma_frac=0.3):
    """Gaussian window."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def init_flat(n):
    """Uniform (flat) window."""
    return np.ones(n, dtype=np.float64)


def init_blackman(n):
    """Blackman window."""
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def init_random_smooth(n, rng, n_bumps=3):
    """Random smooth initialization (sum of Gaussians)."""
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(2, n // 10), max(3, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    return normalize_nonneg(a)


def softmax_grad(a, t):
    """
    Compute gradient of t * logsumexp(conv(a,a) / t) w.r.t. a.
    a: length n, t: temperature
    Returns: gradient of length n
    """
    n = len(a)
    B = np.convolve(a, a)  # length 2n-1
    B_scaled = B / t
    B_scaled -= B_scaled.max()  # numerical stability
    exp_B = np.exp(B_scaled)
    p = exp_B / exp_B.sum()  # length 2n-1
    
    # Term 1: grad1[j] = sum_{i=0}^{n-1} a[i] * p[i+j]
    # Using sliding window view of p
    p_windows = sliding_window_view(p, n)  # shape (n, n)
    grad1 = p_windows @ a  # shape (n,)
    
    # Term 2: grad2[j] = sum_{i=0}^{n-1} a[i] * p[j-i] (valid indices only)
    # This is the standard convolution of a and p, first n elements
    conv_ap = np.convolve(a, p, 'full')  # length n + (2n-1) - 1 = 3n-2
    grad2 = conv_ap[:n]
    
    return grad1 + grad2


def softmax_gd(a, time_limit, t_start=None, t_end=None, verbose=True):
    """
    Log-softmax gradient descent with multiplicative updates.
    Minimizes t*logsumexp(conv(a,a)/t) with annealed temperature.
    """
    start_time = time.monotonic()
    a = normalize_nonneg(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)
    
    B = np.convolve(a, a)
    max_B = B.max()
    if t_start is None:
        t_start = max_B * 0.1
    if t_end is None:
        t_end = max_B * 0.001
    
    t = t_start
    eta = 0.5
    stagnant = 0
    it = 0
    
    while time.monotonic() - start_time < time_limit:
        it += 1
        
        grad = softmax_grad(a, t)
        
        # Backtracking line search
        improved = False
        current_score = score_seq(a)
        for trial in range(15):
            a_new = a * np.exp(-eta * grad)
            a_new = normalize_nonneg(a_new)
            new_score = score_seq(a_new)
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
            if new_score <= current_score - 1e-12:
                a = a_new
                improved = True
                stagnant = 0
                break
            eta *= 0.5
        
        if not improved:
            stagnant += 1
            if stagnant > 30:
                # Try perturbation
                a = normalize_nonneg(a * np.exp(np.random.default_rng(it).normal(0, 0.05, n)))
                stagnant = 0
                eta = 0.5
        
        # Anneal temperature
        if it < 100:
            t = t_start * (t_end / t_start) ** (it / 100.0)
        else:
            t = max(t * 0.98, t_end)
        
        # Reset eta periodically
        if it % 50 == 0:
            eta = 0.5
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    SGD iter {it}: score={best_score:.6f}, t={t:.6f}, eta={eta:.4f}, elapsed={elapsed:.0f}s")
    
    # Final check
    final_score = score_seq(best_a)
    if final_score < best_score:
        best_score = final_score
    
    if verbose:
        print(f"    SGD done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.0f}s")
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build sparse matrix for convolution with a."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def lp_step(a, M, timeout=30.0, relax=0.0):
    """
    Solve LP: maximize sum(g) s.t. conv(a,g) <= M*(1+relax), g >= 0.
    Returns g or None.
    """
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M * (1.0 + relax))
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit, verbose=True):
    """LP-based iterative refinement with adaptive relaxation and step size."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)
    stagnant = 0
    it = 0
    
    while time.monotonic() - start_time < time_limit:
        it += 1
        elapsed = time.monotonic() - start_time
        remaining = time_limit - elapsed
        if remaining < 5.0:
            break
        
        conv = np.convolve(a, a)
        M = conv.max()
        current_score = score_seq(a)
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 50:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break
        
        timeout = max(3.0, min(30.0, remaining * 0.2))
        
        # Try with increasing relaxation
        g = None
        for relax in [0.0, 0.01, 0.05]:
            g = lp_step(a, M, timeout=timeout, relax=relax)
            if g is not None and g.sum() > 1.0 + 1e-10:
                break
        
        if g is None or g.sum() <= 1.0 + 1e-10:
            # Try softmax GD for a few steps instead
            a_tmp, _ = softmax_gd(a, time_limit=min(30.0, remaining * 0.5), verbose=False)
            if score_seq(a_tmp) < best_score:
                best_score = score_seq(a_tmp)
                best_a = a_tmp.copy()
                a = a_tmp.copy()
            stagnant = 0
            continue
        
        g = g / g.sum()
        
        # Backtracking line search for step size
        improved = False
        for alpha in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = normalize_nonneg(a_new)
            new_score = score_seq(a_new)
            if new_score <= current_score + 1e-12:
                a = a_new
                improved = True
                break
        
        if not improved:
            # Try smaller steps
            for alpha in [0.01, 0.005]:
                a_new = (1.0 - alpha) * a + alpha * g
                a_new = normalize_nonneg(a_new)
                new_score = score_seq(a_new)
                if new_score <= current_score + 1e-12:
                    a = a_new
                    improved = True
                    break
        
        if verbose and it % 20 == 0:
            print(f"    LP iter {it}: score={score_seq(a):.6f}, best={best_score:.6f}, elapsed={elapsed:.0f}s")
    
    final_score = score_seq(best_a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.0f}s")
    
    return best_a, best_score


def local_search(a, time_limit, rng, verbose=True):
    """Local search with random perturbations."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)
    accept_count = 0
    
    while time.monotonic() - start_time < time_limit:
        # Random perturbation of a segment
        width = max(2, int(rng.integers(max(2, n // 100), n // 5)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.15, size=stop - start))
        
        # Occasionally enforce symmetry
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = normalize_nonneg(candidate)
        score = score_seq(candidate)
        
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 100 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")
    
    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.0f}s")
    return best_a, best_score
