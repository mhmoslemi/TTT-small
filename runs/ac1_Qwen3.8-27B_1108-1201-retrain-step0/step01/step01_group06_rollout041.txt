def auto_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
combined with mirror descent. The LP finds the 'largest' non-negative sequence
whose cross-convolution with the current sequence is bounded by the current
peak, providing a natural direction to reduce the auto-convolution peak.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_auto_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = fast_auto_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def solve_lp_direction(a, M):
    """
    Solve: max sum(b) s.t. cross_conv(a, b)[k] <= M for all k, b >= 0.
    Returns b (the solution).
    """
    n = len(a)
    num_constraints = 2 * n - 1
    
    # Objective: minimize -sum(b)
    c = -np.ones(n)
    
    # Constraint matrix: A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # (coefficient of b[j] in constraint k)
    A_ub = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        # b[j] appears in constraint k when 0 <= k-j < n
        # i.e., max(0, k-n+1) <= j <= min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[k - j]
    
    b_ub = np.full(num_constraints, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    else:
        # Fallback: return uniform
        return np.ones(n)

    best_a = a.copy()

                conv = conv_new
                best_obj = obj_new

            if obj < best_obj:

                best_a = a.copy()
            stagnation = 0
            t = t_init
