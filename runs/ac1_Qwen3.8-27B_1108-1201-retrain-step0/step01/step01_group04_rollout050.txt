def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)

    idx = np.arange(1, n+1)
    cond = u * idx > (cssum - 1.0)

    if s < 1e-300:

def conv_full(p):
    n = len(p)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))

def score_fn(p):
    n = len(p)
    if n == 0:
        return np.inf
    s = np.sum(p)
    if s < 1e-12:
        return np.inf
    c = conv_full(p)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (s * s))


def normalize(p):
    p = np.maximum(np.where(np.isfinite(p), p, 0.0), 0.0)
    s = p.sum()
    if s < 1e-12:
        return np.full(len(p), 1.0 / len(p))
    return p / s


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.where(np.isfinite(v), v, 0.0).copy()
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s

def shape_fourier(theta, n):
    """
    Build a normalized non-negative vector from Fourier coefficients + endpoint taper.
    theta = [a1, b1, a2, b2, ..., aK, bK, log_taper_left, log_taper_right]
    """
    K = (len(theta) - 2) // 2
    x = np.linspace(0.0, 1.0, n)
    logp = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a = theta[2 * k]
        b = theta[2 * k + 1]
        logp += a * np.cos(2.0 * np.pi * k * x) + b * np.sin(2.0 * np.pi * k * x)
    eps = 1e-4
    logp += theta[-2] * np.log(x + eps)
    logp += theta[-1] * np.log(1.0 - x + eps)
    if not np.all(np.isfinite(logp)):
        return np.full(n, 1.0 / n)
    logp = logp - np.max(logp)
    p = np.exp(logp)
    return normalize(p)
