```python
def lp_direction(a, M, top_k=50):
    """Solve the LP to get a direction that maximizes sum(g) subject to cross-correlation constraints."""
    n = len(a)
    c = fft_convolve_self(a)
    # Get top-K lags
    top_indices = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix A (top_k x n)
    # A[k, j] = a[k - j] if 0 <= k-j < n else 0
    A = np.zeros((top_k, n))
    for i, k in enumerate(top_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = a[k - hi:k - lo + 1][::-1]
    
    # Solve: max 1^T g s.t. A g <= M, g >= 0
    from scipy.optimize import linprog
    c_obj = -np.ones(n)  # minimize -sum(g)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(top_k, M), bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x
    return None
```

```python
"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences using:
1. Smooth log-sum-exp gradient with annealing (main optimization)
2. LP-based direction finding (periodic jumps to escape local minima)
3. Mirror descent for non-negativity
4. Multiple initializations and lengths
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def smooth_gradient_fft(a, c, tau):
    """Compute the smooth gradient of the log-sum-exp approximation using FFT.
    
    Returns grad of shape (n,) where grad_m = 2 * sum_j a_j * w_{m+j}
    with w_k = exp(tau * c_k) / sum_k exp(tau * c_k)
    """
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    # Compute w = softmax(tau * c)
    tc = tau * c
    tc -= tc.max()  # numerical stability
    w = np.exp(tc)
    w /= w.sum()
    
    # Compute grad_m = 2 * sum_j a_j * w_{m+j}
    # This is 2 * cross-correlation of a and w
    # In FFT: cross_corr = ifft(conj(fft(a, L)) * fft(w, L))
    A = fft(a, fsize)
    W = fft(w, fsize)
    cross_corr = np.real(ifft(np.conj(A) * W))
    
    return 2.0 * cross_corr[:n]
```

```python
"""
Minimize 2*n*max_conv/sum^2 using LP-based iterative optimization 
with smooth gradient fine-tuning and multiple initializations.

Key ideas:
- LP direction: solve max sum(g) s.t. a*g <= M, g>=0 (top-K constraints)
- Smooth gradient: log-sum-exp approximation of max for dense gradient
- Annealed tau for the smooth approximation
- Multiple initializations and lengths
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def smooth_grad_fft(a, c, tau):
    """Smooth gradient of log-sum-exp(tau * c) w.r.t. a, via FFT."""
    n = len(a)
    size = 2 * n - 1
    # Need L >= n + size - 1 = 3n - 2
    L = 1 << (3 * n - 2 - 1).bit_length()
    
    # w = softmax(tau * c)
    tc = tau * c
    tc -= tc.max()
    w = np.exp(tc)
    ws = w.sum()
    if ws <= 0:
        w = np.full(size, 1.0 / size)
    else:
        w /= ws
    
    # grad_m = 2 * sum_j a_j * w_{m+j}
    # Using: f[m] = (a_r * w)[m + n - 1] where a_r = reverse(a)
    a_r = a[::-1]
    Ar = fft(a_r, L)
    W = fft(w, L)
    conv = np.real(ifft(Ar * W))
    grad = 2.0 * conv[n - 1: 2 * n - 1]
    return grad


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def lp_direction(a, M, top_k):
    """Solve LP to find direction g maximizing sum(g) s.t. a*g <= M (top-K constraints)."""
    n = len(a)
    c = fft_convolve_self(a)
    top_idx = np.argsort(c)[::-1][:min(top_k, len(c))]
    
    A_ub = np.zeros((len(top_idx), n))
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_ub[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A_ub, b_ub=np.full(len(top_idx), M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        return res.x
    return None


def run_lp_loop(a_init, n, max_time, rng, top_k=100, t_blend=0.3):
    """Iterative LP-based optimization loop."""
    S = np.sqrt(2.0 * float(n))
    a = normalize_seq(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()
    iter_count = 0
    stuck_count = 0
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        M = c.max()
        
        if M < best_score:
            best_score = M
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Solve LP
        g = lp_direction(a, M, top_k)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum <= 1e-15:
            break
        g = g * (S / g_sum)
        
        # Blend
        a = (1.0 - t_blend) * a + t_blend * g
        
        # Occasionally try different blend factors
        if iter_count % 200 == 100:
            t_blend = 0.5
        elif iter_count % 200 == 150:
            t_blend = 0.1
        elif iter_count % 200 == 0:
            t_blend = 0.3
        
        # Perturbation to escape local minima
        if stuck_count > 50:
            noise = rng.normal(0, 0.01 * S / n, n)
            a = project_simplex(a + noise, S)
            stuck_count = 0
        
        iter_count += 1
    
    return best_a, best_score


def run_smooth_adam(a_init, n, max_time, rng, top_tau_start=1.0, top_tau_end=30.0):
    """Adam optimizer with smooth log-sum-exp gradient."""
    S = np.sqrt(2.0 * float(n))
    a = normalize_seq(a_init, n)
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.5 * S / n
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        # Anneal tau
        progress = min(1.0, float(t) / 5000.0)
        tau = top_tau_start * (top_tau_end / top_tau_start) ** progress
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
        
        grad = smooth_grad_fft(a, c, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        
        beta1_t = 1 - beta1 ** t
        beta2_t = 1 - beta2 ** t
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * (grad ** 2)
        
        lr_t = lr * (0.5 ** (float(t) / 3000.0))
        a = project_simplex(a - lr_t * (m_hat / beta1_t) / (np.sqrt(v_hat / beta2_t) + eps), S)
        
        if t % 5000 == 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.005 * S / n, n), S)
    
    return best_a, best_score


def make_random_init(n, rng):
    return rng.random(n)


def make_sinusoidal_init(n, rng):
    t = np.linspace(0, 1, n)
    base = 1.0
    for freq in rng.integers(1, 10, size=3):
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        base += amp * np.sin(2 * np.pi * freq * t + phase)
    return np.maximum(base, 0.1)


def make_bipolar_init(n, rng):
    """Sequence with two distinct levels."""
    base = rng.uniform(0.3, 1.0)
    other = rng.uniform(1.0, 3.0)
    mask = rng.random(n) < 0.5
    a = np.where(mask, base, other)
    return a
```

```python
"""
Minimize 2*n*max_conv/sum^2 using LP-based iterative optimization with
smooth log-sum-exp gradient fine-tuning and multiple initializations.

Key ideas:
- LP direction: solve max sum(g) s.t. a*g <= M (top-K active constraints), g>=0
- Blend update: a <- (1-t)*a + t*g, theoretically non-increasing for small t
- Smooth gradient: log-sum-exp approximation of max for dense gradient (fine-tuning)
- Annealed tau for the smooth approximation
- Multiple initializations and lengths to avoid local minima
- Adaptive perturbation to escape stagnation
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def smooth_grad_fft(a, c, tau):
    """Smooth gradient of log-sum-exp(tau*c) w.r.t. a, via FFT.
    Returns grad of shape (n,) where grad_m = 2 * sum_j a_j * w_{m+j}.
    """
    n = len(a)
    size = 2 * n - 1
    # Need L >= n + size - 1 = 3n - 2
    L = 1 << (3 * n - 2 - 1).bit_length()

    # w = softmax(tau * c) over the convolution lags
    tc = tau * c
    tc -= tc.max()
    w = np.exp(tc)
    ws = w.sum()
    if ws <= 1e-30:
        w = np.full(size, 1.0 / size)
    else:
        w /= ws

    # grad_m = 2 * sum_j a_j * w_{m+j}
    # Using reversal trick: f[m] = (reverse(a) * w)[m + n - 1]
    a_r = a[::-1]
    Ar = fft(a_r, L)
    W = fft(w, L)
    conv_full = np.real(ifft(Ar * W))
    grad = 2.0 * conv_full[n - 1: 2 * n - 1]
    return grad


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    return a * (S / s)


def lp_solve_direction(a, M, top_k):
    """Solve LP: max sum(g) s.t. a*g <= M (top-K lags), g >= 0.
    Returns optimal g or None.
    """
    n = len(a)
    c = fft_convolve_self(a)
    k_eff = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_eff]

    # Build constraint matrix: A[i, j] = a[top_idx[i] - j] if valid
    A_ub = np.zeros((k_eff, n), dtype=np.float64)
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A_ub[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(k_eff, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def run_lp_optimize(a_init, n, max_time, rng, top_k=200, blend=0.4):
    """Iterative LP-based optimization. Returns (best_a, best_maxc)."""
    S = np.sqrt(2.0 * float(n))
    a = normalize_seq(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()
    iters = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        M = c.max()

        if M < best_score - 1e-12:
            best_score = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodically change blend factor
        if iters % 300 == 0:
            blend = rng.choice([0.2, 0.3, 0.4, 0.5, 0.7])
        if iters % 500 == 0:
            top_k = rng.choice([100, 200, 300, 500])

        # Solve LP for direction
        g = lp_solve_direction(a, M, top_k)
        if g is None:
            # Fallback: use smooth gradient step
            grad = smooth_grad_fft(a, c, 5.0)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            a = project_simplex(a - 0.01 * S / n * grad / gn, S)
            iters += 1
            continue

        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g * (S / g_sum)

        # Check if g is too similar to a (stagnation)
        cos_sim = np.dot(g, a) / (np.linalg.norm(g) * np.linalg.norm(a) + 1e-15)
        if cos_sim > 0.9999 and no_improve > 10:
            # Perturb
            a = project_simplex(a + rng.normal(0, 0.02 * S / np.sqrt(n), n), S)
            no_improve = 0
            iters += 1
            continue

        # Blend update
        a = (1.0 - blend) * a + blend * g
        a = np.maximum(a, 0.0)
        s_new = a.sum()
        if s_new < 1e-10:
            a = np.full(n, S / n)
        else:
            a = a * (S / s_new)

        # Periodic perturbation
        if no_improve > 200 and iters % 50 == 0:
            a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)

        iters += 1

    return best_a, best_score


def run_smooth_adam(a_init, n, max_time, rng, tau_start=2.0, tau_end=40.0):
    """Adam optimizer with smooth log-sum-exp gradient. Returns (best_a, best_maxc)."""
    S = np.sqrt(2.0 * float(n))
    a = normalize_seq(a_init, n)
    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    base_lr = 0.3 * S / n

    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, float(t) / 8000.0)
        tau = tau_start * (tau_end / tau_start) ** progress

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        grad = smooth_grad_fft(a, c, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        b1t = 1.0 - beta1 ** t
        b2t = 1.0 - beta2 ** t
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad ** 2)

        lr_t = base_lr * (0.7 ** (float(t) / 5000.0))
        a = project_simplex(a - lr_t * (m_buf / b1t) / (np.sqrt(v_buf / b2t) + eps), S)

        if t % 10000 == 0 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)

    return best_a, best_score


def run_mirror_descent(a_init, n, max_time, rng, eta_start=0.1):
    """Mirror descent (multiplicative weights) with smooth gradient."""
    S = np.sqrt(2.0 * float(n))
    a = normalize_seq(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()
    t = 0

    while time.monotonic() - start < max_time:
        t += 1
        tau = 2.0 + 3.0 * min(1.0, float(t) / 5000.0)
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        grad = smooth_grad_fft(a, c, tau)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break
        grad = grad / gmax

        eta = eta_start * max(0.01, 1.0 / (1.0 + 0.002 * t))
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a = np.full(n, S / n)
        else:
            a = a_new * (S / s)

    return best_a, best_score


def make_random_init(n, rng):
    """Random non-negative initialization."""
    return rng.random(n) + 0.1


def make_sinusoidal_init(n, rng):
    """Smooth sinusoidal initialization."""
    t = np.linspace(0, 1, n)
    base = np.ones(n)
    n_freqs = rng.integers(2, 6)
    for _ in range(n_freqs):
        freq = rng.integers(1, 12)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        base += amp * np.sin(2.0 * np.pi * freq * t + phase)
    return np.maximum(base, 0.05)


def make_two_level_init(n, rng):
    """Two-level (binary-like) initialization."""
    lo = rng.uniform(0.1, 0.5)
    hi = rng.uniform(1.0, 2.5)
    mask = rng.random(n) < rng.uniform(0.3, 0.7)
    return np.where(mask, hi, lo)


def make_tapered_init(n, rng):
    """Tapered (raised-cosine-like) initialization with random modulation."""
    t = np.linspace(0, 1, n)
    taper = 0.5 + 0.5 * np.cos(np.pi * (2 * t - 1))
    mod = 1.0 + rng.uniform(0.2, 0.5) * np.sin(2 * np.pi * rng.integers(2, 8) * t)
    return np.maximum(taper * mod, 0.01)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    # Authoritative verification
    try:
        ver = evaluate_sequence(global_a.tolist())
        if np.isfinite(ver):
            print("Final evaluate_sequence: %.6f" % ver)
        else:
            print("Final evaluate_sequence: inf (unexpected)")
    except Exception as e:
        print("Verification error: %s" % str(e))

    return [float(x) for x in global_a]
```
