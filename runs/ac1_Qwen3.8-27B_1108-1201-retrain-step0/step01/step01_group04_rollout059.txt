"""
LP best-response search for minimizing max self-convolution of nonnegative step functions.
Key ideas: (1) iterated LP best-response with full constraint set, (2) global shape init via
parametric families, (3) running-average tracking for the "dual" estimate, (4) local gradient
polish near the end. The LP directly optimizes the true (non-smooth) objective via best response.

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix


def _fft_conv(p, n):
    """Full convolution p*p, returns array of length 2n-1."""
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def _score(p):
    """Score = 2n * max(p*p) / (sum p)^2."""
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if s < 1e-12:
        return np.inf
    c = _fft_conv(p, n)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def _normalize(p):
    """Normalize to sum=1, non-negative."""
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s < 1e-12:
        return np.full(p.size, 1.0 / max(1, p.size))
    return p / s


def _build_conv_matrix(a, n):
    """Build sparse matrix A (2n-1 x n) where A[m,k] = a[m-k] if 0<=m-k<=n-1 else 0."""
    # For each column k (variable g[k]), it appears in rows m where 0<=m-k<=n-1,
    # i.e., m in [k, k+n-1], with coefficient a[m-k] = a[0], a[1], ..., a[n-1].
    k_vals = np.repeat(np.arange(n), n)
    m_vals = np.concatenate([np.arange(k, k + n) for k in range(n)])
    a_vals = np.tile(a, n)
    A = csr_matrix((a_vals, (m_vals, k_vals)), shape=(2 * n - 1, n))
    return A


def _solve_lp_best_response(a, n, time_limit=10.0):
    """Solve LP: min t s.t. A*g <= t, sum g = 1, g >= 0. Return g and t."""
    A = _build_conv_matrix(a, n)
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t -> c = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    # Inequality constraints: A*g - t <= 0 -> [A, -1] * [g; t] <= 0
    # Build [A, -1] as sparse
    ones_col = np.ones(2 * n - 1)
    A_ub = csr_matrix((np.concatenate([A.data, -ones_col]),
                       (np.concatenate([A.row.indices, np.full(2 * n - 1, -1)]),
                        np.concatenate([A.row.indices, np.arange(2 * n - 1) + A.row.indptr[-1]])),
                       shape=A.shape))
    # Simpler: build manually
    A_ub_dense_parts = []
    row_dirs = []
    col_dirs = []
    # A part
    A_coo = A.tocoo()
    row_dirs = [A_coo.row]
    col_dirs = [A_coo.col]
    data_parts = [A_coo.data]
    # -1 column for t
    row_t = np.arange(2 * n - 1)
    col_t = np.full(2 * n - 1, n)  # t is variable index n
    data_t = -np.ones(2 * n - 1)
    row_dirs.append(row_t)
    col_dirs.append(col_t)
    data_parts.append(data_t)
    
    A_ub = csr_matrix((np.concatenate(data_parts), (np.concatenate(row_dirs), np.concatenate(col_dirs))),
                      shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum g = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, t free
    
    start = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max(1.0, time_limit), 'disp': False})
    except Exception:
        return None, None, None
    
    if res is None or not res.success:
        return None, None, None
    
    g = res.x[:n]
    t_val = float(res.x[n])
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s < 1e-12:
        return None, None, None
    g = g / s
    return g, t_val, res

    n = len(p)
    c = _fft_conv(p, n)

    if cmax <= 0:
        return cmax, np.zeros(n)
    T = T_factor * cmax
    if T < 1e-300:
        T = 1e-300
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return cmax, np.zeros(n)
    w = e / s
    # Gradient: g_j = 2 * sum_m w_m * a[m-j] (valid m-j)
    # = 2 * (w_reversed * a)_j appropriately sliced
    L = next_fast_len(3 * n)
    Fp = fft(p, L)
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    # h[k] = sum_j p[j] * w_rev[k-j] = sum_j p[j] * w[2n-2-j-(k-j)] ... 
    # Actually let me use the direct formula:
    # grad_j = 2 * sum_m w_m * p[m-j] for valid m-j in [0,n-1]
    # This is 2 * (w * rev(p))_{j} where * is convolution, sliced appropriately.
    # Let's just compute it via the convolution matrix approach for clarity.
    # For efficiency, use FFT:
    # sum_m w_m * p[m-j] = (w * p_flipped)_j where p_flipped[k] = p[n-1-k]
    # Actually: sum_m w_m * p[m-j] where 0<=m-j<=n-1
    # Let's define q[k] = p[n-1-k] (reversed p)
    # sum_m w_m * p[m-j] = sum_m w_m * q[n-1-m+j]
    # This is the cross-correlation of w and q at shift (n-1-j).
    # = (w * q_rev)_{n-1-j} where q_rev = q (since we're doing correlation)
    # Simpler: use the FFT approach from the original code.
    # grad_j = 2 * sum_m w_m * p[m-j] for valid (m-j) in [0, n-1]
    # Let r[k] = p[k]. Then sum_m w_m * r[m-j] is the cross-correlation of w and r at lag j.
    # Cross-correlation: (w ⋆ r)[j] = sum_m w[m] * r[m+j]... no.
    # Let's just do it directly for correctness (O(n^2) but n=2048 so ~4M ops, fine).
    # Actually for n=2048, that's too slow per iteration. Let me use FFT.
    # sum_m w_m * p[m-j]: let's substitute l = m-j, m = l+j:
    # = sum_l w[l+j] * p[l] for l in [0,n-1] and l+j in [0, 2n-2]
    # This is the cross-correlation of p and w at shift j.
    # = sum_l p[l] * w[l+j]
    # Using FFT: cross_correlation(p, w)[j] = ifft(conj(fft(w)) * fft(p)) ... 
    # Actually: (f ⋆ g)[k] = sum_n f[n] * g[n+k] = ifft(FFT(f) * conj(FFT(g)))[k]
    # Hmm, let me just use np.convolve for the gradient (slower but correct for moderate n).
    # For n=2048, np.convolve is O(n^2) in worst case but with the FFT-based implementation it's O(n log n).
    
    # Direct FFT computation:
    # grad_j = 2 * sum_m w[m] * p[m-j] where 0<=m-j<=n-1, i.e., j<=m<=j+n-1
    # = 2 * sum_{l=0}^{n-1} p[l] * w[l+j] (substituting l = m-j, m = l+j)
    # = 2 * (w * p_flip)_{n-1-j} ... 
    # Let me just compute it as: convolve(w, p)[n-1+j] for j=0,...,n-1? 
    # convolve(w, p)[k] = sum_i w[i] * p[k-i]
    # We want sum_m w[m] * p[m-j] = sum_i w[i] * p[i-j] (setting i=m)
    # convolve(w, p)[k] = sum_i w[i] * p[k-i]
    # Set k-i = i-j -> k = 2i-j... this doesn't work simply.
    
    # Let me use the approach: define q[k] = p[n-1-k] (reverse of p)
    # sum_m w[m] * p[m-j] = sum_m w[m] * q[n-1-m+j]
    # Let s = n-1-m+j, m = n-1+j-s
    # = sum_s w[n-1+j-s] * q[s]
    # = (w_rev * q)[n-1+j] where w_rev[k] = w[n-1-k]... 
    # This is getting confusing. Let me just use the original code's approach.
    
    # From the original code:
    # Fw = fft(w[::-1], L)  # FFT of reversed w
    # h = ifft(Fp * Fw)
    # G = h[2*n-2 : n-2 : -1]  # this gives n elements
    # This computes: h[k] = sum_j p[j] * w_rev[k-j] = sum_j p[j] * w[n-1-(k-j)] = sum_j p[j] * w[n-1-k+j]
    # G[n-1-j] = h[2n-2-j] = sum_j' p[j'] * w[n-1-(2n-2-j')+j'] = sum_j' p[j'] * w[1-n+j'+j'-... ]
    
    # I'll just use the formula: grad_j = 2 * (w * p_reversed)[j + n - 1] 
    # where * is linear convolution. Let me verify:
    # (w * p_r)[k] = sum_i w[i] * p_r[k-i] = sum_i w[i] * p[n-1-(k-i)] = sum_i w[i] * p[n-1-k+i]
    # We want: grad_j/2 = sum_m w[m] * p[m-j]
    # Set k = j + n - 1: (w * p_r)[j+n-1] = sum_i w[i] * p[n-1-j-n+1+i] = sum_i w[i] * p[i-j]
    # = sum_i w[i] * p[i-j] which is what we want (with i playing the role of m)!
    # So grad_j = 2 * (w * p_reversed)[j + n - 1]
    
    p_rev = p[::-1]
    L2 = next_fast_len(n + n - 1 + n - 1)  # length of w * p_rev is (2n-1) + n - 1 = 3n-2
    Fw2 = fft(w, L2)
    Fp2 = fft(p_rev, L2)
    conv_wp = np.real(ifft(Fw2 * Fp2))[:3*n-2]
    grad = 2.0 * conv_wp[n-1:2*n-1]
    grad = np.maximum(grad, 0.0)
    surr = cmax + T * np.log(s)
    return surr, grad


def _local_gradient_optimize(p0, rng, time_limit, T_factor=0.03):
    """Local optimization using smoothed-max gradient on the simplex."""
    p = _normalize(p0.copy())
    n = len(p)
    best_p = p.copy()
    best_score = _score(best_p)
    current_score = best_score
    prev_p = p.copy()
    
    end = time.monotonic() + time_limit
    since_improve = 0
    T_f = T_factor
    iter_no = 0
    
    while time.monotonic() < end:
        iter_no += 1
        surr, grad = _surrogate_score(p, T_f)
        if not np.all(np.isfinite(grad)):
            break
        
        gabs = float(np.max(np.abs(grad)))
        if gabs < 1e-300:
            break
        
        p_scale = max(1.0 / n, 0.05 * float(np.max(p)), 1e-12)
        base_lr = p_scale / gabs
        
        accepted = False
        lrs = (0.3 * base_lr, 0.1 * base_lr, 0.03 * base_lr, 0.005 * base_lr, 0.001 * base_lr)
        
        for lr in lrs:
            lr = min(max(lr, 1e-10), 5.0)
            v = p - lr * grad
            if iter_no > 50:
                v = v + 0.3 * (p - prev_p)
            pt = _normalize(v)
            sc = _score(pt)
            if np.isfinite(sc) and sc < current_score:
                if sc < best_score:
                    best_score = sc
                    best_p = pt.copy()
                    since_improve = 0
                else:
                    since_improve += 1
                current_score = sc
                prev_p = p
                p = pt
                accepted = True
                break
        
        if not accepted:
            # Random perturbation
            sigma = 0.5 / n if since_improve > 100 else 0.1 / n
            pt = _normalize(p + rng.normal(0, sigma, n))
            sc = _score(pt)
            if np.isfinite(sc) and sc < current_score:
                if sc < best_score:
                    best_score = sc
                    best_p = pt.copy()
                    since_improve = 0
                else:
                    since_improve += 1
                current_score = sc
                prev_p = p
                p = pt
            else:
                since_improve += 1
                T_f = max(0.005, T_f * 0.95)
                if since_improve > 300:
                    p = _normalize(0.7 * best_p + 0.3 * p + rng.normal(0, 0.3/n, n))
                    current_score = _score(p)
                    prev_p = p
                    since_improve = 0
                    T_f = 0.05
        
        if iter_no % 500 == 0:
            print("  grad_opt iter=%d best=%.8f cur=%.8f T=%.4f" % (iter_no, best_score, current_score, T_f), flush=True)
        
        if best_score < 1.500 and since_improve > 5000:
            break
    
    return best_p, best_score


def _make_geometric(n, r):
    """Geometric sequence a[k] = (1-r)r^k / (1-r^n), normalized."""
    if abs(r - 1.0) < 1e-10:
        return np.full(n, 1.0/n)
    p = (1.0 - r) * np.power(r, np.arange(n)) / (1.0 - r**n)
    return _normalize(p)


def _make_power(n, alpha):
    """a[k] proportional to (k+1)^(-alpha)."""
    k = np.arange(1, n+1, dtype=np.float64)
    p = np.power(k, -alpha)
    return _normalize(p)


def _make_bimodal(n, peak_width):
    """Two peaks at the edges, dip in the middle."""
    x = np.linspace(0, 1, n)
    center = 0.5
    p = np.exp(-((x - center) ** 2) / (2 * 0.15**2))
    p = 1.0 - 0.7 * p  # dip in middle
    return _normalize(p)


def _make_hann(n):
    """Hann window shape."""
    k = np.arange(n)
    p = 0.5 * (1 + np.cos(2 * np.pi * k / (n - 1)))
    p = np.maximum(p, 1e-6)
    return _normalize(p)


def _make_chebyshev_taper(n, coeffs):
    """Chebyshev polynomial + log-taper parameterization."""
    x = np.linspace(0, 1, n)
    z = 2 * x - 1
    D = len(coeffs) - 2
    logp = np.zeros(n)
    if D >= 1:
        T_nm = np.ones(n)
        T_n = z.copy()
        logp += coeffs[0] * T_n
        for d in range(2, D + 1):
            T_np = 2 * z * T_n - T_nm
            logp += coeffs[d-1] * T_np
            T_nm = T_n
            T_n = T_np
    eps = 1e-4
    logp += coeffs[D] * np.log(x + eps)
    logp += coeffs[D+1] * np.log(1 - x + eps)
    if not np.all(np.isfinite(logp)):
        return np.full(n, 1.0/n)
    logp -= np.max(logp)
    p = np.exp(logp)
    return _normalize(p)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for nonnegative step function with low evaluate_sequence score."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 15.0)
    
    # Get parent sequence if available
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and parent_arr.sum() > 0:
            parent = _normalize(parent_arr)
    except Exception:
        pass
    
    # Primary sequence length
    n = 2048
    
    # Also try the parent length if it's reasonable
    n_parent = 0
    if parent is not None:
        n_parent = len(parent)
    
    best_p = None
    best_score = np.inf
    
    def try_record(p, sc=None):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0 or not np.all(np.isfinite(p)):
            return
        p = _normalize(p)
        if sc is None or not np.isfinite(sc):
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("NEW BEST: score=%.8f at t=%.0fs" % (best_score, time.monotonic() - start_time), flush=True)
    
    print("=== Starting LP best-response search, n=%d ===" % n, flush=True)
    
    # Phase 1: Generate and evaluate initial candidates
    print("Phase 1: Initial candidates", flush=True)
    candidates = []
    
    # Geometric sequences (both directions)
    for r in (0.98, 0.99, 0.995, 0.999, 0.97, 0.95, 0.9, 0.85, 0.8, 0.7):
        p = _make_geometric(n, r)
        candidates.append(p)
        p = _make_geometric(n, r)[::-1]
        candidates.append(p)
    
    # Power law
    for alpha in (0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        candidates.append(_make_power(n, alpha))
        candidates.append(_make_power(n, alpha)[::-1])
    
    # Bimodal
    candidates.append(_make_bimodal(n, 0.15))
    candidates.append(_make_hann(n))
    
    # Uniform
    candidates.append(np.full(n, 1.0/n))
    
    # Log-taper (geometric-like but smoother at edges)
    for a_taper in (0.5, 1.0, 2.0, 3.0, 5.0):
        x = np.linspace(0, 1, n)
        logp = -a_taper * x  # decreasing
        logp -= logp.max()
        candidates.append(_normalize(np.exp(logp)))
        logp = -a_taper * (1-x)  # increasing
        logp -= logp.max()
        candidates.append(_normalize(np.exp(logp)))
    
    # Two-sided exponential (bimodal)
    for a_exp in (1.0, 2.0, 3.0, 5.0, 8.0, 12.0, 20.0):
        x = np.linspace(0, 1, n)
        logp = -a_exp * (x - 0.5)**2 / 0.05
        logp -= logp.max()
        candidates.append(_normalize(np.exp(logp)))
    
    # Product of two geometric (skew)
    for a1, a2 in ((2.0, 0.5), (3.0, 1.0), (5.0, 2.0), (8.0, 3.0), (12.0, 5.0), (20.0, 8.0)):
        x = np.linspace(0, 1, n)
        logp = -a1 * x - a2 * (1-x)
        logp -= logp.max()
        candidates.append(_normalize(np.exp(logp)))
    
    # Cosine-modulated
    for f, amp in ((2, 0.3), (3, 0.3), (5, 0.2), (10, 0.15), (20, 0.1)):
        x = np.linspace(0, 1, n)
        p = 1.0 + amp * np.cos(2 * np.pi * f * x)
        p = np.maximum(p, 0.01)
        candidates.append(_normalize(p))
    
    # Chebyshev parametric (random inits)
    for _ in range(20):
        D = 6
        coeffs = rng.uniform(-2, 2, D + 2)
        candidates.append(_make_chebyshev_taper(n, coeffs))
    
    # Score all candidates
    scored = []
    for p in candidates:
        p = _normalize(p)
        sc = _score(p)
        if np.isfinite(sc):
            scored.append((sc, p))
    
    scored.sort(key=lambda x: x[0])
    print("Top 5 initial: " + ", ".join("%.4f" % s for s, _ in scored[:5]), flush=True)
    
    # Use top few as LP starting points
    for sc, p in scored[:8]:
        try_record(p, sc)
    
    # If we have a parent, try to extend it
    if parent is not None and n_parent != n:
        # Interpolate parent to length n
        from scipy.interpolate import interp1d
        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, n)
        f = interp1d(x_old, parent, kind='cubic', fill_value='extrapolate')
        p_interp = _normalize(f(x_new))
        try_record(p_interp)
    
    # Phase 2: LP best-response iterations
    print("Phase 2: LP best-response iterations", flush=True)
    
    # Select top 4 starting points for LP
    lp_starts = [(sc, p) for sc, p in scored[:4]]
    
    # Time allocation for LP
    lp_total_time = max(60.0, (deadline - time.monotonic()) * 0.65)
    time_per_lp = lp_total_time / len(lp_starts)
    
    for idx, (sc_init, p_init) in enumerate(lp_starts):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        lp_time = min(time_per_lp, remaining * 0.8)
        print("LP start %d: initial score=%.6f, time budget=%.0fs" % (idx, sc_init, lp_time), flush=True)
        _run_lp_loop(p_init, n, rng, lp_time, try_record, start_time)
    
    # Phase 3: Refine the best found with more LP iterations
    if best_p is not None and time.monotonic() < deadline - 120:
        print("Phase 3: Refining best (score=%.6f)" % best_score, flush=True)
        refine_time = min(600.0, deadline - time.monotonic() - 60)
        _run_lp_loop(best_p, n, rng, refine_time, try_record, start_time)
    
    # Phase 4: Local gradient polish
    if best_p is not None and time.monotonic() < deadline - 30:
        polish_time = max(10.0, deadline - time.monotonic() - 10)
        print("Phase 4: Gradient polish, time=%.0fs" % polish_time, flush=True)
        p_pol, sc_pol = _local_gradient_optimize(best_p, rng, polish_time)
        try_record(p_pol, sc_pol)
    
    # Ensure valid output
    if best_p is None:
        best_p = np.full(n, 1.0/n)
    
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s < 0.01:
        best_p = np.full(n, 1.0/n)
    
    final_score = _score(best_p)
    print("FINAL: score=%.8f, n=%d" % (final_score, len(best_p)), flush=True)
    
    return [float(x) for x in best_p]


def _run_lp_loop(p0, n, rng, time_limit, try_record, start_time):
    """Run LP best-response iterations."""
    p = _normalize(p0.copy())
    n = len(p)
    current_score = _score(p)
    best_local_p = p.copy()
    best_local_score = current_score
    
    # Running average (for the "dual" estimate)
    running_sum = p.copy().astype(np.float128) if hasattr(np, 'float128') else p.copy().astype(np.float64)
    count = 1
    
    end = time.monotonic() + time_limit
    iter_no = 0
    
    print("  LP: n=%d, initial=%.6f" % (n, current_score), flush=True)
    
    while time.monotonic() < end:
        iter_no += 1
        remaining = end - time.monotonic()
        if remaining < 3:
            break
        
        # Solve LP
        lp_time = min(20.0, remaining * 0.5)
        g, t_val, res = _solve_lp_best_response(p, n, lp_time)
        
        if g is None:
            # LP failed, try smaller problem or break
            if iter_no > 5:
                break
            continue
        
        g = _normalize(g)
        
        # Compute score of g
        sc_g = _score(g)
        
        # Try mixing
        best_mix_p = g
        best_mix_score = sc_g
        
        for alpha in (1.0, 0.8, 0.6, 0.4, 0.2, 0.1):
            mixed = _normalize((1 - alpha) * p + alpha * g)
            sc_mix = _score(mixed)
            if sc_mix < best_mix_score:
                best_mix_score = sc_mix
                best_mix_p = mixed
        
        # Update running average
        running_sum += best_mix_p.astype(running_sum.dtype)
        count += 1
        
        # Check running average
        if count >= 10 and count % 10 == 0:
            p_avg = _normalize(np.array(running_sum, dtype=np.float64))
            sc_avg = _score(p_avg)
            if sc_avg < best_local_score:
                best_local_score = sc_avg
                best_local_p = p_avg.copy()
        
        # Update current p
        p = best_mix_p
        current_score = best_mix_score
        
        if current_score < best_local_score:
            best_local_score = current_score
            best_local_p = p.copy()
        
        # Record
        try_record(best_local_p, best_local_score)
        
        if iter_no % 20 == 0 or iter_no < 5:
            print("  LP iter %d: curr=%.6f best=%.6f avg_count=%d t_lp=%.4f time=%.0fs remaining=%.0fs" % (
                iter_no, current_score, best_local_score, count, t_val,
                time.monotonic() - start_time, remaining), flush=True)
        
        # Check convergence
        if best_local_score < 1.501 and iter_no > 50:
            # Do a few more iterations then stop
            if iter_no > 100:
                break
        
        if best_local_score < 1.498:
            break
    
    # Final record
    try_record(best_local_p, best_local_score)
    print("  LP done: best=%.6f after %d iterations" % (best_local_score, iter_no), flush=True)
    return best_local_p, best_local_score
