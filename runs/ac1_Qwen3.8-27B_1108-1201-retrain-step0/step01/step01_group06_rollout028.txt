        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Subgradient: sg[j] = sum_k w[k] * a[k-j]
        sg = np.zeros(n)
        thresh = w.max() * 0.002
        active_idx = np.where(w > thresh)[0]
        
        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]
        
        # Multiplicative update (mirror descent step)
        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.ones(n) / n
        else:
            a = a / a_sum
        
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-8:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
            stall_count = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            # Restart strategy
            stall_count += 1
            patience = 0
            
            if stall_count > 5:
                # Add perturbation and reset
                a = best_a + np.random.normal(0, 0.05, n)
                a = np.maximum(a, 0)
                a_sum = a.sum()
                if a_sum > 1e-12:
                    a /= a_sum
                else:
                    a = best_a.copy()
                conv = fast_convolve(a)
                eta = 0.4
                t = np.clip(t * 0.5, 5.0, 300.0)
                max_patience = 80
                stall_count = 0
            else:
                eta *= 0.65
                if eta < 0.008:
                    eta = 0.35
                    t = np.clip(t * 1.4, 5.0, 500.0)
                    max_patience = max(60, max_patience - 10)
                else:
                    t = np.clip(t * 1.15, 5.0, 400.0)
    
    return best_a, best_obj


def generate_diverse_init(n, seed, method='flat'):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    
    if method == 'flat':
        return flat_spectrum_init(n, gamma=0.3, n_iter=300, seed=seed)
    elif method == 'flat_clip':
        return flat_spectrum_clipped(n, clip_level=2.5, n_iter=300, seed=seed)
    elif method == 'gauss':
        for sf in [0.2, 0.3, 0.4]:
            g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
            g /= g.sum()
            if rng.random() > 0.5:
                return g
        return np.exp(-0.5 * ((x - n * 0.5) / (0.3 * n)) ** 2)
    elif method == 'raised_cos':
        a = (1 + np.cos(2 * np.pi * x / n)) / 2
        a += 1e-6
        a /= a.sum()
        return a
    elif method == 'cos_power':
        alpha = rng.choice([1.5, 2.0, 3.0, 4.0])
        a = np.cos(np.pi * x / (n - 1)) ** (2 * alpha)
        a = np.maximum(a, 0)
        a += 1e-8
        a /= a.sum()
        return a
    elif method == 'ramp_asym':
        a = (x + 1) ** rng.choice([1.0, 1.5, 2.0])
        a /= a.sum()
        return a
    elif method == 'bimodal':
        a = np.zeros(n)
        w = max(2, n // 6)
        a[:w] = 1.0
        a[-w:] = 1.0
        a = a + rng.normal(0, 0.1, n)
        a = np.maximum(a, 0)
        a /= a.sum()
        return a
    else:  # random
        a = rng.dirichlet(np.ones(n) * 0.5)
        return a


def run_optimization(seed, budget_s):
    """Main optimization driver with multiple phases."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Try to access parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass
    
    print(f"Starting optimization with budget {budget_s}s")
    if parent is not None:
        print(f"Parent sequence: n={parent.size}, score={evaluate_sequence(parent.tolist()):.6f}")
    
    # Phase 1: Flat-spectrum initialization + quick mirror descent
    print("\n=== Phase 1: Flat-spectrum init + quick optimization ===")
    lengths = [64, 128, 128, 128, 256, 256, 512]
    methods = ['flat', 'flat_clip', 'raised_cos', 'cos_power', 'flat', 'gauss', 'flat']
    
    phase1_deadline = time.monotonic() + min(600.0, (deadline - time.monotonic()) * 0.3)
    results = []
    
    for i, (n, method) in enumerate(zip(lengths, methods)):
        if time.monotonic() > phase1_deadline - 30:
            break
        
        a0 = generate_diverse_init(n, seed + i * 100, method)
        t_alloc = min(30.0, phase1_deadline - time.monotonic() - 10)
        if t_alloc < 5:
            break
        
        a_opt, obj_norm = mirror_descent_phase(a0, time_limit=t_alloc, t_param=40.0, eta0=0.4)
        score = float(evaluate_sequence(a_opt.tolist()))
        
        if np.isfinite(score):
            results.append((score, n, method, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Phase1 [{i}]: n={n}, {method}, score={score:.6f}")
    
    # Also try from parent
    if parent is not None and time.monotonic() < phase1_deadline - 30:
        for n in [128, 256]:
            if parent.size >= n:
                a0 = parent[:n].copy()
                a0 = np.maximum(a0, 0)
                if a0.sum() > 1e-10:
                    a0 /= a0.sum()
                    t_alloc = min(30.0, phase1_deadline - time.monotonic() - 10)
                    if t_alloc > 5:
                        a_opt, obj_norm = mirror_descent_phase(a0, time_limit=t_alloc, t_param=50.0, eta0=0.3)
                        score = float(evaluate_sequence(a_opt.tolist()))
                        if np.isfinite(score):
                            results.append((score, n, 'parent', a_opt))
                            if score < best_score:
                                best_score = score
                                best_sequence = a_opt.copy()
                                print(f"  Phase1 parent: n={n}, score={score:.6f}")
    
    # Phase 2: Deep optimization on top results
    print(f"\n=== Phase 2: Deep optimization (best so far: {best_score:.6f}) ===")
    results.sort(key=lambda x: x[0])
    
    # Select top configs, ensuring diversity
    top_configs = []
    seen = set()
    for score, n, method, a_opt in results:
        key = n
        if len(top_configs) < 10 or (key not in seen and len(top_configs) < 14):
            top_configs.append((n, method, a_opt))
            seen.add(key)
    
    # Always include current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))
    
    # Also add fresh flat-spectrum inits for the best n
    best_n = len(best_sequence) if best_sequence is not None else 128
    for seed_off in [200, 300]:
        a0 = flat_spectrum_init(best_n, gamma=0.2, n_iter=400, seed=seed + seed_off)
        top_configs.append((best_n, f'flat_deep_{seed_off}', a0))
    
    remaining = deadline - time.monotonic() - 300
    n_configs = len(top_configs)
    time_per_config = max(10.0, remaining / max(1, n_configs))
    
    for i, (n, method, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 300:
            break
        t_alloc = min(time_per_config, deadline - time.monotonic() - 300)
        if t_alloc < 10:
            break
        
        # Try multiple parameter settings
        for t_p, eta_p in [(40.0, 0.35), (80.0, 0.3), (150.0, 0.25)]:
            if time.monotonic() > deadline - 300:
                break
            t_single = t_alloc / 3.0
            if t_single < 5:
                break
            
            a_opt, obj_norm = mirror_descent_phase(a0, time_limit=t_single, t_param=t_p, eta0=eta_p)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Phase2 [{i}] n={n}, {method}, t={t_p:.0f}: score={score:.6f}")
    
    # Phase 3: Intensive polishing with restarts
    print(f"\n=== Phase 3: Intensive polishing (best: {best_score:.6f}) ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 60
        n = len(best_sequence)
        
        # Multiple rounds of polishing with different parameters
        for round_idx in range(6):
            if remaining < 30:
                break
            t_single = remaining / (6 - round_idx) if round_idx < 5 else remaining
            t_single = min(t_single, remaining - 10)
            if t_single < 10:
                break
            
            # Vary parameters each round
            t_params = [30.0, 60.0, 100.0, 200.0, 350.0, 500.0]
            eta_params = [0.3, 0.25, 0.2, 0.15, 0.12, 0.1]
            
            a_opt, obj_norm = mirror_descent_phase(
                best_sequence, time_limit=t_single,
                t_param=t_params[round_idx % len(t_params)],
                eta0=eta_params[round_idx % len(eta_params)]
            )
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Phase3 round {round_idx}: score={score:.6f}")
            
            remaining = deadline - time.monotonic() - 60
        
        # Final round with perturbation-based restart
        if remaining > 60:
            for restart in range(3):
                if remaining < 20:
                    break
                t_single = remaining / 4
                if t_single < 15:
                    break
                
                # Perturb the best sequence
                a_perturbed = best_sequence + rng.normal(0, 0.03, n)
                a_perturbed = np.maximum(a_perturbed, 0)
                a_sum = a_perturbed.sum()
                if a_sum > 1e-10:
                    a_perturbed /= a_sum
                
                a_opt, obj_norm = mirror_descent_phase(
                    a_perturbed, time_limit=t_single, t_param=80.0, eta0=0.3
                )
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Phase3 restart {restart}: score={score:.6f}")
                
                remaining = deadline - time.monotonic() - 15
    
    # Phase 4: Try different sequence lengths for the final push
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print(f"\n=== Phase 4: Length exploration ===")
        best_n = len(best_sequence)
        alt_lengths = [l for l in [64, 96, 128, 160, 192, 256, 320] if l != best_n]
        
        for n in alt_lengths[:4]:
            if time.monotonic() > deadline - 50:
                break
            a0 = flat_spectrum_init(n, gamma=0.25, n_iter=300, seed=seed + 500)
            t_alloc = min(30.0, deadline - time.monotonic() - 50)
            if t_alloc < 10:
                break
            a_opt, _ = mirror_descent_phase(a0, time_limit=t_alloc, t_param=60.0, eta0=0.3)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Phase4 n={n}: score={score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.6f}")
    return best_sequence


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Uses flat-spectrum design combined with mirror descent
    to minimize 2n*max(auto-conv)/(sum)^2 for non-negative sequences.
    """
    result = run_optimization(seed=seed, budget_s=budget_s)
    # Ensure valid output
    result = np.maximum(result, 0)
    result = np.minimum(result, 1000.0)
    result = result[np.isfinite(result)]
    if len(result) == 0:
        result = np.ones(128)
    return [float(x) for x in result]
