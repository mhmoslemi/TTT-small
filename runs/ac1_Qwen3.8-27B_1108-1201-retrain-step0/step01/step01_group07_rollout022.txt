Let me think about the LP implementation more carefully.

The LP is:
- Variables: b_0, ..., b_{n-1} (n variables)
- Objective: maximize sum(b) → minimize -sum(b)
- Constraints:
  - conv(a, b)_k <= M for k = 0, ..., 2n-2 (2n-1 constraints)
  - b_j >= 0 (handled by bounds)

The constraint matrix A has A[k, j] = a[j] if 0 <= k-j <= n-1, else 0.

This is a "convolution matrix" which is upper-triangular with bandwidth n. Specifically:
- Row k has nonzeros in columns max(0, k-n+1) to min(k, n-1).
- The entries in row k, column j are a[j].

I can build this as a sparse matrix:
```python
def build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
```

    n = len(a)

    cols = []

`np.correlate(a, v, mode='valid')`:

- `out[i] = sum_k a[i+k] * conj(v[k])` for k=0,...,N-1.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_soft_max(a, tau):
    """Compute soft-max approximation of max(conv(a,a)) and its gradient."""
    c = np.convolve(a, a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    max_approx = c.max() + tau * np.log(exp_c.sum())  # soft-max value
    grad = 2.0 * np.correlate(q, a, 'valid')
    return max_approx, grad, c


def project_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    s = a.sum()
    if s == 0:
        return np.full(n, S / n)
    if s <= S:
        return a + (S - s) / n
    # Need to find threshold tau such that sum(max(a - tau, 0)) = S
    a_sorted = np.sort(a)[::-1]
    cumulative = np.cumsum(a_sorted)
    # Find the largest k such that a_sorted[k] > (cumulative[k] - S) / (k+1)
    rho = np.where(a_sorted > (cumulative - S) / np.arange(1, n + 1))[0]
    if len(rho) == 0:
        return np.maximum(a - a.mean(), 0) + (S - np.maximum(a - a.mean(), 0).sum()) / n
    k = rho[-1] + 1
    tau = (cumulative[k-1] - S) / k
    return np.maximum(a - tau, 0.0)


def init_ramp(n, tilt=0.3, seed=0):
    """Linear ramp initialization with small tilt."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + tilt * j / (n - 1)
    rng = np.random.default_rng(seed)
    a += 0.02 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_chirp(n, seed=0):
    """Chirp-based initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.5 * np.cos(2 * np.pi * t / n)
    rng = np.random.default_rng(seed)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_exp(n, rate=0.5, seed=0):
    """Exponential ramp initialization."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(rate * j / (n - 1))
    rng = np.random.default_rng(seed)
    a += 0.02 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

    """Hann window initialization."""
    j = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1)))
    rng = np.random.default_rng(seed)
    a += 0.02 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def run_gradient_descent(a0, tau_hi, tau_lo, lr0, t_budget, verbose=False):
    """Soft-max projected gradient descent."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = (a0 / max(a0.sum(), 1e-12) * S).copy()
    a = project_simplex(a, S)
    
    best = a.copy()
    best_max = np.convolve(best, best).max()
    best_score = evaluate_sequence(best.tolist())
    
    t0 = time.monotonic()
    lr = lr0
    step = 0
    stall_count = 0
    last_improve = time.monotonic()
    
    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        f = min(1.0, elapsed / max(t_budget, 1e-6))
        tau = tau_hi * (tau_lo / tau_hi) ** f
        
        c = np.convolve(a, a)
        c_shift = c - c.max()
        exp_c = np.exp(c_shift / tau)
        q = exp_c / exp_c.sum()
        grad = 2.0 * np.correlate(q, a, 'valid')
        
        a_new = a - lr * grad
        a_new = project_simplex(a_new, S)
        step += 1
        
        if step % 5 == 0:
            mc = np.convolve(a_new, a_new).max()
            if mc < best_max - 1e-10:
                best_max = mc
                best = a_new.copy()
                last_improve = time.monotonic()
                if lr < lr0 * 10:
                    lr *= 1.1
                stall_count = 0
            else:
                if time.monotonic() - last_improve > 10:
                    lr = max(0.001, lr * 0.7)
                    stall_count += 1
                else:
                    lr *= 0.95
            
            a = a_new
        else:
            a = a_new
    
    sc = evaluate_sequence(best.tolist())
    return best, sc


def build_conv_matrix(a):
    """Build sparse matrix A where (A @ b)[k] = conv(a, b)[k]."""
    n = len(a)
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    
    idx = 0
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1) + 1
        count = hi - lo
        rows[idx:idx+count] = k
        cols[idx:idx+count] = np.arange(lo, hi, dtype=np.int32)
        vals[idx:idx+count] = a[lo:hi]
        idx += count
    
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def lp_polish(a, verbose=False):
    """One LP polishing step. Returns improved sequence or None."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = a / max(a.sum(), 1e-12) * S
    a = project_simplex(a, S)
    
    M = np.convolve(a, a).max()
    
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, 
                     method='highs', options={'time_limit': 60, 'presolve': True})
    except Exception as e:
        if verbose:
            print(f"  LP error: {e}")
        return None
    
    if res.status != 0:
        if verbose:
            print(f"  LP status: {res.status}")
        return None
    
    b = res.x
    sum_b = b.sum()
    
    if sum_b <= S * (1 + 1e-6):
        if verbose:
            print(f"  LP: no improvement (sum_b={sum_b:.4f}, S={S:.4f})")
        return None
    
    # Rescale b to have sum S
    b = b / sum_b * S
    b = project_simplex(b, S)
    
    # Check if b actually improves
    new_max = np.convolve(b, b).max()
    old_max = M
    new_score = 2 * n * new_max / S**2
    old_score = 2 * n * old_max / S**2
    
    if verbose:
        print(f"  LP: sum_b/S={sum_b/S:.6f}, new_max/old_max={new_max/old_max:.6f}, "
              f"new_score={new_score:.6f}, old_score={old_score:.6f}")
    
    return b
