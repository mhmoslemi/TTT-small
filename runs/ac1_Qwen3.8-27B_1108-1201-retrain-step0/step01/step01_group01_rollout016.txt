def _make_conv_indices(n):
    """Precompute row and column indices for the convolution matrix."""
    j = np.arange(n)
    i = np.arange(n)
    rows = np.repeat(j, n) + np.tile(i, n)
    cols = np.repeat(j, n)
    return rows, cols


def _build_conv_matrix(a, rows, cols):
    """Build the convolution matrix for sequence a using precomputed indices."""
    vals = np.tile(a, len(a))
    n = len(a)
    return sp.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def _score(a):
    """Compute the score for a normalized sequence (sum=1)."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _lp_direction(a, M, rows, cols):
    """
    Solve the LP: max sum(g) s.t. A@g <= M, g >= 0
    where A is the convolution matrix of a.
    Returns the optimal g (unnormalized).
    """
    n = len(a)
    A = _build_conv_matrix(a, rows, cols)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 60})
    if result.success:
        return result.x
    return None


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient_optimize(a, deadline, rng, M_top=10, eta_init=None, cool_rate=0.85):
    """Subgradient descent for local polishing."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    
    if eta_init is None:
        eta_init = 0.05 / np.sqrt(n)
    
    best = a.copy()
    best_score = _score(best)
    eta = eta_init
    no_improve = 0
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 200:
            eta *= cool_rate
            no_improve = 0
            if eta < 1e-10:
                break
        
        # Get top M peak locations
        if M_top >= len(b):
            k_idx = np.arange(len(b))
            wts = np.ones(len(b)) / len(b)
        else:
            k_idx = np.argpartition(b, -M_top)[-M_top:]
            b_sel = b[k_idx]
            wts = np.exp(-5.0 * (b_max - b_sel) / max(b_max, 1e-15))
            wts /= np.sum(wts)
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for ki in range(len(k_idx)):
            w = wts[ki]
            if w < 1e-8:
                continue
            k = int(k_idx[ki])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[k - idx]
        
        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        a = _project_simplex(a, 1.0)

    all_cols = []
    all_vals = []
    for idx, k in enumerate(top_k):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        # Coefficients: a[k-j] for j in [j_lo, j_hi]
        js = np.arange(j_lo, j_hi + 1)
        coeffs = a[k - js]

        all_cols.extend(js.tolist())
        all_vals.extend(coeffs.tolist())
    
    A_r = sp.csc_matrix((all_vals, (all_rows, all_cols)), shape=(len(top_k), n))

    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A_r, b_ub=b_ub_r, bounds=bounds, method='highs',

"""
LP-based iterative optimization (restricted constraints) combined with subgradient
polishing and diversification to minimize the normalized autocorrelation peak of a
non-negative sequence. Uses the state-of-the-art LP direction finding with the top-K
tightest convolution constraints for efficiency, plus local search for polishing.

import numpy as np

import scipy.sparse as sp
from scipy.optimize import linprog


def _score_normalized(a):
    """Compute 2*n*max(convolve(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _lp_direction_restricted(a, M, K=200, time_limit=20):
    """
    Solve restricted LP: max sum(g) s.t. (a*g)[k] <= M for top-K k, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    n = len(a)
    b = np.convolve(a, a)
    if K >= len(b):
        top_k = np.arange(len(b))
    else:
        top_k = np.argpartition(b, -K)[-K:]
    top_k = sorted(top_k.tolist())
    K_actual = len(top_k)

    # Build restricted constraint matrix
    all_data = []
    all_row = []
    all_col = []
    for idx, k in enumerate(top_k):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        js = np.arange(j_lo, j_hi + 1)
        coeffs = a[k - js]
        all_data.extend(coeffs.tolist())
        all_row.extend([idx] * (j_hi - j_lo + 1))
        all_col.extend(js.tolist())

    if not all_data:
        return None

    A_r = sp.csc_matrix((all_data, (all_row, all_col)), shape=(K_actual, n))

    c = -np.ones(n)
    b_ub_r = np.full(K_actual, M + 1e-12)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c, A_ub=A_r, b_ub=b_ub_r, bounds=bounds,
                         method='highs',
                         options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None

    if result.success and result.x is not None:
        g = np.maximum(result.x, 0.0)
        g_sum = np.sum(g)
        if g_sum > 1e-10:
            return g / g_sum
    return None


def _lp_direction_full(a, M, time_limit=60):
    """
    Solve full LP: max sum(g) s.t. (a*g)[k] <= M for ALL k, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    n = len(a)
    # Build full convolution matrix
    j = np.arange(n)
    i = np.arange(n)
    rows = np.repeat(j, n) + np.tile(i, n)
    cols = np.repeat(j, n)
    vals = np.tile(a, n)
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-12)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None

    if result.success and result.x is not None:
        g = np.maximum(result.x, 0.0)
        g_sum = np.sum(g)
        if g_sum > 1e-10:
            return g / g_sum
    return None


def _subgradient_optimize(a, deadline, rng, M_top=15, eta_init=None, cool_rate=0.85, min_eta=1e-10):
    """Subgradient descent to minimize max autocorrelation peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n

    if eta_init is None:
        eta_init = 0.05 / np.sqrt(n)

    best = a.copy()
    best_score = _score_normalized(best)
    eta = eta_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 300:
            eta *= cool_rate
            no_improve = 0
            if eta < min_eta:
                break

        # Get top M peak locations
        if M_top >= len(b):
            k_idx = np.arange(len(b))
            wts = np.ones(len(b)) / len(b)
        else:
            k_idx = np.argpartition(b, -M_top)[-M_top:]
            b_sel = b[k_idx]
            wts = np.exp(-3.0 * (b_max - b_sel) / max(b_max, 1e-15))
            wts /= np.sum(wts)

        # Compute weighted gradient of max b_k w.r.t. a
        grad = np.zeros(n)
        for ki in range(len(k_idx)):
            w = wts[ki]
            if w < 1e-8:
                continue
            k = int(k_idx[ki])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[k - idx]

        # Center for sum constraint
        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        a = a - eta * grad / grad_norm
        a = _project_simplex(a, 1.0)
        S = np.sum(a)
        if S < 1e-10:
            a = best.copy()
            break
        it += 1

    return best, best_score


def _try_mix(a, g, best_score, best_a, n):
    """Try different mixing ratios and return (new_a, new_best_score, new_best_a)."""
    for t in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.02, 0.01]:
        new_a = (1.0 - t) * a + t * g
        new_score = _score_normalized(new_a)
        if new_score < best_score:
            return new_a, new_score, new_a.copy(), t
    return a, best_score, best_a, 0.0
