def _fft_convolve(x, y):
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    need = len(x) + len(y) - 1
    L = 1
    if need > 1:
        L = 1 << (need - 1).bit_length()
    else:
        L = 1
    fx = np.fft.rfft(x, L)
    fy = np.fft.rfft(y, L)
    z = np.fft.irfft(fx * fy, L)
    return z[:need]

def _cross_matrix_with_t(a):

    n = len(a)
    m = 2*n - 1
    cols = np.repeat(np.arange(n, dtype=np.int64), n)
    rows = cols + np.tile(np.arange(n, dtype=np.int64), n)
    vals = np.tile(a, n)
    rows_t = np.arange(m, dtype=np.int64)
    cols_t = np.full(m, n, dtype=np.int64)
    vals_t = np.full(m, -1.0, dtype=np.float64)
    all_vals = np.concatenate((vals, vals_t))
    all_rows = np.concatenate((rows, rows_t))
    all_cols = np.concatenate((cols, cols_t))
    return csr_matrix((all_vals, (all_rows, all_cols)), shape=(m, n+1))

```python
"""
Search for non-negative coefficient sequences minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2.

The main ingredients are:
  - direct L-BFGS-B optimization of a smoothed peak coefficient,
    with non-negativity bounds and a sum penalty,
  - a linear-programming map: given a, find g >= 0, sum(g)=sum(a),
    minimizing max(convolve(a,g)), then line-search a -> blend(a,g),
  - projected soft-peak gradient polishing.

Multiple starting points and several sequence lengths are explored.
The previous best sequence (global height_sequence_1) is used if present.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _fft_len(need):
    """Smallest power of two >= need."""
    if need <= 1:
        return 1
    return 1 << (need - 1).bit_length()


def _fft_convolve(x, y):
    """Linear convolution via FFT."""
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    need = len(x) + len(y) - 1
    L = _fft_len(need)
    fx = np.fft.rfft(x, L)
    fy = np.fft.rfft(y, L)
    z = np.fft.irfft(fx * fy, L)
    return z[:need]


def _conv_square(a):
    """Convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    need = 2 * n - 1
    L = _fft_len(need)
    fa = np.fft.rfft(a, L)
    z = np.fft.irfft(fa * fa, L)
    return z[:need]


def _softmax_prob(x, tau):
    """Softmax of tau*x, robust for large tau."""
    x = np.asarray(x, dtype=np.float64)
    if tau <= 0.0:
        return np.full_like(x, 1.0 / max(1, len(x)), dtype=np.float64)
    t = tau * x
    tm = float(np.max(t))
    if not np.isfinite(tm):
        return np.full_like(x, 1.0 / max(1, len(x)), dtype=np.float64)
    e = np.exp(t - tm)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.full_like(x, 1.0 / max(1, len(x)), dtype=np.float64)
    return e / s


def _top_k_weights(b, k):
    """Uniform weights on the largest k coefficients."""
    b = np.asarray(b, dtype=np.float64)
    L = len(b)
    k = int(max(1, min(L, k)))
    w = np.zeros(L, dtype=np.float64)
    if k >= L:
        w.fill(1.0 / L)
        return w
    idx = np.argpartition(b, L - k)[L - k:]
    w[idx] = 1.0 / k
    return w


def _soft_max_value(x, tau):
    """Smooth approximation to max(x)."""
    x = np.asarray(x, dtype=np.float64)
    if tau <= 0.0:
        return float(np.max(x))
    t = tau * x
    tm = float(np.max(t))
    e = np.exp(t - tm)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return float(tm) / tau
    return float((tm + np.log(s)) / tau)


def _normalize(a, target=None):
    """Normalize to non-negative sum target, default sqrt(2*n)."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    if target is None:
        target = np.sqrt(2.0 * len(a))
    s = float(np.sum(a))
    if s <= 1e-14:
        return np.ones(len(a), dtype=np.float64) * (target / max(1, len(a)))
    return a * (target / s)


def _score_fft(a):
    """Objective computed with FFT convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")
    b = _conv_square(a)
    mb = float(np.max(b)) if b.size else 0.0
    return float(2.0 * n * mb) / (s * s)


def _score_exact(a):
    """Objective computed with exact np.convolve."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")
    b = np.convolve(a, a)
    mb = float(np.max(b)) if b.size else 0.0
    return float(2.0 * n * mb) / (s * s)


def _a_to_y(a):
    """Logit/softplus parameterization."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 1e-15)
    y = np.log(a)
    y = y - float(np.mean(y))
    return y


def _y_to_a(y, target=None):
    """Stable softmax with fixed sum."""
    y = np.asarray(y, dtype=np.float64)
    n = len(y)
    if target is None:
        target = np.sqrt(2.0 * n)
    y = y - float(np.max(y))
    e = np.exp(y)
    s = float(np.sum(e))
    if s <= 1e-18:
        a = np.ones(n, dtype=np.float64)
    else:
        a = e / s
    return a * target


def _soft_loss_grad_a(a, tau, penalty, target_sum):
    """
    Objective in a-space:
        smooth peak of conv(a,a) + penalty*(sum(a)-target_sum)^2.
    Returns (value, gradient).
    """
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    n = len(a)
    b = _conv_square(a)
    t = tau * b
    tm = float(np.max(t))
    e = np.exp(t - tm)
    se = float(np.sum(e))
    if se > 0.0 and np.isfinite(se):
        w = e / se
        loss = (tm + float(np.log(se))) / tau
    else:
        w = np.full(2 * n - 1, 1.0 / max(1, 2 * n - 1), dtype=np.float64)
        loss = float(tm) / tau if tau > 0 else float(tm)

    sum_dev = float(np.sum(a)) - target_sum
    loss += penalty * sum_dev * sum_dev

    # Gradient of soft max b_k = 2 * conv(w, reversed(a)) sliced.
    grad = 2.0 * _fft_convolve(w, a[::-1])[n - 1:2 * n - 1]
    grad += 2.0 * penalty * sum_dev
    return float(loss), grad


def _soft_loss_grad_y(y, tau):
    """
    Softmax-parameter version: a = softmax(y) normalized to sqrt(2n).
    Minimizes smooth peak only; sum is exact by parameterization.
    """
    y = np.asarray(y, dtype=np.float64)
    n = len(y)
    target = np.sqrt(2.0 * n)
    a = _y_to_a(y, target)

    b = _conv_square(a)
    t = tau * b
    tm = float(np.max(t))
    e = np.exp(t - tm)
    se = float(np.sum(e))
    if se > 0.0 and np.isfinite(se):
        w = e / se
        loss = (tm + float(np.log(se))) / tau
    else:
        w = np.full(2 * n - 1, 1.0 / max(1, 2 * n - 1), dtype=np.float64)
        loss = float(tm) / tau if tau > 0 else float(tm)

    grad_a = 2.0 * _fft_convolve(w, a[::-1])[n - 1:2 * n - 1]
    # Gradient wrt y for softmax parameterization.
    grad_y = a * (grad_a - float(np.dot(a, grad_a)) / target)
    grad_y -= float(np.mean(grad_y))
    return float(loss), grad_y


def _lbfgs_direct_phase(a, tau, maxiter, penalty=20.0):
    """One L-BFGS-B phase in a-space with non-negativity and sum penalty."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    target = np.sqrt(2.0 * n)
    a = _normalize(a, target)
    best_a = a.copy()
    best_s = _score_fft(a)

    bounds = [(0.0, None)] * n
    try:
        res = minimize(
            _soft_loss_grad_a,
            a,
            args=(tau, penalty, target),
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": int(max(1, maxiter)),
                "ftol": 1e-12,
                "gtol": 1e-7,
                "disp": False,
            },
        )
    except Exception:
        return best_a, best_s

    if res is not None and getattr(res, "x", None) is not None:
        a2 = np.asarray(res.x, dtype=np.float64)
        a2 = np.maximum(a2, 0.0)
        if float(np.sum(a2)) > 1e-12:
            a2 = _normalize(a2, target)
            s2 = _score_fft(a2)
            if np.isfinite(s2) and s2 < best_s:
                best_a = a2
                best_s = s2
    return best_a, best_s


def _lbfgs_y_phase(a, tau, maxiter):
    """One L-BFGS-B phase in softmax y-space."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    target = np.sqrt(2.0 * n)
    a = _normalize(a, target)
    best_a = a.copy()
    best_s = _score_fft(a)
    y0 = _a_to_y(a)

    try:
        res = minimize(
            _soft_loss_grad_y,
            y0,
            args=(tau,),
            method="L-BFGS-B",
            options={
                "maxiter": int(max(1, maxiter)),
                "ftol": 1e-12,
                "gtol": 1e-7,
                "disp": False,
            },
        )
    except Exception:
        return best_a, best_s

    if res is not None and getattr(res, "x", None) is not None:
        y1 = np.asarray(res.x, dtype=np.float64)
        y1 = np.clip(y1, -60.0, 60.0)
        a2 = _y_to_a(y1, target)
        s2 = _score_fft(a2)
        if np.isfinite(s2) and s2 < best_s:
            best_a = a2
            best_s = s2
    return best_a, best_s


def _pg_refine(a, deadline, max_iters=300, tau=120.0, top_k=0):
    """Projected soft-peak gradient polishing."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    target = np.sqrt(2.0 * n)
    a = _normalize(a, target)
    best_a = a.copy()
    best_s = _score_fft(a)

    lr = 0.2
    no_improve = 0

    for _ in range(int(max_iters)):
        if time.monotonic() > deadline:
            break

        b = _conv_square(a)
        if top_k and top_k > 0:
            w = _top_k_weights(b, top_k)
        else:
            w_soft = _softmax_prob(b, tau)
            w_top = _top_k_weights(b, max(5, min(64, max(1, n // 8))))
            w = 0.65 * w_soft + 0.35 * w_top

        grad = 2.0 * _fft_convolve(w, a[::-1])[n - 1:2 * n - 1]
        grad = grad - float(np.mean(grad))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-12:
            break

        improved = False
        for f in (1.0, 0.5, 0.25, 0.125):
            cand = a - lr * f * grad
            cand = np.maximum(cand, 0.0)
            cs = float(np.sum(cand))
            if cs < 1e-12:
                break
            cand = cand * (target / cs)
            s = _score_fft(cand)
            if np.isfinite(s) and s < best_s - 1e-12:
                best_a = cand.copy()
                best_s = s
                a = cand
                lr = min(lr * 1.05, 2.0)
                improved = True
                no_improve = 0
                break

        if not improved:
            lr *= 0.85
            no_improve += 1
            if no_improve > 25 and lr < 0.01:
                break

    return best_a, best_s


def _cross_matrix_with_t(a):
    """
    Sparse matrix for LP: for every m,
        (a*g)_m - t <row>
    where <row> = [a[g] coefficients, -1 for t].
    """
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    m = 2 * n - 1

    cols = np.repeat(np.arange(n, dtype=np.int64), n)
    rows = cols + np.tile(np.arange(n, dtype=np.int64), n)
    vals = np.tile(a, n)

    rows_t = np.arange(m, dtype=np.int64)
    cols_t = np.full(m, n, dtype=np.int64)
    vals_t = np.full(m, -1.0, dtype=np.float64)

    rows_all = np.concatenate((rows, rows_t))
    cols_all = np.concatenate((cols, cols_t))
    vals_all = np.concatenate((vals, vals_t))

    return csr_matrix((vals_all, (rows_all, cols_all)), shape=(m, n + 1))


def _lp_refine(a, deadline, max_rounds=10):
    """
    LP refinement:
        given a, solve for g >= 0, sum(g)=sum(a), minimizing max(a*g).
        Then line-search a -> blend(a, g).
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    target = np.sqrt(2.0 * n)
    a = _normalize(a, target)

    best_a = a.copy()
    best_s = _score_fft(a)

    for _ in range(int(max_rounds)):
        if time.monotonic() > deadline:
            break

        try:
            A = _cross_matrix_with_t(a)
            c = np.zeros(n + 1, dtype=np.float64)
            c[n] = 1.0
            b_ub = np.zeros(2 * n - 1, dtype=np.float64)

            A_eq = np.zeros((1, n + 1), dtype=np.float64)
            A_eq[0, :n] = 1.0
            b_eq = np.array([target], dtype=np.float64)

            bounds = [(0.0, None)] * n
            bounds.append((0.0, None))

            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
            )
        except Exception:
            break

        if res is None or getattr(res, "status", -1) != 0:
            break

        g = np.asarray(res.x[:n], dtype=np.float64)
        g = np.maximum(g, 0.0)
        gs = float(np.sum(g))
        if gs < 1e-12:
            break
        g = g * (target / gs)

        improved = False
        for frac in (1.0, 0.75, 0.55, 0.4, 0.28, 0.18, 0.1, 0.05):
            cand = (1.0 - frac) * a + frac * g
            cand = _normalize(cand, target)
            s = _score_fft(cand)
            if np.isfinite(s) and s < best_s - 1e-12:
                best_a = cand.copy()
                best_s = s
                a = cand.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best_s


def _find_global_seq():
    """Return the previous best global sequence if available."""
    g = globals().get("height_sequence_1", None)
    if g is None:
        return None
    try:
        arr = np.asarray(g, dtype=np.float64)
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.clip(arr, 0.0, 1000.0)
    if not np.all(np.isfinite(arr)):
        return None
    return arr


def _make_starts(n, rng, global_seq=None, n_random=3):
    """Construct diverse non-negative starting sequences of length n."""
    starts = []

    if global_seq is not None and global_seq.size > 0:
        if global_seq.size >= n:
            starts.append(global_seq[:n].copy())
            starts.append(global_seq[:n][::-1].copy())
        else:
            reps = int(np.ceil(n / global_seq.size))
            starts.append(np.tile(global_seq, reps)[:n].copy())

        if global_seq.size > 1:
            old_x = np.arange(global_seq.size, dtype=np.float64)
            new_x = np.linspace(0.0, global_seq.size - 1.0, n)
            starts.append(np.interp(new_x, old_x, global_seq))

        if global_seq.size > 4:
            # Smoothed-downsampled then upscaled version.
            ds = global_seq[::2]
            if ds.size > 1:
                old_x = np.arange(ds.size, dtype=np.float64)
                new_x = np.linspace(0.0, ds.size - 1.0, n)
                starts.append(np.interp(new_x, old_x, ds))

    i = np.arange(n, dtype=np.float64)
    starts.append(np.ones(n, dtype=np.float64))

    if n > 1:
        starts.append(1.0 + 0.7 * np.cos(np.pi * i / (n - 1)))
        starts.append(1.0 + 0.7 * np.cos(2.0 * np.pi * i / (n - 1)))
        starts.append(0.2 + 0.8 * np.abs(np.sin(np.pi * i / (n - 1))))
        starts.append(0.2 + 0.8 * np.sin(2.0 * np.pi * 3.0 * i / n))
        starts.append(np.exp(-2.0 * np.minimum(i, n - 1 - i) / max(1.0, n)))

    for _ in range(int(n_random)):
        starts.append(rng.random(n) * 1.5 + 0.1)
        freq = float(rng.uniform(0.2, 2.5))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        starts.append(np.maximum(0.05, 1.0 + 0.6 * np.sin(2.0 * np.pi * freq * i / n + phase)))

    cleaned = []
    for arr in starts:
        try:
            arr = np.asarray(arr, dtype=np.float64)
        except Exception:
            continue
        if arr.size != n:
            continue
        arr = np.clip(arr, 0.0, 1000.0)
        if not np.all(np.isfinite(arr)):
            continue
        if float(np.sum(arr)) <= 0.0:
            arr = np.ones(n, dtype=np.float64)
        cleaned.append(arr)

    return cleaned


def _optimize_one(a0, deadline, max_lbfgs_iter=None, pg_iters=300, lp_rounds=8, do_lp=True):
    """Run a full local optimization pipeline from one starting point."""
    if a0 is None:
        return None, float("inf")

    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    if n == 0:
        return None, float("inf")

    a = np.clip(a, 0.0, 1000.0)
    if float(np.sum(a)) < 1e-8:
        a = np.ones(n, dtype=np.float64)
    a = _normalize(a)

    best_a = a.copy()
    best_s = _score_fft(a)

    if max_lbfgs_iter is None:
        max_lbfgs_iter = max(30, min(180, 120000 // max(1, n)))
    max_lbfgs_iter = int(max_lbfgs_iter)

    # Low-to-medium tau L-BFGS to move quickly.
    for tau in (2.0, 8.0, 30.0, 100.0):
        if time.monotonic() > deadline - 30.0:
            break
        a2, s2 = _lbfgs_direct_phase(best_a, tau, max_lbfgs_iter, penalty=20.0)
        if np.isfinite(s2) and s2 < best_s:
            best_a = a2
            best_s = s2

    # LP refinement (exact linear oracle for cross-peak).
    if do_lp and n <= 1024 and time.monotonic() < deadline - 45.0:
        local_ld = min(deadline - 45.0, time.monotonic() + max(60.0, 0.35 * max(60.0, max_lbfgs_iter)))
        a2, s2 = _lp_refine(best_a, local_ld, max_rounds=max(4, int(lp_rounds)))
        if np.isfinite(s2) and s2 < best_s:
            best_a = a2
            best_s = s2

    # High tau L-BFGS for peak focusing.
    for tau in (200.0, 500.0):
        if time.monotonic() > deadline - 30.0:
            break
        a2, s2 = _lbfgs_direct_phase(best_a, tau, max_lbfgs_iter // 2 + 20, penalty=20.0)
        if np.isfinite(s2) and s2 < best_s:
            best_a = a2
            best_s = s2

    # Optional softmax parameterization for a different basin.
    if time.monotonic() < deadline - 30.0:
        a2, s2 = _lbfgs_y_phase(best_a, 80.0, max_lbfgs_iter // 2 + 20)
        if np.isfinite(s2) and s2 < best_s:
            best_a = a2
            best_s = s2

    # Final projected gradient polish.
    if time.monotonic() < deadline - 10.0:
        a2, s2 = _pg_refine(best_a, min(deadline - 5.0, time.monotonic() + 60.0),
                            max_iters=max(50, int(pg_iters)), tau=200.0, top_k=0)
        if np.isfinite(s2) and s2 < best_s:
            best_a = a2
            best_s = s2

    # Final LP pass if short enough and time remains.
    if do_lp and n <= 1024 and time.monotonic() < deadline - 10.0:
        a2, s2 = _lp_refine(best_a, deadline - 2.0, max_rounds=max(2, int(lp_rounds) // 2))
        if np.isfinite(s2) and s2 < best_s:
            best_a = a2
            best_s = s2

    return best_a, best_s


def _consider_candidate(a, best_seq, best_score):
    """Accept candidate only if exact score improves the global best."""
    if a is None:
        return best_seq, best_score
    try:
        a = np.asarray(a, dtype=np.float64)
    except Exception:
        return best_seq, best_score
    if a.size == 0:
        return best_seq, best_score
    a = np.clip(a, 0.0, 1000.0)
    if not np.all(np.isfinite(a)):
        return best_seq, best_score
    if float(np.sum(a)) < 0.01:
        return best_seq, best_score

    s = _score_exact(a)
    if np.isfinite(s) and s < best_score:
        return a.copy(), float(s)
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Main entry point. Searches within budget_s and returns a list of floats.
    """
    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed))

    global_seq = _find_global_seq()
    best_seq = None
    best_score = float("inf")

    print(f"Starting search, budget={budget_s}s, global_seq_available={global_seq is not None}")

    jobs = []

    # First: lean heavily on the previous best construction if available.
    if global_seq is not None:
        for L in (512, 1024, 2048):
            st = _make_starts(L, rng, global_seq, n_random=0)
            if st:
                budget = 360.0 if L == 512 else 220.0
                jobs.append((L, st[0], budget))

    # Other starts, ordered by likely usefulness.
    n_list = [512, 1024, 256, 2048, 4096, 128]
    budgets = {
        128: 70.0,
        256: 100.0,
        512: 150.0,
        1024: 190.0,
        2048: 200.0,
        4096: 240.0,
    }
    take_counts = {
        128: 3,
        256: 3,
        512: 5,
        1024: 4,
        2048: 3,
        4096: 3,
    }

    for n in n_list:
        st = _make_starts(n, rng, global_seq, n_random=2)
        for a0 in st[:take_counts.get(n, 3)]:
            jobs.append((n, a0, budgets.get(n, 150.0)))

    print(f"Initial job count: {len(jobs)}")

    # Process jobs.
    for idx, (n, a0, job_budget) in enumerate(jobs):
        now = time.monotonic()
        if now > deadline - 120.0:
            break

        local_deadline = min(deadline - 80.0, now + float(job_budget))
        if local_deadline - now < 45.0:
            continue

        seq, _ = _optimize_one(
            a0,
            local_deadline,
            max_lbfgs_iter=None,
            pg_iters=300,
            lp_rounds=12 if n <= 1024 else 0,
            do_lp=(n <= 1024),
        )
        best_seq, best_score = _consider_candidate(seq, best_seq, best_score)
        print(f"job {idx:03d} n={n:5d} elapsed={time.monotonic() - t0:7.1f}s best={best_score:.9f}")

        # If we are already much below the stated target, stop broad search
        # and use remaining time for polishing.
        if best_score < 1.5005:
            break

    # Final polishing of the best known sequence.
    if best_seq is not None:
        nbest = len(best_seq)

        # Direct LP polishing is only practical for moderate length.
        if nbest <= 1024 and time.monotonic() < deadline - 50.0:
            seq, _ = _lp_refine(best_seq, deadline - 30.0, max_rounds=30)
            best_seq, best_score = _consider_candidate(seq, best_seq, best_score)
            print(f"final LP: best={best_score:.9f}")

        # High-tau L-BFGS.
        if time.monotonic() < deadline - 60.0:
            maxit = max(50, min(300, 120000 // max(1, nbest)))
            seq, _ = _lbfgs_direct_phase(best_seq, 500.0, maxit, penalty=20.0)
            best_seq, best_score = _consider_candidate(seq, best_seq, best_score)
            print(f"final high L-BFGS: best={best_score:.9f}")

        # Softmax parameterization.
        if time.monotonic() < deadline - 45.0:
            maxit = max(30, min(200, 80000 // max(1, nbest)))
            seq, _ = _lbfgs_y_phase(best_seq, 300.0, maxit)
            best_seq, best_score = _consider_candidate(seq, best_seq, best_score)
            print(f"final y-L-BFGS: best={best_score:.9f}")

        # Projected gradient.
        if time.monotonic() < deadline - 20.0:
            seq, _ = _pg_refine(best_seq, deadline - 10.0, max_iters=800, tau=300.0)
            best_seq, best_score = _consider_candidate(seq, best_seq, best_score)
            print(f"final PG: best={best_score:.9f}")

        # Try a longer tiled version if plenty of time remains.
        if best_seq is not None and time.monotonic() < deadline - 260.0:
            based = np.asarray(best_seq, dtype=np.float64)
            for factor in (2, 4):
                new_n = len(based) * factor
                if new_n <= 4096 and new_n > len(based) and time.monotonic() < deadline - 120.0:
                    tiled = np.tile(based, factor)[:new_n]
                    seq, _ = _optimize_one(
                        tiled,
                        min(deadline - 60.0, time.monotonic() + 180.0),
                        max_lbfgs_iter=None,
                        pg_iters=250,
                        lp_rounds=8 if new_n <= 1024 else 0,
                        do_lp=(new_n <= 1024),
                    )
                    best_seq, best_score = _consider_candidate(seq, best_seq, best_score)
                    print(f"tile factor {factor} -> n={new_n}: best={best_score:.9f}")

    # Fallback.
    if best_seq is None or len(np.asarray(best_seq)) == 0:
        best_seq = np.ones(256, dtype=np.float64)

    # Clean and normalize before returning.
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if not np.all(np.isfinite(best_seq)):
        best_seq = np.ones(256, dtype=np.float64)
    if float(np.sum(best_seq)) < 0.01:
        best_seq = np.ones(256, dtype=np.float64)
    best_seq = _normalize(best_seq)

    final_exact = _score_exact(best_seq)
    print(f"Final exact score: {final_exact:.9f} length={len(best_seq)}")

    # Optional cross-check with the official evaluator if present.
    try:
        ev = float(evaluate_sequence([float(x) for x in best_seq]))
        print(f"evaluate_sequence score: {ev:.9f}")
    except Exception as e:
        print(f"evaluate_sequence not usable in this environment: {e}")

    return [float(x) for x in best_seq]
```
