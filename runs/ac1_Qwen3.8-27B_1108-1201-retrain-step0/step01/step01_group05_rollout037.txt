"""
Minimize 2*n*max_conv / sum^2 using soft-max smoothed optimization.

Key innovations over previous approach:
1. Soft-max gradient computed via FFT (O(n log n) per step) instead of 
   top-k subgradient (which only uses a few lags)
2. Temperature annealing: start smooth (low T), end sharp (high T)
3. Nesterov momentum for faster convergence
4. Multiple diverse initializations including the previous best
5. Mirror descent (multiplicative weights) as complement to Adam
6. Final polish phase with high temperature
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_grad_fft(a, c, T):
    """Compute gradient of soft-max(c) w.r.t. a using FFT.
    
    soft_max(c) = (1/T) * log(sum(exp(T*c)))
    d(soft_max)/d(c_k) = exp(T*c_k) / sum(exp(T*c_j)) =: w_k
    
    d(c_k)/d(a_i) = 2*a[k-i] if 0 <= k-i < n else 0
    
    So grad[i] = 2 * sum_k w_k * a[k-i] = 2 * cross_corr(w, a)[i]
    
    Computed via FFT in O(n log n).
    """
    n = len(a)
    # Soft-max weights
    c_shifted = T * (c - c.max())
    exp_c = np.exp(c_shifted)
    w = exp_c / exp_c.sum()
    
    # cross_corr(w, a)[i] = sum_j w[i+j] * a[j] for j=0..n-1
    # = (a[::-1] * w)[i + n - 1] (linear convolution)
    r = a[::-1]
    conv_size = n + (2*n - 1) - 1  # = 3n - 2
    fsize = 1 << (conv_size - 1).bit_length()
    R = np.real(ifft(fft(r, fsize) * fft(w, fsize)))
    grad = 2.0 * R[n-1 : 2*n-1]
    return grad, w


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n, dtype=np.float64)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clamp to [0, 1000] and rescale to have sum S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.full(len(a), S / len(a), dtype=np.float64)
    return a * (S / s)

    a = normalize_to_sum(a_init.copy(), S)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init if lr_init > 0 else 0.5 * S / n
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    best_soft = (1.0/T_init) * np.log(np.sum(np.exp(T_init*(c - c.max())))) + c.max()
    
    start = time.monotonic()
    no_improve = 0
    REPORT_INTERVAL = 2000
    
    while time.monotonic() - start < max_time:
        t += 1
        # Anneal temperature
        progress = min(1.0, t / max(1, int(0.8 * max_time / 0.005)))  # assume ~5ms per iteration
        T = T_init * (T_end / T_init) ** progress
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute soft-max gradient
        grad, w = softmax_grad_fft(a, c, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr_t = lr * (0.5 ** (float(t) / 3000.0))
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)
        
        # Random restart if stuck
        if no_improve > 10000:
            a = project_simplex(a + rng.normal(0.0, 0.005 * S / n, n), S)
            no_improve = 0
        
        if t % REPORT_INTERVAL == 0:
            elapsed = time.monotonic() - start
            print(f"  Adam t={t} T={T:.1f} maxc={cur:.6f} best={best_score:.6f} "
                  f"elapsed={elapsed:.1f}s")
    
    return best_local, best_score


def run_mirror_softmax(a_init, n, S, max_time, rng, T_init=10.0, T_end=80.0,
                       eta_init=0.5):
    """Mirror descent (multiplicative weights) with soft-max gradient."""
    a = normalize_to_sum(a_init.copy(), S)
    t = 0
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, t / max(1, int(0.8 * max_time / 0.005)))
        T = T_init * (T_end / T_init) ** progress
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        grad, w = softmax_grad_fft(a, c, T)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break
        
        eta = eta_init / (1.0 + 0.002 * t)
        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 1e-15)
        a = a_new * (S / a_new.sum())
        
        if no_improve > 15000:
            a = np.maximum(a + rng.normal(0.0, 0.003 * S / n, n), 0.0)
            a = a * (S / a.sum())
            no_improve = 0
    
    return best_local, best_score


def run_nesterov_softmax(a_init, n, S, max_time, T_init=10.0, T_end=100.0):
    """Nesterov accelerated gradient with soft-max and simplex projection."""
    a = normalize_to_sum(a_init.copy(), S)
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n, dtype=np.float64)  # momentum
    n_elem = len(a)
    t = 0
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    
    # Lipschitz constant estimate (rough)
    L = 2.0 * S  # upper bound on gradient Lipschitz
    
    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, t / max(1, int(0.7 * max_time / 0.005)))
        T_cur = T_init * (T_end / T_init) ** progress
        
        # Nesterov step
        alpha_t = 2.0 / (t + 1.0 + 2.0)
        y_old = y.copy()
        y = a + (alpha_t - 1.0) * (a - y_old)
        y = np.maximum(y, 0.0)
        y = y * (S / max(y.sum(), 1e-15))
        
        c = fft_convolve_self(y)
        grad, w = softmax_grad_fft(y, c, T_cur)
        
        # Gradient step
        gamma_t = 1.0 / (t + 1.0)
        a_new = y - gamma_t * grad
        a = project_simplex(a_new, S)
        
        cur = c.max()  # Using y's convolution (close enough)
        c_a = fft_convolve_self(a)
        if c_a.max() < best_score - 1e-10:
            best_score = c_a.max()
            best_local = a.copy()
    
    return best_local, best_score

"""
Minimize 2*n*max_conv / sum^2 using FFT-accelerated soft-max gradient optimization.

Key innovations:
1. Soft-max smoothed objective with FFT-based gradient (O(n log n) per step)
2. Temperature annealing from smooth to sharp for progressive peak focusing
3. Multiple diverse initializations (given, uniform, smooth, structured)
4. Nesterov momentum variant for faster convergence on the smooth approximation
5. Adaptive temperature based on active lag count
6. Multi-phase polish with increasing sharpness
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_grad_fft(a, c, T):
    """Compute gradient of soft-max(c) w.r.t. a using FFT in O(n log n).
    
    Returns (grad, weights) where grad[i] = 2 * sum_j w[i+j] * a[j].
    """
    n = len(a)
    c_shifted = T * (c - c.max())
    exp_c = np.exp(c_shifted)
    w = exp_c / exp_c.sum()
    
    # g[i] = sum_j w[i+j] * a[j] = (a * w_rev)[2n-2-i]
    w_rev = w[::-1]
    conv_size = n + (2 * n - 1) - 1  # 3n - 2
    fsize = 1 << (conv_size - 1).bit_length()
    conv = np.real(ifft(fft(a, fsize) * fft(w_rev, fsize)))
    g = conv[n - 1:2 * n - 1][::-1]
    
    grad = 2.0 * g
    return grad, w


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n, dtype=np.float64)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clamp to [0, 1000] and rescale to have sum S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.full(len(a), S / len(a), dtype=np.float64)
    return a * (S / s)


def run_adam_sm(a_init, n, S, max_time, rng, T_init=10.0, T_end=100.0,
                lr_scale=0.5, beta1=0.9, beta2=0.999):
    """Adam optimizer with soft-max gradient and simplex projection."""
    a = normalize_to_sum(a_init.copy(), S)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    eps = 1e-8
    lr_base = lr_scale * S / n
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    no_improve = 0
    last_report = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, t / 50000.0)
        T = T_init * (T_end / T_init) ** progress
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        grad, w = softmax_grad_fft(a, c, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr_t = lr_base * (0.5 ** (float(t) / 4000.0))
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)
        
        if no_improve > 15000:
            a = project_simplex(a + rng.normal(0.0, 0.003 * S / n, n), S)
            no_improve = 0
        
        if t - last_report > 20000:
            elapsed = time.monotonic() - start
            print(f"    Adam t={t} T={T:.1f} cur={cur:.6f} best={best_score:.6f} [{elapsed:.0f}s]")
            last_report = t
    
    return best_local, best_score


def run_nesterov_sm(a_init, n, S, max_time, T_init=10.0, T_end=100.0):
    """Nesterov accelerated gradient with soft-max and simplex projection."""
    a = normalize_to_sum(a_init.copy(), S)
    y = a.copy()
    t = 0
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    last_report = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, t / 30000.0)
        T = T_init * (T_end / T_init) ** progress
        
        alpha = 2.0 / (t + 3.0)
        y_old = y.copy()
        y = a + (alpha - 1.0) * (a - y_old)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y * (S / sy)
        
        c = fft_convolve_self(y)
        grad, w = softmax_grad_fft(y, c, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        gamma = 1.0 / (t + 1.0)
        a_new = y - gamma * grad / gnorm * (S / n)
        a = project_simplex(a_new, S)
        
        c_a = fft_convolve_self(a)
        if c_a.max() < best_score - 1e-12:
            best_score = c_a.max()
            best_local = a.copy()
        
        if t - last_report > 20000:
            elapsed = time.monotonic() - start
            print(f"    Nesterov t={t} T={T:.1f} best={best_score:.6f} [{elapsed:.0f}s]")
            last_report = t
    
    return best_local, best_score


def run_mirror_sm(a_init, n, S, max_time, rng, T_init=10.0, T_end=80.0):
    """Mirror descent (multiplicative weights) with soft-max gradient."""
    a = normalize_to_sum(a_init.copy(), S)
    t = 0
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    no_improve = 0
    last_report = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, t / 40000.0)
        T = T_init * (T_end / T_init) ** progress
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        grad, w = softmax_grad_fft(a, c, T)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break
        
        eta = 0.3 / (1.0 + 0.001 * t)
        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 1e-15)
        a = a_new * (S / a_new.sum())
        
        if no_improve > 20000:
            a = np.maximum(a + rng.normal(0.0, 0.002 * S / n, n), 0.0)
            a = a * (S / a.sum())
            no_improve = 0
        
        if t - last_report > 20000:
            elapsed = time.monotonic() - start
            print(f"    Mirror t={t} T={T:.1f} cur={cur:.6f} best={best_score:.6f} [{elapsed:.0f}s]")
            last_report = t
    
    return best_local, best_score
