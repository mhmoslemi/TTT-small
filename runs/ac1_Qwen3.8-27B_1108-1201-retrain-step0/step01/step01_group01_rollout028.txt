def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - target)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target/n)
    theta = (cssv[rho[-1]] - target) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)

def _convolve_full(a, b):
    """Full convolution using FFT."""
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def _grad_lse(a, weights):
    """
    Compute sum_k weights[k] * 2 * a_{k-m} for m=0,...,n-1.
    This is the gradient part for the log-sum-exp smoothed objective.
    weights has length 2n-1.

    fa = np.fft.rfft(a, size)

    conv = np.fft.irfft(fa * fw, size)

# --- Helper functions ---

def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def _grad_weighted(a, w):
    """Compute sum_k w_k * 2 * a_{k-m} for m=0,...,n-1.
    w has length 2n-1 (the weights for each lag)."""
    n = len(a)
    w_rev = w[::-1]
    size = 1 << (n + len(w) - 2).bit_length()
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)
    # g_m = 2 * conv[2n-2-m] for m=0,...,n-1
    g = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return g


def _score(a):
    """Compute 2n * max(convolve(a, a)) assuming sum(a)=1."""
    n = len(a)
    b = _fft_convolve(a, a)
    return 2.0 * n * b.max()


def _fasta(a, tau, deadline, max_iters=10000):
    """FISTA with log-sum-exp smoothing."""
    n = len(a)
    a = _project_simplex(a.copy(), 1.0)
    a_prev = a.copy()
    t = 1.0
    L = 2.0 * (tau + 1)  # Lipschitz estimate
    
    best = a.copy()
    best_score = _score(best)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t_prev = t
        t = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
        y = a + (t_prev - 1.0) / t * (a - a_prev)
        y = _project_simplex(y, 1.0)
        
        b = _fft_convolve(y, y)
        max_tb = tau * b.max()
        exp_vals = np.exp(tau * b - max_tb)
        w = exp_vals / exp_vals.sum()
        
        grad = _grad_weighted(y, w)
        grad -= grad.mean()  # center for sum constraint
        
        a_new = _project_simplex(y - (1.0 / L) * grad, 1.0)
        
        a_prev = a
        a = a_new
        
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best = a.copy()
    
    return best, best_score

The DFT: $\hat{a}_m = \sum_j a_j e^{-2\pi i mj/N}$, $\hat{w}_m = \sum_k w_k e^{-2\pi i mk/N}$.

$\text{conj}(\hat{a}_m) = \sum_j a_j e^{2\pi i mj/N}$

$\text{conj}(\hat{a}_m) \cdot \hat{w}_m = \sum_j \sum_k a_j w_k e^{2\pi i mj/N} e^{-2\pi i mk/N} = \sum_j \sum_k a_j w_k e^{-2\pi i m(k-j)/N}$

$= \sum_j \sum_k a_j w_k \cdot \delta_{\ell, k-j \pmod{N}}$
$= \sum_j a_j w_{j+\ell \pmod{N}}$
