def _compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def _gradient_topk(a, peak_indices):
    n = len(a)
    grad = np.zeros(n)
    for k_idx in peak_indices:
        lo = max(0, k_idx - n + 1)
        hi = min(n - 1, k_idx)
        indices = np.arange(lo, hi + 1)
        grad[indices] += 2.0 * a[k_idx - indices]
    return grad


def _subgradient_optimize(a0, deadline, max_iters=5000, topk=3, use_annealing=True):
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a)

    best = a.copy()
    best_score = _compute_obj(a)
    S_t = np.sqrt(2.0 * n)

    v = np.zeros(n)
    beta = 0.85
    T_init = 0.05
    T_floor = 0.0005
    current_score = best_score

    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break

        c = np.convolve(a, a)
        peak_indices = np.argsort(c)[-topk:]
        grad = _gradient_topk(a, peak_indices)
        grad -= grad.mean()

        progress = it / max_iters
        lr = 0.08 * (1.0 - progress) ** 1.5
        lr = max(lr, 0.0005)

        T = T_init * (T_floor / T_init) ** progress

        v = beta * v + grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S_t)

        new_score = _compute_obj(a_new)

        if use_annealing and T > T_floor * 2:
            if new_score < current_score:
                accept = True
            else:
                delta = new_score - current_score
                accept = np.random.random() < np.exp(-delta / T)
        else:
            accept = new_score < current_score

        if accept:
            a = a_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                if (it + 1) % 500 == 0:
                    print(f"      it={it}, score={best_score:.6f}")
        else:
            v *= 0.5

        if it > 0 and it % 800 == 0:
            perturb_scale = 0.03 * (1.0 - progress)
            a_pert = a + np.random.normal(0, perturb_scale, n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = _normalize(a_pert, S_t)
            pert_score = _compute_obj(a_pert)
            if pert_score < best_score:
                best_score = pert_score
                best = a_pert.copy()
                a = a_pert.copy()
                current_score = pert_score
                v = np.zeros(n)

    return best, best_score


def _coordinate_descent(a0, deadline, max_passes=20):
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    best = a.copy()
    best_score = _compute_obj(a)
    S_t = np.sqrt(2.0 * n)

    for path in range(max_passes):
        if time.monotonic() > deadline - 10.0:
            break

        improved = False
        for j in range(n):
            if time.monotonic() > deadline - 10.0:
                break
            old_val = a[j]
            a[j] = 0.0
            c_base = np.convolve(a, a)
            S_base = a.sum()

            if S_base < 1e-12:
                a[j] = old_val
                continue

            lo = 0.0
            hi = max(2.0 * S_base / n, 0.1)

            for _ in range(30):
                mid = (lo + hi) / 2.0
                a[j] = mid
                c_test = c_base.copy()
                k_valid = 2 * j
                if k_valid < 2 * n - 1:
                    c_test[k_valid] += mid * mid
                for k in range(2 * n - 1):
                    diff = k - j
                    if 0 <= diff < n and diff != j:
                        c_test[k] += 2.0 * mid * a[diff]
                S_test = S_base + mid
                if S_test < 1e-12:
                    obj_test = float('inf')
                else:
                    obj_test = 2.0 * n * c_test.max() / (S_test * S_test)

                lo_mid = mid + (hi - mid) * 0.382
                a[j] = lo_mid
                c_test2 = c_base.copy()
                if k_valid < 2 * n - 1:
                    c_test2[k_valid] += lo_mid * lo_mid
                for k in range(2 * n - 1):
                    diff = k - j
                    if 0 <= diff < n and diff != j:
                        c_test2[k] += 2.0 * lo_mid * a[diff]
                S_test2 = S_base + lo_mid
                obj_lo = 2.0 * n * c_test2.max() / (S_test2 * S_test2) if S_test2 > 1e-12 else float('inf')

                hi_mid = mid + (hi - mid) * (1 - 0.382)
                a[j] = hi_mid
                c_test3 = c_base.copy()
                if k_valid < 2 * n - 1:
                    c_test3[k_valid] += hi_mid * hi_mid
                for k in range(2 * n - 1):
                    diff = k - j
                    if 0 <= diff < n and diff != j:
                        c_test3[k] += 2.0 * hi_mid * a[diff]
                S_test3 = S_base + hi_mid
                obj_hi = 2.0 * n * c_test3.max() / (S_test3 * S_test3) if S_test3 > 1e-12 else float('inf')

                if obj_lo < obj_hi:
                    hi = mid
                else:
                    lo = mid

            a[j] = (lo + hi) / 2.0
            a = _normalize(a, S_t)
            s = _compute_obj(a)
            if s + 1e-12 < best_score:
                best_score = s
                best = a.copy()
                improved = True
            else:
                a[j] = old_val

        if not improved:
            break

    return best, best_score


def _lp_refine(a, deadline, max_iters=30):
    n = len(a)
    if n > 1024:
        return a.copy(), _compute_obj(a)

    S_t = np.sqrt(2.0 * n)
    best = a.copy()
    best_score = _compute_obj(a)
    c = np.convolve(a, a)
    M = float(c.max())

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        A_mat = lil_matrix((2 * n - 1, n))
        for m in range(2 * n - 1):
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            for j in range(lo, hi + 1):
                A_mat[m, j] = a[m - j]
        A_mat = A_mat.tocsr()

        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 30})
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break

        improved = False
        for t in [0.6, 0.4, 0.25, 0.1]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, S_t)
            s_new = _compute_obj(a_new)
            if s_new < best_score - 1e-9:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            break

        c = np.convolve(a, a)
        M = float(c.max())
        print(f"      LP it={it}, M={M:.6f}, score={best_score:.6f}")

    return best, best_score


def _generate_starts(n, rng):
    starts = []
    i = np.arange(n, dtype=np.float64)
    starts.append(np.ones(n) * 0.5 + 0.5 * np.cos(2 * np.pi * i / n))
    starts.append(np.ones(n) * 0.5 + 0.5 * np.cos(2 * np.pi * i * 3.0 / n))
    starts.append(np.ones(n) * 0.5 + 0.5 * np.cos(2 * np.pi * i * 5.0 / n))
    starts.append(np.ones(n) * 0.5 + 0.3 * np.sin(2 * np.pi * i * 7.0 / n))
    starts.append(np.abs(np.sin(np.pi * i / (n - 1))) ** 1.5 + 0.1)
    starts.append(np.exp(-0.5 * ((i - n / 2) / (n * 0.3)) ** 2) + 0.1)
    starts.append(rng.random(n) * 0.5 + 0.25)
    starts.append(0.5 + 0.3 * np.cos(2 * np.pi * i * 2.0 / n) + 0.2 * np.cos(2 * np.pi * i * 5.0 / n))
    starts.append(np.maximum(0.0, 1.0 - 2.0 * np.abs(i / n - 0.5) - np.random.normal(0, 0.05, n)))
    for s in starts:
        s = np.clip(s, 0.0, None)
        starts[-1] = s if s.sum() > 0 else np.ones(n)
    return [np.clip(s, 0.0, None) for s in starts if s.sum() > 0.01]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def _update_best(seq, score):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  *** New best: score={best_score:.6f}, n={len(seq)} ***")

    # Try to use the global height_sequence_1
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)):
            print(f"Using height_sequence_1 with length {gs.size}")
        else:
            gs = None
    except NameError:
        gs = None

    # Phase 1: Quick exploration with various n
    print("Phase 1: Quick exploration")
    n_candidates = [128, 256, 512, 768]

    if gs is not None:
        for n in n_candidates:
            if time.monotonic() > deadline - 300.0:
                break
            if n <= len(gs):
                a0 = gs[:n].copy()
                t0 = time.monotonic()
                seq, score = _subgradient_optimize(a0, deadline - 200.0, max_iters=2000, topk=3)
                _update_best(seq, score)
                print(f"  n={n} (from gs): score={score:.6f}, time={time.monotonic()-t0:.1f}s")
            else:
                a0 = np.zeros(n)
                a0[:len(gs)] = gs
                seq, score = _subgradient_optimize(a0, deadline - 200.0, max_iters=2000, topk=3)
                _update_best(seq, score)
                print(f"  n={n} (padded gs): score={score:.6f}, time={time.monotonic()-t0:.1f}s")

    for n in n_candidates:
        if time.monotonic() > deadline - 300.0:
            break
        starts = _generate_starts(n, rng)
        for si, a0 in enumerate(starts[:4]):
            if time.monotonic() > deadline - 250.0:
                break
            t0 = time.monotonic()
            seq, score = _subgradient_optimize(a0, deadline - 200.0, max_iters=1500, topk=3)
            _update_best(seq, score)
            print(f"  n={n}, start={si}: score={score:.6f}, time={time.monotonic()-t0:.1f}s")

    # Phase 2: Detailed optimization of best candidates
    print("Phase 2: Detailed optimization")
    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        print(f"  Best so far: score={best_score:.6f}, n={n_best}, remaining={remaining:.0f}s")

        # Further subgradient optimization
        seq, score = _subgradient_optimize(best_seq, deadline - 100.0, max_iters=8000, topk=5, use_annealing=True)
        _update_best(seq, score)

        # Coordinate descent polishing
        if time.monotonic() < deadline - 80.0:
            print("  Running coordinate descent...")
            seq, score = _coordinate_descent(best_seq, deadline - 50.0, max_passes=10)
            _update_best(seq, score)

        # LP refinement
        if n_best <= 1024 and time.monotonic() < deadline - 40.0:
            print("  Running LP refinement...")
            seq, score = _lp_refine(best_seq, deadline - 10.0, max_iters=25)
            _update_best(seq, score)

        # After LP, do more coordinate descent
        if time.monotonic() < deadline - 30.0:
            print("  Final coordinate descent pass...")
            seq, score = _coordinate_descent(best_seq, deadline - 15.0, max_passes=15)
            _update_best(seq, score)

        # Final subgradient polish
        if time.monotonic() < deadline - 15.0:
            print("  Final subgradient polish...")
            seq, score = _subgradient_optimize(best_seq, deadline - 5.0, max_iters=2000, topk=5, use_annealing=False)
            _update_best(seq, score)

    # Try a few more n values if time permits
    if time.monotonic() < deadline - 100.0:
        print("Phase 3: Additional exploration")
        for n in [384, 640, 896]:
            if time.monotonic() > deadline - 50.0:
                break
            a0 = _generate_starts(n, rng)[0]
            if gs is not None:
                a0 = np.zeros(n)
                a0[:min(len(gs), n)] = gs[:min(len(gs), n)]
                if a0.sum() < 0.01:
                    a0 = _generate_starts(n, rng)[0]
            seq, score = _subgradient_optimize(a0, deadline - 30.0, max_iters=3000, topk=4)
            if score < best_score:
                _update_best(seq, score)
                print(f"  n={n} new best: score={score:.6f}")
            else:
                print(f"  n={n}: score={score:.6f} (not better)")

    # Final output
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"Final result: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Fallback to uniform sequence")
    return [1.0] * 256
