```python
"""
Multi-strategy optimization for minimizing the peak of linear autocorrelation
of non-negative step functions. Combines LP-based direction finding, 
gradient descent with soft-max smoothing, and simulated annealing polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score computation."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def solve_lp_direction(f, M, max_iters_hint=100):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g.
    """
    n = len(f)
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi+1] = f[j_lo:j_hi+1]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x
    return np.ones(n) / n


def lp_iteration(f, time_limit_per_lp=2.0):
    """
    One LP-based iteration: find direction, normalize, blend with line search.
    Returns (new_f, improved).
    """
    n = len(f)
    S = np.sum(f)
    M = np.max(fft_convolve(f, f))
    
    t0 = time.monotonic()
    g = solve_lp_direction(f, M)
    
    if time.monotonic() - t0 > time_limit_per_lp:
        return f, False
    
    g_sum = np.sum(g)
    if g_sum < 1e-10:
        return f, False
    
    g_norm = g / g_sum * S
    
    # Line search
    best_f = f.copy()
    best_score = score_fast(f)
    for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
        candidate = (1 - t) * f + t * g_norm
        candidate = np.clip(candidate, 0, None)
        s = score_fast(candidate)
        if s < best_score:
            best_score = s
            best_f = candidate
    
    return best_f, best_score < score_fast(f)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def local_search(a_init, time_limit, rng):
    """Simulated annealing local search."""
    current = a_init.astype(np.float64).copy()
    current_score = score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.3:
            # Targeted peak reduction
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            # Find top contributors to the peak
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            # Increase others
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                inc_idx = rng.choice(other, size=min(10, len(other)), replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=len(inc_idx)))
        elif r < 0.5:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.7:
            # Swap mass between positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            delta = current[i] * rng.uniform(-0.1, 0.1)
            candidate[i] = max(0, current[i] - delta)
            candidate[j] = max(0, current[j] + delta)
        elif r < 0.85:
            # Symmetrize partially
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.015, size=n))
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0, 0.01, size=n))
        
        candidate = np.clip(candidate, 0, 1000)
        new_score = score_fast(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
```

```python
Multi-strategy optimization for minimizing the peak of linear autocorrelation
of non-negative step functions. Combines LP-based direction finding (solving
max sum(g) s.t. f*g <= M, g>=0), gradient descent with soft-max smoothing,

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score computation matching evaluate_sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def solve_lp_direction(f, M):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g vector.
    """
    n = len(f)
    # Build constraint matrix: A[k, j] = f[j] if max(0,k-n+1) <= j <= min(n-1,k)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi + 1] = f[j_lo:j_hi + 1]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

    if res.status == 0:
        return res.x
    return np.ones(n) / n


def lp_iteration(f):
    """
    One LP-based iteration. Returns (new_f, improved, score).
    """
    n = len(f)
    S = np.sum(f)
    if S < 1e-10:
        return f, False, np.inf
    conv = fft_convolve(f, f)
    M = np.max(conv)

    g = solve_lp_direction(f, M)
    g_sum = np.sum(g)
    if g_sum < 1e-10:
        return f, False, score_fast(f)

    g_norm = g / g_sum * S

    # Line search over blend parameter
    best_candidate = f.copy()
    best_score = score_fast(f)
    for t in [0.4, 0.6, 0.8, 1.0, 0.2, 0.5, 0.7, 0.9]:
        candidate = (1 - t) * f + t * g_norm
        candidate = np.clip(candidate, 0, None)
        s = score_fast(candidate)
        if s < best_score:
            best_score = s
            best_candidate = candidate

    improved = best_score < score_fast(f) - 1e-12
    return best_candidate, improved, best_score


def grad_smoothed_peak(a, tau):
    """
    Compute gradient of soft_max(conv(a,a), tau) w.r.t. a using FFT.
    Returns (gradient, smoothed_value).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp

    # grad[p] = a[p] + sum_j a[j]*w[j+p]
    # = a[p] + (a_rev * w)[n-1+p]
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    cross_part = corr[n - 1:2 * n - 1]
    grad = a + cross_part

    smooth_val = c_max + tau * np.log(sum_exp)
    return grad, smooth_val


def gradient_optimize(a_init, time_limit, rng, lr_base=5.0):
    """
    Adam gradient descent in log-space with annealed soft-max temperature.
    """
    n = len(a_init)
    S = np.sum(a_init)
    theta = np.log(a_init / S + 1e-10)

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0
    best_theta = theta.copy()
    best_score = score_fast(a_init)
    t0 = time.monotonic()

    init_conv = fft_convolve(a_init, a_init)
    peak_mag = np.max(init_conv)
    tau_start = max(20.0 / n, peak_mag * 0.5)
    tau_end = max(0.2 / n, peak_mag * 0.01)

    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** min(1.0, elapsed * 1.2)

        a = softmax_normalize(theta) * S
        a = np.clip(a, 1e-12, None)

        grad_a, _ = grad_smoothed_peak(a, tau)

        inner = np.dot(grad_a, a)
        grad_theta = a * (grad_a - inner / S)

        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * grad_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (grad_theta * grad_theta)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        lr_current = lr_base / np.sqrt(step)
        theta = theta - lr_current * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 50 == 0:
            a_check = softmax_normalize(theta) * S
            a_check = np.clip(a_check, 1e-12, None)
            sc = score_fast(a_check)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()

    best_a = softmax_normalize(best_theta) * S
    best_a = np.clip(best_a, 1e-10, None)
    return best_a, best_score


def local_search(a_init, time_limit, rng):
    """Simulated annealing with targeted peak-reduction moves."""
    current = a_init.astype(np.float64).copy()
    current_score = score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.003
    cooling = 0.9996

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.35:
            # Targeted: reduce peak contributors, increase others
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(12, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.70:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, size=width))
        elif r < 0.85:
            # Partial symmetrization
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.015, size=n))
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0, 0.012, size=n))

        candidate = np.clip(candidate, 0, 1000)
        new_score = score_fast(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature *= cooling

    return best, best_score


def resample_sequence(a, new_n):
    """Resample a 1D sequence to a new length."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def make_chirp_init(n, alpha, rng):
    """Chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng):
    """Multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0
    for k, amp in [(1, 0.25), (3, 0.15), (5, 0.1), (7, 0.08), (11, 0.05)]:
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def run_lp_phase(a, time_limit, rng, verbose=True):
    """Run LP-based optimization loop."""
    n = len(a)
    t0 = time.monotonic()
    iteration = 0
    stuck = 0
    best_local = a.copy()
    best_local_score = score_fast(a)

    while time.monotonic() - t0 < time_limit:
        a_new, improved, sc = lp_iteration(a)
        if improved:
            a = a_new
            iteration += 1
            stuck = 0
            if sc < best_local_score:
                best_local_score = sc
                best_local = a.copy()
                if verbose and iteration % 5 == 0:
                    print(f"    LP iter={iteration}, score={sc:.6f}")
        else:
            stuck += 1
            if stuck > 5:
                # Perturb to escape
                a = a * np.exp(rng.normal(0, 0.02 + 0.01 * stuck / 10, size=n))
                a = np.clip(a, 0.01, None)
            if stuck > 40:
                break

    return best_local, best_local_score, iteration


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search: LP-based optimization, gradient descent,
    and local search polish to minimize autocorrelation peak.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 30
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Use provided sequence if available
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = score_fast(start_seq)
        print(f"Provided: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # === Phase 1: LP optimization on provided + a few lengths ===
    print("\n=== Phase 1: LP optimization ===")
    lp_lengths = [256, 384, 512]
    if best_seq is not None:
        lp_lengths.insert(0, len(best_seq))
        lp_lengths = list(dict.fromkeys(lp_lengths))  # dedupe preserving order

    for target_n in lp_lengths:
        if time.monotonic() > deadline - 400:
            break
        remaining = deadline - time.monotonic()
        time_per = min(300.0, remaining * 0.35)

        if target_n == len(best_seq) if best_seq is not None else False:
            a0 = best_seq.copy()
        elif best_seq is not None:
            a0 = resample_sequence(best_seq, target_n)
        else:
            a0 = make_chirp_init(target_n, 0.5, rng)

        a0 = np.clip(a0, 0.01, None)
        print(f"  LP on n={target_n}, time_budget={time_per:.0f}s")
        a_best, sc, iters = run_lp_phase(a0, time_per, rng)
        print(f"  n={target_n}: score={sc:.6f}, iterations={iters}")

        if sc < best_score:
            best_score = sc
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 2: Gradient optimization from multiple starts ===
    print("\n=== Phase 2: Gradient optimization ===")
    grad_configs = []
    for n in [256, 384, 512]:
        grad_configs.append((n, 'best'))
        grad_configs.append((n, 'chirp'))
        grad_configs.append((n, 'multitone'))

    for n, init_type in grad_configs:
        if time.monotonic() > deadline - 300:
            break

        if init_type == 'best' and best_seq is not None:
            a0 = resample_sequence(best_seq, n)
        elif init_type == 'chirp':
            a0 = make_chirp_init(n, 0.3 + 0.4 * rng.random(), rng)
        else:
            a0 = make_multitone_init(n, rng)

        a0 = np.clip(a0, 0.01, None)
        grad_time = min(80.0, max(20.0, (deadline - time.monotonic()) * 0.1))
        a_best, sc = gradient_optimize(a0, grad_time, rng, lr_base=3.0)
        print(f"  n={n}, init={init_type}: score={sc:.6f}")

        if sc < best_score:
            best_score = sc
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 3: More LP refinement on best ===
    print("\n=== Phase 3: LP refinement ===")
    if best_seq is not None:
        a = best_seq.copy()
        lp_time = min(400.0, max(60.0, (deadline - time.monotonic()) * 0.5))
        a_best, sc, iters = run_lp_phase(a, lp_time, rng)
        print(f"  LP refinement: score={sc:.6f}, iterations={iters}")
        if sc < best_score:
            best_score = sc
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 4: Local search polish ===
    print("\n=== Phase 4: Local search ===")
    if best_seq is not None:
        polish_time = max(30.0, min(400.0, deadline - time.monotonic() - 10))
        print(f"  Polish time: {polish_time:.0f}s")
        polished, psc = local_search(best_seq, polish_time, rng)
        if psc < best_score:
            best_score = psc
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f} *** New best ***")

    # === Phase 5: Final LP pass if time ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        print("\n=== Phase 5: Final LP pass ===")
        final_time = min(60.0, deadline - time.monotonic() - 10)
        a_best, sc, _ = run_lp_phase(best_seq.copy(), final_time, rng, verbose=False)
        if sc < best_score:
            best_score = sc
            best_seq = a_best.copy()
            print(f"  Final LP: {best_score:.6f} *** New best ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
```
