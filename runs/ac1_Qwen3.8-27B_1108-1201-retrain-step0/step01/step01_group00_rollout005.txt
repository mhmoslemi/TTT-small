"""
Peak-directed gradient optimization with intensive simulated annealing for
minimizing the normalized peak of linear autocorrelation of non-negative sequences.
Uses FFT-based convolution, projected gradient descent with momentum on the
non-negative simplex, and multi-phase search over diverse initializations and
sequence lengths. Includes reheat annealing and peak-targeted perturbations.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def compute_score_normalized(a):
    """Score assuming sum(a) = 1, so score = 2n * max(C)."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def compute_grad_at_peak(peak_idx, a):
    """Compute gradient of C[peak_idx] w.r.t. a. O(n) vectorized."""
    n = len(a)
    k = peak_idx
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    js = np.arange(j_start, j_end + 1)
    grad[js] = 2.0 * a[k - js]
    return grad


def project_to_simplex(a, target_sum=1.0):
    """Project a onto the non-negative simplex with given sum."""
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-15:
        n = len(a)
        return np.ones(n) * target_sum / n
    return a * (target_sum / s)


def gradient_optimize(a_init, time_limit, lr_init=0.01, momentum=0.85,
                      decay=0.9999, min_lr=1e-6, reheat_every=5000):
    """
    Projected gradient descent with momentum on the non-negative simplex.
    Minimizes max of linear autocorrelation.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = project_to_simplex(a, 1.0)
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    step = 0
    no_improve = 0
    target_sum = 1.0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))
        grad = compute_grad_at_peak(peak_idx, a)
        # Project gradient to tangent space of simplex
        grad = grad - np.mean(grad)
        # Momentum step
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = project_to_simplex(a_new, target_sum)

        step += 1
        if step % 1000 == 0:
            score = compute_score_normalized(a_new)
        else:
            score = 2.0 * n * np.max(c)  # approximate (stale)

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive learning rate
        if no_improve > 2000:
            lr *= 0.95
            no_improve = 0
        elif no_improve > 500:
            lr *= 0.99
            v *= 0.9  # dampen momentum

        if lr < min_lr:
            lr = min_lr

        # Reheat: occasionally perturb to escape local minima
        if step % reheat_every == 0 and time.monotonic() - t0 < time_limit * 0.8:
            a = best_a.copy()
            # Small random perturbation
            a = a * np.exp(np.random.standard_normal(n) * 0.05)
            a = project_to_simplex(a, target_sum)
            v = np.zeros(n)
            lr = max(lr, 0.005)
            best_score = compute_score_normalized(a)

    return best_a, best_score


def simulated_annealing(a_init, time_limit, rng, peak_weight=0.5):
    """
    Simulated annealing with peak-directed moves, block moves, and reheat.
    """
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    current = project_to_simplex(current, 1.0)
    current_score = compute_score_normalized(current)
    best = current.copy()
    best_score = current_score

    t0 = time.monotonic()
    temperature = 0.02 / n
    min_temp = 1e-8 / n
    cooling = 0.99995
    reheat_count = 0
    max_reheat = 10
    no_improve = 0
    eval_count = 0

    while time.monotonic() - t0 < time_limit:
        # Check if we should stop early if temperature is too low
        if temperature < min_temp and reheat_count >= max_reheat:
            break

        candidate = current.copy()
        r = rng.random()

        if r < peak_weight:
            # Peak-directed: reduce top contributors to peak, boost others
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            # Compute contributions to peak
            js = np.arange(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
            contrib_idx = peak_idx - js
            contrib_idx = contrib_idx[(contrib_idx >= 0) & (contrib_idx < n)]
            contrib = current[js[peak_idx - js == contrib_idx]] * current[contrib_idx]
            # Simpler: just compute product pairs
            pairs = []
            for j in js:
                if 0 <= peak_idx - j < n:
                    pairs.append((j, peak_idx - j, current[j] * current[peak_idx - j]))
            if pairs:
                pairs.sort(key=lambda x: -x[2])
                top_k = max(3, n // 12)
                top_indices = set()
                for idx, _, _ in pairs[:top_k]:
                    top_indices.add(idx)
                    top_indices.add(peak_idx - idx if 0 <= peak_idx - idx < n else idx)
                top_arr = np.array(list(top_indices))
                candidate[top_arr] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_arr)))
                # Boost non-peak contributors
                other = np.setdiff1d(np.arange(n), top_arr)
                if len(other) > 0:
                    n_boost = min(10, len(other))
                    boost_idx = rng.choice(other, size=n_boost, replace=False)
                    candidate[boost_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_boost))
        elif r < peak_weight + 0.2:
            # Block perturbation
            width = max(1, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width + 1))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        elif r < peak_weight + 0.35:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width + 1))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < peak_weight + 0.45:
            # Symmetrize partially
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.015, size=n))
        elif r < peak_weight + 0.55:
            # Swap-based: exchange mass between peak and off-peak regions
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            center = n // 2
            # Move mass from center outward or from edges inward
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src != dst:
                amount = current[src] * rng.uniform(0.1, 0.3)
                candidate[src] -= amount
                candidate[dst] += amount
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = project_to_simplex(candidate, 1.0)
        new_score = compute_score_normalized(candidate)
        eval_count += 1

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = current.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            accept_prob = np.exp(-(new_score - current_score) / temperature)
            if rng.random() < accept_prob:
                current = candidate
                current_score = new_score

        temperature *= cooling

        # Reheat if stuck
        if no_improve > 5000 and reheat_count < max_reheat:
            temperature = max(temperature, 0.01 / n)
            current = best.copy()
            current = current * np.exp(rng.normal(0.0, 0.1, size=n))
            current = project_to_simplex(current, 1.0)
            current_score = compute_score_normalized(current)
            reheat_count += 1
            no_improve = 0
            if time.monotonic() - t0 < time_limit * 0.1:
                break  # early exit if we're still in first 10%

    return best, best_score


def make_hann_init(n, rng):
    """Hann window initialization."""
    t = np.arange(n) / max(1, n - 1)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_raised_cosine_init(n, rng):
    """Raised cosine with random perturbation."""
    t = np.arange(n) / max(1, n - 1)
    a = 0.5 * (1 + np.cos(np.pi * (2 * t - 1)))
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_gaussian_init(n, rng, sigma_mult=0.4):
    """Gaussian centered initialization."""
    t = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_mult * n / 2.0
    a = np.exp(-0.5 * (t / sigma) ** 2)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_tapered_random_init(n, rng):
    """Random values with tapering at edges."""
    a = rng.uniform(0.5, 1.5, size=n)
    t = np.arange(n) / max(1, n - 1)
    window = 0.5 * (1 + np.cos(np.pi * (2 * t - 1)))
    a = a * (0.3 + 0.7 * window)
    return np.clip(a, 0.001, None)


def make_polyphase_init(n, rng):
    """Based on polyphase / Zadoff-Chu inspired real structure."""
    t = np.arange(n) / n
    a = 1.0 + 0.3 * np.cos(2 * np.pi * t ** 2) + 0.1 * np.cos(4 * np.pi * t ** 2)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)

                c = fft_convolve(current, current)
                peak_idx = int(np.argmax(c))
                js = np.arange(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
                top_k = max(2, n // 20)
                # Find top contributors
                prods = np.array([current[j] * current[peak_idx - j]
                                  for j in js if 0 <= peak_idx - j < n])
                valid_js = [j for j in js if 0 <= peak_idx - j < n]
                if len(prods) > 0:
                    top_local = np.argsort(prods)[-top_k:]
                    top_js = np.array(valid_js)[top_local]
                    candidate[top_js] *= np.exp(rng.normal(-0.015, 0.005, size=len(top_js)))
                    other = np.setdiff1d(np.arange(n), top_js)
                    if len(other) > 3:
                        inc = rng.choice(other, size=min(5, len(other)), replace=False)
                        candidate[inc] *= np.exp(rng.normal(0.015, 0.005, size=len(inc)))
            elif r < 0.8:
                # Small local perturbation
                idx = rng.integers(0, n)
                width = max(1, int(rng.integers(1, 5)))
                start = int(np.clip(idx, 0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
            else:
                # Tiny global
                candidate = current * np.exp(rng.normal(0.0, 0.005, size=n))

            candidate = project_to_simplex(candidate, 1.0)
            new_score = compute_score_normalized(candidate)

            if new_score < current_score:
                current = candidate
                current_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = current.copy()
            elif rng.random() < np.exp(-(new_score - current_score) / (1e-5 / n)):
                current = candidate
                current_score = new_score

        real_score = compute_score(best_local)
        if real_score < best_score:
            best_score = real_score
            best_seq = best_local.copy()
            print(f"  Final polish: {best_score:.6f}")
        else:
            print(f"  Final polish: no improvement (best_local={real_score:.6f}, best={best_score:.6f})")

    # Verify with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        if final_score < best_score:
            best_score = final_score
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
