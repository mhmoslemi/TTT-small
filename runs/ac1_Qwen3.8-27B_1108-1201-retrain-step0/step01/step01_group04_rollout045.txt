def score_of(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)

    c = conv_full(p)
    cmax = float(np.max(c))
    if not np.isfinite(cmax):

def normalize(p):

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (css - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (css[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def softmax_weights(c, T):

        w = np.zeros(m)
        w[int(np.argmax(c))] = 1.0
        return w

    s = e.sum()
    if s <= 0 or not np.isfinite(s):

    return e / s

    Uses FFT for efficiency.
    """
    n = int(p.size)
    Fp = fft(p, L)
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2 : n - 2 : -1]
    if G.size != n:
        tmp = np.zeros(n)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    grad = 2.0 * np.maximum(G, 0.0)
    # Block diagonal: for each peak lag k, H_kk = 2*c_k/n
    # Approximate: 2 * sum of c_k for lags with significant weight
    c = conv_full(p)
    peak_mass = float(np.sum(w * c))  # weighted average of c
    diag = np.full(n, 2.0 * peak_mass / n)
    return grad, diag, w


def hessian_step(p, grad, diag, lr, w, L):
    """One step using block-diagonal Hessian inverse.
    Newton-like: z = D^{-1} grad, then projected step.
    """
    n = int(p.size)
    # D^{-1} * grad
    z = grad / np.maximum(diag, 1e-30)
    # Scale to reasonable step
    z_norm = float(np.sqrt(np.sum(z * z)))
    if z_norm > 1.0:
        z = z * (1.0 / z_norm)
    v = p - lr * z
    return project_simplex(v)


def make_initials(n, rng, parent=None):
    out = []
    if parent is not None:
        pa = np.asarray(parent, dtype=np.float64)
        if pa.size == n and np.all(np.isfinite(pa)) and pa.sum() > 0:
            out.append(normalize(pa))
    out.append(np.full(n, 1.0 / n))
    # Geometric
    for s in (0.3, 0.7, 1.4, 3.0, 6.0, 12.0, 25.0, 50.0):
        x = np.linspace(0, 1, n)
        out.append(normalize(np.exp(-s * x)))
        out.append(normalize(np.exp(-s * (1 - x))))
    # Power
    for a in (-1.5, -0.8, -0.4, 0.3, 1.0, 3.0, 8.0):
        x = np.linspace(1e-5, 1, n)
        out.append(normalize(np.power(x, a)))
        out.append(normalize(np.power(1 - x, a)))
    # Symmetric power (U-shaped)
    for a in (0.5, 1.0, 2.0, 5.0, 15.0):
        x = np.linspace(1e-5, 1 - 1e-5, n)
        out.append(normalize(np.power(x, a) + np.power(1 - x, a)))
    # Beta
    for al, be in [(0.1, 1.0), (0.3, 0.3), (0.5, 0.5), (1.0, 0.1), (1.5, 1.5)]:
        x = np.linspace(1e-5, 1 - 1e-5, n)
        out.append(normalize(np.power(x, al) * np.power(1 - x, be)))
    # Cosine window
    x = np.linspace(0, 1, n)
    out.append(normalize(0.54 + 0.46 * np.cos(2 * np.pi * x)))
    out.append(normalize(np.cos(np.pi * (x - 0.5)) ** 2))
    # Gaussian centered
    for mu, sig in [(0.5, 0.1), (0.5, 0.25), (0.3, 0.15), (0.7, 0.15)]:
        x = np.linspace(0, 1, n)
        out.append(normalize(np.exp(-((x - mu) ** 2) / (2 * sig * sig))))
    # Two-peak
    for m1, m2, s1, s2 in [(0.2, 0.8, 0.05, 0.05), (0.3, 0.7, 0.08, 0.08), (0.1, 0.9, 0.03, 0.03)]:
        x = np.linspace(0, 1, n)
        p = np.exp(-((x - m1) ** 2) / (2 * s1 * s1)) + np.exp(-((x - m2) ** 2) / (2 * s2 * s2))
        out.append(normalize(p))
    # Random smooth
    for _ in range(8):
        x = np.linspace(0, 1, n)
        f1 = rng.integers(1, max(2, n // 100))
        f2 = rng.integers(1, max(2, n // 50))
        ph1 = rng.uniform(0, 2 * np.pi)
        ph2 = rng.uniform(0, 2 * np.pi)
        p = 1.0 + 0.5 * np.sin(2 * np.pi * f1 * x + ph1) + 0.3 * np.sin(2 * np.pi * f2 * x + ph2)
        out.append(normalize(np.maximum(p, 0.01)))
    return out


def lp_best_response(p, best_score, budget_s):
    """LP: max sum(g) s.t. p*g <= M, g >= 0."""
    if not _HAVE_LP:
        return p, best_score
    n = int(p.size)
    if n > LP_NMAX or n < 4:
        return p, best_score
    if budget_s < 1.0:
        return p, best_score
    c = conv_full(p)
    M = float(np.max(c))
    if M <= 0 or not np.isfinite(M):
        return p, best_score
    p_n = normalize(p)
    end = time.monotonic() + budget_s
    try:
        mid = np.arange(n, dtype=np.int32)
        rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel()
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(p_n, n)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=(0, None), method='highs', options={'disp': False})
    except Exception:
        return p, best_score
    if res is None or res.x is None:
        return p, best_score
    g = np.maximum(res.x, 0)
    gs = g.sum()
    if gs <= 1e-12:
        return p, best_score
    g = g / gs
    for t in (1.0, 0.7, 0.5, 0.3, 0.15, 0.05):
        q = project_simplex((1 - t) * p + t * g)
        sc = score_of(q)
        if sc < best_score:
            return q, sc
    return p, best_score


def optimize_core(p0, rng, time_limit, label=""):
    """Main local optimization with Hessian-aware steps and peak-flush."""
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    best_p = p.copy()
    best_score = score_of(p)
    cur_score = best_score
    prev_p = p.copy()
    L = fftlen(3 * n - 2)

    # Adam-style state
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    adam_t = 0
    beta1, beta2, eps_a = 0.85, 0.999, 1e-8

    end = time.monotonic() + max(0.0, float(time_limit))
    since_improve = 0
    iter_no = 0

    # Optional LP at start
    if n <= LP_NMAX and time_limit > 30:
        lp_budget = min(20.0, time_limit * 0.1)
        p_lp, sc_lp = lp_best_response(p, best_score, lp_budget)
        if sc_lp < best_score:
            best_score = sc_lp
            cur_score = sc_lp
            best_p = p_lp.copy()
            p = p_lp.copy()
            prev_p = p_lp.copy()

    while time.monotonic() < end:
        iter_no += 1
        c = conv_full(p)
        cmax = float(np.max(c))
        if cmax <= 0 or not np.isfinite(cmax):
            break

        # Adaptive temperature
        progress = min(1.0, iter_no / max(1, int(time_limit * 3)))
        T_base = 0.5 * (1.0 - progress) + 0.02
        factor = max(0.005, T_base * np.exp(-since_improve * 0.005))
        T = max(factor * cmax, 1e-12)

        w = softmax_weights(c, T)
        n_peak = int(np.sum(w > 1.0 / max(n, 10)))  # number of significant peaks

        # Compute gradient and block diagonal
        grad, diag, w = compute_grad_and_blkc(p, w, L)
        gnorm = float(np.sqrt(np.sum(grad * grad)))

        if gnorm > 1e-15:
            # Adam step
            adam_t += 1
            m_buf = beta1 * m_buf + (1 - beta1) * grad
            v_buf = beta2 * v_buf + (1 - beta2) * grad * grad
            m_hat = m_buf / (1 - beta1 ** adam_t)
            v_hat = v_buf / (1 - beta2 ** adam_t)
            adam_lr = 0.4 / (np.sqrt(v_hat) + eps_a)
            adam_step = -adam_lr * m_hat

            # Hessian step
            hess_lr = 1.0 / max(float(np.max(diag)), 1e-10)
            hess_step = -hess_lr * (grad / np.maximum(diag, 1e-10))

            # Combined step with line search
            best_trial = None
            best_trial_score = cur_score
            since_improve_local = 0

            for blend in (1.0, 0.7, 0.4):  # weight on Hessian step
                for scale in (0.5, 0.2, 0.08, 0.03, 0.01):
                    step = scale * (blend * hess_step + (1 - blend) * adam_step)
                    # Clip step
                    step_max = 0.5 / max(n, 1)
                    step = np.clip(step, -step_max, step_max)
                    trial = project_simplex(p + step)
                    sc = score_of(trial)
                    if sc < best_trial_score - 1e-14:
                        best_trial_score = sc
                        best_trial = trial
                        since_improve_local = 0

            if best_trial is not None:
                if best_trial_score < best_score:
                    best_score = best_trial_score
                    best_p = best_trial.copy()
                    since_improve = 0
                else:
                    since_improve += 1
                prev_p = p.copy()
                p = best_trial
                cur_score = best_trial_score
            else:
                since_improve += 1
                # Peak-flush move
                if since_improve > 30 or iter_no % 20 == 0:
                    c_cur = conv_full(p)
                    peak_indices = np.argsort(c_cur)[::-1][:min(5, n)]
                    flush_applied = False
                    for peak_idx in peak_indices:
                        if c_cur[peak_idx] < cmax * 0.9:
                            continue
                        # Find coordinates contributing to this peak
                        # conv[k] = sum_j p[j]*p[k-j], find j that maximize product
                        j_lo = max(0, int(peak_idx) - n + 1)
                        j_hi = min(n - 1, int(peak_idx))
                        if j_lo >= j_hi:
                            continue
                        j_range = np.arange(j_lo, j_hi + 1)
                        products = p[j_range] * p[int(peak_idx) - j_range]
                        if products.size == 0:
                            continue
                        top_contrib = int(np.argmax(products))
                        hi_j = int(j_range[top_contrib])
                        lo_j = int(peak_idx) - hi_j
                        if hi_j == lo_j:
                            continue
                        # Transfer from higher p to lower p
                        if p[hi_j] > p[lo_j]:
                            src, dst = hi_j, lo_j
                        else:
                            src, dst = lo_j, hi_j
                        for frac in (0.1, 0.05, 0.02, 0.01):
                            delta = frac * p[src]
                            if delta <= 1e-15:
                                continue
                            trial = p.copy()
                            trial[src] -= delta
                            trial[dst] += delta
                            trial = project_simplex(trial)
                            sc = score_of(trial)
                            if sc < cur_score:
                                if sc < best_score:
                                    best_score = sc
                                    best_p = trial.copy()
                                    since_improve = 0
                                else:
                                    since_improve += 1
                                prev_p = p.copy()
                                p = trial
                                cur_score = sc
                                flush_applied = True
                                break
                        if flush_applied:
                            break
                # Random perturbation if stuck
                if not flush_applied and since_improve > 80:
                    sig = 0.3 / n if since_improve < 200 else 0.1 / n
                    trial = project_simplex(p + rng.normal(0, sig, n))
                    sc = score_of(trial)
                    if sc < cur_score:
                        if sc < best_score:
                            best_score = sc
                            best_p = trial.copy()
                            since_improve = 0
                        else:
                            since_improve += 1
                        prev_p = p.copy()
                        p = trial
                        cur_score = sc
                # Cooldown / restart
                if since_improve > 200:
                    mix = 0.6 if since_improve > 400 else 0.3
                    p = project_simplex(mix * best_p + (1 - mix) * p +
                                        rng.normal(0, 0.5 / n, n))
                    cur_score = score_of(p)
                    prev_p = p.copy()
                    since_improve = 0
                    m_buf *= 0.5
                    v_buf *= 0.5

        # Periodic LP
        if n <= LP_NMAX and iter_no % 2000 == 0:
            remaining = end - time.monotonic()
            if remaining > 15:
                p_lp, sc_lp = lp_best_response(p, best_score, min(30.0, remaining * 0.3))
                if sc_lp < best_score:
                    best_score = sc_lp
                    cur_score = sc_lp
                    best_p = p_lp.copy()
                    p = p_lp.copy()
                    prev_p = p_lp.copy()
                    since_improve = 0

        if iter_no % 2000 == 0:
            print(f"{label} iter={iter_no} best={best_score:.8f} cur={cur_score:.8f} "
                  f"n_peak={n_peak} since_imp={since_improve}", flush=True)

    return project_simplex(best_p), best_score


def coarse_to_fine(p_coarse, n_fine):
    """Upsample a sequence from coarse to fine using interpolation."""
    p_coarse = np.asarray(p_coarse, dtype=np.float64)
    nc = int(p_coarse.size)
    n_fine = int(n_fine)
    if nc == n_fine:
        return normalize(p_coarse)
    try:
        if _ndzoom is not None:
            factor = n_fine / nc
            p_fine = _ndzoom(p_coarse, factor, mode='nearest', order=3)
        else:
            x_c = np.linspace(0, 1, nc)
            x_f = np.linspace(0, 1, n_fine)
            p_fine = np.interp(x_f, x_c, p_coarse)
    except Exception:
        x_c = np.linspace(0, 1, nc)
        x_f = np.linspace(0, 1, n_fine)
        p_fine = np.interp(x_f, x_c, p_coarse)
    return normalize(np.maximum(p_fine, 0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for nonnegative step function minimizing evaluate_sequence score."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 12.0)

    # Load parent
    parent = None
    try:
        pa = np.asarray(height_sequence_1, dtype=np.float64)
        pa = np.where(np.isfinite(pa), pa, 0.0)
        pa = np.maximum(pa, 0.0)
        if pa.ndim == 1 and pa.size >= 16 and pa.sum() > 0:
            parent = pa
            print(f"Loaded parent: n={pa.size}, score={score_of(pa):.8f}", flush=True)
    except Exception:
        pass

    # Determine target lengths
    if parent is not None:
        base_n = int(parent.size)
    else:
        base_n = 700

    lengths = sorted(set([
        base_n,
        max(128, base_n // 2),
        base_n * 2,
        min(4096, base_n * 4),
        512,
        1024,
        2048,
    ]))
    lengths = [l for l in lengths if 64 <= l <= 8192]
    if not lengths:
        lengths = [700]

    best_p = None
    best_score = np.inf

    def record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0:
            return
        p = normalize(p)
        if not np.isfinite(sc):
            sc = score_of(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p.copy()
            print(f"  NEW BEST: score={sc:.8f} n={p.size}", flush=True)

    # Phase 1: Quick evaluation of initials across lengths
    print(f"Phase 1: Evaluating initial candidates... (deadline in {deadline - time.monotonic():.0f}s)", flush=True)
    phase1_end = t0 + min(60.0, (deadline - t0) * 0.08)
    candidates = []  # (score, p, n)

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        inits = make_initials(n, rng, parent if parent is not None and parent.size == n else None)
        for p in inits:
            sc = score_of(p)
            candidates.append((sc, p, n))
        candidates.sort(key=lambda x: x[0])
        # Keep top 3 per length
        top = [c for c in candidates if c[2] == n][:3]
        for _, p, _ in top:
            record(p, score_of(p))

    # Phase 2: Coarse-to-fine strategy
    print(f"Phase 2: Coarse-to-fine... (remaining {deadline - time.monotonic():.0f}s)", flush=True)
    if best_p is not None and time.monotonic() < deadline - 60:
        # Try to refine best solution at different lengths
        for n_fine in [l for l in lengths if l > best_p.size][:2]:
            if time.monotonic() > deadline - 60:
                break
            p_interp = coarse_to_fine(best_p, n_fine)
            record(p_interp, score_of(p_interp))

    # Phase 3: Main optimization
    print(f"Phase 3: Local optimization... (remaining {deadline - time.monotonic():.0f}s)", flush=True)
    # Prepare jobs sorted by initial score
    jobs = []
    seen_lengths = set()
    for sc, p, n in candidates:
        if n in seen_lengths and len([j for j in jobs if j[2] == n]) >= 2:
            continue
        if time.monotonic() >= deadline - 120:
            break
        jobs.append((sc, p.copy(), n))
        seen_lengths.add(n)
        if len(jobs) >= 8:
            break

    # Sort by score
    jobs.sort(key=lambda x: x[0])

    # Allocate time
    reserve = min(120.0, max(30.0, (deadline - time.monotonic()) * 0.15))
    n_jobs = max(1, len(jobs))
    available = max(0.0, (deadline - reserve) - time.monotonic())
    time_per_job = max(15.0, available / n_jobs)

    ji = 0
    while ji < len(jobs) and time.monotonic() < deadline - reserve:
        sc_init, p_init, n = jobs[ji]
        remaining = deadline - reserve - time.monotonic()
        if remaining < 15:
            break
        t_alloc = min(time_per_job, remaining)
        label = f"JOB[{n}]"
        p_best, sc_best = optimize_core(p_init, rng, t_alloc, label)
        record(p_best, sc_best)
        # Update job for potential re-optimization
        jobs[ji] = (sc_best, p_best.copy(), n)
        ji += 1

    # Phase 4: Final refinement on best
    if best_p is not None and time.monotonic() < deadline - 20:
        print(f"Phase 4: Final refinement on n={best_p.size}...", flush=True)
        t_final = max(10.0, deadline - 10.0 - time.monotonic())
        p_best, sc_best = optimize_core(best_p, rng, t_final, "FINAL")
        record(p_best, sc_best)

    # Phase 5: Try different length if time remains
    if time.monotonic() < deadline - 60 and best_p is not None:
        print(f"Phase 5: Cross-length try...", flush=True)
        n_alt = None
        for n in lengths:
            if n != best_p.size and 128 <= n <= 4096:
                n_alt = n
                break
        if n_alt is not None:
            p_alt = coarse_to_fine(best_p, n_alt)
            sc_alt = score_of(p_alt)
            record(p_alt, sc_alt)
            t_alt = max(10.0, deadline - 15.0 - time.monotonic())
            if t_alt > 10:
                p_alt, sc_alt = optimize_core(p_alt, rng, t_alt, f"CROSS-{n_alt}")
                record(p_alt, sc_alt)

    if best_p is None:
        n = base_n if parent is None else parent.size
        best_p = np.full(n, 1.0 / n)
        best_score = score_of(best_p)

    # Final safety
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = best_p.sum()
    if s < 1e-10:
        best_p = np.full(max(1, best_p.size), 1.0)

    # Verify with evaluator
    try:
        seq = [float(x) for x in best_p]
        eval_score = evaluate_sequence(seq)
        print(f"FINAL: internal={best_score:.8f}, evaluator={eval_score:.8f}, n={len(best_p)}", flush=True)
        if not np.isfinite(eval_score):
            # Fallback to uniform
            best_p = np.full(max(16, best_p.size), 1.0)
            seq = [float(x) for x in best_p]
            print(f"FALLBACK: evaluator={evaluate_sequence(seq):.8f}", flush=True)
    except Exception as e:
        print(f"Eval check failed: {e}", flush=True)

    print(f"Total time: {time.monotonic() - t0:.1f}s", flush=True)
    return [float(x) for x in best_p]
