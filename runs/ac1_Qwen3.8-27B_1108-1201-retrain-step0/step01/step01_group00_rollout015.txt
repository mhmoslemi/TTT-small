"""
Hybrid optimization for minimizing peak of self-convolution of non-negative sequences.
Uses L-BFGS-B with annealed log-sum-exp smoothing, coordinate descent polish,
and multi-length exploration. Focuses on refining the provided best candidate
with more sophisticated optimization techniques.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_conv(a):
    """Self-convolution."""
    return fft_convolve(a, a)


def compute_score(a):
    """Direct score computation matching evaluate_sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = compute_conv(a)
    return float(2.0 * n * np.max(c) / (s * s))


def soft_max_value_weights(c, tau):
    """Compute soft-max value and weights of array c with temperature tau."""
    m = np.max(c)
    shifted = (c - m) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = m + tau * np.log(sum_exp)
    return value, w


def obj_and_grad(a, n, tau):
    """
    Compute smoothed objective and its gradient.
    Objective: 2n * soft_max(conv(a,a), tau) / (sum(a))^2
    """
    c = compute_conv(a)
    val, w = soft_max_value_weights(c, tau)
    s = np.sum(a)
    if s < 0.01:
        return 1e10, np.zeros_like(a)
    # Gradient of soft_max w.r.t. a[i] = 2 * sum_j w[j+i] * a[j]
    # This equals 2 * conv(a, w)[n-1 : 2n-1]
    corr = fft_convolve(a, w)
    grad_conv = 2.0 * corr[n - 1:2 * n - 1]
    obj = 2.0 * n * val / (s * s)
    grad = 2.0 * n * (grad_conv * s - 2.0 * val) / (s ** 3)
    return obj, grad


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def l_bfgs_run(a0, n, tau, maxiter=500, ftol=1e-12):
    """Run L-BFGS-B with sum constraint and non-negativity bounds."""
    S = np.sqrt(2.0 * n)
    a_start = a0.copy()
    a_start = np.clip(a_start, 1e-10, None)
    a_start = a_start / np.sum(a_start) * S

    def fun(a):
        v, _ = obj_and_grad(a, n, tau)
        return v

    def jac(a):
        _, g = obj_and_grad(a, n, tau)
        return g

    def cons_fun(a):
        return np.sum(a) - S

    def cons_jac(a):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(1e-10, 1000.0)] * n

    result = minimize(fun, a_start, jac=jac, method='L-BFGS-B',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-7,
                               'maxls': 50})
    return result.x


def coordinate_descent_polish(a, n, time_limit, rng, tol=1e-10):
    """
    Coordinate descent: for each element, try small multiplicative changes
    and accept if score improves. Cycles through elements.
    """
    a = a.copy()
    current_score = compute_score(a)
    best = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    n = len(a)
    S = np.sum(a)

    multipliers = [0.99, 0.97, 0.94, 0.90, 0.85, 1.0/0.85, 1.0/0.90, 1.0/0.94, 1.0/0.97, 1.0/0.99]

    stagnant = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Random permutation of indices for diversity
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for mult in multipliers:
                candidate = a.copy()
                candidate[i] *= mult
                # Renormalize to maintain sum
                new_sum = np.sum(candidate)
                if new_sum < 0.01:
                    continue
                new_score = compute_score(candidate)
                if new_score < current_score - 1e-12:
                    a = candidate
                    current_score = new_score
                    improved = True
                    if current_score < best_score:
                        best_score = current_score
                        best = a.copy()
                    break
        if not improved:
            stagnant += 1
            if stagnant > 3:
                break
        else:
            stagnant = 0

    return best, best_score


def peak_targeted_search(a, n, time_limit, rng):
    """
    Targeted search that focuses on reducing the convolution peak.
    Identifies peak contributors and makes targeted adjustments.
    """
    a = a.copy()
    current_score = compute_score(a)
    best = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    S = np.sum(a)

    while time.monotonic() - t0 < time_limit:
        c = compute_conv(a)
        peak_val = np.max(c)
        # Find all near-peak indices
        threshold = peak_val * 0.95
        peak_indices = np.where(c >= threshold)[0]

        # For each peak index, find top contributors
        for pk in peak_indices[:5]:  # limit to 5 peaks
            contribs = np.zeros(n)
            for i in range(n):
                j = pk - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            # Sort by contribution
            top_idx = np.argsort(contribs)[-max(3, n // 12):]
            bottom_idx = np.argsort(contribs)[:max(3, n // 12)]

            # Try reducing top contributors
            candidate = a.copy()
            for i in top_idx:
                candidate[i] *= (1.0 - 0.01 * rng.random())
            if np.sum(candidate) > 0.01:
                new_score = compute_score(candidate)
                if new_score < current_score:
                    a = candidate
                    current_score = new_score
                    if current_score < best_score:
                        best_score = current_score
                        best = a.copy()
                        break

            # Try increasing bottom contributors (redistribute)
            candidate = a.copy()
            total_decrease = 0.0
            for i in top_idx[:3]:
                dec = a[i] * 0.01
                candidate[i] -= dec
                total_decrease += dec
            if total_decrease > 0:
                per_inc = total_decrease / len(bottom_idx)
                for i in bottom_idx:
                    candidate[i] += per_inc
                if np.all(candidate >= 0) and np.sum(candidate) > 0.01:
                    new_score = compute_score(candidate)
                    if new_score < current_score:
                        a = candidate
                        current_score = new_score
                        if current_score < best_score:
                            best_score = current_score
                            best = a.copy()
                            break

    return best, best_score


def block_optimization(a, n, time_limit, rng):
    """
    Optimize blocks of consecutive elements. For each block, try
    multiplicative adjustments to all elements in the block simultaneously.
    """
    a = a.copy()
    current_score = compute_score(a)
    best = a.copy()
    best_score = current_score
    t0 = time.monotonic()

    block_sizes = [2, 4, 8, 16, 32]

    while time.monotonic() - t0 < time_limit:
        bs = block_sizes[rng.integers(0, len(block_sizes))]
        start = rng.integers(0, max(1, n - bs))
        end = min(start + bs, n)

        # Generate random direction for this block
        direction = rng.standard_normal(bs)
        scale = 0.02

        for attempt in range(10):
            step = scale * (0.5 ** attempt)
            candidate = a.copy()
            candidate[start:end] = a[start:end] * np.exp(step * direction)
            candidate = np.clip(candidate, 1e-10, 1000.0)

            if np.sum(candidate) < 0.01:
                continue
            new_score = compute_score(candidate)
            if new_score < current_score:
                a = candidate
                current_score = new_score
                if current_score < best_score:
                    best_score = current_score
                    best = a.copy()
            else:
                break

    return best, best_score

        start_seq = np.ones(256)

    n = len(start_seq)
    best_score = compute_score(start_seq)
    best_a = start_seq.copy()
    print(f"Initial: n={n}, score={best_score:.6f}")

    # Phase 1: L-BFGS-B with annealed temperature (50% of budget)
    phase1_time = 0.50 * time_budget
    print(f"\n=== Phase 1: L-BFGS-B optimization (budget: {phase1_time:.0f}s) ===")
    t_phase = time.monotonic()

    # Temperature schedule: start high (smooth), end low (sharp)
    tau_values = [20.0/n, 10.0/n, 5.0/n, 3.0/n, 2.0/n, 1.5/n, 1.0/n, 0.7/n, 0.5/n, 0.3/n]

    current_a = start_seq.copy()
    for idx, tau in enumerate(tau_values):
        if time.monotonic() - t_phase > phase1_time - 10:
            break
        remaining = phase1_time - (time.monotonic() - t_phase)
        maxiter = max(50, int(remaining * 200))  # adaptive max iterations

        print(f"  tau={tau:.4f} (n={n}), maxiter={maxiter}")
        try:
            result_a = l_bfgs_run(current_a, n, tau, maxiter=maxiter)
            result_a = np.clip(result_a, 0.0, 1000.0)
            result_score = compute_score(result_a)
            print(f"    L-BFGS result: score={result_score:.6f}, converged")

            if result_score < best_score:
                best_score = result_score
                best_a = result_a.copy()
                current_a = result_a.copy()
                print(f"    *** New best: {best_score:.6f} ***")
            else:
                # Keep current_a as is if no improvement
                pass
        except Exception as e:
            print(f"    L-BFGS error: {e}")
            continue

        # Small perturbation to escape potential local minima between temperature changes
        if idx < len(tau_values) - 1 and time.monotonic() - t_phase < phase1_time - 30:
            noise_scale = 0.005 * (1.0 - idx / len(tau_values))
            current_a = current_a * np.exp(rng.standard_normal(n) * noise_scale)
            current_a = np.clip(current_a, 0.01, 1000.0)

    print(f"  Phase 1 best: {best_score:.6f}")

    # Phase 2: Peak-targeted search + block optimization (25% of budget)
    phase2_time = 0.25 * time_budget
    print(f"\n=== Phase 2: Peak-targeted & block search (budget: {phase2_time:.0f}s) ===")
    t_phase = time.monotonic()

    # Alternate between peak-targeted and block optimization
    sub_budget = phase2_time / 3.0
    for method_idx in range(3):
        if time.monotonic() - t_phase > phase2_time:
            break
        t_sub = time.monotonic()
        sub_time = min(sub_budget, phase2_time - (time.monotonic() - t_phase))

        if method_idx == 0:
            print(f"  Peak-targeted search...")
            result_a, result_score = peak_targeted_search(best_a, n, sub_time, rng)
        elif method_idx == 1:
            print(f"  Block optimization...")
            result_a, result_score = block_optimization(best_a, n, sub_time, rng)
        else:
            print(f"  More L-BFGS at low temperature...")
            tau = 0.5 / n
            try:
                result_a = l_bfgs_run(best_a, n, tau, maxiter=300)
                result_a = np.clip(result_a, 0.0, 1000.0)
                result_score = compute_score(result_a)
            except:
                result_a, result_score = best_a, best_score

        if result_score < best_score:
            best_score = result_score
            best_a = result_a.copy()
            print(f"    *** New best: {best_score:.6f} ***")
        else:
            print(f"    No improvement (score={result_score:.6f})")

    print(f"  Phase 2 best: {best_score:.6f}")

    # Phase 3: Try different sequence lengths (15% of budget)
    phase3_time = 0.15 * time_budget
    print(f"\n=== Phase 3: Multi-length exploration (budget: {phase3_time:.0f}s) ===")
    t_phase = time.monotonic()

    target_lengths = [240, 256, 272, 288, 300, 320, 384, 512]
    # Prioritize lengths near the current best
    target_lengths.sort(key=lambda x: abs(x - n))

    for target_n in target_lengths:
        if time.monotonic() - t_phase > phase3_time - 10:
            break
        if target_n == n:
            continue  # already explored

        remaining = phase3_time - (time.monotonic() - t_phase)
        if remaining < 15:
            break

        # Resample best sequence to target length
        a_resampled = resample_sequence(best_a, target_n)
        a_resampled = np.clip(a_resampled, 0.01, 1000.0)

        # Quick L-BFGS optimization
        tau = 3.0 / target_n
        try:
            result_a = l_bfgs_run(a_resampled, target_n, tau, maxiter=200)
            result_a = np.clip(result_a, 0.0, 1000.0)
            result_score = compute_score(result_a)
            print(f"  n={target_n}: score={result_score:.6f}")

            if result_score < best_score:
                best_score = result_score
                best_a = result_a.copy()
                n = target_n
                print(f"    *** New best: {best_score:.6f} at n={n} ***")
        except Exception as e:
            print(f"  n={target_n}: error {e}")

    print(f"  Phase 3 best: {best_score:.6f} at n={n}")

    # Phase 4: Final coordinate descent polish (10% of budget)
    phase4_time = 0.10 * time_budget
    print(f"\n=== Phase 4: Coordinate descent polish (budget: {phase4_time:.0f}s) ===")
    t_phase = time.monotonic()

    polished_a, polished_score = coordinate_descent_polish(
        best_a, len(best_a), phase4_time, rng)

    if polished_score < best_score:
        best_score = polished_score
        best_a = polished_a.copy()
        print(f"  After coord descent: {best_score:.6f}")
    else:
        print(f"  Coord descent: no improvement ({polished_score:.6f})")

    # Final: a few more L-BFGS iterations at very low temperature
    if time.monotonic() < deadline - 10:
        tau_final = 0.2 / len(best_a)
        try:
            result_a = l_bfgs_run(best_a, len(best_a), tau_final, maxiter=200)
            result_a = np.clip(result_a, 0.0, 1000.0)
            result_score = compute_score(result_a)
            if result_score < best_score:
                best_score = result_score
                best_a = result_a.copy()
                print(f"  Final L-BFGS: {best_score:.6f}")
        except:
            pass

    # Final cleanup: ensure all values are valid
    best_a = np.clip(best_a, 0.0, 1000.0)
    final_score = compute_score(best_a)
    print(f"\n=== FINAL: score={final_score:.6f}, n={len(best_a)} ===")
    print(f"  Sum={np.sum(best_a):.4f}, Max element={np.max(best_a):.4f}, Min element={np.min(best_a):.6f}")

    # Verify with evaluate_sequence
    final_list = [float(x) for x in best_a]
    eval_score = evaluate_sequence(final_list)
    print(f"  evaluate_sequence: {eval_score:.6f}")

    return final_list
