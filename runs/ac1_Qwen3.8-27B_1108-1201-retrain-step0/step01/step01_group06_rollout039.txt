```python
    C = np.cumsum(p)

    # Part 2: cross-correlation h[m] = sum_j a[j] * p[j+m]
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam-based gradient descent
with softmax-smoothed max and FFT-based gradient computation.
The gradient of the auto-convolution has a simple cross-correlation structure,
enabling efficient O(n log n) gradient steps.
"""

import numpy as np
import time


def auto_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_gradient_ffT(a, p):
    """
    Compute gradient g[m] = 2 * sum_j a[j] * p[j+m].
    This is 2 * cross_correlation(a, p).
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    corr = np.fft.irfft(A * P, fft_size)
    return 2.0 * corr[n-1:2*n-1]


def softmax(x, tau):
    """Stable softmax with temperature."""
    x = x / tau
    x = x - x.max()
```

```python
def auto_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)

    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_gradient(a, p):
    """
    Compute gradient g[m] = 2 * sum_j a[j] * p[j+m].
    Uses FFT-based cross-correlation for O(n log n) complexity.
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
```

```python
    print(f"\nDeep optimization on top {top_k}:")
    for i in range(top_k):
        if time.monotonic() > deadline - 60:
            break
        score_i, n_i, a_i, _ = all_results[i]
        t_rem = deadline - time.monotonic() - 60
        t_alloc = min(60.0, t_rem / max(1, top_k - i))
        if t_alloc < 5:
            break

        # Multiple parameter variations
        for variant in range(3):
            if time.monotonic() > deadline - 60:
                break
            t_v = t_alloc / 3.0
            if t_v < 3:
                break

            tau_s_v = [5.0, 10.0, 2.0][variant]
            tau_e_v = [150.0, 300.0, 50.0][variant]
            lr_v = [0.03, 0.05, 0.02][variant]

            a_opt, obj_opt = optimize_single(
                a_i, time_limit=t_v,
                tau_start=tau_s_v, tau_end=tau_e_v,
                lr=lr_v, seed=seed + 1000 + i * 10 + variant
            )
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep n={n_i} v={variant}: score={score:.6f}")

    # Final polish
    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_rem = deadline - 10 - time.monotonic()
        n_final = len(best_sequence)
        for v in range(3):
            if t_rem < 5:
                break
            t_v = t_rem / 3.0
            params = [
                (5.0, 200.0, 0.03),
                (10.0, 400.0, 0.05),
                (2.0, 80.0, 0.02),
            ]
            ts, te, lrp = params[v]
            a_opt, obj_opt = optimize_single(
                best_sequence, time_limit=t_v,
                tau_start=ts, tau_end=te, lr=lrp,
                seed=seed + 2000 + v
            )
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish v={v}: score={score:.6f}")
            t_rem = deadline - 10 - time.monotonic()

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f}")
    return [float(x) for x in best_sequence]
```
