import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_objective_grad(a, p):
    """Compute objective and gradient using power approximation."""
    n = len(a)
    b = fast_convolve(a)
    b = np.maximum(b, 1e-30)  # avoid issues with 0^0
    
    if p == 1:
        return 2.0 * n * b.sum(), np.ones(n) * 2.0 * n
    else:
        b_p_sum = np.sum(b ** p)
        obj = 2.0 * n * b_p_sum ** (1.0 / p)
        
        # Gradient
        c = b ** (p - 1)
        a_rev = a[::-1]
        # Compute a_rev * c via FFT
        conv_size = 2 * n - 1
        fft_size = 1 << (conv_size - 1).bit_length()
        Ac = np.fft.rfft(a_rev, fft_size)
        Cc = np.fft.rfft(c, fft_size)
        conv_ac = np.fft.irfft(Ac * Cc, fft_size)[:conv_size]

        return obj, grad

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s

    best_obj = compute_objective(a, 1000)[0]  # Use high p for actual objective
    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:

        a_new = project_simplex(a_new)

- `sum(a)` changes by `delta`

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (S * S)

def compute_power_obj_grad(a, p):
    """Compute power-approximation objective and gradient (assuming sum(a)=1)."""
    n = len(a)
    b = fast_convolve(a)
    b = np.maximum(b, 1e-30)
    
    b_p = b ** p
    f = b_p.sum()
    obj = 2.0 * n * f ** (1.0 / p)
    
    # Gradient: d(obj)/da_i = 4*n * f^(1/p - 1) * (a_rev * c)[n-1+i]
    # where c = b^(p-1)
    if p == 1:
        # d(sum b)/da_i = 2 * sum_k a_{k-i} (for valid k) = 2 * S = 2 (since sum=1)
        # Actually: db_k/da_i = 2*a_{k-i}, so sum_k db_k/da_i = 2*sum_m a_m = 2
        # d(obj)/da_i = 2n * 2 = 4n... wait that's not right either
        # obj = 2n * sum(b) = 2n * (sum a)^2 = 2n (constant!)
        # So gradient is 0. This makes sense - for p=1, the objective is constant on the simplex.
        grad = np.zeros(n)
    else:
        c = b ** (p - 1)
        a_rev = a[::-1]
        conv_size = 2 * n - 1
        fft_size = 1 << (conv_size - 1).bit_length()
        Ac = np.fft.rfft(a_rev, fft_size)
        Cc = np.fft.rfft(c, fft_size)
        conv_ac = np.fft.irfft(Ac * Cc, fft_size)[:conv_size]
        f_pow = f ** (1.0 / p - 1.0)
        grad = 4.0 * n * f_pow * conv_ac[n-1:2*n-1]
    
    return obj, grad

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

import numpy as np
from scipy.optimize import minimize_scalar, minimize

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_true_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (S * S)


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def power_obj_grad(a, p):
    """
    Compute power-approximation objective and gradient.
    obj = 2n * (sum_k b_k^p)^(1/p), where b = conv(a,a)
    Assumes sum(a) = 1.
    """
    n = len(a)
    b = fast_convolve(a)
    b_safe = np.maximum(b, 1e-30)

    if p <= 1:
        return 2.0 * n * b_safe.sum(), np.zeros(n)

    b_p = b_safe ** p
    f = b_p.sum()
    obj = 2.0 * n * f ** (1.0 / p)

    # Gradient: d(obj)/da_i = 4*n * f^(1/p - 1) * (a_rev * c)[n-1+i]
    # where c_k = b_k^(p-1)
    c = b_safe ** (p - 1)
    a_rev = a[::-1]
    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    Ac = np.fft.rfft(a_rev, fft_size)
    Cc = np.fft.rfft(c, fft_size)
    conv_ac = np.fft.irfft(Ac * Cc, fft_size)[:conv_size]

    f_pow = f ** (1.0 / p - 1.0)
    grad = 4.0 * n * f_pow * conv_ac[n - 1:2 * n - 1]

    return obj, grad


def optimize_gradient(a0, time_limit, p_schedule=None, lr_init=0.05):
    """
    Power-continuation projected gradient optimization.
    a0: initial point (will be normalized to simplex)
    Returns optimized point (on simplex).
    """
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s

    n = len(a)
    if p_schedule is None:
        p_schedule = [5, 10, 20, 50, 100, 200]

    best_a = a.copy()
    best_true = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    total_time = time_limit

    for p_idx, p in enumerate(p_schedule):
        remaining = total_time - (time.monotonic() - t0)
        if remaining < 3:
            break
        # Allocate time proportional to remaining stages
        frac = 1.0 / max(1, len(p_schedule) - p_idx)
        p_time = remaining * frac * 1.2  # slight overlap allowed
        p_deadline = time.monotonic() + p_time

        lr = lr_init / (1.0 + 0.05 * p)
        min_lr = 1e-5
        max_iter = 5000

        for it in range(max_iter):
            if time.monotonic() > p_deadline:
                break

            obj, grad = power_obj_grad(a, p)
            if not np.all(np.isfinite(grad)):
                lr *= 0.5
                continue

            # Gradient step with projection
            a_new = a - lr * grad
            a_new = project_simplex(a_new)

            # Evaluate true objective
            true_obj = 2.0 * n * fast_convolve(a_new).max()
            if true_obj < best_true:
                best_true = true_obj
                best_a = a_new.copy()

            # Adaptive learning rate
            obj_new, _ = power_obj_grad(a_new, p)
            if np.isfinite(obj_new):
                if obj_new > obj * 1.001 + 1e-10:
                    lr = max(min_lr, lr * 0.7)
                elif obj_new < obj * 0.999 - 1e-10:
                    lr = min(0.5, lr * 1.03)
            a = a_new

            # Early stop if converged
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break

    return best_a, best_true


def optimize_peak_flatten(a0, time_limit, k_peaks=8):
    """
    Directly flatten top-K peaks of the convolution.
    Uses gradient of sum of squared deviations of top peaks.
    """
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s

    n = len(a)
    best_a = a.copy()
    best_true = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.02

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        top_k = np.argsort(b)[::-1][:k_peaks]
        target = b[top_k].mean()

        # Gradient of sum_j (b_{top_j} - target)^2 w.r.t. a
        grad = np.zeros(n)
        for k_int in top_k:
            k = int(k_int)
            diff = b[k] - target
            if abs(diff) < 1e-12:
                continue
            # db_k/da_i = 2*a_{k-i} for i in valid range
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                grad[i_start:i_end + 1] += 4.0 * diff * a[k - i_end:k - i_start + 1][::-1]

        if np.linalg.norm(grad) < 1e-14:
            break

        a_new = project_simplex(a - lr * grad)
        obj_new = 2.0 * n * fast_convolve(a_new).max()

        if obj_new < best_true:
            best_true = obj_new
            best_a = a_new.copy()

        if obj_new >= 2.0 * n * b.max():
            lr *= 0.7
            a_new = project_simplex(a - lr * grad)
            obj_new = 2.0 * n * fast_convolve(a_new).max()
            if obj_new >= 2.0 * n * b.max():
                lr *= 0.5
                if lr < 1e-8:
                    break
                a_new = project_simplex(a - lr * grad)
        else:
            lr = min(0.1, lr * 1.02)

        a = a_new

    return best_a, best_true


def coordinate_descent(a0, time_limit, max_passes=500):
    """
    Fast coordinate descent with incremental convolution updates.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-12:
        return a
    a /= S
    S = 1.0

    b = fast_convolve(a)
    best_obj = 2.0 * n * b.max()
    best_a = a.copy()
    t0 = time.monotonic()

    for outer in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            # Search range for a[i]
            lower = 0.0
            upper = max(0.1, ai * 3.0 + 0.01)

            if ai < 1e-14:
                # Try a few small values
                candidates = [0.001, 0.003, 0.01, 0.03]
                best_val = best_obj
                best_c = ai
                for c in candidates:
                    if c > 1.0:
                        continue
                    a[i] = c
                    b_new = b.copy()
                    delta = c
                    for k in range(i, min(i + n, 2 * n - 1)):
                        if k == 2 * i:
                            b_new[k] += delta * (2.0 * ai + delta)
                        else:
                            b_new[k] += 2.0 * delta * a[k - i]
                    val = 2.0 * n * b_new.max()
                    if val < best_val - 1e-13:
                        best_val = val
                        best_c = c
                if best_c != ai:
                    a[i] = best_c
                    b = fast_convolve(a)
                    improved = True
                    if 2.0 * n * b.max() < best_obj:
                        best_obj = 2.0 * n * b.max()
                        best_a = a.copy()
                continue

            # Use minimize_scalar with fast evaluation
            def obj_of_delta(delta):
                new_ai = ai + delta
                if new_ai < -1e-14:
                    return 1e10
                new_ai = max(0.0, new_ai)
                delta_actual = new_ai - ai
                if abs(delta_actual) < 1e-14:
                    return 2.0 * n * b.max()

                b_new = b.copy()
                for k in range(i, min(i + n, 2 * n - 1)):

                    else:
                        b_new[k] += 2.0 * delta_actual * a[k - i]
                return 2.0 * n * b_new.max()

            res = minimize_scalar(obj_of_delta, bounds=(-ai * 0.95, upper - ai),
                                  method='bounded', options={'xatol': 1e-10})

            if res.success and res.fun < best_obj - 1e-13:
                a[i] = max(0.0, ai + res.x)

                improved = True
                cur_obj = 2.0 * n * b.max()
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()

        if not improved:
            break

        if (outer + 1) % 20 == 0:
            print(f"    CD pass {outer+1}: obj={best_obj:.8f}")

def generate_initializations(n, rng):

        if t_alloc < 1.0:
            break

        # Quick gradient optimization
        a_opt, _ = optimize_gradient(a0, t_alloc * 0.6,
                                     p_schedule=[10, 30, 80], lr_init=0.04)
        # Quick coordinate descent
        a_cd = coordinate_descent(a_opt, t_alloc * 0.4, max_passes=50)

        score1 = float(evaluate_sequence(a_opt.tolist()))
        score2 = float(evaluate_sequence(a_cd.tolist()))
        best_local = min(score1, score2)
        best_local_seq = a_cd if score2 <= score1 else a_opt

        results.append((best_local, n, name, best_local_seq))

        if np.isfinite(best_local) and best_local < best_score:
            best_score = best_local
            best_sequence = best_local_seq.copy()
            print(f"  [{idx+1:3d}/{n_configs}] n={n:3d} {name:15s}: {best_local:.6f}")

    # Phase 2: Deep optimization on top candidates
    print(f"Phase 2: Deep optimization. Best: {best_score:.6f}")
    results.sort(key=lambda x: x[0])

    # Select top candidates with diversity
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in results:
        if not np.isfinite(score):
            continue
        if len(top_configs) < 20:
            top_configs.append((n, name, a_opt, score))
            seen_lengths.add(n)
        elif n not in seen_lengths and len(top_configs) < 28:
            top_configs.append((n, name, a_opt, score))
            seen_lengths.add(n)

    # Always include current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'current_best', best_sequence.copy(), best_score))

    phase2_end = t_start + budget_s * 0.75
    time_per2 = max(15.0, (phase2_end - time.monotonic()) / max(1, len(top_configs)))

    for i, (n, name, a0, prev_score) in enumerate(top_configs):
        if time.monotonic() > phase2_end:
            break
        t_alloc = min(time_per2, phase2_end - time.monotonic() - 5)
        if t_alloc < 8:
            break

        # Multi-strategy deep optimization
        # Strategy 1: Longer gradient continuation
        a1, _ = optimize_gradient(a0, t_alloc * 0.35,
                                  p_schedule=[5, 10, 20, 50, 100, 200, 500],
                                  lr_init=0.03)
        # Strategy 2: Peak flattening
        a2, _ = optimize_peak_flatten(a0, t_alloc * 0.2, k_peaks=10)
        # Strategy 3: Coordinate descent on best of above
        a_best = a1 if compute_true_objective(a1) <= compute_true_objective(a2) else a2
        a3 = coordinate_descent(a_best, t_alloc * 0.3, max_passes=200)
        # Strategy 4: One more gradient pass on the CD result
        a4, _ = optimize_gradient(a3, t_alloc * 0.15,
                                  p_schedule=[100, 300], lr_init=0.02)

        for a_try, tag in [(a1, 'grad'), (a2, 'flatten'), (a3, 'cd'), (a4, 'grad2')]:
            score = float(evaluate_sequence(a_try.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_try.copy()
                print(f"  Deep [{i}] n={n} {name} ({tag}): {score:.6f}")

    # Phase 3: Final intensive polish
    print(f"Phase 3: Final polish. Best: {best_score:.6f}")
    if best_sequence is not None and time.monotonic() < deadline:
        t_remaining = deadline - time.monotonic()

        # Extended coordinate descent
        a_cd = coordinate_descent(best_sequence, t_remaining * 0.35, max_passes=2000)
        score = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_cd.copy()
            print(f"  Final CD: {best_score:.6f}")

        # Gradient + CD cycle
        if time.monotonic() < deadline - 10:
            t_rem = deadline - time.monotonic() - 10
            for p_sched in [[200, 500, 1000], [100, 300, 1000, 2000]]:
                if time.monotonic() > deadline - 10:
                    break
                t_each = (deadline - time.monotonic() - 10) / 2.0
                a_g = optimize_gradient(best_sequence, t_each * 0.6,
                                        p_schedule=p_sched, lr_init=0.015)
                a_final = coordinate_descent(a_g, t_each * 0.4, max_passes=500)
                score = float(evaluate_sequence(a_final.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_final.copy()
                    print(f"  Grad+CD (p={p_sched}): {best_score:.6f}")

        # Peak flattening on final
        if time.monotonic() < deadline - 5:
            a_pf = optimize_peak_flatten(best_sequence, deadline - time.monotonic() - 5,
                                         k_peaks=15)
            a_pf = coordinate_descent(a_pf, 3, max_passes=100)
            score = float(evaluate_sequence(a_pf.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_pf.copy()
                print(f"  Final flatten: {best_score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f} (length={len(best_sequence)})")
    return best_sequence


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Finds a non-negative sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization
    with power-continuation gradients, peak flattening, and
    coordinate descent.
    """
    result = run_full_optimization(seed=seed, budget_s=budget_s)
    return [float(max(0.0, x)) for x in result]
