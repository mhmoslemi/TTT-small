import numpy as np

try:

except Exception:
    from numpy.fft import fft, ifft
    def next_fast_len(n):
        n = int(max(1, n))
        m = 1
        while m < n:
            m <<= 1
        return m

try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

except Exception:

def convolve_full(p):
    """Full linear self-convolution via FFT."""
    n = int(p.size)
    if n == 0:
        return np.empty(0)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def internal_score(p):
    """Compute the evaluation score for a normalized non-negative sequence."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if s < 1e-12:
        return np.inf
    c = convolve_full(p)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (s * s))


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s < 1e-30:
        return np.full(n, 1.0 / n)
    return out / s


def safe_normalize(p):
    """Normalize to sum=1, clip to non-negative."""
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s < 1e-12:
        return np.full(max(1, p.size), 1.0 / max(1, p.size))
    return p / s


def lp_best_response_active(p, M, K=40, time_limit=30.0):
    """
    LP best-response using only the top-K tightest constraints.
    Solves: max sum(g) s.t. (p*g)[k] <= M for k in active set, g >= 0.
    Returns g (normalized to sum=1) or None on failure.
    """
    if not _HAVE_LP:
        return None
    n = int(p.size)
    if n < 4 or n > 12000:
        return None
    p = np.asarray(p, dtype=np.float64)
    fs = float(p.sum())
    if fs < 1e-12:
        return None
    p = p / fs

    end = time.monotonic() + max(0.1, time_limit)
    if end - time.monotonic() < 1.0:
        return None

    # Find active constraints: top-K lags with highest p*p values
    c = convolve_full(p)
    nc = c.size  # 2n-1
    K = min(K, nc)
    active_idx = np.argsort(c)[::-1][:K]

    try:
        rows = []
        cols_data = []
        for k in active_idx:
            # conv(p, g)[k] = sum_i p[i] * g[k-i] where 0<=i<n, 0<=k-i<n
            # So g[j] appears with coefficient p[k-j] for j in [max(0,k-n+1), min(n-1,k)]
            j_lo = max(0, int(k) - n + 1)
            j_hi = min(n - 1, int(k))
            for j in range(j_lo, j_hi + 1):
                rows.append(int(k) - j_lo)  # local row index
                cols_data.append((j, float(p[int(k) - j])))

        # Build sparse matrix
        row_idx = []
        col_idx = []
        data = []
        for k_pos, k in enumerate(active_idx):
            j_lo = max(0, int(k) - n + 1)
            j_hi = min(n - 1, int(k))
            for j in range(j_lo, j_hi + 1):
                row_idx.append(k_pos)
                col_idx.append(j)
                data.append(float(p[int(k) - j]))

        A = csr_matrix((np.array(data, dtype=np.float64),
                        (np.array(row_idx, dtype=np.int32),
                         np.array(col_idx, dtype=np.int32))),
                       shape=(K, n))
        b_ub = np.full(K, M, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)

        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=(0.0, None), method='highs',
                      options={'disp': False, 'time_limit': max(1.0, end - time.monotonic())})

    except Exception:
        return None

    if res is None or not res.success:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s < 1e-12:
        return None
    return g / s


def gradient_smoothed_max(p, T_factor=0.5):
    """
    Compute gradient of smoothed max of conv(p,p).
    Returns (gradient, smoothed_max_value).
    """
    n = int(p.size)
    c = convolve_full(p)
    nc = c.size
    cmax = float(np.max(c))
    if cmax < 1e-30:
        return np.zeros(n), 0.0

    T = max(T_factor * cmax, 1e-20)
    diff = (c - cmax) / T
    # Clip for numerical stability
    diff = np.clip(diff, -50, 0)
    e = np.exp(diff)
    s = float(e.sum())
    if s < 1e-30:
        e = np.zeros(nc)
        e[int(np.argmax(c))] = 1.0
        s = 1.0
    w = e / s
    surr = cmax + T * np.log(s)

    # Gradient: d(c_k)/d(p_j) = p[k-j] + p[j-k] (for valid indices)
    # grad_j = sum_k w_k * (p[k-j] + p[j-k])... 
    # Actually conv(p,p)[k] = sum_i p[i] p[k-i]
    # d/dp_j conv[k] = p[k-j] + p[j-k] if both valid, else just one term
    # By symmetry of conv: d/dp_j conv[k] = p[(k-j) mod ...] 
    # Simpler: conv = ifft(fft(p)^2), so gradient = 2 * ifft(fft(p) * fft(w_reversed))
    # But let's do it directly for correctness.
    # grad_j = sum_k w_k * (p[k-j] if 0<=k-j<n else 0) + w_k * (p[j-k+n-1] if ...)
    # Actually for linear conv: conv[k] = sum_{i} p[i] p[k-i] (valid range)
    # d conv[k] / d p[j] = p[k-j] (if 0<=k-j<n) + p[j-k+n-1]... 
    # No: d/dp[j] sum_i p[i] p[k-i] = (if j in range and k-j in range) p[k-j]
    #   + (if k-j in range and j in range) ... 
    # It's p[k-j] if 0 <= k-j < n, else 0. Just one term since the other index is fixed.
    # Wait: sum_i p[i] p[k-i]. If i=j, term is p[j] p[k-j]. Derivative w.r.t. p[j] is p[k-j].
    # Also if k-i=j, i.e., i=k-j, term is p[k-j] p[j]. Derivative w.r.t. p[j] is p[k-j].
    # So if j != k-j (i.e., j != k/2), we get 2*p[k-j].
    # If j == k/2, we get p[j] (only one term).
    # For simplicity and since n is large, the factor of 2 vs 1 doesn't matter much.
    # Use FFT: ifft(fft(p, L) * conj(fft(w_padded, L))) gives cross-correlation.
    # Actually, the gradient of sum_k w_k conv(k) w.r.t. p is:
    #   sum_k w_k * d conv(k)/d p = cross-correlation of w with p (circular, padded)
    L = next_fast_len(2 * n - 1)
    Fp = fft(p, L)
    Fw = fft(np.concatenate([w, np.zeros(L - nc)]), L)
    h = np.real(ifft(Fp * Fw))
    # h[j] = sum_k w[k] p[k-j] (with circular wrap in the padded space)
    # We need the linear portion: grad_j = h[j] for j = 0,...,n-1 (this gives sum_k w[k]p[k-j] for valid k-j)
    # Plus the "reflected" part for when k-j < 0 or >= n... 
    # Actually in the padded FFT, h[j] = sum_{k=0}^{nc-1} w[k] * p_padded[k-j mod L]
    # where p_padded is p zero-padded to length L.
    # For j in [0, n-1]: h[j] = sum_k w[k] p_padded[(k-j) mod L]
    # = sum_k w[k] p[k-j] (for 0 <= k-j < n)
    # This is exactly what we want (one-sided). The gradient of sum_k w_k conv[k] w.r.t. p[j]
    # should also include the contribution from when j is the "second" index, which gives
    # the same by symmetry of w (since conv is symmetric). So grad = 2 * h (approximately).
    # Let's just use grad = 2*h for now (ignoring the j=k/2 edge case which is negligible for large n).
    grad = 2.0 * h[:n]
    grad = np.maximum(grad, 0.0)  # Since p >= 0, the gradient of the max should be non-negative
    return grad, surr


def spectral_init(n, rng, n_freqs=None):
    """
    Initialize by shaping the frequency domain.
    Create a target |P(f)|^2 that's flatter than uniform, then find p via IFFT.
    """
    if n_freqs is None:
        n_freqs = max(4, n // 4)
    L = next_fast_len(2 * n - 1)
    
    # Create a target power spectrum that's piecewise flat
    # With some random variation to break symmetry
    target = np.ones(L, dtype=np.float64)
    # Add structured variation
    n_lobes = int(rng.integers(2, 8))
    for _ in range(int(rng.integers(1, 4))):
        center = int(rng.integers(0, L))
        width = int(rng.integers(L // 20, L // 4))
        depth = float(rng.uniform(0.3, 0.9))
        for f in range(L):
            d = min(abs(f - center), L - abs(f - center))
            target[f] *= (1.0 - depth * np.exp(-0.5 * (d / max(1, width)) ** 2))
    
    # Ensure DC component is appropriate
    target[0] = 1.0
    # Make it symmetric (real signal)
    for f in range(1, L // 2 + 1):
        target[f] = 0.5 * (target[f] + target[L - f])
    
    # Take square root to get amplitude, add random phase
    mag = np.sqrt(np.maximum(target, 0.01))
    phase = rng.uniform(0, 2 * np.pi, L)
    phase[0] = 0
    # Make spectrum conjugate symmetric
    for f in range(1, L // 2 + 1):
        phase[L - f] = -phase[f]
    
    P = mag * np.exp(1j * phase)
    p = np.real(ifft(P, L))[:n]
    p = np.maximum(p, 0.0)
    return safe_normalize(p)


def make_structured(n, rng):
    """Generate a diverse set of structured initial sequences."""
    inits = []
    
    # Uniform
    inits.append(np.full(n, 1.0 / n))
    
    # Cosine-based (anti-symmetric pairs)
    x = np.linspace(0, 1, n)
    for r in (0.5, 0.7, 0.85, 0.95):
        p = 1.0 + r * np.cos(2 * np.pi * x)
        p = np.maximum(p, 0.01)
        inits.append(safe_normalize(p))
    
    # Two-frequency
    for f1, f2, r1, r2 in ((1, 2, 0.5, 0.3), (1, 3, 0.4, 0.4), (2, 3, 0.5, 0.5),
                            (1, 5, 0.6, 0.2), (3, 7, 0.4, 0.4)):
        p = 1.0 + r1 * np.cos(2 * np.pi * f1 * x) + r2 * np.cos(2 * np.pi * f2 * x)
        p = np.maximum(p, 0.01)
        inits.append(safe_normalize(p))
    
    # Power-law tapers
    for a in (0.3, 0.7, 1.0, 1.5, 2.0):
        p = np.power(x + 0.01, a)
        p = 0.5 * (p + p[::-1])  # symmetrize
        inits.append(safe_normalize(np.maximum(p, 0.01)))
    
    # Concentrated pairs (anti-symmetric in pairs)
    for gap_ratio in (0.3, 0.5, 0.7):
        gap = int(n * gap_ratio)
        p = np.zeros(n)
        p[:n//4] = 1.0
        p[-n//4:] = 1.0
        inits.append(safe_normalize(p))
    
    # Random smooth
    for _ in range(8):
        f1 = int(rng.integers(1, n // 20))
        f2 = int(rng.integers(1, n // 10))
        f3 = int(rng.integers(1, n // 5))
        p1, p2, p3 = (float(rng.uniform(0, 2 * np.pi)) for _ in range(3))
        p = (1.0 + 0.4 * np.sin(2 * np.pi * f1 * x + p1)
             + 0.25 * np.sin(2 * np.pi * f2 * x + p2)
             + 0.15 * np.sin(2 * np.pi * f3 * x + p3))
        p = np.maximum(p, 0.05)
        inits.append(safe_normalize(p))
    
    # Spectral init
    for _ in range(4):
        inits.append(spectral_init(n, rng))
    
    return inits


def optimize_lp_iterative(p0, rng, time_limit, K_constraints=40):
    """
    Main optimizer: iterative LP best-response with blending and gradient refinement.
    """
    p = safe_normalize(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    best_p = p.copy()
    best_score = internal_score(p)
    current_score = best_score
    
    end = time.monotonic() + max(1.0, time_limit)
    no_improve_count = 0
    iter_count = 0
    lp_time_avg = 5.0  # Running average of LP time
    
    while time.monotonic() < end - 5.0:
        iter_count += 1
        
        # Get current convolution and max
        c = convolve_full(p)
        cmax = float(np.max(c))
        M = cmax * (1.0 + 0.01 * (1.0 + 0.5 * np.sin(iter_count * 0.1)))  # Slightly above max for feasibility
        
        # Solve LP best-response
        t0 = time.monotonic()
        g = lp_best_response_active(p, M, K=K_constraints,
                                    time_limit=min(20.0, max(1.0, (end - time.monotonic()) * 0.3)))
        lp_time = time.monotonic() - t0
        lp_time_avg = 0.8 * lp_time_avg + 0.2 * lp_time
        
        if g is not None:
            # Try multiple blending factors
            improved = False
            for t in (1.0, 0.8, 0.6, 0.45, 0.3, 0.18, 0.1, 0.05):
                q = project_simplex((1.0 - t) * p + t * g)
                sc = internal_score(q)
                if sc < best_score - 1e-10:
                    best_p = q.copy()
                    best_score = sc
                    current_score = sc
                    p = q
                    no_improve_count = 0
                    improved = True
                    break
                elif sc < current_score - 1e-12:
                    # Accept even if not a new best, to escape plateaus
                    p = q
                    current_score = sc
                    no_improve_count += 1
                    improved = True
                    break
            
            if not improved:
                no_improve_count += 1
        else:
            no_improve_count += 1
        
        # Gradient refinement every few iterations
        if iter_count % 3 == 0 and time.monotonic() < end - 10.0:
            grad, _ = gradient_smoothed_max(p, T_factor=0.3 + 0.3 * (no_improve_count / 100.0))
            if np.max(np.abs(grad)) > 1e-15:
                # Backtracking line search
                g_norm = np.linalg.norm(grad)
                if g_norm > 0:
                    grad_norm = grad / g_norm
                    accepted = False
                    for lr in (0.02, 0.008, 0.003, 0.001, 0.0004):
                        q = project_simplex(p - lr * grad_norm * np.max(p) * 10)
                        sc = internal_score(q)
                        if sc < current_score - 1e-12:
                            p = q
                            current_score = sc
                            if sc < best_score - 1e-10:
                                best_p = q.copy()
                                best_score = sc
                                no_improve_count = 0
                            accepted = True
                            break
                    if not accepted:
                        no_improve_count += 1
        
        # Periodic perturbation to escape local optima
        if no_improve_count > 15 and time.monotonic() < end - 30.0:
            # Large kick: blend with random
            p_random = safenormalize_helper(n, rng)
            p = project_simplex(0.7 * best_p + 0.3 * p_random)
            current_score = internal_score(p)
            no_improve_count = 0
            
        elif no_improve_count > 8 and time.monotonic() < end - 30.0:
            # Small kick
            sigma = 0.02 / np.sqrt(n) * (1.0 + no_improve_count * 0.1)
            p = project_simplex(p + rng.normal(0, sigma, n))
            current_score = internal_score(p)
            if current_score < best_score - 1e-10:
                best_p = p.copy()
                best_score = current_score
            no_improve_count = 0
        
        # Log progress
        if iter_count % 20 == 0:
            print("  LP iter=%d best=%.8f cur=%.8f no_improve=%d lp_time=%.1fs" %
                  (iter_count, best_score, current_score, no_improve_count, lp_time_avg), flush=True)
        
        # Early termination if score is very good
        if best_score < 1.49:
            break
    
    return best_p, best_score


def safe_normalize_helper(n, rng):
    """Quick safe normalize for random vectors."""
    v = rng.uniform(0, 1, n)
    s = v.sum()
    if s < 1e-10:
        return np.full(n, 1.0 / n)
    return v / s


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative step function minimizing evaluate_sequence score.
    Uses LP-based best-response iteration with gradient refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(60.0, float(budget_s) - 30.0)
    
    # Try to use previous best as starting point
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 64 and float(parent_arr.sum()) > 0.0:
            parent = safe_normalize(parent_arr)
    except Exception:
        parent = None
    
    # Choose sequence length
    n = 4096
    if parent is not None:
        n = int(parent.size)
        if n < 256:
            n = 4096
        if n > 16000:
            n = 8192
    
    print("Starting search: n=%d, budget=%.0fs" % (n, budget_s), flush=True)
    
    # Generate initial candidates
    candidates = []
    scores = []
    
    if parent is not None and parent.size == n:
        candidates.append(parent.copy())
        scores.append(internal_score(parent))
    
    structured = make_structured(n, rng)
    for s in structured:
        if s.size == n:
            candidates.append(s)
            scores.append(internal_score(s))
    
    # Sort by score and keep top
    idx = np.argsort(scores)[:10]
    best_p = candidates[idx[0]].copy()
    best_score = float(scores[idx[0]])
    print("Best initial: %.8f" % best_score, flush=True)
    
    # Phase 1: LP optimization on the best initial (and a few others)
    phase1_time = min(1800.0, max(300.0, (deadline - time.monotonic()) * 0.75))
    
    # Prioritize the best candidates
    for i in range(min(6, len(idx))):
        if time.monotonic() > deadline - 120.0:
            break
        remaining = deadline - 120.0 - time.monotonic()
        if remaining < 60.0:
            break
        alloc = min(remaining * 0.4, 400.0)
        p_init = candidates[idx[i]].copy()
        sc_init = float(scores[idx[i]])
        
        print("Phase 1: optimizing candidate %d (score=%.6f) for %.0fs" %
              (i, sc_init, alloc), flush=True)
        
        p_best, sc_best = optimize_lp_iterative(p_init, rng, alloc, K_constraints=40)
        
        if sc_best < best_score:
            best_score = sc_best
            best_p = p_best.copy()
            print("  New best: %.8f" % best_score, flush=True)
        
        # Update the candidate for future reference
        candidates[idx[i]] = p_best
        scores[idx[i]] = sc_best
    
    # Phase 2: Continue optimizing the global best with more iterations
    if time.monotonic() < deadline - 60.0:
        remaining = deadline - 60.0 - time.monotonic()
        print("Phase 2: continued optimization for %.0fs, current best=%.8f" %
              (remaining, best_score), flush=True)
        p_best, sc_best = optimize_lp_iterative(best_p, rng, remaining, K_constraints=50)
        if sc_best < best_score:
            best_score = sc_best
            best_p = p_best.copy()
            print("  Phase 2 best: %.8f" % best_score, flush=True)
    
    # Phase 3: Final gradient polish
    if time.monotonic() < deadline - 10.0:
        remaining = deadline - 10.0 - time.monotonic()
        print("Phase 3: final polish for %.0fs" % remaining, flush=True)
        p = best_p.copy()
        for _ in range(min(200, int(remaining * 5))):
            if time.monotonic() > deadline - 10.0:
                break
            grad, _ = gradient_smoothed_max(p, T_factor=0.4)
            g_norm = np.linalg.norm(grad)
            if g_norm < 1e-15:
                break
            grad = grad / g_norm
            for lr in (0.005, 0.002, 0.001, 0.0005, 0.0002):
                q = project_simplex(p - lr * grad * np.max(p) * 10)
                sc = internal_score(q)
                if sc < best_score - 1e-12:
                    p = q
                    best_score = sc
                    best_p = q.copy()
                    break
                elif sc < internal_score(p):
                    p = q
                    break
        
        print("Final: %.8f" % best_score, flush=True)
    
    # Validate and return
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(best_p.sum())
    if s < 1e-12:
        best_p = np.full(n, 1.0 / n)
    best_p = np.minimum(best_p, 1000.0)
    
    final_score = internal_score(best_p)
    print("Returned sequence: n=%d, score=%.8f, time=%.0fs" %
          (best_p.size, final_score, time.monotonic() - start_time), flush=True)
    
    # Also verify with the actual evaluator
    try:
        eval_score = evaluate_sequence([float(x) for x in best_p])
        print("Evaluator confirms: %.8f" % eval_score, flush=True)
    except Exception as e:
        print("Evaluator error: %s" % str(e), flush=True)
    
    return [float(x) for x in best_p]
