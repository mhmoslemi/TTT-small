```python
            #   So c_{2j} = (sum_{i!=j} a_i a_{2j-i}) + a_j^2.
            #   dc_{2j}/da_j = 2*a_j.
            #
            # For k != 2j: c_k = (sum_{i!=j, i!=k-j} a_i a_{k-i}) + a_j a_{k-j} + a_{k-j} a_j
            #   if both j and k-j are in [0,n-1] and j != k-j: c_k = base + 2*a_j*a[k-j]
            #   if j in [0,n-1] but k-j not in [0,n-1]: c_k = base (a_j doesn't appear)
            #   if k-j in [0,n-1] but j not in [0,n-1]: can't happen since j is always in [0,n-1]
            
            # Let me just compute base and d directly:
            # c_k = base_k + a_j * d_k + (a_j^2 if k == 2j else 0)
            # where d_k = 2*a[k-j] if 0<=k-j<n and k!=2j, else 0
            
            for k in range(2*n-1):
                mk = k - j  # the "other index"
                if 0 <= mk < n:
                    if mk == j:
                        # k = 2j: term is a_j^2
                        # base_k = sum of all other terms
                        # d_k = 0 (the a_j contribution is quadratic)
                        d[k] = 0.0
                    else:
                        d[k] = 2.0 * a[mk]
                # else: d[k] = 0
            
            # Now c_k = base_k + a_j * d_k + (a_j^2 if k == 2j else 0)
            # We need base_k. base_k = c_k_current - a_j * d_k - (a_j^2 if k==2j else 0)
            c_curr = fft_convolve(a, a)
            base = c_curr.copy()
            for k in range(2*n-1):
                base[k] -= a_orig * d[k]
            if 2*j < 2*n-1:
                base[2*j] -= a_orig * a_orig
            
            # Now minimize max_k (base_k + t * d_k + t^2 * q_k) where q_k = 1 if k==2j else 0
            # This is minimize max of (base_k + t*d_k + t^2*q_k) over t in [0, some_upper]
            # For the linear terms (q_k=0): f_k(t) = base_k + t*d_k, which is linear in t.
            # For the quadratic term (k=2j): f_{2j}(t) = base_{2j} + t^2.
            
            # The function g(t) = max_k (base_k + t*d_k + t^2*q_k) is convex in t.
            # We can minimize it using bisection on the subgradient.
            
            # Upper bound on t: a_j can be at most... well, sum(a) = 1 so a_j <= 1.
            # But practically, a_j should be around 1/n.
            t_upper = max(10.0 * a_orig, 1.0 / n)
            
            # Bisection: find t where 0 is in the subgradient of g(t)
            lo = 0.0
            hi = t_upper
            
            for _ in range(30):  # 30 bisection steps
                mid = (lo + hi) / 2.0
                # Compute g and subgradient at mid
                vals = base + mid * d
                if 2*j < 2*n-1:
                    vals[2*j] += mid * mid
                g_val = np.max(vals)
                # Subgradient: d_k + 2*mid*q_k for active k
                active = np.where(np.abs(vals - g_val) < 1e-10)[0]
                subgrad = np.mean([d[k] + 2*mid*(1 if k == 2*j else 0) for k in active])
                if subgrad > 0:
                    hi = mid
                elif subgrad < 0:
                    lo = mid
                else:
                    break
            
            t_new = (lo + hi) / 2.0
            t_new = max(0.0, min(t_new, t_upper))
            
            a[j] = t_new
            new_score = compute_score(a)
            if new_score < best_score:
                best_score = new_score
            if abs(t_new - a_orig) > 1e-10:
                improved = True
        
        if not improved and verbose:
            break
    
    return a, best_score
```
