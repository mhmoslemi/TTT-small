def normalize(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    mask = np.isfinite(arr)
    arr[~mask] = 0.0
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return np.full(arr.size, 1.0 / arr.size, dtype=np.float64)
    arr /= s
    return arr
```

def conv_full(a):
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    if n == 0:
        return np.array([0.0])
    m = np.maximum(0.0, arr)
    need = 2*n - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(m, size)
    fa = fa * np.conjugate(fa)
    c = np.fft.irfft(fa, size)
    c = c[:need]
    c = np.maximum(c, 0.0)
    return c
```

"""
Search for a low-peak linear auto-convolution non-negative sequence.

Method:
- Represent sequences as strictly positive densities in the simplex (log/softmax).
- Use FFT to evaluate full linear convolution and its soft-max active-set gradient.
- Perform adaptive mirror-descent / projected gradient steps in log-density space.
- Interpolate and smooth previously found good sequences, adding Fourier / random
  smooth starts to avoid local minima.
- Polish with random interval multiplicative mutations.
"""

import time
import numpy as np
from operator import itemgetter


def normalize(a):
    """Return a non-negative float vector with sum 1."""
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr[~np.isfinite(arr)] = 0.0
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return np.full(arr.size, 1.0 / arr.size, dtype=np.float64)
    arr /= s
    return arr


def conv_full(a):
    """Full linear convolution a * a using FFT. Values are clipped to nonnegative."""
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    arr = np.maximum(arr, 0.0)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(arr, size)
    fa = fa * np.conjugate(fa)
    c = np.fft.irfft(fa, size)
    c = c[:need]
    c = np.maximum(c, 0.0)
    return c


def hard_score(a):
    """Objective for non-negative sequence: 2*n*max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    if n == 0:
        return float("inf")
    w = normalize(a)
    c = conv_full(w)
    if c.size == 0:
        return float("inf")
    val = float(2.0 * n * np.max(c))
    if not np.isfinite(val):
        return float("inf")
    return val


def softmax(z):
    """Numerically stable softmax."""
    arr = np.asarray(z, dtype=np.float64).copy()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr[~np.isfinite(arr)] = 0.0
    arr -= float(np.max(arr))
    w = np.exp(arr)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(arr.size, 1.0 / arr.size, dtype=np.float64)
    return w / s


def crosscorr(w, p):
    """
    For w length n and p length 2n-1, return x_i = sum_j w_j p_{j+i}, i=0..n-1.
    This is the convolution-part of the auto-convolution gradient weighted by p.
    """
    n = len(w)
    L = 2 * n - 1
    need = n + L - 1
    size = 1
    while size < need:
        size <<= 1
    p_rev = p[::-1]
    fw = np.fft.rfft(w, size)
    fp = np.fft.rfft(p_rev, size)
    cc = np.fft.irfft(fw * fp, size)
    seg = cc[n - 1:2 * n - 1]
    seg = np.maximum(seg, 0.0)
    return seg[::-1]


def soft_weights(c, L):
    """
    Normalize exp((c - max(c))/L). Returns normalized weights and soft max value.
    """
    m = float(np.max(c))
    if L <= 0.0 or not np.isfinite(L):
        L = 1e-12
    u = (c - m) / L
    u = np.maximum(u, -745.0)
    p = np.exp(u)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros_like(c, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p, m
    p /= s
    val = m + L * np.log(s)
    return p, val


def initial_soft_L(c, n):
    """Choose a soft-max scale from the width of the peak region."""
    m = float(np.max(c))
    if c.size < 20 or m <= 0.0:
        return max(1e-8, 1e-5 / max(1, n))
    q80 = float(np.percentile(c, 80))
    diff = m - q80
    if diff <= 0.0 or not np.isfinite(diff):
        diff = 0.1 * m
    L = 0.1 * diff
    minL = 1e-6 / float(max(n, 1))
    if L < minL:
        L = minL
    if L > 1e-3:
        L = 1e-3
    return L


def vector_descent(w0, time_limit, rng=None, label="vec", verbose=True):
    """
    FFT-based mirror descent / active-set gradient refinement.
    Maintains a strict positive density w with sum 1.
    """
    start = time.monotonic()
    if rng is None:
        rng = np.random.default_rng(987654321)

    if time_limit <= 0.0:
        w = normalize(w0)
        return w, hard_score(w)

    w = normalize(w0)
    w = np.maximum(w, 1e-12)
    w /= w.sum()
    n = w.size

    z = np.log(w)
    z -= float(z.mean())

    c = conv_full(w)
    cur_score = float(2.0 * n * np.max(c))
    best_score = cur_score
    best_w = w.copy()

    step = 0.15
    L = initial_soft_L(c, n)
    minL = 1e-6 / float(max(n, 1))

    fail = 0
    stagn = 0
    restart_noise = 0.06
    last_print = 0.0
    it = 0

    while time.monotonic() - start < time_limit:
        it += 1

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            stagn = 0
        else:
            stagn += 1

        if verbose and (it <= 3 or it % 1000 == 0 or time.monotonic() - last_print > 5.0):
            print(f"    {label}: it={it} cur={cur_score:.7f} best={best_score:.7f} "
                  f"L={L:.3e} step={step:.3f} fail={fail} stagn={stagn}")
            last_print = time.monotonic()

        if stagn > 12000 and step < 0.03:
            break

        if (stagn > 8000 or fail > 2500) and cur_score > best_score:
            z = np.log(np.maximum(best_w, 1e-12))
            z -= float(z.mean())
            z = np.clip(z, -30.0, 30.0)
            z = z + rng.normal(0.0, restart_noise, n)
            z -= float(z.mean())
            w = softmax(z)
            c = conv_full(w)
            cur_score = float(2.0 * n * np.max(c))
            step = 0.15
            L = initial_soft_L(c, n)
            fail = 0
            stagn = 0
            restart_noise = max(0.005, restart_noise * 0.6)
            continue

        p, _ = soft_weights(c, L)
        act = 1.0 / (float(np.dot(p, p)) + 1e-30)
        if act < 25.0:
            L = min(1e-2, L * 1.5)
            p, _ = soft_weights(c, L)
        elif act > 0.35 * float(n):
            L = max(minL, L * 0.6)
            p, _ = soft_weights(c, L)

        hc = crosscorr(w, p)
        g = float(n) * hc
        dot_wg = float(np.dot(w, g))
        hz = g - w * dot_wg
        hz -= 1e-5 * z

        hmax = float(np.max(np.abs(hz)))
        if (not np.isfinite(hmax)) or hmax < 1e-12:
            hz = rng.normal(0.0, 1.0, n)
            hmax = float(np.max(np.abs(hz)))
            if hmax < 1e-12:
                hz = np.ones(n, dtype=np.float64)
                hmax = 1.0

        dirn = hz / hmax
        alpha = step * min(1.0, hmax)
        if alpha < 1e-7:
            alpha = 1e-7

        accepted = False
        for a in (alpha, 0.5 * alpha):
            z2 = z - a * dirn
            z2 -= float(z2.mean())
            z2 = np.clip(z2, -30.0, 30.0)
            w2 = softmax(z2)
            c2 = conv_full(w2)
            sc2 = float(2.0 * n * np.max(c2))

            if sc2 < cur_score - 1e-13:
                z = z2
                w = w2
                c = c2
                cur_score = sc2
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_w = w.copy()
                    stagn = 0
                step = min(1.0, step * 1.02)
                fail = 0
                accepted = True
                break

        if not accepted:
            step = max(0.01, step * 0.75)
            fail += 1
            if fail > 2500:
                z2 = z + rng.normal(0.0, 0.08, n)
                z2 -= float(z2.mean())
                z2 = np.clip(z2, -30.0, 30.0)
                w2 = softmax(z2)
                c2 = conv_full(w2)
                sc2 = float(2.0 * n * np.max(c2))
                if sc2 < cur_score + 0.002:
                    z = z2
                    w = w2
                    c = c2
                    cur_score = sc2
                    if cur_score < best_score - 1e-12:
                        best_score = cur_score
                        best_w = w.copy()
                        stagn = 0
                    fail = 0

        L = max(minL, L * 0.9995)

    if cur_score < best_score - 1e-12:
        best_score = cur_score
        best_w = w.copy()

    if verbose:
        print(f"    {label}: finished best={best_score:.7f} iterations={it}")

    return best_w, best_score


def smooth_sequence(w, width):
    """Moving average smoothing with edge padding."""
    if width <= 1:
        return w.copy()
    width = int(width)
    n = w.size
    pad = width // 2
    wp = np.pad(w, pad, mode="edge")
    kernel = np.ones(width, dtype=np.float64) / float(width)
    ws = np.convolve(wp, kernel, mode="same")
    ws = ws[pad:pad + n]
    return normalize(ws)


def lowpass_fft(w, max_mode):
    """Keep low Fourier modes of log(w), then exponentiate and normalize."""
    if max_mode <= 0:
        return w.copy()
    n = w.size
    if n <= 3:
        return w.copy()
    w = np.maximum(w, 1e-12)
    w /= w.sum()
    z = np.log(w)
    z -= float(z.mean())
    F = np.fft.fft(z)
    F[0] = 0.0
    keep = np.zeros(n, dtype=bool)
    upper = min(int(max_mode), n // 2 - 1)
    if upper > 0:
        keep[1:upper + 1] = True
        keep[n - upper:] = True
    F[~keep] = 0.0
    z2 = np.fft.ifft(F).real
    if not np.all(np.isfinite(z2)):
        return w.copy()
    z2 -= float(z2.mean())
    return softmax(z2)


def interpolate_to_len(a, n):
    """Linearly interpolate a sequence to length n and normalize."""
    arr = np.asarray(a, dtype=np.float64).ravel()
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    L = arr.size
    if L == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if L == n:
        return normalize(arr)
    if L == 1:
        return np.full(n, 1.0 / n, dtype=np.float64)
    old_x = np.linspace(0.0, 1.0, L)
    new_x = np.linspace(0.0, 1.0, n)
    out = np.interp(new_x, old_x, arr)
    return normalize(out)


def power_sequence(w, p):
    """Apply p-th power transform and normalize."""
    w = np.maximum(w, 1e-12)
    out = np.power(w, p)
    return normalize(out)


def sine_start(n, rng, fixed=False):
    """Asymmetric sinusoidal start in log-density space."""
    t = np.arange(n, dtype=np.float64) / float(n)
    if fixed:
        a1 = 0.55
        a3 = -0.15
        a5 = 0.05
        z = a1 * np.sin(2.0 * np.pi * t) + a3 * np.sin(6.0 * np.pi * t) + a5 * np.sin(10.0 * np.pi * t)
    else:
        a1 = float(rng.uniform(0.35, 0.85))
        a3 = float(rng.uniform(-0.35, 0.35))
        a5 = float(rng.uniform(-0.25, 0.25))
        z = a1 * np.sin(2.0 * np.pi * t) + a3 * np.sin(6.0 * np.pi * t) + a5 * np.sin(10.0 * np.pi * t)
        z += 0.05 * rng.normal(size=n)
    z -= float(z.mean())
    z = np.clip(z, -3.0, 3.0)
    return softmax(z)


def random_smooth_start(n, rng, max_mode=30):
    """Low-passed Gaussian noise plus a sinusoidal asymmetric bias."""
    z = rng.normal(size=n)
    F = np.fft.fft(z)
    F[0] = 0.0
    keep = np.zeros(n, dtype=bool)
    upper = min(int(max_mode), n // 2 - 1)
    if upper > 0:
        keep[1:upper + 1] = True
        keep[n - upper:] = True
    F[~keep] = 0.0
    z = np.fft.ifft(F).real
    t = np.arange(n, dtype=np.float64) / float(n)
    phase = float(rng.uniform(0.0, 2.0 * np.pi))
    amp = float(rng.uniform(0.35, 0.85))
    z = 0.5 * z + amp * np.sin(2.0 * np.pi * t + phase)
    z -= float(z.mean())
    z = np.clip(z, -3.0, 3.0)
    return softmax(z)


def generate_candidates(n, best_w, prev, rng):
    """Generate many transforms / smoothings of base starting shapes."""
    cands = []

    if n <= 4096:
        smooth_widths = [1, 3, 7, 13]
        low_modes = [0, 4, 8, 16, 32]
        max_bases = 9
    elif n <= 8192:
        smooth_widths = [1, 3, 7]
        low_modes = [0, 4, 8, 16]
        max_bases = 6
    else:
        smooth_widths = [1, 3]
        low_modes = [0, 8]
        max_bases = 4

    bases = []

    if best_w is not None:
        b = interpolate_to_len(best_w, n)
        bases.append((b, "best"))
        if n <= 4096:
            bases.append((power_sequence(b, 0.85), "bestp85"))
            bases.append((power_sequence(b, 1.25), "bestp125"))
            bz = np.log(np.maximum(b, 1e-12))
            bz -= float(bz.mean())
            bz = np.clip(bz + rng.normal(0.0, 0.05, n), -30.0, 30.0)
            bases.append((softmax(bz), "bestnoise"))

    if prev is not None:
        b = interpolate_to_len(prev, n)
        bases.append((b, "prev"))
        if n <= 4096:
            bases.append((power_sequence(b, 0.85), "prevp85"))
            bases.append((power_sequence(b, 1.25), "prevp125"))

    if best_w is None and prev is None:
        bases.append((sine_start(n, rng, fixed=True), "sine_fix"))
        bases.append((sine_start(n, rng, fixed=False), "sine_r1"))
        for k in range(6):
            bases.append((random_smooth_start(n, rng, max_mode=30), f"rand{k}"))
    elif prev is not None and best_w is None:
        bases.append((random_smooth_start(n, rng, max_mode=30), "random"))

    if len(bases) > max_bases:
        bases = bases[:max_bases]

    for bw, blab in bases:
        for sw in smooth_widths:
            wsm = bw.copy()
            if sw > 1:
                wsm = smooth_sequence(wsm, sw)
            for lm in low_modes:
                if lm > 0:
                    w2 = lowpass_fft(wsm, lm)
                else:
                    w2 = wsm.copy()
                w2 = normalize(w2)
                c = conv_full(w2)
                sc = float(2.0 * n * np.max(c))
                if not np.isfinite(sc):
                    continue
                label = f"{blab}|sm{sw}|lp{lm}"
                cands.append((sc, label, w2))

    return cands


def select_candidates(cands, k, current_best_score):
    """Pick a few diverse high-likelihood candidates."""
    finite = []
    for item in cands:
        if np.isfinite(item[0]):
            finite.append(item)
    if not finite:
        return []
    finite.sort(key=itemgetter(0))
    if not finite:
        return []

    selected = []
    used = set()
    selected.append(finite[0])
    used.add(0)

    base_score = float(finite[0][0])
    tol = max(0.005, 0.03 * max(1.0, abs(base_score)))

    # Prefer an almost-raw candidate if it is close to the best score.
    for i, item in enumerate(finite):
        if i in used:
            continue
        if item[0] <= base_score + tol and "|sm1|lp0" in item[1]:
            selected.append(item)
            used.add(i)
            break

    for i, item in enumerate(finite):
        if len(selected) >= k:
            break
        if i not in used:
            selected.append(item)
            used.add(i)

    return selected


def optimize_length(n, best_w, prev, rng, deadline, max_per=600.0, label="len"):
    """Generate, rank, and refine candidates at a fixed length."""
    cands = generate_candidates(n, best_w, prev, rng)
    if not cands:
        u = np.full(n, 1.0 / float(n), dtype=np.float64)
        return u, hard_score(u)

    cur_best_w = None
    cur_best_score = float("inf")
    for item in cands:
        sc = float(item[0])
        if np.isfinite(sc) and sc < cur_best_score:
            cur_best_score = sc
            cur_best_w = item[2]

    k = 2
    selected = select_candidates(cands, k, cur_best_score)
    if not selected:
        selected = [cands[0]]

    for idx, cand in enumerate(selected):
        rem = deadline - time.monotonic()
        if rem < 45.0:
            break

        remaining_cands = max(1, len(selected) - idx)
        cap = max(45.0, rem / float(remaining_cands) * 0.95)

        cand_score = float(cand[0])
        if np.isfinite(cand_score) and np.isfinite(cur_best_score):
            deficit = max(0.0, cand_score - cur_best_score)
        else:
            deficit = 0.0
        frac = 1.0 / (1.0 + 15.0 * deficit)

        lim = min(float(max_per) * frac, cap)
        if lim < 30.0:
            lim = min(30.0, rem * 0.5)
        if lim <= 0.0:
            break

        w, sc = vector_descent(cand[2], lim, rng, label=f"{label}:{cand[1]}")
        if np.isfinite(sc) and sc < cur_best_score:
            cur_best_score = sc
            cur_best_w = w

    if cur_best_w is None:
        cur_best_w = cands[0][2]
        cur_best_score = cands[0][0]

    return cur_best_w, cur_best_score

                best_score = s
                best_w = w.copy()
                print(f"  n=4096 best so far: {best_score:.7f}")

        rem = deadline - time.monotonic()
        if rem > 500.0:
            print("\n=== Stage: upscale/refine to n=8192 ===")
            w, s = optimize_length(8192, best_w, prev, rng, deadline, max_per=1800.0, label="n8192")
            if w is not None and np.isfinite(s) and s < best_score:
                best_score = s
                best_w = w.copy()
                print(f"  n=8192 best so far: {best_score:.7f}")

        rem = deadline - time.monotonic()
        if rem > 400.0 and best_w is not None and len(best_w) < 16384:
            print("\n=== Optional stage: n=16384 ===")
            w, s = optimize_length(16384, best_w, prev, rng, deadline, max_per=700.0, label="n16384")
            if w is not None and np.isfinite(s) and s < best_score:
                best_score = s
                best_w = w.copy()
                print(f"  n=16384 best so far: {best_score:.7f}")
    else:
        rem = deadline - time.monotonic()
        if rem > 400.0:
            stage_deadline = min(deadline, time.monotonic() + 550.0)
            print("\n=== No previous construction: explore n=2048 ===")
            w, s = optimize_length(2048, None, None, rng, stage_deadline, max_per=550.0, label="n2048")
            if w is not None and np.isfinite(s) and s < best_score:
                best_score = s
                best_w = w.copy()
                print(f"  n=2048 best so far: {best_score:.7f}")

        rem = deadline - time.monotonic()
        if rem > 700.0:
            stage_deadline = min(deadline, time.monotonic() + 1100.0)
            print("\n=== Explore/refine n=4096 ===")
            w, s = optimize_length(4096, best_w, None, rng, stage_deadline, max_per=850.0, label="n4096")
            if w is not None and np.isfinite(s) and s < best_score:
                best_score = s
                best_w = w.copy()
                print(f"  n=4096 best so far: {best_score:.7f}")

        rem = deadline - time.monotonic()
        if rem > 600.0:
            print("\n=== Explore/refine n=8192 ===")
            w, s = optimize_length(8192, best_w, None, rng, deadline, max_per=1800.0, label="n8192")
            if w is not None and np.isfinite(s) and s < best_score:
                best_score = s
                best_w = w.copy()
                print(f"  n=8192 best so far: {best_score:.7f}")

    # Final upgrades if the current best is short and time remains.
    rem = deadline - time.monotonic()
    if best_w is not None and len(best_w) < 4096 and rem > 450.0:
        print("\n=== Final upgrade to n=4096 ===")
        w, s = optimize_length(4096, best_w, prev, rng, deadline, max_per=900.0, label="upgrade4096")
        if w is not None and np.isfinite(s) and s < best_score:
            best_score = s
            best_w = w.copy()
            print(f"  upgraded best: {best_score:.7f}, n={len(best_w)}")

    rem = deadline - time.monotonic()
    if best_w is not None and len(best_w) < 8192 and rem > 650.0:
        print("\n=== Final upgrade to n=8192 ===")
        w, s = optimize_length(8192, best_w, prev, rng, deadline, max_per=1200.0, label="upgrade8192")
        if w is not None and np.isfinite(s) and s < best_score:
            best_score = s
            best_w = w.copy()
            print(f"  upgraded best: {best_score:.7f}, n={len(best_w)}")

    # Final local refinement.
    rem = deadline - time.monotonic()
    if best_w is not None and rem > 120.0:
        print("\n=== Final vector polish ===")
        w, s = vector_descent(best_w, min(rem * 0.45, 300.0), rng, label="final_vec")
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_w = w.copy()

    rem = deadline - time.monotonic()
    if best_w is not None and rem > 120.0:
        print("\n=== Final random polish ===")
        w, s = local_polish(best_w, rem * 0.8, rng, label="final_polish")
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_w = w.copy()

    if best_w is None or not np.isfinite(best_score):
        if prev is not None:
            best_w = interpolate_to_len(prev, 1024)
        else:
            best_w = np.full(512, 1.0 / 512.0, dtype=np.float64)
        best_score = hard_score(best_w)

    n = len(best_w)
    final = np.asarray(best_w, dtype=np.float64) * float(n)
    final = np.nan_to_num(final, nan=0.0, posinf=1000.0, neginf=0.0)
    final = np.maximum(final, 0.0)
    m = float(np.max(np.abs(final)))
    if m > 1000.0 and m > 0.0:
        final *= 1000.0 / m

    final_list = [float(x) for x in final]

    print(f"\nInternal best: score={best_score:.7f}, n={n}")
    try:
        val = evaluate_sequence(final_list)
        print(f"Verified score via evaluate_sequence: {val:.7f}")
    except Exception as exc:
        print(f"Verification skipped or failed: {exc}")

    print(f"Total time used: {time.monotonic() - t0:.1f}s")
    return final_list
