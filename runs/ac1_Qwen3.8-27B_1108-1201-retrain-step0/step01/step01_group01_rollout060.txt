def _fft_convolve_full(a, b):
    """Full linear convolution via FFT."""
    out_len = len(a) + len(b) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_len]


def _autocorr(a):
    """Compute convolve(a, a) = a * a."""
    return _fft_convolve_full(a, a)


def _smooth_obj(a, alpha):
    """Compute (1/alpha) * log(sum exp(alpha * b)) where b = convolve(a,a)."""
    b = _autocorr(a)
    x = alpha * b
    x -= np.max(x)
    s = np.sum(np.exp(x))
    return np.log(s) / alpha


def _smooth_grad(a, alpha):
    """
    Compute gradient of smooth_obj w.r.t. a.
    grad[i] = 2 * sum_k p[k] * a[k-i] where p = softmax(alpha * b)
    """
    n = len(a)
    b = _autocorr(a)
    x = alpha * b
    x -= np.max(x)
    p = np.exp(x)
    p /= np.sum(p)
    # grad[i] = 2 * sum_k p[k] * a[k-i] = 2 * (p * a_rev)[n-1+i]
    a_rev = a[::-1]
    N = _next_pow2(2 * n - 1 + n)  # need full conv of p (len 2n-1) and a_rev (len n)
    fp = np.fft.rfft(p, N)
    fa = np.fft.rfft(a_rev, N)
    g_full = np.fft.irfft(fp * fa, N)
    g = g_full[n - 1:2 * n - 1]
    return 2.0 * g


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_score_direct(a):
    """Compute 2*n*max_b / sum_a^2 for a sequence (scale-invariant, so use sum=1 form)."""
    n = len(a)
    b = _autocorr(a)
    max_b = np.max(b)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _phase_smooth_optimize(a, deadline, alpha_init, alpha_end, n_iter_max=None):
    """
    Projected gradient descent on the smooth log-sum-exp objective with continuation.
    Returns best a and best score.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_score = _compute_score_direct(best_a)
    print(f"  Smooth opt start: score={best_score:.8f}", flush=True)

    if n_iter_max is None:
        n_iter_max = 1000000

    alpha = float(alpha_init)
    alpha_total_span = alpha_end - alpha_init
    eta = 1.0 / np.sqrt(n)
    no_improve = 0
    patience = 2000

    for it in range(n_iter_max):
        if time.monotonic() >= deadline:
            break

        # Update alpha (linear continuation)
        if it > 0:
            progress = min(1.0, it / (n_iter_max * 0.8))
            alpha = alpha_init + progress * alpha_total_span

        # Compute gradient
        grad = _smooth_grad(a, alpha)
        # Center gradient (for sum constraint)
        grad -= np.mean(grad)

        # Line search (backtracking)
        cur_obj = _smooth_obj(a, alpha)
        step = eta
        improved = False
        for ls in range(10):
            a_try = a - step * grad
            a_try = _project_simplex(a_try, 1.0)
            obj_try = _smooth_obj(a_try, alpha)
            if obj_try < cur_obj:
                a = a_try
                cur_obj = obj_try
                improved = True
                break
            step *= 0.5
        if not improved:
            # Try a smaller step
            step = eta * 0.01
            a_try = a - step * grad
            a_try = _project_simplex(a_try, 1.0)
            a = a_try

        # Check actual score improvement
        if it % 50 == 0:
            cur_score = _compute_score_direct(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                no_improve = 0
                if best_score < 1.51:
                    print(f"    iter {it}: score={best_score:.8f} (alpha={alpha:.1f})", flush=True)
            else:
                no_improve += 1

            if no_improve >= patience:
                # Reheat: perturb and reset
                a = best_a + 0.02 * np.random.randn(n)
                a = _project_simplex(a, 1.0)
                eta = eta * 0.5
                no_improve = 0

        # Slowly decay eta
        if it % 1000 == 0:
            eta *= 0.95

    # Final score check
    final_score = _compute_score_direct(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


def _phase_simulated_annealing(a, deadline, rng, n_iterations=None):
    """
    Simulated annealing with sum-preserving moves (transfer between two positions).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_score = _compute_score_direct(best_a)
    cur_b = _autocorr(a)
    cur_max_b = np.max(cur_b)
    cur_score = 2.0 * n * cur_max_b  # with sum=1

    T_init = 0.001
    T_final = 1e-8
    if n_iterations is None:
        n_iterations = 500000

    t_start = time.monotonic()
    t_budget = deadline - t_start
    cooling_rate = (T_final / T_init) ** (1.0 / max(1, n_iterations))

    T = T_init
    delta_scale = 0.05 / np.sqrt(n)
    no_improve = 0
    patience = 20000

    for it in range(n_iterations):
        if time.monotonic() >= deadline:
            break

        # Pick two random positions
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

        # Pick delta
        delta = rng.uniform(-delta_scale, delta_scale)
        # Ensure non-negativity
        if a[i] - delta < 0:
            delta = a[i] * rng.uniform(0, 1)
        if a[j] + delta < 0:
            delta = -a[j] * rng.uniform(0, 1)
        if abs(delta) < 1e-15:
            continue

        # Apply move
        a[i] -= delta
        a[j] += delta

        # Update b incrementally
        # Effect of a[i] -= delta: for k in [i, i+n-1], b[k] += 2*(-delta)*a[k-i] (approx, ignoring delta^2)
        # Effect of a[j] += delta: for k in [j, j+n-1], b[k] += 2*delta*a[k-j]
        # But we need to account for the fact that a changed. Use the OLD a values for the linear term.
        # For small delta, this is accurate to first order.

        # Actually, let's just recompute b via FFT for accuracy (it's fast enough)
        new_b = _autocorr(a)
        new_max_b = np.max(new_b)
        new_score = 2.0 * n * new_max_b  # sum=1

        # Accept/reject
        delta_score = new_score - cur_score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / T):
            # Accept
            cur_b = new_b
            cur_max_b = new_max_b
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Reject
            a[i] += delta
            a[j] -= delta

        # Cool
        T *= cooling_rate

        # Reheat if stuck
        if no_improve >= patience:
            T = 0.0005
            a = best_a + 0.01 * rng.standard_normal(n)
            a = _project_simplex(a, 1.0)
            cur_b = _autocorr(a)
            cur_max_b = np.max(cur_b)
            cur_score = 2.0 * n * cur_max_b
            no_improve = 0

        if it % 10000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    SA iter {it}: score={best_score:.8f}, T={T:.2e}, elapsed={elapsed:.1f}s", flush=True)

    return best_a, best_score
