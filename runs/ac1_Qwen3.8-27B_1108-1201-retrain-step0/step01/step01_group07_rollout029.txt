    s = a.sum()

    return 2.0 * n * c.max() / (s * s)

    cols = []

def line_search(f, b, max_iters=30):
    """Find best t in [0,1] minimizing ||((1-t)f+tb)*((1-t)f+tb)||_inf."""
    n = len(f)
    ff = np.convolve(f, f)
    fb = np.convolve(f, b)
    bb = np.convolve(b, b)
    
    def obj(t):
        val = (1-t)**2 * ff + 2*t*(1-t) * fb + t**2 * bb
        return val.max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = obj(c1), obj(c2)
    
    for _ in range(max_iters):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = obj(c2)
    
    t_best = (lo + hi) / 2
    return t_best, obj(t_best)

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding
with line search, combined with gradient-based polishing.
The LP approach finds a direction b that maximizes total mass while keeping
(f*b)_k <= M for all lags, then a line search determines the optimal blend.
Fallback to smoothed gradient descent for local refinement.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def build_conv_matrix(f, n):
    """Build sparse matrix C where (C @ b)_k = (f*b)_k for k=0..2n-2."""
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo > m_hi:
            continue
        m_range = np.arange(m_lo, m_hi + 1)
        rows.extend([k] * len(m_range))
        cols.extend(m_range.tolist())
        data.extend(f[k - m_range].tolist())
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_direction_lp(f, M, time_limit=25):
    """Solve: max sum(b) s.t. (f*b)_k <= M for all k, b >= 0."""
    n = len(f)
    C = build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def line_search(f, b, n_iterations=40):
    """Find best t in [0,1] minimizing ||((1-t)f+tb)*((1-t)f+tb)||_inf."""
    ff = np.convolve(f, f)
    fb = np.convolve(f, b)
    bb = np.convolve(b, b)

    def obj(t):
        val = (1.0 - t) ** 2 * ff + 2.0 * t * (1.0 - t) * fb + t ** 2 * bb
        return val.max()

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = obj(c1), obj(c2)

    for _ in range(n_iterations):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = obj(c2)

    t_best = (lo + hi) * 0.5
    return t_best, obj(t_best)


def compute_gradient(a, q):
    """Compute grad[m] = 2 * sum_j a[j] * q[m+j] for m=0..n-1."""
    n = len(a)
    from numpy.lib.stride_tricks import sliding_window_view
    windows = sliding_window_view(q, n)
    grad = 2.0 * (windows @ a)
    return grad


def gradient_polish(a, budget, tau=0.5, lr0=1.0):
    """Projected gradient descent with log-sum-exp smoothing and power smoothing."""
    n = len(a)
    S = a.sum()
    a = a.copy()
    best = a.copy()
    best_cmax = np.convolve(a, a).max()

    t0 = time.monotonic()
    step = 0
    lr = lr0

        c = np.convolve(a, a)
        c_max = c.max()

        # Log-sum-exp smoothing
        c_shift = c - c_max
        exp_c = np.exp(c_shift / tau)
        q = exp_c / exp_c.sum()

        # Gradient
        grad = compute_gradient(a, q)

        # Projected step
        a_new = a - lr * grad * S / c_max
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a + lr * 0.01 * np.random.randn(n)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
        a_new = a_new / s_new * S

        c_new = np.convolve(a_new, a_new)
        if c_new.max() < c_max:
            lr = min(lr * 1.02, 50.0)
        else:
            lr = max(lr * 0.7, 0.001)

        a = a_new
        if c_new.max() < best_cmax:
            best_cmax = c_new.max()
            best = a.copy()

        step += 1

    return best, compute_score(best)


def project_and_normalize(a, S):
    """Project to non-negative and normalize sum to S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:

    return a / s * S
