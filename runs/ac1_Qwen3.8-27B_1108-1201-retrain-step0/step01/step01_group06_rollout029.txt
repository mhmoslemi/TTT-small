def solve_lp_direction(f, M, n):

import numpy as np
from scipy.optimize import linprog
from scipy.signal import fftconvolve
import time

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix A such that (f*g)_k = A[k,:] @ g."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    return A

def lp_direction(f, M, n, g_bound=100.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, 0 <= g <= g_bound."""
    c = -np.ones(n)
    A = build_conv_matrix(f, n)
    b = np.full(2*n-1, M)
    bounds = [(0, g_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

Core idea: At each step, solve an LP to find the best direction g
that keeps (current * g) below the current peak, then take a step
in that direction with line search.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def build_A_matrix(f, n):
    A = np.zeros((2*n-1, n))
    for j in range(n):
        end = min(j + n, 2*n - 1)
        A[j:end, j] = f[:end-j]
    return A

def solve_lp(f, M, n, g_bound=10.0):
    """Solve: max sum(g) s.t. A@f... wait, A@ g <= M, g >= 0, g <= g_bound"""
    n = len(f)
    c = -np.ones(n)
    A = build_A_matrix(f, n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, g_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', 
                  options={'presolve': True, 'time_limit': 5})
    if res.success:
        return res.x
    return None

"""
LP-guided optimization for minimizing 2*n*max(autocorr)/(sum)^2.
Uses linear programming to find descent directions at each step,
combined with line search, diverse initializations, and mirror descent
polishing. Explores multiple sequence lengths (64-1024).
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_A_matrix(f, n):
    """Build matrix A where (f*g)_k = A[k,:] @ g for linear convolution."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = f[:end - j]
    return A


def solve_lp_direction(f, M, n, g_bound=5.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, 0 <= g <= g_bound."""
    c = -np.ones(n)
    A = build_A_matrix(f, n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, g_bound)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a0, time_limit, g_bound=5.0, verbose=False):
    """LP-based optimization with line search. a0 is normalized to sum=1 internally."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    start = time.monotonic()
    stagnation = 0
    max_stagnation = 80
    iter_count = 0

    while time.monotonic() - start < time_limit:
        if time.monotonic() - start < 0:
            break
        conv = fast_convolve(a)
        M = conv.max()
        current_obj = 2.0 * n * M

        g = solve_lp_direction(a, M, n, g_bound)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search: try multiple step sizes
        found_improve = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.7, 0.9, 0.05]:
            trial = (1 - t) * a + t * g
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-12:
                continue
            trial /= ts
            conv_t = fast_convolve(trial)
            obj_t = 2.0 * n * conv_t.max()
            if obj_t < current_obj - 1e-12:
                a = trial
                current_obj = obj_t
                found_improve = True
                break

        if not found_improve:
            # Try even smaller steps
            for t in [0.02, 0.01, 0.005]:
                trial = (1 - t) * a + t * g
                trial = np.maximum(trial, 0.0)
                ts = trial.sum()
                if ts < 1e-12:
                    continue
                trial /= ts
                conv_t = fast_convolve(trial)
                obj_t = 2.0 * n * conv_t.max()
                if obj_t < current_obj - 1e-12:
                    a = trial
                    current_obj = obj_t
                    found_improve = True
                    break

        if found_improve:
            if current_obj < best_obj - 1e-10:
                best_obj = current_obj
                best_a = a.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        iter_count += 1
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"      iter={iter_count}, t={elapsed:.1f}s, obj={best_obj:.6f}, stag={stagnation}")

        if stagnation >= max_stagnation:
            break

    return best_a, best_obj


def mirror_descent_optimize(a0, time_limit, t_param=50.0, eta0=0.4):
    """Multiplicative mirror descent with softmax peak smoothing (fallback/polish)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    patience = 0
    max_patience = 60
    cur_t = t_param
    start = time.monotonic()

    while time.monotonic() - start < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        scaled = cur_t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Compute weighted subgradient
        sg = np.zeros(n)
        thresh = w.max() * 0.005
        active_idx = np.where(w > thresh)[0]
        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]

        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.ones(n) / n
        else:
            a /= a_sum

        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()

        if current_obj < best_obj - 1e-9:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            patience = 0
            eta *= 0.7
            if eta < 0.01:
                cur_t *= 1.5
                eta = 0.3
                max_patience = 40
            else:
                cur_t = min(cur_t * 1.1, 200.0)

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse non-negative initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussians with different widths
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g += 0.05
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Cosine modulations
    for freq in [1, 2, 3, 5]:
        for phase_shift in [0, 0.5]:
            c = np.cos(2 * np.pi * freq * x / n + phase_shift * np.pi)
            a = 1.0 + 0.6 * c
            a = np.maximum(a, 0.01)
            a /= a.sum()
            inits[f'cos_f{freq}_p{int(phase_shift*2)}'] = a

    # Ramp
    ramp = np.linspace(0.05, 1.0, n)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Bimodal (edges high)
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    if n > 2 * w:
        bi[w:n - w] = 0.15
    else:
        bi[w:] = 0.5
    bi /= bi.sum()
    inits['bimodal'] = bi

    # "Donut" shape (low in middle, high on sides)
    donut = 1.0 - 0.7 * np.exp(-0.5 * ((x - n * 0.5) / (0.2 * n)) ** 2)
    donut = np.maximum(donut, 0.05)
    donut /= donut.sum()
    inits['donut'] = donut

    # Random structured (power of sine)
    for p in [1, 2, 3]:
        f_rand = rng.uniform(1.0, 4.0)
        ph_rand = rng.uniform(0, 2 * np.pi)
        r = np.abs(np.sin(2 * np.pi * f_rand * x / n + ph_rand)) ** p
        r += 0.1
        r /= r.sum()
        inits[f'rsin_{p}'] = r

    # Pure random
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits['random'] = r

    # From parent if available
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_clip'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp
        # Also try full parent if size matches
        if parent.size == n:
            pp = np.maximum(parent.astype(np.float64), 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_exact'] = pp

    return inits
