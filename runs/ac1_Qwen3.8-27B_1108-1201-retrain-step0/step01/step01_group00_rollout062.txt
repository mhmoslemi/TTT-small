"""
Hybrid search for low-peak non-negative self-convolution.

The objective is
    2 * n * max(c) / (sum(a)**2),  c = np.convolve(a, a),
which is scale-invariant.  Sequences are normalized to sum 1.

Components:
1. Multi-start optimization over several lengths and structured/random starts.
2. Smooth objective: log-sum-exp of the convolution coefficients, with an
   annealed temperature.  The optimization is done in log-space (softmax
   simplex) using L-BFGS-B with exact gradients.
3. Basin-hopping around the best candidate using perturbations and quick
   high-temperature refinements.
4. Exact-coordinate pair polish: 1-D exact updates for moving mass between
   two coefficients.  This directly optimizes the final unsmeared objective.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import minimize
    HAS_SCIPY = True
except Exception:
    HAS_SCIPY = False


class SearchTimeout(Exception):
    pass


_TIMEOUT_DEADLINE = None
_HARD_DEADLINE = 0.0


def fft_convolve(a, b):
    """Linear convolution of 1D arrays by FFT."""
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    out_len = len(a) + len(b) - 1
    if out_len <= 0:
        return np.array([], dtype=float)
    size = 1
    while size < out_len:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    out = np.fft.irfft(fa * fb, size)[:out_len]
    return out


def softmax(theta):
    """Stable softmax normalizing to sum 1."""
    theta = np.asarray(theta, dtype=float)
    if theta.size == 0:
        return np.array([1.0], dtype=float)
    theta = np.where(np.isfinite(theta), theta, 0.0)
    theta = theta - np.max(theta)
    e = np.exp(theta)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(theta.shape, 1.0 / theta.size, dtype=float)
    return e / s


def normalize_p(p):
    """Clip and normalize to a non-negative simplex vector."""
    p = np.asarray(p, dtype=float).copy()
    if p.size == 0:
        return np.array([1.0], dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-15:
        p = np.ones_like(p, dtype=float)
        s = float(p.sum())
    p /= s
    return p


def score_and_m_with_c(p):
    """Return (score, max_convolution_coefficient, convolution) for sum=1 p."""
    p = normalize_p(p)
    c = fft_convolve(p, p)
    c = np.maximum(c, 0.0)
    m = float(np.max(c))
    score = 2.0 * len(p) * m
    return score, m, c


def update_tracker(tracker, p, m, n):
    """Push an improved non-negative sequence into the tracker dictionary."""
    if tracker is None:
        return
    if not np.isfinite(m):
        return
    score = 2.0 * n * float(m)
    if score < tracker.get("best_score", np.inf):
        tracker["best_score"] = score
        tracker["best_p"] = normalize_p(p).copy()
        # Optional bookkeeping for printing.
        tracker.setdefault("best_n", n)


def hankel_grad(p, w):
    """
    Gradient factor d_i = 2 * sum_j p_j w_{i+j}
    used for quadratic diagonal forms c_t = sum_{i+j=t} p_i p_j.
    """
    n = len(p)
    if n == 0:
        return np.array([], dtype=float)
    rev = p[::-1]
    cw = fft_convolve(rev, w)
    return 2.0 * cw[n - 1:2 * n - 1]


def p_to_theta(p):
    """Logs for softmax parameterization."""
    p = normalize_p(p)
    th = np.log(np.maximum(p, 1e-300))
    th = th - np.max(th)
    return th


def resample_p(p, n):
    """Resample a non-negative sequence to length n."""
    p = normalize_p(p)
    if p.size == n:
        return p.copy()
    old = np.linspace(0.0, 1.0, p.size)
    new = np.linspace(0.0, 1.0, n)
    y = np.interp(new, old, p)
    y = np.maximum(y, 1e-12)
    return normalize_p(y)


def compute_logsumexp_fg(theta, n, alpha, tracker):
    """
    Value and gradient of the smoothed objective in theta-space:
        J = (1/alpha) log sum_t exp(alpha*c_t),  p = softmax(theta),
        c = p * p (linear convolution).
    Also returns p, c, m=max(c) for bookkeeping.
    """
    theta = np.asarray(theta, dtype=float)
    if theta.shape[0] != n:
        theta = np.resize(theta, n)
    if not np.all(np.isfinite(theta)):
        theta = np.where(np.isfinite(theta), theta, 0.0)

    p = softmax(theta)
    c = fft_convolve(p, p)
    c = np.maximum(c, 0.0)
    m = float(np.max(c))

    if not np.isfinite(m) or m <= 1e-300:
        return 1e100, np.zeros(n, dtype=float), p, c, m

    update_tracker(tracker, p, m, n)

    alpha = float(alpha)
    if alpha <= 0.0 or not np.isfinite(alpha):
        alpha = 1.0

    # Stable log-sum-exp around the maximum coefficient.
    shifted = alpha * (c - m)
    z = np.exp(shifted)
    s = float(z.sum())
    if s <= 0.0 or not np.isfinite(s):
        return 1e100, np.zeros(n, dtype=float), p, c, m

    val = (m + math.log(s)) / alpha
    w = z / s
    d = hankel_grad(p, w)
    inner = float(np.dot(p, d))
    grad = p * (d - inner)

    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=float)
        val = 1e100

    return val, grad, p, c, m


def _cache_valid(state, theta, alpha):
    lt = state.get("last_theta")
    if lt is None:
        return False
    if state.get("last_alpha") != alpha:
        return False
    if lt.shape != theta.shape:
        return False
    return bool(np.array_equal(lt, theta))


def _fun_lse(theta, n, alpha, state):
    th = np.asarray(theta, dtype=float)
    try:
        state["last_x"] = th.copy()
    except Exception:
        state["last_x"] = th

    if _cache_valid(state, th, alpha):
        return state["last_val"]

    val, grad, p, c, m = compute_logsumexp_fg(th, n, alpha, state.get("tracker"))
    state["last_theta"] = th.copy()
    state["last_alpha"] = alpha
    state["last_val"] = val
    state["last_grad"] = grad
    return val


def _jac_lse(theta, n, alpha, state):
    th = np.asarray(theta, dtype=float)

    if _cache_valid(state, th, alpha):
        return state["last_grad"]

    val, grad, p, c, m = compute_logsumexp_fg(th, n, alpha, state.get("tracker"))
    state["last_theta"] = th.copy()
    state["last_alpha"] = alpha
    state["last_val"] = val
    state["last_grad"] = grad
    return grad


def lbfgs_callback(xk):
    if _TIMEOUT_DEADLINE is not None and time.monotonic() > _TIMEOUT_DEADLINE:
        raise SearchTimeout()


def run_lbfgs(x0, n, alpha, tracker, time_budget):
    """
    Run L-BFGS-B for one alpha.  Falls back safely if scipy is unavailable.
    """
    global _TIMEOUT_DEADLINE

    x0 = np.asarray(x0, dtype=float).copy()
    x0 = np.where(np.isfinite(x0), x0, 0.0)

    if not HAS_SCIPY:
        return x0
    if time_budget <= 1.5:
        return x0

    start = time.monotonic()
    if _HARD_DEADLINE <= 0.0:
        _HARD_DEADLINE = time.monotonic() + time_budget + 120.0

    run_deadline = min(_HARD_DEADLINE - 5.0, start + time_budget * 1.08)
    if run_deadline <= start:
        return x0

    _TIMEOUT_DEADLINE = run_deadline

    state = {
        "tracker": tracker,
        "last_x": x0.copy(),
        "last_theta": None,
        "last_alpha": None,
        "last_val": None,
        "last_grad": None,
    }

    if n <= 256:
        fcps = 850
    elif n <= 512:
        fcps = 450
    elif n <= 1024:
        fcps = 220
    else:
        fcps = 100

    maxfun = int(max(80, min(6000, time_budget * fcps * 0.38)))
    maxiter = int(max(20, maxfun // 4))

    options = {
        "maxiter": maxiter,
        "maxfun": maxfun,
        "ftol": 1e-12,
        "gtol": 1e-8,
        "maxls": 50,
        "disp": False,
    }

    try:
        res = minimize(
            _fun_lse,
            x0,
            args=(n, alpha, state),
            method="L-BFGS-B",
            jac=_jac_lse,
            callback=lbfgs_callback,
            options=options,
        )
        if res is not None and hasattr(res, "x"):
            rx = np.asarray(res.x, dtype=float)
            if rx.shape == (n,) and np.all(np.isfinite(rx)):
                rx = rx - np.max(rx)
                return rx
    except SearchTimeout:
        pass
    except Exception as e:
        print("    L-BFGS exception:", e)

    lx = state.get("last_x")
    if lx is None:
        return x0
    lx = np.asarray(lx, dtype=float)
    if lx.shape == (n,) and np.all(np.isfinite(lx)):
        return lx - np.max(lx)
    return x0


def generate_inits(n, provided_p, extra_p, rng, count):
    """Create a diverse list of normalized nonnegative starts at length n."""
    inits = []
    if count <= 0:
        return inits

    if provided_p is not None:
        inits.append(resample_p(provided_p, n))
        p = resample_p(provided_p, n)
        p = np.maximum(p, 1e-12) * np.exp(rng.normal(0.0, 0.12, size=n))
        inits.append(normalize_p(p))

    if extra_p is not None and (provided_p is None or extra_p is not provided_p):
        inits.append(resample_p(extra_p, n))

    while len(inits) < count:
        idx = len(inits)
        x = np.linspace(0.0, 1.0, n)
        mode = idx % 8

        if mode == 0:
            p = np.ones(n, dtype=float)
        elif mode == 1:
            beta = float(rng.uniform(0.5, 2.5))
            p = np.exp(beta * x)
        elif mode == 2:
            beta = float(rng.uniform(0.5, 2.5))
            p = np.exp(beta * (1.0 - x))
        elif mode == 3:
            a = float(rng.uniform(0.2, 3.0))
            b = float(rng.uniform(0.2, 3.0))
            xe = np.clip(x, 1e-6, 1.0 - 1e-6)
            p = (xe ** a) * ((1.0 - xe) ** b)
        elif mode == 4:
            f = float(rng.uniform(0.2, 4.0))
            amp = float(rng.uniform(0.3, 1.0))
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            p = 1.0 + amp * np.cos(2.0 * math.pi * f * x * x + ph)
        elif mode == 5:
            t = np.arange(n, dtype=float)
            p = np.ones(n, dtype=float)
            nf = int(rng.integers(2, 6))
            for _ in range(nf):
                maxf = max(2, n // 4)
                freq = int(rng.integers(1, maxf))
                amp = float(rng.uniform(0.05, 1.0 / max(1, freq)))
                ph = float(rng.uniform(0.0, 2.0 * math.pi))
                p += amp * np.cos(2.0 * math.pi * freq * t / max(1, n) + ph)
        elif mode == 6:
            beta = float(rng.uniform(1.5, 6.0))
            x0 = float(rng.uniform(0.35, 0.65))
            p = 1.0 / (1.0 + np.exp(-beta * (x - x0)))
        else:
            if rng.random() < 0.5:
                mu1 = float(rng.uniform(0.15, 0.40))
                mu2 = float(rng.uniform(0.60, 0.85))
                sig = float(rng.uniform(0.03, 0.18))
                w = float(rng.uniform(0.25, 0.75))
                p = (
                    w * np.exp(-0.5 * ((x - mu1) / sig) ** 2)
                    + (1.0 - w) * np.exp(-0.5 * ((x - mu2) / sig) ** 2)
                )
            else:
                spread = float(rng.uniform(0.2, 1.0))
                p = 1.0 + spread * rng.standard_normal(n)

        p = np.maximum(p, 1e-8)
        if rng.random() < 0.5:
            p = p * np.exp(rng.normal(0.0, 0.08, size=n))
        p = np.where(np.isfinite(p), p, 0.0)
        p = normalize_p(p)
        inits.append(p)

    return inits[:count]


def perturb_sequence(base, n, sigma, rng):
    """Multiplicative positive perturbation of a simplex sequence."""
    base = normalize_p(base)
    p = np.maximum(base, 1e-15) * np.exp(rng.normal(0.0, sigma, size=n))
    p = np.where(np.isfinite(p), p, 0.0)
    return normalize_p(p)


def optimize_p(p, n, tracker, time_budget, quick=False):
    """
    Continue a sequence through a temperature anneal of L-BFGS runs.
    Returns the best local sequence found from this start.
    """
    p = normalize_p(resample_p(p, n))

    if time_budget <= 2.0:
        sc, m, c = score_and_m_with_c(p)
        update_tracker(tracker, p, m, n)
        return p

    sc, m, c = score_and_m_with_c(p)
    update_tracker(tracker, p, m, n)
    local_score = sc
    local_p = p.copy()

    if m <= 1e-15:
        return local_p

    theta = p_to_theta(p)

    if quick:
        factors = [4.0, 16.0, 64.0]
        fracs = [0.25, 0.30, 0.45]
    else:
        factors = [0.5, 2.0, 8.0, 32.0, 128.0]
        fracs = [0.10, 0.18, 0.22, 0.25, 0.25]

    for factor, frac in zip(factors, fracs):
        if time.monotonic() > _HARD_DEADLINE - 25.0:
            break
        tb = time_budget * frac
        tb = min(tb, _HARD_DEADLINE - 12.0 - time.monotonic())
        if tb < 2.0:
            continue

        alpha = min(100000.0, max(1.0, factor / max(m, 1e-12)))
        old_theta = theta.copy()
        theta = run_lbfgs(theta, n, alpha, tracker, tb)
        if not np.all(np.isfinite(theta)):
            theta = old_theta

        theta = theta - np.max(theta)
        p = normalize_p(softmax(theta))
        sc2, m2, c2 = score_and_m_with_c(p)
        update_tracker(tracker, p, m2, n)

        if sc2 < local_score:
            local_score = sc2
            local_p = p.copy()

        sc = sc2
        m = m2

    # A final very high-alpha push, if time remains.
    if time_budget >= 12.0 and time.monotonic() < _HARD_DEADLINE - 30.0:
        tb = min(35.0, _HARD_DEADLINE - 25.0 - time.monotonic())
        if tb > 5.0:
            high_factor = 256.0 if quick else 512.0
            alpha = min(100000.0, max(1.0, high_factor / max(m, 1e-12)))
            old_theta = theta.copy()
            theta = run_lbfgs(theta, n, alpha, tracker, tb)
            if not np.all(np.isfinite(theta)):
                theta = old_theta
            theta = theta - np.max(theta)
            p = normalize_p(softmax(theta))
            sc2, m2, c2 = score_and_m_with_c(p)
            update_tracker(tracker, p, m2, n)
            if sc2 < local_score:
                local_score = sc2
                local_p = p.copy()

    return local_p


def pair_polish(p, n, tracker, time_budget, rng=None):
    """
    Exact local polish in the original objective.

    Moves mass from one coordinate to another.  For each one-dimensional move
    the convolution coefficients are updated by exact linear/quadratic updates,
    and the maximum coefficient is minimized by a small grid plus a quadratic
    candidate from the active weighted subgradient.
    """
    if rng is None:
        rng = np.random.default_rng(12345)

    p = normalize_p(resample_p(p, n))
    sc, m, c = score_and_m_with_c(p)
    update_tracker(tracker, p, m, n)

    if time_budget <= 1.0:
        return p

    start_time = time.monotonic()
    t_end = min(start_time + time_budget, _HARD_DEADLINE - 12.0)

    fracs = (0.01, 0.02, 0.05, 0.10, 0.20, 0.35, 0.55, 0.80, 0.97)
    refresh_pairs = 70
    stagnant = 0

    while time.monotonic() < t_end:
        p_start = p.copy()
        c_start = c.copy()
        m_start = float(m)
        if m_start <= 1e-30:
            break

        alpha = min(50000.0, max(20.0, 5000.0 / max(m_start, 1e-12)))
        z = np.exp(alpha * (c - m_start))
        s = float(z.sum())
        if s <= 0.0 or not np.isfinite(s):
            break
        w = z / s
        d = hankel_grad(p, w)
        if not np.all(np.isfinite(d)):
            break

        pairs = []
        seen = set()
        K = min(20, n)
        order = np.argsort(d)
        low = order[:K].astype(int, copy=False)
        high = order[-K:][::-1].astype(int, copy=False)

        if len(low) > 0 and len(high) > 0:
            diff = d[high][:, None] - d[low][None, :]
            flat = np.argsort(diff, kind="mergesort")[::-1]
            max_take = min(refresh_pairs, flat.size)
            for ii in range(max_take):
                pos = int(flat[ii])
                u = int(low[pos % len(low)])
                v = int(high[pos // len(low)])
                if u != v and (u, v) not in seen:
                    pairs.append((u, v))
                    seen.add((u, v))

        max_random = max(0, refresh_pairs - len(pairs))
        for _ in range(max_random):
            u = int(rng.integers(0, n))
            v = int(rng.integers(0, n))
            if u != v:
                if d[u] < d[v]:
                    pair = (u, v)
                else:
                    pair = (v, u)
                if pair not in seen:
                    pairs.append(pair)
                    seen.add(pair)

        accepted_any = False
        L = 2 * n - 1

        for u, v in pairs:
            if time.monotonic() >= t_end:
                break
            if u == v:
                continue
            pv = float(p[v])
            if pv <= 1e-14:
                continue

            A = np.zeros(L, dtype=float)
            A[u:u + n] += 2.0 * p
            A[v:v + n] -= 2.0 * p

            B = np.zeros(L, dtype=float)
            B[2 * u] += 1.0
            B[2 * v] += 1.0
            B[u + v] -= 2.0

            candidate_deltas = [f * pv for f in fracs]

            aw = float(np.dot(w, A))
            bw = float(np.dot(w, B))
            if bw > 1e-12:
                ds = -aw / (2.0 * bw)
                if 1e-14 < ds < pv:
                    candidate_deltas.append(ds)
                    candidate_deltas.append(max(1e-14, 0.5 * ds))
                    candidate_deltas.append(min(pv, 2.0 * ds))

            best_delta = 0.0
            best_m = float(m)

            for delta in candidate_deltas:
                if delta <= 1e-14 or delta >= pv:
                    continue
                # Exact update of all convolution coefficients for this pair move.
                cn = c + A * delta + B * (delta * delta)
                mn = float(np.max(cn))
                if mn < best_m - 1e-15:
                    best_m = mn
                    best_delta = delta

            if best_delta > 0.0 and best_m < m * (1.0 - 1e-13):
                p[u] += best_delta
                p[v] -= best_delta
                c = c + A * best_delta + B * (best_delta * best_delta)
                c = np.maximum(c, 0.0)
                m = best_m

                # Keep sums stable without losing the exact update too often.
                ssum = float(p.sum())
                if ssum <= 1e-15:
                    p = p_start
                    c = c_start
                    m = m_start
                    break
                if abs(ssum - 1.0) > 1e-7 or p[v] < 0.0:
                    p = p / ssum
                    p = np.maximum(p, 0.0)
                    sc2, m2, c2 = score_and_m_with_c(p)
                    c = c2
                    m = m2

                update_tracker(tracker, p, m, n)
                accepted_any = True

        if accepted_any:
            # Re-sync to the exact convolution after a refresh.
            psum = float(p.sum())
            if psum <= 1e-15:
                p = p_start
                c = c_start
                m = m_start
            else:
                p = p / psum
                sc2, m2, c2 = score_and_m_with_c(p)
                if m2 < m_start * (1.0 + 1e-14):
                    c = c2
                    m = m2
                else:
                    p = p_start
                    c = c_start
                    m = m_start
            update_tracker(tracker, p, m, n)
            stagnant = 0
        else:
            stagnant += 1
            if stagnant >= 3:
                break

    p = normalize_p(p)
    sc, m, c = score_and_m_with_c(p)
    update_tracker(tracker, p, m, n)
    return p


def basin_hopping(p, n, tracker, time_budget, rng):
    """
    Basin-hopping around a best sequence: perturb, quickly re-optimize,
    keep improvements.
    """
    p = normalize_p(resample_p(p, n))
    sc, m, c = score_and_m_with_c(p)
    update_tracker(tracker, p, m, n)
    best = p.copy()
    best_score = sc

    if time_budget <= 10.0:
        return best, best_score

    start_time = time.monotonic()
    t_end = min(start_time + time_budget, _HARD_DEADLINE - 25.0)
    fails = 0

    while time.monotonic() < t_end - 20.0:
        if fails > 14:
            break

        tb = min(45.0, t_end - time.monotonic() - 12.0)
        if tb < 8.0:
            break

        r = rng.random()
        if r < 0.65:
            sigma = float(0.03 + 0.35 * rng.random())
            p2 = perturb_sequence(best, n, sigma, rng)
        elif r < 0.85:
            # Mix best with a new random structured candidate.
            fresh = generate_inits(n, tracker.get("best_p"), None, rng, 1)[0]
            p2 = normalize_p(0.6 * best + 0.4 * fresh)
        else:
            fresh = generate_inits(n, None, best, rng, 1)[0]
            sigma = float(0.05 + 0.2 * rng.random())
            p2 = perturb_sequence(fresh, n, sigma, rng)

        p2 = optimize_p(p2, n, tracker, tb, quick=True)
        sc2, m2, c2 = score_and_m_with_c(p2)
        update_tracker(tracker, p2, m2, n)

        if sc2 < best_score - 1e-12:
            best = p2.copy()
            best_score = sc2
            fails = 0
        else:
            fails += 1

    return best, best_score


def propose_candidate(seed=20240524, budget_s=2400.0, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    """
    global _HARD_DEADLINE

    t0 = time.monotonic()
    budget = max(120.0, float(budget_s))
    # Leave a hard safety margin so the function returns before termination.
    _HARD_DEADLINE = t0 + budget - 35.0

    rng = np.random.default_rng(seed)
    tracker = {"best_score": np.inf, "best_p": None, "best_n": 0}

    # Use the supplied previous construction if present.
    provided_p = None
    provided = globals().get("height_sequence_1", None)
    if provided is not None:
        try:
            arr = np.asarray(provided, dtype=float)
            if arr.size > 0:
                provided_p = normalize_p(arr)
                # Keep only reasonably long sequences for direct optimization.
                if provided_p.size > 2048:
                    # Still useful as a resampled start, but not direct length.
                    pass
                sc, m, c = score_and_m_with_c(provided_p)
                update_tracker(tracker, provided_p, m, provided_p.size)
                print(f"Provided sequence: n={provided_p.size}, score={sc:.9f}")
        except Exception as e:
            print("Could not parse provided sequence:", e)

    # Configure length searches.  The provided length 256 is prioritized.
    configs = [
        (256, 650.0, 8),
        (384, 250.0, 5),
        (512, 350.0, 6),
        (768, 160.0, 4),
    ]
    if budget > 2200.0:
        configs.append((1024, 100.0, 2))

    reserve = max(600.0, 0.32 * budget)

    for n, alloc, count in configs:
        now = time.monotonic()
        if now > _HARD_DEADLINE - reserve:
            break
        cfg_start = now
        cfg_end = min(_HARD_DEADLINE - reserve, cfg_start + alloc)
        if cfg_end - cfg_start < 25.0:
            break

        print(f"\n--- Length n={n}, budget~{alloc:.0f}s ---")
        inits = generate_inits(n, provided_p, tracker.get("best_p"), rng, count)

        local_best_p = None
        local_best_score = np.inf

        for idx, p0 in enumerate(inits):
            now = time.monotonic()
            if now > cfg_end - 15.0:
                break
            remaining = len(inits) - idx
            per = (cfg_end - now) / max(1, remaining)
            per = min(per, max(25.0, alloc / max(1, count)))
            if per < 12.0:
                break

            print(f"  start {idx + 1}/{len(inits)}", flush=True)
            popt = optimize_p(p0, n, tracker, per, quick=False)
            sc, m, c = score_and_m_with_c(popt)
            update_tracker(tracker, popt, m, n)

            if sc < local_best_score:
                local_best_score = sc
                local_best_p = popt.copy()

            if sc < 1.51 or local_best_score < 1.51:
                print(f"    n={n} score={sc:.9f}  (local best {local_best_score:.9f})")

        # Use any leftover time inside the length budget for exact polishing.
        if local_best_p is not None and time.monotonic() < cfg_end - 10.0:
            tb = max(0.0, min(35.0, cfg_end - time.monotonic()))
            if tb > 6.0:
                local_best_p = pair_polish(local_best_p, n, tracker, tb, rng)
                sc, m, c = score_and_m_with_c(local_best_p)
                update_tracker(tracker, local_best_p, m, n)
                print(f"  after pair polish n={n}: {sc:.9f}")

    # If nothing was found, ensure a valid fallback.
    if tracker.get("best_p") is None:
        if provided_p is not None:
            tracker["best_p"] = provided_p.copy()
            tracker["best_n"] = len(provided_p)
        else:
            fb = np.ones(256, dtype=float) / 256.0
            tracker["best_p"] = fb
            tracker["best_n"] = 256

    best_p = tracker["best_p"]
    n_best = int(len(best_p))
    print(f"\nBest after length passes: score={tracker['best_score']:.9f}, n={n_best}")

    # Final high-level refinement.
    remaining = _HARD_DEADLINE - time.monotonic() - 25.0
    if best_p is not None and remaining > 40.0:
        tb = min(250.0, remaining * 0.35)
        if tb > 20.0:
            best_p = optimize_p(best_p, n_best, tracker, tb, quick=True)
            print(f"After final quick re-optimization: score={tracker['best_score']:.9f}, n={len(best_p)}")

    remaining = _HARD_DEADLINE - time.monotonic() - 30.0
    if best_p is not None and remaining > 60.0:
        tb = min(1000.0, remaining * 0.65)
        if tb > 30.0:
            best_p, _ = basin_hopping(best_p, n_best, tracker, tb, rng)
            print(f"After basin hopping: score={tracker['best_score']:.9f}, n={len(best_p)}")

    # Final exact pair-polish.
    best_p = tracker.get("best_p")
    if best_p is not None:
        remaining = _HARD_DEADLINE - time.monotonic() - 25.0
        tb = min(350.0, remaining)
        if tb > 8.0:
            best_p = pair_polish(best_p, len(best_p), tracker, tb, rng)
            sc, m, c = score_and_m_with_c(best_p)
            update_tracker(tracker, best_p, m, len(best_p))
            print(f"After final pair polish: score={tracker['best_score']:.9f}, n={len(best_p)}")

    # Build final output.
    if tracker.get("best_p") is None:
        final_p = np.ones(256, dtype=float) / 256.0
    else:
        final_p = np.asarray(tracker["best_p"], dtype=float)

    final_p = np.where(np.isfinite(final_p), final_p, 0.0)
    final_p = np.maximum(final_p, 0.0)
    s = float(final_p.sum())
    if s <= 1e-12:
        final_p = np.ones(256, dtype=float) / 256.0
        s = 1.0

    # Scale so numerator/denominator are well conditioned; score is invariant.
    final_p = final_p * (10.0 / s)
    final_p = np.clip(final_p, 0.0, 1000.0)
    final_list = [float(x) for x in final_p]

    # Optional sanity evaluation if the evaluator is available.
    try:
        if "evaluate_sequence" in globals():
            eval_score = evaluate_sequence(final_list)
            print(
                f"FINAL local={tracker['best_score']:.9f}, "
                f"evaluated={float(eval_score):.9f}, n={len(final_list)}"
            )
    except Exception as e:
        # Do not let final validation affect the returned candidate.
        print("Final validation skipped:", e)

    return final_list
