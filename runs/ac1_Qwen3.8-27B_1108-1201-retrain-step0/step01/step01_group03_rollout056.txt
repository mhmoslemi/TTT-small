def _objective_compute(x, n, mode, beta, tau, weight):
    x=np.asarray(x,float).reshape(n).copy()
    if not np.all(np.isfinite(x)):
        return 1e100, np.zeros(n)
    x -= float(np.mean(x))
    x = np.clip(x, -60.0, 60.0)
    x -= x.max()
    e=np.exp(x)
    s=float(e.sum())
    if s <= 0 or not np.isfinite(s): return 1e100, np.zeros(n)
    a=e/s
    need=2*n-1
    L=_fft_len(need)
    fa=np.fft.rfft(a,L)
    c=np.fft.irfft(fa*fa,L)[:need].real
    np.maximum(c,0.0,out=c)
    D=n*c
    np.maximum(D,0.0,out=D)
    f=0.0; v=np.zeros(n)
    if mode == 'lse':
       if beta <= 0:
          beta = 0.1
       mu=float(D.max()) if D.size else 0.0
       zz=(D-mu)/beta
       # avoid underflow
       em=np.exp(zz)
       Z=float(em.sum())
       if Z <=0 or not np.isfinite(Z):
          p=np.full(need,1.0/need)
       else:
          p=em/Z
       Lvalue = mu + (float(np.log(Z)) if Z>0 else 0.0)
       f = 2.0 * beta * Lvalue
       y=_cross_fft_from(p,a,fa,L)
       v=4.0*n*y
    else:
       # quadratic modes
       if mode == 'hinge':
          z=np.maximum(D - tau, 0.0)
          f = weight * float(np.sum(z*z) / n)
          evec = weight * z
       elif mode == 'mix':
          z=np.maximum(D - tau, 0.0)
          f = float(np.sum(D*D) / n) + weight * float(np.sum(z*z) / n)
          evec = D + weight * z
       else: # energy
          evec = D
          f = float(np.sum(D*D) / n)
       y=_cross_fft_from(evec,a,fa,L)
       v=4.0*y
    if not np.all(np.isfinite(v)):
       val = f if np.isfinite(f) else 1e100
       return val, np.zeros(n)
    grad = a * (v - float(np.dot(a, v)))
    if not np.all(np.isfinite(grad)):
       return (f if np.isfinite(f) else 1e100), np.zeros(n)

"""
Search for a non-negative step-function coefficient sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

Strategy:
- Use FFT-based dense/sparse-ish local optimization around the simplex.
- The main variables are x = log(a) (approximately); a = softmax(x), so sum(a)=1.
- Blend several objectives:
    * energy: mean(D^2), where D = n*conv(a,a), encourages flat convolution.
    * mix / hinge: penalizes lags above a tau near the target.
    * log-sum-exp surrogate for the exact max-lag objective.
- Multiple starting points of several lengths are explored, including resampled
  versions of the provided height_sequence_1 global when available, random/
  structured starts, and pair-orientation starts that try to break symmetry.
- Final polishing uses exact line search in the log-domain, optional sparsification,
  and a small number of LP refinements that solve the linearized Chebyshev problem
    min t  s.t.  (a*g)_m <= t,  sum g = 1,  g >= 0.
"""

import numpy as np
import time
from operator import itemgetter
from scipy.optimize import minimize, linprog

_BOUNDS_CACHE = {}
_FFT_LEN_CACHE = {}
_OBJ_CACHE = {"key": None, "x": None, "g": None}


def _fft_len(need):
    need = int(max(1, need))
    if need in _FFT_LEN_CACHE:
        return _FFT_LEN_CACHE[need]
    L = 1
    while L < need:
        L <<= 1
    _FFT_LEN_CACHE[need] = L
    return L


def _bounds_for(n):
    if n not in _BOUNDS_CACHE:
        _BOUNDS_CACHE[n] = [(-40.0, 40.0) for _ in range(n)]
    return _BOUNDS_CACHE[n]


def _normalize(a, target=1.0):
    a = np.asarray(a, dtype=float).reshape(-1).copy()
    if a.size == 0:
        return np.empty(0, dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.full(a.size, target / a.size, dtype=float)
    return a * (target / s)


def _conv_fft(a):
    a = np.asarray(a, dtype=float).reshape(-1)
    n = len(a)
    if n == 0:
        return np.empty(0, dtype=float)
    if n < 8:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    need = 2 * n - 1
    L = _fft_len(need)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:need].real
    return np.maximum(c, 0.0)


def _score(a):
    a = np.asarray(a, dtype=float).reshape(-1)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return float("inf")
    c = _conv_fft(a)
    if c.size == 0:
        return float("inf")
    return float(2.0 * len(a) * float(c.max()) / (s * s))


def _a_to_x(a):
    a = np.asarray(a, dtype=float).reshape(-1)
    a = np.maximum(a, 0.0)
    eps = max(1e-16, 1e-12 / max(1, a.size))
    a = np.maximum(a, eps)
    x = np.log(a)
    x -= float(np.mean(x))
    return np.clip(x, -40.0, 40.0)


def _softmax_x(x):
    x = np.asarray(x, dtype=float).reshape(-1).copy()
    if x.size == 0:
        return x
    if not np.all(np.isfinite(x)):
        return np.full(x.size, 1.0 / x.size, dtype=float)
    x -= float(np.mean(x))
    x = np.clip(x, -60.0, 60.0)
    x -= x.max()
    e = np.exp(x)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(x.size, 1.0 / x.size, dtype=float)
    return e / s


def _cross_fft_from(weights, a, fa, L):
    fw = np.fft.rfft(weights, L)
    y = np.fft.irfft(fw * np.conj(fa), L)
    return y[:len(a)].real


def _objective_compute(x, n, mode, beta, tau, weight):
    x = np.asarray(x, dtype=float).reshape(n).copy()
    if not np.all(np.isfinite(x)):
        return 1e100, np.zeros(n, dtype=float)

    x -= float(np.mean(x))
    x = np.clip(x, -60.0, 60.0)
    x -= x.max()
    e = np.exp(x)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return 1e100, np.zeros(n, dtype=float)
    a = e / s

    need = 2 * n - 1
    L = _fft_len(need)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:need].real
    c = np.maximum(c, 0.0)
    D = n * c
    D = np.maximum(D, 0.0)

    if mode == "lse":
        if beta <= 0.0:
            beta = 0.1
        mu = float(D.max()) if D.size else 0.0
        zz = (D - mu) / beta
        em = np.exp(zz)
        Z = float(em.sum())
        if Z <= 0.0 or not np.isfinite(Z):
            p = np.full(need, 1.0 / need, dtype=float)
            logZ = 0.0
        else:
            p = em / Z
            logZ = float(np.log(Z))
        f = 2.0 * beta * (mu + logZ)
        y = _cross_fft_from(p, a, fa, L)
        v = 4.0 * n * y
    else:
        if mode == "hinge":
            z = np.maximum(D - tau, 0.0)
            f = weight * float(np.sum(z * z) / n)
            evec = weight * z
        elif mode == "mix":
            z = np.maximum(D - tau, 0.0)
            f = float(np.sum(D * D) / n) + weight * float(np.sum(z * z) / n)
            evec = D + weight * z
        else:
            evec = D
            f = float(np.sum(D * D) / n)
        y = _cross_fft_from(evec, a, fa, L)
        v = 4.0 * y

    if not np.all(np.isfinite(v)):
        val = float(f) if np.isfinite(f) else 1e100
        return val, np.zeros(n, dtype=float)

    grad = a * (v - float(np.dot(a, v)))
    if not np.all(np.isfinite(grad)):
        val = float(f) if np.isfinite(f) else 1e100
        return val, np.zeros(n, dtype=float)

    return float(f), grad


def _objective_f(x, n, mode, beta, tau, weight):
    f, g = _objective_compute(x, n, mode, beta, tau, weight)
    _OBJ_CACHE["key"] = (int(n), str(mode), float(beta), float(tau), float(weight))
    _OBJ_CACHE["x"] = np.asarray(x, dtype=float).reshape(n).copy()
    _OBJ_CACHE["g"] = g
    return f


def _objective_g(x, n, mode, beta, tau, weight):
    key = (int(n), str(mode), float(beta), float(tau), float(weight))
    if _OBJ_CACHE["key"] == key:
        cx = _OBJ_CACHE["x"]
        xx = np.asarray(x, dtype=float).reshape(-1)
        if cx is not None and cx.size == xx.size and np.array_equal(cx, xx):
            return _OBJ_CACHE["g"]
    _, g = _objective_compute(x, n, mode, beta, tau, weight)
    _OBJ_CACHE["key"] = key
    _OBJ_CACHE["x"] = np.asarray(x, dtype=float).reshape(n).copy()
    _OBJ_CACHE["g"] = g
    return g


def _lbfgs_once(x0, n, mode, beta, tau, weight, maxiter, deadline):
    x0 = np.asarray(x0, dtype=float).reshape(n).copy()
    if maxiter <= 0 or time.monotonic() > deadline:
        return x0
    maxiter = int(max(1, maxiter))
    options = {
        "maxiter": maxiter,
        "maxfun": max(200, 10 * maxiter),
        "ftol": 1e-10,
        "gtol": 1e-5,
        "disp": False,
        "maxls": 12,
    }
    try:
        res = minimize(
            _objective_f,
            x0,
            jac=_objective_g,
            method="L-BFGS-B",
            args=(n, mode, beta, tau, weight),
            bounds=_bounds_for(n),
            options=options,
        )
        if res is not None and hasattr(res, "x") and np.all(np.isfinite(res.x)):
            return np.asarray(res.x, dtype=float).reshape(n).copy()
    except Exception:
        pass
    return x0


def _refine_sequence(a0, deadline, full=False):
    a = _normalize(a0)
    n = len(a)
    best_a = a.copy()
    best_s = _score(a)
    cur_a = a.copy()
    cur_s = best_s
    x = _a_to_x(cur_a)

    scale = 1.0
    if n > 1024:
        scale = 0.55
    elif n > 512:
        scale = 0.75
    if not full:
        scale *= 0.65
    scale = max(0.25, scale)

    cm = 0.5 * best_s
    if best_s > 1.505:
        tau1 = 0.775
        tau2 = 0.755
        tau_f = 0.751
    else:
        tau1 = max(0.50, 0.995 * cm)
        tau2 = max(0.50, 0.9875 * cm)
        tau_f = max(0.50, 0.980 * cm)

    phases = [
        ("energy", 0.0, tau1, 1.0, int(180 * scale)),
        ("mix", 0.0, tau1, 2.0, int(120 * scale)),
    ]
    if full:
        phases.append(("mix", 0.0, tau2, 10.0, int(160 * scale)))
        phases.append(("mix", 0.0, tau_f, 25.0, int(160 * scale)))

    phases.append(("lse", 0.12, 0.0, 1.0, int(120 * scale)))
    if full:
        phases.append(("lse", 0.05, 0.0, 1.0, int(160 * scale)))
    phases.append(("lse", 0.02, 0.0, 1.0, int(180 * scale)))
    if full:
        phases.append(("lse", 0.008, 0.0, 1.0, int(220 * scale)))

    hr = float(min(tau_f, max(0.50, 0.985 * cm)))
    phases.append(("hinge", 0.0, hr, (120.0 if full else 60.0), int(150 * scale)))

    no_improve = 0
    for idx, (mode, beta, tau, w, mx) in enumerate(phases):
        if time.monotonic() > deadline - 12.0:
            break
        mx = int(max(15, mx))
        x_prev = x.copy()
        x = _lbfgs_once(x, n, mode, beta, tau, w, mx, deadline - 10.0)
        cand_a = _normalize(_softmax_x(x))
        cand_s = _score(cand_a)

        if np.isfinite(cand_s) and cand_s <= cur_s * 1.10 + 1e-9:
            cur_a = cand_a.copy()
            cur_s = cand_s
            x = _a_to_x(cur_a)
        else:
            x = x_prev
            cand_s = cur_s

        if np.isfinite(cand_s) and cand_s < best_s:
            best_s = cand_s
            best_a = cur_a.copy()
            no_improve = 0
            print(f"    refined n={n} phase {idx} mode={mode} score={best_s:.9f}")
        else:
            no_improve += 1

        if no_improve >= 3 and idx > 0:
            break

    return best_a, best_s, x


def _polish_sequence(a0, deadline, iters=150):
    a = _normalize(a0)
    n = len(a)
    if n < 2:
        return a, _score(a)

    best_a = a.copy()
    best_s = _score(a)
    x = _a_to_x(a)

    betas = [0.08, 0.03, 0.012, 0.005]
    if n > 1024:
        betas = betas[:2]

    iters = int(max(0, iters))
    per_beta = max(4, iters // max(1, len(betas)))

    for beta in betas:
        if time.monotonic() > deadline:
            break
        stuck = 0
        for _ in range(per_beta):
            if time.monotonic() > deadline:
                break

            c = _conv_fft(a)
            D = n * c
            D = np.maximum(D, 0.0)
            if D.size == 0:
                break

            mu = float(D.max())
            zz = (D - mu) / beta
            p = np.exp(zz)
            zsum = float(p.sum())
            if zsum <= 0.0 or not np.isfinite(zsum):
                break
            p = p / zsum

            L = _fft_len(2 * n - 1)
            fa = np.fft.rfft(a, L)
            y = _cross_fft_from(p, a, fa, L)
            v = 4.0 * n * y
            g = a * (v - float(np.dot(a, v)))
            gn = float(np.linalg.norm(g))
            if gn < 1e-12:
                stuck = per_beta + 1
                break
            d = g / gn

            improved = False
            for alpha in (0.05, 0.12, 0.25, 0.5, 0.9, 1.5):
                x_new = x + alpha * d
                a_new = _normalize(_softmax_x(x_new))
                s_new = _score(a_new)
                if s_new < best_s:
                    best_s = s_new
                    best_a = a_new.copy()
                    x = x_new.copy()
                    a = a_new.copy()
                    improved = True
                    break

            if improved:
                stuck = 0
            else:
                stuck += 1
                if stuck > 4:
                    break
        if stuck > per_beta:
            break

    return best_a, best_s


def _sparsify_sequence(a0, deadline):
    a = _normalize(a0)
    best_a = a.copy()
    best_s = _score(a)
    n = a.size

    for frac in (0.005, 0.01, 0.02, 0.05):
        if time.monotonic() > deadline:
            break
        a2 = a.copy()
        if n > 1:
            thr = float(np.quantile(a, frac))
            a2[a2 <= thr] = 0.0
        a2 = _normalize(a2)
        s2 = _score(a2)
        if s2 < best_s:
            best_s = s2
            best_a = a2.copy()
    return best_a, best_s


def _lp_improve(a0, deadline, attempts=2):
    a = _normalize(a0)
    n = len(a)
    best_a = a.copy()
    best_s = _score(a)

    if n < 4 or n > 1024:
        return best_a, best_s

    max_at = int(min(2, attempts))
    if n > 768:
        max_at = min(max_at, 1)

    for attempt in range(max_at):
        if time.monotonic() > deadline:
            break

        try:
            A = np.zeros((2 * n - 1, n + 1), dtype=float)
            A[:, 0] = -1.0
            for j in range(n):
                A[j:j + n, j + 1] = a

            b_ub = np.zeros(2 * n - 1, dtype=float)
            A_eq = np.zeros((1, n + 1), dtype=float)
            A_eq[0, 1:] = 1.0
            b_eq = np.ones(1, dtype=float)
            c_obj = np.zeros(n + 1, dtype=float)
            c_obj[0] = 1.0
            bounds = [(0.0, None)] + [(0.0, None)] * n

            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "disp": False},
            )
        except Exception:
            break

        if res is None or res.status != 0:
            break

        g = np.asarray(res.x[1:], dtype=float)
        g = np.maximum(g, 0.0)
        gs = float(g.sum())
        if gs < 1e-12:
            break
        g = g / gs

        before = best_s
        s_g = _score(g)
        if s_g < best_s:
            best_s = s_g
            best_a = g.copy()

        for alpha in (0.1, 0.25, 0.5, 0.8):
            if time.monotonic() > deadline:
                break
            mix = _normalize((1.0 - alpha) * best_a + alpha * g)
            s_mix = _score(mix)
            if s_mix < best_s:
                best_s = s_mix
                best_a = mix.copy()

        if best_s >= before - 1e-12:
            break
        a = best_a.copy()

    return best_a, best_s


def _resize_global(g, n):
    n = int(n)
    if g is None or len(g) == 0 or n <= 0:
        return _normalize(np.ones(n, dtype=float))
    g = np.asarray(g, dtype=float).reshape(-1)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    if g.size == n:
        return _normalize(g)
    if g.size == 1:
        return _normalize(np.full(n, float(g[0])))
    xo = np.linspace(0.0, 1.0, g.size)
    xn = np.linspace(0.0, 1.0, n)
    a = np.interp(xn, xo, g)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _make_spec_start(n, kind, rng, global_seq, param):
    n = int(n)
    if kind == "global":
        base = _resize_global(global_seq, n)
        if param <= 0.0:
            return _normalize(base)
        x = _a_to_x(base) + rng.normal(0.0, param, n)
        return _normalize(_softmax_x(x))

    if kind == "pairs":
        base = _resize_global(global_seq, n)
        a = base.copy()
        m = n // 2
        if m > 0:
            u = rng.beta(0.8, 4.5, m)
            flip = rng.random(m) < 0.5
            for k in range(m):
                i = k
                j = n - 1 - k
                s = a[i] + a[j]
                if s <= 0.0:
                    continue
                if flip[k]:
                    a[i] = s * u[k]
                    a[j] = s * (1.0 - u[k])
                else:
                    a[i] = s * (1.0 - u[k])
                    a[j] = s * u[k]
        return _normalize(a)

    if kind == "cos":
        i = np.linspace(0.0, 1.0, n)
        k = 2 + int(round(float(param) * 4.0))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a1 = 0.12 + 0.88 * np.abs(np.sin(2.0 * np.pi * k * i + ph))
        a2 = 0.18 + 0.82 * np.abs(np.sin(2.0 * np.pi * (2 * k + 1) * i + 1.3 * ph))
        return _normalize(0.65 * a1 + 0.35 * a2)

    # random start
    a = 0.1 + 0.9 * rng.random(n)
    if float(param) > 0.5:
        a = a * np.exp(rng.normal(0.0, 0.35, n))
    return _normalize(a)

            continue
        if score is not None and np.isfinite(score) and score <= best_score * 1.015 + 1e-9:
            same = False
            try:
                same = (len(seq) == len(top_seqs[0]) and np.allclose(seq, top_seqs[0], rtol=1e-8, atol=1e-10))
            except Exception:
                same = False
            if not same:
                top_seqs.append(seq.copy())

    full_deadline = max(time.monotonic() + 90.0, deadline - 120.0)
    for idx, a0 in enumerate(top_seqs):
        now = time.monotonic()
        if now > full_deadline or now > deadline - 180:
            break
        want = 1400.0 if idx == 0 else 500.0
        dl = min(full_deadline, now + want)
        if dl - time.monotonic() < 45.0:
            continue
        try:
            seq, score, _ = _refine_sequence(a0, dl, full=True)
        except Exception:
            continue
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  full refinement improved: n={len(seq)} score={score:.9f}")

    # Final local polish and special fixes.
    if best_seq is not None:
        if time.monotonic() < deadline - 60.0:
            try:
                a2, s2 = _sparsify_sequence(best_seq, deadline - 60.0)
                if s2 < best_score:
                    best_score = s2
                    best_seq = a2
                    print(f"  sparsify improved: n={len(best_seq)} score={score:.9f}")
            except Exception:
                pass

        if time.monotonic() < deadline - 60.0:
            n = len(best_seq)
            iters = 260 if n <= 512 else (120 if n <= 1024 else 50)
            try:
                a2, s2 = _polish_sequence(best_seq, deadline - 60.0, iters=iters)
                if s2 < best_score:
                    best_score = s2
                    best_seq = a2
                    print(f"  polish improved: n={len(best_seq)} score={s2:.9f}")
            except Exception:
                pass

        if time.monotonic() < deadline - 100.0:
            try:
                a2, s2 = _lp_improve(best_seq, deadline - 100.0, attempts=2)
                if s2 < best_score:
                    best_score = s2
                    best_seq = a2
                    print(f"  LP improved: n={len(best_seq)} score={s2:.9f}")
            except Exception:
                pass

        if best_score < 1.51 and time.monotonic() < deadline - 50.0:
            n = len(best_seq)
            iters = 180 if n <= 512 else (80 if n <= 1024 else 30)
            try:
                a2, s2 = _polish_sequence(best_seq, deadline - 50.0, iters=iters)
                if s2 < best_score:
                    best_score = s2
                    best_seq = a2
                    print(f"  final polish improved: n={len(best_seq)} score={s2:.9f}")
            except Exception:
                pass

    if best_seq is None:
        best_seq = _normalize(np.ones(256, dtype=float))
        best_score = _score(best_seq)

    # Final safety: convert to Python floats, guarantee finite non-negative values.
    final_seq = []
    for v in best_seq:
        try:
            fv = float(v)
        except Exception:
            fv = 0.0
        if not np.isfinite(fv) or fv < 0.0:
            fv = 0.0
        if fv > 1000.0:
            fv = 1000.0
        final_seq.append(fv)

    if sum(final_seq) < 0.01:
        final_seq = [1.0] * 256

    try:
        eval_score = float(evaluate_sequence(final_seq))
        if np.isfinite(eval_score) and eval_score < best_score - 1e-12:
            best_score = eval_score
        print(f"final returned length={len(final_seq)} internal_score={_score(final_seq):.9f} eval_score={eval_score:.9f}")
    except Exception as exc:
        print(f"evaluate_sequence check failed: {exc}")

    return final_seq
