"""
Hybrid optimization for minimizing the peak of self-convolution of non-negative
step functions. Combines spectral initialization (Gerchberg-Saxton style),
subgradient descent with multi-lag focusing, and multi-scale simulated annealing.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def subgradient_at_peak(a, c, num_lags=5):
    """
    Compute a weighted subgradient focusing on the top num_lags peaks.
    Returns (g, peak_val, peak_lags).
    """
    n = len(a)
    peak_val = c.max()
    # Find lags within a fraction of the max
    threshold = peak_val * (1.0 - 0.05 * num_lags)  # lags within 5%, 10%, etc. of max
    peak_lags = np.where(c >= threshold)[0]
    
    if len(peak_lags) == 0:
        peak_lags = np.array([np.argmax(c)])
    
    # Weighted subgradient
    g = np.zeros(n)
    weights = []
    for k in peak_lags:
        # Weight by how close to the max
        w = c[k] / peak_val
        weights.append(w)
        # d c[k] / d a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        # First term: a[k-i] valid for i in [max(0, k-n+1), min(n-1, k)]
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            g[lo1:hi1+1] += a[k - np.arange(lo1, hi1+1)] * w
        # Second term: a[i-k] valid for i in [max(0, k), min(n-1, k+n-1)]
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            g[lo2:hi2+1] += a[np.arange(lo2, hi2+1) - k] * w
    
    total_w = sum(weights)
    g /= total_w
    return g, peak_val, peak_lags

    a /= a.sum()

        A = np.fft.fft(a)

        # For flat r: |A[k]|^2 = n * r[k] = 1 for all k.

        A = np.exp(1j * phase)  # |A| = 1 for all k

        a_new = np.clip(a_new, 0.0, None)

    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score efficiently."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_subgradient(a, c, top_lags=5, smooth_frac=0.1):
    """
    Compute weighted subgradient focusing on top lags.
    Returns (gradient, peak_value, num_active_lags).
    """
    n = len(a)
    peak_val = c.max()
    threshold = peak_val * (1.0 - smooth_frac * top_lags)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])

    g = np.zeros(n)
    total_w = 0.0
    for k in active:
        w = c[k]
        total_w += w
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            g[idx] += w * a[k - idx]
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            g[idx] += w * a[idx - k]

    g /= max(total_w, 1e-30)
    return g, peak_val, len(active)


def gradient_optimize(a_init, time_limit, rng, top_lags=5):
    """
    Subgradient descent with adaptive step size, annealed smoothing,
    and periodic restarts.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = a / a.sum() * target_sum

    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()

    lr = 0.005
    lr_min = 1e-7
    lr_max = 0.05
    patience = 0
    max_patience = 500
    smooth_frac = 0.15

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        peak_val = c.max()
        score = 2.0 * n * peak_val / (a.sum() ** 2)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            a = best_a * np.exp(rng.normal(0, 0.015, n))
            a = np.clip(a, 1e-6, None)
            a = a / a.sum() * target_sum
            lr = 0.005
            smooth_frac = 0.2
            patience = 0
            continue

        g, pv, n_active = compute_subgradient(a, c, top_lags=top_lags, smooth_frac=smooth_frac)

        a_new = a - lr * g
        a_new = np.clip(a_new, 1e-8, None)
        a_new = a_new / a_new.sum() * target_sum

        new_score = local_score(a_new)
        if new_score < score:
            a = a_new
            lr = min(lr * 1.03, lr_max)
            smooth_frac = max(smooth_frac * 0.999, 0.02)
        else:
            lr *= 0.95
            if lr < lr_min:
                lr = lr_min
                patience += 10

    return best_a, best_score


def targeted_local_search(a_init, time_limit, rng):
    """
    Local search that specifically targets peak-contributing coefficients.
    Makes small coordinated adjustments to reduce the convolution peak.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    current_score = local_score(a)
    best = a.copy()
    best_score = current_score
    t0 = time.monotonic()

    update_count = 0
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))

        # Find top contributing pairs to the peak
        n_half = min(n, peak_idx + 1)
        contribs = np.zeros(n_half)
        for i in range(n_half):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        top_k = max(3, n // 16)
        top_idx = np.argsort(contribs)[-top_k:]

        # Strategy: reduce top contributors slightly, increase "safe" positions
        candidate = a.copy()

        # Determine safe positions (low contribution to all top peaks)
        top_peaks = np.argsort(c)[-3:]
        safe_score = np.zeros(n)
        for pk in top_peaks:
            pk_half = min(n, pk + 1)
            for i in range(pk_half):
                j = pk - i
                if 0 <= j < n:
                    safe_score[i] += c[pk] * a[i] * a[j]
                    if i != j:
                        safe_score[j] += c[pk] * a[i] * a[j]

        safe_positions = np.where(safe_score < np.median(safe_score))[0]
        safe_positions = safe_positions[safe_positions != peak_idx]

        if len(safe_positions) == 0:
            safe_positions = np.setdiff1d(np.arange(n), top_idx)

        r = rng.random()

        if r < 0.4:
            # Reduce peak contributors, boost safe positions
            n_reduce = max(2, len(top_idx) // 2)
            reduce_idx = top_idx[:n_reduce]
            factor_red = np.exp(rng.uniform(-0.08, -0.02, len(reduce_idx)))
            candidate[reduce_idx] *= factor_red

            n_boost = max(2, len(reduce_idx))
            if len(safe_positions) >= n_boost:
                boost_idx = rng.choice(safe_positions, n_boost, replace=False)
                factor_boost = np.exp(rng.uniform(0.02, 0.08, len(boost_idx)))
                candidate[boost_idx] *= factor_boost
        elif r < 0.6:
            # Swap weight between peak and safe positions
            n_swap = max(2, n // 20)
            src = rng.choice(top_idx, min(n_swap, len(top_idx)), replace=False)
            dst = rng.choice(safe_positions, min(n_swap, len(safe_positions)), replace=False)
            amount = a[src] * rng.uniform(0.02, 0.08, len(src))
            candidate[src] -= amount
            candidate[dst] += amount
        elif r < 0.75:
            # Block reduction around peak
            half_width = max(1, n // 32)
            center = peak_idx // 2
            for offset in range(-half_width, half_width + 1):
                pos = center + offset
                if 0 <= pos < n and pos in top_idx:
                    candidate[pos] *= np.exp(rng.uniform(-0.06, -0.01))
                elif 0 <= pos < n:
                    candidate[pos] *= np.exp(rng.uniform(0.0, 0.03))
        else:
            # Random small perturbation
            n_pert = max(3, n // 10)
            pert_idx = rng.choice(n, n_pert, replace=False)
            candidate[pert_idx] *= np.exp(rng.normal(0, 0.03, n_pert))

        candidate = np.clip(candidate, 1e-8, None)
        new_score = local_score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            update_count += 1
            stall = 0
            if new_score < best_score:
                best_score = new_score
                best = candidate.copy()
        else:
            stall += 1
            if stall > max_stall:
                # Reheat: take a bigger random step
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.clip(a, 1e-8, None)
                current_score = local_score(a)
                stall = 0

    return best, best_score


def sa_exploration(a_init, time_limit, rng):
    """
    Simulated annealing with diverse perturbation strategies for exploration.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    current_score = local_score(a)
    best = a.copy()
    best_score = current_score
    t0 = time.monotonic()

    temp = 0.005
    cooling = 0.99975
    min_temp = 1e-7
    reheat_count = 0
    max_reheats = 20
    stall = 0

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.25:
            # Peak-targeted reduction
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            n_half = min(n, peak_idx + 1)
            contribs = np.zeros(n_half)
            for i in range(n_half):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 14)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.06, 0.025, len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(8, len(other))
                inc_idx = rng.choice(other, n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.06, 0.025, n_inc))
        elif r < 0.4:
            # Block perturbation
            width = int(rng.integers(1, max(2, n // 6)))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.05, width))
            other = np.setdiff1d(np.arange(n), range(start, start + width))
            if len(other) > 0:
                n_comp = min(width, len(other))
                comp_idx = rng.choice(other, n_comp, replace=False)
                candidate[comp_idx] *= np.exp(rng.normal(0, 0.03, n_comp))
        elif r < 0.55:
            # Symmetry modification
            center = n // 2
            side = int(rng.integers(1, min(center, n // 4) + 1))
            t = rng.uniform(0.3, 0.7)
            left = a[center - side:center]
            right = a[center:center + side]
            candidate[center - side:center] = (1 - t) * left + t * right[::-1]
            candidate[center:center + side] = (1 - t) * right + t * left[::-1]
        elif r < 0.7:
            # Spectral perturbation
            A = np.fft.fft(a)
            n_bins = int(rng.integers(1, 4))
            for _ in range(n_bins):
                k = int(rng.integers(1, n // 2))
                phase_shift = rng.uniform(-0.8, 0.8)
                mag_scale = np.exp(rng.uniform(-0.3, 0.3))
                A[k] *= mag_scale * np.exp(1j * phase_shift)
                A[n - k] = np.conj(A[k])
            candidate = np.fft.ifft(A).real
        elif r < 0.85:
            # Multi-scale perturbation
            for scale in [n // 16, n // 4, n]:
                width = max(1, scale // 4)
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, width))
        else:
            # Global perturbation
            candidate = a * np.exp(rng.normal(0, 0.02, n))

        candidate = np.clip(candidate, 1e-8, None)
        new_score = local_score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            stall = 0
            if new_score < best_score:
                best_score = new_score
                best = candidate.copy()
                temp = max(temp * 0.99, min_temp)
        elif temp > min_temp and rng.random() < np.exp(-(new_score - current_score) / temp):
            a = candidate
            current_score = new_score
            stall += 1
        else:
            stall += 1

        temp = max(temp * cooling, min_temp)

        if stall > 800 and reheat_count < max_reheats:
            temp = 0.003
            stall = 0
            reheat_count += 1
            a = best * np.exp(rng.normal(0, 0.02, n))
            a = np.clip(a, 1e-8, None)
            current_score = local_score(a)

    return best, best_score


def final_polish(a_init, time_limit, rng):
    """
    Very fine-grained greedy local search for final improvement.
    Only accepts improvements.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    current_score = local_score(a)
    best = a.copy()
    best_score = current_score
    t0 = time.monotonic()

    step_size = 0.01
    min_step = 1e-5
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Greedy: try small perturbations, only accept improvements
        found_improvement = False

        for attempt in range(50):
            candidate = a.copy()
            r = rng.random()

            if r < 0.4:
                # Single coefficient adjustment
                idx = int(rng.integers(0, n))
                delta = step_size * rng.uniform(-1, 1)
                candidate[idx] = max(1e-8, a[idx] + delta)
            elif r < 0.7:
                # Pair adjustment
                idx1 = int(rng.integers(0, n))
                idx2 = int(rng.integers(0, n))
                if idx1 != idx2:
                    delta1 = step_size * rng.uniform(-1, 1)
                    delta2 = step_size * rng.uniform(-1, 1)
                    candidate[idx1] = max(1e-8, a[idx1] + delta1)
                    candidate[idx2] = max(1e-8, a[idx2] + delta2)
            else:
                # Small block
                width = int(rng.integers(1, 5))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] += step_size * rng.uniform(-1, 1, width)
                candidate = np.clip(candidate, 1e-8, None)

            new_score = local_score(candidate)
            if new_score < current_score:
                a = candidate
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best = candidate.copy()
                found_improvement = True
                break

        if not found_improvement:
            no_improve += 1
            if no_improve % 20 == 0:
                step_size *= 0.8
                if step_size < min_step:
                    break
            # Occasional random kick
            if no_improve % 100 == 0:
                kick = 0.3 * step_size
                idxs = rng.choice(n, 5, replace=False)
                a[idxs] += kick * rng.uniform(-1, 1, 5)
                a = np.clip(a, 1e-8, None)
                current_score = local_score(a)
        else:
            no_improve = 0
            step_size = min(step_size * 1.02, 0.05)

    return best, best_score


def gerchberg_saxton_init(n, rng, iters=80):
    """
    Initialize using Gerchberg-Saxton to approximate flat circular autocorrelation.
    """
    a = 1.0 + 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a /= a.sum()

    for it in range(iters):
        A = np.fft.fft(a)
        phase = np.angle(A)
        target_mag = np.ones(n)
        target_mag[0] = np.abs(A[0])  # Preserve DC
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new /= a_new.sum()
        a = a_new

    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase hybrid search: gradient optimization, targeted local search,
    SA exploration, and final greedy polish. Returns best sequence found.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # === Phase 1: Gradient-based optimization (400s) ===
    print("\n=== Phase 1: Gradient optimization ===")
    if best_seq is not None:
        n_main = len(best_seq)
    else:
        n_main = 256

    phase1_time = min(400.0, max(60.0, (deadline - time.monotonic()) * 0.25))
    a_opt, grad_score = gradient_optimize(best_seq if best_seq is not None else np.ones(n_main),
                                          phase1_time, rng, top_lags=5)
    actual_score = local_score(a_opt)
    print(f"  n={n_main}, grad score: {actual_score:.6f}")
    if actual_score < best_score:
        best_score = actual_score
        best_seq = a_opt.copy()
        print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 2: Targeted local search (600s) ===
    print(f"\n=== Phase 2: Targeted local search ===")
    if best_seq is not None:
        phase2_time = min(600.0, max(60.0, (deadline - time.monotonic()) * 0.3))
        print(f"  Starting, score={best_score:.6f}, time={phase2_time:.0f}s")
        a_tgt, tgt_score = targeted_local_search(best_seq, phase2_time, rng)
        if tgt_score < best_score:
            best_score = tgt_score
            best_seq = a_tgt.copy()
            print(f"  After targeted search: {best_score:.6f}")
        else:
            print(f"  No improvement (best remains {best_score:.6f})")

    # === Phase 3: SA exploration (600s) ===
    print(f"\n=== Phase 3: SA exploration ===")
    if best_seq is not None:
        phase3_time = min(600.0, max(60.0, (deadline - time.monotonic()) * 0.3))
        print(f"  Starting, score={best_score:.6f}, time={phase3_time:.0f}s")
        a_sa, sa_score = sa_exploration(best_seq, phase3_time, rng)
        if sa_score < best_score:
            best_score = sa_score
            best_seq = a_sa.copy()
            print(f"  After SA: {best_score:.6f}")
        else:
            print(f"  No improvement (best remains {best_score:.6f})")

    # === Phase 4: Try a different length quickly ===
    print(f"\n=== Phase 4: Alternative length exploration ===")
    if best_seq is not None and time.monotonic() < deadline - 120:
        alt_lengths = [220, 300, 384]
        remaining = deadline - time.monotonic() - 120
        for n_alt in alt_lengths:
            if time.monotonic() > deadline - 60:
                break
            t_per = min(60.0, remaining / 3.0)
            if t_per < 15:
                break
            a_alt = gerchberg_saxton_init(n_alt, rng, iters=30)
            a_opt_alt, score_alt = gradient_optimize(a_alt, t_per, rng, top_lags=5)
            actual_alt = local_score(a_opt_alt)
            print(f"  n={n_alt}, score: {actual_alt:.6f}")
            if actual_alt < best_score:
                best_score = actual_alt
                best_seq = a_opt_alt.copy()
                print(f"  *** New best at n={n_alt}: {best_score:.6f} ***")

    # === Phase 5: Final greedy polish (remaining time) ===
    print(f"\n=== Phase 5: Final greedy polish ===")
    if best_seq is not None:
        polish_time = max(20.0, deadline - time.monotonic() - 5)
        print(f"  Starting polish, score={best_score:.6f}, time={polish_time:.0f}s")
        a_pol, pol_score = final_polish(best_seq, polish_time, rng)
        if pol_score < best_score:
            best_score = pol_score
            best_seq = a_pol.copy()
            print(f"  After polish: {best_score:.6f}")

    # Phase 5b: One more SA round if time allows
    if best_seq is not None and time.monotonic() < deadline - 30:
        extra_time = min(120.0, deadline - time.monotonic() - 5)
        if extra_time > 20:
            print(f"\n=== Phase 5b: Extra SA round ===")
            a_extra, ext_score = sa_exploration(best_seq, extra_time, rng)
            if ext_score < best_score:
                best_score = ext_score
                best_seq = a_extra.copy()
                print(f"  After extra SA: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
