import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def smooth_max_and_weights(c, t):
    """Compute smooth max and softmax weights."""
    c_max = c.max()
    shifted = t * (c - c_max)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    log_sum_exp = c_max + np.log(total) / t
    weights = exp_shifted / total
    return log_sum_exp, weights

def compute_gradient_from_weights(a, weights, threshold=1e-6):
    """Compute gradient of smooth max conv using precomputed weights."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    active = np.where(weights > threshold)[0]
    for k in active:
        w = weights[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # grad[lo:hi+1] += w * 2 * a[k-hi : k-lo+1][::-1]
            segment = a[k-hi:k-lo+1][::-1]
            grad[lo:hi+1] += w * 2.0 * segment
    return grad

def softmax(z):
    """Stable softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS with smooth max objective
and multiplicative weights, with annealing temperature and multiple restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (max(size, 2) - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def stable_softmax(z):
    """Numerically stable softmax."""
    z_max = z.max()
    e = np.exp(z - z_max)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(len(z)) / len(z)
    return e / s

def smooth_max_value(c, t):
    """Compute smooth max (log-sum-exp) of array c with temperature t."""
    c_max = c.max()
    if t < 1e-6:
        return c_max
    shifted = t * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_total = np.log(exp_shifted.sum())
    return c_max + log_total / t

def compute_smooth_gradient(a, c, t, threshold=1e-6):
    """
    Compute the gradient of the smooth max of conv(a,a) with respect to a.
    Uses the log-sum-exp weights to form a weighted subgradient.
    """
    n = len(a)
    c_max = c.max()
    shifted = t * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total <= 0:
        return np.zeros(n)
    weights = exp_shifted / total
    
    grad = np.zeros(n, dtype=np.float64)
    # Only process lags with significant weight
    active = np.where(weights > threshold)[0]
    for k in active:
        w = weights[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # dc[k]/da[m] = 2*a[k-m] for m in [lo, hi]
            # a[k-m] for m in [lo,hi] is a[k-hi : k-lo+1][::-1]
            grad[lo:hi+1] += w * 2.0 * a[k-hi:k-lo+1][::-1]
    return grad

def run_lbfgs_smooth(a_init, n, S, t_start, t_end, max_iter, deadline, rng):
    """
    Run L-BFGS-B optimization with annealing temperature.
    Uses softmax parameterization: a = softmax(z) * S.
    """
    n = len(a_init)
    a0 = np.clip(a_init, 1e-10, None)
    a0 = a0 / a0.sum() * S  # normalize to sum S
    
    # Initialize z from a0: z = log(a0/S) = log(a0) - log(S)
    z = np.log(a0 / S + 1e-30)
    z = z - z.mean()  # center to avoid the flat direction
    
    best_obj = np.inf
    best_z = z.copy()
    
    # Anneal temperature from t_start to t_end over the course of optimization
    n_phases = 5
    iters_per_phase = max_iter // n_phases
    
    for phase in range(n_phases):
        if time.monotonic() > deadline:
            break
        t = t_start * (t_end / t_start) ** (phase / max(1, n_phases - 1))
        
        def func_grad(z):
            a_norm = stable_softmax(z)
            a = a_norm * S
            c = fft_convolve_self(a_norm)  # scale invariant, use normalized
            val = smooth_max_value(c, t)
            obj = 2.0 * n * val  # since sum(a_norm)=1, obj = 2n * max_c_norm
            # But we want the actual objective: 2n * max_c / S^2
            # max_c = S^2 * max_c_norm, so obj = 2n * S^2 * max_c_norm / S^2 = 2n * max_c_norm
            # So obj = 2n * val is correct!
            
            grad_a_norm = compute_smooth_gradient(a_norm, c, t)
            center = np.dot(a_norm, grad_a_norm)
            grad_z = a_norm * (grad_a_norm - center)
            
            return obj, grad_z
        
        try:
            result = scipy_minimize(func_grad, z, method='L-BFGS-B',
                                   jac=True, options={'maxiter': iters_per_phase,
                                                      'ftol': 1e-14, 'gtol': 1e-10,
                                                      'maxfun': iters_per_phase * 2})
            z = result.x
            cur_obj = func_grad(z)[0]
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_z = z.copy()
            print(f"  Phase {phase+1}/{n_phases}, t={t:.2f}, obj={cur_obj:.6f}, best={best_obj:.6f}")
        except Exception as e:
            print(f"  L-BFGS error in phase {phase}: {e}")
    
    return best_z, best_obj, stable_softmax(best_z) * S

def run_mirror_descent(a_init, n, S, t, eta_init, max_iter, deadline, rng):
    """
    Multiplicative weights (mirror descent) with smooth subgradient.
    """
    a = np.clip(a_init, 1e-10, 1000.0).copy()
    a = a / a.sum() * S
    best_obj = np.inf
    best_a = a.copy()
    eta = eta_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = fft_convolve_self(a / S)  # normalize for computation
        val = smooth_max_value(c, t)
        obj = 2.0 * n * val
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        grad_a = compute_smooth_gradient(a / S, c, t) * S  # scale back
        
        # Adaptive step size (backtracking)
        for trial in range(20):
            a_new = a * np.exp(-eta * grad_a / S)
            a_new = np.clip(a_new, 1e-15, None)
            s = a_new.sum()
            if s <= 0:
                eta *= 0.5
                continue
            a_new = a_new / s * S
            
            c_new = fft_convolve_self(a_new / S)
            val_new = smooth_max_value(c_new, t)
            obj_new = 2.0 * n * val_new
            
            if obj_new < obj:
                a = a_new
                break
            else:
                eta *= 0.7
        else:
            # Couldn't improve, break
            if eta < 1e-8:
                break
        
        # Decay eta slowly
        eta *= 0.999
        # Occasionally add noise
        if it % 1000 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, 0.01 * S / n, n)
            a = np.clip(a + noise, 1e-10, None)
            a = a / a.sum() * S
    
    return best_a, best_obj

- Smooth differentiable objective via log-sum-exp enables L-BFGS quasi-Newton steps

import numpy as np

def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2*n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (max(size, 2) - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def stable_softmax(z):
    """Numerically stable softmax."""
    z_max = z.max()
    e = np.exp(z - z_max)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(len(z)) / len(z)
    return e / s


def compute_smooth_grad(a, c, t, threshold=1e-8):
    """
    Gradient of log-sum-exp(t*c)/t with respect to a, where c = conv(a,a).
    Uses weighted subgradient: sum_k softmax(t*c)_k * dc_k/da.
    """
    n = len(a)
    c_max = c.max()
    shifted = t * (c - c_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total <= 1e-300:
        return np.zeros(n, dtype=np.float64)
    weights = exp_shifted / total

    grad = np.zeros(n, dtype=np.float64)
    active = np.where(weights > threshold)[0]
    for k in active:
        w = weights[k]
        if w <= 0:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def smooth_max_val(c, t):
    """Compute log-sum-exp(t*c)/t (smooth max)."""
    c_max = c.max()
    if t < 1e-4:
        return c_max
    shifted = t * (c - c_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    return c_max + np.log(exp_shifted.sum()) / t

def run_lbfgs_phase(z, n, S, t, max_iter, deadline, label=""):
    """
    Run one phase of L-BFGS-B on the log-parameter space.
    a = softmax(z) * S. Objective is scale-invariant = 2n * max(conv(softmax(z))).
    """
    def func_grad(z):

        c = fft_convolve_self(a_norm)

        obj = 2.0 * n * val
