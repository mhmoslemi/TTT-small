"""
Hybrid black-box / surrogate optimizer for non-negative step-function heights.

The evaluator is treated as the source of truth: all final accept decisions use
evaluate_sequence.  For speed, several cheap surrogates (linear/circular
self-convolution peak scores) are used for ordering and fast pre-screening.
The main search combines:
  - soft-max gradient descent on LP-inspired surrogate active-peak objectives,
  - exact coordinate line-search directly on evaluate_sequence,
  - finite-difference Adam gradients directly on evaluate_sequence,
  - an LP/LP-mixing phase inspired by the previous state-of-the-art,
  - simulated-annealing block moves,
  - cautious exploration of other sequence lengths.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    SCIPY_AVAILABLE = True
except Exception:
    SCIPY_AVAILABLE = False

MAX_VALUE = 1000.0


def normalize_sum(a, target=1.0):
    """Clip to non-negative finite values and rescale to target sum."""
    a = np.asarray(a, dtype=float)
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, MAX_VALUE)
    s = float(np.sum(a))
    if s <= 1e-15:
        return np.full(a.shape, target / float(a.size), dtype=float)
    return a * (target / s)


def safe_to_python_list(a):
    """Convert to a Python list of floats safe for evaluate_sequence."""
    a = normalize_sum(a, 1.0)
    return [float(x) for x in a]


def direct_score(a):
    """Score using the real evaluator.  Invalid => inf."""
    arr = np.asarray(a, dtype=float)
    if arr.size == 0:
        return float("inf")
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, MAX_VALUE)
    if float(np.sum(arr)) < 0.01:
        return float("inf")
    try:
        val = evaluate_sequence(arr.tolist())
    except Exception:
        return float("inf")
    try:
        val = float(val)
    except Exception:
        return float("inf")
    if np.isnan(val) or np.isinf(val):
        return float("inf")
    return val


def measure_direct_cost(a):
    """Rough median cost of one evaluator call."""
    arr = np.asarray(a, dtype=float)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, MAX_VALUE)
    if arr.size == 0:
        return 0.001
    lst = arr.tolist()
    t0 = time.monotonic()
    cnt = 0
    for _ in range(5):
        try:
            evaluate_sequence(lst)
            cnt += 1
        except Exception:
            break
    if cnt == 0:
        return 0.001
    return max((time.monotonic() - t0) / cnt, 1e-7)


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    need = len(a) + len(b) - 1
    if need <= 0:
        return np.array([], dtype=float)
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:need]


def circular_convolve_self(a):
    """Circular self-convolution of length n."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    if n == 0:
        return np.array([], dtype=float)
    fa = np.fft.rfft(a, n)
    return np.fft.irfft(fa * fa, n).real


def local_hard_score_linear(a, mode):
    """Hard score from linear self-convolution: full or off-center."""
    a = normalize_sum(a, 1.0)
    n = len(a)
    if n == 0:
        return float("inf")
    b = fft_convolve(a, a)
    if mode == "full":
        idx = np.arange(2 * n - 1, dtype=int)
    else:
        if n <= 1:
            idx = np.empty(0, dtype=int)
        else:
            idx = np.concatenate((
                np.arange(n - 1, dtype=int),
                np.arange(n, 2 * n - 1, dtype=int)
            ))
    if idx.size == 0:
        return float("inf")
    val = float(np.max(b[idx]))
    s = float(np.sum(a))
    if s <= 1e-15:
        return float("inf")
    return (2.0 * n * val) / (s * s)


def local_hard_score_circ(a):
    """Hard score from circular self-convolution, excluding lag 0."""
    a = normalize_sum(a, 1.0)
    n = len(a)
    if n <= 1:
        return float("inf")
    b = circular_convolve_self(a)
    if n <= 1:
        return float("inf")
    val = float(np.max(b[1:]))
    s = float(np.sum(a))
    if s <= 1e-15:
        return float("inf")
    return (2.0 * n * val) / (s * s)


def local_hard_score(a, mode):
    """Dispatch local hard score."""
    if mode == "circ_off":
        return local_hard_score_circ(a)
    if mode == "full":
        return local_hard_score_linear(a, "full")
    return local_hard_score_linear(a, "off")


def choose_mode(samples, directs):
    """Choose surrogate mode by matching local scores to direct scores."""
    variants = ["full", "off", "circ_off"]
    best_variant = "off"
    best_err = float("inf")
    for v in variants:
        err = 0.0
        cnt = 0
        for s, d in zip(samples, directs):
            if not np.isfinite(d):
                continue
            lv = local_hard_score(s, v)
            if np.isfinite(lv):
                err += abs(float(lv) - float(d))
                cnt += 1
        if cnt > 0:
            avg = err / cnt
            if avg < best_err:
                best_err = avg
                best_variant = v

    finite_ds = [float(d) for d in directs if np.isfinite(d)]
    if best_variant == "full" and finite_ds and min(finite_ds) < 1.95:
        best_variant = "off"
    return best_variant


def softmax_vector(theta):
    """Stable softmax to sum 1."""
    theta = np.asarray(theta, dtype=float)
    if theta.size == 0:
        return theta
    t = theta - np.max(theta)
    e = np.exp(t)
    s = float(e.sum())
    if s <= 1e-15:
        return np.full(theta.shape, 1.0 / theta.size, dtype=float)
    return e / s


def soft_max_vals(vals, tau):
    """Soft-max value and normalized weights."""
    vals = np.asarray(vals, dtype=float)
    if vals.size == 0:
        return 0.0, np.array([], dtype=float)
    vmax = float(np.max(vals))
    z = (vals - vmax) / tau
    ez = np.exp(z)
    s = float(ez.sum())
    if s <= 1e-15:
        w = np.zeros(vals.shape, dtype=float)
        w[int(np.argmax(vals))] = 1.0
        return vmax, w
    w = ez / s
    return vmax + tau * np.log(s), w


def grad_soft_selected(a, mode, tau):
    """
    Soft-max surrogate and gradient wrt simplex theta.
    Works for linear full/off.  If circular is requested, uses off-linear.
    """
    a = np.asarray(a, dtype=float)
    n = len(a)
    if n == 0:
        return float("inf"), np.array([], dtype=float)
    s = float(np.sum(a))
    if s <= 1e-15:
        return float("inf"), np.zeros(n, dtype=float)
    C = 2.0 * n

    if mode == "full":
        idx = np.arange(2 * n - 1, dtype=int)
    else:
        if n <= 1:
            idx = np.empty(0, dtype=int)
        else:
            idx = np.concatenate((
                np.arange(n - 1, dtype=int),
                np.arange(n, 2 * n - 1, dtype=int)
            ))

    if idx.size == 0:
        return float("inf"), np.zeros(n, dtype=float)

    b = fft_convolve(a, a)
    v, w = soft_max_vals(b[idx], tau)

    wfull = np.zeros(2 * n - 1, dtype=float)
    wfull[idx] = w

    corr = fft_convolve(a[::-1], wfull)
    g_a = 2.0 * corr[n - 1:2 * n - 1]

    inner = float(np.dot(a, g_a))
    scale = C / (s * s)
    g_theta = scale * (a * (g_a - inner / s))
    return scale * v, g_theta


def surrogate_gradient_search(a0, mode, time_limit, rng):
    """Soft-max gradient descent with Adam, periodically verified directly."""
    a = normalize_sum(a0, 1.0)
    best = a.copy()
    best_score = direct_score(a)
    if time_limit <= 0 or a.size <= 1 or not np.isfinite(best_score):
        return best, best_score

    n = a.size
    surrogate_mode = "full" if mode == "full" else "off"

    theta = np.log(a + 1e-15)
    m1 = np.zeros(n, dtype=float)
    m2 = np.zeros(n, dtype=float)
    step = 0
    lr = 0.06
    tau_start = 0.25 / (n + 1.0)
    tau_end = 1e-5 / (n + 1.0)

    t0 = time.monotonic()
    deadline = t0 + max(0.0, time_limit)
    stall = 0

    while time.monotonic() < deadline:
        frac = min(1.0, (time.monotonic() - t0) / max(1e-9, time_limit))
        tau = tau_start * ((tau_end / tau_start) ** (0.85 * frac))

        a = softmax_vector(theta)
        v, g = grad_soft_selected(a, surrogate_mode, tau)
        if not np.isfinite(v) or not np.all(np.isfinite(g)):
            break

        step += 1
        m1 = 0.9 * m1 + 0.1 * g
        m2 = 0.999 * m2 + 0.001 * g * g
        m1h = m1 / (1.0 - 0.9 ** step)
        m2h = m2 / (1.0 - 0.999 ** step)
        lr_eff = lr / (1.0 + step / 500.0)
        theta = theta - lr_eff * m1h / (np.sqrt(m2h) + 1e-8)
        theta = np.clip(theta, -25.0, 25.0)

        if step % 40 == 0:
            cand = softmax_vector(theta)
            d = direct_score(cand)
            if np.isfinite(d) and d < best_score - 1e-12:
                best = cand.copy()
                best_score = d
                stall = 0
            else:
                stall += 1
                if stall > 5:
                    theta = theta + rng.normal(0.0, 0.08, size=n)
                    m1 *= 0.3
                    m2 *= 0.3
                    stall = 0

    cand = softmax_vector(theta)
    d = direct_score(cand)
    if np.isfinite(d) and d < best_score:
        best = cand.copy()
        best_score = d

    return best, best_score


def finite_diff_gradient(theta, eps, central, deadline=None):
    """
    Finite-difference gradient of direct score wrt simplex theta.
    Returns None if a deadline forces abort.
    """
    theta = np.asarray(theta, dtype=float)
    n = int(theta.size)
    grad = np.zeros(n, dtype=float)

    if central:
        for i in range(n):
            if deadline is not None and (i % 24 == 0) and (time.monotonic() > deadline - 1.5):
                return None
            old = theta[i]
            theta[i] = old + eps
            ap = softmax_vector(theta)
            fp = direct_score(ap)
            theta[i] = old - eps
            am = softmax_vector(theta)
            fm = direct_score(am)
            theta[i] = old
            if np.isfinite(fp) and np.isfinite(fm) and (fp - fm) != 0.0:
                grad[i] = (fp - fm) / (2.0 * eps)
            else:
                grad[i] = 0.0
        return grad

    base_a = softmax_vector(theta)
    base = direct_score(base_a)
    if not np.isfinite(base):
        return grad

    for i in range(n):
        if deadline is not None and (i % 24 == 0) and (time.monotonic() > deadline - 1.5):
            return None
        old = theta[i]
        theta[i] = old + eps
        ap = softmax_vector(theta)
        fp = direct_score(ap)
        theta[i] = old
        if np.isfinite(fp):
            grad[i] = (fp - base) / eps
        else:
            grad[i] = 0.0

    return grad


def blackbox_gradient_search(a0, time_limit, rng):
    """Finite-difference Adam directly on evaluate_sequence."""
    a = normalize_sum(a0, 1.0)
    best = a.copy()
    best_score = direct_score(a)
    n = a.size
    if n == 0 or n > 4096 or time_limit <= 0 or not np.isfinite(best_score):
        return best, best_score

    cost = measure_direct_cost(a)
    central = (n <= 512) and ((2 * n + 1) * cost < 30.0)
    eps = 1e-3 if central else 2e-3

    theta = np.log(a + 1e-15)
    m1 = np.zeros(n, dtype=float)
    m2 = np.zeros(n, dtype=float)
    step = 0
    lr = 0.04
    stall = 0

    t0 = time.monotonic()
    deadline = t0 + max(0.0, time_limit)
    est_grad = ((2 if central else 1) * n + 1) * cost + n * 1e-4

    while time.monotonic() < deadline:
        if time.monotonic() + est_grad > deadline - 2.0 and step > 3:
            break

        a = softmax_vector(theta)
        score = direct_score(a)
        if np.isfinite(score) and score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            stall = 0
        else:
            stall += 1

        if stall > 20:
            sigma = 0.05 + 0.05 * min(3.0, stall / 20.0)
            theta = theta + rng.normal(0.0, sigma, size=n)
            m1 *= 0.2
            m2 *= 0.2
            stall = 0
            continue

        g = finite_diff_gradient(theta, eps, central, deadline - 1.0)
        if g is None:
            break
        if not np.any(np.abs(g) > 1e-15):
            theta = theta + rng.normal(0.0, 0.08, size=n)
            m1 *= 0.2
            m2 *= 0.2
            continue

        step += 1
        m1 = 0.9 * m1 + 0.1 * g
        m2 = 0.999 * m2 + 0.001 * g * g
        m1h = m1 / (1.0 - 0.9 ** step)
        m2h = m2 / (1.0 - 0.999 ** step)
        lr_eff = lr / (1.0 + step / 1000.0)
        theta = theta - lr_eff * m1h / (np.sqrt(m2h) + 1e-8)
        theta = np.clip(theta, -25.0, 25.0)

    return best, best_score


def peak_contribution_order(a, mode, rng):
    """Order coordinates by their contribution to the current local peak."""
    a = normalize_sum(a, 1.0)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=int)
    try:
        if mode == "circ_off":
            b = circular_convolve_self(a)
            if n <= 1:
                return rng.permutation(n)
            p = 1 + int(np.argmax(b[1:]))
            contrib = np.zeros(n, dtype=float)
            for i in range(n):
                contrib[i] = 2.0 * float(a[(p - i) % n])
        else:
            b = fft_convolve(a, a)
            if mode == "full":
                idx = np.arange(2 * n - 1, dtype=int)
            else:
                if n <= 1:
                    return rng.permutation(n)
                idx = np.concatenate((
                    np.arange(n - 1, dtype=int),
                    np.arange(n, 2 * n - 1, dtype=int)
                ))
            if idx.size == 0:
                return rng.permutation(n)
            p = int(idx[int(np.argmax(b[idx]))])
            contrib = np.zeros(n, dtype=float)
            for i in range(n):
                j = p - i
                if 0 <= j < n:
                    contrib[i] = 2.0 * float(a[j])

        cmax = float(np.max(contrib)) if contrib.size else 0.0
        if cmax <= 0.0:
            return rng.permutation(n)
        scores = -contrib / cmax + 0.05 * rng.standard_normal(n)
        return np.argsort(scores).astype(int, copy=False)
    except Exception:
        return rng.permutation(n)


def coordinate_direct_search(a0, time_limit, rng, mode):
    """Exact coordinate line search directly on evaluate_sequence."""
    a = normalize_sum(a0, 1.0)
    n = a.size
    if n == 0 or time_limit <= 0:
        return a, direct_score(a)

    best = a.copy()
    best_score = direct_score(a)
    current = a.copy()
    current_score = best_score

    if n <= 512:
        ratios = [0.97, 0.94, 0.9, 0.82, 0.7, 0.55, 0.4, 0.25,
                  1.03, 1.08, 1.15, 1.3, 1.6, 2.2]
    elif n <= 1024:
        ratios = [0.95, 0.9, 0.8, 0.65, 0.5, 1.05, 1.15, 1.35, 1.7, 2.4]
    else:
        ratios = [0.95, 0.8, 0.6, 1.15, 1.5, 2.2]

    zero_vals = [1e-8, 1e-6, 1e-4, 1e-2]
    refine_ratios = [0.9, 0.95, 1.05, 1.1, 1.2]

    t0 = time.monotonic()
    deadline = t0 + max(0.0, time_limit)
    no_improve = 0

    while time.monotonic() < deadline - 1.0:
        if no_improve >= 4:
            break

        order = peak_contribution_order(current, mode, rng)
        if n <= 1024:
            coords = order
        elif n <= 2048:
            coords = order[:min(len(order), 512)]
        else:
            coords = order[:min(len(order), 256)]

        improved_cycle = False

        for i in coords:
            if time.monotonic() > deadline - 1.0:
                break

            base = float(current[i])
            if base <= 1e-10:
                qs = list(zero_vals)
            else:
                qs = []
                for r in ratios:
                    qs.append(base * r)
                qs.append(1e-12)

            best_sc = current_score
            best_cand = None

            if n > 512:
                pairs = []
                for q in qs:
                    cand = current.copy()
                    cand[i] = float(max(0.0, q))
                    cand = normalize_sum(cand, 1.0)
                    loc = local_hard_score(cand, mode)
                    if np.isfinite(loc):
                        pairs.append((loc, cand))
                if pairs:
                    pairs.sort()
                    for k, (loc_val, cand) in enumerate(pairs):
                        if k >= 3:
                            break
                        d = direct_score(cand)
                        if np.isfinite(d) and d < best_sc - 1e-12:
                            best_sc = d
                            best_cand = cand
            else:
                for q in qs:
                    cand = current.copy()
                    cand[i] = float(max(0.0, q))
                    cand = normalize_sum(cand, 1.0)
                    d = direct_score(cand)
                    if np.isfinite(d) and d < best_sc - 1e-12:
                        best_sc = d
                        best_cand = cand

            if best_cand is not None:
                current = best_cand
                current_score = best_sc
                improved_cycle = True
                if current_score < best_score - 1e-12:
                    best = current.copy()
                    best_score = current_score

                # Small refinement around the newly accepted coordinate.
                if n <= 512:
                    ref_base = float(current[i])
                    if ref_base > 1e-10:
                        improve_local = best_sc
                        improve_cand = current
                        for rr in refine_ratios:
                            cand = current.copy()
                            cand[i] = ref_base * rr
                            cand = normalize_sum(cand, 1.0)
                            d = direct_score(cand)
                            if np.isfinite(d) and d < improve_local - 1e-13:
                                improve_local = d
                                improve_cand = cand
                        if improve_cand is not current:
                            current = improve_cand
                            current_score = improve_local
                            if current_score < best_score - 1e-12:
                                best = current.copy()
                                best_score = current_score

        if improved_cycle:
            no_improve = 0
        else:
            no_improve += 1

    return best, best_score


def random_block_search(a0, time_limit, rng):
    """Simulated annealing with multiplicative block moves."""
    a = normalize_sum(a0, 1.0)
    n = a.size
    if n == 0 or time_limit <= 0:
        return a, direct_score(a)

    current = a.copy()
    current_score = direct_score(a)
    best = current.copy()
    best_score = current_score

    temp = 0.002
    sigma = 0.1
    t0 = time.monotonic()
    deadline = t0 + max(0.0, time_limit)
    it = 0

    while time.monotonic() < deadline - 1.0:
        max_block = max(1, n // 12 + 1)
        blen = min(n, int(rng.integers(1, max_block + 1)))
        if blen <= 0:
            blen = 1
        if n > blen:
            start = int(rng.integers(0, n - blen + 1))
        else:
            start = 0

        cand = current.copy()
        cand[start:start + blen] *= np.exp(rng.normal(0.0, sigma, size=blen))
        cand = normalize_sum(cand, 1.0)

        d = direct_score(cand)
        if np.isfinite(d):
            accepted = False
            if d < current_score - 1e-12:
                accepted = True
                sigma = max(0.02, sigma * 1.05)
            else:
                try:
                    acc_prob = np.exp(-(d - current_score) / max(temp, 1e-12))
                except Exception:
                    acc_prob = 0.0
                if rng.random() < acc_prob:
                    accepted = True
                    sigma = max(0.02, sigma * 0.98)

            if accepted:
                current = cand
                current_score = d
                if current_score < best_score - 1e-12:
                    best = current.copy()
                    best_score = current_score

        temp *= 0.9995
        it += 1
        if it % 400 == 0:
            print("      SA it=%d best=%.8f cur=%.8f" % (it, best_score, current_score))

    return best, best_score


def lp_improve_sequence(a0, mode, time_limit, rng):
    """LP-inspired improvement, accepting only direct-score improvements."""
    a = normalize_sum(a0, 1.0)
    n = a.size
    if not SCIPY_AVAILABLE or n <= 1 or n > 1024 or time_limit <= 0:
        return a, direct_score(a)

    best = a.copy()
    best_score = direct_score(a)
    if not np.isfinite(best_score):
        return best, best_score

    linear_full = (mode == "full")
    linear_off = (mode == "off")
    circ_off = (mode == "circ_off")

    t0 = time.monotonic()
    deadline = t0 + max(0.0, time_limit)

    for _ in range(8):
        if time.monotonic() > deadline - 45.0:
            break

        cur = best
        if circ_off:
            b = circular_convolve_self(cur)
            if n <= 1:
                break
            M = float(np.max(b[1:]))
            idx = np.arange(1, n, dtype=int)
        else:
            b = fft_convolve(cur, cur)
            if linear_full:
                idx = np.arange(2 * n - 1, dtype=int)
            else:
                idx = np.concatenate((
                    np.arange(n - 1, dtype=int),
                    np.arange(n, 2 * n - 1, dtype=int)
                ))
            if idx.size == 0:
                break
            M = float(np.max(b[idx]))

        if not np.isfinite(M) or M <= 0.0:
            break
        M = M * (1.0 + 1e-3) + 1e-10

        data = []
        inds = []
        indptr = [0]
        ar = np.arange(n, dtype=int)

        try:
            if circ_off:
                for k in idx:
                    coeffs = cur[(k - ar) % n]
                    inds.extend(range(n))
                    data.extend(coeffs.astype(float, copy=False).tolist())
                    indptr.append(len(data))
            else:
                for k in idx:
                    low = max(0, int(k) - n + 1)
                    high = min(n - 1, int(k))
                    if low <= high:
                        fidx = np.arange(k - low, k - high - 1, -1, dtype=int)
                        vals = cur[fidx]
                        inds.extend(range(low, high + 1))
                        data.extend(vals.astype(float, copy=False).tolist())
                    indptr.append(len(data))

            A = csr_matrix((data, inds, indptr), shape=(len(idx), n))
            c = -np.ones(n, dtype=float)
            res = linprog(
                c,
                A_ub=A,
                b_ub=np.full(len(idx), M, dtype=float),
                bounds=(0.0, 100.0),
                options={"presolve": True, "dual": False, "time_limit": 30}
            )
        except Exception:
            res = None

        best_mix = None
        best_mix_score = best_score

        if res is not None and getattr(res, "success", False):
            g = normalize_sum(np.asarray(res.x, dtype=float), 1.0)
            t_values = (0.1, 0.2, 0.35, 0.5, 0.7, 1.0)
            for t in t_values:
                cand = (1.0 - t) * best + t * g
                cand = normalize_sum(cand, 1.0)
                d = direct_score(cand)
                if np.isfinite(d) and d < best_mix_score - 1e-12:
                    best_mix = cand
                    best_mix_score = d

        if best_mix is None:
            break

        best = best_mix
        best_score = best_mix_score

    return best, best_score


def resample_sequence(a, n):
    """Resample a non-negative sequence to length n."""
    a = normalize_sum(a, 1.0)
    if n <= 0:
        return np.array([], dtype=float)
    if len(a) == n:
        return a.copy()
    if len(a) == 0:
        return np.ones(n, dtype=float) / float(n)
    x_old = np.linspace(0.0, 1.0, len(a), dtype=float)
    x_new = np.linspace(0.0, 1.0, n, dtype=float)
    y = np.interp(x_new, x_old, a)
    return normalize_sum(y, 1.0)


def make_chirp_init(n, rng):
    """A smooth chirp-like non-negative initializer."""
    if n <= 0:
        return np.array([], dtype=float)
    t = np.arange(n, dtype=float)
    alpha = 0.35 + 0.35 * rng.random()
    a = 1.0 + alpha * np.cos(np.pi * t * t / float(n))
    a += 0.04 * rng.standard_normal(n)
    return normalize_sum(np.clip(a, 1e-6, None), 1.0)


def make_random_init(n, rng):
    """Random positive initializer."""
    if n <= 0:
        return np.array([], dtype=float)
    a = 1.0 + 0.5 * rng.standard_normal(n)
    return normalize_sum(np.clip(a, 1e-6, None), 1.0)


def explore_other_lengths(best, best_score, mode, time_limit, rng):
    """Try a few other lengths, with modest local search."""
    if time_limit <= 0 or best is None or best.size == 0:
        return best, best_score

    base_n = int(best.size)
    seen = {base_n}
    cand_lengths = []
    for x in (max(256, base_n + 64), max(512, base_n * 2), 512, 1024, 2048, 4096):
        if x not in seen and 2 <= x <= 8192:
            cand_lengths.append(x)
            seen.add(x)

    cand_lengths.sort(key=lambda z: abs(z - max(512, base_n)))
    cand_lengths = cand_lengths[:3]

    t0 = time.monotonic()
    deadline = t0 + max(0.0, time_limit)

    for n in cand_lengths:
        if time.monotonic() > deadline - 80.0:
            break

        init_candidates = [
            resample_sequence(best, n),
            normalize_sum(np.ones(n, dtype=float), 1.0),
            make_chirp_init(n, rng),
            make_random_init(n, rng),
        ]

        best_init = None
        best_init_score = float("inf")
        for init in init_candidates:
            aa = normalize_sum(init, 1.0)
            d = direct_score(aa)
            if np.isfinite(d):
                if d < best_init_score:
                    best_init = aa
                    best_init_score = d
                if d < best_score - 1e-12:
                    best = aa.copy()
                    best_score = d

        if best_init is None:
            continue

        # Only spend search time if this length is plausibly competitive.
        if best_init_score > best_score + 0.75:
            continue

        a = best_init.copy()

        rem = deadline - time.monotonic()
        if rem > 40.0 and a.size <= 8192:
            t_surr = min(45.0, rem * 0.35)
            a, sc = surrogate_gradient_search(a, mode, t_surr, rng)
            if sc < best_score - 1e-12:
                best = a.copy()
                best_score = sc

        rem = deadline - time.monotonic()
        if rem > 45.0 and a.size <= 2048:
            t_coord = min(75.0, rem * 0.35)
            a, sc = coordinate_direct_search(a, t_coord, rng, mode)
            if sc < best_score - 1e-12:
                best = a.copy()
                best_score = sc

        rem = deadline - time.monotonic()
        if rem > 35.0 and a.size <= 768:
            t_bb = min(45.0, rem * 0.35)
            a, sc = blackbox_gradient_search(a, t_bb, rng)
            if sc < best_score - 1e-12:
                best = a.copy()
                best_score = sc

        print("      length %d best=%.8f" % (n, best_score))

    return best, best_score


def propose_candidate(seed=42, budget_s=2400.0):
    """Main search driver. Returns the best valid sequence found."""
    t_start = time.monotonic()
    try:
        total_budget = max(60.0, min(float(budget_s), 2360.0))
    except Exception:
        total_budget = 2360.0
    deadline = t_start + total_budget
    rng = np.random.default_rng(seed)

    start_arr = None
    if "height_sequence_1" in globals():
        try:
            raw = globals()["height_sequence_1"]
            if raw is not None:
                start_arr = np.asarray(raw, dtype=float)
                start_arr = np.where(np.isfinite(start_arr), start_arr, 0.0)
                start_arr = np.clip(start_arr, 0.0, MAX_VALUE)
                if start_arr.size > 8192:
                    start_arr = resample_sequence(start_arr, 4096)
        except Exception:
            start_arr = None

    base_n = 256
    if start_arr is not None and start_arr.size > 0:
        base_n = int(start_arr.size)

    start_candidates = []
    if start_arr is not None and start_arr.size > 0:
        start_candidates.append(start_arr)
    start_candidates.append(np.ones(base_n, dtype=float))
    start_candidates.append(make_chirp_init(base_n, rng))
    start_candidates.append(make_random_init(base_n, rng))

    best = None
    best_score = float("inf")
    mode_samples = []
    mode_directs = []

    for cand in start_candidates:
        a = normalize_sum(cand, 1.0)
        mode_samples.append(a)
        d = direct_score(a)
        mode_directs.append(d)
        if d < best_score:
            best_score = d
            best = a.copy()

    if best is None:
        best = normalize_sum(np.ones(256, dtype=float), 1.0)
        best_score = direct_score(best)

    print("=== Start: n=%d score=%.8f ===" % (len(best), best_score))

    if not np.isfinite(best_score):
        print("No finite evaluator score available; returning fallback.")
        return safe_to_python_list(best)

    mode = choose_mode(mode_samples, mode_directs)
    print("Chosen surrogate mode:", mode)

    # Phase 1: fast surrogate gradient.
    rem = deadline - time.monotonic()
    if rem > 30.0:
        t_phase = min(90.0, 0.10 * total_budget, rem - 2.0)
        if t_phase > 5.0:
            a, sc = surrogate_gradient_search(best, mode, t_phase, rng)
            if sc < best_score - 1e-12:
                best = a.copy()
                best_score = sc
            print("After surrogate: %.8f" % best_score)

    # Phase 2: exact coordinate line search.
    rem = deadline - time.monotonic()
    if rem > 60.0:
        t_phase = min(300.0, 0.20 * total_budget, rem - 2.0)
        if t_phase > 5.0:
            a, sc = coordinate_direct_search(best, t_phase, rng, mode)
            if sc < best_score - 1e-12:
                best = a.copy()
                best_score = sc
            print("After coordinate #1: %.8f" % best_score)

    # Phase 3: finite-difference Adam directly on evaluator.
    rem = deadline - time.monotonic()
    if rem > 120.0 and len(best) <= 1200:
        t_phase = min(1100.0, 0.45 * total_budget, rem - 2.0)
        if t_phase > 20.0:
            a, sc = blackbox_gradient_search(best, t_phase, rng)
            if sc < best_score - 1e-12:
                best = a.copy()
                best_score = sc
            print("After blackbox gradient: %.8f" % best_score)

    # Phase 4: another coordinate polish.
    rem = deadline - time.monotonic()
    if rem > 60.0:
        t_phase = min(180.0, 0.10 * total_budget, rem - 2.0)
        if t_phase > 5.0:
            a, sc = coordinate_direct_search(best, t_phase, rng, mode)
            if sc < best_score - 1e-12:
                best = a.copy()
                best_score = sc
            print("After coordinate #2: %.8f" % best_score)

    # Phase 5: LP-inspired mixing.
    rem = deadline - time.monotonic()
    if rem > 90.0 and len(best) <= 1024:
        t_phase = min(180.0, 0.10 * total_budget, rem - 2.0)
        if t_phase > 10.0:
            a, sc = lp_improve_sequence(best, mode, t_phase, rng)
            if sc < best_score - 1e-12:
                best = a.copy()
                best_score = sc
            print("After LP mixing: %.8f" % best_score)

    # Phase 6: simulated annealing block moves.
    rem = deadline - time.monotonic()
    if rem > 45.0:
        t_phase = min(90.0, 0.05 * total_budget, rem - 2.0)
        if t_phase > 5.0:
            a, sc = random_block_search(best, t_phase, rng)
            if sc < best_score - 1e-12:
                best = a.copy()
                best_score = sc
            print("After SA: %.8f" % best_score)

    # Phase 7: exploration of other lengths.
    rem = deadline - time.monotonic()
    if rem > 120.0:
        t_phase = max(0.0, rem - 35.0)
        a, sc = explore_other_lengths(best, best_score, mode, t_phase, rng)
        if sc < best_score - 1e-12:
            best = a.copy()
            best_score = sc
        print("After length exploration: %.8f" % best_score)

    # Final safety pass: tiny coordinate polish if time permits.
    rem = deadline - time.monotonic()
    if rem > 25.0:
        t_phase = max(5.0, min(18.0, rem - 5.0))
        a, sc = coordinate_direct_search(best, t_phase, rng, mode)
        if sc < best_score - 1e-12:
            best = a.copy()
            best_score = sc

    final_list = safe_to_python_list(best)
    final_val = direct_score(final_list)
    if not np.isfinite(final_val):
        final_list = safe_to_python_list(np.ones(max(1, len(best)), dtype=float))
        final_val = direct_score(final_list)

    print("FINAL n=%d score=%s" % (len(final_list), str(final_val)))
    return final_list
