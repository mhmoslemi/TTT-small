def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-300:

    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=(0.0, None), method='highs', options={'disp': False, 'time_limit': budget_s})
    except Exception:
        return None
    if res is None or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-12:
        return None
    return g / s


def shape_fn(theta, n):
    x = np.linspace(0.0, 1.0, n)
    z = 2.0 * x - 1.0
    D = min(int(len(theta)) - 2, 12)
    if D < 0:
        D = 0
    logp = np.zeros(n, dtype=np.float64)
    if D >= 1:
        T_nm = np.ones(n, dtype=np.float64)
        T_n = z.copy()
        logp += theta[0] * T_n
        for d in range(2, D + 1):
            T_np = 2.0 * z * T_n - T_nm
            logp += theta[d - 1] * T_np
            T_nm = T_n
            T_n = T_np
    eps = 1e-3
    logp += theta[D] * np.log(x + eps)
    logp += theta[D + 1] * np.log(1.0 - x + eps)
    if not np.all(np.isfinite(logp)):
        return np.full(n, 1.0 / n, dtype=np.float64)
    logp = logp - float(np.max(logp))
    p = np.exp(logp)
    return safe_normalize(p)


def shape_score(theta, n):
    try:
        p = shape_fn(theta, n)
        return internal_score(p)
    except Exception:
        return 1e100


def search_shape(n, rng, budget_s):
    if budget_s < 10:
        return None, np.inf
    D = 8
    dim = D + 2
    end = time.monotonic() + budget_s
    bounds = [(-4.0, 4.0)] * D + [(-5.0, 8.0), (-5.0, 8.0)]
    best_p = None
    best_sc = np.inf

    starts = [np.zeros(dim, dtype=np.float64)]
    for _ in range(4):
        starts.append(rng.uniform(-2.5, 2.5, dim).astype(np.float64))
    t = np.zeros(dim, dtype=np.float64)
    t[D] = 3.0
    t[D + 1] = 3.0
    starts.append(t)
    t = np.zeros(dim, dtype=np.float64)
    t[0] = 2.0
    t[1] = -1.0
    t[2] = 0.5
    t[D] = 2.0
    t[D + 1] = 0.5
    starts.append(t)
    t = np.zeros(dim, dtype=np.float64)
    t[0] = -2.0
    t[D] = 0.5
    t[D + 1] = 3.0
    starts.append(t)

    for x0 in starts:
        if time.monotonic() > end - 2:
            break
        sc = shape_score(x0, n)
        if sc < best_sc:
            best_sc = sc
            best_p = shape_fn(x0, n)

    if time.monotonic() < end - 60:
        try:
            res = differential_evolution(lambda th: shape_score(th, n), bounds,
                                         maxiter=30, popsize=10, tol=0.1,
                                         seed=int(rng.integers(0, 2**31 - 1)),
                                         polish=False, init='latinhypercube')
            if res.x is not None and np.isfinite(res.fun):
                sc = shape_score(res.x, n)
                if sc < best_sc:
                    best_sc = sc
                    best_p = shape_fn(res.x, n)
                starts.append(res.x.copy())
        except Exception:
            pass

    for x0 in starts[:4]:
        if time.monotonic() > end - 3:
            break
        try:
            res = minimize(lambda th: shape_score(th, n), x0, method='Nelder-Mead',
                           options={'maxiter': 300, 'xatol': 1e-5, 'fatol': 1e-7})
            if res.x is not None and np.isfinite(res.fun):
                sc = shape_score(res.x, n)
                if sc < best_sc:
                    best_sc = sc
                    best_p = shape_fn(res.x, n)
        except Exception:
            pass

    return best_p, best_sc


def make_initials(n, rng, parent=None):
    out = []
    if parent is not None:
        pp = np.asarray(parent, dtype=np.float64)
        if pp.size == n:
            out.append(safe_normalize(pp))
    out.append(np.full(n, 1.0 / n, dtype=np.float64))
    x = np.linspace(0, 1, n)
    for a in [0.5, 1.0, 2.0, 4.0, 8.0]:
        out.append(safe_normalize(1.0 + a * x))
        out.append(safe_normalize(1.0 + a * (1 - x)))
    for a in [0.5, 1.0, 2.0, 5.0]:
        out.append(safe_normalize(np.exp(-a * x)))
        out.append(safe_normalize(np.exp(-a * (1 - x))))
    for a in [0.3, 1.0, 3.0]:
        out.append(safe_normalize(1.0 + a * (2 * x - 1.0) ** 2))
    for a in [0.3, 0.7, 0.9]:
        out.append(safe_normalize(np.maximum(1.0 - a * (2 * x - 1.0) ** 2, 0.01)))
    for _ in range(8):
        f1 = int(rng.integers(1, max(2, n // 50)))
        f2 = int(rng.integers(1, max(2, n // 25)))
        p = 1.0 + 0.5 * np.sin(2 * np.pi * f1 * x + rng.uniform(0, 6.28))
        p += 0.3 * np.sin(2 * np.pi * f2 * x + rng.uniform(0, 6.28))
        out.append(safe_normalize(np.maximum(p, 0.01)))
    # Strongly asymmetric
    for a in [2.0, 5.0, 10.0]:
        out.append(safe_normalize(np.exp(a * (x - 0.5))))
        out.append(safe_normalize(np.exp(-a * (x - 0.5))))
    # Two peaks
    out.append(safe_normalize(np.exp(-100 * (x - 0.2) ** 2) + np.exp(-100 * (x - 0.8) ** 2)))
    out.append(safe_normalize(np.exp(-50 * (x - 0.3) ** 2) + 0.5 * np.exp(-50 * (x - 0.7) ** 2)))
    return out


def optimize_hybrid(p0, rng, time_limit, label=""):
    p = project_simplex(safe_normalize(p0))
    n = int(p.size)
    if n < 2:
        return p, np.inf

    best_p = p.copy()
    best_score = internal_score(p)
    curr_score = best_score
    prev_p = p.copy()

    end = time.monotonic() + max(1.0, float(time_limit))
    T_factor = 0.5
    iter_no = 0
    since_improve = 0
    lp_count = 0

    # Initial LP
    if n <= LP_NMAX:
        t_start = time.monotonic()
        q = lp_best_response(p, 1.0, min(30.0, time_limit * 0.1))
        if q is not None:
            for t in [1.0, 0.7, 0.5, 0.3]:
                cand = project_simplex((1 - t) * p + t * q)
                sc = internal_score(cand)
                if sc < best_score:
                    best_score = sc
                    curr_score = sc
                    best_p = cand.copy()
                    p = cand.copy()
                    prev_p = cand.copy()
                    since_improve = 0
                    break
            lp_count += 1
            print(label + " init LP: best=%.8f" % best_score, flush=True)

    while time.monotonic() < end:
        iter_no += 1
        # Adaptive temperature
        if since_improve > 500:
            T_factor = min(2.0, T_factor * 1.1)
        elif since_improve > 100:
            T_factor = max(0.01, T_factor * 0.95)
        else:
            T_factor = max(0.005, T_factor * 0.995)

        g, c, cmax, surr = gradient_smoothed_max(p, T_factor)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)

        gabs = float(np.max(np.abs(g))) if g.size else 0.0
        p_scale = max(1.0 / n, 0.05 * float(np.max(p)), 1e-12)
        if gabs > 1e-300:
            base = p_scale / gabs
            lrs = [0.3 * base, 0.08 * base, 0.01 * base, 0.002 * base, 0.0004 * base]
        else:
            lrs = [0.01, 0.003, 0.001]

        accepted = False
        for lr in lrs:
            v = p - lr * g + 0.3 * (p - prev_p)
            pt = project_simplex(v)
            sc = internal_score(pt)
            _, _, _, surr_t = gradient_smoothed_max(pt, T_factor)
            if np.isfinite(sc) and (surr_t < surr - 1e-12 or sc < curr_score - 1e-12):
                if sc < best_score:
                    best_score = sc
                    best_p = pt.copy()
                    since_improve = 0
                else:
                    since_improve += 1
                curr_score = sc
                prev_p = p
                p = pt
                accepted = True
                break

        if not accepted:
            since_improve += 1
            T_factor = max(0.005, T_factor * 0.7)
            # Random perturbation
            if since_improve > 200:
                sig = max(0.01 / n, 0.3 / n * (since_improve / 500.0))
                pt = project_simplex(p + rng.normal(0, sig, n))
                sc = internal_score(pt)
                if np.isfinite(sc) and sc < curr_score:
                    curr_score = sc
                    prev_p = p
                    p = pt
                    since_improve = 0
            if since_improve > 600:
                p = project_simplex(0.5 * best_p + 0.5 * p + rng.normal(0, 0.5 / n, n))
                curr_score = internal_score(p)
                prev_p = p
                T_factor = 0.5
                since_improve = 0

        # Periodic LP
        lp_interval = 30 if n <= 800 else 60
        if n <= LP_NMAX and iter_no % lp_interval == 0:
            remaining = end - time.monotonic()
            if remaining > max(5.0, 20.0 / (n / 500.0 + 1)):
                t_start = time.monotonic()
                q = lp_best_response(p, 1.0, min(remaining * 0.3, 45.0))
                if q is not None:
                    for t in [1.0, 0.8, 0.6, 0.4, 0.25]:
                        cand = project_simplex((1 - t) * p + t * q)
                        sc = internal_score(cand)
                        if sc < best_score:
                            best_score = sc
                            curr_score = sc
                            best_p = cand.copy()
                            p = cand.copy()
                            prev_p = cand.copy()
                            since_improve = 0
                            break
                    lp_count += 1
                if iter_no % (lp_interval * 10) == 0:
                    print(label + " iter=%d lp=%d best=%.8f cur=%.8f T=%.4f" %
                          (iter_no, lp_count, best_score, curr_score, T_factor), flush=True)

        # Very fine pair moves near convergence
        if since_improve > 400 and n > 2:
            c_curr = conv_full(p)
            cmax_idx = int(np.argmax(c_curr))
            # Find which products contribute to the peak
            if 0 <= cmax_idx < n:
                hi = cmax_idx
                lo = int(np.argmin(p))
            elif cmax_idx < 2 * n - 1:
                hi = n - 1 - (cmax_idx - n + 1)
                lo = int(np.argmin(p))
            else:
                hi = 0
                lo = int(np.argmin(p))
            if hi != lo and p[hi] > p[lo]:
                for frac in [0.15, 0.05, 0.01, 0.003]:
                    delta = frac * p[hi]
                    if delta <= 0:
                        continue
                    pt = p.copy()
                    pt[hi] -= delta
                    pt[lo] += delta
                    pt = project_simplex(pt)
                    sc = internal_score(pt)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        curr_score = sc
                        p = pt
                        prev_p = pt
                        since_improve = 0
                        break

        if best_score < 1.498 and since_improve > 8000:
            break

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    # Load parent
    parent = None
    try:
        pa = np.asarray(height_sequence_1, dtype=np.float64)
        pa = np.where(np.isfinite(pa), pa, 0.0)
        pa = np.maximum(pa, 0.0)
        if pa.ndim == 1 and pa.size >= 50 and pa.sum() > 0:
            parent = safe_normalize(pa)
    except Exception:
        pass

    # Determine n values to try
    n_candidates = [700, 800, 900, 600]
    if parent is not None:
        pn = int(parent.size)
        n_candidates = [pn] + [n for n in [700, 800, 600, 900] if n != pn and n <= LP_NMAX]
    n_candidates = [n for n in n_candidates if LP_NMIN <= n <= LP_NMAX]
    if not n_candidates:
        n_candidates = [700]

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        p = safe_normalize(p)
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("NEW BEST: score=%.10f n=%d elapsed=%.0fs" % (sc, p.size, time.monotonic() - t0), flush=True)

    # Phase 1: Shape search for the primary n
    primary_n = n_candidates[0]
    if deadline - time.monotonic() > 200:
        shape_budget = min(180.0, (deadline - time.monotonic()) * 0.2)
        shape_p, shape_sc = search_shape(primary_n, rng, shape_budget)
        try_record(shape_p, shape_sc)
        if shape_p is not None:
            print("shape search: score=%.6f n=%d" % (shape_sc, primary_n), flush=True)

    # Phase 2: Generate and score initials
    inits = make_initials(primary_n, rng, parent)
    if shape_p is not None and shape_p.size == primary_n:
        inits.insert(0, shape_p)
    if parent is not None and parent.size == primary_n:
        inits.insert(0, parent)

    scored = []
    for p in inits:
        p = safe_normalize(p)
        sc = internal_score(p)
        if np.isfinite(sc):
            scored.append((sc, p))
    scored.sort(key=lambda x: x[0])
    for sc, p in scored[:8]:
        try_record(p, sc)

    # Phase 3: Hybrid optimization on top candidates
    top_k = min(6, len(scored))
    if deadline - time.monotonic() > 100 and top_k > 0:
        final_reserve = 300.0
        available = max(0.0, (deadline - final_reserve) - time.monotonic())
        time_per = max(30.0, available / float(top_k))

        for i in range(top_k):
            if time.monotonic() > deadline - final_reserve:
                break
            sc, p = scored[i]
            limit = min(time_per, deadline - final_reserve - time.monotonic())
            if limit < 15:
                break
            print("optimizing candidate %d: initial_score=%.6f n=%d time=%.0fs" %
                  (i, sc, p.size, limit), flush=True)
            p_best, sc_best = optimize_hybrid(p, rng, limit, "C%d:" % i)
            try_record(p_best, sc_best)
            scored[i] = (sc_best, p_best)

        # Re-score and try next-best
        scored.sort(key=lambda x: x[0])

    # Phase 4: Try other n values if time permits
    for alt_n in n_candidates[1:]:
        if time.monotonic() > deadline - 200:
            break
        if alt_n == primary_n:
            continue
        inits_alt = make_initials(alt_n, rng, None)
        if shape_p is not None:
            inits_alt.insert(0, shape_fn(shape_p, alt_n))  # reshape
        scored_alt = []
        for p in inits_alt[:10]:
            p = safe_normalize(p)
            sc = internal_score(p)
            if np.isfinite(sc):
                scored_alt.append((sc, p))
        scored_alt.sort(key=lambda x: x[0])
        if scored_alt:
            sc_alt, p_alt = scored_alt[0]
            try_record(p_alt, sc_alt)
            if sc_alt < best_score + 0.1 and deadline - time.monotonic() > 150:
                limit = min(120.0, deadline - time.monotonic() - 100)
                p_best, sc_best = optimize_hybrid(p_alt, rng, limit, "ALT_n%d:" % alt_n)
                try_record(p_best, sc_best)

    # Phase 5: Final intensive polish
    if best_p is not None and time.monotonic() < deadline - 30:
        limit = max(10.0, deadline - time.monotonic() - 15.0)
        print("final polish: score=%.10f time=%.0fs" % (best_score, limit), flush=True)
        p_best, sc_best = optimize_hybrid(best_p, rng, limit, "POLISH:")
        try_record(p_best, sc_best)

    # Final: one more round of very small LP steps
    if best_p is not None and time.monotonic() < deadline - 10 and int(best_p.size) <= LP_NMAX:
        p_curr = best_p.copy()
        for _ in range(20):
            if time.monotonic() > deadline - 10:
                break
            q = lp_best_response(p_curr, 1.0, max(1.0, deadline - time.monotonic() - 12))
            if q is None:
                break
            improved = False
            for t in [0.1, 0.05, 0.02, 0.01]:
                cand = project_simplex((1 - t) * p_curr + t * q)
                sc = internal_score(cand)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_p = cand
                    p_curr = cand
                    improved = True
                    break
            if not improved:
                break
        try_record(best_p, best_score)

    if best_p is None:
        best_p = np.full(700, 1.0 / 700, dtype=np.float64)
        best_score = internal_score(best_p)

    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    print("FINAL: score=%.10f n=%d time=%.0fs" % (best_score, best_p.size, time.monotonic() - t0), flush=True)
    return [float(x) for x in best_p]
