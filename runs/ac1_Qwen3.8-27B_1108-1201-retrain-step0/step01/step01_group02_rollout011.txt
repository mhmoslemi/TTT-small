                # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k

Hybrid optimization for minimizing peak auto-convolution of non-negative sequences.

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient(a):
    """Compute subgradient of max_k conv(a,a)[k] w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)
    k_star = np.argmax(conv)
    grad = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(n - 1, k_star)
    grad[j_start:j_end+1] = a[k_star - np.arange(j_start, j_end+1)]
    return grad, k_star, conv

def compute_soft_subgradient(a, T):
    """Compute soft subgradient using softmax weighting."""
    n = len(a)
    conv = np.convolve(a, a)
    weights = np.exp((conv - conv.max()) / T)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-10:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            j_vals = np.arange(j_start, j_end+1)
            grad[j_start:j_end+1] += weights[k] * a[k - j_vals]
    return grad, conv

def build_conv_matrix_rows(a):
    """Build sparse matrix for conv(a, x) = Ax where x is variable."""
    n = len(a)
    n_rows = 2 * n - 1
    # conv(a, x)[k] = sum_j a[j] * x[k-j]
    # So row k of the matrix has a[j] in column (k-j) for valid j
    row_idx = []
    col_idx = []
    data = []
    for k in range(n_rows):
        for j in range(n):
            col = k - j
            if 0 <= col < n:
                row_idx.append(k)
                col_idx.append(col)
                data.append(a[j])
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))

    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix_rows(a)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix_rows(a)

    # Total variables: n + 1
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: conv(a,g)[k] - t <= 0 for all k
    # A_ub shape: (n_rows, n+1)
    A_ub = A.copy()
    A_ub = A_ub.tolil()
    # Add column for -t
    t_col = -np.ones(n_rows)
    A_ub = A_ub.tocsr()
    # We need to add the t column
    from scipy.sparse import hstack, csc_matrix
    t_matrix = csc_matrix((n_rows, 1), data=t_col)
    A_full = hstack([A, t_matrix], format='csr')
    
    b_ub = np.zeros(n_rows)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(None, None)]  # g >= 0, t free
    
    try:
        res = linprog(c, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            g = np.maximum(0.0, res.x[:n])
            s = g.sum()
            if s > 1e-10:
                g = g / s
            return g
    except Exception:
        pass
    return None

"""
Hybrid optimization for minimizing peak auto-convolution of non-negative sequences.
Uses a new "min-peak" LP that directly minimizes the cross-correlation peak,
combined with mirror descent, standard LP, and local search polishing.
"""

import numpy as np
from scipy.sparse import csr_matrix, csc_matrix, hstack
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_conv_matrix(a):
    """Sparse matrix A such that A @ x = conv(a, x)."""
    n = len(a)
    n_rows = 2 * n - 1
    row_list = []
    col_list = []
    data_list = []
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            row_list.append(k)
            col_list.append(k - j)
            data_list.append(a[j])
    return csr_matrix((data_list, (row_list, col_list)), shape=(n_rows, n))


def solve_lp_min_peak(a, timeout=30.0):
    """LP: minimize t s.t. conv(a,g)[k] <= t for all k, sum(g)=1, g>=0."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Inequalities: conv(a,g)[k] - t <= 0
    t_col = csc_matrix((n_rows, 1), data=-np.ones(n_rows))
    A_ub = hstack([A, t_col], format='csr')
    b_ub = np.zeros(n_rows)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            g = np.maximum(0.0, res.x[:n])
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_max_sum(a, M, timeout=30.0):
    """LP: maximize sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def mirror_descent_step(a, eta, soft_k=1):
    """One mirror descent step using (soft) subgradient of max conv(a,a)."""
    n = len(a)
    conv = np.convolve(a, a)
    if soft_k <= 1:
        k_star = np.argmax(conv)
        grad = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        j_vals = np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi + 1] = a[k_star - j_vals]
    else:
        # Use top-k peaks with softmax weighting
        sorted_idx = np.argsort(conv)[::-1]
        top_k = sorted_idx[:soft_k]
        w = np.exp(conv[top_k] - conv[top_k[0]])
        w /= w.sum()
        grad = np.zeros(n)
        for idx, kk in enumerate(top_k):
            j_lo = max(0, kk - n + 1)
            j_hi = min(n - 1, kk)
            j_vals = np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] += w[idx] * a[kk - j_vals]
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def run_mirror_descent(a, time_limit, eta0=0.05, momentum=0.85, soft_k=1, verbose=True):
    """Mirror descent with momentum and adaptive step size."""
    start = time.monotonic()
    a = normalize_nonneg(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    velocity = np.zeros(n)
    eta = eta0
    no_improve = 0

    for t in range(50000000):
        elapsed = time.monotonic() - start
        if elapsed > time_limit:
            break
        grad_component = mirror_descent_step  # just to track
        # Compute gradient inline for speed
        conv = np.convolve(a, a)
        if soft_k <= 1:
            k_star = np.argmax(conv)
            grad = np.zeros(n)
            j_lo = max(0, k_star - n + 1)
            j_hi = min(n - 1, k_star)
            j_vals = np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] = a[k_star - j_vals]
        else:
            sorted_idx = np.argsort(conv)[::-1]
            top_k = sorted_idx[:soft_k]
            w = np.exp(conv[top_k] - conv[top_k[0]])
            w /= w.sum()
            grad = np.zeros(n)
            for idx, kk in enumerate(top_k):
                j_lo = max(0, kk - n + 1)
                j_hi = min(n - 1, kk)
                j_vals = np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi + 1] += w[idx] * a[kk - j_vals]

        velocity = momentum * velocity + (1.0 - momentum) * grad
        a_new = a * np.exp(-eta * velocity)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            velocity = np.zeros(n)
            continue
        a_new = a_new / s
        score = score_sequence(a_new)

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
            eta = min(eta * 1.005, 0.5)
        else:
            no_improve += 1
            if no_improve > 200:
                eta *= 0.9
            if no_improve > 2000:
                velocity = np.zeros(n)
                no_improve = 0
                eta = eta0

        a = a_new
        if verbose and t % 10000 == 0 and t > 0:
            print(f"    MD t={t}, best={best_score:.6f}, eta={eta:.5f}")

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    MD done: best={best_score:.6f}")
    return best_a, best_score


def lp_refine_min_peak(a, time_limit, verbose=True):
    """LP refinement using the min-peak formulation."""
    start = time.monotonic()
    a = normalize_nonneg(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0

    for it in range(5000):
        elapsed = time.monotonic() - start
        remaining = time_limit - elapsed
        if remaining < 8:
            break

        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 50:
            break

        timeout = min(25.0, max(3.0, remaining * 0.3))
        g = solve_lp_min_peak(a, timeout=timeout)
        if g is None:
            break

        # Try multiple alphas via line search
        best_alpha = 0.0
        best_new_score = score
        for alpha in [0.3, 0.2, 0.1, 0.05, 0.5]:
            a_trial = (1.0 - alpha) * a + alpha * g
            a_trial = normalize_nonneg(a_trial)
            trial_score = score_sequence(a_trial)
            if trial_score < best_new_score:
                best_new_score = trial_score
                best_alpha = alpha
                best_trial = a_trial.copy()

        if best_alpha > 0:
            a = best_trial
        else:
            # Fallback: small step
            a = (1.0 - 0.02) * a + 0.02 * g
            a = normalize_nonneg(a)

        if verbose and it % 5 == 0:
            print(f"    LP-MP it={it}, best={best_score:.6f}, alpha={best_alpha}")

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    LP-MP done: best={best_score:.6f}")
    return best_a, best_score

def local_search(a, time_limit, rng, verbose=True):
    """Local search with random multiplicative perturbations."""
    start = time.monotonic()
    a = normalize_nonneg(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    accept_count = 0

    while time.monotonic() - start < time_limit:
        width = max(2, int(rng.integers(n // 100, n // 4)))
        pos = int(rng.integers(0, max(1, n - width)))
        candidate = a.copy()
        candidate[pos:pos + width] *= np.exp(rng.normal(0.0, 0.12, size=width))
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 100 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, accepts={accept_count}")
    return best_a, best_score


def make_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return normalize_nonneg(0.5 * (1.0 + np.cos(2.0 * np.pi * t / n)))


def make_exponential(n, decay=0.3):
    t = np.arange(n, dtype=np.float64)
    return normalize_nonneg(np.exp(-t / (n * decay)))


def make_bimodal(n):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - n * 0.3) / (n * 0.08)) ** 2) + \
        np.exp(-0.5 * ((t - n * 0.7) / (n * 0.08)) ** 2)
    return normalize_nonneg(a)


def make_skewed_gaussian(n, center_frac=0.6, sigma_frac=0.2):
    t = np.arange(n, dtype=np.float64)
    center = n * center_frac
    sigma = n * sigma_frac
    return normalize_nonneg(np.exp(-0.5 * ((t - center) / sigma) ** 2))
