combined with soft-max gradient descent polishing.

def compute_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fftconvolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def build_conv_matrix(a):
    """Build Toeplitz matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    c = np.zeros(2 * n - 1)
    c[:n] = a
    r = np.zeros(n)
    r[0] = a[0]
    return toeplitz(c, r)


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. g >= 0, conv(a, g) <= M."""
    n = len(a)
    A_ub = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

        else:

    res = np.maximum(a - t, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative optimization
combined with soft-max gradient descent polishing.
LP approach: at each step, solve max sum(g) s.t. g>=0, conv(a,g)<=M,
then line-search interpolate towards g. Gradient descent polishes the result.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.signal import fftconvolve, toeplitz
import time


def compute_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fftconvolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def build_conv_matrix(a):
    """Build Toeplitz matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    c = np.zeros(2 * n - 1)
    c[:n] = a
    r = np.zeros(n)
    r[0] = a[0]
    return toeplitz(c, r)


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. g >= 0, conv(a, g) <= M."""
    n = len(a)
    A_ub = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def soft_max_gradient(a, tau):
    """Compute gradient of tau*log(sum(exp(c/tau))) where c=conv(a,a)."""
    n = len(a)
    c = fftconvolve(a, a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    # grad_i = 2 * sum_m a[m] * q[m+i]
    N = 1
    while N < 3 * n:
        N *= 2
    A = np.fft.rfft(a, N)
    Q = np.fft.rfft(q, N)
    r = np.fft.irfft(np.conj(A) * Q, N)[:n]
    grad = 2.0 * r
    return grad, c


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a.copy(), 0.0)
    n = a.size
    s = a.sum()
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, float(a.max())
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    idx = int(res.argmax())
    res[idx] += S - res.sum()
    return res


def run_lp_opt(a0, t_budget, verbose=False):
    """LP-based iterative optimization with line search."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.array(a0, dtype=np.float64), 0.0).copy()
    if a.sum() < 1e-12:
        a = np.ones(n)
    a = a / a.sum() * S

    best = a.copy()
    best_score = compute_score(best)
    best_mc = fftconvolve(best, best).max()

    t0 = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() - t0 < t_budget:
        remaining = t_budget - (time.monotonic() - t0)
        if remaining < 3:
            break

        c = fftconvolve(a, a)
        M = c.max()

        t_start = time.monotonic()
        g = solve_lp(a, M)
        t_lp = time.monotonic() - t_start

        if g is None or g.sum() < 1e-12:
            break

        g = g / g.sum() * S

        # Line search: try aggressive steps first
        best_M = M
        best_t = 0.0
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.5, 0.3]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            M_try = fftconvolve(a_try, a_try).max()
            if M_try < best_M:
                best_M = M_try
                best_t = t

        if best_t > 0:
            a = np.maximum((1.0 - best_t) * a + best_t * g, 0.0)
            stuck = 0
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best = a.copy()
                best_mc = fftconvolve(best, best).max()
        else:
            stuck += 1
            if stuck > 40:
                break

        it += 1
        if verbose and (it % 20 == 0 or it == 1):
            sc = compute_score(a)
            print(f"  LP iter {it}: M={M:.8f}, score={sc:.8f}, best={best_score:.8f}, t={best_t:.2f}, lp={t_lp:.2f}s")

    return best, best_score


def run_grad_opt(a0, tau_hi, tau_lo, lr0, t_budget, verbose=False):
    """Soft-max projected gradient descent."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = (np.maximum(np.array(a0, dtype=np.float64), 0.0) / max(np.sum(a0), 1e-12) * S).copy()
    a = project_simplex(a, S)

    best = a.copy()
    best_score = compute_score(best)
    best_mc = fftconvolve(best, best).max()

    t0 = time.monotonic()
    lr = lr0
    step = 0
    last_mc = best_mc

    while time.monotonic() - t0 < t_budget:
        el = time.monotonic() - t0
        f = min(1.0, el / max(t_budget, 1e-6))
        tau = tau_hi * (tau_lo / tau_hi) ** f

        g, c = soft_max_gradient(a, tau)
        a_new = a - lr * g
        a_new = project_simplex(a_new, S)
        step += 1

        if step % 5 == 1:
            mc = fftconvolve(a_new, a_new).max()
            if mc <= last_mc + 1e-10:
                a = a_new
                if mc < best_mc - 1e-10 * best_mc:
                    best_mc = mc
                    best = a.copy()
                    best_score = compute_score(best)
                last_mc = mc
                if lr < lr0 * 5.0:
                    lr *= 1.03
            else:
                a = a_new
                lr = max(0.001, lr * 0.5)
                last_mc = mc
        else:
            a = a_new

    sc = compute_score(best)
    if verbose:
        print(f"  Grad done: steps={step}, best_score={sc:.8f}")
    return best, sc
