`a_new = argmin_{a >= 0, sum(a)=1} { <g, a> + (1/eta) * D_KL(a || a_old) }`

where `D_KL(a || b) = sum a_i * log(a_i/b_i)` is the KL divergence.

The solution is `a_new[i] = a_old[i] * exp(-eta * g_i) / sum_j a_old[j] * exp(-eta * g_j)`

This is exactly the multiplicative update followed by normalization! So my approach is correct.

For the step size, the standard mirror descent theory says `eta = 1/sqrt(t)` where `t` is the iteration number (for the `O(1/sqrt(T))` convergence rate) or `eta = 1/t` (for strongly convex problems, but our problem isn't strongly convex).

"""
Soft-max mirror descent (multiplicative gradient updates) for minimizing
peak auto-convolution of non-negative sequences. Uses temperature annealing,
multi-length/multi-init search, adaptive step sizes, and local search polishing.
"""

import numpy as np
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_softmax_grad(a, T):
    n = len(a)
    conv = np.convolve(a, a)
    shifted = conv - conv.max()
    exp_vals = np.exp(T * shifted)
    weights = exp_vals / exp_vals.sum()
    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    term1 = np.convolve(weights, a[::-1])
    term2 = np.convolve(a, weights)
    grad = term1[n - 1:2 * n - 1] + term2[:n]
    soft_max = conv.max() + np.log(exp_vals.sum()) / T
    return grad, soft_max, conv


def mirror_descent(a, time_limit, T_start=3.0, T_end=200.0,
                   eta_start=0.8, eta_end=0.05, n_steps=None):
    start_time = time.monotonic()
    if n_steps is None:
        n_steps = max(200, int(time_limit * 20))
    a = normalize(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    best_step = 0
    stagnation = 0
    for step in range(n_steps):
        if time.monotonic() - start_time > time_limit:
            break
        frac = step / max(1, n_steps - 1)
        T = T_start * (T_end / T_start) ** frac
        eta = eta_start * (eta_end / eta_start) ** frac
        grad, soft_max, conv = compute_softmax_grad(a, T)
        S = a.sum()
        grad_score = (2.0 * n / (S * S)) * grad - (4.0 * n * soft_max / (S * S * S))
        # Clip extreme gradient values for stability
        g_max = np.max(np.abs(grad_score))
        if g_max > 0:
            grad_score = np.clip(grad_score / g_max, -5.0, 5.0)
        update = np.exp(-eta * grad_score)
        update = np.clip(update, 1e-12, 100.0)
        a_new = a * update
        a_new = normalize(a_new)
        new_score = score_sequence(a_new)
        if new_score < best_score - 1e-15:
            best_score = new_score
            best_a = a_new.copy()
            best_step = step
            a = a_new
            stagnation = 0
        else:
            a = a_new
            stagnation += 1
            if stagnation > 80:
                eta *= 0.9
                stagnation = 0
    return best_a, best_score


def adaptive_mirror_descent(a, time_limit):
    start_time = time.monotonic()
    a = normalize(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    T = 5.0
    eta = 0.5
    patience = 0
    step = 0
    max_steps = 200000
    while time.monotonic() - start_time < time_limit:
        step += 1
        grad, soft_max, conv = compute_softmax_grad(a, T)
        S = a.sum()
        grad_score = (2.0 * n / (S * S)) * grad - (4.0 * n * soft_max / (S * S * S))
        g_norm = np.linalg.norm(grad_score)
        if g_norm < 1e-15:
            break
        g_scaled = grad_score / g_norm
        update = np.exp(-eta * g_scaled)
        update = np.clip(update, 1e-12, 100.0)
        a_new = a * update
        a_new = normalize(a_new)
        new_score = score_sequence(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            patience = 0
            T = min(500.0, T * 1.01)
            eta = max(0.01, eta * 0.999)
        else:
            patience += 1
            if patience > 30:
                eta *= 0.85
                patience = 0
            if patience > 150:
                T = min(500.0, T * 1.5)
                patience = 0
    return best_a, best_score


def local_search(a, time_limit=30.0, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_score = score_sequence(a)
    while time.monotonic() - start_time < time_limit:
        r = rng.random()
        if r < 0.3:
            width = max(2, int(rng.integers(1, max(2, n // 30))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, width))
        elif r < 0.6:
            width = max(5, int(rng.integers(n // 50, max(6, n // 8))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.1, width))
        elif r < 0.8:
            candidate = a * np.exp(rng.normal(0, 0.02, n))
        else:
            # Symmetrize + small perturbation
            candidate = 0.5 * (a + a[::-1])
            candidate *= np.exp(rng.normal(0, 0.02, n))
        candidate = normalize(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
    return best_a, best_score


def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def init_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def init_blackman(n):
    if n == 1:
        return np.ones(1)
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def init_triangle(n):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(t - center) / (center + 1.0))


def init_flat(n):
    return np.ones(n, dtype=np.float64)


def init_smooth_random(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(2, n // 10), max(3, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return init_raised_cosine(n)
    return a / s
