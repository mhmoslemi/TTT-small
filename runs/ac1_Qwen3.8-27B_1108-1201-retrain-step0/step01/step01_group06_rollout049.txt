def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_objective_normalized(a):
    """Objective assuming sum(a)=1: 2n*max(conv(a,a))."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()

def topk_gradient(c, a, k=20):
    """
    Compute subgradient of max(c) using top-k peaks.
    Returns grad where grad[i] = sum over top peaks p: weight_p * 2 * a[p-i] (valid).
    """
    n = len(a)
    # Get top-k indices
    if k >= len(c):
        top_idx = np.arange(len(c))
        top_vals = c.copy()
    else:
        top_idx = np.argsort(c)[-k:]
        top_vals = c[top_idx]
    
    # Normalize weights (uniform over top-k, or proportional to values)
    # Use proportional to values for better conditioning
    w = top_vals / top_vals.sum()
    
    grad = np.zeros(n)
    for i, pk in enumerate(top_idx):
        p = int(pk)
        wi = w[i] * 2
        # a[p-j] valid when 0 <= p-j <= n-1, i.e., p-(n-1) <= j <= p
        j_start = max(0, p - n + 1)
        j_end = min(n - 1, p)
        if j_start <= j_end:
            # a[p-j] for j = j_start..j_end
            # = a[p-j_start], a[p-j_start-1], ..., a[p-j_end]
            # indices in a: from (p-j_end) to (p-j_start)
            a_idx_start = p - j_end
            a_idx_end = p - j_start
            grad[j_start:j_end+1] += wi * a[a_idx_start:a_idx_end+1][::-1]
    
    return grad

    a = np.maximum(a0.astype(np.float64), 0.0)

        a = np.ones(n) / n
    else:
        a = a / a_sum
    
    c = fast_convolve(a)
    best_obj = compute_objective_normalized(a)
    best_a = a.copy()
    
    eta = eta0
    momentum = momentum0
    v = np.zeros(n)  # momentum buffer

    t0 = time.monotonic()

        grad = topk_gradient(c, a, k=k_peaks)
        
        # Center the gradient (project onto simplex tangent space)
        grad_centered = grad - grad.mean()
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad_centered
        
        # Multiplicative update (mirror descent on simplex)
        log_factor = -eta * v
        log_factor -= log_factor.max()
        a_new = a * np.exp(log_factor)
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-12:
            a_new = a.copy()  # revert
        else:
            a_new = a_new / a_sum
        
        a = a_new
        c = fast_convolve(a)
        curr_obj = compute_objective_normalized(a)

            best_obj = curr_obj
            best_a = a.copy()
            patience = 0

Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses accelerated multiplicative mirror descent with top-k peak focus and
momentum, combined with scipy L-BFGS-B polish and local perturbation.
Tries multiple lengths and diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj_norm(a):
    """Objective with sum(a)=1: 2n * max(conv(a,a))."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def topk_subgrad(c, a, k):
    """Subgradient of max(c) using top-k peaks. Returns gradient w.r.t. a."""
    n = len(a)
    if k >= len(c):
        top_idx = np.arange(len(c))
        top_vals = c.copy()
    else:
        top_idx = np.argpartition(c, -k)[-k:]
        top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    w = top_vals / total
    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        p = int(top_idx[ii])
        wi = w[ii] * 2.0
        j_start = max(0, p - n + 1)
        j_end = min(n - 1, p)
        if j_start <= j_end:
            a_s = p - j_end
            a_e = p - j_start
            grad[j_start:j_end + 1] += wi * a[a_s:a_e + 1][::-1]
    return grad


def smooth_max_grad(c, a, T):
    """Smooth max gradient using soft-max with temperature T."""
    n = len(a)
    c_shifted = c - c.max()
    Tc = T * c_shifted
    Tc -= Tc.max()
    w = np.exp(Tc)
    w_sum = w.sum()
    if w_sum < 1e-30:
        grad = np.zeros(n)
        best_k = int(np.argmax(c))
        j_start = max(0, best_k - n + 1)
        j_end = min(n - 1, best_k)
        a_s = best_k - j_end
        a_e = best_k - j_start
        grad[j_start:j_end + 1] = 2.0 * a[a_s:a_e + 1][::-1]
        return grad
    w /= w_sum
    grad = np.zeros(n)
    thresh = w.max() * 1e-6
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        wi = w[k] * 2.0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            a_s = k - j_end
            a_e = k - j_start
            grad[j_start:j_end + 1] += wi * a[a_s:a_e + 1][::-1]
    return grad


def mirror_descent(a0, time_limit, k_peaks=15, eta0=0.8, mom0=0.4, seed=0):
    """Accelerated multiplicative mirror descent with top-k peak focus."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    c = fast_convolve(a)
    best_obj = compute_obj_norm(a)
    best_a = a.copy()
    eta = eta0
    mom = mom0
    v = np.zeros(n)
    patience = 0
    max_pat = 50
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        grad = topk_subgrad(c, a, k=k_peaks)
        grad_c = grad - grad.mean()
        v = mom * v + (1.0 - mom) * grad_c
        step = -eta * v
        step -= step.max()
        a_new = a * np.exp(step)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        c = fast_convolve(a)
        obj = compute_obj_norm(a)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            eta *= 0.65
            mom = min(0.92, mom * 1.06)
            if eta < 0.008:
                # Reheat with perturbation
                alpha = 0.3 + 0.3 * rng.random()
                perturb = rng.dirichlet(np.ones(n) * 0.2)
                a = (1 - alpha) * a + alpha * perturb
                a /= a.sum()
                c = fast_convolve(a)
                eta = eta0 * (0.3 + 0.4 * rng.random())
                mom = mom0
                max_pat = 35
                v = np.zeros(n)

    return best_a, best_obj


def lbfgs_smooth_polish(a0, max_iter=100):
    """L-BFGS-B polish on log-variables with fixed soft-max T=40."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    x0 = np.log(a)
    x0 -= x0.mean()
    T = 40.0

    def obj_grad(x):
        xc = x - x.mean()
        e = np.exp(xc)
        a_s = e / e.sum()
        c = fast_convolve(a_s)
        c_sh = c - c.max()
        Tc = T * c_sh
        Tc -= Tc.max()
        smooth_max = c.max() + np.log(np.sum(np.exp(Tc))) / T
        obj = 2.0 * n * smooth_max
        # Gradient of smooth max w.r.t. a
        w = np.exp(T * (c_sh - c_sh.max()))
        ws = w.sum()
        if ws < 1e-30:
            grad_a = np.zeros(n)
            bk = int(np.argmax(c))
            js = max(0, bk - n + 1)
            je = min(n - 1, bk)
            a_s_idx = bk - je
            a_e_idx = bk - js
            grad_a[js:je + 1] = 2.0 * a_s[a_s_idx:a_e_idx + 1][::-1]
        else:
            w /= ws
            grad_a = np.zeros(n)
            thresh = w.max() * 1e-7
            active = np.where(w > thresh)[0]
            for ki in active:
                k = int(ki)
                wi = w[k] * 2.0
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                if js <= je:
                    a_s_idx = k - je
                    a_e_idx = k - js
                    grad_a[js:je + 1] += wi * a_s[a_s_idx:a_e_idx + 1][::-1]
        # Chain rule through softmax parameterization
        dot = np.dot(grad_a, a_s)
        grad_x = 2.0 * n * a_s * (grad_a - dot)
        return obj, grad_x

    result = minimize(obj_grad, x0, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'maxfun': max_iter * 3, 'gtol': 1e-5})
    if result.success or result.fun < compute_obj_norm(a):
        x_opt = result.x - result.x.mean()
        a_opt = np.exp(x_opt)
        a_opt = np.maximum(a_opt, 0.0)
        s = a_opt.sum()
        if s > 1e-12:
            a_opt = a_opt / s
            return a_opt
    return a


def local_search(a, budget_s, seed=1):
    """Random perturbation local search."""
    rng = np.random.default_rng(seed)
    n = len(a)
    best = a.copy()
    best_obj = compute_obj_norm(a)
    t0 = time.monotonic()
    scale = 0.05
    while time.monotonic() - t0 < budget_s:
        noise = rng.normal(0, scale, n)
        a_new = best + noise
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            continue
        a_new = a_new / s
        obj = compute_obj_norm(a_new)
        if obj < best_obj - 1e-9:
            best = a_new.copy()
            best_obj = obj
            scale = max(0.005, scale * 1.1)
        else:
            scale = max(0.005, scale * 0.95)
    return best


def generate_inits(n, rng, parent):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    inits.append(('uniform', np.ones(n)))
    for sf in [0.1, 0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        inits.append((f'g{sf}', g))
    bi = np.zeros(n)
    w = max(1, n // 3)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits.append(('bimodal', bi))
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(('ramp', ramp))
    cos_b = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits.append(('cos', cos_b))
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits.append(('rand', r))
    rb = np.zeros(n)
    rb[rng.random(n) < 0.25] = 1.0
    if rb.sum() > 1:
        inits.append(('rand_bin', rb))
    alt = np.ones(n)
    alt[::2] = 0.2
    inits.append(('alt', alt))
    # Two bumps
    for ratio in [0.2, 0.3]:
        tb = np.zeros(n)
        c1 = int(n * ratio)
        c2 = int(n * (1 - ratio))
        tb[c1] = 1.0
        tb[c2] = 1.0
        sigma = n * 0.05
        tb = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
        inits.append((f'twobug{ratio}', tb))
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits.append(('parent', pp))
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits.append(('parent_i', pp))
    return inits

    """Main entry: hybrid optimization for step function coefficient minimization."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 25.0)

    best_sequence = None
    best_score = float('inf')

    parent = None

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    def update_best(a):
        nonlocal best_sequence, best_score
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 1e-10:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  >>> New best: {score:.6f} (n={len(a)})")

    if parent is not None:
        print(f"Parent: n={len(parent)}, sum={parent.sum():.1f}")
        update_best(parent)

    lengths = [64, 128, 256, 512, 1024, 2048]
    all_configs = []
    for n in lengths:
        inits = generate_inits(n, rng, parent)
        for name, a0 in inits:
            all_configs.append((n, name, a0))

    print(f"Phase 1: Scan {len(all_configs)} configs")
    p1_budget = min(500.0, (deadline - time.monotonic()) * 0.25)
    t_per = max(2.0, p1_budget / len(all_configs))
    scan_results = []

    for idx, (n, name, a0) in enumerate(all_configs):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 180)
        if t_alloc < 1.5:
            break
        a_opt, obj = mirror_descent(a0, time_limit=t_alloc, k_peaks=10, eta0=1.0, mom0=0.3, seed=idx)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            update_best(a_opt)

    scan_results.sort(key=lambda z: z[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Deep optimization
    print(f"Phase 2: Deep optimization")
    top_cands = []
    seen_n = set()
    for score, n, name, a_opt in scan_results:
        if len(top_cands) < 6:
            top_cands.append((n, a_opt))
            seen_n.add(n)
        elif n not in seen_n and len(top_cands) < 10:
            top_cands.append((n, a_opt))
            seen_n.add(n)
    if best_sequence is not None:
        top_cands.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 90
    t_deep = max(5.0, remaining / max(1, len(top_cands)))

    for i, (n, a0) in enumerate(top_cands):
        if time.monotonic() > deadline - 90:
            break
        t_alloc = min(t_deep, deadline - time.monotonic() - 90)
        if t_alloc < 4:
            break

        for k_p, eta_v in [(8, 0.9), (20, 0.7), (50, 0.5)]:
            if time.monotonic() > deadline - 90:
                break
            t_s = t_alloc / 3.0
            a_opt, obj = mirror_descent(a0, time_limit=t_s, k_peaks=k_p, eta0=eta_v, mom0=0.5, seed=100 + i)
            update_best(a_opt)

        # L-BFGS polish
        if time.monotonic() < deadline - 90:
            a_opt = lbfgs_smooth_polish(a0, max_iter=80)
            update_best(a_opt)

    # Phase 3: Final intensive polish
    print(f"Phase 3: Final polish. Best: {best_score:.6f}")
    if best_sequence is not None and time.monotonic() < deadline - 20:
        rem = deadline - 20 - time.monotonic()
        n = len(best_sequence)

        # Round-trip: try to improve by re-optimizing with different params
        for attempt in range(4):
            if rem < 8:
                break
            t_s = min(rem / 4.0, 40.0)
            k_p = [10, 25, 5, 40][attempt % 4]
            eta_v = [1.0, 0.8, 1.2, 0.6][attempt % 4]
            a_opt, obj = mirror_descent(best_sequence, time_limit=t_s, k_peaks=k_p,
                                         eta0=eta_v, mom0=0.6, seed=200 + attempt)
            update_best(a_opt)
            rem = deadline - 20 - time.monotonic()

        # L-BFGS
        if rem > 15:
            a_opt = lbfgs_smooth_polish(best_sequence, max_iter=120)
            update_best(a_opt)
            rem = deadline - 20 - time.monotonic()

        # Local search
        if rem > 10:
            a_opt = local_search(best_sequence, budget_s=min(rem - 5, 60), seed=999)
            update_best(a_opt)
            rem = deadline - 20 - time.monotonic()

        # Try sparsification: zero out small elements
        if rem > 5 and best_sequence is not None:
            a_sp = best_sequence.copy()
            thresh = np.sort(a_sp)[int(0.1 * len(a_sp))]
            a_sp[a_sp < thresh] = 0.0
            if a_sp.sum() > 1e-10:
                a_sp /= a_sp.sum()
                update_best(a_sp)

            # Try binary rounding
            a_bin = (best_sequence > np.median(best_sequence[best_sequence > 0])).astype(float)
            if a_bin.sum() > 1:
                s = float(evaluate_sequence(a_bin.tolist()))
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_sequence = a_bin.copy()
                    print(f"  Binary rounding improved: {best_score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
