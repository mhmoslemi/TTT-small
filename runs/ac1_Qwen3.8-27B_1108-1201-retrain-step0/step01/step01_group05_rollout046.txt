"""
Minimize 2*n*max_conv/sum^2 using log-sum-exp smoothed gradient descent
with FFT-accelerated gradient computation and multiple optimization strategies.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c = fft_convolve_self(a)
    # Softmax of beta*c (shifted for numerical stability)
    bc = beta * c
    bc_shifted = bc - bc.max()
    w = np.exp(bc_shifted)
    w /= w.sum()
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (cross-correlation)
    # Using FFT: R = ifft(FFT(a) * conj(FFT(w)))[:n]
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    R = np.real(ifft(A * np.conj(W)))[:n]
    g = 2.0 * R
    # Also return c.max() for tracking
    obj = c.max()
    smooth_obj = (bc.max() + np.log(w.sum())) / beta  # = logsumexp(bc)/beta
    return g, obj, smooth_obj


def run_smooth_adam(a_init, n, S, max_time, rng, beta_start=50.0, beta_end=5000.0,
                    lr_init=None, total_steps=5000):
    """Adam with log-sum-exp smoothing, increasing beta."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s <= 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    if lr_init is None:
        lr_init = 0.3 * S / n
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        # Schedule beta
        progress = min(1.0, t / total_steps)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        g, obj, smooth_obj = compute_smooth_gradient(a, beta)
        if obj < best_score:
            best_score = obj
            best_local = a.copy()
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g = g / gnorm
        
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr = lr_init * (0.5 ** (float(t) / 1000.0))
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        
        # Occasional random perturbation
        if t % 1000 == 0 and rng.random() < 0.05:
            a = project_simplex(a + rng.normal(0, 0.005 * S / n, n), S)
    
    return best_local, best_score

def run_lbfgs_smooth(a_init, n, S, max_time, beta=500.0):

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s <= 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    
    def objective_and_grad(x):
        nonlocal best_score, best_local
        x = np.maximum(x, 0.0)
        xs = x.sum()
        if xs < 1e-15:
            return 1e10, np.zeros(n)
        x = x * (S / xs)
        c = fft_convolve_self(x)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = x.copy()
        # Smooth objective
        bc = beta * c
        bc_shift = bc - bc.max()
        log_sum = bc_shift.max() + np.log(np.exp(bc_shift).sum())
        obj = log_sum / beta
        # Gradient
        w = np.exp(bc_shift)
        w /= w.sum()
        size = 2 * len(x) - 1
        fsize = 1 << (size - 1).bit_length()
        A = fft(x, fsize)
        W = fft(w, fsize)
        R = np.real(ifft(A * np.conj(W)))[:n]
        g = 2.0 * R
        return obj, g
    
    # Use L-BFGS-B but we need to handle constraints
    # We'll use bounds to enforce non-negativity and project to simplex in a callback
    # Actually, simpler: just optimize on the unconstrained space with the
    # re-normalized objective
    
    # Hmm, L-BFGS-B doesn't handle equality constraints. Let's use a 
    # change of variables: a = exp(y) / sum(exp(y)) * S
    # This ensures a > 0 and sum(a) = S automatically.
    
    y0 = np.log(np.maximum(a, 1e-10) * n / S)  # log(a) - log(S/n)
    
    def obj_reparam(y):
        nonlocal best_score, best_local
        # Softmax
        y_shifted = y - y.max()
        exp_y = np.exp(y_shifted)
        a_new = exp_y / exp_y.sum() * S
        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a_new.copy()
        bc = beta * c
        bc_shift = bc - bc.max()
        log_sum = bc_shift.max() + np.log(np.exp(bc_shift).sum())
        obj = log_sum / beta
        # Gradient w.r.t. y
        w = np.exp(bc_shift)
        w /= w.sum()
        size = 2 * n - 1
        fsize = 1 << (size - 1).bit_length()
        A = fft(a_new, fsize)
        W = fft(w, fsize)
        R = np.real(ifft(A * np.conj(W)))[:n]
        g_a = 2.0 * R
        # Chain rule: da_i/dy_j = a_i * (delta_ij - a_j/S)
        # d(obj)/dy_j = sum_i g_a_i * da_i/dy_j
        #              = g_a_j * a_j * (1 - a_j/S) - a_j * sum_i g_a_i * a_i / S
        #              = g_a_j * a_j - a_j * (g_a @ a) / S
        #              = a_j * (g_a_j - (g_a @ a) / S)
        g_a_dot_a = np.dot(g_a, a_new)
        g_y = a_new * (g_a - g_a_dot_a / S)

        if time.monotonic() - start > max_time:
            raise RuntimeError("timeout")
        
        return obj, g_y

        result = minimize(obj_reparam, y0, method='L-BFGS-B',
                         options={'maxiter': 10000, 'maxfun': 5000,
                                  'disp': False, 'gtol': 1e-8})
    except RuntimeError:
        pass
    except Exception:
        pass
    
    return best_local, best_score


def run_multiplicative(a_init, n, S, max_time, rng, eta_init=0.5):
    """Multiplicative update / mirror descent targeted at peak reduction."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s <= 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    eta = eta_init
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Find peak lags
        peak_val = cur
        # Soft assignment: weight each lag by exp(alpha * (c[k] - peak_val))
        alpha = 10.0
        w = np.exp(alpha * (c - peak_val))
        w /= w.sum()
        
        # Gradient (same as before)
        size = 2 * n - 1
        fsize = 1 << (size - 1).bit_length()
        A = fft(a, fsize)
        W = fft(w, fsize)
        R = np.real(ifft(A * np.conj(W)))[:n]
        g = 2.0 * R
        
        # Multiplicative update
        gmax = g.max()
        if gmax < 1e-15:
            break
        a_new = a * np.exp(-eta * g / gmax)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new * (S / a_new.sum())
        a = a_new
        
        eta *= 0.999  # Slowly decay
    
    return best_local, best_score

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using
log-sum-exp smoothed full-gradient optimization with multiple
complementary strategies (Adam, multiplicative, L-BFGS-B) and
progressive beta sharpening for near-exact max tracking.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_full_gradient(a, beta):
    """Compute gradient of logsumexp(beta*c)/beta and the max of c.
    Uses full softmax weighting over all convolution lags."""
    n = len(a)
    c = fft_convolve_self(a)
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    smooth_obj = (bc_max + np.log(exp_bc.sum())) / beta
    # Gradient via cross-correlation: g[i] = 2 * sum_k w[k] * a[k-i]
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    R = np.real(ifft(A * np.conj(W)))[:n]
    g = 2.0 * R
    return g, c.max(), smooth_obj

                    lr_init=None, total_steps=5000):

    s = a.sum()
    if s <= 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    if lr_init is None:
        lr_init = 0.4 * S / n
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, float(t) / total_steps)
        beta = beta_start * (beta_end / beta_start) ** progress
        g, cur_max, _ = compute_full_gradient(a, beta)
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g = g / gnorm
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr_init * (0.5 ** (float(t) / 2000.0))
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        if t % 2000 == 0 and rng.random() < 0.03:
            a = project_simplex(a + rng.normal(0, 0.003 * S / n, n), S)
    return best_local, best_score


def run_multiplicative(a_init, n, S, max_time, rng, eta_init=0.4, alpha=20.0):
    """Multiplicative weights with soft peak targeting."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s <= 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    eta = eta_init
    t = 0
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        # Soft weights concentrated near the peak
        w = np.exp(alpha * (c - cur))
        w /= w.sum()
        size = 2 * n - 1
        fsize = 1 << (size - 1).bit_length()
        A = fft(a, fsize)
        W = fft(w, fsize)
        R = np.real(ifft(A * np.conj(W)))[:n]
        g = 2.0 * R
        gmax = g.max()
        if gmax < 1e-15:
            break
        a = a * np.exp(-eta * g / gmax)
        a = np.maximum(a, 1e-30)
        a = a * (S / a.sum())
        eta *= 0.9995
    return best_local, best_score


def _reparam_obj(y, n, S, beta):
    """Top-level objective+gradient for L-BFGS-B via softmax reparameterization."""
    y_shift = y - y.max()
    exp_y = np.exp(y_shift)
    a = exp_y / exp_y.sum() * S
    c = fft_convolve_self(a)
    bc = beta * c
    bc_shift = bc - bc.max()
    exp_bc = np.exp(bc_shift)
    log_sum = bc_shift.max() + np.log(exp_bc.sum())
    obj = log_sum / beta
    w = exp_bc / exp_bc.sum()
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    R = np.real(ifft(A * np.conj(W)))[:n]
    g_a = 2.0 * R
    g_a_dot_a = np.dot(g_a, a)
    g_y = a * (g_a - g_a_dot_a / S)
    return obj, g_y


def run_lbfgs(a_init, n, S, max_time, beta=300.0, maxfun=3000):
    """L-BFGS-B on reparameterized smooth objective."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s <= 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    # Track best via the final evaluation
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    # Check periodic best
    best_tracking = [best_score, best_local.copy()]
    start = time.monotonic()
    y0 = np.log(np.maximum(a, 1e-10) * n / S)

    # Count evaluations to estimate time
    eval_count = [0]
    deadline = start + max_time

    def _timeout_check(*_args, **_kwargs):
        pass

    try:
        result = minimize(_reparam_obj, y0, method='L-BFGS-B',
                          args=(n, S, beta),
                          options={'maxiter': 10000, 'maxfun': maxfun,
                                   'disp': False, 'gtol': 1e-6})
    except Exception:
        result = None

    if result is not None and hasattr(result, 'x'):
        y_end = result.x
        y_shift = y_end - y_end.max()
        exp_y = np.exp(y_shift)
        a_end = exp_y / exp_y.sum() * S
        c_end = fft_convolve_self(a_end)
        if c_end.max() < best_score:
            best_score = c_end.max()
            best_local = a_end.copy()

    return best_local, best_score


def run_nesterov(a_init, n, S, max_time, beta_start=50.0, beta_end=5000.0,
                 lr_init=None, total_steps=5000):
    """Nesterov-accelerated gradient with smooth objective."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s <= 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    if lr_init is None:
        lr_init = 0.3 * S / n
    y = a.copy()  # Nesterov lookahead
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, float(t) / total_steps)
        beta = beta_start * (beta_end / beta_start) ** progress
        # Evaluate gradient at Nesterov point
        g, cur_max, _ = compute_full_gradient(y, beta)
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g = g / gnorm
        lr = lr_init / (1.0 + 0.1 * t)
        # Update
        a_new = project_simplex(a - lr * g, S)
        # Nesterov momentum
        mt = 1.0 + np.sqrt(1.0 + 4.0 * t * t)
        momentum = 0.5 * (3.0 - mt)
        y = a_new + momentum * (a_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = a_new
    return best_local, best_score


def get_initializations(rng, n):
    """Generate diverse starting points of length n."""
    inits = []
    # Given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given = np.clip(given, 0, 1000)
        if given.size == n and given.sum() > 0:
            inits.append(("given", given.copy()))
        elif given.size >= 64:
            # Resample to length n
            idx = (np.linspace(0, given.size - 1, n)).astype(int)
            inits.append(("given_resample", given[idx].copy()))
    except Exception:
        pass
    t = np.linspace(0, 1, n)
    # Raised cosine
    inits.append(("raised", 0.5 + 0.5 * np.cos(2.0 * np.pi * t)))
    # Broad Gaussian
    center = n / 2.0
    sigma = n / 4.0
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(("gauss", gauss + 0.1))
    # U-shaped (high at edges)
    u_shape = 0.3 + 0.7 * np.abs(2.0 * t - 1.0)
    inits.append(("u_shaped", u_shape))
    # Sinusoidal modulation
    inits.append(("sin2", 1.0 + 0.4 * np.sin(2.0 * np.pi * 2.0 * t)))
    # Random smooth
    freqs = rng.integers(2, 15, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.1, 0.4, size=4)
    sm = np.ones(n)
    for i in range(4):
        sm += amps[i] * np.sin(2.0 * np.pi * freqs[i] * t + phases[i])
    inits.append(("smooth_rand", np.maximum(sm, 0.1)))
    # Pure random
    inits.append(("random", rng.random(n) + 0.05))
    return inits


def polish_sequence(a, budget, rng):
    """Multi-phase polishing of the best sequence found."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.clip(a, 0, 1000).copy()
    s = a.sum()
    a = a * (S / s)
    best = a.copy()
    best_score = fft_convolve_self(a).max()
    t_start = time.monotonic()
    # Phase 1: Smooth Adam with moderate-to-high beta
    seg = min(budget * 0.3, budget - 5)
    if seg > 1:
        a1, s1 = run_smooth_adam(a, n, S, seg, rng,
                                  beta_start=100, beta_end=5000,
                                  lr_init=0.15 * S / n, total_steps=3000)
        if s1 < best_score:
            best, best_score = a1, s1
        print("    polish1 (smooth_adam): %.6f" % best_score)
    # Phase 2: Nesterov
    left = max(0.0, budget - (time.monotonic() - t_start))
    seg2 = min(left * 0.3, left - 5)
    if seg2 > 1:
        a2, s2 = run_nesterov(best, n, S, seg2,
                               beta_start=200, beta_end=10000,
                               lr_init=0.1 * S / n, total_steps=3000)
        if s2 < best_score:
            best, best_score = a2, s2
        print("    polish2 (nesterov): %.6f" % best_score)
    # Phase 3: Multiplicative
    left = max(0.0, budget - (time.monotonic() - t_start))
    seg3 = min(left * 0.3, left - 5)
    if seg3 > 1:
        a3, s3 = run_multiplicative(best, n, S, seg3, rng, eta_init=0.2, alpha=30.0)
        if s3 < best_score:
            best, best_score = a3, s3
        print("    polish3 (multiplicative): %.6f" % best_score)
    # Phase 4: L-BFGS-B with high beta
    left = max(0.0, budget - (time.monotonic() - t_start))
    seg4 = min(left * 0.4, left - 3)
    if seg4 > 3:
        a4, s4 = run_lbfgs(best, n, S, seg4, beta=2000, maxfun=2000)
        if s4 < best_score:
            best, best_score = a4, s4
        print("    polish4 (lbfgs): %.6f" % best_score)
    # Phase 5: Final smooth Adam with highest beta
    left = max(0.0, budget - (time.monotonic() - t_start))
    if left > 3:
        a5, s5 = run_smooth_adam(best, n, S, left, rng,
                                  beta_start=2000, beta_end=50000,
                                  lr_init=0.05 * S / n, total_steps=5000)
        if s5 < best_score:
            best, best_score = a5, s5
        print("    polish5 (final_adam): %.6f" % best_score)
    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: explore multiple initializations, then polish the best."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0
    # Determine primary length
    n_primary = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if 64 <= given.size <= 65536:
            n_primary = given.size
    except Exception:
        pass
    inits = get_initializations(rng, n_primary)
    total_budget = max(1.0, deadline - time.monotonic())
    n_inits = len(inits)
    explore_fraction = 0.55
    explore_time = total_budget * explore_fraction
    per_init_time = explore_time / max(1, n_inits)
    global_best = np.inf
    global_a = None
    for name, arr in inits:
        n = len(arr)
        if n < 64:
            continue
        S = np.sqrt(2.0 * n)
        t0 = time.monotonic()
        remaining = per_init_time
        # Run happy path: smooth Adam
        seg = remaining * 0.45
        a1, s1 = run_smooth_adam(arr, n, S, seg, rng,
                                  beta_start=20, beta_end=800,
                                  lr_init=0.5 * S / n, total_steps=2000)
        if s1 < global_best:
            global_best, global_a = s1, a1.copy()
        print("%s | smooth_adam=%.6f (t=%.1fs)" % (name, s1, time.monotonic() - t0))
        # Multiplicative
        if time.monotonic() < deadline - (total_budget * (1 - explore_fraction)):
            seg2 = remaining * 0.3
            a2, s2 = run_multiplicative(a1, n, S, seg2, rng, eta_init=0.3, alpha=15.0)
            if s2 < global_best:
                global_best, global_a = s2, a2.copy()
            print("%s | multiplicative=%.6f (t=%.1fs)" % (name, s2, time.monotonic() - t0))
        # Nesterov
        if time.monotonic() < deadline - (total_budget * (1 - explore_fraction)):
            seg3 = remaining * 0.25
            a3, s3 = run_nesterov(a1, n, S, seg3,
                                   beta_start=50, beta_end=2000,
                                   lr_init=0.3 * S / n, total_steps=2000)
            if s3 < global_best:
                global_best, global_a = s3, a3.copy()
            print("%s | nesterov=%.6f (t=%.1fs)" % (name, s3, time.monotonic() - t0))
    # Polish phase
    left = deadline - time.monotonic()
    if left > 10 and global_a is not None:
        print("=== Polish phase: %.1fs remaining, best=%.6f ===" % (left, global_best))
        global_a, global_best = polish_sequence(global_a, left, rng)
    # Final verification
    result = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(256, dtype=np.float64)
    ver = evaluate_sequence(result.tolist())
    print("Final verified: %.8f" % ver)
    if not np.isfinite(ver):
        result = np.ones(256, dtype=np.float64)
        ver = evaluate_sequence(result.tolist())
    return [float(x) for x in result]
