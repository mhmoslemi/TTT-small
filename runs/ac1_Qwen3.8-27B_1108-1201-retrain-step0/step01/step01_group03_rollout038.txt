(3) multiplicative (Sinkhorn-like) updates, (4) random restarts.

import numpy as np

from scipy.optimize import linprog

def _obj(a):
    """Compute objective value."""
    n = len(a)
    c = np.convolve(a, a)
    S = a.sum()
    if S < 1e-12:
        return float('inf')
    return 2.0 * n * c.max() / (S * S)

def _normalize(a):
    """Normalize to sum=1."""
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a / S

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-12:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _subgrad_step(a, lr):
    """One subgradient step for minimizing max conv(a,a) subject to a on simplex."""
    n = len(a)
    a = _normalize(a)
    c = np.convolve(a, a)
    # Find argmax (or set of argmax)
    max_val = c.max()
    # Get all indices where c is close to max
    tol = 1e-8
    peak_indices = np.where(c >= max_val - tol)[0]
    
    # Compute subgradient: average of 2*B_k*a for k in peak_indices
    # (B_k*a)[j] = a[k-j] if 0 <= k-j < n, else 0
    grad = np.zeros(n)
    for k in peak_indices:
        # B_k * a: for each j, (B_k*a)[j] = a[k-j] if 0<=k-j<n
        shifted = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            shifted[j] = a[k - j]
        grad += 2.0 * shifted
    grad /= len(peak_indices)
    
    # Projected step
    a_new = _project_simplex(a - lr * grad)
    return a_new

def _mul_step(a, lr):
    """Multiplicative (Sinkhorn-like) step."""
    n = len(a)
    a = _normalize(a)
    c = np.convolve(a, a)
    max_val = c.max()
    peak_indices = np.where(c >= max_val - 1e-8)[0]
    
    # Compute responsibility: for each j, how much does a[j] contribute to the peak?
    resp = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            resp[j] += a[k - j]
    resp /= len(peak_indices)
    
    # Multiplicative update
    a_new = a * np.exp(-lr * resp / max(resp.max(), 1e-12))
    return _normalize(a_new)

def _lp_refine(f, target, max_iters=50):
    """LP-based refinement: given f, find g that keeps conv(f,g) <= target."""
    n = len(f)
    if n > 1024:
        return None
    
    # Build constraint matrix efficiently
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = f[k-j] for k in [j, j+n-1]
        A[j:j+n, j] = f
    
    b = np.full(2 * n - 1, target)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except:
        pass
    return None

            # Try to find a better g

def _compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    S = float(a.sum())
    if S < 1e-12:
        return float('inf')
    return float(2.0 * n * float(c.max()) / (S * S))

def _normalize(a):
    S = a.sum()
    if S < 1e-12:
        return a / max(len(a), 1)
    return a / S

def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _subgrad_step(a, lr, momentum_buf, beta=0.9):
    n = len(a)
    a = _normalize(a)
    c = np.convolve(a, a)
    max_val = float(c.max())
    tol = max_val * 1e-10 + 1e-15
    peak_mask = c >= (max_val - tol)
    peak_indices = np.where(peak_mask)[0]
    
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * a[k - j]
    grad /= len(peak_indices)
    
    momentum_buf[:] = beta * momentum_buf + (1.0 - beta) * grad
    a_new = _project_simplex(a - lr * momentum_buf)
    return a_new

def _lp_min_peak(f, deadline=None):
    n = len(f)
    if n > 2048:
        return None
    
    # Minimize t s.t. conv(f,g)[k] <= t for all k, sum(g)=1, g>=0
    # Variables: [g(0), ..., g(n-1), t]
    # Objective: minimize t -> c_obj = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: conv(f,g)[k] - t <= 0
    # A_ub has shape (2n-1, n+1)
    A_ub = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        A_ub[j:j+n, j] = f
    A_ub[:, -1] = -1.0  # -t term
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0, t unbounded below (but t >= 0 in practice)
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:

            return None
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')

            return res.x[:n]
    except Exception:
        pass
    return None

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses projected subgradient descent with momentum, LP-based min-peak refinement,
simulated annealing, and diverse starting points of varying lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    S = float(a.sum())
    if S < 1e-12:
        return float('inf')
    return float(2.0 * n * float(np.max(c)) / (S * S))


def _normalize(a):
    """Normalize to sum=1."""
    S = a.sum()
    if S < 1e-12:
        return np.ones_like(a) / len(a)
    return a / S


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v_sum = v.sum()
    if v_sum < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgrad_step(a, lr, momentum_buf, beta=0.9):
    """One momentum subgradient step minimizing max conv(a,a) on the simplex."""
    n = len(a)
    a = _normalize(a)
    c = np.convolve(a, a)
    max_val = float(np.max(c))
    tol = max_val * 1e-10 + 1e-15
    peak_indices = np.where(c >= (max_val - tol))[0]

    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi : k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * seg
    grad /= float(len(peak_indices))

    momentum_buf[:] = beta * momentum_buf + (1.0 - beta) * grad
    a_new = _project_simplex(a - lr * momentum_buf)
    return a_new


def _lp_min_peak(f, deadline=None):
    """LP: minimize t s.t. conv(f,g)[k] <= t for all k, sum(g)=1, g>=0."""
    n = len(f)
    if n > 2048:
        return None

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(-1.0, None)]

    try:
        if deadline is not None and time.monotonic() > deadline:
            return None
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(1.0, deadline - time.monotonic()) if deadline else None})
        if res.status == 0:
            return np.maximum(res.x[:n], 0.0)
    except Exception:
        pass
    return None


def _run_subgrad(a0, deadline, max_iters=10000, lr_init=0.5, lr_decay=0.95, decay_every=100):
    """Run subgradient descent with momentum. Returns (best_seq, best_score)."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a)
    score = _compute_obj(a)
    best = a.copy()
    best_score = score

    momentum = np.zeros(n)
    lr = lr_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if (it + 1) % decay_every == 0:
            lr *= lr_decay
            if lr < 0.001:
                lr = 0.001

        a = _subgrad_step(a, lr, momentum)

        if (it + 1) % 300 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score


def _run_lp_refine(a0, deadline, max_lp_iters=40):
    """LP-based refinement. Returns (best_seq, best_score)."""
    n = len(a0)
    a = _normalize(a0.astype(np.float64).copy())
    score = _compute_obj(a)
    best = a.copy()
    best_score = score

    if n > 1024:
        return best, best_score

    momentum = np.zeros(n)

    for it in range(max_lp_iters):
        if time.monotonic() > deadline - 3.0:
            break

        g = _lp_min_peak(a, deadline=deadline - 3.0)
        if g is None:
            break

        g = _normalize(g)

        improved = False
        for t_blend in [0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new)
            s = _compute_obj(a_new)
            if s < best_score - 1e-10:
                best_score = s
                best = a_new.copy()
                a = a_new
                improved = True
                break

        if not improved:
            # Escape with a subgradient step
            a = _subgrad_step(a, 0.15, momentum)
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
            else:
                # Try another blend
                for t_blend in [0.5, 0.3]:
                    a_new = (1.0 - t_blend) * a + t_blend * g
                    a_new = np.maximum(a_new, 0.0)
                    a_new = _normalize(a_new)
                    s = _compute_obj(a_new)
                    if s < best_score:
                        best_score = s
                        best = a_new.copy()
                        a = a_new
                        break

    return best, best_score


def _run_anneal(a0, deadline, rng, n_iter=8):
    """Simulated annealing: perturb and re-optimize. Returns (best_seq, best_score)."""
    a = _normalize(a0.astype(np.float64).copy())
    best = a.copy()
    best_score = _compute_obj(a)
    n = len(a)
    T = 0.15

    for it in range(n_iter):
        if time.monotonic() > deadline - 10.0:
            break
        noise = rng.random(n) * T
        perturbed = np.maximum(a + noise, 0.0)
        perturbed = _normalize(perturbed)

        seq, score = _run_subgrad(perturbed, deadline - 5.0, max_iters=3000,
                                   lr_init=0.3, lr_decay=0.94, decay_every=80)
        if score < best_score:
            best_score = score
            best = seq.copy()
            a = seq.copy()
        T *= 0.65

    return best, best_score
