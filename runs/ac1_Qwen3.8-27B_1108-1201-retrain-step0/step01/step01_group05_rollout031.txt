```python
"""
Minimize 2*n*max(self_convolution) / sum^2 over non-negative sequences.

Uses an LP-based conditional gradient method:
- At each step, solve a reduced LP to find the best direction g
  that stays within the current max-convolution level set.
- Move toward g with a line search.
- This is more informed than plain subgradient descent because the LP
  considers multiple active constraints simultaneously.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def build_lp_matrix(f, active_lags):
    """Build the K x n constraint matrix for the reduced LP."""
    n = len(f)
    K = len(active_lags)
    C = np.zeros((K, n), dtype=np.float64)
    for i, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        # (f*g)[k] = sum_{j=lo}^{hi-1} f[k-j] * g[j]
        # f[k-j] for j in [lo, hi-1] => indices k-hi+1 to k-lo in f
        C[i, lo:hi] = f[k - hi + 1 : k - lo + 1][::-1]
    return C


def solve_lp_direction(f, M, active_lags):
    """Solve: max sum(g) s.t. C@g <= M*1, g >= 0."""
    n = len(f)
    K = len(active_lags)
    C = build_lp_matrix(f, active_lags)
    
    # Minimize -sum(g) subject to C@G <= M*1, g >= 0
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(K, M, dtype=np.float64)
    
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    if res.status != 0 or res.x is None:
        return None
    return np.maximum(res.x, 0.0)

    c = fft_convolve_self(f)

    best_obj = 2.0 * n * best_M / S**2

        cur_obj = 2.0 * n * M / S**2

            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
```

```python
"""
Minimize 2*n*max(self_convolution) / sum^2 over non-negative sequences.

Strategy:
- LP-based conditional gradient: at each step, solve a reduced LP to find
  the best direction g that keeps (f*g) within the current max-convolution
  level set, then move toward g with adaptive step size.
- This is more informed than plain subgradient descent because the LP
  considers multiple active constraints simultaneously.
- Subgradient refinement for final polish.
- Multiple phases with different hyperparameters for robustness.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution of a via FFT (length 2n-1)."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def build_lp_matrix(f, active_lags):
    """Build K x n constraint matrix for reduced LP.
    Row i corresponds to constraint (f*g)[active_lags[i]] <= M.
    """
    n = len(f)
    K = len(active_lags)
    C = np.zeros((K, n), dtype=np.float64)
    for i in range(K):
        k = int(active_lags[i])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            # (f*g)[k] = sum_{j=lo}^{hi-1} f[k-j] * g[j]
            # For j in [lo, hi), k-j ranges from k-lo down to k-hi+1
            C[i, lo:hi] = f[k - hi + 1:k - lo + 1][::-1]
    return C


def solve_lp_direction(f, M, active_lags, g_upper):
    """Solve: max sum(g) s.t. C@g <= M*1, 0 <= g <= g_upper."""
    n = len(f)
    K = len(active_lags)
    if K == 0:
        return None
    C = build_lp_matrix(f, active_lags)
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(K, M, dtype=np.float64)
    bounds = [(0.0, g_upper)] * n
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception:
        try:
            res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                          method='interior-point')
        except Exception:
            return None
    if res.status != 0 or res.x is None:
        return None
    return np.maximum(res.x, 0.0)


def optimize_lp(f_init, max_time, K=50, threshold=0.85, g_upper_mult=2.0):
    """LP-based conditional gradient optimization."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    g_upper = g_upper_mult * S

    f = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
    f_sum = f.sum()
    if f_sum < 1e-15:
        f = np.ones(n, dtype=np.float64)
        f_sum = n
    f = f * (S / f_sum)

    best_f = f.copy()
    c = fft_convolve_self(f)
    best_M = c.max()
    best_obj = 2.0 * n * best_M / (S * S)

    start = time.monotonic()
    t_iter = 0
    no_improve = 0
    patience = 80

    while time.monotonic() - start < max_time:
        t_iter += 1
        elapsed = time.monotonic() - start

        c = fft_convolve_self(f)
        M = c.max()
        cur_obj = 2.0 * n * M / (S * S)

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            break

        # Select active lags
        active_mask = c >= threshold * M
        active_set = np.where(active_mask)[0]
        if len(active_set) > K:
            active_set = np.argsort(c)[::-1][:K]
        elif len(active_set) == 0:
            active_set = np.array([int(np.argmax(c))], dtype=np.int64)

        # Solve LP for direction
        g = solve_lp_direction(f, M, active_set, g_upper)
        if g is None or g.sum() <= 1e-15:
            break

        # Rescale g to match normalization
        g = g * (S / g.sum())

        # Line search for step size
        best_t = 0.01
        best_new_M = M
        step_candidates = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
                           0.015625, 0.0078125, 0.00390625]
        for t in step_candidates:
            new_f = (1.0 - t) * f + t * g
            new_f = np.maximum(new_f, 0.0)
            ns = new_f.sum()
            if ns < 1e-15:
                continue
            new_f = new_f * (S / ns)
            new_M = fft_convolve_self(new_f).max()
            if new_M < best_new_M:
                best_new_M = new_M
                best_t = t
            if new_M <= M * 0.99:
                break

        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        f = f * (S / f.sum())

        if t_iter % 10 == 0:
            print(f"  LP iter {t_iter}: obj={cur_obj:.8f}, best={best_obj:.8f}, "
                  f"t={best_t:.4f}, act={len(active_set)}, time={elapsed:.0f}s")

    return best_f, best_obj


def subgradient_refine(f_init, max_time, top_k=30, lr_decay=0.005):
    """Subgradient method with top-k weighting for final polish."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    lr_init = 0.5 * S / n

    f = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
    fs = f.sum()
    if fs < 1e-15:
        f = np.ones(n)
        fs = n
    f = f * (S / fs)

    best_f = f.copy()
    best_M = fft_convolve_self(f).max()
    best_obj = 2.0 * n * best_M / (S * S)

    start = time.monotonic()
    t = 0

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(f)
        M = c.max()
        cur_obj = 2.0 * n * M / (S * S)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()

        # Compute subgradient from top-k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-15:
            break
        weights = top_vals / total

        grad = np.zeros(n, dtype=np.float64)
        for w, k in zip(weights, top_idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                grad[lo:hi] += w * 2.0 * f[k - hi + 1:k - lo + 1][::-1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        lr = lr_init / (1.0 + lr_decay * t)
        f = f - lr * grad / gnorm
        f = np.maximum(f, 0.0)
        f = f * (S / max(f.sum(), 1e-15))

        if t % 1000 == 0:
            print(f"  subgrad iter {t}: obj={cur_obj:.8f}, best={best_obj:.8f}")

    return best_f, best_obj


def mirror_refine(f_init, max_time, top_k=20, eta0=0.3):
    """Mirror descent (multiplicative weights) for refinement."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)

    f = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
    fs = f.sum()
    if fs < 1e-15:
        f = np.ones(n)
        fs = n
    f = f * (S / fs)

    best_f = f.copy()
    best_M = fft_convolve_self(f).max()
    best_obj = 2.0 * n * best_M / (S * S)

    start = time.monotonic()
    t = 0

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(f)
        M = c.max()
        cur_obj = 2.0 * n * M / (S * S)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()

        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-15:
            break
        weights = top_vals / total

        grad = np.zeros(n, dtype=np.float64)
        for w, k in zip(weights, top_idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                grad[lo:hi] += w * 2.0 * f[k - hi + 1:k - lo + 1][::-1]

        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        eta = eta0 / (1.0 + 0.003 * t)
        f_new = f * np.exp(-eta * grad / gmax)
        f_new = np.maximum(f_new, 0.0)
        ns = f_new.sum()
        if ns < 1e-15:
            f = np.full(n, S / n)
        else:
            f = f_new * (S / ns)

        if t % 1000 == 0:
            print(f"  mirror iter {t}: obj={cur_obj:.8f}, best={best_obj:.8f}")

    return best_f, best_obj
```
