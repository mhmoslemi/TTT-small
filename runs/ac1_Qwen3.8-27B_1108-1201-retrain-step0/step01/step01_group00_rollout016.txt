        elif r < 0.6:
            width = max(1, int(rng.integers(1, max(2, n // 12))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.75:
            for _ in range(2):
                width = max(1, int(rng.integers(1, max(2, n // 20))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, size=width))
        elif r < 0.85:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = 0.02 * a[i]
                candidate[i] -= delta
                candidate[j] += delta
        else:
            sym = 0.5 * (a + a[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.005, size=n))

        candidate = np.clip(candidate, 0, None)
        cs = np.sum(candidate)
        if cs < 1e-6:
            candidate = a.copy()
        else:
            candidate = candidate / cs

        new_score = local_score(candidate)
        if new_score < cur_score:
            a = candidate
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        elif rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-15)):
            a = candidate
            cur_score = new_score
        temp *= cooling

        if iters % 2000 == 0:
            print(f"        LS iter={iters}, score={cur_score:.8f}, best={best_score:.8f}")

    return best, best_score


def init_uniform(n):
    return np.ones(n) / n


def init_edge_peak(n, beta, rng):
    t = np.arange(n, dtype=np.float64)
    dist = np.minimum(t, n - 1 - t)
    a = np.exp(-beta * (dist / n) ** 2)
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / a.sum()


def init_chirp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * (t - 1) / n)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / a.sum()


def init_multitone(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in [1, 2, 3, 5, 8]:
        amp = 0.2 / np.sqrt(k)
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / a.sum()


def init_gaussian_center(n, beta, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-beta * ((t - center) / n) ** 2)
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_global = time.monotonic()
    hard_deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000.0)
        s = local_score(start_seq)
        print(f"Provided: n={len(start_seq)}, score={s:.8f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    print(f"\n=== Phase 1: Projected gradient optimization ===")
    configs = [
        (128, 3), (192, 2), (256, 4), (384, 3), (512, 2), (768, 2), (1024, 1),
    ]

    for n, n_inits in configs:
        if time.monotonic() > hard_deadline - 120:
            break
        remaining = hard_deadline - time.monotonic()
        time_per = remaining / (n_inits + 1.0)
        print(f"\n  n={n}, {n_inits} inits, ~{time_per:.1f}s each")

        inits = []
        inits.append(('uniform', init_uniform(n)))
        inits.append(('edge', init_edge_peak(n, 2.0 + rng.random() * 3.0, rng)))
        inits.append(('chirp', init_chirp(n, 0.4 + 0.4 * rng.random(), rng)))
        inits.append(('multitone', init_multitone(n, rng)))
        inits.append(('gauss', init_gaussian_center(n, 2.0 + rng.random() * 4.0, rng)))

        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000.0)
            if len(start_seq) != n:
                x_old = np.linspace(0, 1, len(start_seq))
                x_new = np.linspace(0, 1, n)
                resampled = np.interp(x_new, x_old, start_seq)
            else:
                resampled = start_seq.copy()
            resampled = project_simplex(resampled)
            inits.append(('provided', resampled))

        inits = inits[:n_inits + 1]

        for name, a0 in inits:
            if time.monotonic() > hard_deadline - 60:
                break
            t0 = time.monotonic()
            t_limit = min(time_per, hard_deadline - time.monotonic() - 30)
            if t_limit < 5:
                break
            print(f"    init='{name}', time={t_limit:.1f}s")
            a_best, score = optimize_simplex(a0, t_limit, rng)
            elapsed = time.monotonic() - t0
            print(f"    -> score={score:.8f} ({elapsed:.1f}s)")
            if score < best_score:
                best_score = score
                best_seq = a_best.copy()
                print(f"    *** NEW BEST: {best_score:.8f} (n={len(a_best)}) ***")

    print(f"\n=== Phase 2: Local search polish ===")
    if best_seq is not None:
        polish_time = min(300.0, max(30.0, hard_deadline - time.monotonic() - 30))
        print(f"  Starting polish ({polish_time:.0f}s), current best={best_score:.8f}")
        polished, p_score = local_search(best_seq, polish_time, rng)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.8f}")

    print(f"\n=== Phase 3: Additional exploration ===")
    if time.monotonic() < hard_deadline - 60:
        remaining = hard_deadline - time.monotonic() - 30
        extra_configs = [
            (320, 2), (448, 2), (640, 1), (896, 1),
        ]
        for n, n_inits in extra_configs:
            if time.monotonic() > hard_deadline - 40:
                break
            for ini in range(n_inits):
                if time.monotonic() > hard_deadline - 30:
                    break
                r = rng.random()
                if r < 0.5:
                    a0 = init_edge_peak(n, 1.0 + 4.0 * rng.random(), rng)
                else:
                    a0 = init_chirp(n, 0.3 + 0.5 * rng.random(), rng)
                t_limit = min(remaining / 3.0, hard_deadline - time.monotonic() - 20)
                if t_limit < 5:
                    break
                a_best, score = optimize_simplex(a0, t_limit, rng)
                print(f"  extra n={n} ini={ini}: score={score:.8f}")
                if score < best_score:
                    best_score = score
                    best_seq = a_best.copy()
                    print(f"  *** NEW BEST: {best_score:.8f} (n={n}) ***")

    print(f"\n=== Final refinement on best ===")
    if best_seq is not None and time.monotonic() < hard_deadline - 20:
        remaining = hard_deadline - time.monotonic() - 10
        if remaining > 20:
            a_best, score = optimize_simplex(best_seq, remaining, rng,
                                             tau_start=2.0 / len(best_seq),
                                             tau_end=0.02 / len(best_seq),
                                             base_k=5)
            if score < best_score:
                best_score = score
                best_seq = a_best.copy()
                print(f"  After final refinement: {best_score:.8f}")

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.8f}, n={len(final_list)} ===")
        return final_list

    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, float(x)))) for x in height_sequence_1]
    return [1.0] * 256
