"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-based iterative method.
The key idea: given current sequence f, solve an LP to find the "largest"
sequence g that, when convolved with f, doesn't exceed the current peak M.
Mix f and g iteratively to reduce the peak.
Combined with diverse initialization and local refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def solve_lp(f, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns g (the optimal solution).
    """
    n = len(f)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix A: A[k, i] = f[k-i] if 0 <= k-i < n
    # We use a sparse-friendly approach: build A row by row
    # For scipy linprog with method='highs', we can use sparse or dense
    # For n=128, dense is fine (255 x 128 = 32640 entries)
    
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            # A[k, j] = f[k - j]
            A[k, j] = f[k - j]
    
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    b = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        # Fallback: return f itself (which is always feasible)
        return f.copy()


def lp_optimize(f, time_limit, t_init=0.3, t_decay=0.999):
    """
    Iteratively apply LP-based updates to minimize max(conv(f,f)).
    f should be normalized to sum=1.
    """
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(len(f)) / len(f)
    else:
        f /= s
    
    n = len(f)
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    t0 = time.monotonic()
    t = t_init
    no_improve = 0
    max_no_improve = 30
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < 1e-15:
            break
        
        # Solve LP
        g = solve_lp(f, M)
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        
        if g_sum < 1e-12:
            # LP didn't find anything useful
            t *= t_decay
            no_improve += 1
            if no_improve >= max_no_improve:
                break
            continue
        
        # Normalize g to sum = 1
        g /= g_sum
        
        # Mix
        new_f = (1.0 - t) * f + t * g
        
        # Check improvement
        new_conv = fast_convolve(new_f)
        new_M = new_conv.max()
        
        if new_M < M * 0.999 or new_M < best_obj:
            f = new_f
            if new_M < best_obj:
                best_obj = new_M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Try smaller step
            no_improve += 1
        
        if no_improve >= max_no_improve:
            t *= 0.5
            no_improve = 0
            if t < 0.001:
                break
        
        # Decay step size slowly
        t = max(0.01, t * t_decay)
    
    return best_f, best_obj


def local_refine(a, time_limit):
    """
    Local refinement using small random perturbations and coordinate-wise adjustments.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a_sum
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    step_size = 0.01
    
    while time.monotonic() - t0 < time_limit:
        # Random perturbation
        n_trials = 100
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Random direction on the simplex
            delta = rng.standard_normal(n) * step_size
            # Project back to simplex (maintain non-negativity and sum=1)
            new_a = a + delta
            new_a = np.maximum(new_a, 0.0)
            ns = new_a.sum()
            if ns < 1e-12:
                continue
            new_a /= ns
            
            obj = compute_objective(new_a)
            if obj < best_obj:
                best_obj = obj
                best_a = new_a.copy()
                a = new_a.copy()
                step_size = max(0.001, step_size * 1.05)
            else:
                step_size = max(0.001, step_size * 0.95)
    
    return best_a, best_obj

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj_normalized(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def solve_lp_fast(f, M):
    """Solve LP: max 1^T g s.t. F g <= M*1, g >= 0."""
    n = len(f)
    num_constraints = 2 * n - 1
    
    # Build sparse constraint matrix
    # Row k has nonzeros at columns max(0,k-n+1) .. min(n-1, k)
    # A[k, j] = f[k-j]
    
    rows = []
    cols = []
    data = []
    
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            val = f[k - j]
            if val > 1e-15:  # Skip near-zeros
                rows.append(k)
                cols.append(j)
                data.append(val)
    
    A_sparse = csr_matrix((data, (rows, cols)), shape=(num_constraints, n))
    
    c = -np.ones(n)
    b = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_sparse, b_ub=b, bounds=bounds, method='highs')
    
    if result.success:
        return np.maximum(result.x, 0)
    return f.copy()


def mirror_descent(a0, time_limit, t_start=30.0, eta=0.5):
    """Improved multiplicative mirror descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a_sum
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t = t_start
    patience = 0
    max_patience = 50
    t0 = time.monotonic()
    lp_counter = 0
    
    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Soft-max weights
        scaled = t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Subgradient
        sg = np.zeros(n)
        thresh = w.max() * 0.005
        active_idx = np.where(w > thresh)[0]
        
        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end+1] += w[k] * a[k-j_end:k-j_start+1][::-1]
        
        # Multiplicative update
        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.ones(n) / n
        else:
            a /= a_sum
        
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-9:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            # LP correction every 50 steps of no improvement
            lp_counter += 1
            if lp_counter % 3 == 0 and n <= 512:
                M = conv.max()
                g = solve_lp_fast(a, M)
                g_sum = g.sum()
                if g_sum > 1e-12:
                    g /= g_sum
                    a = 0.7 * a + 0.3 * g
                    conv = fast_convolve(a)
                    current_obj = 2.0 * n * conv.max()
                    if current_obj < best_obj:
                        best_obj = current_obj
                        best_a = a.copy()
            
            patience = 0
            eta *= 0.7
            if eta < 0.01:
                t *= 2.0
                t = min(t, 300.0)
                eta = 0.3
            else:
                t = min(t * 1.3, 200.0)
    
    return best_a, best_obj


def local_search(a0, time_limit):
    """Local search with adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    a = a / a_sum if a_sum > 1e-12 else np.ones(n) / n
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    step = 0.02
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Generate random perturbation
            idx = rng.integers(0, n, size=min(10, n))
            delta = rng.standard_normal(len(idx)) * step
            new_a = a.copy()
            for i, j in enumerate(idx):
                new_a[j] += delta[i]
            new_a = np.maximum(new_a, 0)
            ns = new_a.sum()
            if ns < 1e-12:
                continue
            new_a /= ns
            
            obj = compute_obj_normalized(new_a)
            if obj < best_obj:
                best_obj = obj
                best_a = new_a.copy()
                a = new_a.copy()
                step = min(0.1, step * 1.1)
                improved = True
        
        if not improved:
            step *= 0.7
            if step < 0.001:
                break
    
    return best_a, best_obj

Actually, here's another idea: **the "peel" method**. 

Observation: The convolution (a*a) has a triangular support (length 2n-1). The peaks are typically near the center (lag n-1). If we can "peel" the mass from the center of the sequence to the edges, we might reduce the center peak of the convolution.

More concretely: if a[j] is large in the middle, then (a*a)[n-1] = sum a[j]*a[n-1-j] is large. By moving mass from the center to the edges, we reduce this sum.

But moving mass to the edges increases (a*a)[0] = a[0]^2 and (a*a)[2n-2] = a[n-1]^2.

The optimal balance is where all the "peak" lags have approximately equal values.

This suggests a "water-filling" approach: repeatedly find the lag with the maximum convolution value, and redistribute mass to reduce it.

Let me implement this as an additional method:
