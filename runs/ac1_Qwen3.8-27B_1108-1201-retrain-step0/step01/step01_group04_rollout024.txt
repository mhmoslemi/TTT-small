except Exception:
    from numpy.fft import fft, ifft
    def next_fast_len(n):

        m = 1
        while m < n:
            m <<= 1
        return m

LP_NMAX = 800


def _fft_len(n, factor=2):
    return int(next_fast_len(max(1, factor * int(n))))


def conv_sq(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(n, 2)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v.copy()
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def gradient_smoothed_max(p, T_fill):
    """Compute gradient of T*logsumexp(c/T) where c = p*p (full convolution)."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.zeros(0), 0.0, 0.0

    L = _fft_len(n, 3)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]

    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.zeros(n), 0.0, 0.0

    T = max(float(T_fill) * cmax, 1e-300)
    diff = (c - cmax) / T
    e = np.exp(diff)
    raw = float(e.sum())
    if not np.isfinite(raw) or raw <= 0.0:
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        raw = 1.0

    w = e / raw
    # Compute gradient: grad[i] = 2 * sum_k w[k] * p[k-i] (valid range)
    # Using FFT cross-correlation trick
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))

    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2:n - 2:-1]

    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, G.size)
        tmp[:m] = np.asarray(G)[:m]
        G = tmp

    G = np.maximum(G, 0.0)
    g = 2.0 * G
    surr = float(cmax + T * np.log(raw))
    return g, cmax, surr


def shape_from_params(theta, n):
    """
    Build positive vector from log-polynomial with asymmetric tapers.
    theta = [c0, c1, ..., cD, alpha_L, alpha_R]
    p(x) = exp(sum_{d=0}^D c_d * x^d) * x^alpha_L * (1-x)^alpha_R
    """
    D = 8
    if len(theta) < D + 3:
        theta = np.concatenate([theta, np.zeros(D + 3 - len(theta))])
    
    x = np.linspace(0.0, 1.0, n)
    xp = 1e-8
    xpm = 1.0 - 1e-8

    logp = np.zeros(n, dtype=np.float64)
    x_pow = np.ones(n, dtype=np.float64)
    for d in range(D + 1):
        logp += theta[d] * x_pow
        x_pow = x_pow * x

    alpha_L = theta[D + 1]
    alpha_R = theta[D + 2]
    alpha_L = float(np.clip(alpha_L, 0.01, 8.0))
    alpha_R = float(np.clip(alpha_R, 0.01, 8.0))

    logp += alpha_L * np.log(xp + 1e-12)
    logp += alpha_R * np.log(xpm + 1e-12)

    if not np.all(np.isfinite(logp)):
        return np.full(n, 1.0 / n)

    m = float(np.max(logp))
    logp = logp - m
    p = np.exp(logp)
    return safe_normalize(p)


def make_shape_from_theta(theta, n):
    return shape_from_params(np.asarray(theta, dtype=np.float64), n)


def shape_objective(theta, n, x_cache=None):

def make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def make_raised_cosine(n, power=1.0):
    x = np.linspace(0.0, 1.0, n)
    p = np.sin(np.pi * x) ** float(power)
    return safe_normalize(p)


def make_beta(n, a, b):
    x = np.linspace(1e-6, 1.0 - 1e-6, n)
    p = np.power(x, a) * np.power(1.0 - x, b)
    return safe_normalize(np.maximum(p, 1e-30))


def make_poly_bump(n, coeffs):
    x = np.linspace(0.0, 1.0, n)
    p = np.ones(n)
    for i, c in enumerate(coeffs):
        p += c * x ** (i + 1)
    p = np.maximum(p, 0.01)
    return safe_normalize(p)


def make_asym_bump(n, peak_frac, left_exp, right_exp):
    """Asymmetric bump peaking at peak_frac of the domain."""
    x = np.linspace(0.0, 1.0, n)
    if peak_frac <= 0.5:
        left = x / peak_frac
        right = (1.0 - x) / (1.0 - peak_frac)
        p = np.power(np.minimum(left, 1.0), left_exp) * np.power(np.minimum(right, 1.0), right_exp)
    else:
        left = x / peak_frac
        right = (1.0 - x) / (1.0 - peak_frac)
        p = np.power(np.minimum(left, 1.0), left_exp) * np.power(np.minimum(right, 1.0), right_exp)
    return safe_normalize(np.maximum(p, 1e-30))


def generate_initials(n, rng):
    out = []
    out.append(make_uniform(n))

                 (1.3, 0.7), (2.0, 3.0), (3.0, 2.0), (0.5, 2.5), (2.5, 0.5)]:

    for pf in (0.3, 0.4, 0.5, 0.6, 0.7):

                out.append(make_asym_bump(n, pf, le, re))
    # Random smooth
    for _ in range(20):
        x = np.linspace(0.0, 1.0, n)
        p = 1.0
        for f in (1, 2, 3, 5):
            p += rng.uniform(0.0, 0.4) * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        p = np.maximum(p, 0.02)
        out.append(safe_normalize(p))
    # Pure random
    for _ in range(5):
        out.append(safe_normalize(rng.uniform(0.1, 1.0, n)))
    return out


def lp_best_response(p, budget_s):
    """Solve LP: max sum(g) s.t. a*g <= M, g >= 0, where M = max(a*a)."""
    if not _HAVE_LP:
        return None
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 0 or n > LP_NMAX or budget_s <= 0.5:
        return None

    p = safe_normalize(p)
    c = conv_sq(p)
    M = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(M) or M <= 0.0:
        return None

    try:
        mid = np.arange(n, dtype=np.int32)
        rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel().astype(np.int32)
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(p, n).astype(np.float64)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                      method="highs", options={"disp": False})
    except Exception:
        return None

    if res is None or not hasattr(res, 'x') or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-12:
        return None
    return g / s


def adam_optimize(p0, time_limit, seed=42, use_lp=False):
    """
    Adam-based optimization on the probability simplex with adaptive temperature.
    """
    rng = np.random.default_rng(seed)
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    if n == 0:
        return p0, np.inf

    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score

    end = time.monotonic() + time_limit

    # Adam state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1 = 0.7
    beta2 = 0.95
    eps_adam = 1e-8

    # Temperature schedule
    T0 = 0.15
    T_min = 0.001
    T_decay = 0.9995

    # Learning rate schedule
    lr0 = 1.0 / np.sqrt(n) * 0.5
    lr_decay = 0.999

    # LP on first call
    if use_lp and n <= LP_NMAX:
        lp_budget = min(20.0, time_limit * 0.1)
        g = lp_best_response(p, lp_budget)
        if g is not None:
            for mix in (1.0, 0.7, 0.5, 0.3):
                q = project_simplex((1 - mix) * p + mix * g)
                sc = internal_score(q)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_p = q.copy()
                    current_score = sc
                    p = q.copy()

    iter_count = 0
    no_improve = 0

    while time.monotonic() < end:
        iter_count += 1
        T = max(T_min, T0 * (T_decay ** iter_count))
        lr = lr0 * (lr_decay ** (iter_count // 50))

        g, cmax, surr = gradient_smoothed_max(p, T)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        p_new = p - step
        p_new = project_simplex(p_new)

        sc_new = internal_score(p_new)
        if np.isfinite(sc_new) and sc_new < current_score + 0.001 * T * cmax:
            p = p_new
            current_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Periodic LP
        if use_lp and n <= LP_NMAX and iter_count % 300 == 0:
            remaining = end - time.monotonic()
            if remaining > 15.0:
                g = lp_best_response(p, min(20.0, remaining * 0.3))
                if g is not None:
                    for mix in (1.0, 0.6, 0.4, 0.2):
                        q = project_simplex((1 - mix) * p + mix * g)
                        sc = internal_score(q)
                        if np.isfinite(sc) and sc < best_score:
                            best_score = sc
                            best_p = q.copy()
                            p = q.copy()
                            current_score = sc
                            no_improve = 0

        # Random kicks to escape local minima
        if no_improve > 500:
            kick_scale = min(0.3, 0.05 * no_improve / 500.0) / np.sqrt(n)
            p = project_simplex(p + rng.normal(0, kick_scale, n))
            current_score = internal_score(p)
            no_improve = 0
            m = np.zeros(n)
            v = np.zeros(n)

        # Reheat occasionally
        if no_improve > 2000:
            p = project_simplex(0.5 * best_p + 0.5 * p + rng.normal(0, 0.1 / np.sqrt(n), n))
            current_score = internal_score(p)
            no_improve = 0
            m = np.zeros(n)
            v = np.zeros(n)

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def shape_search(n, rng, budget_s):
    """Optimize shape parameters using Adam on the shape space."""
    D = 6
    dim = D + 3
    best_theta = None
    best_score = np.inf
    best_p = None

    end = time.monotonic() + budget_s

    # Multiple starts
    starts = []
    starts.append(np.zeros(dim))
    t = np.zeros(dim)
    t[D + 1] = 1.5
    t[D + 2] = 1.5
    starts.append(t)
    t = np.zeros(dim)
    t[D + 1] = 2.0
    t[D + 2] = 1.0
    starts.append(t)
    t = np.zeros(dim)
    t[D + 1] = 1.0
    t[D + 2] = 2.0
    starts.append(t)
    t = np.zeros(dim)
    t[0] = 0.5
    t[1] = -1.0
    t[D + 1] = 1.5
    t[D + 2] = 1.5
    starts.append(t)
    for _ in range(5):
        t = rng.normal(0, 0.8, dim)
        t[D + 1] = abs(t[D + 1]) + 0.5
        t[D + 2] = abs(t[D + 2]) + 0.5
        starts.append(t)

    for x0 in starts:
        if time.monotonic() > end - 5:
            break
        sc = shape_objective(x0, n)
        if sc < best_score:
            best_score = sc
            best_theta = x0.copy()
            best_p = shape_from_params(x0, n)

    # Adam on shape parameters
    for x0 in starts[:4]:
        if time.monotonic() > end - 10:
            break
        theta = x0.copy().astype(np.float64)
        m = np.zeros(dim)
        v = np.zeros(dim)
        t_adam = 0
        lr_s = 0.1
        cur_sc = shape_objective(theta, n)
        no_improve = 0
        step_limit = max(10, int(budget_s / len(starts[:4]) / 0.05))

        for _ in range(step_limit):
            if time.monotonic() > end - 5:
                break
            # Central difference gradient (cheap for small dim)
            eps = 0.01
            grad = np.zeros(dim)
            for j in range(dim):
                tp = theta.copy()
                tp[j] += eps
                tm = theta.copy()
                tm[j] -= eps
                grad[j] = (shape_objective(tp, n) - shape_objective(tm, n)) / (2 * eps)

            t_adam += 1
            m = 0.8 * m + 0.2 * grad
            v = 0.9 * v + 0.1 * grad * grad
            m_hat = m / (1 - 0.8 ** t_adam)
            v_hat = v / (1 - 0.9 ** t_adam)
            theta_new = theta - lr_s * m_hat / (np.sqrt(v_hat) + 1e-8)
            theta_new[D + 1] = float(np.clip(theta_new[D + 1], 0.01, 8.0))
            theta_new[D + 2] = float(np.clip(theta_new[D + 2], 0.01, 8.0))

            sc_new = shape_objective(theta_new, n)
            if sc_new < cur_sc + 1e-8:
                theta = theta_new
                cur_sc = sc_new
                no_improve = 0
                if sc_new < best_score:
                    best_score = sc_new
                    best_theta = theta.copy()
                    best_p = shape_from_params(theta, n)
            else:
                no_improve += 1
                if no_improve > 100:
                    break

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf, best_theta

    # Phase 2: medium resolution

def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a nonnegative step function minimizing 2*n*max(a*a)/(sum a)^2.
    Uses multi-resolution Adam optimization with shape parameterization and LP finishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    # Try to use parent sequence
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and float(parent_arr.sum()) > 0:
            parent = safe_normalize(parent_arr)
            print("Using parent sequence of length %d, score=%.6f" % (
                parent.size, internal_score(parent)), flush=True)
    except Exception:
        parent = None

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = safe_normalize(np.asarray(p, dtype=np.float64))
        if p.size == 0:
            return
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("  New best: %.8f (n=%d)" % (best_score, p.size), flush=True)

    # --- Track 1: Multi-resolution at medium length ---
    n_med = 1000
    budget_t1 = (deadline - time.monotonic()) * 0.40
    if budget_t1 > 60:
        print("Track 1: Multi-resolution search at n=%d, budget=%.0fs" % (n_med, budget_t1), flush=True)
        p1, sc1 = multi_resolution_search(rng, budget_t1, n_med, parent)
        try_record(p1, sc1)

    # --- Track 2: Shape parameter optimization ---
    n_shape = 800
    budget_t2 = (deadline - time.monotonic()) * 0.25
    if budget_t2 > 30:
        print("Track 2: Shape parameter search at n=%d, budget=%.0fs" % (n_shape, budget_t2), flush=True)
        ps, scs, theta = shape_search(n_shape, rng, budget_t2)
        try_record(ps, scs)
        # Upsample shape result to longer length and optimize
        if ps is not None and (deadline - time.monotonic()) > 60:
            n_long = 2000
            p_up = safe_normalize(np.interp(np.linspace(0, 1, n_long),
                                            np.linspace(0, 1, n_shape), ps))
            pb, scb = adam_optimize(p_up, min(80.0, deadline - time.monotonic() - 20),
                                    seed=int(rng.integers(0, 10**6)))
            try_record(pb, scb)

    # --- Track 3: Direct optimization with diverse starts ---
    n_direct = 700
    budget_t3 = max(60.0, (deadline - time.monotonic()) * 0.25)
    if budget_t3 > 30:
        print("Track 3: Direct optimization at n=%d, budget=%.0fs" % (n_direct, budget_t3), flush=True)
        inits = generate_initials(n_direct, rng)
        if parent is not None and parent.size != n_direct:
            inits.insert(0, safe_normalize(np.interp(np.linspace(0, 1, n_direct),
                                                     np.linspace(0, 1, parent.size), parent)))
        scored = sorted([(internal_score(safe_normalize(p)), safe_normalize(p)) for p in inits])
        scored = [(s, p) for s, p in scored if np.isfinite(s)][:8]
        per_start = max(5.0, budget_t3 / len(scored))
        for i, (sc, p) in enumerate(scored):
            if time.monotonic() > deadline - 30:
                break
            limit = min(per_start, deadline - time.monotonic() - 30)
            if limit < 3:
                break
            pb, scb = adam_optimize(p, limit, seed=int(rng.integers(0, 10**6)) + i,
                                    use_lp=True)
            try_record(pb, scb)

    # --- Track 4: LP-focused short sequence ---
    n_lp = 600
    budget_t4 = max(30.0, (deadline - time.monotonic()) * 0.15)
    if budget_t4 > 30 and _HAVE_LP:
        print("Track 4: LP-assisted at n=%d, budget=%.0fs" % (n_lp, budget_t4), flush=True)
        inits = generate_initials(n_lp, rng)[:8]
        scored = sorted([(internal_score(safe_normalize(p)), safe_normalize(p)) for p in inits])
        scored = [(s, p) for s, p in scored if np.isfinite(s)][:4]
        for i, (sc, p) in enumerate(scored):
            if time.monotonic() > deadline - 20:
                break
            limit = min(budget_t4 / len(scored), deadline - time.monotonic() - 20)
            if limit < 5:
                break
            pb, scb = adam_optimize(p, limit, seed=int(rng.integers(0, 10**6)) + 100 + i,
                                    use_lp=True)
            try_record(pb, scb)

    # --- Final polish ---
    if best_p is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        print("Final polish: %.0fs remaining, current best=%.8f" % (remaining, best_score), flush=True)
        n_polish = len(best_p)
        pb, scb = adam_optimize(best_p, remaining, seed=int(rng.integers(0, 10**6)) + 999,
                                use_lp=(n_polish <= LP_NMAX))
        try_record(pb, scb)

    if best_p is None:
        best_p = make_uniform(700)
        best_score = internal_score(best_p)

    # Ensure contract
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s <= 1e-12:
        best_p = make_uniform(700)

    elapsed = time.monotonic() - start_time
    print("Done in %.1fs. Final score=%.8f, length=%d" % (elapsed, best_score, len(best_p)), flush=True)
    return [float(x) for x in best_p]
