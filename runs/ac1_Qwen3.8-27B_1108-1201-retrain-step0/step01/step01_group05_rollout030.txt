```python
        (16384, 'sawtooth', 1),  # Longer sequence
    ]
    
    for length, init_type, n_restarts in configs:
        if time.monotonic() > deadline - 30:
            break
        
        per_restart_time = min(120, (deadline - time.monotonic()) / max(1, n_restarts * remaining_configs))
        
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            
            # Create initialization
            a = make_initialization(init_type, length, rng, r)
            
            # Optimize
            a_opt, val = optimize(a, per_restart_time, rng)
            
            if val < best_val:
                best_val = val
                best_seq = a_opt.copy()
                print(f"New best: {val:.6f} (len={length}, init={init_type}, restart={r})")
    
    # Polish
    if best_seq is not None and time.monotonic() < deadline - 5:
        budget_left = deadline - time.monotonic()
        best_seq = polish(best_seq, budget_left, rng)
        best_val = evaluate_sequence(best_seq.tolist())
        print(f"After polish: {best_val:.6f}")
    
    return [float(x) for x in best_seq]
```

```python
        tau = min(tau * 1.001, tau_min)  # Gradually decrease
```

```python
"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
  - Normalize to sum = sqrt(2n) so objective = max(self-convolution)
  - Use softmax-smoothed gradient computed via FFT cross-correlation
  - Heavy-ball momentum with simplex projection
  - Multiple structured initializations (sawtooth, cosine, edge-enhanced, given)
  - Adaptive temperature, learning rate, and random escape from local minima
  - Final polishing with targeted top-lag gradient
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_c = (cumsum - S) / idx
    valid = theta_c <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_fft_gradient(a, b, tau, n):
    """Gradient of softmax-smoothed max(b) w.r.t. a via FFT cross-correlation.
    
    softmax gradient direction: p_k = exp(tau*b_k)/sum(exp(tau*b_k))
    grad_m = 2 * sum_k p_{k+m} * a_k  (cross-correlation)
    """
    b_shift = b - b.max()
    weights = np.exp(tau * b_shift)
    wsum = weights.sum()
    if wsum <= 0:
        return np.zeros(n)
    weights /= wsum

    # Need N >= (2n-1) + (n-1) = 3n-2 for linear cross-correlation
    N = 1 << (3 * n - 2).bit_length()
    P = fft(weights, N)
    A = fft(a, N)
    cross = np.real(ifft(P * np.conj(A)))
    return 2.0 * cross[:n]


def compute_toplevel_gradient(a, b, n, top_k=16):
    """Gradient using top-k lags only (for polishing phase)."""
    top_indices = np.argsort(b)[::-1][:top_k]
    top_vals = b[top_indices] - b[top_indices].min()
    wsum = top_vals.sum()
    if wsum <= 0:
        return np.zeros(n)
    weights = top_vals / wsum

    grad = np.zeros(n, dtype=np.float64)
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[k - hi:k - lo + 1]
            grad[lo:hi + 1] += w_val * 2.0 * seg[::-1]
    return grad


def make_sawtooth(n, alpha, rng):
    """Linear ramp: large on one side, small on the other."""
    t = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2.0 * t - 1.0)
    a = np.clip(a, 0.01, None)
    return a


def make_cosine(n, freq, alpha, rng):
    """Cosine-modulated sequence."""
    t = np.linspace(0, 1, n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * t)
    a = np.clip(a, 0.01, None)
    return a


def make_edge_enhanced(n, beta, rng):
    """Enhance edges: large at j=0 and j=n-1, small in middle."""
    t = np.linspace(0, 1, n)
    dist = 1.0 - 2.0 * np.abs(t - 0.5)  # 0 at center, 1 at edges
    a = 1.0 + beta * dist
    a = np.clip(a, 0.01, None)
    return a


def make_chirp(n, rng):
    """Quadratic phase chirp, made non-negative by adding offset."""
    t = np.arange(n, dtype=np.float64)
    phase = 2.0 * np.pi * t * t / (2.0 * n)
    a = 1.0 + 0.5 * np.cos(phase)
    a = np.clip(a, 0.01, None)
    return a


def get_initializations(n, rng, given_seq=None):
    """Generate diverse initializations for length n."""
    inits = []

    if given_seq is not None:
        g = np.asarray(given_seq, dtype=np.float64).flatten()
        if len(g) == n and g.sum() > 0:
            inits.append(("given", np.clip(g, 0, 1000)))
        elif len(g) > 0:
            # Resample to length n
            from numpy import interp
            old_t = np.linspace(0, 1, len(g))
            new_t = np.linspace(0, 1, n)
            resized = interp(new_t, old_t, np.clip(g, 0, 1000))
            inits.append(("given_resized", resized))

    # Sawtooth with various alphas
    for alpha in [0.5, 0.8, 0.95]:
        inits.append(("sawtooth_a%.2f" % alpha, make_sawtooth(n, alpha, rng)))

    # Cosine with various frequencies
    for freq, alpha in [(1, 0.5), (1, 0.8), (2, 0.4), (3, 0.3)]:
        inits.append(("cos_f%d_a1" % (freq * 10 + int(alpha * 10)), make_cosine(n, freq, alpha, rng)))

    # Edge-enhanced
    for beta in [0.5, 1.0, 1.5]:
        inits.append(("edge_b%.1f" % beta, make_edge_enhanced(n, beta, rng)))

    # Chirp
    inits.append(("chirp", make_chirp(n, rng)))

    # Random
    inits.append(("random", rng.random(n) + 0.01))

    return inits


def optimize_sequence(a_init, max_time, rng, use_fft_grad=True):
    """Main optimization: minimize max(self-convolution) with sum constraint."""
    n = len(a_init)
    S = np.sqrt(2.0 * float(n))
    a = project_simplex(a_init, S)

    # Parameters
    tau = 8.0 / S
    tau_max = 80.0 / S
    mu = 0.85
    eta = 0.3 * S / np.sqrt(n)
    eta_min = 1e-6 * S / n
    patience_base = 300
    esc = 0

    b = fft_convolve_self(a)
    best_val = b.max()
    best_a = a.copy()

    v = np.zeros(n, dtype=np.float64)
    start = time.monotonic()
    t = 0
    last_improve = 0
    stall = 0

    while time.monotonic() - start < max_time:
        t += 1

        # Compute gradient
        if use_fft_grad:
            grad = compute_fft_gradient(a, b, tau, n)
        else:
            grad = compute_toplevel_gradient(a, b, n, top_k=16)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        # Momentum
        v = mu * v + (1.0 - mu) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-14:
            v = grad
            vnorm = gnorm
        v = v / vnorm

        # Update and project
        a = project_simplex(a - eta * v, S)

        # Evaluate
        b = fft_convolve_self(a)
        cur_val = b.max()

        if cur_val < best_val * (1.0 - 1e-7):
            best_val = cur_val
            best_a = a.copy()
            last_improve = t
            stall = 0

        stall = t - last_improve

        # Escape from local minimum
        if stall > patience_base + 100 * esc:
            sigma = (0.05 + 0.05 * esc) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, sigma, n), S)
            v = np.zeros(n, dtype=np.float64)
            tau = max(tau / 2.0, 8.0 / S)
            eta = max(eta * 2.0, 0.1 * S / np.sqrt(n))
            esc = min(esc + 1, 5)
            last_improve = t
            b = fft_convolve_self(a)

        # Adaptive parameters
        tau = min(tau * 1.0005, tau_max)
        if stall > patience_base // 2:
            eta = max(eta * 0.98, eta_min)

    obj = float(2.0 * n * best_val / S**2)
    return best_a, obj


def polish_sequence(a_init, max_time, rng):
    """Final polishing with top-lag gradient and very small steps."""
    n = len(a_init)
    S = np.sqrt(2.0 * float(n))
    a = project_simplex(a_init, S)

    eta = 0.05 * S / np.sqrt(n)
    mu = 0.8
    tau = 100.0 / S  # Very sharp (near-argmax)

    b = fft_convolve_self(a)
    best_val = b.max()
    best_a = a.copy()

    v = np.zeros(n, dtype=np.float64)
    start = time.monotonic()
    t = 0

    while time.monotonic() - start < max_time:
        t += 1

        grad = compute_toplevel_gradient(a, b, n, top_k=8)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        v = mu * v + (1.0 - mu) * grad
        vnorm = np.linalg.norm(v)
        if vnorm > 1e-14:
            v = v / vnorm
        else:
            v = grad

        a = project_simplex(a - eta * v, S)
        b = fft_convolve_self(a)
        cur_val = b.max()

        if cur_val < best_val * (1.0 - 1e-8):
            best_val = cur_val
            best_a = a.copy()

        eta *= 0.9995

    obj = float(2.0 * n * best_val / S**2)
    return best_a, obj
```
