def object_function(a, tau=1.0):
    """
    Smooth surrogate object function to approximate the original evaluation.
    It uses the log-sum-exp trick to smooth the maximum value of the convolution.
    The score is computed as: 2*n*smooth_max_value / (sum_a)**2.
    """
    n = len(a)
    if n == 0:
        return np.inf
    
    try:
        conv_result = np.convolve(a, a, mode='full')
        conv_result = conv_result.astype(np.float64)
        # Create a smooth estimate of the max value
        # Using τ-slider to control the level of smoothing
        smoothed_max = np.log(np.sum(np.exp(-conv_result / tau))) * (-tau)
        # Smoothed version of smooth_max: if we want tightness, we might need a larger tau.
        # But to be an upper bound on the true max, we can use τ in a sensitive range.
        # We'll compute the numerator part: 2*n*smoothed_max
        S = np.sum(a)
        # S must be at least 0.01 -> otherwise, guarded against below 0.01
        if S < 0.01:
            return np.inf
        score = 2.0 * n * smoothed_max / (S**2)
    except Exception as e:
        print(f"ConvolutionError {n}: {str(e)}")
        return np.inf
    
    return score

            # So for the purpose of this function, we only return the optimized sequence if score_opt is low.
            if score_opt < final_score:
                print("Optimization did not improve score?")
                # Fallback to provided parent
                sequence = global_sequence.copy()
                
    except:
        # Apply uniform fallback or random initialization with smaller n.
        n_initial = 2000
        print(f"Returning fallback uniform sequence of length{n_initial}")
        sequence = np.ones(n_initial) * (np.sqrt(2*n_initial) / n_initial)
        sequence = simplex_proj(sequence, np.sqrt(2*n_initial))
        score_opt = 2.0 * n_initial * object_function(sequence, 1.0)
        
    # Now, we must also check that the sequence is not negative and sum is at least 0.01.
    if len(sequence) == 0 or not np.all(sequence >= 0) or max(sequence) > 1000:
        # Generate a new candidate
        n_initial = 2400
        sequence = np.ones(n_initial) * (np.sqrt(2*n_initial) / n_initial)
        sequence = simplex_proj(sequence, np.sqrt(2*n_initial))
        score_opt = object_function(sequence, 1.0) * 2.0 * n_initial
        
    return sequence[:2400]  # Limit to 2400 if necessary
