"""
Optimized step function design using soft-max gradient descent with Adam,
peak-shaving local search, and multi-scale exploration.
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Key improvements: larger n, better initializations (U-shaped, geometric, chirp),
faster convergence with adaptive temperature, and targeted peak reduction.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score computation (scale-invariant, assumes a >= 0)."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def soft_max_convolve(a, tau):
    """Compute soft-max of convolve(a,a) and its gradient w.r.t. a."""
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    value = c_max + tau * np.log(sum_exp)
    # Gradient: d(value)/da[i] = 2 * sum_k w[k] * a[k-i] (valid indices)
    # = 2 * (convolve(a_rev, w))[i + n - 1]
    n = len(a)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    grad = 2.0 * corr[n - 1: 2 * n - 1]
    return value, grad, c


def make_init_type(n, init_type, rng):
    """Generate various types of initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'uniform':
        a = np.ones(n)
    elif init_type == 'chirp':
        alpha = 2.0 + 3.0 * rng.random()
        a = 1.0 + 0.6 * np.cos(alpha * t * t / n)
        a = np.clip(a, 0.01, None)
    elif init_type == 'u_shape':
        # Large at ends, small in middle
        center = (n - 1) / 2.0
        a = 1.0 + (t - center) ** 2 / (n * n)
        a = np.clip(a, 0.01, None)
    elif init_type == 'geometric':
        # Geometric sequence (highly asymmetric)
        r = 0.95 + 0.04 * rng.random()
        a = r ** t
        a = a / np.mean(a)
    elif init_type == 'geometric_strong':
        r = 0.85 + 0.05 * rng.random()
        a = r ** t
        a = a / np.mean(a)
    elif init_type == 'gaussian':
        center = (n - 1) / 2.0
        width = n * (0.2 + 0.3 * rng.random())
        a = np.exp(-((t - center) ** 2) / (2 * width ** 2))
        a = 0.1 + a
    elif init_type == 'two_lobe':
        c1 = n * 0.2
        c2 = n * 0.8
        w = n * 0.15
        a = np.exp(-((t - c1) ** 2) / (2 * w ** 2)) + np.exp(-((t - c2) ** 2) / (2 * w ** 2))
        a = 0.05 + a
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for k in [1, 2, 3, 5, 7, 11]:
            amp = 0.3 / k
            a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
        a = np.clip(a, 0.01, None)
    elif init_type == 'lorentzian':
        center = (n - 1) / 2.0
        a = 1.0 / (1.0 + (t - center) ** 2 / (n * 0.1))
        a = 0.1 + a
    elif init_type == 'random':
        a = 1.0 + 0.4 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    elif init_type == 'random_sparse':
        a = 0.1 + np.abs(rng.standard_normal(n))
    else:
        a = np.ones(n)
    return a


def optimize_gradient(n, a0, time_limit, rng, tau_start=None, tau_end=None):
    """
    Adam gradient descent on log-softmax parameterization.
    Minimizes soft-max of convolve(a,a) with annealed temperature.
    a is constrained to sum to sqrt(2n) for numerical stability.
    """
    if tau_start is None:
        tau_start = 20.0 / n
    if tau_end is None:
        tau_end = 0.5 / n

    S = np.sqrt(2.0 * n)
    a0 = np.clip(a0, 1e-6, None)
    theta = np.log(a0 / np.sum(a0) * n).astype(np.float64)

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0
    t0 = time.monotonic()

    best_score = score_fast(softmax_normalize(theta) * S)
    best_theta = theta.copy()
    plateau = 0
    best_plateau_score = best_score

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)

        a = softmax_normalize(theta) * S
        value, g_a, c = soft_max_convolve(a, tau)

        # Gradient w.r.t. theta (accounting for softmax constraint)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta ** 2 + 1e-12)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)

        # Adaptive learning rate
        lr = 1.5 / (1.0 + 0.1 * step ** 0.5)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # Periodic score check
        if step % 100 == 0 or step < 10:
            a_check = softmax_normalize(theta) * S
            sc = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (S * S)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()
                plateau = 0
            else:
                plateau += 1
                if plateau > 2000:
                    # Reset with perturbation
                    theta = theta + 0.01 * rng.standard_normal(n)
                    plateau = 0

        # Occasional print
        if step % 5000 == 0:
            a_check = softmax_normalize(theta) * S
            sc = score_fast(a_check)
            print(f"      step={step}, score={sc:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")

    a_best = softmax_normalize(best_theta) * S
    return a_best, score_fast(a_best)


def peak_shaving(a_current, time_limit, rng):
    """
    Local search that specifically targets the top peaks in the convolution.
    Uses multiplicative updates to reduce peak contributions.
    """
    current = a_current.astype(np.float64).copy()
    n = len(current)
    current_score = score_fast(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()

    # Track peaks
    conv = fft_convolve(current, current)
    num_peaks = max(3, n // 32)

    iteration = 0
    eta = 0.02  # Step size

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        conv = fft_convolve(current, current)
        max_c = np.max(conv)
        current_score = 2.0 * n * max_c / np.sum(current) ** 2

        # Find top peak positions
        peak_indices = np.argsort(conv)[-num_peaks:]

        # Build reduction vector: for each peak, compute its gradient contribution
        reduction = np.zeros(n)
        for pk in peak_indices:
            # Gradient of c[pk] w.r.t. a[i] is 2*a[pk-i] for valid i
            for i in range(max(0, pk - n + 1), min(pk, n - 1) + 1):
                reduction[i] += current[i] * current[pk - i]

        # Also track which positions would increase other peaks (to avoid)
        # Simple approach: normalize reduction by max to keep step size controlled
        max_red = np.max(reduction) if np.max(reduction) > 0 else 1.0
        reduction = reduction / max_red

        # Apply multiplicative update: reduce where reduction is high
        # But also add some random exploration
        noise = rng.standard_normal(n) * 0.005
        mult = np.exp(-eta * reduction + noise)
        candidate = current * mult

        # Ensure non-negativity and reasonable range
        candidate = np.clip(candidate, 1e-6, 1000.0)

        # Renormalize to keep same sum (score is scale-invariant)
        target_sum = np.sum(current)
        candidate = candidate * (target_sum / np.sum(candidate))

        new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate)) / target_sum ** 2

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                # Slowly decrease step size
                eta = max(0.001, eta * 0.999)
        else:
            # Occasionally accept bad moves (simulated annealing)
            if rng.random() < np.exp(-(new_score - current_score) * n / 50.0):
                current = candidate
                current_score = new_score
            eta = max(0.001, eta * 0.995)

        # Every 500 iterations, try a more aggressive perturbation
        if iteration % 500 == 0:
            candidate = current.copy()
            # Randomly boost some positions that have low peak contribution
            low_contrib = np.argsort(reduction)[:n // 8]
            candidate[low_contrib] *= np.exp(rng.uniform(0.01, 0.05, size=len(low_contrib)))
            candidate = np.clip(candidate, 1e-6, 1000.0)
            candidate = candidate * (target_sum / np.sum(candidate))
            cs = 2.0 * n * np.max(fft_convolve(candidate, candidate)) / target_sum ** 2
            if cs < best_score:
                current = candidate
                current_score = cs
                best_score = cs
                best = current.copy()

        if iteration % 2000 == 0:
            print(f"      peak_shave iter={iteration}, score={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.1f}s")

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for optimal non-negative step function sequence.
    Phase 1: Gradient optimization across multiple n and initializations.
    Phase 2: Peak-shaving local search.
    Phase 3: Additional exploration if time permits.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided starting sequence
    start_seq = None
    if 'height_sequence_1' in dir() or 'height_sequence_1' in globals():
        try:
            hs = globals().get('height_sequence_1', None)
            if hs is not None and len(hs) > 0:
                start_seq = np.clip(np.asarray(hs, dtype=np.float64), 0.01, 1000.0)
                s = score_fast(start_seq)
                print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
                if s < best_score:
                    best_score = s
                    best_seq = start_seq.copy()
        except Exception:
            pass

    # Define search configurations: (n, list_of_init_types, time_fraction)
    configs = [
        (512, ['geometric', 'chirp', 'u_shape', 'multi_tone', 'random'], 0.15),
        (512, ['geometric_strong', 'two_lobe', 'lorentzian', 'gaussian'], 0.12),
        (1024, ['geometric', 'chirp', 'u_shape', 'random'], 0.15),
        (1024, ['multi_tone', 'two_lobe', 'geometric_strong', 'lorentzian'], 0.12),
        (256, ['geometric', 'u_shape', 'chirp', 'random'], 0.08),
        (2048, ['geometric', 'chirp', 'u_shape'], 0.15),
    ]

    # If we have a provided sequence, add it as a config
    if start_seq is not None:
        n_prov = len(start_seq)
        if n_prov in [256, 384, 512, 768]:
            configs.insert(0, (n_prov, ['provided'], 0.08))

    print(f"\n=== Phase 1: Gradient Optimization ===")
    total_time = time_budget * 0.6  # 60% for gradient optimization

    for n, init_types, time_frac in configs:
        if time.monotonic() > deadline - 120:
            break
        time_for_n = total_time * time_frac
        t_start = time.monotonic()

        for idx, init_type in enumerate(init_types):
            if time.monotonic() - t_start > time_for_n:
                break
            if time.monotonic() > deadline - 60:
                break

            time_remaining = max(5.0, (t_start + time_for_n) - time.monotonic())

            if init_type == 'provided' and start_seq is not None:
                if len(start_seq) == n:
                    a0 = start_seq.copy()
                else:
                    x_old = np.linspace(0, 1, len(start_seq))
                    x_new = np.linspace(0, 1, n)
                    a0 = np.interp(x_new, x_old, start_seq)
            else:
                a0 = make_init_type(n, init_type, rng)

            sc0 = score_fast(a0)
            print(f"    n={n}, init='{init_type}', initial_score={sc0:.6f}")

            a_best, sc_best = optimize_gradient(n, a0, time_remaining, rng)
            elapsed = time.monotonic() - t_start
            print(f"      -> optimized: score={sc_best:.6f} (t={elapsed:.1f}s)")

            if sc_best < best_score:
                best_score = sc_best
                best_seq = a_best.copy()
                print(f"      *** NEW BEST: {best_score:.6f} (n={n}) ***")

    print(f"\n  Best after Phase 1: {best_score:.6f}")

    # Phase 2: Peak shaving
    print(f"\n=== Phase 2: Peak Shaving ===")
    if best_seq is not None:
        polish_time = min(600.0, max(30.0, (deadline - time.monotonic()) * 0.5))
        print(f"  Starting peak shaving: score={best_score:.6f}, time={polish_time:.0f}s")
        polished, polish_score = peak_shaving(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After peak shaving: {best_score:.6f}")
        else:
            print(f"  Peak shaving didn't improve: {polish_score:.6f} vs {best_score:.6f}")

    # Phase 3: Additional exploration with different random seeds
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\n=== Phase 3: Additional Exploration ===")
        remaining = deadline - time.monotonic()
        n_try = len(best_seq)

        for trial in range(3):
            if time.monotonic() > deadline - 30:
                break
            # Perturb the best sequence and re-optimize
            a0 = best_seq * np.exp(rng.normal(0, 0.1, n_try))
            a0 = np.clip(a0, 1e-4, None)
            t_per = max(20.0, remaining / (3 - trial) - 30)
            rng2 = np.random.default_rng(seed + 1000 + trial)
            a_opt, sc_opt = optimize_gradient(n_try, a0, min(t_per, 200.0), rng2)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  Trial {trial}: NEW BEST {best_score:.6f}")
            else:
                print(f"  Trial {trial}: score={sc_opt:.6f}")

        # One more peak shaving pass
        if time.monotonic() < deadline - 20:
            t_pe = min(120.0, deadline - time.monotonic() - 15)
            if t_pe > 10:
                polished2, ps2 = peak_shaving(best_seq, t_pe, rng)
                if ps2 < best_score:
                    best_score = ps2
                    best_seq = polished2.copy()
                    print(f"  Final polish: {best_score:.6f}")

    # Final verification
    print(f"\n=== FINAL RESULT ===")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"  n={len(final_list)}, evaluate_sequence score={final_score:.6f}")
        print(f"  local score={score_fast(best_seq):.6f}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
