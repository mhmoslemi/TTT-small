def solve_lp(p, M, n):
    """
    Solve: max sum(q) s.t. q >= 0, (p*q)[k] <= M for all k
    Returns optimal q.
    """
    # Variables: q_0, ..., q_{n-1}
    # Objective: max sum(q) => min -sum(q)
    c = -np.ones(n)
    
    # Constraints: (p*q)[k] <= M for k = 0, ..., 2n-2
    # (p*q)[k] = sum_j p_j * q_{k-j} (where q_i = 0 if i out of [0,n-1])

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using iterative LP-based optimization.
At each step, solves an LP to find the best "direction" q that keeps the
cross-convolution bounded, then blends with current iterate.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def build_conv_indices(n):
    """Pre-compute (row, col) pairs and p-index for convolution constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        count = i_end - i_start
        rows.extend([k] * count)
        cols.extend(range(i_start, i_end))
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    p_idx = rows - cols  # index into p for each nonzero
    return rows, cols, p_idx


def solve_direction_lp(p, M, rows, cols, p_idx, n):
    """
    Solve: max sum(q) s.t. q >= 0, (p*q)[k] <= M for all k.
    Returns optimal q and optimal value.
    """
    c = -np.ones(n)  # minimize -sum(q)
    
    # Build sparse constraint matrix
    A = csr_matrix((p[p_idx], (rows, cols)), shape=(2*n-1, n))
    b = M * np.ones(2*n-1)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x, -res.fun
    return None, 0


def run_lp_iteration(p, M, rows, cols, p_idx, n, t_step=1.0):
    """One iteration of the LP-based method."""
    q, q_sum = solve_direction_lp(p, M, rows, cols, p_idx, n)
    if q is None or q_sum <= 1.0 + 1e-9:
        return p, q_sum, False  # No improvement
    
    q_norm = q / q_sum
    p_new = (1 - t_step) * p + t_step * q_norm
    p_new = np.maximum(p_new, 0)
    p_new = p_new / p_new.sum()
    return p_new, q_sum, True

LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses the "direction finding" LP at each step to find the most promising
update direction, combined with gradient-based polishing.

import numpy as np
from scipy.optimize import linprog

def build_conv_indices(n):
    rows = np.empty(n*n, dtype=np.int32)
    cols = np.empty(n*n, dtype=np.int32)
    idx = 0
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        cnt = i_end - i_start
        rows[idx:idx+cnt] = k
        cols[idx:idx+cnt] = np.arange(i_start, i_end)
        idx += cnt
    p_idx = rows - cols
    return rows, cols, p_idx


def solve_lp(p, M, rows, cols, p_idx, n):
    c = -np.ones(n)
    A = csr_matrix((p[p_idx], (rows, cols)), shape=(2*n-1, n))
    b = M * np.ones(2*n-1)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        return res.x, -res.fun
    return None, 0.0


def compute_score_from_p(p, n):
    c = np.convolve(p, p)
    return 2.0 * n * c.max()


def project_simplex(a, S):

        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    idx = res.argmax()
    res[idx] += S - res.sum()
    return res

"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses direction-finding LP at each step to find the most promising update,
combined with gradient-based polishing and adaptive step sizes.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def build_conv_indices(n):
    """Pre-compute (row, col) pairs and p-index for the convolution constraint matrix."""
    total_nz = n * n
    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    idx = 0
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        cnt = i_end - i_start
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = np.arange(i_start, i_end, dtype=np.int32)
        idx += cnt
    p_idx = rows - cols
    return rows, cols, p_idx


def solve_direction_lp(p, M, rows, cols, p_idx, n, time_limit=60):
    """
    Solve: max sum(q) s.t. q >= 0, (p*q)[k] <= M for all k = 0,...,2n-2.
    Returns (optimal_q, optimal_sum) or (None, 0.0).
    """
    c_obj = -np.ones(n)
    A = csr_matrix((p[p_idx], (rows, cols)), shape=(2 * n - 1, n))
    b = M * np.ones(2 * n - 1)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'time_limit': time_limit})
    if res.status == 0:
        return res.x, -res.fun
    elif res.status == 3:  # time limit
        if res.x is not None:
            s = res.x.sum()
            if s > 1.0:
                return res.x, s
    return None, 0.0


def compute_score_from_p(p, n):
    """Compute 2*n*max(conv(p,p)) for probability vector p (sum=1)."""
    c = np.convolve(p, p)
    return 2.0 * n * c.max()


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    s = a.sum()
    if s <= S + 1e-15:
        return a + (S - s) / n
    lo, hi = 0.0, a.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    idx = res.argmax()
    res[idx] += S - res.sum()
    return res


def soft_max_grad(p, tau, n):
    """
    Compute gradient of tau*log(sum_k exp(c_k/tau)) where c = conv(p,p).
    Returns (gradient, convolution).
    """
    c = np.convolve(p, p)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    # grad_j = 2 * sum_{m} q_{m+j} * a_m = 2 * correlate(q, p, 'valid')[j]
    grad = 2.0 * np.correlate(q, p, 'valid')
    return grad, c


def run_gradient_polish(p0, n, t_budget, tau_start=0.5, tau_end=0.01):
    """Projected gradient descent with log-sum-exp smoothing and tau annealing."""
    a = p0.copy()
    best = a.copy()
    best_score = compute_score_from_p(best, n)

    t0 = time.monotonic()
    lr = 0.05
    step = 0
    last_score = best_score

    while time.monotonic() - t0 < t_budget:
        f = min(1.0, (time.monotonic() - t0) / max(t_budget, 1e-6))
        tau = tau_start * (tau_end / tau_start) ** f

        grad, c = soft_max_grad(a, tau, n)
        a_new = a - lr * grad
        a_new = project_simplex(a_new, 1.0)
        step += 1

        if step % 10 == 0:
            sc = compute_score_from_p(a_new, n)
            if sc < last_score - 1e-12:
                a = a_new
                if sc < best_score:
                    best_score = sc
                    best = a.copy()
                if lr < 1.0:
                    lr *= 1.08
            else:
                lr = max(0.005, lr * 0.6)
                a = a_new
            last_score = sc
        else:
            a = a_new

    return best, best_score


def init_chirp_prob(n, seed=0):
    """Chirp-based probability vector initialization."""
    t = np.arange(n, dtype=np.float64) - (n - 1) * 0.5
    a = 1.0 + 0.9 * np.cos(np.pi * t * t / n)
    rng = np.random.default_rng(seed)
    a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones(n) / n
    return a / s


def init_random_prob(n, seed=0):
    """Random exponential probability vector."""
    a = np.random.default_rng(seed).exponential(1.0, size=n)
    return a / a.sum()


def run_lp_phase(p0, n, rows, cols, p_idx, end_time, verbose=True):
    """Run LP-based iterative optimization until end_time."""
    p = p0.copy()
    best_p = p.copy()
    best_score = compute_score_from_p(p, n)
    t_step = 1.0
    stall = 0
    it = 0
    last_improve = best_score

    while time.monotonic() < end_time - 5:
        c = np.convolve(p, p)
        M = c.max()
        q, q_sum = solve_direction_lp(p, M, rows, cols, p_idx, n)
        it += 1

        if q is not None and q_sum > 1.0 + 1e-8:
            q_norm = q / q_sum
            p_new = (1.0 - t_step) * p + t_step * q_norm
            p_new = np.maximum(p_new, 0.0)
            s = p_new.sum()
            if s > 1e-12:
                p_new = p_new / s
            else:
                p_new = p.copy()
            p = p_new
            stall = 0
        else:
            stall += 1
            if stall > 15:
                # Perturb to escape local optimum
                rng = np.random.default_rng(it * 7 + 3)
                p = p + 0.003 * rng.standard_normal(n)
                p = np.maximum(p, 0.0)
                p = p / p.sum()
                t_step = max(0.3, t_step * 0.7)
                stall = 0

        sc = compute_score_from_p(p, n)
        if sc < best_score - 1e-10:
            best_score = sc
            best_p = p.copy()
            last_improve = best_score
        elif sc > last_improve + 1e-4:
            # Regress: pull back toward best
            p = 0.5 * p + 0.5 * best_p
            t_step = max(0.2, t_step * 0.8)

        if verbose and it % 10 == 0:
            rem = end_time - time.monotonic()
            print(f"    LP iter {it}: score={sc:.6f}, best={best_score:.6f}, "
                  f"q_sum={q_sum:.4f}, t={t_step:.2f}, rem={rem:.0f}s")

        # Convergence check
        if it > 30 and best_score > last_improve - 1e-9 and stall > 30:
            if verbose:
                print(f"    LP phase converging at iter {it}, score={best_score:.6f}")
            break

    return best_p, best_score, it
