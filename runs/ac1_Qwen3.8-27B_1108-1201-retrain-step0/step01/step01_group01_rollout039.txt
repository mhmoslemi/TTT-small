"""
Mirror descent with adaptive softmax temperature for minimizing
the normalized autocorrelation peak of a non-negative sequence.
"""

import numpy as np
import time

def _score_normalized(a):
    """Score for a sequence with sum=1: 2*n*max(conv(a,a))."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * float(np.max(b))

def _grad_at_peaks(a, b, top_k, temp):
    """
    Compute gradient using softmax-weighted top peaks.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Softmax over top_k peaks
    peak_idx = np.argpartition(b, -top_k)[-top_k:]
    b_top = b[peak_idx]
    w = np.exp(temp * (b_top - np.max(b_top)))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for ki in range(top_k):
        wt = w[ki]
        if wt < 1e-10:
            continue
        k = peak_idx[ki]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += wt * a[k - idx]
    grad *= 2.0
    return grad

def _mirror_descent(a, deadline, rng, top_k=50, temp=100.0, eta=0.1, cool_rate=0.95, min_eta=1e-10):
    """
    Mirror descent (multiplicative weight update) to minimize max(b) subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    
    best = a.copy()
    best_score = _score_normalized(best)
    no_improve = 0
    temp_cur = temp
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            eta *= cool_rate
            temp_cur *= 0.9
            no_improve = 0
            if eta < min_eta:
                break
        
        # Compute gradient
        grad = _grad_at_peaks(a, b, top_k, temp_cur)
        
        # Center the gradient (for sum constraint)
        grad -= np.mean(grad)
        
        # Mirror descent update: a_i *= exp(-eta * grad_i)
        # Scale eta by 1/max(|grad|) for stability
        g_max = np.max(np.abs(grad))
        if g_max < 1e-15:
            break
        a = a * np.exp(-eta * grad / g_max)
        
        # Normalize (this handles the sum constraint in mirror descent)
        S = np.sum(a)
        if S < 1e-10:
            a = best.copy()
            break
        a /= S
        
        # Ensure non-negativity (exp is always positive, but numerical issues)
        a = np.maximum(a, 1e-30)
        a /= np.sum(a)
    
    return best, best_score

    # Let j = k-i, so i = k-j: sum_{j: 0<=k-j<n} f[k-j] * g[j] <= M
