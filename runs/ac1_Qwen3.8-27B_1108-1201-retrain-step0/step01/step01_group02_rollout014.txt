"""
Combined LP + IFT + local search optimization for minimizing peak auto-convolution.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a, target_sum=None):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    if target_sum is not None:
        return a * target_sum / s
    return a / s


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def lp_step(a, M, timeout=30.0):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=300, verbose=True):
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    a = np.maximum(0, a)
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"  LP start: score={best_score:.6f}, n={n}")
    
    stagnant = 0
    for it in range(2000):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and it < 20:
                print(f"  LP iter {it}: score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 30:
            break
        
        remaining = time_limit_s - elapsed
        if remaining < 3.0:

def ift_refine(a, time_limit_s=100, verbose=True):

    n_fft = 1
    while n_fft < 4*n:
        n_fft *= 2
    
    for it in range(500):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break

        power = np.abs(A) ** 2
        
        # Target: flat power spectrum (proportional to total energy)
        target_power = np.full_like(power, power.mean())
        
        # Blend: keep some of the current power, add some flat
        blended_power = 0.7 * power + 0.3 * target_power
        blended_mag = np.sqrt(blended_power)

        phase = np.angle(A)
        A_new = blended_mag * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, n=n_fft)[:n]
        a_new = np.maximum(0, a_new)
        
        if a_new.sum() < 1e-10:
            break
        
        # Normalize to match original sum
        a_new = a_new * a.sum() / a_new.sum()
        
        score = score_sequence(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose and it < 20:
                print(f"  IFT iter {it}: score={score:.6f}")
        else:
            # Try with less blending
            blended_power2 = 0.9 * power + 0.1 * target_power
            blended_mag2 = np.sqrt(blended_power2)

            a_new2 = np.fft.irfft(A_new2, n=n_fft)[:n]

                        best_a = a_new2.copy()

def raised_cosine(n):

```python
"""
Multi-phase optimization for minimizing peak auto-convolution of non-negative sequences.
Uses LP-based direction finding, coordinate descent, and local search.
Score: 2*n*max(conv(a,a))/sum(a)^2 (lower is better).
"""
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))

def lp_step(a, M, timeout=20.0):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

def lp_refine(a, time_limit_s=300.0, verbose=True):

    a = np.maximum(0.0, a.copy().astype(np.float64))

    best_score = score_sequence(a)
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")
    stagnant = 0
    it = 0
    for it in range(500):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it < 10 or it % 50 == 0):
                print(f"    LP iter {it}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 25:
            break
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.25))
        g = lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= a.sum() * (1 + 1e-10):
            break
        g = g * a.sum() / sum_g
        # Line search
        best_alpha = 0.1
        best_s = score_sequence(np.maximum(0, (1-0.1)*a + 0.1*g))
        for alpha in [0.2, 0.3, 0.5, 0.7]:
            cand = np.maximum(0, (1-alpha)*a + alpha*g)
            s = score_sequence(cand)
            if s < best_s:
                best_s = s
                best_alpha = alpha
        a = np.maximum(0, (1-best_alpha)*a + best_alpha*g)
    if verbose:
        print(f"    LP done: best={best_score:.6f}, iters={it+1}, t={time.monotonic()-start_time:.1f}s")
    return best_a, best_score
