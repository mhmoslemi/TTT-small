```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = next_fast_len(n + m - 1)
    fa = fft(a, size)
    fb = fft(b, size)
    return ifft(fa * fb, size).real[:n + m - 1]

def local_score(a):
    n = len(a)
    a = np.clip(a, 0, 1000)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (S * S)

def lp_step(f, M, n):
    """Solve LP: maximize sum(b) s.t. conv(f,b) <= M, b >= 0"""
    # A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    # A has shape (2n-1, n)
    A = np.zeros((2*n-1, n))
    # A[k,j] = f[k-j] for valid (k,j) where 0 <= k-j < n
    for j in range(n):
        # k-j ranges from 0 to n-1, so k ranges from j to j+n-1
        # But k must be in [0, 2n-2]
        k_start = j
        k_end = min(j + n, 2*n-1)
        A[k_start:k_end, j] = f[0:k_end-k_start]
    
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = M * np.ones(2*n-1)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, method='highs', options={'presolve': True})
    if res.status != 0:
        return None
    return res.x

def lp_alternating(f_init, n, max_iter, t_lr, rng, time_limit=None):
    """LP-based alternating minimization."""
    S_target = np.sqrt(2.0 * n)
    f = f_init.copy()
    f = f / f.sum() * S_target
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    
    for it in range(max_iter):
        if time_limit and (time.monotonic() - t0) > time_limit:
            break
        
        M = fft_convolve(f, f).max()
        b = lp_step(f, M, n)
        if b is None:
            break
        
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        b = b / b_sum * S_target
        
        # Update
        f_new = (1 - t_lr) * f + t_lr * b
        f_new = np.clip(f_new, 0, None)
        
        # Renormalize
        f_sum = f_new.sum()
        if f_sum < 1e-10:
            break
        f_new = f_new / f_sum * S_target
        
        score = local_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        
        f = f_new
    
    return best_f, best_score
```

```python
def lp_step(f, M, n):
    """Solve LP: maximize sum(b) s.t. conv(f,b) <= M, b >= 0"""
    # A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        k_min = j  # k-j >= 0 => k >= j
        k_max = min(j + n - 1, 2*n-2)  # k-j <= n-1 => k <= j+n-1, and k <= 2n-2
        A[k_min:k_max+1, j] = f[0:k_max-k_min+1]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, method='highs', 
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)
```

```python
"""
LP-based alternating minimization for minimizing the peak of self-convolution
of non-negative sequences, with gradient-based polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len

def fft_convolve(a, b):
    n = len(a)
    size = next_fast_len(n + len(b) - 1)
    return ifft(fft(a, size) * fft(b, size), size).real[:n + len(b) - 1]

def local_score(a):
    a = np.clip(a, 0, 1000)
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (S * S)

def build_toeplitz(f, n):
    """Build the (2n-1) x n matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        end = min(j + n, 2*n-1)
        A[j:end, j] = f[:end-j]
    return A

def lp_solve(f, M, n):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    A = build_toeplitz(f, n)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, method='highs',
                  options={'presolve': True})
    if res.status != 0 or not np.isfinite(res.fun):
        return None
    return np.maximum(res.x, 0)

def lp_alternate(f_init, n, time_limit, rng, verbose=True):
    """LP-based alternating minimization."""
    S = np.sqrt(2.0 * n)
    f = f_init.copy()
    f = np.clip(f, 0, None)
    if f.sum() < 1e-10:
        f = np.ones(n)
    f = f / f.sum() * S
    
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    
    t_lr = 0.5  # step size
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        elapsed = time.monotonic() - t0
        
        M = fft_convolve(f, f).max()
        b = lp_solve(f, M, n)
        if b is None or b.sum() < 1e-10:
            if verbose:
                print(f"    LP failed at iter {iteration}")
            break
        
        b = b / b.sum() * S
        
        # Try the update
        f_new = np.clip((1 - t_lr) * f + t_lr * b, 0, None)
        f_sum = f_new.sum()
        if f_sum < 1e-10:
            break
        f_new = f_new / f_sum * S
        
        score = local_score(f_new)
        
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Increase step size slightly if improving
            t_lr = min(0.8, t_lr * 1.01)
        else:
            f = f_new  # Still update, but track
            no_improve += 1
            if no_improve > 20:
                # Decrease step size
                t_lr = max(0.05, t_lr * 0.8)
                no_improve = 0
            if no_improve > 100:
                # Random perturbation to escape
                f = f * np.exp(rng.normal(0, 0.05, n))
                f = np.clip(f, 0, None)
                f = f / f.sum() * S
                t_lr = 0.3
                no_improve = 0
        
        if verbose and iteration % 50 == 0:
            print(f"    iter={iteration}, M={M:.4f}, score={score:.6f}, best={best_score:.6f}, t_lr={t_lr:.3f}, t={elapsed:.1f}s")
    
    return best_f, best_score

def gradient_polish(a_init, n, time_limit, rng):
    """Gradient-based polish using soft subgradient."""
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / a.sum() * S
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    lr = 0.01
    tau = 5.0 / n
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        tau_cur = tau * (0.1 ** elapsed_frac)  # anneal tau
        
        c = fft_convolve(a, a)
        c_max = c.max()
        c_shift = (c - c_max) / tau_cur
        w = np.exp(c_shift)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient: g[j] = 2 * sum_i a[i] * w[i+j]
        g = np.zeros(n)
        for j in range(n):
            g[j] = 2.0 * np.dot(a, w[j:j+n])
        
        # Scale gradient
        g = g / (S * S)
        
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        a_sum = a_new.sum()
        if a_sum < 1e-10:
            a_new = a.copy()
        a_new = a_new / a_new.sum() * S
        
        score = local_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            # Keep or slightly increase lr
        else:
            a = a_new
            lr *= 0.995
        
        step += 1
        if step % 200 == 0:
            print(f"    gp: step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau_cur:.2e}, lr={lr:.5f}")
        
        if lr < 1e-6:
            break
    
    return best_a, best_score

def make_init(n, init_type, rng):
    """Create initial sequence."""
    if init_type == 'uniform':
        return np.ones(n)
    elif init_type == 'chirp':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * t * t / (4 * n))
        return np.clip(a, 0.01, None)
    elif init_type == 'random':
        return 1.0 + 0.3 * rng.standard_normal(n)
    elif init_type == 'multi_tone':
        t = np.arange(n, dtype=np.float64)
        a = 1.0
        for k, amp in [(1, 0.3), (2, 0.2), (3, 0.1), (5, 0.05)]:
            a += amp * np.cos(2 * np.pi * k * t / n + rng.uniform(0, 2*np.pi))
        return np.clip(a, 0.01, None)
    elif init_type == 'triangle':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 - np.abs(t - n/2) / (n/2)
        return np.clip(a, 0.01, None)
    else:
        return np.ones(n)
```

```python
"""
Hybrid LP-based alternating minimization with L-BFGS soft-max refinement
for minimizing the peak of self-convolution of non-negative step functions.
Uses Toeplitz LP structure for fast direction finding, scipy L-BFGS-B for
smooth refinement, and targeted local search for final polish.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.fft import fft, ifft, next_fast_len


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = next_fast_len(n + m - 1)
    fa = fft(a, size)
    fb = fft(b, size)
    return ifft(fa * fb, size).real[:n + m - 1]


def local_score(a):
    """Compute the evaluation score (matches evaluate_sequence formula)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (S * S)


def build_toeplitz(f, n):
    """Build (2n-1) x n matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = f[:end - j]
    return A


def lp_solve(f, M, n):
    """Solve LP: maximize sum(b) subject to conv(f,b) <= M, b >= 0."""
    A = build_toeplitz(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs',
                  options={'presolve': True, 'time_limit': 5})
    if res.status != 0 or not np.isfinite(res.fun):
        return None
    return np.maximum(res.x, 0)


def lp_alternate(f_init, n, time_limit, rng, verbose=True):
    """LP-based alternating minimization with adaptive step size."""
    S = np.sqrt(2.0 * n)
    f = np.clip(f_init.copy(), 0, None)
    if f.sum() < 1e-10:
        f = np.ones(n)
    f = f / f.sum() * S

    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()

    t_lr = 0.4
    no_improve = 0
    stagnation = 0
    last_print = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0

        c_ff = fft_convolve(f, f)
        M = c_ff.max()

        b = lp_solve(f, M, n)
        if b is None or b.sum() < 1e-10:
            break

        b = b / b.sum() * S

        # Line search over step sizes
        best_step_score = local_score(f)
        best_t = 0.0
        for t_candidate in [t_lr, t_lr * 0.5, t_lr * 1.5, 0.1]:
            f_try = np.clip((1 - t_candidate) * f + t_candidate * b, 0, None)
            fs = f_try.sum()
            if fs < 1e-10:
                continue
            f_try = f_try / fs * S
            s_try = local_score(f_try)
            if s_try < best_step_score:
                best_step_score = s_try
                best_t = t_candidate

        if best_t > 0:
            f = np.clip((1 - best_t) * f + best_t * b, 0, None)
            fs = f.sum()
            if fs < 1e-10:
                break
            f = f / fs * S
            t_lr = min(0.7, best_t * 1.05)
            no_improve = 0
        else:
            stagnation += 1
            t_lr = max(0.05, t_lr * 0.85)
            if stagnation > 30:
                # Perturb to escape
                f = f * np.exp(rng.normal(0, 0.08, n))
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs < 1e-10:
                    break
                f = f / fs * S
                t_lr = 0.4
                stagnation = 0
            elif int(elapsed) - last_print > 30:
                if verbose:
                    print(f"    lp_stag: t={elapsed:.0f}s, M={M:.4f}, best={best_score:.6f}")
                last_print = int(elapsed)
            continue

        score = best_step_score
        if score < best_score:
            best_score = score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

            print(f"    lp: t={elapsed:.0f}s, iter M={M:.4f}, score={score:.6f}, best={best_score:.6f}, t_lr={t_lr:.3f}")
            last_print = int(elapsed)

    return best_f, best_score


def softmax_obj_and_grad(theta, n, tau):
    """Objective and gradient for L-BFGS: minimize tau*logsumexp(conv(a,a)/tau)."""
    S = np.sqrt(2.0 * n)
    t_shift = theta - theta.max()
    p = np.exp(t_shift)
    p_sum = p.sum()
    p = p / p_sum
    a = p * S

    c = fft_convolve(a, a)
    c_max = c.max()
    c_shift = (c - c_max) / tau
    exp_c = np.exp(c_shift)
    w = exp_c / exp_c.sum()

    # value = tau * logsumexp(c/tau) = tau * (c_max/tau + log(sum(exp((c-c_max)/tau))))
    value = c_max + tau * np.log(exp_c.sum())

    # Gradient: g[j] = 2 * sum_i a[i] * w[i+j]
    # Using correlation: g = correlate(w, a, 'full')[:n]
    g = np.correlate(w, a, 'full')[:n] * 2.0

    # grad_theta[k] = (2/S) * p[k] * (g[k] - sum_j p[j]*g[j])
    pg = np.dot(p, g)
    grad_theta = (2.0 / S) * p * (g - pg)

    return value, grad_theta


def l_bfgs_optimize(theta_init, n, time_limit, rng):
    """Run L-BFGS on the soft-max objective with annealed temperature."""
    tau_start = 20.0 / n
    tau_end = 0.5 / n
    t0 = time.monotonic()

    best_theta = theta_init.copy()
    best_val = np.inf
    best_score = np.inf
    tau = tau_start

    # Phase 1: Coarse optimization with large tau
    time_phase1 = time_limit * 0.4
    time_phase2 = time_limit * 0.4
    time_phase3 = time_limit * 0.2

    def obj_grad(theta_cur):
        return softmax_obj_and_grad(theta_cur, n, tau)

    # Phase 1
    print(f"    lbfgs phase1: tau={tau:.4f}, time={time_phase1:.0f}s")
    res = minimize(obj_grad, theta_init, method='L-BFGS-B',
                   options={'maxiter': 5000, 'maxfun': 5000,
                            'gtol': 1e-4, 'iprint': 0})
    elapsed = time.monotonic() - t0
    if time.monotonic() - t0 > time_phase1:
        return best_theta, best_score

    theta_cur = res.x
    S = np.sqrt(2.0 * n)
    p = np.exp(theta_cur - theta_cur.max())
    a = p / p.sum() * S
    score = local_score(a)
    if score < best_score:
        best_score = score
        best_theta = theta_cur.copy()
    print(f"    lbfgs phase1 done: score={score:.6f}, t={elapsed:.1f}s")

    # Phase 2: Medium tau
    tau = tau_start * 0.1
    print(f"    lbfgs phase2: tau={tau:.4f}, time={time_phase2:.0f}s")
    res = minimize(obj_grad, theta_cur, method='L-BFGS-B',
                   options={'maxiter': 5000, 'maxfun': 5000,
                            'gtol': 1e-5, 'iprint': 0})
    if time.monotonic() - t0 > time_phase1 + time_phase2:
        a = a  # keep current
    else:
        theta_cur = res.x
        p = np.exp(theta_cur - theta_cur.max())
        a = p / p.sum() * S
        score = local_score(a)
        if score < best_score:
            best_score = score
            best_theta = theta_cur.copy()
        print(f"    lbfgs phase2 done: score={score:.6f}, t={time.monotonic()-t0:.1f}s")

    # Phase 3: Fine tau
    tau = tau_end
    print(f"    lbfgs phase3: tau={tau:.4f}, time={time_phase3:.0f}s")
    res = minimize(obj_grad, theta_cur, method='L-BFGS-B',
                   options={'maxiter': 10000, 'maxfun': 10000,
                            'gtol': 1e-6, 'iprint': 0})
    theta_cur = res.x
    p = np.exp(theta_cur - theta_cur.max())
    a = p / p.sum() * S
    score = local_score(a)
    if score < best_score:
        best_score = score
        best_theta = theta_cur.copy()
    print(f"    lbfgs phase3 done: score={score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.1f}s")

    return best_theta, best_score


def local_search_polish(a_init, n, time_limit, rng):
    """Targeted local search with peak-reduction mutations."""
    S = np.sqrt(2.0 * n)
    current = a_init.copy()
    current = current / current.sum() * S
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.4:
            # Targeted: reduce peak contributors
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            max_i = min(n, peak_idx + 1)
            for i in range(max_i):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.012, size=len(top_idx)))
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_inc = min(10, len(others))
                inc_idx = rng.choice(others, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.012, size=n_inc))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.75:
            # Symmetrize partially
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.008, size=n))
        elif r < 0.9:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        else:
            # Global small
            candidate = current * np.exp(rng.normal(0.0, 0.008, size=n))

        candidate = np.clip(candidate, 0.001, 1000.0)
        fs = candidate.sum()
        if fs < 1e-10:
            continue
        candidate = candidate / fs * S
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature *= cooling
        iter_count += 1

        if iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ls: iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, t={elapsed:.0f}s")

    return best, best_score


def make_init(n, init_type, rng):
    """Create various initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'uniform':
        return np.ones(n)
    elif init_type == 'chirp':
        a = 1.0 + 0.5 * np.cos(np.pi * t * t / (2.0 * n))
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif init_type == 'random':
        a = 1.0 + 0.4 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif init_type == 'multi_tone':
        a = 1.0
        for k, amp in [(1, 0.3), (2, 0.2), (3, 0.12), (5, 0.06), (8, 0.03)]:
            a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif init_type == 'triangle':
        a = 1.0 - 0.8 * np.abs(t - n / 2) / (n / 2)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif init_type == 'bilinear':
        # Two-segment with different slopes
        a = np.ones(n)
        half = n // 2
        a[:half] = 1.0 + 0.5 * t[:half] / half
        a[half:] = 1.5 - 0.5 * (t[half:] - half) / half
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    else:
        return np.ones(n)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Hybrid search: LP-based alternating minimization + L-BFGS refinement + local search.
    Explores multiple sequence lengths and initializations.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 40  # safety margin
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # === Phase 1: LP-based optimization ===
    print("\n" + "=" * 60)
    print("Phase 1: LP-based alternating minimization")
    print("=" * 60)

    lp_configs = [
        (256, ['provided', 'chirp', 'multi_tone', 'bilinear']),
        (256, ['random', 'triangle']),
        (384, ['chirp', 'multi_tone']),
        (512, ['chirp', 'bilinear']),
    ]

    for n, init_types in lp_configs:
        if time.monotonic() > deadline - 300:
            break
        time_left = deadline - time.monotonic()
        time_per_init = min(200.0, time_left / (len(init_types) + 1))

        for init_type in init_types:
            if time.monotonic() > deadline - 120:
                break

            if init_type == 'provided' and start_seq is not None:
                if len(start_seq) == n:
                    a0 = start_seq.copy()
                else:
                    x_old = np.linspace(0, 1, len(start_seq))
                    x_new = np.linspace(0, 1, n)
                    a0 = np.interp(x_new, x_old, start_seq)
            else:
                a0 = make_init(n, init_type, rng)

            a0 = np.clip(a0, 0.01, 1000.0)
            t_per = min(time_per_init, 250.0)
            print(f"\n  LP n={n}, init='{init_type}', budget={t_per:.0f}s")

            result_a, result_score = lp_alternate(a0, n, t_per, rng, verbose=True)

            if result_score < best_score:
                best_score = result_score
                best_seq = result_a.copy()
                print(f"    *** New best: {best_score:.6f} (n={n}, init={init_type}) ***")

        print(f"\n  Best score after n={n} configs: {best_score:.6f}")

    # === Phase 2: L-BFGS refinement ===
    print("\n" + "=" * 60)
    print("Phase 2: L-BFGS soft-max refinement")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 200:
        polish_time = min(400.0, deadline - time.monotonic() - 100)
        n = len(best_seq)
        print(f"\n  L-BFGS on n={n}, budget={polish_time:.0f}s")

        theta_init = np.log(best_seq / best_seq.sum() + 1e-10)
        theta_best, lbfgs_score = l_bfgs_optimize(theta_init, n, polish_time, rng)

        S = np.sqrt(2.0 * n)
        p = np.exp(theta_best - theta_best.max())
        a_lbfgs = p / p.sum() * S
        actual_score = local_score(a_lbfgs)
        print(f"  L-BFGS result: score={actual_score:.6f}")

        if actual_score < best_score:
            best_score = actual_score
            best_seq = a_lbfgs.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 3: Local search polish ===
    print("\n" + "=" * 60)
    print("Phase 3: Local search polish")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 60:
        ls_time = min(300.0, max(30.0, deadline - time.monotonic() - 30))
        n = len(best_seq)
        print(f"\n  Local search on n={n}, budget={ls_time:.0f}s")

        polished, ls_score = local_search_polish(best_seq, n, ls_time, rng)
        print(f"  Local search result: score={ls_score:.6f}")

        if ls_score < best_score:
            best_score = ls_score
            best_seq = polished.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: Additional LP runs if time permits ===
    if best_seq is not None and time.monotonic() < deadline - 80:
        print("\n" + "=" * 60)
        print("Phase 4: Additional refinement")
        print("=" * 60)
        n = len(best_seq)
        remaining = deadline - time.monotonic()
        n_runs = max(1, int(remaining / 150))

        for run_i in range(min(3, n_runs)):
            if time.monotonic() > deadline - 30:
                break
            # Perturb and re-run LP
            perturbation_scale = 0.03 + 0.02 * run_i
            perturbed = best_seq * np.exp(rng.normal(0, perturbation_scale, n))
            perturbed = np.clip(perturbed, 0.01, 1000.0)
            t_per = min(150.0, remaining / (n_runs - run_i))
            print(f"\n  Extra LP run {run_i+1}, budget={t_per:.0f}s")
            result_a, result_score = lp_alternate(perturbed, n, t_per, rng, verbose=False)
            if result_score < best_score:
                best_score = result_score
                best_seq = result_a.copy()
                print(f"    *** New best: {best_score:.6f} ***")

            # Also try L-BFGS on the perturbed
            if time.monotonic() < deadline - 60:
                theta_p = np.log(perturbed / perturbed.sum() + 1e-10)
                t_lbfgs = min(100.0, deadline - time.monotonic() - 30)
                theta_best2, _ = l_bfgs_optimize(theta_p, n, t_lbfgs, rng)
                S = np.sqrt(2.0 * n)
                p2 = np.exp(theta_best2 - theta_best2.max())
                a2 = p2 / p2.sum() * S
                s2 = local_score(a2)
                if s2 < best_score:
                    best_score = s2
                    best_seq = a2.copy()
                    print(f"    *** New best (L-BFGS): {best_score:.6f} ***")

    # === Final verification ===
    print("\n" + "=" * 60)
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"Total time: {time.monotonic() - t_global:.1f}s")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
