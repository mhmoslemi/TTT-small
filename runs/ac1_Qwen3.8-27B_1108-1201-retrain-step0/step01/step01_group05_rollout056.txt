def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.

Key improvements over previous approach:
1. FFT-based gradient for log-sum-exp smoothing (uses ALL lags, not just top-k)

4. Spectral initialization + multiple starting points
5. Adaptive learning rate based on gradient norm
6. Restart mechanism to escape local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}. O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient_fft(a, weights, n):
    """Compute gradient of log-sum-exp w.r.t. a using FFT cross-correlation.
    
    grad[m] = 2 * sum_j weights[m+j] * a[j]  for j=0,...,n-1
    (where weights[m+j] is 0 if m+j >= 2n-1)
    """
    fsize = 1 << (3 * n - 2 - 1).bit_length()
    a_flip = a[::-1]
    A = fft(a_flip, fsize)
    W = fft(weights, fsize)
    conv = np.real(ifft(A * W))
    out = conv[n - 1:2 * n - 1]
    return 2.0 * out


def topk_subgradient(a, c, top_k):
    """Compute subgradient using top-k active lags (fast, O(k*n))."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    grad = np.zeros(n, dtype=np.float64)
    for k in top_indices:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def run_adam_lse(a, n, S, max_time, rng, tau, lr_scale=0.3):
    """Adam optimizer using LSE-smoothed gradient (all lags via FFT)."""
    a = project_simplex(np.clip(a, 0.0, 1000.0), S)
    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-10
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()

    c = fft_convolve_self(a)
    c_max = c.max()
    tau = max(tau, 1e-6)

    while time.monotonic() - start < max_time:
        t += 1
        # Compute LSE gradient
        shifted = np.clip((c - c_max) / tau, -50, 50)
        exp_shifted = np.exp(shifted)
        sum_exp = exp_shifted.sum()
        weights = exp_shifted / sum_exp

        grad = lse_gradient_fft(a, weights, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = a.copy()
            break
        grad = grad / gnorm

        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * (grad ** 2)
        m_hat = m_buf / (1 - beta1 ** t)
        v_hat = v_buf / (1 - beta2 ** t)
        lr = lr_scale * S / n
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

        # Update c and track best
        if t % 3 == 0 or t < 10:
            c = fft_convolve_self(a)
            c_max = c.max()
            if c_max < best_score:
                best_score = c_max
                best_local = a.copy()

    c = fft_convolve_self(a)
    cur = c.max()
    if cur < best_score:
        best_score = cur
        best_local = a.copy()
    return best_local, best_score

    a = project_simplex(np.clip(a, 0.0, 1000.0), S)

    best_score = c.max()

    while time.monotonic() - start < max_time:

            break

    if cur < best_score:

def run_mirror_descent(a, n, S, max_time, rng, eta_init=0.5, top_k=8):
    """Mirror descent (multiplicative weights) - naturally respects a >= 0."""
    a = project_simplex(np.clip(a, 0.0, 1000.0), S)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    t = 0
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        t += 1
        grad = topk_subgradient(a, c, top_k)
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax

        eta = eta_init / (1.0 + 0.002 * t)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.full(n, S / n)
        else:
            a = a_new * (S / s)

        if t % 10 == 0:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = a.copy()

    c = fft_convolve_self(a)
    cur = c.max()
    if cur < best_score:
        best_score = cur
        best_local = a.copy()
    return best_local, best_score

def spectral_init(n, rng):
    """Spectral initialization: try to make the power spectrum flat."""
    # Target: flat convolution means flat power spectrum (approximately)
    # Start with a smooth, slightly perturbed uniform
    a = np.ones(n, dtype=np.float64)
    # Add structured perturbation
    freqs = np.arange(n)
    for _ in range(5):
        f0 = rng.integers(1, n // 2)
        amp = rng.uniform(0.1, 0.3)
        a += amp * np.sin(2 * np.pi * f0 * freqs / n + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.0)
    S = np.sqrt(2.0 * n)
    a = a * S / a.sum()
    return a


def get_initializations(n, rng):
    """Generate multiple initializations for the given length n."""
    inits = []
    
    # Provided sequence (if available and right length)
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("provided", np.clip(given, 0, 1000)))
        elif given.size > 16:
            # Resize by interpolation
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, len(given))
            x_new = np.linspace(0, 1, n)
            f = interp1d(x_old, np.clip(given, 0, 1000), kind='cubic', fill_value='extrapolate')
            inits.append(("provided_resized", np.maximum(f(x_new), 0)))
    except Exception:
        pass
    
    # Uniform
    inits.append(("uniform", np.ones(n)))
    
    # Spectral
    inits.append(("spectral", spectral_init(n, rng)))
    
    # Random
    inits.append(("random", rng.random(n) + 0.1))
    
    # Smooth (low-freq dominant)
    t = np.linspace(0, 1, n)
    smooth = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.cos(2 * np.pi * 7 * t)
    inits.append(("smooth", np.maximum(smooth, 0.01)))
    
    # Bump (concentrated in middle)
    center = n / 2
    bump = np.exp(-0.5 * ((np.arange(n) - center) / (n / 6)) ** 2)
    inits.append(("bump", bump))
    
    return inits


def optimize_sequence(a_init, n, S, budget, rng):
    """Run the multi-stage optimization on a single initial sequence."""
    a = project_simplex(np.clip(a_init, 0.0, 1000.0), S)
    best_score = np.inf
    best_a = a.copy()
    
    def update_best(a_cand, score_cand):
        nonlocal best_score, best_a
        if score_cand < best_score:
            best_score = score_cand
            best_a = a_cand.copy()
            print(f"    new best: {best_score:.6f}")
    
    # Compute initial score
    c = fft_convolve_self(a)
    init_max = c.max()
    update_best(a, init_max)
    
    # Stage 1: LSE with large tau (global exploration)
    stage_budget = budget * 0.3
    per_tau = stage_budget / 3
    c = fft_convolve_self(a)
    for tau_frac in [0.5, 0.3, 0.15]:
        tau = tau_frac * c.max()
        if time.monotonic() >= start_time + budget - 30:
            break
        a1, s1 = run_adam_lse(a, n, S, per_tau, rng, tau=tau, lr_scale=0.4)
        update_best(a1, s1)
        a = a1
        c = fft_convolve_self(a)
    
    # Stage 2: LSE with small tau (refinement)
    stage_budget = budget * 0.2
    c = fft_convolve_self(a)
    for tau_frac in [0.08, 0.04, 0.02]:
        tau = tau_frac * c.max()
        a2, s2 = run_adam_lse(a, n, S, stage_budget / 3, rng, tau=tau, lr_scale=0.2)
        update_best(a2, s2)
        a = a2
        c = fft_convolve_self(a)
    
    # Stage 3: Top-k subgradient (sharp local optimization)
    stage_budget = budget * 0.2
    a3, s3 = run_adam_topk(a, n, S, stage_budget, rng, top_k=15, lr_scale=0.15)
    update_best(a3, s3)
    a = a3
    
    # Stage 4: Mirror descent
    stage_budget = budget * 0.1
    a4, s4 = run_mirror_descent(a, n, S, stage_budget, rng, eta_init=0.3, top_k=10)
    update_best(a4, s4)
    if s4 < best_score:
        a = a4
    
    # Stage 5: Coordinate descent polish
    stage_budget = max(2.0, budget * 0.2)
    a5, s5 = coordinate_descent_simple(best_a, n, S, stage_budget, rng)
    update_best(a5, s5)
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    global start_time
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20
    rng = np.random.default_rng(seed)
    
    global_best = np.inf
    global_a = None
    
    # Try multiple lengths
    lengths = [8192, 4096, 16384]
    # Budget allocation: give more time to the provided length (8192)
    time_alloc = {8192: 0.5, 4096: 0.25, 16384: 0.25}
    
    for n in lengths:
        if time.monotonic() > deadline - 30:
            break
        n_budget = (deadline - time.monotonic()) * time_alloc.get(n, 0.3)
        n_budget = max(10.0, min(n_budget, 600.0))
        
        S = np.sqrt(2.0 * n)
        inits = get_initializations(n, rng)
        
        print(f"\n=== n={n}, budget={n_budget:.0f}s ===")
        
        # Run optimization on each initialization
        per_init_time = n_budget / max(1, len(inits))
        for name, a_init in inits:
            if time.monotonic() > deadline - 30:
                break
            a_init = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000)
            if a_init.sum() < 1e-10:
                continue
            a_opt, score = optimize_sequence(a_init, n, S, per_init_time, rng)
            if score < global_best:
                global_best = score
                global_a = a_opt.copy()
                # Verify with evaluate_sequence periodically
                ver = evaluate_sequence(np.clip(global_a, 0, 1000).tolist())
                if np.isfinite(ver):
                    print(f"  [{name}] n={n}: lse_score={score:.6f}, verified={ver:.6f}")
                    global_best = min(global_best, ver)
                else:
                    print(f"  [{name}] n={n}: lse_score={score:.6f}, verified=inf")
            else:
                print(f"  [{name}] n={n}: score={score:.6f} (not better)")
        
        # After exploring this length, try to polish the best from this length
        if global_a is not None and len(global_a) == n:
            polish_time = min(60.0, max(5.0, (deadline - time.monotonic()) * 0.1))
            if polish_time > 5:
                pa, ps = run_adam_topk(global_a, n, S, polish_time, rng, top_k=20, lr_scale=0.1)
                ver = evaluate_sequence(np.clip(pa, 0, 1000).tolist())
                if np.isfinite(ver) and ver < global_best:
                    global_best = ver
                    global_a = pa.copy()
                    print(f"  Polish n={n}: verified={ver:.6f}")
    
    # Final polish with remaining time
    if global_a is not None and time.monotonic() < deadline - 10:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            pa, ps = run_adam_topk(global_a, n, S, remaining * 0.5, rng, top_k=25, lr_scale=0.08)
            ver = evaluate_sequence(np.clip(pa, 0, 1000).tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = pa.copy()
                print(f"Final polish: verified={ver:.6f}")
            if time.monotonic() < deadline - 5:
                cd_time = deadline - time.monotonic() - 5
                if cd_time > 5:
                    ca, cs = coordinate_descent_simple(global_a, n, S, cd_time, rng)
                    ver2 = evaluate_sequence(np.clip(ca, 0, 1000).tolist())
                    if np.isfinite(ver2) and ver2 < global_best:
                        global_best = ver2
                        global_a = ca.copy()
                        print(f"Final CD: verified={ver2:.6f}")
    
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
    
    # Final verification
    final_score = evaluate_sequence(global_a.tolist())
    print(f"\nFinal: score={final_score:.6f}, length={len(global_a)}")
    
    return [float(x) for x in global_a]
