def _fft_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def _fft_len_grad(n):
    return int(next_fast_len(max(1, 3 * int(n) - 2)))


def conv_sq(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[: 2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v.copy()
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def compute_grad_and_c(p, T):
    """Compute gradient of smoothed max surrogate and the convolution values."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0), np.empty(0), 0.0

    L = _fft_len_grad(n)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[: 2 * n - 1]

    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.zeros(n), c.copy(), 0.0

    T = max(float(T), 1e-300)
    diff = (c - cmax) / T
    e = np.exp(diff)
    raw = float(e.sum())
    if not np.isfinite(raw) or raw <= 0.0:
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        raw = 1.0

    w = e / raw
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))

    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2 : n - 2 : -1]

    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp

    g = 2.0 * np.maximum(G, 0.0)
    surr_val = cmax + T * np.log(raw)
    return g, c, surr_val

        base_lr = p_scale / gnorm

        accepted = False
        for lr_mult in [0.5, 0.2, 0.08, 0.03, 0.01, 0.003, 0.001]:
            lr = lr_mult * base_lr
            v = p - lr * g
            pt = project_simplex(v)
            sc_t = internal_score(pt)

            if np.isfinite(sc_t) and sc_t < surr_cur + 0.5 * max(1.0, T_factor):
                if sc_t < best_score:
                    best_score = sc_t
                    best_p = pt.copy()
                    since_improve = 0
                else:
                    since_improve += 1
                cur_score = sc_t
                prev_p = p.copy()
                p = pt
                accepted = True
                break

        if not accepted:
            T_factor = max(0.001, T_factor * 0.8)
            since_improve += 1

        if not accepted and since_improve > 50:
            sigmas = [0.1 / n, 0.02 / n, 0.004 / n]
            for sig in sigmas:
                pt = project_simplex(p + rng.normal(0.0, sig, n))
                sc = internal_score(pt)
                if np.isfinite(sc) and sc < min(best_score, cur_score):
                    if sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        since_improve = 0
                    cur_score = sc
                    prev_p = p.copy()
                    p = pt
                    accepted = True
                    break

        if not accepted and n > 2:
            top_k = min(5, 2 * n - 1)
            idx_top = np.argsort(c_cur)[-top_k:]
            max_idx = int(np.argmax(c_cur))
            grad_max = g[:]
            hi = int(np.argmax(grad_max))
            lo = int(np.argmin(grad_max))
            if hi != lo and grad_max[hi] > grad_max[lo]:
                for frac in [0.15, 0.05, 0.015, 0.005]:
                    delta = frac * float(p[hi])
                    if delta <= 0:
                        continue
                    pt = p.copy()
                    pt[hi] -= delta
                    pt[lo] += delta
                    pt = project_simplex(pt)
                    sc = internal_score(pt)
                    if np.isfinite(sc) and sc < min(best_score, cur_score):
                        if sc < best_score:
                            best_score = sc
                            best_p = pt.copy()
                            since_improve = 0
                        cur_score = sc
                        prev_p = p.copy()
                        p = pt
                        accepted = True
                        break

        if not accepted and since_improve > 100:
            active = c_cur > 0.9 * c_cur.max()
            if np.sum(active) < 10:
                active_lags = np.where(active)[0]
                for lag in active_lags[:3]:
                    lag = int(lag)
                    contrib = np.zeros(n)
                    lo_j = max(0, lag - (n - 1))
                    hi_j = min(n - 1, lag)
                    for j in range(lo_j, hi_j + 1):
                        contrib[j] += p[lag - j]
                    if contrib.max() > 0:
                        hi_c = int(np.argmax(contrib))
                        lo_c = int(np.argmin(contrib))
                        if hi_c != lo_c:
                            for frac in [0.1, 0.05, 0.02]:
                                delta = frac * float(p[hi_c] * contrib[hi_c] / contrib.max())
                                if delta <= 0:
                                    continue
                                pt = p.copy()
                                pt[hi_c] -= delta
                                pt[lo_c] += delta
                                pt = project_simplex(pt)
                                sc = internal_score(pt)
                                if np.isfinite(sc) and sc < best_score:
                                    best_score = sc
                                    best_p = pt.copy()
                                    since_improve = 0
                                    cur_score = sc
                                    prev_p = p.copy()
                                    p = pt
                                    break

        if not accepted:
            since_improve += 1
            if since_improve > patience:
                restart_count += 1
                if restart_count < 50:
                    if since_improve > patience * 2:
                        p = project_simplex(0.3 * best_p + 0.7 * rng.dirichlet(np.ones(n) * 0.5))
                    else:
                        p = project_simplex(0.7 * best_p + 0.3 * p + rng.normal(0, 0.5 / n, n))
                    cur_score = internal_score(p)
                    prev_p = p.copy()
                    T_factor = 0.1
                    since_improve = 0
                else:
                    break
            elif since_improve > 30:
                p = project_simplex(0.9 * best_p + 0.1 * p + rng.normal(0, 0.1 / n, n))
                cur_score = internal_score(p)
                prev_p = p.copy()

        if iter_no % 500 == 0:
            T_factor = max(0.001, T_factor * 0.9)

        if iter_no % 2000 == 0:
            print("  %s iter=%d best=%.10f T=%.6f" % (label, iter_no, best_score, T_factor), flush=True)

        if best_score < 1.50 and since_improve > 5000:
            break

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(60.0, float(budget_s)) - 30.0

    best_p_global = None
    best_score_global = np.inf

    def try_record(p, sc):
        nonlocal best_p_global, best_score_global
        if p is None:
            return
        p = safe_normalize(p)
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score_global:
            best_score_global = float(sc)
            best_p_global = p.copy()
            print("  NEW BEST: %.10f (n=%d)" % (sc, p.size), flush=True)

    # Load previous best if available
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and float(parent_arr.sum()) > 0:
            parent = safe_normalize(parent_arr)
    except Exception:
        parent = None

    # Multi-length search
    lengths_to_try = [512, 700, 1024, 1536, 2048, 3072, 4096]
    if parent is not None:
        p_len = int(parent.size)
        lengths_to_try = [p_len] + [x for x in lengths_to_try if x != p_len]

    # Phase 1: Evaluate initial shapes at each length, pick top candidates
    candidates = []

    for n in lengths_to_try:
        if time.monotonic() > deadline - 300:
            break
        inits = generate_initials(n, rng)
        if parent is not None and n == int(parent.size):
            inits.insert(0, parent.copy())

        scored = []
        for p in inits:
            p = safe_normalize(p)
            sc = internal_score(p)
            if np.isfinite(sc):
                scored.append((sc, p))
        scored.sort(key=lambda x: x[0])
        for sc, p in scored[:3]:
            candidates.append((sc, p, n, "init"))
            try_record(p, sc)

    candidates.sort(key=lambda x: x[0])

    # Phase 2: Local optimization on top candidates
    # Allocate time based on number of candidates
    n_candidates = min(len(candidates), 8)
    if n_candidates == 0:
        n = lengths_to_try[0]
        p = safe_normalize(parent) if parent is not None else np.full(n, 1.0 / n)
        candidates = [(internal_score(p), p, n, "fallback")]

    available = max(60.0, deadline - time.monotonic())
    # Give more time to better candidates
    time_weights = []
    for i in range(n_candidates):
        w = 1.0 / (i + 1)
        time_weights.append(w)
    total_w = sum(time_weights)
    time_per = [available * w / total_w * 0.8 for w in time_weights]

    for i in range(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        sc, p, n, label = candidates[i]
        t_alloc = min(time_per[i], max(30.0, deadline - time.monotonic() - 60.0))
        if t_alloc < 20:
            break
        print("Optimizing candidate %d (n=%d, init_score=%.6f, alloc=%.0fs)" % (i, n, sc, t_alloc), flush=True)
        p_best, sc_best = optimize_adaptive(p, rng, t_alloc, "c%d" % i)
        try_record(p_best, sc_best)

    # Phase 3: Polish the best with more time
    if best_p_global is not None and time.monotonic() < deadline - 30:
        remaining = max(10.0, deadline - time.monotonic() - 10.0)
        print("Final polish (alloc=%.0fs)" % remaining, flush=True)
        p_final, sc_final = optimize_adaptive(best_p_global, rng, remaining, "FINAL")
        try_record(p_final, sc_final)

    # Phase 4: Try a few more random restarts if time permits
    if best_p_global is not None and time.monotonic() < deadline - 60:
        n_best = int(best_p_global.size)
        for _ in range(3):
            if time.monotonic() > deadline - 60:
                break
            p_try = project_simplex(0.6 * best_p_global + 0.4 * rng.dirichlet(np.ones(n_best) * 0.3))
            sc_try = internal_score(p_try)
            if np.isfinite(sc_try) and sc_try < best_score_global + 0.05:
                t_alloc = min(120.0, max(10.0, deadline - time.monotonic() - 30.0))
                p_r, sc_r = optimize_adaptive(p_try, rng, t_alloc, "restart")
                try_record(p_r, sc_r)

    if best_p_global is None:
        n = 700
        best_p_global = np.full(n, 1.0 / n, dtype=np.float64)
        best_score_global = internal_score(best_p_global)

    best_p_global = np.asarray(best_p_global, dtype=np.float64)
    best_p_global = np.where(np.isfinite(best_p_global), best_p_global, 0.0)
    best_p_global = np.maximum(best_p_global, 0.0)
    s = float(best_p_global.sum())
    if s <= 1e-12:
        best_p_global = np.full(max(1, int(best_p_global.size)), 1.0 / max(1, int(best_p_global.size)))
        s = 1.0
    best_p_global = np.minimum(best_p_global, 1000.0)

    elapsed = time.monotonic() - start_time
    print("Done. Best score: %.10f, n=%d, elapsed=%.1fs" % (best_score_global, best_p_global.size, elapsed), flush=True)

    return [float(x) for x in best_p_global]
