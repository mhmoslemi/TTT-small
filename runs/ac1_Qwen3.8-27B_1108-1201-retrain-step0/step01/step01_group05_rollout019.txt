"""
Minimize 2*n*max(self_convolution) / sum^2 over non-negative sequences.

Strategy:
  - Start from the previous best sequence (height_sequence_1) and optimize further.
  - Use FFT-based self-convolution and gradient computation for O(n log n) per step.
  - Combine adaptive top-k subgradient with soft threshold-based subgradient.
  - Use Adam optimizer with simplex projection for the main phase.
  - Use mirror descent (multiplicative weights) for non-negativity-preserving updates.
  - Multiple restarts with random perturbations to escape local minima.
  - Final polish with very small learning rate and full subgradient.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, w):
    """Compute subgradient: grad[m] = 2 * sum_k w[k]*a[k-m] for valid k-m.
    
    Uses identity: grad = 2 * conv(w, a_rev)[n-1 : 2n-1]
    where a_rev = a[::-1].
    """
    n = len(a)
    a_rev = a[::-1]
    # conv(w, a_rev) has length (2n-1) + n - 1 = 3n-2
    target_len = 3 * n - 2
    fsize = 1 << (target_len - 1).bit_length()
    W = fft(w, fsize)
    AR = fft(a_rev, fsize)
    conv_result = np.real(ifft(W * AR))
    return 2.0 * conv_result[n - 1:2 * n - 1]


def make_topk_weights(c, k):
    """Weights proportional to c values on top-k lags."""
    w = np.zeros(len(c), dtype=np.float64)
    if k >= len(c):
        idx = np.argsort(c)[::-1]
    else:
        idx = np.argpartition(c, -k)[-k:]
        idx = idx[np.argsort(c[idx])[::-1]]
    top_vals = c[idx]
    total = top_vals.sum()
    if total <= 1e-15:
        w[np.argmax(c)] = 1.0
        return w
    w[idx] = top_vals / total
    return w


def make_soft_weights(c, M, delta, p=1.0):
    """Soft weights: w[k] proportional to max(0, c[k] - (M-delta))^p."""
    threshold = M - delta
    excess = np.maximum(c - threshold, 0.0)
    if p != 1.0:
        excess = excess ** p
    total = excess.sum()
    if total <= 1e-15:
        w = np.zeros(len(c), dtype=np.float64)
        w[np.argmax(c)] = 1.0
        return w
    return excess / total


def run_adam_phase(a_init, n, S, budget, rng, lr_init=0.3, top_k_start=3,
                   top_k_end=60, switch_to_soft=0.4, soft_delta_frac=0.02):
    """Adam-based optimization with adaptive subgradient.
    
    Phase 1 (first 1-switch_to_soft fraction): top-k subgradient with increasing k.
    Phase 2 (last switch_to_soft fraction): soft threshold subgradient.
    """
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)

    c = fft_convolve_self(a)
    best_M = c.max()
    best_a = a.copy()
    start_time = time.monotonic()
    t = 0

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.8, 0.999, 1e-10
    lr = lr_init

    no_improve_count = 0
    improve_threshold = 1e-9

    while time.monotonic() - start_time < budget:
        t += 1

        c = fft_convolve_self(a)
        M = c.max()
        if M < best_M - improve_threshold:
            best_M = M
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Adaptive top-k
        frac = min(1.0, float(t) / max(1, int(budget * 100)))
        if frac < (1.0 - switch_to_soft):
            # Phase 1: top-k with increasing k
            progress = frac / (1.0 - switch_to_soft)
            k = int(top_k_start + (top_k_end - top_k_start) * progress * progress)
            k = max(top_k_start, min(top_k_end, k))
            w = make_topk_weights(c, k)
        else:
            # Phase 2: soft subgradient
            delta = M * soft_delta_frac * (1.0 + frac)
            p = 1.0 + 2.0 * (frac - (1.0 - switch_to_soft)) / switch_to_soft
            w = make_soft_weights(c, M, delta, p=min(p, 3.0))

        grad = compute_gradient_fft(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_scaled = grad / gnorm

        m = beta1 * m + (1.0 - beta1) * grad_scaled
        v = beta2 * v + (1.0 - beta2) * (grad_scaled ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)

            noise = rng.normal(0.0, noise_scale, n)
            a = project_simplex(a + noise, S)
            no_improve_count = 0

    return best_a, best_M

                # Random with structure

                a_rand = a_rand * (1.0 + 0.5 * np.sin(2 * np.pi * 2.0 * np.linspace(0, 1, n)))
                a_rand = np.abs(a_rand)
                a_restart = a_rand * (S / a_rand.sum())

            a_r, M_r = run_combined(a_restart, n, S, restart_budget, rng)
            if M_r < global_best_M:
                global_best_M = M_r
                global_best_a = a_r.copy()
                print("Restart %d improved: M=%.6f" % (restart_i, M_r))
            else:
                print("Restart %d: M=%.6f (no improvement)" % (restart_i, M_r))

    # Phase 3: Final polish
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        polish_budget = max(5.0, remaining - 5.0)
        a_polished, M_polished = polish(global_best_a, n, S, polish_budget, rng)
        if M_polished < global_best_M:
            global_best_M = M_polished
            global_best_a = a_polished.copy()
        print("After polish: M=%.6f" % global_best_M)

    # Final verification
    final_a = np.clip(global_best_a, 0.0, 1000.0)
    try:
        final_score = evaluate_sequence(final_a.tolist())
        print("Final verified score: %.6f" % final_score)
        if not np.isfinite(final_score):
            # Fallback
            final_a = np.ones(256)
            final_score = evaluate_sequence(final_a.tolist())
            print("Fallback score: %.6f" % final_score)
    except Exception as e:
        print("Final verification error: %s" % e)
        final_a = np.ones(256)

    # Safety check
    if final_a.sum() < 0.01:
        final_a = np.ones(256)

    return [float(x) for x in final_a]
