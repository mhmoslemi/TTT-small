import numpy as np
import time

def _score(a):
    """Compute 2*n*max(convolve(a,a)) for a with sum=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)

def _top_k_grad(a, K, temperature):
    """Compute weighted gradient of top-K peaks."""
    n = len(a)
    b = np.convolve(a, a)
    b_max = b[np.argmax(b)]
    
    # Get top K indices
    if K >= len(b):
        k_idx = np.arange(len(b))
    else:
        k_idx = np.argpartition(b, -K)[-K:]
    
    # Exponential weights based on temperature
    b_sel = b[k_idx]
    wts = np.exp(-temperature * (b_max - b_sel) / max(b_max, 1e-15))
    wts /= np.sum(wts)
    
    # Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (valid range)
    grad = np.zeros(n)
    for ki in range(len(k_idx)):
        if wts[ki] < 1e-10:
            continue
        k = k_idx[ki]
        w = 2.0 * wts[ki]
        # a[k-i] is valid when 0 <= k-i <= n-1, i.e., k-n+1 <= i <= k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # a[k-i] for i in [lo, hi] is a[k-hi : k-lo+1][:reverse]
            # When i=lo: a[k-lo]. When i=hi: a[k-hi].
            # So the slice is a[k-hi : k-lo+1] reversed
            seg = a[k-hi : k-lo+1][::-1]
            grad[lo:hi+1] += w * seg
    
    return grad, b_max

def _mirror_descent_step(a, grad, eta):
    """Multiplicative update: a[i] *= exp(-eta * grad[i]), then normalize."""
    a_new = a * np.exp(-eta * grad)
    S = np.sum(a_new)
    if S < 1e-15:
        return a
    a_new /= S
    return a_new

Uses mirror descent (multiplicative updates), top-K peak selection with exponential

Multiple diverse initializations are explored before intensive refinement.

import numpy as np
import time


def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) for a sequence with sum=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _top_k_grad(a, K, temperature):
    """
    Compute weighted subgradient of max_k b[k] using top-K peaks
    with exponential temperature weighting.
    Returns (gradient vector, current peak value).
    """
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)

    if K >= len(b):
        k_idx = np.arange(len(b))
    else:
        k_idx = np.argpartition(b, -K)[-K:]

    b_sel = b[k_idx]
    denom = max(b_max, 1e-15)
    wts = np.exp(-temperature * (b_max - b_sel) / denom)
    wts /= np.sum(wts)

    grad = np.zeros(n)
    for ki in range(len(k_idx)):
        if wts[ki] < 1e-10:
            continue
        k = int(k_idx[ki])
        w = 2.0 * wts[ki]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[k - hi: k - lo + 1][::-1]
            grad[lo: hi + 1] += w * seg

    return grad, b_max


def _mirror_step(a, grad, eta):
    """Multiplicative update: a[i] *= exp(-eta*grad[i]), normalize to sum=1."""
    log_mult = np.clip(-eta * grad, -50.0, 50.0)
    a_new = a * np.exp(log_mult)
    a_new = np.maximum(a_new, 1e-15)
    S = np.sum(a_new)
    if S < 1e-15:
        return a
    a_new /= S
    return a_new

    """

    S = np.sum(a)

        a = np.ones(n) / n
    else:

    eta = eta_init

    temp = temp_init

    a = np.ones(n, dtype=np.float64)
    for f in freqs:
        phase = 2.0 * np.pi * f * i / n
        a += eps * np.cos(phase)
    a = np.maximum(a, 1e-10)
    a /= np.sum(a)
    return a
