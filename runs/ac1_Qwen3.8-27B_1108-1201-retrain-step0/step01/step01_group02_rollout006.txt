import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def compute_score(a):
    n = len(a)
    b = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * b.max() / (s ** 2)


def compute_conv(a):
    return np.convolve(a, a)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_beta(n, alpha=2.0, beta_param=2.0):
    """Beta-distribution-like shape."""
    t = np.arange(n, dtype=np.float64)
    x = t / max(n - 1, 1)
    x = np.clip(x, 1e-8, 1 - 1e-8)
    return x ** (alpha - 1) * (1 - x) ** (beta_param - 1)


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_inverse_sqrt(n):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = np.abs(t - center) / (n / 2.0)
    return 1.0 / (1.0 + x)


def build_and_solve_lp(a, timeout=30.0):
    """
    Solve LP: minimize M' subject to b_k + 2*conv(a,d)_k <= M', d >= -a, sum(d)=0.
    Returns (d_star, M_prime) or (None, None).
    """
    n = len(a)
    a = normalize_nonneg(a.copy())
    b = np.convolve(a, a)
    
    n_vars = n + 1  # d_0..d_{n-1}, M'
    n_conv_rows = 2 * n - 1
    n_nn_rows = n
    
    row_idx = []
    col_idx = []
    data = []
    
    # Build convolution constraint rows
    # For row k: 2*sum_j a[k-j]*d_j - M' <= -b_k
    for k in range(n_conv_rows):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            idx = k - j
            val = 2.0 * a[idx]
            if val > 1e-15:
                row_idx.append(k)
                col_idx.append(j)
                data.append(val)
        # M' coefficient
        row_idx.append(k)
        col_idx.append(n)
        data.append(-1.0)
    
    # Non-negativity: -d_i <= a_i
    for i in range(n):
        row_idx.append(n_conv_rows + i)
        col_idx.append(i)
        data.append(-1.0)
    
    A_ub = csr_matrix((data, (row_idx, col_idx)), shape=(n_conv_rows + n_nn_rows, n_vars))
    b_ub = np.concatenate([-b, a])
    
    # Objective: minimize M'
    c = np.zeros(n_vars)
    c[n] = 1.0
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: d_i in [-a_i, 1-a_i], M' in [0, inf)
    bounds = [(-a[i], 1.0 - a[i]) for i in range(n)] + [(0.0, None)]
    
    try:

def local_search(a, time_limit_s=60.0, seed=123, verbose=False):

        print(f"    LS start: score={best_score:.8f}")

        # Random multiplicative perturbation on a random segment
        width = max(3, int(rng.integers(n // 100, n // 6)))
        start = int(rng.integers(0, max(1, n - width)))

            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = normalize_nonneg(candidate)
        sc = compute_score(candidate)
        
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 100 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.8f}")
    
    if verbose:
        print(f"    LS done: best={best_score:.8f}, {accept_count} accepts, {time.monotonic()-start_time:.1f}s")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses direct LP peak-reduction, subgradient pre-optimization, and local search.
    """
    total_start = time.monotonic()
    deadline = total_start + budget_s - 60.0  # safety margin
    
    rng = np.random.default_rng(seed)
    
    best_score = float('inf')
    best_seq = None
    
    def track(result_a, result_score, label=""):
        nonlocal best_score, best_seq
        if result_score < best_score:
            best_score = result_score
            best_seq = result_a.copy()
            elapsed = time.monotonic() - total_start
            print(f"  *** NEW BEST: score={best_score:.8f}, n={len(result_a)}, "
                  f"{label}, elapsed={elapsed:.1f}s")
    
    # Try to use previous best if available
    prev_sequences = []
    try:
        if 'height_sequence_1' in dir() or True:
            prev = height_sequence_1
            if prev is not None and len(prev) > 0:
                prev_arr = np.array(prev, dtype=np.float64)
                prev_arr = np.maximum(0, prev_arr)
                if prev_arr.sum() > 0.01:
                    prev_sequences.append((prev_arr, "prev_best"))
    except Exception:
        pass
    
    # Define candidate configurations: (n, init_func, label)
    configs = []
    
    # From previous best
    for prev_arr, label in prev_sequences:
        configs.append((len(prev_arr), lambda n, p=prev_arr: p.copy(), label))
    
    # Raised cosine with various lengths
    for n in [301, 401, 501, 701]:
        configs.append((n, initial_raised_cosine, f"raised_cos_{n}"))
    
    # Beta distributions
    for n in [401, 601]:
        configs.append((n, lambda n: initial_beta(n, 3.0, 3.0), f"beta_3_3_{n}"))
        configs.append((n, lambda n: initial_beta(n, 4.0, 4.0), f"beta_4_4_{n}"))
    
    # Gaussian
    for n in [401, 601]:
        configs.append((n, lambda n: initial_gaussian(n, 0.35), f"gauss_{n}"))
    
    # Inverse sqrt
    configs.append((501, lambda n: initial_inverse_sqrt(n), "inv_sqrt_501"))
    
    # Allocate time: sort by priority (previous best first, then shorter sequences)
    # Give more time to promising configs
    time_per_config = max(60.0, (budget_s - 200.0) / max(len(configs), 1))
    
    # Cap total configs to try
    max_configs = min(len(configs), 8)
    
    print(f"Total configs to try: {max_configs}, time per config ~{time_per_config:.0f}s")
    print(f"Deadline in {budget_s - 60:.0f}s")
    
    for cfg_idx in range(max_configs):
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break
        
        n, init_func, label = configs[cfg_idx]
        
        # Allocate time for this config
        configs_remaining = max_configs - cfg_idx
        time_alloc = min(time_per_config, remaining * 0.7 / configs_remaining)
        if time_alloc < 20.0:
            time_alloc = max(20.0, remaining * 0.3)
        
        print(f"\n=== Config {cfg_idx+1}/{max_configs}: n={n}, init={label}, "
              f"alloc={time_alloc:.0f}s, remaining={remaining:.0f}s ===")
        
        try:
            a = init_func(n)
            a = normalize_nonneg(a)
        except Exception as e:
            print(f"    Init error: {e}")
            continue
        
        init_score = compute_score(a)
        print(f"    Init score: {init_score:.8f}")
        
        # Phase 1: Subgradient pre-optimization (fast)
        sg_time = min(30.0, time_alloc * 0.15)
        if sg_time > 5.0:
            a_sg, sg_score = subgradient_refine(a, time_limit_s=sg_time, verbose=False)
            if sg_score < init_score:
                print(f"    After subgrad: {sg_score:.8f}")
                a = a_sg
        
        # Phase 2: LP refinement (precise)
        lp_time = time_alloc * 0.65
        if time.monotonic() - total_start > deadline - 30:
            break
        a_lp, lp_score = lp_refine(a, time_limit_s=lp_time, verbose=True)
        track(a_lp, lp_score, label=f"{label}_lp")
        
        # Phase 3: Local search polish
        ls_time = min(30.0, time_alloc * 0.2)
        if ls_time > 5.0 and a_lp is not None:
            a_ls, ls_score = local_search(a_lp, time_limit_s=ls_time, 
                                           seed=seed + cfg_idx, verbose=False)
            track(a_ls, ls_score, label=f"{label}_ls")
            
            # Quick LP polish after LS
            if time.monotonic() - total_start < deadline - 20:
                a_pl, pl_score = lp_refine(a_ls, time_limit_s=min(30.0, 
                    deadline - time.monotonic() - 10), verbose=False)
                track(a_pl, pl_score, label=f"{label}_final")
    
    # Final: if we have a best, do one more round of polish
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60.0:
            print(f"\n=== Final polish round: {remaining:.0f}s remaining ===")
            
            # LP polish
            lp_final_time = remaining * 0.5
            a_f, score_f = lp_refine(best_seq, time_limit_s=lp_final_time, verbose=True)
            track(a_f, score_f, label="final_lp")
            
            # LS polish
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20.0:
                a_fs, score_fs = local_search(a_f, time_limit_s=min(40.0, remaining2 * 0.5),
                                               seed=seed + 999, verbose=True)
                track(a_fs, score_fs, label="final_ls")
    
    if best_seq is None:
        # Fallback: raised cosine
        best_seq = normalize_nonneg(initial_raised_cosine(500))
        best_score = compute_score(best_seq)
        print(f"  Fallback: score={best_score:.8f}")
    
    # Final verification
    result_list = [float(x) for x in best_seq]
    verify_score = evaluate_sequence(result_list)
    print(f"\nFinal: score={best_score:.8f}, n={len(best_seq)}")
    print(f"Verified via evaluate_sequence: {verify_score:.8f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s / {budget_s}s")
    
    return result_list
