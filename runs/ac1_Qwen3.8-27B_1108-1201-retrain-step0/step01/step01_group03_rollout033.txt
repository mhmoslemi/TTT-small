Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive softmax-gradient with backtracking line search, aggressive LP-based
refinement, diverse initialization, and coordinate descent polishing.

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def grad_soft(a, tau):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    # Softmax weights
    w = np.exp(tau * (c - c_max))
    w /= w.sum()
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (with valid indices)
    # For efficiency, use FFT
    a_padded = np.zeros(2*n - 1)
    a_padded[:n] = a
    w_padded = np.zeros(2*n - 1)
    w_padded[:] = w
    
    # We need: grad[j] = 2 * sum_{k} w[k] * a[k-j] for valid k, k-j
    # = 2 * (correlation of w and a) at lag j
    # Use FFT for efficiency
    N = 1
    while N < 4*n:
        N *= 2
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # corr = ifft(W * conj(A)) gives sum_k w[k] * conj(a[k-j]) = sum_k w[k] * a[k-j] (a is real)
    corr = np.fft.irfft(W * np.conj(A), N)
    grad = 2.0 * corr[:n]
    
    return grad, c.max()


def grad_hard(a):
    """Compute hard-max gradient of the peak autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(c.argmax())
    
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[k_star - j]
    
    return grad, float(c.max())


def gradient_step(a, lr, tau=None):
    """One gradient step with projection."""
    n = len(a)
    if tau is not None and tau > 0:
        grad, _ = grad_soft(a, tau)
    else:
        grad, _ = grad_hard(a)
    
    # Project out the mean (maintain sum constraint)
    grad = grad - grad.mean()
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-12:
        a_new = np.ones(n)
    a_new = normalize(a_new)
    return a_new


def line_search(a, lr, tau, max_iter=20):
    """Backtracking line search for the gradient step."""
    S = a.sum()
    obj_current = compute_obj(a)
    for i in range(max_iter):
        a_new = gradient_step(a, lr, tau)
        obj_new = compute_obj(a_new)
        if obj_new < obj_current:
            return a_new, obj_new
        lr = lr * 0.5
    return a, obj_current


def lp_solve(f, M, n_max=2048):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M (elementwise), g >= 0."""
    n = len(f)
    if n > n_max:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix efficiently
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

import numpy as np

from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def softmax_grad(a, tau):
    """Compute softmax-weighted gradient of peak autocorrelation using FFT."""
    n = len(a)
    N = 1
    while N < 4 * n:
        N *= 2
    A = np.fft.rfft(a, N)
    c_full = np.fft.irfft(A * A.conj(), N)
    c = c_full[:2 * n - 1]
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w /= w.sum()
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(W * A.conj(), N)
    grad = 2.0 * corr[:n]
    return grad, float(c.max())

    """Compute hard-max gradient of peak autocorrelation."""
    n = len(a)

def spectral_iter(a, alpha, N=None):
    """
    Gerchberg-Saxton style spectral iteration.
    Modifies the power spectrum to flatten the autocorrelation,
    then projects back to non-negative time domain.
    """
    n = len(a)
    if N is None:
        N = 1
        while N < 4 * n:
            N *= 2

    A = np.fft.rfft(a, N)
    P = np.abs(A) ** 2
    c_full = np.fft.irfft(P, N)
    c = c_full[:2 * n - 1]

    S = a.sum()
    c_mean = (S * S) / (2 * n - 1)

    # Reduce excess above mean
    excess = np.maximum(0.0, c - c_mean)
    c_target = c - alpha * excess
    c_target = np.maximum(c_target, 0.0)

    # Pad c_target back to full length N
    c_full_target = np.zeros(N)
    c_full_target[:2 * n - 1] = c_target
    # Mirror for symmetry (autocorrelation is symmetric)
    c_full_target[2 * n - 1:N] = c_target[2 * n - 2 :: -1][:N - (2 * n - 1)]

    P_new = np.fft.rfft(c_full_target, N).real
    P_new = np.maximum(P_new, 0.0)

    mag = np.abs(A)

                if res.status != 0:
                    continue
                g = res.x
            except Exception:
                continue

            # Try blending
            improved = False
            current_score = compute_obj(a)
            for t in [0.8, 0.6, 0.4, 0.2, 0.1, 0.9]:
                a_try = (1.0 - t) * a + t * g
                a_try = np.maximum(a_try, 0.0)
                a_try = normalize(a_try, S_t)
                s_try = compute_obj(a_try)
                if s_try < current_score:
                    a = a_try
                    current_score = s_try
                    improved = True
                    break

            if improved:
                break

        if not improved:
            # Try a small gradient nudge
            a_try, _, _ = backtrack_step(a, 0.03, 50.0)
            s_try = compute_obj(a_try)
            if s_try < compute_obj(a):
                a = a_try
            else:
                break

    return a


def coordinate_descent(a, deadline, max_rounds=15):
    """Coordinate descent polishing with small perturbations."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)

    for _ in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        current_score = compute_obj(a)

        for j in range(n):
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-8:
                # Try turning on
                for frac in [0.05, 0.1, 0.2]:
                    a_try = a.copy()
                    a_try[j] = a.mean() * frac
                    a_try = normalize(a_try, S_t)
                    s = compute_obj(a_try)
                    if s < current_score:
                        a = a_try
                        current_score = s
                        improved = True
                        break
            else:
                # Try adjusting
                for delta_frac in [0.1, 0.05, 0.2, 0.02]:
                    for sign in [-1, 1]:
                        a_try = a.copy()
                        a_try[j] = max(0.0, a_old * (1.0 + sign * delta_frac))
                        a_try = normalize(a_try, S_t)
                        s = compute_obj(a_try)
                        if s < current_score:
                            a = a_try
                            current_score = s
                            improved = True
                            break
                    if improved:
                        break

        if not improved:
            break

    return a


def optimize_one(a0, deadline):
    """Full optimization pipeline for a single starting sequence."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize(a)
    S_t = np.sqrt(2.0 * n)

    best = a.copy()
    best_score = compute_obj(a)

    # Phase 1: Softmax gradient with backtracking
    tau = 15.0
    lr = 0.3
    it = 0
    max_grad = 3000
    patience = 0
    patience_limit = 500

    while it < max_grad:
        if time.monotonic() > deadline - max(30.0, (deadline - time.monotonic()) * 0.3):
            break

        a_try, obj_try, lr_used = backtrack_step(a, lr, tau)

        if obj_try < best_score:
            best = a_try.copy()
            best_score = obj_try
            patience = 0
        else:
            patience += 1

        a = a_try
        it += 1

        # Anneal
        frac = it / max_grad
        tau = 15.0 + 85.0 * frac
        lr = max(0.01, 0.3 * (1.0 - 0.8 * frac))

        if patience > patience_limit:
            # Reset with a kick
            a = best.copy() + 0.01 * np.random.default_rng(it).random(n)
            a = np.maximum(a, 0)
            a = normalize(a, S_t)
            patience = 0

    # Phase 2: Spectral iteration
    if time.monotonic() < deadline - max(20.0, (deadline - time.monotonic()) * 0.3):
        a_spec, spec_score = spectral_optimize(best, deadline - 30.0, alpha_init=0.05, max_iters=300)
        if spec_score < best_score:
            best = a_spec.copy()
            best_score = spec_score

    # Phase 3: LP refinement
    if n <= 768 and time.monotonic() < deadline - 20.0:
        a_lp = lp_refine(best, deadline - 10.0, max_iters=100)
        lp_score = compute_obj(a_lp)
        if lp_score < best_score:
            best = a_lp.copy()
            best_score = lp_score

    # Phase 4: Coordinate descent
    if time.monotonic() < deadline - 5.0:
        best = coordinate_descent(best, deadline - 2.0, max_rounds=5)
        cd_score = compute_obj(best)
        if cd_score < best_score:
            best_score = cd_score

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses adaptive softmax-gradient with backtracking,
    spectral iteration, LP refinement, and coordinate descent.
    Explores diverse starting points including height_sequence_1.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    print(f"=== Starting optimization, budget={budget_s:.0f}s ===")

    starts = []

    # Use height_sequence_1 if available
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)):
            for L in [256, 384, 512, 640, 768, 1024]:
                if L <= len(gs):
                    starts.append(gs[:L].copy())
            # Reversed
            if len(gs) >= 512:
                starts.append(gs[:512][::-1].copy())
            # Slightly perturbed
            if len(gs) >= 512:
                perturbed = gs[:512].copy()
                noise = rng.random(512) * 0.1
                perturbed = perturbed + noise
                starts.append(np.maximum(perturbed, 0).copy())
            print(f"  Loaded height_sequence_1 (len={len(gs)})")
    except NameError:
        print("  height_sequence_1 not available, using other starts")

    # Diverse structured starts
    for n in [256, 512]:
        i = np.arange(n, dtype=np.float64)
        # Chirp
        starts.append(0.4 + 0.6 * np.cos(2 * np.pi * i * 3.0 / n))
        # Raised cosine
        starts.append(0.4 + 0.6 * np.cos(2 * np.pi * i / n))
        # Double bump
        starts.append(np.exp(-((i - n * 0.25) / (n * 0.12)) ** 2) +
                      np.exp(-((i - n * 0.75) / (n * 0.12)) ** 2) + 0.1)
        # Sine modulated
        starts.append(np.abs(np.sin(2 * np.pi * i * 5.0 / n)) * 0.5 + 0.5)
        # Power decay
        starts.append((1.0 - i / n) ** 0.5 + 0.1)
        # Random
        starts.append(rng.random(n) * 0.4 + 0.3)

    # Sort by length (shorter first for quick results)
    starts.sort(key=lambda x: len(x))

    print(f"  Total starting sequences: {len(starts)}")

    # Time allocation: reserve time for final polish
    reserve_time = 60.0
    effective_deadline = deadline - reserve_time
    available_time = max(60.0, effective_deadline - t_start)
    time_per = available_time / len(starts)

    for idx, a0 in enumerate(starts):
        if time.monotonic() > effective_deadline:
            break

        t0 = time.monotonic()
        alloc = min(time_per, effective_deadline - t0)
        if alloc < 30.0:
            alloc = 30.0

        try:
            seq, score = optimize_one(a0, t0 + alloc)
        except Exception as e:
            print(f"  Start {idx} error: {e}")
            continue

        elapsed = time.monotonic() - t0
        status = ""
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            status = " *** NEW BEST ***"
        print(f"  [{idx:2d}] n={len(seq):4d} score={score:.6f} t={elapsed:6.1f}s{status}")

    # Final intensive polish on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 10.0:
        print(f"\n  === Final polish (score={best_score:.6f}) ===")
        remain = deadline - 10.0 - time.monotonic()

        # Multiple rounds of combined optimization
        for round_i in range(3):
            if time.monotonic() > remain:
                break
            round_deadline = remain if round_i == 2 else remain - (remain / 3) * (2 - round_i)
            seq, score = optimize_one(best_seq, round_deadline)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
            print(f"  Round {round_i}: score={best_score:.6f}")

        # Final coordinate descent
        if time.monotonic() < deadline - 5.0:
            best_seq = coordinate_descent(best_seq, deadline - 3.0, max_rounds=10)
            final_cd = compute_obj(best_seq)
            print(f"  After final CD: {final_cd:.6f}")
            if final_cd < best_score:
                best_score = final_cd

    # Validate with evaluate_sequence
    if best_seq is not None:
        final_check = compute_obj(best_seq)
        print(f"\n  Final internal score: {final_check:.6f}")
        print(f"  Final best tracked: {best_score:.6f}")
        print(f"  Sequence length: {len(best_seq)}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"  evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"  Evaluation error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("  FALLBACK: uniform sequence")
    return [1.0] * 256
