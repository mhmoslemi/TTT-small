Minimize 2*n*max_conv / sum^2 over non-negative sequences using
smooth max (log-sum-exp) with FFT-based gradient computation.

Key improvements over previous approach:
- O(n log n) gradient via single FFT convolution (vs O(k*n) loop)
- Nesterov accelerated gradient descent
- Adaptive temperature annealing for smooth -> sharp max transition
- Multiple diverse initializations including flat-spectrum constructions
- Careful polishing phase with fine-tuned hyperparameters

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = fft(a, size)
    return np.real(ifft(A * A))[:2 * n - 1]


def fft_cross_convolve(w, a):
    """Compute conv(w, a_reversed) via FFT. Returns array of length len(w)+n-1."""
    n = len(a)
    a_rev = a[::-1]
    size = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, size)
    A = fft(a_rev, size)
    return np.real(ifft(W * A))[:len(w) + n - 1]


def compute_grad(a, w):
    """Gradient of smooth max of conv(a,a) w.r.t. a.
    grad[m] = 2 * sum_k w[k] * a[k-m] = 2 * conv(w, a_rev)[m+n-1]"""
    n = len(a)
    conv_res = fft_cross_convolve(w, a)
    return 2.0 * conv_res[n - 1:2 * n - 1]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cand = u - (css - S) / rho
    idx = np.where(cand > 0)[0]
    if len(idx) == 0:
        theta = css[-1] / n
    else:
        theta = (css[idx[-1]] - S) / (idx[-1] + 1)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    """Numerically stable softmax."""
    x_max = x.max()
    e = np.exp(x - x_max)
    return e / e.sum()

Actually, one more thought: I should also try a "dual" approach where I directly optimize the constraint. The problem is:
min t s.t. (a*a)[k] <= t for all k, a >= 0, sum(a) = S

The Lagrangian is: L = t + sum_k lambda_k * ((a*a)[k] - t) - sum_i mu_i * a_i - nu * (sum(a) - S)

At the optimum, KKT conditions give:
- lambda_k >= 0, lambda_k * ((a*a)[k] - t) = 0 (complementary slackness)
- For active constraints (where (a*a)[k] = t): the gradient of (a*a)[k] must be in the span of the constraint gradients
- sum_k lambda_k = 1 (from dt/dt = 1 - sum lambda_k = 0)

So at the optimum, the weighted sum of gradients of active constraints (weighted by lambda_k which sum to 1) must equal zero (in the tangent space of the feasible set). This means the active constraint gradients must "balance" each other.

This is exactly what the smooth max gradient does: it computes a weighted average of the constraint gradients (weighted by softmax), and we drive this to zero. So the smooth max approach is essentially a primal-dual method.

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
- Use smooth max (log-sum-exp with temperature T) as differentiable surrogate for max
- Compute gradients in O(n log n) via a single FFT cross-convolution
- Nesterov-accelerated projected gradient with adaptive temperature annealing
- Multiple diverse initializations (flat-spectrum, cosine, Gaussian, random, given)
- Multiple sequence lengths to explore the solution landscape
- Final polishing phase with very fine hyperparameters
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = fft(a, size)
    return np.real(ifft(A * A))[:2 * n - 1]


def compute_grad(a, w):
    """Gradient of smooth max of conv(a,a) w.r.t. a.
    grad[m] = 2 * sum_k w[k] * a[k-m] = 2 * conv(w, a_rev)[m+n-1]
    Computed via a single FFT cross-convolution: O(n log n)."""
    n = len(a)
    a_rev = a[::-1]
    size = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, size)
    A = fft(a_rev, size)
    conv_res = np.real(ifft(W * A))[:len(w) + n - 1]
    return 2.0 * conv_res[n - 1:2 * n - 1]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cand = u - (css - S) / rho
    idx = np.where(cand > 0)[0]
    if len(idx) == 0:
        theta = css[-1] / n
    else:
        theta = (css[idx[-1]] - S) / (idx[-1] + 1)
    return np.maximum(v - theta, 0.0)


def stable_softmax(x):
    """Numerically stable softmax."""
    x_max = x.max()
    e = np.exp(x - x_max)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def normalize_to_S(a, n):
    """Clip to [0, 1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
    return a * (S / s)


def make_flat_spectrum_init(n, rng):
    """Initialize with flat DFT magnitude (random phases, conjugate symmetry)."""
    half_n = n // 2 + 1
    phases = rng.uniform(0.0, 2.0 * np.pi, half_n)
    phases[0] = 0.0
    if n % 2 == 0:
        phases[-1] = 0.0
    A = np.zeros(n, dtype=np.complex128)
    A[:half_n] = np.exp(1j * phases[:half_n])
    for k in range(1, half_n):
        if n - k < n:
            A[n - k] = np.conj(A[k])
    a = np.real(ifft(A))
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-10:
        a = a / a.sum() * np.sqrt(2.0 * n)
    else:
        a = np.ones(n) * np.sqrt(2.0 / n)
    return a


def make_cosine_init(n, rng, num_harmonics=3):
    """Initialize with sum of random cosine harmonics (non-negative)."""
    t = np.linspace(0.0, 2.0 * np.pi, n, endpoint=False)
    a = np.ones(n)
    for _ in range(num_harmonics):
        freq = rng.integers(1, n // 4 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        a += amp * np.cos(freq * t + phase)
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-10:
        a = a / a.sum() * np.sqrt(2.0 * n)
    else:
        a = np.ones(n) * np.sqrt(2.0 / n)
    return a


def make_gaussian_init(n, rng):
    """Gaussian bump (possibly shifted)."""
    center = rng.uniform(0.2 * n, 0.8 * n)
    sigma = rng.uniform(0.15 * n, 0.4 * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    if a.sum() > 1e-10:
        a = a / a.sum() * np.sqrt(2.0 * n)
    else:
        a = np.ones(n) * np.sqrt(2.0 / n)
    return a


def make_ramp_init(n, rng):
    """Linear ramp (direction random)."""
    if rng.random() < 0.5:
        a = np.linspace(0.01, 1.0, n)
    else:
        a = np.linspace(1.0, 0.01, n)
    a = a / a.sum() * np.sqrt(2.0 * n)
    return a


def generate_initializations(rng, lengths):
    """Generate diverse initializations."""
    inits = []
    # Given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 16 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            inits.append(("given", normalize_to_S(given, given.size)))
    except Exception:
        pass

    for n in lengths:
        inits.append(("uniform_%d" % n, np.ones(n) * np.sqrt(2.0 / n)))
        inits.append(("cos_%d" % n, make_cosine_init(n, rng, 4)))
        inits.append(("gauss_%d" % n, make_gaussian_init(n, rng)))
        inits.append(("flat_%d" % n, make_flat_spectrum_init(n, rng)))
        inits.append(("ramp_%d" % n, make_ramp_init(n, rng)))
        inits.append(("rand_%d" % n, rng.random(n) * np.sqrt(2.0 / n)))

    return inits


def run_adam_smooth(a_init, n, max_time, T_start, T_end, lr, tau_anneal,
                    use_nesterov=True, seed=0):
    """Adam optimizer with smooth max objective and temperature annealing."""
    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * float(n))
    a = normalize_to_S(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()

    # Adam state
    m_hat = np.zeros(n, dtype=np.float64)
    v_hat = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    start = time.monotonic()
    no_improve = 0
    patience = 20000

    # Nesterov velocity
    v_nest = np.zeros(n, dtype=np.float64)
    mu = 0.9

    while time.monotonic() - start < max_time:
        t += 1

        # Adaptive temperature
        T = max(T_end, T_start * np.exp(-float(t) / tau_anneal))

        # Compute convolution and gradient
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Softmax weights
        w = stable_softmax(c / T)

        # Gradient via FFT
        grad = compute_grad(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Nesterov look-ahead
        if use_nesterov:
            a_y = a + mu * v_nest
            c_y = fft_convolve_self(a_y)
            w_y = stable_softmax(c_y / T)
            grad_y = compute_grad(a_y, w_y)
            gn_y = np.linalg.norm(grad_y)
            if gn_y < 1e-15:
                grad = grad_y
            else:
                grad = grad_y / gn_y

        # Adam update
        m_hat = beta1 * m_hat + (1.0 - beta1) * grad
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad ** 2)
        m_b = m_hat / (1.0 - beta1 ** t)
        v_b = v_hat / (1.0 - beta2 ** t)

        # Learning rate with decay
        lr_t = lr / (1.0 + 0.001 * t)

        if use_nesterov:
            v_nest = mu * v_nest - lr_t * m_b / (np.sqrt(v_b) + eps)
            a = a + v_nest
        else:
            a = a - lr_t * m_b / (np.sqrt(v_b) + eps)

        # Project to simplex
        a = project_simplex(a, S)

        # Random restart if stagnating
        if no_improve > patience and rng.random() < 0.01:
            perturbation = rng.normal(0.0, 0.005 * S / n, n)
            a = project_simplex(a + perturbation, S)
            no_improve = 0

    return best_a, best_score


def run_adam_topk(a_init, n, max_time, top_k=10, lr=1.0, seed=0):
    """Adam with top-k subgradient (for polishing when T is very small)."""
    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * float(n))
    a = normalize_to_S(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()

    m_hat = np.zeros(n, dtype=np.float64)
    v_hat = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()

        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-15:
            break
        weights = top_vals / total

        grad = np.zeros(n, dtype=np.float64)
        for w_val, k in zip(weights, top_idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                segment = a[lo:hi + 1]
                if k - hi != k - lo:
                    grad[lo:hi + 1] += w_val * 2.0 * segment[::-1]
                else:
                    grad[lo:hi + 1] += w_val * 2.0 * segment

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        m_hat = beta1 * m_hat + (1.0 - beta1) * grad
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad ** 2)
        m_b = m_hat / (1.0 - beta1 ** t)
        v_b = v_hat / (1.0 - beta2 ** t)
        lr_t = lr / (1.0 + 0.01 * t)
        a = project_simplex(a - lr_t * m_b / (np.sqrt(v_b) + eps), S)

    return best_a, best_score


def polish(a_best, n, budget_left, seed=123):
    """Polish the best sequence with fine-tuned hyperparameters."""
    if budget_left < 2.0 or a_best is None:
        return a_best, np.inf

    S = np.sqrt(2.0 * float(n))
    a = normalize_to_S(a_best, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()

    # Phase 1: Smooth max with low temperature
    t1 = budget_left * 0.6
    if t1 > 2.0:
        T_start = max(0.01, 0.1 * c.max())
        T_end = max(0.001, 0.01 * c.max())
        a, s1 = run_adam_smooth(a, n, t1, T_start, T_end,
                                lr=0.5, tau_anneal=t1 * 1000.0,
                                use_nesterov=True, seed=seed)
        if s1 < best_score:
            best_score = s1
            best_a = a.copy()
        print("  polish smooth: %.6f" % s1)

    # Phase 2: Top-k subgradient
    remaining = budget_left - t1 - 1.0
    if remaining > 2.0:
        a2, s2 = run_adam_topk(best_a, n, remaining, top_k=20, lr=0.3, seed=seed + 1)
        if s2 < best_score:
            best_score = s2
            best_a = a2.copy()
        print("  polish topk:   %.6f" % s2)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety_margin = 15.0
    effective_budget = max(60.0, budget_s - safety_margin)
    t_start = time.monotonic()

    # Try different lengths
    lengths = [8192, 16384]

    inits = generate_initializations(rng, lengths)
    n_inits = len(inits)

    # Allocate time: 65% exploration, 35% polishing
    explore_frac = 0.65
    explore_budget = explore_frac * effective_budget
    polish_budget = effective_budget - explore_budget

    global_best = np.inf
    global_a = None
    global_n = 0

    # Sort inits: put "given" first, then shorter lengths (faster to process)
    inits.sort(key=lambda x: (0 if x[0] == "given" else 1, len(x[1])))

    per_init_time = explore_budget / max(1, n_inits)

    print("Starting optimization with %d initializations, budget=%.0fs"
          % (n_inits, budget_s))

    for name, a_init in inits:
        n = len(a_init)
        elapsed = time.monotonic() - t_start
        remaining_explore = explore_budget - elapsed
        if remaining_explore < 10.0:
            break

        # Don't let a single init take more than its allocated time
        alloc = min(per_init_time, remaining_explore)

        # Compute initial max for temperature
        c_init = fft_convolve_self(a_init)
        init_max = c_init.max()

        # Run smooth max optimization
        T_start = max(0.5, 0.5 * init_max)
        T_end = max(0.01, 0.05 * init_max)
        tau = max(100.0, alloc * 500.0)

        a_res, score = run_adam_smooth(a_init, n, alloc, T_start, T_end,
                                       lr=1.0, tau_anneal=tau,
                                       use_nesterov=True, seed=seed)
        if score < global_best:
            global_best = score
            global_a = a_res.copy()
            global_n = n
            # Verify periodically
            ver = evaluate_sequence(np.clip(a_res, 0, 1000).tolist())
            print("%s (n=%d): smooth=%.6f verify=%s" %
                  (name, n, score, "%.6f" % ver if np.isfinite(ver) else "inf"))

    print("After exploration: best=%.6f" % global_best)

    # Polish phase
    if global_a is not None:
        elapsed = time.monotonic() - t_start
        left = effective_budget - elapsed + safety_margin - 5.0
        left = max(5.0, left)
        global_a, pol_score = polish(global_a, global_n, left, seed=seed + 999)
        if pol_score < global_best:
            global_best = pol_score
        print("After polish: %.6f" % pol_score)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256)
        final_ver = evaluate_sequence(global_a.tolist())
        print("Final verified objective: %.6f" % final_ver)
    else:
        global_a = np.ones(256)
        final_ver = evaluate_sequence(global_a.tolist())
        print("Fallback uniform, objective: %.6f" % final_ver)

    elapsed_total = time.monotonic() - t_start
    print("Total time: %.1fs" % elapsed_total)

    return [float(x) for x in global_a]
