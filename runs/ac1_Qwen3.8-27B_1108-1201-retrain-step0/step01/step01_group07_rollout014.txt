Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Fienup frequency-domain pre-conditioning for good initialization
2. ADAM-optimized soft-max gradient descent for main optimization
3. L-BFGS-B polish for final refinement
Multi-length exploration with the provided sequence as a starting point.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def project_nonneg_sum(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = a.size
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return np.full(n, S / n, dtype=np.float64)
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, a.max()
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    res = np.maximum(a - t, 0.0)
    diff = S - res.sum()
    if diff > 1e-15:
        idx = int(np.argmax(res))
        res[idx] += diff
    elif diff < -1e-15:
        idx = int(np.argmax(res))
        res[idx] += diff
    return res

def fienup_precondition(a, n_iter=100):
    """Fienup-style iterations to flatten the magnitude spectrum."""
    n = a.size
    N = 2 * n
    a_work = np.zeros(N, dtype=np.float64)
    a_work[:n] = a
    S = a.sum()
    
    for i in range(n_iter):
        A = np.fft.rfft(a_work)
        # Target: flat magnitude
        target_mag = S / np.sqrt(N)
        phase = np.angle(A)
        # Reconstruct with target magnitude and current phase
        A_new = target_mag * np.exp(1j * phase)
        A_new[0] = complex(S, 0)
        a_work = np.fft.irfft(A_new, N)[:n]
        a_work = np.maximum(a_work, 0.0)
        s2 = a_work.sum()
        if s2 > 1e-15:
            a_work = a_work * (S / s2)
    
    return a_work.copy()

def compute_softmax_grad(a, tau):
    """Compute soft-max of conv(a,a) and its gradient w.r.t. a."""
    c = np.convolve(a, a)
    c_max = c.max()
    c_shift = c - c_max
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    sm_val = c_max + tau * np.log(exp_c.sum())
    # Gradient: 2 * correlate(q, a, 'valid')
    grad = 2.0 * np.correlate(q, a, 'valid')
    return sm_val, grad

def adam_softmax_opt(a0, t_budget, tau_init=5.0, tau_min=0.1, lr_init=1e-3, verbose=False):
    """ADAM optimizer with soft-max smoothing and tau annealing."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = (a0 / max(a0.sum(), 1e-15) * S).copy()
    a = project_nonneg_sum(a, S)
    
    best_a = a.copy()
    best_mc = np.convolve(best_a, best_a).max()
    best_score = 2.0 * n * best_mc / (S * S)
    
    # ADAM state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    lr = lr_init
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(t_budget, 1e-6))
        tau = tau_init * (tau_min / tau_init) ** frac
        
        sm_val, grad = compute_softmax_grad(a, tau)
        
        # ADAM update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_nonneg_sum(a_new, S)
        
        # Evaluate
        if t % 10 == 0:
            mc = np.convolve(a_new, a_new).max()
            if mc < best_mc - 1e-10:
                best_mc = mc
                best_a = a_new.copy()
                best_score = 2.0 * n * best_mc / (S * S)
        
        a = a_new
    
    # Final evaluation
    final_mc = np.convolve(a, a).max()
    final_score = 2.0 * n * final_mc / (S * S)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score

    n = a0.size
    S = a0.sum()

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Fienup frequency-domain pre-conditioning for flat-spectrum initialization
2. ADAM-optimized soft-max gradient descent with tau annealing
3. L-BFGS-B polish for final refinement
Multi-length exploration with periodic perturbation to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def project_nonneg_sum(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = a.size
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return np.full(n, S / n, dtype=np.float64)
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, a.max()
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    res = np.maximum(a - t, 0.0)
    diff = S - res.sum()
    if abs(diff) > 1e-15:
        idx = int(np.argmax(res))
        res[idx] += diff
    return res


def fienup_precondition(a, n_iter=100, N_factor=2):
    """Fienup-style iterations to flatten the magnitude spectrum."""
    n = a.size
    N = N_factor * n
    # Round N to even
    if N % 2 != 0:
        N += 1
    a_work = np.zeros(N, dtype=np.float64)
    a_work[:n] = a
    S = a.sum()

    for i in range(n_iter):
        A = np.fft.rfft(a_work)
        target_mag = S / np.sqrt(N)
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        A_new[0] = complex(S, 0)
        a_work = np.fft.irfft(A_new, N)[:n]
        a_work = np.maximum(a_work, 0.0)
        s2 = a_work.sum()
        if s2 > 1e-15:
            a_work = a_work * (S / s2)

    return a_work.copy()


def compute_softmax_grad(a, tau):
    """Compute soft-max of conv(a,a) and its gradient w.r.t. a."""
    c = np.convolve(a, a)
    c_max = c.max()
    c_shift = c - c_max
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    sm_val = c_max + tau * np.log(exp_c.sum())
    grad = 2.0 * np.correlate(q, a, 'valid')
    return sm_val, grad, c_max


def compute_score(a):
    """Compute the actual score 2*n*max(conv)/sum^2."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    c_max = np.convolve(a, a).max()
    return 2.0 * n * c_max / (s * s)


def adam_softmax_opt(a0, t_budget, tau_init=5.0, tau_min=0.05, lr_init=0.01,
                     verbose=False, perturb_interval=500):
    """ADAM optimizer with soft-max smoothing, tau annealing, and perturbation."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = (a0 / max(a0.sum(), 1e-15) * S).copy()
    a = project_nonneg_sum(a, S)

    best_a = a.copy()
    best_mc = np.convolve(best_a, best_a).max()
    best_score = 2.0 * n * best_mc / (S * S)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    lr = lr_init
    no_improve = 0
    rng = np.random.default_rng(12345)

    t0 = time.monotonic()
    last_print = t0

    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(t_budget, 1e-6))
        tau = tau_init * (tau_min / tau_init) ** frac

        sm_val, grad, c_max = compute_softmax_grad(a, tau)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_nonneg_sum(a_new, S)

        if t % 20 == 0:
            mc = np.convolve(a_new, a_new).max()
            if mc < best_mc - 1e-12:
                best_mc = mc
                best_a = a_new.copy()
                best_score = 2.0 * n * best_mc / (S * S)
                no_improve = 0
            else:
                no_improve += 20

            # Perturbation to escape local minima
            if no_improve > perturb_interval:
                noise = rng.standard_normal(n) * 0.05 * a.std()
                a_new = a_new + noise
                a_new = project_nonneg_sum(a_new, S)
                no_improve = 0
                lr = min(lr * 1.5, lr_init * 2.0)

            # Adaptive learning rate
            if no_improve > 200:
                lr = max(lr * 0.7, lr_init * 0.1)

        a = a_new

        if verbose and time.monotonic() - last_print > 10:
            print(f"    adam: t={t}, tau={tau:.4f}, lr={lr:.5f}, best_score={best_score:.6f}")
            last_print = time.monotonic()

    # Final check
    final_mc = np.convolve(a, a).max()
    final_score = 2.0 * n * final_mc / (S * S)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


def grad_lbfgs(x, tau, n):
    """Objective and gradient for L-BFGS-B (score with soft-max smoothing)."""
    S = x.sum()
    if S < 0.01:
        return 1e10, np.zeros_like(x)
    sm_val, grad_sm, _ = compute_softmax_grad(x, tau)
    score = 2.0 * n * sm_val / (S * S)
    grad_score = (2.0 * n / (S * S)) * (grad_sm - 2.0 * sm_val / S)
    return score, grad_score
