import numpy as np

def softmax_weights(c, tau):
    """Compute softmax weights of convolution."""
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    return exp_c / exp_c.sum()

def compute_gradient(a, tau):
    """Compute gradient of tau*logsumexp(conv(a,a)/tau) with respect to a."""
    c = np.convolve(a, a)
    w = softmax_weights(c, tau)
    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
    # This is 2 * correlate(w, a, 'valid')
    grad = 2.0 * np.correlate(w, a, 'valid')
    return grad, c

def project_nonneg_sum(a, S):
    """Project onto {x >= 0, sum(x) = S} using bisection on threshold."""
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s == 0:
        return np.full_like(a, S / len(a))
    if s <= S:
        return a + (S - s) / len(a)
    lo, hi = 0.0, a.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    # Fix rounding
    diff = S - res.sum()
    if abs(diff) > 1e-12:
        idx = res.argmax()
        res[idx] += diff
    return res

def adam_step(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    return a - lr * m_hat / (np.sqrt(v_hat) + eps), m, v

def coordinate_descent(a, n_iters, time_budget, verbose=False):

        a[pick] -= delta

                a[other] = 0
                a[pick] += delta

        else:

            a[other] -= delta

    a_new = a_new / a_new.sum() * S
    return a_new


def run_adam_opt(a0, tau_hi, tau_lo, lr, t_budget, verbose=False, use_mirror=False):
    """
    Adam (or mirror descent) optimization with tau annealing.
    Returns (best_a, best_max_c).
    """
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = (a0.astype(np.float64) / max(a0.sum(), 1e-12) * S).copy()
    a = project_simplex(a, S)

    best_a = a.copy()
    best_mc = float(np.convolve(best_a, best_a).max())

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < t_budget:
        step += 1
        # Anneal tau linearly in log space
        f = min(1.0, step / max(3000, int(t_budget * 5)))
        tau = tau_hi * (tau_lo / tau_hi) ** f

        grad, c = compute_soft_loss_grad(a, tau)

        if use_mirror:
            a_new = mirror_step(a, grad, lr * 0.01, S)
        else:
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** step)
            v_hat = v / (1 - beta2 ** step)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = project_simplex(a_new, S)

        # Periodic evaluation
        if step % 30 == 1:
            mc = float(np.convolve(a_new, a_new).max())
            if mc < best_mc - 1e-12:
                best_mc = mc
                best_a = a_new.copy()

        a = a_new

    if verbose:
        sc = evaluate_sequence(best_a.tolist())
        print(f"    run_adam: steps={step}, max_c={best_mc:.6f}, score={sc:.6f}")
    return best_a, best_mc


def subgradient_refine(a, t_budget, lr_init=0.05, verbose=False):
    """
    Subgradient method: at each step, compute subgradient at the argmax set
    of the convolution and take a step. Uses diminishing step size.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = (a.astype(np.float64) / max(a.sum(), 1e-12) * S).copy()
    a = project_simplex(a, S)

    best_a = a.copy()
    best_mc = float(np.convolve(best_a, best_a).max())

    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < t_budget:
        step += 1
        c = np.convolve(a, a)
        max_c = c.max()
        # Active set: indices where c is close to max
        tol = max_c * 0.01 + 1e-10
        mask = c >= max_c - tol
        n_active = max(1, int(mask.sum()))

        # Subgradient: average of dc_k/da for k in active set
        # dc_k/da_j = 2*a[k-j] (if 0 <= k-j < n)
        # grad_j = (2/n_active) * sum_{k in active} a[k-j]
        w = np.zeros(2 * n - 1)
        w[mask] = 2.0 / n_active
        grad = np.correlate(w, a, 'valid')

        lr = lr_init / np.sqrt(step + 1)
        a = a - lr * grad
        a = project_simplex(a, S)

        mc = float(np.convolve(a, a).max())
        if mc < best_mc - 1e-12:
            best_mc = mc
            best_a = a.copy()

    if verbose:
        print(f"    subgrad: steps={step}, max_c={best_mc:.6f}")
    return best_a


def perturb_search(a, t_budget, verbose=False):
    """
    Random perturbation + acceptance. Tries to nudge the sequence
    in random directions and keeps improvements.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_mc = float(np.convolve(best_a, best_a).max())

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    iters = 0
    scale = 0.05

    while time.monotonic() - t0 < t_budget:
        # Batch of perturbations
        a_trial = best_a.copy()
        n_perturb = int(np.clip(rng.integers(5, 50), 5, n))
        idx = rng.choice(n, size=n_perturb, replace=False)
        a_trial[idx] *= (1.0 + scale * (rng.random(n_perturb) - 0.5))
        a_trial = np.maximum(a_trial, 0.0)
        a_trial = project_simplex(a_trial, S)

        mc = float(np.convolve(a_trial, a_trial).max())
        iters += 1
        if mc < best_mc - 1e-12:
            best_mc = mc
            best_a = a_trial.copy()
            scale = min(scale * 1.02, 0.2)
        else:
            # Occasionally reverse direction or try larger scale
            if iters % 500 == 0:
                scale = max(0.01, scale * 0.9)

    if verbose:
        print(f"    perturb: iters={iters}, max_c={best_mc:.6f}")
    return best_a


def init_chirp(n, seed=0):
    """Chirp-based initialization aiming for flat spectrum."""
    t = np.arange(n, dtype=np.float64) - (n - 1) * 0.5
    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t * t / (2.0 * n))
    rng = np.random.default_rng(seed)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_block(n, seed=0):
    """Block structure: alternating high/low blocks."""
    rng = np.random.default_rng(seed)
    block_size = max(1, n // 20)
    a = np.empty(n)
    for i in range(0, n, block_size):
        val = 0.5 + rng.random()
        a[i:i + block_size] = val
    return np.maximum(a, 0.0)


def init_from_prov(prov, n, seed=0):
    """Resample provided sequence to length n with small perturbation."""
    if len(prov) == n:
        a = prov.copy()
    else:
        x_old = np.linspace(0, 1, len(prov))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, prov)
    rng = np.random.default_rng(seed)
    a += 0.02 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search: multi-phase optimization exploring different lengths,
    initializations, and optimization methods.
    """
    deadline = time.monotonic() + budget_s - 15.0
    best_seq = None
    best_score = np.inf

    # Get provided sequence
    prov = None
    try:
        prov = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        prov = np.maximum(prov, 0.0)
        if prov.size > 0 and prov.sum() >= 0.01:
            pass
        else:
            prov = None
    except Exception:
        prov = None

    # Determine lengths to explore
    lengths = [2000, 3000, 4000]
    if prov is not None and len(prov) not in lengths:
        lengths.append(len(prov))
    lengths = sorted(set(lengths))

    print(f"Lengths to explore: {lengths}")
    print(f"Total time budget: {budget_s - 15:.0f}s")

    # ============================================================
    # Phase 1: Quick exploration (~20% of time)
    # ============================================================
    p1_frac = 0.20
    p1_time = (deadline - time.monotonic()) * p1_frac
    p1_end = time.monotonic() + p1_time

    # Build candidate list
    candidates = []
    for n in lengths:
        if prov is not None:
            candidates.append((n, 'prov', 0))
        candidates.append((n, 'chirp', 0))
        candidates.append((n, 'chirp', 1))
        candidates.append((n, 'block', 0))

    # Sort: prefer provided, then shorter
    def sort_key(x):
        prio = 0 if x[1] == 'prov' else 1
        return (prio, x[0])
    candidates.sort(key=sort_key)

    t_per = p1_time / max(1, len(candidates))
    print(f"\nPhase 1: {len(candidates)} candidates, ~{t_per:.0f}s each")

    for i, (n, itype, sd) in enumerate(candidates):
        tb = p1_end - time.monotonic()
        if tb < 20:
            break
        tb = min(t_per * 1.5, tb)

        if itype == 'prov' and prov is not None:
            a0 = init_from_prov(prov, n, sd)
        elif itype == 'chirp':
            a0 = init_chirp(n, sd)
        elif itype == 'block':
            a0 = init_block(n, sd)
        else:
            a0 = np.ones(n)

        a_r, mc = run_adam_opt(a0, 1.0, 0.1, 0.5, tb)
        sc = evaluate_sequence(a_r.tolist())
        print(f"  [{i+1}/{len(candidates)}] n={n} {itype}_{sd}: max_c={mc:.6f} score={sc:.6f}")

        if sc < best_score:
            best_score = sc
            best_seq = a_r.copy()
            print(f"  *** NEW BEST: {best_score:.6f}")

    # ============================================================
    # Phase 2: Focused multi-round optimization (~50% of time)
    # ============================================================
    rem = deadline - time.monotonic()
    if rem > 60 and best_seq is not None:
        n_best = len(best_seq)
        print(f"\nPhase 2: polishing n={n_best}, {rem:.0f}s remaining")

        # Define rounds: (tau_hi, tau_lo, lr, time_fraction, use_mirror)
        rounds = [
            (2.0, 0.2, 0.5, 0.30, False),
            (1.0, 0.1, 0.3, 0.25, False),
            (0.5, 0.05, 0.2, 0.20, False),
            (0.3, 0.02, 0.1, 0.15, False),
            (1.0, 0.05, 0.3, 0.10, True),   # mirror descent round
        ]

        for ri, (tau_hi, tau_lo, lr, frac, mir) in enumerate(rounds):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            tb = min(rem * frac, rem - 30)
            label = "mirror" if mir else "adam"
            print(f"  Round {ri+1}: tau=[{tau_hi},{tau_lo}] lr={lr} {label} t={tb:.0f}s")

            a_r, mc = run_adam_opt(best_seq, tau_hi, tau_lo, lr, tb, use_mirror=mir)
            sc = evaluate_sequence(a_r.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = a_r.copy()
                print(f"  *** improved to {best_score:.6f}")
            else:
                print(f"  score={sc:.6f} (no improvement, keeping prev)")

    # ============================================================
    # Phase 3: Subgradient + perturbation refinement (~20% of time)
    # ============================================================
    rem = deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        print(f"\nPhase 3: subgradient + perturbation, {rem:.0f}s remaining")

        # Subgradient refinement
        sg_time = rem * 0.4
        print(f"  Subgradient: {sg_time:.0f}s")
        a_r = subgradient_refine(best_seq, sg_time, lr_init=0.03, verbose=True)
        sc = evaluate_sequence(a_r.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a_r.copy()
            print(f"  *** subgrad improved to {best_score:.6f}")

        # Perturbation search
        rem = deadline - time.monotonic()
        if rem > 20:
            pt_time = rem * 0.8
            print(f"  Perturbation: {pt_time:.0f}s")
            a_r = perturb_search(best_seq, pt_time, verbose=True)
            sc = evaluate_sequence(a_r.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = a_r.copy()
                print(f"  *** perturb improved to {best_score:.6f}")

    # ============================================================
    # Phase 4: Final quick Adam polish
    # ============================================================
    rem = deadline - time.monotonic()
    if rem > 15 and best_seq is not None:
        print(f"\nPhase 4: final polish, {rem:.0f}s remaining")
        a_r, mc = run_adam_opt(best_seq, 0.5, 0.01, 0.1, rem * 0.8)
        sc = evaluate_sequence(a_r.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a_r.copy()
            print(f"  *** final improved to {best_score:.6f}")

    # Fallback
    if best_seq is None:
        n = 3000
        best_seq = np.ones(n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"Fallback: uniform n={n}, score={best_score:.6f}")

    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
