"""
LP-guided descent for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
At each iteration, solves a linear program to find the direction that maximally
increases the sequence sum while keeping the cross-convolution bounded, then
performs golden-section line search. Combines multiple initializations,
sequence lengths, and perturbation strategies for robustness.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def objective_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build matrix A where (A @ g) = f * g (linear convolution)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            A[k, i_start:i_end + 1] = f[k - i_end:k - i_start + 1]
    return A


def solve_lp(f, M):
    """Solve max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def golden_line_search(a, g, current_obj):

        a_try /= s
        return objective_normalized(a_try)

        return 1.0, obj_full

    # Golden section on [0.05, 0.95]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = lo + (1.0 - gr) * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    f_c = eval_t(c_pt)
    f_d = eval_t(d_pt)
    best_t = c_pt if f_c < f_d else d_pt

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(('uniform', np.ones(n) / n))

    # Gaussians
    for sf in [0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits.append((f'gauss_{sf}', g))

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits.append(('ramp', ramp))

    # Bimodal (edges)
    for frac in [0.2, 0.3, 0.4]:
        bi = np.zeros(n)
        m = max(1, int(n * frac))
        bi[:m] = 1.0
        bi[-m:] = 1.0
        bi /= bi.sum()
        inits.append((f'bimodal_{frac}', bi))

    # Cosine
    cos_b = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_b /= cos_b.sum()
    inits.append(('cosine', cos_b))

    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append((f'random_a{alpha}', r))

    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            pp /= pp.sum()
            inits.append(('parent', pp))

    return inits

    lengths = [64, 96, 128, 160, 192, 256, 384]
    scan_results = []

    phase1_deadline = time.monotonic() + min(300.0, (deadline - time.monotonic()) * 0.2)

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        inits = generate_initializations(n, rng, parent)
        # Limit number of inits for scan
        inits = inits[:8]

        time_per = max(5.0, (phase1_deadline - time.monotonic()) / max(1, len(inits) * 1))

        for name, a0 in inits:
            if time.monotonic() > phase1_deadline:
                break
            t_alloc = min(30.0, max(5.0, phase1_deadline - time.monotonic() - 5))
            a_opt, obj_opt = optimize_lp(a0, time_limit=t_alloc, seed=seed, verbose=False)
            seq = a_opt.tolist()
            score = float(evaluate_sequence(seq))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                print(f"  Scan n={n}, {name}: score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"\nPhase 1 done. Best so far: {best_score:.6f}")

    # Phase 2: Deep optimization on top candidates
    print("Phase 2: Deep optimization...")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, name, a_opt, score))
            seen.add(key)
        if len(top_configs) >= 10:
            break

    # Always include parent if available
    if parent is not None:
        n_p = min(len(parent), 256)
        pp = parent[:n_p].copy()
        pp = np.maximum(pp, 0)
        pp /= pp.sum()
        if not any(c[2] is pp for c in top_configs):
            top_configs.append((n_p, 'parent_deep', pp, best_score))

    # Also add some fresh random inits for diversity
    for n in [128, 256]:
        r = rng.dirichlet(np.ones(n) * 1.0)
        top_configs.append((n, 'random_deep', r, best_score))

    # Shuffle to diversify
    rng.shuffle(top_configs)

    phase2_deadline = time.monotonic() + min(1500.0, (deadline - time.monotonic()) * 0.6)
    time_per_deep = max(30.0, (phase2_deadline - time.monotonic()) / max(1, len(top_configs)))

    for i, (n, name, a0, prev_score) in enumerate(top_configs):
        if time.monotonic() > phase2_deadline - 30:
            break
        t_alloc = min(time_per_deep, max(20.0, phase2_deadline - time.monotonic() - 30))
        if t_alloc < 10:
            break

        # Multiple seeds for diversity
        for s in [seed + i * 100, seed + i * 100 + 1]:
            if time.monotonic() > phase2_deadline - 10:
                break
            t_single = t_alloc / 2.0
            a_opt, obj_opt = optimize_lp(a0, time_limit=t_single, seed=s, verbose=False)
            seq = a_opt.tolist()
            score = float(evaluate_sequence(seq))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep n={n}, {name}, seed={s}: score={score:.6f}")

    print(f"\nPhase 2 done. Best so far: {best_score:.6f}")

    # Phase 3: Polish the best
    if best_sequence is not None:
        print("Phase 3: Polish...")
        remaining = deadline - time.monotonic() - 10
        n_best = len(best_sequence)

        # Run multiple polish attempts with different seeds
        n_polish = min(6, max(2, int(remaining / 60)))
        time_per_polish = remaining / n_polish

        for i in range(n_polish):
            if time.monotonic() > deadline - 10:
                break
            a_polish = best_sequence + 0.001 * np.random.default_rng(seed + 9000 + i).standard_normal(n_best)
            a_polish = np.maximum(a_polish, 0)
            a_polish /= a_polish.sum()

            a_opt, obj_opt = optimize_lp(a_polish, time_limit=time_per_polish,
                                         seed=seed + 9000 + i, verbose=False)
            seq = a_opt.tolist()
            score = float(evaluate_sequence(seq))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish {i}: score={score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (length={len(best_sequence)})")
    return best_sequence


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Uses LP-guided descent with golden-section line search
    to minimize 2n*max(conv(a,a))/(sum(a))^2. Multiple phases: scan, deep
    optimization, and polish. Explores multiple sequence lengths and
    initializations.
    """
    result = run_search(seed=seed, budget_s=budget_s)
    return [float(x) for x in result]
