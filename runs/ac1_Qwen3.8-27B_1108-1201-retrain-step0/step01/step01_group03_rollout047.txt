"""
Optimizes non-negative sequence to minimize 2*n*max(autocorr)/(sum)^2.
Uses FFT-based autocorrelation, soft-attention gradient, Adam optimizer,
and LP-based refinement. Explores multiple starting points and lengths.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog


def _next_pow2(n):
    return 1 << (n - 1).bit_length()


def _autocorr_fft(a):
    """Full autocorrelation via FFT: c[m] = sum_j a[j]*a[m-j]."""
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)
    return C[:size]


def _compute_obj(a):
    """Compute 2*n*max(autocorr)/(sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    c = _autocorr_fft(a)
    max_c = float(c.max())
    return 2.0 * n * max_c / (S * S)


def _soft_grad(a, tau):
    """
    Gradient of soft-max(autocorr) w.r.t. a.
    Uses temperature tau for tempered attention.
    Returns (gradient, c, max_c, w).
    """
    n = len(a)
    c = _autocorr_fft(a)
    max_c = c.max()
    
    # Tempered softmax
    c_shifted = c - max_c
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # grad[j] = 2 * sum_{k=0}^{min(n-1, 2n-2-j)} w[j+k] * a[k]
    grad = np.zeros(n)
    for j in range(n):
        k_end = min(n, 2 * n - 1 - j)
        if k_end > 0:
            grad[j] = 2.0 * np.dot(a[:k_end], w[j:j + k_end])
    
    return grad, c, max_c, w


def _hard_grad(a):
    """Gradient at the hard maximum of the autocorrelation."""
    n = len(a)
    c = _autocorr_fft(a)
    m_star = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        k = m_star - j
        if 0 <= k < n:
            grad[j] = 2.0 * a[k]
    return grad, c, float(c[m_star])


def _adam_step(a, grad, m_v, v_v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update step. m_v and v_v are running averages (modified in place)."""
    m_v[:] = beta1 * m_v + (1 - beta1) * grad
    v_v[:] = beta2 * v_v + (1 - beta2) * grad * grad
    m_hat = m_v / (1 - beta1 ** t)
    v_hat = v_v / (1 - beta2 ** t)
    a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a


def _optimize_adam(a0, deadline, max_iters=5000, lr_init=0.05, tau_init=5.0, tau_end=50.0):

def _next_pow2(n):
    """Next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _autocorr_fft(a):
    """Full autocorrelation c[m] = sum_j a[j]*a[m-j] via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)
    return C[:size]


def _compute_obj(a):
    """Compute 2*n*max(autocorr)/(sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    c = _autocorr_fft(a)
    max_c = float(c.max())
    return 2.0 * n * max_c / (S * S)


def _soft_gradient(a, tau):
    """
    Compute gradient of soft-max(autocorr) w.r.t. a using tempered softmax.
    Returns (grad, autocorr, max_c, weights).
    grad[j] = 2 * sum_k a[k] * w[j+k] for valid k.
    """
    n = len(a)
    c = _autocorr_fft(a)
    max_c = float(c.max())

    # Tempered softmax weights
    c_shifted = c - max_c
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum

    grad = np.zeros(n)

                improved = True

        c = _autocorr_fft(a)
        M = float(c.max())

    return best, best_score


def _optimize_hybrid(a0, deadline, max_grad_iters=4000):
    """Full hybrid optimization: Adam gradient + LP refinement."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    target_S = math.sqrt(2.0 * n)
    a = _project_nonneg_fixedsum(a, target_S)

    # Phase 1: Adam gradient descent
    a, best_score = _optimize_adam(a, deadline - 60.0,
                                    max_iters=max_grad_iters, lr_init=0.04)

    # Phase 2: LP refinement
    best, best_score = _optimize_with_lp(a, deadline, target_S, max_lp_iters=60)

    return best, best_score


def _fourier_seq(n, freqs, phases, amplitudes):
    """Generate non-negative sequence from sum of cosines."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n) * 0.5
    for f, p, amp in zip(freqs, phases, amplitudes):
        a += amp * np.cos(2.0 * np.pi * f * i / n + p)
    return np.maximum(a, 0.01)


def _interpolate_seq(a, factor):
    """Upsample a sequence by inserting values (linear interpolation)."""
    n = len(a)
    new_n = n * factor
    old_idx = np.arange(n)
    new_idx = np.arange(new_n) * (n / new_n)
    return np.interp(new_idx, old_idx, a)


def _perturb_seq(a, sigma, rng):
    """Multiplicative perturbation preserving sum."""
    target_sum = a.sum()
    noise = np.maximum(1.0 + sigma * rng.standard_normal(len(a)), 0.05)
    a_new = a * noise
    a_new = _project_nonneg_fixedsum(a_new, target_sum)
    return a_new


def _record_result(a, score, best_score, best_seq, label):
    """Print and update best if improved. Returns (best_seq, best_score)."""
    if score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: score={best_score:.6f}, n={len(a)}")
    else:
        print(f"  [{label}]: score={score:.6f}, n={len(a)} (best={best_score:.6f})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: FFT-accelerated autocorrelation, Adam with soft-attention gradient,
    LP refinement, multi-resolution starts, and perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    # --- Build diverse starting points ---
    starts = []  # list of (sequence, label)

    # From provided global
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)):
            for L in [128, 256, 512]:
                if L <= len(gs):
                    starts.append((gs[:L].copy(), f"global_{L}"))
            if 16 <= len(gs) <= 2048:
                starts.append((gs.copy(), f"global_{len(gs)}"))
            # Multi-resolution: interpolate to larger sizes
            if len(gs) >= 32:
                base = gs[:min(len(gs), 256)]
                starts.append((_interpolate_seq(base, 2), f"interp_512"))
                starts.append((_interpolate_seq(base, 4), f"interp_1024"))
    except NameError:
        pass

    # Random starts
    for n in [128, 256, 512, 1024]:
        starts.append((rng.random(n) * 0.4 + 0.3, f"rand_{n}"))

    # Structured: Fourier-based
    for n in [256, 512, 1024]:
        for trial in range(2):
            n_freqs = rng.integers(3, 8)
            freqs = rng.choice(np.arange(1, n // 4 + 1), size=n_freqs, replace=False)
            phases = rng.uniform(0, 2 * np.pi, n_freqs)
            amps = rng.uniform(0.1, 0.4, n_freqs)
            starts.append((_fourier_seq(n, freqs, phases, amps), f"fourier_{n}_{trial}"))

    # Structured: cosine taper
    for n in [256, 512, 1024]:
        i = np.arange(n, dtype=np.float64)
        starts.append((0.4 + 0.6 * (0.5 + 0.5 * np.cos(2 * np.pi * i / n)),
                       f"cos_taper_{n}"))

    # Structured: modulated
    for n in [256, 512]:
        i = np.arange(n, dtype=np.float64)
        starts.append((np.abs(np.sin(2.0 * np.pi * i * 2.0 / n)) + 0.4,
                       f"mod_{n}"))
        starts.append((0.3 + 0.7 * np.abs(np.sin(np.pi * i * i / (2.0 * n))),
                       f"chirp_{n}"))

    # Sort shorter first
    starts.sort(key=lambda x: len(x[0]))

    print(f"Total starting points: {len(starts)}, budget={budget_s}s")
    print(f"Starting optimization at t=0...")

    # --- Main optimization loop ---
    # Allocate time per start based on length
    for idx, (a0, label) in enumerate(starts):
        if time.monotonic() > deadline - 50.0:
            break
        elapsed = time.monotonic() - t_start
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break

        # Allocate time proportionally: shorter sequences get less time per start
        n = len(a0)
        time_per_start = min(remaining * 0.4, max(30.0, 150.0 * (n / 512.0)))
        local_deadline = time.monotonic() + time_per_start

        t0 = time.monotonic()
        seq, score = _optimize_hybrid(a0, local_deadline,
                                       max_grad_iters=3000 if n <= 512 else 2000)
        elapsed_s = time.monotonic() - t0
        best_seq, best_score = _record_result(seq, score, best_score, best_seq,
                                               f"{label} t={elapsed_s:.0f}s")

    # --- Refinement phase on best ---
    if best_seq is not None and time.monotonic() < deadline - 30.0:
        print(f"\n=== Refinement phase === (current best={best_score:.6f})")

        for pass_i in range(10):
            if time.monotonic() > deadline - 15.0:
                break

            local_dl = min(deadline - 5.0, time.monotonic() + 120.0)

            # Direct refinement
            seq, score = _optimize_hybrid(best_seq, local_dl, max_grad_iters=2000)
            improved = score < best_score
            if improved:
                best_seq, best_score = _record_result(seq, score, best_score, best_seq,
                                                       f"refine_{pass_i}")
            else:
                # Try perturbation
                sigma = 0.15 * (1.0 - pass_i / 12.0)
                perturbed = _perturb_seq(best_seq, sigma, rng)
                seq2, score2 = _optimize_hybrid(perturbed, local_dl, max_grad_iters=1500)
                if score2 < best_score:
                    best_seq, best_score = _record_result(seq2, score2, best_score, best_seq,
                                                           f"perturb_{pass_i}")
                else:
                    # Try Adam-only with different params
                    seq3, score3 = _optimize_adam(best_seq, local_dl,
                                                   max_iters=3000, lr_init=0.02,
                                                   tau_init=10.0, tau_end=80.0)
                    if score3 < best_score:
                        best_seq, best_score = _record_result(seq3, score3, best_score, best_seq,
                                                               f"adam2_{pass_i}")
                    else:
                        if pass_i > 3:
                            break

    # --- Final: try extending best to larger n ---
    if best_seq is not None and time.monotonic() < deadline - 40.0:
        print(f"\n=== Extension phase === (n={len(best_seq)})")
        best_n = len(best_seq)
        for target_n in [best_n * 2, best_n * 4]:
            if target_n > 4096 or time.monotonic() > deadline - 30.0:
                break
            ext = _interpolate_seq(best_seq, target_n // best_n)
            local_dl = min(deadline - 5.0, time.monotonic() + 180.0)
            seq, score = _optimize_hybrid(ext, local_dl, max_grad_iters=2000)
            best_seq, best_score = _record_result(seq, score, best_score, best_seq,
                                                   f"extend_{target_n}")

    # --- Final validation ---
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"\n=== FINAL ===")
        print(f"n={len(best_seq)}, internal_score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence score: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Fallback: uniform 256")
    return [1.0] * 256
