"""
Hybrid optimizer for minimizing normalized autocorrelation peak of non-negative sequences.
Combines: (1) parameterized smooth profile initialization via L-BFGS-B,
(2) log-sum-exp smoothed gradient with Nesterov momentum and backtracking,
(3) subgradient descent with adaptive step size and heavy-ball momentum,
(4) LP-based obstacle space direction finding for large jumps.
Uses mode-switching to alternate between methods based on progress.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) for a sequence assumed to have sum=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _hankel_matvec(p, a):
    """Compute P @ a where P[i,j] = p[i+j] (Hankel matrix)."""
    n = len(a)
    # p has length 2n-1
    # Use sliding window view for BLAS-efficient matvec
    Q = np.lib.stride_tricks.sliding_window_view(p, n)
    return Q @ a


def _smoothed_obj(a, temp):
    """Log-sum-exp smoothed max of convolution."""
    b = np.convolve(a, a)
    b_max = np.max(b)
    return b_max + np.log(np.sum(np.exp(temp * (b - b_max)))) / temp


def _smoothed_grad(a, temp):
    """Gradient of log-sum-exp smoothed max of convolution."""
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    p = np.exp(temp * (b - b_max))
    s = p.sum()
    if s < 1e-300:
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
    else:
        p /= s
    return 2.0 * _hankel_matvec(p, a)


def _subgrad_topM(a, M, sharpness=5.0):
    """Subgradient using top-M active lags with exponential weighting."""
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    if M >= len(b):
        idx = np.arange(len(b))
        wts = np.ones(len(b)) / len(b)
    else:
        idx = np.argpartition(b, -M)[-M:]
        b_sel = b[idx]
        denom = max(b_max, 1e-15)
        wts = np.exp(-sharpness * (b_max - b_sel) / denom)
        wts /= wts.sum()
    p = np.zeros(len(b))
    p[idx] = wts
    return 2.0 * _hankel_matvec(p, a)


def _lp_direction(a, M_threshold, max_time=2.0):
    """
    Solve the obstacle space LP to find a direction that maximizes sum(g)
    subject to (a*g)[k] <= M for active lags and g >= 0.
    Returns the direction g (normalized to sum=1) or None if LP fails.
    """
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    M = b_max + M_threshold

    # Select active lags (those with b[k] >= b_max - M_threshold)
    active_mask = b >= (b_max - M_threshold)
    active_idx = np.nonzero(active_mask)[0]
    K = len(active_idx)

    if K == 0:
        return None

    # Build constraint matrix A_active: K x n
    # A_active[k,i] = a[k-i] if 0 <= k-i < n, else 0
    # Use efficient construction
    A = np.zeros((K, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[row, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    # Solve: max sum(g) s.t. A @ g <= M, g >= 0
    # linprog: min -sum(g) s.t. A @ g <= M, -g <= 0
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'time_limit': max_time})

    if result.status == 0:
        g = result.x
        S = g.sum()
        if S > 1e-10:
            return g / S
    return None


def _make_smooth_profile(n, params):
    """Create profile: a[i] = max(0, 1 + sum_k params[k]*cos((k+1)*pi*i/(n-1))), normalized."""

    x = np.pi * i / (n - 1)
    a = np.ones(n)
    for k, s in enumerate(params):
        a += s * np.cos((k + 1) * x)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a

def _nesterov_smoothed(a, deadline, temp_init=3.0, temp_final=50.0,
                        nesterov_mu=0.9, lr_init=None, n):
    """
    Nesterov-accelerated gradient on log-sum-exp smoothed objective.
    Temperature anneals from temp_init to temp_final.
    """
    if lr_init is None:
        lr_init = 1.0 / (2.0 * n * temp_init)

    a = a.copy()
    a_prev = a.copy()
    t_k = 1.0
    lr = lr_init
    best = a.copy()
    best_score = _score_norm(best)
    temp = temp_init
    total_iters = 10000
    no_improve = 0
    max_no_improve = 200

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Anneal temperature
        frac = min(1.0, it / total_iters)
        temp = temp_init + (temp_final - temp_init) * frac

        # Nesterov extrapolation
        t_prev = t_k
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev))
        momentum = (t_prev - 1.0) / t_k
        y = a + momentum * (a - a_prev)

        # Gradient at y
        grad = _smoothed_grad(y, temp)
        # Center for sum constraint
        grad -= np.mean(grad)

        # Backtracking line search
        a_new = _project_simplex(y - lr * grad)
        f_new = _smoothed_obj(a_new, temp)
        f_y = _smoothed_obj(y, temp)
        backtrack_count = 0
        while f_new > f_y + 1e-10 and backtrack_count < 10:
            lr *= 0.5
            a_new = _project_simplex(y - lr * grad)
            f_new = _smoothed_obj(a_new, temp)
            backtrack_count += 1

        a_prev = a
        a = a_new

        # Track best
        s = _score_norm(a)
        if s < best_score - 1e-10:
            best_score = s
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            # Restart with smaller lr
            lr *= 0.5
            a_prev = a.copy()
            t_k = 1.0
            no_improve = 0
            if lr < 1e-12:
                break

    return best, best_score


def _momentum_subgrad(a, deadline, n, M=30, mu=0.85, sharpness=5.0,
                      lr_init=None, max_no_improve=500):
    """
    Subgradient descent with heavy-ball momentum and adaptive step size.
    """
    if lr_init is None:
        lr_init = 1.0 / np.sqrt(n)

    a = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best = a.copy()
    best_score = _score_norm(best)
    no_improve = 0
    adapt_lr = lr_init

    while time.monotonic() < deadline:
        # Compute subgradient
        grad = _subgrad_topM(a, M, sharpness)
        grad -= np.mean(grad)

        # Heavy ball momentum
        v = mu * v + grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        # Step
        a_new = _project_simplex(a - lr * v / v_norm)
        s_new = _score_norm(a_new)

        if s_new < best_score - 1e-12:
            best_score = s_new
            best = a_new.copy()
            no_improve = 0
            a = a_new
            # Occasionally increase lr slightly if making good progress
            if no_improve == 0 and lr < adapt_lr * 2:
                lr = min(lr * 1.01, adapt_lr * 2)
        else:
            no_improve += 1
            if s_new < _score_norm(a):
                a = a_new  # Accept if better than current even if not best
            if no_improve > 50:
                lr *= 0.9
            if no_improve > max_no_improve:
                # Reheat: perturb and reset momentum
                v = np.zeros(n)
                lr = adapt_lr
                no_improve = 0
                # Small perturbation
                noise = np.random.randn(n) * 0.01
                a = _project_simplex(a + noise)

    return best, best_score


def _lp_refine(a, deadline, n, n_lp_steps=20):
    """
    Use LP-based obstacle space direction finding to make large improvements.
    """
    a = a.copy()
    best = a.copy()
    best_score = _score_norm(best)

    for step in range(n_lp_steps):
        if time.monotonic() >= deadline:
            break

        # Solve LP
        g = _lp_direction(a, M_threshold=0.01 * np.max(np.convolve(a, a)), max_time=1.0)
        if g is None:

"""
Hybrid optimizer for minimizing normalized autocorrelation peak of non-negative sequences.
Combines: (1) parameterized smooth profile initialization via L-BFGS-B,
(2) log-sum-exp smoothed gradient with Nesterov momentum and backtracking,
(3) subgradient descent with adaptive step size and heavy-ball momentum,
(4) LP-based obstacle space direction finding for large jumps.
Uses mode-switching to alternate between methods based on progress.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) for a sequence assumed to have sum=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _hankel_matvec(p, a):
    """Compute P @ a where P[i,j] = p[i+j] (Hankel matrix), p has length 2n-1."""
    n = len(a)
    Q = np.lib.stride_tricks.sliding_window_view(p, n)
    return Q @ a


def _smoothed_obj(a, temp):
    """Log-sum-exp smoothed max of convolution."""
    b = np.convolve(a, a)
    b_max = np.max(b)
    return b_max + np.log(np.sum(np.exp(temp * (b - b_max)))) / temp


def _smoothed_grad(a, temp):
    """Gradient of log-sum-exp smoothed max of convolution."""
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    exponents = temp * (b - b_max)
    # Clip for numerical stability
    exponents = np.clip(exponents, -500, 0)
    p = np.exp(exponents)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
    else:
        p /= s
    return 2.0 * _hankel_matvec(p, a)


def _subgrad_topM(a, M, sharpness=5.0):
    """Subgradient using top-M active lags with exponential weighting."""
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    if M >= len(b):
        idx = np.arange(len(b))
        wts = np.ones(len(b)) / len(b)
    else:
        idx = np.argpartition(b, -M)[-M:]
        b_sel = b[idx]
        denom = max(b_max, 1e-15)
        wts = np.exp(-sharpness * (b_max - b_sel) / denom)
        wts /= wts.sum()
    p = np.zeros(len(b))
    p[idx] = wts
    return 2.0 * _hankel_matvec(p, a)


def _lp_direction(a, M_threshold, max_time=2.0):
    """
    Solve the obstacle space LP to find a direction g that maximizes sum(g)
    subject to (a*g)[k] <= b_max + M_threshold for active lags and g >= 0.
    Returns g normalized to sum=1, or None.
    """
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    M = b_max + M_threshold

    # Select active lags
    active_mask = b >= (b_max - M_threshold)
    active_idx = np.nonzero(active_mask)[0]
    K = len(active_idx)

    if K == 0 or K > 500:
        # If too many active constraints, reduce
        if K > 500:
            top = np.argpartition(b, -500)[-500:]
            active_idx = top
            K = 500
        else:
            return None

    # Build constraint matrix A_active: K x n
    A = np.zeros((K, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[row, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    # Solve: max sum(g) s.t. A @ g <= M, g >= 0
    c = -np.ones(n)
    try:
        result = linprog(c, A_ub=A, b_ub=np.full(K, M),
                         bounds=[(0, None)] * n, method='highs',
                         options={'time_limit': max_time})
        if result.status == 0:
            g = result.x
            S = g.sum()
            if S > 1e-10:
                return g / S
    except Exception:
        pass
    return None
