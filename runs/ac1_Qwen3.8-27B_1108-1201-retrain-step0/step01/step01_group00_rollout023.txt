import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def subgradient_descent(n, time_limit, rng):
    # Initialize with linear gradient
    t = np.linspace(0, 1, n)
    a = 1.0 + 0.5 * t  # linear from 1 to 1.5
    a = a / np.sum(a)  # normalize to sum = 1
    
    best_score = np.inf
    best_a = a.copy()
    
    # Compute initial score
    conv = fft_convolve(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    score = 2.0 * n * max_conv / (sum_a ** 2)
    best_score = score
    
    lr = 0.01
    momentum = np.zeros(n)
    beta = 0.9
    t0 = time.monotonic()
    step = 0
    best_ever = score
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        
        # Subgradient: d/da[j] conv[a,a][max_idx]
        g = np.zeros(n)
        for j in range(n):
            k1 = max_idx - j
            k2 = 2*(n-1) - max_idx - j
            if 0 <= k1 < n:
                g[j] += a[k1]
            if 0 <= k2 < n:
                g[j] += a[k2]
        
        # Projected gradient step on the simplex
        step += 1
        lr_t = lr / np.sqrt(step)
        momentum = beta * momentum + (1-beta) * g
        a_candidate = a - lr_t * momentum
        a_candidate = np.maximum(a_candidate, 0)
        a_candidate = a_candidate / np.sum(a_candidate)
        a = a_candidate
        
        # Evaluate
        conv = fft_convolve(a, a)
        max_val = np.max(conv)
        sum_a = np.sum(a)
        score = 2.0 * n * max_val / (sum_a ** 2)
        
        if score < best_ever:
            best_ever = score
            best_a = a.copy()
        
        if step % 1000 == 0:
            print(f"  step={step}, score={score:.6f}, best={best_ever:.6f}, t={time.monotonic()-t0:.1f}s")
        
        # Adaptive learning rate
        if step % 5000 == 0:
            if score > best_ever * 1.1:
                lr *= 0.5
            elif score < best_ever * 0.99:
                lr *= 1.1
    
    return best_a, best_ever
