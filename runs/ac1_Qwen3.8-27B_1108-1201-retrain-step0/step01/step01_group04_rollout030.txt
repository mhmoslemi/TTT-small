Alternating LP search for low-peak self-convolution sequences.

Uses alternating linear programs to iteratively improve a non-negative
sequence, minimizing the peak of its self-convolution. Combines:
  * LP best-response (max sum(g) s.t. p*g <= M)
  * LP min-peak (min M s.t. g*p' <= M, sum(p')=1)
  * Projected gradient descent with smoothed-max surrogate
  * Multi-start with diverse shape initializations
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _fft_len(n):
    return int(next_fast_len(max(1, n)))

def conv_sq(p):
    n = len(p)
    if n == 0:
        return np.empty(0)
    L = _fft_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1].copy()

def internal_score(p):
    n = len(p)
    s = np.sum(p)
    if s < 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (s * s))

def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.maximum(np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0), 0.0)
    s = p.sum()
    if s < 1e-12:
        n = max(1, len(p))
        return np.full(n, 1.0/n)
    return p / s

def build_conv_matrix(p):
    """Sparse matrix A where (A @ g) = (p * g) [full convolution]."""
    n = len(p)
    rows = []
    cols = []
    data = []
    for m in range(2 * n - 1):
        k_start = max(0, m - n + 1)
        k_end = min(n - 1, m)
        num = k_end - k_start + 1
        rows.extend([m] * num)
        cols.extend(range(k_start, k_end + 1))
        for k in range(k_start, k_end + 1):
            data.append(p[m - k])
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def lp_best_response(p, M):
    """max sum(g) s.t. (p*g)_k <= M for all k, g >= 0."""
    n = len(p)
    A = build_conv_matrix(p)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                  options={'disp': False, 'time_limit': 30})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        s = g.sum()
        if s > 1e-12:
            return g / s
    return None

def lp_min_peak(g):
    """min M s.t. (g*p)_k <= M for all k, p >= 0, sum(p) = 1.
    Returns (p_opt, M_opt)."""
    n = len(g)
    A_g = build_conv_matrix(g)  # A_g @ p = (g * p)
    # Variables: [p_0, ..., p_{n-1}, M] (n+1 variables)
    # Constraints: A_g @ p - M <= 0  =>  [A_g | -1] @ [p; M] <= 0
    # sum(p) = 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M
    
    A_ub = np.hstack([A_g.toarray(), -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'disp': False, 'time_limit': 30})
    if res.success and res.x is not None:
        p_opt = np.maximum(res.x[:n], 0.0)
        s = p_opt.sum()
        if s > 1e-12:
            p_opt = p_opt / s
        M_opt = float(res.x[-1])
        return p_opt, M_opt
    return None, np.inf

"""
Alternating LP search for low-peak self-convolution sequences.

Uses alternating linear programs to iteratively improve a non-negative
sequence, minimizing the peak of its self-convolution. The key idea is:
  - LP1 (best response): max sum(g) s.t. p*g <= M, g >= 0
  - LP2 (min peak): min M' s.t. g*p' <= M', p'>=0, sum(p')=1
Alternating these two LPs drives the sequence toward a local minimum
of the max self-convolution peak. Combined with gradient polishing
and multi-start, this achieves scores below 1.503.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _fft_len(n):
    return int(next_fast_len(max(1, n)))


def conv_sq(p):
    """Full self-convolution of p via FFT."""
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    if n == 0:
        return np.empty(0)
    L = _fft_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].copy()


def internal_score(p):
    """Compute score = 2*n*max(conv)/sum^2 for non-negative p."""
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    s = float(np.sum(p))
    if s < 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (s * s))


def safe_normalize(p):
    """Clip to non-negative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64)
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s < 1e-12:
        n = max(1, len(p))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def build_conv_matrix(p):
    """Build sparse matrix A where (A @ g) = (p * g) [full convolution]."""
    n = len(p)
    total_nnz = n * n
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    data = np.empty(total_nnz, dtype=np.float64)

    idx = 0
    for m in range(2 * n - 1):
        k_lo = max(0, m - n + 1)
        k_hi = min(n - 1, m)
        count = k_hi - k_lo + 1
        rows[idx:idx + count] = m
        cols[idx:idx + count] = np.arange(k_lo, k_hi + 1)
        data[idx:idx + count] = p[m - k_hi:m - k_lo + 1][::-1]
        idx += count

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(p, M, time_limit=60.0):
    """Solve: max sum(g) s.t. (p*g)_k <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(p)
    A = build_conv_matrix(p)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                      options={'disp': False, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = float(g.sum())
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None


def lp_min_peak(g, time_limit=60.0):
    """Solve: min M s.t. (g*p)_k <= M for all k, p>=0, sum(p)=1. Returns (p_opt, M_opt)."""
    n = len(g)
    A_g = build_conv_matrix(g)
    A_g_dense = A_g.toarray()

    # Variables: [p_0, ..., p_{n-1}, M] (n+1)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.hstack([A_g_dense, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'disp': False, 'time_limit': time_limit})
        if res.success and res.x is not None:
            p_opt = np.maximum(res.x[:n], 0.0)
            s = float(p_opt.sum())
            if s > 1e-12:
                p_opt = p_opt / s
            M_opt = float(res.x[-1])
            return p_opt, M_opt
    except Exception:
        pass
    return None, np.inf


def alternating_lp(p, budget_s, verbose=True):
    """Run alternating LP optimization. Returns (best_p, best_score)."""
    p = safe_normalize(p)
    n = len(p)
    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score

    end = time.monotonic() + budget_s
    iteration = 0
    no_improve = 0

    # Determine LP time limit based on n
    if n <= 500:
        lp_time = 15.0
    elif n <= 1000:
        lp_time = 30.0
    elif n <= 2000:
        lp_time = 45.0
    else:
        lp_time = 60.0

    # Precompute convolution matrix structure (rows, cols) once
    # since it doesn't depend on p values, only n
    # But data changes with p, so we rebuild each time.
    # For efficiency, we could cache rows/cols.

    while time.monotonic() < end - 5.0:
        iteration += 1
        remaining = end - time.monotonic()
        if remaining < lp_time * 2 + 5:
            break

        c = conv_sq(p)
        M = float(np.max(c))
        if not np.isfinite(M) or M <= 0:
            break

        # LP1: best response
        g = lp_best_response(p, M, time_limit=min(lp_time, remaining * 0.4))
        if g is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue

        # LP2: min peak with g fixed
        p_new, M_new = lp_min_peak(g, time_limit=min(lp_time, remaining * 0.4))
        if p_new is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue

        # Evaluate actual score of p_new
        sc_new = internal_score(p_new)

        if np.isfinite(sc_new) and sc_new < best_score - 1e-10:
            best_p = p_new.copy()
            best_score = sc_new
            no_improve = 0
        else:
            no_improve += 1

        # Try blending to find better point
        if np.isfinite(sc_new):
            for t in (1.0, 0.7, 0.5, 0.3):
                p_blend = project_simplex((1.0 - t) * best_p + t * p_new)
                sc_blend = internal_score(p_blend)
                if np.isfinite(sc_blend) and sc_blend < best_score - 1e-10:
                    best_p = p_blend
                    best_score = sc_blend
                    break

        p = best_p.copy()
        current_score = best_score

        if verbose and iteration % 5 == 0:
            print("  ALT-LP iter=%d n=%d score=%.10f M=%.6e no_improve=%d t=%.0fs" %
                  (iteration, n, best_score, M, no_improve, time.monotonic() - (end - budget_s)),
                  flush=True)

        if no_improve > 15:
            # Try a perturbation to escape
            p = project_simplex(best_p + np.random.default_rng(iteration).normal(0, 1.0 / n, n))
            no_improve = 0

        if best_score < 1.50:
            break

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def gradient_polish(p, budget_s, verbose=True):
    """Polish using smoothed-max gradient on the probability simplex."""
    p = project_simplex(p)
    n = len(p)
    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score

    end = time.monotonic() + budget_s
    factor = 0.5
    iter_no = 0
    since_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() < end - 2.0:
        iter_no += 1
        factor = max(0.001, 0.5 * (0.995 ** iter_no))
        if since_improve > 200:
            factor = max(0.001, factor * 0.7)

        # Compute smoothed max gradient
        c = conv_sq(p)
        cmax = float(np.max(c))
        if not np.isfinite(cmax) or cmax <= 0:
            break

        T = max(factor * cmax, 1e-300)
        diff = (c - cmax) / T
        e = np.exp(diff)
        raw = float(e.sum())
        if not np.isfinite(raw) or raw <= 0:
            e = np.zeros_like(c)
            e[int(np.argmax(c))] = 1.0
            raw = 1.0
        w = e / raw

        # Gradient: g_j = 2 * sum_k w_k * p_{k-j} (with appropriate bounds)
        # This is 2 * (w_reversed * p) evaluated appropriately
        L = _fft_len(3 * n - 2)
        Fp = fft(p, L)
        Fw = fft(w[::-1], L)
        h = np.real(ifft(Fp * Fw))
        G = h[2 * n - 2:n - 2:-1]
        if G.size != n:
            tmp = np.zeros(n)
            m = min(n, G.size)
            tmp[:m] = G[:m]
            G = tmp
        g = 2.0 * np.maximum(G, 0.0)

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        # Line search with multiple step sizes
        gabs = float(np.max(np.abs(g))) if g.size else 0.0
        p_scale = max(1.0 / n, 1e-12)
        if gabs > 1e-300:
            base = p_scale / gabs
            lrs = (0.3 * base, 0.1 * base, 0.03 * base, 0.01 * base, 0.003 * base)
        else:
            lrs = (0.1 / n, 0.03 / n, 0.01 / n)

        accepted = False
        for lr in lrs:
            v = p - max(lr, 1e-10) * g
            pt = project_simplex(v)
            sc = internal_score(pt)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_p = pt.copy()
                best_score = sc
                since_improve = 0
                p = pt
                current_score = sc
                accepted = True
                break
            elif np.isfinite(sc) and sc < current_score - 1e-10:
                p = pt
                current_score = sc
                accepted = True
                break

        if not accepted:
            since_improve += 1
            factor = max(0.001, factor * 0.5)

            # Random perturbation
            if since_improve > 50:
                sigma = max(0.5 / n, 1e-6)
                pt = project_simplex(p + rng.normal(0, sigma, n))
                sc = internal_score(pt)
                if np.isfinite(sc) and sc < best_score:
                    best_p = pt.copy()
                    best_score = sc
                    since_improve = 0
                    p = pt

            if since_improve > 200:
                p = project_simplex(0.5 * best_p + 0.5 * p + rng.normal(0, 2.0 / n, n))
                current_score = internal_score(p)
                since_improve = 0
                factor = 0.3

        if verbose and iter_no % 100 == 0:
            remaining = end - time.monotonic()
            print("  GRAD iter=%d score=%.10f T=%.4f rem=%.0fs" %
                  (iter_no, best_score, factor, remaining), flush=True)

        if best_score < 1.498:
            break

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def make_initial(n, parent=None, rng=None):
    """Generate a diverse set of initial sequences."""
    if rng is None:
        rng = np.random.default_rng(0)
    inits = []

    # Uniform
    inits.append(np.full(n, 1.0 / n))

    # Parent
    if parent is not None:
        parent = np.asarray(parent, dtype=np.float64)
        if parent.size == n:
            inits.append(safe_normalize(parent))
        elif parent.size > n:
            # Downsample
            indices = np.linspace(0, parent.size - 1, n).astype(int)
            inits.append(safe_normalize(parent[indices]))
        elif parent.size < n:
            # Upsample via interpolation
            x_orig = np.linspace(0, 1, parent.size)
            x_new = np.linspace(0, 1, n)
            inits.append(safe_normalize(np.interp(x_new, x_orig, np.maximum(parent, 0))))

    # Power laws
    x = np.linspace(0, 1, n)
    for a in (0.5, 1.0, 1.5, 2.0, 3.0):
        p = np.power(x + 1e-6, a)
        inits.append(safe_normalize(p))
        inits.append(safe_normalize(p[::-1]))

    # Beta distributions
    for alpha, beta in ((0.5, 0.5), (1, 1), (2, 0.5), (0.5, 2), (3, 3), (0.3, 0.3)):
        p = np.power(x + 1e-6, alpha - 1) * np.power(1 - x + 1e-6, beta - 1)
        inits.append(safe_normalize(p))

    # Triangular / tent
    for peak in (0.3, 0.5, 0.7):
        p = np.maximum(0, 1 - np.abs(x - peak) / min(peak, 1 - peak + 1e-6))
        if p.sum() > 0:
            inits.append(safe_normalize(p))

    # Cosine windows
    for w in (0.5, 1.0, 2.0):
        p = np.cos(np.pi * w * (x - 0.5)) ** 2
        inits.append(safe_normalize(np.maximum(p, 0)))

    # Random smooth
    for strength in (0.3, 0.8, 1.5):
        f1 = rng.integers(1, max(2, n // 100))
        f2 = rng.integers(1, max(2, n // 50))
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        p = 1.0 + strength * (0.5 * np.sin(2 * np.pi * f1 * x + p1) + 0.3 * np.sin(2 * np.pi * f2 * x + p2))
        inits.append(safe_normalize(p))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative step function minimizing evaluate_sequence score.
    Uses alternating LP + gradient polish with multi-start and multiple lengths.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 20.0)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
        parent = np.maximum(parent, 0.0)
        if parent.size >= 10 and parent.sum() > 0:
            parent = safe_normalize(parent)
        else:
            parent = None
    except Exception:
        parent = None

    best_p = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = safe_normalize(p)
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("NEW BEST: score=%.10f n=%d t=%.0fs" % (sc, len(p), time.monotonic() - start_time),
                  flush=True)

    # Sequence lengths to try, with time allocations
    # Strategy: focus on medium lengths where LP is most effective
    tracks = [
        (500, 0.20),
        (800, 0.25),
        (1200, 0.25),
        (2000, 0.15),
        (4096, 0.10),
    ]

    total_budget = deadline - start_time
    if total_budget <= 0:
        return [1.0]

    for n, frac in tracks:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        t_budget = min(frac * total_budget, remaining * 0.7)
        if t_budget < 30:
            break

        print("\n=== Track n=%d, budget=%.0fs ===" % (n, t_budget), flush=True)

        # Generate initializations
        inits = make_initial(n, parent, rng)
        scored = []
        for p in inits:
            p = safe_normalize(p)
            sc = internal_score(p)
            if np.isfinite(sc):
                scored.append((sc, p))
        scored.sort(key=lambda x: x[0])

        # Try top initializations with alternating LP
        n_attempts = max(1, min(4, len(scored)))
        attempts_budget = t_budget / n_attempts

        for i in range(n_attempts):
            if time.monotonic() > deadline - 30:
                break
            sc0, p0 = scored[i]
            try_update(p0, sc0)

            remaining_attempt = min(attempts_budget, deadline - time.monotonic() - 20)
            if remaining_attempt < 10:
                break

            # Alternating LP
            lp_time = remaining_attempt * 0.7
            gp_time = remaining_attempt * 0.3

            p_lp, sc_lp = alternating_lp(p0, lp_time, verbose=True)
            try_update(p_lp, sc_lp)

            # Gradient polish
            if time.monotonic() < deadline - 25:
                p_gp, sc_gp = gradient_polish(p_lp, gp_time, verbose=True)
                try_update(p_gp, sc_gp)

    # Final polish on the best found
    if best_p is not None and time.monotonic() < deadline - 10:
        final_time = max(5.0, deadline - time.monotonic() - 5)
        print("\n=== Final polish, time=%.0fs ===" % final_time, flush=True)
        p_final, sc_final = gradient_polish(best_p, final_time, verbose=True)
        try_update(p_final, sc_final)

    if best_p is None:
        best_p = np.full(500, 1.0 / 500)
        best_score = internal_score(best_p)

    # Ensure valid output
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.nan_to_num(best_p, nan=0.0, posinf=0.0, neginf=0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(best_p.sum())
    if s < 1e-12:
        best_p = np.full(max(1, len(best_p)), 1.0 / max(1, len(best_p)))
        s = 1.0
    best_p = np.minimum(best_p, 1000.0)

    print("\n=== FINAL: score=%.10f n=%d ===" % (best_score, len(best_p)), flush=True)
    return [float(x) for x in best_p]
