    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]

"""
Optimization of non-negative sequence to minimize normalized autocorrelation peak.
Uses smooth-max gradient descent followed by LP-based active set refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score_normalized(a):
    """Compute the objective for a normalized sequence (sum=1)."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * float(np.max(b))


def _score_raw(a):
    """Compute the raw objective (same as evaluate_sequence)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * n * np.max(b) / (S * S))


def _compute_gradient(a, beta):
    """Compute gradient of smoothed max of convolve(a,a) w.r.t. a. Returns (smooth_max_value, gradient)."""
    n = len(a)
    b = np.convolve(a, a)
    b_mv = np.max(b)
    # Numerical stability
    exp_terms = np.exp(beta * (b - b_mv))
    sum_exp = np.sum(exp_terms)
    smooth_val = (b_mv + np.log(sum_exp)) / beta
    w = exp_terms / sum_exp
    # grad[j] = 2 * sum_m w[m+j] * a[m]
    idx = np.arange(n)
    W = w[idx[:, None] + idx[None, :]]
    grad = 2.0 * (W @ a)
    # Center for sum constraint
    grad = grad - np.mean(grad)
    return smooth_val, grad
