```python
"""
Hybrid optimizer for minimizing peak auto-convolution of non-negative sequences.
Combines: (1) subgradient descent in log-space, (2) LP-based directional refinement,
(3) simulated annealing for exploration, (4) block-wise local search for polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a, target_sum=None):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    return a / s * target_sum


def get_subgradient(a, k_star):
    """Compute subgradient of max_k conv(a,a)[k] at position k_star."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = a[k_star - idx]
    return g


def subgradient_refine(a, time_limit_s=300.0, verbose=False):
    """Subgradient descent with adaptive step size (additive)."""
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    lr = 0.005
    patience = 0
    
    for it in range(10000000):
        if time.monotonic() - start_time > time_limit_s:
            break
        b = np.convolve(a, a)
        k_star = int(b.argmax())
        g = get_subgradient(a, k_star)
        g = g - g.mean()  # Center to preserve sum
        
        a_new = a - lr * g
        a_new = np.maximum(0.0, a_new)
        a_new = normalize(a_new)
        
        new_score = score_sequence(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            patience = 0
            lr = min(lr * 1.005, 0.02)
        else:
            patience += 1
            lr = max(lr * 0.995, 1e-5)
            # Explore: occasionally take the step anyway
            if patience % 500 == 250 and new_score < best_score + 0.01:
                a = a_new
        
        a_score = score_sequence(a)
        if a_score > best_score * 1.01:
            a = best_a.copy()  # Reset if drifts too far
        
        if patience > 20000:
            break
    
    return best_a, best_score


def multiplicative_refine(a, time_limit_s=300.0, verbose=False):
    """Multiplicative (mirror descent) update: a[i] *= exp(-lr * g[i])."""
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    lr = 0.01
    patience = 0
    
    for it in range(10000000):
        if time.monotonic() - start_time > time_limit_s:
            break
        b = np.convolve(a, a)
        k_star = int(b.argmax())
        g = get_subgradient(a, k_star)
        g = g - g.mean()
        
        a_new = a * np.exp(-lr * g)
        a_new = normalize(a_new)
        
        new_score = score_sequence(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            patience = 0
            lr = min(lr * 1.01, 0.1)
        else:
            patience += 1
            lr = max(lr * 0.99, 1e-5)
        
        if patience > 20000:
            break
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix A where (A @ g)[k] = (conv(a, g))[k]."""
    n = len(a)
    n_rows = 2 * n - 1
    cols = np.arange(n_rows)[:, None] - np.arange(n)[None, :]
    mask = (cols >= 0) & (cols < n)
    A = np.zeros((n_rows, n))
    A[mask] = a[cols[mask]]
    return A


def solve_lp(a, M, timeout=15.0):
    """Solve: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=300.0, verbose=False):
    """LP-based iterative refinement with adaptive step sizes."""
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    
    for it in range(500):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        b = np.convolve(a, a)
        M = b.max()
        score = 2.0 * n * M / (a.sum() ** 2)
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 50:
            break
        
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        
        timeout = max(2.0, min(30.0, remaining * 0.15))
        g = solve_lp(a, M, timeout=timeout)
        if g is None or g.sum() <= 1.0 + 1e-10:
            break
        
        g = g / g.sum() * np.sqrt(2.0 * n)
        
        # Try multiple step sizes
        best_candidate = a
        best_candidate_score = score
        for alpha in np.concatenate([np.linspace(0.1, 1.0, 10), np.array([1.0])]):
            a_try = (1 - alpha) * a + alpha * g
            a_try = normalize(a_try)
            s_try = score_sequence(a_try)
            if s_try < best_candidate_score:
                best_candidate_score = s_try
                best_candidate = a_try
        
        a = best_candidate
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score


def local_search(a, time_limit_s=120.0, rng=None, verbose=False):

        stop = min(n, start + width)
        
        candidate = a.copy()
        # Smooth random perturbation
        t = np.arange(stop - start, dtype=np.float64)
        noise = np.cumsum(rng.normal(0, step_scale, size=stop - start)) * 0.3

        candidate = np.maximum(0, candidate)
        candidate = normalize(candidate)

            best_a = candidate.copy()
            a = candidate.copy()

        else:

    T = 0.1
    T_min = 0.001
    cooling = 0.999

        T = max(T * cooling, T_min)
        
        # Generate candidate
        width = max(2, int(rng.integers(n // 100, n // 5)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0, 0.1 * (1 + T), size=stop - start))
        candidate = normalize(candidate)
        
        new_score = score_sequence(candidate)
        delta = new_score - score_sequence(a)
        
        # Accept with probability exp(-delta/T) if worse
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = candidate
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
```

```python
"""
Hybrid optimizer for minimizing peak auto-convolution of non-negative sequences.
Combines: (1) hard & soft subgradient descent, (2) LP-based directional refinement,
(3) multiplicative updates, (4) local search with adaptive steps, (5) simulated annealing.
Uses vectorized convolution for smooth subgradient computation.
"""

import numpy as np
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a, target_sum=None):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    return a / s * target_sum


def get_hard_subgradient(a, k_star):
    """Subgradient at the argmax position: db[k*]/da[j] = 2*a[k*-j] if valid."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k_star - idx]
    return g


def get_soft_subgradient(a, alpha=5.0):
    """Weighted subgradient using softmax over all convolution peaks."""
    n = len(a)
    b = np.convolve(a, a)
    b_max = b.max()
    weights = np.exp(alpha * (b - b_max))
    weights /= weights.sum()
    w_rev = weights[::-1]
    conv_result = np.convolve(w_rev, a, mode='full')
    # g[j] = 2 * conv_result[2n-2-j] for j=0..n-1
    g = 2.0 * conv_result[n - 1:2 * n - 1][::-1]
    return g


def subgradient_refine(a, time_limit_s=300.0, soft_alpha=5.0, verbose=False):
    """Subgradient descent with adaptive step size (hard + soft)."""
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    lr = 0.003
    patience = 0
    use_soft = (n <= 1500)

    for it in range(100000000):
        if time.monotonic() - start_time > time_limit_s:
            break

        if use_soft:
            g = get_soft_subgradient(a, alpha=soft_alpha)
        else:
            b = np.convolve(a, a)
            k_star = int(b.argmax())
            g = get_hard_subgradient(a, k_star)

        g = g - g.mean()
        a_new = a - lr * g
        a_new = np.maximum(0.0, a_new)
        a_new = normalize(a_new)

        new_score = score_sequence(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            patience = 0
            lr = min(lr * 1.003, 0.02)
        else:
            patience += 1
            lr = max(lr * 0.997, 1e-6)
            if patience % 300 == 150 and new_score < best_score + 0.005:
                a = a_new

        if patience > 30000:
            break

        if verbose and it % 50000 == 0 and it > 0:
            print(f"      SG iter {it}: best={best_score:.6f}, lr={lr:.6f}, patience={patience}")

    return best_a, best_score


def multiplicative_refine(a, time_limit_s=200.0, verbose=False):
    """Multiplicative update: a[i] *= exp(-lr * g[i]) - maintains non-negativity naturally."""
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    lr = 0.01
    patience = 0

    for it in range(100000000):
        if time.monotonic() - start_time > time_limit_s:
            break

        b = np.convolve(a, a)
        k_star = int(b.argmax())
        g = get_hard_subgradient(a, k_star)
        g = g - g.mean()

        a_new = a * np.exp(-lr * g)
        a_new = normalize(a_new)

        new_score = score_sequence(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            patience = 0
            lr = min(lr * 1.01, 0.1)
        else:
            patience += 1
            lr = max(lr * 0.99, 1e-5)

        if patience > 20000:
            break

    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix A where (A @ g)[k] = (conv(a, g))[k]."""
    n = len(a)
    n_rows = 2 * n - 1
    cols = np.arange(n_rows)[:, None] - np.arange(n)[None, :]
    mask = (cols >= 0) & (cols < n)
    A = np.zeros((n_rows, n))
    A[mask] = a[cols[mask]]
    return A


def solve_lp(a, M, timeout=10.0):
    """Solve LP: maximize sum(g) subject to conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            gv = np.maximum(0.0, res.x)
            if gv.sum() > 0:
                return gv
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=300.0, verbose=False):
    """LP-based iterative refinement with line search over step sizes."""
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0

    for it in range(300):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break

        b = np.convolve(a, a)
        M = b.max()
        score = 2.0 * n * M / (a.sum() ** 2)

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 40:
            break

        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break

        timeout = max(2.0, min(20.0, remaining * 0.1))
        g = solve_lp(a, M, timeout=timeout)
        if g is None or g.sum() <= 1.0 + 1e-10:
            break

        g = g / g.sum() * np.sqrt(2.0 * n)

        best_candidate = a.copy()
        best_candidate_score = score
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            a_try = (1.0 - alpha) * a + alpha * g
            a_try = normalize(a_try)
            s_try = score_sequence(a_try)
            if s_try < best_candidate_score:
                best_candidate_score = s_try
                best_candidate = a_try

        a = best_candidate

        if verbose and it % 10 == 0:
            print(f"      LP iter {it}: best={best_score:.6f}, current={score:.6f}")

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


def local_search(a, time_limit_s=120.0, rng=None, verbose=False):
    """Adaptive block-wise local search."""
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    step_scale = 0.03
    accept_count = 0

    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(rng.integers(n // 300, max(3, n // 10))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)

        candidate = a.copy()
        noise = rng.normal(0.0, step_scale, size=stop - start)
        candidate[start:stop] *= np.exp(noise)
        candidate = np.maximum(0.0, candidate)
        candidate = normalize(candidate)

        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            step_scale = min(step_scale * 1.05, 0.15)
        else:
            step_scale = max(step_scale * 0.98, 0.003)

        if verbose and accept_count % 100 == 0 and accept_count > 0:
            print(f"      LS accept {accept_count}: score={best_score:.6f}")

    return best_a, best_score


def simulated_annealing(a, time_limit_s=150.0, rng=None, verbose=False):
    """Simulated annealing with block multiplicative moves."""
    if rng is None:
        rng = np.random.default_rng(123)
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    current_score = best_score

    T = 0.05
    T_min = 0.0005

    it = 0
    while time.monotonic() - start_time < time_limit_s:
        T = max(T * 0.9995, T_min)
        it += 1

        width = max(2, int(rng.integers(n // 200, max(3, n // 6))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)

        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.05 + T, size=stop - start))
        candidate = normalize(candidate)

        new_score = score_sequence(candidate)
        delta = new_score - current_score

        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                if verbose and it % 5000 == 0:
                    print(f"      SA iter {it}: best={best_score:.6f}, T={T:.5f}")

    return best_a, best_score


def peak_reduction(a, time_limit_s=100.0, verbose=False):
    """Directly reduce peaks by modifying the a-values that contribute most to them."""
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    lr = 0.01

    for it in range(100000):
        if time.monotonic() - start_time > time_limit_s:
            break

        b = np.convolve(a, a)
        b_max = b.max()
        # Find top peaks
        top_k = np.argsort(b)[-min(5, len(b)):]

        g = np.zeros(n)
        for k in top_k:
            g += get_hard_subgradient(a, int(k))

        g = g - g.mean()
        a_new = a - lr * g
        a_new = np.maximum(0.0, a_new)
        a_new = normalize(a_new)

        new_score = score_sequence(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.05)
        else:
            lr = max(lr * 0.95, 0.001)

    return best_a, best_score


def combined_refine(a, time_limit_s=600.0, rng=None, verbose=True):
    """Run multiple refinement methods in sequence, keeping the best."""
    if rng is None:
        rng = np.random.default_rng(42)
    start_time = time.monotonic()
    current = a.copy().astype(np.float64)
    best_a = current.copy()
    best_score = score_sequence(current)

    if verbose:
        print(f"    Combined start: n={len(current)}, score={best_score:.6f}")

    # Phase 1: Soft subgradient (fast, uses all peaks)
    remaining = time_limit_s - (time.monotonic() - start_time)
    if remaining > 50:
        t = min(180, remaining * 0.25)
        a1, s1 = subgradient_refine(current, time_limit_s=t, soft_alpha=5.0)
        if s1 < best_score:
            best_score = s1
            best_a = a1.copy()
            current = a1.copy()
            if verbose:
                print(f"    After soft SG: {best_score:.6f}")

    # Phase 2: Multiplicative
    remaining = time_limit_s - (time.monotonic() - start_time)
    if remaining > 40:
        t = min(100, remaining * 0.15)
        a2, s2 = multiplicative_refine(current, time_limit_s=t)
        if s2 < best_score:
            best_score = s2
            best_a = a2.copy()
            current = a2.copy()
            if verbose:
                print(f"    After mult: {best_score:.6f}")

    # Phase 3: LP refinement
    remaining = time_limit_s - (time.monotonic() - start_time)
    if remaining > 40:
        t = min(250, remaining * 0.35)
        a3, s3 = lp_refine(current, time_limit_s=t, verbose=verbose)
        if s3 < best_score:
            best_score = s3
            best_a = a3.copy()
            current = a3.copy()
            if verbose:
                print(f"    After LP: {best_score:.6f}")

    # Phase 4: Peak reduction
    remaining = time_limit_s - (time.monotonic() - start_time)
    if remaining > 30:
        t = min(80, remaining * 0.15)
        a4, s4 = peak_reduction(current, time_limit_s=t)
        if s4 < best_score:
            best_score = s4
            best_a = a4.copy()
            current = a4.copy()
            if verbose:
                print(f"    After peak red: {best_score:.6f}")

    # Phase 5: Local search
    remaining = time_limit_s - (time.monotonic() - start_time)
    if remaining > 20:
        t = min(100, remaining * 0.2)
        a5, s5 = local_search(current, time_limit_s=t, rng=rng)
        if s5 < best_score:
            best_score = s5
            best_a = a5.copy()
            if verbose:
                print(f"    After LS: {best_score:.6f}")

    # Phase 6: Final LP
    remaining = time_limit_s - (time.monotonic() - start_time)
    if remaining > 20:
        a6, s6 = lp_refine(best_a, time_limit_s=remaining * 0.8, verbose=verbose)
        if s6 < best_score:
            best_score = s6
            best_a = a6.copy()
            if verbose:
                print(f"    After final LP: {best_score:.6f}")

    return best_a, best_score


def get_initial(n, rng):
    """Generate diverse initial sequences."""
    t = np.arange(n, dtype=np.float64)
    choice = int(rng.integers(0, 7))
    if choice == 0:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))
    elif choice == 1:
        center = (n - 1) / 2.0
        sigma = 0.25 * n / 2.0
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif choice == 2:
        x = t / max(1, n - 1)
        a = np.maximum(0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))
    elif choice == 3:
        a = np.sin(np.pi * t / n) ** 2
    elif choice == 4:
        n_bumps = int(rng.integers(2, 5))
        a = np.zeros(n)
        for _ in range(n_bumps):
            center = float(rng.integers(0, n))
            width = float(max(2, n // int(rng.integers(5, 15))))
            a += np.exp(-0.5 * ((t - center) / width) ** 2)
    elif choice == 5:
        freq = int(rng.integers(1, 4))
        a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * t / n)
        a = np.maximum(0, a)
    else:
        a = np.ones(n)
        a *= np.exp(-0.5 * ((t - n / 2) / (n / 4)) ** 2)
    return normalize(a)


def propose_candidate(seed=42, budget_s=2400):
    """
    Hybrid optimizer: soft subgradient + multiplicative + LP + peak reduction + local search.
    Starts from provided parent and explores multiple lengths/initializations.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20.0

    best_score = float('inf')
    best_seq = None

    # Load parent
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if len(parent) > 0 and parent.sum() > 0.01:
            parent = normalize(parent)
            parent_score = score_sequence(parent)
            best_score = parent_score
            best_seq = parent.copy()
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.6f}")
        else:
            parent = None
    except (NameError, AttributeError, TypeError):
        parent = None

    # Determine lengths to try
    if parent is not None:
        p_len = len(parent)
        lengths = sorted(set([
            p_len,
            p_len + 10, p_len + 50, p_len - 50,
            600, 700, 800, 1000, 1200
        ]))
        lengths = [l for l in lengths if 200 <= l <= 2000]
    else:
        lengths = [500, 600, 700, 800, 1000, 1200]

    # Build candidate list
    candidates = []
    if parent is not None:
        candidates.append((parent, "parent", 1.0))

    for n in lengths[:10]:
        for idx in range(2):
            a_init = get_initial(n, rng)
            s_init = score_sequence(a_init)
            candidates.append((a_init, f"init_n{n}_{idx}", s_init))

    # Sort by initial score (skip parent)
    non_parent = [(a, label, s) for a, label, s in candidates if label != "parent"]
    non_parent.sort(key=lambda x: x[2])
    candidates = [(candidates[0][0], "parent", candidates[0][2])] + non_parent[:8]

    # Process candidates with decreasing time allocation
    for i, (a_init, label, init_score) in enumerate(candidates):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Time allocation: more time for earlier/better candidates
        if i == 0:
            time_alloc = min(900, remaining * 0.45)
        elif i == 1:
            time_alloc = min(500, remaining * 0.30)
        elif i <= 3:
            time_alloc = min(300, remaining * 0.20)
        else:
            time_alloc = min(200, remaining * 0.12)

        if time_alloc < 30:
            continue

        # Skip if init score is much worse than current best and not the first
        if i > 0 and best_score < init_score - 0.1:
            time_alloc = min(time_alloc, 100)

        print(f"\n--- Candidate [{i}]: {label}, n={len(a_init)}, init_score={init_score:.6f}, time={time_alloc:.0f}s ---")

        a_refined, score = combined_refine(a_init, time_limit_s=time_alloc, rng=rng, verbose=(i < 2))

        if score < best_score:
            best_score = score
            best_seq = a_refined.copy()
            print(f"  *** NEW BEST: {best_score:.6f}, n={len(a_refined)}, elapsed={time.monotonic()-total_start:.0f}s")

    # Final intensive polish on the best
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== Final intensive polish (remaining={remaining:.0f}s) ===")

            # Round 1: Full combined
            a_pol, s_pol = combined_refine(best_seq, time_limit_s=remaining * 0.5, rng=rng, verbose=True)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = a_pol.copy()

            # Round 2: Focused LP + local search
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                a_p2, s_p2 = lp_refine(best_seq, time_limit_s=remaining2 * 0.4, verbose=True)
                if s_p2 < best_score:
                    best_score = s_p2
                    best_seq = a_p2.copy()

            remaining3 = deadline - time.monotonic()
            if remaining3 > 15:
                a_p3, s_p3 = local_search(best_seq, time_limit_s=remaining3 * 0.6, rng=rng)
                if s_p3 < best_score:
                    best_score = s_p3
                    best_seq = a_p3.copy()

            # Round 3: Simulated annealing for final escape
            remaining4 = deadline - time.monotonic()
            if remaining4 > 15:
                a_p4, s_p4 = simulated_annealing(best_seq, time_limit_s=remaining4 * 0.7, rng=rng)
                if s_p4 < best_score:
                    best_score = s_p4
                    best_seq = a_p4.copy()

    # Fallback
    if best_seq is None:
        best_seq = get_initial(500, rng)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Final verification
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    verify_score = evaluate_sequence(result)
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, internal_score={best_score:.6f}, verified_score={verify_score:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s / {budget_s}s")
    print(f"{'='*60}")

    return result
```
