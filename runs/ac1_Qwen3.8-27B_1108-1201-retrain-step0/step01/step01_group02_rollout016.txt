"""
Hybrid L-BFGS + LP optimization for minimizing peak auto-convolution of non-negative sequences.
Uses log-sum-exp smoothing with softmax parameterization for efficient gradient-based optimization,
combined with LP-based direction finding and multi-length/multi-init exploration.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_convolve(a, b):
    """Fast linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    size *= 2
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n]


def score(a):
    """Compute the objective score 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return float('inf')
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-15:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    """Normalize to sum 1 and clip negatives."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def softmax_a(x):
    """Compute softmax of x (numerically stable)."""
    xm = x - x.max()
    e = np.exp(xm)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def lsm_objective_and_grad(x, beta):
    """
    Objective and gradient in x-space for the log-sum-exp smoothed score.
    a = softmax(x), objective = 2n * (1/beta) * log(sum_k exp(beta * conv_k(a)))
    """
    a = softmax_a(x)
    n = len(a)
    conv = np.convolve(a, a)  # length 2n-1
    # Numerically stable log-sum-exp
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    e = np.exp(shifted)
    e_sum = e.sum()
    p = e / e_sum
    soft_val = (conv_max + np.log(e_sum)) / beta
    obj = 2.0 * n * soft_val

    # Gradient w.r.t. a: (4n/beta) * sum_k p_k * a_{k-i}
    g_a = 2.0 * np.convolve(a[::-1], p)[(n - 1):(2 * n - 1)]
    df_da = (4.0 * n / beta) * g_a

    # Convert to x-space using softmax Jacobian
    s_term = np.dot(df_da, a)
    df_dx = a * (df_da - s_term)

    return obj, df_dx


def lsm_optimize(a_init, time_budget, beta_start=50.0, beta_end=300.0, verbose=True):
    """Optimize using L-BFGS-B in softmax x-space with beta annealing."""
    start_time = time.monotonic()
    a = normalize(a_init.copy().astype(np.float64))
    n = len(a)

    # Initialize x from a
    eps = 1e-10
    x = np.log(a + eps)

    best_a = a.copy()
    best_score = score(a)

    # Run L-BFGS-B with increasing beta
    n_phases = 3
    phase_times = [time_budget * t for t in [0.4, 0.3, 0.3]]
    betas = [beta_start, (beta_start + beta_end) / 2, beta_end]

    for phase in range(n_phases):
        phase_start = time.monotonic()
        remaining = phase_times[phase] - (phase_start - start_time)
        if remaining < 5.0:
            break
        beta = betas[phase]

        def obj_fun(x):
            val, _ = lsm_objective_and_grad(x, beta)
            return val

        def grad_fun(x):
            _, grad = lsm_objective_and_grad(x, beta)
            return grad

        def obj_grad_fun(x):
            return lsm_objective_and_grad(x, beta)

        bounds = [(-30.0, 30.0)] * n

        try:
            result = minimize(
                obj_grad_fun, x, jac=True, method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 5000, 'maxfun': 10000,
                         'ftol': 1e-12, 'gtol': 1e-8,
                         'iprint': -1, 'maxls': 50}
            )
            if result.x is not None:
                x = result.x
                a_new = softmax_a(x)
                a_new = normalize(a_new)
                sc = score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                else:
                    a = a_new
                # Update x for next phase (re-derive from a)
                x = np.log(a + eps)

                if verbose:
                    elapsed = time.monotonic() - start_time
                    print(f"      Phase {phase+1} (beta={beta:.0f}): score={best_score:.8f}, "
                          f"lsm_obj={result.fun:.8f}, elapsed={elapsed:.1f}s, "
                          f"nfev={result.nfev}, niters={result.nit}")
        except Exception as e:
            if verbose:
                print(f"      Phase {phase+1} error: {e}")

    # Final evaluation
    final_score = score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"      LSM done: best={best_score:.8f}, total_time={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def build_conv_matrix(a):
    """Build sparse matrix for convolution with a."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_parts = []
    for j in range(n):
        row_parts.append(np.arange(j, j + n, dtype=np.int32))
    row_idx = np.concatenate(row_parts)
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_direction(a, M, timeout=10.0):
    """Solve LP to find direction g maximizing sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_budget, verbose=True):
    """LP-based iterative refinement."""
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score(a)
    stagnant = 0

    it = 0
    max_iters = 3000
    while time.monotonic() - start_time < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start_time)
        if remaining < 3.0:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        sum_a = a.sum()
        sc = 2.0 * n * M / (sum_a ** 2)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 50:
            if verbose:
                print(f"      LP stagnant at iter {it}, best={best_score:.8f}")
            break

        timeout = max(1.0, min(15.0, remaining * 0.2))
        g = solve_lp_direction(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"      LP optimal at iter {it}, best={best_score:.8f}")
            break
        g = g / sum_g

        # Line search for step size
        best_alpha = 0.1
        best_sc = sc
        for alpha in [0.5, 0.3, 0.1, 0.05, 0.01]:
            a_try = (1.0 - alpha) * a + alpha * g
            a_try = normalize(a_try)
            sc_try = score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g
        a = normalize(a)
        it += 1

        if verbose and it % 100 == 0:
            print(f"      LP iter {it}: score={best_score:.8f}, alpha={best_alpha}")

    final_score = score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"      LP done: best={best_score:.8f}, {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_polish(a, time_budget, rng, verbose=True):
    """Local search polishing with small perturbations."""
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score(a)
    accept_count = 0

    while time.monotonic() - start_time < time_budget:
        # Random smooth perturbation
        width = max(2, int(rng.integers(n // 300, n // 5)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)

        candidate = a.copy()
        sigma = 0.05
        candidate[start:stop] *= np.exp(rng.normal(0.0, sigma, size=stop - start))

        # Occasionally try global smooth perturbation
        if rng.random() < 0.2:
            t = np.arange(n, dtype=np.float64)
            freq = rng.uniform(1, 5)
            candidate *= np.exp(0.05 * np.sin(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi)))

        candidate = normalize(candidate)
        sc = score(candidate)

        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            # Decrease sigma as we improve
            sigma = max(0.005, sigma * 0.95)
            if verbose and accept_count % 100 == 0:
                print(f"      Polish accept {accept_count}: score={best_score:.8f}")

    if verbose:
        print(f"      Polish done: best={best_score:.8f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def init_asymmetric_gaussian(n, frac=0.4):
    """Asymmetric Gaussian to encourage non-symmetric solutions."""
    t = np.arange(n, dtype=np.float64)
    center = frac * (n - 1)
    sigma = 0.25 * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def init_fourier_flat(n, rng):
    """Initialize with approximately flat circular auto-convolution using random phases."""
    # Target: |A(f)| approximately constant
    size = 1
    while size < 2 * n - 1:
        size *= 2
    size *= 2

    A = np.zeros(size, dtype=np.complex128)
    A[0] = np.sqrt(n)  # DC component
    # Set random phases for other frequencies, flat magnitude
    freqs = np.arange(1, size // 2 + 1)
    A[freqs] = np.exp(1j * rng.uniform(0, 2 * np.pi, len(freqs)))
    A[size - freqs] = np.conj(A[freqs])

    a = np.fft.irfft(A)[:n]
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-10:
        return init_raised_cosine(n)
    return a / s


def init_random_bumps(n, rng, n_bumps=4):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 15), max(2, n // 4)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-10:
        return init_raised_cosine(n)
    return a / s

        a_refined, sc_refined = lsm_optimize(a, time_budget=lsm_time, beta_start=50, beta_end=300, verbose=True)
        if sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined.copy()
            print(f"    *** NEW BEST (L-BFGS): score={best_score:.8f}, n={n}")
        candidates.append((sc_refined, a_refined, n))

        # Phase 2: LP refinement
        t_remaining = deadline - time.monotonic()
        if t_remaining > 20.0:
            lp_time = min(time_limit * 0.25, t_remaining * 0.5)
            if lp_time > 10.0:
                a_lp, sc_lp = lp_refine(a_refined, time_budget=lp_time, verbose=False)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.copy()
                    print(f"    *** NEW BEST (LP): score={best_score:.8f}, n={n}")
                candidates.append((sc_lp, a_lp, n))

        # Phase 3: Local polish
        t_remaining = deadline - time.monotonic()
        if t_remaining > 10.0:
            polish_time = min(time_limit * 0.15, t_remaining * 0.4)
            if polish_time > 5.0:
                a_pol, sc_pol = local_polish(a_lp if sc_lp < sc_refined else a_refined,
                                             time_budget=polish_time, rng=rng, verbose=False)
                if sc_pol < best_score:
                    best_score = sc_pol
                    best_seq = a_pol.copy()
                    print(f"    *** NEW BEST (Polish): score={best_score:.8f}, n={n}")

    # Deep refinement on the best candidate
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60.0:
            print(f"\n=== Deep refinement on best (n={len(best_seq)}, score={best_score:.8f}) ===")
            print(f"    Remaining: {remaining:.0f}s")

            # L-BFGS with high beta for fine-tuning
            deep_time = remaining * 0.4
            a_deep, sc_deep = lsm_optimize(best_seq, time_budget=deep_time,
                                           beta_start=200, beta_end=800, verbose=True)
            if sc_deep < best_score:
                best_score = sc_deep
                best_seq = a_deep.copy()
                print(f"    *** DEEP L-BFGS: score={best_score:.8f}")

            # LP refinement
            remaining2 = deadline - time.monotonic()
            if remaining2 > 30.0:
                lp_time2 = remaining2 * 0.35
                a_lp2, sc_lp2 = lp_refine(best_seq, time_budget=lp_time2, verbose=True)
                if sc_lp2 < best_score:
                    best_score = sc_lp2
                    best_seq = a_lp2.copy()
                    print(f"    *** DEEP LP: score={best_score:.8f}")

            # Final polish
            remaining3 = deadline - time.monotonic()
            if remaining3 > 15.0:
                a_fin, sc_fin = local_polish(best_seq, time_budget=remaining3 * 0.5,
                                             rng=rng, verbose=True)
                if sc_fin < best_score:
                    best_score = sc_fin
                    best_seq = a_fin.copy()
                    print(f"    *** FINAL POLISH: score={best_score:.8f}")

            # One more L-BFGS pass at very high beta
            remaining4 = deadline - time.monotonic()
            if remaining4 > 20.0:
                a_vf, sc_vf = lsm_optimize(best_seq, time_budget=remaining4 * 0.7,
                                           beta_start=500, beta_end=2000, verbose=True)
                if sc_vf < best_score:
                    best_score = sc_vf
                    best_seq = a_vf.copy()
                    print(f"    *** VERY FINE L-BFGS: score={best_score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = init_raised_cosine(500)
        best_score = score(best_seq)
        print(f"  Fallback: score={best_score:.8f}")

    # Final verification
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"\nFinal result: score={best_score:.8f}, n={len(best_seq)}")
    print(f"Verified score via evaluate_sequence: {verify_score:.8f}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")

    return [float(x) for x in best_seq]
