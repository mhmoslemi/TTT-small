```python
import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def conv_matrix(f, n):
    """Construct the matrix A such that A @ g = conv(f, g)."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    return A

def score_fast(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def lp_direction(f, M, n):
    """
    Find g >= 0 that maximizes sum(g) subject to conv(f,g) <= M element-wise.
    Returns (g, success).
    """
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    
    c = -np.ones(n)  # minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        g = res.x
        return g, True
    return None, False
```

```python
"""
LP-guided optimization for minimizing the peak of self-convolution of non-negative sequences.
Combines linear programming for direction finding, gradient descent for refinement,
and targeted peak reduction for local improvements.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score computation (matches evaluate_sequence formula)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def conv_matrix(f, n):
    """Construct matrix A such that A @ g = conv(f, g) for len(f)=len(g)=n."""
    A = np.zeros((2 * n - 1, n))
    for k in range(n):
        A[k, :k + 1] = f[k :: -1]
    for k in range(n, 2 * n - 1):
        A[k, k - n + 1:] = f[n - 1 : k - n : -1]
    return A


def lp_solve(f, M, n, method='highs'):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    A = conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=method)
    if res.success:
        return res.x
    return None


def gradient_step(f, tau, n):
    """Compute gradient of soft_max(conv(f,f), tau) w.r.t. f."""
    c = fft_convolve(f, f)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    rev_f = f[::-1]
    corr = fft_convolve(rev_f, w)
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return g
```

```python
import numpy as np

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score computation matching evaluate_sequence formula."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def conv_matrix(f, n):
    """Construct matrix A such that A @ g = conv(f, g) for len(f)=len(g)=n."""
    A = np.zeros((2 * n - 1, n))
    for k in range(n):
        A[k, :k + 1] = f[k :: -1]
    for k in range(n, 2 * n - 1):
        A[k, k - n + 1:] = f[n - 1 : k - n : -1]
    return A


def lp_solve(f, M, n, time_limit_s=5.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M element-wise, g >= 0."""
    A = conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_s, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def gradient_descent_step(f, tau, n):
    """Compute gradient of soft_max(conv(f,f), tau) w.r.t. f."""
    c = fft_convolve(f, f)

    corr = fft_convolve(rev_f, w)
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return g

            pol2_score = score_fast(f_pol2)
            print(f"  Result: {pol2_score:.8f}")
            if pol2_score < best_score:
                best_score = pol2_score
                best_seq = f_pol2.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Final verification with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
