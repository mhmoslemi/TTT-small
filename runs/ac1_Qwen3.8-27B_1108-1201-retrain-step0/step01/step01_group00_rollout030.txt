"""
LP-based optimization for minimizing the peak autocorrelation of non-negative
step functions. Uses the primal LP to find descent directions, adaptive step
sizes, and local search polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def solve_lp_direction(f, M):
    """
    Solve the LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Returns the optimal g.
    """
    n = len(f)
    # Build constraint matrix A where A[k, i] = f[i] if 0 <= k-i < n else 0
    # A has shape (2n-1, n)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        A[k, i_start:i_end + 1] = f[i_start:i_end + 1]
    
    # LP: min -sum(g) s.t. A @ g <= M * ones, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.status != 0:
        # Fallback: return uniform
        return np.ones(n)
    
    g = res.x
    # Clean up numerical noise
    g = np.clip(g, 0, None)
    return g

def lp_search(f_init, time_limit, rng, verbose=True):
    """
    Run the LP-based iterative search.
    """
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    # Normalize to sum = sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    f = f * (target_sum / sum_f)
    
    current_score = compute_score(f)
    best_score = current_score
    best_f = f.copy()
    t0 = time.monotonic()
    
    if verbose:
        print(f"  LP search: n={n}, initial score={current_score:.6f}")
    
    for iteration in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute M = max(f*f)
        conv_f = fft_convolve(f, f)
        M = np.max(conv_f)
        
        # Solve LP
        g = solve_lp_direction(f, M)
        
        if np.sum(g) < 1e-10:
            if verbose:
                print(f"    LP returned trivial solution at iter {iteration}")
            break
        
        # Rescale g to match target_sum
        g = g * (target_sum / np.sum(g))
        
        # Try full step first
        candidate = g
        cand_score = compute_score(candidate)
        
        if cand_score < current_score:
            f = candidate
            current_score = cand_score
            if current_score < best_score:
                best_score = current_score
                best_f = f.copy()
                if verbose and iteration < 10:
                    print(f"    iter {iteration}: full step, score={current_score:.6f}")
            continue
        
        # Try blending
        t_values = [0.5, 0.3, 0.1, 0.05]
        improved = False
        for t in t_values:
            candidate = (1 - t) * f + t * g
            cand_score = compute_score(candidate)
            if cand_score < current_score:
                f = candidate
                current_score = cand_score
                if current_score < best_score:
                    best_score = current_score
                    best_f = f.copy()
                improved = True
                break
        
        if not improved:
            # Try even smaller steps
            for t in [0.02, 0.01, 0.005]:
                candidate = (1 - t) * f + t * g
                cand_score = compute_score(candidate)
                if cand_score < current_score:
                    f = candidate
                    current_score = cand_score
                    if current_score < best_score:
                        best_score = current_score
                        best_f = f.copy()
                    improved = True
                    break
            
            if not improved:
                # No improvement possible with this direction, add small noise
                f = f * np.exp(rng.normal(0, 0.01, n))
                f = np.clip(f, 1e-6, None)
                f = f * (target_sum / np.sum(f))
                current_score = compute_score(f)
        
        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter {iteration}: score={current_score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")
    
    return best_f, best_score


def local_polish(a_init, time_limit, rng):
    """Local search polish with targeted peak reduction."""
    current = a_init.astype(np.float64).copy()
    n = len(current)
    target_sum = np.sqrt(2.0 * n)
    current = current * (target_sum / np.sum(current))
    
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.3:
            # Targeted peak reduction
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            # Find contributors to the peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, top_k))
            # Boost non-contributors
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 2:
                inc_idx = rng.choice(other, min(5, len(other)), replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, len(inc_idx)))
        elif r < 0.5:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, width))
        elif r < 0.7:
            # Symmetry push
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.005, n))
        elif r < 0.85:
            # Multi-point
            for _ in range(4):
                idx = int(rng.integers(0, n))
                candidate[idx] *= np.exp(rng.normal(0, 0.04))
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0, 0.005, n))
        
        candidate = np.clip(candidate, 1e-6, 1000.0)
        new_score = compute_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current = candidate
            current_score = new_score
        
        temp *= cooling
    
    return best, best_score


def make_initial_sequence(n, rng, method='chirp'):
    """Create initial sequences of different types."""
    if method == 'chirp':
        t = np.arange(n, dtype=np.float64)
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (2 * n))
    elif method == 'v_shape':
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * (2 * np.abs(t - 0.5) - 0.5)  # V-shaped
    elif method == 'cosine_taper':
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * (t - 0.5))  # Bell-shaped (opposite of what we want)
        # Actually we want U-shape: high at edges
        a = 0.5 + 0.5 * np.cos(np.pi * t) ** 2
    elif method == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif method == 'two_tone':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * t / n) + 0.1 * np.cos(2 * np.pi * 7 * t / n + 1.0)
    else:
        a = np.ones(n)
    
    a = np.clip(a, 0.01, None)
    return a

"""
LP-based iterative optimization for minimizing peak autocorrelation of
non-negative step functions. Uses the primal linear program (maximize sum(g)
subject to f*g <= M) to find descent directions, with adaptive step sizes,
iterated local search with mutations, and simulated annealing polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_lp_matrix(f):
    """
    Build the (2n-1) x n matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0.
    This encodes the linear convolution (f*g)_k as (A @ g)_k.
    """
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        length = k_end - k_start + 1
        A[k_start:k_end + 1, j] = f[:length]
    return A


def solve_lp_direction(f, M):
    """
    Solve LP: maximize sum(g) subject to (f*g)_k <= M for all k, g >= 0.
    Returns optimal g (or None if infeasible/failed).
    """
    n = len(f)
    A = build_lp_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            return None
        g = np.clip(res.x, 0.0, None)
        if np.sum(g) < 1e-15:
            return None
        return g
    except Exception:
        return None


def lp_search(f_init, time_limit, rng, max_iters=2000, verbose=True):
    """
    Run LP-based iterative search to reduce peak autocorrelation.
    Returns (best_sequence, best_score).
    """
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    if sum_f < 1e-10:
        return f, np.inf
    target_sum = np.sqrt(2.0 * n)
    f = f * (target_sum / sum_f)

    current_score = compute_score(f)
    best_score = current_score
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve_count = 0

    if verbose:
        print(f"    LP init: n={n}, score={current_score:.6f}")

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute M = max(f*f)
        conv_f = fft_convolve(f, f)
        M = np.max(conv_f)

        # Solve LP
        g = solve_lp_direction(f, M)
        if g is None:
            break

        # Rescale g to match target_sum
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g * (target_sum / g_sum)

        # Check if g is essentially f (convergence)
        if np.allclose(g, f, rtol=1e-4, atol=1e-6):
            if verbose and iteration > 10:
                print(f"    LP converged (g ~ f) at iter {iteration}")
            break

        # Try full step
        improved = False
        cand_score = compute_score(g)
        if cand_score < current_score * (1 - 1e-8):
            f = g.copy()
            current_score = cand_score
            improved = True
            no_improve_count = 0
            if current_score < best_score:
                best_score = current_score
                best_f = f.copy()
                if verbose and iteration < 20:
                    print(f"      iter {iteration}: full step, score={current_score:.6f}")
        else:
            # Try sequential blending
            for t in [0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
                candidate = (1.0 - t) * f + t * g
                cs = compute_score(candidate)
                if cs < current_score * (1 - 1e-9):
                    f = candidate
                    current_score = cs
                    improved = True
                    no_improve_count = 0
                    if current_score < best_score:
                        best_score = current_score
                        best_f = f.copy()
                        if verbose and iteration < 20:
                            print(f"      iter {iteration}: t={t}, score={current_score:.6f}")
                    break

        if not improved:
            no_improve_count += 1
            if no_improve_count > 50:
                # Stuck: add perturbation
                noise_scale = 0.05 + 0.05 * (no_improve_count / 100.0)
                f = f * np.exp(rng.normal(0.0, noise_scale, n))
                f = np.clip(f, 1e-8, None)
                f = f * (target_sum / np.sum(f))
                current_score = compute_score(f)
                no_improve_count = 0

        if verbose and iteration % 200 == 0 and iteration > 0:
            elapsed = time.monotonic() - t0
            print(f"      iter {iteration}: score={current_score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")

    return best_f, best_score


def make_initial_sequence(n, rng, method='chirp'):
    """Create initial sequences of various types."""
    t = np.linspace(0, 1.0, n)
    if method == 'chirp':
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t * 2)
    elif method == 'v_shape':
        # High at edges, low in middle
        a = 0.5 + 0.5 * (2.0 * np.abs(t - 0.5)) ** 0.5
    elif method == 'u_shape':
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
    elif method == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif method == 'two_tone':
        tt = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.15 * np.cos(2 * np.pi * 3 * tt / n + rng.uniform(0, 6.28))
        a += 0.1 * np.cos(2 * np.pi * 7 * tt / n + rng.uniform(0, 6.28))
    elif method == 'three_tone':
        tt = np.arange(n, dtype=np.float64)
        a = 1.0
        for k, amp in [(2, 0.12), (5, 0.08), (11, 0.05)]:
            a += amp * np.cos(2 * np.pi * k * tt / n + rng.uniform(0, 6.28))
    elif method == 'linear_taper':
        a = 0.5 + 0.5 * np.abs(2.0 * t - 1.0) ** 1.5
    else:
        a = np.ones(n)

    a = np.clip(a, 0.01, None)
    return a


def local_polish(a_init, time_limit, rng):
    """Simulated annealing local search with targeted peak reduction."""
    current = a_init.astype(np.float64).copy()
    n = len(current)
    target_sum = np.sqrt(2.0 * n)
    current = current * (target_sum / np.sum(current))

    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9993
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.30:
            # Targeted: reduce peak contributors
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            i_start = max(0, peak_idx - n + 1)
            i_end = min(n - 1, peak_idx)
            for i in range(i_start, i_end + 1):
                j = peak_idx - i
                contribs[i] = current[i] * current[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.06, 0.02, len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 3:
                n_inc = min(6, len(other))
                inc_idx = rng.choice(other, n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.06, 0.02, n_inc))
        elif r < 0.45:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, width))
        elif r < 0.55:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 10))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, width))
        elif r < 0.70:
            # Symmetry push
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.003, n))
        elif r < 0.85:
            # Point mutations
            n_mut = int(rng.integers(2, max(3, n // 10)))
            idx = rng.choice(n, n_mut, replace=False)
            candidate[idx] *= np.exp(rng.normal(0.0, 0.05, n_mut))
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.004, n))

        candidate = np.clip(candidate, 1e-8, 1000.0)
        new_score = compute_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current = candidate
            current_score = new_score

        temp *= cooling
        iter_count += 1

        if iter_count % 5000 == 0:
            print(f"      SA iter {iter_count}: score={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.1f}s")

    return best, best_score


def iterated_lp_with_mutation(f_init, time_limit, rng, verbose=True):
    """
    Iterated local search: LP optimization, then mutation, then repeat.
    Keeps track of the best solution found.
    """
    f = f_init.astype(np.float64).copy()
    n = len(f)
    target_sum = np.sqrt(2.0 * n)
    f = f * (target_sum / np.sum(f))

    best_score = compute_score(f)
    best_f = f.copy()
    t0 = time.monotonic()
    round_num = 0
    mutation_scale = 0.05

    while time.monotonic() - t0 < time_limit:
        round_num += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        # LP optimization from current f
        lp_time = min(remaining * 0.7, max(5.0, remaining / 3.0))
        if verbose and round_num <= 3:
            print(f"    ILS round {round_num}: starting LP, score={compute_score(f):.6f}, time_left={remaining:.0f}s")

        f_opt, score_opt = lp_search(f, lp_time, rng, max_iters=500, verbose=verbose and round_num == 1)

        if score_opt < best_score:
            best_score = score_opt
            best_f = f_opt.copy()
            if verbose:
                print(f"      Improved to: {best_score:.6f}")

        # Mutation for next round
        if time.monotonic() - t0 > time_limit - 5.0:
            break

        r = rng.random()
        if r < 0.4:
            # Gaussian noise
            f = f_opt * np.exp(rng.normal(0.0, mutation_scale, n))
        elif r < 0.6:
            # Block mutation
            f = f_opt.copy()
            n_blocks = int(rng.integers(2, max(3, n // 20)))
            for _ in range(n_blocks):
                width = int(rng.integers(1, max(2, n // 8)))
                start = int(rng.integers(0, n - width))
                f[start:start + width] *= np.exp(rng.normal(0.0, mutation_scale * 1.5, width))
        elif r < 0.8:
            # Targeted mutation on peak
            conv = fft_convolve(f_opt, f_opt)
            peak_idx = int(np.argmax(conv))
            f = f_opt.copy()
            i_start = max(0, peak_idx - n + 1)
            i_end = min(n - 1, peak_idx)
            n_mut = min(10, i_end - i_start + 1)
            idx = rng.choice(range(i_start, i_end + 1), n_mut, replace=False)
            f[idx] *= np.exp(rng.normal(-mutation_scale, mutation_scale, n_mut))
            other = np.setdiff1d(np.arange(n), idx)
            if len(other) > 3:
                inc = rng.choice(other, min(5, len(other)), replace=False)
                f[inc] *= np.exp(rng.normal(mutation_scale, mutation_scale, len(inc)))
        else:
            # Restart with fresh random init (keep best so far)
            f = make_initial_sequence(n, rng, method='random')
            mutation_scale = max(0.03, mutation_scale * 0.8)

        f = np.clip(f, 1e-8, None)
        f = f * (target_sum / np.sum(f))

        # Slowly increase mutation
        mutation_scale = min(0.15, mutation_scale * 1.02)

    return best_f, best_score

            print(f"  Score before polish: {best_score:.6f}, time={polish_time:.0f}s")
            polished, polish_score = local_polish(best_seq, polish_time, rng)
            if polish_score < best_score:
                best_score = polish_score
                best_seq = polished.copy()
                print(f"  After polish: {best_score:.6f}")

    # === Phase 4: Final LP pass on polished result ===
    if time.monotonic() < deadline - 30:
        print(f"\n=== Phase 4: Final LP refinement ===")
        remaining = deadline - time.monotonic()
        t4 = min(remaining * 0.7, 200.0)
        if t4 > 10:
            f_opt, s_opt = lp_search(best_seq, t4, rng, max_iters=500, verbose=False)
            if s_opt < best_score:
                best_score = s_opt
                best_seq = f_opt.copy()
                print(f"  After final LP: {best_score:.6f}")

        # Tiny final polish
        if time.monotonic() < deadline - 15:
            ft = deadline - time.monotonic() - 5
            if ft > 5:
                polished, ps = local_polish(best_seq, ft, rng)
                if ps < best_score:
                    best_score = ps
                    best_seq = polished.copy()
                    print(f"  Final tiny polish: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
