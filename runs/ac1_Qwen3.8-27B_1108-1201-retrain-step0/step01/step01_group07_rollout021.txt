```python
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), 
                  bounds=[(0, None)] * n, method='highs')
    
    if res.status != 0:
        return None
    return res.x
```

Wait, this might be slow to construct the sparse matrix in a Python loop. Let me use a more efficient construction.

The matrix `A` is a banded Toeplitz matrix. Row `k` has nonzeros at columns `j` where `0 <= k-j < n`, i.e., `k-n+1 <= j <= k` (intersected with `[0, n-1]`).

I can construct it using `scipy.sparse.diags` or by directly computing the row/column indices vectorized.

```python
def build_conv_matrix(f, n):
    """Build the matrix A where (A @ g)[k] = conv(f, g)[k]."""
    # For each k in 0..2n-2, the row has entries f[k-j] at column j
    # where 0 <= k-j < n, i.e., k-n+1 <= j <= k
    # Intersect with [0, n-1]: j in [max(0, k-n+1), min(n-1, k)]
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        rows.extend(np.full(len(js), k))
        cols.extend(js)
        vals.extend(f[k - js])
    
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
```

This should be fast enough. For `n = 3000`, the total number of nonzeros is `sum_{k=0}^{2n-2} (min(k+1, n) - max(0, k-n+1) + 1)` = `n^2` (exactly, since it's a convolution matrix).

For `n = 3000`, that's 9 million nonzeros. The sparse matrix should be manageable.

Actually, I realize there might be an issue with the LP approach. Let me reconsider.

The LP finds `g` that maximizes `sum(g)` subject to `conv(f, g) <= M` and `g >= 0`. The optimal `g` will typically have many zero entries (since we're maximizing the sum, the solver will push mass to where it's "cheapest" in terms of the convolution constraints).

After updating `f' = (1-t)*f + t*g` and re-normalizing, the new sequence should have a lower peak in its self-convolution. The re-normalization (scaling down) helps because `sum(g) > S` typically.

Let me also think about what happens if the LP is infeasible or unbounded. It shouldn't be infeasible (since `g = 0` is always feasible), and it shouldn't be unbounded (since `conv(f,g)[0] = f[0]*g[0] <= M` bounds `g[0]`, and similarly for other entries by induction).

One more idea: instead of the LP, I could use a simpler "water-filling" approach. The constraint `conv(f, g)[k] <= M` means that for each `k`, the dot product of `g` with the "shifted" `f` is bounded by `M`. This is like a set of "knapsack" constraints. A greedy approach would be to fill in `g` one coordinate at a time, at each step choosing the coordinate that gives the most "sum" per "constraint budget".

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding,
projected gradient descent with correct gradient, and coordinate descent polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = a.size
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, a.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    # Fix rounding
    deficit = S - res.sum()
    if deficit > 0:
        idx = res.argmax()
        res[idx] += deficit
    elif deficit < 0:
        idx = np.where(res > 0)[0]
        # Remove from the largest
        res[idx[np.argmax(res[idx])]] += deficit
    return res


def compute_conv(a):
    """Compute full convolution of a with itself."""
    return np.convolve(a, a)


def soft_lse_grad(a, tau):
    """Compute gradient of tau * logsumexp(conv(a,a)) with respect to a.
    
    Returns (gradient, convolution).
    """
    c = np.convolve(a, a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    # Correct gradient: 2 * correlate(q, a, 'valid')
    grad = 2.0 * np.correlate(q, a, 'valid')
    return grad, c


def build_conv_matrix(f, n):
    """Build sparse matrix A where (A @ g)[k] = sum_j f[k-j] * g[j]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        rows.extend(np.full(len(js), k, dtype=np.int32))
        cols.extend(js.astype(np.int32))
        vals.extend(f[k - js])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g >= 0, conv(f,g) <= M."""
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, 
                  method='highs', options={'disp': False, 'presolve': True})
    if res.status != 0:
        return None
    return res.x


def lp_optimize(f, S, t_budget, t_lr=0.3, verbose=False):
    """LP-based optimization loop."""
    n = f.size
    a = f.copy()
    best = a.copy()
    best_M = np.convolve(best, best).max()
    
    t0 = time.monotonic()
    iteration = 0
    
    while time.monotonic() - t0 < t_budget:
        c = np.convolve(a, a)
        M = c.max()
        
        g = solve_lp_direction(a, M, n)
        if g is None:
            break
        
        # Update: a' = (1-t)*a + t*g, then re-normalize
        t = t_lr
        a_new = (1 - t) * a + t * g
        a_new = project_simplex(a_new, S)
        
        # Check if improved
        new_M = np.convolve(a_new, a_new).max()
        
        if new_M < M or new_M < best_M:
            a = a_new
            if new_M < best_M:
                best_M = new_M
                best = a_new.copy()
                if verbose:
                    print(f"    LP iter {iteration}: M={new_M:.6f}")
        else:
            # Try smaller step
            t = t_lr * 0.1
            a_new = (1 - t) * a + t * g
            a_new = project_simplex(a_new, S)
            new_M = np.convolve(a_new, a_new).max()
            if new_M < best_M:
                a = a_new
                best_M = new_M
                best = a_new.copy()
            else:
                a = a_new  # Still move, might help escape
        
        iteration += 1
    
    return best, best_M


def gradient_optimize(a0, tau_hi, tau_lo, t_budget, verbose=False):
    """FISTA with soft LSE smoothing."""
    n = a0.size
    S = np.sqrt(2 * n)
    a = (a0.astype(np.float64) / max(a0.sum(), 1e-12) * S).copy()
    a = project_simplex(a, S)
    
    best = a.copy()
    best_M = np.convolve(best, best).max()
    
    t0 = time.monotonic()
    # FISTA parameters
    z = a.copy()
    a_prev = a.copy()
    t_k = 1.0
    
    # Lipschitz constant estimate
    L = 2.0 * n / max(tau_lo, 1e-3) + 10.0
    
    step_count = 0
    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(t_budget, 1e-6))
        tau = tau_hi * (tau_lo / tau_hi) ** frac
        
        # Compute gradient at z
        g, c = soft_lse_grad(z, tau)
        
        # FISTA update
        a_new = z - g / L
        a_new = project_simplex(a_new, S)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = a_new + (t_k - 1.0) / t_new * (a_new - a_prev)
        z = project_simplex(z, S)
        
        a_prev = a_new
        a = a_new
        t_k = t_new
        step_count += 1
        
        if step_count % 20 == 0:
            M = np.convolve(a, a).max()
            if M < best_M:
                best_M = M
                best = a.copy()
                if verbose and step_count % 200 == 0:
                    print(f"    FISTA step {step_count}: M={M:.6f}, tau={tau:.4f}")

def coordinate_polish(a0, S, t_budget, verbose=False):
    """Coordinate descent: for each coordinate, find optimal value."""
    n = a0.size
    a = a0.copy()
    best_M = np.convolve(a, a).max()
    best = a.copy()
    
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < t_budget:
        c = np.convolve(a, a)
        M = c.max()
        improved = False
        
        for j in range(n):
            # Try to optimize a[j]
            # The affected convolution coefficients are c[k] for k in [j, j+n-1]
            # Changing a[j] by delta changes c[k] by 2*delta*a[k-j] for k in [j, j+n-1]
            
            # Simple approach: try a few values of a[j]
            old_val = a[j]
            if old_val < 1e-15:
                continue
            
            # Binary search for optimal a[j]
            # The objective max_c is piecewise linear in a[j]
            # Try: reduce a[j] and distribute to other coordinates
            
            # Simple heuristic: try reducing a[j] by a small amount
            # and adding it to the coordinate with smallest contribution
            for frac_try in [0.9, 0.8, 0.5, 0.3, 0.1]:
                delta = old_val * (1 - frac_try)
                a[j] = old_val * frac_try
                # Redistribute delta to a random other coordinate
                # Actually, project to simplex
                a = project_simplex(a, S)
                new_M = np.convolve(a, a).max()
                if new_M < M:
                    M = new_M
                    improved = True
                    break
                else:
                    a[j] = a[j] + delta  # Revert (approximately)
                    a = project_simplex(a, S)
            
            if time.monotonic() - t0 > t_budget:
                break
        
        if M < best_M:
            best_M = M
            best = a.copy()
            if verbose and sweep % 5 == 0:
                print(f"    Coord sweep {sweep}: M={best_M:.6f}")
        
        sweep += 1
    
    return best, best_M

    rng = np.random.default_rng(seed)
