def compute_softmax_loss_grad(a, tau):
    """Compute soft-max smoothed loss and gradient of max(convolve(a,a))."""
    c = np.convolve(a, a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    grad = 2.0 * np.correlate(q, a, mode='valid')
    loss = tau * np.log(exp_c.sum()) + c.max()
    return loss, grad, c


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    s = a.sum()
    if s == 0:
        return np.full(n, S / n)
    if abs(s - S) < 1e-15:
        return a
    if s < S:
        return a + (S - s) / n
    lo, hi = 0.0, float(a.max())
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    diff = S - res.sum()
    if diff > 0:
        idx = int(np.argmax(res))
        res[idx] += diff
    elif diff < 0 and res.sum() > 0:
        # Remove excess from the largest element
        idx = int(np.argmax(res))
        res[idx] += diff
    return res

    a += 0.01 * rng.standard_normal(n)

    """Chirp-based initialization."""

    a += 0.05 * rng.standard_normal(n)

    Returns (best_sequence, best_score).
    """
    n = a.size
    S = 1.0  # Fix sum to 1
    a = (a / a.sum() * S).copy()
    a = project_simplex(a, S)
    
    best = a.copy()
    best_mc = np.convolve(best, best).max()
    best_score = evaluate_sequence(best.tolist())
    
    v = np.zeros(n)
    beta = beta_momentum
    lr = lr_init
    t0 = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 200
    last_best_mc = best_mc
    
    while time.monotonic() - t0 < t_budget:

                    # Don't call evaluate_sequence too often (it's slow due to safety checks)

                best = a.copy()

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using projected gradient descent
with soft-max smoothing, Nesterov momentum, adaptive learning rate, and
multi-start exploration over diverse initializations (exponential, power-law,
chirp, spectral, random) and multiple sequence lengths. Includes tau annealing
for progressive peak concentration and post-optimization local search.
"""

import numpy as np
import time


def compute_softmax_loss_grad(a, tau):
    """Compute soft-max smoothed max of convolve(a,a) and its gradient w.r.t. a."""
    c = np.convolve(a, a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    q = exp_c / exp_c.sum()
    grad = 2.0 * np.correlate(q, a, mode='valid')
    loss = c_max + tau * np.log(exp_c.sum())
    return loss, grad, c


def project_simplex(a, S):
    """Euclidean projection onto simplex {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    if n == 0:
        return a
    s = a.sum()
    if s < 1e-18:
        return np.full(n, S / n)
    if abs(s - S) < 1e-14:
        return a
    if s < S:
        return a + (S - s) / n
    # s > S: need to subtract a threshold
    lo, hi = 0.0, float(a.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    res = np.maximum(a - t, 0.0)
    diff = S - res.sum()
    if abs(diff) > 1e-15:
        idx = int(np.argmax(res))
        res[idx] += diff
        res[idx] = max(0.0, res[idx])
    return res


def init_exponential(n, beta, seed=0):
    """Exponential decay initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-beta * t / n)
    rng = np.random.default_rng(seed)
    a += 0.005 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_power_law(n, alpha, seed=0):
    """Power-law decay initialization."""
    t = np.arange(n, dtype=np.float64) + 1.0
    a = t ** (-alpha)
    rng = np.random.default_rng(seed)
    a += 0.005 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_chirp(n, seed=0):
    """Chirp-based initialization with varying frequency."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.7 * np.cos(np.pi * t * t / (0.3 * n))
    rng = np.random.default_rng(seed)
    a += 0.03 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_spectral(n, seed=0):
    """Spectral initialization: IFFT of random phases with constant magnitude."""
    rng = np.random.default_rng(seed)
    phases = rng.uniform(0.0, 2.0 * np.pi, size=n)
    F = np.exp(1j * phases)
    a = np.real(np.fft.ifft(F))
    return np.abs(a)


def init_random(n, seed=0):
    """Random exponential initialization."""
    return np.random.default_rng(seed).exponential(1.0, size=n)


def optimize_sequence(a, tau_start, tau_end, t_budget, beta_momentum=0.85,
                      lr_init=1.0, target_step_frac=0.01, eval_every=100):
    """
    Optimize a sequence using projected gradient descent with soft-max smoothing,
    Nesterov momentum, and adaptive learning rate.
    Returns (best_sequence, best_score).
    """
    n = a.size
    S = 1.0
    a = (a / max(a.sum(), 1e-15) * S).copy()
    a = project_simplex(a, S)

    best = a.copy()
    best_mc = np.convolve(best, best).max()
    best_score = evaluate_sequence(best.tolist())

    v = np.zeros(n)
    beta = beta_momentum
    lr = lr_init
    t0 = time.monotonic()
    step = 0
    last_best_mc = best_mc
    plateau_counter = 0
    max_plateau = 300

    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / t_budget)
        # Smooth annealing with a floor
        tau = tau_start * (tau_end / tau_start) ** frac
        tau = max(tau, 1e-6)

        loss, grad, c = compute_softmax_loss_grad(a, tau)

        # Nesterov momentum
        v = beta * v + (1.0 - beta) * grad
        v_norm = np.linalg.norm(v)
        if v_norm > 1e-15:
            # Adaptive lr: target step size as fraction of S
            adaptive_lr = target_step_frac * S / v_norm
            lr = min(2.0, max(0.05, lr * 0.95 + adaptive_lr * 0.05))
        a_new = a - lr * v
        a_new = project_simplex(a_new, S)

        # Check actual max convolution
        mc = np.convolve(a_new, a_new).max()

        a = a_new
        step += 1

        if step % 5 == 0:
            if mc < last_best_mc - 1e-12:
                plateau_counter = 0
                last_best_mc = mc
                if mc < best_mc - 1e-12:
                    best_mc = mc
                    best = a.copy()
                # Slightly increase lr on progress
                lr = min(3.0, lr * 1.01)
            else:
                plateau_counter += 5
                # Reduce lr on stagnation
                lr = max(0.05, lr * 0.85)
                if plateau_counter > max_plateau:
                    # Dampen momentum to help escape
                    v *= 0.3
                    plateau_counter = 0

        # Periodic true evaluation
        if step % eval_every == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best = a.copy()

    # Final evaluation
    sc = evaluate_sequence(best.tolist())
    if sc < best_score:
        best_score = sc

    return best, best_score


def local_perturb_search(a, t_budget, n_trials=200):
    """Random perturbation local search to escape local minima."""
    n = a.size
    S = a.sum()
    best = a.copy()
    best_mc = np.convolve(best, best).max()
    t0 = time.monotonic()

    for trial in range(n_trials):
        if time.monotonic() - t0 > t_budget:
            break
        rng = np.random.default_rng(42 + trial)
        # Geometrically scaled perturbations
        scale = 0.02 * best.mean() * (1.0 + 0.5 * trial / n_trials)
        a_pert = best + scale * rng.standard_normal(n)
        a_pert = project_simplex(a_pert, S)
        mc = np.convolve(a_pert, a_pert).max()
        if mc < best_mc:
            best = a_pert.copy()
            best_mc = mc

    return best
