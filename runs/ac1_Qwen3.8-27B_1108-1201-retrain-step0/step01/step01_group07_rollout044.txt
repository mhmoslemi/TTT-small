import numpy as np
import time


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    s = a.sum()
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, a.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    idx = res.argmax()
    res[idx] += S - res.sum()
    return res


def compute_soft_max_grad(a, tau):
    """Compute gradient of tau*log-sum-exp(conv(a,a)) with respect to a."""
    c = np.convolve(a, a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    # grad[k] = 2 * sum_i q[i] * a[i+k-n+1] (correlation form)
    # Actually: d/da[j] of (a*a)[k] = a[k-j] for valid k-j
    # sum_k q[k] * d(a*a)[k]/da[j] = sum_k q[k] * a[k-j] (for valid indices)
    # This is a correlation
    grad = 2.0 * np.correlate(q, a, mode='full')
    # np.correlate(q, a, 'full')[k] = sum_i q[i] * a[i + k - (len(a)-1)]
    # We need the part where indices align with a (length n)
    # The full correlation has length len(q) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    # We want the middle n values
    grad = grad[n - 1:2 * n - 1]
    return grad, c

    # This is a cross-correlation
    grad = 2.0 * np.correlate(q, a, mode='full')
    # np.correlate(q, a, 'full')[m] = sum_i q[i] * a[i + m - (n-1)]
    # We want grad[j] = 2 * sum_i q[j + i] * a[i] for i in [0, n-1]
    # = 2 * sum_i q[i + j] * a[i]
    # Let's verify: correlate(q, a, 'full')[j + n - 1] = sum_i q[i] * a[i + j + n - 1 - n + 1] = sum_i q[i] * a[i+j]
    # Hmm that's not quite right. Let me use a different approach.
    # 
    # Actually, let me just use the direct formula:
    # grad[j] = 2 * sum_{i: j+i < 2n-1} q[j+i] * a[i]
    # The full correlation np.correlate(q, a, 'full') has length 3n-2
    # Full correlation: fc[m] = sum_i q[i] * a[i - m + (n-1)] for valid i
    # 
    # Let me just compute it directly using the fact that:
    # sum_k q[k] * a[k-j] where k ranges over [max(0,j), min(2n-2, j+n-1)]
    # This is equivalent to: for each j, take the slice q[j:j+2n-1] and multiply by a
    # 
    # Actually the simplest correct formula:
    # grad = 2 * (q correlated with a) where we take the right slice
    # Let me verify with a small example in code.
    # 
    # For now, use the 'valid' mode correlation which gives:
    # np.correlate(q, a, 'valid')[j] = sum_i q[j+i] * a[i] for j in [0, 2n-1-n] = [0, n-1]
    # Wait, 'valid' returns length max(len(q), len(a)) - min(len(q), len(a)) + 1
    # len(q) = 2n-1, len(a) = n
    # 'valid' length = (2n-1) - n + 1 = n
    # correlate(q, a, 'valid')[j] = sum_{i=0}^{n-1} q[j+i] * a[i]
    # That's exactly what we want!
    grad = 2.0 * np.correlate(q, a, mode='valid')
    return grad, c


def init_chirp(n, seed=0):
    """Chirp-based initialization."""
    t = np.arange(n, dtype=np.float64) - (n - 1) * 0.5
    a = 1.0 + 0.9 * np.cos(np.pi * t * t / n)
    rng = np.random.default_rng(seed)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_random(n, seed=0):
    """Random exponential initialization."""
    return np.random.default_rng(seed).exponential(1.0, size=n)


def init_u_shape(n):
    """U-shaped initialization (large at edges, small in middle)."""
    t = np.linspace(-1, 1, n)
    a = 1.0 + 2.0 * t * t
    return a


def init_w_shape(n):
    """W-shaped initialization."""
    t = np.linspace(0, 1, n)
    a = 1.0 + 1.5 * np.cos(4 * np.pi * t)
    return np.maximum(a, 0.1)


def init_hamming(n):
    """Hamming window initialization (inverted - large in middle)."""
    w = np.hamming(n)
    a = 2.0 - w  # inverted hamming = large at edges
    return np.maximum(a, 0.1)


def init_inverse_hamming(n):
    """Inverted Hamming - large at edges, small in middle."""
    w = np.hamming(n)
    a = 1.5 - 0.8 * w
    return np.maximum(a, 0.05)


def init_gauss_center(n):
    """Gaussian centered."""
    t = np.linspace(-3, 3, n)
    a = np.exp(-t * t / 2) + 0.2
    return a


def init_gauss_edges(n):
    """Gaussian at edges."""
    t = np.linspace(-3, 3, n)
    a = 1.5 * np.exp(-(t - 2.5) ** 2 / 0.5) + 1.5 * np.exp(-(t + 2.5) ** 2 / 0.5) + 0.3
    return a


def run_nesterov_pgd(a0, tau_hi, tau_lo, lr0, t_budget, momentum=0.9,
                      perturbation_interval=500, perturbation_size=0.05,
                      verbose=False):
    """Nesterov-accelerated projected gradient with soft-max, backtracking, and perturbation."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = (a0.astype(np.float64) / max(a0.sum(), 1e-12) * S).copy()
    a = project_simplex(a, S)
    a_prev = a.copy()
    
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    
    t0 = time.monotonic()
    lr = lr0
    step = 0
    worst_consecutive = 0
    max_worst = 300
    
    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        f = min(1.0, elapsed / max(t_budget, 1e-6))
        # Aggressive tau annealing: start high, go very low
        tau = tau_hi * (tau_lo / tau_hi) ** (f ** 0.7)
        
        # Nesterov step
        v = a - a_prev
        y = a + momentum * v
        
        # Compute gradient at y
        g, c = compute_soft_max_grad_v2(y, tau)
        
        # Backtracking line search
        current_obj = np.log(np.exp((c - c.max()) / tau).sum()) * tau
        # Actually, let's just use the max of c as objective (simpler)
        current_max = c.max()
        
        accepted = False
        for backtrack in range(10):
            a_new = y - lr * (1.0 + backtrack * 0.5) * g
            a_new = project_simplex(a_new, S)
            
            # Check if objective improved (using max of conv)
            c_new = np.convolve(a_new, a_new)
            max_new = c_new.max()
            
            if max_new < current_max * (1 + 1e-10) or backtrack == 9:
                a_prev = a
                a = a_new
                accepted = True
                break
        
        if not accepted:
            break
        
        step += 1
        
        # Track best
        if step % 20 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best = a.copy()
                worst_consecutive = 0
            else:
                worst_consecutive += 20
        
        # Periodic perturbation to escape local minima
        if perturbation_interval > 0 and step % perturbation_interval == 0:
            if worst_consecutive >= max_worst:
                rng = np.random.default_rng(step)
                noise = perturbation_size * rng.standard_normal(n)
                a = project_simplex(a + noise, S)
                a_prev = a.copy()
                worst_consecutive = 0
                if verbose:
                    print(f"      Perturbation at step {step}")
            # Decay momentum slightly after perturbation
            momentum = min(0.95, momentum * 1.01)
        
        # Decay learning rate slowly
        lr *= 0.9999
        
    return best, best_score


def run_pgd_simple(a0, tau_hi, tau_lo, lr0, t_budget, verbose=False):
    """Simple projected gradient with adaptive lr (faster per iteration)."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = (a0.astype(np.float64) / max(a0.sum(), 1e-12) * S).copy()
    a = project_simplex(a, S)
    
    best = a.copy()
    best_mc = np.convolve(best, best).max()
    best_score = evaluate_sequence(best.tolist())
    
    t0 = time.monotonic()
    lr = lr0
    step = 0
    last_improve_step = 0
    
    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        f = min(1.0, elapsed / max(t_budget, 1e-6))
        tau = tau_hi * (tau_lo / tau_hi) ** (f ** 0.5)
        
        g, c = compute_soft_max_grad_v2(a, tau)
        a_new = a - lr * g
        a_new = project_simplex(a_new, S)
        step += 1
        
        if step % 15 == 1:
            mc = np.convolve(a_new, a_new).max()
            if mc < best_mc * (1 + 1e-8):
                a = a_new
                if mc < best_mc - 1e-10:
                    best_mc = mc
                    best = a.copy()
                    best_score = evaluate_sequence(best.tolist())
                    last_improve_step = step
                    if lr < lr0 * 5.0:
                        lr *= 1.03
            else:
                a = a_new
                lr = max(0.01, lr * 0.7)
                
            # Stagnation detection - add perturbation
            if step - last_improve_step > 500:
                rng = np.random.default_rng(step)
                noise = 0.03 * rng.standard_normal(n)
                a = project_simplex(a + noise, S)
                a_new = a.copy()
                last_improve_step = step
                if verbose:
                    print(f"      Perturbation at step {step}")
        else:
            a = a_new
    
    return best, best_score


def run_multiscale(a0, t_budget, verbose=False):
    """Multi-scale optimization: start with coarse (large tau), refine with fine (small tau)."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = (a0.astype(np.float64) / max(a0.sum(), 1e-12) * S).copy()
    a = project_simplex(a, S)
    
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    
    t0 = time.monotonic()
    
    # Schedule: (tau_hi, tau_lo, lr, fraction_of_time)
    schedule = [
        (5.0, 0.5, 2.0, 0.15),
        (0.5, 0.1, 1.0, 0.25),
        (0.1, 0.02, 0.5, 0.30),
        (0.02, 0.003, 0.2, 0.30),
    ]
    
    for tau_hi, tau_lo, lr, frac in schedule:
        elapsed = time.monotonic() - t0
        remaining = t_budget - elapsed
        if remaining < 10:
            break
        tb = remaining * frac
        if verbose:
            print(f"      Scale: tau=[{tau_hi},{tau_lo}] lr={lr} t={tb:.0f}s")
        
        a_r, sc = run_pgd_simple(a, tau_hi, tau_lo, lr, tb)
        if sc < best_score:
            best_score = sc
            best = a_r.copy()
        a = a_r
    
    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search: multi-phase exploration and optimization."""
    dl = time.monotonic() + budget_s - 5.0
    bs = None
    bsc = np.inf
    
    # Get provided sequence
    prov = None
    try:
        prov = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if prov.size > 0 and np.isfinite(prov).all():
            prov = np.maximum(prov, 0.0)
            if prov.sum() < 0.01:
                prov = None
        else:
            prov = None
    except Exception:
        prov = None
    
    print(f"Starting search. Budget: {budget_s}s. Prov: {prov is not None}")
    if prov is not None:
        print(f"  Prov length: {len(prov)}, score: {evaluate_sequence(prov.tolist()):.6f}")
    
    # Phase 1: Quick exploration (~250s)
    p1_end = time.monotonic() + 250.0
    print(f"\nPhase 1: Exploration until t={p1_end - time.monotonic():.0f}s")
    
    n_opts = [5000, 8000, 12000]
    if prov is not None:
        n_opts.append(len(prov))
    n_opts = sorted(set(n_opts))
    
    init_funcs = [
        ('chirp0', lambda n, s: init_chirp(n, s)),
        ('rand', lambda n, s: init_random(n, s)),
        ('u_shape', lambda n, s: init_u_shape(n)),
        ('w_shape', lambda n, s: init_w_shape(n)),
        ('inv_hamm', lambda n, s: init_inverse_hamming(n)),
        ('edge_gauss', lambda n, s: init_gauss_edges(n)),
    ]
    
    candidates = []
    for n in n_opts:
        for name, func in init_funcs[:4]:  # Use first 4 inits for speed
            candidates.append((n, name, func, 0))
    if prov is not None:
        candidates.insert(0, (len(prov), 'provided', None, 0))
    
    t_per = max(20.0, (p1_end - time.monotonic()) / max(1, len(candidates)))
    print(f"  {len(candidates)} candidates, ~{t_per:.0f}s each")
    
    for i, (n, name, func, sd) in enumerate(candidates):
        tb = p1_end - time.monotonic()
        if tb < 15:
            break
        tb = min(t_per, tb)
        
        if name == 'provided' and prov is not None:
            a0 = prov.copy()
        else:
            a0 = func(n, sd)
        
        sc_est = evaluate_sequence(a0.tolist())
        a_r, sc = run_pgd_simple(a0, 1.0, 0.05, 1.0, tb)
        
        tag = f"  P1[{i+1}/{len(candidates)}] n={n} {name} init_sc={sc_est:.4f}"
        if sc < bsc:
            bsc = sc
            bs = a_r.copy()
            tag += f" *** NEW BEST: {bsc:.6f}"
            print(tag)
        else:
            print(tag + f" -> sc={sc:.6f}")
    
    # Phase 2: Multi-scale focused optimization (~1200s)
    rem = dl - time.monotonic()
    if rem > 100 and bs is not None:
        n_best = bs.size
        print(f"\nPhase 2: Multi-scale optimization n={n_best}, {rem:.0f}s remaining")
        
        # Run multi-scale on best
        a_r, sc = run_multiscale(bs, rem * 0.7, verbose=True)
        if sc < bsc:
            bsc = sc
            bs = a_r.copy()
            print(f"  Multi-scale improved: {bsc:.6f}")
        
        # Also try Nesterov on best
        rem2 = dl - time.monotonic()
        if rem2 > 60:
            a_r2, sc2 = run_nesterov_pgd(bs, 0.5, 0.01, 0.5, rem2 * 0.85,
                                          momentum=0.9, verbose=True)
            if sc2 < bsc:
                bsc = sc2
                bs = a_r2.copy()
                print(f"  Nesterov improved: {bsc:.6f}")
    
    # Phase 3: Fine-tuning with perturbations (~300s)
    rem = dl - time.monotonic()
    if rem > 30 and bs is not None:
        n_best = bs.size
        print(f"\nPhase 3: Fine-tuning {rem:.0f}s remaining")
        
        # Multiple short runs with tiny perturbations
        n_runs = min(6, max(2, int(rem / 40)))
        per_run = rem / n_runs * 0.85
        
        for r in range(n_runs):
            rem = dl - time.monotonic()
            if rem < 20:
                break
            rng = np.random.default_rng(1000 + r)
            noise = 0.01 * rng.standard_normal(n_best)
            a_pert = project_simplex(bs + noise, np.sqrt(2.0 * n_best))
            
            a_r, sc = run_pgd_simple(a_pert, 0.05, 0.002, 0.1, rem - 10)
            if sc < bsc:
                bsc = sc
                bs = a_r.copy()
                print(f"  Fine-tune {r+1}: {bsc:.6f}")
            else:
                print(f"  Fine-tune {r+1}: {sc:.6f} (no improvement)")
    
    # Final: try the provided sequence upscaled + optimized
    rem = dl - time.monotonic()
    if rem > 60 and prov is not None:
        print(f"\nBonus: Optimizing provided sequence, {rem:.0f}s")
        a_r, sc = run_multiscale(prov, rem - 10, verbose=True)
        if sc < bsc:
            bsc = sc
            bs = a_r.copy()
            print(f"  Provided improved: {bsc:.6f}")
    
    # Final check
    if bs is None:
        bs = init_uniform(2000)
        bsc = evaluate_sequence(bs.tolist())
    
    print(f"\n=== FINAL: score={bsc:.6f}, n={len(bs)} ===")
    return [float(x) for x in bs]


def init_uniform(n):
    """Uniform initialization."""
    return np.ones(n, dtype=np.float64)
