Now the question is: what sequence achieves a convolution that is as flat as possible?

The convolution of a with itself is flat (constant) if and only if... hmm, this is a well-studied problem. A sequence whose autocorrelation is as flat as possible is related to "flat partial autocorrelation" or "low autopower" sequences.

For a sequence of length n, the full autocorrelation (convolution with itself) has 2n-1 values. We want to minimize the maximum, given that the sum of all 2n-1 values is (sum a)^2.

The minimum possible maximum is (sum a)^2 / (2n-1), achieved when all 2n-1 values are equal. This is the "perfect" autocorrelation.

But can we actually achieve this? For the full autocorrelation to be perfectly flat, we need:
sum_j a[j] * a[k-j] = constant for all k = 0, 1, ..., 2n-2.

This is equivalent to saying the Fourier transform of a has constant magnitude (a "flat spectrum" or "allpass" sequence). In the discrete case, this is related to Zadoff-Chu sequences and other "CAZAC" (Constant Amplitude Zero Autocorrelation) sequences, but those are for complex sequences.

For real non-negative sequences, achieving a perfectly flat autocorrelation is generally impossible. But we can get close.

from scipy.sparse import csr_matrix

import time

def normalize(a):
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-18:
        return a
    return a / s

def score(a):

    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (a.sum()**2)

def gradient_l2(a):
    """Compute gradient of ||conv(a,a)||_2^2 with respect to a."""
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1

    # and sum(d) = 0 (we'll handle normalization separately)
    d = d.copy()
    d -= d.mean()  # ensure sum = 0

    else:
        t_max = 1.0

        else:
            t_max = 10.0

            eta *= 0.7

where softmax_k = exp(beta * c_k) / sum_m exp(beta * c_m).

For large beta, this concentrates on the argmax indices, recovering the subgradient.

def adam_optimize(a, time_limit=60.0, beta=20.0, lr=0.01, verbose=True):
    start = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score(a)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    for it in range(10000):
        if time.monotonic() - start > time_limit:
            break
        t += 1
        
        c = np.convolve(a, a)
        cur_score = 2.0 * n * c.max() / (a.sum()**2)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
        
        # Softmax weights
        shifted = beta * c - beta * c.max()
        w = np.exp(shifted)
        w /= w.sum()
        
        # Gradient
        g = np.zeros(n)
        for j in range(n):
            g[j] = 2.0 * np.dot(a, w[j:j+n])
        
        # Adam update
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Update (in log-space for non-negativity)
        a_new = a * np.exp(-lr * m_hat / np.sqrt(v_hat + eps))
        a_new = normalize(a_new)
        
        # Check constraint satisfaction (a_new >= 0 is guaranteed by exp)
        a = a_new
        
        if verbose and it % 500 == 0:
            print(f"    Adam iter {it}: score={cur_score:.6f}")
    
    return best_a, best_score

```python
"""
Minimizes peak auto-convolution of non-negative sequences using
Adam-based gradient optimization with softmax smoothing, LP refinement,
and local search polishing. Uses multiple sequence lengths and
initializations for robustness.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_sequence(a):
    n = len(a)
    c = np.convolve(a, a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


def softmax_gradient(a, beta=20.0):
    """Compute gradient of softmax-approximated max convolution."""
    n = len(a)
    c = np.convolve(a, a)
    shifted = beta * c - beta * c.max()
    w = np.exp(shifted)
    w /= w.sum()
    g = np.zeros(n)
    for j in range(n):
        g[j] = 2.0 * np.dot(a, w[j:j+n])
    return g, c


def lp_solve(a, M, timeout=10.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    n_rows = 2 * n - 1
    # Build convolution matrix: (C g)_k = sum_j a_j * g_{k-j}
    # C is n_rows x n
    # C[k, j] = a_{k-j} if 0 <= k-j < n, else 0
    row_idx = []
    col_idx = []
    data = []
    for k in range(n_rows):
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            row_idx.append(k)
            col_idx.append(j)
            data.append(a[k - j])
    
    C = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))
    c_obj = -np.ones(n)
    b_ub = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
        if res.success:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def adam_optimize(a, time_limit=60.0, beta_softmax=20.0, lr=0.05, verbose=False):
    """Optimize using Adam with soft
