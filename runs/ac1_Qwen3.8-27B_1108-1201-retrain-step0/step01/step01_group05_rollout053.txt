    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]

    """Compute softmax weights and gradient of t*logsumexp(c/t)."""

        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * (grad ** 2)
        m_u = m_hat / (1 - beta1 ** t)
        v_u = v_hat / (1 - beta2 ** t)
        
        # Learning rate decay
        lr_t = lr / (1.0 + 0.001 * t)
        
        a = project_simplex(a - lr_t * m_u / (np.sqrt(v_u) + eps), S)
        
        # Periodic noise
        if t % 2000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
        
        # Log progress
        if t % 5000 == 0:
            print("  smooth_adam t=%d maxc=%.6f temp=%.4f" % (t, max_c, temp))
    
    return best_a, best_maxc


def run_mirror_smooth(a_init, n, max_time, rng, eta_init=0.5, t_ratio_start=0.5, t_ratio_end=0.05):
    """Mirror descent with smooth objective (multiplicative weights)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(a_init.astype(np.float64).copy(), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a * (S / s)
    
    fsize = 1 << (3 * n - 1 - 1).bit_length()
    t = 0
    start = time.monotonic()
    best_maxc = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max_time)
        
        c_curr = fft_convolve_self(a)
        max_c = c_curr.max()
        
        if max_c < best_maxc:
            best_maxc = max_c
            best_a = a.copy()
        
        temp = max_c * t_ratio_start * (t_ratio_end / t_ratio_start) ** frac
        temp = max(temp, 1e-10)
        
        weights, grad, c, _ = smooth_grad(a, temp, fsize)
        
        # Mirror descent: a <- a * exp(-eta * grad), then rescale
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break
        grad_normed = grad / gmax
        
        eta = eta_init / (1.0 + 0.002 * t)
        a = a * np.exp(-eta * grad_normed)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)
        
        if t % 5000 == 0:
            print("  mirror t=%d maxc=%.6f temp=%.4f" % (t, max_c, temp))
    
    return best_a, best_maxc
