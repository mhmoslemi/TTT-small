"""
Multi-phase optimization for minimizing the peak of linear autocorrelation
of non-negative step functions. Uses softmax parameterization, Adam with
annealed temperature, scipy L-BFGS-B, and targeted peak-shaving local search.
Explores multiple sequence lengths (256-768) with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import minimize

# ============================================================
# Core computation functions
# ============================================================

def fft_convolve(a, b):
    """FFT-based linear convolution of two 1-D arrays."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_from_a(a):
    """Compute the exact score for a non-negative sequence."""
    n = len(a)
    a = np.clip(a, 0.0, 1000.0)
    s = np.sum(a)
    if s < 0.01:
        return 1e10
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax_a(theta):
    """Convert log-parameters to scaled probability vector (sum = sqrt(2n))."""
    n = len(theta)
    t = theta - np.max(theta)
    e = np.exp(t)
    p = e / np.sum(e)
    return p * np.sqrt(2.0 * n)


def softmax_p(theta):
    """Stable softmax returning unit-sum probabilities."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


# ============================================================
# Soft-max objective and gradient
# ============================================================

def soft_max_obj(theta, tau):
    """Differentiable soft-max approximation of the score."""
    n = len(theta)
    a = softmax_a(theta)
    S = np.sqrt(2.0 * n)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    logsumexp = c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau)))
    return 2.0 * n * logsumexp / (S * S)


def soft_max_obj_grad(theta, tau):
    """Value and gradient of soft-max objective w.r.t. theta."""
    n = len(theta)
    a = softmax_a(theta)
    S = np.sqrt(2.0 * n)
    p = a / S  # unit-sum probabilities
    c = fft_convolve(a, a)
    c_max = np.max(c)
    w = np.exp((c - c_max) / tau)
    w_sum = np.sum(w)
    w = w / w_sum

    # g_a[j] = 2 * (cross-correlation of w and a)[j]
    # = 2 * sum_k w[k] * a[k-j]
    # Computed as: 2 * conv(a[::-1], w)[n-1 : 2n-1]
    corr = fft_convolve(a[::-1], w)
    g_a = 2.0 * corr[n - 1: 2 * n - 1]

    # d(score)/dtheta_k = (2n/S) * p_k * (g_a[k] - <g_a, p>)
    dot_gp = np.dot(g_a, p)
    g_theta = (2.0 * n / S) * p * (g_a - dot_gp)

    val = 2.0 * n * (c_max + tau * np.log(w_sum)) / (S * S)
    return val, g_theta

# Adam optimizer with annealed temperature
# ============================================================

def adam_optimize(n, theta_init, time_limit, tau_start, tau_end, lr_base=1.5, seed=0):
    """Adam optimization on theta with annealed soft-max temperature."""
    rng = np.random.default_rng(seed)
    theta = theta_init.astype(np.float64).copy()
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    best_theta = theta.copy()
    best_score = score_from_a(softmax_a(theta))
    t0 = time.monotonic()
    consecutive_no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Exponential annealing of tau
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac

        val, g = soft_max_obj_grad(theta, tau)

        step += 1
        m1 = beta1 * m1 + (1 - beta1) * g
        m2 = beta2 * m2 + (1 - beta2) * (g * g)
        m1_hat = m1 / (1.0 - beta1 ** step)
        m2_hat = m2 / (1.0 - beta2 ** step)

        # Learning rate with decay
        lr = lr_base / (1.0 + 0.05 * np.sqrt(step))
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # Periodically check actual score
        if step % 200 == 0 or step <= 10:
            a = softmax_a(theta)
            actual = score_from_a(a)
            if actual < best_score:
                best_score = actual
                best_theta = theta.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1

        # If stuck for too long, add small noise to escape
        if consecutive_no_improve > 100:
            theta += rng.normal(0, 0.01, size=n)
            consecutive_no_improve = 0

    return best_theta, best_score

    if hasattr(result, 'x') and result.x is not None:

        else:
            no_improve_count += 1

        temperature *= cooling
        if no_improve_count > 500:
            temperature = max(temperature, 0.001)
            no_improve_count = 0

    return best, best_score


# ============================================================
# Coordinate descent polish
# ============================================================

def coord_descent_polish(a_init, time_limit, seed=0):
    """Coordinate descent: optimize one or two coordinates at a time."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    current = np.clip(current, 1e-6, 1000.0)
    S = np.sum(current)
    current = current / S * np.sqrt(2.0 * n)
    current_score = score_from_a(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    order = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        # Shuffle order
        rng.shuffle(order)
        progress = False
        for idx in order:
            if time.monotonic() - t0 > time_limit:
                break
            i = int(idx)
            # Try modifying a[i] paired with another element
            # To maintain sum, decrease a[i] by delta and increase a[j] by delta
            for _ in range(3):
                j = int(rng.integers(0, n))
                if j == i:
                    continue
                # Try several step sizes
                for sign in [1, -1]:
                    for frac in [0.01, 0.03, 0.1]:
                        delta = sign * frac * current[i]
                        if current[i] + delta < 1e-8:
                            continue
                        candidate = current.copy()
                        candidate[i] += delta
                        candidate[j] -= delta
                        if candidate[j] < 1e-8:
                            continue
                        new_s = score_from_a(candidate)
                        if new_s < current_score:
                            current = candidate
                            current_score = new_s
                            progress = True
                            if current_score < best_score:
                                best_score = current_score
                                best = current.copy()
                            break
                    if current_score < best_score:
                        break
                if current_score < best_score:
                    break
        if not progress:
            # Occasionally add noise
            current = current * np.exp(rng.normal(0, 0.005, size=n))
            current = np.clip(current, 1e-6, 1000.0)
            current_score = score_from_a(current)

    return best, best_score


# ============================================================
# Initialization functions
# ============================================================

def init_random(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def init_chirp(n, rng, alpha=0.5):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def init_multitone(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    freqs = [1, 2, 3, 5, 8, 13]
    amps = [0.3, 0.2, 0.15, 0.1, 0.08, 0.05]
    for f, amp in zip(freqs, amps):
        a += amp * np.cos(2.0 * np.pi * f * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def init_gaussian_envelope(n, rng):
    """Gaussian envelope with random oscillation."""
    t = np.arange(n, dtype=np.float64)
    center = n // 2
    sigma = n / (4 + 3 * rng.random())
    envelope = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a = envelope * (1.0 + 0.3 * np.cos(2 * np.pi * rng.uniform(2, 8) * t / n))
    a = np.clip(a, 0.01, None)
    return a


def init_from_resample(seq, n):
    """Resample a sequence to target length n."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, seq)
    return np.clip(a, 0.01, None)


def a_to_theta(a):
    """Convert a non-negative sequence to log-softmax parameters."""
    a = np.clip(a, 1e-10, None)
    return np.log(a / np.sum(a))

def propose_candidate(seed=42, budget_s=2380, **kwargs):
    """
    Multi-phase search combining gradient optimization (Adam + L-BFGS-B),
    multiple initializations, and targeted local search polishing.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Initialize from provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = score_from_a(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Define search configurations
    # (n, [init_types], time_per_init)
    configs = [
        (256, ['provided', 'chirp', 'multitone', 'random'], 280),
        (384, ['chirp', 'multitone', 'random'], 250),
        (512, ['chirp', 'multitone', 'random', 'gauss'], 220),
        (768, ['chirp', 'random'], 180),
    ]

    print("\n=== Phase 1: Gradient optimization (Adam) ===")
    for n, init_types, t_per_init in configs:
        if time.monotonic() > deadline - 300:
            break
        print(f"\n  --- n={n} ---")

        for init_type in init_types:
            if time.monotonic() > deadline - 250:
                break
            time_left = min(t_per_init, deadline - time.monotonic() - 200)
            if time_left < 30:
                continue

            # Generate initial sequence
            if init_type == 'provided' and start_seq is not None:
                a0 = init_from_resample(start_seq, n)
            elif init_type == 'chirp':
                alpha = 0.3 + 0.4 * rng.random()
                a0 = init_chirp(n, rng, alpha)
            elif init_type == 'multitone':
                a0 = init_multitone(n, rng)
            elif init_type == 'gauss':
                a0 = init_gaussian_envelope(n, rng)
            else:
                a0 = init_random(n, rng, spread=0.2 + 0.4 * rng.random())

            a0 = np.clip(a0, 0.01, 1000.0)
            theta_init = a_to_theta(a0)

            # Adam phase with annealed tau
            tau_start = 20.0 / n
            tau_end = 0.5 / n
            t0 = time.monotonic()
            theta_best, adam_score = adam_optimize(n, theta_init, time_left, tau_start, tau_end, lr_base=1.5, seed=seed)
            elapsed = time.monotonic() - t0

            a_best = softmax_a(theta_best)
            actual_score = score_from_a(a_best)
            print(f"    n={n}, '{init_type}': adam={adam_score:.6f}, actual={actual_score:.6f}, t={elapsed:.0f}s")

            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} (n={n}) ***")

    print(f"\n  Best after Phase 1: {best_score:.6f}")

    # Phase 2: L-BFGS-B refinement on top candidates
    print("\n=== Phase 2: L-BFGS-B refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        n = len(best_seq)
        theta_init = a_to_theta(best_seq)
        tau = 1.0 / n
        t_per = min(200.0, deadline - time.monotonic() - 150)
        print(f"  Refining n={n}, tau={tau:.4f}, time={t_per:.0f}s")
        theta_ref, ref_score = lbfgsb_optimize(n, theta_init, t_per, tau)
        a_ref = softmax_a(theta_ref)
        actual = score_from_a(a_ref)
        print(f"  L-BFGS result: {actual:.6f}")
        if actual < best_score:
            best_score = actual
            best_seq = a_ref.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # Phase 3: Local search polish
    print(f"\n=== Phase 3: Local search polish ===")
    if best_seq is not None and time.monotonic() < deadline - 50:
        polish_time = min(300.0, deadline - time.monotonic() - 30)
        print(f"  Time available: {polish_time:.0f}s")
        polished, polish_score = polish_local_search(best_seq, polish_time, seed=seed + 1)
        print(f"  After SA polish: {polish_score:.6f}")
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  *** New best: {best_score:.6f} ***")

        # Coordinate descent on the polished result
        if time.monotonic() < deadline - 30:
            cd_time = min(100.0, deadline - time.monotonic() - 10)
            print(f"  Coordinate descent: {cd_time:.0f}s")
            cd_result, cd_score = coord_descent_polish(best_seq, cd_time, seed=seed + 2)
            print(f"  After coord descent: {cd_score:.6f}")
            if cd_score < best_score:
                best_score = cd_score
                best_seq = cd_result.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Phase 4: Try one more angle if time permits
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 4: Final exploration ===")
        remaining = deadline - time.monotonic() - 20
        if remaining > 100:
            # Try a different n with fresh start
            n_try = 448
            a0 = init_multitone(n_try, rng)
            theta_init = a_to_theta(a0)
            t_avail = min(remaining * 0.6, 200.0)
            tau_s, tau_e = 10.0 / n_try, 0.3 / n_try
            theta_b, s_b = adam_optimize(n_try, theta_init, t_avail, tau_s, tau_e, seed=seed + 10)
            a_b = softmax_a(theta_b)
            score_b = score_from_a(a_b)
            print(f"  n={n_try}: score={score_b:.6f}")
            if score_b < best_score:
                best_score = score_b
                best_seq = a_b.copy()
                print(f"  *** New best: {best_score:.6f} ***")

            # Quick polish
            if time.monotonic() < deadline - 30:
                pt = min(60.0, deadline - time.monotonic() - 10)
                pol, pol_s = polish_local_search(best_seq, pt, seed=seed + 20)
                if pol_s < best_score:
                    best_score = pol_s
                    best_seq = pol.copy()
                    print(f"  *** New best after polish: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        print(f"{'='*50}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
