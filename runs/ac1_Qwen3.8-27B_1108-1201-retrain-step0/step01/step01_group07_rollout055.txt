"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using projected gradient descent
with correct gradient, Adam optimizer, tau annealing, and multi-start.
"""

import numpy as np

from scipy.optimize import minimize

def compute_conv(a):
    """Compute full convolution a*a."""
    return np.convolve(a, a)

def compute_gradient(a, tau):
    """Compute gradient of tau*log(sum(exp(conv(a,a)/tau))) w.r.t. a."""
    c = np.convolve(a, a)
    c_shift = (c - c.max()) / tau
    exp_c = np.exp(c_shift)
    p = exp_c / exp_c.sum()
    # grad[j] = 2 * sum_k p[k] * a[k-j] for valid k,j
    # = 2 * np.correlate(p, a, 'valid')[j]
    grad = 2.0 * np.correlate(p, a, 'valid')
    return grad, c, p

def project_onto_simplex(a, S):
    """Project a onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= S:
        return a + (S - s) / len(a)
    # Binary search for threshold
    lo, hi = 0.0, a.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    res = np.maximum(a - t, 0.0)
    # Fix rounding
    diff = S - res.sum()
    if abs(diff) < 1e-10:
        return res
    # Add to the largest element
    idx = res.argmax()
    res[idx] += diff
    return res

def normalize(a, S):
    """Normalize a to have sum S and be non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a)) * S / len(a)
    return a / s * S

    c = np.convolve(a, a)
    max_c = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a ** 2)

    a = 1.0 + c * np.cos(phase)
    rng = np.random.default_rng(seed)
    a += 0.02 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

    rng = np.random.default_rng(seed)
    a += 0.03 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

def run_adam_pgd(a0, tau_hi, tau_lo, t_budget, lr_base=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    """Projected gradient descent with Adam in a-space, with simplex projection."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = normalize(a0.astype(np.float64).copy(), S)
    
    best = a.copy()
    best_score = score(best)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    
    t0 = time.monotonic()
    last_eval = 0.0
    eval_interval = 50  # steps between evaluations
    
    while time.monotonic() - t0 < t_budget:
        t_elapsed = time.monotonic() - t0
        progress = min(1.0, t_elapsed / t_budget)
        tau = tau_hi * (tau_lo / tau_hi) ** progress
        
        grad, c, p = compute_gradient(a, tau)
        
        t_step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        
        # Adaptive learning rate: decrease as we go
        lr = lr_base / (1.0 + 0.5 * progress)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a_new

            s = score(a)
            if s < best_score:

                best = a.copy()

    return best, final_score

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses projected gradient descent with corrected gradient, Adam optimizer,

Key fix: correct gradient computation (2*correlate(p, a, 'valid')).

def compute_gradient_correct(a, tau):
    """Compute gradient of soft-max objective w.r.t. a.
    Returns (grad, max_conv_value) where grad = d/d a [tau * log sum exp(conv(a,a)/tau)]
    """
    c = np.convolve(a, a)
    c_max = c.max()
    c_shift = (c - c_max) / tau
    exp_c = np.exp(c_shift)
    p = exp_c / exp_c.sum()
    # grad[j] = 2 * sum_k p[k] * a[k-j] for valid indices
    # = 2 * np.correlate(p, a, 'valid')[j]
    grad = 2.0 * np.correlate(p, a, 'valid')
    return grad, c_max

def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, a.max()
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    diff = S - res.sum()
    if abs(diff) > 1e-12:
        idx = res.argmax()
        res[idx] += diff
    return res

    a = np.maximum(a, 0.0)

    if s < 1e-15:

    return a * (S / s)

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Key improvement: corrected gradient (2*correlate(p,a,'valid')) fixing a bug
in the parent code, combined with Adam optimizer, geometric tau annealing,
multiplicative (mirror) descent mixing, and multi-start exploration.
"""

import numpy as np
import time


def compute_gradient_correct(a, tau):
    """Compute gradient of soft-max objective w.r.t. a.
    Objective: tau * log(sum(exp(conv(a,a)/tau)))
    Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] for valid k,j
             = 2 * np.correlate(p, a, 'valid')[j]
    """
    c = np.convolve(a, a)
    c_max = c.max()
    c_shift = (c - c_max) / tau
    exp_c = np.exp(c_shift)
    p = exp_c / exp_c.sum()
    grad = 2.0 * np.correlate(p, a, 'valid')
    return grad, c_max


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = a.size
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, a.max()
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    diff = S - res.sum()
    if abs(diff) > 1e-12:
        idx = res.argmax()
        res[idx] += diff
    return res


def normalize_sum(a, S):
    """Normalize a to sum S, keeping non-negative."""
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.full(a.size, S / a.size)
    return a * (S / s)


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def init_chirp_asym(n, seed=0):
    """Asymmetric quadratic chirp - breaks palindromic symmetry."""
    t = np.arange(n, dtype=np.float64)
    phase = np.pi * t * t / n
    a = 1.0 + 0.5 * np.cos(phase) + 0.2 * np.sin(2.0 * phase)
    rng = np.random.default_rng(seed)
    a += 0.02 * rng.standard_normal(n)
    return np.maximum(a, 1e-10)


def init_chirp_multi(n, seed=0):
    """Multi-frequency chirp."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0
    a += 0.4 * np.cos(2.0 * np.pi * 0.3 * t * t / n)
    a += 0.3 * np.sin(2.0 * np.pi * 0.7 * t * t / n)
    a += 0.15 * np.cos(2.0 * np.pi * 0.1 * t)
    rng = np.random.default_rng(seed)
    a += 0.02 * rng.standard_normal(n)
    return np.maximum(a, 1e-10)


def init_onesided(n, seed=0):
    """Exponentially decaying - strongly non-palindromic."""
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-1.5 * t / n)
    rng = np.random.default_rng(seed)
    a += 0.03 * rng.standard_normal(n)
    return np.maximum(a, 1e-10)


def init_bump_offcenter(n, seed=0):
    """Gaussian bump off-center to break symmetry."""
    t = np.arange(n, dtype=np.float64)
    center = 0.35 * n
    width = 0.25 * n
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    rng = np.random.default_rng(seed)
    a += 0.02 * rng.standard_normal(n)
    return np.maximum(a, 1e-10)


def init_uniform(n):
    return np.ones(n, dtype=np.float64)


def optimize_hybrid(a0, tau_sched, t_budget, lr_adam=0.5, lr_mirror=0.1,
                    mix_ratio=0.3, beta1=0.9, beta2=0.999, eps=1e-8):
    """Hybrid Adam + multiplicative (mirror) descent with tau annealing.
    
    tau_sched: list of (tau_hi, tau_lo, time_fraction) tuples.
    mix_ratio: fraction of steps using pure Adam (rest uses mirror descent blend).
    """
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a0.copy(), S)

    best = a.copy()
    best_sc = compute_score(best)

    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    t0 = time.monotonic()

    total_time = sum(f for _, _, f in tau_sched)
    cumulative = 0.0

    for tau_hi, tau_lo, frac in tau_sched:
        cumulative += frac
        phase_progress_start = (cumulative - frac) / total_time
        phase_progress_end = cumulative / total_time

        while True:
            elapsed = time.monotonic() - t0
            if elapsed >= t_budget:
                break

            # Interpolate tau within this phase
            phase_elapsed = elapsed - sum(t for _, _, t in tau_sched[:tau_sched.index((tau_hi, tau_lo, frac))] * t_budget)
            phase_dur = frac * t_budget
            if phase_dur > 0:
                local_progress = min(1.0, max(0.0, phase_elapsed / phase_dur))
            else:
                local_progress = 1.0
            # Geometric interpolation
            tau = tau_hi * (tau_lo / tau_hi) ** local_progress

            grad, c_max = compute_gradient_correct(a, tau)

            t_step += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_step)
            v_hat = v / (1.0 - beta2 ** t_step)

            if t_step % 1000 < int(1000 * mix_ratio):
                # Mirror (multiplicative) descent step
                eta = lr_mirror * tau
                a = a * np.exp(-eta * grad)
                a = normalize_sum(a, S)
            else:
                # Adam step + projection
                a_new = a - lr_adam * m_hat / (np.sqrt(v_hat) + eps)
                a = project_simplex(a_new, S)

            if t_step % 300 == 0:
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best = a.copy()

    return best, best_sc


def optimize_adam_pure(a0, tau_hi, tau_lo, t_budget, lr=0.5,
                       beta1=0.9, beta2=0.999, eps=1e-8):
    """Pure Adam projected gradient descent with geometric tau annealing."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a0.copy(), S)

    best = a.copy()
    best_sc = compute_score(best)

    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    t0 = time.monotonic()
    last_improve_step = 0
    lr_current = lr

    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / t_budget)
        tau = tau_hi * (tau_lo / tau_hi) ** progress

        grad, c_max = compute_gradient_correct(a, tau)

        t_step += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_step)
        v_hat = v / (1.0 - beta2 ** t_step)

        a_new = a - lr_current * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)

        if t_step % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc - 1e-9:
                best_sc = sc
                best = a.copy()
                last_improve_step = t_step
                lr_current = min(lr_current * 1.1, 5.0)
            elif t_step - last_improve_step > 5000:
                lr_current = max(lr_current * 0.5, 0.01)
                last_improve_step = t_step

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best = a.copy()

    return best, best_sc


def subgradient_polish(a0, t_budget, max_iters=50000):
    """Polish using direct subgradient on the argmax of convolution.
    More aggressive than soft-max for the final stage."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a0.copy(), S)
    best = a.copy()
    best_sc = compute_score(best)

    t0 = time.monotonic()
    step = 0
    lr = 0.05

    while time.monotonic() - t0 < t_budget and step < max_iters:
        c = np.convolve(a, a)
        k_star = c.argmax()

        # Subgradient of c[k_star] w.r.t. a[j] = 2*a[k_star - j] for valid j
        # This is 2 * a[k_star-j] where 0 <= k_star-j <= n-1
        j_min = max(0, k_star - (n - 1))
        j_max = min(n - 1, k_star)
        subgrad = np.zeros(n)
        for j in range(j_min, j_max + 1):
            subgrad[j] = 2.0 * a[k_star - j]

        a = project_simplex(a - lr * subgrad, S)
        step += 1

        if step % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best = a.copy()
                lr = min(lr * 1.1, 0.5)
            else:
                lr = max(lr * 0.9, 0.001)

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best = a.copy()

    return best, best_sc
