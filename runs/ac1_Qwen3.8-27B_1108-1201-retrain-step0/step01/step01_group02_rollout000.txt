def solve_lp(a, M, timeout=30.0):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            g = np.maximum(0.0, res.x)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_conv_indices(n):
    """Precompute row and column indices for the convolution matrix."""
    row_idx = []
    col_idx = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1, dtype=np.int32)
            row_idx.append(np.full(len(js), k, dtype=np.int32))
            col_idx.append(js)
    row_idx = np.concatenate(row_idx)
    col_idx = np.concatenate(col_idx)
    return row_idx, col_idx
