import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def compute_score(a):
    n = len(a)
    c = fft_convolve(a, a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * np.max(c) / (s*s)

def build_constraint_matrix(a):
    """Build A such that A @ g gives the convolution a*g (full)."""
    n = len(a)
    n_out = 2 * n - 1
    A = np.zeros((n_out, n))
    for j in range(n):
        # g[j] contributes to outputs k from j to j+n-1
        # A[k, j] = a[k-j]
        for k in range(j, min(j + n, n_out)):
            A[k, j] = a[k - j]
    return A

def solve_lp_direction(a):
    """Find g >= 0 maximizing sum(g) subject to (a*g)_k <= M for all k."""
    n = len(a)
    M = np.max(fft_convolve(a, a))
    A = build_constraint_matrix(a)
    # Minimize -sum(g) subject to A@g <= M, g >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), 
                  bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x, M
    return None, M

def lp_iterate(a, max_iters=100, t_start=0.1, t_end=0.01):
    """Iteratively apply LP-based direction finding."""
    n = len(a)
    S = np.sum(a)
    a = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(max_iters):
        g, M = solve_lp_direction(a)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        # Rescale g to have sum S
        g = g * S / g_sum
        
        # Line search for t
        best_t = 0
        best_new_score = compute_score(a)
        for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            new_a = (1 - t) * a + t * g
            new_score = compute_score(new_a)
            if new_score < best_new_score:
                best_new_score = new_score
                best_t = t
        
        if best_t == 0:
            break
        
        a = (1 - best_t) * a + best_t * g
        
        if compute_score(a) < best_score:
            best_score = compute_score(a)
            best_a = a.copy()
    
    return best_a, best_score

def fft_convolve(a, b):
    """FFT-based linear convolution."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    """Compute the score locally (matches evaluate_sequence formula)."""
    n = len(a)

    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def build_A(a):
    """Build constraint matrix for LP: A @ g = convolve(a, g)."""
    n = len(a)
    n_out = 2 * n - 1
    A = np.zeros((n_out, n))
