"""
LP-based iterative optimization for minimizing peak autocorrelation of non-negative sequences.
Uses linear programming to find optimal direction updates, combined with local search polishing.
"""

import numpy as np
import time
import scipy.sparse
import scipy.optimize


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def build_constraint_matrix(f):
    """Build C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    # C is (2n-1) x n
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            C[k, j] = f[k - j]
    return C


def solve_lp(f, M):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    C = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2*n - 1)
    bounds = [(0, None)] * n
    result = scipy.optimize.linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return f.copy()


def lp_optimize(a_init, time_limit, seed=42):
    """Run LP-based iterative optimization."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    
    # Normalize
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    a = a / np.sum(a) * S
    
    best_a = a.copy()
    best_score = local_score(a)
    M = np.max(fft_convolve(a, a))
    
    iteration = 0
    no_improve = 0
    step_size = 0.3
    
    while time.monotonic() - t0 < time_limit:
        # Solve LP
        g = solve_lp(a, M)
        if g is None or not np.all(np.isfinite(g)):
            break
        g = np.clip(g, 0, None)
        g_sum = np.sum(g)
        if g_sum < 1e-6:
            break
        # Rescale g
        g = g / g_sum * S
        
        # Line search for step size
        best_t = 0
        best_new_score = np.inf
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.clip(a_trial, 0, 1000.0)
            sc = local_score(a_trial)
            if sc < best_new_score:
                best_new_score = sc
                best_t = t
        
        if best_t == 0:
            # Try smaller steps
            for t in [0.01, 0.05]:
                a_trial = (1 - t) * a + t * g
                a_trial = np.clip(a_trial, 0, 1000.0)
                sc = local_score(a_trial)
                if sc < best_new_score:
                    best_new_score = sc
                    best_t = t
        
        if best_t == 0 or best_new_score >= local_score(a):
            no_improve += 1
            # Try a random perturbation to escape
            a_pert = a * np.exp(rng.normal(0, 0.02, n))
            a_pert = np.clip(a_pert, 0, 1000.0)
            sc = local_score(a_pert)
            if sc < local_score(a):
                a = a_pert
            else:
                if no_improve > 20:
                    break
            continue
        
        a = (1 - best_t) * a + best_t * g
        a = np.clip(a, 0, 1000.0)
        sc = local_score(a)
        
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 30:
            break
        
        M = np.max(fft_convolve(a, a))
        iteration += 1
        if iteration % 50 == 0:
            print(f"    LP iter {iteration}: score={sc:.6f}, best={best_score:.6f}, M={M:.4f}")
    
    return best_a, best_score


def local_search(a_init, time_limit, seed=42):
    """Targeted local search with peak reduction."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0, 1000.0)
    n = len(a)
    current_score = local_score(a)
    best_a = a.copy()
    best_score = current_score
    temp = 0.002
    cooling = 0.9995
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.4:
            # Peak-targeted: identify peak lag, reduce its contributors
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Find contributing pairs
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            # Reduce top contributors
            top_k = max(2, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            # Increase some others
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(5, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.6:
            # Random block
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.8:
            # Pair swap (move mass from one position to another)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = a[i] * 0.1
                candidate[i] -= delta
                candidate[j] += delta
        else:
            # Small global
            candidate *= np.exp(rng.normal(0, 0.01, n))
        
        candidate = np.clip(candidate, 0, 1000.0)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-15)):
            a = candidate
            current_score = new_score
        
        temp *= cooling
        iter_count += 1
        if iter_count % 10000 == 0:
            print(f"    LS iter {iter_count}: score={current_score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score


def make_chirp(n, seed=42):

"""
LP-based iterative optimization for minimizing peak autocorrelation of non-negative sequences.
Uses linear programming to find optimal direction updates (maximizing sum under peak constraint),
combined with peak-targeted local search polishing. This differs from the previous approach by
using exact LP solutions for direction finding rather than approximate gradients.
"""

import numpy as np
import time
import scipy.optimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local computation of the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def build_constraint_matrix(f):
    """Build C where C[k,j] = f[k-j] if 0<=k-j<n else 0. Shape: (2n-1, n)."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            C[k, j] = f[k - j]
    return C


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. C(f)@g <= M*1, g >= 0. Returns optimal g."""
    n = len(f)
    C = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    result = scipy.optimize.linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success and result.x is not None:
        return result.x
    return f.copy()


def solve_lp_min_peak(f, S_target):
    """
    Solve: min M s.t. C(f)@g <= M*1, sum(g) = S_target, g >= 0.
    Returns (M_opt, g_opt). This directly minimizes the peak.
    """
    n = len(f)
    C = build_constraint_matrix(f)
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: minimize M -> c = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    # Inequality constraints: C@g - M <= 0 for each row -> [C, -I_ones] @ [g; M] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = S_target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S_target])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    result = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                    bounds=bounds, method='highs')
    if result.success and result.x is not None:
        return result.x[n], result.x[:n]
    return None, None


def lp_optimize(a_init, time_limit, seed=42):
    """
    Run LP-based iterative optimization.
    Alternates between:
    1. LP to maximize sum under peak constraint (direction finding)
    2. LP to minimize peak under sum constraint (peak reduction)
    With line search and occasional random perturbations.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a_init)
    S = np.sqrt(2.0 * n)

    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    if np.sum(a) < 1e-10:
        a = np.ones(n)
    a = a / np.sum(a) * S

    best_a = a.copy()
    best_score = local_score(a)
    M = np.max(fft_convolve(a, a))

    iteration = 0
    no_improve = 0
    use_min_peak = False

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit * 0.85:
            break

        # Alternate between the two LP formulations
        if iteration % 3 == 2:
            use_min_peak = True
        else:
            use_min_peak = False

        if use_min_peak:
            # Minimize peak with fixed sum
            M_new, g = solve_lp_min_peak(a, S)
            if M_new is None:
                g = solve_lp_direction(a, M)
                g = np.clip(g, 0, None)
                gs = np.sum(g)
                if gs < 1e-6:
                    g = a.copy()
                else:
                    g = g / gs * S
            else:
                g = np.clip(g, 0, None)
                gs = np.sum(g)
                if gs < 1e-6:
                    break
                # g already has sum = S_target
        else:
            # Maximize sum under peak constraint
            g = solve_lp_direction(a, M)
            g = np.clip(g, 0, None)
            gs = np.sum(g)
            if gs < 1e-6:
                break
            g = g / gs * S

        # Check if g is meaningfully different from a
        diff = np.max(np.abs(g - a)) / S
        if diff < 1e-6:
            no_improve += 2
            if no_improve > 30:
                break
            # Add random perturbation to escape
            a = a * np.exp(rng.normal(0, 0.03, n))
            a = np.clip(a, 0, 1000.0)
            a = a / np.sum(a) * S
            continue

        # Line search
        best_t = 0.0
        best_trial_score = np.inf
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            sc = local_score(a_trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_t = t

        if best_t == 0.0 or best_trial_score >= local_score(a):
            no_improve += 1
            if no_improve > 25:
                # Try a bolder perturbation
                a_pert = a * np.exp(rng.normal(0, 0.05, n))
                a_pert = np.clip(a_pert, 0, 1000.0)
                sc_pert = local_score(a_pert)
                if sc_pert < local_score(a):
                    a = a_pert
                    a = a / np.sum(a) * S
                elif no_improve > 50:
                    break
            continue

        a = (1.0 - best_t) * a + best_t * g
        a = np.clip(a, 0.0, 1000.0)
        sc = local_score(a)

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0

        M = np.max(fft_convolve(a, a))
        iteration += 1

        if iteration % 30 == 0:
            print(f"    LP iter {iteration}: score={sc:.6f}, best={best_score:.6f}, M={M:.4f}, t={best_t:.2f}")

    return best_a, best_score


def local_search(a_init, time_limit, seed=42):
    """
    Peak-targeted simulated annealing local search.
    Uses knowledge of which lags are near the peak to craft effective perturbations.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    current_score = local_score(a)
    best_a = a.copy()
    best_score = current_score
    temp = 0.003
    cooling = 0.9998
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        if (time.monotonic() - t0) % 5 > 4:
            pass  # just checking time

        candidate = a.copy()
        r = rng.random()

        if r < 0.35:
            # Peak-targeted: identify peak lag, reduce its top contributors
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            contributions = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contributions[i] = a[i] * a[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contributions)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.06, 0.025, size=len(top_idx)))
            # Boost non-contributors
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.06, 0.025, size=n_inc))
        elif r < 0.50:
            # Second-peak reduction
            conv = fft_convolve(a, a)
            idx = int(np.argmax(conv))
            conv_copy = conv.copy()
            conv_copy[max(0, idx-1):idx+2] = 0
            if np.max(conv_copy) > 0:
                second_idx = int(np.argmax(conv_copy))
                contributions = np.zeros(n)
                for i in range(n):
                    j = second_idx - i
                    if 0 <= j < n:
                        contributions[i] = a[i] * a[j]
                top_k = max(2, n // 12)
                top_idx = np.argsort(contributions)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.02, size=len(top_idx)))
        elif r < 0.65:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))
        elif r < 0.80:
            # Asymmetric shift: move mass from one region to another
            half = n // 2
            i = int(rng.integers(0, half))
            j = int(rng.integers(half, n))
            delta = min(a[i], a[j]) * 0.15
            candidate[i] -= delta
            candidate[j] += delta
        elif r < 0.90:
            # Multi-point perturbation
            for _ in range(4):
                idx = int(rng.integers(0, n))
                candidate[idx] *= np.exp(rng.normal(0.0, 0.03))
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.015, n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        elif temp > 1e-8 and rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-15)):
            a = candidate
            current_score = new_score

        temp *= cooling
        iter_count += 1

        if iter_count % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LS iter {iter_count}: score={current_score:.6f}, best={best_score:.6f}, t={elapsed:.0f}s")

    return best_a, best_score


def gradient_polish(a_init, time_limit, seed=42):
    """
    Short gradient-based polish using subgradient of max(conv(a,a)).
    Used for final fine-tuning after LP and local search.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    a = a / np.sum(a) * S
    best_a = a.copy()
    best_score = local_score(a)

    lr = 0.01
    momentum = np.zeros(n)
    beta = 0.9

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_val = np.max(conv)
        # Find all lags at (or near) the peak
        peak_mask = conv >= peak_val * 0.999
        peak_lags = np.where(peak_mask)[0]

        # Subgradient: average of gradients from near-peak lags
        grad = np.zeros(n)
        for k in peak_lags:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad /= len(peak_lags)

        # Also penalize the sum-of-squares component (to keep spread)
        grad += 0.001 * a

        # Normalized gradient step
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            grad = grad / grad_norm

        momentum = beta * momentum + (1 - beta) * grad
        a_new = a - lr * momentum
        a_new = np.clip(a_new, 0.0, 1000.0)
        a_new = a_new / np.sum(a_new) * S

        new_score = local_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.95

    return best_a, best_score


def make_chirp_init(n, seed=42):
    """Create a chirp-like initial sequence with asymmetric amplitude."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)
    # Asymmetric chirp: amplitude varies quadratically
    center = n * 0.35  # asymmetric center
    a = 1.0 + 0.6 * np.cos(np.pi * (t - center) ** 2 / (n * 0.4))
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_taper_init(n, seed=42):
    """Create a tapered initial sequence (high in middle, low at edges)."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64) / (n - 1)
    # Expanding sinusoidal taper
    a = 0.3 + 0.7 * np.sin(np.pi * t) ** 0.7
    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_two_bump_init(n, seed=42):
    """Create a two-bump initial sequence (helps break symmetry)."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)
    c1 = n * 0.3
    c2 = n * 0.7
    w = n * 0.12
    a = 0.2 + np.exp(-(t - c1) ** 2 / (2 * w ** 2)) + 0.8 * np.exp(-(t - c2) ** 2 / (2 * w ** 2))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP-based direction finding,
    peak-targeted local search, and gradient polishing.

    Phase 1: LP-based iterative optimization (main computational work)
    Phase 2: Peak-targeted simulated annealing local search
    Phase 3: Gradient-based fine polishing
    """
    t_global = time.monotonic()
    time_budget = max(60.0, float(budget_s) - 25.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence if available
    a_provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:

        if len(a_provided) > 0 and np.sum(a_provided) > 0.01:
            s = local_score(a_provided)
            print(f"Provided sequence: n={len(a_provided)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = a_provided.copy()

    # Phase 1: LP-based optimization
    print("\n=== Phase 1: LP-based optimization ===")

    # Define search configurations
    configs = []
    if a_provided is not None:
        configs.append((len(a_provided), 'provided', 0))
    configs.extend([
        (256, 'chirp', 1),
        (256, 'taper', 2),
        (256, 'two_bump', 3),
        (256, 'provided', 4),
        (384, 'chirp', 5),
        (384, 'taper', 6),
        (512, 'chirp', 7),
        (512, 'two_bump', 8),
    ])

    # Remove duplicates
    seen = set()
    unique_configs = []
    for cfg in configs:
        key = (cfg[0], cfg[1])
        if key not in seen:
            seen.add(key)
            unique_configs.append(cfg)
    configs = unique_configs

    for n, init_type, idx in configs:
        if time.monotonic() > deadline - 400:
            print("  Time running out, stopping LP phase")
            break

        # Generate initialization
        if init_type == 'provided' and a_provided is not None:

            a0 = make_taper_init(n, seed=seed + idx * 100)

        else:
            a0 = np.ones(n)

        # Allocate time for this configuration
        remaining = deadline - time.monotonic() - 400
        n_remaining_configs = len(configs) - configs.index((n, init_type, idx))
        t_per = max(20.0, min(150.0, remaining / max(1, n_remaining_configs)))

        print(f"  Config: n={n}, init={init_type}, time={t_per:.0f}s")
        a_opt, score = lp_optimize(a0, t_per, seed=seed + idx * 7)

    # Phase 3: Gradient polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Phase 3: Gradient polish ===")
        gt = min(120.0, deadline - time.monotonic() - 15)
        refined, grad_score = gradient_polish(best_seq, gt, seed=seed + 800)
        if grad_score < best_score:
            best_score = grad_score

            print(f"  After gradient polish: {best_score:.6f}")

    # Phase 4: One final LP round if time permits
    if best_seq is not None and time.monotonic() < deadline - 25:

        ft = min(100.0, deadline - time.monotonic() - 10)
        if ft > 10:
            final_a, final_score = lp_optimize(best_seq, ft, seed=seed + 1000)
            if final_score < best_score:
                best_score = final_score
                best_seq = final_a.copy()
                print(f"  After final LP: {best_score:.6f}")

    # Final validation
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    # Fallback
    if a_provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in a_provided]
    return [1.0] * 256
