One more thought: the **softmax trick**. Instead of `max_k b[k]`, use:
`soft_max(b, tau) = (1/tau) * log(sum_k exp(tau * b[k] - tau * max(b)))` (shifted for numerical stability)

The gradient of this is:
`grad soft_max = (1/tau) * sum_k (tau * exp(tau * b[k]) / sum_j exp(tau * b[j])) * grad b[k]`
`= sum_k softmax(tau * b) * grad b[k]`

This is a smooth approximation to the max, with a well-defined gradient. For large `tau`, it's close to the max. For small `tau`, it's more like the mean.

By starting with small `tau` and increasing it, we get a "continuation" method that can navigate the landscape more effectively.

```python
def _soft_max_grad(a, tau):
    """Compute the gradient of soft_max(convolve(a,a), tau)."""
    n = len(a)
    b = np.convolve(a, a)
    # Softmax weights
    b_shifted = b - np.max(b)  # For numerical stability
    weights = np.exp(tau * b_shifted)
    weights /= np.sum(weights)
    
    # Gradient is weighted sum of individual gradients
    grad = np.zeros(n)
    for k in range(len(b)):
        if weights[k] < 1e-8:
            continue
        # grad b[k] / grad a = 2 * R_k(a) where R_k shifts a
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * weights[k] * a[k - idx]
    
    return grad, np.max(b)
```

import numpy as np
import time

def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) for a sequence with sum=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_gradient(a, tau):
    """
    Compute gradient of soft_max(convolve(a,a), tau) and the actual max.
    Returns (gradient, max_b).
    """
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    # Softmax weights (shifted for stability)
    shifted = b - b_max
    w = np.exp(tau * shifted)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    # Only loop over significant weights
    threshold = 1e-6
    significant = w > threshold
    for k in np.nonzero(significant)[0]:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * w[k] * a[k - idx]
    
    return grad, b_max

def _adam_step(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update step. Returns updated (a, m, v)."""
    n = len(a)
    # Center gradient (for sum constraint)
    grad_centered = grad - np.mean(grad)
    
    # Update moments
    m = beta1 * m + (1 - beta1) * grad_centered
    v = beta2 * v + (1 - beta2) * grad_centered ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    
    # Update
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v

    Continuation method: start with small tau (smooth) and increase to large tau (sharp).
    Uses Adam optimizer.
    """
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a /= S
    
    best_a = a.copy()
    best_score = _compute_score(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    total_iters = 5000
    iters_per_phase = total_iters // n_phases
    
    for phase in range(n_phases):
        tau = tau_start * (tau_end / tau_start) ** (phase / max(1, n_phases - 1))
        lr = lr_start * (lr_end / lr_start) ** (phase / max(1, n_phases - 1))
        
        for it in range(iters_per_phase):
            if time.monotonic() > deadline:
                break
            
            t += 1
            grad, b_max = _softmax_gradient(a, tau)
            cur_score = 2.0 * n * b_max
            
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
            
            a_new, m, v = _adam_step(a, grad, m, v, t, lr)
            a = _project_simplex(a_new, 1.0)
            S = np.sum(a)
            if S < 1e-10:
                a = best_a.copy()
                break
    
    return best_a, best_score

def _local_search_transfer(a, deadline, rng, max_iters=10000, step_init=0.01):

    a_new = a + scale * noise
    return _project_simplex(a_new, 1.0)
