"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences using
log-sum-exp smooth gradient descent with adaptive temperature.

Key improvement over previous approach: instead of using top-k subgradients,
we use the smooth log-sum-exp approximation of the max, which gives a true
gradient that weights all convolution peaks proportionally to their
exp(c_k/tau). This provides a much richer descent direction.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, tau):
    """Compute gradient of tau*log(sum exp(c_k/tau)) where c = a*a (full conv).
    
    Returns (gradient, max_c, lse_value).
    """
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    
    # Numerically stable softmax
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient: dL/da_m = 2 * sum_k w_k * a_{k-m} (for 0 <= k-m < n)
    # = 2 * cross_correlate(w, a)[m]
    # Using the derivation above:
    a_rev = a[::-1]
    fsize = 1 << (2 * n - 2).bit_length()  # need at least 2n-1 points
    A = fft(a_rev, fsize)
    W = fft(w, fsize)
    conv = np.real(ifft(A * W))
    grad = 2.0 * conv[n - 1:2 * n - 1]
    
    lse_val = c_max + tau * np.log(exp_shifted.sum())
    return grad, c_max, lse_val


def optimize_lse(a_init, n, S, max_time, tau_init, tau_min, decay_rate,
                 lr_init, lr_decay, momentum=0.9):
    """Log-sum-exp gradient descent with simplex projection.
    
    Parameters:
        a_init: initial sequence (will be normalized)
        n: length
        S: target sum (sqrt(2n))
        max_time: time budget in seconds
        tau_init: initial temperature
        tau_min: minimum temperature
        decay_rate: rate at which tau decays
        lr_init: initial learning rate
        lr_decay: learning rate decay
        momentum: momentum coefficient
    """
    a = np.clip(a_init, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a * (S / s)
    
    v_mom = np.zeros(n)
    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        
        grad, c_max, lse_val = lse_gradient(a, tau_init * max(tau_min / tau_init, (decay_rate ** t)))
        
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        
        # Momentum update
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_normed = grad / gnorm
        
        v_mom = momentum * v_mom + (1 - momentum) * grad_normed
        lr = lr_init / (1.0 + lr_decay * t)
        
        a = project_simplex(a - lr * v_mom, S)
    
    return best_a, best_score

def gerchberg_saxton_init(n, rng, iterations=50):
    """Generate initialization using GS-like algorithm targeting flat spectrum."""
    a = rng.random(n)
    S = np.sqrt(2.0 * n)
    a = a / a.sum() * S
    
    fsize = 1 << (n - 1).bit_length()
    
    for _ in range(iterations):
        A = fft(a, fsize)
        # Modify magnitude to be more flat
        mag = np.abs(A)
        phase = np.angle(A)
        # Target: constant magnitude
        target_mag = np.full(n, mag.mean())
        # Blend between current and target
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fsize))[:n]
        # Project onto non-negative simplex
        a = project_simplex(a_new, S)
    
    return a

def smooth_random_init(n, rng):
    """Smooth random initialization using low-frequency components."""
    fsize = 1 << (n - 1).bit_length()
    # Random spectrum with low-pass filtering
    spectrum = rng.normal(size=fsize) + 1j * rng.normal(size=fsize)
    # Low-pass: zero out high frequencies
    for i in range(fsize // 4, fsize // 2 + 1):
        spectrum[i] = 0
        spectrum[fsize - i] = 0 if i > 0 else spectrum[0]
    a = np.real(ifft(spectrum, fsize))[:n]
    a = np.abs(a) + 0.1
    S = np.sqrt(2.0 * n)
    a = a / a.sum() * S
    return a


def binary_like_init(n, rng):
    """Initialize with a binary-like pattern (0 or 1) then smooth."""
    a = (rng.random(n) > 0.5).astype(float)
    # Smooth with a boxcar
    k = max(1, n // 64)
    kernel = np.ones(k) / k
    a = np.convolve(a, kernel, mode='same')
    a = a + 0.01  # Ensure positive
    S = np.sqrt(2.0 * n)
    a = a / a.sum() * S
    return a

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences using
log-sum-exp smooth gradient descent with adaptive temperature.

Key innovation: Instead of top-k subgradients, use the smooth log-sum-exp
approximation of max which gives a true gradient weighting ALL convolution
peaks proportionally to exp(c_k/tau). Combined with adaptive tau decay
(simulated-annealing style), this provides richer descent directions and
better convergence than sparse subgradient methods.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_full(a, tau):
    """Compute LSE gradient, max conv value, and convolution.
    
    Returns (grad, c_max, c) where grad is dLSE/da.
    """
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    
    # Numerically stable softmax weights
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    exp_sum = exp_shifted.sum()
    w = exp_shifted / exp_sum
    
    # Gradient: dLSE/da_m = 2 * sum_k w_k * a_{k-m} (for 0 <= k-m < n)
    # = 2 * (a_rev * w)[n-1 : 2n-1] where a_rev = a[::-1]
    a_rev = a[::-1]
    fsize = 1 << (3 * n - 2 - 1).bit_length()
    A = fft(a_rev, fsize)
    W = fft(w, fsize)
    conv = np.real(ifft(A * W))
    grad = 2.0 * conv[n - 1:2 * n - 1]
    
    return grad, c_max, c


def run_adam_lse(a_init, n, S, max_time, rng, lr_init, tau_init, tau_min,
                 tau_decay=0.995, use_momentum=True):
    """Adam optimizer with log-sum-exp gradient and simplex projection."""
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a * (S / s)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    tau = tau_init
    
    no_improve = 0
    patience = 2000
    
    while time.monotonic() - start < max_time:
        t += 1
        
        grad, c_max, _ = lse_full(a, tau)
        
        if c_max < best_score - 1e-10:
            best_score = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Decay tau (annealing)
        tau = max(tau_min, tau * tau_decay)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)

            v = np.zeros(n)

                   tau_decay=0.997):
    """Mirror descent (multiplicative weights) with LSE gradient."""
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a * (S / s)
    
    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    t = 0
    tau = tau_init
    
    while time.monotonic() - start < max_time:
        t += 1
        
        grad, c_max, _ = lse_full(a, tau)
        
        if c_max < best_score - 1e-10:
            best_score = c_max
            best_a = a.copy()
        
        tau = max(tau_min, tau * tau_decay)
        
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        
        eta = eta_init / (1.0 + 0.003 * t)
        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.full(n, S / n)
        else:
            a = a_new * (S / s)
    
    return best_a, best_score


def run_topk_subgrad(a_init, n, S, max_time, rng, lr_init, top_k=10):
    """Adam with top-k subgradient (similar to previous SOTA but improved)."""
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a * (S / s)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
        
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-15:
            break
        weights = top_vals / total
        
        grad = np.zeros(n)
        for w_val, k in zip(weights, top_indices):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr = lr_init / (1.0 + 0.0005 * t)
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
    
    return best_a, best_score


def gerchberg_saxton_init(n, rng, iterations=100, blend=0.7):
    """GS-like initialization targeting flat magnitude spectrum."""
    a = rng.random(n) + 0.1
    S = np.sqrt(2.0 * n)
    a = a / a.sum() * S
    
    fsize = 1 << (n - 1).bit_length()
    
    for _ in range(iterations):
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)
        target_mag = mag.mean() * np.ones(n)
        A_new = (blend * target_mag + (1 - blend) * mag) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fsize))[:n]
        a = project_simplex(a_new, S)
    
    return a


def smooth_init(n, rng):
    """Smooth initialization with multiple frequency components."""
    t = np.linspace(0, 1, n, endpoint=False)
    freqs = rng.choice(np.arange(1, 10), size=5, replace=False)
    amps = rng.uniform(0.1, 0.5, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    
    a = np.ones(n)
    for f, am, ph in zip(freqs, amps, phases):
        a += am * np.sin(2 * np.pi * f * t + ph)
    
    a = np.abs(a) + 0.1
    S = np.sqrt(2.0 * n)
    a = a / a.sum() * S
    return a


def lowpass_random_init(n, rng, cutoff_frac=0.3):
    """Random spectrum with low-pass filtering for smooth result."""
    fsize = 1 << (n - 1).bit_length()
    spectrum = rng.normal(size=fsize) + 1j * rng.normal(size=fsize)
    cutoff = int(cutoff_frac * fsize // 2)
    for i in range(cutoff, fsize // 2 + 1):
        spectrum[i] = 0
        if i > 0:
            spectrum[fsize - i] = 0
    a = np.real(ifft(spectrum, fsize))[:n]
    a = np.abs(a) + 0.1
    S = np.sqrt(2.0 * n)
    a = a / a.sum() * S
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function.
    
    Uses log-sum-exp smooth gradient descent as the primary optimizer,
    combined with multiple initializations and progressive refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 12.0)
    
    global_best = np.inf
    global_a = None
    
    def try_update(a, label=""):
        nonlocal global_best, global_a
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        ver = evaluate_sequence(a.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            print("  NEW BEST %s: %.6f" % (label, ver))
    
    # --- Load given sequence ---
    given_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 16:
            given_a = np.clip(given, 0, 1000)
            if given_a.sum() > 0.01:
                try_update(given_a, "given_initial")
    except Exception:
        pass
    
    # --- Phase 1: Exploration (30% budget) ---
    # Try multiple initializations and short optimization runs
    usable = max(1.0, deadline - time.monotonic())
    explore_time = 0.30 * usable
    explore_deadline = time.monotonic() + explore_time
    
    n_main = 8192
    S_main = np.sqrt(2.0 * n_main)
    
    inits = []
    if given_a is not None:
        n_g = len(given_a)
        S_g = np.sqrt(2.0 * n_g)
        a_g = given_a / given_a.sum() * S_g
        inits.append(("given", a_g, n_g, S_g))
    
    inits.append(("smooth", smooth_init(n_main, rng), n_main, S_main))
    inits.append(("gs", gerchberg_saxton_init(n_main, rng), n_main, S_main))
    inits.append(("lowpass", lowpass_random_init(n_main, rng), n_main, S_main))
    inits.append(("rand", rng.random(n_main), n_main, S_main))
    
    # Also try smaller length
    n_sm = 4096
    S_sm = np.sqrt(2.0 * n_sm)
    inits.append(("smooth_sm", smooth_init(n_sm, rng), n_sm, S_sm))
    
    per_init = explore_time / max(1, len(inits))
    
    phase1_best = None
    phase1_score = np.inf
    
    for name, a_init, n, S in inits:
        if time.monotonic() > explore_deadline - 1.0:
            break
        
        seg = min(per_init * 0.8, explore_deadline - time.monotonic() - 0.5)
        if seg < 0.3:
            break
        
        t0 = time.monotonic()
        # Quick LSE-Adam run
        a_best, score = run_adam_lse(a_init, n, S, seg, rng,
                                      lr_init=0.4 * S / n,
                                      tau_init=1.0, tau_min=0.01,
                                      tau_decay=0.99)
        
        elapsed = time.monotonic() - t0
        print("Explore %-10s n=%4d score=%.6f time=%.1fs" % (name, n, score, elapsed))
        
        try_update(a_best, "explore_%s" % name)
        
        if score < phase1_score:
            phase1_score = score
            phase1_best = (a_best.copy(), n, S)
    
    # --- Phase 2: Focused optimization on best candidate (55% budget) ---
    if phase1_best is not None:
        a_cur, n_best, S_best = phase1_best
    elif global_a is not None:
        a_cur = global_a.copy()
        n_best = len(a_cur)
        S_best = np.sqrt(2.0 * n_best)
    else:
        a_cur = np.ones(n_main) * S_main / n_main
        n_best = n_main
        S_best = S_main
    
    focus_deadline = time.monotonic() + 0.55 * usable
    
    # Progressive refinement: multiple passes with decreasing tau and lr
    passes = [
        # (type, lr_scale, tau_init, tau_min, tau_decay, label)
        ("adam", 0.3, 0.8, 0.01, 0.995, "coarse"),
        ("adam", 0.2, 0.4, 0.005, 0.996, "medium"),
        ("mirror", 0.3, 0.3, 0.005, 0.997, "mirror"),
        ("adam", 0.15, 0.2, 0.003, 0.997, "fine"),
        ("topk", 0.2, 1.0, 1.0, 1.0, "topk"),
        ("adam", 0.1, 0.1, 0.001, 0.998, "veryfine"),
    ]
    
    n_passes = len(passes)
    per_pass = (focus_deadline - time.monotonic()) / max(1, n_passes)
    
    for ptype, lr_scale, t_init, t_min, t_dec, plabel in passes:
        if time.monotonic() > focus_deadline - 2.0:
            break
        
        seg = min(per_pass, focus_deadline - time.monotonic() - 1.0)
        if seg < 1.0:
            break
        
        t0 = time.monotonic()
        
        if ptype == "adam":
            a_cur, score = run_adam_lse(a_cur, n_best, S_best, seg, rng,
                                         lr_init=lr_scale * S_best / n_best,
                                         tau_init=t_init, tau_min=t_min,
                                         tau_decay=t_dec)
        elif ptype == "mirror":
            a_cur, score = run_mirror_lse(a_cur, n_best, S_best, seg, rng,
                                           eta_init=lr_scale,
                                           tau_init=t_init, tau_min=t_min,
                                           tau_decay=t_dec)
        elif ptype == "topk":
            a_cur, score = run_topk_subgrad(a_cur, n_best, S_best, seg, rng,
                                             lr_init=lr_scale * S_best / n_best,
                                             top_k=15)
        else:
            a_cur, score = run_adam_lse(a_cur, n_best, S_best, seg, rng,
                                         lr_init=lr_scale * S_best / n_best,
                                         tau_init=t_init, tau_min=t_min,
                                         tau_decay=t_dec)
        
        elapsed = time.monotonic() - t0
        print("Focus %-8s score=%.6f time=%.1fs" % (plabel, score, elapsed))
        
        try_update(a_cur, "focus_%s" % plabel)
    
    # --- Phase 3: Final intensive polish (remaining budget) ---
    remaining = deadline - time.monotonic()
    if remaining > 3.0 and global_a is not None:
        a_cur = global_a.copy()
        n_best = len(a_cur)
        S_best = np.sqrt(2.0 * n_best)
        
        # Very fine LSE-Adam
        seg1 = remaining * 0.5
        a_cur, score = run_adam_lse(a_cur, n_best, S_best, seg1, rng,
                                     lr_init=0.05 * S_best / n_best,
                                     tau_init=0.05, tau_min=0.001,
                                     tau_decay=0.999)
        print("Polish1: score=%.6f" % score)
        try_update(a_cur, "polish1")
        
        remaining2 = deadline - time.monotonic()
        if remaining2 > 2.0 and global_a is not None:
            a_cur = global_a.copy()
            # Fine mirror descent
            seg2 = remaining2 * 0.6
            a_cur, score = run_mirror_lse(a_cur, n_best, S_best, seg2, rng,
                                           eta_init=0.1,
                                           tau_init=0.05, tau_min=0.001,
                                           tau_decay=0.999)
            print("Polish2: score=%.6f" % score)
            try_update(a_cur, "polish2")
        
        # Final topk refinement
        remaining3 = deadline - time.monotonic()
        if remaining3 > 2.0 and global_a is not None:
            a_cur = global_a.copy()
            a_cur, score = run_topk_subgrad(a_cur, n_best, S_best, remaining3 * 0.7,
                                             rng, lr_init=0.05 * S_best / n_best,
                                             top_k=20)
            print("Polish3(topk): score=%.6f" % score)
            try_update(a_cur, "polish3")
    
    # --- Final verification and return ---
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_ver = evaluate_sequence(global_a.tolist())
    print("FINAL: verified_score=%.6f, length=%d" % (final_ver, len(global_a)))
    
    return [float(x) for x in global_a]
