    n = len(p)
    L = fft_len(n, 3)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2*n-1]
    cmax = np.max(c)
    if cmax <= 0:
        return np.zeros(n), 0.0, 1.0
    T = max(T * cmax, 1e-30)
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = np.sum(e)
    if s <= 0 or not np.isfinite(s):
        w = np.zeros(2*n-1)
        w[np.argmax(c)] = 1.0
        s = 1.0
    else:
        w = e / s
    # Gradient: grad_j = 2 * sum_k w_k * p_{k-j} (with boundary)
    # Using FFT: ifft(fft(p) * fft(w[::-1]))
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    # Extract the right part
    G = h[2*n-2:n-2:-1]
    if len(G) != n:
        tmp = np.zeros(n)
        m = min(n, len(G))
        tmp[:m] = G[:m]
        G = tmp
    g = 2.0 * np.maximum(G, 0.0)
    sm = cmax + T * np.log(s)
    return g, sm, T

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0/n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = np.sum(out)
    if s <= 1e-300:
        return np.full(n, 1.0/n)
    return out / s

def mirror_step(p, g, eta):
    """Multiplicative update: p * exp(-eta * g), normalized."""
    # Clip eta * g to avoid overflow
    ag = eta * g
    ag = np.clip(ag, -50.0, 50.0)
    new_p = p * np.exp(-ag)
    s = np.sum(new_p)
    if s <= 0 or not np.isfinite(s):
        return p.copy()
    return new_p / s

def normalize(p):
    p = np.maximum(p, 0.0)
    s = np.sum(p)
    if s <= 1e-12:
        return np.full(len(p), 1.0/len(p))
    return p / s

def make_flat_top(n, frac=0.5):
    """Raised cosine flat-top: flat in middle, cosine taper at edges."""
    x = np.linspace(0, 1, n)
    half = frac / 2.0
    p = np.ones(n)
    # Taper at left
    left = x < (1 - frac) / 2.0
    p[left] = 0.5 * (1 + np.cos(np.pi * x[left] / ((1-frac)/2.0)))
    # Taper at right
    right = x > (1 + frac) / 2.0
    p[right] = 0.5 * (1 + np.cos(np.pi * (1-x[right]) / ((1-frac)/2.0)))
    return normalize(p)

def make_gaussian(n, sigma=0.2):
    x = np.linspace(0, 1, n)
    p = np.exp(-(x - 0.5)**2 / (2 * sigma**2))
    return normalize(p)

def make_beta_shape(n, a, b):
    x = np.linspace(1e-6, 1-1e-6, n)
    p = x**a * (1-x)**b
    return normalize(p)

def make_raised_cosine(n):
    x = np.linspace(0, 1, n)
    p = 0.5 * (1 + np.cos(2 * np.pi * x))
    p = np.maximum(p, 1e-8)
    return normalize(p)

def make_two_peak(n, d=0.2, w=0.08):
    x = np.linspace(0, 1, n)
    p = np.exp(-(x - 0.5 + d)**2 / (2*w**2)) + np.exp(-(x - 0.5 - d)**2 / (2*w**2))
    return normalize(p)

def make_sinusoidal(n, freq=2.0, amp=0.3):
    x = np.linspace(0, 1, n)
    p = 1.0 + amp * np.sin(2 * np.pi * freq * x)
    return normalize(np.maximum(p, 1e-8))

def generate_diverse_inits(n, rng):
    """Generate diverse initial points."""
    inits = []
    # Flat-top variants
    for frac in (0.3, 0.5, 0.7, 0.85, 1.0):
        inits.append(make_flat_top(n, frac))
    # Gaussian variants
    for sig in (0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4):
        inits.append(make_gaussian(n, sig))
    # Beta shapes
    for a, b in [(1,1), (2,2), (3,3), (1,2), (2,1), (0.5,0.5), (1,3), (3,1), (0.3,0.3)]:
        inits.append(make_beta_shape(n, a, b))
    # Raised cosine
    inits.append(make_raised_cosine(n))
    # Two-peak
    for d in (0.1, 0.15, 0.2, 0.25):
        for w in (0.05, 0.08, 0.12):
            inits.append(make_two_peak(n, d, w))
    # Sinusoidal
    for freq in (1, 2, 3, 4):
        for amp in (0.1, 0.2, 0.3, 0.5):
            inits.append(make_sinusoidal(n, freq, amp))
    # Random smooth
    for _ in range(8):
        x = np.linspace(0, 1, n)
        f1 = rng.integers(1, 8)
        f2 = rng.integers(2, 15)
        p = 1.0 + 0.4*np.sin(2*np.pi*f1*x + rng.uniform(0, 2*np.pi)) + 0.2*np.sin(2*np.pi*f2*x + rng.uniform(0, 2*np.pi))
        inits.append(normalize(np.maximum(p, 1e-6)))
    # Uniform
    inits.append(np.full(n, 1.0/n))
    return inits

def lp_best_response(p, M, max_time=30.0):
    """Solve LP: max sum(g) s.t. (p*g)_k <= M, g >= 0."""
    n = len(p)
    if not _HAVE_LP or n > 800:
        return None
    start = time.monotonic()
    try:
        # Build constraint matrix: A has rows for each lag k (0..2n-2), columns for g_j
        # (p*g)_k = sum_j p_j * g_{k-j} (valid indices)
        # A[k, j] = p_{k-j} if 0 <= k-j < n, else 0
        # This is a "banded" Toeplitz-like matrix
        rows = []
        cols = []
        data = []
        for k in range(2*n-1):
            jmin = max(0, k - n + 1)
            jmax = min(n-1, k)
            if jmin > jmax:
                continue
            for j in range(jmin, jmax+1):
                val = p[k-j]
                if val > 1e-15:
                    rows.append(k)
                    cols.append(j)
                    data.append(val)
        A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs', options={'time_limit': max_time, 'disp': False})
        if time.monotonic() - start > max_time:
            return None
        if res is not None and res.status == 0:
            g = np.asarray(res.x, dtype=np.float64)
            g = np.maximum(g, 0)
            s = np.sum(g)
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

def optimize_mirror(p0, rng, time_limit, label=""):
    """
    Optimize using mirror descent + smoothed max.
    Returns (best_p, best_score).
    """
    p = normalize(np.asarray(p0, dtype=np.float64))
    n = len(p)
    best_p = p.copy()
    best_sc = score_normalized(p)
    cur_sc = best_sc

    end = time.monotonic() + time_limit
    T = 0.1  # initial temperature (fraction of cmax)
    eta = 0.5  # initial step size
    since_best = 0
    iter_no = 0
    lp_counter = 0

    while time.monotonic() < end:
        iter_no += 1
        
        # Adaptive temperature: cool down over time, but not too fast
        T = max(0.005, 0.1 * (0.998 ** iter_no))
        
        # Compute gradient of smoothed max
        g, sm_val, T_used = grad_smoothed_max(p, T)
        
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        
        gmax = np.max(np.abs(g))
        if gmax < 1e-15:
            # At a "stationary" point of the smoothed max
            since_best += 1
            if since_best > 500:
                # Try a random kick
                p = normalize(p + rng.normal(0, 0.1/n, n))
                cur_sc = score_normalized(p)
                T = 0.1
                since_best = 0
            continue
        
        # Mirror descent step
        accepted = False
        for factor in (1.0, 0.5, 0.25, 0.1, 0.05):
            step = factor * eta / gmax
            p_new = mirror_step(p, g, step)
            sc_new = score_normalized(p_new)
            sm_new = smoothed_max(conv_full(p_new), T_used)
            
            # Accept if smoothed max decreased or actual score decreased
            if sm_new < sm_val - 1e-10 * max(1, abs(sm_val)):
                accepted = True
                break
            elif sc_new < cur_sc - 1e-10:
                if sm_new < sm_val + 0.5 * max(1, abs(sm_val)):
                    accepted = True
                    break
            elif sc_new < best_sc - 1e-10:
                accepted = True
                break
        
        if accepted:
            if sc_new < best_sc:
                best_sc = sc_new
                best_p = p_new.copy()
                since_best = 0
            else:
                since_best += 1
            cur_sc = sc_new
            p = p_new
            # Slightly increase step size on success
            eta = min(eta * 1.01, 2.0)
        else:
            eta = max(eta * 0.7, 0.01)
            since_best += 1
            
            if since_best > 300:
                # Random perturbation
                sig = 0.5 / n
                p = normalize(p + rng.normal(0, sig, n))
                cur_sc = score_normalized(p)
                T = max(T, 0.05)
                since_best = 0
            elif since_best > 150:
                # Pair move: transfer mass from high-gradient to low-gradient
                hi = np.argmax(g)
                lo = np.argmin(g)
                if hi != lo:
                    delta = 0.1 * p[hi]
                    p_new = p.copy()
                    p_new[hi] -= delta
                    p_new[lo] += delta
                    p_new = normalize(p_new)
                    sc_new = score_normalized(p_new)
                    if sc_new < cur_sc:
                        p = p_new
                        cur_sc = sc_new
                        if sc_new < best_sc:
                            best_sc = sc_new
                            best_p = p_new.copy()
                            since_best = 0
        
        # Periodic LP teleportation
        lp_counter += 1
        if _HAVE_LP and n <= 600 and lp_counter % 500 == 0 and time.monotonic() < end - 40:
            c_cur = conv_full(p)
            M_cur = np.max(c_cur)
            g_lp = lp_best_response(p, M_cur, max_time=min(20.0, end - time.monotonic() - 10))
            if g_lp is not None:
                # Blend and check
                for t in (1.0, 0.7, 0.5, 0.3, 0.1):
                    p_blend = normalize((1-t)*p + t*g_lp)
                    sc_blend = score_normalized(p_blend)
                    if sc_blend < best_sc:
                        best_sc = sc_blend
                        best_p = p_blend.copy()
                        p = p_blend.copy()
                        cur_sc = sc_blend
                        since_best = 0
                        break
        
        # Restart from best if stuck
        if since_best > 800:
            p = normalize(0.7 * best_p + 0.3 * p + rng.normal(0, 0.2/n, n))
            cur_sc = score_normalized(p)
            T = 0.1
            since_best = 0
        
        if iter_no % 500 == 0:
            print(f"  {label} iter={iter_no} best={best_sc:.6f} cur={cur_sc:.6f} T={T:.4f} eta={eta:.4f}", flush=True)
        
        if best_sc < 1.498 and since_best > 2000:
            break
    
    return best_p, best_sc

def optimize_euclidean(p0, rng, time_limit, label=""):
    """
    Alternative: Euclidean (projected gradient) optimization.
    Sometimes escapes local minima that mirror descent gets stuck in.
    """
    p = normalize(np.asarray(p0, dtype=np.float64))
    n = len(p)
    best_p = p.copy()
    best_sc = score_normalized(p)
    cur_sc = best_sc

    end = time.monotonic() + time_limit
    T = 0.1
    lr = 1.0
    since_best = 0
    iter_no = 0

    while time.monotonic() < end:
        iter_no += 1
        T = max(0.005, 0.1 * (0.999 ** iter_no))
        
        g, sm_val, T_used = grad_smoothed_max(p, T)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        
        gmax = np.max(np.abs(g))
        if gmax < 1e-15:
            since_best += 1
            if since_best > 400:
                p = normalize(p + rng.normal(0, 0.1/n, n))
                cur_sc = score_normalized(p)
                since_best = 0
            continue
        
        accepted = False
        for factor in (1.0, 0.5, 0.2, 0.1, 0.05, 0.02):
            step = factor * lr / gmax
            p_new = project_simplex(p - step * g)
            sc_new = score_normalized(p_new)
            sm_new = smoothed_max(conv_full(p_new), T_used)
            
            if sm_new < sm_val - 1e-10:
                accepted = True
                break
            elif sc_new < cur_sc - 1e-10 and sm_new < sm_val + 0.5:
                accepted = True
                break
            elif sc_new < best_sc - 1e-10:
                accepted = True
                break
        
        if accepted:
            if sc_new < best_sc:
                best_sc = sc_new
                best_p = p_new.copy()
                since_best = 0
            else:
                since_best += 1
            cur_sc = sc_new
            p = p_new
            lr = min(lr * 1.005, 5.0)
        else:
            lr = max(lr * 0.6, 0.01)
            since_best += 1
            if since_best > 300:
                p = normalize(p + rng.normal(0, 0.3/n, n))
                cur_sc = score_normalized(p)
                T = 0.08
                since_best = 0
        
        if since_best > 700:
            p = normalize(0.6 * best_p + 0.4 * p + rng.normal(0, 0.15/n, n))
            cur_sc = score_normalized(p)
            T = 0.1
            since_best = 0
        
        if iter_no % 500 == 0:
            print(f"  {label} iter={iter_no} best={best_sc:.6f} cur={cur_sc:.6f}", flush=True)
        
        if best_sc < 1.498 and since_best > 2000:
            break
    
    return best_p, best_sc

def _fft_len(n, factor=2):
    total = int(max(1, factor * n))
    if _HAVE_SCIPY_FFT:
        return int(next_fast_len(total))
    m = 1
    while m < total:

    return m


def conv_full(a):
    n = len(a)
    if n == 0:
        return np.empty(0)
    L = _fft_len(n, 2)
    F = fft(a, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def score_norm(p):
    n = len(p)
    c = conv_full(p)
    return 2.0 * n * float(np.max(c))


def smoothed_max(c, T):
    cmax = float(np.max(c))
    if cmax <= 0:
        return 0.0
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        return cmax
    return cmax + T * np.log(s)


def grad_smoothed(p, T_frac):
    n = len(p)
    L = _fft_len(n, 3)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]
    cmax = float(np.max(c))
    if cmax <= 0:
        return np.zeros(n), 0.0, 1.0
    T = max(T_frac * cmax, 1e-30)
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = e / s
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))

        tmp = np.zeros(n)

        tmp[:m] = G[:m]
        G = tmp

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(np.sum(out))
    if s <= 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def mirror_update(p, g, step):
    ag = np.clip(step * g, -40.0, 40.0)
    new_p = p * np.exp(-ag)
    s = float(np.sum(new_p))
    if s <= 0 or not np.isfinite(s):
        return p.copy()
    return new_p / s


def normalize(p):
    p = np.maximum(np.asarray(p, dtype=np.float64), 0.0)
    s = float(np.sum(p))
    if s <= 1e-12:
        return np.full(len(p), 1.0 / max(1, len(p)))
    return p / s


def make_flat_top(n, frac=0.6):
    x = np.linspace(0, 1, n)
    p = np.ones(n)
    edge = (1.0 - frac) / 2.0
    if edge > 0.01:
        left_mask = x < edge
        p[left_mask] = 0.5 * (1.0 + np.cos(np.pi * x[left_mask] / edge))
        right_mask = x > 1.0 - edge
        p[right_mask] = 0.5 * (1.0 + np.cos(np.pi * (1.0 - x[right_mask]) / edge))
    return normalize(p)


def make_gauss(n, sigma):
    x = np.linspace(0, 1, n)
    p = np.exp(-(x - 0.5) ** 2 / (2.0 * sigma ** 2))
    return normalize(p)


def make_beta_shap(n, a, b):
    x = np.linspace(1e-6, 1 - 1e-6, n)
    p = x ** a * (1 - x) ** b
    return normalize(p)


def make_two_peak(n, d, w):
    x = np.linspace(0, 1, n)
    p = np.exp(-(x - 0.5 + d) ** 2 / (2 * w ** 2)) + np.exp(-(x - 0.5 - d) ** 2 / (2 * w ** 2))
    return normalize(p)


def make_windowed(n, method='hann'):
    if method == 'hann':
        x = np.linspace(0, 1, n)
        p = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif method == 'hamming':
        x = np.linspace(0, 1, n)
        p = 0.54 + 0.46 * np.cos(2 * np.pi * x)
    elif method == 'blackman':
        x = np.linspace(0, 1, n)
        p = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    else:
        p = np.ones(n)
    return normalize(np.maximum(p, 1e-10))


def make_riemann(n, k):
    x = np.linspace(1e-4, 1.0, n)
    p = np.zeros(n)
    for m in range(1, k + 1):
        p += np.abs(1.0 / (x - m / (k + 1.0))) + 1.0 / (m * m)
    return normalize(p)

def lp_solve(p, M, max_time=20.0):
    n = len(p)
    if not _HAVE_LP or n > 700:
        return None
    start = time.monotonic()
    try:
        rows = []
        cols = []
        data = []
        for k in range(2 * n - 1):
            jmin = max(0, k - n + 1)
            jmax = min(n - 1, k)
            for j in range(jmin, jmax + 1):
                val = p[k - j]
                if val > 1e-14:
                    rows.append(k)
                    cols.append(j)
                    data.append(val)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'time_limit': max_time, 'disp': False})
        if res is not None and res.status == 0:
            g = np.asarray(res.x, dtype=np.float64)
            g = np.maximum(g, 0)
            s = float(np.sum(g))
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

def opt_grad(p0, rng, tlimit, label=""):
    p = normalize(np.asarray(p0, dtype=np.float64))
    n = len(p)
    best_p = p.copy()
    best_sc = score_norm(p)
    cur_sc = best_sc
    end = time.monotonic() + tlimit
    T_frac = 0.12
    lr = 1.5
    since_best = 0
    it = 0
    lp_cd = 0
    prev_p = p.copy()
    mom = 0.0

    while time.monotonic() < end:
        it += 1
        T_frac = max(0.004, 0.12 * (0.998 ** it))
        mom = 0.3 if it > 100 else 0.0

        g, sm, T_used = grad_smoothed(p, T_frac)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        gmax = float(np.max(np.abs(g)))

        if gmax < 1e-14:
            since_best += 1
            if since_best % 200 == 0:
                p = normalize(p + rng.normal(0, 0.3 / n, n))
                cur_sc = score_norm(p)
                since_best = 0
            continue

        accepted = False
        for fac in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
            step = fac * lr / gmax
            v = p - step * g
            if mom > 0:
                v = v + mom * (p - prev_p)
            p_new = project_simplex(v)
            sc_new = score_norm(p_new)
            c_new = conv_full(p_new)
            sm_new = smoothed_max(c_new, T_used)

            if sm_new < sm - 1e-9:
                accepted = True
                break
            elif sc_new < cur_sc - 1e-9 and sm_new < sm + 0.4:
                accepted = True
                break
            elif sc_new < best_sc - 1e-9:
                accepted = True
                break

        if accepted:
            prev_p = p.copy()
            if sc_new < best_sc:
                best_sc = sc_new
                best_p = p_new.copy()
                since_best = 0
            else:
                since_best += 1
            cur_sc = sc_new
            p = p_new
            lr = min(lr * 1.006, 5.0)
        else:
            lr = max(lr * 0.6, 0.05)
            since_best += 1
            if since_best > 250:
                p = normalize(p + rng.normal(0, 0.4 / n, n))
                cur_sc = score_norm(p)
                T_frac = max(T_frac, 0.05)
                since_best = 0
                prev_p = p.copy()
            elif since_best > 100:
                hi = int(np.argmax(g))
                lo = int(np.argmin(g))
                if hi != lo:
                    d = 0.12 * p[hi]
                    pn = p.copy()
                    pn[hi] -= d
                    pn[lo] += d
                    pn = project_simplex(pn)
                    sc_n = score_norm(pn)
                    if sc_n < cur_sc:
                        p = pn
                        cur_sc = sc_n
                        prev_p = p.copy()
                        if sc_n < best_sc:
                            best_sc = sc_n
                            best_p = p.copy()
                            since_best = 0

        lp_cd += 1
        if _HAVE_LP and n <= 600 and lp_cd % 400 == 0 and time.monotonic() < end - 35:
            c_c = conv_full(p)
            M_c = float(np.max(c_c))
            g_lp = lp_solve(p, M_c, max_time=min(15.0, end - time.monotonic() - 15))
            if g_lp is not None:
                for t in (1.0, 0.7, 0.5, 0.3):
                    pb = normalize((1 - t) * p + t * g_lp)
                    sb = score_norm(pb)
                    if sb < best_sc:
                        best_sc = sb
                        best_p = pb.copy()
                        p = pb.copy()
                        cur_sc = sb
                        since_best = 0
                        prev_p = p.copy()
                        break

        if since_best > 700:
            p = normalize(0.6 * best_p + 0.4 * p + rng.normal(0, 0.2 / n, n))
            cur_sc = score_norm(p)
            T_frac = 0.1
            since_best = 0
            prev_p = p.copy()

        if it % 800 == 0:
            print(f"    {label} it={it} best={best_sc:.6f} cur={cur_sc:.6f}", flush=True)
        if best_sc < 1.497 and since_best > 3000:
            break

    return best_p, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    global _global_best_p, _global_best_score
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 40

    _global_best_p = None
    _global_best_score = np.inf

    parent = None
    try:
        pa = np.asarray(height_sequence_1, dtype=np.float64)
        pa = np.where(np.isfinite(pa), pa, 0.0)
        pa = np.maximum(pa, 0.0)
        if pa.ndim == 1 and len(pa) >= 64 and np.sum(pa) > 0:
            parent = normalize(pa)
            sp = score_norm(parent)
            print(f"Parent: n={len(parent)}, score={sp:.6f}", flush=True)
            _global_best_score = sp
            _global_best_p = parent.copy()
    except Exception:
        pass

    n_list = [512, 1024, 2048]
    if parent is not None:
        pn = len(parent)
        if pn <= 4096 and pn >= 128:
            n_list.insert(0, pn)
    n_list = sorted(set(n_list))
    print(f"Lengths to try: {n_list}", flush=True)

    # Phase 1: Evaluate inits
    print("Phase 1: Scoring initializations", flush=True)
    scored_by_n = {}
    for n in n_list:
        if time.monotonic() > deadline - 100:
            break
        inits = gen_inits(n, rng)
        if parent is not None:
            if len(parent) == n:
                inits.append(parent)
            else:
                xo = np.linspace(0, 1, len(parent))
                xn = np.linspace(0, 1, n)
                inits.append(normalize(np.interp(xn, xo, parent)))
        scored = []
        for p in inits:
            p = normalize(p)
            sc = score_norm(p)
            if np.isfinite(sc):
                scored.append((sc, p))
        scored.sort(key=lambda x: x[0])
        scored_by_n[n] = scored
        for sc, p in scored[:2]:
            if sc < _global_best_score:
                _global_best_score = sc
                _global_best_p = p.copy()
        print(f"  n={n}: best={scored[0][0]:.6f} ({len(scored)} inits)", flush=True)

    # Phase 2: Build and execute optimization jobs
    print("Phase 2: Optimizing", flush=True)
    jobs = []
    for n in n_list:
        if n not in scored_by_n:
            continue
        for i, (sc, p) in enumerate(scored_by_n[n][:5]):
            use_mirror = (i % 2 == 0)
            lbl = f"n{n}_i{i}_{'M' if use_mirror else 'G'}"
            jobs.append((sc, p, use_mirror, lbl))
    jobs.sort(key=lambda x: x[0])

    n_jobs = len(jobs)
    final_res = 200.0
    avail = max(0.0, (deadline - time.monotonic()) - final_res)
    tper = max(25.0, min(250.0, avail / max(1, n_jobs)))

    for ji, (sc, p, um, lbl) in enumerate(jobs):
        if time.monotonic() > deadline - final_res:
            break
        rem = deadline - time.monotonic() - final_res
        if rem < 20:
            break
        lim = min(tper, rem)
        print(f"  Job {ji+1}/{n_jobs} [{lbl}] t={lim:.0f}s", flush=True)
        if um:
            pb, sb = opt_mirror(p, rng, lim, lbl)
        else:
            pb, sb = opt_grad(p, rng, lim, lbl)
        if sb < _global_best_score:
            _global_best_score = sb
            _global_best_p = pb.copy()
            print(f"  *** NEW BEST {sb:.8f} (n={len(pb)})", flush=True)
        # Cross-optimize with other method
        if time.monotonic() < deadline - final_res - 25 and sb < sc - 1e-5:
            oth_lim = min(45.0, deadline - time.monotonic() - final_res)
            if um:
                pb2, sb2 = opt_grad(pb, rng, oth_lim, lbl + "_x")
            else:
                pb2, sb2 = opt_mirror(pb, rng, oth_lim, lbl + "_x")
            if sb2 < _global_best_score:
                _global_best_score = sb2
                _global_best_p = pb2.copy()
                print(f"  *** CROSS BEST {sb2:.8f}", flush=True)

    # Phase 3: Final polish
    print("Phase 3: Final polish", flush=True)
    if _global_best_p is not None and time.monotonic() < deadline - 10:
        rem = deadline - time.monotonic() - 5
        half = rem / 2
        pb, sb = opt_mirror(_global_best_p, rng, half, "pol_M")
        if sb < _global_best_score:
            _global_best_score = sb
            _global_best_p = pb.copy()
        if time.monotonic() < deadline - 10:
            pb2, sb2 = opt_grad(_global_best_p, rng, min(half, deadline - time.monotonic() - 5), "pol_G")
            if sb2 < _global_best_score:
                _global_best_score = sb2
                _global_best_p = pb2.copy()

    # Phase 4: LP on best if short
    if _global_best_p is not None and len(_global_best_p) <= 600 and _HAVE_LP and time.monotonic() < deadline - 25:
        print("Phase 4: LP", flush=True)
        cc = conv_full(_global_best_p)
        Mc = float(np.max(cc))
        gl = lp_solve(_global_best_p, Mc, max_time=min(50.0, deadline - time.monotonic() - 15))
        if gl is not None:
            for t in (1.0, 0.8, 0.6, 0.4, 0.2, 0.1):
                pbl = normalize((1 - t) * _global_best_p + t * gl)
                sbl = score_norm(pbl)
                if sbl < _global_best_score:
                    _global_best_score = sbl
                    _global_best_p = pbl.copy()
                    print(f"  LP t={t}: {sbl:.8f}", flush=True)
            if time.monotonic() < deadline - 12:
                pbo, sbo = opt_mirror(_global_best_p, rng, min(25.0, deadline - time.monotonic() - 8), "lpopt")
                if sbo < _global_best_score:
                    _global_best_score = sbo
                    _global_best_p = pbo.copy()

    # Output
    if _global_best_p is None:
        _global_best_p = np.full(512, 1.0 / 512)
        _global_best_score = 2.0

    _global_best_p = np.asarray(_global_best_p, dtype=np.float64)
    _global_best_p = np.where(np.isfinite(_global_best_p), _global_best_p, 0.0)
    _global_best_p = np.maximum(_global_best_p, 0.0)
    _global_best_p = np.minimum(_global_best_p, 1000.0)
    if float(np.sum(_global_best_p)) < 0.01:
        _global_best_p = np.full(512, 1.0 / 512)

    print(f"\nFINAL: score={_global_best_score:.10f}, n={len(_global_best_p)}", flush=True)
    return [float(x) for x in _global_best_p]
