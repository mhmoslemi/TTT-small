    phase = A / (np.abs(A) + 1e-30)
    A_new = new_mag * phase

    return a_new


def _optimize_one(a0, deadline, method='soft', max_iters=5000):

    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    S = float(a.sum())
    if S < 1e-8:
        a = np.ones(len(a))
        S = float(a.sum())

    S_target = np.sqrt(2.0 * n)
    a = _water_filling(a, S_target)
    
    c = _autocorr(a)
    score = float(2.0 * n * c.max() / (S_target * S_target))
    best = a.copy()
    best_score = score

    tau = 10.0  # softmax temperature
    lr_base = 0.1
    spec_interval = 50  # do spectral flattening every 50 iters

            break
        
        # Cosine schedule for learning rate
        progress = it / max_iters
        lr = lr_base * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, 0.001)
        
        # Anneal temperature (increase to focus on peak)
        tau = 10.0 + 50.0 * progress

        c = _autocorr(a)

            grad = _grad_softmax(a, c, tau)
        else:

            grad = _grad_peak(a, c, k_peak)

def _lp_refine(a, deadline, max_iters=50):
    """LP-based refinement: maximize sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(a)
    if n > 512:
        return a, _objective(a)
    
    S_target = np.sqrt(2.0 * n)
    best = a.copy()
    best_score = _objective(a)
    
    c = _autocorr(a)
    M = float(c.max())
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        # Build LP
        L = 2 * n - 1
        c_obj = -np.ones(n)
        A_mat = np.zeros((L, n))
        for m_idx in range(L):
            lo = max(0, m_idx - n + 1)
            hi = min(n - 1, m_idx)
            for j in range(lo, hi + 1):
                A_mat[m_idx, j] = a[m_idx - j]
        
        b_vec = np.full(L, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_mat, b_ub=b_vec, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break
        
        # Try blending
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = _water_filling(a_new, S_target)
            c_new = _autocorr(a_new)
            s_new = float(2.0 * n * c_new.max() / (S_target * S_target))
            if s_new < best_score - 1e-9:
                a = a_new
                best = a.copy()
                best_score = s_new
                M = float(c_new.max())
                improved = True
                break
        
        if not improved:
            break
    
    return best, best_score
