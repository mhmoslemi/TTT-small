    rows = np.empty(n * n + 2 * n, dtype=np.int32)
    cols = np.empty(n * n + 2 * n, dtype=np.int32)
    vals = np.empty(n * n + 2 * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        cnt = j_hi - j_lo + 1
        if cnt <= 0:
            continue
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = np.arange(j_lo, j_hi + 1)
        vals[idx:idx + cnt] = f[k - j_lo:k - j_lo + cnt]
        idx += cnt
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))
    return A


def _lp_solve(f, M, n):
    """Solve max sum(g) s.t. (f*g)_k <= M for all k, g >= 0. Returns g or None."""
    A = _build_lp_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                      method='highs', options={'disp': False, 'presolve': True,
                                               'time_limit': 60.0})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if np.all(np.isfinite(g)) and g.sum() > 1e-30:
                return g
    except Exception:
        pass
    return None


def _lp_iterate(p, rng, time_limit, n, verbose=True):
    """LP-based fixed-point iteration."""
    start = time.monotonic()
    best_p = p.copy()
    best_score = _score(p)
    cur_p = p.copy()
    cur_score = best_score
    no_improve = 0
    it = 0
    
    while time.monotonic() - start < time_limit:
        it += 1
        c = _fft_conv(cur_p)
        M = float(c.max())
        if M < 1e-30:
            break
        
        g = _lp_solve(cur_p, M, n)
        if g is None:
            break
        s_g = float(g.sum())
        if s_g < 1e-30:
            break
        g = g / s_g
        
        # Try several blending fractions
        best_t_score = cur_score
        best_t_p = cur_p
        for t in (1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05):
            candidate = (1.0 - t) * cur_p + t * g
            candidate = np.maximum(candidate, 0.0)
            candidate = _normalize(candidate)
            sc = _score(candidate)
            if np.isfinite(sc) and sc < best_t_score:
                best_t_score = sc
                best_t_p = candidate.copy()
        
        if best_t_score < cur_score - 1e-14:
            cur_p = best_t_p
            cur_score = best_t_score
            no_improve = 0
            if cur_score < best_score:
                best_score = cur_score
                best_p = cur_p.copy()
                if verbose:
                    elapsed = time.monotonic() - start
                    print(f"    LP[{n}] it={it} t_improve: score={best_score:.8f} elapsed={elapsed:.0f}s", flush=True)
        else:
            no_improve += 1
            if no_improve % 5 == 0:
                # Perturb to escape
                for sig in (0.05 / n, 0.02 / n, 0.005 / n):
                    pert = cur_p + rng.normal(0, sig, n)
                    pert = np.maximum(pert, 0.0)
                    pert = _normalize(pert)
                    sc = _score(pert)
                    if np.isfinite(sc) and sc < cur_score:
                        cur_p = pert
                        cur_score = sc
                        no_improve = 0
                        break
            if no_improve > 30:
                if verbose:
                    print(f"    LP[{n}] converged at it={it}, score={best_score:.8f}", flush=True)
                break
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP[{n}] it={it} best={best_score:.8f} cur={cur_score:.8f} elapsed={elapsed:.0f}s", flush=True)
    
    return best_p, best_score


def _grad_smooth_max(p, T_factor=0.1):
    """Gradient of smoothed max of convolution."""
    n = len(p)
    c = _fft_conv(p)
    cmax = float(c.max())
    if cmax < 1e-30:
        return np.zeros(n), cmax
    T = max(T_factor * cmax, 1e-12)
    w = np.exp((c - cmax) / T)
    w = w / w.sum()
    # Gradient: g_j = 2 * sum_k w_k * p_{k-j} (valid indices)
    # This is 2 * (w * p) evaluated appropriately
    L = next_fast_len(3 * n - 2)
    Fp = fft(p, L)
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    # h[j] = sum_k w_k * p_{k-j} (circular, but we want linear)
    # Need to extract the right part
    # Actually: (w * p)[m] = sum_k w_k p_{m-k}
    # We want g_j = 2 * sum_k w_k * p_{k-j} for k in [0, 2n-2]
    # This equals 2 * (w reversed corr p)[j]
    # Let me just compute directly for small n, or use the FFT trick
    # h = IFFT(FFT(p, L) * FFT(w_flipped, L)) gives circular correlation
    # h[j] = sum_k p[k] * w_flipped[(j-k) mod L]
    # w_flipped[k] = w[2n-2-k] (reversed)
    # So h[j] = sum_k p[k] * w[2n-2-(j-k) mod L]
    # For j in [0, n-1]: h[j] = sum_{k=0}^{n-1} p[k] * w[2n-2-j+k] (if in range)
    # This is getting complex. Let me use a direct approach.
    
    # Direct computation (O(n^2) but works for moderate n)
    g = np.zeros(n, dtype=np.float64)
    for j in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            idx = k - j
            if 0 <= idx < n:
                s += w[k] * p[idx]
        g[j] = 2.0 * s
    return g, cmax


def _gradient_descent(p, rng, time_limit, n, verbose=True):
    """Smoothed-max gradient descent on the simplex."""
    start = time.monotonic()
    best_p = p.copy()
    best_score = _score(p)
    cur_p = p.copy()
    cur_score = best_score
    no_improve = 0
    it = 0
    T_factor = 0.15
    
    while time.monotonic() - start < time_limit:
        it += 1
        g, cmax = _grad_smooth_max(cur_p, T_factor)
        g = g - g.mean()  # subtract mean for simplex
        
        # Try several step sizes
        best_step_score = cur_score
        best_step_p = cur_p
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        steps = [0.5 / gnorm, 0.2 / gnorm, 0.05 / gnorm, 0.01 / gnorm]
        for lr in steps:
            candidate = cur_p - lr * g
            candidate = _normalize(candidate)
            sc = _score(candidate)
            if np.isfinite(sc) and sc < best_step_score:
                best_step_score = sc
                best_step_p = candidate.copy()
        
        if best_step_score < cur_score - 1e-14:
            cur_p = best_step_p
            cur_score = best_step_score
            no_improve = 0
            T_factor = max(0.02, T_factor * 0.98)
            if cur_score < best_score:
                best_score = cur_score
                best_p = cur_p.copy()
                if verbose and (it % 100 == 0 or best_score < 1.51):
                    print(f"    GD[{n}] it={it} score={best_score:.8f}", flush=True)
        else:
            no_improve += 1
            T_factor = max(0.01, T_factor * 0.9)
            if no_improve % 20 == 0:
                pert = cur_p + rng.normal(0, 0.02 / n, n)
                pert = _normalize(pert)
                sc = _score(pert)
                if np.isfinite(sc) and sc < cur_score:
                    cur_p = pert
                    cur_score = sc
                    no_improve = 0
            if no_improve > 150:
                break
    
    return best_p, best_score


def _make_initials(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    
    # Parent
    if parent is not None and len(parent) == n and parent.sum() > 0:
        inits.append(_normalize(parent))
    
    # Uniform
    inits.append(np.full(n, 1.0 / n))
    
    # Linear ramps
    x = np.linspace(0, 1, n)
    for a in (0.5, 1.0, 2.0, 4.0, 8.0):
        inits.append(_normalize(1.0 + a * x))
        inits.append(_normalize(1.0 + a * (1.0 - x)))
    
    # Quadratic
    for a in (0.5, 1.0, 2.0, 4.0):
        inits.append(_normalize(1.0 + a * (2 * x - 1.0) ** 2))
        inits.append(_normalize(np.maximum(1.0 - a * (2 * x - 1.0) ** 2, 0.01)))
    
    # Power
    for a in (0.5, 1.0, 2.0, 3.0, 5.0):
        inits.append(_normalize(np.power(x + 1e-6, a)))
        inits.append(_normalize(np.power(1.0 - x + 1e-6, a)))
    
    # Two Gaussians (asymmetric)
    for mu1, sig1, mu2, sig2, w1, w2 in [
        (0.25, 0.08, 0.75, 0.06, 1.0, 0.8),
        (0.2, 0.1, 0.7, 0.07, 1.2, 0.7),
        (0.3, 0.06, 0.8, 0.05, 0.9, 1.1),
        (0.15, 0.12, 0.65, 0.09, 1.0, 0.6),
        (0.35, 0.05, 0.75, 0.08, 1.3, 0.8),
    ]:
        p = w1 * np.exp(-((x - mu1) ** 2) / (2 * sig1 ** 2)) + \
            w2 * np.exp(-((x - mu2) ** 2) / (2 * sig2 ** 2))
        inits.append(_normalize(p))
    
    # Skew Gaussian
    for mu, sig, skew in [(0.3, 0.15, 2.0), (0.4, 0.12, -1.5), (0.5, 0.2, 1.0)]:
        p = np.exp(-((x - mu) ** 2) / (2 * sig ** 2) + skew * (x - mu))
        inits.append(_normalize(p))
    
    # Random smooth
    for _ in range(5):
        f1, f2 = int(rng.integers(1, 5)), int(rng.integers(1, 8))
        p = 1.0 + 0.5 * np.sin(2 * np.pi * f1 * x + rng.uniform(0, 6.28)) + \
            0.3 * np.sin(2 * np.pi * f2 * x + rng.uniform(0, 6.28))
        inits.append(_normalize(np.maximum(p, 0.05)))
    
    # "Bump" shapes: concentrated in middle, asymmetric
    for center, width, asym in [(0.5, 0.15, 0.3), (0.4, 0.2, -0.5), (0.6, 0.12, 0.4)]:
        p = np.exp(-((x - center) ** 2) / (2 * width ** 2) + asym * (x - center))
        inits.append(_normalize(p))
    
    return inits


def _resample(p, n_target):
    """Resample a sequence to a different length via linear interpolation."""
    n_src = len(p)
    if n_src == n_target:
        return p.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    p_interp = np.interp(x_tgt, x_src, p)
    return _normalize(p_interp)


def propose_candidate(seed=42, budget_s=2340):
    """
    Search for a non-negative step function minimizing 2n*max(a*a)/(sum(a))^2.
    Uses LP-based fixed-point iteration as the primary optimizer,
    combined with gradient descent polishing and multiple restarts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0  # 30s safety buffer
    
    # Try to use parent sequence
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and parent_arr.sum() > 0:
            parent = parent_arr
            print(f"Parent sequence: n={len(parent_arr)}, score={_score(_normalize(parent_arr)):.6f}", flush=True)
    except Exception:
        parent = None
    
    best_p = None
    best_score = np.inf
    
    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None or len(p) == 0:
            return
        p = _normalize(p)
        if not np.isfinite(sc):
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print(f"  NEW BEST: n={len(p)} score={best_score:.8f}", flush=True)
    
    # Sequence lengths to try
    lengths = [500, 700, 1000]
    
    # Allocate time: roughly 30%, 35%, 25%, 10% polish
    time_alloc = {}
    remaining_frac = [0.30, 0.35, 0.25]
    for i, n in enumerate(lengths):
        time_alloc[n] = remaining_frac[i] * (budget_s - 100)
    
    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        
        t_alloc = time_alloc[n]
        print(f"\n=== Starting n={n}, budget={t_alloc:.0f}s ===", flush=True)
        
        # Generate and score initial points
        inits = _make_initials(n, rng, parent)
        # Also resample parent if available and different length
        if parent is not None and len(parent) != n:
            inits.append(_resample(parent, n))
        
        scored = []
        for p in inits:
            sc = _score(p)
            if np.isfinite(sc):
                scored.append((sc, p))
                try_record(p, sc)
        scored.sort(key=lambda x: x[0])
        
        print(f"  Initial scores: best={scored[0][0]:.6f}, worst={scored[-1][0]:.6f}, count={len(scored)}", flush=True)
        
        # Run LP optimization from top initial points
        n_starts = min(3, len(scored))
        t_per_start = t_alloc / max(1, n_starts)
        
        for i in range(n_starts):
            if time.monotonic() > deadline - 60:
                break
            sc0, p0 = scored[i]
            t_limit = min(t_per_start, deadline - time.monotonic() - 30)
            if t_limit < 30:
                break
            
            print(f"  LP start {i+1}/{n_starts}: initial score={sc0:.6f}, time={t_limit:.0f}s", flush=True)
            
            # LP phase
            p_lp, sc_lp = _lp_iterate(p0, rng, t_limit * 0.7, n, verbose=True)
            try_record(p_lp, sc_lp)
            
            # Gradient polish on the LP result
            t_gd = min(t_limit * 0.3, deadline - time.monotonic() - 30)
            if t_gd > 10:
                p_gd, sc_gd = _gradient_descent(p_lp, rng, t_gd, n, verbose=True)
                try_record(p_gd, sc_gd)
            
            # Feed back into scored list for next start
            if np.isfinite(sc_lp) and sc_lp < scored[0][0]:
                scored.insert(0, (sc_lp, p_lp.copy()))
    
    # Final polish on the best solution found
    if best_p is not None and time.monotonic() < deadline - 10:
        n_best = len(best_p)
        t_final = min(200.0, deadline - time.monotonic() - 10)
        if t_final > 10:
            print(f"\n=== Final polish: n={n_best}, time={t_final:.0f}s ===", flush=True)
            # Alternate between LP and GD
            cur_p = best_p.copy()
            cur_score = best_score
            t_lp_part = t_final * 0.6
            t_gd_part = t_final * 0.4
            
            if t_lp_part > 15 and n_best <= 1000:
                cur_p, cur_score = _lp_iterate(cur_p, rng, t_lp_part, n_best, verbose=True)
                try_record(cur_p, cur_score)
            
            t_gd_part = min(t_gd_part, deadline - time.monotonic() - 10)
            if t_gd_part > 10:
                cur_p, cur_score = _gradient_descent(cur_p, rng, t_gd_part, n_best, verbose=True)
                try_record(cur_p, cur_score)
    
    # Ensure valid output
    if best_p is None:
        best_p = np.full(500, 1.0 / 500)
    
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(best_p.sum())
    if s < 1e-12:
        best_p = np.full(500, 1.0 / 500)
    best_p = np.minimum(best_p, 1000.0)
    
    final_score = _score(best_p)
    elapsed = time.monotonic() - start_time
    print(f"\n=== DONE: n={len(best_p)}, score={final_score:.8f}, time={elapsed:.0f}s ===", flush=True)
    
    return [float(x) for x in best_p]
