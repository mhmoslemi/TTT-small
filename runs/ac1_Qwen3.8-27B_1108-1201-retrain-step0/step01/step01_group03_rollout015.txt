"""
Water-filling + gradient optimization for low-peak-autocorrelation sequences.
Combines fast coordinate-wise water-filling with soft-max gradient refinement.

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def water_fill_pass(a, M, n):
    """One Gauss-Seidel pass of water-filling. Returns updated a."""
    c = np.convolve(a, a)
    changed = False
    
    for j in range(n):
        old_aj = a[j]
        if old_aj <= 0:
            # Try to set it to a small positive value if feasible
            # For now, skip
            continue
        
        # Find max feasible a[j]
        max_aj = 1000.0
        
        # Constraint from k = 2j
        if 2*j < 2*n - 1:
            c_2j_without = c[2*j] - old_aj*old_aj
            bound = M - c_2j_without
            if bound < 0:
                max_aj = 0.0
            else:
                max_aj = min(max_aj, np.sqrt(bound))
        
        # Constraints from other k
        if max_aj > 0:
            # Vectorized computation
            k_vals = np.arange(j, min(j + n, 2*n - 1))
            k_vals = k_vals[k_vals != 2*j]
            if len(k_vals) > 0:
                idx_vals = k_vals - j
                a_sub = a[idx_vals]
                valid = a_sub > 1e-15
                if valid.any():
                    c_sub = c[k_vals[valid]]
                    a_valid = a_sub[valid]
                    c_without = c_sub - 2.0 * old_aj * a_valid
                    # Check feasibility
                    infeasible = c_without >= M - 1e-12
                    if infeasible.any():
                        max_aj = 0.0
                    else:
                        bounds = (M - c_without) / (2.0 * a_valid)
                        max_aj = min(max_aj, float(bounds.min()))
        
        new_aj = max(0.0, max_aj)
        if abs(new_aj - old_aj) > 1e-12:
            a[j] = new_aj
            # Update c incrementally
            delta = new_aj - old_aj
            k_update = np.arange(j, min(j + n, 2*n - 1))
            for k in k_update:
                if k == 2*j:
                    c[k] = c[k] - old_aj*old_aj + new_aj*new_aj
                else:
                    c[k] += 2.0 * delta * a[k - j]
            changed = True
    
    return a, changed

def water_fill_optimize(a_init, M_init, max_passes=200):
    """Iterate water-filling passes until convergence. Returns (a, M)."""
    a = a_init.copy().astype(np.float64)
    M = M_init
    n = len(a)
    
    for p in range(max_passes):
        a, changed = water_fill_pass(a, M, n)
        if not changed:
            break
    
    new_M = float(np.convolve(a, a).max())
    return a, new_M

def optimize_with_waterfill(a_init, deadline, max_rounds=30):
    """Main water-filling optimization loop."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    M = float(np.convolve(a, a).max())
    S = float(a.sum())
    best_score = 2.0 * n * M / (S * S)
    best_a = a.copy()
    
    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a, new_M = water_fill_optimize(a, M, max_passes=100)
        new_S = float(a.sum())
        if new_S < 0.01:
            break
        new_score = 2.0 * n * new_M / (new_S * new_S)
        
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
        
        if new_M >= M * (1 - 1e-6):
            # M didn't decrease; can't improve further with water-filling
            # Try reducing M slightly and projecting
            M = new_M * 0.99
            # Project a back
            scale = np.sqrt(M / float(np.convolve(a, a).max()))
            a = a * min(1.0, scale)
            M = float(np.convolve(a, a).max())
        else:
            M = new_M
    
    return best_a, best_score
