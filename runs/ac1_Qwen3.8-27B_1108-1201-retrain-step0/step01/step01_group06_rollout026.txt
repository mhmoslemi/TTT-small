def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_gradient(w, a):
    """Compute g[j] = 2 * sum_m w[j+m] * a[m] using FFT cross-correlation."""
    n = len(a)
    # w has length 2n-1, a has length n
    # We need (w ⋆ a)[j] for j = 0..n-1
    # (w ⋆ a)[j] = sum_k w[k+j] * a[k]
    fft_size = 1 << (2*n - 1 + n - 1 - 1).bit_length()  # >= (2n-1) + n - 1 = 3n-2
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(W) * A, fft_size)
    return 2.0 * corr[:n]

def spectral_project(a, n_iters=500):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a)
    L = 1 << (2*n).bit_length()  # power of 2 > 2n
    a_curr = np.zeros(L)
    a_curr[:n] = a
    
    for _ in range(n_iters):
        A = np.fft.fft(a_curr)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        # Target: flat magnitude
        target = np.sqrt(np.sum(mag**2) / L)
        A_new = target * phase
        a_curr = np.fft.ifft(A_new).real
        a_curr = np.maximum(a_curr, 0)
        a_curr_sum = a_curr[:n].sum()
        if a_curr_sum < 1e-15:
            break
        # Normalize but keep the full array
        a_curr[:n] /= a_curr_sum
        a_curr[n:] = 0
    
    result = a_curr[:n]
    s = result.sum()
    if s > 1e-15:
        result /= s
    else:
        result = np.ones(n) / n
    return result
