"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS-B with softmax
parameterization and multi-scale soft-max annealing.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def soft_max_obj_grad(u, tau, n):
    """
    Compute F_tau(softmax(u)) and its gradient with respect to u.
    u: array of shape (n,)
    tau: soft-max temperature
    n: sequence length
    """
    # Compute a = softmax(u)
    u_shifted = u - u.max()
    exp_u = np.exp(u_shifted)
    a = exp_u / exp_u.sum()
    
    # Compute c = conv(a, a)
    c = np.convolve(a, a)  # length 2n-1
    
    # Soft-max value
    c_shifted = c - c.max()
    lse = c.max() + tau * np.log(np.sum(np.exp(c_shifted / tau)))
    
    # Compute q = softmax(c / tau)
    q = np.exp(c_shifted / tau)
    q = q / q.sum()
    
    # Gradient of F w.r.t. a: (q * a)_valid + a
    # (q * a)[j] = sum_k q[k] * a[k-j] for valid k-j
    # This is conv(q[::-1], a)[n-1 : 2n-1]
    full_conv = np.convolve(q[::-1], a)
    grad_a = full_conv[n-1:2*n-1] + a
    
    # Gradient of F w.r.t. u via chain rule for softmax
    # dF/du[i] = a[i] * (grad_a[i] - sum_j a[j]*grad_a[j])
    mean_grad = np.dot(a, grad_a)
    grad_u = a * (grad_a - mean_grad)
    
    return lse, grad_u

def optimize_sequence(u0, n, tau_schedule, max_time_per_tau=30):
    """
    Run L-BFGS-B with decreasing tau.
    u0: initial parameter vector (length n)
    n: sequence length
    tau_schedule: list of tau values (decreasing)
    """
    u = u0.copy()
    best_score = np.inf
    best_a = None
    t_start = time.monotonic()
    
    for tau in tau_schedule:
        t_tau_start = time.monotonic()
        
        def fun(u):
            val, _ = soft_max_obj_grad(u, tau, n)
            return val
        
        def jac(u):
            _, g = soft_max_obj_grad(u, tau, n)
            return g
        
        try:
            result = minimize(
                fun, u, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': 200, 'maxfun': 1000, 'ftol': 1e-10, 'gtol': 1e-8}
            )
            u = result.x
        except Exception as e:
            print(f"  Optimization error at tau={tau}: {e}")
            break
        
        # Check time
        if time.monotonic() - t_start > max_time_per_tau * len(tau_schedule):
            break
        
        # Evaluate actual score
        u_shifted = u - u.max()
        a = np.exp(u_shifted) / np.exp(u_shifted).sum()
        c = np.convolve(a, a)
        score = 2 * n * c.max()  # sum(a) = 1
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score
