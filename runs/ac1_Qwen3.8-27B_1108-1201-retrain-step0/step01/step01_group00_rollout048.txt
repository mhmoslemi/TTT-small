   Actually, let me think about this more carefully. When we do f_new = (1-t)f + tg:
   (f_new * f_new)[k] = (1-t)^2 (f*f)[k] + 2t(1-t)(f*g)[k] + t^2 (g*g)[k]

   At the peak k* of f*f:
   (f_new*f_new)[k*] = (1-t)^2 * M + 2t(1-t)*(f*g)[k*] + t^2*(g*g)[k*]

   We know (f*g)[k*] <= M. So:
   <= (1-t)^2 * M + 2t(1-t)*M + t^2*(g*g)[k*]
   = M(1-t)^2 + 2t(1-t)*M + t^2*(g*g)[k*]
   = M(1-t)^2 + 2t(1-t)*M + t^2*(g*g)[k*]
   = M[(1-t)^2 + 2t(1-t)] + t^2*(g*g)[k*]
   = M(1-t)[(1-t) + 2t] + t^2*(g*g)[k*]
   = M(1-t)(1+t) + t^2*(g*g)[k*]
   = M(1-t^2) + t^2*(g*g)[k*]

   So the new peak at k* is at most M(1-t^2) + t^2*(g*g)[k*]. This is less than M if (g*g)[k*] < M, which is likely for small t.

   But the new peak might be at a different k! We need to check all k. For any k:
   (f_new*f_new)[k] = (1-t)^2 (f*f)[k] + 2t(1-t)(f*g)[k] + t^2 (g*g)[k]
   <= (1-t)^2 * M + 2t(1-t)*M + t^2 * max(g*g)
   = M(1-t^2) + t^2 * max(g*g)

   So the new peak is bounded by M(1-t^2) + t^2 * max(g*g). If max(g*g) < M, the peak decreases.

   The LP ensures max((f*g)[k]) <= M, but doesn't constrain max(g*g). So we should check the actual new peak after the update and use a small t if needed.

   Alternatively, we could add g*g constraints to the LP, but that would make it non-linear (quadratic constraints). Not feasible with linprog.

   I think the practical approach is:
   1. Solve the LP to get g
   2. Try a few values of t (e.g., 0.01, 0.05, 0.1, 0.2)
   3. Evaluate the actual new peak
   4. Use the best t

   Or simply use a small fixed t (like 0.05) and rely on the fact that for small t, the peak should decrease.

"""
Hybrid LP-guided descent and gradient optimization for minimizing the peak
of self-convolution of non-negative step functions. Uses linear programming
to find optimal descent directions, combined with active-constraint gradient
descent and simulated annealing polish.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score_fast(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def build_lp_matrix(f):
    """Build the constraint matrix for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    """
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M (element-wise), g >= 0."""
    n = len(f)
    c = -np.ones(n)  # minimize -sum(g)
    A = build_lp_matrix(f)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

def lp_descent(f_init, time_limit, rng, verbose=True):
    """LP-guided descent."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f = f_init.astype(float).copy()
    f = np.clip(f, 0, None)
    f_sum = np.sum(f)
    if f_sum < 1e-10:
        f = np.ones(n)
        f_sum = n
    f = f / f_sum * S
    
    best_f = f.copy()
    best_score = local_score_fast(f)
    t0 = time.monotonic()
    
    for step in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        g = solve_lp_direction(f, M)
        if g is None:
            break
        
        g = np.clip(g, 0, None)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum * S
        
        # Try different step sizes
        best_t = 0.05
        best_new_score = np.inf
        for t in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2]:
            f_new = (1-t) * f + t * g
            f_new = np.clip(f_new, 0, None)
            f_new_sum = np.sum(f_new)
            if f_new_sum < 1e-10:
                continue
            f_new = f_new / f_new_sum * S
            s = local_score_fast(f_new)
            if s < best_new_score:
                best_new_score = s
                best_t = t
        
        f = (1-best_t) * f + best_t * g
        f = np.clip(f, 0, None)
        f_sum = np.sum(f)
        f = f / f_sum * S
        
        score = local_score_fast(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
            if verbose and step % 20 == 0:
                print(f"  LP step {step}: score={score:.6f}, t={best_t:.3f}, elapsed={time.monotonic()-t0:.1f}s")
        elif verbose and step % 50 == 0:
            print(f"  LP step {step}: score={score:.6f}, best={best_score:.6f}, elapsed={time.monotonic()-t0:.1f}s")
    
    return best_f, best_score

def gradient_descent(f_init, time_limit, rng, verbose=True):
    """Active-constraint gradient descent."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f = f_init.astype(float).copy()
    f = np.clip(f, 0, None)
    f_sum = np.sum(f)
    if f_sum < 1e-10:
        f = np.ones(n)
        f_sum = n
    f = f / f_sum * S
    
    best_f = f.copy()
    best_score = local_score_fast(f)
    t0 = time.monotonic()
    lr = 0.01
    momentum = np.zeros(n)
    beta = 0.9
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fft_convolve(f, f)
        M = np.max(conv)
        peak_idx = int(np.argmax(conv))
        
        # Gradient of conv[f, f][peak_idx] w.r.t. f
        # grad[j] = 2 * f[peak_idx - j] if 0 <= peak_idx - j < n, else 0
        grad = np.zeros(n)
        for j in range(n):
            idx = peak_idx - j
            if 0 <= idx < n:
                grad[j] = 2.0 * f[idx]
        
        # Also include near-peak constraints
        threshold = M * 0.95
        near_peaks = np.where(conv >= threshold)[0]
        # Average gradient over near peaks
        if len(near_peaks) > 1:
            grad_total = np.zeros(n)
            for pk in near_peaks:
                for j in range(n):
                    idx = pk - j
                    if 0 <= idx < n:
                        grad_total[j] += 2.0 * f[idx]
            grad = grad_total / len(near_peaks)
        
        # Project gradient to maintain sum constraint
        grad_mean = np.mean(grad)
        grad_centered = grad - grad_mean
        
        # Adam-like update
        momentum = beta * momentum + (1-beta) * grad_centered
        f = f - lr * momentum
        f = np.clip(f, 0, None)
        
        # Renormalize
        f_sum = np.sum(f)
        if f_sum < 1e-10:
            f = best_f.copy()
            continue
        f = f / f_sum * S
        
        score = local_score_fast(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        # Adaptive learning rate
        if score > best_score * 1.01:
            lr *= 0.9
        elif step % 100 == 0:
            lr = min(lr * 1.05, 0.05)
        
        if verbose and step % 200 == 0:
            print(f"  GD step {step}: score={score:.6f}, best={best_score:.6f}, lr={lr:.5f}, elapsed={time.monotonic()-t0:.1f}s")
    
    return best_f, best_score

def sa_polish(f_init, time_limit, rng, verbose=True):
    """Simulated annealing polish."""
    n = len(f_init)
    f = f_init.astype(float).copy()
    best_f = f.copy()
    best_score = local_score_fast(f)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9999
    
    while time.monotonic() - t0 < time_limit:
        candidate = f.copy()
        r = rng.random()
        
        if r < 0.4:
            # Targeted peak reduction
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            # Find top contributors to the peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=top_k))
            # Boost others
            other = np.setdiff1d(np.arange(n), top_idx)
            n_inc = min(10, len(other))
            if n_inc > 0:
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.03, size=width))
        elif r < 0.75:
            # Symmetrize
            f_sym = 0.5 * (f + f[::-1])
            candidate = f_sym * np.exp(rng.normal(0, 0.01, size=n))
        else:
            # Global small perturbation
            candidate = f * np.exp(rng.normal(0, 0.01, size=n))
        
        candidate = np.clip(candidate, 0, 1000.0)
        new_score = local_score_fast(candidate)
        
        if new_score < current_score:
            f = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_f = f.copy()
                if verbose:
                    print(f"  SA: new best={best_score:.6f}, t_elapsed={time.monotonic()-t0:.1f}s")
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            f = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best_f, best_score

def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_woodbury_init(n, rng):
    """Initialize using Woodbury-like sequence structure."""
    t = np.arange(n, dtype=np.float64)
    # Use a quadratic phase (chirp) with some randomization
    a = np.exp(-0.5 * (t - n/2)**2 / (n/4)**2)  # Gaussian envelope
    a += 0.3 * np.cos(2*np.pi * 3 * t / n + rng.uniform(0, 2*np.pi))
    a += 0.2 * np.cos(2*np.pi * 7 * t / n + rng.uniform(0, 2*np.pi))
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

"""
Hybrid LP-guided descent and active-constraint gradient optimization for
minimizing the peak of self-convolution of non-negative step functions.
Uses linear programming to find optimal descent directions (SOTA-inspired),
combined with vectorized gradient descent on active/near-active constraints,
and simulated annealing polish. Multi-start over different initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score_fast(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_lp_matrix(f):
    """Build constraint matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A = build_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 5})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_gradient_at_peak(f, peak_idx):
    """Compute gradient of conv(f,f)[peak_idx] w.r.t. f (vectorized)."""
    n = len(f)
    grad = np.zeros(n)
    j_min = max(0, peak_idx - n + 1)
    j_max = min(peak_idx, n - 1)
    indices = np.arange(j_min, j_max + 1)
    grad[indices] = 2.0 * f[peak_idx - indices]
    return grad


def lp_descent(f_init, time_limit, rng, verbose=False):
    """LP-guided descent: use LP to find optimal direction at each step."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f = f_init.astype(np.float64).copy()
    f = np.clip(f, 0, None)
    f_sum = np.sum(f)
    if f_sum < 1e-10:
        f = np.ones(n)
        f_sum = n
    f = f / f_sum * S

    best_f = f.copy()
    best_score = local_score_fast(f)
    t0 = time.monotonic()
    stagnant = 0

    for step in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fft_convolve(f, f)
        M = np.max(conv)

        g = solve_lp_direction(f, M)
        if g is None or np.sum(g) < 1e-10:
            break

        g = np.clip(g, 0, None)
        g_sum = np.sum(g)
        g = g / g_sum * S

        # Try multiple step sizes and pick the best
        best_t = 0.05
        best_new_score = np.inf
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
            f_new = (1 - t) * f + t * g
            f_new = np.clip(f_new, 0, None)
            s = np.sum(f_new)
            if s < 1e-10:
                continue
            f_new = f_new / s * S
            score = local_score_fast(f_new)
            if score < best_new_score:
                best_new_score = score
                best_t = t

        f = (1 - best_t) * f + best_t * g
        f = np.clip(f, 0, None)
        s = np.sum(f)
        if s < 1e-10:
            break
        f = f / s * S

        score = local_score_fast(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
            stagnant = 0
            if verbose and step % 10 == 0:
                print(f"    LP step {step}: score={score:.6f}, t={best_t:.3f}, "
                      f"elapsed={time.monotonic()-t0:.1f}s")
        else:
            stagnant += 1
            if stagnant > 200:
                # Perturb to escape
                f = f * np.exp(rng.normal(0, 0.05, size=n))
                f = np.clip(f, 0, None)
                f = f / np.sum(f) * S
                stagnant = 0

    return best_f, best_score


def gradient_descent(f_init, time_limit, rng, verbose=False):
    """Active-constraint gradient descent with momentum."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f = f_init.astype(np.float64).copy()
    f = np.clip(f, 0, None)
    f_sum = np.sum(f)
    if f_sum < 1e-10:
        f = np.ones(n)
        f_sum = n
    f = f / f_sum * S

    best_f = f.copy()
    best_score = local_score_fast(f)
    t0 = time.monotonic()
    lr = 0.02
    momentum = np.zeros(n)
    beta = 0.85
    step = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fft_convolve(f, f)
        M = np.max(conv)

        # Find near-peak indices (within 5% of max)
        threshold = M * 0.95
        near_peaks = np.where(conv >= threshold)[0]
        # Limit to top 5 for efficiency
        if len(near_peaks) > 5:
            peak_vals = conv[near_peaks]
            top_idx = near_peaks[np.argsort(peak_vals)[-5:]]
        else:
            top_idx = near_peaks

        # Average gradient over near-peak constraints
        grad = np.zeros(n)
        for pk in top_idx:
            grad += compute_gradient_at_peak(f, pk)
        grad /= len(top_idx)

        # Center the gradient (to maintain sum constraint)
        grad = grad - np.mean(grad)

        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        f = f - lr * momentum
        f = np.clip(f, 0, None)

        f_sum = np.sum(f)
        if f_sum < 1e-10:
            f = best_f.copy()
            stagnant += 10
            continue
        f = f / f_sum * S

        score = local_score_fast(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Adaptive learning rate
        if stagnant > 50:
            lr *= 0.95
            if stagnant > 200:
                # Restart with perturbation
                f = best_f * np.exp(rng.normal(0, 0.03, size=n))
                f = np.clip(f, 0, None)
                f = f / np.sum(f) * S
                stagnant = 0
                lr = 0.02
        elif step % 100 == 0:
            lr = min(lr * 1.02, 0.1)

        if verbose and step % 200 == 0:
            print(f"    GD step {step}: score={score:.6f}, best={best_score:.6f}, "
                  f"lr={lr:.5f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_f, best_score


def sa_polish(f_init, time_limit, rng, verbose=False):
    """Simulated annealing with targeted peak-reduction moves."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = local_score_fast(f)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.0015
    cooling = 0.99995

    while time.monotonic() - t0 < time_limit:
        candidate = f.copy()
        r = rng.random()

        if r < 0.45:
            # Targeted: reduce top contributors to peak
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            j_min = max(0, peak_idx - n + 1)
            j_max = min(peak_idx, n - 1)
            for i in range(j_min, j_max + 1):
                contribs[i] = f[i] * f[peak_idx - i]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.06, 0.025, size=top_k))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(12, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.06, 0.025, size=n_inc))
        elif r < 0.65:
            # Block perturbation
            width = max(1, int(rng.integers(2, max(3, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.035, size=width))
        elif r < 0.8:
            # Multi-point
            n_pts = int(rng.integers(3, 8))
            pts = rng.choice(n, size=n_pts, replace=False)
            candidate[pts] *= np.exp(rng.normal(0, 0.04, size=n_pts))
        else:
            # Symmetrize + noise
            f_sym = 0.5 * (f + f[::-1])
            candidate = f_sym * np.exp(rng.normal(0, 0.01, size=n))

        candidate = np.clip(candidate, 0, 1000.0)
        new_score = local_score_fast(candidate)

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_f = f.copy()
                if verbose:
                    print(f"    SA: new best={best_score:.6f}, "
                          f"t={time.monotonic()-t0:.1f}s")
        elif temperature > 1e-10:
            accept_prob = np.exp(-(new_score - current_score) / temperature)
            if rng.random() < accept_prob:
                f = candidate
                current_score = new_score

        temperature *= cooling

    return best_f, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    freqs = [2, 5, 8, 13, 21]
    amps = [0.2, 0.15, 0.1, 0.07, 0.05]
    for freq, amp in zip(freqs, amps):
        a += amp * np.cos(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_gaussian_chirp_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    envelope = np.exp(-0.3 * (t - n / 2) ** 2 / (n / 2) ** 2)
    chirp = 1.0 + 0.4 * np.cos(2 * np.pi * 3 * t * t / n ** 2)
    a = envelope * chirp + 0.05
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
