```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft

def _fft_convolve(a, b):
    """Linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = fft(a, size)
    fb = fft(b, size)
    result = ifft(fa * fb).real[:n]
    return result

def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) for a sequence with sum=1."""
    n = len(a)
    b = _fft_convolve(a, a)
    return 2.0 * n * np.max(b), b

def _grad_bk(a, k):
    """Gradient of b_k = sum_i a_i*a_{k-i} w.r.t. a_j for all j."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
"""
FISTA-based optimizer with adaptive log-sum-exp smoothing to minimize
the normalized autocorrelation peak of a non-negative sequence.
Uses FFT for fast convolution and gradient computation.
"""

import numpy as np
import time


def _get_fft_size(n_linear):
    """Get next power of 2 >= n_linear."""
    return 1 << (n_linear - 1).bit_length()


def _compute_conv(a, fa, fft_size):
    """Compute linear convolution a*a using precomputed FFT."""
    b = np.fft.irfft(fa * fa, fft_size)[:2*len(a)-1]
    return b


def _compute_grad_from_weights(a, weights, fft_size_grad):
    """
    Compute grad[j] = 2 * sum_i a[i] * weights[i+j] for j=0..n-1.
    """
    n = len(a)
    w_rev = weights[::-1]
    fa = np.fft.rfft(a, fft_size_grad)
    fw = np.fft.rfft(w_rev, fft_size_grad)
    conv = np.fft.irfft(fa * fw, fft_size_grad)
    return 2.0 * conv[2*n-2 : n-2 : -1]


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _make_two_level(n, eps):
    """Two-level anti-symmetric sequence."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    return a / np.sum(a)


def _make_cosine(n, eps):
    """Smooth cosine anti-symmetric profile."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    return a / np.sum(a)


def _make_gaussian_taper(n, width=0.3):
    """Gaussian taper (high in center, low at edges)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / (width * n))**2)
    return a / np.sum(a)

def _polish_peaks(a, deadline, M=5):

    Polishing phase: directly target the top M peaks using weighted gradient.
    """

    a = np.asarray(a, dtype=np.float64).copy()

    if S < 1e-10:
        return a, np.inf
    a /= S
    
    best = a.copy()
    b = np.convolve(a, a)
    best_score = 2.0 * n * np.max(b)
    
    step = 0.1 / np.sqrt(n)
    no_improve = 0

        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            step *= 0.7
            if step < 1e-10:
                break
            no_improve = 0

        if M >= len(b):
            idx = np.arange(len(b))
            wts = np.ones(len(b)) / len(b)
        else:
            idx = np.argpartition(b, -M)[-M:]
            idx = idx[np.argsort(b[idx])[::-1]]
            b_sel = b[idx]
            wts = np.exp(-3.0 * (b_max - b_sel) / max(b_max, 1e-15))
            wts /= wts.sum()
        
        # Weighted gradient
        grad = np.zeros(n)
        for ii in range(len(idx)):
            w = wts[ii]
            if w < 1e-8:

            lo = max(0, k - n + 1)

                grad[j_idx] += 2.0 * w * a[k - j_idx]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        a = _project_simplex(a - step * grad / grad_norm, 1.0)
    
    return best, best_score


def _perturb(a, rng, scale=0.05):
    """Perturb with smooth noise."""
    n = len(a)
    freqs = rng.integers(1, max(2, n // 4), size=min(8, n // 2))
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)

        noise /= np.max(np.abs(noise))
    a_new = a + scale * noise
    return _project_simplex(a_new, 1.0)
```

```python
"""
Adaptive FISTA with log-sum-exp smoothing and peak polishing for minimizing
the normalized autocorrelation peak of a non-negative sequence.
Uses FFT-accelerated convolution and gradient computation.
"""

import numpy as np
import time


def _next_pow2(n):
    return 1 << (n - 1).bit_length()


def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _auto_corr(a, fft_size):
    """Compute autocorrelation using FFT."""
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:2 * len(a) - 1]
```

    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)

def _autocorr(a, fft_size):

    fa = np.fft.rfft(a, fft_size)

def _grad_from_weights(a, weights, fft_size):
    """
    Compute gradient vector where grad[j] = 2 * sum_{i=0}^{n-1} a[i] * weights[i+j].
