def project_simplex(v, S):
    n=v.size
    if n==0: return v
    w=np.sort(v)[::-1]
    cum=np.cumsum(w)
    idx=np.arange(1,n+1, dtype=float)
    theta=(cum - S)/idx
    cond=theta <= w
    if np.any(cond):
        rho=np.max(np.nonzero(cond)[0])+1
    else: rho=n
    theta=(cum[rho-1]-S)/rho
    return np.maximum(v-theta,0.0)
```

    n=len(a)

    A=fft(a,size)
    c=np.real(ifft(A*A))[:2*n-1]
    c=np.where(c>0,c,0.0) # tiny negatives
    return c

Standard projection formula:
```
def project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0: return v
    v = np.where(np.isfinite(v), v, 0.0)
    if S <= 0: return np.zeros(n)
    w = np.sort(v)[::-1]
    cum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta = (cum - S) / idx
    cond = theta <= w
    if np.any(cond):
        rho = int(np.max(np.nonzero(cond)[0]) + 1)
    else:
        rho = n
    theta = (cum[rho-1] - S) / rho
    proj = v - theta
    proj = np.maximum(proj, 0.0)
    # fix sum due rounding
    s = proj.sum()
    if s <= 0:
        proj = np.full(n, S/n)
    else:
        proj *= S / s
    return proj
```
This extra normalize ensures sum exactly.

        w[0] = alpha * p[0]
    if n>1:

    H = np.zeros(2*n-1, dtype=np.float64)

    G = irfft(Hf * af, size)[center:center+n]
    G = np.where(G > 0.0, G, 0.0)
    return G

```python
"""
Search for a non-negative sequence minimizing the evaluation objective.

The code combines:
  * runtime probing of evaluate_sequence to infer a weighted-max model;
  * mirror-descent / multiplicative weights on log-heights with annealed
    smoothing over active autocorrelation lags;
  * short L-BFGS-B refinements on a smoothed log-sum-exp surrogate;
  * a final small black-box hill-climb using the true evaluator directly;
  * multiple alpha candidates and an optional larger-length polish.

All helpers are top-level and avoid closures/lambdas.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# Globals used only by the top-level L-BFGS callback.
_OPT_N = 0
_OPT_S = 1.0
_OPT_ALPHA = 1.0
_OPT_BETA = 60.0


def next_pow2_int(x):
    if x <= 1:
        return 1
    return 1 << (int(x) - 1).bit_length()


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    if S <= 0.0:
        return np.zeros(n, dtype=np.float64)

    w = np.sort(v)[::-1]
    cum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cum - S) / idx
    cond = theta <= w
    if np.any(cond):
        rho = int(np.max(np.nonzero(cond)[0]) + 1)
    else:
        rho = n
    theta = (cum[rho - 1] - S) / rho
    proj = v - theta
    proj = np.maximum(proj, 0.0)
    s = proj.sum()
    if s <= 0.0:
        proj = np.full(n, S / n, dtype=np.float64)
    else:
        proj *= S / s
    return proj


def convolve_self(a):
    """Linear self-convolution via real FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    size = next_pow2_int(2 * n - 1)
    A = rfft(a, size)
    c = irfft(A * A, size)[: 2 * n - 1]
    c = np.where(c > 0.0, c, 0.0)
    return c


def make_values(c, alpha):
    """Distinct-lag weighted values: [alpha*c0, c1, ..., c_{n-1}]."""
    n = (c.size + 1) // 2
    v = np.zeros(n, dtype=np.float64)
    if n <= 0:
        return v
    v[0] = float(alpha) * float(c[0])
    if n > 1:
        v[1:] = c[1:n]
    return v


def hard_score(c, alpha):
    """Proxy hard score max(alpha*c0, max side-lag)."""
    n = (c.size + 1) // 2
    if n <= 0:
        return 0.0
    val = float(alpha) * float(c[0])
    if n > 1 and c[1:n].size > 0:
        val = max(val, float(np.max(c[1:n])))
    return float(val)


def softmax_weights(values, beta):
    """Softmax weights over values; non-negative."""
    values = np.asarray(values, dtype=np.float64)
    if values.size == 0:
        return values
    m = float(np.max(values))
    if m <= 1e-300:
        return np.zeros_like(values)
    e = np.exp(float(beta) * (values - m))
    s = e.sum()
    if s <= 0.0 or not np.isfinite(s):
        k = max(1, int(0.01 * values.size))
        idx = np.argsort(values)[::-1][:k]
        p = np.zeros_like(values)
        p[idx] = 1.0 / k
        return p
    return e / s


def grad_from_p(a, p, alpha):
    """
    Gradient of sum_k p_k q_k c_k, where q_0=alpha and q_k=1 for k>0.
    Uses FFT with a symmetric signed-lag kernel.
    """
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if p.size != n:
        p = np.resize(p, n)

    w = np.zeros(n, dtype=np.float64)
    if n > 0:
        w[0] = float(alpha) * p[0]
    if n > 1:
        w[1:] = p[1:]

    H = np.zeros(2 * n - 1, dtype=np.float64)
    center = n - 1
    H[center] = 2.0 * w[0]
    if n > 1:
        H[: n - 1] = w[1:][::-1]
        H[n:] = w[1:]

    size = next_pow2_int(3 * n - 2)
    G = irfft(rfft(H, size) * rfft(a, size), size)[center : center + n]
    G = np.where(np.isfinite(G), G, 0.0)
    G = np.where(G > 0.0, G, 0.0)
    return G


def smooth_direction(rng, n, order=4):
    """Random smooth zero-mean direction."""
    idx = np.arange(n, dtype=np.float64)
    u = np.zeros(n, dtype=np.float64)
    for _ in range(order):
        maxf = max(2, n // 8)
        freq = max(1, int(rng.integers(1, maxf + 1)))
        phase = 2.0 * np.pi * float(rng.random())
        amp = float(rng.normal(scale=1.0))
        u += amp * np.cos(2.0 * np.pi * freq * idx / float(n) + phase)
    u -= u.mean()
    std = u.std()
    if std <= 1e-15:
        u = rng.normal(size=n)
        u -= u.mean()
        std = u.std()
    if std > 1e-15:
        u /= std
    return u


def upsample_sequence(arr, n, S):
    """Interpolate/rescale a non-negative sequence to length n and sum S."""
    arr = np.asarray(arr, dtype=np.float64).reshape(-1)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == n:
        s = arr.sum()
        if s <= 0.0:
            return np.full(n, S / n, dtype=np.float64)
        return arr * (S / s)
    if arr.size >= 2:
        x_old = np.linspace(0.0, 1.0, arr.size)
        x_new = np.linspace(0.0, 1.0, n)
        out = np.interp(x_new, x_old, arr)
        out = np.maximum(out, 0.0)
        s = out.sum()
        if s > 0.0:
            return out * (S / s)
    return np.full(n, S / n, dtype=np.float64)


def safe_evaluate(a):
    """Wrapper around evaluate_sequence with basic safety."""
    arr = np.asarray(a, dtype=np.float64).reshape(-1)
    if arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if float(arr.sum()) < 0.01:
        arr = np.ones(256, dtype=np.float64)
    try:
        vals = [float(x) for x in arr]
        return float(evaluate_sequence(vals))
    except Exception:
        return np.inf


def fit_alpha():
    """
    Infer a relative self-peak weight alpha in a model
      score * sum^2 / (2n) ~= r * max(alpha*c0, max_side).
    Falls back to 1.0 if fitting is unsuccessful.
    """
    m = 512
    rng = np.random.default_rng(12345)

    seqs = []

    const = np.full(m, 1.0 / m, dtype=np.float64)
    seqs.append(const)

    spike = np.zeros(m, dtype=np.float64)
    spike[0] = 1.0
    seqs.append(spike)

    block = np.zeros(m, dtype=np.float64)
    k = m // 2
    block[:k] = 1.0 / k
    seqs.append(block)

    two = np.zeros(m, dtype=np.float64)
    two[0] = 0.5
    two[-1] = 0.5
    seqs.append(two)

    rad = rng.random(m, dtype=np.float64)
    rad = rad / (rad.sum() + 1e-300)
    seqs.append(rad)

    t = np.linspace(0.0, 1.0, m)
    sm = 1.0 + 0.6 * np.sin(2.0 * np.pi * t)
    sm = np.maximum(sm, 1e-6)
    sm = sm / sm.sum()
    seqs.append(sm)

    alt = 1.0 + 0.7 * np.sin(np.pi * np.arange(m, dtype=np.float64) / m)
    alt = np.maximum(alt, 1e-6)
    alt = alt / alt.sum()
    seqs.append(alt)

    ys = []
    c0s = []
    sides = []

    for s in seqs:
        sc = safe_evaluate(s)
        if not np.isfinite(sc):
            continue
        A = float(np.sum(s))
        if A <= 0.0:
            continue
        c = convolve_self(s)
        y = sc * A * A / (2.0 * m)
        c0 = float(c[0]) if c.size > 0 else 0.0
        side = float(np.max(c[1:m])) if m > 1 and c.size >= m else 0.0
        if c0 <= 0.0 and side <= 0.0:
            continue
        ys.append(y)
        c0s.append(c0)
        sides.append(side)

    if len(ys) < 3:
        print("fit_alpha: insufficient probe data, fallback alpha=1.0")
        return 1.0

    ys = np.asarray(ys, dtype=np.float64)
    c0s = np.asarray(c0s, dtype=np.float64)
    sides = np.asarray(sides, dtype=np.float64)

    grid = np.linspace(0.0, 4.0, 201)
    best_err = 1e100
    best_alpha = 1.0
    best_r = 1.0

    for alpha in grid:
        M = np.maximum(float(alpha) * c0s, sides)
        mask = M > 1e-30
        if not np.any(mask):
            continue
        Mm = M[mask]
        ym = ys[mask]
        den = float(np.dot(Mm, Mm))
        if den <= 1e-300:
            continue
        r = float(np.dot(ym, Mm) * np.dot(Mm, Mm) / (den * den)) if False else float(np.dot(ym, Mm)) / den
        if r < 0.0 or not np.isfinite(r):
            r = 0.0
        pred = r * M
        denom = float(np.dot(ys, ys))
        if denom <= 1e-300:
            continue
        err = float(np.dot(pred - ys, pred - ys)) / denom
        if err < best_err:
            best_err = err
            best_alpha = float(alpha)
            best_r = float(r)

    print("fit_alpha: alpha=%.5f r=%.5f rel_err=%e" % (best_alpha, best_r, best_err))
    if best_err > 50.0:
        print("fit_alpha: poor fit, fallback alpha=1.0")
        return 1.0
    return max(0.0, best_alpha)


def alpha_candidates(alpha_fit):
    """Produce a few alpha values to optimize, with priority weights."""
    base = [
        float(max(0.0, alpha_fit)),
        1.0,
        0.75,
        0.5,
        0.25,
        0.1,
        2.0,
        3.0,
    ]
    if alpha_fit > 0.0:
        base.append(0.5 * alpha_fit)
        base.append(2.0 * alpha_fit)
    if alpha_fit < 0.1:
        base.append(0.0)
        base.append(0.01)
        base.append(0.05)

    cleaned = []
    for x in base:
        if not np.isfinite(x):
            continue
        if x < 0.0:
            x = 0.0
        if x > 4.0:
            x = 4.0
        cleaned.append(float(x))

    unique = []
    for x in cleaned:
        ok = True
        for y in unique:
            if abs(x - y) < 0.05:
                ok = False
                break
        if ok:
            unique.append(x)

    key = float(np.log1p(max(1e-6, alpha_fit)))
    scored = []
    for x in unique:
        d = abs(float(np.log1p(x)) - key)
        scored.append((d, x))
    scored.sort()

    cands = []
    weights = []
    for d, x in scored[:4]:
        cands.append(float(x))
        weights.append(1.0 / (1.0 + 5.0 * d))
    return cands, weights


def make_start_sequence(n, start_raw, rng):
    """Create an initial non-negative sequence of length n."""
    S = float(np.sqrt(2.0 * float(n)))

    if start_raw is not None:
        arr = np.asarray(start_raw, dtype=np.float64).reshape(-1)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        if arr.size > 0:
            s = arr.sum()
            if s > 0.0:
                return upsample_sequence(arr, n, S)

    t = np.linspace(0.0, 1.0, n)
    arr = (
        1.0
        + 0.18 * np.sin(2.0 * np.pi * 3.0 * t)
        + 0.08 * np.cos(2.0 * np.pi * 7.0 * t + 0.7)
        + 0.04 * rng.standard_normal(n)
    )
    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if s <= 0.0:
        return np.full(n, S / n, dtype=np.float64)
    return arr * (S / s)


def mirror_main(a, alpha, S, max_time, rng):
    """
    Multiplicative / mirror descent in log-space.

    Updates log a_j -= eta * centered_subgradient_j and renormalizes.
    The active-lag weights are annealed from dense to peak-focused.
    """
    a = project_simplex(a, S)
    n = a.size
    c = convolve_self(a)
    T = hard_score(c, alpha)
    best = a.copy()
    best_T = T

    if max_time <= 0.0:
        return best, best_T

    beta = 3.0
    eta = 0.7
    min_eta = 1e-4
    no_imp = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < max_time:
        it += 1
        if it % 2000 == 0:
            print(
                "mirror alpha=%.4f it=%d T=%.6f beta=%.2f eta=%.4f no_imp=%d"
                % (alpha, it, T, beta, eta, no_imp)
            )

        if T < best_T - 1e-12:
            best_T = T
            best = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        values = make_values(c, alpha)
        vmax = float(np.max(values)) if values.size else 0.0
        if vmax <= 1e-14:
            break

        p = softmax_weights(values, beta)
        G = grad_from_p(a, p, alpha)
        G = np.maximum(G, 0.0)
        gmax = float(np.max(G)) if G.size else 0.0
        if gmax <= 1e-14:
            break

        g = G / gmax
        g = g - g.mean()
        gscale = float(np.max(np.abs(g)))
        if gscale <= 1e-14:
            u = smooth_direction(rng, n, 4)
            a = a * np.exp(0.2 * u)
            ss = a.sum()
            if ss > 0.0:
                a *= S / ss
            else:
                a = best.copy()
            c = convolve_self(a)
            T = hard_score(c, alpha)
            beta = min(80.0, beta * 1.05 + 0.01)
            continue
        g /= gscale

        accepted = False
        for factor in (1.0, 0.5):
            step = eta * factor
            if step <= 0.0:
                continue
            loga = np.log(np.maximum(a, 1e-300))
            new_log = loga - step * g
            new_log -= np.max(new_log)
            new_log = np.maximum(new_log, -700.0)
            new_log -= np.max(new_log)
            na = np.exp(new_log)
            s = na.sum()
            if s <= 0.0:
                continue
            na *= S / s
            nc = convolve_self(na)
            nT = hard_score(nc, alpha)
            if np.isfinite(nT) and nT <= T + 1e-11:
                a = na
                c = nc
                T = nT
                accepted = True
                break

        if accepted:
            if T <= best_T + 1e-12:
                no_imp = 0
            eta = min(1.8, eta * 1.04)
            beta = min(100.0, beta * 1.012 + 0.01)
        else:
            eta = max(min_eta, eta * 0.5)
            no_imp += 1

            if no_imp > 600:
                beta = max(1.0, beta * 0.7)

            if no_imp > 1000:
                if rng.random() < 0.35:
                    u = smooth_direction(rng, n, 4)
                    a_try = a * np.exp(0.12 * u)
                    s_try = a_try.sum()
                    if s_try > 0.0:
                        a_try *= S / s_try
                        t_try = hard_score(convolve_self(a_try), alpha)
                        if t_try < T:
                            a = a_try
                            c = convolve_self(a)
                            T = t_try

            if no_imp > 5000:
                a = best.copy()
                c = convolve_self(a)
                T = best_T
                no_imp = 0
                beta = max(2.0, beta * 0.5)
                eta = 0.7

    return best, best_T


def _lbfgs_objgrad(x):
    """Top-level callback for L-BFGS-B on softmax log-parameters."""
    x = np.asarray(x, dtype=np.float64)
    n = int(_OPT_N)
    if x.size != n:
        x = np.resize(x, n)

    z = np.clip(x, -50.0, 50.0)
    z -= z.max()
    ez = np.exp(z)
    s = ez.sum()
    if s <= 0.0 or not np.isfinite(s):
        return 0.0, np.zeros(n, dtype=np.float64)

    a = ez / s * float(_OPT_S)
    c = convolve_self(a)
    v = make_values(c, float(_OPT_ALPHA))
    beta = float(_OPT_BETA)

    m = float(np.max(v)) if v.size else 0.0
    if m <= 1e-30:
        return 0.0, np.zeros(n, dtype=np.float64)

    e = np.exp(beta * (v - m))
    es = e.sum()
    if es <= 0.0 or not np.isfinite(es):
        k = max(1, int(0.01 * v.size))
        idx = np.argsort(v)[::-1][:k]
        p = np.zeros_like(v)
        p[idx] = 1.0 / k
    else:
        p = e / es

    phi = (m + float(np.log(es))) / beta
    if not np.isfinite(phi):
        phi = m / beta

    G = grad_from_p(a, p, float(_OPT_ALPHA))
    mu = float(np.dot(a, G))
    g = a * (G - mu)
    g = np.where(np.isfinite(g), g, 0.0)
    return float(phi), g


def lbfgs_refine(a, alpha, S, max_time, beta):
    """Refine a sequence using L-BFGS-B on a smoothed log-sum-exp surrogate."""
    a = project_simplex(a, S)
    n = a.size
    T0 = hard_score(convolve_self(a), alpha)
    best = a.copy()
    best_T = T0

    if n < 16 or max_time < 5.0:
        return best, best_T

    global _OPT_N, _OPT_S, _OPT_ALPHA, _OPT_BETA
    _OPT_N = n
    _OPT_S = float(S)
    _OPT_ALPHA = float(alpha)
    _OPT_BETA = float(max(10.0, beta))

    rel = a / (a.sum() + 1e-300)
    rel = np.maximum(rel, 1e-300)
    z0 = np.log(rel)
    z0 -= z0.mean()

    if n <= 10000:
        maxfun = 500
    else:
        maxfun = 180

    try:
        res = minimize(
            _lbfgs_objgrad,
            z0,
            jac=True,
            method="L-BFGS-B",
            options={
                "maxiter": max(20, maxfun // 2),
                "maxfun": maxfun,
                "maxcor": 10,
                "maxls": 8,
                "ftol": 1e-10,
                "gtol": 1e-8,
                "disp": 0,
            },
        )
        z = np.asarray(res.x, dtype=np.float64)
        z = np.clip(z, -50.0, 50.0)
        z -= z.max()
        ez = np.exp(z)
        s = ez.sum()
        if s > 0.0:
            na = ez / s * S
            na = project_simplex(na, S)
            nT = hard_score(convolve_self(na), alpha)
            if np.isfinite(nT) and nT < best_T:
                best = na
                best_T = nT
    except Exception:
        pass

    return best, best_T


def black_box_refine(a, score, max_time, rng):
    """
    Small direct hill-climb using the true evaluate_sequence.
    This helps if the inferred proxy model is imperfect.
    """
    a = project_simplex(a, float(np.sum(a)))
    S = float(np.sum(a))
    n = a.size
    if not np.isfinite(score):
        score = safe_evaluate(a)
    best = a.copy()
    best_score = score
    if not np.isfinite(best_score):
        best_score = 1e300

    amp = 0.3
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < max_time:
        it += 1
        if it % 50 == 0:
            a = best.copy()

        if it % 25 == 0:
            u = rng.normal(size=n)
            if np.max(np.abs(u)) > 0:
                thr = 0.45 * float(np.max(np.abs(u)))
                u[np.abs(u) < thr] = 0.0
                u -= u.mean()
                st = u.std()
                if st > 1e-15:
                    u /= st
                else:
                    u = smooth_direction(rng, n, 3)
        else:
            u = smooth_direction(rng, n, 3)

        a_try = a * np.exp(amp * u)
        s_try = a_try.sum()
        if s_try <= 0.0:
            amp = max(0.02, amp * 0.7)
            continue
        a_try *= S / s_try

        sc = safe_evaluate(a_try)
        if np.isfinite(sc) and sc < best_score:
            best = a_try
            best_score = sc
            if sc < score:
                score = sc
                amp = min(1.0, amp * 1.05)
            else:
                amp = max(0.05, amp * 0.9)
        else:
            amp = max(0.02, amp * 0.8)

    return best, best_score


def propose_candidate(seed=42, budget_s=2380):
    """Main entry point: search for a good non-negative sequence."""
    t_start = time.monotonic()
    budget = float(budget_s)
    deadline = t_start + max(120.0, budget - 20.0)
    rng = np.random.default_rng(seed)

    # Determine working length from the provided construction if reasonable.
    start_raw = None
    try:
        start_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
    except Exception:
        start_raw = None

    n = 8192
    if start_raw is not None:
        L = start_raw.size
        if 12288 <= L <= 16384:
            n = 16384
        elif 4096 <= L <= 8192:
            n = L
        elif L > 16384:
            n = 16384

    S = float(np.sqrt(2.0 * float(n)))
    a_start = make_start_sequence(n, start_raw, rng)

    # Try to infer objective model.
    alpha_fit = fit_alpha()
    if not np.isfinite(alpha_fit):
        alpha_fit = 1.0

    alphas, weights = alpha_candidates(alpha_fit)
    total_w = float(sum(weights))
    if total_w <= 0.0:
        alphas = [alpha_fit]
        weights = [1.0]
        total_w = 1.0

    # Time allocation: exploration with proxy, then actual-score polish.
    if n >= 12000:
        main_frac = 0.62
    else:
        main_frac = 0.72
    main_time = max(60.0, min(deadline - time.monotonic(), budget * main_frac))

    candidates = []

    for alpha, wgt in zip(alphas, weights):
        left = deadline - time.monotonic()
        alloc = main_time * (float(wgt) / total_w)
        alloc = min(alloc, max(0.0, left - 30.0))
        if alloc < 10.0:
            break

        print("optimizing alpha=%.5f allocated=%.1fs" % (alpha, alloc))
        a, proxy_T = mirror_main(a_start, alpha, S, alloc * 0.90, rng)
        sc = safe_evaluate(a)
        if np.isfinite(sc):
            candidates.append((sc, a.copy(), float(alpha), float(proxy_T)))
            print("  actual=%.8f proxy=%.6f" % (sc, proxy_T))
        else:
            candidates.append((inf, a.copy(), float(alpha), float(proxy_T)))

    # If no valid actual score, keep the start.
    if not candidates:
        candidates.append((1e300, a_start.copy(), float(alpha_fit), 0.0))

    # Select best by true evaluator when possible.
    finite = [cand for cand in candidates if np.isfinite(cand[0])]
    if finite:
        best = min(finite, key=lambda x: x[0])
    else:
        best = min(candidates, key=lambda x: x[3])
    best_score, best_a, best_alpha, _ = best
    print("current best actual=%.8f alpha=%.4f" % (best_score, best_alpha))

    # L-BFGS polish the best and perhaps the second-best actual candidate.
    left = deadline - time.monotonic()
    if left > 90.0 and np.isfinite(best_score):
        finite.sort(key=lambda x: x[0])
        for cand in finite[:2]:
            left = deadline - time.monotonic()
            if left < 40.0:
                break
            sc_c, a_c, al_c, _ = cand
            t_alloc = min(max(25.0, left * 0.25), 180.0)
            beta_lb = 60.0
            a_p = lbfgs_refine(a_c, al_c, S, t_alloc, beta_lb)
            sc_p = safe_evaluate(a_p)
            if np.isfinite(sc_p) and sc_p < best_score:
                best = (sc_p, a_p.copy(), float(al_c), float(hard_score(convolve_self(a_p), al_c)))
                best_score, best_a, best_alpha, _ = best
                print("lbfgs improved to %.8f alpha=%.4f" % (best_score, best_alpha))

    # Optional larger-length polish if time remains.
    left = deadline - time.monotonic()
    if left > 260.0 and n < 16384 and np.isfinite(best_score):
        n2 = min(16384, 2 * n)
        S2 = float(np.sqrt(2.0 * float(n2)))
        a2 = upsample_sequence(best_a, n2, S2)
        alloc2 = min(max(80.0, left * 0.35), 300.0)
        a2, proxy_T2 = mirror_main(a2, best_alpha, S2, alloc2, rng)
        sc2 = safe_evaluate(a2)
        if np.isfinite(sc2) and sc2 < best_score:
            best = (sc2, a2.copy(), float(best_alpha), float(proxy_T2))
            best_score, best_a, best_alpha, _ = best
            n = n2
            S = S2
            print("larger length improved to %.8f n=%d" % (best_score, n))

    # Final black-box refinement using the real evaluator.
    left = deadline - time.monotonic()
    if left > 30.0:
        if n >= 12000:
            bb_time = min(max(15.0, left * 0.25), 160.0)
        else:
            bb_time = min(max(20.0, left * 0.35), 300.0)
        a_bb, sc_bb = black_box_refine(best_a, best_score, bb_time, rng)
        if np.isfinite(sc_bb) and sc_bb < best_score:
            best = (sc_bb, a_bb.copy(), float(best_alpha), hard_score(convolve_self(a_bb), best_alpha))
            best_score, best_a, best_alpha, _ = best
            print("black-box improved to %.8f" % best_score)

    # Final safety normalization and positivity.
    final = np.asarray(best_a, dtype=np.float64).reshape(-1)
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.maximum(final, 0.0)
    code_sum = float(np.sum(np.clip(final, 0.0, 1000.0)))
    if code_sum < 0.01:
        final = np.ones(256, dtype=np.float64)

    # A tiny floor avoids exactly zero heights without materially changing score.
    final = np.maximum(final, 1e-12)

    # Keep safely below evaluator caps.
    final = np.clip(final, 0.0, 1000.0)

    # Verify one last time and fall back if something went wrong.
    final_score = safe_evaluate(final)
    if not np.isfinite(final_score):
        final = np.ones(8192, dtype=np.float64)
        final_score = safe_evaluate(final)

    print("final n=%d score=%s" % (final.size, final_score))
    return [float(x) for x in final]
