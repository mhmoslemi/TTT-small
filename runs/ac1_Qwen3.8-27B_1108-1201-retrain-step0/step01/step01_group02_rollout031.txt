Hybrid optimization for minimizing peak auto-convolution of non-negative sequences.
Combines spectral flattening (Gerchberg-Saxton), LP refinement, subgradient descent,
and local search across multiple sequence lengths and initializations.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_uniform(n):
    return np.ones(n, dtype=np.float64)


def initial_two_bump(n, sep_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    c1 = sep_frac * n
    c2 = (1.0 - sep_frac) * n
    w = max(1.0, n * 0.05)
    a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    return a


def spectral_flatten(a, n_iters=1000, relax=0.5, time_limit=30.0):
    """Gerchberg-Saxton style spectral flattening."""
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    best_a = a.copy()
    best_score = score_sequence(a)
    
    for it in range(n_iters):
        if time.monotonic() - start > time_limit:
            break
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        # Target: flat magnitude
        target = np.full_like(mag, mag.mean())
        # Avoid division by zero
        mask = mag > 1e-15
        A_new = A.copy()
        A_new[mask] = A[mask] * target[mask] / mag[mask]
        # Relaxation
        A_relax = (1 - relax) * A + relax * A_new
        a_new = np.fft.irfft(A_relax, N)[:n]
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
        sc = score_sequence(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def line_search_mix(a, g, max_alpha=0.8):
    """Find best mixing parameter for a + alpha*(g - a)."""
    best_alpha = 0.0
    best_score = score_sequence(a)
    alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, max_alpha]
    for alpha in alphas:
        candidate = (1 - alpha) * a + alpha * g
        candidate = normalize_nonneg(candidate)
        sc = score_sequence(candidate)
        if sc < best_score:
            best_score = sc
            best_alpha = alpha
    return best_alpha, best_score


def lp_refine(a, time_limit_s=300.0, max_iters=200, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 3):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 30:
            if verbose:
                print(f"    LP stagnant at {it}, best={best_score:.6f}")
            break
        
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.25))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g
        
        # Line search
        alpha, new_score = line_search_mix(a, g)
        if alpha > 0:
            a = normalize_nonneg((1 - alpha) * a + alpha * g)
        else:
            # Try smaller alpha
            a = normalize_nonneg(0.95 * a + 0.05 * g)
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def subgradient_descent(a, time_limit_s=120.0, verbose=True):
    """Subgradient descent on the score function."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    eta = 0.01  # initial step size
    
    for it in range(10000):
        if time.monotonic() - start_time > time_limit_s:
            break
        
        b = np.convolve(a, a)
        S = a.sum()
        k_star = np.argmax(b)
        score = 2.0 * n * b[k_star] / (S ** 2)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            if verbose and it % 100 == 0:
                print(f"    SG iter {it:4d}: score={score:.6f}")
        
        # Subgradient
        # df/da[i] = (2n/S^2) * (db[k*]/da[i] - 2*b[k*]/S)
        # db[k*]/da[i] = 2*a[k*-i] if valid and k*!=2i, or a[i] if k*=2i
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if j == i:
                    db_da = a[i]
                else:
                    db_da = 2.0 * a[j]
            else:
                # Check if a[i] appears as the second factor
                # b[k*] = sum_j a[j]*a[k*-j], so a[i] appears as a[k*-j] when k*-j=i, j=k*-i
                # which is the same condition: 0 <= k*-i < n
                db_da = 0.0
            
            grad[i] = (2.0 * n / (S ** 2)) * (db_da - 2.0 * b[k_star] / S)
        
        # Projected subgradient step
        a_new = a - eta * grad
        a_new = np.maximum(0, a_new)
        a_new = normalize_nonneg(a_new)
        
        # Adaptive step size
        new_score = score_sequence(a_new)
        if new_score < score:
            a = a_new
            eta = min(eta * 1.05, 0.1)
        else:
            eta = eta * 0.7
            if eta < 1e-6:
                break
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    SG done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=120.0, rng=None, verbose=True):
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(rng.integers(n // 300, n // 5)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        
        # Random perturbation strategy
        strategy = rng.random()
        if strategy < 0.4:
            # Multiplicative perturbation
            candidate[start:stop] *= np.exp(rng.normal(0.0, 0.05, size=stop - start))
        elif strategy < 0.7:
            # Add small noise
            candidate[start:stop] += rng.normal(0.0, 0.01 * a[start:stop].mean(), size=stop - start)
        else:
            # Swap-like: redistribute energy
            total = candidate[start:stop].sum()
            candidate[start:stop] = rng.dirichlet(np.ones(stop - start) * 0.5) * total
        
        candidate = np.maximum(0, candidate)
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
    
    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

Hybrid optimization for minimizing peak auto-convolution ratio of non-negative sequences.
Combines spectral flattening (Gerchberg-Saxton), LP-based refinement with line search,
vectorized subgradient descent, and diverse local search across multiple lengths.

import numpy as np

from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_uniform(n):
    return np.ones(n, dtype=np.float64)


def initial_two_bump(n, sep_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    c1 = sep_frac * n
    c2 = (1.0 - sep_frac) * n
    w = max(1.0, n * 0.05)
    return np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)


def initial_multi_bump(n, n_bumps=3, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 15), max(2, n // 4)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    return np.maximum(0, a)


def spectral_flatten(a, n_iters=3000, relax=0.5, time_limit=40.0):
    """Gerchberg-Saxton style: alternate between flat-spectrum constraint and non-negativity."""
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    start = time.monotonic()
    best_a = a.copy()
    best_score = score_sequence(a)
    
    for it in range(n_iters):
        if time.monotonic() - start > time_limit:
            break
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        mean_mag = mag.mean()
        mask = mag > 1e-15
        A_new = A.copy()
        A_new[mask] = A[mask] * mean_mag / mag[mask]
        A_relax = (1.0 - relax) * A + relax * A_new
        a_new = np.fft.irfft(A_relax, N)[:n]
        a_new = np.maximum(0.0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
        sc = score_sequence(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def spectral_flatten_with_targets(a, n_iters=2000, time_limit=40.0):
    """Spectral flattening with adaptive target (reduce peaks more aggressively)."""
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    start = time.monotonic()
    best_a = a.copy()
    best_score = score_sequence(a)
    
    for it in range(n_iters):
        if time.monotonic() - start > time_limit:
            break
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        # Use a target that emphasizes reducing the largest bins
        # Power-law flattening: target[k] = mean * (mean/mag[k])^alpha
        alpha = 0.3 + 0.3 * (it / max(1, n_iters))  # anneal
        mean_mag = mag.mean()
        mask = mag > 1e-15
        A_new = A.copy()
        ratio = mean_mag / mag[mask]
        A_new[mask] = A[mask] * ratio ** alpha
        A_relax = 0.5 * A + 0.5 * A_new
        a_new = np.fft.irfft(A_relax, N)[:n]
        a_new = np.maximum(0.0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
        sc = score_sequence(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def build_conv_matrix_sparse(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_parts = [np.arange(j, j + n, dtype=np.int32) for j in range(n)]
    row_idx = np.concatenate(row_parts)
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_basic(a, M, timeout=20.0):
    """Standard LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    n_rows = 2 * n - 1
    A_mat = build_conv_matrix_sparse(a)
    c = -np.ones(n)
    b_vec = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_vec, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def solve_lp_asymmetric(a, b_conv, timeout=20.0):
    """Asymmetric LP: tighter bounds where b is large, looser where small."""
    n = len(a)
    n_rows = 2 * n - 1
    A_mat = build_conv_matrix_sparse(a)
    c = -np.ones(n)
    # Create asymmetric bounds: tighter where b is above mean
    mean_b = b_conv.mean()
    max_b = b_conv.max()
    # Scale: where b[k] > mean, use tighter bound
    b_vec = np.where(b_conv > mean_b, 
                     mean_b + 0.3 * (b_conv - mean_b),
                     max_b)
    b_vec = np.maximum(b_vec, max_b * 0.5)  # don't go too tight
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_vec, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def line_search_mix(a, g, n_trials=12):
    """Find optimal mixing parameter."""
    best_alpha = 0.0
    best_score = score_sequence(a)
    alphas = np.concatenate([np.linspace(0.01, 0.3, 6), np.linspace(0.4, 0.9, 6)])
    for alpha in alphas:
        candidate = normalize_nonneg((1.0 - alpha) * a + alpha * g)
        sc = score_sequence(candidate)
        if sc < best_score:
            best_score = sc
            best_alpha = alpha
    return best_alpha, best_score

def local_search(a, time_limit_s=120.0, rng=None, verbose=True):

            # Multiplicative perturbation
            sigma = 0.02 + 0.03 * t_anneal
            candidate[pos:end] *= np.exp(rng.normal(0.0, sigma, size=end - pos))
        elif strat < 0.5:

            w = max(2, (end - pos) // 3)
            center = pos + (end - pos) / 2.0
            t_arr = np.arange(n, dtype=np.float64)
            amplitude = rng.normal(0.0, 0.003)
            candidate += amplitude * np.exp(-0.5 * ((t_arr - center) / w) ** 2)
        elif strat < 0.7:
            # Reshape segment (power transform)
            seg = candidate[pos:end].copy()
            total = seg.sum()
            if total > 1e-15:
                alpha = rng.lognormal(0.0, 0.5)
                alpha = np.clip(alpha, 0.3, 3.0)
                new_seg = np.maximum(0.0, seg - seg.min()) ** alpha + 1e-10

                candidate[pos:end] = new_seg * total
        elif strat < 0.85:

            sigma_global = 0.005 + 0.005 * t_anneal
            candidate *= np.exp(rng.normal(0.0, sigma_global, size=n))
        else:

            i1 = pos
            i2 = min(end - 1, pos + rng.integers(1, max(2, end - pos)))

            a_flat, sc = spectral_flatten_with_targets(a, n_iters=2000, time_limit=20.0)
        
        # Also try a second spectral pass on the result
        if sc < best_score or sc < init_sc - 0.01:
            a_flat2, sc2 = spectral_flatten(a_flat, n_iters=1000, relax=0.3, time_limit=10.0)
            if sc2 < sc:
                a_flat, sc = a_flat2, sc2
        
        label = f"(spectral, n={n}, {init_type}, init={init_sc:.4f})"
        print(f"  {label} -> {sc:.6f}")
        
        if sc < best_score:
            best_score = sc
            best_a = a_flat.copy()
            print(f"  *** NEW BEST: {best_score:.6f}")
    
    # Phase 2: LP refinement on best candidates
    print("\n=== Phase 2: LP Refinement ===")
    lp_candidates = []
    if best_a is not None:
        lp_candidates.append(best_a.copy())
    
    # Add a few diverse LP starting points
    lp_configs = [
        (500, 'raised_cosine'),
        (500, 'uniform'),
        (1000, 'raised_cosine'),
        (1000, 'uniform'),
        (1000, 'gaussian'),
        (2000, 'uniform'),
    ]
    for n, init_type in lp_configs:
        if init_type == 'uniform':
            a = initial_uniform(n)
        elif init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif init_type == 'gaussian':
            a = initial_gaussian(n, 0.25)
        else:
            a = initial_uniform(n)
        lp_candidates.append(normalize_nonneg(a))
    
    for idx, a_cand in enumerate(lp_candidates):
        remaining = deadline - time.monotonic()
        if remaining < 40.0:
            break
        
        n = len(a_cand)
        time_limit = min(remaining * 0.25, max(30.0, 200.0 * n / 2000.0))
        use_asym = (idx % 2 == 1)
        
        print(f"  LP candidate {idx}: n={n}, init_score={score_sequence(a_cand):.6f}")
        a_ref, sc = lp_refine(a_cand, time_limit_s=time_limit, verbose=False,
                              use_asymmetric=use_asym)
        print(f"    -> {sc:.6f}")
        
        if sc < best_score:
            best_score = sc
            best_a = a_ref.copy()
            print(f"  *** NEW BEST: {best_score:.6f}")
    
    # Phase 3: Subgradient descent on best
    print("\n=== Phase 3: Subgradient Descent ===")
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_sg, sc = subgrad_descent(best_a, time_limit_s=min(remaining * 0.25, 60.0), verbose=False)
            print(f"  SG: {sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_a = a_sg.copy()
                print(f"  *** NEW BEST: {best_score:.6f}")
    
    # Phase 4: Local search
    print("\n=== Phase 4: Local Search ===")
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_ls, sc = local_search(best_a, time_limit_s=min(remaining * 0.3, 120.0),
                                    rng=rng, verbose=True)
            print(f"  LS: {sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_a = a_ls.copy()
                print(f"  *** NEW BEST: {best_score:.6f}")
    
    # Phase 5: Final LP polish
    print("\n=== Phase 5: Final LP Polish ===")
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_final, sc = lp_refine(best_a, time_limit_s=min(remaining * 0.5, 180.0),
                                    verbose=True, use_asymmetric=True)
            print(f"  Final LP: {sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_a = a_final.copy()
                print(f"  *** NEW BEST: {best_score:.6f}")
    
    # Phase 6: Final local search
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_final2, sc = local_search(best_a, time_limit_s=min(remaining * 0.7, 60.0),
                                        rng=np.random.default_rng(seed + 1), verbose=False)
            if sc < best_score:
                best_score = sc
                best_a = a_final2.copy()
                print(f"  *** FINAL LS: {best_score:.6f}")
    
    # Phase 7: One more quick LP if time allows
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_final3, sc = lp_refine(best_a, time_limit_s=min(remaining * 0.8, 60.0),
                                     verbose=False, use_asymmetric=False)
            if sc < best_score:
                best_score = sc
                best_a = a_final3.copy()
                print(f"  *** FINAL LP2: {best_score:.6f}")
    
    if best_a is None:
        best_a = initial_raised_cosine(500)
        best_score = score_sequence(best_a)
    
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_a)} ===")
    verify = evaluate_sequence([float(x) for x in best_a])
    print(f"Verified via evaluate_sequence: {verify:.6f}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")
    
    return [float(x) for x in best_a]
