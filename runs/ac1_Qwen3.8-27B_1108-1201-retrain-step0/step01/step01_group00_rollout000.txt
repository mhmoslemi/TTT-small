Hybrid p-norm L-BFGS optimization with peak-targeted local search for minimizing
the autocorrelation peak of non-negative step functions. Uses homotopy in the p-norm
exponent, multiple starting points, and a final polish phase with direct peak reduction.

import numpy as np
from scipy.optimize import minimize
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sa = np.sum(a)
    if sa < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sa * sa)


def softmax_fn(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def pnorm_obj_and_grad(theta, n, p):
    """
    Compute the p-norm of conv(a,a) and its gradient w.r.t. theta,
    where a = softmax(theta) * sqrt(2n).
    The score with this normalization equals max(conv(a,a)), and the
    p-norm approximates it from below.
    """
    S = np.sqrt(2.0 * n)
    a = softmax_fn(theta) * S
    c = fft_convolve(a, a)
    
    c_p = np.sum(c ** p)
    val = c_p ** (1.0 / p)
    
    # Gradient of (sum c^k^p)^(1/p) w.r.t. a
    w = c ** (p - 1)
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g_a = 2.0 * corr[n - 1: 2 * n - 1]
    g_a *= c_p ** (1.0 / p - 1.0)
    
    # Chain rule through softmax * S
    inner = np.dot(a, g_a)
    g_theta = a * (g_a - inner / S)
    
    return val, g_theta


def lbfgs_pnorm(theta_init, n, p_schedule):
    """
    Run L-BFGS with homotopy in p: start with small p (smooth) and
    increase to large p (approaches true max).
    """
    theta = theta_init.copy()
    best_theta = theta.copy()
    best_val = np.inf
    
    for p in p_schedule:
        # Set up the objective/gradient for this p
        def obj_and_grad(t):
            return pnorm_obj_and_grad(t, n, p)
        
        def obj_only(t):
            return pnorm_obj_and_grad(t, n, p)[0]
        
        res = minimize(
            obj_and_grad,
            theta,
            jac=True,
            method='L-BFGS-B',
            options={
                'maxiter': 500,
                'maxfun': 2000,
                'gtol': 1e-10,
                'ftol': 1e-12,
                'maxls': 50,
            }
        )
        theta = res.x
        val = obj_only(theta)
        if val < best_val:
            best_val = val
            best_theta = theta.copy()
        print(f"    p={p}: val={val:.8f}, converged={res.success}")
    
    return best_theta, best_val

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if int((time.monotonic() - t0) % 10) == 0:
                    print(f"    peak_red: score={best_score:.8f}, t={time.monotonic()-t0:.0f}s")
        elif rng.random() < np.exp(-(new_score - current_score) / max(0.001, 0.005 * np.exp(-0.001 * (time.monotonic() - t0)))):
            current = candidate
            current_score = new_score
    
    return best, best_score


def block_local_search(a_init, n, time_limit, rng):
    """
    Block-based local search: perturb small contiguous blocks to find
    local improvements.
    """
    current = a_init.astype(np.float64).copy()
    S = np.sqrt(2.0 * n)
    current = current / np.sum(current) * S
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    temp = 0.002
    
    while time.monotonic() - t0 < time_limit:
        r = rng.random()
        candidate = current.copy()
        
        if r < 0.4:
            # Single block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, width))
        elif r < 0.7:
            # Two-point swap-like perturbation
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            delta = rng.normal(0, 0.02)
            candidate[i1] *= np.exp(delta)
            candidate[i2] *= np.exp(-delta)
        elif r < 0.85:
            # Peak-targeted
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            # Perturb around the peak position
            center = peak_idx // 2 if peak_idx < n else n - 1 - (2*n-2-peak_idx)//2
            radius = max(2, n // 32)
            lo = max(0, center - radius)
            hi = min(n, center + radius + 1)
            candidate[lo:hi] *= np.exp(rng.normal(0, 0.02, hi - lo))
        else:
            # Symmetry exploitation: try (current + reversed) / 2 perturbation
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.01, n))
        
        candidate = np.clip(candidate, 1e-10, 1000.0)
        candidate = candidate / np.sum(candidate) * S
        new_score = compute_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-10)):
            current = candidate
            current_score = new_score
        
        temp *= 0.99995
    
    return best, best_score


def lp_based_improvement(a_init, n, time_limit, rng):
    """
    LP-based approach inspired by the state of the art:
    find g that maximizes sum(g) subject to conv(a, g) <= M,
    then blend.
    """
    from scipy.optimize import linprog
    
    current = a_init.astype(np.float64).copy()
    S = np.sqrt(2.0 * n)
    current = current / np.sum(current) * S
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    
    for iteration in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fft_convolve(current, current)
        M = np.max(c)
        
        # Build LP: maximize sum(g) subject to:
        # (a * g)[k] <= M for all k (where a*g is the cross-correlation)
        # g[j] >= 0
        # Note: (a*g)[k] = sum_i a[i]*g[k-i] for the cross-convolution
        
        # The cross-convolution a*g has length 2n-1
        # (a*g)[k] = sum_{i} a[i] * g[k-i] where 0<=i<n, 0<=k-i<n
        
        # c_matrix: c_matrix[k, j] = coefficient of g[j] in (a*g)[k]
        # (a*g)[k] = sum_j a[k-j] * g[j] (with a[m]=0 for m outside [0,n-1])
        # So c_matrix[k, j] = a[k-j] if 0 <= k-j < n, else 0
        
        # This is a Toeplitz-like matrix
        # Building it explicitly for n up to ~512 is feasible
        
        if n > 512:
            # Too large for LP, skip
            break
        
        c_mat = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            for j in range(n):
                m = k - j
                if 0 <= m < n:
                    c_mat[k, j] = current[m]
        
        # LP: maximize -sum(g) (scipy minimizes)
        # Subject to -c_mat @ g <= -M (i.e., c_mat @ g >= M ... wait)
        # We want c_mat @ g <= M, which is c_mat @ g <= M*ones
        
        c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
        A_ub = c_mat  # c_mat @ g <= M
        b_ub = np.full(2 * n - 1, M)
        
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                continue
        except Exception:
            continue
        
        g = res.x
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            continue
        
        # Rescale g to match normalization
        g_norm = g / sum_g * S
        
        # Blend
        t_blend = 0.1
        candidate = (1 - t_blend) * current + t_blend * g_norm
        candidate = np.clip(candidate, 1e-10, 1000.0)
        candidate = candidate / np.sum(candidate) * S
        
        new_score = compute_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                print(f"    LP iter {iteration}: score={best_score:.8f}")
        else:
            # Try smaller blend
            for t_try in [0.05, 0.02, 0.01]:
                candidate2 = (1 - t_try) * current + t_try * g_norm
                candidate2 = np.clip(candidate2, 1e-10, 1000.0)
                candidate2 = candidate2 / np.sum(candidate2) * S
                s2 = compute_score(candidate2)
                if s2 < current_score:
                    current = candidate2
                    current_score = s2
                    if current_score < best_score:
                        best_score = current_score
                        best = current.copy()
                    break
    
    return best, best_score


def make_barker_like(n, rng):
    """Create a Barker-like initial sequence (±1 pattern mapped to non-negative)."""
    # Barker sequences have minimum aperiodic autocorrelation
    # Map to [0,1] range
    seq = rng.choice([0.0, 1.0], size=n)
    # Smooth it out a bit
    kernel = np.ones(3) / 3.0
    seq = np.convolve(seq, kernel, mode='same')
    seq = 0.1 + 0.8 * seq / np.max(seq)
    return seq


def make_random_init(n, rng, spread=0.5):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_chirp_init(n, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    alpha = 0.3 + 0.4 * rng.random()
    a = 1.0 + alpha * np.cos(np.pi * t * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    n_tones = rng.integers(2, 6)
    for _ in range(n_tones):
        k = int(rng.integers(1, n // 3))
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining p-norm L-BFGS, LP-based improvement,
    and peak-targeted local search.
    """
    t_global = time.monotonic()
    deadline = t_global + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Load provided sequence if available
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = compute_score(provided)
        print(f"Provided: n={len(provided)}, score={s:.8f}")
        if s < best_score:
            best_score = s
            best_seq = provided.copy()
    
    # Phase 1: L-BFGS with p-norm homotopy on multiple start points
    print("\n=== Phase 1: L-BFGS p-norm optimization ===")
    
    # Primary: optimize the provided sequence
    if provided is not None:
        n = len(provided)
        # Try a few sequence lengths
        lengths = [n]
        if n <= 300:
            lengths.extend([n * 2, int(n * 1.5)])
        elif n <= 600:
            lengths.extend([n * 2, int(n * 1.5), n + 128])
        
        for n in lengths:
            if time.monotonic() > deadline - 120:
                break
            
            inits = []
            # From provided (resampled if needed)
            if provided is not None:
                if n == len(provided):
                    inits.append(provided.copy())
                else:
                    x_old = np.linspace(0, 1, len(provided))
                    x_new = np.linspace(0, 1, n)
                    inits.append(np.interp(x_new, x_old, provided))
            # Random inits
            inits.append(make_chirp_init(n, rng))
            inits.append(make_multitone_init(n, rng))
            inits.append(make_random_init(n, rng, spread=0.4))
            
            if n <= 384:
                inits.append(make_barker_like(n, rng))
            
            time_per_init = min(60.0, max(15.0, (deadline - time.monotonic() - 60) / len(inits)))
            
            for i, a0 in enumerate(inits):
                if time.monotonic() > deadline - 90:
                    break
                
                a0 = np.clip(a0, 0.01, 1000.0)
                theta_init = np.log(a0 / np.sum(a0) + 1e-10)
                
                # P-schedule: start smooth, end sharp
                if n <= 256:
                    p_sched = [2, 4, 8, 16, 32, 64, 128]
                elif n <= 512:
                    p_sched = [2, 4, 8, 16, 32, 64]
                else:
                    p_sched = [2, 4, 8, 16, 32]
                
                print(f"  n={n}, init {i}/{len(inits)}, t_remaining={deadline-time.monotonic():.0f}s")
                
                t0 = time.monotonic()
                theta_best, pnorm_val = lbfgs_pnorm(theta_init, n, p_sched)
                elapsed = time.monotonic() - t0
                
                a_best = softmax_fn(theta_best) * np.sqrt(2.0 * n)
                actual_score = compute_score(a_best)
                print(f"    pnorm={pnorm_val:.8f}, actual={actual_score:.8f}, time={elapsed:.1f}s")
                
                if actual_score < best_score:
                    best_score = actual_score
                    best_seq = a_best.copy()
                    print(f"    *** New best: {best_score:.8f} ***")
    
    # Phase 2: LP-based improvement
    print(f"\n=== Phase 2: LP-based improvement (score={best_score:.8f}) ===")
    if best_seq is not None:
        n = len(best_seq)
        lp_time = min(180.0, max(30.0, (deadline - time.monotonic()) * 0.25))

                best_score = lp_score
                best_seq = lp_result.copy()
                print(f"  After LP: {best_score:.8f}")
    
    # Phase 3: Peak reduction search
    print(f"\n=== Phase 3: Peak reduction (score={best_score:.8f}) ===")

        n = len(best_seq)
