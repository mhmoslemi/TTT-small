```python
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:

    unconstrained over z.
  * Use a log-sum-exp (LSE) smooth approximation of the max-convolution
    objective with a decreasing temperature schedule.
  * Optimize z with Adam + periodic random restarts.
  * Multiple initializations (random, smooth cosine, previous best) and
    multiple sequence lengths for global search.
  * Final polish with very low temperature and small learning rate.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    A = fft(a, size)
    B = fft(b, size)
    return np.real(ifft(A * B))[:n]


def compute_self_conv(a):
    """Self-convolution a * a, length 2n-1."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = fft(a, size)
    return np.real(ifft(A * A))[:2 * n - 1]


def lse_gradient_weights(c, beta):
    """Compute softmax weights w_k = exp(c_k/beta) / sum(exp(c_k/beta))."""
    c_max = c.max()
    exp_c = np.exp((c - c_max) / beta)
    w = exp_c / exp_c.sum()
    return w

    """
    Compute G_m = sum_k w_k * dc_k/da_m.
    
    G_m = 2 * (w * a_rev)[m + n - 1] - w[2m] * a_m
    """
    a_rev = a[::-1]
    # Convolution of w (length 2n-1) with a_rev (length n), result length 3n-2
    conv = fft_convolve(w, a_rev)
    # G_m = 2 * conv[m + n - 1] for m = 0..n-1
    G = 2.0 * conv[n - 1:2 * n - 1]
    # Correction: subtract w[2m] * a_m
    k_diag = np.arange(n) * 2
    valid = k_diag < 2 * n - 1
    G[valid] -= w[k_diag[valid]] * a[valid]
    return G


def softmax(z):
    """Numerically stable softmax."""
    z_shift = z - z.max()
    e = np.exp(z_shift)
    return e / e.sum()


def gradient_through_softmax(G, p):
    """
    Compute dF/dz where a = S * softmax(z), p = a/S = softmax(z).
    dF/dz_m = S * p_m * (G_m - sum_j p_j * G_j)
    """
    mean_G = np.dot(p, G)
    return p * (G - mean_G)


def run_lse_adam(n, z_init, time_budget, rng, beta_init=1.0, beta_final=0.02,
                 lr_init=1.0, patience=5000):
    """
    Optimize softmax parameters z minimizing LSE approximation to max self-conv.
    """
    z = z_init.copy().astype(np.float64)
    S = np.sqrt(2.0 * n)
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    best_score = np.inf
    best_p = None
    no_improve = 0
    start = time.monotonic()
    
    total_iters = estimate_iters(time_budget, n)
    
    while time.monotonic() - start < time_budget:
        t += 1
        
        # Anneal beta
        frac = min(1.0, t / total_iters)
        beta = beta_init * (beta_final / beta_init) ** frac
        
        # Current point
        p = softmax(z)
        a = S * p
        c = compute_self_conv(a)
        cur_score = c.max()
        
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute LSE objective and gradient
        if beta > 0.01:
            w = lse_gradient_weights(c, beta)
        else:
            # Use top-1 (hard max) subgradient
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0
        
        G = compute_gradient(a, w, n)
        grad_z = gradient_through_softmax(G, p)
        
        # Adam update
        lr = lr_init * (1.0 / (1.0 + 0.001 * t)) ** 0.5
        m_hat = beta1 * m_hat + (1 - beta1) * grad_z
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_z ** 2)
        m_u = m_hat / (1 - beta1 ** t)
        v_u = v_hat / (1 - beta2 ** t)
        z = z - lr * m_u / (np.sqrt(v_u) + eps)
        
        # Random restart if stuck
        if no_improve > patience and rng.random() < 0.1:
            z = z + rng.normal(0, 0.5, n)
            no_improve = 0
        
        if t % 50000 == 0:
            print(f"  iter={t}, beta={beta:.4f}, score={cur_score:.6f}, best={best_score:.6f}")
    
    return best_p, best_score


def estimate_iters(budget, n):
    """Rough estimate of how many iterations we can do."""
    # Each iteration involves 2-3 FFTs of size ~2n or 3n
    # For n=8192, each FFT pair takes ~0.1ms
    # So ~10000 iters per second is a rough guess
    # Be conservative: 2000 per second
    return int(budget * 2000)


def make_initialization(n, init_type, rng):
    """Create initial z (log-probability) vector."""
    if init_type == 'uniform':
        return np.zeros(n)
    elif init_type == 'random':
        return rng.normal(0, 1.0, n)
    elif init_type == 'smooth':
        t = np.linspace(0, 1, n, endpoint=False)
        z = (2.0 * np.sin(2 * np.pi * rng.integers(1, 8) * t) +
             1.0 * np.cos(2 * np.pi * rng.integers(2, 12) * t + rng.uniform(0, 2*np.pi)))
        return z
    elif init_type == 'triangle':
        t = np.linspace(0, 1, n, endpoint=False)
        return 2.0 * np.abs(t - 0.5) - 0.5  # V-shape
    elif init_type == 'spike':
        z = np.zeros(n)
        nz = n // 4
        idx = rng.choice(n, nz, replace=False)
        z[idx] = 2.0
        return z
    else:
        return np.zeros(n)
```

```python
"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
  * Parameterize a = S * softmax(z) with S = sqrt(2n) so optimization is
    unconstrained over z (no projection needed).
  * Use log-sum-exp (LSE) smooth approximation of the max self-convolution
    objective, with an annealing schedule from smooth (large beta) to sharp
    (small beta, approaching the true max subgradient).
  * Adam optimizer on z; gradient computed via two FFT-based convolutions
    per iteration (self-conv + cross-correlation for the LSE gradient).
  * Multiple initializations (previous best, random, cosine, uniform) at
    several sequence lengths for global search.
  * Polish phase: take the globally best candidate, refine with very low
    beta and small learning rate.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def _fft_convolve(a, b):
    """Linear convolution via FFT. Returns array of length len(a)+len(b)-1."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    A = fft(a, size)
    B = fft(b, size)
    return np.real(ifft(A * B))[:n]


def _self_conv(a):
    """Self-convolution a*a, length 2n-1."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = fft(a, size)
    return np.real(ifft(A * A))[:2 * n - 1]


def _lse_weights(c, beta):
    """Softmax weights w_k = exp(c_k/beta) / sum(exp). Numerically stable."""
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / beta)
    return exp_c / exp_c.sum()


def _compute_G(a, w, n, fft_size_g):
    """
    Compute G_m = 2 * sum_u w_{u+m} * a_u  for m=0..n-1.
    This equals 2 * xcorr(w, a)[m], computed via convolution of w with reversed a.
    G = 2 * (w * a_rev)[n-1 : 2n-1]
    """
    a_rev = a[::-1]
    # w has length 2n-1, a_rev has length n, conv has length 3n-2
    conv = _fft_convolve(w, a_rev)
    # Extract the n elements starting at index n-1
    G = 2.0 * conv[n - 1:2 * n - 1]
    return G


def _softmax(z):
    """Stable softmax."""
    z_shift = z - z.max()
    e = np.exp(z_shift)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(z), 1.0 / len(z))
    return e / s


def _grad_z(G, p):
    """
    Gradient of F(a) w.r.t. z where a = S*softmax(z).
    dF/dz_m = S * p_m * (G_m - p.G)
    We omit the constant factor S (absorbed into learning rate).
    """
    pG = np.dot(p, G)
    return p * (G - pG)


def _run_lse_adam(n, z_init, time_budget, rng, beta_start, beta_end, lr_init,
                  patience, verbose_every):
    """
    Core optimizer: minimize LSE approx to max self-conv via softmax param + Adam.
    Returns (best_p, best_max_c) where best_p is the softmax probability vector.
    """
    z = z_init.astype(np.float64).copy()
    m_hat = np.zeros(n, dtype=np.float64)
    v_hat = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    best_score = np.inf
    best_p = _softmax(z)
    no_improve = 0

    start = time.monotonic()
    total_est = int(time_budget * 3000)  # rough iter estimate

    while time.monotonic() - start < time_budget:
        t += 1
        # Anneal beta (exponential)
        frac = min(1.0, t / max(1, total_est))
        beta = beta_start * (beta_end / beta_start) ** frac

        p = _softmax(z)
        S = np.sqrt(2.0 * n)
        a = S * p
        c = _self_conv(a)
        cur = c.max()

        if cur < best_score:
            best_score = cur
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute LSE gradient
        if beta > 0.005:
            w = _lse_weights(c, beta)
        else:
            # Approximate with top-3 lags for speed
            w = np.zeros(2 * n - 1, dtype=np.float64)
            top3 = np.argsort(c)[::-1][:3]
            w[top3] = 1.0 / len(top3)

        G = _compute_G(a, w, n, 0)
        grad_z = _grad_z(G, p)

        # Adam step
        lr = lr_init * (1.0 + 0.0005 * t) ** (-0.5)
        m_hat = beta1 * m_hat + (1.0 - beta1) * grad_z
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_z ** 2)
        m_u = m_hat / (1.0 - beta1 ** t)
        v_u = v_hat / (1.0 - beta2 ** t)
        z = z - lr * m_u / (np.sqrt(v_u) + eps)

        # Random restart if stuck
        if no_improve > patience:
            if rng.random() < 0.15:
                z = z + rng.normal(0.0, 0.8, n)
                no_improve = 0

        if verbose_every > 0 and t % verbose_every == 0:
            elapsed = time.monotonic() - start
            print(f"    t={t}, beta={beta:.4f}, cur={cur:.6f}, best={best_score:.6f}, "
                  f"elapsed={elapsed:.0f}s/{time_budget:.0f}s")

    return best_p, best_score


def _make_init(n, init_type, rng):
    """Generate initial z vector."""
    if init_type == 'uniform':
        return np.zeros(n, dtype=np.float64)
    elif init_type == 'random':
        return rng.normal(0.0, 1.0, n).astype(np.float64)
    elif init_type == 'cosine':
        t = np.linspace(0.0, 1.0, n, endpoint=False)
        k1 = rng.integers(1, 10)
        k2 = rng.integers(2, 15)
        phi = rng.uniform(0, 2 * np.pi)
        return (2.0 * np.sin(2 * np.pi * k1 * t) +
                1.0 * np.cos(2 * np.pi * k2 * t + phi)).astype(np.float64)
    elif init_type == 'bimodal':
        t = np.linspace(0.0, 1.0, n, endpoint=False)
        sigma = rng.uniform(0.05, 0.15)
        center = rng.uniform(0.1, 0.9)
        return ((-(t - center) ** 2) / (2 * sigma * sigma) -
                (-(t + 1 - center) ** 2) / (2 * sigma * sigma)).astype(np.float64)
    elif init_type == 'linear':
        t = np.linspace(0.0, 1.0, n, endpoint=False)
        sign = 1.0 if rng.random() < 0.5 else -1.0
        return (sign * (t - 0.5) * 4.0).astype(np.float64)
    else:
        return np.zeros(n, dtype=np.float64)
```
