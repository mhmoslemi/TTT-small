The issue is that for non-negative sequences, the convolution tends to have a peak in the middle (by some form of concentration). The question is how flat we can make it.

Let me think about the problem in terms of the Fourier transform. If we zero-pad `a` to length `N >= 2n-1`, then:
`DFT_N(conv(a,a)) = |DFT_N(a)|^2`

The peak of `conv(a,a)` is at index `k*` where the inverse DFT has its maximum. The inverse DFT of `|A|^2` (where `A = DFT_N(a)`) is:
`(1/N) * sum_k |A[k]|^2 * exp(2πi*k*m/N)` for each output position `m`.

We want this to be as flat as possible. This is related to making `|A[k]|^2` have a flat inverse DFT, which means `|A[k]|^2` should be close to constant (i.e., `a` should have a flat power spectrum).

But `a` is non-negative and of length `n` (with zeros outside). So `A[k] = sum_{j=0}^{n-1} a[j] * exp(-2πi*k*j/N)`.

If `a` is constant (= 1/n after normalization), then `|A[k]|^2` is a sinc^2 function, which is peaked at k=0. This gives the triangular convolution.

To flatten the convolution, we want `|A[k]|^2` to be flatter. But the constraint that `a >= 0` and `a` has support [0, n-1] limits what we can do.

        b_ub[k] = M - c[k]

    # d >= -a  =>  -d <= a
    A_ub[num_c:, :n] = -np.eye(n)
    b_ub[num_c:] = a

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    t0 = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq, method='highs',
                      options={'time_limit': max_time})
        if time.monotonic() - t0 > max_time + 1:
            return None, 0.0
        if res.status == 0:
            d = res.x[:n]
            t_val = float(res.x[n])
            if np.all(d >= -a - 1e-6):
                return d, t_val
    except Exception:
        pass
    return None, 0.0


def _lp_direction_simple(a, M_target, max_time=5.0):
    """
    Simpler LP: find d that reduces peak below M_target.
    max sum(d) s.t. 2*conv(a,d)[k] <= M_target - c[k], d >= -a, sum(d)=0
    """
    n = len(a)
    c = np.convolve(a, a)

    c_obj = -np.ones(n)  # maximize sum(d)

    num_c = 2 * n - 1
    A_ub = np.zeros((num_c + n, n))
    b_ub = np.zeros(num_c + n)

    for k in range(num_c):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[k, j] = 2.0 * a[k - j]
        b_ub[k] = M_target - c[k]

    A_ub[num_c:, :] = -np.eye(n)
    b_ub[num_c:] = a

    A_eq = np.array([[1.0] * n])
    b_eq = np.zeros(1)

    t0 = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq, method='highs',
                      options={'time_limit': max_time})
        if res.status == 0:
            d = res.x
            if np.all(d >= -a - 1e-6):
                return d
    except Exception:
        pass
    return None


def _optimize_single(a0, deadline, use_lp=True, lp_every=300, max_iters=10000):
    """
    Optimize a single sequence using hybrid gradient + LP.
    Returns (best_sequence, best_obj).
    """
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    a = _normalize(a)

    c = np.convolve(a, a)
    M = float(c.max())
    best = a.copy()
    best_M = M

    lr = 2.0 / n
    k = 5
    tau = 0.0
    patience = 0
    max_patience = 400

    for it in range(max_iters):
        if time.monotonic() > deadline - 20.0:
            break

        # Adaptive parameters
        progress = it / max_iters
        tau = 10.0 * progress  # gradually increase softmax sharpness
        k_eff = int(5 + 15 * progress)  # increase top-k over time

        # Gradient step
        grad = _subgradient(a, k=k_eff, tau=tau)
        grad = grad - grad.mean()

        # Adaptive step size
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new)

        c_new = np.convolve(a_new, a_new)
        M_new = float(c_new.max())

        if M_new < M:
            a = a_new
            M = M_new
            patience = 0
            if M < best_M:
                best_M = M
                best = a.copy()
        else:
            patience += 1
            if patience % 100 == 0:
                lr *= 0.85
                if lr < 1e-6:
                    break
            if patience > max_patience:
                # Try a random perturbation to escape
                rng = np.random.default_rng(it)
                pert = rng.normal(0, 0.1, n)
                a_pert = a + pert
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = _normalize(a_pert)
                c_pert = np.convolve(a_pert, a_pert)
                M_pert = float(c_pert.max())
                if M_pert < M:
                    a = a_pert
                    M = M_pert
                patience = 0
                lr = max(lr, 1.0 / n)

        # LP refinement periodically
        if use_lp and n <= 1024 and (it + 1) % lp_every == 0:
            d, t_val = _lp_direction(a, max_time=3.0)
            if d is not None and t_val > 1e-8:
                # Try different step sizes
                for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
                    a_lp = a + alpha * d
                    a_lp = np.maximum(a_lp, 0.0)
                    a_lp = _normalize(a_lp)
                    c_lp = np.convolve(a_lp, a_lp)
                    M_lp = float(c_lp.max())
                    if M_lp < M:
                        a = a_lp
                        M = M_lp
                        if M < best_M:
                            best_M = M
                            best = a.copy()
                        patience = 0
                        break

    return best, best_M


def _optimize_spectral(a0, deadline, max_iters=5000):
    """
    Optimize using spectral domain projection.
    Idea: in frequency domain, conv(a,a) has DFT = |DFT(a_padded)|^2.
    We project the power spectrum to reduce temporal peak.
    """
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    a = _normalize(a)

    # Zero-pad to length >= 2n-1
    N = 1
    while N < 2 * n - 1:
        N *= 2

    best = a.copy()
    c = np.convolve(a, a)
    best_M = float(c.max())

    eta = 0.5  # projection strength

    for it in range(max_iters):
        if time.monotonic() > deadline - 20.0:
            break

        # FFT of zero-padded sequence
        A = np.fft.fft(a, N)
        P = np.abs(A) ** 2  # power spectrum

        # Current convolution (via IFFT of P)
        c_current = np.fft.ifft(P).real[:2 * n - 1]
        M_current = float(c_current.max())

        if M_current < best_M:
            best_M = M_current
            best = a.copy()

        # Identify temporal peaks
        peak_threshold = M_current * 0.9
        peak_mask = (c_current > peak_threshold).astype(np.float64)

        # Project power spectrum: reduce power at frequencies that contribute to peaks
        # Use gradient ascent on the "flatness" of the temporal convolution
        # Simple approach: reduce P where its IFFT contribution to peaks is high
        grad_P = np.zeros_like(P)
        for k_idx in range(len(peak_mask)):
            if peak_mask[k_idx] > 0:
                # Contribution of frequency f to temporal position k_idx
                # d c[k_idx] / d P[f] = exp(2*pi*i*f*k_idx/N) / N
                grad_P += np.exp(2j * np.pi * np.arange(N) * k_idx / N) / N

        # Project P in direction of reducing peak
        # P_new = max(0, P - eta * Re(grad_P) * M_current)
        P_new = P - eta * np.abs(grad_P) * M_current
        P_new = np.maximum(P_new, 0)

        # Ensure Hermitian symmetry
        P_new[(N // 2) + 1:] = P_new[1:(N // 2)][::-1].copy() if N % 2 == 0 else P_new[1:N // 2 + 1][::-1].copy()
        # Actually, for real signal, P should be symmetric: P[k] = P[N-k]
        for k_idx in range(1, N // 2 + 1):
            P_new[N - k_idx] = P_new[k_idx]

        # Compute magnitude and phase
        mag_new = np.sqrt(P_new)
        phase = np.angle(A)

        # Inverse FFT
        A_new = mag_new * np.exp(1j * phase)
        a_new_full = np.fft.ifft(A_new).real
        a_new = np.maximum(a_new_full[:n], 0.0)
        a_new = _normalize(a_new)

        c_new = np.convolve(a_new, a_new)
        M_new = float(c_new.max())

        if M_new < M_current:
            a = a_new
        else:
            # Gradient step fallback
            eta *= 0.9
            if eta < 0.01:
                break

    return best, best_M


def _generate_starts(rng, n):
    """Generate diverse starting sequences of length n."""
    starts = []
    i = np.arange(n, dtype=np.float64)

    # Uniform
    starts.append(np.ones(n) / n)

    # Random
    starts.append(rng.random(n) / (n * rng.random(n).mean()))

    # Chirp-like
    for freq in [1, 2, 3, 5, 7]:
        starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * freq / n))

    # Edge-concentrated
    starts.append(np.exp(-4.0 * np.abs(i - n / 2.0) / n))
    starts.append(np.exp(-4.0 * np.minimum(i, n - 1 - i) / n))

    # Two humps
    starts.append(np.exp(-10.0 * ((i - n * 0.25) / n) ** 2) + np.exp(-10.0 * ((i - n * 0.75) / n) ** 2))

    # Periodic pattern
    starts.append((1 + np.sign(np.sin(2 * np.pi * i * 3 / n))) * 0.5 + 0.1)

    # Triangular
    starts.append(1.0 - 2.0 * np.abs(i - n / 2.0) / (n / 2.0))

    # Inverse triangular
    starts.append(2.0 * np.abs(i - n / 2.0) / (n / 2.0) + 0.1)

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Explores multiple starting sequences of various lengths
    using hybrid gradient+LP optimization and spectral methods.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def _update_best(seq, score):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  NEW BEST: obj={best_score:.6f}, n={len(seq)}, time={time.monotonic()-t_start:.1f}s")

    # Try to use previous solution as a starting point
    prev_starts = []
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)) and np.all(gs >= 0):
            S = gs.sum()
            if S > 0.01:
                for n_try in [min(256, len(gs)), min(512, len(gs)), min(1024, len(gs))]:
                    prev_starts.append(gs[:n_try].copy())
                print(f"Using previous solution of length {len(gs)} as starting point.")
    except (NameError, Exception):
        pass

    # Define sequence lengths to try
    lengths = [256, 512, 1024]

    # Collect all candidates
    candidates = []  # (sequence, priority)

    # Previous solutions get highest priority
    for ps in prev_starts:
        candidates.append((ps, 0))

    # Generate starts for each length
    for n in lengths:
        starts = _generate_starts(rng, n)
        for s in starts:
            s = np.maximum(s, 0)
            if s.sum() > 0:
                candidates.append((s, 1))

    # Additional random starts
    for n in lengths:
        for _ in range(3):
            s = rng.random(n)
            s = np.maximum(s, 0)
            if s.sum() > 0:
                candidates.append((s, 2))

    # Sort by priority, then by length (shorter first for quick wins)
    candidates.sort(key=lambda x: (x[1], len(x[0])))

    print(f"Starting optimization: {len(candidates)} candidates, budget={budget_s}s")

    # Phase 1: Quick scan of all candidates (short optimization)
    quick_deadline = t_start + (deadline - t_start) * 0.3
    for idx, (a0, priority) in enumerate(candidates):
        if time.monotonic() > quick_deadline:
            break
        if len(a0) > 1024 and time.monotonic() > t_start + (deadline - t_start) * 0.1:
            continue

        use_lp = len(a0) <= 512
        seq, score = _optimize_single(a0, quick_deadline,
                                       use_lp=use_lp, lp_every=200,
                                       max_iters=2000)
        # Convert score (which is max(conv) with sum=sqrt(2n)) to actual objective
        actual_obj = _compute_obj(seq)
        _update_best(seq, actual_obj)

    print(f"After Phase 1: best={best_score:.6f}, time={time.monotonic()-t_start:.1f}s")

    # Phase 2: Focused optimization on best candidates
    if best_seq is not None and time.monotonic() < deadline - 60.0:
        # Refine the best sequence with longer optimization
        n_best = len(best_seq)
        focus_deadline = t_start + (deadline - t_start) * 0.8

        # Try multiple LP-heavy optimizations on the best
        for attempt in range(5):
            if time.monotonic() > focus_deadline:
                break
            a0 = best_seq.copy()
            # Add small random perturbation for diversity
            if attempt > 0:
                pert = rng.normal(0, 0.05, n_best)
                a0 = np.maximum(a0 + pert, 0)
                a0 = _normalize(a0)

            use_lp = n_best <= 1024
            seq, score = _optimize_single(a0, focus_deadline,
                                           use_lp=use_lp, lp_every=100,
                                           max_iters=5000)
            actual_obj = _compute_obj(seq)
            _update_best(seq, actual_obj)

    # Phase 3: Try spectral optimization on the best
    if best_seq is not None and time.monotonic() < deadline - 60.0:
        n_best = len(best_seq)
        if n_best <= 512:
            spectral_deadline = min(deadline - 10, t_start + (deadline - t_start) * 0.9)
            seq, score = _optimize_spectral(best_seq, spectral_deadline, max_iters=2000)
            actual_obj = _compute_obj(seq)
            _update_best(seq, actual_obj)

    # Phase 4: Final LP polish on the best
    if best_seq is not None and time.monotonic() < deadline - 15.0:
        n_best = len(best_seq)
        if n_best <= 512:
            for _ in range(20):
                if time.monotonic() > deadline - 10.0:
                    break
                d, t_val = _lp_direction(best_seq, max_time=2.0)
                if d is not None and t_val > 1e-6:
                    for alpha in [0.5, 0.3, 0.1, 0.05]:
                        a_new = best_seq + alpha * d
                        a_new = np.maximum(a_new, 0.0)
                        a_new = _normalize(a_new)
                        obj_new = _compute_obj(a_new)
                        if obj_new < best_score:
                            best_seq = a_new
                            best_score = obj_new
                            print(f"  LP polish: obj={best_score:.6f}")
                            break
                else:
                    break

    # Final validation
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"Final: n={len(best_seq)}, obj={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
