    if reverse:
        p = p[::-1]
    return safe_normalize(p)


def make_power_law(n, a):
    n = max(2, int(n))
    j = np.arange(1, n + 1, dtype=np.float64)
    p = j ** (-float(a))
    return safe_normalize(p)


def make_waterfall(n, steep, tail):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-float(steep) * x) + float(tail)
    return safe_normalize(p)


def make_two_bump(n, mu1, s1, w1, mu2, s2, w2):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    p = w1 * np.exp(-((x - mu1) ** 2) / (2.0 * s1 ** 2))
    p += w2 * np.exp(-((x - mu2) ** 2) / (2.0 * s2 ** 2))
    return safe_normalize(np.maximum(p, 0.0))


def make_asym_gauss(n, a, mu, b):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    dx = x - mu
    p = np.exp(-a * dx * dx + b * dx)
    return safe_normalize(np.maximum(p, 0.0))


def make_cosine_taper(n, alpha):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + float(alpha) * np.cos(np.pi * x)
    return safe_normalize(np.maximum(p, 0.01))


def make_sine_boost(n, f, amp, phase):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + float(amp) * np.sin(2.0 * np.pi * float(f) * x + float(phase))
    return safe_normalize(np.maximum(p, 0.01))


def make_cheb_shape(n, coeffs, tail_l, tail_r):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    z = 2.0 * x - 1.0
    logp = np.zeros(n, dtype=np.float64)
    for i, c in enumerate(coeffs):
        if i == 0:
            T = np.ones(n)
        elif i == 1:
            T = z.copy()
        else:
            T_prev2 = np.ones(n)
            T_prev1 = z.copy()
            for d in range(2, i + 1):
                T_curr = 2.0 * z * T_prev1 - T_prev2
                T_prev2 = T_prev1
                T_prev1 = T_curr
            T = T_prev1
        logp += float(c) * T
    eps = 1e-5
    logp += float(tail_l) * np.log(x + eps)
    logp += float(tail_r) * np.log(1.0 - x + eps)
    m = float(np.max(logp))
    p = np.exp(logp - m)
    return safe_normalize(p)


def generate_initials(n, rng, parent=None):
    n = max(16, int(n))
    out = []
    if parent is not None:
        parent = np.asarray(parent, dtype=np.float64)
        if parent.size == n:
            out.append(safe_normalize(parent))
    out.append(np.full(n, 1.0 / n, dtype=np.float64))
    for s in (0.5, 1.0, 2.0, 4.0, 7.0, 12.0, 20.0, 35.0):
        out.append(make_geometric(n, s, False))
        out.append(make_geometric(n, s, True))
    for a in (0.3, 0.7, 1.2, 2.0, 3.5, 5.0):
        out.append(make_power_law(n, a))
    for steep, tail in ((3, 0.1), (5, 0.05), (8, 0.02), (12, 0.01), (20, 0.005)):
        out.append(make_waterfall(n, steep, tail))
        out.append(make_waterfall(n, steep, tail))  # reversed versions differ
    out.append(make_waterfall(n, 5, 0.05))
    x = np.linspace(0, 1, n)
    rev_waterfall = safe_normalize(np.exp(-5.0 * (1.0 - x)) + 0.05)
    out.append(rev_waterfall)
    out.append(make_two_bump(n, 0.15, 0.06, 1.0, 0.85, 0.08, 0.3))
    out.append(make_two_bump(n, 0.85, 0.06, 1.0, 0.15, 0.08, 0.3))
    out.append(make_two_bump(n, 0.3, 0.04, 1.0, 0.05, 0.15, 0.5))
    out.append(make_two_bump(n, 0.7, 0.04, 1.0, 0.95, 0.15, 0.5))
    out.append(make_two_bump(n, 0.5, 0.03, 1.0, 0.5, 0.2, 0.3))
    for a, mu, b in ((3, 0.2, 2), (3, 0.8, -2), (6, 0.15, 3), (6, 0.85, -3),
                     (10, 0.1, 2), (10, 0.9, -2), (15, 0.08, 2.5), (15, 0.92, -2.5),
                     (25, 0.05, 2), (25, 0.95, -2)):
        out.append(make_asym_gauss(n, a, mu, b))
    for alpha in (0.3, 0.7, 1.2, 2.0):
        out.append(make_cosine_taper(n, alpha))
        out.append(make_cosine_taper(n, -alpha))
    for f, amp, ph in ((1, 0.3, 0), (1, 0.5, 1), (2, 0.2, 0), (2, 0.4, 1.5),
                       (3, 0.15, 0), (3, 0.3, 2)):
        out.append(make_sine_boost(n, f, amp, ph))
    for coeffs_list, tl, tr in (
        ([0, 1.5, 0, -0.5], 0, 1),
        ([0, -1.5, 0, 0.5], 1, 0),
        ([0.5, 2, -1, 0], 0.5, 0.5),
        ([-0.5, -2, 1, 0], 0.5, 0.5),
        ([0, 3, 0, -1], 0, 2),
        ([0, -3, 0, 1], 2, 0),
        ([1, -1, 1, -1], 0, 0),
        ([-1, 1, -1, 1], 0, 0),
    ):
        out.append(make_cheb_shape(n, coeffs_list, tl, tr))
    for _ in range(8):
        coeffs = rng.uniform(-2, 2, 6).tolist()
        tl = float(rng.uniform(0, 3))
        tr = float(rng.uniform(0, 3))
        out.append(make_cheb_shape(n, coeffs, tl, tr))
    for _ in range(5):
        s1 = float(rng.uniform(0.5, 15))
        p = make_geometric(n, s1, rng.random() < 0.5)
        p = project_simplex(p + rng.normal(0, 0.1 / n, n))
        out.append(p)
    return out


def lp_best_response(p, M_target, rng):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 1 or n > LP_NMAX:
        return None
    if M_target <= 0 or not np.isfinite(M_target):
        return None
    f = p + 1e-15
    fs = float(f.sum())
    if fs <= 0:
        return None
    f = f / fs
    mid = np.arange(n, dtype=np.int32)
    rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel().astype(np.int32)
    rows = rows[rows < 2 * n - 1]
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    cols = cols[rows < 2 * n - 1]
    data = np.tile(f, n).astype(np.float64, copy=False)
    data = data[rows < 2 * n - 1]
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M_target, dtype=np.float64)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                      method="highs", options={"disp": False})
    except Exception:
        return None
    if res is None or not getattr(res, "success", False):
        return None
    g = getattr(res, "x", None)
    if g is None:
        return None
    g = np.asarray(g, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    return g / s


def optimize_local(p0, rng, time_limit, label=""):
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    if n == 0:
        return p, np.inf
    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score
    prev_p = p.copy()
    end = time.monotonic() + max(0.0, float(time_limit))
    iter_no = 0
    since_improve = 0
    factor = 0.5
    lp_count = 0

    n_lp = min(n, LP_NMAX)
    use_lp = n_lp >= 64

    while time.monotonic() < end:
        iter_no += 1
        factor = max(0.002, 0.4 * (0.998 ** iter_no))
        if since_improve > 400:
            factor = max(0.001, factor * 0.7)

        g, c_cur, T, surr = gradient_smoothmax(p, factor)
        gabs = float(np.max(np.abs(g))) if g.size else 0.0
        p_scale = max(1.0 / float(n), 0.05 * float(np.max(p)), 1e-12)

        accepted = False
        if gabs > 1e-300:
            base = p_scale / gabs
            lrs = [0.5 * base, 0.15 * base, 0.05 * base, 0.01 * base, 0.003 * base]
        else:
            lrs = [0.1, 0.05, 0.01]

        momentum = 0.3 if iter_no > 200 else 0.0

        for lr in lrs:
            lr = float(min(max(lr, 1e-9), 10.0))
            v = p - lr * g
            if momentum > 0:
                v = v + momentum * (p - prev_p)
            pt = project_simplex(v)
            c_t = conv_sq(pt)
            cmax_t = float(np.max(c_t))
            score_t = float(2.0 * n * cmax_t)  # p is normalized to sum=1
            T_t = max(T, 1e-300)
            diff_t = (c_t - cmax_t) / T_t
            surv = float(np.sum(np.exp(diff_t)))
            surr_t = float(cmax_t + T_t * np.log(surv)) if surv > 0 else cmax_t
            accept = False
            if np.isfinite(score_t) and np.isfinite(surr_t):
                if surr_t < surr - 1e-10:
                    accept = True
                elif score_t < best_score - 1e-10 and surr_t < surr + 0.5 * max(1.0, surr):
                    accept = True
                elif score_t < current_score - 1e-10 and surr_t < surr + 0.2 * max(1.0, surr):
                    accept = True
            if accept:
                if score_t < best_score:
                    best_score = score_t
                    best_p = pt.copy()
                    since_improve = 0
                else:
                    since_improve += 1
                current_score = score_t
                prev_p = p.copy()
                p = pt
                accepted = True
                break

        if not accepted:
            factor = max(0.001, factor * 0.5)
            if since_improve > 300:
                sigmas = [1.0 / n, 0.2 / n, 0.05 / n, 0.01 / n]
            else:
                sigmas = [0.05 / n, 0.01 / n, 0.002 / n]
            for sigma in sigmas:
                pt = project_simplex(p + rng.normal(0.0, sigma, n))
                sc = internal_score(pt)
                if np.isfinite(sc) and sc < current_score - 1e-12:
                    if sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        since_improve = 0
                    else:
                        since_improve += 1
                    current_score = sc
                    prev_p = p.copy()
                    p = pt
                    accepted = True
                    break

            if not accepted and n > 1:
                hi = int(np.argmax(g))
                lo = int(np.argmin(g))
                if hi != lo and g[hi] > g[lo] + 1e-14:
                    for frac in (0.2, 0.07, 0.02, 0.005):
                        delta = frac * p[hi]
                        if delta <= 0:
                            continue
                        pt = p.copy()
                        pt[hi] -= delta
                        pt[lo] += delta
                        pt = project_simplex(pt)
                        sc = internal_score(pt)
                        if np.isfinite(sc) and sc < current_score - 1e-12:
                            if sc < best_score:
                                best_score = sc
                                best_p = pt.copy()
                                since_improve = 0
                            else:
                                since_improve += 1
                            current_score = sc
                            prev_p = p.copy()
                            p = pt
                            accepted = True
                            break

        if not accepted:
            since_improve += 1
            if since_improve > 800:
                p = project_simplex(0.5 * best_p + 0.3 * p + 0.2 * rng.dirichlet(np.ones(n)))
                current_score = internal_score(p)
                prev_p = p.copy()
                factor = 0.4
                since_improve = 0
            elif since_improve > 350:
                p = project_simplex(0.8 * best_p + 0.2 * p + rng.normal(0, 0.05 / n, n))
                current_score = internal_score(p)
                prev_p = p.copy()

        # LP best-response
        if use_lp and iter_no % 400 == 0:
            remaining = end - time.monotonic()
            if remaining > 8.0:
                c_cur = conv_sq(p)
                M = float(np.max(c_cur))
                for M_frac in (1.0, 0.95, 0.85):
                    M_t = M * M_frac
                    if time.monotonic() > end - 5:
                        break
                    g_lp = lp_best_response(p[:n_lp] if n > n_lp else p, M_t, rng)
                    if g_lp is not None:
                        if n > n_lp:
                            g_full = np.zeros(n)
                            g_full[:n_lp] = g_lp
                            g_full = safe_normalize(g_full)
                        else:
                            g_full = g_lp
                        for t in (1.0, 0.7, 0.4, 0.2):
                            q = project_simplex((1 - t) * p + t * g_full)
                            sc = internal_score(q)
                            if np.isfinite(sc) and sc < best_score:
                                best_score = sc
                                best_p = q.copy()
                                p = q.copy()
                                prev_p = q.copy()
                                current_score = sc
                                since_improve = 0
                                lp_count += 1
                                break

        if iter_no % 2000 == 0:
            print("  %s iter=%d best=%.8f cur=%.8f T=%.5f lp=%d" % (
                label, iter_no, best_score, current_score, factor, lp_count), flush=True)

        if best_score < 1.4950 and since_improve > 5000:
            break

    best_p = safe_normalize(best_p)
    return best_p, best_score


def upsample_shape(p_small, n_large):
    p_small = np.asarray(p_small, dtype=np.float64)
    n_s = len(p_small)
    if n_s == 0:
        return np.full(n_large, 1.0 / n_large)
    x_s = np.linspace(0, 1, n_s)
    x_l = np.linspace(0, 1, n_large)
    p_interp = np.interp(x_l, x_s, p_small)
    return safe_normalize(p_interp)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = start_time + total_budget - 30.0

    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and float(parent_arr.sum()) > 0:
            parent = parent_arr
    except Exception:
        parent = None

    n_large = 4096
    if parent is not None and 64 <= parent.size <= 8192:
        n_large = parent.size
    n_small = 256

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = safe_normalize(np.asarray(p, dtype=np.float64))
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("  NEW BEST: %.8f (n=%d)" % (best_score, p.size), flush=True)

    # Phase 1: Small-n exploration
    t0 = time.monotonic()
    small_budget = min(120.0, (deadline - t0) * 0.12)
    if small_budget > 20:
        print("Phase 1: Small-n search (n=%d) budget=%.0fs" % (n_small, small_budget), flush=True)
        small_inits = generate_initials(n_small, rng, None)
        scored = []
        for p in small_inits:
            p = safe_normalize(p)
            sc = internal_score(p)
            if np.isfinite(sc):
                scored.append((sc, p))
        scored.sort(key=lambda x: x[0])
        for sc, p in scored[:3]:
            p_opt, sc_opt = optimize_local(p, rng, min(30.0, small_budget / 4.0), "SM")
            try_record(upsample_shape(p_opt, n_large), sc_opt)
        # Also score the upsampled parent
        if parent is not None:
            p_par = upsample_shape(parent, n_large)
            sc_par = internal_score(p_par)
            try_record(p_par, sc_par)
    print("Phase 1 done, best=%.8f, elapsed=%.0fs" % (best_score, time.monotonic() - start_time), flush=True)

    # Phase 2: Large-n local optimization from diverse starts
    t1 = time.monotonic()
    large_budget_max = (deadline - t1) * 0.75
    large_inits = generate_initials(n_large, rng, parent)
    scored_large = []
    for p in large_inits:
        p = safe_normalize(p)
        sc = internal_score(p)
        if np.isfinite(sc):
            scored_large.append((sc, p))
    scored_large.sort(key=lambda x: x[0])

    # Add the best small-n result upsampled
    if best_p is not None:
        scored_large.insert(0, (best_score, best_p))

    top_k = min(6, len(scored_large))
    jobs = [(sc, p, "L%d" % i) for i, (sc, p) in enumerate(scored_large[:top_k])]
    jobs.sort(key=lambda x: x[0])

    print("Phase 2: Large-n optimization, %d jobs, budget=%.0fs" % (len(jobs), large_budget_max), flush=True)
    n_jobs = len(jobs)
    final_reserve = min(180.0, max(60.0, (deadline - t1) * 0.2))
    available = max(30.0, (deadline - final_reserve) - t1)
    time_per_job = available / max(1, n_jobs)

    for i, (sc, p, label) in enumerate(jobs):
        remaining = deadline - final_reserve - time.monotonic()
        if remaining < 15:
            break
        t_alloc = min(time_per_job * (1 + 0.3 * (top_k - i)), remaining)
        t_alloc = max(20.0, t_alloc)
        print("  Job %s: initial score=%.8f, time=%.0fs" % (label, sc, t_alloc), flush=True)
        p_opt, sc_opt = optimize_local(p, rng, t_alloc, label)
        try_record(p_opt, sc_opt)
        elapsed = time.monotonic() - start_time
        print("  Job %s done: %.8f, total elapsed=%.0fs" % (label, sc_opt, elapsed), flush=True)

    # Phase 3: Final polish
    if best_p is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print("Phase 3: Final polish, time=%.0fs" % remaining, flush=True)
            p_final, sc_final = optimize_local(best_p, rng, remaining - 10, "FIN")
            try_record(p_final, sc_final)

    if best_p is None:
        best_p = np.full(n_large, 1.0 / n_large, dtype=np.float64)
        best_score = internal_score(best_p)

    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s <= 1e-12:
        best_p = np.full(n_large, 1.0 / n_large, dtype=np.float64)
    final_sc = internal_score(best_p)
    print("FINAL: score=%.10f, n=%d, elapsed=%.0fs" % (final_sc, len(best_p), time.monotonic() - start_time), flush=True)
    return [float(x) for x in best_p]
