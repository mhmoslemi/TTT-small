from scipy.fft import next_fast_len

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = next_fast_len(n + m - 1)
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def local_score(a):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def solve_best_response(f, M):
    """
    Solve: max sum(g) s.t. conv(f, g)[k] <= M for all k, g >= 0
    Using the dual: min M * sum(y) s.t. A^T y >= 1, y >= 0
    where A is the convolution matrix (2n-1 rows, n cols)
    """
    n = len(f)
    # Dual: min M * 1^T y s.t. A^T y >= 1, y >= 0
    # A^T y >= 1 means: for each i in 0..n-1, sum_k y[k] * f[k-i] >= 1
    # A is (2n-1) x n, where A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    
    # Actually, let's work with the primal directly since it's smaller
    # Primal: max -sum(g) (minimize) s.t. A@g <= M, g >= 0
    # c = -1 (minimize -sum = maximize sum)
    # A shape: (2n-1, n), A[k, i] = f[i] if k-i is in [0, n-1], i.e., i in [k-(n-1), k]
    
    # Build A matrix
    # A[k, i] = f[i] if max(0, k-n+1) <= i <= min(k, n-1), else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        A[k, i_min:i_max+1] = f[i_min:i_max+1]
    
    c_vec = -np.ones(n)  # minimize -sum(g)
    b_vec = np.full(2*n-1, M)
    
    res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=[(0, None)] * n, method='highs')
    if res.success:
        g = res.x
    else:
        g = np.ones(n)  # fallback
    return g

def lp_iteration(f, n, t):
    """One LP-based iteration: find best response g, then blend."""
    c = fft_convolve(f, f)
    M = np.max(c)
    g = solve_best_response(f, M)
    # Normalize g to have same sum as f
    if np.sum(g) > 1e-10:
        g = g * np.sum(f) / np.sum(g)
    f_new = (1 - t) * f + t * g
    return np.maximum(f_new, 0), M

"""
Hybrid optimization combining LP-based best response dynamics with gradient
refinement and peak-targeted local search to minimize the normalized peak
of the self-convolution of non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = np.fft.next_fast_len(n + m - 1)
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def compute_score(a):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def get_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix A where (A@g)[k] = conv(f,g)[k]."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        A[k, i_lo:i_hi+1] = f[i_lo:i_hi+1]
    return A

def lp_best_response(f, M):
    """Solve max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = get_conv_matrix(f)
    c_vec = -np.ones(n)
    b_vec = np.full(2*n-1, M)
    res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return f / np.sum(f) * np.sum(f) if np.sum(f) > 0 else np.ones(n)

def lp_iteration(f, M=None, t=0.3):
    c = fft_convolve(f, f)
    if M is None:
        M = np.max(c)
    g = lp_best_response(f, M)
    S_f = np.sum(f)
    S_g = np.sum(g)
    if S_g > 1e-10:
        g = g * S_f / S_g
    f_new = (1 - t) * f + t * g
    return np.maximum(f_new, 0)

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def soft_max_conv_grad(a, tau):
    """Compute soft-max of conv(a,a) and its gradient w.r.t. a."""
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    w = np.exp(shifted)
    w = w / np.sum(w)
    value = c_max + tau * np.log(np.sum(w))
    # grad = 2 * corr(a, w) = 2 * conv(a_rev, w)[n-1 : 2n-1]
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    n = len(a)
    g = 2.0 * corr[n-1:2*n-1]
    return value, g

def gradient_refine(a, max_time, rng):
    """Adam gradient descent on softmax-parameterized sequence."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    theta = np.log(a / np.sum(a) + 1e-10)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    step = 0
    best = a.copy()
    best_score = compute_score(a)
    tau_start = 20.0 / n
    tau_end = 0.5 / n
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < max_time:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / max_time)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        a_curr = softmax_normalize(theta) * S
        value, g_a = soft_max_conv_grad(a_curr, tau)
        inner = np.dot(a_curr, g_a)
        g_theta = a_curr * (g_a - inner / S)
        
        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * g_theta**2
        m1_hat = m1 / (1 - b1**step)
        m2_hat = m2 / (1 - b2**step)
        lr = 1.0 / np.sqrt(step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        a_check = softmax_normalize(theta) * S
        score = compute_score(a_check)
        if score < best_score:
            best_score = score
            best = a_check.copy()
    
    return best, best_score

def peak_targeted_polish(a, max_time, rng):
    """Local search targeting the current peak of the convolution."""
    n = len(a)
    current = a.copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    temp = 0.002
    t0 = time.monotonic()

            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            # Compute contributions
            contribs = np.zeros(n)
            i_lo = max(0, peak_idx - n + 1)
            i_hi = min(peak_idx, n - 1)
            for i in range(i_lo, i_hi + 1):
                j = peak_idx - i
                contribs[i] = current[i] * current[j]
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.6:
            # Block perturbation
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.8:

            shift = int(rng.integers(1, max(2, n // 4)))
            candidate = np.roll(current, shift)
            candidate *= np.exp(rng.normal(0, 0.01, size=n))
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0, 0.01, size=n))
        
        candidate = np.clip(candidate, 0.001, 1000.0)
        new_score = compute_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score

                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-15)):
            current = candidate
            current_score = new_score
        
        temp *= 0.9998
    
    return best, best_score

    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.05, None)

Actually, for a pure exponential a[i] = r^i (0 < r < 1):
conv[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} r^i * r^{k-i} = sum of (number of valid terms) * r^k

So conv[k] = m(k) * r^k where m(k) is the number of valid terms.
For k = 0: m = 1, conv = r^0 = 1
For k = 1: m = 2, conv = 2r
...
For k = n-1: m = n, conv = n*r^{n-1}
For k = n: m = n-1, conv = (n-1)*r^n
...

The peak is at some k that balances m(k) and r^k. For r close to 1, the peak is near k=n-1 (center). For r close to 0, the peak is at k=0.

This is not great because the peak is always at one extreme.

A better structure might be: a[i] = exp(-c*i*(n-i)/n) which is a "bump" function that's larger in the middle. But this is symmetric, so we're back to the Cauchy-Schwarz bound.

"""
Hybrid optimization combining LP-based steepest descent for the L-infinity
convolution peak, best-response dynamics, gradient refinement, and
peak-targeted local search to minimize the normalized peak of the
self-convolution of non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = np.fft.next_fast_len(n + m - 1)
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def compute_score(a):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        A[k, i_lo:i_hi + 1] = f[i_lo:i_hi + 1]
    return A


def lp_steepest_descent(f, alpha=1.0):
    """
    Linearized steepest descent: minimize t s.t.
      conv(f,f)[k] + 2*conv(f, delta)[k] <= t for all k
      sum(delta) = 0
      delta[i] >= -f[i]
    Returns the step delta (scaled by alpha).
    """
    n = len(f)
    conv_ff = fft_convolve(f, f)
    A_conv = build_conv_matrix(f)

    # Variables: [delta_0, ..., delta_{n-1}, t]
    # Objective: minimize t
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0

    # Inequality: A_conv @ delta - t <= -conv_ff
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = 2.0 * A_conv
    A_ub[:, n] = -1.0
    b_ub = -conv_ff

    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    # Bounds: delta[i] >= -f[i], t free
    bounds = [(-f[i], None) for i in range(n)] + [(None, None)]

    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success:
        delta = res.x[:n] * alpha
        return delta
    return np.zeros(n)


def lp_best_response(f, M):
    """Solve max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_vec = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return f.copy()


def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def soft_max_conv_grad(a, tau):
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    value = c_max + tau * np.log(sum_exp)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    n = len(a)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return value, g


def gradient_refine(a, max_time, rng):
    n = len(a)
    S = np.sqrt(2.0 * n)
    theta = np.log(a / np.sum(a) + 1e-10)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    step = 0
    best = a.copy()
    best_score = compute_score(a)
    tau_start = 30.0 / n
    tau_end = 0.3 / n
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < max_time:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / max_time)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        a_curr = softmax_normalize(theta) * S
        value, g_a = soft_max_conv_grad(a_curr, tau)
        inner = np.dot(a_curr, g_a)
        g_theta = a_curr * (g_a - inner / S)

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * g_theta ** 2
        m1_hat = m1 / (1 - b1 ** step)
        m2_hat = m2 / (1 - b2 ** step)
        lr = 1.5 / np.sqrt(step + 1)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        a_check = softmax_normalize(theta) * S
        score = compute_score(a_check)
        if score < best_score:
            best_score = score
            best = a_check.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                # Perturb to escape local min
                theta += rng.normal(0, 0.1, size=n)
                no_improve = 0

    return best, best_score


def peak_targeted_polish(a, max_time, rng):
    n = len(a)
    current = a.copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    temp = 0.0015
    t0 = time.monotonic()
    iter_count = 0

    while time.monotonic() - t0 < max_time:
        r = rng.random()
        candidate = current.copy()

        if r < 0.45:
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            i_lo = max(0, peak_idx - n + 1)
            i_hi = min(peak_idx, n - 1)
            for i in range(i_lo, i_hi + 1):
                j = peak_idx - i
                contribs[i] = current[i] * current[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.06, 0.02, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(12, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.06, 0.02, size=n_inc))
        elif r < 0.65:
            width = max(2, int(rng.integers(2, max(3, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.05, size=width))
        elif r < 0.8:
            shift = int(rng.integers(1, max(2, n // 6)))
            candidate = np.roll(current, shift)
            candidate *= np.exp(rng.normal(0, 0.015, size=n))
        elif r < 0.9:
            # Antisymmetric perturbation
            antisym = (current - current[::-1]) * 0.1
            candidate = current + antisym
            candidate *= np.exp(rng.normal(0, 0.01, size=n))
        else:
            candidate = current * np.exp(rng.normal(0, 0.012, size=n))

        candidate = np.clip(candidate, 0.001, 1000.0)
        new_score = compute_score(candidate)
        iter_count += 1

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-15)):
            current = candidate
            current_score = new_score

        temp *= 0.99985

    return best, best_score


def make_chirp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.05, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_ramp_cos(n, rng):
    t = np.linspace(0, 1, n)
    phase = rng.uniform(0, 2 * np.pi)
    a = (0.5 + 0.5 * t) * (1.0 + 0.4 * np.cos(4 * np.pi * t + phase))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_bump_osc(n, rng):
    t = np.linspace(0, 1, n)
    a = np.exp(-4.0 * (t - 0.5 + 0.2 * (rng.random() - 0.5)) ** 2)
    a *= (1.0 + 0.3 * np.cos(2 * np.pi * rng.integers(3, 12) * t + rng.uniform(0, 6.28)))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_random(n, rng):
    a = 1.0 + 0.3 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def run_lp_search(a0, max_time, use_steepest=True, use_bestresp=True, rng=None):
    """Run LP-based iterations on a candidate."""
    n = len(a0)
    a_curr = a0.astype(np.float64).copy()
    S = np.sum(a_curr)
    curr_score = compute_score(a_curr)
    best = a_curr.copy()
    best_score = curr_score
    t0 = time.monotonic()
    no_improve = 0
    alpha = 1.0
    iter_count = 0

    while time.monotonic() - t0 < max_time:
        use = use_steepest and (not use_bestresp or rng.random() < 0.7) if rng else use_steepest
        if use:
            delta = lp_steepest_descent(a_curr, alpha=alpha)
            a_new = a_curr + delta
        else:
            c = fft_convolve(a_curr, a_curr)
            M = np.max(c)
            g = lp_best_response(a_curr, M)
            S_g = np.sum(g)
            if S_g > 1e-10:
                g = g * S / S_g
            a_new = 0.5 * a_curr + 0.5 * g

        a_new = np.maximum(a_new, 0.0)
        S_new = np.sum(a_new)
        if S_new < 0.01:
            a_new = a_curr
        else:
            a_new = a_new * S / S_new
            new_score = compute_score(a_new)

            if new_score < curr_score:
                a_curr = a_new
                curr_score = new_score
                no_improve = 0
                if curr_score < best_score:
                    best_score = curr_score
                    best = a_curr.copy()
                alpha = min(1.0, alpha * 1.05)
            else:
                no_improve += 1
                if no_improve > 50:
                    alpha = max(0.1, alpha * 0.7)
                    no_improve = 0

        iter_count += 1
        if iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iter_count}, score={curr_score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s, alpha={alpha:.3f}")

    return best, best_score
