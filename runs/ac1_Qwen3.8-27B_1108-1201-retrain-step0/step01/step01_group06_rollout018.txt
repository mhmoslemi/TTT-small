from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    max_rho = rho[-1]
    tau = (cv[max_rho] - 1) / (max_rho + 1.0)
    return np.maximum(v - tau, 0)

def subgradient_max_conv(a):
    """Compute a (sub)gradient of max_k (a*a)[k] w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return np.zeros(n), 0.0, conv
    
    # Soft-max weights
    # Use top-k peaks with exponential weighting
    top_k = max(1, min(n, int(0.05 * len(conv))))
    sorted_idx = np.argsort(conv)[::-1]
    active = sorted_idx[:top_k]
    
    # Exponential weights (temperature)
    t = conv[active]
    beta = 10.0 / max_conv  # temperature parameter
    w = np.exp(beta * (t - t.max()))
    w /= w.sum()

    for idx, k in enumerate(active):
        k_int = int(k)

def solve_lp_direction(f, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    f: current sequence (sum = sqrt(2n))
    M: max(f*f)
    Returns: g (the LP solution)
    """
    n = len(f)
    # c = -ones(n) (we're minimizing -sum(g))
    c = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # A_ub[i, j] = f[k-j] if valid, else 0
    # where k = i (the i-th constraint)
    
    # Build the constraint matrix
    # For constraint k: sum_j f[k-j] * g[j] <= M (where f[m]=0 for m outside [0,n-1])
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # This is a banded matrix. Let's build it.
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        A_ub[k, j_start:j_end+1] = f[k-j_end : k-j_start+1]  # f[k-j_end], ..., f[k-j_start]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return None

`= M(1-t)^2 + 2t(1-t)M + t^2(g*g)[k*]`
`= M(1-2t+t^2) + 2tM - 2t^2 M + t^2(g*g)[k*]`
`= M - 2tM + t^2M + 2tM - 2t^2M + t^2(g*g)[k*]`
`= M + t^2((g*g)[k*] - M)`

So if (g*g)[k*] < M, the k*-th entry decreases (for the new sequence). But we don't know (g*g)[k*] in advance.

However, for the sum: sum(f_new) = (1-t)sum(f) + t*sum(g) = (1-t)S + t*sum(g) where S = sum(f).
Since the LP maximizes sum(g), and sum(f) = S, we have sum(g) >= S (because f itself is a feasible solution with sum = S). Actually, is f feasible? (f*f)[k] <= M for all k, yes. And f >= 0. So sum(g*) >= sum(f) = S.

So sum(f_new) = (1-t)S + t*sum(g*) >= (1-t)S + tS = S. The sum increases (or stays the same).

The objective E = 2n * max(conv) / (sum)^2. If max(conv) doesn't increase and sum increases, E decreases.

But max(conv) might increase at indices other than k*! The LP only bounds (f*g)[k], not (g*g)[k] or the full (f_new * f_new)[k].

So the LP approach is a heuristic, not a guaranteed descent method. But in practice, it often works well, especially for small t.

"""
Hybrid subgradient + LP-based optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses projected gradient descent with momentum on the simplex, soft-max smoothing
with temperature annealing, and periodic LP-based direction finding (from SOTA)
for local refinement. Tries multiple sequence lengths and initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    tau = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - tau, 0.0)


def compute_gradient(a, conv, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*conv[k])) w.r.t. a.
    This equals sum_k p_k * grad(conv[k]) where p_k = softmax(beta*conv).
    grad(conv[k])[j] = 2*a[k-j] (or a[j] if k=2j) for valid k-j in [0,n-1].
    """
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)

    scaled = beta * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    thresh = w.max() * 1e-8
    active = np.where(w > thresh)[0]

    for k_idx in active:
        k = int(k_idx)
        wk = w[k]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        # a[k-j] for j = j_start..j_end is a[k-j_start], a[k-j_start-1], ..., a[k-j_end]
        # = a[k-j_end : k-j_start+1][::-1]
        vals = a[k - j_end : k - j_start + 1][::-1]
        if k % 2 == 0:
            j_mid = k // 2
            if j_start <= j_mid <= j_end:
                pos = j_mid - j_start
                if pos > 0:
                    grad[j_start : j_start + pos] += wk * 2.0 * vals[:pos]
                grad[j_mid] += wk * vals[pos]
                if pos + 1 < len(vals):
                    grad[j_mid + 1 : j_end + 1] += wk * 2.0 * vals[pos + 1:]
            else:
                grad[j_start : j_end + 1] += wk * 2.0 * vals
        else:
            grad[j_start : j_end + 1] += wk * 2.0 * vals

    return grad


def solve_lp_direction(f, M):
    """
    LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns g or None if infeasible.
    """
    n = len(f)
    c = -np.ones(n)
    rows = 2 * n - 1
    A_ub = np.zeros((rows, n))
    for k in range(rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A_ub[k, j_start:j_end + 1] = f[k - j_end : k - j_start + 1]
    b_ub = np.full(rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if res.success:
            return res.x
    except Exception:
        pass
    return None
