def compute_smax_grad(a, tau):
    """Compute gradient of soft-max of conv(a,a)."""
    c = np.convolve(a, a, mode='full')
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    grad = (2.0 / tau) * np.correlate(q, a, mode='valid')
    return grad, c

def compute_trunc_grad(a, threshold):
    """Compute gradient of sum(max(0, c_k - T)^2) where c = conv(a,a)."""
    c = np.convolve(a, a, mode='full')
    excess = np.maximum(0, c - threshold)
    grad = 4.0 * np.correlate(excess, a, mode='valid')
    return grad, c

def project_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    s = a.sum()
    if abs(s - S) < 1e-12:
        return a
    if s < S:
        return a + (S - s) / n
    # Binary search for offset
    lo, hi = 0.0, a.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    # Fix rounding
    res[res.argmax()] += S - res.sum()
    return np.maximum(res, 0.0)

    t_k = 1.0

    # Estimate initial step size

    for step in range(n_steps):

        beta_new = (t_k - 1.0) / t_new
        t_k = t_new

        tau = tau_hi * (tau_lo / tau_hi) ** prog

            c_curr = np.convolve(a, a, mode='full')

def soft_max_grad(a, tau):
    """Gradient of soft-max approximation to max(conv(a,a))."""
    c = np.convolve(a, a, mode='full')
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    q = exp_c / exp_c.sum()
    grad = (2.0 / tau) * np.correlate(q, a, mode='valid')
    soft_max_val = tau * np.log(exp_c.sum()) + c_max
    return grad, soft_max_val, c


def trunc_l2_grad(a, T):

    c = np.convolve(a, a, mode='full')
    excess = np.maximum(0.0, c - T)
    grad = 4.0 * np.correlate(excess, a, mode='valid')
    obj = excess.sum(excess)  # This is the objective value
    return grad, obj, c


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    a = np.asarray(a, dtype=np.float64).copy()

    if s < S:
        a += (S - s) / n
        return a

        return a
    lo, hi = 0.0, a.max()

        else:

        res[idx] += diff

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using FISTA with correct soft-max gradient,
geometric tau decay, adaptive step size, and truncated-L2 polishing.
Explores multiple lengths and initializations, then focuses on the best candidate.
"""

import numpy as np
import time


def soft_max_grad(a, tau):
    """Compute gradient and value of soft-max approximation to max(conv(a,a))."""
    c = np.convolve(a, a, mode='full')
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    exp_sum = exp_c.sum()
    q = exp_c / exp_sum
    grad = (2.0 / tau) * np.correlate(q, a, mode='valid')
    val = tau * np.log(exp_sum) + c_max
    return grad, val, c


def trunc_l2_grad(a, T):
    """Gradient of sum(max(0, c_k - T)^2) where c = conv(a,a)."""
    c = np.convolve(a, a, mode='full')
    excess = np.maximum(0.0, c - T)
    grad = 4.0 * np.correlate(excess, a, mode='valid')
    obj = np.dot(excess, excess)
    return grad, obj, c


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= S:
        a += (S - s) / n
        return a
    lo, hi = 0.0, a.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    res = np.maximum(a - t, 0.0)
    diff = S - res.sum()
    if abs(diff) > 1e-14:
        idx = res.argmax()
        res[idx] += diff
    res = np.maximum(res, 0.0)
    return res


def run_fista(a0, tau_hi, tau_lo, t_budget, lr_init=None, verbose=False):
    """FISTA with soft-max objective, simplex projection, adaptive lr."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = project_simplex(a0 / max(a0.sum(), 1e-30) * S, S)
    a_prev = a.copy()
    t_k = 1.0

    # Initial step size estimation
    if lr_init is not None:
        lr = lr_init
    else:
        # L ~ 4*sqrt(2n)/tau_hi is a rough upper bound
        lr = tau_hi / (4.0 * np.sqrt(2.0 * n))
        lr = max(lr, 1e-6)

    best_a = a.copy()
    best_mc = np.convolve(best_a, best_a, mode='full').max()
    best_smax = best_mc  # soft-max value at best

    t0 = time.monotonic()
    step = 0
    stuck_count = 0

    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        prog = min(1.0, elapsed / max(t_budget, 1e-9))
        tau = tau_hi * (tau_lo / tau_hi) ** prog

        # Nesterov extrapolation
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        beta = (t_k - 1.0) / t_new
        t_k = t_new

        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0.0)

        # Gradient
        g, smax_val, c = soft_max_grad(y, tau)

        # Step with backtracking
        a_prev = a.copy()
        lr_trial = lr
        for bt in range(15):
            a_new = project_simplex(y - lr_trial * g, S)
            # Evaluate soft-max at new point (cheap check via max)
            c_new = np.convolve(a_new, a_new, mode='full')
            mc_new = c_new.max()
            smax_new = tau * np.log(np.exp((c_new - mc_new) / tau).sum()) + mc_new
            # Arc-condition (simplified)
            if smax_new <= smax_val + 0.01 * np.dot(g, a_new - y):
                break
            lr_trial *= 0.5

        a = a_new
        lr = lr_trial
        step += 1

        # Track best
        if step % 50 == 1:
            mc = np.convolve(a, a, mode='full').max()
            if mc < best_mc:
                best_mc = mc
                best_a = a.copy()
                stuck_count = 0
            else:
                stuck_count += 1
                if stuck_count > 20:
                    # Stuck: reduce lr and reset momentum
                    lr *= 0.7
                    t_k = 1.0
                    a_prev = a.copy()
                    stuck_count = 0

        if verbose and step % 5000 == 0:
            mc = np.convolve(a, a, mode='full').max()
            print(f"    step={step}, mc={mc:.6f}, best={best_mc:.6f}, tau={tau:.4f}, lr={lr:.2e}")

    return best_a, best_mc
