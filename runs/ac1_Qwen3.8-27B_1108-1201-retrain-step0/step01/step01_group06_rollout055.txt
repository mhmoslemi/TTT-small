def project_simplex(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    v = np.array(v, dtype=np.float64)
    n = len(v)
    # Use the standard projection algorithm
    if v.sum() <= 0:
        return np.full(n, target_sum / n)
    
    # Bisection on the threshold
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.where(v_sorted > (cumsum - target_sum) / np.arange(1, n+1))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)

    theta = (cumsum[rho] - target_sum) / (rho + 1)
    result = np.maximum(v - theta, 0)
    # Ensure exact sum
    result_sum = result.sum()
    if result_sum > 1e-15:

    return result

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smoothed-max (softmax) subgradient with FISTA acceleration,
window-function initializations, spectral diversification, and discrete perturbations.
Key insight: optimal sequences resemble asymmetric window shapes; softmax-softened
peak targeting with Nesterov acceleration escapes flat plateaus better than
pure multiplicative mirror descent.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_nonneg_sum(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum} via sorting."""
    v = np.array(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    v_clipped = np.maximum(v, 0.0)
    s = v_clipped.sum()
    if s < 1e-15:
        return np.full(n, target_sum / n)
    # Exact projection via bisection on threshold
    v_sorted = np.sort(v_clipped)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho_candidates = np.where(v_sorted > (cumsum - target_sum) / np.arange(1, n + 1))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cumsum[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    result_sum = result.sum()
    if result_sum > 1e-15:
        result *= target_sum / result_sum
    else:
        result = np.full(n, target_sum / n)
    return result


def softmax_weights(conv, t):
    """Compute softmax weights over convolution values with temperature t."""
    scaled = t * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    return w


def compute_grad_smoothed(a, conv, t, n):
    """
    Gradient of smoothed-max objective (1/t)*log(sum exp(t*conv)) wrt a.
    Uses softmax weights over convolution positions.
    grad_a[m] = sum_k w[k] * 2 * a[k-m]  (with a[i]=0 for invalid i)
    """
    w = softmax_weights(conv, t)
    grad = np.zeros(n)
    # Only process significant weights for efficiency
    thresh = w.max() * 1e-4
    active_idx = np.where(w > thresh)[0]
    for k_int in active_idx:
        k = int(k_int)
        wk = w[k]
        # a[k-m] valid for m in [max(0, k-n+1), min(n-1, k)]
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            # a[k-m] for m=m_lo..m_hi gives a[k-m_hi], a[k-m_hi-1], ..., a[k-m_lo]
            # = a[k-m_hi : k-m_lo+1] reversed
            grad[m_lo:m_hi + 1] += 2.0 * wk * a[k - m_hi:k - m_lo + 1][::-1]
    return grad


def fista_optimize(a0, time_limit, t_start=50.0, t_end=500.0, eta0=0.05, name=""):
    """
    FISTA (accelerated proximal gradient) on the smoothed auto-convolution peak.
    a0: initial sequence (will be normalized to sum=1)
    t: softmax temperature (annealed from t_start to t_end)
    eta: step size
    Returns (best_a, best_objective_normalized).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = obj_norm(a)
    best_a = a.copy()

    t0 = time.monotonic()
    total_time = time_limit
    elapsed = 0.0
    patience = 0
    max_patience = 150
    L = 4.0  # Lipschitz constant estimate (will adapt)
    y = a.copy()
    beta_prev = 1.0
    eta = eta0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = elapsed / total_time  # 0 to 1

        # Anneal temperature: start smooth, end sharp
        current_t = t_start + (t_end - t_start) * (progress ** 0.5)

        # Compute gradient at y
        conv_y = fast_convolve(y)
        grad_y = compute_grad_smoothed(y, conv_y, current_t, n)

        # Center gradient to approximately preserve sum
        grad_centered = grad_y - grad_y.mean()

        # BFal Anschluss: step size = 1/L
        step = 1.0 / L

        # Proximal step (projection onto simplex)
        x_new = project_nonneg_sum(y - step * grad_centered)

        # FISTA update
        beta = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * beta_prev ** 2))
        y = x_new + ((beta_prev - 1.0) / beta) * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-12:
            y = x_new.copy()
        else:
            y /= y_sum
        beta_prev = beta
        a = x_new

        # Evaluate
        conv_a = fast_convolve(a)
        current_obj = obj_norm(a)

        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Adaptive step size
        if patience > 30:
            L *= 1.5
            patience = 0
            if L > 100.0:
                L = 100.0
                # Try to escape: perturb
                a += eta0 * 0.1 * np.random.randn(n)
                a = project_nonneg_sum(a)
                y = a.copy()
                beta_prev = 1.0
                patience = 0

        if patience >= max_patience:
            # Reset momentum and try different parameters
            L = 10.0
            eta = eta0 * 0.7
            y = a.copy()
            beta_prev = 1.0
            patience = 0
            max_patience = 100
            if eta < 0.001:
                break

    return best_a, best_obj


def subgradient_descent(a0, time_limit, eta0=0.01, top_k=10, t_param=100.0):
    """
    Basic subgradient descent with softmax-smoothed peak targeting.
    Simpler than FISTA but more stable in some regimes.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = obj_norm(a)
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    patience = 0
    eta = eta0
    current_t = t_param

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        # Use softmax gradient
        grad = compute_grad_smoothed(a, conv, current_t, n)
        grad_centered = grad - grad.mean()

        step = eta / (1.0 + 0.005 * iteration)
        a_new = project_nonneg_sum(a - step * grad_centered)

        conv_new = fast_convolve(a_new)
        obj_new = obj_norm(a_new)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new
        iteration += 1

        if patience > 80:
            eta *= 0.75
            current_t *= 1.5
            patience = 0
            if eta < 1e-5:
                break

    return best_a, best_obj


def spectral_diversify(a, n_variants=5):
    """
    Generate diverse variants by modifying the frequency spectrum.
    This helps escape local minima by exploring different spectral shapes.
    """
    n = len(a)
    variants = []
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    power = np.abs(A) ** 2
    phase = np.angle(A)

    for i in range(n_variants):
        new_power = power.copy()
        # Apply different spectral modifications
        if i == 0:
            # Low-pass: keep only low frequencies
            cutoff = len(power) // 4
            new_power[cutoff:] *= 0.3
        elif i == 1:
            # High-pass emphasis
            new_power[:max(1, len(power)//8)] *= 0.5
        elif i == 2:
            # Random phase perturbation
            phase_new = phase + 0.1 * np.random.randn(len(phase))
            new_power = power
            A_new = np.sqrt(new_power) * np.exp(1j * phase_new)
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            if a_new.sum() > 1e-12:
                a_new /= a_new.sum()
            variants.append(a_new)
            continue
        elif i == 3:
            # Bandpass
            lo = len(power) // 8
            hi = 3 * len(power) // 4
            mask = np.ones_like(new_power)
            mask[:lo] *= 0.5
            mask[hi:] *= 0.5
            new_power *= mask
        elif i == 4:
            # Sharpen spectrum
            new_power = new_power ** 1.5 / (new_power ** 1.5).sum() * power.sum()

        new_power = np.maximum(new_power, 0)
        A_new = np.sqrt(new_power) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        if a_new.sum() > 1e-12:
            a_new /= a_new.sum()
            variants.append(a_new)

    return variants


def discrete_perturb(a, rng, n_perturbs=20):
    """Apply random discrete perturbations to explore nearby solutions."""
    n = len(a)
    results = []
    for _ in range(n_perturbs):
        a_new = a.copy()
        op = rng.integers(0, 4)
        if op == 0:
            # Swap two elements
            i, j = rng.choice(n, 2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]
        elif op == 1:
            # Transfer mass from one to another
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                amt = a[i] * rng.uniform(0.05, 0.4)
                a_new[i] -= amt
                a_new[j] += amt
        elif op == 2:
            # Local smoothing (average with neighbor)
            i = rng.integers(1, n - 1)
            t = rng.uniform(0.3, 0.7)
            a_new[i] = t * a[i] + (1 - t) * (a[i-1] + a[i+1]) / 2
            a_new[i-1] = (1 - t/2) * a[i-1]
            a_new[i+1] = (1 - t/2) * a[i+1]
        elif op == 3:
            # Add small noise to a subset
            subset = rng.choice(n, size=max(2, n//10), replace=False)
            a_new[subset] += rng.normal(0, 0.01, size=len(subset))

        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-12:
            a_new /= s
            results.append(a_new)
    return results


def generate_inits(n, rng):
    """Generate diverse initializations including window functions and asymmetric shapes."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Hann window
    if n > 1:
        h = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
        inits['hann'] = h / h.sum()

    # Hamming window
    if n > 1:
        h = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / (n - 1))
        inits['hamming'] = h / h.sum()

    # Blackman window
    if n > 2:
        b = 0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * x / (n - 1))
        b = np.maximum(b, 0)
        if b.sum() > 1e-12:
            inits['blackman'] = b / b.sum()

    # Gaussian (centered)
    g = np.exp(-0.5 * ((x - n / 2.0) / (n * 0.25)) ** 2)
    inits['gauss'] = g / g.sum()

    # Asymmetric ramp (key: asymmetry helps reduce center lag)
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    # Inverse ramp
    ir = (n + 1 - np.arange(n + 1)).astype(np.float64)
    ir = ir[1:]  # remove the zero
    inits['inv_ramp'] = ir / ir.sum()

    # Exponential decay
    e = np.exp(-x / (n * 0.2))
    inits['exp_decay'] = e / e.sum()

    # Bimodal (edges)
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Asymmetric bimodal
    abi = np.zeros(n)
    abi[:n // 4] = 1.0
    abi[-n // 8:] = 0.5
    inits['asym_bimodal'] = abi / abi.sum()

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Sine bump (asymmetric)
    if n > 2:
        s = np.sin(np.pi * x**0.7 / (n**0.7))
        s = np.maximum(s, 0)
        if s.sum() > 1e-12:
            inits['sine_asym'] = s / s.sum()

    # Power law
    p = (x + 1) ** (-0.5)
    inits['power'] = p / p.sum()

    # Two-scale: large values at a few positions, small elsewhere
    ts = np.full(n, 0.01)
    peaks = rng.choice(n, size=max(2, n // 8), replace=False)
    ts[peaks] = 1.0
    inits['sparse'] = ts / ts.sum()

    return inits

                                             t_start=50.0, t_end=400.0, eta0=0.04)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Diversify: n={len(a_opt)}, score={score:.6f}")

    # Phase 4: Deep refinement on the overall best
    if best_sequence is not None and time.monotonic() < deadline - 15:
        print(f"Phase 4: Deep refinement. Best: {best_score:.6f}")
        remaining = deadline - time.monotonic() - 10
        n_chunks = 8
        chunk_time = remaining / n_chunks

        for i in range(n_chunks):
            if time.monotonic() > deadline - 10:
                break
            t_alloc = min(chunk_time, deadline - time.monotonic() - 10)
            if t_alloc < 2:
                break

            # Vary parameters for each chunk
            t_s = 50.0 + 50.0 * i
            t_e = 300.0 + 100.0 * i
            e0 = 0.04 * (0.85 ** i)

            a_opt, obj_opt = fista_optimize(best_sequence, time_limit=t_alloc,
                                             t_start=t_s, t_end=t_e, eta0=e0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep {i}: n={len(a_opt)}, score={score:.6f}")

        # Ultra-fine final pass
        if time.monotonic() < deadline - 5:
            t_remaining = deadline - time.monotonic() - 5
            if t_remaining > 3:
                a_opt, obj_opt = subgradient_descent(best_sequence, time_limit=t_remaining,
                                                      eta0=0.001, top_k=5, t_param=200.0)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final SGD: score={score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Clip to valid range
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    print(f"FINAL: n={len(best_sequence)}, score={best_score:.6f}")
    return best_sequence


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Main entry point. Uses FISTA with softmax-smoothed peak minimization,
    window-function initializations, spectral diversification, and discrete
    perturbations to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    result = run_solver(seed=seed, budget_s=budget_s)
    return [float(x) for x in result]
