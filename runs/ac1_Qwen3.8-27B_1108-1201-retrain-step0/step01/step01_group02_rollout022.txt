    start = time.monotonic()

    best_a = a.copy()
    best_score = score_sequence(a)

        if time.monotonic() - start > time_limit_s:
            break

            best_score = score
            best_a = a.copy()

        # grad f_smooth / a[j] = sum_k exp(tau*conv[k]) * dconv[k]/da[j] / (tau * sum exp(tau*conv[k]))

```python
"""
Multi-strategy optimization for minimizing peak auto-convolution of non-negative sequences.
Combines LP refinement, smoothed gradient descent, and local search with convolution guidance.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_blackman(n):
    if n == 1:
        return np.ones(1)
    x = np.linspace(0, 1, n)
    w = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    return np.maximum(0.0, w)


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_asymmetric(n, rng):
    """Create an asymmetric initial sequence."""
    t = np.arange(n, dtype=np.float64)
    # Asymmetric bump
    center = n * rng.uniform(0.2, 0.8)
    sigma = n * rng.uniform(0.1, 0.3)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    # Add asymmetry
    a *= (1.0 + rng.uniform(0.1, 0.5) * np.sin(2 * np.pi * t / n))
    return np.maximum(0.0, a)


def build_conv_matrix(a):
    """Build sparse matrix A such that A @ g = np.convolve(a, g)."""
    n = len(a)
    n_rows = 2 * n - 1
    row_parts = []
    col_parts = []
    data_parts = []
    
    for k in range(n):
        row_parts.append(np.full(k + 1, k, dtype=np.int32))
        col_parts.append(np.arange(k + 1, dtype=np.int32))
        data_parts.append(a[:k+1])
    
    for k in range(n, 2 * n - 1):
        j_min = k - n + 1
        cnt = n - j_min
        row_parts.append(np.full(cnt, k, dtype=np.int32))
        col_parts.append(np.arange(j_min, n, dtype=np.int32))
        data_parts.append(a[j_min:])
    
    row_idx = np.concatenate(row_parts)
    col_idx = np.concatenate(col_parts)
    data = np.concatenate(data_parts)
    
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp(a, M, time_limit=30.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None
