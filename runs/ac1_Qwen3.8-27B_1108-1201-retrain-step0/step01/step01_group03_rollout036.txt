    n = len(a)

    if S < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (S * S)

def normalize_to_sum(a, target):
    S = a.sum()
    if S < 1e-15:
        return a
    return a * (target / S)

def get_subgradient(a, k_star, epsilon=0.0):
    """Get subgradient of max_k (a*a)[k] at the given peak lag(s)."""
    n = len(a)
    c = np.convolve(a, a)
    peak = c.max()
    # Find all lags within epsilon of peak
    active_lags = np.where(c >= peak - epsilon)[0]

            # d(conv(a,a)[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)

    n = len(a)
    if n > n_max:
        return None

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based softmax subgradient descent, LP-based refinement,
and coordinate descent for final polishing. Multiple lengths explored.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (S * S)

def softmax_subgrad(a, tau=10.0):
    """Compute softmax-weighted subgradient of max_k (a*a)[k] using FFT."""
    n = len(a)
    c = np.convolve(a, a)
    peak = c.max()
    w = np.exp(tau * (c - peak))
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Fallback: use just the argmax
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        for j in range(max(0, k_star - n + 1), min(n - 1, k_star) + 1):
            grad[j] = 2.0 * a[k_star - j]
        return grad
    
    w /= w_sum
    
    # grad[j] = 2 * sum_{m=0}^{min(n-1, 2n-2-j)} w[m+j] * a[m]
    # This is 2 * cross_correlation(w, a)[j]
    # Using FFT: cross_correlation(x, y)[m] = sum_k x[k+m] * y[k]
    #         = ifft(conj(fft(x, N)) * fft(y, N))[m]
    
    N = 1
    while N < 3 * n:
        N *= 2
    
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(np.conj(W) * A, N)
    grad = 2.0 * corr[:n]
    return grad

def gradient_step(a, lr, tau=10.0):
    """One projected gradient step."""
    n = len(a)
    grad = softmax_subgrad(a, tau)
    # Project out mean component (for fixed-sum constraint)
    grad -= np.mean(grad)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-15:
        a_new = np.ones(n) / n
    else:
        a_new = a_new / S  # Normalize to sum=1
    return a_new

def lp_direction(a, M, n_max=2048):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M elementwise, g >= 0."""
    n = len(a)
    if n > n_max:
        return None
    
    c_obj = -np.ones(n)
    # Build A matrix: A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a[k-j] is valid when 0 <= k-j < n, i.e., j <= k <= j+n-1
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        A[k_lo:k_hi + 1, j] = a[0:k_hi - k_lo + 1]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def optimize_subgrad(a0, deadline, max_iters=5000, tau_start=5.0, tau_end=30.0):
    """Optimize using subgradient descent with annealed tau."""
    a = normalize_to_1(np.maximum(a0, 0.0))
    n = len(a)
    S_target = 1.0
    best = a.copy()
    best_score = compute_obj(a)
    
    lr = 0.1
    lr_min = 0.001
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        tau = tau_start + (tau_end - tau_start) * (it / max_iters)
        # Cosine decay of learning rate
        lr = lr_min + 0.5 * (0.1 - lr_min) * (1 + np.cos(np.pi * it / max_iters))
        
        a = gradient_step(a, lr, tau)
        
        if (it + 1) % 200 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    
    return best, best_score

def optimize_lp(a0, deadline, max_iters=200):
    """Optimize using LP-based refinement."""
    a = normalize_to_1(np.maximum(a0, 0.0))
    n = len(a)
    best = a.copy()
    best_score = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a)
        M = float(c.max())
        
        g = lp_direction(a, M)
        if g is None:
            # Try with a slightly larger M
            g = lp_direction(a, M * 1.01)
            if g is None:
                break
        
        if g.sum() < 1e-10:
            break
        
        # Line search for best blend
        improved = False
        for t in [0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to_1(a_new)
            s_new = compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break
        
        if not improved:
            # Try a small gradient step to escape
            a = gradient_step(a, 0.02, tau=15.0)
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
    
    return best, best_score

def normalize_to_1(a):
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S

def coordinate_descent(a, deadline, max_sweeps=5):
    """Simple coordinate descent: try small perturbations to individual elements."""
    a = normalize_to_1(np.maximum(a, 0.0))
    n = len(a)
    best = a.copy()
    best_score = compute_obj(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            # Try increasing and decreasing a[j]
            for delta_frac in [0.01, 0.05, 0.1, -0.01, -0.05, -0.1]:
                a_new = a.copy()
                a_new[j] = max(0.0, a[j] * (1 + delta_frac))
                if a_new.sum() < 1e-15:
                    continue
                a_new = normalize_to_1(a_new)
                s_new = compute_obj(a_new)
                if s_new < best_score:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    improved = True
                    break
    
    return best, best_score

    A[k_lo:k_hi+1, j] = a[0:k_hi-k_lo+1]

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based softmax subgradient descent, LP-based refinement,
perturbation restarts, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (S * S)


def normalize_to_1(a):
    """Normalize array to sum to 1."""
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def softmax_subgrad_fft(a, tau=10.0):
    """Compute softmax-weighted subgradient of max_k (a*a)[k] using FFT."""
    n = len(a)
    c = np.convolve(a, a)
    peak = c.max()
    w = np.exp(tau * (c - peak))
    w_sum = w.sum()
    if w_sum < 1e-30:
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        for j in range(j_lo, j_hi + 1):
            grad[j] = 2.0 * a[k_star - j]
        return grad
    w /= w_sum
    # grad[j] = 2 * cross_correlation(w, a)[j]
    # cross_correlation(x, y)[j] = sum_m x[m+j] * y[m]
    # = ifft(conj(fft(x, N)) * fft(y, N))[j]
    N = 1
    need = 3 * n
    while N < need:
        N *= 2
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(np.conj(W) * A, N)
    grad = 2.0 * corr[:n]
    return grad


def gradient_step(a, lr, tau=10.0):
    """One projected gradient step (fixed sum, non-negative)."""
    n = len(a)
    grad = softmax_subgrad_fft(a, tau)
    grad -= np.mean(grad)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    return normalize_to_1(a_new)


def lp_direction(a, M, n_max=2048):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M elementwise, g >= 0."""
    n = len(a)
    if n > n_max:
        return None
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        A[k_lo:k_hi + 1, j] = a[:k_hi - k_lo + 1]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_subgrad(a0, deadline, max_iters=5000, tau_start=5.0, tau_end=40.0):
    """Subgradient descent with annealed tau and cosine LR decay."""
    a = normalize_to_1(np.maximum(np.asarray(a0, dtype=np.float64), 0.0))
    n = len(a)
    best = a.copy()
    best_score = compute_obj(a)
    lr_max = 0.12
    lr_min = 0.002
    patience = 0
    patience_limit = 1500

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start + (tau_end - tau_start) * frac
        lr = lr_min + 0.5 * (lr_max - lr_min) * (1.0 + np.cos(np.pi * frac))
        a = gradient_step(a, lr, tau)
        if (it + 1) % 300 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
                patience = 0
            else:
                patience += 300
            if patience >= patience_limit:
                # Add perturbation to escape
                noise = np.random.default_rng(it).normal(0, 0.02, n)
                a = np.maximum(a + noise, 0.0)
                a = normalize_to_1(a)
                patience = 0
    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    return best, best_score


def optimize_lp_refine(a0, deadline, max_iters=100):
    """LP-based refinement: alternately solve LP and blend."""
    a = normalize_to_1(np.maximum(np.asarray(a0, dtype=np.float64), 0.0))
    n = len(a)
    best = a.copy()
    best_score = compute_obj(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = np.convolve(a, a)
        M = float(c.max())
        g = lp_direction(a, M)
        if g is None or g.sum() < 1e-10:
            break
        improved = False
        for t in [0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to_1(a_new)
            s_new = compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break
        if not improved:
            # Try gradient step to escape
            a = gradient_step(a, 0.015, tau=20.0)
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
    return best, best_score


def optimize_combined(a0, deadline, subgrad_time_frac=0.5):
    """Combined subgradient + LP optimization."""
    t_now = time.monotonic()
    total_time = max(1.0, deadline - t_now)
    subgrad_deadline = t_now + total_time * subgrad_time_frac
    lp_deadline = deadline

    seq, score = optimize_subgrad(a0, subgrad_deadline, max_iters=4000)
    print(f"    After subgrad: score={score:.6f}")

    seq2, score2 = optimize_lp_refine(seq, lp_deadline, max_iters=80)
    print(f"    After LP: score={score2:.6f}")

    if score2 < score:
        return seq2, score2
    return seq, score


def get_initial_sequence():

    Main search function. Explores multiple sequence lengths using

    t_start = time.monotonic()

    best_seq = None

    def update_best(seq, score):

    # Phase 1: Refine the known best (n=512) with subgradient + LP
    print("Phase 1: Refining n=512 from known best...")
    a512 = a_init[:512].copy() if len(a_init) >= 512 else np.pad(a_init, (0, 512 - len(a_init)))
    phase1_deadline = t_start + 600.0
    seq, score = optimize_combined(a512, phase1_deadline, subgrad_time_frac=0.5)
    update_best(seq, score)

    # Phase 2: Continue refining with more LP iterations
    print("Phase 2: Further LP refinement...")

    phase3_deadline = t_start + 1400.0

        noise_scale = [0.01, 0.02, 0.03, 0.05][restart]

        a_perturbed = normalize_to_1(a_perturbed)

        lp_deadline = min(phase4_deadline, time.monotonic() + 60.0)
