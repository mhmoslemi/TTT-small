"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LP-based alternating iterations
with simulated annealing perturbations to escape local minima.
Multiple starting points of varying lengths are explored.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize to target sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = float(np.sum(a))
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def _build_conv_matrix(f):
    """Build matrix A where (A @ g)[m] = (f*g)[m] for full convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    return A


def _lp_solve(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    n = len(f)
    A = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_iterate(a_init, max_iters, deadline, rng, verbose=False):

    perturb_count = 0

        if time.monotonic() > deadline:
            break

def _gradient_step(a, lr, n_active=5):
    """
    Subgradient step on peak autocorrelation.
    Uses the top n_active lag positions for a softer gradient.
    """

    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t)
    
    c = np.convolve(a, a)
    c_max = c.max()
    
    # Get top n_active positions
    idx = np.argsort(c)[::-1][:n_active]
    
    grad = np.zeros(n)
    for m_star in idx:
        m_star = int(m_star)
        lo = max(0, m_star - n + 1)
        hi = min(n - 1, m_star)
        for j in range(lo, hi + 1):
            grad[j] += a[m_star - j]
    
    # Normalize gradient
    grad_norm = np.linalg.norm(grad)
    if grad_norm > 1e-12:
        grad = grad / grad_norm
    
    # Project out mean (to maintain sum)
    grad = grad - np.mean(grad)
    
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = _normalize(a, S_t)
    return a

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LP-based alternating iterations
with simulated annealing perturbations to escape local minima.
Uses multiple diverse starting points and adaptive length selection.
Core algorithm: solve LP max sum(g) s.t. conv(f,g)<=M, g>=0, blend result back.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute the objective value: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize to target sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = float(np.sum(a))
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def _build_conv_matrix(f):
    """Build matrix A where (A @ g)[m] = (f*g)[m] for full convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    return A


def _lp_solve(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    n = len(f)
    A = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_iterate(a_init, max_iters, deadline, rng):
    """
    Run LP-based iterations to minimize peak autocorrelation.
    Uses simulated annealing perturbations to escape local minima.
    Returns (best_sequence, best_score).
    """
    n = len(a_init)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(np.maximum(a_init, 0.0), S_t).copy()

    c = np.convolve(a, a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    best_score = _compute_obj(best_a)

    no_improve = 0
    perturb_count = 0
    temperature = 0.1  # Annealing temperature

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Solve LP
        g = _lp_solve(a, M)
        if g is None:
            break

        g_sum = float(np.sum(g))
        if g_sum < 1e-12:
            break

        g = g * (S_t / g_sum)

        # Try to blend with decreasing t
        improved = False
        for t in [1.0, 0.8, 0.5, 0.3, 0.15, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, S_t)
            c_new = np.convolve(a_new, a_new)
            M_new = float(np.max(c_new))
            if M_new < M * (1.0 - 1e-10):
                a = a_new
                M = M_new
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                    best_score = _compute_obj(best_a)
                improved = True
                no_improve = 0
                temperature = max(0.01, temperature * 0.98)
                break

        if not improved:
            no_improve += 1

            # Perturbation to escape local minimum
            if no_improve >= 2:
                perturb_count += 1
                strategy = perturb_count % 4

                if strategy == 1:
                    # Additive Gaussian noise
                    noise = rng.normal(0, temperature * float(np.mean(a)), n)
                    a_pert = a + noise
                elif strategy == 2:
                    # Multiplicative noise
                    a_pert = a * (1.0 + temperature * (rng.random(n) - 0.5) * 2)
                elif strategy == 3:
                    # Interpolate with random sequence
                    a_rand = rng.random(n) * 0.5 + 0.25
                    a_rand = _normalize(a_rand, S_t)
                    mix = min(0.3, temperature * 2)
                    a_pert = (1.0 - mix) * a + mix * a_rand
                else:
                    # Random pair swap
                    a_pert = a.copy()
                    idx = rng.permutation(n)
                    for _ in range(max(1, n // 32)):
                        i, j = idx[_], idx[(_ + 1) % n]
                        a_pert[i], a_pert[j] = a_pert[j], a_pert[i]

                a_pert = np.maximum(a_pert, 0.0)
                a_pert = _normalize(a_pert, S_t)
                c_pert = np.convolve(a_pert, a_pert)
                M_pert = float(np.max(c_pert))

                # Accept with annealing criterion
                if M_pert <= M or (rng.random() < np.exp(-(M_pert - M) / max(temperature * 0.01, 1e-6))):
                    a = a_pert
                    M = M_pert
                    temperature = max(0.01, temperature * 0.95)

                no_improve = 0

            if no_improve >= 40:
                break

    return best_a, best_score


def _gradient_polish(a_init, deadline, rng, max_iters=2000):
    """
    Subgradient-based polish using top-k active constraints.
    Returns (best_sequence, best_score).
    """
    n = len(a_init)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(np.maximum(a_init, 0.0), S_t).copy()

    best_a = a.copy()
    best_score = _compute_obj(best_a)
    best_M = float(np.max(np.convolve(a, a)))

    lr = 0.08
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = np.convolve(a, a)
        M = float(np.max(c))

        # Use top-k peaks for smoother gradient
        k = max(1, min(5, n // 32))
        top_idx = np.argsort(c)[::-1][:k]
        weights = np.ones(k) / k

        grad = np.zeros(n)
        for w, m_star in zip(weights, top_idx):
            m_star = int(m_star)
            lo = max(0, m_star - n + 1)
            hi = min(n - 1, m_star)
            for j in range(lo, hi + 1):
                grad[j] += w * a[m_star - j]

        # Project out mean component
        grad = grad - np.mean(grad)

        # Adaptive learning rate
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            a = a - lr * grad / gnorm * (M / max(S_t, 1e-6))
            a = np.maximum(a, 0.0)
            a = _normalize(a, S_t)

        c_new = np.convolve(a, a)
        M_new = float(np.max(c_new))
        if M_new < best_M * (1.0 - 1e-8):
            best_M = M_new
            best_a = a.copy()
            best_score = _compute_obj(best_a)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Perturb and reset
                a = a + rng.normal(0, 0.02 * np.mean(a), n)
                a = np.maximum(a, 0.0)
                a = _normalize(a, S_t)
                no_improve = 0
                lr = max(0.01, lr * 0.7)

        # Decay learning rate
        lr = max(0.005, lr * 0.999)

    return best_a, best_score


def _generate_starts(rng, provided_seq=None):
    """Generate diverse starting sequences of various lengths."""
    starts = []

    if provided_seq is not None and len(provided_seq) >= 16:
        for L in [64, 128, 256, 512, min(1024, len(provided_seq))]:
            if L <= len(provided_seq):
                starts.append(provided_seq[:L].copy())

    # Random starts
    for n in [128, 256, 512, 1024]:
        starts.append(rng.random(n) * 0.5 + 0.25)

    # Structured starts
    for n in [128, 256, 512]:
        i = np.arange(n, dtype=np.float64)
        # Various chirps
        for freq in [1.0, 3.0, 5.0, 7.0]:
            starts.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * i / n))
        # Gaussian
        center = n / 2.0
        starts.append(np.exp(-0.5 * ((i - center) / (n * 0.25)) ** 2))
        # Two-lobe
        starts.append(np.exp(-0.5 * ((i - n * 0.25) / (n * 0.12)) ** 2) +
                      np.exp(-0.5 * ((i - n * 0.75) / (n * 0.12)) ** 2))
        # Linear ramp
        starts.append(np.linspace(0.2, 0.8, n))
        # Bimodal
        starts.append(0.3 + 0.7 * np.abs(np.sin(np.pi * i / n)))

    # Sort by length (shorter first for quick evaluation)
    starts.sort(key=len)
    return starts
