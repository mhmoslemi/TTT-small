import numpy as np
import time
import cvxpy as cp

def _compute_score_normalized(a):
    """Compute 2*n*max(convolve(a,a))/sum(a)^2 for non-negative a."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (S * S)

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        k_idx = np.argpartition(b, -M)[-M:]
        k_sel = b[k_idx]

    grad -= np.mean(grad)
    
    # Step

    a_new = _project_simplex(a_new, 1.0)

    """
    n = len(a)
    M_peak = np.max(b)
    
    # Get top lags
    if M_lags >= len(b):
        lag_idx = np.arange(len(b))
    else:
        lag_idx = np.argpartition(b, -M_lags)[-M_lags:]
    
    # Build LP: maximize sum(g) subject to:
    #   A @ g <= b_ub (conv constraints)
    #   g >= 0

    # The coefficient of g_j in constraint k is: a_{k-j} (if valid)

    # A has shape (len(lag_idx), n)
    # A[r, j] = a[k_r - j] if 0 <= k_r - j < n, else 0
    
    A = np.zeros((len(lag_idx), n))

Multi-phase optimizer for minimizing normalized autocorrelation peak of
non-negative sequences. Combines adaptive subgradient descent, LP-based
direction finding, and diversification to improve upon previous SOTA.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_score(a):
    """Compute 2*n*max(convolve(a,a))/sum(a)^2 for non-negative a."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (S * S)


def _compute_peak_score(a):
    """Assume sum(a)=1, return 2*n*max(convolve(a,a))."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_weighted_peaks(a, b, M, beta):
    """Compute weighted gradient of top-M autocorrelation peaks."""
    n = len(a)
    b_max = np.max(b)
    if M >= len(b):
        k_idx = np.arange(len(b))
        wts = np.ones(len(b)) / len(b)
    else:
        k_idx = np.argpartition(b, -M)[-M:]
        k_sel = b[k_idx]
        wts = np.exp(beta * (k_sel - b_max))
        w_sum = np.sum(wts)
        if w_sum < 1e-30:
            wts = np.ones(len(k_idx)) / len(k_idx)
        else:
            wts = wts / w_sum

    grad = np.zeros(n)
    for i, k in enumerate(k_idx):
        c = wts[i] * 2.0
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += c * a[k - idx]

    # Center for equality constraint sum(a)=1
    grad -= np.mean(grad)
    return grad


def _subgradient_step(a, b, M, beta, eta):
    """One subgradient step with projection onto simplex."""
    n = len(a)
    grad = _grad_weighted_peaks(a, b, M, beta)
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return a
    a_new = a - eta * grad
    a_new = _project_simplex(a_new, 1.0)
    return a_new


def _lp_direction(a, b, num_lags, max_g=10.0):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, g <= max_g, conv(a,g)_k <= peak for top lags.
    Returns normalized direction or None.
    """
    n = len(a)
    peak_val = np.max(b)

    if num_lags >= len(b):
        lag_idx = np.arange(len(b))
    else:
        lag_idx = np.argpartition(b, -num_lags)[-num_lags:]

    # Build constraint matrix: for each lag k, sum_j a[k-j] * g_j <= peak
    # Coefficient of g_j in row r (lag k_r) is a[k_r - j] if valid
    A_ub = np.zeros((len(lag_idx), n))
    for r, k in enumerate(lag_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # g indices from lo to hi, coefficient is a[k - j]
            js = np.arange(lo, hi + 1)
            A_ub[r, js] = a[k - js]

    b_ub = np.full(len(lag_idx), peak_val)
    c_vec = -np.ones(n)  # minimize -sum(g)
    bounds = [(0, max_g)] * n

    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': 15, 'presolve': True})
        if res.success and res.x is not None:
            g = res.x
            S = np.sum(g)
            if S > 1e-10:
                return g / S
    except Exception:
        pass
    return None


def _smooth_perturb(a, rng, scale):
    """Add smooth low-frequency random perturbation and re-project."""
    n = len(a)
    freqs = rng.integers(1, max(2, n // 8), size=min(8, n // 4))
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2.0 * np.pi)
        amp = rng.uniform(-1.0, 1.0)
        noise += amp * np.cos(2.0 * np.pi * f * np.arange(n) / n + phase)
    max_abs = np.max(np.abs(noise))
    if max_abs > 1e-15:
        noise /= max_abs
    a_new = a + scale * noise
    return _project_simplex(a_new, 1.0)


def _make_two_level(n, eps):
    """Anti-symmetric two-level sequence."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_smooth_profile(n, eps):
    """Smooth anti-symmetric: 1 + eps*cos(pi*i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / max(1, n - 1))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_geometric(n):
    """Geometric sequence with r^(n-1) = n (balanced peak)."""
    r = np.exp(np.log(n) / (n - 1))
    a = r ** np.arange(n, dtype=np.float64)
    S = np.sum(a)
    return a / S


def _run_subgradient(a_init, deadline, rng, eta_init, beta_init, M_peaks,
                     patience, tag=""):
    """
    Run subgradient descent until deadline.
    Returns (best_a, best_score).
    """
    n = len(a_init)
    a = a_init.copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    best_a = a.copy()
    best_s = _compute_peak_score(best_a)
    eta = eta_init
    beta = beta_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        cur_s = 2.0 * n * np.max(b)

        if cur_s < best_s - 1e-12:
            best_s = cur_s
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            eta *= 0.75
            beta = min(beta * 1.2, 50.0)
            no_improve = 0
            if eta < 1e-11:
                break

        a = _subgradient_step(a, b, M_peaks, beta, eta)

        # Simulated annealing: occasionally accept slightly worse solutions
        if iter_count > 100 and iter_count % 50 == 0:
            b_check = np.convolve(a, a)
            s_check = 2.0 * n * np.max(b_check)
            if s_check > best_s:
                delta = (s_check - best_s) / best_s
                if delta < 0.005 and rng.random() < np.exp(-50.0 * delta):
                    pass  # Keep the worse solution (annealing)
                else:
                    a = best_a.copy()

        iter_count += 1

    return best_a, best_s


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Adaptive subgradient from multiple starts
    Phase 2: LP-assisted direction finding + subgradient
    Phase 3: Diversification with smooth perturbations
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)
    total_time = deadline - t0

    # Get initial sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size

    # Normalize parent
    a_parent = parent.copy()
    S_p = np.sum(a_parent)
    if S_p > 1e-10:
        a_parent = a_parent / S_p
    else:
        a_parent = np.ones(n) / n

    # Global best
    global_best = a_parent.copy()
    global_best_score = _compute_peak_score(global_best)
    print(f"n={n}, Initial score: {2*n*global_best_score/(np.sum(global_best)**2):.6f}"
          if np.sum(global_best) != 1 else f"n={n}, Initial score: {global_best_score:.6f}")
    # Actually since sum=1, score = 2*n*max_b
    print(f"Initial peak score (sum=1): {global_best_score:.6f}")

    # === Phase 1: Multi-start subgradient (45% time) ===
    phase1_end = t0 + 0.45 * total_time
    print(f"\n=== Phase 1: Multi-start subgradient ({(phase1_end-t0):.0f}s) ===")

    # Prepare starting points
    starts = [a_parent]
    starts.append(_make_two_level(n, 0.5))
    starts.append(_make_smooth_profile(n, 0.7))
    starts.append(_make_geometric(n))
    # Random smooth start
    a_rand = np.ones(n) / n
    a_rand = _smooth_perturb(a_rand, rng, 0.3)
    starts.append(a_rand)

    # Allocate time: parent gets more
    n_starts = len(starts)
    time_per_start = (phase1_end - t0) / n_starts
    time_parent = time_per_start * 2
    time_others = (phase1_end - t0 - time_parent) / (n_starts - 1)

    for i, a_start in enumerate(starts):
        if time.monotonic() >= phase1_end:
            break
        t_alloc = time_parent if i == 0 else time_others
        local_deadline = min(time.monotonic() + t_alloc, phase1_end)

        eta_init = 0.5 / np.sqrt(n)
        best_a, best_s = _run_subgradient(
            a_start, local_deadline, rng,
            eta_init=eta_init, beta_init=4.0, M_peaks=25,
            patience=300, tag=f"start{i}"
        )
        if best_s < global_best_score - 1e-12:
            global_best_score = best_s
            global_best = best_a.copy()
            print(f"  Start {i+1}: NEW BEST! Score: {global_best_score:.6f}")
        else:
            print(f"  Start {i+1}: Score {best_s:.6f} (best={global_best_score:.6f})")

    # === Phase 2: LP-assisted optimization (35% time) ===
    phase2_end = t0 + 0.80 * total_time
    remaining2 = phase2_end - time.monotonic()
    print(f"\n=== Phase 2: LP-assisted ({remaining2:.0f}s) ===")

    if remaining2 > 5:
        a = global_best.copy()
        eta = 0.3 / np.sqrt(n)
        beta = 5.0
        M_peaks = 30
        lp_interval = 80  # try LP every 80 subgradient steps
        step_count = 0
        no_improve = 0

        while time.monotonic() < phase2_end:
            b = np.convolve(a, a)
            cur_s = 2.0 * n * np.max(b)

            if cur_s < global_best_score - 1e-12:
                global_best_score = cur_s
                global_best = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Subgradient step
            a = _subgradient_step(a, b, M_peaks, beta, eta)
            step_count += 1

            # Periodic parameter adjustment
            if no_improve > 200:
                eta *= 0.7
                beta = min(beta * 1.15, 40.0)
                no_improve = 0

            # LP direction finding
            if step_count % lp_interval == 0 and time.monotonic() < phase2_end - 10:
                b_cur = np.convolve(a, a)
                g = _lp_direction(a, b_cur, num_lags=80)
                if g is not None:
                    # Try different mix ratios
                    for t_mix in [0.2, 0.4, 0.6]:
                        a_try = (1 - t_mix) * a + t_mix * g
                        a_try = _project_simplex(a_try, 1.0)
                        s_try = _compute_peak_score(a_try)
                        if s_try < global_best_score - 1e-12:
                            global_best_score = s_try
                            global_best = a_try.copy()
                            a = a_try.copy()
                            no_improve = 0
                            print(f"  LP (t={t_mix}): NEW BEST! Score: {global_best_score:.6f}")
                            break
                        elif s_try < cur_s:
                            a = a_try.copy()
                            break

    # === Phase 3: Diversification and polish (remaining time) ===
    remaining3 = deadline - time.monotonic()
    print(f"\n=== Phase 3: Diversification ({remaining3:.0f}s) ===")

    if remaining3 > 20:
        # First: polish the best with very fine steps
        polish_time = min(remaining3 * 0.4, 300.0)
        if polish_time > 10:
            polish_deadline = time.monotonic() + polish_time
            best_a_p, best_s_p = _run_subgradient(
                global_best, polish_deadline, rng,
                eta_init=0.1 / np.sqrt(n), beta_init=8.0,
                M_peaks=50, patience=500, tag="polish"
            )
            if best_s_p < global_best_score - 1e-12:
                global_best_score = best_s_p
                global_best = best_a_p.copy()
                print(f"  Polish: NEW BEST! Score: {global_best_score:.6f}")

        # Then: diversification
        div_time = deadline - time.monotonic()
        if div_time > 30:
            n_div = max(3, int(div_time / 40))
            per_div = div_time / n_div
            print(f"  Running {n_div} diversification rounds, {per_div:.1f}s each")

            for d in range(n_div):
                if time.monotonic() >= deadline:
                    break

                # Generate diverse perturbation
                scale = 0.05 + 0.1 * (d % 3) / 3.0
                a_pert = _smooth_perturb(global_best, rng, scale)

                # Also try a structural change: flip a random segment
                if d % 2 == 1:
                    seg_len = rng.integers(n // 10, n // 3)
                    seg_start = rng.integers(0, n - seg_len)
                    a_seg = a_pert[seg_start:seg_start + seg_len]
                    a_pert[seg_start:seg_start + seg_len] = a_seg[::-1]
                    a_pert = _project_simplex(a_pert, 1.0)

                div_deadline = min(time.monotonic() + per_div, deadline)
                best_a_d, best_s_d = _run_subgradient(
                    a_pert, div_deadline, rng,
                    eta_init=0.3 / np.sqrt(n), beta_init=5.0,
                    M_peaks=30, patience=200, tag=f"div{d}"
                )
                if best_s_d < global_best_score - 1e-12:
                    global_best_score = best_s_d
                    global_best = best_a_d.copy()
                    print(f"  Div {d+1}/{n_div}: NEW BEST! Score: {global_best_score:.6f}")
                else:
                    print(f"  Div {d+1}/{n_div}: Score {best_s_d:.6f}")

    # Final result
    result = [float(x) for x in global_best]
    elapsed = time.monotonic() - t0
    final_score = float(evaluate_sequence(result))
    print(f"\nTotal time: {elapsed:.1f}s")
    print(f"FINAL SCORE: {final_score:.6f}")
    return result
