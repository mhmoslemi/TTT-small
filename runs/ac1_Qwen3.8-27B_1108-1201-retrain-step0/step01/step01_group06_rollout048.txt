def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_obj(a):
    """Compute 2n * max(conv) / sum^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def multiplicative_update(a, t=50.0, eta=0.3, top_k=0):
    """
    One step of multiplicative mirror descent.
    a: current sequence (non-negative, sum=1)
    t: temperature for peak smoothing (higher = focus more on the single peak)
    eta: step size
    top_k: if > 0, only use top-k peaks (0 = use all)
    Returns updated a (sum=1).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return a
    
    # Compute weights
    if top_k > 0 and top_k < len(conv):
        # Only use top-k peaks
        indices = np.argsort(conv)[-top_k:]
        w = np.zeros(len(conv))
        w[indices] = conv[indices]
        # Normalize
        w_max = w.max()
        scaled = t * w / w_max - t
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[w.argmax()] = 1.0
        else:
            w /= w_sum
    else:
        scaled = t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

    sg = np.zeros(n)

        k = int(k_int)

        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        wk = w[k]

        # a[k-j] for j in [j_start, j_end] is a[k-j_start : k-j_end+1][::-1]
        # But we need 2* this (since a[i] appears twice in conv[k] for i != k-i)
        # Actually, the gradient of conv[k] w.r.t. a[i] is:
        # 2*a[k-i] if i != k-i (both in range)
        # 2*a[i] if i = k-i (i.e., k = 2i)
        # For the vector form: sg[j] += 2 * wk * a[k-j] for j in range
        # But when k=2j, a[k-j] = a[j], and the gradient is 2*a[j], which is the same formula.
        # So: sg[j] += 2 * wk * a[k-j] for all valid j.
        # 
        # Wait, I need to double-check. conv[k] = sum_j a[j]*a[k-j].
        # d/d a[i] conv[k] = a[k-i] (j=i term) + a[k-i] (j=k-i term) = 2*a[k-i] if i != k-i
        # or 2*a[i] if i = k-i.
        # In both cases: 2*a[k-i] (since when i=k-i, a[k-i]=a[i]).
        # So the gradient is simply 2*a[k-i] for all valid i.
        
        # For j in [j_start, j_end], k-j ranges from k-j_start to k-j_end.
        sg[j_start:j_end+1] += 2.0 * wk * a[k-j_end:k-j_start+1][::-1]
    
    # Multiplicative update
    sg_centered = sg - sg.mean()
    log_factor = -eta * sg_centered
    log_factor -= log_factor.max()
    a_new = a * np.exp(log_factor)
    a_new = np.maximum(a_new, 0.0)

    else:

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Gaussians
    for sf in [0.08, 0.12, 0.15, 0.2, 0.25, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - n/2) / (sf * n))**2)
        # Truncate to [0, n-1]
        g[x < 0] = 0
        g[x >= n] = 0
        s = g.sum()
        if s > 1e-10:
            inits[f'gauss_{sf:.2f}'] = g / s
    
    # Ramp
    ramp = np.arange(1, n+1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    ramp_rev = np.arange(n, 0, -1, dtype=np.float64)
    inits['ramp_rev'] = ramp_rev / ramp_rev.sum()
    
    # Bimodal (edges)
    bi = np.zeros(n)
    width = max(1, n // 6)
    bi[:width] = 1.0
    bi[-width:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Cosine
    cos_bump = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cosine'] = cos_bump / cos_bump.sum()
    
    # Alternating with different ratios
    alt1 = np.zeros(n)
    alt1[::2] = 2.0
    alt1[1::2] = 1.0
    inits['alt_2_1'] = alt1 / alt1.sum()
    
    alt2 = np.zeros(n)
    alt2[::2] = 3.0
    alt2[1::2] = 1.0
    inits['alt_3_1'] = alt2 / alt2.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Random smooth
    noise = rng.standard_normal(n)
    # Smooth with a window
    from scipy.ndimage import gaussian_filter1d
    smooth = gaussian_filter1d(noise, sigma=max(1.0, n*0.05))
    smooth = np.abs(smooth)
    s = smooth.sum()
    if s > 1e-10:
        inits['random_smooth'] = smooth / s
    
    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].astype(np.float64).copy()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        pp = np.maximum(pp, 0)
        s = pp.sum()
        if s > 1e-10:
            inits['parent'] = pp / s
    
    return inits
