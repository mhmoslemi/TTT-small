def mirror_descent_optimize(a0, time_limit, n_iter_max=1000000):
    """
    Nesterov-accelerated mirror descent on the probability simplex.
    Minimizes max_k conv_k(a) where conv_k is the auto-convolution at lag k.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autoconvolve(a).max()
    
    x = a.copy()
    x_prev = a.copy()
    t = 2.0
    
    eta = 2.0 / np.sqrt(n)
    threshold = 0.99
    patience = 0
    max_patience = 300
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-12:
            y = x.copy()
        else:
            y = y / y_sum
        t = t_new
        
        # Evaluate at y
        conv_y = fast_autoconvolve(y)
        obj_y = 2.0 * n * conv_y.max()
        
        if obj_y < best_obj - 1e-12:
            best_obj = obj_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1
        
        # Compute subgradient
        grad = subgradient_peak(conv_y, y, threshold)
        grad_centered = grad - grad.mean()
        
        # Mirror descent update
        log_factor = -eta * grad_centered
        log_factor -= log_factor.max()
        factor = np.exp(log_factor)
        x_new = x * factor
        x_new_sum = x_new.sum()
        if x_new_sum < 1e-12:
            x_new = np.ones(n) / n
        else:
            x_new = x_new / x_new_sum
        
        x_prev = x
        x = x_new
        
        # Adaptive step size
        if patience >= max_patience:
            patience = 0
            eta *= 0.6
            if eta < 0.005:
                eta = 2.0 / np.sqrt(n)
                threshold = min(1.0, threshold + 0.005)
                max_patience = 150
            else:
                threshold = max(0.9, threshold - 0.01)
    
    return best_a, best_obj

"""
Minimizes 2*n*max(autoconv(a))/(sum(a))^2 using Nesterov-accelerated mirror
descent with efficient peak-subgradient computation on the probability simplex.
Multi-start over sequence lengths with diverse initializations, adaptive
step-size, and periodic restarts. Exploits the structure that the subgradient
at peak k is simply 2*reversed-truncated(a).

import numpy as np
import time


def fast_autoconvolve(a):
    """Compute linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def subgradient_single_peak(a, k):
    """
    Efficient subgradient of conv_k at point a.
    k is the convolution index in [0, 2n-2].
    Returns gradient vector g where g[i] = 2*a[k-i] if valid, else 0.
    """
    n = len(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    g = np.zeros(n)
    start = max(0, k - (n - 1))
    end = min(n - 1, k)
    g[i_start:i_end + 1] = 2.0 * a[start:end + 1][::-1]
    return g


def subgradient_weighted(a, conv, top_k=8, temperature=0.5):
    """
    Weighted subgradient using top-k peaks with softmax weighting.
    """
    n = len(a)
    max_val = conv.max()
    if max_val < 1e-15:
        return np.zeros(n)
    
    # Get top-k peaks
    sorted_idx = np.argsort(conv)[::-1][:top_k]
    peak_vals = conv[sorted_idx]
    
    # Softmax weighting
    scaled = peak_vals / max_val
    weights = np.exp(temperature * (scaled - 1.0))
    w_sum = weights.sum()
    if w_sum < 1e-15:
        weights = np.zeros(top_k)
        weights[0] = 1.0
    else:
        weights /= w_sum
    
    # Compute weighted subgradient
    g = np.zeros(n)
    for idx, w in zip(sorted_idx, weights):
        g += w * subgradient_single_peak(a, int(idx))
    
    return g


def mirror_descent_optimize(a0, time_limit, top_k=8, temperature=0.5,
                            use_nesterov=True, eta_scale=1.0):
    """
    Mirror descent (exponential weights) on the probability simplex.
    Minimizes max_k conv_k(a) with optional Nesterov acceleration.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    # Initial eval
    conv = fast_autoconvolve(a)
    obj = 2.0 * n * conv.max()
    best_obj = obj
    best_a = a.copy()

    x = a.copy()
    x_prev = a.copy()
    if use_nesterov:
        t_val = 2.0
    else:
        t_val = 1.0

    eta = eta_scale * 2.0 / np.sqrt(n)
    patience = 0
    max_patience = 250
    restart_count = 0
    max_restarts = 20
    t0 = time.monotonic()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        # Nesterov extrapolation
        if use_nesterov:
            t_prev = t_val
            t_val = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
            theta = (t_prev - 1.0) / t_val
            y = x + theta * (x - x_prev)
            y = np.maximum(y, 0.0)
            y_sum = y.sum()
            if y_sum < 1e-12:
                y = x.copy()
            else:
                y = y / y_sum
        else:
            y = x

        # Evaluate at y
        conv_y = fast_autoconvolve(y)
        obj_y = 2.0 * n * conv_y.max()

        if obj_y < best_obj - 1e-12:
            best_obj = obj_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        # Compute subgradient
        grad = subgradient_weighted(y, conv_y, top_k=top_k, temperature=temperature)
        grad_centered = grad - grad.mean()

        # Mirror descent update
        log_factor = -eta * grad_centered
        log_max = log_factor.max()
        if log_max > 50:
            log_factor = log_factor - log_max
        factor = np.exp(log_factor)
        x_new = x * factor
        x_new_sum = x_new.sum()
        if x_new_sum < 1e-12:
            x_new = np.ones(n) / n
        else:
            x_new = x_new / x_new_sum

        x_prev = x
        x = x_new

        # Adaptive step size
        if patience >= max_patience:
            patience = 0
            eta *= 0.6
            if eta < 0.003:
                # Restart with perturbation
                restart_count += 1
                if restart_count > max_restarts:
                    break
                eta = eta_scale * 2.0 / np.sqrt(n)
                # Perturb the best solution
                if use_nesterov:
                    t_val = 2.0
                noise = np.random.default_rng(iter_count).dirichlet(np.ones(n) * 0.3, size=1)[0]
                x = 0.7 * best_a + 0.3 * noise
                x = np.maximum(x, 0)
                x /= x.sum()
                x_prev = x.copy()
                top_k = min(20, top_k + 4)
                temperature = max(0.1, temperature * 0.8)
                max_patience = 150

    return best_a, best_obj


def get_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # U-shaped (concentrated at edges)
    u = 1.0 + 0.8 * np.cos(2.0 * np.pi * x / n)
    u = np.maximum(u, 0.01)
    u /= u.sum()
    inits['u_shape'] = u

    # Strong U
    su = 1.0 + 2.0 * np.cos(2.0 * np.pi * x / n)
    su = np.maximum(su, 0.01)
    su /= su.sum()
    inits['strong_u'] = su

    # Centered
    c = 1.0 - 0.5 * np.cos(2.0 * np.pi * x / n)
    c = np.maximum(c, 0.01)
    c /= c.sum()
    inits['centered'] = c

    # Gaussian
    g = np.exp(-0.5 * ((x - n / 2) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gaussian'] = g

    # Wide Gaussian
    wg = np.exp(-0.5 * ((x - n / 2) / (0.45 * n)) ** 2)
    wg /= wg.sum()
    inits['wide_gauss'] = wg

    # Sine
    s = np.sin(np.pi * (x + 0.5) / n) ** 2
    s = np.maximum(s, 0.001)
    s /= s.sum()
    inits['sine'] = s

    # Two humps
    th = np.exp(-0.5 * ((x - n * 0.25) / (n * 0.15)) ** 2) + \
         np.exp(-0.5 * ((x - n * 0.75) / (n * 0.15)) ** 2)
    th = np.maximum(th, 0.01)
    th /= th.sum()
    inits['two_humps'] = th

    # Random Dirichlet
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits['random'] = r

    # Random with bias toward edges
    rb = np.ones(n)
    rb[:n // 4] = 2.0
    rb[-n // 4:] = 2.0
    rb /= rb.sum()
    r2 = rng.dirichlet(rb * 3.0)
    inits['random_edge'] = r2

    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            pp /= pp.sum()
            inits['parent'] = pp

    return inits
