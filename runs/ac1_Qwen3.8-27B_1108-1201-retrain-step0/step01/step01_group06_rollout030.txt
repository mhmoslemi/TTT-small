"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a minimax game iteration
with Nesterov-accelerated projected gradient for the inner QP subproblem,
combined with adaptive temperature weight updates, multiple sequence lengths,
and perturbative restarts. Also includes a direct smoothed-max gradient
descent mode with Euclidean geometry (projected onto simplex) as an
alternative to the previous multiplicative mirror descent.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def hankel_grad(w, a):
    """Compute 2 * h where h[i] = sum_j w[i+j]*a[j]. w has len 2n-1, a has len n."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    w_flip = np.zeros(fft_size)
    w_flip[:size] = w[::-1]
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    Wf = np.fft.rfft(w_flip)
    A = np.fft.rfft(a_pad)
    cross = np.fft.irfft(Wf * A, fft_size)
    h = cross[:size][::-1]
    return 2.0 * h


def solve_qp_nesterov(w, a_init, n_iters=100):
    """Solve min sum_k w[k]*(a*a)[k] s.t. sum(a)=1, a>=0 using Nesterov PG."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    eta = 0.4  # step size (Lipschitz ~ 2, so 1/L = 0.5, use 0.4 for safety)
    a_prev = a.copy()

    for k in range(1, n_iters + 1):
        if k == 1:
            z = a.copy()
        else:
            momentum = (k - 1) / (k + 2)
            z = a + momentum * (a - a_prev)
        a_prev = a.copy()

        z = project_simplex(z)
        grad = hankel_grad(w, z)
        a_new = project_simplex(z - eta * grad)
        a = a_new

    return a


def minimax_iterate(a_init, n, time_limit, tau_outer=10.0, qp_iters=80,
                    restart_every=30, perturb_scale=0.05):
    """Minimax game iteration: alternate between QP solve and weight update."""
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    size = 2 * n - 1
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    # Initialize weights: concentrate at peak
    w = np.ones(size) / size
    tau_w = tau_outer
    patience = 0
    max_patience = restart_every
    gen = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        gen += 1

        # Inner: solve QP with current weights
        a = solve_qp_nesterov(w, a, n_iters=qp_iters)

        # Evaluate
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()

        if current_obj < best_obj - 1e-9:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Outer: update weights
        scaled = tau_w * (conv - conv.max())
        scaled = np.clip(scaled, -500, 500)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(size)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Increase temperature gradually (sharpen focus)
        tau_w = min(tau_w * 1.15, 200.0)

        # Restart if stuck
        if patience >= max_patience:
            patience = 0
            tau_w = max(5.0, tau_w * 0.3)
            # Perturb
            a = best_a.copy()
            noise = np.random.default_rng(gen).normal(0, perturb_scale, n)
            a = a + noise * a.sum() / n
            a = project_simplex(a)
            w = np.ones(size) / size
            perturb_scale = max(0.01, perturb_scale * 0.8)

    return best_a, best_obj


def accelerated_smoothed_max(a_init, n, time_limit, tau_start=30.0, tau_end=800.0):
    """Nesterov accelerated projected gradient on smoothed max objective."""
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    size = 2 * n - 1
    eta = 0.4
    current_tau = tau_start
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    a_prev = a.copy()
    patience = 0
    max_patience = 60
    k = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        k += 1
        if k == 1:
            z = a.copy()
        else:
            momentum = (k - 1) / (k + 2)
            z = a + momentum * (a - a_prev)
        a_prev = a.copy()
        z = project_simplex(z)

        conv_z = fast_convolve(z)
        # Softmax weights
        scaled = current_tau * (conv_z - conv_z.max())
        scaled = np.clip(scaled, -500, 500)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(size)
            w[np.argmax(conv_z)] = 1.0
        else:
            w /= w_sum

        grad = hankel_grad(w, z)
        a_new = project_simplex(z - eta * grad)
        a = a_new

        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()

        if current_obj < best_obj - 1e-9:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Adapt
        if patience >= max_patience:
            patience = 0
            eta *= 0.7
            if eta < 0.05:
                eta = 0.35
                current_tau = max(20.0, current_tau * 0.5)
            current_tau = min(current_tau * 1.8, tau_end)
        elif k % 50 == 0:
            current_tau = min(current_tau * 1.3, tau_end)

    return best_a, best_obj


def coordinate_descent_polish(a_init, n, n_rounds=3):
    """Simple coordinate descent: for each coordinate, try small perturbations."""
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s

    conv = fast_convolve(a)
    best_obj = conv.max()
    rng = np.random.default_rng(123)

    for round_i in range(n_rounds):
        idx_order = np.arange(n)
        rng.shuffle(idx_order)
        for i in idx_order:
            # Try moving mass from i to other positions
            for delta_frac in [0.01, 0.03, 0.05, 0.1, -0.01, -0.03, -0.05]:
                a_trial = a.copy()
                move = a[i] * delta_frac
                if move == 0:
                    continue
                a_trial[i] -= move
                # Distribute to others proportionally
                others = np.ones(n)
                others[i] = 0
                others_sum = a_trial.sum() - a_trial[i]
                if others_sum < 1e-15:
                    continue
                a_trial[others > 0] += move * a_trial[others > 0] / others_sum
                a_trial = np.maximum(a_trial, 0)
                ts = a_trial.sum()
                if ts < 1e-15:
                    continue
                a_trial /= ts
                c = fast_convolve(a_trial)
                if c.max() < best_obj - 1e-10:
                    best_obj = c.max()
                    a = a_trial.copy()

    return a

            g = np.maximum(g, 0)
            g /= g.sum()
            inits.append((f'gauss_{sf}', g))
        # Bimodal
        bi = np.zeros(n)
        w = max(1, n // 8)
        bi[:w] = 1.0
        bi[-w:] = 1.0
        bi /= bi.sum()
        inits.append(('bimodal', bi))
        # Cosine
