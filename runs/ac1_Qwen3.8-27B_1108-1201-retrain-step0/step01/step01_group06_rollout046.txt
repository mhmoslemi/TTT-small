import numpy as np

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_objective(a):
    """Compute the full objective 2n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def subgradient_topk(a, k=5, temperature=1.0):
    """
    Compute a weighted subgradient focusing on the top-k peaks of the convolution.
    Returns the subgradient vector (same length as a).
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Weight the top peaks
    max_conv = conv.max()
    if max_conv < 1e-15:
        return np.zeros(n), conv
    
    # Soft-max weighting with temperature control
    scaled = conv / max_conv
    # Only weight the top-k
    threshold_idx = np.argsort(scaled)[-k:]
    weights = np.zeros(len(conv))
    for idx in threshold_idx:
        weights[idx] = scaled[idx]
    w_sum = weights.sum()
    if w_sum < 1e-15:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= w_sum
    
    # Compute subgradient: sg[j] = 2 * sum_k weights[k] * a_sym[k-j]
    # where a_sym[m] = a[m] if 0<=m<n else 0
    sg = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    
    for k_idx in active:
        k = int(k_idx)
        wk = weights[k]
        # sg[j] += 2 * wk * a[k-j] for valid j
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # a[k-j] for j = j_start, ..., j_end
            # = a[k-j_end], ..., a[k-j_start]
            # = a[k-j_end : k-j_start+1][::-1]
            sg[j_start:j_end+1] += 2.0 * wk * a[k-j_end:k-j_start+1][::-1]
    
    return sg, conv

def mirror_descent_accelerated(a0, time_limit, eta0=0.5, k_peaks=5, momentum_init=0.5):

    Nesterov-accelerated mirror descent on the simplex.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a_sum = a.sum()
    if a_sum < 1e-15:
        a = np.ones(n) / n
    else:
        a /= a_sum
    
    # Nesterov momentum variable (the "lookahead" point)
    y = a.copy()
    momentum = momentum_init
    
    eta = eta0
    best_obj = compute_objective(a)
    best_a = a.copy()
    
    patience = 0
    max_patience = 100
    t0 = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Compute subgradient at y (Nesterov lookahead)
        sg, conv = subgradient_topk(y, k=k_peaks)
        
        # Center the subgradient (for mirror descent on simplex)
        sg_centered = sg - sg.mean()
        
        # Mirror descent step from y
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a_new = y * np.exp(log_factor)
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= a_sum
        
        # Nesterov update
        momentum = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y = a_new + (momentum - 1.0) / (momentum + 1.0) * (a_new - a)
        # Ensure y is on the simplex
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a_new.copy()
        else:
            y /= y_sum
        
        a = a_new
        
        # Evaluate
        current_obj = compute_objective(a)
        
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Gaussians
    for sf in [0.1, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g
    
    # Bimodal (edges)
    bi = np.zeros(n)
    m = max(1, n // 4)
    bi[:m] = 1.0
    bi[-m:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Cosine
    cos_b = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_b /= cos_b.sum()
    inits['cosine'] = cos_b
    
    # Random (Dirichlet with concentration)
    for alpha in [0.3, 1.0, 3.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits[f'rand_{alpha}'] = r
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # "W" shape (4 peaks)
    w = np.zeros(n)
    quarter = n // 4
    for center in [quarter, n//2, 3*quarter]:
        sigma = n // 16
        w += np.exp(-0.5 * ((x - center) / sigma) ** 2)
    w /= w.sum()
    inits['w_shape'] = w
    
    # From parent
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp
    
    return inits

    small_factors = [0.92, 0.95, 0.97, 1.03, 1.05, 1.08]
    large_factors = [0.7, 0.8, 1.2, 1.4, 1.6]

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Try small perturbations on random coords
        n_try = min(200, n)
        idxs = rng.choice(n, size=n_try, replace=False)

        for idx in idxs:
            if time.monotonic() - t0 > time_limit:
                break
            for factor in small_factors:
                a_test = a.copy()
                a_test[idx] *= factor
                a_test = np.maximum(a_test, 0.0)
                s = a_test.sum()
                if s < 1e-15:
                    continue
                a_test /= s
                obj = compute_score_normalized(a_test)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_test.copy()
                    a = a_test.copy()
                    improved = True
                    break

        if not improved:
            # Try larger perturbations
            n_try = min(50, n)
            idxs = rng.choice(n, size=n_try, replace=False)
            for idx in idxs:
                if time.monotonic() - t0 > time_limit:
                    break
                for factor in large_factors:
                    a_test = a.copy()
                    a_test[idx] *= factor
                    a_test = np.maximum(a_test, 0.0)
                    s = a_test.sum()
                    if s < 1e-15:
                        continue
                    a_test /= s
                    obj = compute_score_normalized(a_test)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a_test.copy()
                        a = a_test.copy()
                        improved = True
                        break
            if not improved:
                break

    return best_a, best_obj


def generate_inits(n, rng, parent):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uni'] = np.ones(n) / n

    # Gaussians
    for sf in [0.12, 0.2, 0.3, 0.45]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits['g%d' % int(sf * 100)] = g

    # Bimodal
    bi = np.zeros(n)
    m = max(1, n // 4)
    bi[:m] = 1.0
    bi[-m:] = 1.0
    bi /= bi.sum()
    inits['bi'] = bi

    # Cosine bump
    cb = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    cb /= cb.sum()
    inits['cos'] = cb

    # Random
    for alpha in [0.5, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['r%d' % int(alpha * 10)] = r

    # W-shape
    w = np.zeros(n)
    for frac in [0.25, 0.5, 0.75]:
        center = n * frac
        sigma = max(1.0, n * 0.08)
        w += np.exp(-0.5 * ((x - center) / sigma) ** 2)
    w /= w.sum()
    inits['W'] = w

    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].astype(np.float64).copy()
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['par'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['par'] = pp

    return inits


def refine_sequence(a_init, time_limit):
    """
    Multi-stage refinement: mirror descent with decreasing eta, then polish.
    Returns (best_a, best_score_normalized).
    """
    a = a_init.copy()
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-15:
        a = np.ones(n) / n
    else:
        a /= a_sum

    best_obj = compute_score_normalized(a)
    best_a = a.copy()
    t0 = time.monotonic()

    # Stage 1: Moderate optimization
    stage1_time = min(60.0, time_limit * 0.4)
    a_opt, obj_opt = mirror_descent_opt(a, stage1_time, eta0=0.4, power0=5.0)
    if obj_opt < best_obj:
        best_obj = obj_opt
        best_a = a_opt.copy()
    a = a_opt

    remaining = time_limit - (time.monotonic() - t0)

    # Stage 2: Fine optimization with smaller steps
    if remaining > 10:
        stage2_time = min(remaining * 0.4, 120.0)
        a_opt2, obj_opt2 = mirror_descent_opt(a, stage2_time, eta0=0.15, power0=8.0, avg_alpha=0.02)
        if obj_opt2 < best_obj:
            best_obj = obj_opt2
            best_a = a_opt2.copy()
        a = a_opt2 if obj_opt2 < best_obj else best_a

    remaining = time_limit - (time.monotonic() - t0)

    # Stage 3: Very fine
    if remaining > 10:
        stage3_time = min(remaining * 0.3, 90.0)
        a_opt3, obj_opt3 = mirror_descent_opt(a, stage3_time, eta0=0.08, power0=12.0, avg_alpha=0.01)
        if obj_opt3 < best_obj:
            best_obj = obj_opt3
            best_a = a_opt3.copy()
        a = a_opt3 if obj_opt3 < best_obj else best_a

    remaining = time_limit - (time.monotonic() - t0)

    # Stage 4: Coordinate polish
    if remaining > 5:
        polish_time = remaining * 0.8
        a_pol, obj_pol = coordinate_polish(a, polish_time)
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()

    return best_a, best_obj


def run_full_search(seed, budget_s):
    """Main search: scan lengths/inits, then deep optimization."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Loaded parent sequence: n=%d, sum=%.4f" % (p.size, p.sum()))
    except Exception:
        pass

    def try_update(seq, label):
        nonlocal best_sequence, best_score
        if seq is None:
            return
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = seq.copy()
            print("  %s: score=%.8f (n=%d)" % (label, score, len(seq)))

    # Phase 1: Quick scan across lengths
    print("=== Phase 1: Scan ===")
    lengths = [96, 128, 160, 192, 256, 320, 384, 512]

    scan_results = []
    phase1_deadline = time.monotonic() + min(500.0, (deadline - time.monotonic()) * 0.35)

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            print("  Phase 1 deadline reached at n=%d" % n)
            break

        inits = generate_inits(n, rng, parent)
        time_per = max(8.0, min(25.0, (phase1_deadline - time.monotonic()) / max(1, len(inits))))

        for name, a0 in inits.items():
            if time.monotonic() > phase1_deadline - 5:
                break
            t_alloc = min(time_per, phase1_deadline - time.monotonic() - 3)
            if t_alloc < 5:
                break

            a_opt, obj_opt = mirror_descent_opt(a0, t_alloc, eta0=0.5, power0=5.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print("  Scan n=%d %s: %.8f" % (n, name, score))

    # Sort results
    scan_results.sort(key=lambda x: x[0])

    # Phase 2: Deep optimization
    print("\n=== Phase 2: Deep ===")
    print("  Best so far: %.8f" % best_score)

    top_configs = []
    seen_keys = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen_keys and len(top_configs) < 20:
            top_configs.append((n, name, a_opt))
            seen_keys.add(key)

    # Always include current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'BEST', best_sequence.copy()))
    # Include parent if available
    if parent is not None:
        top_configs.append((parent.size, 'PARENT', parent.copy()))

    remaining = deadline - time.monotonic() - 60
    time_per_deep = max(20.0, remaining / max(1, len(top_configs)))

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 30:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 30)
        if t_alloc < 15:
            break

        a_opt, obj_opt = refine_sequence(a0, t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print("  Deep n=%d %s: %.8f" % (n, name, score))

    # Phase 3: Final polish on best
    if best_sequence is not None and time.monotonic() < deadline - 15:
        print("\n=== Phase 3: Final polish ===")
        remaining = deadline - time.monotonic() - 5
        a_opt, obj_opt = refine_sequence(best_sequence, remaining)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print("  Final: %.8f" % score)

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print("\n=== FINAL: %.8f (n=%d) ===" % (best_score, len(best_sequence)))
    return best_sequence


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses power-weighted mirror descent with iterative
    averaging, multi-stage refinement, and coordinate descent polishing
    to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    result = run_full_search(seed=seed, budget_s=budget_s)
    # Ensure valid output
    result = np.maximum(result, 0.0)
    result = np.minimum(result, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    return [float(x) for x in result]
