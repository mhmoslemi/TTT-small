"""
Hybrid optimization for minimizing peak autocorrelation of non-negative step functions.
Combines gradient-based optimization (Adam with adaptive softmax temperature),
LP-guided restarts, and peak-directed simulated annealing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:len(a)+len(b)-1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def score_normalized(a, S):
    """Score assuming sum(a) = S."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def grad_smoothed(a, tau):
    """
    Compute soft-max smoothed max(conv(a,a)) and gradient w.r.t. a.
    """
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    value = c_max + tau * np.log(np.sum(exp_s))
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[len(a)-1 : 2*len(a)-1]
    return value, g, c, w

def adam_step(theta, g, m, v, step, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * g
    v = beta2 * v + (1 - beta2) * g * g
    m_hat = m / (1 - beta1**step)
    v_hat = v / (1 - beta2**step)
    theta_new = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
    return theta_new, m, v

def lp_find_direction(a, M, n):
    """
    Solve LP: max sum(b) subject to b >= 0, conv(a, b) <= M.
    Returns the direction b (normalized to sum = S).
    """
    # conv(a, b)_k = sum_i a[i] * b[k-i] for valid indices
    # This is a Toeplitz-like system. Build the constraint matrix A.
    # A has (2n-1) rows and n columns.
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            b = result.x
            b = np.maximum(b, 0)
            s = np.sum(b)
            if s > 1e-10:
                return b / s  # Normalize to sum = 1
        else:
            return None
    except:
        return None

    n = len(a_init)
    S = np.sum(a_init)
    current = a_init.copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temp = 0.005
    cooling = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

            # Peak-directed swap
            c = fft_convolve(current, current)
            peak_idx = np.argmax(c)
            M = c[peak_idx]
            # Find top contributors to the peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] += current[i] * current[j]
            top_k = max(3, n // 16)
