FISTA with log-sum-exp smoothing to minimize normalized autocorrelation peak.
Uses adaptive temperature continuation, FFT-based gradient computation,
and multiple restarts with diverse initializations.

import numpy as np
import time

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_b(a):
    """Compute convolution b = a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next fast FFT size
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    B = A * A
    b = np.fft.irfft(B, fft_size)[:size]
    return b

def _compute_score(a):
    """Compute 2*n*max(convolve(a,a))/sum(a)^2 for sequence a."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    a_norm = a / S
    b = _compute_b(a_norm)
    return 2.0 * n * np.max(b)

def _fista_smoothed(a_start, T, eta, max_iter, deadline, verbose=False):
    """
    FISTA to minimize the log-sum-exp smoothed objective on the simplex.
    """
    n = len(a_start)
    a = a_start.astype(np.float64).copy()
    S = np.sum(a)
    if S > 1e-15:
        a = a / S
    
    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    
    best_a = a.copy()
    best_score = _compute_score(a)
    
    for k in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        b_y = _compute_b(y)
        grad = _compute_grad_smoothened(y, b_y, T)
        
        # FISTA update
        z = y - eta * grad
        a = _project_simplex(z, 1.0)
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + (t - 1.0) / t_new * (a - a_prev)
        y = np.maximum(y, 0.0)  # Keep y non-negative (optional, helps stability)
        
        # Track best
        # (Checking exact score every iteration is expensive, check periodically)
        if k % 50 == 0 or k == max_iter - 1:
            score = _compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        a_prev = a
        t = t_new
    
    # Final check
    score = _compute_score(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

def _make_two_level(n, eps):
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    return a / np.sum(a)

def _make_smooth_profile(n, eps):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)

FISTA (Nesterov-accelerated proximal gradient) with log-sum-exp smoothing

import numpy as np
import time


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _convolve_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)[:size]
    return b


def _grad_lse(a, b, T):
    """
    Gradient of (1/T)*log(sum_k exp(T*b_k)) w.r.t. a.
    Uses FFT for O(n log n) computation.
    g[i] = 2 * sum_j w[i+j] * a[j] = 2 * (a_rev * w)[i+n-1]

    w = exp_Tb / s

    conv = np.fft.irfft(AR * W, fft_size)
    g = 2.0 * conv[n - 1:n - 1 + n]

    """
    FISTA to minimize log-sum-exp smoothed objective on the simplex.
    Step size: 1/(2T) (theoretical Lipschitz constant).
    """
    n = len(a_init)
    if eta is None:
        eta = 1.0 / (2.0 * T)

    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    if S > 1e-15:
        a /= S
    else:
        a = np.ones(n) / n

    y = a.copy()
    a_prev = a.copy()
    t = 1.0

    best_a = a.copy()
    best_score = _score(a)
    stagnant = 0

    it = 0
    while time.monotonic() < deadline:
        b_y = _convolve_fft(y)
        g = _grad_lse(y, b_y, T)

        z = y - eta * g
        a_new = _project_simplex(z, 1.0)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + (t - 1.0) / t_new * (a_new - a_prev)
        y = np.maximum(y, 0.0)

        if it % 200 == 0 or it == 0:
            sc = _score(a_new)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a_new.copy()
                stagnant = 0
            else:
                stagnant += 200

        a_prev = a
        a = a_new
        t = t_new
        it += 1

        if stagnant >= max_stagnant:
            break

    sc = _score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _fista_subgrad(a_init, deadline, M=20, eta_init=None):
    """
    FISTA with exact max subgradient (top-M peaks with softmax weighting).
    For final polishing.
    """
    n = len(a_init)
    if eta_init is None:
        eta_init = 1.0 / np.sqrt(n)

    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    if S > 1e-15:
        a /= S

    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    eta = eta_init

    best_a = a.copy()
    best_score = _score(a)

    it = 0
    while time.monotonic() < deadline:
        b_y = _convolve_fft(y)
        b_max = np.max(b_y)

        if M < len(b_y):
            k_idx = np.argpartition(b_y, -M)[-M:]
            b_sel = b_y[k_idx]
            diff = b_max - b_sel
            scale = np.max(diff)
            if scale < 1e-15:
                weights = np.ones(M) / M
            else:
                weights = np.exp(-15.0 * diff / scale)
                weights /= np.sum(weights)
            w_full = np.zeros(len(b_y))
            w_full[k_idx] = weights
        else:
            w_full = np.ones(len(b_y)) / len(b_y)

        a_rev = y[::-1]
        fft_size = 1
        while fft_size < 3 * n - 2:
            fft_size *= 2
        AR = np.fft.rfft(a_rev, fft_size)
        W = np.fft.rfft(w_full, fft_size)
        conv = np.fft.irfft(AR * W, fft_size)
        g = 2.0 * conv[n - 1:n - 1 + n]
        g -= np.mean(g)

        z = y - eta * g
        a_new = _project_simplex(z, 1.0)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + (t - 1.0) / t_new * (a_new - a_prev)
        y = np.maximum(y, 0.0)

        eta *= 0.9995

        if it % 300 == 0:
            sc = _score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        a_prev = a
        a = a_new
        t = t_new
        it += 1

    sc = _score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _make_two_level(n, eps):
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = a.sum()
    return a / S if S > 0 else np.ones(n) / n


def _make_smooth_profile(n, eps):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    S = a.sum()
    return a / S if S > 0 else np.ones(n) / n


def _make_fourier_smooth(n, rng, n_modes=10):
    """Create a smooth random sequence from low-frequency Fourier components."""
    coeffs = rng.standard_normal(n_modes) * 0.1
    phases = rng.uniform(0, 2 * np.pi, n_modes)
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, n_modes + 1):
        a += coeffs[k - 1] * np.cos(2 * np.pi * k * i / n + phases[k - 1])
    a = np.maximum(a, 0.0)
    S = a.sum()
    return a / S if S > 0 else np.ones(n) / n


def _perturb(a, rng, scale=0.03):
    noise = rng.standard_normal(len(a)) * scale
    a_new = a + noise
    return _project_simplex(a_new, 1.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    FISTA with log-sum-exp smoothing, temperature continuation,
    and diverse initializations to minimize normalized autocorrelation peak.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 15.0)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size
    print(f"Sequence length: {n}")

    a0 = parent.copy()
    S0 = np.sum(a0)
    if S0 > 1e-15:
        a0 = a0 / S0
    else:
        a0 = np.ones(n) / n

    global_best = a0.copy()
    global_best_score = _score(global_best)
    print(f"Initial score: {global_best_score:.6f}")

    # Diverse initializations
    inits = [
        a0,
        _make_two_level(n, 0.2),
        _make_two_level(n, 0.5),
        _make_smooth_profile(n, 0.4),
        _make_smooth_profile(n, 0.8),
        _make_fourier_smooth(n, rng, 15),
        _make_fourier_smooth(n, rng, 25),
        _make_fourier_smooth(n, rng, 5),
        np.ones(n) / n,
    ]
    # Add random
    r = rng.random(n)
    inits.append(r / r.sum())

    n_inits = len(inits)
    temps = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0]

    # Phase 1: FISTA with temperature continuation for each init
    # Give first init more time
    total_time = deadline - start_time
    time_first = 0.35 * total_time
    time_rest = total_time / (n_inits - 1)

    print("\n=== Phase 1: FISTA with temperature continuation ===")
    for i, a_init in enumerate(inits):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if i == 0:
            t_alloc = time_first
        else:
            t_alloc = time_rest

        phase_deadline = min(time.monotonic() + t_alloc, deadline)
        current = a_init.copy()

        for T in temps:
            t_rem = phase_deadline - time.monotonic()
            if t_rem < 5:
                break
            a_new, sc = _fista(current, T, t_rem, max_stagnant=300)
            if sc < global_best_score:
                global_best = a_new.copy()
                global_best_score = sc
                print(f"  Init {i}, T={T:.0f}: score={sc:.6f} (NEW BEST)")
            current = a_new

    print(f"\nAfter Phase 1: best={global_best_score:.6f}, time={time.monotonic()-start_time:.1f}s")

    # Phase 2: Polish with subgradient FISTA
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 2: Subgradient FISTA polishing ===")
        t_sleep = remaining * 0.5
        polish_deadline = time.monotonic() + t_sleep
        a_pol, sc = _fista_subgrad(global_best, polish_deadline, M=30,
                                   eta_init=0.5 / np.sqrt(n))
        if sc < global_best_score:
            global_best = a_pol.copy()
            global_best_score = sc
            print(f"  Polish: score={sc:.6f} (NEW BEST)")

    # Phase 3: High-T FISTA restarts from perturbed best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 3: Diversification ({remaining:.0f}s) ===")
        n_div = max(1, int(remaining / 40))
        per_div = remaining / n_div

        for d in range(n_div):
            if time.monotonic() >= deadline:
                break
            a_pert = _perturb(global_best, rng, scale=0.02 + 0.01 * d)
            div_deadline = min(time.monotonic() + per_div, deadline)

            # Use highest temperatures for refinement
            current = a_pert
            for T in [50.0, 100.0, 200.0]:
                t_rem = div_deadline - time.monotonic()
                if t_rem < 5:
                    break
                a_new, sc = _fista(current, T, t_rem, max_stagnant=200)
                if sc < global_best_score:
                    global_best = a_new.copy()
                    global_best_score = sc
                    print(f"  Div {d}, T={T:.0f}: score={sc:.6f} (NEW BEST)")
                current = a_new

    # Phase 4: Final exact subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 4: Final polish ({remaining:.0f}s) ===")
        a_fin, sc = _fista_subgrad(global_best, deadline, M=50,
                                   eta_init=0.2 / np.sqrt(n))
        if sc < global_best_score:
            global_best = a_fin.copy()
            global_best_score = sc
            print(f"  Final: score={sc:.6f} (NEW BEST)")

    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"\nFINAL SCORE: {final_score:.6f}")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")
    return result
