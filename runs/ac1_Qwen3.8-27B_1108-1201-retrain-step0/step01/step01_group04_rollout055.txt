def fft_length(n):

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssum - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    k = np.count_nonzero(cond) - 1
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = np.sum(out)
    if s <= 0:
        return np.ones(n) / n
    return out / s

Key: n=700 is small enough for fast LP solves, enabling many best-response iterations.

def _next_pow2(n):
    n = int(max(1, n))
    m = 1
    while m < n:
        m <<= 1
    return m


def _self_conv(p):
    """Full self-convolution of p via FFT, returned as length 2n-1 array."""
    n = len(p)
    L = _next_pow2(2 * n - 1)
    F = np.fft.fft(p, L)
    c = np.fft.ifft(F * F)
    return c[:2 * n - 1].real


def _score(p):
    """Score = 2n * max(p*p) assuming sum(p)=1, p>=0."""
    n = len(p)
    s = np.sum(p)
    if s < 1e-15:
        return np.inf
    pn = p / s
    c = _self_conv(pn)
    return 2.0 * n * np.max(c)


def _project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-30:
        return np.full(n, 1.0 / n)
    return out / s


def _lp_solve(p, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0.
    A[k,i] = p[k-i] if 0<=k-i<n else 0.
    Returns normalized g or None.
    """
    n = len(p)
    # Build sparse constraint matrix
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        # i ranges from max(0, k-n+1) to min(k, n-1)
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            idx = k - i
            if 0 <= idx < n:
                rows.append(k)
                cols.append(i)
                data.append(p[idx])
    
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=(0, None),
                      method='highs', options={'disp': False, 'dual_feasibility_tolerance': 1e-10})
    except Exception as e:
        return None
    
    if res is None or res.x is None or not res.success:
        return None
    
    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s


def _grad_smoothed_max(p, T):
    """Gradient of smoothed max of p*p w.r.t. p.
    Uses FFT-based computation.
    """
    n = len(p)
    c = _self_conv(p)
    cmax = np.max(c)
    if cmax < 1e-15:
        return np.zeros(n), c, 0.0
    T = max(T, 1e-12 * cmax)
    
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = e.sum()
    if s < 1e-30 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    
    surr = cmax + T * np.log(s) if s > 0 else cmax
    
    # Compute gradient via FFT
    L = _next_pow2(3 * n - 2)
    P = np.fft.fft(p, L)
    
    # Term 2: (w * p)_j for j=0..n-1 (standard convolution)
    W = np.fft.fft(w, L)
    conv_wp = np.fft.ifft(W * P).real
    term2 = conv_wp[:n]
    
    # Term 1: a_j = sum_m p_m * w_{j+m} = (p * w_rev)_{2n-2-j}
    w_rev = w[::-1]
    W_rev = np.fft.fft(w_rev, L)
    conv_pwrev = np.fft.ifft(P * W_rev).real
    term1 = conv_pwrev[2 * n - 2 : n - 1 : -1]
    
    grad = term1 + term2
    return grad, c, surr

    s = p.sum()
    if s < 1e-15:
        return np.ones(len(p)) / len(p)
    return p / s

"""
Iterated LP best-response with gradient refinement for minimum peak autocorrelation.
"""

import time
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _pw2(n):
    """Next power of 2 >= n."""
    n = int(max(1, n))
    m = 1
    while m < n:
        m <<= 1
    return m


def _conv_self(p):
    """Self-convolution via FFT. Returns length 2n-1 array."""
    n = len(p)
    L = _pw2(2 * n - 1)
    F = np.fft.fft(p, L)
    return np.fft.ifft(F * F)[:2*n-1].real


def _score1(p):
    """Score for normalized (sum=1, nonneg) p."""
    n = len(p)
    c = _conv_self(p)
    return 2.0 * n * c.max()
