import time

def score(a):
    """Compute the objective."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def smooth_score_and_grad(a, beta):
    """Compute smooth score and gradient using log-sum-exp approximation."""
    n = len(a)
    conv = np.convolve(a, a)  # length 2n-1
    sum_a = a.sum()
    
    # Smooth max via log-sum-exp
    max_conv = conv.max()
    shifted = beta * (conv - max_conv)
    log_sum_exp = max_conv + np.log(np.sum(np.exp(shifted)))
    
    # Score (smooth)
    smooth_score = 2.0 * n * log_sum_exp / (sum_a ** 2)
    
    # Gradient
    # p = softmax(beta * conv)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    # dM/da[j] = 2 * sum_m p[m+j] * a[m] (for m=0..n-1)
    # Use FFT-based cross-correlation
    N = 2 * (2 * n)  # FFT length
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    # Cross-correlation: r[k] = sum_t p[t+k] * a[t]
    # r = ifft(P * conj(A))
    r = np.fft.irfft(P * np.conj(A), N)
    dM_da = 2.0 * r[:n]
    
    # Full gradient: dS/da[j] = 2n * (dM_da[j] * sum_a - 2 * M) / sum_a^3
    # where M = log_sum_exp (smooth max)
    grad = 2.0 * n * (dM_da * sum_a - 2.0 * log_sum_exp) / (sum_a ** 3)
    
    return smooth_score, grad

def projected_gradient_descent(a, beta, lr, max_iters, time_limit):
    """PGD with backtracking line search and non-negativity projection."""
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_limit:
            break
        
        s, grad = smooth_score_and_grad(a, beta)
        
        # Backtracking line search
        step = lr
        for _ in range(20):
            a_new = np.maximum(0, a - step * grad)
            s_new, _ = smooth_score_and_grad(a_new, beta)
            if s_new < s:
                break
            step *= 0.5
        else:
            break
        
        a = a_new
        if it % 100 == 0:
            actual_score = score(a)
            print(f"  PGD iter {it}: smooth={s:.6f}, actual={actual_score:.6f}")
    
    return a, score(a)

    obj = 2.0 * n * log_sum_exp / (sum_a ** 2)

def score(a):
    n = len(a)
    conv = np.convolve(a, a)

    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def smooth_obj_grad(x, beta, n):
    """Smooth objective and gradient for x (length n)."""
    N = 2 * n
    A = np.fft.rfft(x, N)
    conv = np.fft.irfft(A * A, N)[:2*n-1]
    conv = np.maximum(0, conv)
    
    s = x.sum()
    if s < 1e-10:
        s = 1e-10
    mc = conv.max()
    tmp = beta * (conv - mc)
    e = np.exp(tmp)
    se = e.sum()
    lse = mc + np.log(se)
    obj = 2.0 * n * lse / (s * s)
    
    p = e / se

    return obj, grad
