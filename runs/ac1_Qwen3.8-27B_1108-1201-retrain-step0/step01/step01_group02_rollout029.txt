"""
Optimized LP-based iterative refinement + gradient polishing for minimizing
peak auto-convolution of non-negative sequences.
Uses symmetry, adaptive step sizes, and multi-scale approach.
"""

import numpy as np
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_sinc_squared(n):
    """sinc^2 function, which has a known flat-ish autocorrelation."""
    t = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    return np.sinc(t / (n / 4.0)) ** 2  # np.sinc(x) = sin(pi*x)/(pi*x)


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = a[k - j]
    return A


def solve_lp(a, M):
    """Solve: max sum(g) s.t. conv(a, g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60.0})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception as e:
        print(f"    LP error: {e}")
    return None


def lp_refine(a, time_limit_s=60.0, verbose=True):
    """LP-based refinement."""
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    it = 0
    stagnant = 0
    while time.monotonic() - start_time < time_limit_s:
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 30:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break
        
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        
        g = solve_lp(a, M)
        if g is None:
            if verbose:
                print(f"    LP solve failed at iter {it}")
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-8:
            if verbose:
                print(f"    LP no improvement (sum_g={sum_g:.8f}) at iter {it}")
            break
        
        g = g / sum_g
        
        # Adaptive step size with backtracking
        alpha = 0.5
        while alpha > 0.01:
            a_new = normalize((1.0 - alpha) * a + alpha * g)
            new_score = score_sequence(a_new)
            if new_score <= score + 1e-10:
                break
            alpha *= 0.5
        
        a = normalize((1.0 - alpha) * a + alpha * g)
        it += 1
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def gradient_polish(a, time_limit_s=30.0, verbose=True):
    """Gradient-based polishing using soft-max approximation."""
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    beta = 10.0  # soft-max temperature
    lr = 0.1
    
    it = 0
    while time.monotonic() - start_time < time_limit_s:
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        
        # Soft-max weights
        b_shifted = conv - conv.max()  # for numerical stability
        p = np.exp(beta * b_shifted)
        p = p / p.sum()
        
        # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i]
        # This is the cross-correlation of p and a
        grad = 2.0 * np.correlate(p, a, mode='full')[n - 1:n - 1 + n]
        
        # Normalized gradient (account for the sum constraint)
        # Project gradient onto the tangent space of the simplex
        grad_mean = grad.mean()
        grad = grad - grad_mean
        
        # Multiplicative update (maintains non-negativity)
        a_new = a * np.exp(-lr * grad / (2.0 * np.sqrt(np.mean(grad**2)) + 1e-10))
        a_new = normalize(a_new)
        
        new_score = score_sequence(a_new)
        if new_score < score:
            a = a_new
            lr = min(lr * 1.05, 1.0)
        else:
            lr *= 0.7
            if lr < 0.001:
                break
        
        it += 1
    
    if verbose:
        print(f"    Grad done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

"""
Optimized LP-based iterative refinement + gradient polishing for minimizing
peak auto-convolution of non-negative sequences.
Key improvements over previous:
- Better initialization (bump-shaped windows)
- Adaptive step size with backtracking
- Gradient-based polishing phase
- Multi-scale search (multiple n values)
- Symmetry-aware optimization
"""

import numpy as np
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def initial_raised_cosine_dip(n):
    """Original 'dip' shape: high at edges, low in middle."""
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_raised_cosine_bump(n):
    """Bump shape: low at edges, high in middle."""
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * t / (n - 1) if n > 1 else 1))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_triangular(n):
    """Triangular window: 0 at edges, 1 at center."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return np.maximum(0, 1.0 - np.abs(t - center) / (center + 1))


def initial_bohman(n):
    """Bohman window (idempotent under convolution)."""
    if n == 1:
        return np.ones(1)
    t = np.arange(n, dtype=np.float64)
    x = 2.0 * t / (n - 1) - 1.0
    return np.maximum(0, (1 - x**2) * (1 - 2 * np.abs(1 - x)))


def initial_uniform(n):

    """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    # Use efficient construction
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = a[k - j]
    return A
