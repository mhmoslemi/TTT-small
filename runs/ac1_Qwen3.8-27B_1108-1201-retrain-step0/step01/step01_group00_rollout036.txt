def lp_step(f, M, tol=1e-8):
    """
    Solve: maximize sum(b) subject to b >= 0 and (f*b)_k <= M for all k.
    Returns the optimal b.
    """
    n = len(f)
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Inequality constraints: (f*b)_k <= M for k = 0, ..., 2n-2
    # (f*b)_k = sum_j f_j * b_{k-j} where 0 <= j < n and 0 <= k-j < n
    # A_ub has shape (2n-1, n), b_ub has shape (2n-1,)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]
    b_ub = np.full(2*n-1, M)
    
    # Non-negativity: b >= 0 (handled by bounds)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    
    return None

    cols = []
    vals = []

        j_end = min(n - 1, k)

"""
LP-based peak reduction with gradient polish for minimizing autocorrelation
of non-negative sequences. Uses linear programming to find descent directions
that reduce the peak of self-convolution, combined with Adam gradient descent
and local search for final polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score (same formula as evaluate_sequence)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = fft_convolve(a, a)
    return 2.0 * n * np.max(b) / (S * S)


def build_lp_constraint_matrix(f):
    """
    Build sparse constraint matrix A for the LP:
    (convolve(f, b))_k = sum_j f_{k-j} * b_j <= M
    A has shape (2n-1, n), A[k,j] = f[k-j] if 0 <= k-j < n else 0.
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_solve(f, M):
    """
    Solve LP: maximize sum(b) subject to convolve(f,b)_k <= M for all k, b >= 0.
    Returns optimal b or None if infeasible.
    """
    n = len(f)
    c = -np.ones(n)
    A_ub = build_lp_constraint_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

            f = f_new

def gradient_phase(f_init, time_limit, rng, verbose=True):
    """
    Phase 2: Adam gradient descent with projection onto {a >= 0, sum(a) = S}.
    Minimizes soft_max(conv(a,a), tau) with annealed temperature.
    """
    n = len(f_init)
    S = np.sum(f_init)
    a = f_init.astype(np.float64).copy()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t0 = time.monotonic()
    tau_start = 5.0 / n
    tau_end = 0.2 / n
    lr_base = 0.5
    step = 0
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    stagnation = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        # Compute gradient
        b = fft_convolve(a, a)
        b_shifted = b - np.max(b)
        w = np.exp(b_shifted / tau)
        w /= np.sum(w)
        
        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g = 2.0 * corr[n - 1: 2 * n - 1]
        
        # Adam update
        step += 1
        m1 = beta1 * m1 + (1 - beta1) * g
        m2 = beta2 * m2 + (1 - beta2) * (g * g)
        m1_hat = m1 / (1.0 - beta1 ** step)
        m2_hat = m2 / (1.0 - beta2 ** step)
        lr = lr_base / np.sqrt(step)
        a = a - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Project: non-negative and fixed sum
        a = np.maximum(a, 0.0)
        a = a * S / (np.sum(a) + 1e-30)
        
        # Check score
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        if step % 200 == 0 and verbose:
            print(f"    Grad step {step}: score={score:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")
    
    return best_a, best_score

"""
LP-based peak reduction with gradient polish for minimizing the peak
autocorrelation of non-negative sequences. The core idea is to solve a
linear program at each iteration that finds the "most spread" non-negative
vector b whose cross-correlation with the current f is bounded by the peak
of f*f. Blending f with b reduces the peak, and iteration drives the score
down. Gradient descent and local search provide final polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = fft_convolve(a, a)
    return 2.0 * n * np.max(b) / (S * S)


def build_lp_constraint_matrix(f):
    """
    Build sparse matrix A for LP constraints: (convolve(f,b))_k <= M.
    A[k,j] = f[k-j] if 0 <= k-j < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            js = np.arange(j_min, j_max + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_solve(f, M):
    """Solve: max sum(b) s.t. convolve(f,b) <= M, b >= 0. Returns b or None."""
    n = len(f)
    c = -np.ones(n)
    A_ub = build_lp_constraint_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None


def lp_phase(f_init, time_limit, verbose=True):
    """LP-based peak reduction iterations."""
    n = len(f_init)
    S_target = np.sqrt(2.0 * n)
    f = f_init.astype(np.float64).copy()
    f = f * S_target / np.sum(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    t0 = time.monotonic()
    blend = 0.4
    stagnation = 0
    iter_count = 0
    A_cached = None
    f_cached = None
    
    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(f, f))
        
        # Cache constraint matrix if f hasn't changed much
        if A_cached is None or not np.allclose(f, f_cached, rtol=1e-6):
            A_cached = build_lp_constraint_matrix(f)
            f_cached = f.copy()
        
        # Solve LP
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        result = linprog(c, A_ub=A_cached, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not result.success:
            stagnation += 2
            if stagnation > 20:
                break
            continue
        
        b_opt = result.x
        if np.sum(b_opt) < 1e-10:
            break
        
        b_sum = np.sum(b_opt)
        b_scaled = b_opt * S_target / b_sum
        f_new = (1.0 - blend) * f + blend * b_scaled
        new_score = compute_score(f_new)
        
        if new_score < best_score - 1e-9:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            stagnation = 0
            blend = min(0.7, blend + 0.005)
            if verbose and iter_count % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter {iter_count}: score={new_score:.6f}, M={M:.4f}, "

            stagnation += 1
            blend *= 0.85
            if blend < 0.05:

                stagnation += 5

            print(f"    LP iter {iter_count}: best={best_score:.6f}, t={time.monotonic()-t0:.1f}s")
    
    if verbose:
        print(f"    LP phase done: {iter_count} iterations, best={best_score:.6f}")
    return best_f, best_score


def gradient_phase(f_init, time_limit, rng, verbose=True):
    """Adam gradient descent with projection."""
    n = len(f_init)

    a = f_init.astype(np.float64).copy()

    best_score = compute_score(a)

        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        b = fft_convolve(a, a)
        b_shifted = b - np.max(b)
        w = np.exp(b_shifted / tau)
        w /= np.sum(w)
        
        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g = 2.0 * corr[n - 1: 2 * n - 1]
        
        step += 1
        m1 = beta1 * m1 + (1 - beta1) * g

        m1_hat = m1 / (1.0 - beta1 ** step)
        m2_hat = m2 / (1.0 - beta2 ** step)
        a = a - lr / np.sqrt(step) * m1_hat / (np.sqrt(m2_hat) + eps)
        
        a = np.maximum(a, 0.0)

        candidate = np.maximum(candidate, 0.0)
        total = np.sum(candidate)
        if total < 1e-10:
            continue
        candidate = candidate * S / total

"""
LP-based peak reduction with gradient polish for minimizing the peak
autocorrelation of non-negative step functions. The core innovation is
solving a linear program at each iteration to find the most "spread out"
non-negative direction b whose cross-correlation with the current f is
bounded by the peak of f*f. Blending f toward b reduces the peak
monotonically. Combined with Adam gradient descent and local search
for final polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score (same formula as evaluate_sequence)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = fft_convolve(a, a)
    return 2.0 * n * np.max(b) / (S * S)


def build_lp_constraint_matrix(f):
    """
    Build sparse constraint matrix A for LP:
    (convolve(f, b))_k = sum_j f[k-j] * b_j <= M for k=0,...,2n-2.
    Returns A with shape (2n-1, n).
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            js = list(range(j_min, j_max + 1))
            rows.extend([k] * len(js))
            cols.extend(js)
            vals.extend([f[k - j] for j in js])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_phase(f_init, time_limit, verbose=True):
    """
    LP-based peak reduction: iteratively solve
      max sum(b) s.t. convolve(f,b)_k <= M, b >= 0
    and blend f toward the LP solution to reduce peak of f*f.
    """
    n = len(f_init)
    S_target = np.sqrt(2.0 * n)
    f = f_init.astype(np.float64).copy()
    f = f * S_target / np.sum(f)

    best_f = f.copy()
    best_score = compute_score(f)

    t0 = time.monotonic()
    blend = 0.35
    stagnation = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(f, f))
        A_ub = build_lp_constraint_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        if not result.success or result.x is None:
            stagnation += 2
            if stagnation > 30:
                break
            continue

        b_opt = result.x
        b_sum = np.sum(b_opt)
        if b_sum < 1e-10:
            break

        b_scaled = b_opt * S_target / b_sum
        f_new = (1.0 - blend) * f + blend * b_scaled
        new_score = compute_score(f_new)

        if new_score < best_score - 1e-10:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            stagnation = 0
            blend = min(0.65, blend + 0.003)
            if verbose and iter_count % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter {iter_count}: score={new_score:.6f}, M={M:.4f}, "
                      f"blend={blend:.3f}, t={elapsed:.1f}s")
        else:
            stagnation += 1
            blend *= 0.82
            if blend < 0.03:
                blend = 0.35
                stagnation += 10
            if stagnation > 100:
                break

        iter_count += 1

    if verbose:
        print(f"    LP phase done: {iter_count} iters, best={best_score:.6f}, "
              f"time={time.monotonic()-t0:.1f}s")
    return best_f, best_score


def gradient_phase(f_init, time_limit, rng, verbose=True):
    """
    Adam gradient descent on soft_max(conv(a,a), tau) with
    projection onto {a >= 0, sum(a) = S}.
    """
    n = len(f_init)
    S = np.sum(f_init)
    a = f_init.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    t0 = time.monotonic()
    tau_start = 4.0 / n
    tau_end = 0.05 / n
    lr_base = 0.5
    step = 0
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    stall = 0

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** frac

        b = fft_convolve(a, a)
        b_shifted = b - np.max(b)
        w = np.exp(b_shifted / tau)
        w /= np.sum(w)

        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g = 2.0 * corr[n - 1: 2 * n - 1]

        step += 1
        m1 = beta1 * m1 + (1.0 - beta1) * g
        m2 = beta2 * m2 + (1.0 - beta2) * (g * g)
        m1_hat = m1 / (1.0 - beta1 ** step)
        m2_hat = m2 / (1.0 - beta2 ** step)
        a = a - (lr_base / np.sqrt(step)) * m1_hat / (np.sqrt(m2_hat) + eps)

        a = np.maximum(a, 0.0)
        total = np.sum(a)
        if total < 1e-15:
            a = np.full(n, S / n)
        else:
            a = a * S / total

        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if step % 500 == 0 and verbose:
            print(f"    Grad step {step}: score={score:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")

        if stall > 5000:
            # Restart with perturbation
            stall = 0
            a = best_a * np.exp(rng.normal(0, 0.05, size=n))
            a = np.maximum(a, 0)
            a = a * S / (np.sum(a) + 1e-30)

    return best_a, best_score


def local_search_phase(a_init, time_limit, rng, verbose=True):
    """Simulated annealing with targeted peak reduction."""
    n = len(a_init)
    S = np.sum(a_init)
    a = a_init.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t0 = time.monotonic()
    temp = 0.0015
    iter_count = 0
    current_score = best_score

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.35:
            # Targeted: reduce peak contributors
            b = fft_convolve(a, a)
            peak_idx = int(np.argmax(b))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            k = max(3, n // 14)
            top_idx = np.argsort(contribs)[-k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.008, size=k))
            other = np.setdiff1d(np.arange(n), top_idx)
            n_inc = min(6, len(other))
            if n_inc > 0:
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.012, 0.008, size=n_inc))
        elif r < 0.6:
            # Block modification
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.8:
            # Symmetrize partially
            sym = 0.5 * (a + a[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.008, size=n))
        else:
            # Small global
            candidate = a * np.exp(rng.normal(0.0, 0.005, size=n))

        candidate = np.maximum(candidate, 0.0)
        total = np.sum(candidate)
        if total < 1e-10:
            continue
        candidate = candidate * S / total

        new_score = compute_score(candidate)

        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
            a = candidate
            current_score = new_score
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            a = candidate
            current_score = new_score

        temp = max(temp * 0.9997, 1e-7)
        iter_count += 1

        if verbose and iter_count % 5000 == 0:
            print(f"    LS iter {iter_count}: best={best_score:.6f}, "
                  f"temp={temp:.2e}, t={time.monotonic()-t0:.1f}s")

    if verbose:
        print(f"    Local search done: {iter_count} iters, best={best_score:.6f}, "
              f"time={time.monotonic()-t0:.1f}s")
    return best_a, best_score


def make_chirp_init(n, alpha, rng):
    """Create a chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
