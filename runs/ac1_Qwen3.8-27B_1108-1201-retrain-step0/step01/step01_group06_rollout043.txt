def correlation_from_weights(a, p):
    n = a.size
    L = 2*n - 1
    p_rev = p[::-1]
    conv_size = 1
    conv_size_needed = n + L - 1
    while conv_size < conv_size_needed:
        conv_size *= 2
    A = np.fft.rfft(a, conv_size)
    B = np.fft.rfft(p_rev, conv_size)
    c = np.fft.irfft(A * B, conv_size)
    idx = (2*n - 2) - np.arange(n)
    return c[idx]
```

    if n == 0: return np.zeros(0)
    L = 2*n-1
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size != L:
        # trim/pad?
        if p.size > L: p = p[:L]
        else: p = np.pad(p, (0, L-p.size))
    s = p.sum()
    if s <= 0:

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2*n - 1
    size = 1
    while size < L:
        size <<= 1
    A = np.fft.rfft(a, size)
    b = np.fft.irfft(A * A, size)[:L]
    b = np.where(b > 0.0, b, 0.0)
    return b

```python
"""
Hybrid continuous/nonconvex search for minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2,
a >= 0.

The program combines:
  * fast FFT-based smooth-exponentiated-gradient ("Adam on softmax") local searches,
  * a successive linear programming "local sublevel" step that solves a first-order
    LP to find a direction that lowers the active convolution peaks while keeping
    all other coefficients below the current peak to first order,
  * aggressive multi-start, multi-length, parent-seed, and polishing stages.

The objective is scale-invariant, so internal optimization normalizes sequences to
sum 1.  The returned sequence is normalized as well.
"""

import numpy as np
import time

try:
    import scipy.optimize as _scipy_optimize
    import scipy.sparse as _scipy_sparse
    _SCIPY_OPT_AVAILABLE = True
except Exception:
    _SCIPY_OPT_AVAILABLE = False

_CONV_CACHE = {}


def normalize_nonnegative(a):
    """Clip to nonnegative finite values and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def symmetrize_sequence(a):
    """Make a sequence palindromic and renormalize."""
    a = normalize_nonnegative(a)
    out = 0.5 * (a + a[::-1])
    return normalize_nonnegative(out)


def resize_sequence(a, n):
    """Linearly resample a nonnegative probability vector to length n."""
    a = normalize_nonnegative(a)
    n = int(n)
    if n <= 0:
        n = 1
    if a.size == n:
        return a.copy()
    if a.size == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(x_new, x_old, a)
    return normalize_nonnegative(out)


def fast_convolve(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    size = 1
    while size < L:
        size <<= 1
    A = np.fft.rfft(a, size)
    b = np.fft.irfft(A * A, size)[:L]
    b = np.where(b > 0.0, b, 0.0)
    return b


def score_internal(a):
    """
    Internal objective assuming a is normalized to sum 1:
        2*n*max(conv(a,a)).
    """
    a = normalize_nonnegative(a)
    n = a.size
    if n == 0:
        return float("inf")
    b = fast_convolve(a)
    if b.size == 0:
        return float("inf")
    m = float(b.max())
    if not np.isfinite(m):
        return float("inf")
    return 2.0 * float(n) * m


def softmax_func(x):
    """Stable softmax."""
    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, -700.0, 700.0)
    x = x - float(np.max(x))
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        out = np.zeros_like(x)
        out[int(np.argmax(e))] = 1.0
        return out
    return e / s


def correlation_from_weights(a, p):
    """
    For a length n and p length 2n-1, return s_l = sum_j a_j p_{j+l}
    for l=0..n-1.  This is used for the gradient of weighted convolution peaks.
    """
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size != L:
        if p.size > L:
            p = p[:L]
        else:
            p = np.pad(p, (0, L - p.size), mode="constant")
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-300:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(p))] = 1.0
    else:
        p = p / s

    p_rev = p[::-1]
    needed = n + L - 1
    size = 1
    while size < needed:
        size <<= 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(p_rev, size)
    c = np.fft.irfft(A * B, size)
    idx = (2 * n - 2) - np.arange(n, dtype=np.int64)
    return c[idx]


def make_beta_init(n, alpha):
    """Smooth beta-distribution-like initialization on [0,1]."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    if abs(alpha - 1.0) < 1e-12:
        a = np.ones(n, dtype=np.float64)
    else:
        xc = np.clip(x, 1e-4, 1.0 - 1e-4)
        a = np.exp((alpha - 1.0) * (np.log(xc) + np.log(1.0 - xc)))
        a = np.maximum(a, 0.0)
    return normalize_nonnegative(a)


def make_gauss_init(n, sigma):
    """Gaussian bump initialization."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = np.exp(-0.5 * ((x - 0.5) / max(1e-6, float(sigma))) ** 2)
    return normalize_nonnegative(a)


def make_random_smooth_init(n, rng, symmetric=False):
    """Smoothed random log-density initialization."""
    z = rng.standard_normal(n)
    kern = np.array([0.2, 0.6, 0.2], dtype=np.float64)
    for _ in range(3):
        z = np.convolve(z, kern, mode="same")
    z = np.clip(z, -5.0, 5.0)
    a = np.exp(0.6 * z)
    if symmetric:
        a = 0.5 * (a + a[::-1])
    return normalize_nonnegative(a)


def make_spectral_init(n):
    """
    Spectral initialization: choose a DFT whose squared magnitude approximates
    the DFT of a flat length-(2n-1) convolution, then inverse FFT and clip.
    """
    L = 2 * n - 1
    size = 1
    while size < L:
        size <<= 1
    theta = 2.0 * np.pi * np.arange(size, dtype=np.float64) / float(size)
    half = 0.5 * L * theta
    denom = 0.5 * theta
    with np.errstate(divide="ignore", invalid="ignore"):
        sinc = np.sin(half) / np.sin(denom)
    sinc[0] = float(L)
    mag = np.abs(sinc)
    phase = np.exp(1j * (float(n - 1) * theta) / 2.0)
    A = mag * phase
    a = np.fft.irfft(A, size)[:n]
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    return a / s


def generate_initializations(n, rng, parent):
    """Generate a diverse set of initial sequences of length n."""
    n = int(n)
    inits = []

    inits.append(np.ones(n, dtype=np.float64) / float(n))

    for alpha in (0.4, 0.6, 0.8, 1.2, 2.0, 4.0):
        inits.append(make_beta_init(n, alpha))

    for sigma in (0.15, 0.30, 0.60):
        inits.append(make_gauss_init(n, sigma))

    edge_len = max(1, n // 8)
    edge = np.zeros(n, dtype=np.float64)
    edge[:edge_len] = 1.0
    edge[-edge_len:] = 1.0
    inits.append(normalize_nonnegative(edge))

    for sym in (False, True):
        inits.append(make_random_smooth_init(n, rng, sym))
        inits.append(make_random_smooth_init(n, rng, sym))

    spec = make_spectral_init(n)
    if spec is not None:
        inits.append(spec)
        inits.append(symmetrize_sequence(spec))

    if parent is not None:
        p = resize_sequence(parent, n)
        inits.append(p)
        inits.append(symmetrize_sequence(p))

    return inits


def get_parent_sequence():
    """Try to read the global height_sequence_1 if it exists."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if p.size > 0 and s > 1e-12:
            return p / s
    except Exception:
        pass
    return None


def adam_search(a_init, time_limit, seed=0, focus_start=8.0, focus_end=300.0,
                lr=0.05, beta1=0.9, beta2=0.99, patience_max=100):
    """
    Adam on the log-softmax parameterization.  The weights p_k are a softmax of
    the convolution coefficients, giving a smoothed max derivative.
    Returns (best_sequence, best_internal_score).
    """
    a_init = normalize_nonnegative(a_init)
    n = a_init.size
    if n == 0 or time_limit <= 0:
        return a_init.copy(), score_internal(a_init)

    eps = 1e-12
    x = np.log(np.maximum(a_init, eps))
    x = x - float(np.mean(x))

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    steps = 0

    focus = float(focus_start)
    lr_cur = float(lr)
    patience = 0
    restart_count = 0

    best_a = a_init.copy()
    best_score = score_internal(a_init)
    best_x = x.copy()

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        a = softmax_func(x)
        b = fast_convolve(a)
        if b.size == 0:
            break
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax <= 1e-15:
            break

        score = 2.0 * float(n) * bmax
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_x = x.copy()
            patience = 0
        else:
            patience += 1

        elapsed = time.monotonic() - t0
        base_progress = min(1.0, elapsed / max(1.0, time_limit * 0.6))
        base_focus = float(focus_start) + (float(focus_end) - float(focus_start)) * base_progress
        if focus < base_focus:
            focus = base_focus

        if patience > patience_max:
            patience = 0
            lr_cur *= 0.75
            focus = min(float(focus_end) * 2.0, focus * 1.35)
            if lr_cur < 0.004 or (focus > float(focus_end) * 1.8 and restart_count < 12):
                restart_count += 1
                rng = np.random.default_rng(int(seed + 1000 * restart_count + steps))
                sigma = 0.05 + 0.02 * restart_count
                x = best_x + sigma * rng.standard_normal(n)
                x = x - float(np.mean(x))
                x = np.clip(x, -50.0, 50.0)
                m = np.zeros(n, dtype=np.float64)
                v = np.zeros(n, dtype=np.float64)
                steps = 0
                lr_cur = float(lr)
                focus = float(focus_start)
                patience = 0
                if restart_count > 12:
                    break

        scaled = focus * b / bmax
        scaled = scaled - float(scaled.max())
        e = np.exp(np.clip(scaled, -745.0, 0.0))
        ssum = float(e.sum())
        if not np.isfinite(ssum) or ssum <= 1e-15:
            p = np.zeros_like(b)
            p[int(np.argmax(b))] = 1.0
        else:
            p = e / ssum

        s = correlation_from_weights(a, p)
        mu = float(np.dot(p, b))
        grad = 2.0 * a * (s - mu)
        grad -= float(np.mean(grad))

        steps += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad

        b1 = 1.0 - beta1 ** steps
        b2 = 1.0 - beta2 ** steps
        x = x - lr_cur * (m / b1) / (np.sqrt(v / b2) + 1e-8)
        x = x - float(np.mean(x))
        x = np.clip(x, -50.0, 50.0)

    return best_a, best_score


def conv_matrix_indices(n):
    """
    Precompute row/col indices for the n x n cross-convolution matrix A where
    (A delta)_k = sum_j a_{k-j} delta_j.  Values are not stored.
    """
    cached = _CONV_CACHE.get(n)
    if cached is not None:
        return cached

    rows = np.empty(n * n, dtype=np.int64)
    cols = np.empty(n * n, dtype=np.int64)
    idx = 0
    for j in range(n):
        start = idx
        stop = idx + n
        rows[start:stop] = np.arange(j, j + n, dtype=np.int64)
        cols[start:stop] = j
        idx = stop

    _CONV_CACHE[n] = (rows, cols)
    return rows, cols


def lp_direction_step(a, active_tol=1e-5, delta_scale=50.0):
    """
    One successive-linear-programming local step.

    Given current a (sum 1), compute scaled a_s = a / sqrt(max conv(a,a)) so that
    the current peak is 1.  Find delta and margin t maximizing t subject to:
        b_k + 2(A delta)_k <= 1 - t_k (active peaks),
        b_k + 2(A delta)_k <= 1        (others),
        delta_i >= -a_s_i,
        sum delta = 0,
        0 <= t <= 1.
    Then line-search a + alpha*delta/scale.
    """
    if not _SCIPY_OPT_AVAILABLE:
        return None

    a = normalize_nonnegative(a)
    n = a.size
    if n < 2 or n > 600:
        return None

    b = fast_convolve(a)
    bmax = float(b.max())
    if not np.isfinite(bmax) or bmax <= 1e-15:
        return None

    scale = 1.0 / np.sqrt(bmax)
    as_ = a * scale
    bs = b / bmax
    bs = np.clip(bs, 0.0, 1.0)

    active = bs >= (1.0 - max(1e-8, float(active_tol)))
    if not np.any(active):
        active[int(np.argmax(bs))] = True

    rows, cols = conv_matrix_indices(n)
    avals = np.tile(as_, n)
    vals = 2.0 * avals
    active_idx = np.where(active)[0]

    rows_all = np.concatenate((rows, active_idx))
    cols_all = np.concatenate((cols, np.full(active_idx.size, n, dtype=np.int64)))
    vals_all = np.concatenate((vals, np.ones(active_idx.size, dtype=np.float64)))

    L = 2 * n - 1
    A = _scipy_sparse.csr_matrix((vals_all, (rows_all, cols_all)), shape=(L, n + 1))

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_ub = 1.0 - bs

    upper_base = np.maximum(as_ * float(delta_scale), 2.0 * scale / float(n))
    bounds = []
    for i in range(n):
        bounds.append((-float(as_[i]), float(upper_base[i])))
    bounds.append((0.0, 1.0))

    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = -1.0

    res = None
    for relax in (0.0, 1e-7, 1e-5):
        try:
            b_try = b_ub + relax
            res = _scipy_optimize.linprog(
                c,
                A_ub=A,
                b_ub=b_try,
                A_eq=A_eq,
                b_eq=[0.0],
                bounds=bounds,
                method="highs",
                options={"disp": False, "presolve": True},
            )
            if res.status == 0:
                break
        except Exception:
            res = None

    if res is None or res.status != 0:
        return None

    delta = np.asarray(res.x[:n], dtype=np.float64)
    tval = float(res.x[n])
    if not np.isfinite(tval) or tval <= 1e-9:
        return None
    if not np.all(np.isfinite(delta)):
        return None

    current_score = 2.0 * float(n) * bmax
    best_score = current_score
    best_cand = None

    alphas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.0]
    alpha_t = min(1.0, max(0.01, 0.25 / (tval + 0.05)))
    alphas.append(float(alpha_t))

    alphas_sorted = []
    for val in sorted(alphas):
        if val < 0.001 or val > 1.0:
            continue
        if not alphas_sorted or abs(val - alphas_sorted[-1]) > 1e-4:
            alphas_sorted.append(float(val))

    for alpha in alphas_sorted:
        cand = as_ + alpha * delta
        if not np.all(np.isfinite(cand)):
            continue
        if cand.min() < 0.0:
            cand = np.where(cand > 0.0, cand, 0.0)
        s = float(cand.sum())
        if s <= 1e-15:
            continue
        cand_norm = cand / s
        bm = float(fast_convolve(cand_norm).max())
        score = 2.0 * float(n) * bm
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_cand = cand_norm

    if best_cand is None or best_score >= current_score - 1e-12:
        return None

    return best_cand, best_score


def lp_search(a_init, time_limit, seed=0, active_tol=1e-5, delta_scale=50.0):
    """
    Iteratively apply LP-direction steps, with perturbations and occasional Adam
    fallbacks when the LP step stalls.
    """
    a_init = normalize_nonnegative(a_init)
    a = a_init.copy()
    score = score_internal(a)
    best_a = a.copy()
    best_score = score
    n = a.size

    if time_limit <= 0:
        return best_a, best_score

    if (not _SCIPY_OPT_AVAILABLE) or n < 2 or n > 512:
        return adam_search(a_init, time_limit, seed=seed,
                           focus_start=8.0, focus_end=400.0)

    t0 = time.monotonic()
    no_improve = 0
    param_idx = 0
    restart = 0
    rng = np.random.default_rng(seed)

    param_list = [
        (float(active_tol), float(delta_scale)),
        (float(active_tol) * 2.0, float(delta_scale) * 1.5),
        (max(1e-4, float(active_tol) * 10.0), float(delta_scale) * 0.7),
        (max(1e-6, float(active_tol) * 0.5), float(delta_scale) * 1.2),
    ]

    while time.monotonic() - t0 < time_limit:
        tol, ds = param_list[param_idx % len(param_list)]
        res = lp_direction_step(a, active_tol=tol, delta_scale=ds)

        if res is not None:
            a_new, score_new = res
            if np.isfinite(score_new) and score_new < best_score - 1e-12:
                best_score = score_new
                best_a = a_new.copy()
            if np.isfinite(score_new) and score_new < score - 1e-12:
                a = a_new
                score = score_new
                no_improve = 0
                param_idx = 0
                continue

        no_improve += 1

        if no_improve > 6:
            no_improve = 0
            param_idx += 1
            improved = False

            base_x = np.log(np.maximum(best_a, 1e-12))
            base_x = base_x - float(np.mean(base_x))

            attempts = 2 if n > 256 else 3
            amps = [0.05, 0.10, 0.20, 0.40]
            amp_list = amps[:2] if n > 256 else amps

            for _ in range(attempts):
                noise = rng.standard_normal(n)
                klen = max(3, n // 20 + 1)
                kernel = np.ones(klen, dtype=np.float64)
                kernel /= kernel.sum()
                noise = np.convolve(noise, kernel, mode="same")

                for amp in amp_list:
                    cand_x = np.clip(base_x + amp * noise, -50.0, 50.0)
                    cand = np.exp(cand_x)
                    cand = normalize_nonnegative(cand)

                    sc1 = score_internal(cand)
                    if np.isfinite(sc1) and sc1 < best_score - 1e-12:
                        best_score = sc1
                        best_a = cand.copy()
                        a = cand
                        score = sc1
                        improved = True
                        break

                    cand_s = symmetrize_sequence(cand)
                    sc2 = score_internal(cand_s)
                    if np.isfinite(sc2) and sc2 < best_score - 1e-12:
                        best_score = sc2
                        best_a = cand_s.copy()
                        a = cand_s
                        score = sc2
                        improved = True
                        break

                if improved:
                    break

            if not improved:
                rem = time_limit - (time.monotonic() - t0)
                if rem > 15:
                    a_try, sc_try = adam_search(
                        best_a,
                        min(30.0, rem * 0.5),
                        seed=int(seed) + restart * 131 + 7,
                        focus_start=15.0,
                        focus_end=400.0,
                    )
                    if np.isfinite(sc_try) and sc_try < best_score - 1e-12:
                        best_score = sc_try
                        best_a = a_try.copy()
                    a = best_a
                    score = best_score

            restart += 1
            if restart > 24:
                break

    return best_a, best_score


def hybrid_search(a_init, time_limit, seed=0):
    """
    A short Adam phase followed by LP polishing.
    """
    a = normalize_nonnegative(a_init)
    best_a = a.copy()
    best_score = score_internal(a)
    n = a.size

    if time_limit <= 0:
        return best_a, best_score

    if n > 512:
        a1, s1 = adam_search(a, time_limit, seed=seed,
                             focus_start=8.0, focus_end=400.0)
        if np.isfinite(s1) and s1 < best_score:
            best_score = s1
            best_a = a1.copy()
        return best_a, best_score

    t0 = time.monotonic()

    if time_limit > 80:
        t1 = min(0.35 * time_limit, 50.0)
    else:
        t1 = 0.25 * time_limit

    if t1 > 3:
        a1, s1 = adam_search(a, t1, seed=seed,
                             focus_start=8.0, focus_end=300.0)
        if np.isfinite(s1) and s1 < best_score:
            best_score = s1
            best_a = a1.copy()

    rem = max(0.0, time_limit - (time.monotonic() - t0))
    if rem > 8:
        a2, s2 = lp_search(best_a, rem, seed=seed + 1)
        if np.isfinite(s2) and s2 < best_score:
            best_score = s2
            best_a = a2.copy()

    return best_a, best_score


def choose_top_candidates(candidates, max_per_length=3, max_total=16):
    """
    Select top candidates by score while preserving some length diversity.
    candidates: list of (score, n, sequence).
    """
    by_length = {}
    for item in candidates:
        try:
            score, n, _a = item
        except Exception:
            continue
        if not np.isfinite(score):
            continue
        n = int(n)
        if n not in by_length:
            by_length[n] = []
        by_length[n].append(item)

    selected = []
    for n, lst in by_length.items():
        if not lst:
            continue
        scores = np.array([float(x[0]) for x in lst], dtype=np.float64)
        order = np.argsort(scores)
        count = 0
        for idx in order:
            selected.append(lst[int(idx)])
            count += 1
            if count >= max_per_length:
                break

    if not selected:
        return []

    scores = np.array([float(x[0]) for x in selected], dtype=np.float64)
    order = np.argsort(scores)

    chosen = []
    counts = {}
    for idx in order:
        item = selected[int(idx)]
        n = int(item[1])
        c = counts.get(n, 0)
        if c < max_per_length:
            chosen.append(item)
            counts[n] = c + 1
            if len(chosen) >= max_total:
                break

    return chosen


def propose_candidate(seed=42, budget_s=2350, **kwargs):
    """
    Main entry point.  Hybrid multi-start search with LP polishing.
    """
    start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2350.0
    budget = max(60.0, min(budget, 2380.0))
    deadline = start + max(60.0, budget - 25.0)

    best_a = None
    best_score = float("inf")
    explored = []

    parent = get_parent_sequence()

    # --------------------------
    # 1) Strong initial push from the previous best, if available.
    # --------------------------
    if parent is not None:
        a_par = normalize_nonnegative(parent)
        sc1 = score_internal(a_par)
        a_par_sym = symmetrize_sequence(a_par)
        sc2 = score_internal(a_par_sym)
        if sc2 < sc1:
            a_par = a_par_sym
            sc1 = sc2

        a_run = a_par
        if a_run.size > 512:
            a_run = resize_sequence(a_par, 512)

        par_time = min(500.0, max(180.0, budget * 0.22))
        now = time.monotonic()
        if now + par_time < deadline:
            a1, s1 = lp_search(a_run, par_time, seed=1001)
            if np.isfinite(s1) and s1 < best_score:
                best_score = s1
                best_a = a1.copy()
            if np.isfinite(s1):
                explored.append((s1, int(a1.size), a1.copy()))

            rem_par = min(120.0, max(30.0, budget * 0.06),
                          deadline - time.monotonic() - 350.0)
            if rem_par > 20 and best_a is not None:
                a2, s2 = adam_search(best_a, rem_par, seed=1002,
                                     focus_start=20.0, focus_end=500.0)
                if np.isfinite(s2) and s2 < best_score:
                    best_score = s2
                    best_a = a2.copy()
                if np.isfinite(s2):
                    explored.append((s2, int(a2.size), a2.copy()))

    # --------------------------
    # 2) Generate diverse initial candidates.
    # --------------------------
    lengths = [64, 96, 128, 192, 256, 384, 512]
    if parent is not None:
        pp = int(parent.size)
        for cand_n in (pp, pp // 2, pp * 2):
            if 32 <= cand_n <= 1024:
                lengths.append(int(cand_n))
    lengths = sorted(set([n for n in lengths if 32 <= n <= 1024]))
    if not lengths:
        lengths = [128]

    rng = np.random.default_rng(seed)
    candidates = []
    for n in lengths:
        inits = generate_initializations(n, rng, parent)
        for a0 in inits:
            sc = score_internal(a0)
            if np.isfinite(sc):
                candidates.append((sc, int(n), a0.copy()))

    for item in explored:
        candidates.append(item)

    # --------------------------
    # 3) Quick multi-start Adam scanning.
    # --------------------------
    reserve_final = min(700.0, max(250.0, budget * 0.28))
    phase2_budget = min(700.0, max(250.0, budget * 0.30))
    phase1_end = max(time.monotonic() + 30.0,
                     deadline - reserve_final - phase2_budget)

    cand_list = choose_top_candidates(candidates, max_per_length=2, max_total=14)
    results = list(explored)

    idx = 0
    while idx < len(cand_list) and time.monotonic() < phase1_end:
        sc0, n, a0 = cand_list[idx]
        time_left = phase1_end - time.monotonic()
        if time_left < 5:
            break
        remaining = max(1, len(cand_list) - idx)
        t_alloc = max(5.0, time_left / float(remaining))
        t_alloc = min(45.0, t_alloc)
        cutoff = deadline - reserve_final - phase2_budget
        if time.monotonic() + t_alloc > cutoff:
            t_alloc = max(3.0, cutoff - time.monotonic())
        if t_alloc < 3:
            break

        if t_alloc > 35:
            a1, s1 = hybrid_search(a0, t_alloc, seed=int(seed) + 2000 + idx)
        else:
            a1, s1 = adam_search(a0, t_alloc, seed=int(seed) + 2000 + idx)

        if np.isfinite(s1) and s1 < best_score:
            best_score = s1
            best_a = a1.copy()
        if np.isfinite(s1):
            results.append((s1, int(a1.size), a1.copy()))
        idx += 1

    # --------------------------
    # 4) Deeper LP polishing on promising candidates.
    # --------------------------
    phase2_end = deadline - reserve_final
    phase2_items = []
    if best_a is not None:
        phase2_items.append((best_score, int(best_a.size), best_a.copy()))
    for item in choose_top_candidates(results + candidates, max_per_length=2, max_total=8):
        if len(phase2_items) < 8:
            phase2_items.append(item)

    for i, item in enumerate(phase2_items):
        if time.monotonic() > phase2_end:
            break
        sc0, n, a0 = item
        time_left = phase2_end - time.monotonic()
        if time_left < 15:
            break
        remaining = max(1, len(phase2_items) - i)
        t_alloc = max(25.0, time_left / float(remaining))
        t_alloc = min(150.0, t_alloc)
        if time.monotonic() + t_alloc > deadline - reserve_final:
            t_alloc = max(10.0, deadline - reserve_final - time.monotonic())
        if t_alloc < 10:
            break

        a1, s1 = lp_search(a0, t_alloc, seed=int(seed) + 3000 + i)
        if np.isfinite(s1) and s1 < best_score:
            best_score = s1
            best_a = a1.copy()
        if np.isfinite(s1):
            results.append((s1, int(a1.size), a1.copy()))

    # --------------------------
    # 5) Final cyclic polishing.
    # --------------------------
    final_pool = []
    if best_a is not None:
        final_pool.append((best_score, int(best_a.size), best_a.copy()))
    for item in choose_top_candidates(results + candidates, max_per_length=2, max_total=7):
        if len(final_pool) < 5:
            final_pool.append(item)

    cycle = 0
    while time.monotonic() < deadline - 10 and final_pool:
        item = final_pool[cycle % len(final_pool)]
        sc0, n, a0 = item
        remaining = deadline - 10 - time.monotonic()
        if remaining < 15:
            break
        chunk = min(90.0, max(20.0, remaining / 3.0))

        if n > 512 or chunk < 30:
            a1, s1 = adam_search(a0, chunk, seed=int(seed) + 4000 + cycle,
                                 focus_start=10.0, focus_end=500.0)
        else:
            if cycle % 2 == 0:
                a1, s1 = lp_search(a0, chunk, seed=int(seed) + 4000 + cycle)
            else:
                a1, s1 = hybrid_search(a0, chunk, seed=int(seed) + 4000 + cycle)

        if np.isfinite(s1) and s1 < best_score:
            best_score = s1
            best_a = a1.copy()
            results.append((s1, int(a1.size), a1.copy()))

        cycle += 1
        if cycle > 10000:
            break

    # --------------------------
    # 6) Final cheap transformations / polish.
    # --------------------------
    if best_a is not None:
        final_candidates = [best_a, symmetrize_sequence(best_a)]
        if best_a.size > 10:
            for frac in (0.02, 0.05, 0.10, 0.20):
                thresh = float(np.quantile(best_a, frac))
                ah = best_a * (best_a > thresh)
                ah = normalize_nonnegative(ah)
                final_candidates.append(ah)
                final_candidates.append(symmetrize_sequence(ah))

        for ac in final_candidates:
            sc = score_internal(ac)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = ac.copy()

        if time.monotonic() < deadline - 15 and best_a is not None and best_a.size <= 512:
            a1, s1 = lp_search(best_a, max(5.0, min(30.0, deadline - 15 - time.monotonic())),
                               seed=int(seed) + 5000)
            if np.isfinite(s1) and s1 < best_score:
                best_score = s1
                best_a = a1.copy()

    if best_a is None:
        best_a = np.ones(128, dtype=np.float64) / 128.0
        best_score = score_internal(best_a)

    best_a = normalize_nonnegative(best_a)
    a_sym = symmetrize_sequence(best_a)
    if score_internal(a_sym) < score_internal(best_a):
        best_a = a_sym

    out = [float(x) for x in best_a]
    cleaned = []
    for x in out:
        if not np.isfinite(x):
            x = 0.0
        if x < 0.0:
            x = 0.0
        if x > 1000.0:
            x = 1000.0
        cleaned.append(float(x))

    ssum = float(sum(cleaned))
    if ssum > 1e-15:
        cleaned = [x / ssum for x in cleaned]

    try:
        print("final internal score:", score_internal(np.asarray(cleaned, dtype=np.float64)))
        ev = evaluate_sequence(cleaned)
        print("evaluate_sequence(score):", ev)
        if not np.isfinite(ev):
            best_a = np.ones(128, dtype=np.float64) / 128.0
            cleaned = [float(x) for x in best_a]
    except Exception:
        pass

    print("length:", len(cleaned))
    return cleaned
