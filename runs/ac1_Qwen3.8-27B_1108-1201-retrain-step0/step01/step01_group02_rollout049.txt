```python
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def normalize(a):
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def lp_step(a, M, timeout=10.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    n = len(a)
    # Build sparse convolution matrix: A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For efficiency, build as a sparse matrix
    rows = []
    cols = []
    data = []
    for j in range(n):
        # For column j, non-zero entries are at rows k where 0 <= k-j < n, i.e., j <= k < j+n
        k_start = max(0, j)  # Actually k-j >= 0 means k >= j. And k-j < n means k < j+n.
        # So k ranges from j to min(j+n-1, 2n-2)
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        for k in range(k_lo, k_hi + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)  # Minimize -sum(b) = maximize sum(b)
    b_vec = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': timeout})
    if res.success and res.x is not None:
        return np.maximum(0, res.x)
    return None

def lp_refine(a, time_limit=600, verbose=True):
    a = normalize(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    start = time.monotonic()
    
    stagnant = 0
    for it in range(1000):
        if time.monotonic() - start > time_limit:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum()**2)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or score < 1.52):
                print(f"  LP iter {it}: score={score:.6f}")
        else:
            stagnant += 1
            if stagnant > 30:
                break
        
        b = lp_step(a, M, timeout=max(2, min(30, (time_limit - (time.monotonic()-start))*0.2)))
        if b is None:
            break
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-10:
            break
        b = b / sum_b
        
        # Try several step sizes
        best_alpha = 0
        best_new_score = score
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1-alpha)*a + alpha*b
            a_trial = normalize(a_trial)
            trial_score = score_sequence(a_trial)
            if trial_score < best_new_score:
                best_new_score = trial_score
                best_alpha = alpha
```
