"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses Adam optimizer with FFT-accelerated soft-max loss and gradient,
simplex projection, multi-scale search over sequence lengths,
and peak-shaving post-processing.
"""

import numpy as np
import time


def fft_convolve_self(a):
    """Compute linear self-convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fc = fa * fa
    c = np.fft.irfft(fc, size)
    return c[:2 * n - 1]


def compute_conv(a):
    """Compute self-convolution, using FFT for large inputs."""
    n = len(a)
    if n > 4000:
        return fft_convolve_self(a)
    else:
        return np.convolve(a, a)


def fft_correlate(w, a):
    """Compute correlate(w, a, 'valid') using FFT for efficiency.
    w has length 2n-1, a has length n. Returns array of length n.
    """
    n = len(a)
    ar = a[::-1]
    size = 1
    while size < 3 * n - 1:
        size *= 2
    fa = np.fft.rfft(ar, size)
    fw = np.fft.rfft(w, size)
    fc = np.fft.irfft(fa * fw, size)
    return fc[n - 1:2 * n - 1]


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    if abs(s - S) < 1e-12:
        return a
    lo, hi = 0.0, a.max()
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    diff = S - res.sum()
    if abs(diff) > 1e-12:
        idx = res.argmax()
        res[idx] += diff
        if res[idx] < 0:
            res[idx] = 0
            res += diff / n
            res = np.maximum(res, 0.0)
    return res


def soft_max_weights(conv, tau):
    """Compute soft-max weights w[k] proportional to exp(conv[k]/tau)."""
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    exp_s = np.exp(shifted)
    s = exp_s.sum()
    if s < 1e-300:
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
    else:
        w = exp_s / s
    return w


def compute_grad_fft(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a using FFT."""
    n = len(a)
    if n > 4000:
        corr = fft_correlate(w, a)
    else:
        corr = np.correlate(w, a, 'valid')
    return 2.0 * corr


def init_chirp(n, seed=0):
    """Chirp-based non-negative initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.8 * np.cos(2.0 * np.pi * t * t / (n * 0.3))
    rng = np.random.default_rng(seed)
    a += 0.03 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_cosine(n, seed=0):
    """Cosine-window based initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (t - 0.5 * (n - 1)) / n) ** 2
    rng = np.random.default_rng(seed)
    a += 0.02 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_random(n, seed=0):
    """Random positive initialization."""
    return np.random.default_rng(seed).exponential(1.0, size=n)


def init_linear(n, seed=0):
    """Monotonically varying initialization (exploits asymmetry)."""
    t = np.arange(n, dtype=np.float64)
    a = 0.3 + 0.7 * (t / max(n - 1, 1)) ** 0.5
    rng = np.random.default_rng(seed)
    a += 0.02 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def optimize_adam(a0, n, t_budget, verbose=False):
    """
    Optimize sequence using Adam with soft-max loss and simplex projection.
    Minimizes max(conv(a,a)) subject to sum(a)=sqrt(2n), a>=0.
    """
    S = np.sqrt(2.0 * n)
    a = (a0.astype(np.float64) / max(a0.sum(), 1e-12) * S).copy()
    a = project_simplex(a, S)

    best_a = a.copy()
    best_conv = compute_conv(best_a)
    best_M = best_conv.max()

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Hyperparameters
    lr_max = 0.5
    tau_start = max(0.5, best_M * 0.05)
    tau_end = 0.005
    check_interval = 20

    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(t_budget, 1e-6))

        # Cosine LR schedule
        lr = lr_max * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, 0.001)

        # Anneal tau geometrically
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.6)

        # Compute convolution
        conv = compute_conv(a)

        # Compute soft-max weights and gradient
        w = soft_max_weights(conv, tau)
        grad = compute_grad_fft(a, w)

        # Adam step
        step += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad ** 2
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new, S)

        # Check for improvement
        if step % check_interval == 0:
            conv_new = compute_conv(a_new)
            M_new = conv_new.max()
            if M_new < best_M - 1e-10:
                best_M = M_new
                best_a = a_new.copy()
            a = a_new
        else:
            a = a_new

    # Final evaluation

    if verbose:
        print(f"    optimize_adam: steps={step}, M={final_M:.6f}, score={final_score:.6f}")

    return best_a, final_score


def peak_shave(a, n, iterations=100, verbose=False):
    """
    Direct peak-shaving: identify max of conv, reduce contributing elements.
    """
    S = np.sqrt(2.0 * n)
    a = (a / max(a.sum(), 1e-12) * S).copy()
    a = project_simplex(a, S)

    best_a = a.copy()
    best_M = compute_conv(best_a).max()

    for _ in range(iterations):
        conv = compute_conv(a)
        k_star = conv.argmax()

        # Compute contributions to conv[k_star]
        # conv[k_star][k] = sum_j a[j] * a[k_star - j] for valid j
        # Reduce elements that contribute most
        contribs = np.zeros(n)
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contribs[j] = 2.0 * a[j] * a[k2]

        # Find top contributors
        n_reduce = max(1, n // 20)
        top_idx = np.argsort(contribs)[-n_reduce:]

        # Reduce them slightly and add to low contributors
        reduce_amount = contribs[top_idx] * 0.05
        total_reduce = reduce_amount.sum()

        low_mask = contribs < np.median(contribs)
        if low_mask.sum() > 0:
            low_idx = np.where(low_mask)[0]
            add_weight = np.ones(len(low_idx))
            add_weight /= add_weight.sum()
            a[low_idx] += total_reduce * add_weight
            a[top_idx] -= reduce_amount

        a = np.maximum(a, 0.0)
        a = project_simplex(a, S)

        conv_new = compute_conv(a)
        if conv_new.max() < best_M - 1e-12:
            best_M = conv_new.max()
            best_a = a.copy()

    final_score = evaluate_sequence(best_a.tolist())
    if verbose:
        print(f"    peak_shave: M={best_M:.6f}, score={final_score:.6f}")
    return best_a, final_score
