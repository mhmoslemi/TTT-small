def conv_full(p):
    """Full linear convolution p*p via FFT."""
    n = int(p.size)
    if n == 0:

    L = _fft_len_conv(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1].astype(np.float64, copy=False)
