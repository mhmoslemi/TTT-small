Minimizes 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS in log-space with
softmax parameterization, followed by coordinate descent for polishing.
Uses adaptive tau annealing and multi-start strategy.
"""

import numpy as np
import time
from scipy.optimize import minimize, brent

# Global to track best
_global_best = [None, np.inf]

def compute_conv(a):
    """Compute linear convolution of a with itself."""
    return np.convolve(a, a)

def soft_max_and_weights(c, tau):
    """Compute soft-max value and weights."""
    c_max = c.max()
    c_shift = c - c_max
    exp_c = np.exp(c_shift / tau)
    w = exp_c / exp_c.sum()
    val = c_max + tau * np.log(exp_c.sum())
    return val, w

def grad_conv(a, q):
    """
    Gradient of sum_k q_k * c_k (where c = conv(a,a)) with respect to a.
    grad_j = 2 * sum_m a_m * q_{m+j}
    """
    n = len(a)
    return 2.0 * np.correlate(q, a, 'full')[n-1:2*n-1]

def softmax(x):
    """Numerically stable softmax."""
    x_shift = x - x.max()
    e = np.exp(x_shift)
    return e / e.sum()

def log_param_obj_grad(x, tau, S):
    """
    Objective and gradient for the log-parameterized problem.
    x: unconstrained parameters
    tau: soft-max temperature
    S: target sum = sqrt(2n)
    """
    n = len(x)
    a_norm = softmax(x)
    a = S * a_norm
    
    c = np.convolve(a, a)
    c_max = c.max()
    c_shift = c - c_max
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    obj = c_max + tau * np.log(exp_c.sum())
    
    grad_a = 2.0 * np.correlate(q, a, 'full')[n-1:2*n-1]
    
    # Chain rule through softmax
    dot = np.dot(grad_a, a_norm)
    grad_x = a_norm * (grad_a - dot)
    
    return obj, grad_x

    """
    Run L-BFGS-B optimization.
    Returns (x_best, obj_best).
    """
    t0 = time.monotonic()
    x_best = x0.copy()

        return log_param_obj_grad(x, tau, S)[0]

        return log_param_obj_grad(x, tau, S)[1]
