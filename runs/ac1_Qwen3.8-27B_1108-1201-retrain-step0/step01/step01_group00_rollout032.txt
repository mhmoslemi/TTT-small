Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions. Uses L-BFGS with smooth max, coordinate descent, and

import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute the score directly (scale-invariant form)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def smooth_score_and_grad(a, tau):

    # = ifft(conj(fft(a)) * fft(v))[i] ... hmm not quite.

    # conv(a_reversed, v)[i + n - 1] = sum_j a_reversed[j] * v[i+n-1-j]
    # = sum_j a[n-1-j] * v[i+n-1-j]
    # Let k = n-1-j: = sum_k a[k] * v[i+k]
    # = corr[i]! 
    
    a_rev = a[::-1]

    """

    a_init = np.clip(a_init, 0.01, 1000.0)
    n = len(a_init)
    
    # We optimize in log-space to maintain positivity
    log_a = np.log(a_init)
    
    t0 = time.monotonic()
    best_score = local_score(a_init)
    best_a = a_init.copy()
    
    def objective(x):
        if time.monotonic() - t0 > time_limit:
            return 1e10
        a = np.exp(x)
        a = np.clip(a, 1e-10, 1000.0)
        s, _ = smooth_score_and_grad(a, tau)
        return s
    
    def objective_grad(x):
        if time.monotonic() - t0 > time_limit:
            return 1e10, np.zeros(n)
        a = np.exp(x)
        a = np.clip(a, 1e-10, 1000.0)
        s, g = smooth_score_and_grad(a, tau)
        # Chain rule: d(score)/d(log_a[i]) = a[i] * g[i]
        return s, a * g
    
    # Use L-BFGS-B
    try:
        result = scipy_minimize(

            method='L-BFGS-B',

"""
LP-based iterative peak reduction for minimizing peak auto-convolution
of non-negative step functions. Uses linear programming to find
"smoother" sequences at each iteration, combined with local search
polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute the score directly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def build_conv_constraint_matrix(a):
    """
    Build the matrix C such that (a * b)[k] = C[k, :] @ b
    for b of length n = len(a).
    C has shape (2n-1, n).
    """
    n = len(a)
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # (a*b)[k] = sum_j a[k-j] * b[j] where 0 <= k-j < n
        j_start = max(0, k - (n-1))
        j_end = min(k, n-1)
        for j in range(j_start, j_end + 1):
            C[k, j] = a[k - j]
    return C

def lp_find_smother(a, M):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns optimal b.
    """
    n = len(a)
    c = -np.ones(n)  # minimize -sum(b)
    
    C = build_conv_constraint_matrix(a)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
        else:
            # Fallback: uniform
            return np.ones(n)
    except Exception:
        return np.ones(n)

def lp_iteration(a, t_max=1.0):
    """
    One LP-based iteration: find smoother direction and update.
    Returns updated sequence.
    """
    n = len(a)
    sum_a = np.sum(a)
    conv = fft_convolve(a, a)
    M = np.max(conv)
    
    # Solve LP
    b = lp_find_smother(a, M)
    sum_b = np.sum(b)
    
    if sum_b <= 0:
        return a.copy()
    
    # Scale b to have same sum as a
    g = b * (sum_a / sum_b)
    
    # Line search for mixing parameter
    best_a = a.copy()
    best_peak = M
    for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        if np.sum(a_new) < 0.01:
            continue
        conv_new = fft_convolve(a_new, a_new)
        peak_new = np.max(conv_new)
        if peak_new < best_peak:
            best_peak = peak_new
            best_a = a_new
    
    return best_a

            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            a_new[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=top_k))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(5, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                a_new[inc_idx] *= np.exp(rng.normal(0.03, 0.015, size=n_inc))
        elif r < 0.6:
            # Block
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - width))
            a_new[start:start+width] *= np.exp(rng.normal(0, 0.03, size=width))
        elif r < 0.75:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 20))))
                start = int(rng.integers(0, n - width))
                a_new[start:start+width] *= np.exp(rng.normal(0, 0.02, size=width))
        else:
            # Global
            a_new = a * np.exp(rng.normal(0, 0.01, size=n))
        
        a_new = np.clip(a_new, 0.001, 1000.0)
        new_score = local_score(a_new)
        
        if new_score < current_score:
            a = a_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(T, 1e-12)):
            a = a_new
            current_score = new_score
        
        T *= cooling
    
    return best_a, best_score
