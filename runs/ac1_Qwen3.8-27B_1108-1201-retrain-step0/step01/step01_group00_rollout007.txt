import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

    t = theta - np.max(theta)
    e = np.exp(t)

def grad_softmax_conv(a, tau):
    """Gradient of soft-max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    value = c_max + tau * np.log(np.sum(exp_shifted))
    # Gradient: d/d(a[j]) sum_i w[i] * C[i] where C[i] = sum_k a[k]*a[i-k]
    # = sum_i w[i] * (a[i-j] + a[j+(n-1-i)]) ... 
    # Actually: dC[i]/da[j] = a[(i-j) mod ...] 
    # Using the correlation trick:
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n-1:2*n-1]
    return value, g, c

def optimize_with_adam(a_init, n_steps, tau_start, tau_end, lr, time_limit):
    """Adam optimization with annealed soft-max."""
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a = a_init / np.sum(a_init) * S_target
    theta = np.log(a + 1e-10)
    
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    step = 0
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = step / n_steps
        tau = tau_start * (tau_end / tau_start) ** progress
        
        a = softmax_normalize(theta) * S_target
        value, g_a, c = grad_softmax_conv(a, tau)
        
        # Project gradient to tangent space (maintain sum constraint)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S_target)
        
        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * g_theta**2
        m1_hat = m1 / (1 - b1**step)
        m2_hat = m2 / (1 - b2**step)
        
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        a_new = softmax_normalize(theta) * S_target
        score = compute_score(a_new)
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
    
    return best_a, best_score

def local_search_peak(a_init, time_limit, rng):
    """Local search that targets the peak of the convolution."""
    a = a_init.copy()
    n = len(a)
    S = np.sum(a)
    score = compute_score(a)
    best_a = a.copy()
    best_score = score
    t0 = time.monotonic()
    temp = 0.001
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        peak_val = np.max(c)
        
        # Find top contributors to the peak
        peak_idx = np.argmax(c)
        # For peak at index k, C[k] = sum_i a[i]*a[k-i]
        # Contributors: for each i, the product a[i]*a[k-i]
        lo = max(0, peak_idx - n + 1)
        hi = min(peak_idx, n - 1)
        contribs = np.zeros(n)
        for i in range(lo, hi + 1):
            j = peak_idx - i
            contribs[i] += a[i] * a[j]
        
        # Strategy: reduce top contributors, increase others
        strategy = rng.random()
        
        if strategy < 0.4:
            # Reduce top-5% contributors at peak, boost random others
            threshold = np.percentile(contribs[contribs > 0], 95)
            top_mask = contribs > threshold
            n_top = np.sum(top_mask)
            if n_top > 0:
                a[top_mask] *= np.exp(rng.normal(-0.05, 0.02, n_top))
            other_mask = ~top_mask
            n_other = np.sum(other_mask)
            if n_other > 0:
                n_boost = min(10, n_other)
                boost_idx = rng.choice(np.where(other_mask)[0], n_boost, replace=False)
                a[boost_idx] *= np.exp(rng.normal(0.05, 0.02, n_boost))
        elif strategy < 0.7:
            # Block modification
            block_size = max(1, n // 16)
            start = rng.integers(0, n - block_size)
            scale = np.exp(rng.normal(0, 0.05, block_size))
            a[start:start+block_size] *= scale
        else:
            # Small random perturbation
            a *= np.exp(rng.normal(0, 0.03, n))
        
        a = np.clip(a, 1e-6, 1000.0)
        new_score = compute_score(a)
        
        if new_score < score or rng.random() < np.exp(-(new_score - score) / temp):
            a = a.copy()
            score = new_score
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        temp *= 0.9995
    
    return best_a, best_score
