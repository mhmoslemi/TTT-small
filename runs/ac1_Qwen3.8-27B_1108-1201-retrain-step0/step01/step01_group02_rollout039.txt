```python
"""
LSE-smoothed L-BFGS optimization for minimizing peak auto-convolution of non-negative sequences.
Uses log-sum-exp smoothing of the max, with annealing of the smoothing parameter β.
Multiple initializations and sequence lengths are explored.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and gradient."""
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = conv.max()
    exp_conv = np.exp(beta * (conv - max_conv))
    lse = (max_conv + np.log(exp_conv.sum()) / beta)
    w = exp_conv / exp_conv.sum()
    sum_a = a.sum()
    score = 2.0 * n * lse / (sum_a ** 2)
    
    # Gradient of LSE w.r.t. a
    # (grad_lse)[j] = sum_k w[k] * a[k-j] (cross-correlation)
    N = 1
    while N < 3 * n:
        N *= 2
    W = np.zeros(N, dtype=np.complex128)
    W[:len(w)] = w
    A = np.zeros(N, dtype=np.complex128)
    A[:n] = a
    grad_lse = np.real(ifft(W * np.conj(A)))[:n]
    
    grad = (2.0 * n / (sum_a ** 2)) * grad_lse - (4.0 * n * lse / (sum_a ** 3)) * np.ones(n)
    return score, grad

    a = np.array(a0, dtype=np.float64).copy()
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s  # normalize
    
    best_a = a.copy()
    best_score = score_sequence(a)
    
    for beta in betas:
        result = minimize(
            fun=lambda x: lse_score_and_grad(x, beta)[0],
            x0=a,
            jac=lambda x: lse_score_and_grad(x, beta)[1],
            method='L-BFGS-B',
            bounds=[(0, None)] * len(a),
            options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol, 'maxls': 50}
        )
        a = np.maximum(0, result.x)
        s = a.sum()
        if s < 1e-15:

            best_a = a.copy()
        if verbose:

    return best_a, best_score
```

```python
"""
LSE-annealed L-BFGS-B optimization for minimizing 2n*max(conv(a,a))/sum(a)^2
over non-negative sequences. Uses log-sum-exp smoothing with annealing,
multiple initializations, and sequence lengths. Polishes with local search.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def lse_obj_and_grad(a, beta):
    n = len(a)
    conv = np.convolve(a, a)
    max_c = np.max(conv)
    shifted = beta * (conv - max_c)
    exp_shifted = np.exp(shifted)
    lse = max_c + np.log(np.sum(exp_shifted)) / beta
    w = exp_shifted / np.sum(exp_shifted)
    sum_a = np.sum(a)
    val = 2.0 * n * lse / (sum_a * sum_a)
    # Gradient via FFT cross-correlation
    N = 1
    while N < 3 * n:
        N *= 2
    W = np.zeros(N, dtype=np.complex128)
    W[:len(w)] = w
    A = np.zeros(N, dtype=np.complex128)
    A[:n] = a
    grad_lse = np.real(ifft(W * np.conj(A)))[:n]
    grad = (2.0 * n / (sum_a * sum_a)) * grad_lse - (4.0 * n * lse / (sum_a ** 3)) * np.ones(n)
    return val, grad

def lse_anneal_optimize(a0, betas, max_iter_per_beta=100, ftol=1e-12, gtol=1e-10):
    a = np.maximum(0.0, np.asarray(a0, dtype=np.float64).copy())
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_exact = score_sequence(a)
    for beta in betas:
        try:
            res = minimize(
                lambda x: lse_obj_and_grad(x, beta)[0],
                a,
                jac=lambda x: lse_obj_and_grad(x, beta)[1],
                method='L-BFGS-B',
                bounds=[(0.0, None)] * len(a),
                options={'maxiter': max_iter_per_beta, 'ftol': ftol, 'gtol': gtol, 'maxls': 30}
            )
            a = np.maximum(0.0, res.x)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            cur = score_sequence(a)
            if cur < best_exact:
                best_exact = cur
                best_a = a.copy()
        except Exception:
            pass
    return best_a, best_exact

def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))

def init_gaussian(n, sigma_frac):
    t = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    sig = sigma_frac * n / 2.0
    return np.exp(-0.5 * ((t - c) / sig) ** 2)

def init_blackman(n):
    if n == 1:
        return np.ones(1)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2*np.pi*x) + 0.08 * np.cos(4*np.pi*x))

def init_bimodal(n, rng, center_frac=0.5, width_frac=0.15):
    t = np.arange(n, dtype=np.float64)
    c1 = center_frac * n - width_frac * n
    c2 = center_frac * n + width_frac * n
    sig = width_frac * n / 4.0
    return np.exp(-0.5*((t-c1)/sig)**2) + np.exp(-0.5*((t-c2)/sig)**2)

def init_flat_with_noise(n, rng, noise_frac=0.1):
    a = np.ones(n, dtype=np.float64)
    a += noise_frac * rng.standard_normal(n)
    return np.maximum(0.0, a)

def interp_sequence(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return np.interp(new_x, old_x, a)
```

```python
def softmax_obj_and_grad(z, beta):
    """Compute LSE-smoothed score and gradient w.r.t. z where a = softmax(z)."""
    z_shifted = z - z.max()  # for numerical stability
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    n = len(a)
    
    # Compute LSE of convolution
    conv = np.convolve(a, a)
    max_c = np.max(conv)
    shifted = beta * (conv - max_c)
    exp_shifted = np.exp(shifted)
    lse = max_c + np.log(np.sum(exp_shifted)) / beta
    w = exp_shifted / np.sum(exp_shifted)
    
    score = 2.0 * n * lse  # since sum(a) = 1
    
    # Gradient of score w.r.t. a
    N = 1
    while N < 3 * n:
        N *= 2
    W = np.zeros(N, dtype=np.complex128)
    W[:len(w)] = w
    A = np.zeros(N, dtype=np.complex128)
    A[:n] = a
    grad_a = 2.0 * n * np.real(ifft(W * np.conj(A)))[:n]
    
    # Gradient of score w.r.t. z through softmax
    # d(score)/dz_j = (d(score)/da_j - sum_k d(score)/da_k * a_k) * a_j
    grad_a_dot_a = np.dot(grad_a, a)
    grad_z = (grad_a - grad_a_dot_a * a) * a
    
    return score, grad_z
```

"""
LSE-annealed L-BFGS-B optimization for minimizing peak auto-convolution.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses log-sum-exp smoothing with beta annealing, multiple inits and lengths.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time


def score_sequence(a):
    """Compute the exact score."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def lse_value_and_gradient(a, beta):
    """Compute LSE-smoothed objective and its gradient.
    
    f(a) = 2n * LSE(beta * conv(a,a)) / (sum(a))^2
    """
    n = len(a)
    conv = np.convolve(a, a)
    mc = np.max(conv)
    shifted = beta * (conv - mc)
    exp_s = np.exp(shifted)
    s_exp = np.sum(exp_s)
    lse = mc + np.log(s_exp) / beta
    w = exp_s / s_exp
    sum_a = np.sum(a)
    val = 2.0 * n * lse / (sum_a * sum_a)
    
    # FFT-based cross-correlation for gradient
    N = 1
    while N < 3 * n:
        N <<= 1
    W = np.zeros(N, dtype=np.complex128)
    W[:len(w)] = w
    A = np.zeros(N, dtype=np.complex128)
    A[:n] = a
    cross_corr = np.real(ifft(W * np.conj(A)))[:n]
    
    # grad = (2n/sum^2) * 2*cross_corr - (4n*lse/sum^3) * 1
    grad = (4.0 * n / (sum_a * sum_a)) * cross_corr - (4.0 * n * lse / (sum_a ** 3)) * np.ones(n)
    return val, grad


def run_lse_optimize(a0, betas, max_iter=100, ftol=1e-13, gtol=1e-11):
    """Run LSE-annealed L-BFGS-B optimization."""
    a = np.maximum(0.0, np.asarray(a0, dtype=np.float64).copy())
    s = a.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_sequence(a)
    n = len(a)
    
    for beta in betas:
        # Create objective and gradient functions
        beta_val = beta  # capture
        
        def _fun(x, _beta=beta_val):
            return lse_value_and_gradient(x, _beta)[0]
        
        def _grad(x, _beta=beta_val):
            return lse_value_and_gradient(x, _beta)[1]
        
        try:
            res = minimize(_fun, a, jac=_grad, method='L-BFGS-B',
                          bounds=[(0.0, None)] * n,
                          options={'maxiter': max_iter, 'ftol': ftol, 'gtol': gtol, 'maxls': 30})
            a = np.maximum(0.0, res.x)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            cur = score_sequence(a)
