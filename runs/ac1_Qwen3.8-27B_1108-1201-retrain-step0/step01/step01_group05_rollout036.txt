def project_simplex(v, S):

import numpy as np

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient(weights, a, n):
    """Compute gradient of weighted sum of convolution values.
    
    grad[m] = 2 * sum_j weights[m+j] * a[j] for m = 0, ..., n-1
    Computed efficiently via FFT-based convolution.
    """
    # cross_corr[m] = sum_j weights[m+j] * a[j]
    # = convolve(weights, a_reversed)[m + n - 1]
    a_rev = a[::-1]
    size = len(weights) + n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a_rev, fsize)
    conv_full = np.real(ifft(W * A))
    grad = 2.0 * conv_full[n - 1: 2 * n - 1]
    return grad


def softmax_weights(c, temperature):
    """Compute softmax weights over convolution values."""
    c_shifted = c - c.max()
    w = np.exp(c_shifted / temperature)
    s = w.sum()
    if s < 1e-300:
        # All weights are zero (extremely spread out); use uniform
        return np.ones_like(c) / len(c)
    return w / s


def spectral_equalize(a, n):
    """Flatten the power spectrum of a to move toward flat autocorrelation."""
    A = fft(a, n)
    abs_A = np.abs(A)
    mean_power = np.mean(np.abs(A)**2)
    # New amplitude: constant magnitude, keep phase
    phase = A / np.maximum(abs_A, 1e-30)
    A_new = np.sqrt(mean_power) * phase
    a_new = np.real(ifft(A_new, n))
    return a_new


def safe_normalize(a, n):
    """Clip to [0, 1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.full(n, S / n, dtype=np.float64)
    return a * (S / s)


def fista_optimize(a_init, max_time, temp_init=1.0, lr_init=0.5):
    """FISTA (Nesterov-accelerated projected gradient) with smoothed subgradient.
    
    Returns (best_a, best_maxc).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    a_prev = a.copy()
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    best_score = c.max()
    y = a.copy()  # FISTA lookahead variable
    
    lr = lr_init
    temp = temp_init
    t = 0
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        
        # Compute convolution at lookahead point
        c = fft_convolve_self(y)
        cur = c.max()
        
        # Track best at actual point
        if t % 100 == 0:
            c_actual = fft_convolve_self(a)
            if c_actual.max() < best_score:
                best_score = c_actual.max()
                best_a = a.copy()
        
        # Compute smoothed subgradient
        temp = max(temp_init * (0.97 ** (t / 1000.0)), 0.01)
        weights = softmax_weights(c, temp)
        grad = compute_gradient(weights, y, n)
        
        # FISTA update
        beta = (t - 1) / (t + 2.0)
        a_prev = a
        a_new = project_simplex(y - lr * grad, S)
        y = a_new + beta * (a_new - a_prev)
        a = a_new
        
        # Learning rate decay
        lr = lr_init / (1.0 + 0.005 * t)
        
        # Periodic spectral equalization
        if t % 5000 == 0 and t > 0:

            a_se = safe_normalize(a_se, n)
            c_se = fft_convolve_self(a_se)
            if c_se.max() < best_score:
                best_score = c_se.max()
                best_a = a_se.copy()
            # Blend: move toward spectral-equalized version

            a = project_simplex(a, S)
            y = a.copy()
            a_prev = a.copy()
    
    # Final check
    c = fft_convolve_self(a)
    if c.max() < best_score:

        best_a = a.copy()

"""
Minimize 2*n*max(convolve(a,a)) / sum(a)^2 over non-negative sequences.

Key improvements over previous approach:
1. FISTA (Nesterov-accelerated projected gradient) with softmax-smoothed subgradient
2. Periodic frequency-domain spectral equalization to flatten the power spectrum
3. Temperature annealing for smoother gradient early, sharper later
4. Multiple diverse initializations including cosine-modulated and U-shaped profiles
5. Adaptive learning rate with decay
6. Dedicated polish phase with low temperature and small steps
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient(weights, a, n):
    """Compute gradient: grad[m] = 2 * sum_{j=0}^{n-1} weights[m+j] * a[j].
    
    Uses FFT-based convolution: grad = 2 * convolve(weights, a[::-1])[n-1:2n-1]
    """
    a_rev = a[::-1]
    size = len(weights) + n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a_rev, fsize)
    conv_full = np.real(ifft(W * A))
    grad = 2.0 * conv_full[n - 1: 2 * n - 1]
    return grad


def softmax_weights(c, temperature):
    """Compute softmax weights over convolution values for smoothing."""
    c_shifted = c - c.max()
    w = np.exp(c_shifted / temperature)
    s = w.sum()
    if s < 1e-300:
        return np.ones_like(c) / len(c)
    return w / s


def spectral_equalize(a, n):
    """Flatten the power spectrum of a to move toward flat autocorrelation."""
    A = fft(a, n)
    abs_A = np.abs(A)
    mean_power = np.mean(np.abs(A) ** 2)
    if mean_power < 1e-30:
        return a.copy()
    phase = A / np.maximum(abs_A, 1e-30)
    A_new = np.sqrt(mean_power) * phase
    a_new = np.real(ifft(A_new, n))
    return a_new


def safe_normalize(a, n):
    """Clip to [0, 1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.full(n, S / n, dtype=np.float64)
    return a * (S / s)


def fista_optimize(a_init, max_time, temp_init=1.0, lr_init=0.05,
                   spectral_interval=3000, spectral_blend=0.3):
    """FISTA with smoothed subgradient and periodic spectral equalization.
    
    Returns (best_a, best_maxc).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    a_prev = a.copy()
    best_a = a.copy()

    c = fft_convolve_self(a)
    best_score = c.max()
    y = a.copy()

    lr = lr_init
    t = 0
    start = time.monotonic()
    check_interval = max(1, int(np.sqrt(n)))

    while time.monotonic() - start < max_time:
        t += 1

        # Compute convolution at FISTA lookahead point
        c = fft_convolve_self(y)
        cur = c.max()

        # Periodic best tracking at actual point
        if t % check_interval == 0:
            c_actual = fft_convolve_self(a)
            if c_actual.max() < best_score:
                best_score = c_actual.max()
                best_a = a.copy()

        # Temperature annealing: start broad, sharpen over time
        temp = max(temp_init * (0.95 ** (t / 2000.0)), 0.05)
        weights = softmax_weights(c, temp)
        grad = compute_gradient(weights, y, n)

        # FISTA update with Nesterov momentum
        beta = (t - 1) / (t + 2.0)
        a_prev = a
        a_new = project_simplex(y - lr * grad, S)
        y = a_new + beta * (a_new - a_prev)
        a = a_new

        # Adaptive learning rate decay
        lr = lr_init / (1.0 + 0.003 * t)

        # Periodic spectral equalization
        if t > 0 and t % spectral_interval == 0:
            a_se = spectral_equalize(a, n)
            a_se = safe_normalize(a_se, n)
            c_se = fft_convolve_self(a_se)
            if c_se.max() < best_score:
                best_score = c_se.max()
                best_a = a_se.copy()
            # Partial blend toward spectral-equalized version
            a = (1.0 - spectral_blend) * a + spectral_blend * a_se
            a = project_simplex(a, S)
            y = a.copy()
            a_prev = a.copy()

    # Final evaluation
    c = fft_convolve_self(a)
    if c.max() < best_score:
        best_score = c.max()
        best_a = a.copy()
    c = fft_convolve_self(a)
    if c.max() < best_score:
        best_score = c.max()
        best_a = a.copy()

    return best_a, best_score


def make_initializations(rng, n):
    """Generate diverse initializations of length n."""
    inits = []

    # 1. Given sequence (if available and matching length)
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0, 1000)))
        elif given.size > 16:
            # Resample to target length using linear interpolation
            x_old = np.linspace(0, 1, given.size)
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, np.clip(given, 0, 1000))
            inits.append(("given_interp", np.maximum(0.0, interp)))
    except Exception:
        pass

    # 2. Uniform
    inits.append(("uniform", np.ones(n)))

    # 3. Cosine-modulated (U-shaped: high at edges, low in middle)
    jj = np.arange(n, dtype=np.float64)
    for alpha in (0.4, 0.7):
        arr = 1.0 + alpha * np.cos(2.0 * np.pi * jj / n)
        inits.append(("cos_a%.1f" % alpha, np.maximum(0.1, arr)))

    # 4. Inverted cosine (bump: high in middle, low at edges)
    arr = 1.0 - 0.5 * np.cos(2.0 * np.pi * jj / n)
    inits.append(("bump", np.maximum(0.1, arr)))

    # 5. U-shaped parabola
    t_param = 2.0 * jj / max(n - 1, 1) - 1.0
    inits.append(("parabola", 1.0 + 0.8 * t_param ** 2))

    # 6. Smooth random
    r = rng.random(n)
    kernel = np.ones(32) / 32.0
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(("smooth_rand", r_smooth + 0.2))

    # 7. Pure random
    inits.append(("random", rng.random(n) + 0.05))

    return inits

    else:
        print("WARNING: evaluate returned inf, using fallback")
        global_a = np.ones(256, dtype=np.float64)
        ver = evaluate_sequence(global_a.tolist())
        print("Fallback score: %.8f" % ver)

    return [float(x) for x in global_a]
