    n = len(p)
    s = np.sum(p)
    if s < 1e-12:
        return np.inf
    c = _full_conv(p)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (s * s))


def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _grad_at_lag(p, k):
    """Gradient of c[k] = sum_i p[i]*p[k-i] w.r.t. p."""
    n = len(p)
    g = np.zeros(n)
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k)
    for i in range(i_lo, i_hi + 1):
        j = k - i
        if 0 <= j < n:
            g[i] += p[j]
            if j != i:
                g[j] += p[i]
    return g


def _optimize_subgrad(p0, time_limit, rng, label=""):
    """
    Projected subgradient with exact line search for minimizing max(c).
    p is normalized to sum=1.
    """
    p = _project_simplex(np.asarray(p0, dtype=np.float64))
    n = len(p)
    if n == 0:
        return p, np.inf

    c = _full_conv(p)
    cmax = float(np.max(c))
    best_p = p.copy()
    best_cmax = cmax
    best_score = 2.0 * n * cmax  # sum(p) = 1

    cur_cmax = cmax
    end = time.monotonic() + max(0.0, float(time_limit))
    it = 0
    stall = 0

    while time.monotonic() < end:
        it += 1

        # Compute gradient at the active lag
        k_star = int(np.argmax(c))
        g = _grad_at_lag(p, k_star)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Normalize gradient for step size
        g = g / g_norm

        # Line search: find best alpha
        # phi(alpha) = max(c) at project_simplex(p - alpha * g)
        best_alpha = 0.0
        best_phi = cur_cmax
        best_p_trial = p.copy()

        # Try a range of alphas
        base_alpha = 1.0 / max(n, 10)
        alphas = base_alpha * np.array([4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005])
        alphas = np.clip(alphas, 1e-8, 2.0)
        alphas = np.unique(alphas)

        for alpha in alphas:
            p_trial = _project_simplex(p - alpha * g)
            c_trial = _full_conv(p_trial)
            phi = float(np.max(c_trial))
            if phi < best_phi:
                best_phi = phi
                best_alpha = alpha
                best_p_trial = p_trial

        if best_alpha == 0.0 or best_p_trial is p:
            # Try smaller/multiplier step
            for mult in [0.5, 0.2, 0.1]:
                alpha = base_alpha * mult
                p_trial = _project_simplex(p - alpha * g)
                c_trial = _full_conv(p_trial)
                phi = float(np.max(c_trial))
                if phi < best_phi:
                    best_phi = phi
                    best_alpha = alpha
                    best_p_trial = p_trial

        if best_alpha == 0.0:
            # No improvement found; try multiplicative update
            p_trial = p * np.exp(-0.1 * g)
            p_trial = np.maximum(p_trial, 0.0)
            s = p_trial.sum()
            if s > 1e-12:
                p_trial = p_trial / s
                c_trial = _full_conv(p_trial)
                phi = float(np.max(c_trial))
                if phi < best_phi:
                    best_phi = phi
                    best_p_trial = p_trial
            else:
                break

        # Update
        p = best_p_trial
        c = _full_conv(p)
        cur_cmax = float(np.max(c))

        if cur_cmax < best_cmax - 1e-12:
            best_cmax = cur_cmax
            best_p = p.copy()
            best_score = 2.0 * n * cur_cmax
            stall = 0
        else:
            stall += 1

        if stall > 500:
            # Restart with perturbation
            p = _project_simplex(best_p + rng.normal(0, 2.0/n, n))
            c = _full_conv(p)
            cur_cmax = float(np.max(c))
            stall = 0

        if it % 2000 == 0:
            print(f"{label} iter={it} best_cmax={best_cmax:.8f} score={best_score:.6f} cur={cur_cmax:.8f}", flush=True)

        if best_score < 1.498:
            break

    return best_p, best_score


def _optimize_mult(p0, time_limit, rng, label=""):
    """Multiplicative weight (mirror descent) approach."""
    p = _project_simplex(np.asarray(p0, dtype=np.float64))
    n = len(p)
    if n == 0:
        return p, np.inf

    c = _full_conv(p)
    cmax = float(np.max(c))
    best_p = p.copy()
    best_cmax = cmax
    best_score = 2.0 * n * cmax

    end = time.monotonic() + max(0.0, float(time_limit))
    it = 0
    stall = 0
    eta = 1.0 / n

    while time.monotonic() < end:
        it += 1
        k_star = int(np.argmax(c))
        g = _grad_at_lag(p, k_star)

        # Multiplicative update
        log_p = np.log(np.maximum(p, 1e-15))
        log_p -= eta * g
        p_new = np.exp(log_p - np.max(log_p))
        p_new = p_new / p_new.sum()

        c_new = _full_conv(p_new)
        cmax_new = float(np.max(c_new))

        p = p_new
        c = c_new

        if cmax_new < best_cmax - 1e-12:
            best_cmax = cmax_new
            best_p = p.copy()
            best_score = 2.0 * n * cmax_new
            stall = 0
        else:
            stall += 1
            # Reduce eta
            eta *= 0.999

        if stall > 300:
            p = _project_simplex(best_p + rng.normal(0, 1.0/n, n))
            c = _full_conv(p)
            cmax = float(np.max(c))
            stall = 0
            eta = 1.0 / n

        if it % 2000 == 0:
            print(f"{label} mult iter={it} best_cmax={best_cmax:.8f} score={best_score:.6f}", flush=True)

        if best_score < 1.498:
            break

    return best_p, best_score


def _lp_best_response(p, budget_s=30.0):
    """
    Solve LP: max sum(g) s.t. conv(p, g)[k] <= max(conv(p,p))[k] for all k, g >= 0.
    Then mix p with g.
    """
    if not _HAVE_LP:
        return p, _score(p)
    
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    if n <= 0 or n > 800:
        return p, _score(p)

    c_p = _full_conv(p)
    M = float(np.max(c_p))
    if not np.isfinite(M) or M <= 0:
        return p, _score(p)

    # conv(p, g)[k] = sum_i p[i]*g[k-i]
    # Build constraint matrix: for each k, sum_i p[i]*g[k-i] <= M
    # Variables: g[0], ..., g[n-1]
    # For lag k: g[k-i] appears with coefficient p[i] for valid i
    # i.e., g[j] appears with coefficient p[k-j] for valid j
    
    # Matrix A[k, j] = p[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix
    
    # For efficiency, only use lags that are binding (close to M)
    binding = np.where(c_p >= 0.9 * M)[0]
    if len(binding) == 0:
        binding = np.array([int(np.argmax(c_p))])
    
    # Build constraint for each binding lag
    cols_k = []
    cols_j = []
    data = []
    b_ub = []
    
    for ki, k in enumerate(binding):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                cols_k.append(ki)
                cols_j.append(j)
                data.append(p[i])
        b_ub.append(M)
    
    if len(data) == 0:
        return p, _score(p)
    
    A = csr_matrix((data, (cols_k, cols_j)), shape=(len(binding), n))
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs', disp=False)
    except Exception:
        return p, _score(p)
    
    if res is None or res.status != 0:
        return p, _score(p)
    
    g = res.x
    if g is None:
        return p, _score(p)
    g = np.maximum(g, 0)
    s = g.sum()
    if s < 1e-12:
        return p, _score(p)
    g = g / s
    
    sc_p = _score(p)
    for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
        q = _project_simplex((1-t)*p + t*g)
        sc = _score(q)
        if sc < sc_p:
            sc_p = sc
            p = q
    
    return p, sc_p


def _make_shape(n, alpha, beta, gamma, delta):
    """
    Parameterized shape: p(x) ∝ x^alpha * (1-x)^beta * (1 + gamma*sin(2πx) + delta*sin(4πx))
    on x in [0, 1], sampled at n points.
    """
    n = max(2, int(n))
    x = np.linspace(0.5/n, 1.0 - 0.5/n, n)
    p = np.power(x, alpha) * np.power(1.0 - x, beta)
    p = p * (1.0 + gamma * np.sin(2*np.pi*x) + delta * np.sin(4*np.pi*x))
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-12:
        return np.full(n, 1.0/n)
    return p / s
