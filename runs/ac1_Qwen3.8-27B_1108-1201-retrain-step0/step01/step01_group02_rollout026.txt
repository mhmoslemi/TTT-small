"""
Hybrid optimization for minimizing peak auto-convolution of non-negative sequences.
Combines projected subgradient descent with softmax smoothing, LP refinement,
and local search polishing. Explores multiple sequence lengths.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a, target_sum=None):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    return a / s * target_sum


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s * np.sqrt(2.0 * n)


def psd_refine(a, time_limit_s=300.0, beta=50.0, verbose=True):
    """
    Projected subgradient descent with softmax smoothing.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = sqrt(2n).
    """
    start_time = time.monotonic()
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_nonneg(a.copy(), target_sum)
    
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    PSD start: score={best_score:.6f}, n={n}")
    
    eta = 0.01  # initial learning rate (scaled)
    stagnant = 0
    max_stagnant = 200
    
    it = 0
    max_iters = 50000
    while it < max_iters:
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        current_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        
        if current_score < best_score - 1e-10:
            best_score = current_score
            best_a = a.copy()
            stagnant = 0
            if verbose and it % 500 == 0:
                print(f"    PSD iter {it:5d}: score={current_score:.6f}, eta={eta:.5f}")
        else:
            stagnant += 1
        
        if stagnant > max_stagnant:
            # Try reducing beta (smoother) and eta
            beta = max(5.0, beta * 0.5)
            eta = max(0.001, eta * 0.5)
            stagnant = 0
            if verbose:
                print(f"    PSD stagnant at {it}, reducing beta={beta:.1f}, eta={eta:.5f}")
            if beta < 5.1:
                break
        
        # Compute softmax weights
        conv_scaled = conv / (1.0 / beta + 1e-10)
        conv_scaled -= conv_scaled.max()
        w = np.exp(conv_scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            break
        w /= w_sum
        
        # Compute gradient via FFT cross-correlation
        N_fft = 2 * n
        W = np.fft.rfft(w, N_fft)
        A = np.fft.rfft(a, N_fft)
        cross = np.fft.irfft(W * np.conj(A), N_fft)
        grad = cross[:n]
        
        # Gradient step
        b = a - eta * grad
        
        # Project onto {a >= 0, sum(a) = target_sum}
        # Bisection for threshold
        lo = b.min() - 1.0
        hi = b.max() + 1.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            s_mid = np.maximum(b - mid, 0.0).sum()
            if s_mid > target_sum:
                lo = mid
            else:
                hi = mid
        tau = (lo + hi) / 2.0
        a = np.maximum(b - tau, 0.0)
        
        # Adaptive learning rate
        it += 1
        if it % 100 == 0:
            eta *= 0.97
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    PSD done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=300.0, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")
    
    stagnant = 0
    it = 0
    max_iters = 200
    while it < max_iters:
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and it % 20 == 0:
                print(f"    LP iter {it:3d}: score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 30:
            break
        
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.2))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g * np.sqrt(2.0 * n)
        alpha = 0.3
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new, np.sqrt(2.0 * n))
        a = a_new
        it += 1
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=60.0, rng=None, verbose=True):
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_nonneg(a.copy(), target_sum)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    LS start: score={best_score:.6f}")
    
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        # Try different mutation strategies
        strategy = rng.integers(3)
        candidate = a.copy()
        
        if strategy == 0:
            # Block perturbation
            width = max(2, int(rng.integers(n // 100, n // 5)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            candidate[start:stop] *= np.exp(rng.normal(0.0, 0.15, size=stop - start))
        elif strategy == 1:
            # Single element perturbation
            idx = int(rng.integers(n))
            candidate[idx] *= np.exp(rng.normal(0.0, 0.3))
        else:
            # Squeeze: reduce peak-contributing elements
            conv = np.convolve(a, a)
            peak_idx = conv.argmax()
            # Elements contributing most to the peak
            j_range = range(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
            for j in j_range:
                if candidate[j] > 0:
                    candidate[j] *= (1.0 - rng.uniform(0.01, 0.1))
        
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = normalize_nonneg(candidate, target_sum)
        score = score_sequence(candidate)
        
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 20 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")
    
    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

"""
Hybrid optimization for minimizing peak auto-convolution of non-negative sequences.
Combines projected subgradient descent (PSD) with softmax smoothing, LP-based
alternating refinement, and local search polishing. Explores multiple sequence
lengths and initializations within the time budget.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def score_sequence(a):
    """Compute the score: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a, target_sum=None):
    """Normalize to non-negative with given target sum."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    return a / s * target_sum


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_hann(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / (n - 1)))


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(2, n // 15), max(3, n // 4)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s * np.sqrt(2.0 * n)


def project_onto_simplex(b, target_sum):
    """Project vector b onto {a >= 0, sum(a) = target_sum} via bisection."""
    lo = b.min() - 1.0
    hi = b.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_mid = np.maximum(b - mid, 0.0).sum()
        if s_mid > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.maximum(b - tau, 0.0)


def compute_gradient_fft(w, a):
    """
    Compute grad[j] = sum_k w[k] * a[k-j] for j=0..n-1
    using FFT-based cross-correlation.
    """
    n = len(a)
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft *= 2
    W = np.fft.rfft(w, N_fft)
    A = np.fft.rfft(a, N_fft)
    cross = np.fft.irfft(W * np.conj(A), N_fft)
    return cross[:n]


def psd_refine(a, time_limit_s=300.0, beta_init=50.0, verbose=True):
    """
    Projected Subgradient Descent with softmax smoothing.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = sqrt(2n).
    """
    start_time = time.monotonic()
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_nonneg(a.copy(), target_sum)

    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"    PSD start: score={best_score:.6f}, n={n}")

    eta = 0.5 / np.sqrt(n)
    beta = beta_init
    stagnant = 0
    max_stagnant = 150
    it = 0
    max_iters = 100000
    no_improve_count = 0

    while it < max_iters:
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break

        conv = np.convolve(a, a)
        current_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if current_score < best_score - 1e-9:
            best_score = current_score
            best_a = a.copy()
            stagnant = 0
            no_improve_count = 0
            if verbose and (it % 200 == 0 or it < 5):
                print(f"    PSD iter {it:6d}: score={current_score:.6f}, eta={eta:.5f}, beta={beta:.1f}")
        else:
            stagnant += 1
            no_improve_count += 1

        if stagnant > max_stagnant:
            if beta > 10.0:
                beta = max(10.0, beta * 0.7)
                eta = min(eta * 1.5, 2.0 / np.sqrt(n))
            else:
                eta = max(eta * 0.5, 0.001 / np.sqrt(n))
            stagnant = 0
            if verbose and it % 500 == 0:
                print(f"    PSD adjust at {it}: beta={beta:.1f}, eta={eta:.5f}")
            if eta < 0.002 / np.sqrt(n) and beta <= 10.1:
                break

        # Compute softmax weights on convolution
        inv_beta = 1.0 / max(beta, 1.0)
        conv_scaled = conv * inv_beta
        conv_scaled -= conv_scaled.max()
        exp_vals = np.exp(conv_scaled)
        w_sum = exp_vals.sum()
        if w_sum < 1e-300:
            w = np.full(len(conv), 1.0 / len(conv))
        else:
            w = exp_vals / w_sum

        # Compute gradient via FFT
        grad = compute_gradient_fft(w, a)

        # Gradient step with projection
        b = a - eta * grad
        a = project_onto_simplex(b, target_sum)

        # Adaptive learning rate decay
        it += 1
        if it % 200 == 0:
            eta *= 0.96

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    PSD done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def build_conv_matrix(a):
    """Build sparse matrix for convolution a * x (as linear system)."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    """Solve LP: maximize sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=300.0, verbose=True):
    """LP-based alternating refinement."""
    start_time = time.monotonic()
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_nonneg(a.copy(), target_sum)
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")

    stagnant = 0
    it = 0
    max_iters = 300
    alpha = 0.4

    while it < max_iters:
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break

        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            alpha = min(0.5, alpha * 1.1)
            if verbose and (it % 10 == 0 or it < 3):
                print(f"    LP iter {it:3d}: score={score:.6f}")
        else:
            stagnant += 1
            alpha = max(0.05, alpha * 0.8)

        if stagnant > 30:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break

        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(40.0, remaining * 0.25))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement (sum_g={sum_g:.10f})")
            break
        g = g / sum_g * target_sum
        a_new = (1.0 - alpha) * a + alpha * g
        a = normalize_nonneg(a_new, target_sum)
        it += 1

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

    best_score = score_sequence(a)

    if verbose:
        print(f"    LS start: score={best_score:.6f}, n={n}")

    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        strategy = rng.integers(5)
        candidate = a.copy()

        if strategy == 0:
            # Block perturbation
            width = max(2, int(rng.integers(n // 200, n // 5)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            candidate[start:stop] *= np.exp(rng.normal(0.0, 0.1, size=stop - start))
        elif strategy == 1:
            # Multiple small blocks
            for _ in range(rng.integers(1, 4)):
                width = max(1, int(rng.integers(1, n // 50)))
                start = int(rng.integers(0, max(1, n - width)))
                stop = min(n, start + width)
                candidate[start:stop] *= np.exp(rng.normal(0.0, 0.2, size=stop - start))
        elif strategy == 2:
            # Squeeze peak contributors
            conv = np.convolve(a, a)
            peak_idx = int(conv.argmax())
            j_lo = max(0, peak_idx - n + 1)
            j_hi = min(n, peak_idx + 1)
            for j in range(j_lo, j_hi):
                if candidate[j] > 1e-10:
                    candidate[j] *= (1.0 - rng.uniform(0.005, 0.15))
        elif strategy == 3:
            # Symmetrize + perturb
            candidate = 0.5 * (candidate + candidate[::-1])
            candidate *= np.exp(rng.normal(0.0, 0.05, size=n))
        else:
            # Global scaling of random subset
            mask = rng.random(n) < 0.3
            candidate[mask] *= np.exp(rng.normal(0.0, 0.3, size=int(mask.sum())))

        candidate = normalize_nonneg(candidate, target_sum)
        score = score_sequence(candidate)

        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 30 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {accept_count} accepts, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def psd_lp_alternate(a, time_limit_s=600.0, verbose=True):
    """
    Alternate between PSD and LP steps for better convergence.
    """
    start_time = time.monotonic()
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_nonneg(a.copy(), target_sum)
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"    ALTERNATE start: score={best_score:.6f}, n={n}")

    eta = 0.5 / np.sqrt(n)
    beta = 50.0
    lp_alpha = 0.3
    stagnant = 0
    max_stagnant = 50
    cycle = 0

    while time.monotonic() - start_time < time_limit_s:
        elapsed = time.monotonic() - start_time
        remaining = time_limit_s - elapsed

        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > max_stagnant:
            if verbose:
                print(f"    ALTERNATE stagnant at cycle {cycle}, best={best_score:.6f}")
            break

        # Alternate: 3 PSD steps then 1 LP step
        phase = cycle % 4
        if phase < 3:
            # PSD step
            conv_scaled = conv / (1.0 / max(beta, 1.0) + 1e-10)
            conv_scaled -= conv_scaled.max()
            exp_vals = np.exp(conv_scaled)
            w = exp_vals / exp_vals.sum()
            grad = compute_gradient_fft(w, a)
            b = a - eta * grad
            a = project_onto_simplex(b, target_sum)
            eta *= 0.98
        else:
            # LP step
            M = conv.max()
            timeout = max(1.0, min(20.0, remaining * 0.1))
            g = solve_lp_step(a, M, timeout=timeout)
            if g is not None:
                sum_g = g.sum()
                if sum_g > 1.0 + 1e-10:
                    g = g / sum_g * target_sum
                    a_new = (1.0 - lp_alpha) * a + lp_alpha * g
                    a = normalize_nonneg(a_new, target_sum)

        cycle += 1
        if cycle % 100 == 0 and verbose:
            s = score_sequence(a)
            print(f"    ALTERNATE cycle {cycle}: score={s:.6f}, best={best_score:.6f}")

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    ALTERNATE done: best={best_score:.6f} after {cycle} cycles, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=2350, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses hybrid PSD + LP + local search with multiple lengths and inits.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 30.0)

    best_score = float('inf')
    best_seq = None

    # Check for provided starting point
    parent = None
    try:
        parent = height_sequence_1
        if parent is not None and len(parent) > 0:
            parent = np.array(parent, dtype=np.float64)
            parent_score = score_sequence(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            if parent_score < best_score:
                best_score = parent_score
                best_seq = parent.copy()
    except NameError:
        pass

    # Define search configurations
    # (n, init_type, time_psd, time_lp, time_ls)
    configs = []

    if parent is not None and len(parent) >= 50:
        np_ = len(parent)
        configs.append((np_, 'parent', 150, 150, 40))

    # Core configurations
    configs.extend([
        (500, 'raised_cosine', 200, 180, 50),
        (700, 'raised_cosine', 200, 180, 50),
        (1000, 'raised_cosine', 250, 200, 50),
        (1500, 'raised_cosine', 250, 200, 50),
        (2000, 'raised_cosine', 300, 200, 50),
        (1000, 'gaussian', 200, 180, 50),
        (1500, 'gaussian', 250, 200, 50),
        (2000, 'gaussian', 250, 200, 50),
        (1000, 'blackman', 200, 180, 50),
        (500, 'hann', 150, 150, 40),
        (1000, 'hann', 200, 180, 50),
    ])

    # Add random inits for diversity
    configs.extend([
        (1000, 'random', 200, 150, 40),
        (2000, 'random', 250, 150, 40),
    ])

    for n, init_type, t_psd, t_lp, t_ls in configs:
        remaining = deadline - time.monotonic()
        needed = t_psd + t_lp + t_ls
        if remaining < needed * 0.4:
            scale = max(0.3, remaining * 0.85 / needed)
            t_psd = int(t_psd * scale)
            t_lp = int(t_lp * scale)
            t_ls = int(t_ls * scale)
            if t_psd < 15:
                break

        print(f"\n=== n={n}, init={init_type}, budget=({t_psd},{t_lp},{t_ls})s, remaining={remaining:.0f}s ===")

        # Initialize
        if init_type == 'parent':
            a = parent.copy()
        elif init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif init_type == 'gaussian':
            a = initial_gaussian(n, sigma_frac=0.25)
        elif init_type == 'blackman':
            a = initial_blackman(n)
        elif init_type == 'hann':
            a = initial_hann(n)
        elif init_type == 'random':
            a = random_smooth_init(n, rng, n_bumps=int(rng.integers(2, 6)))
        else:
            a = initial_raised_cosine(n)

        a = normalize_nonneg(a, np.sqrt(2.0 * n))
        init_score = score_sequence(a)
        print(f"    Init: score={init_score:.6f}")

        # Phase 1: PSD refinement
        a_r, s1 = psd_refine(a, time_limit_s=t_psd, beta_init=50.0, verbose=True)

        # Phase 2: PSD-LP Alternation
        remaining2 = deadline - time.monotonic()
        t_alt = min(t_lp, max(0, int(remaining2 * 0.5)))
        if t_alt > 15:
            a_r, s2 = psd_lp_alternate(a_r, time_limit_s=t_alt, verbose=True)
            if s2 < s1:
                s1 = s2

        # Phase 3: LP refinement
        remaining3 = deadline - time.monotonic()
        t_lp2 = min(t_lp, max(0, int(remaining3 * 0.4)))
        if t_lp2 > 15:
            a_r, s3 = lp_refine(a_r, time_limit_s=t_lp2, verbose=True)
            if s3 < s1:
                s1 = s3

        # Phase 4: Local search
        remaining4 = deadline - time.monotonic()
        t_ls2 = min(t_ls, max(0, int(remaining4 * 0.4)))
        if t_ls2 > 10:
            a_r, s4 = local_search_refine(a_r, time_limit_s=t_ls2, rng=rng, verbose=True)
            if s4 < s1:
                s1 = s4

        # Track best
        final_s = score_sequence(a_r)
        if final_s < s1:
            s1 = final_s
        if s1 < best_score:
            best_score = s1
            best_seq = a_r.copy()
            print(f"  *** NEW BEST: score={best_score:.6f}, n={n}, elapsed={time.monotonic()-total_start:.1f}s")

    # Final intensive polish on the best sequence
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\n=== FINAL POLISH (remaining={remaining:.0f}s) ===")

            # Round 1: PSD with high beta
            t1 = int(remaining * 0.3)
            if t1 > 20:
                p, s = psd_refine(best_seq, time_limit_s=t1, beta_init=100.0, verbose=True)
                if s < best_score:
                    best_score = s
                    best_seq = p.copy()

            # Round 2: LP
            remaining -= (time.monotonic() - total_start) - (deadline - remaining)
            remaining = deadline - time.monotonic()
            t2 = int(remaining * 0.35)
            if t2 > 20:
                p, s = lp_refine(best_seq, time_limit_s=t2, verbose=True)
                if s < best_score:
                    best_score = s
                    best_seq = p.copy()

            # Round 3: Local search
            remaining = deadline - time.monotonic()
            t3 = int(remaining * 0.4)
            if t3 > 10:
                p, s = local_search_refine(best_seq, time_limit_s=t3, rng=rng, verbose=True)
                if s < best_score:
                    best_score = s
                    best_seq = p.copy()

            # Round 4: Another PSD pass
            remaining = deadline - time.monotonic()
            if remaining > 20:
                p, s = psd_refine(best_seq, time_limit_s=int(remaining * 0.8), beta_init=200.0, verbose=True)
                if s < best_score:
                    best_score = s
                    best_seq = p.copy()

    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Final verification
    print(f"\nFinal result: score={best_score:.6f}, n={len(best_seq)}")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified score: {verify_score:.6f}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")

    return [float(x) for x in best_seq]
