import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Binary search for lambda
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def score_normalized(a):
    """Compute score for a normalized sequence (sum=1)."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

def softmax_weights(c, tau):
    """Compute softmax weights for the convolution peak."""
    c_shifted = (c - np.max(c)) / tau
    e = np.exp(c_shifted)
    return e / np.sum(e)

"""
Projected gradient descent with Nesterov acceleration on the probability simplex
for minimizing peak linear autocorrelation of non-negative sequences.
Key differences from prior approach: direct simplex projection (no softmax
parameterization of a), Nesterov momentum, adaptive perturbation schedule,
targeted peak-equalization polish, and structured initializations.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    csv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_candidates = np.nonzero(u * idx > (csv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (csv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def score_normalized(a):
    """Compute score for normalized sequence (sum=1). Scale-invariant."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def compute_grad_and_conv(a, tau):
    """
    Compute gradient of tau*log(sum(exp(c/tau))) w.r.t. a,
    where c = conv(a,a). Also returns c.
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_shifted = (c - np.max(c)) / tau
    e = np.exp(c_shifted)
    w = e / np.sum(e)
    # grad[j] = 2 * sum_k w[k] * a[k-j] (valid where 0<=k-j<n)
    # = 2 * conv(a_reversed, w)[n-1 : 2n-1]
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, c


def make_init_uniform(n, rng):
    """Uniform initialization with small noise."""
    a = np.ones(n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_init_tapered(n, rng):
    """Cosine-tapered initialization (zero at edges)."""
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / (n - 1) - 1.0)))
    a = np.maximum(a, 0.01)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_init_gaussian(n, rng, sigma_frac=0.4):
    """Gaussian-bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / (sigma_frac * n)) ** 2)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_init_multipart(n, rng):
    """Multipart: high in middle, low at edges, with oscillation."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    envelope = np.exp(-0.5 * ((t - center) / (0.35 * n)) ** 2)
    osc = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n + rng.uniform(0, 2 * np.pi))
    a = envelope * osc
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def optimize_nesterov(n, a_init, time_limit, rng, verbose=True):
    """
    Projected gradient descent with Nesterov acceleration on the simplex.
    Minimizes smooth approximation of max(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    a = a_init.copy()
    a = np.maximum(a, 1e-6)
    a = a / np.sum(a)
    a_prev = a.copy()

    nesterov_mu = 0.85
    lr = 0.08
    lr_min = 0.001
    tau_start = 10.0 / n
    tau_end = 0.3 / n

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    step = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)

        # Temperature schedule: fast decay early, slow decay late
        if frac < 0.3:
            tau = tau_start * (tau_end / tau_start) ** (frac / 0.3) ** 0.7
        else:
            tau_mid = tau_start * (tau_end / tau_start)
            tau = tau_mid * (tau_end / tau_mid) ** ((frac - 0.3) / 0.7)

        # Nesterov extrapolation
        y = a + nesterov_mu * (a - a_prev)
        a_prev = a.copy()

        # Ensure y is on or near simplex for gradient computation
        y_clean = np.maximum(y, 0.0)
        y_sum = np.sum(y_clean)
        if y_sum < 1e-6:
            y_clean = a
        else:
            y_clean = y_clean / y_sum

        # Compute gradient
        grad, c = compute_grad_and_conv(y_clean, tau)

        # Gradient step
        a_new = y_clean - lr * grad

        # Project onto simplex
        a_new = project_simplex(a_new)

        a = a_new
        step += 1

        # Evaluate periodically
        if step % 20 == 0:
            score = score_normalized(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stagnation = 0
            else:
                stagnation += 20

            if verbose and step % 10000 == 0:
                print(f"      step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, lr={lr:.4f}")

        # Adaptive perturbation when stuck
        if step % 2000 == 0 and stagnation > 10000:
            pert_scale = 0.02 + 0.03 * min(1.0, stagnation / 50000.0)
            pert = rng.standard_normal(n) * pert_scale
            a = project_simplex(a + pert)
            a_prev = a.copy()
            stagnation = 0
            if verbose:
                print(f"      [perturb at step {step}, scale={pert_scale:.3f}]")

        # Learning rate decay
        if step % 5000 == 0:
            lr = max(lr_min, lr * 0.92)

    final_score = score_normalized(best_a)
    if final_score < best_score:
        best_score = final_score
    return best_a, best_score


def polish_targeted(a_init, time_limit, rng):
    """
    Targeted local search: specifically reduce the top peak contributors
    while increasing other elements to maintain sum.
    """
    current = a_init.astype(np.float64).copy()
    current = np.maximum(current, 0.0)
    current_sum = np.sum(current)
    if current_sum < 1e-6:
        current = np.ones(len(current)) / len(current)
    current = current / np.sum(current)

    n = len(current)
    current_score = score_normalized(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9995

    while time.monotonic() - t0 < time_limit:
        r = rng.random()

        if r < 0.4:
            # Targeted: reduce top peak contributors
            candidate = current.copy()
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            # Compute contributions to peak
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            for i in range(lo, hi + 1):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] += current[i] * current[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.06, 0.02, size=len(top_idx)))
            # Increase others proportionally
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.6:
            # Block swap: move mass from one block to another
            candidate = current.copy()
            block_size = max(2, n // 16)
            src_start = int(rng.integers(0, n - block_size))
            dst_start = int(rng.integers(0, n - block_size))
            src = candidate[src_start:src_start + block_size]
            dst = candidate[dst_start:dst_start + block_size]
            amount = 0.1 * np.mean(src)
            candidate[src_start:src_start + block_size] = src - amount
            candidate[dst_start:dst_start + block_size] = dst + amount
        elif r < 0.75:
            # Symmetrize partially
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.01, size=n))
        elif r < 0.9:
            # Multi-block perturbation
            candidate = current.copy()
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.015, size=n))

        candidate = np.maximum(candidate, 0.0)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-6:
            candidate = current
            continue
        candidate = candidate / cand_sum

        new_score = score_normalized(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current = candidate
            current_score = new_score

        temp *= cooling

    return best, best_score

            print(f"  n={n_alt} result: {eval_alt:.6f}")
            if eval_alt < best_score:
                best_score = eval_alt
                best_seq = np.array(final_alt)
                print(f"  *** New global best at n={n_alt}: {best_score:.6f} ***")

    # ===================== Final verification =====================
    print(f"\n{'='*60}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*60}")
        # Make sure we return the better one
        if final_score < best_score:
            best_score = final_score
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
