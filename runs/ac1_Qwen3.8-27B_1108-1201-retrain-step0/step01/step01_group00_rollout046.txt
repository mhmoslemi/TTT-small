"""
LP-guided optimization for minimizing peak self-convolution of non-negative sequences.
Uses linear programming to find descent directions that keep cross-correlation bounded,

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)

    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={it}, M={current_M:.8f}, best_M={best_M:.8f}, "
                  f"score={best_score:.6f}, t={elapsed:.1f}s")

            # Periodic check: try a random restart
            if it % 500 == 0 and stall_count > 50:
                f_rand = np.abs(rng.standard_normal(n))
                f_rand = f_rand / np.sum(f_rand)
                M_rand = np.max(fft_convolve(f_rand, f_rand))
                if M_rand < best_M * 1.05:
                    f = f_rand
                    current_M = M_rand
                    stall_count = 0

    return best_f, best_score


def coordinate_polish(a, time_budget, rng):
    """
    Coordinate descent polish: try adjusting individual elements.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    current_score = compute_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        # Try adjusting a random coordinate
        i = rng.integers(0, n)
        for delta in [0.5, 0.7, 0.8, 1.0, 1.2, 1.5, 0.1, 0.9]:
            candidate = a.copy()
            candidate[i] *= delta
            candidate = np.clip(candidate, 0.0, 1000.0)
            s = compute_score(candidate)
            if s < current_score:
                a = candidate
                current_score = s
        # Also try small random perturbation
        if rng.random() < 0.3:
            candidate = a * np.exp(0.05 * rng.standard_normal(n))
            candidate = np.clip(candidate, 0.0, 1000.0)
            s = compute_score(candidate)
            if s < current_score:
                a = candidate
                current_score = s

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

        if time.monotonic() - t0 > time_budget * 0.8 and (int(time.monotonic() - t0) % 10 == 0):
            pass  # just continue

    return best_a, best_score


def gradient_polish(a, time_budget, rng):
    """
    Subgradient method on the max-convolution objective.
    """
    n = len(a)
    f = a.astype(np.float64).copy()
    S0 = np.sum(f)
    f = f / S0  # normalize to sum 1

    best_f = f.copy()
    best_score = compute_score(best_f)
    t0 = time.monotonic()
    lr = 0.1
    decay = 0.9995

    for it in range(100000):
        if time.monotonic() - t0 > time_budget:
            break

        conv = fft_convolve(f, f)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]

        # Subgradient of max(conv) w.r.t. f
        # d/d(f[j]) of conv[peak_idx] = f[j + peak_idx] + f[j - peak_idx] (with bounds)
        grad = np.zeros(n)
        for j in range(n):
            # Contribution to conv[peak_idx] from f[j]:
            # f[j] * f[peak_idx - j] + f[j] * f[j - peak_idx] (if valid)
            val = 0.0
            k1 = peak_idx - j
            if 0 <= k1 < n:
                val += f[k1]
            k2 = j - peak_idx
            if 0 <= k2 < n:
                val += f[k2]
            grad[j] = val

        # Projected gradient step: minimize grad^T f subject to sum(f)=1, f>=0
        # Simple approach: f_new = f - lr * grad, then project to simplex
        f_new = f - lr * grad
        f_new = np.clip(f_new, 0.0, None)
        if np.sum(f_new) > 0:
            f_new = f_new / np.sum(f_new)
        else:
            f_new = np.ones(n) / n

        # Check improvement
        new_conv = fft_convolve(f_new, f_new)
        new_peak = np.max(new_conv)
        if new_peak < peak_val * 0.999:
            f = f_new
        else:
            lr *= decay

        score = 2.0 * n * np.max(fft_convolve(f, f))
        if score < best_score:
            best_score = score
            best_f = f.copy()

        lr = max(lr, 0.001)

    return best_f * S0, best_score


def make_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Random
    for _ in range(3):
        a = np.abs(rng.standard_normal(n))
        inits.append(a / np.sum(a))

    # Chirp
    t = np.arange(n, dtype=np.float64)
    for alpha in [0.3, 0.5, 0.7]:
        a = 1.0 + alpha * np.cos(np.pi * t * t / n)
        a = np.clip(a, 0.01, None)
        inits.append(a / np.sum(a))

    # Multi-tone
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k, amp in [(1, 0.3), (3, 0.2), (5, 0.1)]:
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a = np.clip(a, 0.01, None)
    inits.append(a / np.sum(a))

    # Triangular
    t = np.arange(n, dtype=np.float64)
    a = 1.0 - 0.5 * np.abs(2.0 * t / n - 1.0)
    inits.append(a / np.sum(a))

    # Bimodal
    a = np.ones(n)
    a[:n//4] *= 0.5
    a[3*n//4:] *= 0.5
    inits.append(a / np.sum(a))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search using LP-guided optimization, gradient methods,
    and coordinate polish. Returns best sequence found.
    """
    t_global = time.monotonic()
    time_budget = max(10.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Try the provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start_seq) > 0 and np.sum(start_seq) > 0.01:
            s = compute_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()

    # Try uniform sequences of various lengths (score should be 2.0)
    for n_try in [64, 128, 256, 512, 1024]:
        a_uniform = np.ones(n_try)
        s = compute_score(a_uniform)
        print(f"Uniform n={n_try}: score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = a_uniform.copy()

    # Phase 1: LP-based optimization
    print("\n=== Phase 1: LP-based optimization ===")
    lengths_to_try = [128, 256, 512, 1024]

    for n in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break

        inits = make_initializations(n, rng)

        # If we have a provided sequence, resample to this length
        if start_seq is not None and len(start_seq) != n:
            x_old = np.linspace(0, 1, len(start_seq))
            x_new = np.linspace(0, 1, n)
            a_resampled = np.interp(x_new, x_old, start_seq)
            a_resampled = np.clip(a_resampled, 0.0, None)
            if np.sum(a_resampled) > 0:
                inits.append(a_resampled / np.sum(a_resampled))

        time_per_init = min(60.0, max(5.0, (deadline - time.monotonic()) / (len(inits) * 2)))

        for init_i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break

            t_per = time_per_init
            f_best, score = lp_optimize(f_init, t_per, rng, verbose=(init_i == 0))

            # Convert back (we normalized to sum=1 in lp_optimize)
            # Score is scale-invariant so this doesn't matter
            if score < best_score:
                best_score = score
                best_seq = f_best.copy()
                print(f"  *** n={n}, init={init_i}: score={best_score:.6f} ***")

    print(f"\nAfter Phase 1: best_score={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0}")

    # Phase 2: Gradient polish on best candidate
    print("\n=== Phase 2: Gradient polish ===")
    if best_seq is not None:
        polish_time = min(300.0, max(30.0, deadline - time.monotonic() - 30))
        polished, polish_score = gradient_polish(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After gradient polish: score={best_score:.6f}")

    # Phase 3: Coordinate polish
    print("\n=== Phase 3: Coordinate polish ===")
    if best_seq is not None:
        coord_time = min(200.0, max(10.0, deadline - time.monotonic()))
        polished, coord_score = coordinate_polish(best_seq, coord_time, rng)
        if coord_score < best_score:
            best_score = coord_score
            best_seq = polished.copy()
            print(f"  After coordinate polish: score={best_score:.6f}")

    # Phase 4: Try a few more LP runs with different params if time remains
    if time.monotonic() < deadline - 60:
        print("\n=== Phase 4: Additional LP exploration ===")
        for n_extra in [192, 320, 448, 640]:
            if time.monotonic() > deadline - 30:
                break
            f_init = np.ones(n_extra) / n_extra
            # Perturb
            f_init = f_init * np.exp(0.2 * rng.standard_normal(n_extra))
            f_init = np.clip(f_init, 0.01, None)
            f_init = f_init / np.sum(f_init)

            t_per = min(120.0, max(10.0, (deadline - time.monotonic()) / 3))
            f_best, score = lp_optimize(f_init, t_per, rng, verbose=False)
            if score < best_score:
                best_score = score
                best_seq = f_best.copy()
                print(f"  Extra n={n_extra}: score={best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback: uniform sequence
    fallback = [1.0] * 256
    print(f"\n=== FALLBACK: uniform, score={evaluate_sequence(fallback):.6f} ===")
    return fallback
