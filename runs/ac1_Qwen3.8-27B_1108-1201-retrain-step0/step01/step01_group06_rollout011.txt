def fast_auto_convolve(a):

        wk = w[k]
        # sg[j] += wk * (a[k-j] + a[j-k]) for valid j
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        # a[j-k] is valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
        j_low = max(0, k - n + 1)
        j_high = min(n - 1, k + n - 1)
        if j_low <= j_high:
            # a[k-j] for j in [j_low, j_high]: indices k-j range from k-j_low to k-j_high
            # a[k-j] = a[k - j_low : k - j_high + 1][::-1]
            idx1_start = k - j_high
            idx1_end = k - j_low + 1
            if idx1_start >= 0 and idx1_end <= n:
                sg[j_low:j_high + 1] += wk * a[idx1_start:idx1_end][::-1]
            # a[j-k] for j in [j_low, j_high]: indices j-k range from j_low-k to j_high-k
            idx2_start = j_low - k
            idx2_end = j_high - k + 1
            if idx2_start >= 0 and idx2_end <= n:
                sg[j_low:j_high + 1] += wk * a[idx2_start:idx2_end]

    return sg


def mirror_descent_optimize(a0, time_limit, temp0=40.0, eta0=0.3):
    """
    Multiplicative mirror descent (exponential weights) on the simplex.
    Minimizes max(conv(a,a)) for a >= 0, sum(a) = 1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_auto_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    temperature = temp0
    patience = 0
    max_patience = 60
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        sg = softmax_subgradient(a, conv, temperature)
        # Center the subgradient
        sg_c = sg - sg.mean()

        # Multiplicative update
        log_update = -eta * sg_c
        log_update -= log_update.max()  # numerical stability
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        conv = fast_auto_convolve(a)
        current_obj = 2.0 * n * conv.max()

        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            patience = 0
            eta *= 0.5
            temperature *= 1.5
            if eta < 0.005:
                eta = 0.2
                temperature = temp0
                max_patience = 30

    return best_a, best_obj


def nesterov_subgradient(a0, time_limit, lr0=0.01):
    """
    Projected subgradient with Nesterov acceleration.
    Minimizes max(conv(a,a)) for a >= 0, sum(a) = 1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_auto_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    y = a.copy()
    momentum = 0.5
    lr = lr0
    t0 = time.monotonic()
    step_count = 0
    patience = 0

    while time.monotonic() - t0 < time_limit:
        conv_y = fast_auto_convolve(y)
        max_val = conv_y.max()
        sg = softmax_subgradient(y, conv_y, temperature=30.0)

        # Nesterov momentum
        momentum = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y_new = a + momentum * (a - y)
        # Project y_new to simplex
        y_new = project_to_simplex(y_new)

        # Gradient step
        a_new = y_new - lr * (sg - sg.mean())
        a_new = project_to_simplex(a_new)

        y = a
        a = a_new
        step_count += 1

        if step_count % 10 == 0:
            conv = fast_auto_convolve(a)
            current_obj = 2.0 * n * conv.max()
            if current_obj < best_obj - 1e-10:
                best_obj = current_obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience >= 50:
                lr *= 0.7
                patience = 0
                if lr < 1e-6:
                    break

    return best_a, best_obj


def project_to_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    w_sum = w.sum()
    if w_sum < 1e-15:
        return np.ones(n) / n
    return w / w_sum


def coordinate_descent(a0, time_limit, iters_per_coord=3):
    """
    Coordinate descent for minimizing max(conv(a,a)) on the simplex.
    Cyclic through coordinates with line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_auto_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            # Try changing a[i] while keeping sum = 1
            # a[i] can range from 0 to 1 - sum(a[!=i]) = 1 - (1 - a[i]) = a[i]... 
            # Actually sum of others = 1 - a[i], so a[i] can be [0, 1]
            # But to maintain non-negativity of others: a[i] <= 1
            old_val = a[i]
            # Try a few values
            for frac in [0.0, 0.25, 0.5, 0.75, 1.5, 2.0, 3.0]:
                new_val = old_val * frac if frac != 0 else 0.0
                if new_val < 0 or new_val > 1:
                    continue
                a[i] = new_val
                # Renormalize
                s_curr = a.sum()
                if s_curr < 1e-12:
                    a[i] = old_val
                    continue
                a_norm = a / s_curr
                conv_c = fast_auto_convolve(a_norm)
                obj = conv_c.max()
                if obj < fast_auto_convolve(best_a).max() - 1e-12:
                    # Accept (we'd need to recompute best, but this is expensive)
                    pass
                a[i] = old_val
            # Simple: just do a small random perturbation
            delta = np.randomUniform if hasattr(np, 'randomUniform') else np.random.uniform
            a[i] = max(0.0, a[i] + np.random.uniform(-0.01, 0.01) * a.mean())
            a = np.maximum(a, 0)
            a /= a.sum()
            iter_count += 1

        if iter_count % n == 0:
            conv = fast_auto_convolve(a)
            current_obj = 2.0 * n * conv.max()
            if current_obj < best_obj - 1e-10:
                best_obj = current_obj
                best_a = a.copy()

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussian with different widths
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 1e-10)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Bimodal (edges)
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random Dirichlet (concentrated)
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits['random_conc'] = r

    # Random Dirichlet (spread)
    r2 = rng.dirichlet(np.ones(n) * 5.0)
    inits['random_spread'] = r2

    # Fourier-based: |sum of few sinusoids|^2
    k1 = rng.integers(1, max(2, n // 4))
    k2 = rng.integers(1, max(2, n // 4))
    while k2 == k1:
        k2 = rng.integers(1, max(2, n // 4))
    f = np.abs(np.exp(2j * np.pi * k1 * x / n) + np.exp(2j * np.pi * k2 * x / n)) ** 2
    f = np.maximum(f, 0)
    if f.sum() > 0:
        f /= f.sum()
        inits['fourier'] = f

    # Parent-derived
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            pp /= pp.sum()
            inits['parent'] = pp

    return inits


def get_parent_sequence():
    """Try to access the parent sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def accelerated_mirror_descent(a0, time_limit, temp0=20.0, eta0=0.5):
    """
    Improved mirror descent with:
    - Adaptive temperature (starts low for focus, increases for exploration)
    - Periodic restarts with perturbation
    - Nesterov-like momentum on the log-scale
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_auto_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    temperature = temp0
    momentum = 0.0
    velocity = np.zeros(n)
    patience = 0
    max_patience = 80
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        sg = softmax_subgradient(a, conv, temperature)
        sg_c = sg - sg.mean()

        # Momentum
        momentum = 0.9 * momentum + 0.1 * (1.0 / (1.0 + step * 0.01))
        velocity = momentum * velocity + (1 - momentum) * sg_c

        # Multiplicative update with momentum
        log_update = -eta * velocity
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        a = a_new
        step += 1

        if step % 5 == 0:
            conv = fast_auto_convolve(a)
            current_obj = 2.0 * n * conv.max()

            if current_obj < best_obj - 1e-10:
                best_obj = current_obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

            if patience >= max_patience:
                patience = 0
                # Adaptive parameter adjustment
                if eta > 0.05:
                    eta *= 0.6
                else:
                    # Restart: perturb the current best
                    a = best_a.copy()
                    noise = rng_global.standard_normal(n) * 0.1 * a.mean()
                    a = np.maximum(a + noise, 0)
                    a /= a.sum()
                    eta = eta0 * 0.5
                    temperature = temp0
                    max_patience = 40
                    velocity = np.zeros(n)
                    conv = fast_auto_convolve(a)

            # Periodically increase temperature for exploration
            if step % 200 == 0:
                temperature = min(temperature * 1.3, 200.0)

    return best_a, best_obj

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Uses hybrid optimization (mirror descent + Nesterov +
    coordinate refinement) with multi-length exploration to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    global rng_global
    rng_global = np.random.default_rng(seed)
    
    start_time = time.monotonic()
    # Leave 60 seconds margin for return processing
    deadline = start_time + max(10.0, float(budget_s) - 60.0)

    best_sequence = None
    best_score = float('inf')

    parent = get_parent_sequence()

    def try_update(a, label=""):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return
        a_clip = np.clip(a, 0.0, 1000.0)
        score = float(evaluate_sequence(a_clip.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_clip.copy()
            elapsed = time.monotonic() - start_time
            print(f"  [{elapsed:.0f}s] {label}: score={score:.6f}, n={len(a)}")

    # Phase 1: Quick scan across lengths and initializations (30% budget)
    print("Phase 1: Quick scan across lengths and initializations...")
    lengths = [64, 128, 256, 512, 1024, 2048]
    if parent is not None:
        lengths.append(len(parent))

    scan_configs = []
    for n in lengths:
        inits = generate_initializations(n, rng_global, parent)
        for name, a0 in inits.items():
            scan_configs.append((n, name, a0))

    phase1_budget = 0.30 * (deadline - start_time)
    time_per_scan = max(2.0, phase1_budget / max(1, len(scan_configs)))

    scan_results = []
    for i, (n, name, a0) in enumerate(scan_configs):
        if time.monotonic() > deadline - 120:
            break
        t_alloc = min(time_per_scan, deadline - time.monotonic() - 120)
        if t_alloc < 1.5:
            break

        a_opt, _ = accelerated_mirror_descent(a0, time_limit=t_alloc, temp0=30.0, eta0=0.4)
        try_update(a_opt, f"scan n={n} {name}")
        a_norm = a_opt / a_opt.sum() if a_opt.sum() > 0 else a_opt
        scan_results.append((a_opt.copy(), len(a_opt)))

    # Phase 2: Deep optimization on best candidates (50% budget)
    print(f"Phase 2: Deep optimization. Best so far: {best_score:.6f}")
    remaining = deadline - time.monotonic() - 60
    phase2_budget = 0.50 * (deadline - start_time)

    # Select top candidates for deep optimization
    candidates = []
    if best_sequence is not None:
        candidates.append(best_sequence.copy())
    
    # Add diverse candidates from scan
    seen_n = set()
    for a, n in scan_results:
        if n not in seen_n and len(candidates) < 8:
            candidates.append(a.copy())
            seen_n.add(n)
    
    # Also try from parent with perturbation
    if parent is not None:
        p = parent.copy()
        noise = rng_global.standard_normal(len(p)) * 0.05 * p.mean()
        p = np.maximum(p + noise, 0)
        p /= p.sum()
        candidates.append(p)

    # Add some random candidates at promising lengths
    for n in [256, 512, 1024, 2048]:
        if len(candidates) < 12:
            r = rng_global.dirichlet(np.ones(n) * rng_global.uniform(0.2, 3.0))
            candidates.append(r)

    time_per_deep = max(5.0, remaining / max(1, len(candidates) * 2))

    for i, a0 in enumerate(candidates):
        if time.monotonic() > deadline - 30:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 30)
        if t_alloc < 3:
            break

        n = len(a0)
        label = f"deep_{i}_n{n}"

        # Strategy A: Mirror descent with different temps
        t_half = t_alloc * 0.5
        a_opt, _ = accelerated_mirror_descent(a0, time_limit=t_half, temp0=20.0, eta0=0.5)
        try_update(a_opt, label + "_A")

        # Strategy B: Nesterov subgradient from the result
        if time.monotonic() < deadline - 30:
            t_b = min(t_half, deadline - time.monotonic() - 30)
            a_opt2, _ = nesterov_subgradient(a_opt, time_limit=t_b, lr0=0.005)
            try_update(a_opt2, label + "_B")

    # Phase 3: Final polish (remaining budget)
    print(f"Phase 3: Final polish. Best: {best_score:.6f}")
    if best_sequence is not None and time.monotonic() < deadline - 10:
        remaining = deadline - 10 - time.monotonic()
        if remaining > 5:
            n_best = len(best_sequence)
            # Multiple short polish runs with different parameters
            n_runs = min(4, int(remaining // 5))
            time_per_run = remaining / max(1, n_runs)
            for r in range(n_runs):
                if time.monotonic() > deadline - 10:
                    break
                t_single = time_per_run * 0.9
                a_start = best_sequence.copy()
                # Small perturbation
                if r > 0:
                    noise = rng_global.standard_normal(n_best) * 0.02 * a_start.mean()
                    a_start = np.maximum(a_start + noise, 0)
                    a_start /= a_start.sum()
                
                temps = [15.0, 40.0, 80.0, 150.0]
                a_opt, _ = accelerated_mirror_descent(a_start, time_limit=t_single,
                                                       temp0=temps[r % 4], eta0=0.3)
                try_update(a_opt, f"polish_{r}")

    # Final validation
    if best_sequence is None:
        # Fallback: uniform sequence
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"  Fallback uniform: score={best_score:.6f}")
    else:
        # Ensure validity
        best_sequence = np.clip(best_sequence, 0.0, 1000.0)
        final_score = float(evaluate_sequence(best_sequence.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        print(f"FINAL SCORE: {best_score:.6f}, length={len(best_sequence)}")

    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s")

    return [float(x) for x in best_sequence]
