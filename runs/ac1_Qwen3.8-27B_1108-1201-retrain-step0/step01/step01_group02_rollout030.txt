        pass
    return None

def lp_refine(a, time_limit_s=300.0, verbose=True, m_scale=1.0):
    start = time.monotonic()
    a = normalize_nonneg(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LP start: n={n}, score={best_score:.6f}")

    stagnant = 0
    it = 0
    for it in range(5000):
        if time.monotonic() - start > time_limit_s:
            break
        conv = np.convolve(a, a)
        M = conv.max() * m_scale
        score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 100 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1

        if stagnant > 60:
            # Try perturbation to escape
            if it % 30 == 0:
                rng_p = np.random.default_rng(it)
                a_pert = a * (1.0 + 0.05 * rng_p.normal(0, 1, n))
                a_pert = normalize_nonneg(a_pert)
                if score_sequence(a_pert) < best_score:
                    a = a_pert
                    stagnant = 0
            if stagnant > 100:
                if verbose:
                    print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
                break

        remaining = time_limit_s - (time.monotonic() - start)
        if remaining < 5.0:
            break

        timeout = max(0.5, min(10.0, remaining * 0.05))
        g = solve_lp(a, M, timeout=timeout)
        if g is None:
            break

        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-6:
            if verbose:
                print(f"    LP converged (sum_g={sum_g:.8f}) at iter {it}")
            break

        g = g / sum_g

        # Line search for optimal step size
        best_alpha = 0.1
        best_new_score = score_sequence(a)
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            a_try = normalize_nonneg((1.0 - alpha) * a + alpha * g)
            s_try = score_sequence(a_try)
            if s_try < best_new_score:
                best_new_score = s_try
                best_alpha = alpha

        a = normalize_nonneg((1.0 - best_alpha) * a + best_alpha * g)

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    LP done: best={best_score:.6f}, iters={it+1}, time={time.monotonic()-start:.1f}s")
    return best_a, best_score

def local_search(a, time_limit_s=60.0, rng=None, verbose=False):
    if rng is None:
        rng = np.random.default_rng(0)
    start = time.monotonic()
    a = normalize_nonneg(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    accept_count = 0

    while time.monotonic() - start < time_limit_s:
        # Multiple mutation strategies
        strategy = rng.integers(0, 4)
        candidate = a.copy()

        if strategy == 0:
            # Segment scaling
            width = int(rng.integers(max(2, n // 100), max(3, n // 5)))
            pos = int(rng.integers(0, max(1, n - width)))
            candidate[pos:pos + width] *= np.exp(rng.normal(0, 0.2, size=width))
        elif strategy == 1:
            # Global noise
            candidate *= np.exp(rng.normal(0, 0.1, n))
        elif strategy == 2:
            # Two-point perturbation
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            delta = rng.uniform(-0.3, 0.3)
            candidate[i] = max(0, candidate[i] + delta)
            candidate[j] = max(0, candidate[j] - delta)
        else:
            # Shift-like perturbation
            shift = int(rng.integers(max(1, n // 50), max(2, n // 10)))
            candidate = 0.7 * candidate + 0.3 * np.roll(candidate, shift)

        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 30 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, time={time.monotonic()-start:.1f}s")
    return best_a, best_score

def spectral_refine(a, time_limit_s=30.0, verbose=False):
    start = time.monotonic()
    a = normalize_nonneg(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    for alpha in [0.8, 0.9, 0.7, 0.95, 0.6]:
        if time.monotonic() - start > time_limit_s:
            break
        for _ in range(100):
            if time.monotonic() - start > time_limit_s:
                break
            F = np.fft.fft(a)
            mag = np.abs(F)
            mag = np.maximum(mag, 1e-15)
            F_new = F * (mag ** (alpha - 1))
            a_new = np.real(np.fft.ifft(F_new))
            a_new = np.maximum(0, a_new)
            a_new = normalize_nonneg(a_new)
            score = score_sequence(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            a = a_new

    if verbose:
        print(f"    Spectral done: best={best_score:.6f}, time={time.monotonic()-start:.1f}s")
    return best_a, best_score

def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))

def init_ramp(n):
    t = np.arange(n, dtype=np.float64)
    return (t / (n - 1)) ** 0.5

def init_shifted_cos(n, shift_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    return np.maximum(0, 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - n * shift_frac) / n)))

def init_two_bump(n, pos1=0.3, pos2=0.7, w1=0.1, w2=0.15):
    t = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((t - pos1 * n) / (w1 * n)) ** 2) + np.exp(-0.5 * ((t - pos2 * n) / (w2 * n)) ** 2)

def init_random_bump(n, rng):
    t = np.arange(n, dtype=np.float64)
    center = rng.uniform(0.2, 0.8) * n
    width = rng.uniform(0.05, 0.25) * n
    return np.exp(-0.5 * ((t - center) / width) ** 2)

def init_comb(n, period):
    a = np.zeros(n)
    a[0:n:period] = 1.0
    # Add some smoothness
    kernel = np.ones(period) / period
    return np.convolve(a, kernel, mode='same')

def propose_candidate(seed=42, budget_s=1800):
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30
    rng = np.random.default_rng(seed)

    best_score = float('inf')
    best_seq = None

    # Generate diverse candidates
    candidates = []

    # Previous best (most important starting point)
    if 'height_sequence_1' in globals():
        prev = np.array(height_sequence_1, dtype=np.float64)
        candidates.append(('prev_best', prev.copy()))
        # Slightly perturbed versions of prev best
        for i in range(2):
            pert = prev * (1.0 + 0.1 * rng.normal(0, 1, len(prev)))
            candidates.append((f'prev_pert{i}', np.maximum(0, pert)))

    # Asymmetric initializations at various lengths
    for n in [400, 500, 600, 800, 1000]:
        t = np.arange(n, dtype=np.float64)
        candidates.append((f'ramp_{n}', init_ramp(n)))
        candidates.append((f'shift_cos_{n}', init_shifted_cos(n, 0.25)))
        candidates.append((f'2bump_{n}', init_two_bump(n)))
        candidates.append((f'rand_{n}', init_random_bump(n, rng)))
        if n >= 200:
            candidates.append((f'comb_{n}', init_comb(n, max(2, n // 40))))

    # Sort by likely quality (previous best first, then smaller n)
    candidates.sort(key=lambda x: (0 if 'prev' in x[0] else 1, len(x[1])))

    # Time allocation: give more time to promising candidates
    total_time = deadline - total_start
    # First 3 candidates get more time
    time_per_early = total_time * 0.45 / 3
    time_per_late = total_time * 0.35 / max(1, len(candidates) - 3)
    polish_time = total_time * 0.20

    print(f"=== Optimization started. {len(candidates)} candidates, budget={budget_s}s ===")
    print(f"Initial best: {best_score}")

    for idx, (name, a) in enumerate(candidates):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if idx < 3:
            time_alloc = min(time_per_early, remaining * 0.5)
        else:
            time_alloc = min(time_per_late, remaining * 0.4)
        if time_alloc < 8:
            time_alloc = max(5, remaining * 0.3)

        n = len(a)
        init_score = score_sequence(a)

        if init_score >= best_score and 'prev' not in name:
            # Skip if already worse than best and not a perturbation of best
            if init_score > best_score * 1.1:
                continue

        verbose = idx < 3
        if verbose:
            print(f"\n[{idx+1}/{len(candidates)}] {name}, n={n}, init={init_score:.6f}, alloc={time_alloc:.0f}s")

        # LP refinement
        a_lp, score_lp = lp_refine(a, time_limit_s=time_alloc * 0.65, verbose=verbose)
        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"  *** NEW BEST (LP): {score_lp:.6f} from {name} (n={len(a_lp)})")

        # Spectral refinement on LP result
        remaining = deadline - time.monotonic()
        if remaining > 15 and time_alloc > 30:
            a_spec, score_spec = spectral_refine(a_lp, time_limit_s=min(20, remaining * 0.15), verbose=verbose)
            if score_spec < best_score:
                best_score = score_spec
                best_seq = a_spec.copy()
                print(f"  *** NEW BEST (Spectral): {score_spec:.6f} from {name}")
                # Continue LP from spectral result
                remaining = deadline - time.monotonic()
                if remaining > 20:
                    a2, score2 = lp_refine(best_seq, time_limit_s=min(30, remaining * 0.5), verbose=verbose)
                    if score2 < best_score:
                        best_score = score2
                        best_seq = a2.copy()
                        print(f"  *** NEW BEST (LP after spectral): {score2:.6f}")

    # === Final polishing phase ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        print(f"\n=== Final polishing (remaining={remaining:.0f}s, best={best_score:.6f}, n={len(best_seq)}) ===")

        if remaining > 60:
            # Multiple rounds of polish with different methods
            round_time = remaining / 4

            # Round 1: Local search
            a, score = local_search(best_seq, time_limit_s=round_time * 0.8, rng=rng, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"  After LS round 1: {score:.6f}")

            # Round 2: LP from polished result
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a, score = lp_refine(best_seq, time_limit_s=remaining * 0.6, verbose=True)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"  After LP round 2: {score:.6f}")

            # Round 3: Spectral + LP
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a, score = spectral_refine(best_seq, time_limit_s=remaining * 0.2, verbose=True)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                a, score = lp_refine(best_seq, time_limit_s=remaining * 0.4, verbose=True)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"  After spectral+LP round 3: {score:.6f}")

            # Round 4: Final local search
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a, score = local_search(best_seq, time_limit_s=remaining * 0.7, rng=rng, verbose=True)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"  After final LS: {score:.6f}")

    if best_seq is None:
        # Fallback: uniform sequence
        best_seq = np.ones(500)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Ensure all values are valid for evaluate_sequence
    best_seq = np.clip(best_seq, 0, 1000)
    # Scale to avoid sum being too small
    s = best_seq.sum()
    if s < 1.0:
        best_seq = best_seq * (1.0 / s)

    print(f"\nFinal result: score={best_score:.6f}, n={len(best_seq)}")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified via evaluate_sequence: {verify_score:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")

    return [float(x) for x in best_seq]
