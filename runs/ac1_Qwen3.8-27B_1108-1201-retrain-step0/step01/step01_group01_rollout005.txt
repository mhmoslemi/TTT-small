Smoothed optimization with log-sum-exp, softmax parameterization, and L-BFGS-B.
Multiple phases with increasing beta for progressive sharpening, followed by
exact subgradient polishing.

import numpy as np
import time
from scipy.optimize import minimize

def _compute_score(a):
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)

def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax(x):
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    s = np.sum(e)
    if s < 1e-300:
        return np.full(len(x), 1.0 / len(x))
    return e / s

def _smoothed_obj(x, beta):
    a = _softmax(x)
    b = np.convolve(a, a)
    b_max = np.max(b)
    shifted = beta * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return b_max + log_sum / beta

def _smoothed_grad_x(x, beta):
    a = _softmax(x)
    b = np.convolve(a, a)
    b_max = np.max(b)
    shifted = beta * (b - b_max)

    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        return np.zeros_like(x)
    weights /= w_sum
    grad_a = 2.0 * np.correlate(weights, a, mode='valid')
    grad_a_centered = grad_a - np.mean(grad_a)  # Not needed for softmax, but for numerical stability
    # Actually, for softmax parameterization, we don't need to center.
    # The gradient formula handles the constraint automatically.
    ga = np.sum(grad_a * a)
    grad_x = a * (grad_a - ga)
    return grad_x

def _smoothed_obj_grad(x, beta):

    b = np.convolve(a, a)
    b_max = np.max(b)
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -700, 700)
    log_sum = np.log(np.sum(np.exp(shifted)))
    val = b_max + log_sum / beta
    weights = np.exp(shifted)

    ga = np.sum(grad_a * a)

def _lbfgs_phase(a_init, beta, deadline, maxiter=200):
    """Run L-BFGS-B optimization in softmax space."""
    # Convert to softmax parameters
    a = a_init.copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.full(len(a), 1.0/len(a))
        S = 1.0
    a = a / S
    # Log with small offset for zeros
    eps = 1e-15
    x0 = np.log(a + eps)
    
    result = minimize(
        _smoothed_obj_grad, x0, args=(beta,),
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': maxiter, 'maxfun': 50000, 'gtol': 1e-8}
    )
    
    a_new = _softmax(result.x)
    return a_new, result

def _subgradient_polish(a, deadline, M=10, eta_init=None):
    """Exact subgradient descent for final polishing."""
    n = len(a)
    if eta_init is None:
        eta_init = 0.05 / np.sqrt(n)
    
    a = a.astype(np.float64).copy()
    S = np.sum(a)
    a = a / S

    best_score = _compute_score(best)

Smoothed optimization for minimizing normalized autocorrelation peak.
Uses log-sum-exp smoothing, softmax parameterization, L-BFGS-B, and
exact subgradient polishing with adaptive parameters.
"""

import numpy as np
import time

def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) for a normalized sequence (sum=1)."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    s = np.sum(e)
    if s < 1e-300:
        return np.full(len(x), 1.0 / len(x))
    return e / s
