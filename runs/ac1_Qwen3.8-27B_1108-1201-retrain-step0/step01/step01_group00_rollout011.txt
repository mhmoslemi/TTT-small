def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:

    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def topk_grad(a, tau):
    """
    Compute soft-max value and gradient of the convolution with temperature tau.
    Smaller tau focuses more on the peak.
    Returns (value, gradient_wrt_a, convolution, weights).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)

    # Gradient via FFT: g[j] = 2 * (a_reversed * w)[j + n - 1]
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1:2 * n - 1]

    return value, g, c, w


def exact_peak_grad(a):

    Compute exact subgradient of max(convolution) at the argmax.
    Returns (peak_value, gradient, peak_index).

            val += a[k_star - j]
        if 0 <= j - k_star < n:
            val += a[j - k_star]

    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))

        # Compute contribution of each element to the peak
        contribs = np.zeros(n)
        for i in range(n):
            if 0 <= peak_idx - i < n:
                contribs[i] = current[i] * current[peak_idx - i]

        # Strategy: reduce top contributors, increase low contributors
        r = rng.random()

        if r < 0.5:
            # Reduce top contributors, redistribute to bottom
            top_k = max(4, n // 16)
            bottom_k = max(4, n // 8)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            bottom_idx = np.argsort(contribs)[:bottom_k]

            reduction = rng.uniform(0.02, 0.06)
            current[top_idx] *= (1.0 - reduction)
            current[bottom_idx] *= (1.0 + reduction * len(top_idx) / len(bottom_idx))

        elif r < 0.7:
            # Pair swap: reduce a top contributor, increase a random other
            top_k = max(4, n // 12)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            j_red = top_idx[int(rng.integers(0, top_k))]
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red:
                amt = current[j_red] * rng.uniform(0.02, 0.05)
                current[j_red] -= amt
                current[j_inc] += amt
                current[j_red] = max(0.0, current[j_red])

        elif r < 0.85:
            # Scale a random block
            width = max(2, int(rng.integers(2, n // 6)))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.03))
            current[start:start + width] *= factor

        else:
            # Small global perturbation
            current *= np.exp(rng.normal(0, 0.005, size=n))

        current = np.clip(current, 0.0, 1000.0)
        new_score = local_score(current)

        if new_score < current_score:
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current_score = new_score

        temp *= cooling
        if step % 1000 == 0:
            print(f"    PS step={step}, cur={current_score:.6f}, best={best_score:.6f}, temp={temp:.5f}")

    return best, best_score


def make_structured_init(n, rng, method='broadband'):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)

    if method == 'broadband':
        # Broadband: sum of many incommensurate frequencies
        a = np.ones(n)
        for k in range(1, 15):
            freq = k * (1 + 0.7 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.15 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'two_block':
        # Two-block: high values in two separated regions
        a = np.ones(n)
        block1 = slice(0, n // 3)
        block2 = slice(2 * n // 3, n)
        a[block1] = 1.5
        a[block2] = 1.5
        a[n // 3:2 * n // 3] = 0.5
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'periodic':
        # Based on a simple periodic pattern with low autocorrelation
        period = max(3, int(np.sqrt(n)))
        pattern = np.ones(period)
        pattern[0] = 1.5
        pattern[period // 2] = 0.5
        a = np.tile(pattern, n // period + 1)[:n]
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimizing the evaluation score.
    Phase 1: Refined gradient optimization at n=256 (primary)
    Phase 2: Peak-shave local search polish
    Phase 3: Coordinate descent polish
    Phase 4: Secondary exploration at other lengths
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    n_main = 256

    # === PHASE 1: Primary gradient optimization at n=256 ===
    print(f"\n{'='*50}")
    print(f"PHASE 1: Gradient optimization (n={n_main})")
    print(f"{'='*50}")

    # Prepare initial theta from best available
    if start_seq is not None:
        if len(start_seq) == n_main:
            a0 = start_seq.copy()
        else:
            x_old = np.linspace(0, 1, len(start_seq))
            x_new = np.linspace(0, 1, n_main)
            a0 = np.interp(x_new, x_old, start_seq)
    else:
        a0 = make_structured_init(n_main, rng, 'broadband')

    theta_init = np.log(a0 / np.sum(a0) + 1e-10)
    S_target = np.sqrt(2.0 * n_main)

    # Run 55% of time budget for primary optimization
    t_phase1 = 0.55 * time_budget
    theta_best, score1 = optimize_refined(n_main, theta_init, t_phase1, rng)
    a_best = softmax_normalize(theta_best) * S_target
    actual1 = local_score(a_best)
    print(f"  Phase 1 result: score={actual1:.6f}")

    if actual1 < best_score:
        best_score = actual1
        best_seq = a_best.copy()
        print(f"  *** New best: {best_score:.6f} ***")

    # Also try a fresh start with different initialization
    if time.monotonic() < deadline - 60:
        a0_2 = make_structured_init(n_main, rng, 'broadband')
        theta_init_2 = np.log(a0_2 / np.sum(a0_2) + 1e-10)
        t_phase1b = min(0.15 * time_budget, deadline - time.monotonic() - 60)
        if t_phase1b > 10:
            theta_best2, score1b = optimize_refined(n_main, theta_init_2, t_phase1b, rng)
            a_best2 = softmax_normalize(theta_best2) * S_target
            actual1b = local_score(a_best2)
            print(f"  Phase 1b result: score={actual1b:.6f}")
            if actual1b < best_score:
                best_score = actual1b
                best_seq = a_best2.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === PHASE 2: Peak-shave polish ===
    print(f"\n{'='*50}")
    print(f"PHASE 2: Peak-shave polish")
    print(f"{'='*50}")

    if best_seq is not None:
        t_phase2 = min(0.15 * time_budget, max(20.0, deadline - time.monotonic() - 60))
        print(f"  Time allocated: {t_phase2:.0f}s, starting score: {best_score:.6f}")
        polished, ps_score = peak_shave_polish(best_seq, t_phase2, rng)
        if ps_score < best_score:
            best_score = ps_score
            best_seq = polished.copy()
            print(f"  After peak-shave: {best_score:.6f}")
        else:
            print(f"  After peak-shave: no improvement (still {best_score:.6f})")

    # === PHASE 3: Coordinate descent ===
    print(f"\n{'='*50}")
    print(f"PHASE 3: Coordinate descent")
    print(f"{'='*50}")

    if best_seq is not None:
        t_phase3 = min(0.10 * time_budget, max(10.0, deadline - time.monotonic() - 30))
        if t_phase3 > 10:
            print(f"  Time allocated: {t_phase3:.0f}s, starting score: {best_score:.6f}")
            cd_result, cd_score = coordinate_descent(best_seq, t_phase3, rng)
            if cd_score < best_score:
                best_score = cd_score
                best_seq = cd_result.copy()
                print(f"  After CD: {best_score:.6f}")
            else:
                print(f"  After CD: no improvement (still {best_score:.6f})")

    # === PHASE 4: Secondary exploration at other lengths ===
    print(f"\n{'='*50}")
    print(f"PHASE 4: Secondary exploration")
    print(f"{'='*50}")

    for n_try in [320, 384]:
        if time.monotonic() > deadline - 30:
            break
        t_try = min(25.0, deadline - time.monotonic() - 20)
        if t_try < 8:
            break

        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a0_try = np.interp(x_new, x_old, best_seq)
        else:
            a0_try = make_structured_init(n_try, rng, 'broadband')

        theta_try = np.log(a0_try / np.sum(a0_try) + 1e-10)
        S_try = np.sqrt(2.0 * n_try)

        theta_res, s_try = optimize_refined(n_try, theta_try, t_try, rng)
        a_res = softmax_normalize(theta_res) * S_try
        actual_try = local_score(a_res)
        print(f"  n={n_try}: score={actual_try:.6f}")

        if actual_try < best_score:
            best_score = actual_try
            best_seq = a_res.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === FINAL: Additional short polish if time remains ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_final = deadline - time.monotonic() - 10
        if t_final > 15:
            print(f"\n{'='*50}")
            print(f"FINAL POLISH (t={t_final:.0f}s)")
            print(f"{'='*50}")
            final_polished, final_score = peak_shave_polish(best_seq, t_final, rng)
            if final_score < best_score:
                best_score = final_score
                best_seq = final_polished.copy()
                print(f"  Final score: {best_score:.6f}")

    # Verify with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'='*50}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
