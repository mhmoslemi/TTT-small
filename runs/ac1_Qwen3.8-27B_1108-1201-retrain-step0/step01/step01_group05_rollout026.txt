"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences using
FFT-based full subgradient with softmax weighting, momentum, and
Polyak step size. Key improvement over top-k subgradient: uses ALL
active constraints via efficient FFT computation of the weighted
subgradient, giving a much better gradient direction.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_full_gradient(a, w):
    """Compute the full weighted subgradient g = 2 * (a cross-correlate w_shifted).
    
    g[m] = 2 * sum_{d=0}^{n-1} w[m+d] * a[d]
    
    Computed via FFT: g[m] = 2 * convolve(a, w_reversed)[2n-2-m]
    """
    n = len(a)
    w_rev = w[::-1].copy()
    # convolve(a, w_rev) has length n + (2n-1) - 1 = 3n-2
    size = 3 * n - 2
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    WR = fft(w_rev, fsize)
    conv = np.real(ifft(A * WR))[:size]
    # g[m] = 2 * conv[2n-2-m] for m = 0, ..., n-1
    # So g = 2 * conv[n-1 : 2n-1][::-1]
    g = 2.0 * conv[n-1:2*n-1][::-1]
    return g


def softmax_weights(c, beta):
    """Compute softmax weights for the convolution values."""
    c_shifted = c - c.max()
    w = np.exp(beta * c_shifted)
    s = w.sum()
    if s < 1e-300:
        # Fallback: uniform
        return np.ones(len(c)) / len(c)
    return w / s


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def run_subgradient(a_init, n, S, max_time, rng, 
                    beta_init=5.0, momentum_init=0.5,
                    step_init=None, shake_interval=500,
                    shake_magnitude=0.05):
    """Full FFT-based subgradient method with momentum and adaptive step size."""
    a = safe_normalize(a_init, n)
    momentum = np.zeros(n, dtype=np.float64)
    best_score = np.inf
    best_local = a.copy()
    
    if step_init is None:
        step_init = 0.1 * S / n
    
    step = step_init
    mu = momentum_init  # momentum coefficient
    lower_bound = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound on max conv
    
    start = time.monotonic()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        
        # Compute convolution
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Compute softmax weights
        w = softmax_weights(c, beta_init)
        
        # Compute full gradient via FFT
        g = compute_full_gradient(a, w)
        
        # Normalize gradient
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g = g / gnorm
        
        # Polyak step size: step = (f - f*) / ||g||^2, but we normalized g
        # so ||g|| = 1, giving step = (f - f*). We scale this down.
        polyak_step = (cur - lower_bound)
        step_new = min(step_init * 2, max(step_init * 0.1, 0.5 * polyak_step))
        
        # Momentum update
        momentum = mu * momentum + g
        a_new = a - step_new * momentum
        a = project_simplex(a_new, S)
        
        # Decay momentum slightly
        mu = max(0.3, momentum_init * (0.999 ** t))
        
        # Periodic shake to escape local minima
        if t % shake_interval == 0:
            noise = rng.normal(0.0, shake_magnitude * S / n, n)
            a = project_simplex(a + noise, S)
    
    return best_local, best_score


def run_annealed_subgradient(a_init, n, S, max_time, rng):
    """Subgradient method with annealed beta (starts smooth, gets sharper)."""
    a = safe_normalize(a_init, n)
    momentum = np.zeros(n, dtype=np.float64)
    best_score = np.inf
    best_local = a.copy()
    
    S_val = S
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    
    start = time.monotonic()
    t = 0
    step_init = 0.1 * S_val / n
    step = step_init
    mu = 0.7
    
    while time.monotonic() - start < max_time:
        t += 1
        
        # Anneal beta: start low (smooth), increase over time
        fraction = t / max(1, int(max_time / 0.05))  # ~20s per "epoch"
        beta = 1.0 + 20.0 * min(1.0, fraction)
        
        # Compute convolution
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Compute softmax weights
        w = softmax_weights(c, beta)
        
        # Compute full gradient via FFT
        g = compute_full_gradient(a, w)
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g = g / gnorm
        
        # Adaptive step size
        step = step_init * max(0.1, 1.0 / (1.0 + 0.01 * t))
        
        # Momentum
        momentum = mu * momentum + g
        a_new = a - step * momentum
        a = project_simplex(a_new, S_val)
        
        # Decay momentum

    return best_local, best_score


def run_lp_based(a_init, n, S, max_time, rng):
    """LP-based direction finding (simplified: use top constraints)."""
    from scipy.optimize import linprog
    
    a = safe_normalize(a_init, n)
    best_score = np.inf
    best_local = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        M = cur * 1.001  # slight slack
        
        # Build constraint matrix A_ub: each row k gives (a * b)_k <= M
        # (a*b)_k = sum_j a_j * b_{k-j} for valid j and k-j
        # This is a Toeplitz matrix
        
        # For efficiency, only use top constraints
        n_constraints = 50
        top_k = np.argsort(c)[::-1][:n_constraints]
        
        # Build A_ub (n_constraints x n)
        A_ub = np.zeros((n_constraints, n))
        for i, k in enumerate(top_k):
            k = int(k)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    A_ub[i, j] = a[j]
        
        # Maximize sum(b) subject to A_ub @ b <= M, b >= 0
        # linprog minimizes, so minimize -sum(b)
        c_lin = -np.ones(n)
        res = linprog(c_lin, A_ub=A_ub, b_ub=np.full(n_constraints, M),
                      bounds=[(0, None)] * n, method='highs')
        
        if res.status == 0:
            b_opt = res.x
            b_sum = b_opt.sum()
            if b_sum > 1e-10:
                b_normalized = b_opt * (S / b_sum)
                # Line search / interpolation
                t_val = 0.1
                a_new = (1 - t_val) * a + t_val * b_normalized
                a = project_simplex(a_new, S)
        
        # Occasionally random restart
        if (time.monotonic() - start) % 60 < 0.1:
            pass
    
    return best_local, best_score

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Key improvement over previous approach: uses FULL subgradient computed via
FFT (all convolution lags weighted by softmax) instead of top-k, giving a
dramatically better gradient direction at O(n log n) cost per step.
Combines heavy-ball momentum, adaptive step size, periodic shakes, and
multi-start with coordinate refinement polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_full_gradient(a, w):
    """Compute g[m] = 2 * sum_{d=0}^{n-1} w[m+d] * a[d] via FFT.
    
    This is the full weighted subgradient of max(c) using all lags.
    """
    n = len(a)
    w_rev = w[::-1].copy()
    size = 3 * n - 2
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    WR = fft(w_rev, fsize)
    conv = np.real(ifft(A * WR))[:size]
    # g[m] = 2 * conv[2n-2-m] for m=0..n-1
    g = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return g


def softmax_weights(c, beta):
    """Softmax weights over convolution values."""
    c_shifted = c - c.max()
    w = np.exp(beta * c_shifted)
    s = w.sum()
    if s < 1e-300:
        return np.ones(len(c)) / len(c)
    return w / s


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def run_subgradient(a_init, n, S, max_time, rng,
                    beta=5.0, mu=0.7, alpha_init=1.0,
                    shake_interval=400, shake_mag=0.05):
    """Full FFT-based subgradient method with heavy-ball momentum."""
    a = safe_normalize(a_init, n)
    v = np.zeros(n, dtype=np.float64)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()

    start = time.monotonic()
    t = 0
    last_improve = 0
    alpha = alpha_init

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()

        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()
            last_improve = t

        stale = t - last_improve

        # Adaptive step: shrink when stuck
        alpha = alpha_init * max(0.05, 1.0 / (1.0 + 0.002 * stale))

        # Softmax weights over all lags
        w = softmax_weights(c, beta)

        # Full gradient via FFT
        g = compute_full_gradient(a, w)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn

        # Heavy-ball momentum
        v = mu * v + g
        a = project_simplex(a - alpha * v, S)

        # Periodic shake to escape plateaus
        if t % shake_interval == 0 and stale > 80:
            noise = rng.normal(0.0, shake_mag * S / n, n)
            a = project_simplex(a + noise, S)
            v *= 0.3

    return best_local, best_score


def run_annealed(a_init, n, S, max_time, rng,
                 beta_start=2.0, beta_end=20.0,
                 mu=0.7, alpha_init=1.0):
    """Subgradient with annealed softmax temperature."""
    a = safe_normalize(a_init, n)
    v = np.zeros(n, dtype=np.float64)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()

    start = time.monotonic()
    t = 0
    total_steps_est = max(1, int(max_time / 0.003))  # estimate

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()

        # Anneal beta
        frac = min(1.0, t / (total_steps_est * 0.5))
        beta = beta_start + (beta_end - beta_start) * frac

        w = softmax_weights(c, beta)
        g = compute_full_gradient(a, w)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn

        alpha = alpha_init / (1.0 + 0.0005 * t)
        v = mu * v + g
        a = project_simplex(a - alpha * v, S)

        if t % 500 == 0 and rng.random() < 0.2:
            noise = rng.normal(0.0, 0.03 * S / n, n)
            a = project_simplex(a + noise, S)
            v *= 0.5

    return best_local, best_score


def coordinate_refine(a, n, S, max_time, rng):
    """Coordinate-wise greedy refinement: try small perturbations
    on random coordinates and keep if improvement."""
    a = safe_normalize(a, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()

    start = time.monotonic()
    # Precompute for fast update
    while time.monotonic() - start < max_time:
        batch = rng.integers(0, n, size=512)
        for idx in batch:
            idx = int(idx)
            if a[idx] < 1e-10:
                continue
            # Try decreasing
            for frac in (0.95, 0.9, 0.5):
                old = a[idx]
                a[idx] = old * frac
                s = a.sum()
                if s < 1e-10:
                    a[idx] = old
                    break
                a = a * (S / s)
                c = fft_convolve_self(a)
                cur = c.max()
                if cur < best_score - 1e-12:
                    best_score = cur
                    best_local = a.copy()
                    break
                a[idx] = old
            # Try increasing (if room)
            if a[idx] < 100:
                a[idx] = a[idx] * 1.1 + 0.01
                s = a.sum()
                a = a * (S / s)
                c = fft_convolve_self(a)
                cur = c.max()
                if cur < best_score - 1e-12:
                    best_score = cur
                    best_local = a.copy()
                else:
                    a = best_local.copy()

    return best_local, best_score
