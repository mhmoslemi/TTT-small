"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
- Top-K subgradient method with Nesterov momentum
- FFT-based fast convolution
- Adaptive step size with backtracking
- Multiple initialization strategies (provided, chirp, random, wavelet)
- Size exploration (n = 3000 to 10000)
- Coordinate descent polishing phase

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve(a):
    """Compute full autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1
    while fsize < size + n:  # oversize a bit for numerical safety
        fsize *= 2
    A = fft(a, fsize)
    result = ifft(A * A).real[:size]
    # Clean up numerical noise
    result[np.abs(result) < 1e-12] = 0.0
    return result


def compute_topk_grad(a, c, K):
    """
    Compute the average gradient of the top-K terms of c = conv(a,a).
    Returns a vector of length n.
    """
    n = len(a)
    top_k = np.argsort(c)[-K:]
    grad = np.zeros(n)
    for k in top_k:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            grad[indices] += a[k - indices]
            # Handle the diagonal: when k == 2j, dc_k/da_j = 2*a_j
            if k % 2 == 0:
                j_diag = k // 2
                if lo <= j_diag <= hi:
                    grad[j_diag] += a[j_diag]
    return grad / K


def project_nonneg_sum(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s == 0:
        return np.full(len(a), S / len(a))
    if s <= S:
        # Already satisfies sum <= S, just add the remainder uniformly
        return a + (S - s) / len(a)
    # Need to threshold
    a_sorted = np.sort(a)[::-1]
    cumsum = np.cumsum(a_sorted)
    rho = np.where(cumsum - a_sorted < S)[0]
    if len(rho) == 0:
        return a / s * S
    rho = rho[-1]
    theta = (cumsum[rho] - S) / (rho + 1)
    result = np.maximum(a - theta, 0.0)
    # Fix numerical issues
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        idx = result.argmax()
        result[idx] += diff
    return result


def init_chirp(n, seed=0):
    """Chirp-based initialization."""
    t = np.arange(n, dtype=np.float64)
    freq = 0.02
    a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * t)
    rng = np.random.default_rng(seed)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_random(n, seed=0):
    """Random initialization."""
    rng = np.random.default_rng(seed)
    return rng.exponential(1.0, size=n)


def init_wavelet(n, seed=0):
    """Gaussian wavelet initialization."""
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    n_central = 3
    for i in range(n_central):
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(0.05, 0.15) * n
        amp = 1.0
        t = np.arange(n)
        a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.01  # Small floor
    return a


def init_flat_with_mod(n, seed=0):
    """Flat with low-frequency modulation."""
    t = np.arange(n, dtype=np.float64) / n
    rng = np.random.default_rng(seed)
    a = 1.0
    # Add a few random low-frequency cosine components
    for freq in [1, 2, 3, 5, 7]:
        amp = rng.uniform(0.1, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t + phase)
    return np.maximum(a, 0.01)


def subgradient_optimize(a0, S, t_budget, K_start=50, K_end=3, 
                         lr_start=1.0, momentum_start=0.9, 
                         momentum_end=0.7, verbose=False):
    """
    Top-K subgradient method with Nesterov momentum and adaptive lr.
    """
    n = len(a0)
    a = (a0.astype(np.float64) / max(a0.sum(), 1e-12) * S).copy()
    a = project_nonneg_sum(a, S)
    
    v = np.zeros(n)  # Momentum
    best = a.copy()
    c = fft_convolve(best)
    best_mc = c.max()
    
    t0 = time.monotonic()
    lr = lr_start
    step = 0
    last_improve = 0
    
    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        f = elapsed / t_budget  # Progress fraction [0, 1]
        
        # Anneal K
        K = max(K_end, int(K_start * (K_end / K_start) ** f))
        # Anneal momentum
        mu = momentum_start + (momentum_end - momentum_start) * f
        
        # Compute gradient
        c = fft_convolve(a)
        mc = c.max()
        
        grad = compute_topk_grad(a, c, K)
        
        # Nesterov momentum
        v = mu * v + (1 - mu) * grad
        a_new = a - lr * v
        a_new = project_nonneg_sum(a_new, S)
        
        step += 1
        a = a_new
        
        # Check improvement periodically
        if step % 50 == 0:
            c = fft_convolve(a)
            mc = c.max()
            if mc < best_mc - 1e-10:
                best_mc = mc
                best = a.copy()
                last_improve = step
                # Increase lr slightly
                lr = min(lr * 1.1, lr_start * 3.0)
            else:
                # Decrease lr
                lr = max(lr * 0.7, lr_start * 0.01)
                # If stuck for too long, try a larger step
                if step - last_improve > 500:
                    lr = min(lr * 2.0, lr_start * 1.5)
                    v *= 0.5  # Dampen momentum
    
    # Final evaluation
    c = fft_convolve(best)
    best_mc = c.max()
    score = evaluate_sequence(best.tolist())
    if verbose:
        print(f"  subgrad done: steps={step}, max_c={best_mc:.6f}, score={score:.6f}")
    return best, score, best_mc

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
- Top-K subgradient method with Nesterov momentum and adaptive step size
- FFT-based fast convolution
- Power-method gradient smoothing (sharper than log-sum-exp)

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve(a):
    """Compute full autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize *= 2
    A = fft(a, fsize)
    result = ifft(A * A).real[:size]
    return result


def power_grad(a, c, p):
    """
    Compute gradient using power-method weighting.
    w_k = (c_k / M)^p normalized to sum to 1.
    grad = sum_k w_k * dc_k/da
    This is sharper than exp-based soft-max for the same "focus" level.
    """
    n = len(a)
    M = c.max()
    if M < 1e-15:
        return np.zeros(n)
    # Only use terms above a threshold to save computation
    threshold = 0.5 * M
    above = c > threshold
    if not above.any():
        # Fallback to top-1
        k_idx = c.argmax()
        return _single_grad(a, k_idx, n)
    
    c_above = c[above]
    w = (c_above / M) ** p
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum
    
    # Get indices
    indices = np.where(above)[0]
    
    grad = np.zeros(n)
    for k_idx, w_k in zip(indices, w):
        _accumulate_grad(a, k_idx, n, grad, w_k)
    return grad


def _accumulate_grad(a, k, n, grad, weight):
    """Accumulate gradient contribution from convolution term at position k."""
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo > hi:
        return
    indices = np.arange(lo, hi + 1)
    grad[indices] += weight * a[k - indices]
    # Diagonal correction: when k == 2j, extra a_j term
    if k % 2 == 0:
        j_diag = k // 2
        if lo <= j_diag <= hi:
            grad[j_diag] += weight * a[j_diag]


def _single_grad(a, k, n):
    """Gradient of a single convolution term c_k with respect to a."""
    grad = np.zeros(n)
    _accumulate_grad(a, k, n, grad, 1.0)
    return grad


def project_nonneg_sum(a, S):
    """Project onto {x >= 0, sum(x) = S} using bisection on threshold."""
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        return np.full(n, S / n)
    if s <= S:
        return a + (S - s) / n
    # Need to threshold: find theta such that sum(max(a - theta, 0)) = S
    lo = 0.0
    hi = a.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    result = np.maximum(a - theta, 0.0)
    # Fix rounding
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        idx = result.argmax()
        result[idx] += diff
    return result


def init_chirp(n, seed=0):
    """Chirp-based non-negative initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.4 * np.cos(2 * np.pi * 0.03 * t) + 0.2 * np.cos(2 * np.pi * 0.07 * t)
    rng = np.random.default_rng(seed)
    a += 0.03 * rng.standard_normal(n)
    return np.maximum(a, 0.01)


def init_random(n, seed=0):
    """Random exponential initialization."""
    return np.random.default_rng(seed).exponential(1.0, size=n)


def init_structured(n, seed=0):
    """
    Structured initialization using product of low-frequency cosines.
    Designed to have a relatively flat power spectrum.
    """
    t = np.arange(n, dtype=np.float64) / n
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    # Product of several cosine factors

        phase = rng.uniform(0, 2 * np.pi)
        a *= (1.0 + 0.3 * np.cos(2 * np.pi * freq * t + phase))
    return np.maximum(a, 0.001)


def init_complement(n, seed=0):
    """
    'Complement palindrome' initialization: a[j] + a[n-1-j] = const.
    This structure can reduce the central peak.
    """
    rng = np.random.default_rng(seed)
    half = n // 2
    x = rng.exponential(1.0, size=half)
    C = 2.0
    first_half = x
    second_half = C - x[::-1]
    a = np.concatenate([first_half, second_half])
    if n % 2 == 1:
        a = np.concatenate([a, [C / 2]])
    return np.maximum(a, 0.001)

def peak_shaving_polish(a, S, t_budget, verbose=False):
    """
    Peak shaving: iteratively identify the peak, find the contributing
    pair (j, k-j), and move mass to reduce the peak.
    """

        c = fft_convolve(a)
        k_peak = c.argmax()
        M = c[k_peak]
        
        # Find the pair (j, k_peak - j) that contributes most to c[k_peak]
        # c[k_peak] = sum_j a[j] * a[k_peak - j]
        lo = max(0, k_peak - (n - 1))
        hi = min(n - 1, k_peak)
        
        if lo > hi:
            break
        
        indices = np.arange(lo, hi + 1)
        contribs = a[indices] * a[k_peak - indices]
        j_max = indices[np.argmax(contribs)]
        j_mirror = k_peak - j_max
        
        if j_max == j_mirror:
            # Diagonal: c[k_peak] has a[j]^2 term
            # Reduce a[j_max] and add to a position that doesn't contribute to peak
            j_reduce = j_max
        else:
            # Reduce the larger of the two
            if a[j_max] >= a[j_mirror]:
                j_reduce = j_max
            else:
                j_reduce = j_mirror
        
        # Find a good target position to move mass to
        # We want a position j' such that a[k_peak - j'] is small
        # and a[j'] is small (so we don't create new peaks)

        # But exclude j_reduce and its mirror
        
        # Try a few random positions and the position with minimum "peak sensitivity"
        candidates = []

        for _ in range(20):
            candidates.append(rng.integers(0, n))
        # Position of minimum a

        best_score = np.inf
        
        for j_t in candidates:
            if j_t == j_reduce or j_t == j_mirror:

            sens = a[k_peak - j_t] if 0 <= k_peak - j_t < n else 0.0

        a[best_target] += delta

    score = evaluate_sequence(best.tolist())

        n, init_type, sd, p_s, p_e, lr, tb = cfg
        if time.monotonic() + tb > total_deadline - 120:
            break
        if tb < 10:
            break
        
        # Generate initialization
        if init_type == 'prov' and prov is not None:
            a0 = prov.copy()
        elif init_type == 'chirp':
            a0 = init_chirp(n, sd)
        elif init_type == 'random':
            a0 = init_random(n, sd)
        elif init_type == 'structured':
            a0 = init_structured(n, sd)
        elif init_type == 'complement':
            a0 = init_complement(n, sd)
        else:
            a0 = init_chirp(n, sd)
        
        S = (2.0 * n) ** 0.5
        print(f"  n={n} init={init_type}_{sd} p=[{p_s},{p_e}] lr={lr} t={tb:.0f}s")
        a_r, score, mc = subgradient_optimize(a0, S, tb,
                                               p_start=p_s, p_end=p_e,
                                               lr_start=lr,
                                               verbose=True)
        _record(a_r, score, mc)
    
    # Phase 2: Peak shaving polish
    print("\n=== Phase 2: Peak shaving polish ===")
    rem = total_deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        n_best = len(best_seq)
        S_best = (2.0 * n_best) ** 0.5
        polish_time = min(rem * 0.35, 300)
        print(f"  Polishing n={n_best}, t={polish_time:.0f}s")
        a_r, score, mc = peak_shaving_polish(best_seq, S_best, polish_time, verbose=True)
        _record(a_r, score, mc)
    
    # Phase 3: Final subgradient refinement with tight parameters
    print("\n=== Phase 3: Final refinement ===")
    rem = total_deadline - time.monotonic()
    if rem > 20 and best_seq is not None:
        n_best = len(best_seq)
        S_best = (2.0 * n_best) ** 0.5
        ref_time = rem * 0.8
        print(f"  Refining n={n_best}, t={ref_time:.0f}s")
        # Use tight parameters for final polish
        a_r, score, mc = subgradient_optimize(best_seq, S_best, ref_time,
                                               p_start=5.0, p_end=1.5,
                                               lr_start=0.3, momentum=0.80,
                                               noise_every=1500, noise_scale=0.01,
                                               verbose=True)
        _record(a_r, score, mc)
    
    # Phase 4: If time remains, try peak shaving again on the refined result
    rem = total_deadline - time.monotonic()
    if rem > 15 and best_seq is not None:
        n_best = len(best_seq)
        S_best = (2.0 * n_best) ** 0.5
        print(f"\n=== Phase 4: Second peak shave (t={rem:.0f}s) ===")
        a_r, score, mc = peak_shaving_polish(best_seq, S_best, rem * 0.8, verbose=True)
        _record(a_r, score, mc)
    
    # Final safety check
    if best_seq is None:
        best_seq = np.ones(2000)
        best_score = evaluate_sequence(best_seq.tolist())
        print("  WARNING: No valid result found, returning uniform.")
    
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_seq)}, max_c={best_mc_global:.4f} ===")
    return [float(x) for x in best_seq]
