    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)

def fast_convolve(a):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent
with 1D golden-section sub-problems, multiple diverse initializations,
adaptive length selection, and perturbation-based restarts.

Key insight: asymmetric sequences prevent large values from aligning in the
self-convolution, allowing objectives well below the symmetric lower bound of 2.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_from_conv(a, conv):
    """Compute 2*n*max(conv)/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 1e15
    return 2.0 * n * conv.max() / (s * s)

def coordinate_descent(a, time_limit, t0, rng, max_stall=100):
    """
    Exact coordinate descent. Returns (best_a, best_obj).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n) / n
    conv = fast_convolve(a)
    size = 2 * n - 1
    s_total = a.sum()
    best_obj = obj_from_conv(a, conv)
    best_a = a.copy()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        s_total = a.sum()
        
        order = rng.permutation(n)
        
        for i in order:
            x = a[i]
            
            k_start = i
            k_end = min(i + n - 1, size - 1)
            quad_pos = i  # position in the affected array where k = 2*i
            
            # Max of unaffected convolution values
            max_unaff = 0.0
            if k_start > 0:
                max_unaff = float(np.max(conv[0:k_start]))
            if k_end < size - 1:
                v = float(np.max(conv[k_end + 1:size]))
                if v > max_unaff:
                    max_unaff = v
            
            # 1D objective: f(xp) = 2n * max(max_unaff, affected(xp)) / (s - x + xp)^2
            # affected(xp) at position pos:
            #   pos == quad_pos: conv[k_start+pos] - x^2 + xp^2
            #   pos != quad_pos: conv[k_start+pos] + 2*(xp-x)*a[pos]
            # (since j_indices = [0,...,n-1])
            
            j_arr = a  # j_indices[pos] = pos, so a[j_indices[pos]] = a[pos]
            affected_conv = conv[k_start:k_end + 1]  # view
            
            def f_1d(xp):
                ds = s_total - x + xp
                if ds < 1e-15:
                    return 1e15
                delta = xp - x
                # Compute max of affected values
                # Linear terms: affected_conv + 2*delta*a[0:n]
                # But we need to handle the quadratic term separately
                lin_vals = affected_conv + 2.0 * delta * j_arr[0:k_end - k_start + 1]
                lin_vals[quad_pos] = affected_conv[quad_pos] - x * x + xp * xp
                m = max(max_unaff, float(np.max(lin_vals)))
                return 2.0 * n * m / (ds * ds)
            
            # Golden section search
            upper = max(10.0 / n, 3.0 * x + 0.01, 3.0 * s_total / n)
            
            # Coarse grid (15 points)
            n_grid = 15
            grid = np.linspace(0.0, upper, n_grid)
            best_gx = min(max(x, 0.0), upper)
            best_gf = f_1d(best_gx)
            for idx in range(n_grid):
                gf = f_1d(grid[idx])
                if gf < best_gf:
                    best_gf = gf
                    best_gx = grid[idx]
            
            # Golden section refinement
            step = upper / (n_grid - 1) if n_grid > 1 else upper
            lo = max(0.0, best_gx - step)
            hi = min(upper, best_gx + step)
            
            if hi - lo < 1e-15:
                xp = best_gx
            else:
                gr = 0.6180339887498949
                c_g = hi - gr * (hi - lo)
                d_g = lo + gr * (hi - lo)
                fc = f_1d(c_g)

                        hi = d_g
                        d_g = c_g
                        fd = fc
                        c_g = hi - gr * (hi - lo)
                        fc = f_1d(c_g)
                    else:
                        lo = c_g
                        c_g = d_g
                        fc = fd
                        d_g = lo + gr * (hi - lo)
                        fd = f_1d(d_g)
                    if hi - lo < 1e-13 * max(1.0, upper):
                        break
                
                xp = (lo + hi) / 2.0
            
            if xp < 0:
                xp = 0.0
            
            # Accept if improvement
            fp = f_1d(xp)

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Decreasing ramp
    ramp = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    
    # Exponential decay
    exp_dec = np.exp(-x / (n * 0.1))
    inits.append(exp_dec / exp_dec.sum())
    
    # Gaussian (centered)
    for sf in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        inits.append(g / g.sum())
    
    # Gaussian (off-center)
    for center in [0.25, 0.75]:
        for sf in [0.15, 0.3]:
            g = np.exp(-0.5 * ((x - n * center) / (sf * n)) ** 2)
            inits.append(g / g.sum())
    
    # Cosine window (raised cosine, peaked in middle)
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits.append(cos_w / cos_w.sum())
    
    # Linear + noise
    base = np.linspace(1, 0.1, n)
    noisy = base + 0.1 * rng.standard_normal(n)
    noisy = np.maximum(noisy, 0)
    if noisy.sum() > 0:
        inits.append(noisy / noisy.sum())
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)
    
    # "Edge" concentration (mass at both ends)
    edge = np.zeros(n)
    edge[0] = 1.0
    edge[-1] = 1.0
    edge[max(0, n//2 - 1):n//2 + 1] = 0.5
    inits.append(edge / edge.sum())
    
    # Parent-derived
    if parent is not None:
        if len(parent) >= n:
            p = np.maximum(parent[:n].copy(), 0)
            if p.sum() > 1e-10:
                inits.append(p / p.sum())
        else:
            # Interpolate
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
            if p.sum() > 1e-10:
                inits.append(p / p.sum())
    
    return inits
