1. Compute conv = a * a (via FFT): O(n log n)

4. Update: a_new = a * exp(-η * gradient_centered) (multiplicative update)
5. Normalize: a_new = a_new / sum(a_new)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive multiplicative mirror descent
with soft-max peak smoothing, FFT-accelerated gradients, and multi-scale search.
Uses larger sequence lengths (up to 4096) for lower achievable objectives.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_gradient_w(a, w):
    """
    Compute gradient of (4n/tau) * sum_k w[k] * 2 * a[k-i] for valid k-i.
    This is cross-correlation of w (length 2n-1) with a (length n).
    (grad)[i] = sum_k w[k] * a[k-i] for 0 <= k-i < n
    = sum_k w[k] * a[k-i] where k ranges over max(0,i) to min(2n-2, i+n-1)

    So k ranges from max(0, i) to min(2n-2, i+n-1).
    
    Let's define: for each i, grad[i] = sum_{k=i}^{i+n-1} w[k] * a[k-i] (with w[k]=0 outside [0,2n-2])
    Let j = k-i, then grad[i] = sum_j w[i+j] * a[j] for j in [0, n-1] (with w out of range = 0)
    
    This is the cross-correlation of w and a: (w ⊛ a)[i] = sum_j w[i+j] * a[j]
    
    This can be computed as: if we define v = w padded, then grad = correlation(v, a).
    In terms of convolution: correlation(v, a)[i] = convolution(flipped_a, v)[i + n - 1]

    # = (a * w_flipped)[i + len(w) - 1] ... let me think.

    # For real a: = sum_j a[j] * w[j+i]

    # cross_corr(a, w) = ifft(conj(fft(a)) * fft(w))  (with appropriate zero-padding)
    # But we need linear (not circular) correlation.
    
    # Zero-pad a to length >= n + (2n-1) - 1 = 3n-2, and w to same length.
    # Actually for linear cross-correlation:
    # We need output length = n + (2n-1) - 1 = 3n-2, but we only need the first n values.
    
    # Simpler: use the convolution formulation.
    # cross_corr(a, w)[i] = sum_j a[j] * w[j+i]
    # = sum_j a[j] * w[j+i]

    # Let k = n-1-j: = sum_k a[k] * w[i+k]

def objective_value(a):
    """Compute the objective value (normalized)."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()

                       eta_min=0.01, anneal_rate=0.999, patience_max=100):

    Multiplicative mirror descent with adaptive temperature.
    a0: initial non-negative sequence (will be normalized to sum=1).
    Returns best sequence and objective.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a_sum
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    tau = tau_init
    eta = eta0
    patience = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Soft-max weights
        scaled = conv / tau
        scaled -= scaled.max()
        exp_scaled = np.exp(scaled)
        w_sum = exp_scaled.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_scaled / w_sum
        
        # Compute gradient (without the 4n/tau factor, just the correlation part)
        grad = compute_gradient_w(a, w)
        
        # Scale gradient: full gradient is (4n/tau) * grad
        # For the multiplicative update, we use: a *= exp(-eta * grad_centered)
        # The 4n/tau factor is absorbed into eta effectively
        grad_centered = grad - grad.mean()
        
        # Adaptive learning rate
        log_factor = -eta * grad_centered
        # Clamp to avoid underflow
        log_factor = np.clip(log_factor, -50, 50)
        a = a * np.exp(log_factor)
        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.ones(n) / n
        else:
            a /= a_sum
        
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        # Anneal
        tau *= anneal_rate
        tau = max(tau, tau_final)
        
        if patience >= patience_max:
            patience = 0
            eta *= 0.7
            if eta < eta_min:
                eta = eta0 * 0.5
