"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent (multiplicative weights)
with soft-max smoothed subgradients. Key insight: monotone (ramp-like) sequences
are near-optimal by the rearrangement inequality. Starts from polynomial ramps
with various exponents and optimizes using adaptive mirror descent across multiple
sequence lengths. Asymptotic target is 4/3 for the ramp family.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def compute_subgradient(a, w, n):
    """
    Compute the subgradient of max_k b[k] w.r.t. a, weighted by soft-max weights w.
    w[k] for k=0..2n-2, a[j] for j=0..n-1.
    sg[j] = sum_k w[k] * (a[k-j] + a[j-k]) with a[i]=0 for i outside [0,n-1].
    """
    sg = np.zeros(n)
    active_idx = np.where(w > w.max() * 1e-6)[0]
    for k_int in active_idx:
        k = int(k_int)
        wk = w[k]
        # Term 1: a[k-j] for j in [max(0, k-n+1), min(k, n-1)]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # a[k-j] for j=j_start..j_end gives a[k-j_start], a[k-j_start-1], ..., a[k-j_end]
            # = a[k-j_end : k-j_start+1][::-1]
            sg[j_start:j_end + 1] += wk * a[k - j_end:k - j_start + 1][::-1]
        # Term 2: a[j-k] for j in [max(k, 0), min(n-1, k+n-1)]
        j_start2 = max(k, 0)
        j_end2 = min(n - 1, k + n - 1)
        if j_start2 <= j_end2:
            # a[j-k] for j=j_start2..j_end2 gives a[j_start2-k], a[j_start2-k+1], ..., a[j_end2-k]
            sg[j_start2:j_end2 + 1] += wk * a[j_start2 - k:j_end2 - k + 1]
    return sg


def mirror_descent_optimize(a0, time_limit, t_param=50.0, eta0=0.5, max_patience=60):
    """
    Multiplicative mirror descent on the probability simplex.
    Minimizes 2n * max(conv(a,a)) with sum(a)=1, a>=0.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a_sum

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    patience = 0
    current_t = t_param
    t0 = time.monotonic()
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-18:
            break

        # Soft-max weights over convolution entries
        scaled = current_t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Compute weighted subgradient
        sg = compute_subgradient(a, w, n)

        # Center the subgradient (invariant to adding constants under sum=1 constraint)
        sg_centered = sg - sg.mean()

        # Multiplicative update
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()  # numerical stability
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.ones(n) / n
        else:
            a /= a_sum

        # Evaluate
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        step_count += 1

        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Adaptive parameter updates
        if patience >= max_patience:
            patience = 0
            eta *= 0.7
            if eta < 0.01:
                current_t *= 1.5
                eta = max(eta, 0.05)
            else:
                current_t = np.clip(current_t * 1.1, 5.0, 500.0)

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Polynomial ramps: a[j] ∝ (j+1)^p
    for p in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0]:
        r = (x + 1) ** p
        inits[f'ramp_p{p}'] = r / r.sum()

    # Exponential ramps
    for rate in [0.3, 0.5, 0.8, 1.2]:
        r = np.exp(rate * x / n)
        inits[f'exp_{rate}'] = r / r.sum()

    # Symmetric shapes (for comparison)
    g = np.exp(-0.5 * ((x - n * 0.5) / (0.2 * n)) ** 2)
    inits['gauss_center'] = g / g.sum()

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Bimodal (edges)
    bi = np.exp(-((x - n * 0.5) / (0.15 * n)) ** 2) + 0.3 * np.exp(-((x - n * 0.1) / (0.05 * n)) ** 2) + 0.3 * np.exp(-((x - n * 0.9) / (0.05 * n)) ** 2)
    inits['bimodal'] = bi / bi.sum()

    # From parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits['parent_interp'] = pp / pp.sum()

    return inits
