    n = len(a)
    S_target = a.sum()
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break
        
        c = _fft_convolve(a)
        M = c.max()
        
        # Find active set (top lags)
        threshold = M * 0.99
        active = np.where(c >= threshold)[0]
        if len(active) > 20:
            # Take top 20
            top_idx = np.argsort(c)[::-1][:20]
            active = top_idx
        
        # Build LP: find d such that a+alpha*d reduces peak
        # Linearized: c_new[m] ≈ c[m] + 2*conv(a,d)[m]
        # Minimize max_m (c[m] + 2*conv(a,d)[m])
        # Subject to: (a + d)[i] >= 0 => d[i] >= -a[i]
        #             sum(d) = 0
        
        # Variables: d[0..n-1], t
        # Min t
        # s.t. 2*conv(a,d)[m] - t <= -c[m]  for m in active
        #      -d[i] <= a[i]  for all i
        #      sum(d) = 0
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        n_act = len(active)
        A_ub = np.zeros((n_act + n, n + 1))
        b_ub = np.zeros(n_act + n)
        
        for idx, m in enumerate(active):
            b_ub[idx] = -c[m]
            A_ub[idx, -1] = -1.0
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            for j in range(lo, hi + 1):
                A_ub[idx, j] = 2.0 * a[m - j]
        
        # d[i] >= -a[i] => -d[i] <= a[i]
        for i in range(n):
            A_ub[n_act + i, i] = -1.0
            b_ub[n_act + i] = a[i]
        
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0  # t doesn't appear in the equality
        b_eq = np.zeros(1)
        
        bounds = [(None, None)] * n + [(0.0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'presolve': True})
            if res.status == 0:
                d = res.x[:n]
                # Line search
                best_alpha = 0.1
                best_s = _objective(a)
                for alpha in [0.5, 0.3, 0.1, 0.05, 0.01]:
                    a_try = a + alpha * d
                    a_try = np.maximum(a_try, 0.0)
                    a_try *= S_target / a_try.sum()
                    s_try = _objective(a_try)
                    if s_try < best_s:
                        best_s = s_try
                        best_alpha = alpha
                a = a + best_alpha * d
                a = np.maximum(a, 0.0)
                a *= S_target / a.sum()
        except Exception:
            break
    
    return a
