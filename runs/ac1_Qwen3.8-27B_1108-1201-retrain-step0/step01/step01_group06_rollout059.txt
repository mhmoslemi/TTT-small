import numpy as np
import time

def fft_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]

def compute_objective_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fft_convolve(a)
    return 2.0 * n * conv.max()

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def softmax_stable(x, T):
    """Compute softmax(T*x) stably."""
    x_scaled = T * x
    x_scaled -= x_scaled.max()
    e = np.exp(x_scaled)
    s = e.sum()
    if s < 1e-300:
        # Fallback: one-hot at argmax
        w = np.zeros_like(x_scaled)
        w[np.argmax(x_scaled)] = 1.0
        return w
    return e / s

def optimize_mirror_descent(a0, time_limit, T_init=5.0, T_max=500.0, eta_init=1.0, eta_min=0.01):
    """
    Multiplicative mirror descent with adaptive temperature and learning rate.
    a0: initial non-negative vector (will be normalized to sum=1).
    Returns (best_a, best_obj) where best_obj = 2n*max(conv) for sum=1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a_sum
    
    conv = fft_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    T = T_init
    eta = eta_init
    stagnation = 0
    max_stagnation = 50
    t_start = time.monotonic()
    
    # FFT size for gradient computation
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    
    while time.monotonic() - t_start < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Compute softmax weights
        w = softmax_stable(conv, T)
        
        # Compute gradient: g[j] = 2 * sum_k w[k] * a[k-j]
        # Using FFT: g = 2 * (a_rev * w)[n-1 : 2n-2+1]
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, fft_size)
        W = np.fft.rfft(w, fft_size)
        conv_aw = np.fft.irfft(A_rev * W, fft_size)
        g = 2.0 * conv_aw[n-1 : 2*n-1]  # length n
        
        # Multiplicative update
        log_update = -eta * g
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-12:
            a_new = np.ones(n) / n
        else:
            a_new /= a_new_sum
        
        a = a_new
        conv = fft_convolve(a)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

def fft_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]

def softmax_stable(x, T):
    """Compute softmax(T*x) with numerical stability."""
    x_scaled = T * x
    x_scaled -= x_scaled.max()
    e = np.exp(x_scaled)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(x_scaled)
        w[np.argmax(x_scaled)] = 1.0
        return w
    return e / s

    phase2_start = time.monotonic()
    phase2_budget = 0.45 * (budget_s - 8)
    print(f"\nPhase 2: Deep optimization (budget={phase2_budget:.0f}s). Best: {best_score:.6f}")

    scan_results.sort(key=lambda x: x[0])

    # Select top configs with diversity in length
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if len(top_configs) < 8:
            top_configs.append((n, name, a_opt))
            seen_lengths.add(n)
        elif n not in seen_lengths and len(top_configs) < 12:
            top_configs.append((n, name, a_opt))
            seen_lengths.add(n)

    # Always include current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'BEST', best_sequence.copy()))

    # Also try parent-derived at different lengths
    if parent is not None:
        for n in [128, 256, 512]:
            if n not in seen_lengths:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    pp /= pp.sum()
                    top_configs.append((n, 'parent_deep', pp))

    remaining = _global_deadline - time.monotonic() - 30
    time_per_deep = max(15.0, remaining / max(1, len(top_configs)))

    for n, name, a0 in top_configs:
        if time.monotonic() > _global_deadline - 30:
            break
        t_remaining = _global_deadline - time.monotonic() - 30
        t_alloc = min(time_per_deep, t_remaining)
        if t_alloc < 5:
            break

        # Multiple parameter settings for diversity
        param_sets = [
            (5.0, 0.8),
            (10.0, 0.6),
            (20.0, 0.4),
            (40.0, 0.25),
        ]
        t_single = t_alloc / len(param_sets)

        for T_init, eta_init in param_sets:
            if time.monotonic() > _global_deadline - 30:
                break
            a_opt, obj_norm = optimize_mirror_descent(
                a0, time_limit=t_single,
                T_init=T_init, eta_init=eta_init, rng=rng)

            if obj_norm < best_score / (2.0 * n):
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    elapsed = time.monotonic() - t0
                    print(f"  [{elapsed:.0f}s] Deep: n={n}, {name}, T={T_init:.0f}, score={score:.6f}")

    # ================================================================
    # Phase 3: Polishing (25% of time)
    # ================================================================
    print(f"\nPhase 3: Polishing. Best: {best_score:.6f}")

    if best_sequence is not None:
        remaining = _global_deadline - time.monotonic() - 5
        if remaining > 10:
            polish_params = [
                (30.0, 0.3), (50.0, 0.2), (80.0, 0.15),
                (120.0, 0.1), (200.0, 0.08), (300.0, 0.05),
            ]
            for T_init, eta_init in polish_params:
                if time.monotonic() > _global_deadline - 5:
                    break
                t_single = min(remaining / 3.0, _global_deadline - time.monotonic() - 5)
                if t_single < 3:
                    break

                a_opt, obj_norm = optimize_mirror_descent(
                    best_sequence, time_limit=t_single,
                    T_init=T_init, eta_init=eta_init, T_max=1000.0,
                    eta_min=0.002, rng=rng)

                if obj_norm < best_score / (2.0 * len(best_sequence)):
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        elapsed = time.monotonic() - t0
                        print(f"  [{elapsed:.0f}s] Polish: T={T_init:.0f}, eta={eta_init}, score={score:.6f}")
                remaining = _global_deadline - time.monotonic() - 5

            # Final micro-polish with very small steps
            if time.monotonic() < _global_deadline - 3:
                t_final = _global_deadline - time.monotonic() - 3
                if t_final > 2:
                    a_opt, obj_norm = optimize_mirror_descent(
                        best_sequence, time_limit=t_final,
                        T_init=500.0, eta_init=0.05, T_max=1000.0,
                        eta_min=0.001, rng=rng)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Final micro-polish: score={score:.6f}")

    if best_sequence is None:
        # Fallback
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    print(f"Total time: {time.monotonic() - t0:.1f}s")
    print(f"{'='*50}")

    return [float(x) for x in best_sequence]
