```python
def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (a.sum() ** 2)

def solve_lp(f, normalize_sum=1.0):
    """Solve LP: minimize t s.t. conv(f, g) <= t, sum(g) = normalize_sum, g >= 0"""
    n = len(f)
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t => c_obj = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints:
    # 1. conv(f, g)[m] - t <= 0 for m = 0, ..., 2n-2
    # 2. sum(g) = normalize_sum (equality)
    
    # Build conv constraint matrix
    # conv(f, g)[m] = sum_j f[j] * g[m-j] for valid j
    # So the row for constraint m has f[j] in column (m-j) for valid j, and -1 in the t column
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    for m in range(2 * n - 1):
        for j in range(n):
            idx = m - j
            if 0 <= idx < n:
                A_ub[m, idx] = f[j]
        A_ub[m, -1] = -1.0
    
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = normalize_sum
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([normalize_sum])
```

```python
"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2
using SQP-LP ( Sequential Quadratic Programming with LP subproblems ).
The key idea: linearize the peak autocorrelation at the current point,
solve the resulting LP for the best descent direction, and take a step.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * c.max() / (S * S))


def build_conv_matrix_a(a_k, n):
    """Build matrix C such that C @ d = conv(a_k, d) for d of length n.
    C has shape (2n-1, n). C[m, j] = a_k[m-j] if 0 <= m-j < n, else 0."""
    C = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            C[m, j] = a_k[m - j]
    return C


def sqp_lp_step(a_k, deadline):
    """One SQP-LP iteration. Returns (a_new, t_opt, t_curr)."""
    n = len(a_k)
    c_k = np.convolve(a_k, a_k)  # length 2n-1
    t_curr = float(c_k.max())
    
    # Build constraint matrix efficiently
    # conv(a_k, d)[m] = sum_j a_k[m-j] * d[j] for valid j
    # Constraint: 2*conv(a_k, d)[m] + c_k[m] <= t
    # i.e., sum_j 2*a_k[m-j] * d[j] - t <= -c_k[m]
    
    C = build_conv_matrix_a(a_k, n)
    A_ub = np.hstack([2.0 * C, -np.ones((2 * n - 1, 1))])
    b_ub = -c_k
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-a_k[i], None) for i in range(n)] + [(0.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return a_k.copy(), t_curr, t_curr
    
    d = res.x[:n]
    t_opt = res.x[-1]
    
    # Line search
    best_a = a_k.copy()
    best_peak = t_curr
    
    for step in [1.0, 0.7, 0.5, 0.3, 0.1]:
        a_new = a_k + step * d
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-10:
            continue
        a_new = a_new / S
        peak = float(np.convolve(a_new, a_new).max())
        if peak < best_peak:
            best_peak = peak
            best_a = a_new.copy()
    
    return best_a, t_opt, best_peak
```

Ignoring the quadratic term `conv(d, d)`, we get:

```python
"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2
using SQP-LP (Sequential Quadratic Programming with LP subproblems).

Key idea: At each iteration, linearize the peak autocorrelation constraint
around the current point and solve the resulting LP exactly for the optimal
descent direction. This is more principled than gradient-based methods as
it finds the globally optimal step for the linearized subproblem.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * c.max() / (S * S))


def build_conv_constraint_matrix(a_k, n):
    """Build matrix C (2n-1 x n) such that (C @ d)[m] = conv(a_k, d)[m]."""
    C = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            C[m, j] = a_k[m - j]
    return C


def sqp_lp_step(a_k):
    """One SQP-LP iteration. Returns (best_a, improved_peak, curr_peak)."""
    n = len(a_k)
    c_k = np.convolve(a_k, a_k)
    curr_peak = float(c_k.max())

    # Build LP: minimize t s.t. 2*conv(a_k, d)[m] + c_k[m] <= t, sum(d)=0, d >= -a_k
    C = build_conv_constraint_matrix(a_k, n)
    A_ub = np.hstack([2.0 * C, -np.ones((2 * n - 1, 1))])
    b_ub = -c_k

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = [(-a_k[i], None) for i in range(n)] + [(0.0, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})

    if res.status != 0:
        return a_k.copy(), curr_peak, curr_peak

    d = res.x[:n]

    # Line search with multiple step sizes
    best_a = a_k.copy()
    best_peak = curr_peak

    for step in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
        a_new = a_k + step * d
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-10:
            continue
        a_new = a_new / S
        peak = float(np.convolve(a_new, a_new).max())
        if peak < best_peak:
            best_peak = peak
            best_a = a_new.copy()

    return best_a, best_peak, curr_peak


def cross_correlation_lp_step(a_k):
    """Alternative LP step: minimize max(conv(a_k, g)) s.t. sum(g)=1, g>=0.
    Then blend a_k towards g."""
    n = len(a_k)
    # Variables: g[0..n-1], t => n+1 vars
    # Minimize t
    # s.t. conv(a_k, g)[m] - t <= 0 for all m
    #      sum(g) = 1
    #      g >= 0

    C = build_conv_constraint_matrix(a_k, n)
    A_ub = np.hstack([C, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})

    if res.status != 0:
        return a_k.copy()

    g = res.x[:n]

    # Blend
    for alpha in [1.0, 0.8, 0.6, 0.4, 0.2]:
        a_new = (1.0 - alpha) * a_k + alpha * g
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-10:
            continue
        a_new = a_new / S
        return a_new

    return a_k.copy()


def gradient_step(a, lr=0.1, smoothing=0.0):
    """Gradient descent step on peak autocorrelation with optional smoothing."""
    n = len(a)
    c = np.convolve(a, a)

    if smoothing > 0:
        # Soft-max gradient
        c_shifted = c - c.max()
        w = np.exp(smoothing * c_shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for j in range(n):
            lo = max(0, j - n + 1)
            hi = min(2 * n - 1, j + n - 1)
            if hi >= lo:
                grad[j] = 2.0 * np.dot(a[max(0, n-1-hi):-lo if lo < n-1-hi else n-1-hi+1],
                                       w[lo:hi+1])
    else:
        m_star = int(c.argmax())
        grad = np.zeros(n)
        lo = max(0, m_star - n + 1)
        hi = min(n - 1, m_star)
        for j in range(lo, hi + 1):
            grad[j] = 2.0 * a[m_star - j]

    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 1e-10:
        a = a / S
    return a

        starts.append(np.exp(-0.5 * ((i - center) / (n * 0.2)) ** 2))
    # Two-tone
    starts.append(0.3 + 0.7 * np.abs(np.sin(np.pi * i / 4.0)))
    # Power law
    starts.append((1.0 + i) ** (-0.5))

    # Sort by length (shorter first for quick feedback)
    starts.sort(key=lambda x: len(x))

    print(f"=== Starting optimization ===")
    print(f"Budget: {budget_s}s, Start sequences: {len(starts)}")
    print(f"Target: 1.5027")

    # --- Main optimization loop ---
    # Allocate time based on sequence length
    # Shorter sequences: more candidates, less time each
    # Longer sequences: fewer candidates, more time each

    for idx, a0 in enumerate(starts):
        if time.monotonic() > deadline - 60.0:
            break

        n = len(a0)
        t0 = time.monotonic()
        remaining = deadline - t_start

        # Determine iterations based on remaining time and sequence length
        # Estimate time per iteration: ~n^0.5 * 0.005 seconds
        est_time_per_iter = 0.002 * np.sqrt(n) + 0.01
        max_iters = max(20, int((remaining * 0.15) / est_time_per_iter))

        use_sqp = n <= 2048
        use_cross = n <= 1024

        seq, score = optimize_sequence(
            a0, t0 + max(30.0, remaining * 0.15),
            use_sqp=use_sqp, use_cross_lp=use_cross,
            max_sqp_iters=min(max_iters, 500),
            max_grad_iters=min(max_iters // 2, 300)
        )

        record(seq, score)
        elapsed = time.monotonic() - t0
        print(f"  [{idx+1}/{len(starts)}] n={n:4d}, t={elapsed:6.1f}s, "
              f"score={score:.6f}, best={best_score:.6f}")

        # If we found a good score, do perturbation restarts
        if best_score < 1.52 and time.monotonic() < deadline - 120.0:
            for rest in range(3):
                if time.monotonic() > deadline - 60.0:
                    break
                a_pert = perturb_sequence(best_seq, rng, strength=0.05)
                seq2, score2 = optimize_sequence(
                    a_pert, min(deadline - 30.0, time.monotonic() + 120.0),
                    use_sqp=use_sqp, use_cross_lp=False,
                    max_sqp_iters=200, max_grad_iters=150
                )
                record(seq2, score2)

    # --- Final intensive refinement ---
    if best_seq is not None and time.monotonic() < deadline - 20.0:
        print(f"\n=== Final refinement (best={best_score:.6f}) ===")
        n_best = len(best_seq)

        # Multiple passes with different strategies
        for pass_i in range(5):
            if time.monotonic() > deadline - 10.0:
                break

            # Pass A: SQP
            if time.monotonic() < deadline - 15.0:
                sub_deadline = min(deadline - 5.0, time.monotonic() + 180.0)
                seq, score = optimize_sequence(
                    best_seq, sub_deadline,
                    use_sqp=True, use_cross_lp=False,
                    max_sqp_iters=200, max_grad_iters=0
                )
                if score < best_score:
                    record(seq, score)
                    best_seq = seq
                print(f"  Pass {pass_i} SQP: {best_score:.6f}")

            # Pass B: Cross-LP
            if n_best <= 1024 and time.monotonic() < deadline - 15.0:
                sub_deadline = min(deadline - 5.0, time.monotonic() + 120.0)
                seq, score = optimize_sequence(
                    best_seq, sub_deadline,
                    use_sqp=True, use_cross_lp=True,
                    max_sqp_iters=100, max_grad_iters=100
                )
                if score < best_score:
                    record(seq, score)
                    best_seq = seq
                print(f"  Pass {pass_i} CrossLP: {best_score:.6f}")

            # Pass C: Perturbation + short optimization
            if time.monotonic() < deadline - 15.0:
                a_pert = perturb_sequence(best_seq, rng, strength=0.02)
                sub_deadline = min(deadline - 5.0, time.monotonic() + 60.0)
                seq, score = optimize_sequence(
                    a_pert, sub_deadline,
                    use_sqp=True, use_cross_lp=False,
                    max_sqp_iters=100, max_grad_iters=100
                )
                if score < best_score:
                    record(seq, score)
                    best_seq = seq
                print(f"  Pass {pass_i} Perturb: {best_score:.6f}")

    # --- Final evaluation ---
    if best_seq is not None:
        final_score = compute_score(best_seq)
        print(f"\n=== FINAL ===")
        print(f"Sequence length: {len(best_seq)}")
        print(f"Computed score: {final_score:.6f}")
        print(f"Track best: {best_score:.6f}")
        try:
            eval_score = evaluate_sequence([float(x) for x in best_seq])
            print(f"evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

        # Ensure all values are valid
        result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
        return result

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
```
