def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
  * Use L-BFGS on a softmax-parameterized objective (a = S*softmax(z)) to avoid
    explicit constraint handling. The objective is a temperature-continuation
    smoothed max of the self-convolution.
  * Multi-restart with diverse initializations (given sequence, random, smooth,
    sparse patterns).
  * Nesterov-accelerated proximal gradient as a secondary method.
  * Support-reduction polishing: zero out small elements and re-optimize.
  * All FFT-based for O(n log n) per gradient evaluation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    rho_vals = np.where(cond)[0]
    if len(rho_vals) == 0:
        return np.full(n, S / n, dtype=np.float64)
    rho = rho_vals[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2*n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def smoothed_gradient_fft(a, c, T):
    """
    Gradient of T*log(sum(exp(c/T))) w.r.t. a, where c = convolve(a,a).
    Uses FFT cross-correlation: grad[m] = 2 * sum_k w[k] * a[k-m].
    """
    n = len(a)
    c_scaled = c / T
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    w = exp_c / exp_c.sum()
    # Cross-correlation via FFT: corr[lag] = sum_j w[j+lag] * a[j]
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    grad = 2.0 * corr[:n]
    return grad


def z_to_a(z, S):
    """Convert z to a = S * softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    total = exp_z.sum()
    if total < 1e-300:
        return np.full(len(z), S / len(z), dtype=np.float64)
    return exp_z / total * S


def safe_z_from_a(a):
    """Convert a to z = log(a + eps) - mean, for numerical stability."""
    eps = 1e-15
    z = np.log(np.maximum(a, eps))
    z = z - z.mean()
    return z


def objective_and_grad_z(z, S, T):
    """Compute objective and gradient w.r.t. z for the smoothed problem."""
    n = len(z)
    a = z_to_a(z, S)
    c = fft_convolve_self(a)
    # Smoothed max value
    c_scaled = c / T
    c_max = c_scaled.max()
    obj_val = T * (c_max + np.log(np.sum(np.exp(c_scaled - c_max))))
    # Gradient w.r.t. a
    c_scale2 = c / T
    exp_c = np.exp(c_scale2 - c_scale2.max())
    w = exp_c / exp_c.sum()
    # grad F / da via FFT cross-correlation
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    grad_a = 2.0 * corr[:n]
    # Chain rule: dF/dz[k] = a[k] * (grad_a[k] - mean_weighted)
    center = np.sum(a * grad_a) / S
    grad_z = a * (grad_a - center)
    return obj_val, grad_z


def run_lbfgs_z(z_init, n, S, max_time, T_schedule):
    """
    Run L-BFGS with temperature continuation on the z-parameterization.
    T_schedule: list of temperatures to use in sequence.
    Returns (best_a, best_hard_max).
    """
    z = z_init.copy()
    best_hard_max = np.inf
    best_a = None
    start = time.monotonic()
    
    for T in T_schedule:
        if time.monotonic() - start > max_time * 0.9:
            break
        
        T_time_budget = max_time / len(T_schedule) * 0.9
        
        def obj_func(z_local):
            val, _ = objective_and_grad_z(z_local, S, T)
            return val
        
        def grad_func(z_local):
            _, g = objective_and_grad_z(z_local, S, T)
            return g
        
        try:
            result = scipy_minimize(
                obj_func, z, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'maxfun': 3000,
                         'ftol': 1e-12, 'gtol': 1e-8, 'past': 40}
            )
            z = result.x
        except Exception:
            pass
        
        # Evaluate hard max
        a = z_to_a(z, S)
        c = fft_convolve_self(a)
        hard_max = c.max()
        if hard_max < best_hard_max:
            best_hard_max = hard_max
            best_a = a.copy()
        print("  T=%.3f: hard_max=%.6f" % (T, hard_max))
