```python
def proj_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Convex optimization of non-negative sequence auto-convolution peak.
Uses projected gradient descent with soft-max smoothing and FFT acceleration,
combined with L-BFGS refinement on the simplex {a >= 0, sum(a) = 1}.
The score 2n*max(a*a)/(sum(a))^2 reduces to 2n*max(a*a) on the simplex,
which is a convex objective, guaranteeing global convergence for fixed n.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_autoconv(a):
    """Compute auto-convolution a*a via FFT. Returns array of length 2*n-1."""
    n = len(a)
    out_len = 2 * n - 1
    N = 1
    while N < out_len:
        N <<= 1
    af = np.fft.rfft(a, N)
    return np.fft.irfft(af * af, N)[:out_len]


def score_of(a):
    """Compute the score 2n*max(a*a)/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fft_autoconv(a)
    return 2.0 * n * c.max() / (s * s)


def smooth_max_and_grad(a, beta):
    """
    Compute soft-max of auto-convolution and its gradient w.r.t. a.
    Returns (soft_max_value, gradient_array).
    """
    n = len(a)
    c = fft_autoconv(a)
    
    # Soft-max with overflow protection
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    Z = exp_c.sum()
    w = exp_c / Z
    soft_max = c_max + np.log(Z) / beta
    
    # Gradient of soft_max w.r.t. a: cross-correlation of a with w
    # g[j] = sum_t a[t] * w[t+j]
    m = len(w)  # = 2n-1
    N = 1
    while N < n + m - 1:
        N <<= 1
    af = np.fft.fft(a, N).astype(complex)
    wf = np.fft.fft(w, N).astype(complex)
    g = np.fft.ifft(np.conj(af) * wf)[:n].real
    
    return soft_max, g


def proj_simplex(v):
    """Euclidean projection onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find largest rho such that u[rho] > (cssv[rho] - 1) / (rho + 1)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gd_optimize(a0, time_limit, seed=0, verbose=False):
    """
    Projected gradient descent with soft-max smoothing and Adam optimizer.
    Minimizes 2n*max(a*a) on the simplex {a >= 0, sum(a) = 1}.
    """
    start = time.monotonic()
    n = len(a0)
    
    # Initialize on simplex
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fft_autoconv(a).max()
    
    # Adam optimizer
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    
    # Beta (smoothing) schedule
    beta_start = 3.0
    beta_end = 40.0
    
    while time.monotonic() - start < time_limit:
        progress = min(1.0, (time.monotonic() - start) / max(time_limit * 0.8, 1.0))
        beta = beta_start + (beta_end - beta_start) * progress
        lr = 0.5 / (1.0 + 0.1 * t)  # Decaying learning rate
        
        soft_max, grad_sm = smooth_max_and_grad(a, beta)
        
        # Score gradient: d(2n*SM)/da = 2n * grad_sm (on the simplex where sum=1)
        grad = 2.0 * n * grad_sm
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a_new)
        
        # Evaluate
        c = fft_autoconv(a_new)
        sc = 2.0 * n * c.max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            if verbose and (t % 200 == 0 or t < 10):
                print(f"        GD t={t}: score/2n={best_score/(2*n):.6f}, beta={beta:.1f}")
        
        a = a_new
    
    return best_a, best_score


def make_init(n, init_type, rng=None):
    """Create initial sequence on the simplex."""
    if init_type == 'uniform':
        return np.ones(n) / n
    elif init_type == 'raised_cosine':
        t = np.arange(n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))
        a = a / a.sum()
        return a
    elif init_type == 'gaussian':
        t = np.arange(n, dtype=float)
        center = (n - 1) / 2.0
        sigma = 0.25 * (n / 2.0)
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        return a / a.sum()
    elif init_type == 'edge':
        t = np.arange(n, dtype=float)
        center = (n - 1) / 2.0
        sigma = 0.3 * (n / 2.0)
        a = 1.0 - np.exp(-0.5 * ((t - center) / sigma) ** 2)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif init_type == 'random':
        if rng is None:
            rng = np.random.default_rng(0)
        a = np.abs(rng.normal(0, 1, n))
        # Smooth it
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        return a / a.sum()
    elif init_type == 'prev_best':
        # Use height_sequence_1 if available
        try:
            a = np.array(height_sequence_1, dtype=float)
            n2 = len(a)
            if n2 != n:
                # Resize by interpolation
                a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n2), a)
            a = np.maximum(a, 0)
            return a / a.sum()
        except:
            return np.ones(n) / n
    else:
        return np.ones(n) / n


def lbfgs_refine(a0, time_limit, seed=0):
    """
    L-BFGS refinement on the simplex. Fixes the last coordinate to
    satisfy the sum constraint, and uses bounds [0, inf) for the rest.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()  # Ensure on simplex
    
    # Work with first n-1 variables (last is determined by sum constraint)
    x0 = a[:-1].copy()
    
    def objective(x):
        a_full = np.append(x, 1.0 - x.sum())
        if (a_full < -1e-10).any():
            return 1e10  # Penalty for infeasibility
        a_full = np.maximum(a_full, 0.0)
        c = fft_autoconv(a_full)
        return 2.0 * n * c.max()
    
    def gradient(x):
        a_full = np.append(x, 1.0 - x.sum())
        a_full = np.maximum(a_full, 0.0)
        beta = 30.0
        soft_max, grad_sm = smooth_max_and_grad(a_full, beta)
        # Gradient w.r.t. first n-1 components
        g = 2.0 * n * grad_sm[:-1]
        # The last component's gradient contributes through the constraint
        # d/dx_i [f(a)] where a_{n-1} = 1 - sum(x):
        # = g_i - g_{n-1}
        g = g - g[-1]
        return g
    
    start = time.monotonic()
    bounds = [(0.0, None)] * (n - 1)
    
    result = minimize(objective, x0, jac=gradient, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': 5000, 'maxfun': 10000,
                                               'time_limit': time_limit, 'ftol': 1e-12,
                                               'gtol': 1e-10})
    
    x_opt = result.x
    a_full = np.append(x_opt, 1.0 - x_opt.sum())
    a_full = np.maximum(a_full, 0.0)
    a_full = a_full / a_full.sum()
    
    sc = 2.0 * n * fft_autoconv(a_full).max()
    print(f"        L-BFGS: score/2n={sc/(2*n):.6f}, time={time.monotonic()-start:.1f}s")
    return a_full, sc

            remaining = deadline - time.monotonic()
            if remaining < time_per_run + 5:
                break
            
            t_limit = min(time_per_run, remaining - 10)
            if t_limit < 10:
                continue
            
            a = make_init(n, init_type, rng)
            a, sc = gd_optimize(a, t_limit, seed=seed + n + hash(init_type) % 1000)
            
            # Convert to actual score (with sum, not normalized to 1)
            actual_score = 2.0 * n * fft_autoconv(a).max() / (a.sum() ** 2)
            # Since a is on the simplex (sum=1), actual_score = 2n * max(a*a) = sc
            # But let's verify
            assert abs(actual_score - sc) < 1e-6, f"Score mismatch: {actual_score} vs {sc}"
            
            if sc < best_score:
                prev_best = best_score
                best_score = sc
                best_seq = a.copy()
                print(f"  Phase 1: n={n}, init={init_type}, score={sc:.6f} "
                      f"(was {prev_best:.6f}), time={time.monotonic()-total_start:.0f}s")
    
    # Phase 2: Focused search - re-optimize best few with longer time
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\nPhase 2: Focused refinement, {remaining:.0f}s remaining")
            
            # Try different beta schedules on the best solution
            n_best = len(best_seq)
            
            # Re-run GD with longer time and different params
            t2 = min(remaining * 0.4, 200)
            a2, sc2 = gd_optimize(best_seq, t2, seed=seed + 999)
            if sc2 < best_score:
                best_score = sc2
                best_seq = a2.copy()
                print(f"  Phase 2 (extended GD): score={sc2:.6f}")
            
            # Try L-BFGS
            remaining2 = deadline - time.monotonic()
            if remaining2 > 30:
                t3 = min(remaining2 * 0.6, 300)
                a3, sc3 = lbfgs_refine(best_seq, t3)
                if sc3 < best_score:
                    best_score = sc3
                    best_seq = a3.copy()
                    print(f"  Phase 2 (L-BFGS): score={sc3:.6f}")
    
    # Phase 3: Final polish
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\nPhase 3: Final polish, {remaining:.0f}s remaining")
            # One more L-BFGS pass
            a4, sc4 = lbfgs_refine(best_seq, remaining * 0.7)
            if sc4 < best_score:
                best_score = sc4
                best_seq = a4.copy()
                print(f"  Phase 3: score={sc4:.6f}")
    
    # Final verification
    if best_seq is None:
        best_seq = np.ones(500) / 500
        best_score = score_of(best_seq)
    
    verify = evaluate_sequence([float(x) for x in best_seq])
    print(f"\nFinal: n={len(best_seq)}, internal_score={best_score:.6f}, "
          f"verified_score={verify:.6f}, total_time={time.monotonic()-total_start:.0f}s")
    
    return [float(x) for x in best_seq]
```
