```python
def _mass_redistribute(a, deadline, max_rounds=100):

        peak_idx = int(c.argmax())
        peak_val = c[peak_idx]
        
        # Find which positions contribute most to the peak
        contrib = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                contrib[j] = a[j] * a[k]
        
        # Find positions that do NOT contribute to the peak
        non_contrib = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if not (0 <= k < n):
                non_contrib[j] = 1.0
        
        # Move mass from high-contrib to high-non-contrib
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break
        
        # Fraction to move
        eps = 0.01 * S_target / n
        
        # Reduce from top contributors, increase to non-contributors
        top_idx = np.argsort(contrib)[-max(1, n//10):]
        bottom_idx = np.where(non_contrib > 0)[0]
        
        if len(bottom_idx) == 0:
            break
        
        # Random selection from non-contributors
        n_move = min(len(bottom_idx), len(top_idx))
        for _ in range(n_move):
            j_don = top_idx[_]
            j_rec = bottom_idx[np.random.randint(len(bottom_idx))]
            move = min(eps, a[j_don] * 0.1)
            a[j_don] -= move
            a[j_rec] += move
        
        a = np.maximum(a, 0.0)
        a = _normalize_to_sum(a, S_target)
        
        score = _compute_obj(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def _multi_scale_optimize(a_coarse, n_target, deadline):
```

```python
"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses smooth peak minimization via log-sum-exp gradients with Toeplitz-matrix
acceleration, adaptive temperature annealing, subgradient peak shaving,
multi-scale optimization, and multi-start with restarts.
"""

import numpy as np
import time


def _compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize_to_sum(a, S):
    """Scale a to have sum exactly S."""
    cur = a.sum()
    if cur < 1e-15:
        return a.copy()
    return a * (S / cur)


def _smooth_peak_grad(a, t):
    """
    Compute t * grad of log-sum-exp(t * conv(a,a)) w.r.t. a.
    Uses Toeplitz matrix for fast matvec.
    Returns (gradient, convolution, max of convolution).
    """
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())

    # Stable softmax weighting
    tc = t * (c - max_c)
    w = np.exp(tc)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum

    # grad[j] = t * sum_m a[m] * w[j+m]
    # Build Toeplitz-like matrix: T[j,m] = w[j+m] (j,m in 0..n-1, j+m in 0..2n-2)
    idx = np.arange(n)
    T = w[idx[:, None] + idx[None, :]]
    grad = t * (T.dot(a))

    return grad, c, max_c


def _optimize_smooth_peak(a_init, deadline, t_start=1.0, t_end=200.0,
                          max_iters=5000, lr_init=0.5, lr_end=0.01,
                          verbose=False):
    """
    Minimize smooth peak of conv(a,a) via gradient descent with projected non-negativity.
    """
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, None)
    if a.sum() < 1e-12:
        a = np.ones(n)
    S_target = 1.0
    a = _normalize_to_sum(a, S_target)

    best_score = _compute_obj(a)
    best_a = a.copy()
    stale_count = 0
    best_stale = best_score

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(max_iters - 1, 1)
        # Anneal temperature exponentially: start low, end high
        t = t_start * (t_end / t_start) ** frac
        # Learning rate: cosine annealing
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * frac))
        lr = max(lr, lr_end * 0.5)

        grad, c, max_c = _smooth_peak_grad(a, t)

        # Project out mean (preserve sum constraint)
        grad = grad - grad.mean()

        # Update with non-negativity projection
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize_to_sum(a_new, S_target)

        a = a_new

        # Check improvement periodically
        if (it + 1) % 150 == 0:
            score = _compute_obj(a)
            if score < best_score:
                if verbose:
                    print(f"      iter {it}: score={score:.6f}, t={t:.1f}, peak={max_c:.6f}")
                best_score = score
                best_a = a.copy()
                stale_count = 0
                best_stale = score
            else:
                stale_count += 1
                # If stuck for a while, add small perturbation
                if stale_count > 8:
                    noise = 0.01 * np.random.default_rng(it).random(n)
                    a = a + noise
                    a = np.maximum(a, 0.0)
                    a = _normalize_to_sum(a, S_target)
                    stale_count = 0

    score = _compute_obj(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def _optimize_subgrad_peak(a_init, deadline, max_iters=5000):
    """
    Subgradient/accelerated method targeting the specific peak of the convolution.
    """
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, None)
    if a.sum() < 1e-12:
        a = np.ones(n)
    S_target = 1.0
    a = _normalize_to_sum(a, S_target)

    best_score = _compute_obj(a)
    best_a = a.copy()

    # FISTA-style momentum
    y = a.copy()
    momentum = 1.0
    it_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = np.convolve(y, y)
        peak_idx = int(c.argmax())

        # Subgradient of max at peak_idx: (grad)[j] = 2*a[peak_idx - j]
        grad = np.zeros(n)
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        for j in range(lo, hi + 1):
            grad[j] = 2.0 * y[peak_idx - j]

        grad = grad - grad.mean()

        lr = 0.3 / (1.0 + 0.02 * it)
        y_new = np.maximum(y - lr * grad, 0.0)
        y_new = _normalize_to_sum(y_new, S_target)

        # FISTA update
        momentum_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum))
        a_new = y_new + ((momentum - 1.0) / momentum_new) * (y_new - a)
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize_to_sum(a_new, S_target)

        a = a_new
        y = y_new
        momentum = momentum_new
        it_count += 1

        if (it + 1) % 500 == 0:
            score = _compute_obj(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"      subgrad iter {it}: score={score:.6f}")

    score = _compute_obj(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def _optimize_mixed(a_init, deadline, n_smooth=3000, n_subgrad=2000):
    """
    Alternate between smooth peak and subgradient peak methods.
    """
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, None)
    if a.sum() < 1e-12:
        a = np.ones(n)
    S_target = 1.0
    a = _normalize_to_sum(a, S_target)

    best_score = _compute_obj(a)
    best_a = a.copy()

    # Phase A: Smooth peak
    sub_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.5)
    a, s = _optimize_smooth_peak(a, sub_deadline, t_start=5.0, t_end=150.0,
                                  max_iters=n_smooth, lr_init=0.6, lr_end=0.01)
    if s < best_score:
        best_score = s
        best_a = a.copy()

    # Phase B: Subgradient peak
    sub_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.5)
    a2, s2 = _optimize_subgrad_peak(a, sub_deadline, max_iters=n_subgrad)
    if s2 < best_score:
        best_score = s2
        best_a = a2.copy()
    elif s < best_score:
        pass  # keep a

    # Phase C: Continue smooth from best
    a3, s3 = _optimize_smooth_peak(best_a, deadline, t_start=50.0, t_end=500.0,
                                    max_iters=n_smooth, lr_init=0.3, lr_end=0.005)
    if s3 < best_score:
        best_score = s3
        best_a = a3.copy()

    return best_a, best_score


def _try_starts(starts, deadline, best_score, best_seq, tag=""):
    """Try optimizing multiple starts, updating best."""
    for idx, a0 in enumerate(starts):
        if time.monotonic() > deadline - 15.0:
            break
        n = len(a0)
        a0 = np.clip(a0.astype(np.float64), 0.0, None)
        if a0.sum() < 1e-12:
            continue

        sub_deadline = min(deadline - 10.0, time.monotonic() + 45.0)
        seq, score = _optimize_smooth_peak(a0, sub_deadline,
                                            t_start=5.0, t_end=100.0,
                                            max_iters=2000, lr_init=0.6, lr_end=0.02)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  {tag}[{idx}] n={n}: score={score:.6f}")

    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses smooth peak minimization with Toeplitz-gradients,
    subgradient peak shaving, multi-scale refinement, and multi-start.
    Returns the best non-negative sequence found within the time budget.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # --- Collect starting points ---
    starts = []

    # Try global height_sequence_1
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 32 and np.all(np.isfinite(gs)):
            for L in [min(256, len(gs)), min(512, len(gs)), min(768, len(gs)), min(1024, len(gs))]:
                starts.append(gs[:L].copy())
            print(f"  Using height_sequence_1 (len={len(gs)})")
    except NameError:
        pass

    # Structured and random starts of various lengths
    for n in [256, 512, 768, 1024]:
        i = np.arange(n, dtype=np.float64)
        x = i / n

        starts.append(np.ones(n))
        starts.append(1.0 + 0.4 * np.cos(2 * np.pi * 5.0 * x))
        starts.append(1.0 + 0.4 * np.cos(2 * np.pi * 13.0 * x))
        starts.append(np.exp(-3.0 * (x - 0.5) ** 2))
        starts.append(1.0 / (1.0 + 8.0 * (x - 0.5) ** 2))
        starts.append(0.5 + 0.5 * np.sin(np.pi * x) ** 2)
        starts.append(rng.random(n) + 0.3)
        # "Comb" pattern
        starts.append(1.0 + 0.5 * ((i % 3 == 0).astype(np.float64)))
        # Two-bump
        starts.append(np.exp(-8.0 * (x - 0.25) ** 2) + np.exp(-8.0 * (x - 0.75) ** 2) + 0.1)

    # Sort by length (shorter first for quick wins)
    starts.sort(key=len)

    print(f"Total starts: {len(starts)}, budget={budget_s}s")

    # --- Phase 1: Quick screening ---
    print("\n  Phase 1: Quick screening")
    for idx, a0 in enumerate(starts):
        if time.monotonic() > deadline - 120.0:
            break
        n = len(a0)
        a0 = np.clip(a0.astype(np.float64), 0.0, None)
        if a0.sum() < 1e-12:
            continue

        sub_deadline = min(deadline - 60.0, time.monotonic() + 20.0)
        seq, score = _optimize_smooth_peak(a0, sub_deadline,
                                            t_start=10.0, t_end=80.0,
                                            max_iters=800, lr_init=0.8, lr_end=0.05)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"    [{idx}] n={n}: score={score:.6f}")

    if best_seq is None:
        # Fallback: use uniform
        best_seq = np.ones(512)
        best_score = _compute_obj(best_seq)
        print(f"  Fallback to uniform: score={best_score:.6f}")

    # --- Phase 2: Deep optimization on best ---
    print(f"\n  Phase 2: Deep optimization (current best={best_score:.6f}, n={len(best_seq)})")
    n_best = len(best_seq)

    # Deep smooth peak with aggressive annealing
    sub_deadline = min(deadline - 60.0, time.monotonic() + 240.0)
    seq, score = _optimize_smooth_peak(best_seq, sub_deadline,
                                        t_start=2.0, t_end=400.0,
                                        max_iters=8000, lr_init=0.8, lr_end=0.005,
                                        verbose=True)
    if score < best_score:
        best_score = score
        best_seq = seq.copy()

    # Subgradient refinement
    sub_deadline = min(deadline - 30.0, time.monotonic() + 120.0)
    seq2, score2 = _optimize_subgrad_peak(best_seq, sub_deadline, max_iters=5000)
    if score2 < best_score:
        best_score = score2
        best_seq = seq2.copy()

    # --- Phase 3: Continue with higher temperature ---
    print(f"\n  Phase 3: High-temperature refinement (current best={best_score:.6f})")
    for pass_i in range(4):
        if time.monotonic() > deadline - 20.0:
            break
        remaining = deadline - 10.0 - time.monotonic()
        if remaining < 10:
            break
        sub_deadline = min(deadline - 5.0, time.monotonic() + remaining / (4 - pass_i))
        t_s = 50.0 * (pass_i + 1)
        t_e = 300.0 * (pass_i + 1)
        seq, score = _optimize_smooth_peak(best_seq, sub_deadline,
                                            t_start=t_s, t_end=t_e,
                                            max_iters=4000, lr_init=0.2, lr_end=0.002)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"    pass {pass_i}: score={score:.6f}")

    # --- Phase 4: Try larger n via upsampling ---
    if time.monotonic() < deadline - 60.0:
        print(f"\n  Phase 4: Upsampling (current best={best_score:.6f}, n={len(best_seq)})")
        n_cur = len(best_seq)
        for n_try in [n_cur * 2, n_cur + 256]:
            if n_try > 2048 or time.monotonic() > deadline - 30.0:
                break
            x_c = np.linspace(0, 1, n_cur)
            x_f = np.linspace(0, 1, n_try)
            a_up = np.interp(x_f, x_c, best_seq)
            sub_deadline = min(deadline - 15.0, time.monotonic() + 60.0)
            seq, score = _optimize_smooth_peak(a_up, sub_deadline,
                                                t_start=5.0, t_end=150.0,
                                                max_iters=3000, lr_init=0.5, lr_end=0.01)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"    n={n_try}: score={score:.6f}")

    # --- Phase 5: Final polish ---
    if best_seq is not None and time.monotonic() < deadline - 10.0:
        print(f"\n  Phase 5: Final polish (current best={best_score:.6f})")
        remaining = deadline - 5.0 - time.monotonic()
        if remaining > 5:
            t_s = max(100.0, 200.0 * (best_score - 1.0))
            seq, score = _optimize_smooth_peak(best_seq, deadline - 3.0,
                                                t_start=t_s, t_end=t_s * 5.0,
                                                max_iters=5000, lr_init=0.15, lr_end=0.001)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"    Final: score={score:.6f}")

    # --- Final validation ---
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"\n  Result: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"  evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"  evaluate_sequence error: {e}")
        # Ensure all values are non-negative
        best_seq = np.maximum(best_seq, 0.0)
        return [float(x) for x in best_seq]

    # Ultimate fallback
    print("  Fallback to uniform 256.")
    return [1.0] * 256
```
