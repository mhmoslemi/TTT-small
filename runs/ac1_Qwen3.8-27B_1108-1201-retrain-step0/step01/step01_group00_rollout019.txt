def fft_convolve(a, b):

    size = 1
    while size < n + m - 1:

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:

    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

            else:

                a = a * S / a.sum()
                alpha = S * 0.005
                no_improve = 0

    return best_a, best_score

"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses: (1) softmax-parameterized gradient descent for coarse optimization,
(2) direct-parameterization gradient with projection for fine-tuning (allows zeros),
(3) structured peak-shaving local search with gradient-guided redistribution.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-6:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def stable_softmax(theta):
    """Numerically stable softmax."""
    t = theta - theta.max()
    e = np.exp(t)
    return e / e.sum()


def grad_conv(a, w):
    """
    Compute g[k] = sum_m w[m] * a[m-k] (convolution-based).
    Uses: g = (w * a_rev)[n-1 : 2n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    corr = fft_convolve(w, a_rev)
    return corr[n - 1:2 * n - 1]


def project_nn_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S} via bisection on threshold."""
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.maximum(v - mid, 0.0)
        s = proj.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    proj = np.maximum(v - 0.5 * (lo + hi), 0.0)
    s = proj.sum()
    if s > 1e-12:
        proj *= S / s
    else:
        proj = np.full(len(v), S / len(v))
    return proj


def optimize_softmax_phase(n, a_init, time_limit, rng):
    """
    Adam gradient descent with softmax parameterization and annealed temperature.
    Returns best (a, score).
    """
    S = np.sqrt(2.0 * n)
    a0 = np.clip(a_init, 1e-4, None)
    theta = np.log(a0 / a0.sum() + 1e-12).astype(np.float64)

    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    step = 0
    lr = 0.8

    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = score_fast(a_init)

    # Temperature: controls smoothing of the max
    tau_init = 15.0 / n * S
    tau_final = 0.3 / n * S
    patience = 0
    max_patience = 1500

    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        frac = min(1.0, elapsed * 1.4)
        tau = tau_init * (tau_final / tau_init) ** frac

        a = stable_softmax(theta) * S
        c = fft_convolve(a, a)

        # Smoothed max gradient
        c_s = c / tau
        c_s -= c_s.max()
        exp_c = np.exp(c_s)
        w = exp_c / exp_c.sum()

        # Gradient w.r.t. a: g_a[k] = 2 * sum_m w[m] * a[m-k]
        g_a = 2.0 * grad_conv(a, w)

        # Gradient w.r.t. theta (Jacobian of softmax):
        # g_theta = a * (g_a - <a, g_a> / S) * S  (chain rule with normalization)
        # Actually: a = softmax(theta) * S
        # da/dtheta = S * softmax_j * (delta_jk - softmax_k)
        # g_theta_j = sum_k (da_k/dtheta_j) * g_a_k = S * a_j * (g_a_j - sum_k a_k g_a_k / S)
        inner = np.dot(a, g_a) / S
        g_theta = S * a * (g_a - inner)

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * g_theta ** 2
        m1_hat = m1 / (1 - b1 ** step)
        m2_hat = m2 / (1 - b2 ** step)

        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        a_new = stable_softmax(theta) * S
        sc = score_fast(a_new)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                theta += 0.3 * rng.standard_normal(n)
                patience = 0

        if step % 10000 == 0:
            print(f"      softmax step={step}, score={sc:.6f}, best={best_score:.6f}, tau={tau:.2e}")

    return best_a, best_score


def optimize_direct_phase(n, a_init, time_limit, rng):
    """
    Direct gradient descent with projection onto {a>=0, sum(a)=S}.
    Uses top-K peak gradient with backtracking line search.
    """
    S = a_init.sum()
    a = np.clip(a_init, 0.0, 1000.0)
    a = a * S / a.sum()

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()

    alpha = 0.01
    K = 3
    no_improve = 0
    max_no_improve = 300
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        peak_val = c.max()

        # Top-K peaks
        top_idx = np.argsort(c)[::-1][:K]
        w = np.zeros(len(c))
        for m in top_idx:
            w[m] = 1.0 / K

        # Gradient: g[k] = 2 * sum_m w[m] * a[m-k]
        g = 2.0 * grad_conv(a, w)

        # Zero-mean to preserve sum
        g = g - g.mean()

        # Backtracking
        improved = False
        for _ in range(25):
            a_try = a - alpha * g
            a_try = project_nn_sum(a_try, S)
            sc = score_fast(a_try)
            if sc < best_score:
                best_score = sc
                best_a = a_try.copy()
                a = a_try
                no_improve = 0
                alpha = min(alpha * 1.05, 0.2)
                improved = True
                break
            alpha *= 0.65

        if not improved:
            no_improve += 1
            if no_improve % 50 == 0:
                # Adaptive: reduce K or perturb
                if K > 1:
                    K = max(1, K - 1)
                elif no_improve > max_no_improve:
                    # Perturb and restart
                    a = best_a + 0.02 * S * rng.standard_normal(n)
                    a = project_nn_sum(a, S)
                    alpha = 0.005
                    K = 2
                    no_improve = 0

        iter_count += 1
        if iter_count % 5000 == 0:
            print(f"      direct step={iter_count}, score={best_score:.6f}, alpha={alpha:.4f}, K={K}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def peak_shave_search(a_init, time_limit, rng):
    """
    Structured local search targeting peak reduction.
    Uses gradient information to guide Redistribution.
    """
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()

    temp = 0.003
    cooling = 0.99985
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        peak_val = c.max()
        peak_lag = int(c.argmax())

        r = rng.random()
        a_new = a.copy()

        if r < 0.45:
            # Gradient-guided: reduce top contributors to peak, boost others
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_lag - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]

            K_red = max(4, n // 10)
            top_idx = np.argsort(contribs)[::-1][:K_red]
            # Reduce these
            reduction = contribs[top_idx] * rng.uniform(0.01, 0.04)
            a_new[top_idx] -= reduction
            total_red = reduction.sum()

            # Redistribute to positions with low "pressure"
            pressure = np.zeros(n)
            for i in range(n):
                j = peak_lag - i
                if 0 <= j < n:
                    pressure[i] = a[j]
            bottom_idx = np.argsort(pressure)[:max(4, n // 8)]
            a_new[bottom_idx] += total_red / len(bottom_idx) * rng.uniform(0.8, 1.2)

        elif r < 0.70:
            # Multi-peak: address top-3 peaks simultaneously
            top_peaks = np.argsort(c)[::-1][:3]
            for m in top_peaks:
                contribs = np.zeros(n)
                for i in range(n):
                    j = m - i
                    if 0 <= j < n:
                        contribs[i] += a[i] * a[j]
                K_red = max(2, n // 20)
                top_idx = np.argsort(contribs)[::-1][:K_red]
                a_new[top_idx] *= np.exp(-rng.uniform(0.02, 0.06, K_red))
                # Compensate
                a_new *= S / max(a_new.sum(), 1e-10)

        elif r < 0.85:
            # Gradient direction with noise
            w = np.zeros(len(c))
            top_idx = np.argsort(c)[::-1][:max(2, K_red if 'K_red' in dir() else 5)]
            for m in top_idx:
                w[m] = 1.0 / len(top_idx)
            g = 2.0 * grad_conv(a, w)
            g = g - g.mean()
            noise_scale = rng.uniform(0.005, 0.02)
            a_new = a - noise_scale * g
            a_new = a_new + noise_scale * 0.5 * rng.standard_normal(n)
            a_new = a_new - a_new.mean() + a.mean()

        else:
            # Block reshuffling: move mass from high-autocorrelation blocks to low
            n_blocks = max(2, n // 32)
            block_size = n // n_blocks
            block_energies = np.zeros(n_blocks)
            for b in range(n_blocks):
                block_energies[b] = a[b * block_size:(b + 1) * block_size].sum()
            max_b = int(np.argmax(block_energies))
            min_b = int(np.argmin(block_energies))
            amt = 0.03 * block_energies[max_b] * rng.uniform(0.5, 1.5)
            a_new[max_b * block_size:(max_b + 1) * block_size] -= amt / block_size
            a_new[min_b * block_size:(min_b + 1) * block_size] += amt / block_size

        a_new = np.maximum(a_new, 0.0)
        a_new = a_new * S / max(a_new.sum(), 1e-10)

        new_score = score_fast(a_new)
        cur_score = score_fast(a)

        accept = new_score < cur_score or (
            rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12))
        )
        if accept:
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

        temp *= cooling
        iter_count += 1

        if iter_count % 20000 == 0:
            print(f"      shave iter={iter_count}, score={best_score:.6f}, temp={temp:.5f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def make_init(n, rng, style):
    """Generate diverse initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if style == 'chirp':
        alpha = rng.uniform(0.3, 0.7)
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.5))
    elif style == 'sawtooth':
        a = 1.0 + 0.3 * np.sin(2 * np.pi * t / n) + 0.15 * np.sin(4 * np.pi * t / n + rng.uniform(0, 2))
    elif style == 'multi_tone':
        a = np.ones(n)
        for _ in range(rng.integers(3, 6)):
            freq = rng.uniform(0.1, 3.0)
            amp = rng.uniform(0.1, 0.3)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    elif style == 'bump':
        centers = rng.uniform(n * 0.1, n * 0.9, size=rng.integers(2, 5))
        widths = rng.uniform(n * 0.05, n * 0.2, size=len(centers))
        a = np.ones(n) * 0.5
        for c, w in zip(centers, widths):
            a += np.exp(-0.5 * ((t - c) / w) ** 2)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

    # === Phase 1: Softmax gradient (50% budget) ===

    phase1_time = total_budget * 0.50
