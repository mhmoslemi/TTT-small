def lp_solve(f, M, n_max=2048):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M (elementwise), g >= 0."""
    n = len(f)
    if n > n_max:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix: A[m,j] = f[m-j] if 0<=m-j<n else 0
    # A[m, j] = f[m-j] when 0 <= m-j < n, i.e., m-n+1 <= j <= m
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated smoothed gradient descent (log-sum-exp annealing) combined
with LP-based alternating refinement. Exploits scale invariance (fixes sum=1).
Multiple diverse starting points and sequence lengths are explored.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def compute_conv(a):
    """Compute full linear convolution using FFT for O(n log n) speed."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize *= 2
    fa = fft(a, fsize)
    conv = np.real(ifft(fa * fa, fsize))[:size]
    return conv


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    c = compute_conv(a)
    max_c = float(c.max())
    return float(2.0 * n * max_c / (S * S))


def normalize_sum(a):
    """Normalize sequence to have sum exactly 1."""
    S = a.sum()
    if S < 1e-15:
        return np.ones_like(a) * (1.0 / len(a))
    return a / S


def grad_smoothed(a, tau):
    """Gradient of smooth max (log-sum-exp) of conv(a,a)."""
    n = len(a)
    c = compute_conv(a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        m_star = int(c.argmax())
        w = np.zeros_like(c)
        w[m_star] = 1.0
    else:
        w /= w_sum

    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[j+m]
    # This is cross-correlation: r[j] = sum_m a[m]*w[j+m]
    # Using FFT: r = ifft(fft(w, N) * conj(fft(a, N)))[:n]
    max_len = n + (2 * n - 1)
    N = 1
    while N < max_len:
        N *= 2
    r = np.real(ifft(fft(w, N) * np.conj(fft(a, N)), N))[:n]
    return 2.0 * r


def grad_hard(a):
    """Subgradient of max(conv(a,a)) at the single peak."""
    n = len(a)
    c = compute_conv(a)
    m_star = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m_star - j]
    return grad


def grad_multi_peak(a, k=5):
    """Subgradient using convex combination of top-k peaks (more robust)."""
    n = len(a)
    c = compute_conv(a)
    k = min(k, len(c))
    top_indices = np.argsort(c)[-k:]
    top_vals = c[top_indices]
    # Soft weights over top peaks
    w_peaks = top_vals - top_vals.min()
    if w_peaks.sum() < 1e-15:
        w_peaks = np.ones(k) / k
    else:
        w_peaks /= w_peaks.sum()

    grad = np.zeros(n)
    for idx, m_star in enumerate(top_indices):
        lo = max(0, m_star - n + 1)
        hi = min(n - 1, m_star)
        for j in range(lo, hi + 1):
            grad[j] += w_peaks[idx] * 2.0 * a[m_star - j]
    return grad


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    n = len(f)
    if n > 2048:
        return None

    c_obj = -np.ones(n)

    # Build banded constraint matrix
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if hi >= lo:
            A[m, lo:hi + 1] = f[m - lo:m - hi + 1][::-1]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        starts.append(('cos5', n, 0.3 + 0.7 * (0.5 + 0.5 * np.cos(2 * np.pi * 5 * i / n))))
        # Chirp
        starts.append(('chirp', n, 0.3 + 0.7 * (0.5 + 0.5 * np.cos(np.pi * i * i / n))))
        # Sinc-like
        center = n / 2.0
        starts.append(('sinc', n, np.maximum(0.1, np.sinc((i - center) / (n * 0.3)))))
        # Periodic pulses
        starts.append(('pulses', n, (np.cos(2 * np.pi * 7 * i / n) > -0.3).astype(float)))

    # Sort: existing first (most promising), then by length
    starts.sort(key=lambda x: (0 if x[0] == 'existing' else 1, x[1]))

    n_starts = len(starts)
    print(f"Total candidates to try: {n_starts}, time budget: {total_budget - 15:.0f}s")
    print(f"Starting at t=0")

    # Time allocation: give more time to existing sequences and moderate lengths
    for idx, (name, n, a0) in enumerate(starts):
        time_used = time.monotonic() - t_start
        time_remaining = deadline - time.monotonic()
        if time_remaining < 30:
            print(f"  Time running out ({time_remaining:.0f}s left), stopping.")
            break

        # Allocate time: existing gets more, larger n gets more
        base_time = 120.0
        if name == 'existing':
            base_time = 180.0
        if n >= 512:
            base_time += 60.0
        # Scale down if not enough time for all remaining
        n_remaining = n_starts - idx
        if n_remaining > 1:
            base_time = min(base_time, max(30.0, (time_remaining - 30) / n_remaining))

        alloc = min(base_time, time_remaining - 20)
        if alloc < 20:
            break

        print(f"\n[{idx+1}/{n_starts}] {name}, n={n}, alloc={alloc:.0f}s "
              f"(score_so_far={best_score:.6f})")

        t0 = time.monotonic()
        # Scale iterations based on allocated time
        max_grad = int(alloc * 8)
        max_lp = int(alloc * 0.15)
        seq, score = optimize_sequence(a0, t0 + alloc,
                                       max_grad_iters=max_grad,
                                       max_lp_rounds=max_lp)
        elapsed = time.monotonic() - t0
        print(f"    Result: {score:.6f} ({elapsed:.1f}s)")

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"    *** NEW BEST: {best_score:.6f} (n={n}, {name}) ***")

    # Final refinement pass on the best sequence
    if best_seq is not None:
        time_remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: score={best_score:.6f}, "
              f"time_remaining={time_remaining:.0f}s")

        if time_remaining > 40:
            alloc = min(time_remaining - 10, 300.0)
            seq, score = optimize_sequence(best_seq, time.monotonic() + alloc,
                                           max_grad_iters=int(alloc * 10),
                                           max_lp_rounds=int(alloc * 0.2))
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  After refinement: {best_score:.6f}")

            # One more short polish
            time_remaining2 = deadline - time.monotonic()
            if time_remaining2 > 20:
                alloc2 = min(time_remaining2 - 5, 120.0)
                seq2, score2 = optimize_sequence(best_seq, time.monotonic() + alloc2,
                                                 max_grad_iters=int(alloc2 * 15),
                                                 max_lp_rounds=10)
                if score2 < best_score:
                    best_score = score2
                    best_seq = seq2.copy()
                    print(f"  After polish: {best_score:.6f}")

    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n{'='*60}")
        print(f"FINAL: n={len(best_seq)}, internal_score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        print(f"{'='*60}")
        return [float(x) for x in best_seq]

    # Fallback: uniform sequence
    print("FALLBACK: uniform 256")
    return [1.0] * 256
