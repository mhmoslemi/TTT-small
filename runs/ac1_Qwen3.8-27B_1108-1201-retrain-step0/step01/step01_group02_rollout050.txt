import numpy as np
import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def smoothed_gradient_descent(a, tau_start, tau_end, lr_start, lr_end, n_iters, use_symmetry=False):
    a = np.maximum(0, a).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    N = 1
    while N < 3 * n:
        N *= 2
    
    for it in range(n_iters):
        frac = it / max(1, n_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        conv = np.convolve(a, a)
        # Softmax weights
        conv_shifted = conv - conv.max()
        w = np.exp(conv_shifted / tau)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w = w / w_sum
        
        # Compute gradient via cross-correlation
        w_padded = np.zeros(N)
        w_padded[:len(w)] = w
        a_padded = np.zeros(N)
        a_padded[:n] = a
        
        W = np.fft.rfft(w_padded)
        A = np.fft.rfft(a_padded)
        grad_full = np.fft.irfft(W * np.conj(A), N)
        grad = 2.0 * grad_full[:n]
        
        # Take gradient step
        a_new = a - lr * grad
        if use_symmetry:
            a_new = 0.5 * (a_new + a_new[::-1])
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s
        
        score = score_sequence(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

"""
Smoothed gradient descent with log-sum-exp approximation to minimize peak auto-convolution.
Uses FFT-based gradient computation, adaptive learning rates, and multi-start strategy.
"""

import numpy as np
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    if n == 1:
        return np.ones(1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_hamming(n):
    if n == 1:
        return np.ones(1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    return 0.54 - 0.46 * np.cos(2 * np.pi * x)


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def smoothed_gd(a, tau_start=0.5, tau_end=0.01, lr_start=0.1, lr_end=0.001,
                 n_iters=5000, use_symmetry=True, verbose=False, time_limit=None):
    """Smoothed gradient descent to minimize peak autocorrelation."""
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    # FFT size
    N = 1
    while N < 3 * n:
        N *= 2
    
    start_time = time.monotonic()
    it = 0
    no_improve = 0
    
    for it in range(n_iters):
        if time_limit is not None and time.monotonic() - start_time > time_limit:
            break
            
        frac = it / max(1, n_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Compute convolution
        conv = np.convolve(a, a)
        
        # Softmax weights (shifted to avoid overflow)
        conv_shifted = conv - conv.max()
        exp_vals = np.exp(conv_shifted / tau)
        w = exp_vals / exp_vals.sum()
        
        # Compute gradient via FFT cross-correlation
        w_padded = np.zeros(N)
        w_padded[:len(w)] = w
        a_padded = np.zeros(N)
        a_padded[:n] = a
        
        W = np.fft.rfft(w_padded)
        A = np.fft.rfft(a_padded)
        grad_full = np.fft.irfft(W * np.conj(A), N)
        grad = 2.0 * grad_full[:n]
        
        # Gradient step
        a_new = a - lr * grad
        if use_symmetry:
            a_new = 0.5 * (a_new + a_new[::-1])
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            # Revert
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        score = score_sequence(a_new)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
            if verbose and (it % 1000 == 0 or it < 10):
                print(f"      iter {it}: score={score:.6f}, tau={tau:.4f}, lr={lr:.6f}")
        else:
            no_improve += 1
        
        a = a_new
        
        # If no improvement for too long, reduce learning rate further
        if no_improve > 500:
            lr *= 0.5  # This doesn't actually affect the schedule... let me handle differently
            if no_improve > 2000:
                break
    
    return best_a, best_score
