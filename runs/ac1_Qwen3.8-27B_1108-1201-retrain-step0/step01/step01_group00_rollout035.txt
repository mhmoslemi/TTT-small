```python
import cvxpy as cp
import numpy as np

def solve_lp(a, M):
    n = len(a)
    b = cp.Variable(n)
    # Build constraint matrix
    # C[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Constraint: C @ b <= M
    constraints = [b >= 0]
    for k in range(2*n - 1):
        # sum_j C[k,j] * b[j] <= M
        row = np.zeros(n)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                row[j] = a[idx]
        constraints.append(row @ b <= M)
    
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)  # or another solver
    return b.value
```

```python
from scipy.optimize import linprog

def solve_lp_scipy(a, M):
    n = len(a)
    # Build C matrix: C[k, j] = a[k-j] if 0 <= k-j < n
    # C has shape (2n-1, n)
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                C[k, j] = a[idx]
    
    # Minimize -sum(b) subject to C@b <= M, b >= 0
    c_obj = -np.ones(n)
    A_ub = C
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x
```

```python
def build_conv_matrix(a):
    """Build the matrix C such that C @ b = convolve(a, b) for b of same length as a."""
    n = len(a)
    # C[k, j] = a[k-j] if 0 <= k-j < n
    # This is a Toeplitz-like matrix
    # We can build it using scipy.linalg.toeplitz or manually
    C = np.zeros((2*n-1, n))
    for j in range(n):
        # C[k, j] = a[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        C[j:j+n, j] = a
    return C
```

```python
def build_conv_matrix(a):
    """Build matrix C such that C @ b = convolve(a, b) for len(b) == len(a)."""

def local_polish(a_init, time_limit=100, rng=None):
    """Simulated annealing local search."""
    if rng is None:
        rng = np.random.default_rng(999)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s > 0:
        a = a / s
    current_score = compute_score(a)
    best_a = a.copy()
    best_score = current_score
    n = len(a)
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9995

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.4:
            # Targeted: reduce peak
            c = convolve_fft(a, a)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_inc = min(10, len(others))
                inc_idx = rng.choice(others, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.8:
            # Symmetrize
            a_sym = 0.5 * (a + a[::-1])
            candidate = a_sym * np.exp(rng.normal(0, 0.01, size=n))
        else:
            # Global small
            candidate = a * np.exp(rng.normal(0, 0.015, size=n))

        candidate = np.clip(candidate, 0, 1000)
        s_c = np.sum(candidate)
        if s_c < 0.01:
            continue
        new_score = compute_score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score

        temperature *= cooling

    return best_a, best_score

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

            if time.monotonic() > deadline - 150:
                break
            t_budget_lp = min(300.0, (deadline - time.monotonic()) * 0.3)
            if t_budget_lp < 30:
                break
            if start_a is not None:
                a0 = resample_sequence(start_a, n_try)
            else:
                a0 = make_chirp_init(n_try, rng)
            a0 = a0 / np.sum(a0)
            print(f"  Trying n={n_try}, time={t_budget_lp:.0f}s")
            a_lp2, lp_score2 = lp_fixed_point(a0, time_limit=t_budget_lp, verbose=False, rng=rng)
            sc2 = compute_score(a_lp2)
            print(f"  n={n_try}: score={sc2:.6f}")
            if sc2 < best_score:
                best_score = sc2
                best_seq = a_lp2.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Phase 2: Gradient refinement
    print("\n=== Phase 2: Gradient refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        grad_time = min(300.0, max(30.0, (deadline - time.monotonic()) * 0.3))
        print(f"  time_limit={grad_time:.0f}s")
        a_grad, grad_score = gradient_refine(best_seq, time_limit=grad_time, rng=rng)
        if grad_score < best_score:
            best_score = grad_score
            best_seq = a_grad.copy()
            print(f"  *** Gradient best: {best_score:.6f} ***")
        else:
            print(f"  Gradient did not improve: {grad_score:.6f}")

    # Phase 3: Local polish
    print("\n=== Phase 3: Local polish ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        polish_time = min(150.0, max(20.0, (deadline - time.monotonic()) * 0.4))
        print(f"  time_limit={polish_time:.0f}s")
        a_pol, pol_score = local_polish(best_seq, time_limit=polish_time, rng=rng)
        if pol_score < best_score:
            best_score = pol_score
            best_seq = a_pol.copy()
            print(f"  *** Polish best: {best_score:.6f} ***")
        else:
            print(f"  Polish did not improve: {pol_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_a is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_a]
    return [1.0] * 256
```
