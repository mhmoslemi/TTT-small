def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def gs_initialize(n, n_iters=500, seed=0):

        S = a.sum()

def softmax_gradient_fft(a, tau):
    """Compute softmax-weighted gradient using FFT for efficiency."""
    n = len(a)
    c = np.convolve(a, a)
    
    # Softmax weights
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k,j
    # This is 2 * cross_correlation(w, a) at lag j
    # cross_correlation(w, a)[j] = sum_k w[k] * a[k-j]
    
    # Use FFT: pad to length >= len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    L = 1
    while L < 3 * n - 2:
        L *= 2
    
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    
    # Cross-correlation: ifft(fft(w) * conj(fft(a)))
    # This gives R[j] = sum_k w[k] * a[k-j] for j >= 0
    R = np.fft.irfft(W * np.conj(A), L)
    
    # Extract the first n elements (j = 0 to n-1)
    grad = 2.0 * R[:n]
    
    return grad, float(c_max)


def adam_update(a, grad, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step."""
    t = state['t'] + 1
    m = beta1 * state['m'] + (1 - beta1) * grad
    v = beta2 * state['v'] + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, state

        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return np.full_like(a, target / len(a))
    return a * (target / S)

            if mean_mag > 1e-14:
                target_mag = np.abs(A).copy()
                target_mag[1:] = (1.0 - blend) * non_dc_mag + blend * mean_mag
                target_mag[0] = np.abs(A[0])
                target_mag = np.maximum(target_mag, 1e-10)
                A_new = target_mag * np.exp(1j * np.angle(A))
                a_padded = np.fft.irfft(A_new, N_pad)
                a = np.real(a_padded[:n])

        a = np.maximum(a, 0.0)
        a = normalize(a)

    return a


def softmax_gradient_fft(a, tau):
    """
    Compute softmax-weighted gradient of max convolution using FFT.
    Returns (gradient, peak_value).
    """
    n = len(a)
    c = np.convolve(a, a)

    # Softmax weights
    c_max = float(c.max())
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum

    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid indices
    # This is 2 * cross_correlation(w, a)[j]
    # Use FFT: need L >= len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    L = 1
    while L < 3 * n - 2:
        L *= 2

    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)

    # Cross-correlation: ifft(fft(w) * conj(fft(a)))
    R = np.fft.irfft(W * np.conj(A), L)

    # Extract first n elements
    grad = 2.0 * R[:n]

    return grad, c_max


def hard_max_gradient(a):
    """Compute gradient at the argmax of the convolution."""
    n = len(a)
    c = np.convolve(a, a)
    k = int(np.argmax(c))

    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[k - j]

    # Handle the diagonal case (j == k-j)
    if k % 2 == 0:
        j_diag = k // 2
        if 0 <= j_diag < n:
            grad[j_diag] = a[j_diag]  # Only one contribution, not two

    return grad, float(c.max())


def adam_update(a, grad, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. Returns (new_a, updated_state)."""
    t = state['t'] + 1
    m = beta1 * state['m'] + (1.0 - beta1) * grad
    v = beta2 * state['v'] + (1.0 - beta2) * grad ** 2
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, state


def optimize_adam(a_init, deadline, max_iters=50000, tau_start=5.0, tau_end=100.0,
                  lr_start=0.3, lr_end=0.005, use_soft=True):
    """
    Optimize using Adam with softmax (or hard max) gradient.
    Includes periodic perturbation to escape local minima.
    """
    n = len(a_init)
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    a = normalize(a)

    best = a.copy()
    best_score = compute_obj(a)

    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    rng = np.random.default_rng(42 + n)

    stagnant = 0
    stagnant_threshold = 5000

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = min(1.0, it / max_iters)
        tau = tau_start + (tau_end - tau_start) * progress ** 0.7
        lr = lr_start + (lr_end - lr_start) * (1.0 - progress) ** 0.5

        # Compute gradient
        if use_soft:
            grad, c_max = softmax_gradient_fft(a, tau)
        else:
            grad, c_max = hard_max_gradient(a)

        # Project gradient (remove mean for fixed-sum constraint)
        grad = grad - grad.mean()

        # Adam step
        a_new, state = adam_update(a, grad, state, lr)

        # Project: non-negativity + normalization
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        a = a_new

        # Check improvement periodically
        if (it + 1) % 100 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-8:
                best_score = s
                best = a.copy()
                stagnant = 0
            else:
                stagnant += 100

            # Perturbation to escape local minima
            if stagnant > stagnant_threshold:
                perturbation = 0.05 * (1.0 + stagnant / stagnant_threshold)
                a = a + perturbation * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = normalize(a)
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                stagnant = 0
                stagnant_threshold = int(stagnant_threshold * 1.3)

    return best, best_score


def lp_solve(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    n = len(f)
    if n > 512:
        return None

    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a, deadline, max_iters=80):
    """LP-based refinement: iteratively find best direction g and blend."""
    n = len(a)
    if n > 512:
        return a, compute_obj(a)

    a = a.copy()
    best = a.copy()
    best_score = compute_obj(a)
    S_t = np.sqrt(2.0 * n)

    c = np.convolve(a, a)
    M = float(c.max()) * 1.001  # Slight slack

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = lp_solve(a, M)
        if g is None:
            break

        improved = False
        for t in [0.9, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            S = a_new.sum()
            if S > 1e-10:
                a_new = a_new * (S_t / S)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-9:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            # Try reducing M slightly and retrying
            M = M * 0.99
            c = np.convolve(a, a)
            M = max(M, float(c.max()) * 1.001)

        c = np.convolve(a, a)
        M = float(c.max()) * 1.001

    return best, best_score


def multi_tau_optimize(a_init, deadline, max_iters_per_tau=10000):
    """Optimize with multiple tau values, gradually increasing sharpness."""
    n = len(a_init)
    a = a_init.copy()
    best = a.copy()
    best_score = compute_obj(a)

    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    rng = np.random.default_rng(77 + n)

    tau_values = [3.0, 5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0]

    for tau in tau_values:
        if time.monotonic() > deadline:
            break

        # Reset optimizer state for each tau
        state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
        stagnant = 0

        for it in range(max_iters_per_tau):
            if time.monotonic() > deadline:
                break

            lr = 0.1 * (1.0 + np.cos(np.pi * it / max_iters_per_tau))
            lr = max(lr, 0.002)

            grad, c_max = softmax_gradient_fft(a, tau)
            grad = grad - grad.mean()

            a_new, state = adam_update(a, grad, state, lr)
            a_new = np.maximum(a_new, 0.0)
            a = normalize(a_new)

            if (it + 1) % 200 == 0:
                s = compute_obj(a)
                if s < best_score - 1e-9:
                    best_score = s
                    best = a.copy()
                    stagnant = 0
                else:
                    stagnant += 200

                if stagnant > 5000:
                    a = a + 0.03 * rng.standard_normal(n)
                    a = np.maximum(a, 0.0)
                    a = normalize(a)
                    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                    stagnant = 0

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function combining GS initialization, multi-tau Adam
    gradient descent, perturbation restarts, and LP refinement.
    Explores multiple sequence lengths and initialization strategies.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, score):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  *** NEW BEST: score={score:.6f}, n={len(a)} ***")

    # Try to load previous best
    prev_seq = None
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size >= 8 and np.all(np.isfinite(prev)):
            prev = np.clip(prev, 0, 1000)
            score = compute_obj(prev)
            print(f"Previous best: n={len(prev)}, score={score:.6f}")
            best_seq = prev.copy()
            best_score = score
            prev_seq = prev
    except (NameError, Exception):
        pass

    # Phase 1: GS initialization + Adam optimization for multiple lengths
    lengths_to_try = [256, 512, 768, 1024]

    # Allocate time: ~60% for main optimization, ~25% for refinement, ~15% reserve
    main_deadline = t_start + float(budget_s) * 0.60
    refine_deadline = t_start + float(budget_s) * 0.85

    print(f"\n=== Phase 1: GS + Adam optimization ===")
    for n in lengths_to_try:
        if time.monotonic() > main_deadline:
            break

        # Check if we have a previous sequence of this length to use
        starts = []

        # GS initialization
        for gs_seed in [seed, seed + 1, seed + 2]:
            starts.append(gs_initialize(n, n_iters=400, seed=gs_seed))

        # Soft GS
        starts.append(gs_initialize_soft(n, n_iters=400, seed=seed + 5, alpha=0.6))

        # Previous best (resampled)
        if prev_seq is not None:
            if len(prev_seq) == n:
                starts.append(prev_seq.copy())
            elif len(prev_seq) > n:
                starts.append(prev_seq[:n].copy())
            else:
                # Pad by tiling
                tiled = np.tile(prev_seq, n // len(prev_seq) + 1)[:n]
                starts.append(tiled)

        # Chirp / structured
        i = np.arange(n, dtype=np.float64)
        starts.append(np.maximum(0.5 + 0.45 * np.cos(2 * np.pi * i * 3.0 / n), 0.01))
        starts.append(np.maximum(0.3 + 0.7 * np.abs(np.sin(np.pi * i / n)), 0.01))

        # Random
        rng = np.random.default_rng(seed + n * 13)
        starts.append(rng.random(n) + 0.1)

        print(f"\n  n={n}: trying {len(starts)} initializations")

        for idx, a0 in enumerate(starts):
            if time.monotonic() > main_deadline - 30:
                break

            a0 = np.clip(a0, 0, 1000)
            a0 = normalize(a0)
            init_score = compute_obj(a0)
            print(f"    Start {idx}: init_score={init_score:.6f}", end="")

            # Multi-tau Adam optimization
            a_opt, score = multi_tau_optimize(a0, main_deadline - 10,
                                              max_iters_per_tau=8000)
            print(f" → opt_score={score:.6f}")

            update_best(a_opt, score)

            # Also try a quick hard-max refinement on the result
            if score < best_score + 0.05 and time.monotonic() < main_deadline - 20:
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                a = a_opt.copy()
                for hmax_it in range(2000):
                    if time.monotonic() > main_deadline - 5:
                        break
                    grad, c_max = hard_max_gradient(a)
                    grad = grad - grad.mean()
                    lr = 0.05 * (1.0 + np.cos(np.pi * hmax_it / 2000))
                    lr = max(lr, 0.001)
                    a_new, state = adam_update(a, grad, state, lr)
                    a = normalize(np.maximum(a_new, 0.0))
                    if (hmax_it + 1) % 200 == 0:
                        s = compute_obj(a)
                        if s < best_score:
                            update_best(a, s)

    # Phase 2: LP refinement on the best sequence
    print(f"\n=== Phase 2: LP refinement (best score={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < refine_deadline:
        n = len(best_seq)
        if n <= 512:
            a_lp, lp_score = lp_optimize(best_seq, refine_deadline, max_iters=60)
            update_best(a_lp, lp_score)
        else:
            # For larger n, truncate to 512 for LP, then use as guide
            print(f"  n={n} too large for LP, skipping.")

    # Phase 3: Final refinement - restart with best and push harder
    print(f"\n=== Phase 3: Final refinement ===")
    final_deadline = deadline
    if best_seq is not None and time.monotonic() < final_deadline:
        n = len(best_seq)
        # Do one more round of multi-tau with more iterations
        a_final, final_score = multi_tau_optimize(
            best_seq, final_deadline - 10, max_iters_per_tau=20000)
        update_best(a_final, final_score)

        # Quick LP if applicable
        if n <= 512 and time.monotonic() < final_deadline - 5:
            a_lp, lp_score = lp_optimize(best_seq, final_deadline - 2, max_iters=30)
            update_best(a_lp, lp_score)

    # Phase 4: Try a completely different approach - random search with evaluation
    print(f"\n=== Phase 4: Random perturbation search ===")
    if best_seq is not None and time.monotonic() < deadline:
        n = len(best_seq)
        rng = np.random.default_rng(999)
        for trial in range(50):
            if time.monotonic() > deadline - 5:
                break
            # Random perturbation of the best
            a_pert = best_seq + 0.1 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = normalize(a_pert)
            s = compute_obj(a_pert)
            if s < best_score:
                update_best(a_pert, s)

    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n{'='*50}")
        print(f"FINAL: n={len(best_seq)}, score={final_score:.6f}")
        print(f"{'='*50}")

        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
            if eval_score < final_score:
                # evaluate_sequence clamps to [0, 1000], so it might differ
                print(f"  (using evaluate_sequence value)")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

        # Ensure all values are valid
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform sequence of length 256.")
    return [1.0] * 256
