"""
LP-driven search for low peak-to-average ratio of self-convolution.
Core idea: iteratively solve the LP best-response (max sum(g) s.t. p*g <= M, g>=0),
blend with current iterate, and perturb to escape local optima.
Combines with smoothed-max gradient steps for fine-tuning.
"""

import time
import numpy as np

try:
    from scipy.fft import fft, ifft, next_fast_len
except Exception:
    from numpy.fft import fft, ifft
    def next_fast_len(n):
        n = int(max(1, n))
        m = 1
        while m < n:
            m <<= 1
        return m

try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _HAVE_LP = True
except Exception:
    _HAVE_LP = False

try:
    from scipy.optimize import minimize, differential_evolution
    _HAVE_MINIMIZE = True
except Exception:
    _HAVE_MINIMIZE = False


def _NHF(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def _NHFg(n):
    return int(next_fast_len(max(1, 3 * int(n) - 2)))


def conv_linear(p):
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _NHF(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def score_norm(p):
    """Score for normalized non-negative p (sum=1)."""
    n = int(p.size)
    if n == 0:
        return np.inf
    c = conv_linear(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax)


def normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-15 or p.size == 0:
        return np.full(max(1, int(p.size)), 1.0 / max(1, int(p.size)), dtype=np.float64)
    return p / s


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (css - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (css[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-30:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def grad_smoothed_max(p, T_frac=0.05):
    """Gradient of smoothed max of p*p with temperature T_frac * cmax."""
    n = int(p.size)
    if n == 0:
        return np.empty(0)
    c = conv_linear(p)
    cmax = float(np.max(c))
    if cmax <= 0 or not np.isfinite(cmax):
        return np.zeros(n)
    T = max(T_frac * cmax, 1e-10)
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    L = _NHFg(n)
    Fp = fft(p, L)
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2:n - 2:-1]
    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    return 2.0 * np.maximum(G, 0.0)


def solve_lp(p, M):
    """Solve max sum(g) s.t. p*g <= M (element-wise), g >= 0. Returns normalized g or None."""
    n = int(p.size)
    if n == 0 or not _HAVE_LP:
        return None
    # Build sparse constraint matrix: (p*g)_k = sum_{i+j=k} p_i * g_j
    mid = np.arange(n, dtype=np.int32)
    rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel().astype(np.int32)
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    data = np.tile(p.astype(np.float64), n)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=(0, None), method='highs',
                      options={'disp': False, 'time_limit': 30})
    except Exception:
        return None
    if res is None or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-12:
        return None
    return g / s


def lp_blend_step(p, rng, max_time=2.0):
    """One LP iteration: solve best response, blend, return best (p, score)."""
    t0 = time.monotonic()
    n = int(p.size)
    c = conv_linear(p)
    M = float(np.max(c))
    if M <= 0 or not np.isfinite(M):
        return p, score_norm(p)
    g = solve_lp(p, M)
    if g is None:
        return p, score_norm(p)
    best_p = p.copy()
    best_sc = score_norm(p)
    for t in (1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05, 0.02):
        q = project_simplex((1.0 - t) * p + t * g)
        sc = score_norm(q)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_p = q.copy()
        if time.monotonic() - t0 > max_time:
            break
    return best_p, best_sc


def gradient_step(p, rng, step_fracs=None):
    """One gradient descent step on the smoothed max."""
    n = int(p.size)
    g = grad_smoothed_max(p, T_frac=0.03)
    if not np.all(np.isfinite(g)):
        return p, score_norm(p)
    gabs = float(np.max(np.abs(g))) if g.size else 0.0
    if gabs < 1e-15:
        return p, score_norm(p)
    if step_fracs is None:
        step_fracs = (0.3, 0.1, 0.03, 0.01, 0.003)
    base = max(1.0 / n, 1e-6)
    best_p = p.copy()
    best_sc = score_norm(p)
    for frac in step_fracs:
        lr = frac * base / gabs
        v = p - lr * g
        q = project_simplex(v)
        sc = score_norm(q)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_p = q.copy()
    return best_p, best_sc


def random_perturb(p, rng, strength=0.05):
    """Add random perturbation and project."""
    n = int(p.size)
    noise = rng.normal(0, strength / np.sqrt(n), n)
    return project_simplex(p + noise)


def make_shape(p, n, rng, kind, params=None):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'uniform':
        p = np.ones(n)
    elif kind == 'geometric':
        s = params if params is not None else 1.0
        p = np.exp(-s * x)
    elif kind == 'power':
        a = params if params is not None else 1.0
        p = np.power(np.maximum(x, 1e-8), a)
    elif kind == 'cup':
        a = params if params is not None else 1.0
        p = 1.0 + a * (2 * x - 1) ** 2
    elif kind == 'cap':
        a = params if params is not None else 0.5
        p = np.maximum(1.0 - a * (2 * x - 1) ** 2, 0.01)
    elif kind == 'beta':
        al = params[0] if params else 1.0
        be = params[1] if params else 1.0
        p = np.power(np.maximum(x, 1e-6), al) * np.power(np.maximum(1 - x, 1e-6), be)
    elif kind == 'gauss_center':
        a = params if params is not None else 5.0
        p = np.exp(-a * (x - 0.5) ** 2)
    elif kind == 'gauss_edge':
        a = params if params is not None else 10.0
        p = np.maximum(np.exp(-a * x ** 2), 1e-6) + np.maximum(np.exp(-a * (1 - x) ** 2), 1e-6)
    elif kind == 'sine':
        f = params if params is not None else 2.0
        p = np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * f * x), 0.01)
    elif kind == 'two_peak':
        m1, m2, w = (params[0], params[1], params[2]) if params else (0.25, 0.75, 3.0)
        p = np.exp(-w * (x - m1) ** 2) + np.exp(-w * (x - m2) ** 2)
    elif kind == 'random_smooth':
        f1 = int(rng.integers(1, max(2, n // 50)))
        f2 = int(rng.integers(1, max(2, n // 25)))
        ph1 = float(rng.uniform(0, 2 * np.pi))
        ph2 = float(rng.uniform(0, 2 * np.pi))
        p = 1.0 + 0.4 * np.sin(2 * np.pi * f1 * x + ph1) + 0.25 * np.sin(2 * np.pi * f2 * x + ph2)
        p = np.maximum(p, 0.01)
    else:
        p = np.ones(n)
    return normalize(np.maximum(p, 0.0))

    grad_count = 0
    pert_count = 0

        do_lp = use_lp and (iter_no % max(1, lp_interval) == 0 or since_improve > 50)
        do_grad = not do_lp

            p_new, sc_new = lp_blend_step(p, rng, max_time=min(3.0, max(0.5, (deadline - time.monotonic()) * 0.3)))
            lp_count += 1

            grad_count += 1
        else:
            p_new, sc_new = gradient_step(p, rng)
            grad_count += 1

        if np.isfinite(sc_new):
            if sc_new < best_sc - 1e-12:
                best_sc = sc_new
                best_p = p_new.copy()
                since_improve = 0
            elif sc_new < cur_sc - 1e-14:
                since_improve += 1
            else:
                since_improve += 1
            cur_sc = sc_new
            p = p_new
        else:
            since_improve += 1

                    break
            since_improve = 0

        if iter_no2 % 1000 == 0:
            print("  [%s] phase2 iter=%d best=%.8f lp=%d rem=%.0fs" % (
                label, iter_no2, best_sc, lp_count, phase2_deadline - time.monotonic()), flush=True)

    # Phase 3: Final polish with very small steps
    if time.monotonic() < deadline - 1.0:
        p = best_p.copy()
        for _ in range(50):
            if time.monotonic() > deadline - 0.5:
                break
            if use_lp:
                p_new, sc_new = lp_blend_step(p, rng, max_time=1.0)
            else:
                g = grad_smoothed_max(p, T_frac=0.002)
                gabs = float(np.max(np.abs(g))) if g.size else 0.0
                if gabs < 1e-15:
                    break
                lr = 0.1 * max(1.0 / n, 1e-6) / gabs
                p_new = project_simplex(p - lr * g)
                sc_new = score_norm(p_new)
            if np.isfinite(sc_new) and sc_new < best_sc:
                best_sc = sc_new
                best_p = p_new.copy()
                p = p_new

    return best_p, best_sc


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative step function with low evaluate_sequence score."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)

    # Get parent if available
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and float(parent_arr.sum()) > 0:
            parent = parent_arr
            print("Using parent sequence of length %d" % parent_arr.size, flush=True)
    except Exception:
        pass

    best_p_global = None
    best_sc_global = np.inf

    def record(p, sc):
        nonlocal best_p_global, best_sc_global
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0 or not np.all(np.isfinite(p)):
            return
        p = normalize(p)
        if not np.isfinite(sc):
            sc = score_norm(p)
        if np.isfinite(sc) and sc < best_sc_global:
            best_sc_global = float(sc)
            best_p_global = p.copy()
            print("NEW BEST: %.8f (n=%d)" % (sc, p.size), flush=True)

    # Try multiple sequence lengths
    lengths = []
    if parent is not None:
        lengths.append(int(parent.size))
    for n_try in [500, 700, 1000, 1500, 2000, 3000, 4096, 6000, 8192]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))

    # Allocate time: spend more on mid-sized sequences where LP is effective
    total_budget = max(60.0, deadline - time.monotonic())
    # Weight: prefer n in [500, 4096]
    weights = []
    for n in lengths:
        if 400 <= n <= 3000:
            w = 3.0
        elif 3000 < n <= 5000:
            w = 2.0
        elif n > 5000:
            w = 1.0
        else:
            w = 1.5
        weights.append(w)
    total_w = sum(weights)
    time_alloc = [total_budget * w / total_w for w in weights]

    # If parent exists, give it extra time
    if parent is not None:
        pi = lengths.index(int(parent.size))
        time_alloc[pi] *= 1.5
        total_w2 = sum(time_alloc)
        time_alloc = [t * total_budget / total_w2 for t in time_alloc]

    # Quick shape search for best initial points
    shape_candidates = {}
    for i, n in enumerate(lengths):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        if time_alloc[i] < 15:
            continue
        # Quick parametric search (limited time)
        shape_time = min(30.0, time_alloc[i] * 0.15, remaining * 0.1)
        if shape_time > 3:
            sp, ss = shape_param_search(n, rng, shape_time)
            if sp is not None:
                shape_candidates[n] = (sp, ss)
                if ss < best_sc_global:
                    record(sp, ss)

    # Main optimization for each length
    for i, n in enumerate(lengths):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        t_alloc = min(time_alloc[i], remaining * 0.85)
        if t_alloc < 15:
            continue

        print("Optimizing n=%d with %.0f seconds" % (n, t_alloc), flush=True)

        # Generate initial points
        parent_n = None
        if parent is not None and int(parent.size) != n:
            # Resample parent to this length
            from numpy import interp
            xp = np.linspace(0, 1, parent.size)
            xp_new = np.linspace(0, 1, n)
            parent_n = normalize(interp(xp_new, xp, parent))
        elif parent is not None:
            parent_n = normalize(parent)

        inits = generate_initials(n, rng, parent_n, num_extra=6)

        # Add shape candidate if available
        if n in shape_candidates:
            sp, _ = shape_candidates[n]
            inits.append(sp)

        # Score and sort
        scored = []
        for p in inits:
            p = normalize(p)
            sc = score_norm(p)
            if np.isfinite(sc):
                scored.append((sc, p))
        scored.sort(key=lambda x: x[0])

        # Record best initial
        if scored:
            record(scored[0][1], scored[0][0])

        # Pick top initials to optimize
        top_k = min(4, len(scored))
        opt_time_each = max(5.0, t_alloc / (top_k + 1))

        for k in range(top_k):
            if time.monotonic() > deadline - 20:
                break
            sc_k, p_k = scored[k]
            t_remember = max(5.0, min(opt_time_each, deadline - time.monotonic() - 10))
            if t_remember < 5:
                break
            # Use hybrid optimizer
            p_best, sc_best = optimize_hybrid(p_k, rng, t_remember, "n=%d/init%d" % (n, k))
            record(p_best, sc_best)

        # If we have a good solution, do additional LP polish on the best
        if best_p_global is not None and best_p_global.size == n:
            t_polish = max(5.0, min(30.0, deadline - time.monotonic() - 5))
            if t_polish > 5:
                p_best, sc_best = optimize_hybrid(best_p_global, rng, t_polish, "n=%d/polish" % n)
                record(p_best, sc_best)

    # Final: if we have time, do one more round on the global best
    if best_p_global is not None and time.monotonic() < deadline - 15:
        t_final = max(5.0, deadline - time.monotonic() - 3)
        p_best, sc_best = optimize_hybrid(best_p_global, rng, t_final, "FINAL")
        record(p_best, sc_best)

    if best_p_global is None:
        # Fallback: uniform sequence
        best_p_global = np.full(1000, 0.001, dtype=np.float64)
        best_sc_global = score_norm(best_p_global)

    # Final validation and formatting
    best_p_global = np.asarray(best_p_global, dtype=np.float64)
    best_p_global = np.where(np.isfinite(best_p_global), best_p_global, 0.0)
    best_p_global = np.maximum(best_p_global, 0.0)
    s = float(best_p_global.sum())
    if s <= 1e-12:
        best_p_global = np.full(1000, 0.001)
        s = 1.0
    best_p_global = np.minimum(best_p_global, 1000.0)

    elapsed = time.monotonic() - t_start
    print("DONE: best_score=%.8f, n=%d, elapsed=%.1fs" % (best_sc_global, best_p_global.size, elapsed), flush=True)

    return [float(x) for x in best_p_global]
