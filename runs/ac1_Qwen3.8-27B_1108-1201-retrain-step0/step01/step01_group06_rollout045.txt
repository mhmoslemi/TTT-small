def fast_conv(a, b=None):

        b = a

def compute_obj_smoothed(a, t):
    """Compute smoothed objective 2n*smooth_max(conv(a,a)) with soft parameter t."""
    n = len(a)
    conv = fast_conv(a)
    scaled = t * conv
    shifted = scaled - scaled.max()
    lse = scaled.max() + np.log(np.exp(shifted).sum())
    smooth_max = lse / t
    return 2.0 * n * smooth_max


def compute_grad_smoothed(a, t):
    """Compute gradient of smoothed objective w.r.t. a."""
    n = len(a)
    conv = fast_conv(a)
    scaled = t * conv
    shifted = scaled - scaled.max()

        bi /= bi.sum()
        candidates.append((bi, n, f"bimodal_n{n}"))

    # Random candidates
    for n in [256, 512, 1024]:
        r = rng.dirichlet(np.ones(n) * 0.3)
        candidates.append((r, n, f"random_n{n}"))

    # "Cosine" candidates
    for n in [256, 512, 1024]:
        x = np.arange(n, dtype=np.float64)
        c = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        c = np.maximum(c, 1e-10)
        c /= c.sum()
        candidates.append((c, n, f"cosine_n{n}"))

    print(f"Total candidates: {len(candidates)}")

    # Phase 1: Quick screen
    print("Phase 1: Quick screening...")
    phase1_deadline = min(time.monotonic() + 300, deadline - 600)
    time_per_candidate = 30.0

    screened = []
    for i, (a0, n, name) in enumerate(candidates):
        if time.monotonic() > phase1_deadline:
            break
        t_alloc = min(time_per_candidate, phase1_deadline - time.monotonic() - 5)
        if t_alloc < 3:
            break

        a_opt, score = optimize_from_init(a0, time_limit=t_alloc,
                                           t_schedule=[10.0, 50.0, 200.0, 1000.0, 5000.0])
        # True evaluation
        conv = fast_conv(a_opt)
        true_score = 2.0 * n * conv.max()
        screened.append((true_score, n, name, a_opt))

        if true_score < best_score:
            best_score = true_score
            best_sequence = a_opt.copy()
            print(f"  [{name}] n={n}, score={true_score:.8f} (new best)")

    # Phase 2: Deep optimization on best candidates
    print(f"Phase 2: Deep optimization. Best so far: {best_score:.8f}")
    screened.sort(key=lambda x: x[0])

    # Take top candidates, ensuring diversity in n
    top_candidates = []
    seen_n = set()
    for score, n, name, a_opt in screened:
        if len(top_candidates) < 10:
            top_candidates.append((n, name, a_opt))
            seen_n.add(n)
        elif n not in seen_n and len(top_candidates) < 15:
            top_candidates.append((n, name, a_opt))
            seen_n.add(n)

    # Always include current best
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), 'current_best', best_sequence.copy()))

    remaining = deadline - time.monotonic() - 60
    time_per_deep = max(20.0, remaining / max(1, len(top_candidates)))

    for i, (n, name, a0) in enumerate(top_candidates):
        remaining = deadline - time.monotonic() - 30
        if remaining < 10:
            break
        t_alloc = min(time_per_deep, remaining)

        a_opt, score = optimize_with_restarts(a0, time_limit=t_alloc, n_restarts=2)
        conv = fast_conv(a_opt)
        true_score = 2.0 * n * conv.max()

        if true_score < best_score:
            best_score = true_score
            best_sequence = a_opt.copy()
            print(f"  Deep [{name}] n={n}, score={true_score:.8f} (new best)")

    # Phase 3: Final polish on the best
    if best_sequence is not None:
        print(f"Phase 3: Final polish. Best: {best_score:.8f}")
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            n = len(best_sequence)
            # Try very high temperature for final sharpening
            schedules_final = [
                [100.0, 500.0, 2000.0, 10000.0, 50000.0, 200000.0],
                [200.0, 1000.0, 5000.0, 25000.0, 100000.0, 500000.0],
            ]
            for sched in schedules_final:
                remaining = deadline - time.monotonic() - 10
                if remaining < 10:
                    break
                t_alloc = remaining * 0.8
                a_opt, score = optimize_from_init(best_sequence, time_limit=t_alloc,
                                                   t_schedule=sched)
                conv = fast_conv(a_opt)
                true_score = 2.0 * n * conv.max()
                if true_score < best_score:
                    best_score = true_score
                    best_sequence = a_opt.copy()
                    print(f"  Polish: score={true_score:.8f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = 2.0
        print("WARNING: Using fallback uniform sequence!")

    print(f"FINAL: n={len(best_sequence)}, score={best_score:.8f}")
    return best_sequence


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses L-BFGS-B with softmax parameterization and
    temperature annealing to minimize 2n*max(conv)/(sum)^2.
    Multi-phase search: quick screen, deep optimization, final polish.
    """
    result = run_full_search(seed=seed, budget_s=budget_s)
    # Ensure all values are in valid range
    result = np.clip(result, 0.0, 1000.0)
    return [float(x) for x in result]
