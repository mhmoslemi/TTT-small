"""
Optimized step function search using gradient-based optimization with FFT
cross-correlation gradients, LP-based peak reduction refinement, and
targeted local search. Novel approach combines softmax parameterization
with linear programming for exact peak direction finding.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return ifft(fft(a, size) * fft(b, size), size)[:n + m - 1].real


def local_score(a):
    """Compute evaluation score for a non-negative sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def softmax(theta):
    """Numerically stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def compute_gradient(a, tau):
    """
    Compute gradient of soft_max(conv(a,a), tau) w.r.t. a.
    Returns (soft_value, grad) where grad[i] = d(soft_max)/d(a_i).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    soft_val = c_max + tau * np.log(np.sum(exp_shifted))
    
    # grad[i] = sum_k w[k] * 2 * a[k-i] * I(0<=k-i<n)
    # = 2 * cross_corr(w, a)[i]
    # where cross_corr(w, a)[i] = sum_m w[m+i] * a[m]
    
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    W = fft(w, L)
    A = fft(a, L)
    # cross_corr(w, a)[i] = ifft(W * conj(A))[i]
    grad = 2.0 * ifft(W * np.conj(A), L)[:n].real
    
    return soft_val, grad, c


def grad_optimize(n, theta_init, time_limit, rng):
    """
    Adam gradient descent on theta to minimize soft_max(conv(softmax(theta), ...)).
    """
    theta = theta_init.astype(np.float64).copy()
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    step = 0
    best_theta = theta.copy()
    S = np.sqrt(2.0 * n)
    best_score = local_score(softmax(theta) * S)
    
    tau_start = 2.0 / n
    tau_end = 0.05 / n
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** min(1.0, elapsed * 1.2)
        
        a = softmax(theta)
        soft_val, grad_a, c = compute_gradient(a, tau)
        
        # Convert gradient w.r.t. a to gradient w.r.t. theta
        # d a_i / d theta_j = a_i * (delta_ij - a_j)
        # grad_theta_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
        # = grad_a_j * a_j - a_j * sum_i grad_a_i * a_i
        inner = np.dot(grad_a, a)
        grad_theta = a * (grad_a - inner)
        
        step += 1
        m1 = b1 * m1 + (1 - b1) * grad_theta
        m2 = b2 * m2 + (1 - b2) * (grad_theta * grad_theta)
        m1_hat = m1 / (1 - b1 ** step)
        m2_hat = m2 / (1 - b2 ** step)
        lr = 1.0 / np.sqrt(step + 1)
        theta -= lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        if step % 200 == 0:
            score = local_score(softmax(theta) * S)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
    
    final_score = local_score(softmax(theta) * S)
    if final_score < best_score:
        best_score = final_score
        best_theta = theta.copy()
    
    return best_theta, best_score

        c_obj[-1] = 1.0

        # Solve

            break

            a_new = a + t_step * d
            a_new = np.clip(a_new, 0, None)
            S_new = np.sum(a_new)

                a = a_new
                c = c_new
                M = M_actual

"""
Multi-phase optimization for finding non-negative step functions with minimized
peak autocorrelation ratio. Combines FFT-accelerated gradient descent with
softmax parameterization, LP-based peak reduction (active-set linear programming),
and peak-aware simulated annealing. Novel use of convolution cross-correlation
for O(n log n) gradient computation and reduced-constraint LP for fast refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return ifft(fft(a, size) * fft(b, size), size)[:n + m - 1].real


def local_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def softmax(theta):
    """Numerically stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def compute_gradient(a, tau):
    """
    Compute soft-max value and gradient of soft_max(conv(a,a), tau) w.r.t. a.
    Uses FFT cross-correlation for O(n log n) computation.
    Returns (soft_value, grad_array, convolution).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    soft_val = c_max + tau * np.log(np.sum(exp_shifted))

    # grad[i] = 2 * sum_k w[k] * a[k-i] * I(0<=k-i<n)
    # = 2 * cross_correlation(w, a)[i]
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    W = fft(w, L)
    A = fft(a, L)
    grad = 2.0 * ifft(W * np.conj(A), L)[:n].real

    return soft_val, grad, c


def grad_optimize(n, theta_init, time_limit, rng):
    """
    Adam gradient descent on theta (softmax parameterization) to minimize
    soft_max(conv(a,a), tau) where a = softmax(theta).
    """
    theta = theta_init.astype(np.float64).copy()
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    step = 0
    S = np.sqrt(2.0 * n)
    best_theta = theta.copy()
    best_score = local_score(softmax(theta) * S)
    tau_start = 2.0 / n
    tau_end = 0.03 / n
    t0 = time.monotonic()
    plateau = 0
    plateau_best = best_score

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** min(1.0, elapsed_frac * 1.3)

        a = softmax(theta)
        soft_val, grad_a, c = compute_gradient(a, tau)

        # Chain rule: grad_theta_j = a_j * (grad_a_j - sum_i grad_a_i * a_i)
        inner = np.dot(grad_a, a)
        grad_theta = a * (grad_a - inner)

        step += 1
        m1 = b1 * m1 + (1 - b1) * grad_theta
        m2 = b2 * m2 + (1 - b2) * (grad_theta * grad_theta)
        m1_hat = m1 / (1 - b1 ** step)
        m2_hat = m2 / (1 - b2 ** step)
        lr = 1.0 / np.sqrt(step + 1)
        theta -= lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 200 == 0:
            score = local_score(softmax(theta) * S)
            if score < best_score - 1e-9:
                best_score = score
                best_theta = theta.copy()
                plateau = 0
            else:
                plateau += 1
                if plateau > 2000 and plateau % 500 == 0:
                    # Restart with perturbation
                    theta += np.log(softmax(theta)) + rng.normal(0, 0.05, n)
                    plateau = 0

    final_score = local_score(softmax(theta) * S)
    if final_score < best_score:
        best_score = final_score
        best_theta = theta.copy()

    return best_theta, best_score
