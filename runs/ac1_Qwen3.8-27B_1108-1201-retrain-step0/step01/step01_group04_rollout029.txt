def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u*idx > cssum - 1
    k = np.count_nonzero(cond) - 1
    if k < 0:
        return np.ones(n)/n
    theta = (cssum[k] - 1)/(k+1)

    cols = []

        for j in range(jmin, jmax+1):

    g = np.maximum(g, 0)

"""
Step-function search minimizing 2n*max(conv(p,p))/(sum p)^2 via aggressive
LP best-response iteration, subgradient peak reduction, and ideal-shape
initialization. Different from parent: uses exact subgradient of the true max
(not smoothed), much more frequent LP calls, "flat convolution" recurrence
initialization, and simulated-annealing acceptance.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _fft_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def conv_sq(p):
    n = len(p)
    if n == 0:
        return np.empty(0)
    L = _fft_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def internal_score(p):
    n = len(p)
    if n == 0:
        return np.inf
    s = np.sum(p)
    if s < 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = np.max(c)
    return float(2.0 * n * cmax / (s * s))


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-12:
        return np.full(len(p), 1.0 / max(1, len(p)))
    return p / s


def build_lp_matrix(p):
    n = len(p)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        jmin = max(0, k - (n - 1))
        jmax = min(n - 1, k)
        for j in range(jmin, jmax + 1):
            rows.append(k)
            cols.append(j)
            data.append(p[k - j])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(p, M, time_limit=30.0):
    n = len(p)
    if n > 3000:
        return None
    A = build_lp_matrix(p)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    t0 = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'disp': False,
                      'time_limit': time_limit})
    except Exception:
        return None
    if res is None or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0)
    s = g.sum()
    if s < 1e-12:
        return None
    return g / s


def ideal_flat_init(n):
    """Generate sequence from flat-convolution recurrence."""
    p = np.zeros(n)
    p[0] = 1.0
    C = 1.0
    for k in range(1, n):
        val = C
        for j in range(1, k):
            if j < k - j:
                val -= 2.0 * p[j] * p[k - j]
            elif j == k - j:
                val -= p[j] * p[k - j]
        if k % 2 == 0:
            val -= p[k // 2] * p[k // 2]
        p[k] = val / (2.0 * p[0])
        if p[k] < 0:
            p[k] = 0.0
    p = np.maximum(p, 0)
    if p.sum() < 1e-12:
        return np.full(n, 1.0 / n)
    return p / p.sum()


def make_smooth_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    if mode == 0:
        p = np.exp(-2.0 * (x - 0.5) ** 2 / 0.15)
    elif mode == 1:
        p = np.exp(-3.0 * x) + np.exp(-3.0 * (1 - x))
    elif mode == 2:
        p = np.sin(np.pi * x) ** 2 + 0.1
    elif mode == 3:
        p = np.exp(-1.5 * x) * (1 + 0.5 * np.cos(4 * np.pi * x))
        p = np.maximum(p, 0.01)
    elif mode == 4:
        f1 = rng.integers(2, 8)
        f2 = rng.integers(3, 12)
        ph1 = rng.uniform(0, 2 * np.pi)
        ph2 = rng.uniform(0, 2 * np.pi)
        p = 1.0 + 0.4 * np.sin(2 * np.pi * f1 * x + ph1) + 0.2 * np.cos(2 * np.pi * f2 * x + ph2)
        p = np.maximum(p, 0.05)
    else:
        p = np.ones(n)
    return safe_normalize(p)


def subgradient_peak_step(p, c, lr):
    """Move in negative subgradient direction of max of convolution."""
    n = len(p)
    k_star = int(np.argmax(c))
    # Gradient of (p*p)[k_star] w.r.t. p[j] = 2*p[k_star - j] for valid j
    grad = np.zeros(n)
    for j in range(n):
        k2 = k_star - j
        if 0 <= k2 < n:
            grad[j] = 2.0 * p[k2]
    v = p - lr * grad
    return project_simplex(v)


def peak_directed_move(p, c, rng, strength):
    """Identify top peaks and reduce contributing positions."""
    n = len(p)
    c_sorted = np.sort(c)[::-1]
    n_peaks = max(1, min(5, n // 10))
    top_k = np.argsort(c)[::-1][:n_peaks]
    # Accumulate "pressure"
    pressure = np.zeros(n)
    for k in top_k:
        for j in range(n):
            k2 = int(k) - j
            if 0 <= k2 < n:
                pressure[j] += p[k2]
    # Reduce p where pressure is high, increase where low
    p_new = p.copy()
    reduce_amt = strength * p.sum() * 0.01 * rng.uniform(0.5, 1.5)
    mask_high = pressure > np.median(pressure)
    mask_low = ~mask_high
    total_high = p[mask_high].sum()
    if total_high > 1e-15:
        p_new[mask_high] -= reduce_amt * (p[mask_high] / total_high)
        p_new[mask_low] += reduce_amt * (p[mask_low] / max(p[mask_low].sum(), 1e-15))
    p_new = np.maximum(p_new, 0)
    return project_simplex(p_new)


def optimize_sequence(p0, rng, time_limit, label=""):
    """Main optimization loop: LP + subgradient + peak moves + annealing."""
    p = safe_normalize(p0.copy())
    n = len(p)
    c = conv_sq(p)
    best_p = p.copy()
    best_score = internal_score(p)
    current_score = best_score
    T_anneal = 0.02
    iter_no = 0
    end = time.monotonic() + time_limit
    lp_interval = max(3, n // 150)

    while time.monotonic() < end:
        iter_no += 1
        c = conv_sq(p)
        cmax = float(np.max(c))
        current_score = float(2.0 * n * cmax)

        if current_score < best_score:
            best_score = current_score
            best_p = p.copy()
            T_anneal = max(0.01, T_anneal * 0.98)

        remaining = end - time.monotonic()
        if remaining < 1.0:
            break

        # LP step
        if iter_no % lp_interval == 1 and n <= 2000:
            M = cmax
            g = lp_best_response(p, M, time_limit=min(15.0, remaining * 0.3))
            if g is not None:
                best_t = 0.5
                best_sc = np.inf
                for t in (1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1):
                    p_t = project_simplex((1 - t) * p + t * g)
                    sc_t = internal_score(p_t)
                    if sc_t < best_sc:
                        best_sc = sc_t
                        best_t = t
                if best_sc < current_score + T_anneal:
                    p = project_simplex((1 - best_t) * p + best_t * g)
                    c = conv_sq(p)
                    cmax = float(np.max(c))
                    current_score = float(2.0 * n * cmax)
                    if current_score < best_score:
                        best_score = current_score
                        best_p = p.copy()

        # Subgradient step on the peak
        k_star = int(np.argmax(c))
        lr_base = 0.05 / (n * cmax + 1e-15)
        accepted = False
        for lr_mult in (1.0, 0.5, 0.2, 0.1, 0.05):
            lr = lr_base * lr_mult
            p_new = subgradient_peak_step(p, c, lr)
            sc_new = internal_score(p_new)
            if sc_new < current_score + T_anneal * rng.standard_normal() * 0.1:
                if sc_new < current_score:
                    p = p_new
                    current_score = sc_new
                    if sc_new < best_score:
                        best_score = sc_new
                        best_p = p.copy()
                        T_anneal *= 0.95
                    accepted = True
                    break

        if not accepted:
            # Peak-directed move
            p_new = peak_directed_move(p, c, rng, 1.0)
            sc_new = internal_score(p_new)
            if sc_new < current_score + T_anneal:
                if sc_new < current_score:
                    p = p_new
                    current_score = sc_new
                    if sc_new < best_score:
                        best_score = sc_new
                        best_p = p.copy()
                        T_anneal *= 0.97

        # Random perturbation (annealing)
        if iter_no % 50 == 0:
            sigma = T_anneal / n
            p_new = project_simplex(p + rng.normal(0, sigma, n))
            sc_new = internal_score(p_new)
            if sc_new < current_score + T_anneal * 2:
                p = p_new
                current_score = sc_new

        # Cool down
        T_anneal = max(0.005, T_anneal * 0.999)

        if iter_no % 200 == 0:
            print(label + " iter=%d best=%.10f cur=%.10f T=%.5f remaining=%.0fs" % (
                iter_no, best_score, current_score, T_anneal, remaining), flush=True)

        if best_score < 1.5000:
            print(label + " EXCELLENT! score=%.10f at iter %d" % (best_score, iter_no), flush=True)

    return safe_normalize(best_p), best_score
