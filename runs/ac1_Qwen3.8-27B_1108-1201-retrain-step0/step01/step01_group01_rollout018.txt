def _score(a):
    """Compute 2n * max(convolve(a,a)) for unit-sum a."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _subgradient_phase(a, deadline, rng, M=5, eta_init=None):

    n = len(a)
    a = a.copy()
    S = np.sum(a)
    a /= S
    best = a.copy()
    best_score = _score(best)
    
    if eta_init is None:
        eta_init = 0.05 / np.sqrt(n)
    eta = eta_init
    no_improve = 0
    patience = 500
    
    while time.monotonic() < deadline:
        grad, b_max = _grad_peak(a, M)
        cur_score = 2.0 * n * b_max
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= patience:
            eta *= 0.8
            no_improve = 0
            if eta < 1e-12:
                break
        
        # Center gradient
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Backtracking line search

    return best, best_score

    """Simulated annealing with block perturbations."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    a /= S
    best = a.copy()
    best_score = _score(best)

            best_score = new_score
            a = a_pert

"""
Multi-phase optimization for minimizing normalized autocorrelation peak.
Combines subgradient descent, smooth Fourier-basis perturbations with
simulated-annealing-style acceptance, and adaptive polishing.
"""

import numpy as np
import time


def _score(a):
    """Compute 2n * max(convolve(a,a)) for unit-sum a."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_topk(a, k_idx):
    """Compute sum of gradients at given lags. grad[j] = 2*sum_k a[k-j] (valid)."""
    n = len(a)
    grad = np.zeros(n)
    for k in k_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            lo2 = k - hi
            hi2 = k - lo
            grad[lo:hi + 1] += 2.0 * a[lo2:hi2 + 1][::-1]
    return grad


def _subgradient_phase(a, deadline, rng, M=5, eta_init=None, patience=300):
    """Subgradient descent with adaptive step size and top-M peak selection."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    a /= S
    best = a.copy()
    best_score = _score(best)

    if eta_init is None:
        eta_init = 0.05 / np.sqrt(n)
    eta = eta_init
    no_improve = 0

    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            eta *= 0.8
            no_improve = 0
            if eta < 1e-12:
                break

        # Select top M lags with soft weighting
        if M >= len(b):
            k_idx = np.arange(len(b))
            weights = np.ones(len(b)) / len(b)
        else:
            k_idx = np.argpartition(b, -M)[-M:]
            b_sel = b[k_idx]
            weights = np.exp(-4.0 * (b_max - b_sel) / max(b_max, 1e-15))
            weights /= np.sum(weights)

        # Weighted gradient
        grad = np.zeros(n)
        for ki, w in zip(k_idx, weights):
            if w < 1e-10:
                continue
            k = int(ki)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                lo2 = k - hi
                hi2 = k - lo
                grad[lo:hi + 1] += 2.0 * w * a[lo2:hi2 + 1][::-1]

        # Center for sum constraint
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Backtracking
        step = eta
        a_new = a
        for _ in range(8):
            a_trial = a - step * grad / gn
            a_trial = _project_simplex(a_trial)
            sc = _score(a_trial)
            if sc <= cur_score:
                a_new = a_trial
                break
            step *= 0.5
        else:
            a_new = a - step * grad / gn
            a_new = _project_simplex(a_new)

        a = a_new
        S = np.sum(a)
        if S < 1e-10:
            a = best.copy()
            break

    return best, best_score


def _fourier_perturb(a, rng, n_modes=8, amplitude=0.1):
    """Perturb a by adding a smooth Fourier polynomial, then project."""
    n = len(a)
    x = np.arange(n, dtype=np.float64)
    pert = np.zeros(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        pert += amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    if np.max(np.abs(pert)) > 1e-15:
        pert /= np.max(np.abs(pert))
    a_new = a + amplitude * pert
    return _project_simplex(a_new)


def _block_perturb(a, rng, min_len=3, max_len=None):
    """Perturb a random contiguous block by scaling."""
    n = len(a)
    if max_len is None:
        max_len = max(min_len + 1, n // 10)
    blen = int(rng.integers(min_len, min(max_len, n - 1)))
    start = int(rng.integers(0, n - blen))
    a_new = a.copy()
    factors = 1.0 + rng.uniform(-0.25, 0.25, blen)
    a_new[start:start + blen] *= factors
    a_new = _project_simplex(a_new)
    return a_new
