def compute_soft_max_grad(a, tau):
    """Compute gradient of soft_max_tau(conv(a,a)) w.r.t. a. Returns (grad, conv)."""
    c = np.convolve(a, a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)

def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S} using water-filling."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    s = a.sum()
    if abs(s - S) < 1e-15:
        return a
    if s < S:
        return a + (S - s) / n
    # Water-filling: find threshold t such that sum(max(a - t, 0)) = S
    a_sorted = np.sort(a)[::-1]
    css = np.cumsum(a_sorted)
    cond = a_sorted > (css - S) / np.arange(1, n + 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    t = (css[rho] - S) / (rho + 1)
    return np.maximum(a - t, 0.0)
