def _compute_convolution(a):

def _compute_score(a):
    n = len(a)
    b = np.convolve(a, a)
    s = np.sum(a)
    return 2.0 * n * np.max(b) / (s*s)

def _lp_direction(a, M_lags, M_val):
    """
    Solve LP: maximize sum(g) s.t. (a*g)(k) <= M_val for k in M_lags, g >= 0.
    Returns the optimal g (unnormalized).

    n = len(a)
    # Build constraint matrix: (len(M_lags) x n)
    # Row k: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(M_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(row_idx)
            cols.append(j)
            vals.append(a[k - j])

def _solve_lp(a, top_lags, M, n):
    """Solve the LP for the direction."""
    m = len(top_lags)
    
    # Build constraint matrix A (m x n)
    # A[k_row, j] = a[k - j] if 0 <= k - j < n
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(top_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(row_idx)
            cols.append(j)
            vals.append(a[k - j])
    
    A = sp.coo_matrix((vals, (rows, cols)), shape=(m, n)).tocsr()
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(m, M)
    
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, method='highs',
                      options={'presolve': True})
    
    if res.success:
        return res.x
    return None

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        if staleness > 200:
            break

        # Get top lags
        if M_lags >= len(b):
            top_lags = np.arange(len(b))
        else:
            top_lags = np.argpartition(b, -M_lags)[-M_lags:]

        # Solve LP (try dual first for speed, fall back to primal)
        g = None
        if use_dual:
            g = _solve_lp_direction_dual(a, top_lags, M, n, timeout=5.0)
        if g is None:
            g = _solve_lp_direction(a, top_lags, M, n, timeout=10.0)

        if g is None:
            # Try with fewer lags
            M_lags_small = max(10, M_lags // 2)
            top_lags_small = np.argpartition(b, -M_lags_small)[-M_lags_small:]
            g = _solve_lp_direction(a, top_lags_small, M, n, timeout=10.0)
            if g is None:
                break

        # Line search
        t = _line_search(a, g, n)

        if t < 0.005:
            # Try with more lags for a better direction
            M_lags_more = min(len(b), M_lags * 2)
            top_lags_more = np.argpartition(b, -M_lags_more)[-M_lags_more:]
            g2 = _solve_lp_direction(a, top_lags_more, M, n, timeout=10.0)
            if g2 is not None:
                t2 = _line_search(a, g2, n)
                if t2 > t:
                    t = t2
                    g = g2
            if t < 0.005:
                break

        a = (1.0 - t) * a + t * g
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-10:
            a = best.copy()
            break
        a = a / s_a

    return best, best_score


def _subgradient_phase(a, deadline, n, M=30, eta_init=None):
    """Subgradient descent refinement."""
    if eta_init is None:
        eta_init = 0.03 / np.sqrt(n)

    best = a.copy()
    best_score = _score(a)
    eta = eta_init
    no_improve = 0

    while time.monotonic() < deadline:
        b = _conv(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            eta *= 0.6
            no_improve = 0
            if eta < 1e-12:
                break

        # Weighted gradient from top M lags
        if M >= len(b):
            k_idx = np.arange(len(b))
            wts = np.ones(len(k_idx)) / len(k_idx)
        else:
            k_idx = np.argpartition(b, -M)[-M:]
            b_sel = b[k_idx]
            wts = np.exp(-4.0 * (b_max - b_sel) / max(b_max, 1e-15))
            wts /= wts.sum()

        grad = np.zeros(n)
        for wi, ki in zip(wts, k_idx):
            if wi < 1e-9:
                continue
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += wi * 2.0 * a[k - idx]

        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        a = a - eta * grad / gn
        a = _project_simplex(a, 1.0)

    return best, best_score


def _make_two_level(n, eps):
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_smooth_anti(n, eps):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_four_level(n, eps):
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2 * q] = 1.0 + eps / 3.0
    a[2 * q:3 * q] = 1.0 - eps / 3.0
    a[3 * q:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _perturb_smooth(a, rng, scale=0.05):
    n = len(a)
    noise = np.zeros(n)
    n_freqs = min(5, n // 4)
    for _ in range(n_freqs):
        f = rng.integers(1, max(2, n // 6))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    mx = np.max(np.abs(noise))
    if mx > 1e-15:
        noise /= mx
    a_new = a + scale * noise
    return _project_simplex(a_new, 1.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    LP-guided optimization with subgradient refinement and diversification
    to minimize the normalized autocorrelation peak of a non-negative sequence.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(60.0, float(budget_s) - 20.0)
    deadline = time.monotonic() + total_budget
    t_start = time.monotonic()

    # Get starting sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)

    n = len(parent)
    a = np.clip(parent, 0.0, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    init_score = _score(a)
    print(f"Initial: n={n}, score={init_score:.8f}, eval={evaluate_sequence([float(x) for x in a]):.8f}")

    global_best = a.copy()
    global_best_score = init_score

    # Phase 1: LP optimization on parent (45% budget)
    p1_time = total_budget * 0.45
    p1_deadline = t_start + p1_time
    print(f"Phase 1: LP on parent ({p1_time:.0f}s)")
    a1, s1 = _lp_phase(a, p1_deadline, n, verbose=True)
    if s1 < global_best_score:
        global_best = a1.copy()
        global_best_score = s1
        print(f"  Phase 1: {s1:.8f} (improved)")
    else:
        print(f"  Phase 1: {s1:.8f} (no improvement)")

    # Phase 2: LP from anti-symmetric start (20% budget)
    remaining2 = min(total_budget * 0.20, deadline - time.monotonic())
    if remaining2 > 10:
        p2_deadline = time.monotonic() + remaining2
        print(f"Phase 2: LP from anti-symmetric ({remaining2:.0f}s)")
        a2 = _make_two_level(n, 0.7)
        a2b, s2 = _lp_phase(a2, p2_deadline, n, verbose=True)
        if s2 < global_best_score:
            global_best = a2b.copy()
            global_best_score = s2
            print(f"  Phase 2: {s2:.8f} (improved)")
        else:
            print(f"  Phase 2: {s2:.8f} (no improvement)")

    # Phase 3: LP from smooth anti-symmetric (10% budget)
    remaining3 = min(total_budget * 0.10, deadline - time.monotonic())
    if remaining3 > 10:
        p3_deadline = time.monotonic() + remaining3
        print(f"Phase 3: LP from smooth anti ({remaining3:.0f}s)")
        a3 = _make_smooth_anti(n, 0.8)
        a3b, s3 = _lp_phase(a3, p3_deadline, n, verbose=False)
        if s3 < global_best_score:
            global_best = a3b.copy()
            global_best_score = s3
            print(f"  Phase 3: {s3:.8f} (improved)")
        else:
            print(f"  Phase 3: {s3:.8f} (no improvement)")

    # Phase 4: Subgradient refinement of best (15% budget)
    remaining4 = min(total_budget * 0.15, deadline - time.monotonic())
    if remaining4 > 10:
        p4_deadline = time.monotonic() + remaining4
        print(f"Phase 4: Subgradient refinement ({remaining4:.0f}s)")
        a4, s4 = _subgradient_phase(global_best, p4_deadline, n, M=50,
                                    eta_init=0.015 / np.sqrt(n))
        if s4 < global_best_score:
            global_best = a4.copy()
            global_best_score = s4
            print(f"  Phase 4: {s4:.8f} (improved)")
        else:
            print(f"  Phase 4: {s4:.8f} (no improvement)")

    # Phase 5: Diversification (10% budget)
    remaining5 = deadline - time.monotonic()
    if remaining5 > 15:
        print(f"Phase 5: Diversification ({remaining5:.0f}s)")
        n_attempts = max(1, int(remaining5 / 60))
        per_attempt = remaining5 / n_attempts

        for d in range(n_attempts):
            if time.monotonic() >= deadline:
                break
            a_d = _perturb_smooth(global_best, rng, scale=0.08)
            d_deadline = min(time.monotonic() + per_attempt, deadline)
            # Do a short LP phase then subgradient
            lp_split = (d_deadline - time.monotonic()) * 0.5
            a_d2, _ = _lp_phase(a_d, time.monotonic() + lp_split, n, verbose=False)
            a_d3, s_d = _subgradient_phase(a_d2, d_deadline, n, M=30,
                                           eta_init=0.01 / np.sqrt(n))
            if s_d < global_best_score:
                global_best = a_d3.copy()
                global_best_score = s_d
                print(f"  Phase 5 iter {d+1}: {s_d:.8f} (improved)")

    # Final output
    result = [float(max(0.0, x)) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"\nFINAL: score(normalized)={global_best_score:.8f}, eval={final_score:.8f}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    return result
