"""
Step-function search minimizing 2*n*max(conv(p,p))/sum(p)^2.
Uses smoothed-max FFT gradient descent with adaptive temperature,
structured multi-shape global search, multi-scale refinement,
and LP best-response for medium-length sequences.

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-15:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def conv_full(p):
    n = int(np.asarray(p).size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    c = conv_full(p)
    cmax = float(np.max(c)) if c.size > 0 else 0.0
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def gradient_smoothed_max(p, T_frac):
    """Gradient of T*logsumexp(c/T) w.r.t. p, where c = conv(p,p)."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0), 0.0
    L = next_fast_len(3 * n - 2)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]
    cmax = float(np.max(c)) if c.size > 0 else 0.0
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.zeros(n), 0.0
    T = max(T_frac * cmax, 1e-30)
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    G = h[2 * n - 2:n - 2:-1]
    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    g = 2.0 * np.maximum(G, 0.0)
    surr = cmax + T * np.log(max(s, 1e-300))
    return g, surr


def make_uniform(n):
    return np.full(max(1, n), 1.0 / max(1, n), dtype=np.float64)


def make_linear_p(n, a, reverse=False):
    n = max(2, n)

    p = 1.0 + float(a) * x
    if reverse:

def make_quadratic_p(n, a, center=0.5, reverse=False):

def make_semicircle(n, power=0.5):
    n = max(2, n)
    x = np.linspace(-1.0, 1.0, n)
    p = np.power(np.maximum(1.0 - x * x, 0.0), float(power))

    x = np.linspace(0.0, np.pi, n)

def make_power_p(n, a, reverse=False):

def make_two_peak(n, mu1, w1, mu2, w2):
    n = max(2, n)
    x = np.linspace(0.0, 1.0, n)
    p = float(w1)

    p += float(w2) * np.exp(-((x - float(mu2)) ** 2) / 0.02)

def make_random_smooth(n, rng, nfreq=5):

def shape_param_to_p(theta, n):
    """Convert shape parameters to a probability vector using a flexible basis."""
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    z = 2.0 * x - 1.0
    logp = np.zeros(n, dtype=np.float64)
    T_km = np.ones(n)
    T_k = z.copy()
    nd = int(min(len(theta), 12))
    for d in range(nd):
        if d == 0:
            logp += theta[d] * T_k
        elif d == 1:
            T_kp = 2.0 * z * T_k - T_km
            logp += theta[d] * T_kp
            T_km, T_k = T_k, T_kp
        else:
            T_kp = 2.0 * z * T_k - T_km
            logp += theta[d] * T_kp
            T_km, T_k = T_k, T_kp
    if len(theta) > nd:
        logp += theta[nd] * np.log(x + 1e-5)
    if len(theta) > nd + 1:
        logp += theta[nd + 1] * np.log(1.0 - x + 1e-5)
    if not np.all(np.isfinite(logp)):
        return np.full(n, 1.0 / n)
    logp -= float(np.max(logp))
    p = np.exp(logp)
    return safe_normalize(p)


def lp_best_response(p, n, budget_s):
    """LP: max sum(g) s.t. conv(p,g)[k] <= M for all k, g >= 0."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n < 4 or n > 800:
        return p, np.inf
    if budget_s < 1.0:
        return p, np.inf
    c = conv_full(p)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0:
        return p, np.inf
    fs = float(p.sum())
    if fs <= 0:
        return p, np.inf
    f = p / fs
    end = time.monotonic() + max(0.5, budget_s)
    if end - time.monotonic() < 0.5:
        return p, np.inf
    try:
        rows = []
        cols = []
        data = []
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for i in range(lo, hi + 1):
                rows.append(k)
                cols.append(i)
                data.append(f[i])
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        res = linprog(obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'disp': False, 'time_limit': max(1.0, budget_s - 1.0)})
    except Exception:
        return p, np.inf
    if res is None or res.x is None:
        return p, np.inf
    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    gs = float(g.sum())
    if gs <= 1e-15:
        return p, np.inf
    g = g / gs
    best_p = p.copy()
    best_sc = np.inf
    for t in (1.0, 0.7, 0.5, 0.3, 0.15, 0.05):
        q = project_simplex((1.0 - t) * p + t * g)
        sc = internal_score(q)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_p = q
    return best_p, best_sc

def shape_search(n, rng, budget_s):
    """Low-dimensional shape parameter search."""
    if budget_s < 5.0:
        return None, np.inf
    dim = 14
    best_theta = np.zeros(dim)
    best_p = None
    best_score = np.inf
    end = time.monotonic() + budget_s

    starts = [np.zeros(dim)]
    for _ in range(4):
        starts.append(rng.uniform(-2.0, 2.0, dim))
    t = np.zeros(dim)
    t[0] = 0.0
    t[1] = 1.5
    t[2] = -0.5
    starts.append(t)
    t = np.zeros(dim)
    t[1] = 2.0
    t[3] = -1.0
    t[5] = 0.5
    starts.append(t)
    t = np.zeros(dim)
    t[0] = 0.0
    t[12] = 2.0
    t[13] = 2.0
    starts.append(t)

    def eval_shape(theta):
        p = shape_param_to_p(np.asarray(theta, dtype=np.float64), n)
        return internal_score(p)

    for x0 in starts:
        if time.monotonic() > end - 2:
            break
        sc = eval_shape(x0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_theta = x0.copy()
            best_p = shape_param_to_p(x0, n)

    bounds = [(-4.0, 4.0)] * 12 + [(-4.0, 6.0), (-4.0, 6.0)]
    if time.monotonic() < end - 60:
        try:
            from scipy.optimize import differential_evolution
            res = differential_evolution(
                eval_shape, bounds, maxiter=30, popsize=15,
                tol=0.3, seed=int(rng.integers(0, 2**31 - 1)),
                polish=False, init='latinhypercube')
            if res is not None and np.isfinite(res.fun):
                sc = eval_shape(res.x)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_theta = res.x.copy()
                    best_p = shape_param_to_p(res.x, n)
        except Exception:
            pass

    for x0 in starts + [best_theta]:
        if time.monotonic() > end - 3:
            break
        try:
            res = minimize(eval_shape, x0, method='Nelder-Mead',
                           options={'maxiter': 300, 'xatol': 1e-5, 'fatol': 1e-7})
            if res is not None and np.isfinite(res.fun):
                sc = eval_shape(res.x)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_theta = res.x.copy()
                    best_p = shape_param_to_p(res.x, n)
        except Exception:
            pass

    return best_p, best_score


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """Main search function: finds best nonnegative sequence minimizing the evaluation score."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s) - 15.0)

    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and parent_arr.sum() > 0:
            parent = safe_normalize(parent_arr)
            print("Parent loaded: n=%d, score=%.6f" % (parent_arr.size, internal_score(parent_arr)), flush=True)
    except Exception:
        pass

    best_p = None
    best_score = np.inf

    def record(p, sc):
        nonlocal best_p, best_score
        if p is None or p.size == 0:
            return
        p = safe_normalize(p)
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("NEW BEST: score=%.8f n=%d" % (best_score, p.size), flush=True)

    # Phase 1: Shape search at medium resolution
    shape_n = 512
    if deadline - time.monotonic() > 120:
        shape_budget = min(180.0, (deadline - time.monotonic()) * 0.2)
        shape_p, shape_sc = shape_search(shape_n, rng, shape_budget)
        record(shape_p, shape_sc)
        print("Shape search: score=%.6f n=%d" % (shape_sc, shape_n), flush=True)
    else:
        shape_p = None
        shape_sc = np.inf

    # Phase 2: Multi-scale local optimization
    n_scales = [256, 512, 1024, 2048, 4096]
    if parent is not None:
        n_parent = int(parent.size)
        if n_parent not in n_scales:
            n_scales.append(n_parent)
            n_scales = sorted(set(n_scales))

    # Extra initializations
    extra_inits = []
    if shape_p is not None:
        for ns in n_scales:
            if shape_p.size != ns:
                # Reshape
                x_old = np.linspace(0, 1, shape_p.size)
                x_new = np.linspace(0, 1, ns)
                interp = np.interp(x_new, x_old, shape_p)
                extra_inits.append((ns, safe_normalize(interp)))
            else:
                extra_inits.append((ns, safe_normalize(shape_p)))

    # Track best per scale
    scale_best = {}

    for ns in n_scales:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        inits = generate_diverse_initials(ns, rng, parent if parent is not None and parent.size == ns else None)
        # Add scaled shape
        for ens, ep in extra_inits:
            if ens == ns:
                inits.insert(0, safe_normalize(ep))

        # Score and sort
        scored = []
        for p in inits:
            p = safe_normalize(p)
            sc = internal_score(p)
            if np.isfinite(sc):
                scored.append((sc, p))
        scored.sort(key=lambda x: x[0])

        for sc, p in scored[:4]:
            record(p, sc)

        # Pick top candidates for local optimization
        n_opt = max(1, min(4, len(scored)))
        time_per = max(20.0, min(180.0, (remaining - 30) / float(n_opt * 1.5)))

        for i in range(n_opt):
            if time.monotonic() > deadline - 30:
                break
            sc_cand, p_cand = scored[i]
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            limit = max(10.0, min(time_per, rem))
            allow_lp = ns <= 700
            p_best, sc_best = optimize_local(p_cand, rng, limit, allow_lp, 700)
            record(p_best, sc_best)
            if ns not in scale_best or sc_best < scale_best[ns][1]:
                scale_best[ns] = (p_best, sc_best)

        print("Scale n=%d done, best so far: %.6f" % (ns, best_score), flush=True)

    # Phase 3: Use best from medium scale as seed for fine-tuning at larger scale
    if time.monotonic() < deadline - 80:
        # Find best scale result
        if scale_best:
            best_ns = min(scale_best.keys(), key=lambda k: scale_best[k][1])
            seed_p = scale_best[best_ns][0]
            # Try at a larger scale
            for target_n in [2048, 4096, 8192]:
                if time.monotonic() > deadline - 60:
                    break
                if target_n <= best_ns:
                    continue
                x_old = np.linspace(0, 1, seed_p.size)
                x_new = np.linspace(0, 1, target_n)
                interp = np.interp(x_new, x_old, seed_p)
                p_interp = safe_normalize(interp)
                rem = deadline - time.monotonic() - 30
                if rem < 20:
                    break
                limit = min(rem, 200.0)
                p_best, sc_best = optimize_local(p_interp, rng, limit, target_n <= 700, 700)
                record(p_best, sc_best)
                break

    # Phase 4: Final polish on the global best
    if best_p is not None and time.monotonic() < deadline - 10:
        rem = deadline - time.monotonic() - 5
        limit = max(5.0, min(rem, 120.0))
        p_best, sc_best = optimize_local(best_p, rng, limit, best_p.size <= 700, 700)
        record(p_best, sc_best)

    if best_p is None:
        best_p = make_uniform(1024)
        best_score = internal_score(best_p)

    # Final validation and formatting
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s <= 1e-12:
        best_p = make_uniform(max(1, int(best_p.size)))
    # Normalize to avoid overflow in evaluator (sum should be reasonable)
    if s > 1e6:
        best_p = best_p * (1000.0 / s)

    final_sc = internal_score(best_p)
    print("FINAL: score=%.8f n=%d" % (final_sc, best_p.size), flush=True)
    return [float(x) for x in best_p]
