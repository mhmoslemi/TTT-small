            a[i] += delta
            a[j] -= delta

            a /= a.sum()
    
    return best_a, best_obj

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent, projected gradient descent, and simulated annealing
local search. Tries multiple sequence lengths and diverse initializations.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_norm(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(conv, a, t, thresh_frac=0.001):
    """Compute smoothed subgradient of max(conv) using softmax with temperature t."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = t * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        idx = np.argmax(conv)
        w = np.zeros_like(conv)
        w[idx] = 1.0
    else:
        w /= w_sum
    
    sg = np.zeros(n)
    thresh = w.max() * thresh_frac
    active = np.where(w > thresh)[0]
    for k in active:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je+1] += w[k] * a[k-je:k-js+1][::-1]
    return sg


def mirror_descent(a0, time_limit, t_init=50.0, eta_init=0.3, verbose=False):
    """Multiplicative mirror descent on the simplex."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t = t_init
    eta = eta_init
    patience = 0
    max_pat = 50
    t0 = time.monotonic()
    n_iter = 0
    
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient(conv, a, t)
        
        # Center the subgradient
        sg_c = sg - sg.mean()
        
        # Multiplicative update
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        n_iter += 1
        
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

                eta = eta_init * (1.0 - frac * 0.5)
                t = t_init * (1.0 + frac * 2.0)
    
    return best_a, best_obj


def pgd_step(a, conv, t, lr):

    n = len(a)
    sg = compute_subgradient(conv, a, t)
    a_new = a - lr * sg
    a_new = project_simplex(a_new)
    return a_new

def anneal_search(a0, time_limit, rng, verbose=False):
    """Simulated annealing with random mass transfers."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    T = 0.05  # Initial temperature (in objective units)
    
    while time.monotonic() - t0 < time_limit:
        # Adaptive temperature
        frac = (time.monotonic() - t0) / time_limit
        T = 0.05 * (1.0 - frac) ** 2 + 1e-5
        
        # Choose perturbation
        # Mode 1: single mass transfer
        if rng.random() < 0.7:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            frac_move = rng.uniform(0.005, 0.05)
            delta = a[i] * frac_move
            if delta < 1e-8:
                continue
            # Try moving from i to j
            a[i] -= delta
            a[j] += delta
            if a[i] < 0:
                # Revert and try other direction
                a[i] += delta
                a[j] -= delta
                delta = a[j] * frac_move
                if delta < 1e-8:
                    continue
                a[j] -= delta
                a[i] += delta
        else:
            # Mode 2: smooth perturbation
            idx = int(rng.integers(n))
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            lo = max(0, idx - width)
            hi = min(n, idx + width + 1)
            amt = a[idx] * rng.uniform(0.01, 0.1)
            if amt < 1e-8:
                continue
            # Spread amt to neighbors
            a[lo:hi] += amt * np.exp(-0.5 * ((np.arange(lo, hi) - idx) / width) ** 2)
            a[idx] -= amt
            a = np.maximum(a, 0)
            s2 = a.sum()
            if s2 < 1e-10:
                a = best_a.copy()
                conv = fast_convolve(a)
                cur_obj = best_obj
                continue
            a /= s2
        
        conv = fast_convolve(a)
        new_obj = 2.0 * n * conv.max()
        
        # Accept/reject
        delta_obj = new_obj - cur_obj
        if delta_obj < 0:
            cur_obj = new_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            # Metropolis criterion
            if rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
                cur_obj = new_obj
            else:
                # Revert
                a = best_a.copy()
                a /= a.sum()
                conv = fast_convolve(a)
                cur_obj = 2.0 * n * conv.max()
    
    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian (center)
    for sf in [0.1, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - n/2) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g
    
    # Sinusoidal (half-wave rectified)
    for freq in [1, 2, 3]:
        s = 0.5 * (1 + np.cos(2 * np.pi * freq * x / n))
        s /= s.sum()
        inits[f'sin_{freq}'] = s
    
    # Bimodal (edges)
    bi = np.zeros(n)
    w = max(1, n // 6)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Ramp
    ramp = np.arange(1, n+1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Random Dirichlet
    for alpha in [0.3, 1.0, 3.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits[f'rand_{alpha}'] = r
    
    # "Chirp" - quadratic phase
    chirp = 0.5 * (1 + np.cos(2 * np.pi * x**2 / (2 * n)))
    chirp = np.maximum(chirp, 0)
    if chirp.sum() > 1e-10:
        chirp /= chirp.sum()
        inits['chirp'] = chirp
    
    # From parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0).copy()
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp
    
    return inits

"""
Hybrid optimizer combining mirror descent, projected gradient descent,
and informed simulated annealing to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses diverse initializations, multi-stage optimization, and adaptive parameters.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_norm(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(conv, a, t, thresh_frac=0.002):
    """Smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = t * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    thresh = w.max() * thresh_frac
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    return sg


def mirror_descent(a0, time_limit, t_init=50.0, eta_init=0.3):
    """Multiplicative mirror descent on the simplex with adaptive params."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    max_pat = 50
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient(conv, a, t)
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            eta *= 0.65
            t *= 1.3
            if eta < 0.004:
                elapsed = time.monotonic() - t0
                frac = min(1.0, elapsed / max(time_limit, 1.0))
                eta = eta_init * (1.0 - 0.5 * frac)
                t = t_init * (1.0 + 3.0 * frac)
    return best_a, best_obj


def pgd_optimize(a0, time_limit, t_init=30.0, lr_init=0.08):
    """Projected gradient descent with logsumexp smoothing."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    lr = lr_init
    patience = 0
    max_pat = 40
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient(conv, a, t)
        a_new = a - lr * sg
        a_new = project_simplex(a_new)
        if a_new.sum() < 0.5:
            a_new = 0.5 * a + 0.5 * project_simplex(a)
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            lr *= 0.7
            t *= 1.2
            if lr < 0.002:
                lr = lr_init * 0.5
                t = t_init * 2.0
    return best_a, best_obj


def anneal_informed(a0, time_limit, rng):
    """Informed simulated annealing targeting peak reduction."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.015 * (1.0 - frac) ** 2 + 0.0005
        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.75
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        # Choose target: prefer positions with low contribution to peaks
        target_j = int(rng.integers(n))
        for _ in range(5):
            candidate = int(rng.integers(n))
            contrib = 0.0
            for pk in peak_positions[:8]:
                d = pk - candidate
                if 0 <= d <= n - 1:
                    contrib += a[d]
                d2 = candidate - pk + 2 * n - 2
                if 0 <= d2 <= n - 1:
                    contrib += a[d2]
            if contrib < a.mean() * 2.0:
                target_j = candidate
                break
        if target_j == top_j:
            continue
        delta = a[top_j] * rng.uniform(0.01, 0.06)
        if delta < 1e-9:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
    return best_a, best_obj


def anneal_random(a0, time_limit, rng):
    """Random pairwise transfer annealing (simpler, for diversity)."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.03 * (1.0 - frac) ** 1.5 + 0.001
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        delta = a[i] * rng.uniform(0.005, 0.04)
        if delta < 1e-9:
            continue
        a[i] -= delta
        a[j] += delta
        if a[i] < 0:
            a[i] += delta
            a[j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[i] += delta
                a[j] -= delta
    return best_a, best_obj


def generate_initializations(n, rng, parent):
    """Generate diverse initializations for a given length."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n
    for sf in [0.1, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_' + str(sf)] = g
    for freq in [1, 2, 3]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        s /= s.sum()
        inits['sin_' + str(freq)] = s
    bi = np.zeros(n)
    w = max(1, n // 6)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    for alpha in [0.3, 1.0, 3.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['rand_' + str(alpha)] = r
    chirp = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * x / (2.0 * n)))
    chirp = np.maximum(chirp, 0.0)
    if chirp.sum() > 1e-10:
        chirp /= chirp.sum()
        inits['chirp'] = chirp
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0).copy()
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp
    return inits


def get_parent():
    """Try to access parent sequence from global variable."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def safe_evaluate(a):
    """Safely evaluate a sequence, returning inf on failure."""
    try:
        a_clean = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a_clean = np.minimum(a_clean, 1000.0)
        if a_clean.sum() < 0.01:
            a_clean = a_clean * (1.0 / max(a_clean.sum(), 1e-15))
        if not np.all(np.isfinite(a_clean)):
            return float('inf')
        score = float(evaluate_sequence(a_clean.tolist()))
        return score
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Hybrid optimizer using mirror descent, PGD,
    and informed annealing to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)

    parent = get_parent()

    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d" % (score, len(a)))

    # Phase 1: Quick scan
    print("Phase 1: Quick scan...")
    scan_lengths = [64, 128, 256, 512, 1024]
    scan_results = []
    phase1_budget = min(350.0, (deadline - time.monotonic()) * 0.18)

    for n in scan_lengths:
        if time.monotonic() > deadline - 1800:
            break
        inits = generate_initializations(n, rng, parent)
        n_configs = len(inits)
        time_per = max(2.0, phase1_budget / max(1, len(scan_lengths) * n_configs))

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 1800:
                break
            t_alloc = min(time_per, deadline - time.monotonic() - 1800)
            if t_alloc < 1.5:
                break
            a_opt, _ = mirror_descent(a0, time_limit=t_alloc, t_init=40.0, eta_init=0.4)
            score = safe_evaluate(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt))
            record(a_opt)

    print("  Scan complete. Best: %.6f" % best_score)
    scan_results.sort(key=lambda x: x[0])

    # Phase 2: Deep optimization
    print("Phase 2: Deep optimization...")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 14:
            top_configs.append((n, name, a_opt))
            seen.add(key)

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'current_best', best_sequence.copy()))

    phase2_budget = min(1400.0, (deadline - time.monotonic()) * 0.55)
    n_deep = len(top_configs)
    time_per_deep = max(8.0, phase2_budget / max(1, n_deep))

    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 800:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 800)
        if t_alloc < 5.0:
            break
        print("  [%d/%d] n=%d, %s, budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))

        sub = t_alloc / 5.0
        a_cur = a0.copy()

        a_cur, _ = mirror_descent(a_cur, time_limit=sub, t_init=50.0, eta_init=0.3)
        record(a_cur)
        a_cur, _ = pgd_optimize(a_cur, time_limit=sub, t_init=30.0, lr_init=0.07)
        record(a_cur)
        a_cur, _ = mirror_descent(a_cur, time_limit=sub, t_init=100.0, eta_init=0.15)
        record(a_cur)
        a_cur, _ = pgd_optimize(a_cur, time_limit=sub, t_init=80.0, lr_init=0.04)
        record(a_cur)
        a_cur, _ = mirror_descent(a_cur, time_limit=sub, t_init=200.0, eta_init=0.08)
        record(a_cur)

    print("  Deep opt complete. Best: %.6f" % best_score)

    # Phase 3: Annealing local search
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("Phase 3: Annealing...")
        phase3_budget = min(500.0, deadline - time.monotonic() - 100)
        n_anneal = 4
        anneal_per = phase3_budget / n_anneal

        for i in range(n_anneal):
            if time.monotonic() > deadline - 100:
                break
            a_start = best_sequence.copy()
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
            rng_a = np.random.default_rng(seed + 5000 + i)
            if i % 2 == 0:
                a_opt, _ = anneal_informed(a_start, time_limit=anneal_per, rng=rng_a)
            else:
                a_opt, _ = anneal_random(a_start, time_limit=anneal_per, rng=rng_a)
            record(a_opt)

        print("  Annealing complete. Best: %.6f" % best_score)

    # Phase 4: Final polish
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_best = best_sequence.copy()
            a_best = np.maximum(a_best, 0.0)
            a_best /= a_best.sum()

            half = remaining * 0.4
            a_opt, _ = mirror_descent(a_best, time_limit=half, t_init=300.0, eta_init=0.05)
            record(a_opt)

            if time.monotonic() < deadline - 5:
                rem2 = deadline - time.monotonic() - 5
                if rem2 > 5:
                    a_opt2, _ = pgd_optimize(a_best, time_limit=rem2, t_init=200.0, lr_init=0.02)
                    record(a_opt2)

            if time.monotonic() < deadline - 5:
                rem3 = deadline - time.monotonic() - 5
                if rem3 > 3:
                    rng_f = np.random.default_rng(seed + 9999)
                    a_opt3, _ = anneal_informed(a_best, time_limit=rem3, rng=rng_f)
                    record(a_opt3)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
