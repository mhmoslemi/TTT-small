```python
"""
Minimize 2*n*max(conv(a,a)) / sum(a)^2 over non-negative sequences.

Strategy:
  1. Start from the provided construction (length 8192).
  2. Use L-BFGS-B with a log-sum-exp smoothing of the max, annealed over 
     multiple runs with increasing beta for progressively more exact optimization.
  3. Try multiple initializations: the given sequence, perturbed versions,
     and multi-scale (downsample-optimize-upsample) variants.
  4. Polish with a subgradient method using adaptive step size.
  5. Return the best sequence found.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def get_fft_size(n):
    """Return the FFT size for convolution (>= 2n-1) and correlation (>= 3n-2)."""
    conv_size = 1 << (2 * n - 1).bit_length()
    corr_size = 1 << (3 * n - 2).bit_length()
    return conv_size, corr_size


def compute_conv(a, conv_size):
    """Compute full self-convolution via FFT."""
    A = fft(a, conv_size)
    c = np.real(ifft(A * A))
    return c[:2 * len(a) - 1]


def compute_smooth_obj_and_grad(a, beta, conv_size, corr_size):
    """Compute smooth objective and gradient for L-BFGS-B.
    
    Objective: f(a) = logsumexp(b) / beta  (scale-invariant, so we can fix scale)
    where b = convolve(a, a).
    """
    n = len(a)
    b = compute_conv(a, conv_size)
    b_max = b.max()
    shifted = beta * (b - b_max)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    sum_exp = exp_shifted.sum()
    if sum_exp < 1e-300:
        value = b_max / beta
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        value = (b_max + max_shifted / beta + np.log(sum_exp) / beta)
        p = exp_shifted / sum_exp
    
    # Gradient: dL/da[i] = sum_j p[i+j] * a[j]
    # Use FFT cross-correlation
    A = np.zeros(corr_size, dtype=np.complex128)
    A[:n] = a
    P = np.zeros(corr_size, dtype=np.complex128)
    P[:2 * n - 1] = p
    grad = np.real(ifft(np.conj(A) * P))[:n]
    
    return value, grad
```

```python
"""
Minimize 2*n*max(conv(a,a)) / sum(a)^2 over non-negative sequences.

Algorithm:
  1. Start from provided 8192-length construction.
  2. Multi-phase L-BFGS-B optimization with annealed log-sum-exp smoothing:
     - Phase A: beta=10,20,50 for fast descent into a good basin
     - Phase B: beta=100,300,1000,3000 for progressive refinement
     - Each phase uses the previous phase's result as initial point
  3. Random restarts: perturb the current best and re-optimize with high beta
  4. Multi-length exploration: try n=4096 and n=16384
  5. Subgradient polish with AdaGrad on the final best
  All convolution/gradient computations use FFT for O(n log n) complexity.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def next_pow2(m):
    return 1 << (m - 1).bit_length()


def compute_conv(a, conv_size=None):
    n = len(a)
    if conv_size is None:
        conv_size = next_pow2(2 * n - 1)
    A = fft(a, conv_size)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def smooth_max_and_probs(b, beta):
    """Return (logsumexp(b)/beta, softmax(beta*b) probabilities)."""
    b_max = b.max()
    shifted = beta * (b - b_max)
    # Further shift for numerical stability
    s_max = shifted.max()
    exp_vals = np.exp(shifted - s_max)
    sum_exp = exp_vals.sum()
    if sum_exp <= 0:
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
        return b_max, p
    value = b_max + (s_max + np.log(sum_exp)) / beta
    p = exp_vals / sum_exp
    return value, p


def obj_grad_lbfgsb(a, beta, conv_size, corr_size):
    """Objective and gradient for L-BFGS-B.
    f = logsumexp(b) / beta, where b = conv(a,a)
    """
    n = len(a)
    b = compute_conv(a, conv_size)
    value, p = smooth_max_and_probs(b, beta)
    
    # Gradient: dL/da[i] = sum_j p[i+j] * a[j]
    # = cross-correlation of a and p at lag i
    A = np.zeros(corr_size, dtype=np.complex128)
    A[:n] = a
    P = np.zeros(corr_size, dtype=np.complex128)
    P[:2 * n - 1] = p
    grad = np.real(ifft(np.conj(A) * P))[:n]
    
    return value, grad

def subgradient_polish(a, max_time, conv_size, top_k=10):
    """AdaGrad subgradient method to polish the result."""
    n = len(a)
    a = a.copy()
    S = a.sum()
    start = time.monotonic()
    G = np.zeros(n)
    lr = 1.0
    best = a.copy()
    best_obj = exact_objective(a)
    
    while time.monotonic() - start < max_time:
        b = compute_conv(a, conv_size)
        cur = 2.0 * n * b.max() / (a.sum() ** 2)
        if cur < best_obj:
            best_obj = cur
            best = a.copy()
        
        # Subgradient of max(b)
        top_k_idx = np.argpartition(b, -top_k)[-top_k:]
        top_vals = b[top_k_idx]
        w = top_vals / top_vals.sum()
        
        # Gradient of max(b) at these indices
        grad = np.zeros(n)
        for wk, k in zip(w, top_k_idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                # db[k]/da[lo:hi+1] = a[k-hi : k-lo+1] reversed
                grad[lo:hi+1] += wk * 2.0 * a[k-hi:k-lo+1][::-1]
        
        # AdaGrad update
        G += grad ** 2
        a = np.maximum(a - lr * grad / (np.sqrt(G) + 1e-8), 0.0)
        a = normalize(a, S)
        lr *= 0.999
        
        if time.monotonic() - start > max_time * 0.9:
            break
    
    return best

    if ratio < 2:
        return a_large
    blocks = a_large[:n_small * ratio].reshape(n_small, ratio)
    a_small = blocks.mean(axis=1)

    xe = np.linspace(0, 1, n_target)
    a_up = np.interp(xe, xs, a_small)
```

```python
Minimize 2*n*max(conv(a,a)) / sum(a)^2 over non-negative sequences.

Algorithm:
  1. Start from the provided 8192-length construction.
  2. Multi-phase L-BFGS-B optimization with annealed log-sum-exp smoothing of max.
  3. Random restarts with perturbation from the current best to escape local minima.
  4. Subgradient polish with AdaGrad for final refinement.
  5. Multi-length exploration (try n=16384 via upsampled initialization).
  All convolution/gradient computations use FFT for O(n log n) per evaluation.

import numpy as np

from scipy.fft import fft, ifft
from scipy.optimize import minimize


def next_pow2(m):
    """Smallest power of 2 >= m."""
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def compute_conv(a, conv_size=None):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if conv_size is None:
        conv_size = next_pow2(2 * n - 1)
    A = fft(a, conv_size)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def smooth_max_and_probs(b, beta):
    """Return (logsumexp(b)/beta, softmax(beta*b)). Numerically stable."""
    b_max = b.max()
    shifted = beta * (b - b_max)
    s_max = shifted.max()
    exp_vals = np.exp(shifted - s_max)
    sum_exp = exp_vals.sum()
    if sum_exp <= 1e-300:
        p = np.zeros_like(b)
        idxs = np.where(b == b_max)[0]
        p[idxs] = 1.0 / len(idxs)
        return b_max, p
    value = b_max + (s_max + np.log(sum_exp)) / beta
    p = exp_vals / sum_exp
    return value, p


def obj_grad_smooth(a, beta, conv_size, corr_size):
    """Compute smooth objective f=logsumexp(b)/beta and its gradient."""
    n = len(a)
    b = compute_conv(a, conv_size)
    value, p = smooth_max_and_probs(b, beta)
    # Gradient: dL/da[i] = sum_j p[i+j] * a[j] (cross-correlation)
    A = np.zeros(corr_size, dtype=np.complex128)

    P = np.zeros(corr_size, dtype=np.complex128)

    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    start = time.monotonic()
    G = np.zeros(n)
    lr_base = 1.0
    best = a.copy()

    t = 0

    while time.monotonic() - start < max_time:
        t += 1
        b = compute_conv(a, conv_size)
        cur = 2.0 * n * b.max() / (a.sum() ** 2)

            best_obj = cur
            best = a.copy()
            if t % 500 == 0:
                print("    subgrad polish t=%d obj=%.6f" % (t, best_obj))

        # Weighted subgradient using top-k active lags
        if top_k >= len(b):
            top_idx = np.arange(len(b))
            w = np.ones(len(b)) / len(b)
        else:
```
