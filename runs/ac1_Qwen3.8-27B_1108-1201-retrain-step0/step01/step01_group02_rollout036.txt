"""
Multi-strategy optimization for minimizing peak auto-convolution of non-negative sequences.
Combines LP-based refinement, log-domain gradient descent, and local search.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a.astype(np.float64))
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_watermelon(n, width_frac=0.6):
    """Watermelon slice: flat top with smooth edges."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    half_width = width_frac * n / 2.0
    x = np.abs(t - center) / half_width
    result = np.ones(n)
    mask = x > 1.0
    result[mask] = 0.5 * (1.0 + np.cos(np.pi * (x[mask] - 1.0) / (2.0 - 1.0)))
    # Actually let's make it simpler: raised cosine from 0 to 1 at edges
    result = np.ones(n)
    edge = int(n * (1 - width_frac) / 2)
    if edge > 0:
        x = np.arange(edge) / edge
        result[:edge] = 0.5 * (1 - np.cos(np.pi * x))
        result[n-edge:] = 0.5 * (1 - np.cos(np.pi * x[::-1]))
    return np.maximum(0, result)


def initial_two_bumps(n, sep_frac=0.3):
    """Two Gaussian bumps separated by sep_frac of the sequence."""
    t = np.arange(n, dtype=np.float64)
    center1 = (1.0 - sep_frac) * (n - 1) / 2.0
    center2 = (1.0 + sep_frac) * (n - 1) / 2.0
    sigma = 0.15 * n
    return np.exp(-0.5 * ((t - center1) / sigma) ** 2) + np.exp(-0.5 * ((t - center2) / sigma) ** 2)


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    # For each column j (variable a_j), it appears in rows j to j+n-1
    # with value a_j in each
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=600.0, max_iters=1000, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")
    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 60:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(45.0, remaining * 0.25))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement at iter {it}")
            break
        g = g / sum_g
        # Adaptive step size with line search
        alphas = [0.5, 0.3, 0.2, 0.1, 0.05]
        best_alpha = 0.1
        best_new_score = score_sequence((1 - 0.1) * a + 0.1 * g)
        for alpha in alphas:
            a_new = normalize_nonneg((1.0 - alpha) * a + alpha * g)
            s = score_sequence(a_new)
            if s < best_new_score:
                best_new_score = s
                best_alpha = alpha
        a = normalize_nonneg((1.0 - best_alpha) * a + best_alpha * g)
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def log_gradient_descent(a, time_limit_s=60.0, verbose=True):
    """Gradient descent in log-domain to optimize the shape."""
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    # Work in log domain: a_j = exp(x_j)
    # Use a floor to avoid log(0)
    floor = 1e-8
    a_clipped = np.maximum(a, floor * a.max() if a.max() > 0 else floor)
    x = np.log(a_clipped / a_clipped.sum())  # log of normalized values
    
    best_a = a.copy()
    best_score = score_sequence(a)
    
    lr = 0.01
    patience = 0
    
    while time.monotonic() - start_time < time_limit_s:
        # Compute convolution
        a_current = np.exp(x - np.max(x))  # scale for numerical stability
        a_current = a_current / a_current.sum()
        
        conv = np.convolve(a_current, a_current)
        max_b = conv.max()
        score = 2.0 * n * max_b / (a_current.sum() ** 2)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_current.copy()
            patience = 0
        else:
            patience += 1
            if patience > 50:
                break
        
        # Find the peak lag
        peak_lag = np.argmax(conv)
        
        # Compute gradient of conv(peak_lag) with respect to x
        # conv(k) = sum_j a_j * a_{k-j}
        # d conv(k) / d a_j = sum over valid pairs = a_{k-j} + (if j != k-j) a_{k-j}
        # Actually: conv(k) = sum_{j: 0<=j<n, 0<=k-j<n} a_j * a_{k-j}
        # d/d a_j: if 0 <= k-j < n, then a_{k-j} appears
        # If j != k-j (i.e., 2j != k), then a_{k-j} also contributes: so gradient is 2*a_{k-j}
        # If j == k-j (i.e., j = k/2), then gradient is a_{k/2} = a_j
        
        grad_conv = np.zeros(n)
        for j in range(n):
            k_j = peak_lag - j
            if 0 <= k_j < n:
                if j == k_j:
                    grad_conv[j] += a_current[j]
                else:
                    grad_conv[j] += a_current[k_j]
        
        # Gradient of log(a_j) is grad_conv[j] / a_current[j]
        grad_log = grad_conv / (a_current + 1e-15)
        
        # Update
        x -= lr * grad_log
        
        # Renormalize x (shift doesn't matter since we normalize a)
        x -= x.mean()
        
        # Decay learning rate
        lr *= 0.999
        
        if lr < 1e-6:
            break
    
    if verbose:
        print(f"    LogGD done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=120.0, rng=None, verbose=True):
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LS start: score={best_score:.6f}")
    accept_count = 0
    current = a.copy()
    current_score = best_score
    
    while time.monotonic() - start_time < time_limit_s:
        # Random perturbation of a random segment
        width = max(2, int(rng.integers(max(2, n // 100), max(3, n // 6))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = current.copy()
        
        # Multiplicative perturbation
        noise = rng.normal(0.0, 0.15, size=stop - start)
        candidate[start:stop] *= np.exp(noise)
        
        # Occasionally mirror
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])
        
        # Occasionally zero out a small region
        if rng.random() < 0.1:
            z_start = int(rng.integers(0, n))
            z_width = max(1, n // 50)
            candidate[z_start:min(n, z_start + z_width)] *= 0.5
        
        candidate = np.maximum(0, candidate)
        candidate_norm = candidate.sum()
        if candidate_norm < 1e-10:
            continue
        candidate = candidate / candidate_norm
        
        score = score_sequence(candidate)
        
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            current = candidate.copy()
            current_score = score
            accept_count += 1
            if verbose and accept_count % 30 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")
        elif score < current_score + 0.001 * best_score:
            # Accept with some probability (simulated annealing-like)
            if rng.random() < 0.3:
                current = candidate.copy()
                current_score = score
    
    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

"""
Multi-strategy optimization for minimizing peak auto-convolution of non-negative sequences.
Combines LP-based refinement, multiplicative weight updates, log-domain gradient descent,
and local search with diverse initializations and adaptive time management.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a.astype(np.float64))
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_watermelon(n, width_frac=0.6):
    t = np.arange(n, dtype=np.float64)
    result = np.ones(n)
    edge = max(1, int(n * (1 - width_frac) / 2))
    if edge > 0 and edge < n:
        x = np.arange(edge, dtype=np.float64) / edge
        ramp = 0.5 * (1.0 - np.cos(np.pi * x))
        result[:edge] = ramp
        result[n - edge:] = ramp[::-1]
    return np.maximum(0.0, result)


def initial_asymmetric_bump(n, center_frac=0.4):
    t = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    sigma = 0.2 * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_parts = []
    for j in range(n):
        row_parts.append(np.arange(j, j + n, dtype=np.int32))
    row_idx = np.concatenate(row_parts)
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=600.0, max_iters=2000, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")
    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 80:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(50.0, remaining * 0.25))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:

        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:

def log_gradient_refine(a, time_limit_s=60.0, verbose=True):
    """Gradient descent in log-domain focusing on the peak convolution lag."""
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    floor = 1e-10
    a_safe = np.maximum(a, floor)
    x = np.log(a_safe / a_safe.sum())

    best_a = a.copy()
    best_score = score_sequence(a)
    lr = 0.05
    stagnant = 0

    if verbose:
        print(f"    LogGD start: score={best_score:.6f}, n={n}")

    while time.monotonic() - start_time < time_limit_s:
        a_current = np.exp(x - np.max(x))
        a_current = a_current / a_current.sum()
        a_current = np.maximum(a_current, 0.0)

        conv = np.convolve(a_current, a_current)
        score = 2.0 * n * conv.max() / (a_current.sum() ** 2)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_current.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 60:
                break

        # Compute gradient for peak lag
        peak_lag = int(np.argmax(conv))
        grad_x = np.zeros(n)
        for m in range(n):
            km = peak_lag - m
            if 0 <= km < n:
                grad_x[m] = 2.0 * a_current[m] * a_current[km]

        # Soft max over all lags for smoother optimization
        conv_clipped = np.maximum(conv, 0.0)
        beta = 20.0
        conv_scaled = beta * conv_clipped / max(1e-15, conv_clipped.max())
        weights = np.exp(conv_scaled - conv_scaled.max())
        weights = weights / weights.sum()

        # Weighted gradient
        grad_weighted = np.zeros(n)
        for k in range(len(conv)):
            if weights[k] < 1e-6:
                continue
            for m in range(n):
                km = k - m
                if 0 <= km < n:
                    grad_weighted[m] += weights[k] * 2.0 * a_current[m] * a_current[km]

        # Update
        x -= lr * grad_weighted
        x -= x.mean()

        # Decay learning rate
        lr *= 0.995
        if lr < 1e-5:
            break

    if verbose:
        print(f"    LogGD done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

                current_score = score

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def generate_asymmetric_init(n, rng, center_frac=None, sigma_frac=0.22):
    if center_frac is None:
        center_frac = float(rng.uniform(0.3, 0.7))
    t = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP-based refinement, multiplicative weights, log-domain gradient,
    and local search with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 25.0)

    best_score = float('inf')
    best_seq = None

    # Try to use previous best as starting point
    try:
        global height_sequence_1
        if height_sequence_1 is not None and len(height_sequence_1) > 0:
            hs = np.array(height_sequence_1, dtype=np.float64)
            hs = np.maximum(0, hs)
            if hs.sum() > 0.01:
                hs_norm = hs / hs.sum()
                hs_score = score_sequence(hs_norm)
                print(f"Using height_sequence_1: n={len(hs)}, score={hs_score:.6f}")
                best_score = hs_score
                best_seq = hs_norm.copy()
    except Exception:
        pass

    # Phase 1: Quick screening of diverse initializations
    print("\n=== Phase 1: Screening ===")
    init_configs = []
    if best_seq is not None:
        init_configs.append((len(best_seq), 'prev'))

    for n in [400, 600, 800, 1000, 1500, 2000, 3000]:
        init_configs.append((n, 'rc'))
        init_configs.append((n, 'gauss'))
        init_configs.append((n, 'asym'))
        if n >= 800:
            init_configs.append((n, 'water'))

    for n, init_type in init_configs:
        remaining = deadline - time.monotonic()
        if remaining < 40.0:
            break

        if init_type == 'prev':
            a = best_seq.copy()
            n = len(a)
        elif init_type == 'rc':
            a = initial_raised_cosine(n)
        elif init_type == 'gauss':
            a = initial_gaussian(n, sigma_frac=0.22)
        elif init_type == 'asym':
            a = generate_asymmetric_init(n, rng, sigma_frac=0.20)
        elif init_type == 'water':
            a = initial_watermelon(n, width_frac=0.55)
        else:
            a = initial_raised_cosine(n)

        a = normalize_nonneg(a)
        init_score = score_sequence(a)

        if init_score >= best_score + 0.1:
            continue

        time_limit = min(45.0, remaining * 0.12)
        if time_limit < 8.0:
            continue

        # Quick LP
        a_ref, score = lp_refine(a, time_limit_s=time_limit, verbose=False)
        # Quick MW
        remaining2 = deadline - time.monotonic()
        if score < best_score and remaining2 > 20:
            a_ref, score = multiplicative_weight_refine(a_ref, time_limit_s=min(20.0, remaining2 * 0.3), verbose=False)

        if score < best_score:
            best_score = score
            best_seq = a_ref.copy()
            print(f"  Phase 1 new best: score={score:.6f}, n={n}, init={init_type}")

    print(f"  Phase 1 best: score={best_score:.6f}")

    # Phase 2: Intensive LP optimization
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        print(f"\n=== Phase 2: Intensive LP (remaining={remaining:.0f}s) ===")
        lp_time = min(remaining * 0.45, 1000.0)
        if lp_time > 30.0:
            a_ref, score = lp_refine(best_seq, time_limit_s=lp_time, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = a_ref.copy()
                print(f"  *** After intensive LP: score={best_score:.6f}")

        # Phase 2b: Multiplicative weight refinement
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            mw_time = min(remaining * 0.15, 150.0)
            a_ref, score = multiplicative_weight_refine(best_seq, time_limit_s=mw_time, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = a_ref.copy()
                print(f"  *** After MW: score={best_score:.6f}")

        # Phase 2c: Log-domain gradient
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            gd_time = min(remaining * 0.15, 120.0)
            a_ref, score = log_gradient_refine(best_seq, time_limit_s=gd_time, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = a_ref.copy()
                print(f"  *** After LogGD: score={best_score:.6f}")

        # Phase 2d: Second LP round
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            lp_time2 = min(remaining * 0.4, 500.0)
            a_ref, score = lp_refine(best_seq, time_limit_s=lp_time2, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = a_ref.copy()
                print(f"  *** After LP round 2: score={best_score:.6f}")

    # Phase 3: Local search polishing
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20.0:
            print(f"\n=== Phase 3: Local Search (remaining={remaining:.0f}s) ===")
            ls_time = min(remaining * 0.5, 300.0)
            polished, score = local_search_refine(best_seq, time_limit_s=ls_time, rng=rng, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = polished.copy()
                print(f"  *** After LS: score={best_score:.6f}")

        # Final LP polish
        remaining = deadline - time.monotonic()
        if remaining > 20.0:
            final_lp_time = min(remaining * 0.85, 400.0)
            print(f"\n=== Final LP polish (time={final_lp_time:.0f}s) ===")
            polished, score = lp_refine(best_seq, time_limit_s=final_lp_time, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = polished.copy()
                print(f"  *** Final LP: score={best_score:.6f}")

    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Verify
    print(f"\nFinal result: score={best_score:.6f}, n={len(best_seq)}")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified score via evaluate_sequence: {verify_score:.6f}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")

    return [float(x) for x in best_seq]
