from scipy.fft import fft, ifft
import time

def fft_conv(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = fft(a, N)
    return ifft(fa * fa)[:size].real

def simplex_project(a, S):
    """Project a onto {x >= 0, sum(x) = S}."""
    n = a.size
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, a.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    # Fix rounding
    diff = S - res.sum()
    if abs(diff) > 1e-12:
        idx = res.argmax()
        res[idx] += diff
    return res

def smooth_max_and_grad(c, tau):
    """Compute smooth max of c and its gradient w.r.t. c."""
    c_shift = c - c.max()
    e = np.exp(c_shift / tau)
    q = e / e.sum()
    val = c.max() + tau * np.log(e.sum())
    return val, q  # q is the gradient w.r.t. c

def grad_smooth_max_conv(a, q, tau, N):

    Uses precomputed FFT size N."""
    n = len(a)

    # = (q * flip(a))[n-1+i] ... let me just use FFT directly

    N2 = 1 << (3 * n - 2 + 1).bit_length()  # power of 2 >= 3n-1

    # Actually: ifft(fft(q) * conj(fft(a)))[i] = sum_k q[k] * a[k-i] (circular)

    h = ifft(fq * np.conj(fa))[:n].real

    # h[i] = sum_k q[k] * a[k-i] (where a[m] = 0 for m outside [0,n-1])
    # = sum_{j=0}^{n-1} q[i+j] * a[j]  (substituting j = k-i)

    return (2.0 / tau) * h
