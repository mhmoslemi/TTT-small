"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Frank-Wolfe with LP oracle.
The LP at each step solves: max sum(b) s.t. b >= 0, conv(a,b) <= M.
This gives the exact best feasible direction, leading to faster convergence
than gradient-based methods. Combined with diverse initializations and
adaptive step sizes, this achieves the best known bound.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a, b=None):
    """Linear convolution using FFT."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def build_lp_matrix(a):
    """Build the constraint matrix for the LP oracle."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2*n - 2)
        length = k_end - k_start + 1
        A[k_start:k_end+1, j] = a[:length]
    return A

def solve_lp(a, M):
    """
    Solve: max sum(b) s.t. b >= 0, A @ b <= M * 1
    where A is the convolution matrix.
    Returns (b_solution, y_dual) or (None, None) on failure.
    """
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x, res.ineqlin  # primal, dual (might not be exactly dual)
    return None, None

def frank_wolfe_lp(a0, time_limit, verbose=False):
    """
    Frank-Wolfe with LP oracle to minimize max(conv(a,a)) on the simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Solve LP
        b, _ = solve_lp(a, M)
        if b is None:
            # Fallback: uniform
            b = np.ones(n) / n
        else:
            b_sum = b.sum()
            if b_sum < 1e-12:
                b = np.ones(n) / n
            else:
                b /= b_sum  # Put on simplex
        
        # Frank-Wolfe step size
        iter_count += 1
        # Try line search for better step
        # Simplified: use standard FW step size
        # t = 2 / (iter_count + 2)
        
        # Or: try a few step sizes
        best_t = 0
        best_val = best_obj
        for t_candidate in [2.0/(iter_count + 2), 0.5, 0.7, 1.0]:
            a_trial = (1 - t_candidate) * a + t_candidate * b
            a_trial = np.maximum(a_trial, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-12:
                continue
            a_trial /= s_trial
            conv_trial = fast_convolve(a_trial)
            val = 2.0 * n * conv_trial.max()
            if val < best_val:
                best_val = val
                best_t = t_candidate
        
        if best_t == 0:
            best_t = 2.0 / (iter_count + 2)
        
        a = (1 - best_t) * a + best_t * b
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        
        conv = fast_convolve(a)
        M = conv.max()
        current_obj = 2.0 * n * M
        
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and iter_count % 100 == 0:
            print(f"  FW iter {iter_count}: obj={current_obj:.6f}, best={best_obj:.6f}")
        
        if no_improve > 200:
            # Restart with a perturbation
            a = a + 0.01 * np.random.randn(n)
            a = np.maximum(a, 0)
            a /= a.sum()
            no_improve = 0
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Frank-Wolfe with LP oracle
(solve: max sum(b) s.t. b>=0, conv(a,b)<=M at each step) combined with
hybrid mirror descent for large n. Diverse initializations and adaptive

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a, b=None):
    """Linear convolution using FFT."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_lp_matrix(n, a):
    """Build A matrix where A[k,j] = a[k-j] if valid, else 0. Shape (2n-1, n)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A[k_start:k_end + 1, j] = a[:length]
    return A


def solve_lp_direction(a, M):
    """Solve max sum(b) s.t. b>=0, conv(a,b)<=M. Returns b or None."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def frank_wolfe_lp(a0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle on the probability simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Solve LP for direction
        b = solve_lp_direction(a, M)
        if b is None or b.sum() < 1e-12:
            b = np.ones(n) / n
        else:
            b = b / b.sum()

        iter_count += 1

        # Line search over step size
        best_t = 2.0 / (iter_count + 2)
        best_val = best_obj
        candidates = [2.0 / (iter_count + 2), 0.3, 0.5, 0.7, 1.0]
        for tc in candidates:
            a_trial = (1 - tc) * a + tc * b
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-12:
                continue
            a_trial /= st
            conv_trial = fast_convolve(a_trial)
            val = 2.0 * n * conv_trial.max()
            if val < best_val:
                best_val = val
                best_t = tc

        a = (1 - best_t) * a + best_t * b
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s

        conv = fast_convolve(a)
        M = conv.max()
        current_obj = 2.0 * n * M

        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and iter_count % 50 == 0:
            print(f"    FW iter {iter_count}: obj={current_obj:.6f}, best={best_obj:.6f}, t={best_t:.4f}")

        if no_improve > 300:
            # Perturb to escape local optimum
            a = a + 0.005 * np.random.default_rng(iter_count).standard_normal(n)
            a = np.maximum(a, 0)
            a /= a.sum()
            no_improve = 0

    return best_a, best_obj


def mirror_descent(a0, time_limit, t_param=40.0, eta0=0.5):
    """Multiplicative mirror descent with softmax peak smoothing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    patience = 0
    max_patience = 100
    current_t = t_param
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Softmax weights over convolution
        scaled = current_t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Subgradient via top peaks
        sg = np.zeros(n)
        active_idx = np.where(w > w.max() * 0.005)[0]
        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]

        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s

        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()

        if current_obj < best_obj - 1e-9:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            patience = 0
            eta *= 0.5
            if eta < 0.005:
                current_t *= 2.0
                eta = 0.3
                max_patience = 50
            else:
                current_t = np.clip(current_t * 1.3, 5.0, 500.0)

    return best_a, best_obj

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussians
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Edge Gaussian
    g_edge = np.exp(-0.5 * (x / (0.15 * n)) ** 2) + np.exp(-0.5 * ((x - n + 1) / (0.15 * n)) ** 2)
    g_edge /= g_edge.sum()
    inits['edge_gauss'] = g_edge

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Bimodal (edges)
    bi = np.zeros(n)
    bi[:max(1, n // 4)] = 1.0
    bi[-max(1, n // 4):] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Cosine
    cos_b = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_b /= cos_b.sum()
    inits['cosine'] = cos_b

    # Random
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits['random'] = r

    # Comb pattern (sparse)
    comb = np.zeros(n)
    step = max(1, n // (n // 3))
    for i in range(0, n, step):
        comb[i] = 1.0
    comb /= comb.sum()
    inits['comb'] = comb

    # Block pattern
    block = np.zeros(n)
    block[:n // 3] = 1.0
    block[2 * n // 3:] = 1.0
    block /= block.sum()
    inits['block'] = block

    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n].copy(), 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp

    return inits

                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish MD: n={n_best}, t={tp:.0f}, score={score:.6f}")

        # Try coordinate descent polish
        if time.monotonic() < deadline - 10 and best_sequence is not None:
            a_cd = best_sequence.copy()
            n_cd = len(a_cd)
            a_cd /= a_cd.sum()
            t_cd_start = time.monotonic()
            cd_deadline = deadline - 10
            improved = True
            cd_iter = 0
            while improved and time.monotonic() < cd_deadline:
                improved = False
                cd_iter += 1
                # Try small perturbations on random coordinates
                for _ in range(min(200, max(50, int((cd_deadline - time.monotonic()) * 2))):
                    if time.monotonic() > cd_deadline:
                        break
                    idx = rng.integers(0, n_cd)
                    for delta in [-0.01, -0.005, 0.005, 0.01, 0.02]:
                        a_try = a_cd.copy()
                        a_try[idx] = max(0, a_try[idx] + delta)
                        st = a_try.sum()
                        if st < 1e-10:
                            continue
                        a_try /= st
                        cv = fast_convolve(a_try)
                        obj_try = 2.0 * n_cd * cv.max()
                        cv_cur = fast_convolve(a_cd)
                        obj_cur = 2.0 * n_cd * cv_cur.max()
                        if obj_try < obj_cur - 1e-10:
                            a_cd = a_try
                            improved = True
                            break

            if improved:
                score = float(evaluate_sequence(a_cd.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_cd.copy()
                    print(f"  CD Polish: score={score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
