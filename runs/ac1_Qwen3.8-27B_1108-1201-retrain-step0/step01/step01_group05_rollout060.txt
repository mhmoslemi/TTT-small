"""
Search for non-negative unit-sum sequences minimizing
    2 * n * max(conv(a,a)) / (sum(a))^2
using annealed lp-norm smoothing of the convolution coefficients.

Key ideas:
  * Work on the probability simplex (sum(a)=1), so the objective is 2*n*max(c).
  * Use real FFTs for fast convolution and for the corresponding gradient.
  * Replace the nonsmooth max by an lp-norm surrogate; weights are
    proportional to u_k^(p-1), with p annealed from mild to very high.
  * Update using either:
        - mirror / multiplicative weights on the simplex, or
        - Adam in log/softmax coordinates.
  * Start from the provided best construction when available, refine at 8192,
    then transfer the best profile to 16384 for a finer discrete approximation.
  * Keep the best reliable sequence, with authoritative evaluate_sequence calls
    only a few times to avoid spending the runtime budget on direct convolution.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft

_GLOBAL_FAST_SEQ = None
_GLOBAL_FAST_F = np.inf
_GLOBAL_EXACT_SEQ = None
_GLOBAL_EXACT_F = np.inf
_GLOBAL_LAST_EXACT = 0.0

PHASES_8192 = [
    (4,   6, "mirror"),
    (8,   8, "mirror"),
    (12,  8, "mirror"),
    (16, 10, "mirror"),
    (24, 10, "mirror"),
    (32, 10, "mirror"),
    (48, 10, "logadam"),
    (64, 12, "logadam"),
    (96, 12, "logadam"),
    (128,14, "mirror"),
    (192,10, "logadam"),
]

PHASES_16384 = [
    (32,  6, "mirror"),
    (48,  8, "mirror"),
    (64, 12, "logadam"),
    (96, 12, "logadam"),
    (128,14, "mirror"),
    (192,16, "logadam"),
    (256,16, "mirror"),
    (384, 8, "logadam"),
    (512, 8, "mirror"),
]

PHASES_ALT = [
    (16, 15, "mirror"),
    (32, 15, "mirror"),
    (48, 15, "logadam"),
    (64, 15, "logadam"),
    (96, 15, "mirror"),
    (128,15, "logadam"),
    (192,10, "mirror"),
]

PHASES_FINAL = [
    (128,20, "mirror"),
    (192,30, "logadam"),
    (256,30, "mirror"),
    (384,20, "logadam"),
]


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def resample_unit(x, n):
    arr = np.asarray(x, dtype=np.float64).reshape(-1)
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if not np.isfinite(s) or s <= 1e-12:
        return np.full(int(n), 1.0 / max(1, int(n)), dtype=np.float64)
    arr = arr / s
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    if arr.size == n:
        return arr.astype(np.float64, copy=True)
    if arr.size == 1:
        return np.full(n, float(arr[0]), dtype=np.float64)
    old = np.linspace(0.0, 1.0, arr.size)
    new = np.linspace(0.0, 1.0, n)
    y = np.interp(new, old, arr)
    y = np.clip(y, 0.0, None)
    s2 = y.sum()
    if not np.isfinite(s2) or s2 <= 1e-12:
        y = np.ones(n, dtype=np.float64)
        s2 = float(n)
    return y / s2


def safe_normalize(x, n=None):
    arr = np.asarray(x, dtype=np.float64).reshape(-1)
    if arr.size == 0:
        if n is None:
            n = 256
        arr = np.ones(int(n), dtype=np.float64)
    else:
        if n is not None and arr.size != int(n):
            arr = resample_unit(arr, int(n))
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if not np.isfinite(s) or s <= 1e-12:
        arr = np.ones(arr.size, dtype=np.float64)
        s = float(arr.size)
    return arr / s


def symmetrize_unit(x):
    a = safe_normalize(x)
    a = 0.5 * (a + a[::-1])
    s = a.sum()
    if s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def get_base_unit():
    try:
        h = height_sequence_1
    except Exception:
        return None
    arr = np.asarray(h, dtype=np.float64).reshape(-1)
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if arr.size > 1 and np.isfinite(s) and s > 1e-12:
        return arr / s
    return None


def random_smooth_init(n, rng):
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    y = np.ones(n, dtype=np.float64)
    freqs = (1, 2, 3, 5, 7)
    for i, f in enumerate(freqs):
        amp = 0.12 / (1.0 + 0.35 * i)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        y = y + amp * np.cos(2.0 * np.pi * f * t + phase)
    y = np.clip(y, 1e-3, None)
    s = y.sum()
    if not np.isfinite(s) or s <= 1e-12:
        y = np.ones(n, dtype=np.float64)
        s = float(n)
    return y / s


def compute_fast_mu(a, use_unit=True):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if use_unit:
        s = a.sum()
        if not np.isfinite(s) or s <= 1e-12:
            return np.inf
        a = a / s
    n = a.size
    if n < 2:
        return np.inf
    m = 2 * n - 1
    N = next_pow2(m)
    A = rfft(a, N)
    c = irfft(A * A, N)[:m]
    c = 0.5 * (c + c[::-1])
    c = np.maximum(c, 0.0)
    return float(n * np.max(c))


def consider_sequence(a, fast_f=None, force_exact=False, tag=""):
    global _GLOBAL_FAST_SEQ, _GLOBAL_FAST_F
    global _GLOBAL_EXACT_SEQ, _GLOBAL_EXACT_F
    global _GLOBAL_LAST_EXACT

    a = safe_normalize(a)
    n = a.size
    if n < 2:
        return

    if fast_f is None or not np.isfinite(fast_f):
        mu = compute_fast_mu(a)
        if not np.isfinite(mu):
            return
        fast_f = 2.0 * mu

    fast_f = float(fast_f)

    if fast_f < _GLOBAL_FAST_F - 1e-12:
        _GLOBAL_FAST_F = fast_f
        _GLOBAL_FAST_SEQ = a.copy()
        print("fast  %-14s F=%.8f n=%d" % (tag, fast_f, n))

    now = time.monotonic()
    do_exact = bool(force_exact)
    if not do_exact:
        do_exact = (
            now - _GLOBAL_LAST_EXACT > 600.0 and
            n <= 9000 and
            fast_f < 1.55
        )

    if do_exact:
        try:
            exact = evaluate_sequence(a.tolist())
        except Exception:
            exact = np.inf
        _GLOBAL_LAST_EXACT = now
        if np.isfinite(exact):
            print("exact %-14s F=%.8f fast=%.8f n=%d" % (tag, float(exact), fast_f, n))
            if float(exact) < _GLOBAL_EXACT_F - 1e-12:
                _GLOBAL_EXACT_F = float(exact)
                _GLOBAL_EXACT_SEQ = a.copy()


def run_pnorm_phase(a0, n, time_budget, rng, p, method="mirror",
                    noise_sigma=0.018, verbose_tag=""):
    phase_time = max(0.0, float(time_budget))
    a = safe_normalize(a0, int(n))
    n = int(a.size)

    if phase_time <= 0.5:
        return a, compute_fast_mu(a)

    L = 2 * n - 1
    m = 2 * n - 2
    N = next_pow2(3 * n - 3)
    start = time.monotonic()
    deadline = start + phase_time

    A = rfft(a, N)
    c = irfft(A * A, N)[:L]
    c = 0.5 * (c + c[::-1])
    c = np.maximum(c, 0.0)
    best_mu = float(n * np.max(c))
    best_a = a.copy()

    if verbose_tag:
        print("%s phase p=%s method=%s budget=%.1f startF=%.8f"
              % (verbose_tag, p, method, phase_time, 2.0 * best_mu))

    first = True
    it = 0
    no_imp = 0

    if method == "logadam":
        z = np.log(np.maximum(a, 1e-300))
        z = z - np.mean(z)
        mom1 = np.zeros(n, dtype=np.float64)
        mom2 = np.zeros(n, dtype=np.float64)
        t_adam = 0
        b1 = 0.80
        b2 = 0.99
    else:
        z = None
        mom1 = None
        mom2 = None
        t_adam = 0
        b1 = b2 = None

    while time.monotonic() < deadline - 1.0:
        if first:
            u = n * c
            first = False
        else:
            A = rfft(a, N)
            c = irfft(A * A, N)[:L]
            c = 0.5 * (c + c[::-1])
            c = np.maximum(c, 0.0)
            u = n * c

        cur_mu = float(np.max(u))
        if cur_mu < best_mu - 1e-12:
            best_mu = cur_mu
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        max_u = cur_mu if cur_mu > 1e-15 else 1.0
        z_w = u / max_u
        z_w = np.clip(z_w, 0.0, 1.0)

        p_int = int(p)
        if p_int <= 2:
            w = z_w.astype(np.float64, copy=True)
        else:
            w = np.power(z_w, float(p_int - 1))

        if p_int >= 256:
            floor_amount = 1e-6 / float(L)
        elif p_int >= 128:
            floor_amount = 1e-7 / float(L)
        else:
            floor_amount = 1e-9 / float(L)

        w = w + floor_amount
        ws = w.sum()
        if not np.isfinite(ws) or ws <= 1e-300:
            w = np.full(L, 1.0 / float(L), dtype=np.float64)
        else:
            w = w / ws
        w = 0.5 * (w + w[::-1])
        ws = w.sum()
        if np.isfinite(ws) and ws > 1e-300:
            w = w / ws
        else:
            w = np.full(L, 1.0 / float(L), dtype=np.float64)

        W = rfft(w, N)
        H = irfft(A * W, N)
        grad = 2.0 * H[m - n + 1:m + 1][::-1]
        grad = np.maximum(grad, 0.0)
        gmax = float(np.max(grad))
        if not np.isfinite(gmax) or gmax <= 1e-30:
            break

        g = grad / gmax
        g = g - np.mean(g)
        gmax_c = float(np.max(np.abs(g)))
        if gmax_c <= 1e-15:
            break
        g = g / gmax_c

        now = time.monotonic()
        rem_frac = max(0.0, (deadline - now) / max(1e-9, phase_time))

        if method == "mirror":
            lr = 0.42 / (1.0 + 0.08 * float(p_int - 2))
            lr_eff = lr * (0.45 + 0.55 * rem_frac)
            if no_imp > 2500:
                lr_eff *= 0.55
            if lr_eff <= 1e-8:
                break
            a = a * np.exp(-lr_eff * g)
            a = np.maximum(a, 1e-16)
            s = a.sum()
            if not np.isfinite(s) or s <= 1e-30:
                a = np.ones(n, dtype=np.float64)
                s = float(n)
            a = a / s

        elif method == "logadam":
            t_adam += 1
            ag = float(np.dot(a, g))
            gz = a * (g - ag)
            rms = float(np.sqrt(np.mean(gz * gz)))
            if rms > 1e-30:
                gz = gz / rms

            mom1 = b1 * mom1 + (1.0 - b1) * gz
            mom2 = b2 * mom2 + (1.0 - b2) * (gz * gz)
            mh = mom1 / (1.0 - b1 ** t_adam)
            vh = mom2 / (1.0 - b2 ** t_adam)

            lr = 0.18 / (1.0 + 0.05 * float(p_int - 2))
            lr_eff = lr * (0.50 + 0.50 * rem_frac)
            if no_imp > 2500:
                lr_eff *= 0.70

            z = z + lr_eff * (mh / (np.sqrt(vh) + 1e-8))
            z = np.clip(z - np.mean(z), -50.0, 50.0)
            a = np.exp(z)
            s = a.sum()
            if not np.isfinite(s) or s <= 1e-30:
                z = np.zeros(n, dtype=np.float64)
                a = np.ones(n, dtype=np.float64)
                s = float(n)
            a = a / s
        else:
            break

        if no_imp > 3000 and rng.random() < 0.03:
            if method == "logadam" and z is not None:
                z = z + rng.normal(0.0, noise_sigma, n)
                z = np.clip(z - np.mean(z), -50.0, 50.0)
                a = np.exp(z)
                s = a.sum()
                if not np.isfinite(s) or s <= 1e-30:
                    a = np.ones(n, dtype=np.float64)
                    s = float(n)
                a = a / s
            else:
                a = a * np.exp(rng.normal(0.0, noise_sigma, n))
                a = np.maximum(a, 1e-16)
                s = a.sum()
                if not np.isfinite(s) or s <= 1e-30:
                    a = np.ones(n, dtype=np.float64)
                    s = float(n)
                a = a / s
            no_imp = 0

        it += 1
        if verbose_tag and (it == 1 or it % 3000 == 0):
            print("%s p=%s it=%d curF=%.8f bestF=%.8f"
                  % (verbose_tag, p_int, it, 2.0 * cur_mu, 2.0 * best_mu))

    return best_a, best_mu


def optimize_phases(a0, n, phases, time_budget, rng, verbose_tag=""):
    n = int(n)
    a = safe_normalize(a0, n)
    budget = max(0.0, float(time_budget))
    if budget <= 1.0:
        return a, compute_fast_mu(a)

    start = time.monotonic()
    deadline = start + budget
    best_a = a.copy()
    best_mu = compute_fast_mu(best_a)

    total_frac = sum(float(fr) for _, fr, _ in phases)
    if total_frac <= 1e-12:
        return best_a, best_mu

    remaining_frac = total_frac
    for p, frac, method in phases:
        now = time.monotonic()
        if deadline - now < 3.0:
            break
        phase_frac = float(frac)
        if remaining_frac <= 1e-12:
            break
        phase_time = (deadline - now) * (phase_frac / remaining_frac)
        remaining_frac -= phase_frac
        if remaining_frac < 0.0:
            remaining_frac = 0.0

        a_phase, mu_phase = run_pnorm_phase(
            a, n, phase_time, rng, p, method=method, verbose_tag=verbose_tag
        )

        if not np.isfinite(mu_phase):
            a_phase, mu_phase = best_a, best_mu

        if mu_phase < best_mu:
            best_a = a_phase.copy()
            best_mu = float(mu_phase)
            a = best_a.copy()
        else:
            # Allow a very slight temporary increase to preserve a better
            # smoothed iterate for the next, sharper lp phase.
            if mu_phase <= best_mu + 5e-4:
                a = a_phase.copy()
            else:
                a = best_a.copy()

    return best_a, best_mu


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    global _GLOBAL_FAST_SEQ, _GLOBAL_FAST_F
    global _GLOBAL_EXACT_SEQ, _GLOBAL_EXACT_F
    global _GLOBAL_LAST_EXACT

    start = time.monotonic()
    try:
        user_budget = float(budget_s)
    except Exception:
        user_budget = 2400.0
    total_budget = max(600.0, min(2390.0, user_budget))
    hard_deadline = start + total_budget
    soft_deadline = hard_deadline - 30.0

    _GLOBAL_FAST_SEQ = None
    _GLOBAL_FAST_F = np.inf
    _GLOBAL_EXACT_SEQ = None
    _GLOBAL_EXACT_F = np.inf
    _GLOBAL_LAST_EXACT = start

    rng = np.random.default_rng(seed)

    base = get_base_unit()
    if base is None:
        base = random_smooth_init(8192, rng)
    base = safe_normalize(base)

    a8192 = resample_unit(base, 8192)
    mu8 = compute_fast_mu(a8192)
    consider_sequence(a8192, 2.0 * mu8, force_exact=False, tag="init8192")

    # Main refinement on the known good 8192 length.
    t8 = min(0.32 * total_budget, max(0.0, soft_deadline - time.monotonic()))
    a8 = a8192
    if t8 > 120.0:
        a8, mu8 = optimize_phases(a8192, 8192, PHASES_8192, t8, rng, "main8192")
        consider_sequence(a8, 2.0 * mu8, force_exact=True, tag="main8192")

    # Transfer the best profile to 16384 and refine with sharper norms.
    remaining = max(0.0, soft_deadline - time.monotonic())
    t16 = min(0.44 * total_budget, max(0.0, remaining - 80.0))
    a16_start = None
    if t16 > 220.0:
        seq_for_16 = _GLOBAL_FAST_SEQ if _GLOBAL_FAST_SEQ is not None else a8
        a16_start = resample_unit(seq_for_16, 16384)
        a16, mu16 = optimize_phases(a16_start, 16384, PHASES_16384,
                                    t16, rng, "main16384")
        consider_sequence(a16, 2.0 * mu16, force_exact=False, tag="main16384")

    # Alternative 8192-length starts to avoid a single local basin.
    remaining = max(0.0, soft_deadline - time.monotonic())
    if remaining > 260.0:
        t_alt_total = min(0.55 * remaining, 360.0)
        t_sym = 0.55 * t_alt_total
        if t_sym > 80.0:
            a_sym = symmetrize_unit(resample_unit(base, 8192))
            a_s, mu_s = optimize_phases(a_sym, 8192, PHASES_ALT,
                                        t_sym, rng, "alt_sym")
            consider_sequence(a_s, 2.0 * mu_s, force_exact=False, tag="alt_sym")

        remaining = max(0.0, soft_deadline - time.monotonic())
        t_rand = min(0.60 * remaining, 250.0)
        if t_rand > 80.0:
            a_rand = random_smooth_init(8192, rng)
            a_r, mu_r = optimize_phases(a_rand, 8192, PHASES_ALT,
                                        t_rand, rng, "alt_rand")
            consider_sequence(a_r, 2.0 * mu_r, force_exact=False, tag="alt_rand")

    # Final short polish at 16384 of the current best.
    remaining = max(0.0, soft_deadline - time.monotonic())
    if remaining > 160.0:
        seq_pol = _GLOBAL_FAST_SEQ
        if seq_pol is None:
            seq_pol = a16_start if a16_start is not None else a8
        a_pol = resample_unit(seq_pol, 16384)
        t_pol = min(remaining - 40.0, 420.0)
        if t_pol > 90.0:
            a_p, mu_p = optimize_phases(a_pol, 16384, PHASES_FINAL,
                                        t_pol, rng, "final16384")
            consider_sequence(a_p, 2.0 * mu_p, force_exact=False, tag="final16384")

    # Choose and return the best validated sequence.
    if _GLOBAL_FAST_SEQ is None:
        final = np.ones(256, dtype=np.float64)
        final = safe_normalize(final)
        return [float(x) for x in final]

    fast_return = safe_normalize(_GLOBAL_FAST_SEQ)

    exact_seq = None
    exact_val = np.inf
    if _GLOBAL_EXACT_SEQ is not None and np.isfinite(_GLOBAL_EXACT_F):
        exact_seq = safe_normalize(_GLOBAL_EXACT_SEQ)
        exact_val = float(_GLOBAL_EXACT_F)

    # One final authoritative evaluation of the best fast iterate, if time exists.
    if time.monotonic() < hard_deadline - 20.0:
        try:
            ex_fast = evaluate_sequence(fast_return.tolist())
        except Exception:
            ex_fast = np.inf
        if np.isfinite(ex_fast):
            if not np.isfinite(exact_val) or float(ex_fast) < exact_val:
                exact_val = float(ex_fast)
                exact_seq = fast_return

    if exact_seq is not None and np.isfinite(exact_val):
        # If the fast surrogate is clearly better, prefer it; otherwise
        # return the sequence for which we have an authoritative value.
        if np.isfinite(_GLOBAL_FAST_F) and _GLOBAL_FAST_F < exact_val - 0.0015:
            final = fast_return
        else:
            final = exact_seq
    else:
        final = fast_return

    final = np.asarray(final, dtype=np.float64).reshape(-1)
    final = np.clip(final, 0.0, 1000.0)
    s = final.sum()
    if not np.isfinite(s) or s < 0.01:
        final = np.ones(256, dtype=np.float64)
    else:
        final = final / s

    try:
        print("final fastF=%.8f exactKnown=%.8f n=%d"
              % (2.0 * compute_fast_mu(final), loose_exact if (loose_exact := _GLOBAL_EXACT_F) else -1.0, final.size))
    except Exception:
        print("final n=%d" % final.size)

    return [float(x) for x in final]
