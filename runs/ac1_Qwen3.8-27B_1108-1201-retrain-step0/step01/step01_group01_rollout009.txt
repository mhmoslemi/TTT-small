import cvxpy as cp

def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) / (sum(a))^2"""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (S * S)

def _compute_b(a):
    return np.convolve(a, a)

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _lp_direction(a, M, solver=None):
    """
    Solve the LP: maximize sum(b) subject to b >= 0, (a*b)_k <= M for all k.
    Returns the optimal b vector.
    """
    n = len(a)
    b = cp.Variable(n)
    
    # Build constraint matrix (sparse)
    # Constraint k: sum_j a_{k-j} * b_j <= M for k = 0, ..., 2n-2

    # Use cvxpy's convolve if available, otherwise build manually

    constraints = []
    conv = a @ b  # This is the linear convolution
    constraints.append(conv <= M)
    constraints.append(b >= 0)
    
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    try:

        try:
            prob.solve()
        except Exception:

        return None

"""
LP-guided optimization with adaptive line search for minimizing normalized autocorrelation peak.
Uses CVXPY to solve direction-finding LPs, with subgradient fallback, multi-restart diversity,
and coordinate-descent polishing. Builds on the SoTA LP-framework approach.

import numpy as np
from scipy import sparse
import time
import cvxpy as cp


def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) / (sum(a))^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (S * S)


def _compute_peak(a):
    """Return max of convolve(a,a) and the score assuming sum(a)=sqrt(2n)."""
    b = np.convolve(a, a)
    return np.max(b)


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(v, 0.0)
    S = np.sum(v)
    if S < 1e-15:
        return np.full(n, target / n)
    if abs(S - target) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _build_conv_matrix(n):
    """Build sparse matrix C (shape (2n-1, n)) such that C @ b = convolve(a, b) for any a.
    Actually this returns the STRUCTURE; values need to be set from a each time."""
    col_idx = np.repeat(np.arange(n), n)
    row_idx = np.tile(np.arange(n), n) + col_idx
    # We'll set data = np.tile(a, n) each time
    return row_idx, col_idx


def _make_conv_csr(a, row_idx, col_idx):
    """Build CSR convolution matrix for a given a."""
    n = len(a)
    vals = np.tile(a, n)
    return sparse.csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))


def _solve_lp(a, M, row_idx, col_idx, solver_kwargs=None):
    """
    Solve: max sum(b) s.t. C(a) @ b <= M, b >= 0
    where C(a) is the convolution matrix for a.
    Returns optimal b vector or None.
    """
    n = len(a)
    C = _make_conv_csr(a, row_idx, col_idx)
    b = cp.Variable(n)
    constraints = [C @ b <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    solvers_to_try = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PIPS, cp.SCS]
    if solver_kwargs:
        solvers_to_try.insert(0, solver_kwargs)

    last_err = None
    for s in solvers_to_try:
        try:
            t0 = time.monotonic()
            prob.solve(solver=s)
            t1 = time.monotonic()
            if b.value is not None:
                val = np.maximum(np.asarray(b.value).flatten(), 0.0)
                return val, (t1 - t0)
        except Exception as e:
            last_err = e
            continue
    return None, 0.0


def _subgradient_step(a, eta, top_k=10):
    """One subgradient step to reduce max of convolve(a,a), preserving sum and non-negativity."""
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)

    # Get top-k peak indices
    if top_k >= len(b):
        k_idx = np.arange(len(b))
        wts = np.ones(len(b)) / len(b)
    else:
        k_idx = np.argpartition(b, -top_k)[-top_k:]
        b_sel = b[k_idx]
        margin = b_max - b_sel
        wts = np.exp(-3.0 * margin / max(b_max, 1e-15))
        wts /= np.sum(wts)

    # Compute weighted subgradient of max_k b_k
    grad = np.zeros(n)
    for ki in range(len(k_idx)):
        w = wts[ki]
        if w < 1e-8:
            continue
        k = int(k_idx[ki])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += w * 2.0 * a[k - idx]

    # Center for sum constraint
    grad -= np.mean(grad)

    # Normalized update
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a.copy()

    S = np.sum(a)
    a_new = a - eta * grad / gnorm
    a_new = _project_simplex(a_new, S)
    return a_new


def _perturb(a, rng, scale=0.05, mode='smooth'):
    """Perturb a sequence for diversity."""
    n = len(a)
    S = np.sum(a)
    if mode == 'smooth':
        freqs = rng.integers(1, max(2, n // 3), size=min(5, n // 2))
        noise = np.zeros(n)
        for f in freqs:
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(-1, 1)
            ii = np.arange(n)
            noise += amp * np.cos(2 * np.pi * f * ii / n + phase)
        noise /= (np.max(np.abs(noise)) + 1e-15)
        a_new = a + scale * noise * (S / n)
    elif mode == 'local':
        # Swap mass between random pairs
        a_new = a.copy()
        n_swaps = max(1, n // 50)
        for _ in range(n_swaps):
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            delta = scale * (S / n) * rng.uniform(-1, 1)
            a_new[i] -= delta
            a_new[j] += delta
        a_new = np.maximum(a_new, 0)
    else:
        a_new = a + rng.normal(0, scale * S / n, n)
        a_new = np.maximum(a_new, 0)
    a_new = _project_simplex(a_new, S)
    return a_new


def _make_two_level(n, eps):
    """Anti-symmetric two-level sequence normalized to sum=1."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0)
    return a / np.sum(a)


def _make_cosine(n, eps):
    """Cosine anti-symmetric profile."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / max(1, n - 1))
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main optimization: LP-guided descent with adaptive step size,
    subgradient fallback, multi-restart diversity, and polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 15.0)

    # Get starting sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size

    # Precompute matrix structure (indices don't change)
    row_idx, col_idx = _build_conv_matrix(n)
    print(f"Sequence length: {n}, matrix nonzeros: {n*n}")

    # Normalize to sum = 1 for score computation
    a = parent.copy()
    S_orig = np.sum(a)
    if S_orig < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S_orig
    best_a = a.copy()
    best_score = _compute_score(a)
    print(f"Initial score: {best_score:.6f}")

    # Normalize to sum = sqrt(2n) for LP framework (score = max(conv))
    target_sum = np.sqrt(2.0 * n)
    a_lp = a * target_sum

    lp_count = 0
    sg_count = 0
    no_improve = 0
    t_blend = 0.3  # initial blending for LP update
    reheat_counter = 0

    # Phase 1: Main LP optimization loop
    print(f"Phase 1: LP optimization (budget until t={deadline - time.monotonic():.0f}s)")

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        # Compute current peak
        b_conv = np.convolve(a_lp, a_lp)
        M = np.max(b_conv)
        cur_score = M  # Since sum(a_lp) = sqrt(2n), score = M

        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best_a = (a_lp / target_sum).copy()
            no_improve = 0
            if lp_count < 20 or lp_count % 50 == 0:
                print(f"  [{lp_count+sg_count}] New best: {best_score:.6f} (t={time.monotonic()+2400-deadline:.0f}s)")

        # Try LP solve
        b_dir, lp_time = _solve_lp(a_lp, M, row_idx, col_idx)

        if b_dir is not None and np.sum(b_dir) > 1e-10:
            lp_count += 1
            # Rescale direction to have sum = target_sum
            b_dir = target_sum / np.sum(b_dir) * b_dir

            # Line search over blending parameter
            best_t = t_blend
            best_t_score = cur_score
            for t_test in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                a_try = (1.0 - t_test) * a_lp + t_test * b_dir
                a_try = np.maximum(a_try, 0)
                S_try = np.sum(a_try)
                if S_try < 1e-10:
                    continue
                a_try = target_sum / S_try * a_try
                b_try = np.convolve(a_try, a_try)
                s_try = np.max(b_try)
                if s_try < best_t_score:
                    best_t_score = s_try
                    best_t = t_test

            a_lp = (1.0 - best_t) * a_lp + best_t * b_dir
            a_lp = np.maximum(a_lp, 0)
            S_lp = np.sum(a_lp)
            if S_lp < 1e-10:
                a_lp = best_a * target_sum
            else:
                a_lp = target_sum / S_lp * a_lp

            # Adapt t_blend
            if best_t_score < cur_score:
                t_blend = min(0.9, t_blend * 1.2)
            else:
                t_blend = max(0.05, t_blend * 0.7)

        else:
            # Fallback: subgradient step
            sg_count += 1
            eta = 0.05 / np.sqrt(n)
            a_lp = _subgradient_step(a_lp, eta, top_k=10)

        # Reheat: periodically perturb
        no_improve += 1
        if no_improve >= 80:
            reheat_counter += 1
            scale = 0.03 * min(reheat_counter, 5)
            a_lp = _perturb(a_lp, rng, scale=scale, mode='smooth' if reheat_counter % 2 == 0 else 'local')
            # Re-normalize
            S_lp = np.sum(a_lp)
            if S_lp > 1e-10:
                a_lp = target_sum / S_lp * a_lp
            t_blend = 0.3
            no_improve = 0
            print(f"  Reheat #{reheat_counter} at iter {lp_count+sg_count}")

    # Phase 2: Polishing with smaller steps
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"Phase 2: Polishing ({remaining:.0f}s remaining), best={best_score:.6f}")
        a_lp = best_a * target_sum
        polish_deadline = time.monotonic() + remaining * 0.7
        t_polish = 0.05
        for _ in range(200):
            if time.monotonic() > polish_deadline:
                break
            b_conv = np.convolve(a_lp, a_lp)
            M = np.max(b_conv)
            cur_score = M
            if cur_score < best_score:
                best_score = cur_score
                best_a = (a_lp / target_sum).copy()

            b_dir, _ = _solve_lp(a_lp, M, row_idx, col_idx)
            if b_dir is not None and np.sum(b_dir) > 1e-10:
                b_dir = target_sum / np.sum(b_dir) * b_dir
                a_lp = (1.0 - t_polish) * a_lp + t_polish * b_dir
                a_lp = np.maximum(a_lp, 0)
                S_lp = np.sum(a_lp)
                if S_lp > 1e-10:
                    a_lp = target_sum / S_lp * a_lp
                t_polish = max(0.01, t_polish * 0.95)
            else:
                a_lp = _subgradient_step(a_lp, 0.02 / np.sqrt(n), top_k=5)

    # Phase 3: Coordinate descent polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"Phase 3: Coordinate descent ({remaining:.0f}s remaining)")
        a_cd = best_a.copy()  # sum = 1
        n_cd = len(a_cd)
        cd_deadline = time.monotonic() + remaining - 5
        rng2 = np.random.default_rng(seed + 1)
        while time.monotonic() < cd_deadline:
            order = rng2.permutation(n_cd)
            for i in order:
                if time.monotonic() > cd_deadline:
                    break
                # Try moving mass from i to reduce peak
                b_cd = np.convolve(a_cd, a_cd)
                peak_idx = np.argmax(b_cd)
                # Gradient of peak w.r.t. a[i]
                k = peak_idx
                j = k - i
                if 0 <= j < n_cd:
                    # b_k includes a_i * a_j term
                    # Reduce a_i slightly, increase the minimum element
                    delta = a_cd[i] * 0.01
                    if delta > 1e-15:
                        j_add = np.argmin(a_cd)
                        if j_add != i:
                            a_try = a_cd.copy()
                            a_try[i] -= delta
                            a_try[j_add] += delta
                            s_try = _compute_score(a_try)
                            if s_try < _compute_score(a_cd) - 1e-12:
                                a_cd = a_try
                if _compute_score(a_cd) < best_score:
                    best_score = _compute_score(a_cd)
                    best_a = a_cd.copy()
                    print(f"  CD: New best {best_score:.6f}")

    # Final result
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    final_score = float(evaluate_sequence(result))
    print(f"FINAL SCORE: {final_score:.6f} (n={len(result)}, lp_iters={lp_count}, sg_iters={sg_count})")
    return result
