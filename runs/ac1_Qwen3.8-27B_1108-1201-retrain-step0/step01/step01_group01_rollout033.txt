"""
Smoothed gradient descent with temperature annealing to minimize the
normalized autocorrelation peak of a non-negative sequence.
Uses log-sum-exp smoothing of the max for a true gradient, FFT-based
convolution for speed, and periodic diversification to escape local minima.
"""

import numpy as np
import time
from scipy.signal import fftconvolve


def _score(a):
    """Compute the score for a normalized sequence (sum=1)."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _smooth_max_and_weights(b, T):
    """Compute smooth max and softmax weights."""
    b_max = np.max(b)
    shifted = (b - b_max) / T
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    smooth_val = b_max + T * np.log(np.sum(exp_shifted))
    return smooth_val, weights


def _gradient_from_weights(weights, a):
    """Compute gradient of weighted sum of convolution coefficients."""
    n = len(a)
    # grad[j] = 2 * sum_{k} weights[k] * a[k-j] (valid indices only)
    # Equivalent to: grad = 2 * correlate(weights, a[::-1], 'valid')
    grad = 2.0 * np.correlate(weights, a[::-1], mode='valid')
    return grad


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _make_ramp(n, t):
    """Anti-symmetric ramp: high at start, low at end."""
    x = np.linspace(-1, 1, n)
    a = 1.0 - t * x  # goes from 1+t to 1-t
    a = np.maximum(a, 0.01)  # keep positive
    a /= np.sum(a)
    return a


def _make_exp_ramp(n, t):
    """Exponential anti-symmetric ramp."""
    x = np.linspace(-1, 1, n)
    a = np.exp(t * x)  # goes from exp(-t) to exp(t)
    a /= np.sum(a)
    return a


def _make_quadratic(n, t1, t2):
    """Quadratic profile."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + t1 * x + t2 * x * x
    a = np.maximum(a, 0.01)
    a /= np.sum(a)
    return a

```python
"""
Smoothed gradient descent with FFT-based convolution and temperature annealing
to minimize the normalized autocorrelation peak of a non-negative sequence.
"""

import numpy as np
import time
import math


def _score(a):
    """Compute 2*n*max(conv(a,a)) for normalized sequence."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _gradient_smoothed(a, T):
    """Compute gradient of T*log(sum(exp(b_k/T))) w.r.t. a using FFT.
    
    Returns (smooth_value, gradient).
    """
    n = len(a)
    b = np.convolve(a, a)
    
    # Softmax weights
    b_max = np.max(b)
    shifted = (b - b_max) / T
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w /= w_sum
    smooth_val = b_max + T * np.log(w_sum)
    
    # Gradient via FFT cross-correlation
    # grad[j] = 2 * sum_m w[j+m] * a[m] for j = 0, ..., n-1
    N = 1
    while N < 3 * n:
        N *= 2
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    r = np.fft.irfft(np.conj(A) * W, N)
    grad = 2.0 * r[:n]
    
    return smooth_val, grad


def _project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _make_ramp(n, t):
    """Anti-symmetric linear ramp."""
    x = np.linspace(-1, 1, n)
    a = 1.0 - t * x
    a = np.maximum(a, 1e-6)
    return a / np.sum(a)


def _make_exp_ramp(n, t):
    """Anti-symmetric exponential ramp."""
    x = np.linspace(-1, 1, n)
    a = np.exp(t * x)
    return a / np.sum(a)


def _make_quadratic(n, t1, t2):
    """Quadratic anti-symmetric profile."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + t1 * x + t2 * x * x
    a = np.maximum(a, 1e-6)
    return a / np.sum(a)


def _smooth_optimize(a, deadline, T_init_frac=0.1, T_min_frac=0.01, seed=42):
    """
    Smoothed gradient descent with backtracking and temperature annealing.
    """
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a /= np.sum(a)
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    step = 1.0  # Will be adapted by backtracking
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() < deadline:
        iter_count += 1
        
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Perturb if stuck
        if no_improve > 500:
            noise = rng.normal(0, 0.02, n)
            noise -= np.mean(noise)
            a = best_a + noise
            a = _project_simplex(a, 1.0)
            no_improve = 0
            step = 0.1
            continue
        
        # Temperature: start moderate, decrease over time
        T = max(T_min_frac, T_init_frac) * b_max * np.exp(-0.003 * no_improve)
        
        # Compute smoothed gradient
        smooth_val, grad = _gradient_smoothed(a, T)
        
        # Center for sum constraint
        grad -= np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search (limit to 8 iterations for speed)
        a_new = a
        f_new = cur_score
        s = step
        improved = False
        for _ in range(8):
            a_try = _project_simplex(a - s * grad / gnorm, 1.0)
            f_try = _score(a_try)
            if f_try < cur_score - 1e-12:
                a_new = a_try
                f_new = f_try
                improved = True
                break
            s *= 0.5
            if s < 1e-10:
                break
        
        if improved:
            a = a_new
            step = min(s * 1.5, 5.0)
        else:
            step = max(s * 0.5, 1e-10)
            # If no improvement even with tiny step, perturb
            if step <= 1e-10:
                noise = rng.normal(0, 0.01, n)
                noise -= np.mean(noise)
                a = a + noise
                a = _project_simplex(a, 1.0)
                step = 0.1
    
    return best_a, best_score


def _subgrad_optimize(a, deadline, seed=42, M=10):
    """Standard subgradient method for final polishing."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a /= np.sum(a)
    
    best_a = a.copy()
    best_score = _score(best_a)
    eta = 0.05 / np.sqrt(n)
    no_improve = 0
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-12:
                break
        
        # Top M peaks with exponential weighting
        if M >= len(b):
            k_idx = np.arange(len(b))
            b_sel = b.copy()
        else:
            k_idx = np.argpartition(b, -M)[-M:]
            k_idx = k_idx[np.argsort(b[k_idx])[::-1]]
            b_sel = b[k_idx]
        
        wts = np.exp(-5.0 * (b_max - b_sel) / max(b_max, 1e-15))
        wts /= np.sum(wts)
        
        # Weighted convolution coefficients
        b_weighted = np.zeros(len(b))
        for i, k in enumerate(k_idx):
            b_weighted[k] = wts[i]
        
        # Gradient via FFT
        N = 1
        while N < 3 * n:
            N *= 2
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(b_weighted, N)
        r = np.fft.irfft(np.conj(A) * W, N)
        grad = 2.0 * r[:n]
        
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        a = a - eta * grad / gnorm
        a = _project_simplex(a, 1.0)
    
    return best_a, best_score
