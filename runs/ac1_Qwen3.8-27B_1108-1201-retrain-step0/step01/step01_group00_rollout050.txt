def fft_convolve(a, b):

    sum_exp = np.sum(exp_shifted)

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def nesterov_grad_descent(theta, n, time_limit, rng):
    """Nesterov-accelerated gradient descent with cosine annealing."""
    S = np.sqrt(2.0 * n)
    theta = theta.astype(np.float64).copy()
    
    v = np.zeros(n)  # velocity for Nesterov
    mu = 0.9  # momentum
    
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf
    step = 0
    plateau = 0
    plateau_best = best_score
    
    n_total = 5000  # planned total steps
    lr_base = 5.0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        # Cosine annealing for temperature
        frac = min(1.0, step / n_total)
        tau = (0.5 / n) + (30.0 / n - 0.5 / n) * 0.5 * (1 + np.cos(np.pi * frac))
        
        # Nesterov step
        theta_nest = theta + mu * v
        a_nest = softmax_normalize(theta_nest) * S
        
        # Compute gradient at Nesterov position
        c = fft_convolve(a_nest, a_nest)
        sm_val, w = soft_max_val_grad(c, tau)
        a_rev = a_nest[::-1]
        corr = fft_convolve(a_rev, w)
        g_a = 2.0 * corr[n-1 : 2*n-1]
        
        # Project gradient to tangent space of the simplex
        a = softmax_normalize(theta) * S
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)
        
        # Adam-like adaptive learning rate
        # Actually let's use simple Nesterov with adaptive lr
        lr = lr_base / (1.0 + 0.1 * step)
        
        v = mu * v - lr * g_theta
        theta = theta + v
        
        # Evaluate
        a = softmax_normalize(theta) * S
        c = fft_convolve(a, a)
        score = 2.0 * n * np.max(c) / (S * S)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
            plateau = 0
        else:
            plateau += 1
        
        # Random kick when stuck
        if plateau > 500 and step % 100 == 0:
            kick = rng.standard_normal(n) * 0.1
            theta = theta + kick
            v *= 0.5  # reset velocity
            plateau = 0
        
        if step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={elapsed:.1f}s")
    
    return best_theta, best_score

"""
Optimization minimizing autocorrelation peak of non-negative step functions.
Uses L-BFGS with annealed top-k/soft-max smoothing, asymmetric initializations,
and targeted local search polish. Key insight: avoid symmetry (score >= 2 for symmetric).
"""

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def obj_grad_topk(theta, n, k, S):
    """Objective and gradient using top-k mean of convolution."""
    a = softmax_normalize(theta) * S
    c = fft_convolve(a, a)
    
    # Get top k indices
    top_idx = np.argsort(c)[-k:]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / k
    
    score = np.dot(w, c) * 2.0 * n / (S * S)
    
    # Gradient: d(score)/d theta
    # d(score)/d a_j = (2*n/S^2) * sum_m w_m * dc_m/da_j
    # dc_m/da_j = 2*a_{m-j} if 0 <= m-j < n
    # d(a)/d(theta_j) = a_j * (delta_ji - a_i) * ... (from softmax)
    # Actually: a_i = softmax(theta)_i * S
    # d(a_i)/d(theta_j) = softmax(theta)_i * (delta_ij - softmax(theta)_j) * S
    # = a_i * (delta_ij - a_j / S)
    # = a_i * delta_ij - a_i * a_j / S
    
    # Let s_i = softmax(theta)_i = a_i / S
    # d(a_i)/d(theta_j) = S * s_i * (delta_ij - s_j)
    
    # d(score)/d(theta_j) = sum_i d(score)/d(a_i) * d(a_i)/d(theta_j)
    # = sum_i [(2*n/S^2) * 2 * (a * w)[i]] * S * s_i * (delta_ij - s_j)
    # where (a * w)[i] = sum_m w_m * a_{m-i} ... 
    
    # Let me define g_a[i] = d(score)/d(a_i) = (2*n/S^2) * 2 * sum_m w_m * a_{m-i}
    # (with the understanding that a_{m-i} = 0 if m-i not in [0,n-1])
    
    # g_a[i] = (4*n/S^2) * (a_rev * w)[n-1 + i] where a_rev = a[::-1]
    
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g_a = (4.0 * n / (S * S)) * corr[n-1 : 2*n-1]
    
    # d(score)/d(theta_j) = sum_i g_a[i] * S * s_i * (delta_ij - s_j)
    # = S * s_j * (g_a[j] - sum_i g_a[i] * s_i)
    # = S * s_j * (g_a[j] - dot(g_a, s))
    # = a_j * (g_a[j] - dot(g_a, a) / S)
    
    s_vec = a / S
    inner = np.dot(g_a, s_vec)
    g_theta = a * (g_a - inner)
    
    return score, g_theta

def obj_grad_softmax(theta, n, tau, S):
    """Objective and gradient using soft-max of convolution."""
    a = softmax_normalize(theta) * S
    c = fft_convolve(a, a)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    sm_val = c_max + tau * np.log(sum_exp)
    
    score = 2.0 * n * sm_val / (S * S)
    
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g_a = (4.0 * n / (S * S)) * corr[n-1 : 2*n-1]
    
    s_vec = a / S
    inner = np.dot(g_a, s_vec)
    g_theta = a * (g_a - inner)
    
    return score, g_theta

    """Run L-BFGS with time limit."""

        if mode == 'topk':

        else:

        )
        theta_best = result.x
        a_best = softmax_normalize(theta_best) * S
        best_score = local_score(a_best)
    except (TimeoutError, ValueError, np.linalg.LinAlgError):

Minimizes autocorrelation peak of non-negative step functions using

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)
