            a_new = project_simplex(a + v, S)
        else:
            a_new = project_simplex(a - lr * grad, S)
            v = np.zeros_like(a)

        # Line search
        score_new = compute_score(a_new)
        eval_count += 1

        if score_new < best_score - 1e-12:
            best = a_new.copy()
            best_score = score_new
            if verbose and step_count % 1000 == 0:
                print(f"      step={step_count}, score={best_score:.6f}, tau={tau:.4f}")
        elif score_new > compute_score(a) + 1e-8:
            # Step was bad, reduce
            if use_nesterov:
                a_new = project_simplex(a - lr * 0.1 * grad, S)
                v *= 0.5
            else:
                a_new = project_simplex(a - lr * 0.1 * grad, S)
                lr = max(0.001, lr * 0.5)

        # Periodic LP direction
        if use_lp and step_count > 0 and step_count % lp_interval == 0:
            remaining = t_budget - (time.monotonic() - t0)
            if remaining > 40:
                M = c.max()
                g = lp_direction(a, M, timeout=min(20.0, remaining * 0.3))
                if g is not None and g.sum() > 0:
                    g = project_simplex(g, S)
                    # Blend with current
                    for alpha in [0.3, 0.5, 0.7]:
                        a_try = project_simplex((1 - alpha) * a + alpha * g, S)
                        sc_try = compute_score(a_try)
                        if sc_try < best_score:
                            best = a_try.copy()
                            best_score = sc_try
                            a = a_try.copy()
                            break

        a = a_new
        step_count += 1

    # Final evaluation
    final_score = evaluate_sequence(best.tolist())
    if final_score < best_score:
        best_score = final_score
    if verbose:
        print(f"    grad_descent done: steps={step_count}, best_score={best_score:.6f}")
    return best, best_score


def local_search(a, S, t_budget, verbose=False):
    """Random local search: perturb random coordinates and keep improvements."""
    best = a.copy()
    best_score = compute_score(best)
    n = len(best)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < t_budget:
        a_try = best.copy()
        # Perturb a random subset of coordinates
        n_perturb = max(1, n // 100)
        idx = rng.choice(n, size=n_perturb, replace=False)
        a_try[idx] *= np.exp(rng.normal(0, 0.1, size=n_perturb))
        a_try = np.maximum(a_try, 0)
        a_try = project_simplex(a_try, S)
        sc = compute_score(a_try)
        if sc < best_score - 1e-12:
            best = a_try.copy()
            best_score = sc
            if verbose:
                print(f"      local_search improved: {best_score:.6f}")

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search: explore multiple inits/lengths, then polish best."""
    deadline = time.monotonic() + budget_s - 3.0
    bs = None
    bsc = np.inf

    # Get provided sequence
    prov = None
    try:
        prov = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if prov.size > 0 and np.isfinite(prov).all():
            prov = np.maximum(prov, 0.0)
            if prov.sum() < 0.01:
                prov = None
        else:
            prov = None
    except Exception:
        prov = None

    n_opts = [2500, 3000, 4000, 5000]
    if prov is not None:
        n_opts.append(len(prov))
    n_opts = list(set(n_opts))
    n_opts.sort()

    # Phase 1: Quick exploration with diverse initializations
    p1_dur = min(200.0, max(80.0, (deadline - time.monotonic()) * 0.15))
    p1_end = time.monotonic() + p1_dur

    inits = []
    for n in n_opts:
        inits.append((n, 'u_shape', 0, None))
        inits.append((n, 'v_shape', 1, None))
        inits.append((n, 'flat_spec', 2, None))
        inits.append((n, 'random', 42, None))
    if prov is not None:
        for n in n_opts:
            inits.append((n, 'provided', 0, prov))

    print(f"Phase 1: {len(inits)} candidates, budget={p1_dur:.0f}s")
    t_per = max(10.0, p1_dur / max(1, len(inits)))

    for i, (n, itype, sd, prov_seq) in enumerate(inits):
        tb = p1_end - time.monotonic()
        if tb < 8:
            break
        tb = min(t_per, tb)

        if itype == 'provided' and prov_seq is not None:
            a0 = init_from_sequence(prov_seq, n)
        elif itype == 'u_shape':
            a0 = init_u_shape(n, sd)
        elif itype == 'v_shape':
            a0 = init_v_shape(n, sd)
        elif itype == 'flat_spec':
            a0 = init_flat_spectrum(n, sd)
        else:
            a0 = init_random_exp(n, sd)

        S = np.sqrt(2.0 * n)
        print(f"  P1[{i+1}/{len(inits)}] n={n} {itype} t={tb:.0f}s", end=" ")
        a_r, sc = gradient_descent(a0, S, 1.0, 0.1, 0.5, tb,
                                    use_nesterov=True, use_lp=False, verbose=False)
        print(f"-> score={sc:.6f}")

        if sc < bsc:
            bsc, bs = sc, a_r.copy()
            print(f"  *** NEW BEST: {bsc:.6f} (n={n})")

    # Phase 2: LP-based Frank-Wolfe on best candidate
    rem = deadline - time.monotonic()
    if rem > 60 and bs is not None:
        n_best = bs.size
        S = np.sqrt(2.0 * n_best)
        a_curr = (bs / bs.sum() * S).copy()
        a_curr = project_simplex(a_curr, S)

        print(f"\nPhase 2: LP Frank-Wolfe on n={n_best}, remaining={rem:.0f}s")
        fw_budget = min(120.0, rem * 0.25)
        t0 = time.monotonic()
        best_fw = a_curr.copy()
        best_fw_score = compute_score(best_fw)
        n_fw_steps = 0

        while time.monotonic() - t0 < fw_budget:
            c = compute_conv(a_curr)
            M = c.max()
            remaining = fw_budget - (time.monotonic() - t0)
            if remaining < 15:
                break
            g = lp_direction(a_curr, M, timeout=min(25.0, remaining * 0.5))
            if g is None or g.sum() <= 0:
                break
            g = project_simplex(g, S)
            # Try several step sizes
            improved = False
            for alpha in [0.5, 0.3, 0.2, 0.1, 0.7, 0.8]:
                a_try = project_simplex((1 - alpha) * a_curr + alpha * g, S)
                sc = compute_score(a_try)
                if sc < best_fw_score - 1e-10:
                    best_fw = a_try.copy()
                    best_fw_score = sc
                    a_curr = a_try.copy()
                    improved = True
                    n_fw_steps += 1
                    break
            if not improved:
                # Try pure LP direction
                sc = compute_score(g)
                if sc < best_fw_score - 1e-10:
                    best_fw = g.copy()
                    best_fw_score = sc
                    a_curr = g.copy()
                    n_fw_steps += 1
                else:
                    break

        print(f"  FW steps={n_fw_steps}, score={best_fw_score:.6f}")
        if best_fw_score < bsc:
            bs, bsc = best_fw.copy(), best_fw_score
            print(f"  *** NEW BEST after FW: {bsc:.6f}")

    # Phase 3: Fine gradient polishing
    rem = deadline - time.monotonic()
    if rem > 30 and bs is not None:
        n_best = bs.size
        S = np.sqrt(2.0 * n_best)
        print(f"\nPhase 3: Fine polishing n={n_best}, remaining={rem:.0f}s")

        rounds = [
            (0.3, 0.05, 0.3, 0.4),
            (0.1, 0.02, 0.15, 0.35),
            (0.05, 0.01, 0.1, 0.3),
            (0.02, 0.005, 0.05, 0.25),
        ]
        for tau_hi, tau_lo, lr, frac in rounds:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            tb = rem * frac
            print(f"  Round: tau=[{tau_hi},{tau_lo}] lr={lr} t={tb:.0f}s", end=" ")
            a_r, sc = gradient_descent(bs, S, tau_hi, tau_lo, lr, tb,
                                        use_nesterov=True, use_lp=False, verbose=False)
            print(f"-> score={sc:.6f}")
            if sc < bsc:
                bsc, bs = sc, a_r.copy()
                print(f"  *** improved to {bsc:.6f}")

    # Phase 4: Local search
    rem = deadline - time.monotonic()
    if rem > 15 and bs is not None:
        n_best = bs.size
        S = np.sqrt(2.0 * n_best)
        tb = rem * 0.7
        print(f"\nPhase 4: Local search, t={tb:.0f}s")
        a_r, sc = local_search(bs, S, tb, verbose=False)
        if sc < bsc:
            bsc, bs = sc, a_r.copy()
            print(f"  *** improved to {bsc:.6f}")

    # Final: try a few more n values with the best score as reference
    rem = deadline - time.monotonic()
    if rem > 60 and bs is not None:
        # Try one more polish round with different structure
        n_try = 6000
        S = np.sqrt(2.0 * n_try)
        a0 = init_u_shape(n_try, 99)
        # Interpolate best into this length
        if bs is not None and len(bs) != n_try:
            x_old = np.linspace(0, 1, len(bs))
            x_new = np.linspace(0, 1, n_try)
            a0 = np.maximum(np.interp(x_new, x_old, bs), 0.0)
        print(f"\nPhase 5: Try n={n_try}, t={rem*0.5:.0f}s")
        a_r, sc = gradient_descent(a0, S, 0.5, 0.02, 0.2, rem * 0.5,
                                    use_nesterov=True, use_lp=False, verbose=False)
        print(f"  n={n_try} score={sc:.6f}")
        if sc < bsc:
            bsc, bs = sc, a_r.copy()
            print(f"  *** NEW BEST: {bsc:.6f} (n={n_try})")

    # Final evaluation
    if bs is None:
        bs = np.ones(2000)
        bsc = evaluate_sequence(bs.tolist())

    final_sc = evaluate_sequence(bs.tolist())
    if final_sc < bsc:
        bsc = final_sc

    print(f"\n=== FINAL: score={bsc:.6f}, n={len(bs)} ===")
    return [float(x) for x in bs]
