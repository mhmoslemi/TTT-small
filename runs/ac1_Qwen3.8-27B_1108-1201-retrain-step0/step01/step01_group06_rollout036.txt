import numpy as np

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def mirror_descent_optimize(a0, time_limit, t0=30.0, eta0=0.3, momentum=0.7):
    """
    Mirror descent (exponential weights) with momentum and adaptive step size.
    Work on the probability simplex (sum=1, a>=0).
    Returns the best (a, normalized_obj) found.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    momentum_buf = np.zeros(n)
    patience = 0
    max_patience = 50
    temperature = t0
    t_start = time.monotonic()
    no_improve_steps = 0

    while time.monotonic() - t_start < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-18:
            break

        # Softmax weights over convolution peaks
        scaled = temperature * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum

        # Compute subgradient: sg[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
        sg = np.zeros(n)
        # Use threshold to skip negligible weights
        thresh = w.max() * 0.001
        active_idx = np.where(w > thresh)[0]

        for k_int in active_idx:
            k = int(k_int)
            # sg[i] += w[k] * 2 * a[k-i] for valid i (0 <= k-i < n)
            # i ranges from max(0, k-n+1) to min(k, n-1)
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_start <= i_end:
                # a[k-i] for i = i_start..i_end gives a[k-i_start], ..., a[k-i_end]
                # = a[k-i_end : k-i_start+1][::-1]
                sg[i_start:i_end+1] += 2.0 * w[k] * a[k - i_end : k - i_start + 1][::-1]

        # Center the subgradient (for mirror descent on simplex)
        sg_centered = sg - sg.mean()

        # Momentum
        momentum_buf = momentum * momentum_buf + (1 - momentum) * sg_centered

        # Multiplicative update
        log_factor = -eta * momentum_buf
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

        # Evaluate
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
            no_improve_steps = 0
        else:
            patience += 1
            no_improve_steps += 1

        # Adaptive step size and temperature
        if patience >= max_patience:
            patience = 0
            if no_improve_steps < max_patience * 3:
                eta *= 0.5
            else:
                # Restart with higher temperature
                temperature *= 2.0
                eta = eta0
                momentum_buf *= 0.1
                no_improve_steps = 0
                if temperature > 500.0:
                    temperature = t0
                    # Perturb
                    a = a + 0.01 * np.random.randn(n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    conv = fast_convolve(a)

        # Gradually cool
        if patience % 10 == 0 and temperature > t0 * 0.3:
            temperature *= 0.95

    return best_a, best_obj

def generate_initializations(n, rng, parent=None):
