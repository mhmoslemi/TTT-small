Iterative LP water-filling approach to minimize the normalized autocorrelation peak

from scipy.sparse import csr_matrix
from scipy.sparse.linalg import spsolve
import time
import cvxpy as cp


def _build_conv_matrix(a):
    """Build the sparse convolution matrix C where (C @ x)_k = sum_i a_i * x_{k-i}."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[i])
    C = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return C


def _solve_lp_waterfill(a, M):

    Solve: max sum(b) s.t. C @ b <= M, b >= 0
    where C is the convolution matrix with a.
    Returns the optimal b.
    """
    n = len(a)
    C = _build_conv_matrix(a)
    
    b_var = cp.Variable(n)
    constraints = [b_var >= 0, C @ b_var <= M]
    obj = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(obj, constraints)
    
    # Try solving
    try:
        prob.solve(solver=cp.SCS, max_iter=100000)
        if b_var.value is not None:
            return np.maximum(b_var.value, 0)
    except:
        pass
    
    # Fallback: use scipy
    try:
        from scipy.optimize import linprog
        # max 1^T b s.t. C b <= M, b >= 0
        c = -np.ones(n)  # minimize -sum(b)
        A_ub = C.toarray()
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except:
        pass
    
    return np.ones(n) / n

Nesterov momentum, and backtracking line search to minimize the normalized
autocorrelation peak of a non-negative sequence. Includes periodic annealing

import numpy as np
import time


def _fft_convolve(a, b):
    """Linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size=size)
    fb = np.fft.rfft(b, size=size)
    result = np.fft.irfft(fa * fb, size=size)
    return result[:n + m - 1]


def _fft_cross_corr(w, a):
    """
    Compute cross-correlation (w cross a)[j] = sum_n w[j+n] * a[n] for j=0,...,n-1.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    size = 1
    while size < len(w) + n - 1:
        size *= 2
    W = np.fft.rfft(w, size=size)
    A = np.fft.rfft(a, size=size)
    # Cross-correlation: ifft(FFT(w) * conj(FFT(a)))
    corr = np.fft.irfft(W * np.conj(A), size=size)
    return corr[:n]


def _score_normalized(a):
    """Score = 2*n*max(fft_convolve(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = _fft_convolve(a, a)
    return 2.0 * n * float(np.max(b))


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)

        return v

    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
