def solve_lp(a, M):
    """Solve LP: max sum(b) s.t. a*b <= M, b >= 0"""
    n = len(a)
    # Build constraint matrix A where (A @ b)_j = (a*b)_j
    # A has shape (2n-1, n)
    # A[j, k] = a[k] if 0 <= j-k <= n-1
    rows = []
    cols = []
    vals = []

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative method.
Each iteration solves an LP: max sum(b) s.t. conv(a,b) <= M, b >= 0,
then blends the result back into a. Multiple initializations and
gradient polishing are used for refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _build_indices(n):
    """Pre-compute row/col indices for convolution matrix of length-n sequences."""
    m = 2 * n - 1
    counts = np.array([min(n, m - k) for k in range(n)], dtype=np.int64)
    rows = np.concatenate([np.arange(k, k + c, dtype=np.int64) for k, c in zip(range(n), counts)])
    cols = np.repeat(np.arange(n, dtype=np.int64), counts)
    return rows, cols, counts


def _make_matrix(a, indices):
    """Create sparse convolution matrix from values and pre-computed indices."""
    n = len(a)
    m = 2 * n - 1
    rows, cols, counts = indices
    vals = np.repeat(a, counts)
    return csr_matrix((vals, (rows, cols)), shape=(m, n))


def _solve_lp(a, M, indices):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = _make_matrix(a, indices)
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def _lp_search(a0, budget_s, t=0.5, M_scale=1.0, verbose=False):
    """Run LP-based iterations."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0).astype(np.float64)
    a = a / a.sum() * S
    
    indices = _build_indices(n)
    best = a.copy()
    best_mc = np.convolve(a, a).max()
    best_score = evaluate_sequence(a.tolist())
    
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < budget_s:
        mc = np.convolve(a, a).max()
        M = M_scale * mc
        
        b = _solve_lp(a, M, indices)
        if b is None or b.sum() < 1e-10:
            if verbose:
                print(f"    LP failed at it={it}")
            break
        
        b = b / b.sum() * S
        a = (1.0 - t) * a + t * b
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            break
        a = a / s * S
        
        it += 1
        if it % 3 == 0:
            mc_new = np.convolve(a, a).max()
            if mc_new < best_mc - 1e-15:
                best_mc = mc_new
                best = a.copy()
                best_score = evaluate_sequence(best.tolist())
                if verbose:
                    print(f"    it={it}, max_c={best_mc:.8f}, score={best_score:.8f}")
    
    mc_final = np.convolve(a, a).max()
    if mc_final < best_mc:
        best = a.copy()
        best_mc = mc_final
        best_score = evaluate_sequence(best.tolist())
    
    return best, best_score


def _soft_max_grad(a, tau):
    """Gradient of tau * log(sum(exp(conv(a,a)/tau)))."""
    c = np.convolve(a, a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    # grad of sum_j q_j * (2 * (q * a)_j) ... 
    # Actually, d/da_k [tau * log sum_j exp((a*a)_j / tau)]
    # = tau * (1/sum) * sum_j exp((a*a)_j/tau) * (1/tau) * d(a*a)_j/d(a_k)
    # = (1/sum_j exp(...)) * sum_j exp(...) * d(a*a)_j/d(a_k)
    # = sum_j q_j * d(a*a)_j/d(a_k)
    # d(a*a)_j/d(a_k) = a_{j-k} + a_k... wait
    # (a*a)_j = sum_i a_i * a_{j-i}
    # d/d(a_k) (a*a)_j = a_{j-k} + a_{j-k} = 2*a_{j-k} (if j-k is valid)
    # Wait, that's only if k != j-k. If k = j-k (i.e., j=2k), then it's just a_k (one term, not two).
    # Actually no: d/d(a_k) [sum_i a_i * a_{j-i}] = a_{j-k} + (if k appears as j-i, i.e., i=j-k, which it does) a_{j-k}
    # Hmm, let me be more careful.
    # (a*a)_j = sum_i a_i * a_{j-i} where i ranges over valid indices.
    # d/d(a_k) = sum_i [delta_{i,k} * a_{j-i} + a_i * delta_{j-i,k}]
    # = a_{j-k} + a_{j-k} (if both j-k and j-k are valid indices)
    # = 2*a_{j-k} (generically)
    # But if j-k = k (i.e., j = 2k), then it's a_k + a_k = 2*a_k, same thing.
    # Actually wait: the first term is a_{j-k} (from delta_{i,k}), and the second term is a_{j-k} (from delta_{j-i,k} which means i=j-k).
    # So it's always 2*a_{j-k} (when j-k is a valid index).
    
    # So grad_k = sum_j q_j * 2 * a_{j-k} (for valid j-k)
    # = 2 * sum_j q_j * a_{j-k}
    # = 2 * (q * a)_{... } -- this is a correlation
    
    # (q * a) has length 2n-1, and (q*a)_m = sum_i q_i * a_{m-i}
    # We want sum_j q_j * a_{j-k} = (q * a)_{... }
    # Let me define: (q * a)_m = sum_{j} q_j * a_{m-j}
    # We want sum_j q_j * a_{j-k} = sum_j q_j * a_{-(k-j)} = (q * a_{reversed})_{...}
    # Hmm, this is getting confusing. Let me just use the correlate function.
    
    # grad_k = 2 * sum_j q_j * a_{j-k} for |j-k| < n

    # np.correlate(q, a, 'valid') gives corr[k] = sum_j q[j+k] * a[j] for k=0..n-1
    # That's sum_j q[j+k] * a[j] = sum_i q[i] * a[i-k] = sum_j q[j] * a[j-k] (reindexing)
    # Yes! So grad_k = 2 * np.correlate(q, a, 'valid')[k]
    
    grad = 2.0 * np.correlate(q, a, 'valid')
    return grad


def _project_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    s = a.sum()
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, a.max()
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    # Distribute remaining mass
    rem = S - res.sum()
    if rem > 0:
        idx = res.argmax()
        res[idx] += rem
    return res


def _grad_polish(a0, budget_s, tau0=1.0, tau_end=0.02, lr0=1.0, verbose=False):
    """Projected gradient descent with soft-max smoothing."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0).astype(np.float64)
    a = a / a.sum() * S
    a = _project_simplex(a, S)
    
    best = a.copy()
    best_mc = np.convolve(a, a).max()
    best_score = evaluate_sequence(a.tolist())
    
    t0 = time.monotonic()
    lr = lr0
    step = 0
    while time.monotonic() - t0 < budget_s:
        el = time.monotonic() - t0
        f = min(1.0, el / max(budget_s, 1e-6))
        tau = tau0 * (tau_end / tau0) ** f
        
        g = _soft_max_grad(a, tau)
        a_new = a - lr * g
        a_new = _project_simplex(a_new, S)
        step += 1
        
        if step % 5 == 0:
            mc = np.convolve(a_new, a_new).max()
            if mc < best_mc - 1e-15:
                best_mc = mc
                best = a_new.copy()
                best_score = evaluate_sequence(best.tolist())
                if lr < lr0 * 8:
                    lr *= 1.03
            else:
                lr = max(0.01, lr * 0.5)
            a = a_new
        else:
            a = a_new
    
    return best, best_score


def _init_chirp(n, seed=0):
    t = np.arange(n, dtype=np.float64) - (n - 1) * 0.5
    a = 1.0 + 0.9 * np.cos(np.pi * t * t / n)
    rng = np.random.default_rng(seed)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative refinement.
Each iteration solves an LP: max sum(b) s.t. conv(a,b) <= M, b >= 0,
then blends the result back into a. Multiple initializations, adaptive
step sizes, and gradient polishing are used for refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _build_indices(n):
    """Pre-compute row/col indices for convolution matrix of length-n sequences."""
    m = 2 * n - 1
    counts = np.array([min(n, m - k) for k in range(n)], dtype=np.int64)
    rows = np.concatenate([np.arange(k, k + c, dtype=np.int64) for k, c in zip(range(n), counts)])
    cols = np.repeat(np.arange(n, dtype=np.int64), counts)
    return rows, cols, counts


def _make_matrix(a, indices):
    """Create sparse convolution matrix from values and pre-computed indices."""
    n = len(a)
    m = 2 * n - 1
    rows, cols, counts = indices
    vals = np.repeat(a, counts)
    return csr_matrix((vals, (rows, cols)), shape=(m, n))


def _solve_lp(a, M, indices):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns b or None."""
    n = len(a)
    A = _make_matrix(a, indices)
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def _lp_search(a0, budget_s, t_init=0.5, M_scale=1.0, verbose=False):
    """Run LP-based iterations with adaptive step size."""

def _soft_max_grad(a, tau):
    """Gradient of tau * log(sum(exp(conv(a,a)/tau))) w.r.t. a."""
    c = np.convolve(a, a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    grad = 2.0 * np.correlate(q, a, 'valid')
    return grad


def _project_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    s = a.sum()
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, a.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    res = np.maximum(a - tau, 0.0)
    rem = S - res.sum()
    if rem > 0:
        idx = res.argmax()
        res[idx] += rem
    return res


def _grad_polish(a0, budget_s, tau0=0.5, tau_end=0.01, lr0=0.8, verbose=False):
    """Projected gradient descent with soft-max smoothing and adaptive lr."""
    n = len(a0)

    a = np.maximum(a0, 0.0).astype(np.float64)
    a = a / a.sum() * S
    a = _project_simplex(a, S)

    best = a.copy()

    t0 = time.monotonic()

        f = min(1.0, el / max(budget_s, 1e-6))
        tau = tau0 * (tau_end / tau0) ** f

        a_new = a - lr * g

def _init_chirp(n, seed=0):
    t = np.arange(n, dtype=np.float64) - (n - 1) * 0.5
    a = 1.0 + 0.9 * np.cos(np.pi * t * t / n)
    rng = np.random.default_rng(seed)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def _init_random(n, seed=0):
    return np.random.default_rng(seed).exponential(1.0, size=n)


def _init_smooth(n, seed=0):
    t = np.linspace(0, 1, n)
