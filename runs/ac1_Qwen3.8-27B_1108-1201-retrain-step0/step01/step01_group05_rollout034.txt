Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses smoothing (log-sum-exp) + L-BFGS with softmax parameterization.
The problem is convex and scale-invariant, so we fix sum(a)=1 via softmax
and minimize max(conv(a,a)) using a smoothed surrogate.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve(x, y):
    """Linear convolution of x and y via FFT."""
    n = len(x) + len(y) - 1
    fsize = 1 << (n - 1).bit_length()
    X = fft(x, fsize)
    Y = fft(y, fsize)
    return np.real(ifft(X * Y))[:n]


def softmax_obj_grad(x, beta):
    """
    Compute smoothed objective and gradient w.r.t. x (softmax params).
    a = softmax(x), so sum(a) = 1, a > 0.
    f_beta = (1/beta) * log(sum_k exp(beta * b_k)) where b = conv(a,a).
    """
    n = len(x)
    
    # Compute a = softmax(x)
    x_shift = x - x.max()
    exp_x = np.exp(x_shift)
    a = exp_x / exp_x.sum()
    
    # Compute b = conv(a, a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    b_full = np.real(ifft(A * A))[:size]
    b_full = np.maximum(b_full, 0.0)  # numerical safety
    
    # Smooth objective
    beta_b = beta * b_full
    max_bb = beta_b.max()
    shifted = beta_b - max_bb
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    f_val = (max_bb + np.log(sum_exp)) / beta
    
    # Softmax weights
    w = exp_shifted / sum_exp
    
    # Gradient of f w.r.t. a
    # grad_a[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # = c[n-1:2n-1][::-1] where c = conv(a, w[::-1])
    w_rev = w[::-1]
    c = fft_convolve(a, w_rev)
    grad_a = 2.0 * c[n-1:2*n-1][::-1]
    
    # Gradient w.r.t. x via softmax Jacobian
    grad_x = a * (grad_a - np.dot(grad_a, a))
    
    return f_val, grad_x


def run_smoothed_lbfgs(x_init, n, betas, max_iter_per_beta=200, tol=1e-10):
    """
    Run L-BFGS for a sequence of beta values (increasing).
    Returns the best x found.
    """
    x = x_init.copy()
    start = time.monotonic()
    
    for beta in betas:
        res = minimize(
            softmax_obj_grad,
            x,
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': max_iter_per_beta,
                'ftol': tol,
                'gtol': 1e-8,
                'iprint': -1,
                'maxls': 50,
            }
        )
        x = res.x
        a = np.exp(x - x.max())
        a = a / a.sum()
        b = fft_convolve(a, a)
        print(f"  beta={beta:.1f}: f_smooth={res.fun:.8f}, max_b={b.max():.8f}")
        
        if time.monotonic() - start > 1800:
            break
    
    return x

    """Convert softmax params to a sequence (sum=1)."""
    x_shift = x - x.max()
    a = np.exp(x_shift)
    return a / a.sum()

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses log-sum-exp smoothing + L-BFGS with softmax parameterization.
The problem is convex and scale-invariant; we fix sum(a)=1 via softmax
and minimize the peak of the self-convolution using a smooth surrogate,
with a geometric beta schedule from broad to sharp.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve(x, y):
    """Linear convolution of x and y via FFT."""
    n = len(x) + len(y) - 1
    fsize = 1 << (n - 1).bit_length()
    X = fft(x, fsize)
    Y = fft(y, fsize)
    return np.real(ifft(X * Y))[:n]


def fft_self_conv(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]


def softmax_obj_grad(x, beta):
    """
    Smoothed objective and gradient w.r.t. x (softmax params).
    a = softmax(x), sum(a) = 1.
    f_beta = (1/beta) * log(sum_k exp(beta * b_k)), b = conv(a,a).
    Returns (f_val, grad_x).
    """
    n = len(x)
    x_shift = x - x.max()
    exp_x = np.exp(x_shift)
    a = exp_x / exp_x.sum()

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    b_full = np.real(ifft(A * A))[:size]
    b_full = np.maximum(b_full, 0.0)

    beta_b = beta * b_full
    max_bb = beta_b.max()
    shifted = beta_b - max_bb
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    f_val = (max_bb + np.log(sum_exp)) / beta

    w = exp_shifted / sum_exp

    # grad_a[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # = c[n-1:2n-1][::-1] where c = conv(a, w[::-1])
    w_rev = w[::-1]
    c = fft_convolve(a, w_rev)
    grad_a = 2.0 * c[n - 1:2 * n - 1][::-1]

    # Softmax Jacobian: grad_x = a * (grad_a - dot(grad_a, a))
    grad_x = a * (grad_a - np.dot(grad_a, a))

    return f_val, grad_x


def direct_score(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fft_self_conv(a)
    return 2.0 * n * b.max()


def run_smoothed_lbfgs(x_init, n, beta_list, max_iter_total=None, time_limit=None):
    """
    Sequential L-BFGS with increasing beta.
    Returns best x found.
    """
    x = x_init.copy()
    start = time.monotonic()
    best_x = x.copy()
    best_direct = np.inf

    for beta in beta_list:
        if time_limit is not None and time.monotonic() - start > time_limit:
            break
        t0 = time.monotonic()
        res = minimize(
            softmax_obj_grad,
            x,
            args=(beta,),
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': max_iter_total if max_iter_total else 300,
                'ftol': 1e-14,
                'gtol': 1e-10,
                'iprint': -1,
                'maxls': 50,
            }
        )
        x = res.x
        # Compute direct score
        x_shift = x - x.max()
        a = np.exp(x_shift)
        a = a / a.sum()
        dscore = direct_score(a)
        if dscore < best_direct:
            best_direct = dscore
            best_x = x.copy()
        elapsed = time.monotonic() - t0
        print(f"    beta={beta:.0f}: smooth={res.fun:.8f}, direct={dscore:.8f}, "
              f"iter={res.nit}, time={elapsed:.1f}s")

    return best_x, best_direct
