"""
LSE-smoothed gradient descent with adaptive temperature and 
simplex projection to minimize the normalized autocorrelation peak.
"""

import numpy as np
import time


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_true_score(a):
    """Compute the true score: 2*n*max(convolve(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (S * S)


def _compute_lse_and_grad(a, tau):
    """
    Compute LSE-smoothed max of autocorrelation and its gradient.
    Returns (lse_value, gradient).
    """
    n = len(a)
    b = np.convolve(a, a)
    tb = tau * b
    max_tb = np.max(tb)
    exp_vals = np.exp(tb - max_tb)
    w = exp_vals / np.sum(exp_vals)  # softmax weights, length 2n-1
    
    # LSE value
    lse_val = (max_tb + np.log(np.sum(exp_vals))) / tau
    
    # Gradient using FFT-based cross-correlation
    # grad[j] = 2 * sum_m w[m+j] * a[m] for j=0,...,n-1
    size = 1
    while size < (2 * n - 1) + n - 1:
        size *= 2
    
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    cross = np.fft.irfft(np.conj(A) * W, size)
    grad = 2.0 * cross[:n]
    
    return lse_val, grad, w


def _smoothed_gradient_descent(a, deadline, tau_start=50.0, tau_end=300.0, 
                                eta_const=1.0, min_tau=30.0):
    """
    LSE-smoothed gradient descent with adaptive temperature.
    Uses the average gradient norm rule for step size.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    a /= S
    
    best = a.copy()
    best_score = _compute_true_score(best)
    
    G = 0.0  # cumulative squared gradient norm
    n_iter = 0
    t0 = time.monotonic()
    total_time = deadline - t0
    last_print = t0
    
    while time.monotonic() < deadline:
        # Adaptive temperature
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / total_time)
        # Use a slower schedule: tau increases as sqrt(frac)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        
        # Compute LSE and gradient
        lse_val, grad, w = _compute_lse_and_grad(a, tau)
        
        # Center gradient for sum constraint
        grad -= np.mean(grad)
        
        # Update G
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq > 1e-30:
            G += grad_norm_sq
        
        # Step size
        if G > 1e-30:
            eta = eta_const / np.sqrt(G)
        else:
            break
        
        # Take step and project
        a_new = a - eta * grad
        a_new = _project_simplex(a_new, 1.0)
        
        # Check true score periodically
        if n_iter % 100 == 0:
            score = _compute_true_score(a_new)
            if score < best_score:
                best = a_new.copy()
                best_score = score
        
        a = a_new
        n_iter += 1
        
        if time.monotonic() - last_print > 5.0:
            print(f"  SGD iter={n_iter}, tau={tau:.1f}, G={G:.4f}, eta={eta:.6f}, best={best_score:.6f}")
            last_print = time.monotonic()
    
    return best, best_score


def _subgradient_refine(a, deadline, M=5, eta_init=None):
    """
    Subgradient refinement using top-M peaks.
    """
    n = len(a)
    if eta_init is None:
        eta_init = 0.5 / np.sqrt(n)
    
    a = np.asarray(a, dtype=np.float64).copy()
    a /= np.sum(a)
    
    best = a.copy()
    best_score = _compute_true_score(best)
    eta = eta_init
    G = 0.0
    n_iter = 0
    t0 = time.monotonic()
    last_print = t0
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        
        # Get top M peaks
        if M >= len(b):
            k_idx = np.arange(len(b))
            b_sel = b
        else:
            k_idx = np.argpartition(b, -M)[-M:]
            b_sel = b[k_idx]
        
        # Softmax weights over top-M
        tb = b_sel * 20.0  # moderate temperature for weighting
        tb -= np.max(tb)
        w = np.exp(tb) / np.sum(np.exp(tb))
        
        # Compute gradient from top-M peaks
        grad = np.zeros(n)
        for i, k in enumerate(k_idx):
            if w[i] < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w[i] * 2.0 * a[k - idx]
        
        # Center
        grad -= np.mean(grad)
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        G += grad_norm_sq
        eta = 1.0 / np.sqrt(G)

        a_new = _project_simplex(a_new, 1.0)

                best = a_new.copy()
                best_score = score

    return best, best_score


def _make_perturbation(a, rng, scale=0.05):
    """Add smooth random perturbation."""
    n = len(a)

    noise = np.zeros(n)

        f = rng.integers(1, max(2, n // 4))

    noise_max = np.max(np.abs(noise))
    if noise_max > 1e-10:
        noise /= noise_max
    a_new = a + scale * noise
    return _project_simplex(a_new, 1.0)

import numpy as np
import time


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_true_score(a):
    """Compute 2*n*max(convolve(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * n * np.max(b) / (S * S))


def _lse_gradient(a, tau):
    """
    Compute the gradient of the LSE-smoothed max autocorrelation.
    Returns (smoothed_value, gradient).
    """
    n = len(a)
    b = np.convolve(a, a)
    tb = tau * b
    max_tb = np.max(tb)
    exp_vals = np.exp(tb - max_tb)
    w = exp_vals / np.sum(exp_vals)
    
    lse_val = (max_tb + np.log(np.sum(exp_vals))) / tau
    
    # FFT-based gradient: grad[j] = 2 * sum_m a[m] * w[m+j]
    a_rev = a[::-1]
    size = 1
    while size < n + (2 * n - 1) - 1:
        size *= 2
    A = np.fft.rfft(a_rev, size)
    W = np.fft.rfasta(w, size)
    conv = np.fft.irfft(A * W, size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    
    return lse_val, grad

```python
"""
LSE-smoothed gradient descent with FFT-accelerated gradients to minimize
the normalized autocorrelation peak of a non-negative sequence.
Uses adaptive temperature annealing and multi-phase refinement.
"""

import numpy as np
import time


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _autocorr(a):
    """Compute autocorrelation (linear convolution) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def _true_score(a):
    """Compute 2*n*max(autocorr)/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = _autocorr(a)
    return float(2.0 * n * np.max(b) / (S * S))


def _lse_grad(a, tau):
    """
    Compute LSE-smoothed max autocorrelation value and its gradient.
    
    The LSE smoothing: f(a) = (1/tau) * log(sum_k exp(tau * b_k))
    where b = autocorrelation of a.
    
    The gradient: grad_j = 2 * sum_k w_k * a_{k-j} (valid terms)
    where w_k = exp(tau*b_k) / sum_m exp(tau*b_m) (softmax weights).
    
    Uses FFT for O(n log n) gradient computation.
    """
    n = len(a)
    b = _autocorr(a)
    
    # Softmax weights
    tb = tau * b
    max_tb = np.max(tb)
    exp_vals = np.exp(tb - max_tb)
    w = exp_vals / np.sum(exp_vals)
    
    # LSE value
    lse_val = (max_tb + np.log(np.sum(exp_vals))) / tau
    
    # Gradient via FFT cross-correlation:
    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]
    # = 2 * (a_rev * w)[j + n - 1] where a_rev = a[::-1]
    a_rev = a[::-1]
    size = 1
    while size < n + (2 * n - 1) - 1:
        size *= 2
    A = np.fft.rfft(a_rev, size)
    W = np.fft.rfft(w, size)
    conv = np.fft.irfft(A * W, size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    
    return lse_val, grad


def _subgrad_grad_top(a, k_indices):
    """
    Compute subgradient from a set of peak indices.
    grad[j] = 2 * sum_{k in k_indices} a[k-j] (valid terms) / len(k_indices)
    """
    n = len(a)
    grad = np.zeros(n)
    for k in k_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    grad /= len(k_indices)
    return grad


def _lse_gd_main(a, deadline, tau_start=20.0, tau_end=500.0, eta_const=1.0,
                 score_check_interval=20, restart_iters=300, restart_scale=0.02,
                 seed=123):
    """
    Main LSE-smoothed gradient descent loop.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    a /= S
    
    best = a.copy()
    best_score = _true_score(best)
    G = 0.0
    n_iter = 0
    last_improve = 0
    t0 = time.monotonic()
    total_time = deadline - t0
    last_print = t0
    
    print(f"  Starting LSE-GD: n={n}, initial_score={best_score:.6f}")
    
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / total_time)
        # Quadratic tau schedule
        tau = tau_start + (tau_end - tau_start) * frac * frac
        
        lse_val, grad = _lse_grad(a, tau)
        
        # Center gradient for sum=1 constraint
        grad -= np.mean(grad)
        
        # Average gradient norm rule
        g2 = np.dot(grad, grad)
        if g2 < 1e-30:
            break
        G += g2
        eta = eta_const / np.sqrt(G)
        
        # Step + project
        a_new = a - eta * grad
        a_new = _project_simplex(a_new, 1.0)
        
        # Score check
        if n_iter % score_check_interval == 0:
            s = _true_score(a_new)
            if s < best_score:
                best = a_new.copy()
                best_score = s
                last_improve = n_iter
        
        a = a_new
        n_iter += 1
        
        # Restart if stuck
        if n_iter - last_improve > restart_iters:
            pert = best + restart_scale * (rng.random(n) - 0.5)
            a = _project_simplex(pert, 1.0)
            G = max(G * 0.1, 1e-10)  # Effectively reheat
            last_improve = n_iter
        
        # Progress report
        if time.monotonic() - last_print > 5.0:
            print(f"    iter={n_iter}, tau={tau:.1f}, eta={eta:.2e}, score={best_score:.6f}")
            last_print = time.monotonic()
    
    return best, best_score
