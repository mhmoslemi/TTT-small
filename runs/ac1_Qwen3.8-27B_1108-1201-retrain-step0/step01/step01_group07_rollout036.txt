Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses Adam-optimized soft-max gradient descent with backtracking line search,
multi-resolution initialization, coordinate descent polishing, and
multi-start strategy across different sequence lengths.

import numpy as np

def compute_softmax_and_grad(a, tau):
    """Compute soft-max of conv(a,a) and its gradient wrt a."""
    c = np.convolve(a, a)
    c_max = c.max()
    c_shift = (c - c_max) / tau
    # Clip to avoid overflow
    c_shift = np.clip(c_shift, -500, 500)
    exp_c = np.exp(c_shift)
    exp_sum = exp_c.sum()
    p = exp_c / exp_sum
    soft_max_val = c_max + tau * np.log(exp_sum)
    # Gradient: 2 * cross_correlate(p, a)
    grad = 2.0 * np.correlate(p, a, 'valid')
    return soft_max_val, grad, c


def project_nonneg_sum(a, S):
    """Project onto {x >= 0, sum(x) = S} via bisection."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        return np.full(a.shape, S / a.size)
    if s == S:
        return a
    if s < S:
        # Need to add (S-s) to make sum = S, but we also need non-negativity
        # Just add uniformly (simplest)
        return a + (S - s) / a.size
    
    # s > S: need to subtract
    lo, hi = 0.0, a.max()
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    res = np.maximum(a - t, 0.0)
    # Fix rounding error
    diff = S - res.sum()
    if diff > 0:
        # Add to the largest element
        idx = res.argmax()
        res[idx] += diff
    elif diff < 0:
        # Subtract from the largest positive element
        mask = res > 0
        idx = np.where(mask)[0][np.argmax(res[mask])]
        res[idx] += diff  # diff is negative
    return res


def init_chirp(n, seed=0):
    """Chirp-based initialization."""
    t = np.arange(n, dtype=np.float64)
    # Use a raised-cosine times a chirp
    envelope = 0.5 * (1 + np.cos(2 * np.pi * t / n))
    a = 0.5 + 0.5 * envelope * np.cos(2 * np.pi * t * t / (4*n))
    rng = np.random.default_rng(seed)
    a += 0.1 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_random(n, seed=0):
    """Random initialization."""
    return np.random.default_rng(seed).exponential(1.0, size=n)


def init_fourier(n, seed=0):
    """
    Fourier-based initialization: create a sequence with approximately flat 
    power spectrum (which implies flatter autocorrelation).
    """
    rng = np.random.default_rng(seed)
    # Create a random phase spectrum with flat magnitude
    N = 2 * n
    spectrum = np.zeros(N, dtype=np.complex128)
    # Flat magnitude with random phase
    phases = rng.uniform(0, 2*np.pi, size=N)
    magnitudes = np.ones(N)
    spectrum = magnitudes * np.exp(1j * phases)
    # Make it conjugate symmetric for real signal
    spectrum[N//2] = 1.0  # Nyquist
    for k in range(1, N//2):
        spectrum[N-k] = np.conj(spectrum[k])
    a = np.real(np.fft.ifft(spectrum))[:n]
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    return a

    t = 0

    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(t_budget, 1e-6))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Compute gradient
        soft_val, grad, c = compute_softmax_and_grad(a, tau)
        
        t += 1

        v = beta2 * v + (1 - beta2) * (grad ** 2)
