```python
import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def solve_lp_for_g(f, M, n):
    """
    Solve: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    (f*g)[k] = sum_i f[i] * g[k-i] (with g[j] = 0 for j outside [0,n-1])
    """
    # A matrix: each row k gives the coefficients of g in (f*g)[k]
    # (f*g)[k] = sum_i f[i] * g[k-i] for valid i, k-i
    # For a given k, the valid (i, j) pairs are i in [0, n-1], j = k-i in [0, n-1]
    # So: (f*g)[k] = sum_{j: 0<=k-j<n} f[k-j] * g[j]
    
    # Build the constraint matrix A (2n-1 rows, n columns)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    
    # c = -1 (we want to maximize sum(g), which is minimize -sum(g))
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    
    if result.success:
        return result.x
    return None
```

```python
"""
Hybrid optimization: LP-based peak reduction + L_p norm gradient descent + local search.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def build_A_matrix(f, n):
    """Build the constraint matrix A for the LP. A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2*n-1, n))
    # Vectorized construction
    # For row k, column j: A[k,j] = f[k-j] if 0 <= k-j < n
    rows = np.arange(2*n-1)[:, None]
    cols = np.arange(n)[None, :]
    idx = rows - cols  # idx[k,j] = k-j
    valid = (idx >= 0) & (idx < n)
    A[valid] = f[idx[valid]]
    return A

def solve_lp(f, M, n):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = build_A_matrix(f, n)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if result.success:
        return result.x
    return None

def lp_optimize(f_init, time_limit, rng):
    """
    LP-based optimization: iteratively find g that maximizes sum subject to
    cross-correlation bound, then blend.
    """
    f = f_init.copy()
    n = len(f)
    S = np.sum(f)
    f = f / S  # normalize to sum = 1
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    
    # Build A once (it changes as f changes, so we need to rebuild)
    alpha = 0.5  # initial blending parameter
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        # Solve LP
        g = solve_lp(f, M, n)
        if g is None:
            break
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        # Rescale g to sum = 1
        g = g / sum_g
        
        # Blend
        f_new = (1 - alpha) * f + alpha * g
        f_new = np.clip(f_new, 0, None)
        sum_new = np.sum(f_new)
        if sum_new > 0:
            f_new = f_new / sum_new
        
        new_score = score_fast(f_new)
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
        
        f = f_new
        iteration += 1
        
        # Decay alpha
        alpha = max(0.05, 0.5 * (0.95 ** (iteration // 10)))
        
        if iteration % 50 == 0:
            print(f"    LP iter {iteration}: score={best_score:.6f}, alpha={alpha:.3f}, t={time.monotonic()-t0:.1f}s")
    
    return best_f, best_score
```

```python
    # This is 2 * corr(w, a_reversed) at appropriate position
    a_rev = a[::-1]
    corr = fft_convolve(w, a_rev)
    grad_a = 2.0 * corr[len(a)-1 : 2*len(a)-1]
    # Project gradient to account for normalization
    grad_a = grad_a - np.sum(a * grad_a)  # zero-mean in the tangent space
    # Gradient w.r.t. theta: da_i/dtheta_j = a_i * (delta_ij - a_j)
    # grad_theta_j = sum_i grad_a_i * da_i/dtheta_j = grad_a_j * a_j - a_j * sum_i grad_a_i * a_i
    # Since sum(a*grad_a) = 0 after projection, grad_theta_j = a_j * grad_a_j
    grad_theta = a * grad_a
    return val, grad_theta
```

```python
"""
Hybrid optimization combining LP-based peak reduction, L-p norm gradient descent,
and targeted local search to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: uses linear programming to find optimal blending directions,
which is fundamentally different from pure gradient methods.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score computation matching evaluate_sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def build_A_matrix(f):
    """Build constraint matrix for LP: A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    rows = np.arange(2 * n - 1)[:, None]
    cols = np.arange(n)[None, :]
    idx = rows - cols
    valid = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = f[idx[valid]]
    return A


def solve_lp_for_g(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_A_matrix(f)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if result.success:
        return result.x
    return None


def lp_optimize(f_init, time_limit, rng):
    """
    LP-based optimization: iteratively solve LP to find optimal g,
    then blend f <- (1-t)f + t*g.
    """
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    if S < 1e-10:
        return f, np.inf
    f = f / S
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    alpha = 0.7
    iteration = 0
    stuck_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)
        if M < 1e-30:
            break

        g = solve_lp_for_g(f, M)
        if g is None:
            break

        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break

        g = g / sum_g

        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.clip(f_new, 0.0, None)
        sum_new = np.sum(f_new)
        if sum_new > 1e-10:
            f_new = f_new / sum_new
            new_score = score_fast(f_new)
        else:
            break

        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        f = f_new
        iteration += 1

        # Adaptive alpha: decrease when stuck, increase when making progress
        if stuck_count > 20:
            alpha = max(0.05, alpha * 0.9)
        elif stuck_count == 0:
            alpha = min(0.8, alpha * 1.02)

        if iteration % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {iteration}: best={best_score:.6f}, alpha={alpha:.3f}, "
                  f"stuck={stuck_count}, t={elapsed:.1f}s")

    return best_f, best_score


def lp_optimize_symmetric(f_init, time_limit, rng):
    """LP optimization with symmetric sequence constraint."""
    f = f_init.copy().astype(np.float64)
    n = len(f)
    if n % 2 != 0:
        return f, np.inf

    # Work with first half
    half = n // 2
    # Make symmetric
    f = 0.5 * (f + f[::-1])
    S = np.sum(f)
    f = f / S

    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    alpha = 0.5
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        # Build A matrix for symmetric case
        # g is determined by its first half: g[j] = g[n-1-j]
        # (f*g)[k] = sum_j f[k-j] * g[j] (with symmetric g)
        # = sum_{j < n/2} f[k-j]*g[j] + f[k-(n-1-j)]*g[j]  (for j in second half, g[j]=g[n-1-j])
        # Let's just build it explicitly
        A = np.zeros((2 * n - 1, half))
        for k in range(2 * n - 1):
            for j in range(half):
                val = 0.0
                if 0 <= k - j < n:
                    val += f[k - j]
                # Symmetric part: g[n-1-j] = g[j]
                j2 = n - 1 - j
                if 0 <= k - j2 < n and j2 != j:
                    val += f[k - j2]
                A[k, j] = val

        conv = fft_convolve(f, f)
        M = np.max(conv)

        c_obj = -A.sum(axis=0)  # maximize sum of full g = 2*sum of half g
        b = np.full(2 * n - 1, M)
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * half, method='highs')
        if not result.success:
            break

        g_half = result.x
        # Reconstruct full g
        g = np.zeros(n)
        g[:half] = g_half
        g[half:] = g_half[::-1]

        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        g = g / sum_g

        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.clip(f_new, 0.0, None)
        f_new = 0.5 * (f_new + f_new[::-1])  # enforce symmetry
        sum_new = np.sum(f_new)
        if sum_new > 1e-10:
            f_new = f_new / sum_new
            new_score = score_fast(f_new)
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()

        f = f_new
        iteration += 1
        alpha = max(0.1, 0.5 * (0.97 ** (iteration // 20)))

        if iteration % 30 == 0:
            print(f"      SymLP iter {iteration}: best={best_score:.6f}, t={time.monotonic()-t0:.1f}s")

    return best_f, best_score


def gradient_lp_hybrid(f_init, time_limit, p_val=20.0, rng=None):
    """
    Gradient descent on L_p norm of convolution with periodic LP restarts.
    Optimizes over free parameters (no normalization constraint during optimization,
    normalizes at evaluation).
    """
    if rng is None:
        rng = np.random.default_rng(0)

    a = f_init.copy().astype(np.float64)
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    a = a / np.sum(a) * S_target

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()

    # Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.5
    step = 0
    lp_every = 100  # do LP every 100 gradient steps

    while time.monotonic() - t0 < time_limit:
        # Compute L_p objective and gradient
        c = fft_convolve(a, a)
        c_clipped = np.maximum(c, 1e-30)
        c_p = c_clipped ** p_val
        sum_cp = np.sum(c_p)
        obj = sum_cp ** (1.0 / p_val)

        # Gradient of L_p norm w.r.t. a
        # d(L_p)/da_i = (1/L_p^(p-1)) * sum_k c_k^(p-1) * dc_k/da_i
        # dc_k/da_i = 2 * a[k-i] (for valid range)
        w = c_clipped ** (p_val - 1) / sum_cp  # normalized weights
        w_sum = np.sum(w)
        w = w / w_sum  # ensure weights sum to 1

        # grad_a[i] = 2 * sum_k w[k] * a[k-i]
        # This is 2 * (w * a_flipped) correlated appropriately
        a_flip = a[::-1]
        corr = fft_convolve(w, a_flip)
        grad_a = 2.0 * corr[n - 1:2 * n - 1]

        # Scale gradient by 1/L_p^(p-1)
        grad_a = grad_a / (obj ** (p_val - 1))

        # Normalize: we want to keep sum(a) = S_target
        # Project gradient onto tangent space of {sum = S_target}
        mean_grad = np.mean(grad_a)
        grad_a = grad_a - mean_grad

        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad_a
        v = beta2 * v + (1 - beta2) * (grad_a * grad_a)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.01, None)

        # Renormalize
        a = a / np.sum(a) * S_target

        # Periodic LP step
        if step % lp_every == 0:
            conv = fft_convolve(a, a)
            M = np.max(conv)
            g = solve_lp_for_g(a / S_target, M / (S_target * S_target))
            if g is not None:
                sum_g = np.sum(g)
                if sum_g > 1e-10:
                    g = g / sum_g * S_target
                    a = 0.5 * a + 0.5 * g
                    a = np.clip(a, 0.01, None)
                    a = a / np.sum(a) * S_target

        # Evaluate
        sc = score_fast(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Decay learning rate
        lr = max(0.01, 0.5 / (1.0 + 0.01 * step))

        if step % 500 == 0:
            print(f"      Grad iter {step}: score={sc:.6f}, best={best_score:.6f}, lr={lr:.4f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def local_search_polish(a_init, time_limit, rng):
    """Targeted local search with peak reduction."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = np.sum(a)
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    temp = 0.002
    cool = 0.9999
    current_score = best_score

    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        batch = 10  # do 10 perturbations per check
        for _ in range(batch):
            candidate = a.copy()
            r = rng.random()

            if r < 0.4:
                # Peak-targeted: reduce top contributors to the max peak
                conv = fft_convolve(a, a)
                peak_idx = int(np.argmax(conv))
                # Find which a[i] contribute most to this peak
                contribs = np.zeros(n)
                for i in range(n):
                    j = peak_idx - i
                    if 0 <= j < n:
                        contribs[i] = a[i] * a[j]
                top_k = max(3, n // 12)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
                # Boost some others
                other = np.setdiff1d(np.arange(n), top_idx)
                if len(other) > 5:
                    boost_idx = rng.choice(other, size=min(8, len(other)), replace=False)
                    candidate[boost_idx] *= np.exp(rng.normal(0.04, 0.02, size=len(boost_idx)))
            elif r < 0.6:
                # Neighbor swap
                i = rng.integers(0, n - 1)
                delta = rng.normal(0, 0.03)
                candidate[i] = max(0.01, candidate[i] * np.exp(delta))
                candidate[i + 1] = max(0.01, candidate[i + 1] * np.exp(-delta * 0.8))
            elif r < 0.8:
                # Block perturbation
                width = max(2, int(rng.integers(2, max(3, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.025, size=width))
            else:
                # Small global
                candidate *= np.exp(rng.normal(0, 0.008, size=n))

            candidate = np.clip(candidate, 0.0, 1000.0)
            new_score = score_fast(candidate)

            if new_score < current_score:
                a = candidate
                current_score = new_score
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
            elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-15)):
                a = candidate
                current_score = new_score

        temp *= cool
        iter_count += 1
        if iter_count % 50 == 0:
            print(f"      PS iter {iter_count}: best={best_score:.6f}, temp={temp:.6f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def make_good_init(n, rng):
    """Create a good initial sequence using chirp + random."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.3 * np.cos(np.pi * t * t / n + rng.uniform(0, np.pi))
    a += 0.15 * np.cos(2 * np.pi * 3 * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.1 * np.cos(2 * np.pi * 7 * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_barker_like(n, rng):
    """Create initialization inspired by Barker/Lempel-Golay sequences."""
    t = np.arange(n, dtype=np.float64)
    # Use a product of low-frequency cosines (like a windowed signal)
    a = np.ones(n)
    for k in range(1, 8):
        a *= (1.0 + 0.1 * np.cos(2 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi)))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)

        print(f"\n=== Phase 5: Final refinement ({t_final:.0f}s) ===")
        nf = len(best_seq)
        af, sf = gradient_lp_hybrid(best_seq, t_final, p_val=30.0, rng=rng)
        sf_actual = score_fast(af)
        if sf_actual < best_score:
            best_score = sf_actual
            best_seq = af.copy()
            print(f"  Final gradient: {best_score:.6f} *** New best ***")
        else:
            print(f"  Final gradient: {sf_actual:.6f} (no improvement)")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        # Keep the one that actually evaluates better
        if final_score < best_score:
            best_score = final_score
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
