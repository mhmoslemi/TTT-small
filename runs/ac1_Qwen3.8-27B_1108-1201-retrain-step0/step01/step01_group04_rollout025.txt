"""
Step-function search for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses smoothed-max gradient descent with adaptive temperature, peak-directed
coordinate moves, LP best-response, frequency-domain initialization,
and multi-scale refinement.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.optimize import minimize

_LP_NMAX = 800


def _fft_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def _fft_len_grad(n):
    return int(next_fast_len(max(1, 3 * int(n) - 2)))


def conv_sq(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        return np.full(max(1, int(p.size)), 1.0 / max(1, int(p.size)), dtype=np.float64)
    return p / s


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def smoothed_max_and_grad(p, T_factor):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return 0.0, np.empty(0)

    L = _fft_len_grad(n)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]

    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax <= 0.0:
        return 0.0, np.zeros(n, dtype=np.float64)

    T = max(float(T_factor) * cmax, 1e-300)
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        s = 1.0
    w = e / s

    surr = float(cmax + T * np.log(s))

    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2:n - 2:-1]
    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    G = np.maximum(G, 0.0)
    g = 2.0 * G
    return surr, g


def lp_best_response(p, budget_s=30.0):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 0 or n > _LP_NMAX or budget_s <= 0.5:
        return None

    c = conv_sq(p)
    M = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(M) or M <= 0.0:
        return None

    f = p + 1e-14
    fs = float(f.sum())
    if fs <= 0:
        return None
    f = f / fs

    try:
        mid = np.arange(n, dtype=np.int32)
        rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel().astype(np.int32)
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(f, n)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs",
                      options={"disp": False, "time_limit": max(0.5, budget_s)})
    except Exception:
        return None

    if res is None or not res.success:
        g = getattr(res, "x", None)
        if g is None:
            return None
    else:
        g = res.x

    if g is None:
        return None
    g = np.asarray(g, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-12:
        return None
    return g / s


def make_freq_init(n, rng, n_freqs=8):
    x = np.linspace(0.0, 1.0, n)
    p = np.ones(n)
    for _ in range(n_freqs):
        f = int(rng.integers(1, max(2, n // 20)))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(0.1, 0.6))
        p += amp * np.sin(2 * np.pi * f * x + phase)
    p = np.maximum(p, 0.01)
    return safe_normalize(p)


def make_two_hump(n, gap, width):
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-((x - 0.15) ** 2) / (2 * width ** 2))
    p += np.exp(-((x - 0.85) ** 2) / (2 * width ** 2))
    p += 0.1 * np.ones(n)
    return safe_normalize(p)


def make_w_shape(n, depth):
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + depth * (2 * x - 1) ** 2
    p *= 1.0 + depth * 0.3 * np.cos(4 * np.pi * x)
    p = np.maximum(p, 0.01)
    return safe_normalize(p)


def make_geometric(n, s):
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-s * x)
    return safe_normalize(p)


def make_beta(n, a, b):
    x = np.linspace(1e-6, 1 - 1e-6, n)
    p = np.power(x, a) * np.power(1 - x, b)
    p = np.maximum(p, 1e-10)
    return safe_normalize(p)


def make_comb(n, ratio):
    p = np.ones(n)
    if n > 1:
        p[::2] = ratio
    return safe_normalize(p)


def generate_inits(n, rng):
    inits = []
    inits.append(np.full(n, 1.0 / n))
    inits.append(make_geometric(n, 1.0))
    inits.append(make_geometric(n, 3.0))
    inits.append(make_geometric(n, 0.3))
    inits.append(make_two_hump(n, 0.5, 0.15))
    inits.append(make_two_hump(n, 0.5, 0.08))
    inits.append(make_w_shape(n, 0.5))
    inits.append(make_w_shape(n, 1.0))
    inits.append(make_beta(n, 0.5, 0.5))
    inits.append(make_beta(n, 1.0, 3.0))
    inits.append(make_beta(n, 3.0, 1.0))
    inits.append(make_comb(n, 0.3))
    inits.append(make_comb(n, 0.6))
    for _ in range(10):
        inits.append(make_freq_init(n, rng, int(rng.integers(4, 12))))
    for _ in range(5):
        x = np.linspace(0, 1, n)
        p = 1.0
        for _ in range(int(rng.integers(2, 5))):
            mu = float(rng.uniform(0.2, 0.8))
            sig = float(rng.uniform(0.03, 0.15))
            w = float(rng.uniform(0.5, 1.5))
            p += w * np.exp(-((x - mu) ** 2) / (2 * sig ** 2))
        p = np.maximum(p, 0.01)
        inits.append(safe_normalize(p))
    return inits


def optimize_local(p0, rng, time_limit, label=""):
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    if n == 0:
        return p, np.inf

    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score
    prev_p = p.copy()

    end = time.monotonic() + max(0.1, float(time_limit))
    T_factor = 0.3
    iter_no = 0
    since_improve = 0
    momentum = 0.0

    # Try LP first for short sequences
    if n <= _LP_NMAX and time_limit > 20:
        lp_budget = min(25.0, time_limit * 0.2)
        g = lp_best_response(p, lp_budget)
        if g is not None:
            for t in (1.0, 0.8, 0.6, 0.4, 0.25, 0.1):
                q = project_simplex((1 - t) * p + t * g)
                sc = internal_score(q)
                if sc < best_score:
                    best_score = sc
                    best_p = q.copy()
                    p = q.copy()
                    prev_p = q.copy()
                    current_score = sc
                    break

    while time.monotonic() < end:
        iter_no += 1
        # Adaptive temperature: start high, decay, with kicks
        T_factor = max(0.003, 0.3 * (0.9995 ** iter_no))
        if since_improve > 200:
            T_factor = max(0.002, T_factor * 0.8)
        if since_improve > 500:
            T_factor = max(0.001, T_factor * 0.5)

        surr, g = smoothed_max_and_grad(p, T_factor)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        accepted = False
        gmax = float(np.max(np.abs(g))) if g.size else 0.0
        p_scale = max(1.0 / n, 0.02)

        if gmax > 1e-300:
            base = p_scale / gmax
            lrs = (0.3 * base, 0.1 * base, 0.03 * base, 0.006 * base, 0.001 * base)
        else:
            lrs = (0.05, 0.02, 0.005)

        if iter_no > 60:
            momentum = min(0.4, 0.3 * (1 - 0.999 ** (iter_no - 60)))

        for lr in lrs:
            v = p - lr * g
            if momentum > 0:
                v = v + momentum * (p - prev_p)
            pt = project_simplex(v)
            sc = internal_score(pt)
            surr_t, _ = smoothed_max_and_grad(pt, T_factor)

            if np.isfinite(sc) and (surr_t < surr - 1e-13 or sc < best_score - 1e-12 or
                                    (sc < current_score - 1e-12 and surr_t < surr + 0.5)):
                if sc < best_score:
                    best_score = sc
                    best_p = pt.copy()
                    since_improve = 0
                else:
                    since_improve += 1
                current_score = sc
                prev_p = p
                p = pt
                accepted = True
                break

        if not accepted:
            # Peak-directed move
            if n > 1:
                c = conv_sq(p)
                k_star = int(np.argmax(c))
                # Gradient of c[k_star] w.r.t. a
                # d(c[k])/d(a[j]) = 2*a[k-j] if 0<=k-j<n
                lo = max(0, k_star - n + 1)
                hi = min(n - 1, k_star)
                grad_peak = np.zeros(n)
                for j in range(lo, hi + 1):
                    grad_peak[j] = 2.0 * p[k_star - j]
                # Find best coordinate to decrease (high grad) and increase (low/zero grad)
                j_high = int(np.argmax(grad_peak))
                # Find j with zero or lowest gradient that can accept mass
                candidates = np.where(grad_peak < 0.1 * grad_peak[j_high])[0]
                if len(candidates) > 0:
                    j_low = int(candidates[int(rng.integers(len(candidates)))])
                else:
                    j_low = int(np.argmin(grad_peak))

                if j_high != j_low and p[j_high] > 1e-8:
                    for frac in (0.15, 0.05, 0.02, 0.005):
                        delta = frac * p[j_high]
                        pt = p.copy()
                        pt[j_high] -= delta
                        pt[j_low] += delta
                        pt = project_simplex(pt)
                        sc = internal_score(pt)
                        if sc < best_score:
                            best_score = sc
                            best_p = pt.copy()
                            since_improve = 0
                            current_score = sc
                            prev_p = p
                            p = pt
                            accepted = True
                            break
                        elif sc < current_score * 1.01:
                            pass  # don't accept worse than current for peak moves

            # Random perturbations
            if not accepted:
                sigmas = (0.5 / n, 0.1 / n, 0.02 / n, 0.004 / n)
                if since_improve > 150:
                    sigmas = (1.0 / n, 0.3 / n, 0.05 / n, 0.01 / n)
                for sigma in sigmas:
                    pt = project_simplex(p + rng.normal(0, sigma, n))
                    sc = internal_score(pt)
                    if sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        since_improve = 0
                        current_score = sc
                        prev_p = p
                        p = pt
                        accepted = True
                        break

            # Pair swap move
            if not accepted and n > 2:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j:
                        continue
                    for frac in (0.1, 0.03, 0.01):
                        delta = frac * min(p[i], p[j])
                        if delta <= 0:
                            break
                        pt = p.copy()
                        pt[i] -= delta
                        pt[j] += delta
                        pt = project_simplex(pt)
                        sc = internal_score(pt)
                        if sc < best_score:
                            best_score = sc
                            best_p = pt.copy()
                            since_improve = 0
                            current_score = sc
                            prev_p = p
                            p = pt
                            accepted = True
                            break

            if not accepted:
                since_improve += 1
                # Periodic kick to escape
                if since_improve > 400:
                    blend = 0.4 + 0.3 * float(rng.uniform(0, 1))
                    p = project_simplex(blend * best_p + (1 - blend) * p +
                                        rng.normal(0, 0.5 / n, n))
                    current_score = internal_score(p)
                    prev_p = p
                    since_improve = 0
                    T_factor = 0.2
                elif since_improve > 150:
                    p = project_simplex(0.8 * best_p + 0.2 * p +
                                        rng.normal(0, 0.1 / n, n))
                    current_score = internal_score(p)
                    prev_p = p

        # Periodic LP for short sequences
        if n <= _LP_NMAX and iter_no % 800 == 0:
            remaining = end - time.monotonic()
            if remaining > 15:
                g = lp_best_response(p, min(30.0, remaining * 0.5))
                if g is not None:
                    for t in (1.0, 0.7, 0.5, 0.3, 0.15):
                        q = project_simplex((1 - t) * p + t * g)
                        sc = internal_score(q)
                        if sc < best_score:
                            best_score = sc
                            best_p = q.copy()
                            p = q.copy()
                            prev_p = q.copy()
                            current_score = sc
                            since_improve = 0
                            break

        if iter_no % 2000 == 0:
            print("  %s iter=%d best=%.10f cur=%.10f Tf=%.5f" % (label, iter_no, best_score, current_score, T_factor), flush=True)

        if best_score < 1.49 and since_improve > 8000:
            break

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def upsample(p, new_n):
    old_n = int(p.size)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    p_new = np.interp(x_new, x_old, p)
    return safe_normalize(p_new)


def propose_candidate(seed=42, budget_s=1800):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 15.0)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and parent_arr.sum() > 0:
            parent = safe_normalize(parent_arr)
    except Exception:
        parent = None

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0:
            return
        p = safe_normalize(p)
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("  new best: %.10f (n=%d)" % (sc, p.size), flush=True)

    # Phase 1: If we have a parent, try to improve it directly
    if parent is not None:
        pn = int(parent.size)
        print("Parent: n=%d, score=%.10f" % (pn, internal_score(parent)), flush=True)
        try_record(parent, internal_score(parent))

        # Quick optimization on parent
        t_rem = deadline - time.monotonic()
        if t_rem > 60:
            opt_time = min(300.0, t_rem * 0.3)
            p_opt, sc_opt = optimize_local(parent, rng, opt_time, "parent")
            try_record(p_opt, sc_opt)

    # Phase 2: Multi-length exploration
    lengths = [256, 512, 1024, 2048, 4096]
    if parent is not None:
        pn = int(parent.size)
        if pn not in lengths:
            # Insert parent length
            lengths.append(pn)
            lengths.sort()

    t_rem = deadline - time.monotonic()
    if t_rem > 60:
        # Allocate time to each length
        n_lengths = len(lengths)
        time_per = max(30.0, (t_rem * 0.5) / n_lengths)

        for n in lengths:
            t_rem = deadline - time.monotonic()
            if t_rem < 30:
                break
            print("Exploring n=%d ... (time_left=%.0f)" % (n, t_rem), flush=True)

            inits = generate_inits(n, rng)
            if parent is not None and int(parent.size) != n:
                inits.append(upsample(parent, n))

            # Score and sort initializations
            scored = []
            for p in inits:
                sc = internal_score(p)
                if np.isfinite(sc):
                    scored.append((sc, p))
            scored.sort(key=lambda x: x[0])

            # Try top initializations
            top_k = min(4, len(scored))
            for i in range(top_k):
                t_rem = deadline - time.monotonic()
                if t_rem < 20:
                    break
                sc_init, p_init = scored[i]
                try_record(p_init, sc_init)
                opt_time = min(time_per / max(1, top_k - i) * 1.5, t_rem * 0.6)
                if opt_time < 10:
                    continue
                p_opt, sc_opt = optimize_local(p_init, rng, opt_time, "n%d_%d" % (n, i))
                try_record(p_opt, sc_opt)

    # Phase 3: Intensive refinement of best so far
    if best_p is not None:
        t_rem = deadline - time.monotonic()
        if t_rem > 60:
            print("Refining best at n=%d (score=%.10f), time_left=%.0f" % (
                best_p.size, best_score, t_rem), flush=True)
            # Try at current length
            p_opt, sc_opt = optimize_local(best_p, rng, min(t_rem * 0.5, 600), "refine")
            try_record(p_opt, sc_opt)

            # Try upsampled version
            t_rem = deadline - time.monotonic()
            if t_rem > 60 and best_p.size < 8192:
                new_n = min(8192, int(best_p.size) * 2)
                p_up = upsample(best_p, new_n)
                sc_up = internal_score(p_up)
                try_record(p_up, sc_up)
                if t_rem > 40:
                    p_opt2, sc_opt2 = optimize_local(p_up, rng, min(t_rem * 0.4, 400), "up%d" % new_n)
                    try_record(p_opt2, sc_opt2)

    # Phase 4: Final fresh starts if time remains
    t_rem = deadline - time.monotonic()
    if t_rem > 120:
        print("Final fresh starts, time_left=%.0f" % t_rem, flush=True)
        for n in (1024, 2048):
            t_rem = deadline - time.monotonic()
            if t_rem < 40:
                break
            for trial in range(3):
                t_rem = deadline - time.monotonic()
                if t_rem < 20:
                    break
                p_init = make_freq_init(n, rng, int(rng.integers(6, 15)))
                sc_init = internal_score(p_init)
                if sc_init < best_score + 0.3:
                    opt_time = min(80.0, t_rem * 0.5)
                    p_opt, sc_opt = optimize_local(p_init, rng, opt_time, "fresh%d_%d" % (n, trial))
                    try_record(p_opt, sc_opt)

    # Phase 5: Final polish
    if best_p is not None:
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            p_opt, sc_opt = optimize_local(best_p, rng, t_rem - 5, "polish")
            try_record(p_opt, sc_opt)

    if best_p is None:
        best_p = np.full(512, 1.0 / 512)

    # Final validation
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s <= 1e-12:
        best_p = np.full(512, 1.0 / 512)

    final_sc = internal_score(best_p)
    print("FINAL: n=%d, internal_score=%.10f" % (best_p.size, final_sc), flush=True)

    return [float(x) for x in best_p]
