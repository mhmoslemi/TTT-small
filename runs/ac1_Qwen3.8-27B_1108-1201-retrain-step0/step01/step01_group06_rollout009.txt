def solve_lp_direction(f, M=None):

    Given f (non-negative, sum=1), find g (non-negative, sum=1)
    that minimizes max(f*g). Returns (g, t_opt).
    """

    if M is None:
        conv = fast_convolve(f)
        M = conv.max()
    
    n_constraints = 2 * n - 1

    # Objective: minimize t -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # For each k: sum_j f_{k-j} * g_j - t <= 0
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    
    for k in range(n_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
        A_ub[k, -1] = -1.0  # coefficient of t
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # g_j >= 0, t >= 0
    
    from scipy.optimize import linprog

        g = res.x[:n]
        t_opt = res.x[-1]

    else:

def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

        conv = fast_convolve(f)
        top_pos = np.argsort(conv)[-sparse_k:]

        top_pos = np.arange(2 * n - 1)

    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based directional search.
At each step, solves an LP (Chebyshev center on the simplex) to find the best
direction g that minimizes max(f*g), then takes a convex combination step.
Uses multiple starting points, lengths, and a three-phase time allocation.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def build_constraint_matrix(f, positions):
    """Build A_ub rows for given convolution positions."""
    n = len(f)
    n_c = len(positions)
    A = np.zeros((n_c, n + 1))
    for i, k in enumerate(positions):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[i, j_start:j_end + 1] = f[k - j_end:k - j_start + 1][::-1]
        A[i, -1] = -1.0
    return A


def solve_lp_direction(f, M, sparse_k=None):
    """
    Solve LP: minimize t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt). g is on the simplex.
    """
    n = len(f)
    if sparse_k is not None and sparse_k < 2 * n - 1:
        conv = fast_convolve(f)
        top_pos = np.argsort(conv)[-sparse_k:]
    else:
        top_pos = np.arange(2 * n - 1)

    A_ub = build_constraint_matrix(f, top_pos)
    b_ub = np.zeros(len(top_pos))
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status == 0:
        g = np.maximum(res.x[:n], 0.0)
        gs = g.sum()
        if gs > 1e-15:
            g /= gs
        return g, res.x[-1]
    return None, M

    LP-based directional optimization on the simplex.
    Returns (best_f, best_normalized_obj).
    """
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    alpha = alpha_init
    t0 = time.monotonic()
    stall = 0
    max_stall = 80
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        g, t_opt = solve_lp_direction(f, M, sparse_k=sparse_k)
        if g is None:
            break

        iteration += 1
        gap = M - t_opt
        rel_gap = gap / max(M, 1e-15)

        if rel_gap < 1e-7:
            stall += 1
            if stall > 10:
                # Try perturbation to escape
                noise = rng.dirichlet(np.ones(n) * 0.1)
                f = 0.95 * f + 0.05 * noise
                f = np.maximum(f, 0)
                fs = f.sum()
                if fs < 1e-12:
                    break
                f = f / fs
                conv = fast_convolve(f)
                stall = 0
                alpha = alpha_init * 0.5
                continue
            break

        # Line search: try several step sizes
        best_a = alpha
        best_co = None
        alphas_try = [alpha * 0.5, alpha * 0.75, alpha, alpha * 1.25, alpha * 1.5]
        for try_alpha in alphas_try:
            if try_alpha <= 0 or try_alpha >= 1:
                continue
            f_try = (1 - try_alpha) * f + try_alpha * g
            f_try = np.maximum(f_try, 0)
            s_try = f_try.sum()
            if s_try < 1e-12:
                continue
            f_try = f_try / s_try
            co = fast_convolve(f_try)
            obj_try = 2.0 * n * co.max()
            if best_co is None or obj_try < best_co:
                best_co = obj_try
                best_a = try_alpha

        if best_co is None or best_co >= best_obj - 1e-12:
            stall += 1
            alpha *= 0.6
            if alpha < 0.005:
                break
            if stall > max_stall:
                break
            continue

        # Update
        f = (1 - best_a) * f + best_a * g
        f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-12:
            break
        f = f / s
        conv = fast_convolve(f)
        current_obj = 2.0 * n * conv.max()

        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_f = f.copy()
            stall = 0
            # Gradually increase step size if making good progress
            if rel_gap > 0.01:
                alpha = min(alpha * 1.1, 0.5)
        else:
            alpha *= 0.7
            stall += 1
            if stall > max_stall:
                break

    return best_f, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    for sf in [0.12, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_up'] = ramp

    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    ramp_d /= ramp_d.sum()
    inits['ramp_down'] = ramp_d

    # Bimodal edges
    w = max(1, n // 4)
    bi = np.zeros(n)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Triangular
    tri = 1.0 - np.abs(x - n * 0.5) / (n * 0.5)
    tri = np.maximum(tri, 0.01)
    tri /= tri.sum()
    inits['triangular'] = tri

    # Cosine
    cos_bump = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_bump /= cos_bump.sum()
    inits['cosine'] = cos_bump

    # Random Dirichlet (spread)
    inits['random_spread'] = rng.dirichlet(np.ones(n) * 0.3)

    # Random Dirichlet (concentrated)
    inits['random_conc'] = rng.dirichlet(np.ones(n) * 1.0)

    # "Checkerboard-like" for even n
    if n % 2 == 0:
        cb = np.zeros(n)
        cb[::2] = 1.0
        cb /= cb.sum()
        inits['checkerboard'] = cb

    # From parent
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp

    return inits


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Main entry point. Uses LP-based Chebyshev center search to minimize
    2n*max(conv(a,a))/(sum(a))^2. Three-phase: scan, deep, polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 25.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            ps = float(evaluate_sequence(p.tolist()))
            print(f"Parent loaded: len={p.size}, score={ps:.6f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_sequence = p.copy()
    except Exception as e:
        print(f"No parent: {e}")

    # Phase 1: Scan diverse configurations
    print("Phase 1: Scanning diverse configurations...")
    lengths = [64, 96, 128, 160, 192, 256, 320, 384, 512]
    scan_configs = []

    for n in lengths:
        inits = generate_initializations(n, rng, parent)
        for name, a0 in inits.items():
            scan_configs.append((n, name, a0))

    n_total = len(scan_configs)
    phase1_frac = 0.25
    phase1_time = min(500.0, (deadline - time.monotonic()) * phase1_frac)
    time_per_scan = max(4.0, phase1_time / max(1, n_total))

    scan_results = []
    for i, (n, name, a0) in enumerate(scan_configs):
        if time.monotonic() > deadline - 200:
            break
        t_alloc = min(time_per_scan, deadline - time.monotonic() - 200)
        if t_alloc < 3:
            break

        # Use sparse LP for larger n
        sparse_k = None if n <= 256 else min(60, 2 * n - 1)
        a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc,
                                     alpha_init=0.15, sparse_k=sparse_k,
                                     seed=seed + i)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan[{i}]: n={n}, {name}, score={score:.6f}")

    print(f"  Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Deep optimization on promising configs
    print("Phase 2: Deep optimization...")
    scan_results.sort(key=lambda x: x[0])

    top_configs = []
    seen_keys = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen_keys and len(top_configs) < 12:
            top_configs.append((n, name, a_opt, score))
            seen_keys.add(key)

    # Ensure current best is included
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy(), best_score))

    remaining = deadline - time.monotonic() - 100
    time_per_deep = max(15.0, remaining / max(1, len(top_configs)))

    for i, (n, name, a0, sc) in enumerate(top_configs):
        if time.monotonic() > deadline - 80:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 80)
        if t_alloc < 10:
            break

        sparse_k = None if n <= 256 else min(80, 2 * n - 1)

        # Try multiple alpha settings
        for alpha in [0.1, 0.2, 0.35]:
            if time.monotonic() > deadline - 80:
                break
            t_single = t_alloc / 3.0
            if t_single < 5:
                continue
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_single,
                                         alpha_init=alpha, sparse_k=sparse_k,
                                         seed=seed + 100 + i * 10)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep[{i},{name},a={alpha}]: n={n}, score={score:.6f}")

    print(f"  Phase 2 done. Best: {best_score:.6f}")

    # Phase 3: Final polish on the best with restarts
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("Phase 3: Final polish...")
        remaining = deadline - 10 - time.monotonic()
        n = len(best_sequence)
        sparse_k = None if n <= 256 else min(80, 2 * n - 1)

        # Multiple polish attempts with different alphas and perturbations
        n_attempts = min(6, int(remaining / 15))
        for att in range(n_attempts):
            if time.monotonic() > deadline - 10:
                break
            t_single = (deadline - 10 - time.monotonic()) / max(1, n_attempts - att)
            if t_single < 5:
                break

            # Sometimes perturb the starting point
            if att > 0 and att % 3 == 0:
                noise = rng.dirichlet(np.ones(n) * 0.05)
                start = 0.9 * best_sequence + 0.1 * noise
                start = np.maximum(start, 0)
                ss = start.sum()
                if ss > 1e-12:
                    start = start / ss
                else:
                    start = best_sequence.copy()
            else:
                start = best_sequence.copy()

            alpha = 0.05 + 0.1 * (att % 4)
            a_opt, obj_opt = lp_optimize(start, time_limit=t_single,
                                         alpha_init=alpha, sparse_k=sparse_k,
                                         seed=seed + 500 + att)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish[{att}]: alpha={alpha:.2f}, score={score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"FINAL SCORE: {best_score:.6f} (length={len(best_sequence)})")
    return [float(x) for x in best_sequence]
