"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses soft-max gradient descent with Nesterov momentum, adaptive temperature,
and LP-based refinement. Multiple lengths and starting points are explored.

from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_a(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def softmax_gradient(a, tau):
    """Compute the soft-max gradient of the peak autocorrelation.
    Returns (conv, weights, gradient)."""
    conv = np.convolve(a, a)
    # Numerically stable soft-max
    conv_shifted = conv - conv.max()
    w = np.exp(tau * conv_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Fallback: uniform weights
        w = np.ones(len(conv)) / len(conv)
    else:
        w /= w_sum
    # Gradient: grad[i] = 2 * sum_j a[j] * w[i+j]
    grad = 2.0 * np.correlate(w, a, mode='valid')
    # Center the gradient to preserve the sum constraint
    grad -= grad.mean()
    return conv, w, grad


def nesterov_step(a, v, lr, tau, momentum):
    """One Nesterov momentum step with soft-max gradient."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    
    # Nesterov lookahead
    a_look = a + momentum * v
    a_look = np.maximum(a_look, 0.0)
    a_look = normalize_a(a_look, S_target)
    
    # Compute gradient at lookahead point
    conv, w, grad = softmax_gradient(a_look, tau)
    
    # Update velocity
    v = momentum * v + lr * grad
    
    # Update position
    a_new = a - v
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_a(a_new, S_target)
    
    return a_new, v, conv


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    n = len(f)
    if n > 2048:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_soft(a0, deadline, max_iters=5000):
    """Optimize using soft-max gradient with Nesterov momentum."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize_a(a)
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    S_target = np.sqrt(2.0 * n)
    
    # Adaptive parameters
    tau = 5.0  # Start with moderate temperature
    lr = 0.05
    momentum = 0.85
    v = np.zeros(n)
    
    # Schedule
    tau_final = 100.0
    lr_final = 0.005
    momentum_final = 0.95
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break
        
        # Anneal parameters
        progress = min(1.0, it / max_iters)
        tau = tau + (tau_final - tau) * 0.02
        lr = lr + (lr_final - lr) * 0.01
        momentum = momentum + (momentum_final - momentum) * 0.005
        momentum = min(momentum, 0.98)
        
        a, v, conv = nesterov_step(a, v, lr, tau, momentum)
        
        # Check improvement periodically
        if (it + 1) % 200 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

        if (it + 1) % 2000 == 0:
            v *= 0.5
    
    # Final check
    s = compute_obj(a)

        best = a.copy()
    
    return best, best_score

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses soft-max gradient descent with Nesterov momentum, adaptive temperature annealing,
LP-based best-response refinement, and multi-length multi-start exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_a(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = float(np.sum(a))
    if S < 1e-15:
        return a
    return a * (target / S)


def softmax_gradient(a, tau):
    """Compute soft-max approximation of max(conv) and its gradient."""
    conv = np.convolve(a, a)
    conv_shifted = conv - np.max(conv)
    # Clamp for numerical stability
    conv_shifted = np.clip(conv_shifted, -50.0 / max(tau, 1.0), 0.0)
    w = np.exp(tau * conv_shifted)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        w = np.ones(len(conv)) / len(conv)
    else:
        w = w / w_sum
    # grad[i] = 2 * sum_j a[j] * w[i+j] = 2 * correlate(w, a, 'valid')[i]
    grad = 2.0 * np.correlate(w, a, mode='valid')
    # Center to preserve sum constraint
    grad = grad - np.mean(grad)
    return conv, w, grad


def nesterov_step(a, v, lr, tau, momentum):
    """One Nesterov momentum step."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    # Lookahead
    a_look = a + momentum * v
    a_look = np.maximum(a_look, 0.0)
    a_look = normalize_a(a_look, S_target)
    # Gradient at lookahead
    conv, w, grad = softmax_gradient(a_look, tau)
    # Velocity update
    v = momentum * v - lr * grad
    # Position update
    a_new = a + v
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_a(a_new, S_target)
    return a_new, v, conv


def soft_gradient_descent(a0, deadline, max_iters=5000, tau_init=3.0, tau_max=80.0,
                          lr_init=0.08, lr_min=0.002, momentum_init=0.85, momentum_max=0.97):
    """Soft-max gradient descent with Nesterov momentum and adaptive parameters."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize_a(a)
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    S_target = np.sqrt(2.0 * n)

    v = np.zeros(n)
    tau = tau_init
    lr = lr_init
    momentum = momentum_init
    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive schedule
        progress = min(1.0, it / max(1, max_iters))
        tau = tau_init + (tau_max - tau_init) * progress ** 0.7
        lr = lr_init * (1.0 - progress) + lr_min * progress
        momentum = momentum_init + (momentum_max - momentum_init) * progress

        a, v, conv = nesterov_step(a, v, lr, tau, momentum)

        # Periodic evaluation
        if (it + 1) % 300 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-9:
                best_score = s
                best = a.copy()
                stagnation = 0
            else:
                stagnation += 1
            # If stagnant, add small noise
            if stagnation > 3:
                a = a + 0.01 * np.random.default_rng(it).standard_normal(n)
                a = np.maximum(a, 0.0)
                a = normalize_a(a, S_target)
                v *= 0.3
                stagnation = 0

    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score


def solve_lp_fast(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M componentwise, g >= 0."""
    n = len(f)
    if n > 2048:
        return None
    c_obj = -np.ones(n)
    # Build banded constraint matrix efficiently
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        A[m, lo:hi + 1] = f[m - hi:m - lo + 1][::-1]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_refine(a0, deadline, max_iters=200):
    """LP-based best-response refinement."""
    n = len(a0)
    if n > 1024:
        return a0.copy(), compute_obj(a0)

    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    a = normalize_a(a)
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    S_target = np.sqrt(2.0 * n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = np.convolve(a, a)
        M = float(np.max(conv))

        g = solve_lp_fast(a, M)
        if g is None:
            break

        improved = False
        for t in [0.4, 0.25, 0.15, 0.08, 0.03]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_a(a_new, S_target)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-10:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            # Small gradient nudge to escape
            a_tmp, _, _ = nesterov_step(a, np.zeros(n), 0.02, 10.0, 0.9)
            s_try = compute_obj(a_tmp)
            if s_try < best_score:
                a = a_tmp
                best = a.copy()
                best_score = s_try
            else:
                break

    return best, best_score


def gen_starts(n, rng):
    """Generate diverse starting sequences of length n."""
    starts = []
    i = np.arange(n, dtype=np.float64)
    starts.append(np.ones(n) * 0.5)
    starts.append(rng.random(n) * 0.7 + 0.15)
    starts.append(0.1 + 0.8 * i / (n - 1))
    starts.append(np.exp(-3.0 * i / n) + 0.1)
    starts.append(1.0 - 2.0 * np.abs(i - n * 0.4) / n)
    starts.append(0.5 + 0.4 * np.cos(2.0 * np.pi * i / n + 0.5))
    starts.append(0.4 + 0.5 * np.sin(np.pi * i * 0.7 / n))
    starts.append(0.3 + 0.6 * np.sin(2.0 * np.pi * i / n - 0.3) ** 2)
    starts.append(0.5 + 0.3 * np.cos(2.0 * np.pi * i * i / n))
    starts.append(np.exp(-0.5 * ((i - n * 0.2) / (n * 0.15)) ** 2) +
                  0.3 * np.exp(-0.5 * ((i - n * 0.7) / (n * 0.12)) ** 2))
    starts.append(0.5 * (1.0 + np.tanh(3.0 * (i / n - 0.5))))
    starts.append(0.2 + 0.8 * (i / n) ** 0.5)
    return starts
