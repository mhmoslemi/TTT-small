import numpy as np
import time

def compute_objective_and_grad(a, tau):
    """Compute the smoothed objective and its gradient."""
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1
    c_max = c - c.max()
    exp_c = np.exp(c_max / tau)
    q = exp_c / exp_c.sum()
    obj = tau * (np.log(exp_c.sum()) + c.max() / tau)
    # Gradient: 2 * correlate(q, a, 'valid')
    grad = 2.0 * np.correlate(q, a, 'valid')
    return obj, grad, c.max()

def project_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(a)
    u = np.maximum(a, 0).copy()
    s = u.sum()
    if s == 0:
        return np.full(n, S / n)
    if s <= S:
        return u + (S - s) / n
    # Binary search for threshold
    lo, hi = 0.0, u.max()
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.maximum(u - mid, 0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    res = np.maximum(u - t, 0.0)
    res[res.argmax()] += S - res.sum()
    return res

def optimize(a0, tau_init, tau_final, lr_init, time_budget, seed=0):

    n = len(a0)

    a = project_simplex(a0.astype(np.float64), S)
    
    best_a = a.copy()
    best_mc = np.convolve(a, a).max()
    best_score = evaluate_sequence(best_a.tolist())
    
    t0 = time.monotonic()
    step = 0
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_final / tau_init) ** frac
        
        obj, grad, mc = compute_objective_and_grad(a, tau)
        a_new = a - lr * grad
        a_new = project_simplex(a_new, S)
        
        step += 1
        if step % 5 == 0:
            mc = np.convolve(a_new, a_new).max()
            if mc < best_mc - 1e-10:
                best_mc = mc
                best_a = a_new.copy()
                no_improve = 0
                if lr < lr_init * 3:
                    lr *= 1.02
            else:
                no_improve += 1
                lr = max(0.01, lr * (0.9 if no_improve > 10 else 0.95))
            
            # Stochastic perturbation every so often
            if no_improve > 50 and step % 50 == 0:
                sigma = 0.1 * S / n
                noise = sigma * rng.standard_normal(n)
                a_perturbed = project_simplex(a + noise, S)
                mc_p = np.convolve(a_perturbed, a_perturbed).max()
                if mc_p < best_mc:

                    best_mc = mc_p
                    best_a = a.copy()
                    no_improve = 0
                # Always accept perturbation if it's not too bad (simulated annealing)
                elif mc_p < best_mc * 1.01 and rng.random() < 0.3:
                    a = a_perturbed
                    no_improve = 0
            
            a = a_new
    
    # Final evaluation
    best_score = evaluate_sequence(best_a.tolist())
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses FFT-accelerated gradient descent with adaptive tau, learning rate,
stochastic restarts, and local search polish.
"""

import numpy as np
import time

def fast_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def fast_correlate_valid(q, a):
    n = len(a)
    full = fast_convolve(q, a[::-1])
    return full[n-1:2*n-1]

def compute_grad(a, tau):
    n = len(a)
    c = fast_convolve(a, a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    grad = 2.0 * fast_correlate_valid(q, a)
    return grad, c.max()

def project_simplex(a, S):
    n = len(a)
    u = np.maximum(a, 0.0)
    s = u.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    if s <= S:
        return u + (S - s) / n
    lo, hi = 0.0, u.max()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.maximum(u - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(u - t, 0.0)
    deficit = S - res.sum()
    if deficit > 0:
        idx = res.argmax()
        res[idx] += deficit
    return res

def init_chirp(n, seed=0):
    t = np.arange(n, dtype=np.float64) - (n - 1) * 0.5
    a = 1.0 + 0.8 * np.cos(2.0 * np.pi * t * t / (2.0 * n))
    rng = np.random.default_rng(seed)
    a += 0.1 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

def init_random(n, seed=0):
    return np.random.default_rng(seed).exponential(1.0, size=n)

def init_burst(n, seed=0):
    """Initialize with bursty structure."""
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    n_bursts = max(3, n // 200)
    for _ in range(n_bursts):
        center = rng.integers(0, n)
        width = rng.integers(n // 20, n // 5)
        for i in range(max(0, center - width), min(n, center + width)):
            a[i] += np.exp(-0.5 * ((i - center) / (width * 0.3))**2)
    return a

def run_opt(a0, tau_hi, tau_lo, lr0, t_budget, seed=0, verbose=False):
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = project_simplex(a0.astype(np.float64).copy(), S)
    rng = np.random.default_rng(seed)
    
    c = fast_convolve(a, a)
    best_mc = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr0
    step = 0
    stall = 0
    last_mc = best_mc
    
    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(t_budget, 1e-6))
        tau = tau_hi * (tau_lo / tau_hi) ** frac
        
        grad, mc = compute_grad(a, tau)
        grad_norm = np.abs(grad).max() + 1e-12
        adaptive_lr = lr * (lr0 / (grad_norm + 1e-3))  # Scale by gradient
        
        a_new = a - min(adaptive_lr, 3.0 * S / n) * grad
        a_new = project_simplex(a_new, S)
        step += 1
        
        if step % 10 == 1:
            c = fast_convolve(a_new, a_new)
            mc = c.max()
            if mc < last_mc - 1e-10:
                a = a_new
                stall = 0
                if mc < best_mc - 1e-12:
                    best_mc = mc
                    best_a = a.copy()
                    if verbose and step % 100 == 1:
                        print(f"      step={step} mc={mc:.6f}")
            else:
                a = a_new
                stall += 1
                lr = max(0.01, lr * 0.9)
            last_mc = mc
            
            # Stochastic restart
            if stall > 100 and step % 100 == 1:
                sigma = 0.3 * S / np.sqrt(n)
                noise = sigma * rng.standard_normal(n)
                a_p = project_simplex(a + noise, S)
                c_p = fast_convolve(a_p, a_p)
                mc_p = c_p.max()
                if mc_p < best_mc * 1.005:
                    a = a_p
                    stall = 0
                    lr = lr0
                if mc_p < best_mc:
                    best_mc = mc_p
                    best_a = a_p.copy()
    
    return best_a, best_mc

def local_polish(a, t_budget, seed=0):
    """Pairwise local search to polish the solution."""
    rng = np.random.default_rng(seed)
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = project_simplex(a.copy(), S)
    c = fast_convolve(a, a)
    best_mc = c.max()
    
    t0 = time.monotonic()
    iters = 0
    while time.monotonic() - t0 < t_budget:
        # Pick peak lag and find contributing pairs
        peak_lag = c.argmax()
        # Pairs contributing to c[peak_lag]: (i, peak_lag - i) for valid i
        valid_ks = [k for k in range(max(0, peak_lag - n + 1), min(n, peak_lag + 1))]
        if len(valid_ks) < 2:
            j, k = rng.integers(0, n), rng.integers(0, n)
        else:
            idx = rng.integers(0, len(valid_ks))
            j = valid_ks[idx]
            k = peak_lag - j
        
        if j == k or j < 0 or j >= n or k < 0 or k >= n:
            continue
        
        a_j, a_k = a[j], a[k]
        if a_j < 1e-10 and a_k < 1e-10:
            continue
        
        # Try moving weight from the larger to smaller
        best_d = 0.0
        best_local = best_mc
        scale = 0.05 * max(a_j, a_k, S/n)
        for d in np.linspace(-scale, scale, 15):
            nj, nk = a_j + d, a_k - d
            if nj < 0 or nk < 0:
                continue
            a[j], a[k] = nj, nk
            c = fast_convolve(a, a)
            mc = c.max()
            if mc < best_local:
                best_local = mc
                best_d = d
        a[j], a[k] = a_j + best_d, a_k - best_d
        c = fast_convolve(a, a)
        mc = c.max()
        
        if mc < best_mc:
            best_mc = mc
            iters += 1
            if iters % 50 == 0:
                print(f"    polish: mc={mc:.6f} iters={iters}")
    
    a = project_simplex(a, S)

Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses FFT-accelerated projected gradient descent with adaptive tau annealing,
learning rate control, stochastic restarts, and pairwise local search.
"""

import numpy as np
import time


def fast_convolve(a, b):
    """FFT-based linear convolution."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def fast_correlate_valid(q, a):
    """Compute np.correlate(q, a, 'valid') via FFT. q has length 2*len(a)-1."""
    n = len(a)
    full = fast_convolve(q, a[::-1])
    return full[n - 1:2 * n - 1]


def compute_grad(a, tau):
    """Compute gradient of smoothed max of conv(a,a)."""
    n = len(a)
    c = fast_convolve(a, a)
    if tau < 1e-4:
        q = np.zeros_like(c)
        q[np.argmax(c)] = 1.0
    else:
        c_shift = c - c.max()
        exp_c = np.exp(c_shift / tau)
        s = exp_c.sum()
        if s < 1e-300:
            q = np.zeros_like(c)
            q[np.argmax(c)] = 1.0
        else:
            q = exp_c / s
    grad = 2.0 * fast_correlate_valid(q, a)
    return grad, c.max()


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(a)
    u = np.maximum(a, 0.0)
    s = u.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    if s <= S + 1e-15:
        return u + (S - s) / n
    lo, hi = 0.0, u.max()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.maximum(u - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(u - t, 0.0)
    deficit = S - res.sum()
    if deficit > 0:
        res[res.argmax()] += deficit
    return res


def init_chirp(n, seed=0):
    t = np.arange(n, dtype=np.float64) - (n - 1) * 0.5
    a = 1.0 + 0.7 * np.cos(np.pi * t * t / n + 0.3 * np.sin(2 * np.pi * t / n))
    rng = np.random.default_rng(seed)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_random(n, seed=0):
    return np.random.default_rng(seed).exponential(1.0, size=n)


def init_gaussian_mix(n, seed=0):
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    n_mix = max(3, n // 500)
    for _ in range(n_mix):
        center = rng.uniform(0, n)
        width = rng.uniform(n * 0.01, n * 0.1)
        amp = rng.uniform(0.5, 2.0)
        x = np.arange(n)
        a += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    return a


def run_opt(a0, tau_hi, tau_lo, lr0, t_budget, seed=0, verbose=False):
    """Projected gradient descent with adaptive parameters."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = project_simplex(a0.astype(np.float64).copy(), S)
    rng = np.random.default_rng(seed)

    c = fast_convolve(a, a)
    best_mc = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr0
    step = 0
    stall = 0
    last_mc = best_mc

    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(t_budget, 1e-6))
        tau = tau_hi * (tau_lo / tau_hi) ** frac

        grad, mc = compute_grad(a, tau)
        
        # Adaptive step size
        gmax = np.abs(grad).max() + 1e-12
        step_size = lr / (1.0 + gmax * n / S)
        
        a_new = a - step_size * grad
        a_new = project_simplex(a_new, S)
        step += 1

        if step % 8 == 1:
            c = fast_convolve(a_new, a_new)
            mc = c.max()
            if mc < last_mc - 1e-12:
                stall = max(0, stall - 1)
                if mc < best_mc - 1e-12:
                    best_mc = mc
                    best_a = a_new.copy()
                    if verbose and step % 80 == 1:
                        print(f"        step={step} mc={mc:.7f} tau={tau:.4f}")
            else:
                stall += 1
                lr = max(0.005, lr * 0.92)
            last_mc = mc
            a = a_new

            # Stochastic perturbation when stuck
            if stall > 80 and step % 80 == 1:
                sigma = 0.2 * S / np.sqrt(n)
                noise = sigma * rng.standard_normal(n)
                a_p = project_simplex(a + noise, S)
                c_p = fast_convolve(a_p, a_p)
                mc_p = c_p.max()
                if mc_p <= best_mc * 1.002:
                    a = a_p.copy()
                    stall = 0
                    lr = lr0
                if mc_p < best_mc:
                    best_mc = mc_p
                    best_a = a_p.copy()

    return best_a, best_mc


def local_polish(a, t_budget, seed=0):

    rng = np.random.default_rng(seed)
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = project_simplex(a.copy(), S)

    improvements = 0

    t0 = time.monotonic()
    total_iters = 0
    while time.monotonic() - t0 < t_budget:
        peak_lag = int(c.argmax())
        
        # Find contributing pairs for the peak lag
        lo = max(0, peak_lag - n + 1)
        hi = min(n - 1, peak_lag)
        candidates = list(range(lo, hi + 1))
        
        if len(candidates) < 2:

        if a_j < 1e-12 and a_k < 1e-12:

            continue

        best_d = 0.0

        scale = 0.03 * max(a_j, a_k, 2.0 * S / n)

            if mc < best_local - 1e-14:

This is a non-convex problem (quadratic constraints). An SDP relaxation would replace a[i]*a[j] with X[i,j] where X is PSD. But this gives an n x n matrix, which for n=3000 is 9 million variables. Too large for most SDP solvers.

def fast_convolve(a, b):
    """FFT-based linear convolution."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def fast_correlate_valid(q, a):
    """Compute np.correlate(q, a, 'valid') via FFT. len(q) should be 2*len(a)-1."""
    n = len(a)
    full = fast_convolve(q, a[::-1])
    return full[n - 1:2 * n - 1]
