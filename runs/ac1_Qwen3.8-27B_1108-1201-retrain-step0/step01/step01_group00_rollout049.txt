```python
    Returns (g, M_optimal).
    """
    n = len(f)

    # Minimize M -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
```

```python
"""
Hybrid LP + gradient optimization for minimizing peak autocorrelation
of non-negative step functions. Uses a linear program to find the optimal
next iterate at each step, combined with gradient-based refinement and
coordinate descent for final polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_normalized(a):
    """Compute the score assuming sum(a) = 1."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def lp_iteration(f):
    """
    Given f with sum(f)=1, solve LP to find g minimizing M=max(f*g)
    subject to sum(g)=1, g>=0.
    Returns (g, M).
    """
    n = len(f)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            A_ub[k, j] += f[i]
        A_ub[k, n] = -1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status == 0:
        return result.x[:n], result.x[n]
    return None, None


def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / e.sum()


def grad_soft_max_conv(a, tau):
    """Compute value and gradient of soft_max(conv(a,a), tau)."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    value = c_max + tau * np.log(exp_shifted.sum())
    corr = fft_convolve(a[::-1], w)
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return value, g


def optimize_gradient_nesterov(a_init, time_limit, rng):

    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()
    
    # Nesterov momentum
    y = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)

    lr = 0.1

    best_a = a.copy()
    best_score = score_normalized(a)

        a = softmax_normalize(theta)
        value, g_a = grad_soft_max_conv(a, tau)
        
        # Project gradient to tangent space of simplex
        avg_g = g_a.mean()
        g_proj = g_a - avg_g
        
        step += 1
        m = beta1 * m + (1 - beta1) * g_proj
        v = beta2 * v + (1 - beta2) * g_proj ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        lr_t = lr / (1.0 + 0.1 * step)

        sc = score_normalized(a_new)
        
        if sc < best_score:

            best_a = a_new.copy()
```

```python
"""
Hybrid LP + Adam gradient optimization for minimizing peak autocorrelation
of non-negative step functions. Uses a linear program (via HiGHS) to find
optimal next iterates in a power-iteration fashion, combined with Adam-based
gradient refinement and coordinate descent for final polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_normalized(a):
    """Compute score = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def compute_conv_matrix(f, n):
    """Build the (2n-1) x n matrix A where (A @ g)[k] = (f*g)[k]."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            A[k, j] = f[i]
    return A


def lp_iteration(f, n, A_cache=None):
    """
    Given f with sum(f)=1, solve LP:
      min M
      s.t. (f*g)[k] <= M, k=0..2n-2
           sum(g) = 1
           g >= 0
    Returns (g, M).
    """
    if A_cache is None:
        A_conv = compute_conv_matrix(f, n)
    else:
        A_conv = A_cache

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')

    if result.status == 0:
        return result.x[:n], result.x[n]
    return None, None


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = e.sum()
    return e / s


def grad_softmax_conv(a, tau):
    """
    Compute value and gradient of soft_max(conv(a,a), tau) w.r.t. a.
    a is assumed to be on the simplex (sum=1, a>=0).
    Returns (value, gradient_wrt_a).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    value = c_max + tau * np.log(exp_shifted.sum())
    # Gradient: g[j] = 2 * sum_m a[m] * w[m + j - (n-1)]
    # = 2 * (a[::-1] * w)[n-1-j] ... let me use correlation
    corr = fft_convolve(a[::-1], w)
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return value, g


def gradient_descent_phase(a_init, time_limit, rng):
    """
    Adam gradient descent on log-softmax parameterization.
    Minimizes soft_max(conv(a,a), tau) with annealed temperature.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 1e-10, None)
    a = a / a.sum()

    theta = np.log(a) - np.log(a.sum())
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_normalized(a)
    tau_start = 10.0 / n
    tau_end = 0.5 / n
    no_improve = 0
    max_no_improve = 5000

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac

        a = softmax_normalize(theta)
        value, g_a = grad_softmax_conv(a, tau)

        # Project to tangent space of simplex
        g_proj = g_a - g_a.mean()

        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g_proj
        v_buf = beta2 * v_buf + (1 - beta2) * (g_proj * g_proj)
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)

        lr = 0.5 / (1.0 + 0.05 * np.sqrt(step))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if step % 50 == 0 or step < 10:
            a_new = softmax_normalize(theta)
            sc = score_normalized(a_new)
            if sc < best_score - 1e-8:
                best_score = sc
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 50

            if no_improve >= max_no_improve:
                # Restart with perturbation
                theta += rng.normal(0, 0.05, n)
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                step = 0
                no_improve = 0
                max_no_improve = int(max_no_improve * 0.7)

            if step % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    grad step={step}, score={sc:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={elapsed:.1f}s")

    return best_a, best_score


def coordinate_descent_phase(a_init, time_limit, rng):
    """
    Coordinate descent: randomly pick a coordinate, try several multipliers,
    keep the best. Renormalize after each accepted move.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()
    current_score = score_normalized(a)
    best_a = a.copy()
    best_score = current_score

    t0 = time.monotonic()
    iters = 0

    # Precompute the convolution matrix structure for fast updates
    # Actually, let's just recompute each time (FFT is fast enough for n<=512)

    factors = np.array([0.5, 0.6, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5, 2.0])

    while time.monotonic() - t0 < time_limit:
        batch_size = 10
        i = rng.integers(n)
        best_factor = 1.0
        best_sc = current_score
        orig_val = a[i]

        for f in factors:
            trial = a.copy()
            trial[i] = orig_val * f
            if trial[i] < 0:
                continue
            s = trial.sum()
            if s < 1e-10:
                continue
            trial = trial / s
            sc = score_normalized(trial)
            if sc < best_sc:
                best_sc = sc
                best_factor = f

        if best_factor != 1.0:
            a[i] = orig_val * best_factor
            a = a / a.sum()
            current_score = best_sc
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()

        iters += batch_size
        if iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iters}, score={current_score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")

    return best_a, best_score


def block_perturbation_phase(a_init, time_limit, rng):
    """
    Local search with block perturbations and peak-targeted moves.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()
    current_score = score_normalized(a)
    best_a = a.copy()
    best_score = current_score

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.4:
            # Peak-targeted: reduce contributions to the peak
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Find top contributors
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.02, size=top_k))
            # Boost some others
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_inc = min(5, len(others))
                inc_idx = rng.choice(others, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.02, size=n_inc))
        elif r < 0.7:
            # Block perturbation
            width = max(1, rng.integers(1, max(2, n // 8)))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.85:
            # Symmetrize partially
            sym = 0.5 * (a + a[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.02, size=n))
        else:
            # Global small perturbation
            candidate = a * np.exp(rng.normal(0, 0.015, size=n))

        candidate = np.clip(candidate, 1e-10, None)
        candidate = candidate / candidate.sum()
        sc = score_normalized(candidate)

        if sc < current_score:
            a = candidate
            current_score = sc
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        elif rng.random() < np.exp(-(sc - current_score) / 0.005):
            a = candidate
            current_score = sc

        it += 1
        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Block iter={it}, score={current_score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")

    return best_a, best_score


def make_chirp_init(n, rng):
    """Chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    alpha = 0.3 + 0.3 * rng.random()
    a = 1.0 + alpha * np.cos(np.pi * t * t / n + rng.uniform(0, np.pi))
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / a.sum()


def make_multitone_init(n, rng):
    """Multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in [1, 2, 3, 5, 8]:
        amp = 0.3 / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / a.sum()


def run_lp_phase(a0, n, time_limit, rng):
    """Run the LP-based power iteration for a given time."""
    a = a0.copy()
    a = np.clip(a, 1e-10, None)
    a = a / a.sum()

    t0 = time.monotonic()
    it = 0
    best_a = a.copy()
    best_score = score_normalized(a)
    plateau = 0
    alpha = 0.5  # blending parameter

    while time.monotonic() - t0 < time_limit:
        # Solve LP
        g, M_lp = lp_iteration(a, n)
        if g is None:
            print(f"    LP failed at iter {it}, stopping")
            break

        it += 1

        # Decaying blend
        alpha = max(0.15, 0.5 * (0.995 ** it))
        a_new = (1 - alpha) * a + alpha * g
        a_new = np.clip(a_new, 1e-10, None)
        a_new = a_new / a_new.sum()

        sc = score_normalized(a_new)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            plateau = 0
        else:
            plateau += 1

        a = a_new

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            rate = it / elapsed
            print(f"    LP iter={it}, score={sc:.6f}, best={best_score:.6f}, M={M_lp:.6f}, "
                  f"alpha={alpha:.3f}, rate={rate:.1f}/s, t={elapsed:.1f}s")

        if plateau > 1000:
            # Perturb and reset counting
            a = a + rng.normal(0, 0.01, n)
            a = np.clip(a, 1e-10, None)
            a = a / a.sum()
            plateau = 0
            alpha = max(0.3, alpha + 0.1)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid LP + gradient + local search optimization.
    Phase 1: LP-based power iteration for multiple sequence lengths.
    Phase 2: Adam gradient refinement.
    Phase 3: Block perturbation + coordinate descent polish.
    """
    t_global = time.monotonic()
    deadline = t_global + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None
    best_n = 0

    def update_best(a, score):
        nonlocal best_score, best_seq, best_n
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            best_n = len(a)
            print(f"    *** NEW BEST: score={score:.6f}, n={len(a)} ***")

    # Initialize from provided sequence or chirp
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 1e-6, 1000.0)
        a0 = a0 / a0.sum()
        n_prov = len(a0)
        s_prov = score_normalized(a0)
        print(f"Provided sequence: n={n_prov}, score={s_prov:.6f}")
        update_best(a0, s_prov)
    else:
        a0 = make_chirp_init(256, rng)
        s_prov = score_normalized(a0)
        print(f"Chirp initialization: n=256, score={s_prov:.6f}")
        update_best(a0, s_prov)

    # Phase 1: LP-based iteration
    print(f"\n{'='*60}")
    print(f"=== Phase 1: LP-based power iteration ===")
    print(f"{'='*60}")

    # Try multiple lengths
    lp_configs = [
        (256, 700),   # (length, time budget in seconds)
        (384, 350),
        (512, 300),
    ]

    for n, lp_time in lp_configs:
        if time.monotonic() > deadline - 500:
            print("  Not enough time remaining for more LP configs, skipping.")
            break

        # Initialize for this length
        if n == best_n and best_seq is not None:
            a_init = best_seq.copy()
        elif n == n_prov and 'height_sequence_1' in globals() and height_sequence_1 is not None:
            a_init = np.asarray(height_sequence_1, dtype=np.float64)
            a_init = np.clip(a_init, 1e-6, None)
            a_init = a_init / a_init.sum()
        else:
            a_init = make_chirp_init(n, rng)

        actual_time = min(lp_time, max(30.0, deadline - time.monotonic() - 500.0))
        print(f"\n  Starting LP phase for n={n}, time_budget={actual_time:.0f}s")

        a_lp, score_lp = run_lp_phase(a_init, n, actual_time, rng)
        update_best(a_lp, score_lp)

        print(f"  LP phase done for n={n}: score={score_lp:.6f}")

    # Phase 2: Gradient refinement
    print(f"\n{'='*60}")
    print(f"=== Phase 2: Adam gradient refinement ===")
    print(f"{'='*60}")

    if best_seq is not None:
        grad_time = min(400.0, max(60.0, deadline - time.monotonic() - 300.0))
        print(f"  Best: n={best_n}, score={best_score:.6f}, time_budget={grad_time:.0f}s")

        a_grad, score_grad = gradient_descent_phase(best_seq, grad_time, rng)
        update_best(a_grad, score_grad)

        # Also try from a fresh chirp at the best length
        a_chirp = make_chirp_init(best_n, rng)
        a_grad2, score_grad2 = gradient_descent_phase(a_chirp, min(200.0, grad_time * 0.5), rng)
        update_best(a_grad2, score_grad2)

    # Phase 3: Local search polish
    print(f"\n{'='*60}")
    print(f"=== Phase 3: Local search polish ===")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 30:
            # Split between block perturbation and coordinate descent
            block_time = remaining * 0.5
            cd_time = remaining * 0.5

            print(f"  Block perturbation: time={block_time:.0f}s")
            a_block, score_block = block_perturbation_phase(best_seq, block_time, rng)
            update_best(a_block, score_block)

            if time.monotonic() < deadline - 20:
                cd_time = min(cd_time, deadline - time.monotonic() - 20)
                print(f"  Coordinate descent: time={cd_time:.0f}s")
                a_cd, score_cd = coordinate_descent_phase(best_seq, cd_time, rng)
                update_best(a_cd, score_cd)

    # Final validation
    print(f"\n{'='*60}")
    print(f"=== FINAL: best_score={best_score:.6f}, n={best_n} ===")
    print(f"{'='*60}")

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"  evaluate_sequence confirms: {final_eval:.6f}")
        return final_list

    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
```
