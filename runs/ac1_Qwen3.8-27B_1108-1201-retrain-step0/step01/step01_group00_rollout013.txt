"""
Multi-phase optimization for flat autocorrelation of non-negative step functions.
Uses peak-targeted mass redistribution, gradient-based soft-max optimization,

import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score_fast(a):
    """Fast computation of the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def soft_max_and_weights(x, tau):
    """Compute soft-max value and softmax weights."""
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    value = x_max + tau * np.log(sum_exp)
    return value, weights


def grad_soft_max_conv(a, tau):
    """
    Compute value and gradient of soft_max(conv(a,a), tau) w.r.t. a.
    """
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_and_weights(c, tau)
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return value, g, c


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def get_multi_peak_contributions(a, top_lags, weights):
    """
    Compute combined contribution of each element to multiple peaks.
    top_lags: list of lag indices
    weights: corresponding importance weights
    """
    n = len(a)
    contribs = np.zeros(n)
    for lag, w in zip(top_lags, weights):
        for j in range(n):
            val = 0.0
            # a[j] contributes to c[lag] as:
            # From position i=j in a[i]*a[lag-i]: a[lag-j] if valid
            # From position lag-i=j, i=lag-j in a[i]*a[lag-i]: a[lag-j] if valid
            if 0 <= lag - j and lag - j < n:
                if 2 * j == lag:
                    val += 2.0 * a[j]
                else:
                    val += 2.0 * a[lag - j]
            contribs[j] += w * val
    return contribs


def peak_redistribute(a, n_top_peaks=5, transfer_frac=0.1, n_iter=500, time_limit=60.0):
    """
    Iteratively redistribute mass from high-peak-contributing elements
    to low-peak-contributing elements.
    """
    t0 = time.monotonic()
    current = a.astype(np.float64).copy()
    current_score = local_score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    
    step_size = transfer_frac
    patience = 0
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fft_convolve(current, current)
        S = np.sum(current)
        if S < 0.01:
            break
        
        # Find top peaks
        sorted_idx = np.argsort(c)[::-1]
        top_lags = sorted_idx[:n_top_peaks].tolist()
        top_vals = c[top_lags]
        
        # Weight peaks by their excess over the mean
        mean_c = S * S / (2 * n - 1)
        excess = np.maximum(0, top_vals - mean_c)
        if np.sum(excess) < 1e-10:
            break
        weights = excess / np.sum(excess)
        
        # Compute contributions
        contribs = get_multi_peak_contributions(current, top_lags, weights)
        
        if np.max(contribs) - np.min(contribs) < 1e-10:
            break
        
        # Transfer mass: reduce high-contribution elements, increase low-contribution
        # Use a smooth transfer based on rank
        n_elems = len(current)
        sorted_by_contrib = np.argsort(contribs)
        n_transfer = max(1, n_elems // 10)
        
        high_idx = sorted_by_contrib[-n_transfer:]  # highest contribution
        low_idx = sorted_by_contrib[:n_transfer]  # lowest contribution
        
        # Amount to transfer
        high_vals = current[high_idx]
        low_vals = current[low_idx]
        
        # Only transfer from elements that are positive
        valid_high = high_vals > 0.01
        if not np.any(valid_high):
            break
        
        transfer_amount = step_size * S / (2 * n_transfer)
        
        candidate = current.copy()
        candidate[high_idx[valid_high]] -= transfer_amount
        # Distribute to low elements proportionally to their (inverse) contribution
        low_contribs = np.maximum(contribs[low_idx], 1e-10)
        low_weights = (1.0 / low_contribs) / np.sum(1.0 / low_contribs)
        candidate[low_idx] += transfer_amount * sum(valid_high) / len(valid_high) * low_weights * len(low_idx)
        
        # Simpler: just add evenly to low elements
        candidate = current.copy()
        total_transfer = transfer_amount * np.sum(valid_high)
        candidate[high_idx[valid_high]] -= transfer_amount
        candidate[low_idx] += total_transfer / len(low_idx)
        
        # Clip and normalize
        candidate = np.clip(candidate, 0.0, 1000.0)
        if np.sum(candidate) < 0.01:
            break
        candidate = candidate * (S / np.sum(candidate))
        
        new_score = local_score_fast(candidate)

            current = candidate
            current_score = new_score
            patience = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            patience += 1
            step_size *= 0.95
            if patience > 100:
                break
    
    return best, best_score


def coordinate_descent(a_init, time_limit=60.0, n_sweeps=200):
    """
    Coordinate descent: for each element, try to optimize its value
    while keeping others fixed. Uses the analytical formula:
    optimal a[j] ≈ S0 - 2*c_peak / g_j (clamped to [0, max_val])
    """
    t0 = time.monotonic()
    current = a_init.astype(np.float64).copy()
    current_score = local_score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fft_convolve(current, current)
        S = np.sum(current)
        peak_val = np.max(c)
        
        # Randomize the order for diversity
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute g_j: contribution coefficient of a[j] to the peak
            k_peak = int(np.argmax(c))
            g_j = 0.0
            if 0 <= k_peak - j and k_peak - j < n:
                if 2 * j == k_peak:
                    g_j = 2.0 * current[j]
                else:
                    g_j = 2.0 * current[k_peak - j]
            
            if g_j < 1e-10:
                continue
            
            # Analytical optimum: a[j] = S0 - 2*c_peak/g_j
            # But we need to be more careful: score = 2n * c_peak / S^2
            # d(score)/d(a[j]) = 2n/S^2 * g_j - 2*n*c_peak*2/S^3 = 0
            # => g_j * S = 2 * c_peak
            # => S_new = 2*c_peak / g_j
            # => a[j]_new = S_new - S + a[j] = 2*c_peak/g_j - S + a[j]
            
            a_j_opt = 2.0 * peak_val / g_j - S + current[j]
            a_j_opt = max(0.0, min(1000.0, a_j_opt))
            
            # Try a few candidates around the optimum
            candidates = [a_j_opt, 
                          max(0.0, a_j_opt * 0.5), 
                          max(0.0, a_j_opt * 1.5),
                          current[j] * 0.8,
                          current[j] * 1.2]
            
            best_j_val = current[j]
            best_j_score = current_score
            
            for cv in candidates:
                if cv == current[j]:
                    continue
                test = current.copy()
                test[j] = cv
                ts = local_score_fast(test)
                if ts < best_j_score:
                    best_j_score = ts
                    best_j_val = cv
            
            if best_j_val != current[j]:
                current[j] = best_j_val
                current_score = best_j_score
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
        
        if sweep % 20 == 0:
            print(f"    CD sweep {sweep}: score={current_score:.6f}, best={best_score:.6f}")
    
    return best, best_score


def optimize_log_space(a_init, time_limit, tau_start=None, tau_end=None):
    """
    Optimize in log-space using Adam on log(a) directly.
    This avoids the softmax constraint and allows any positive value.
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    
    # Work in log space
    log_a = np.log(np.maximum(a_init, 1e-8)).astype(np.float64)
    
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    
    if tau_start is None:
        tau_start = 5.0 / n
    if tau_end is None:
        tau_end = 0.1 / n
    
    t0 = time.monotonic()
    best_log_a = log_a.copy()
    best_score = np.inf
    plateau = 0
    plateau_best = np.inf
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        a = np.exp(log_a)
        # Normalize to target sum
        a_norm = a * (S_target / np.sum(a))
        
        value, g_a, c = grad_soft_max_conv(a_norm, tau)
        
        # Gradient of score w.r.t. log_a
        # score = 2n * soft_max(c, tau) / S^2
        # d(score)/d(log_a[j]) = d(score)/d(a[j]) * a[j]
        # d(a_norm[j])/d(log_a[j]) = a_norm[j]/S_target * S_target = a_norm[j] (approximately, ignoring normalization coupling)
        # More precisely: a_norm[j] = a[j] * S_target / S where S = sum(a)
        # d(a_norm[j])/d(log_a[j]) = a[j] * S_target / S - a[j]^2 * S_target / S^2 = a_norm[j] * (1 - a[j]/S)
        
        S_actual = np.sum(a)
        da_norm_d_log_a = a_norm * (1.0 - a / S_actual)
        
        # d(score)/d(a_norm[j]) = 2n/S^2 * g_a[j] - 2*score*a_norm[j]/S
        score_approx = 2.0 * n * value / (S_target ** 2)
        g_score = (2.0 * n / (S_target ** 2)) * g_a - 2.0 * score_approx * a_norm / S_target
        
        g_log_a = g_score * da_norm_d_log_a
        
        # Adam update
        step += 1
        lr = 0.5 / (1.0 + 0.05 * step)
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_log_a
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_log_a * g_log_a)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        log_a = log_a - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Evaluate
        a_check = np.exp(log_a)
        a_check = a_check * (S_target / np.sum(a_check))
        a_check = np.clip(a_check, 0.0, 1000.0)
        score = local_score_fast(a_check)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_log_a = log_a.copy()
            plateau = 0
        else:
            plateau += 1
            if score < plateau_best or plateau_best == np.inf:
                plateau_best = score
        
        if plateau > 500:
            # Perturb to escape
            log_a += 0.1 * np.random.standard_normal(n)
            plateau = 0
    
    best_a = np.exp(best_log_a)
    best_a = best_a * (S_target / np.sum(best_a))
    best_a = np.clip(best_a, 0.0, 1000.0)
    return best_a, best_score


def make_sigmoid_init(n, c, w, base=1.0, rng=None):
    """Create a sigmoid-shaped sequence."""
    i = np.arange(n, dtype=np.float64)
    a = base + 1.0 / (1.0 + np.exp(-(i - c * n) / (w * n)))
    if rng is not None:
        a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_poly_init(n, coeffs, rng=None):
    """Create a polynomial-shaped sequence. coeffs are for polynomial in t = i/(n-1)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1) if n > 1 else np.array([0.5])
    a = np.ones(n)
    for k, c in enumerate(coeffs):
        a += c * (t - 0.5) ** (k + 1)
    if rng is not None:
        a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_wave_init(n, freqs, amps, phase=None, rng=None):
    """Create a multi-wave sequence."""
    i = np.arange(n, dtype=np.float64)
    t = i / n
    a = np.ones(n)
    for f, amp in zip(freqs, amps):
        ph = 0.0 if phase is None else phase[f]
        a += amp * np.sin(2.0 * np.pi * f * t + ph)
    if rng is not None:
        a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining gradient optimization, peak redistribution,
    coordinate descent, and simulated annealing.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Load provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score_fast(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()
    
    # Define search configurations
    # (length, time_fraction, description)
    configs = [
        (start_seq is not None and len(start_seq) or 256, 0.25, "primary"),
        (256, 0.15, "short"),
        (384, 0.15, "medium"),
        (512, 0.15, "long"),
        (128, 0.10, "very_short"),
        (0, 0.15, "redistribute_and_cd"),  # special: work on best found
        (0, 0.05, "final_polish"),
    ]
    
    # Phase 1: Gradient optimization on multiple initializations
    print("\n=== Phase 1: Gradient optimization ===")
    
    # For the primary length, use multiple inits
    primary_n = len(start_seq) if start_seq is not None else 256
    
    inits_primary = []
    if start_seq is not None and len(start_seq) == primary_n:
        inits_primary.append(('provided', start_seq.copy()))
    
    # Sigmoid ramps
    for c in [0.3, 0.5, 0.7]:
        for w in [0.1, 0.2, 0.3]:
            inits_primary.append(('sigmoid', make_sigmoid_init(primary_n, c, w, rng=rng)))
    
    # Polynomial shapes
    for coeffs in [[-0.5], [0.5], [-0.3, -0.2], [0.3, 0.2], [-0.5, 0.3]]:
        inits_primary.append(('poly', make_poly_init(primary_n, coeffs, rng=rng)))
    
    # Wave shapes
    for freqs, amps in [([1, 3], [0.3, 0.1]), ([2, 5], [0.2, 0.15]), ([1, 2, 4], [0.2, 0.15, 0.1])]:
        inits_primary.append(('wave', make_wave_init(primary_n, freqs, amps, rng=rng)))
    
    t_phase1_start = time.monotonic()
    time_phase1 = time_budget * 0.55
    
    # Score all inits and pick top ones
    scored_inits = []
    for name, a_init in inits_primary:
        a_norm = a_init.copy()
        S = np.sqrt(2.0 * primary_n)
        a_norm = a_norm * (S / np.sum(a_norm))
        s = local_score_fast(a_norm)
        scored_inits.append((s, name, a_init.copy()))
    
    scored_inits.sort(key=lambda x: x[0])
    print(f"  Top initializations: {[f'{name}={s:.4f}' for s, name, _ in scored_inits[:5]]}")
    
    # Run gradient optimization on top initializations
    n_top = min(8, len(scored_inits))
    time_per_init = time_phase1 / (n_top + 1)
    
    for idx in range(n_top):
        if time.monotonic() > t_phase1_start + time_phase1:
            break
        s_init, name, a_init = scored_inits[idx]
        time_left = min(time_per_init, t_phase1_start + time_phase1 - time.monotonic())
        if time_left < 2.0:
            break
        
        print(f"  Optimize init '{name}' (initial score={s_init:.4f}), time={time_left:.1f}s")
        a_opt, score_opt = optimize_log_space(a_init, time_left)
        print(f"    Result: score={score_opt:.6f}")
        
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 2: Peak redistribution
    print(f"\n=== Phase 2: Peak redistribution === (current best: {best_score:.6f})")
    
    if best_seq is not None:
        time_phase2 = min(time_budget * 0.15, deadline - time.monotonic())
        if time_phase2 > 5.0:
            # Try different numbers of top peaks
            for n_peaks in [1, 3, 5]:
                if time.monotonic() > deadline - 30:
                    break
                a_red, score_red = peak_redistribute(best_seq, n_top_peaks=n_peaks, 
                                                     transfer_frac=0.05, n_iter=1000,
                                                     time_limit=time_phase2 / 3)
                if score_red < best_score:
                    best_score = score_red
                    best_seq = a_red.copy()
                    print(f"  Redistribution (peaks={n_peaks}): {best_score:.6f} *** NEW BEST ***")
                else:
                    print(f"  Redistribution (peaks={n_peaks}): {score_red:.6f}")
    
    # Phase 3: Coordinate descent
    print(f"\n=== Phase 3: Coordinate descent === (current best: {best_score:.6f})")
    
    if best_seq is not None:
        time_phase3 = min(time_budget * 0.15, deadline - time.monotonic())
        if time_phase3 > 5.0:
            a_cd, score_cd = coordinate_descent(best_seq, time_limit=time_phase3, n_sweeps=500)
            if score_cd < best_score:
                best_score = score_cd
                best_seq = a_cd.copy()
                print(f"  After CD: {best_score:.6f} *** NEW BEST ***")
            else:
                print(f"  After CD: {score_cd:.6f}")
    
    # Phase 4: Additional gradient optimization on the best
    print(f"\n=== Phase 4: Further gradient optimization === (current best: {best_score:.6f})")
    
    if best_seq is not None:
        time_phase4 = min(time_budget * 0.1, deadline - time.monotonic())
        if time_phase4 > 5.0:
            a_opt2, score_opt2 = optimize_log_space(best_seq, time_phase4, 
                                                     tau_start=1.0/len(best_seq),
                                                     tau_end=0.05/len(best_seq))
            if score_opt2 < best_score:
                best_score = score_opt2
                best_seq = a_opt2.copy()
                print(f"  After further grad: {best_score:.6f} *** NEW BEST ***")
            
            # CD again
            remaining = deadline - time.monotonic() - 10
            if remaining > 10:
                a_cd2, score_cd2 = coordinate_descent(best_seq, time_limit=remaining * 0.5, n_sweeps=300)
                if score_cd2 < best_score:
                    best_score = score_cd2
                    best_seq = a_cd2.copy()
                    print(f"  After CD2: {best_score:.6f} *** NEW BEST ***")
    
    # Phase 5: Simulated annealing polish
    print(f"\n=== Phase 5: SA polish === (current best: {best_score:.6f})")
    
    if best_seq is not None:
        time_phase5 = max(5.0, deadline - time.monotonic() - 10)
        if time_phase5 > 5.0:
            current = best_seq.copy()
            current_score = best_score
            n = len(current)
            temperature = 0.002
            cooling = 0.9998
            t0_sa = time.monotonic()
            
            while time.monotonic() - t0_sa < time_phase5:
                candidate = current.copy()
                r = rng.random()
                
                if r < 0.4:
                    # Peak-targeted perturbation
                    conv = fft_convolve(current, current)
                    peak_idx = int(np.argmax(conv))
                    # Reduce top contributors to peak
                    contribs = get_peak_contributions(current, peak_idx)
                    top_k = max(2, n // 8)
                    top_idx = np.argsort(contribs)[-top_k:]
                    candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
                    # Boost others
                    other = np.setdiff1d(np.arange(n), top_idx)
                    if len(other) > 0:
                        n_inc = min(10, len(other))
                        inc_idx = rng.choice(other, size=n_inc, replace=False)
                        candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
                elif r < 0.6:
                    # Block perturbation
                    width = max(1, int(rng.integers(1, max(2, n // 8))))
                    start = int(rng.integers(0, n - width))
                    candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
                elif r < 0.8:
                    # Pair swap (move mass from one element to another)
                    i1 = int(rng.integers(n))
                    i2 = int(rng.integers(n))
                    if i1 != i2:
                        delta = 0.02 * current[i1]
                        candidate[i1] -= delta
                        candidate[i2] += delta
                else:
                    # Global small perturbation
                    candidate *= np.exp(rng.normal(0.0, 0.005, size=n))
                
                candidate = np.clip(candidate, 0.0, 1000.0)
                new_score = local_score_fast(candidate)
                
                if new_score < current_score:
                    current = candidate
                    current_score = new_score
                    if current_score < best_score:
                        best_score = current_score
                        best_seq = current.copy()
                elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
                    current = candidate
                    current_score = new_score
                
                temperature *= cooling
            
            print(f"  After SA: best={best_score:.6f}")
    
    # Phase 6: Try a few different lengths if time permits
    print(f"\n=== Phase 6: Other lengths === (current best: {best_score:.6f})")
    
    if time.monotonic() < deadline - 20:
        for n_try in [192, 320, 448]:
            if time.monotonic() > deadline - 15:
                break
            # Quick optimization
            a0 = make_sigmoid_init(n_try, 0.4, 0.2, rng=rng)
            time_left = min(15.0, deadline - time.monotonic() - 10)
            if time_left < 3:
                break
            a_opt, score_opt = optimize_log_space(a0, time_left)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.6f} *** NEW BEST ***")
            else:
                print(f"  n={n_try}: {score_opt:.6f}")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        
        if final_score < best_score:
            best_score = final_score
            best_seq = np.array(final_list)
        
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
