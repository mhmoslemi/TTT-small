from scipy.fft import fft, ifft

    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:

from scipy.fft import fft, ifft

    """Euclidean projection onto {x >= 0, sum(x) = S}."""

    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]

    n = len(a)

    w = np.exp((c - cmax) / tau)

    W = fft(w, fsize)
    conv = np.real(ifft(A * W))
    return 2.0 * conv[n - 1:2 * n - 1]

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
  - Normalize to sum = sqrt(2*n) so objective = max(self-convolution).
  - Use log-sum-exp smoothed max with annealing for a true gradient.
  - Compute the LSE gradient efficiently via FFT cross-correlation.
  - Multiple structured initializations to avoid local minima.
  - Two-phase: coarse annealing (exploration) then fine top-k polishing.
  - Nesterov-style momentum with adaptive step decay.
  - Periodic verification with evaluate_sequence.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cs - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cs[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2*n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, tau):
    """
    Compute gradient of tau * log(sum_k exp(c[k]/tau)).
    Result: g[m] = 2 * sum_{j=0}^{n-1} a[j] * w[m+j]
    where w = softmax(c / tau).
    Computed via FFT cross-correlation.
    """
    n = len(a)
    cmax = c.max()
    w = np.exp((c - cmax) / tau)
    w /= w.sum()
    # g[m] / 2 = sum_j a[j] * w[m+j] = (a_rev * w)[m + n - 1]
    conv_len = 3 * n - 2
    fsize = 1 << (conv_len - 1).bit_length()
    A = fft(a[::-1], fsize)
    W = fft(w, fsize)
    conv = np.real(ifft(A * W))
    return 2.0 * conv[n - 1:2 * n - 1]


def topk_subgradient(a, c, top_k):
    """Subgradient using top-k active lags, weighted by their values."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    grad = np.zeros(n, dtype=np.float64)
    vals = c[top_idx]
    total = vals.sum()
    if total < 1e-15:
        return grad
    weights = vals / total
    for wval, k in zip(weights, top_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += wval * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def run_lse_optimize(a_init, n, S, max_time, rng, tau_start=1.0, tau_end=0.005,
                     n_phases=30, eta_base=0.5, beta=0.9):
    """
    LSE-smoothed optimization with annealing and momentum.
    Returns (best_a, best_maxc).
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    a = a * (S / s)

    best_score = np.inf
    best_a = a.copy()
    momentum = np.zeros(n, dtype=np.float64)
    start = time.monotonic()
    t = 0
    no_improve_count = 0
    patience = 8000

    while time.monotonic() - start < max_time:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max_time)

        # Anneal tau
        tau = tau_start * (tau_end / tau_start) ** frac

        c = fft_conv_self(a)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > patience:
            # Random restart within the neighborhood
            if rng.random() < 0.3:
                noise = rng.normal(0.0, 0.05 * S / n, n)
                a = project_simplex(a + noise, S)
            no_improve_count = 0
            continue

        # Compute gradient
        g = lse_gradient(a, c, tau)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        # Adaptive step: start large, decay over time
        eta = eta_base * (1.0 + 0.5 * (1.0 - frac))

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        a_new = a - eta * momentum

        a = project_simplex(a_new, S)

        # Occasional perturbation for exploration
        if t % 10000 == 0 and frac < 0.8:
            if rng.random() < 0.05:
                noise = rng.normal(0.0, 0.02 * S / n, n)
                a = project_simplex(a + noise, S)

    return best_a, best_score


def run_topk_polish(a_init, n, S, max_time, rng, top_k=32, eta_init=0.3):
    """Fine polishing with top-k subgradient and adaptive step."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    a = a * (S / s)

    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    t = 0
    eta = eta_init

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_conv_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        g = topk_subgradient(a, c, top_k)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        # Decay step size
        eta = eta_init * max(0.05, 1.0 / (1.0 + 0.002 * t))

        a_new = a - eta * g
        a = project_simplex(a_new, S)

    return best_a, best_score


def make_initializations(rng, n):
    """Generate diverse initializations of length n."""
    S = np.sqrt(2.0 * float(n))
    inits = []

    # Try the given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0.01:
            g = np.clip(given, 0, 1000)
            g = g * (S / g.sum())
            inits.append(("given", g))
    except Exception:
        pass

    # Uniform
    inits.append(("uniform", np.full(n, S / n, dtype=np.float64)))

    # Random
    r = rng.random(n)
    r = r / r.sum() * S
    inits.append(("random", r))

    # Cosine modulated
    t_arr = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3]:
        alpha = 0.3
        mod = 1.0 + alpha * np.cos(2.0 * np.pi * freq * t_arr / n)
        mod = np.maximum(mod, 0.01)
        mod = mod / mod.sum() * S
        inits.append(("cos%d_%d" % (freq, round(alpha * 100)), mod))

    # Two-frequency
    mod = 1.0 + 0.25 * np.cos(2.0 * np.pi * 1.0 * t_arr / n) + 0.15 * np.cos(2.0 * np.pi * 3.0 * t_arr / n)
    mod = np.maximum(mod, 0.01)
    mod = mod / mod.sum() * S
    inits.append(("two_freq", mod))

    # Gaussian window
    center = n / 2.0
    width = n / 4.0
    mod = np.exp(-0.5 * ((t_arr - center) / width) ** 2)
    mod = mod / mod.sum() * S
    inits.append(("gaussian", mod))

    # Triangular (inverted)
    t_norm = t_arr / (n - 1)
    mod = 1.0 - 0.5 * np.abs(2.0 * t_norm - 1.0)
    mod = mod / mod.sum() * S
    inits.append(("trig_inv", mod))

    # Barkar-like (alternating with taper)
    mod = np.ones(n)
    taper = np.linspace(0.3, 1.0, n)
    mod = taper * (1.0 + 0.2 * np.cos(2.0 * np.pi * t_arr / (n / 4.0)))
    mod = np.maximum(mod, 0.01)
    mod = mod / mod.sum() * S
    inits.append(("barker_like", mod))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):

    S = np.sqrt(2.0 * float(n))

    global_best = np.inf
    global_a = None

    inits = make_initializations(rng, n)

    # Phase 1: LSE optimization on each initialization
    n_inits = len(inits)
    explore_time = (deadline - time.monotonic()) * 0.65
    per_init_time = explore_time / max(1, n_inits)

    for name, a_init in inits:
        if time.monotonic() > deadline - (deadline - time.monotonic()) * 0.35 - 5.0:
            break
        t_left = max(1.0, per_init_time - (time.monotonic() - (deadline - explore_time)))
        a_result, score = run_lse_optimize(
            a_init, n, S, t_left, rng,
            tau_start=1.0, tau_end=0.005,
            n_phases=30, eta_base=0.5, beta=0.9
        )
        if score < global_best:
            global_best = score
            global_a = a_result.copy()
            print("Phase1 %s: maxc=%.6f (global=%.6f) t=%.0fs" % (
                name, score, global_best, time.monotonic() - (deadline - explore_time)))

    # Phase 2: Polish the best with top-k subgradient
    left = deadline - time.monotonic()
    if left > 5.0 and global_a is not None:
        polish_time = left * 0.6
        a_polished, score_pol = run_topk_polish(
            global_a, n, S, polish_time, rng,
            top_k=32, eta_init=0.3
        )
        if score_pol < global_best:
            global_best = score_pol
            global_a = a_polished.copy()
            print("Phase2 polish: maxc=%.6f" % score_pol)

        # Phase 3: Further LSE refinement with very small tau
        left2 = deadline - time.monotonic()
        if left2 > 3.0 and global_a is not None:
            a_final, score_fin = run_lse_optimize(
                global_a, n, S, left2, rng,
                tau_start=0.01, tau_end=0.001,
                n_phases=10, eta_base=0.2, beta=0.95
            )
            if score_fin < global_best:
                global_best = score_fin
                global_a = a_final.copy()
                print("Phase3 refine: maxc=%.6f" % score_fin)

    # Verify
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:

        ver = evaluate_sequence(global_a.tolist())

        else:
            print("Verification failed, using fallback")
            global_a = np.full(n, S / n)
    else:
        global_a = np.full(n, S / n)
        ver = evaluate_sequence(global_a.tolist())
        print("Fallback uniform: %.6f" % ver)

    return [float(x) for x in global_a]
