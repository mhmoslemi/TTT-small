"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses corrected projected gradient descent with tau annealing, 
Nesterov momentum, and multiple initializations.

import numpy as np
import time
from scipy.optimize import minimize


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (a.sum() ** 2)


def compute_smooth_obj(a, tau):
    """Compute smooth approximation of max(conv(a,a)) and its gradient."""
    c = np.convolve(a, a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    # Correct gradient: 2 * correlate(q, a, 'valid')
    grad = 2.0 * np.correlate(q, a, 'valid')
    return tau * (np.log(exp_c.sum()) + c.max() / tau), grad, c


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = a.size
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s == 0:
        return np.full(n, S / n)
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, a.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    res[res.argmax()] += S - res.sum()
    return res

    """Projected gradient descent with tau annealing, momentum, and adaptive lr."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = (a0.astype(np.float64) / max(a0.sum(), 1e-12) * S).copy()
    a = project_simplex(a, S)
    
    best = a.copy()
    best_score = compute_obj(best)
    
    t0 = time.monotonic()
    lr = lr0
    step = 0
    v = np.zeros(n)  # momentum
    beta = 0.9
    
    # Track the actual max for convergence
    last_cmax = np.convolve(a, a).max()
    
    while time.monotonic() - t0 < t_budget:
        el = time.monotonic() - t0
        f = min(1.0, el / max(t_budget, 1e-6))
        # Cosine annealing for tau
        tau = tau_lo + 0.5 * (tau_hi - tau_lo) * (1 + np.cos(np.pi * f))
        
        _, grad, c = compute_smooth_obj(a, tau)
        
        # Project gradient onto tangent space of simplex
        grad_tangent = grad - np.mean(grad)
        
        # Momentum update
        v = beta * v + (1 - beta) * grad_tangent
        direction = -v
        
        # Line search
        alpha = lr
        for _ in range(10):
            a_new = project_simplex(a + alpha * direction, S)
            c_new = np.convolve(a_new, a_new)
            cmax_new = c_new.max()
            if cmax_new < last_cmax * (1 - 1e-10):
                break
            alpha *= 0.5
        
        a = a_new
        last_cmax = c_new.max()
        
        step += 1
        
        if step % 20 == 0:
            score = compute_obj(a)
            if score < best_score:
                best_score = score
                best = a.copy()
                if lr * 1.1 < lr0 * 5:
                    lr *= 1.1
            else:
                lr = max(0.01, lr * 0.7)
        
        if verbose and step % 500 == 0:
            print(f"      step={step}, cmax={last_cmax:.4f}, score={compute_obj(a):.6f}, tau={tau:.4f}, lr={lr:.4f}")
    
    final_score = compute_obj(best)
    if verbose:
        print(f"    run_opt done: steps={step}, cmax={np.convolve(best,best).max():.6f}, score={final_score:.6f}")
    return best, final_score

def compute_obj(a):
    n = len(a)
    s = a.sum()

        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def smooth_max_and_grad(a, tau):
    """Compute soft-max of conv(a,a) and its gradient w.r.t. a."""
    c = np.convolve(a, a)
    c_max = c.max()
    c_shift = c - c_max
    exp_c = np.exp(c_shift / tau)
    Z = exp_c.sum()
    q = exp_c / Z
    # Correct gradient: d/d a[i] smooth_max = 2 * sum_k q[k] * a[k-i] * [valid]
    # = 2 * np.correlate(q, a, 'valid')[i]

def init_chirp(n, seed=0):

def init_random(n, seed=0):
    return np.random.default_rng(seed).exponential(1.0, size=n)


def init_bimodal(n, seed=0):
    """Bimodal initialization - two Gaussian bumps."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64) / n
    center1 = rng.uniform(0.2, 0.4)
    center2 = rng.uniform(0.6, 0.8)
    width = rng.uniform(0.05, 0.15)
    a = 1.0 + 2.0 * np.exp(-((t - center1) / width) ** 2) + 2.0 * np.exp(-((t - center2) / width) ** 2)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

    """Log-uniform initialization."""
    rng = np.random.default_rng(seed)
    a = np.exp(rng.uniform(-1, 1, size=n))
    return a


def run_projected_grad(a0, t_budget, tau_hi=1.0, tau_lo=0.01, lr0=0.5, momentum=0.9, verbose=False):
    """Projected gradient descent with Nesterov momentum and cosine annealing."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = (a0.astype(np.float64) / max(a0.sum(), 1e-12) * S).copy()
    a = project_simplex(a, S)
    
    best = a.copy()
    best_cmax = np.convolve(best, best).max()
    best_score = compute_obj(best)
    
    t0 = time.monotonic()
    lr = lr0
    step = 0
    v = np.zeros(n)  # velocity for Nesterov momentum
    
    while time.monotonic() - t0 < t_budget:
        el = time.monotonic() - t0
        f = min(1.0, el / max(t_budget, 1e-6))
        # Cosine annealing
        tau = tau_lo + 0.5 * (tau_hi - tau_lo) * (1.0 + np.cos(np.pi * f))
        
        # Nesterov: look-ahead
        a_look = project_simplex(a - momentum * v, S)
        _, grad = smooth_max_and_grad(a_look, tau)
        
        # Project gradient to tangent space
        grad_tan = grad - np.mean(grad)
        
        # Update velocity
        v = momentum * v + lr * grad_tan
        
        # Update
        a_new = project_simplex(a - v, S)
        
        step += 1
        
        # Periodic evaluation
        if step % 10 == 0:
            c = np.convolve(a_new, a_new)
            cmax = c.max()
            score = 2.0 * n * cmax / S**2
            if score < best_score:
                best_score = score
                best_cmax = cmax
                best = a_new.copy()
                if lr < lr0 * 3:
                    lr *= 1.05
            elif step % 50 == 0:
                lr = max(0.01, lr * 0.8)
        
        a = a_new
    
    final_score = compute_obj(best)
    if verbose:
        print(f"      steps={step}, cmax={np.convolve(best,best).max():.4f}, score={final_score:.6f}")
    return best, final_score


def run_lbfgs(a0, t_budget, verbose=False):
    """L-BFGS-B optimization on log-transformed variables."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a_norm = np.maximum(a0.astype(np.float64), 1e-10)
    a_norm = a_norm / a_norm.sum() * S
    x0 = np.log(a_norm)
    
    best_x = x0.copy()
    best_score = compute_obj(np.exp(x0))
    
    tau = 0.03
    
    def obj_and_grad(x):
        a = np.exp(x)
        s = a.sum()
        a_n = a / s * S
        c = np.convolve(a_n, a_n)
        c_max = c.max()
        c_shift = c - c_max
        exp_c = np.exp(c_shift / tau)
        Z = exp_c.sum()
        val = tau * np.log(Z) + c_max
        
        q = exp_c / Z
        grad_an = 2.0 * np.correlate(q, a_n, 'valid')
        
        # Chain rule for log transform
        e = np.exp(x)
        s = e.sum()
        # a_n_i = S * e_i / s
        # d a_n_i / d x_j = S * (delta_ij * e_i * s - e_i * e_j) / s^2
        # = (delta_ij - e_j/s) * S * e_i / s
        # = (delta_ij - a_n_j/S) * a_n_i
        # grad_x_j = sum_i grad_an_i * d a_n_i / d x_j
        # = sum_i grad_an_i * (delta_ij - a_n_j/S) * a_n_i
        # = grad_an_j * a_n_j - (a_n_j/S) * sum_i grad_an_i * a_n_i
        
        dot = np.dot(grad_an, a_n)
        grad_x = grad_an * a_n - (a_n / S) * dot
        
        return val, grad_x
    
    t0 = time.monotonic()
    x = x0.copy()
    
    while time.monotonic() - t0 < t_budget:
        try:
            res = minimize(obj_and_grad, x, method='L-BFGS-B',
                          options={'maxiter': 200, 'maxfun': 500, 'ftol': 1e-10})
            x = res.x
            a_test = np.exp(x)
            a_test = a_test / a_test.sum() * S
            score = compute_obj(a_test)
            if score < best_score:
                best_score = score
                best_x = x.copy()
            if res.status == 0:
                # Try with smaller tau
                tau = max(0.005, tau * 0.5)
            if res.status not in (0, 1):
                break
        except Exception:
            break
    
    a_best = np.exp(best_x)
    a_best = a_best / a_best.sum() * S
    a_best = np.maximum(a_best, 0.0)
    final_score = compute_obj(a_best)
    if verbose:
        print(f"      L-BFGS: score={final_score:.6f}")
    return a_best, final_score

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Key improvements: corrected gradient (removed erroneous term), Nesterov momentum,
cosine annealing, L-BFGS polishing, perturbation restarts, diverse initializations.

import numpy as np

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def smooth_max_and_grad(a, tau):
    """Compute soft-max approx of max(conv(a,a)) and correct gradient w.r.t. a.
    
    Gradient derivation: d/d a[i] conv(a,a)[k] = 2*a[k-i] for valid k-i in [0,n-1].
    So grad[i] = 2 * sum_k q[k] * a[k-i] = 2 * correlate(q, a, 'valid')[i].
    """
    c = np.convolve(a, a)
    c_max = c.max()
    c_shift = c - c_max
    exp_c = np.exp(c_shift / tau)
    Z = exp_c.sum()
    q = exp_c / Z
    grad = 2.0 * np.correlate(q, a, 'valid')
    smax = tau * np.log(Z) + c_max
    return smax, grad


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = a.size
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, a.max()
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    deficit = S - res.sum()
    if deficit > 0:
        res[res.argmax()] += deficit
    return res


def normalize_to_simplex(a, S):
    """Normalize a to lie on the simplex {x>=0, sum=S}."""
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.full(a.size, S / a.size)
    return project_simplex(a / s * S, S)


def init_chirp(n, seed=0):
    """Chirp initialization targeting flat spectrum."""
    t = np.arange(n, dtype=np.float64) - (n - 1) * 0.5
    a = 1.0 + 0.9 * np.cos(np.pi * t * t / n)
    rng = np.random.default_rng(seed)
    a += 0.1 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_chirp_inv(n, seed=0):
    """Inverse chirp (conjugate symmetry)."""
    t = np.arange(n, dtype=np.float64) - (n - 1) * 0.5
    a = 1.0 + 0.9 * np.cos(-np.pi * t * t / n + np.pi * t / n)
    rng = np.random.default_rng(seed)
    a += 0.1 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_random(n, seed=0):
    return np.random.default_rng(seed).exponential(1.0, size=n)


def init_bimodal(n, seed=0):
    """Two Gaussian bumps."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64) / max(n, 1)
    c1 = rng.uniform(0.15, 0.35)
    c2 = rng.uniform(0.65, 0.85)
    w = rng.uniform(0.04, 0.12)
    a = 1.0 + 2.0 * np.exp(-((t - c1) / w) ** 2) + 2.0 * np.exp(-((t - c2) / w) ** 2)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_log_uniform(n, seed=0):
    rng = np.random.default_rng(seed)
    return np.exp(rng.uniform(-1.5, 1.5, size=n))


def init_ramp(n, seed=0):
    """Ramp with noise."""
    rng = np.random.default_rng(seed)
    t = np.linspace(0, 1, n)
    a = 0.5 + t  # linear ramp
    a += 0.3 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def run_projected_grad(a0, t_budget, tau_hi=1.0, tau_lo=0.01, lr0=0.5,
                       momentum=0.9, verbose=False):
    """Projected gradient with Nesterov momentum, cosine tau annealing, adaptive lr."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = normalize_to_simplex(a0.astype(np.float64), S)

    best = a.copy()
    best_score = compute_obj(best)

    t0 = time.monotonic()
    lr = lr0
    step = 0
    v = np.zeros(n)

    while time.monotonic() - t0 < t_budget:
        el = time.monotonic() - t0
        f = min(1.0, el / max(t_budget, 1e-6))
        # Cosine annealing
        tau = tau_lo + 0.5 * (tau_hi - tau_lo) * (1.0 + np.cos(np.pi * f))

        # Nesterov look-ahead
        a_look = project_simplex(a - momentum * v, S)
        _, grad = smooth_max_and_grad(a_look, tau)

        # Tangent projection (stay on simplex)
        grad_tan = grad - np.mean(grad)

        # Momentum update
        v = momentum * v + lr * grad_tan

        # Update with backtracking
        a_new = project_simplex(a - v, S)

        step += 1

        if step % 10 == 0:
            score = compute_obj(a_new)
            if score < best_score:
                best_score = score
                best = a_new.copy()
                if lr < lr0 * 4.0:
                    lr *= 1.08
            elif step % 100 == 0:
                lr = max(0.005, lr * 0.7)

        a = a_new

    final_score = compute_obj(best)
    if verbose:
        cmax = np.convolve(best, best).max()
        print(f"      steps={step}, cmax={cmax:.4f}, score={final_score:.6f}")
    return best, final_score


def run_lbfgs(a0, t_budget, tau_start=0.05, verbose=False):
    """L-BFGS-B on log-reparameterized variables with decreasing tau."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a_pos = np.maximum(a0.astype(np.float64), 1e-8)
    x0 = np.log(a_pos / a_pos.sum() * S)

    best_x = x0.copy()
    best_score = compute_obj(normalize_to_simplex(np.exp(x0), S))
    tau = tau_start

    t0 = time.monotonic()
    x = x0.copy()

    while time.monotonic() - t0 < t_budget:
        # Define objective and gradient with current tau
        def obj_and_grad(x_val, tau_val=tau):
            e = np.exp(x_val)
            s = e.sum()
            a_n = e / s * S
            c = np.convolve(a_n, a_n)
            c_max = c.max()
            c_shift = c - c_max
            exp_c = np.exp(c_shift / tau_val)
            Z = exp_c.sum()
            val = tau_val * np.log(Z) + c_max
            q = exp_c / Z
            grad_an = 2.0 * np.correlate(q, a_n, 'valid')
            # Chain rule: grad_x = a_n * (grad_an - dot(grad_an, a_n)/S)
            dot = np.dot(grad_an, a_n)
            grad_x = a_n * (grad_an - dot / S)
            return val, grad_x

        try:
            res = minimize(obj_and_grad, x, method='L-BFGS-B',
                          jac=True,
                          options={'maxiter': 300, 'maxfun': 1000, 'ftol': 1e-12})
            x = np.maximum(res.x, -20.0)  # clip to avoid underflow
            a_test = normalize_to_simplex(np.exp(x), S)
            score = compute_obj(a_test)
            if score < best_score:
                best_score = score
                best_x = x.copy()
            # Decrease tau for sharper max approximation
            tau = max(0.005, tau * 0.5)
            if res.status == 0 and tau <= 0.01:
                break
        except Exception:
            break

    a_best = normalize_to_simplex(np.exp(best_x), S)
    final_score = compute_obj(a_best)
    if verbose:
        print(f"      L-BFGS: score={final_score:.6f}, tau_final={tau:.4f}")
    return a_best, final_score


def perturb_and_reopt(bs, sigma, t_budget, tau_hi, tau_lo, lr0):
    """Perturb best solution and re-optimize."""
    n = bs.size
    S = np.sqrt(2.0 * n)
    rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
    a_pert = bs + sigma / S * rng.standard_normal(n)
    a_pert = normalize_to_simplex(a_pert, S)
    return run_projected_grad(a_pert, t_budget, tau_hi, tau_lo, lr0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: 4 phases with corrected gradient.
    Phase 1: Diverse exploration (300s)
    Phase 2: Deep optimization on best (700s)
    Phase 3: L-BFGS + perturb restarts (600s)
    Phase 4: Final micro-polish (500s + buffer)
    """
    t_start = time.monotonic()
    t_deadline = t_start + budget_s - 20.0
    bs = None
    bsc = np.inf

    # Get provided sequence
    prov = None
    try:
        prov = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if prov.size > 0 and np.isfinite(prov).all():
            prov = np.maximum(prov, 0.0)
            if prov.sum() >= 0.01:
                pass
            else:
                prov = None
    except Exception:
        prov = None

    print(f"Provided: {prov is not None}, len={len(prov) if prov is not None else 'N/A'}")
    if prov is not None:
        ps = compute_obj(prov)
        print(f"Provided score: {ps:.6f}")
        if ps < bsc:
            bsc = ps
            bs = prov.copy()

    # ===== Phase 1: Quick exploration =====
    print("\n=== Phase 1: Exploration ===")

    candidates = []
    if prov is not None:
        candidates.append((len(prov), 'provided', 0))

    init_fns = {
        'chirp': init_chirp,
        'chirp_inv': init_chirp_inv,
        'random': init_random,
        'bimodal': init_bimodal,
        'log_uni': init_log_uniform,
        'ramp': init_ramp,
    }

    for n in [2400, 3000, 4000]:
        for name, fn in init_fns.items():
            candidates.append((n, name, 42 if name != 'provided' else 0))

    # Limit candidates to fit time
    p1_time = 300
    time_per = max(15.0, p1_time / len(candidates))
    t_p1_start = time.monotonic()

    for i, (n, init_name, sd) in enumerate(candidates):
        rem = t_deadline - time.monotonic()
        if rem < 100:
            break
        tb = min(time_per, p1_time - (time.monotonic() - t_p1_start))
        if tb < 10:
            break

        if init_name == 'provided' and prov is not None:
            a0 = prov.copy()
        else:
            fn = init_fns.get(init_name, init_random)
            a0 = fn(n, sd)

        print(f"  P1[{i+1}/{len(candidates)}] n={n:4d} {init_name:10s} t={tb:4.0f}s", end=" ")
        a_r, sc = run_projected_grad(a0, tb, tau_hi=1.0, tau_lo=0.05, lr0=0.5)
        print(f"-> {sc:.6f}")

        if sc < bsc:
            bsc = sc
            bs = a_r.copy()
            print(f"    *** NEW BEST: {bsc:.6f}")

    # ===== Phase 2: Deep optimization =====
    print(f"\n=== Phase 2: Deep optimization (score={bsc:.6f}) ===")

    if bs is not None:
        n_best = bs.size
        rounds = [
            (0.5, 0.02, 0.3, 0.30),
            (0.25, 0.01, 0.2, 0.25),
            (0.1, 0.005, 0.15, 0.20),
            (0.05, 0.003, 0.08, 0.15),
        ]
        for tau_hi, tau_lo, lr, frac in rounds:
            rem = t_deadline - time.monotonic()
            if rem < 50:
                break
            tb = min(150.0, rem * frac)
            if tb < 20:
                break
            print(f"  tau=[{tau_hi:.3f},{tau_lo:.4f}] lr={lr:.3f} n={n_best} t={tb:.0f}s", end=" ")
            a_r, sc = run_projected_grad(bs, tb, tau_hi, tau_lo, lr)
            print(f"-> {sc:.6f}")
            if sc < bsc:
                bsc = sc
                bs = a_r.copy()
                print(f"    *** BEST: {bsc:.6f}")

    # ===== Phase 3: L-BFGS + perturbation =====
    print(f"\n=== Phase 3: L-BFGS + perturbation (score={bsc:.6f}) ===")

    if bs is not None:
        rem = t_deadline - time.monotonic()
        if rem > 100:
            # L-BFGS
            tb_lbfgs = min(300.0, rem * 0.4)
            print(f"  L-BFGS: n={len(bs)} t={tb_lbfgs:.0f}s")
            a_r, sc = run_lbfgs(bs, tb_lbfgs, tau_start=0.05, verbose=True)
            if sc < bsc:
                bsc = sc
                bs = a_r.copy()
                print(f"    *** BEST: {bsc:.6f}")

        # Perturbation restarts
        for sigma_frac in [0.05, 0.02, 0.01]:
            rem = t_deadline - time.monotonic()
            if rem < 40:
                break
            tb = min(60.0, rem * 0.3)
            if tb < 15:
                break
            sigma = sigma_frac * np.sqrt(2.0 * len(bs))
            print(f"  Perturb sigma={sigma_frac:.3f} t={tb:.0f}s", end=" ")
            a_r, sc = perturb_and_reopt(bs, sigma, tb, 0.1, 0.01, 0.2)
            print(f"-> {sc:.6f}")
            if sc < bsc:
                bsc = sc
                bs = a_r.copy()
                print(f"    *** BEST: {bsc:.6f}")

    # ===== Phase 4: Final polish =====
    print(f"\n=== Phase 4: Final polish (score={bsc:.6f}) ===")

    if bs is not None:
        rem = t_deadline - time.monotonic()
        if rem > 30:
            tb = rem * 0.85
            print(f"  Final: n={len(bs)} t={tb:.0f}s")
            a_r, sc = run_projected_grad(bs, tb, tau_hi=0.03, tau_lo=0.002, lr0=0.05,
                                         momentum=0.7, verbose=True)
            if sc < bsc:
                bsc = sc
                bs = a_r.copy()
            else:
                # Try L-BFGS one more time
                rem2 = t_deadline - time.monotonic()
                if rem2 > 20:
                    a_r2, sc2 = run_lbfgs(bs, rem2 * 0.8, tau_start=0.02, verbose=True)
                    if sc2 < bsc:
                        bsc = sc2
                        bs = a_r2.copy()

    # Fallback
    if bs is None:
        bs = np.ones(2000, dtype=np.float64)
        bsc = compute_obj(bs)

    # Final validation
    final_eval = evaluate_sequence([float(x) for x in bs])
    total_time = time.monotonic() - t_start
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(bs)}, compute_score={bsc:.6f}, eval={final_eval:.6f}")
    print(f"Total time: {total_time:.1f}s")
    print(f"{'='*50}")

    return [float(x) for x in bs]
