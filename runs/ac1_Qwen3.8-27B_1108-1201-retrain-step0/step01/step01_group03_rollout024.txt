- max conv = 2 (for i=j) or 1 (for i≠j), so max = 2 (from the diagonal)

    Solve: minimize M
    subject to: conv(f, g)[k] <= M for all k
    g[i] >= 0
    sum(g) = S_target

    # 2. -g[i] <= 0 for each i (i.e., g[i] >= 0)

    num_constraints = (2 * n - 1) + n + 2

        for i in range(max(0, k - n + 1), min(n, k + 1)):

            if 0 <= j < n:

        A[row, n] = -1.0  # coefficient of M
        b[row] = 0.0
        row += 1

    for i in range(n):

        b[row] = 0.0
        row += 1

    for i in range(n):
        A[row, i] = 1.0
    b[row] = S_target
    row += 1

    for i in range(n):
        A[row, i] = -1.0
    b[row] = -S_target
    row += 1

            best_score = best_blend_score
            f = best_blend.copy()
        
        # Update f for next iteration
        f = best_blend.copy()

Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses alternating LP (minimize peak of cross-convolution), gradient polishing,

import time
from scipy.optimize import linprog

    n = len(a)

    max_c = float(c.max())

    if S < 1e-15:
        return a
    return a * (target / S)

"""

Uses alternating LP (directly minimizing the cross-convolution peak), 
softmax gradient polishing, and iterative refinement from known good initializations.
Key improvement over prior work: the LP directly minimizes the peak M subject to
fixed-sum normalization, rather than using an indirect max-sum formulation.
"""

import numpy as np

from scipy.optimize import linprog


def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return float('inf')
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=1.0):
    """Normalize sequence to have a specific sum."""
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (target / S)


def _lp_minimize_peak(f, target_sum=1.0):
    """
    Solve LP: minimize M
    subject to: conv(f, g)[k] - M <= 0 for k = 0, ..., 2n-2
                g[i] >= 0 for all i
                sum(g) = target_sum
    Returns (g, M_value) or (None, None) on failure.
    """
    n = len(f)
    # Variables: [g_0, ..., g_{n-1}, M], total n+1
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M

    num_rows = (2 * n - 1) + n + 2

            # Also try hardmax gradient
            for _ in range(grad_count // 2):
                if time.monotonic() > deadline - 8.0:
                    break
                a = _gradient_step_hardmax(a, lr=0.01)
                s = _compute_obj(a)
                if s < best_score:
                    best_score = s
                    best = a.copy()

        if (it + 1) % 500 == 0:
            print(f"      LP iter {it+1}: best={best_score:.8f}")

    return best, best_score


def _optimize_gradient(a0, deadline, max_iters=20000):
    """Pure gradient optimization with adaptive learning rate and tau."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = _normalize(a, 1.0)

    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 8.0:
            break

        # Anneal parameters
        progress = min(1.0, it / max_iters)
        tau = 100.0 + 400.0 * progress  # Increase focus on peak
        lr = 0.03 * (1.0 + np.cos(np.pi * progress)) * 0.5 + 0.003

        a = _gradient_step(a, lr=lr, tau=tau)
        s = _compute_obj(a)

        if s < best_score - 1e-10:
            best_score = s
            best = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 200:
                # Reset to best and try different tau
                a = best.copy()
                tau = 50.0 + 100.0 * np.random.default_rng(it).random()
                no_improve_count = 0

        if (it + 1) % 5000 == 0:
            print(f"      Grad iter {it+1}: best={best_score:.8f}")

    return best, best_score


def _polish_coordinate(a0, deadline):
    """Final coordinate descent polish: try small perturbations of individual elements."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = _normalize(a, 1.0)
    best_score = _compute_obj(a)
    best = a.copy()
    rng = np.random.default_rng(123)

    for it in range(50000):
        if time.monotonic() > deadline - 5.0:
            break

        # Pick a random coordinate to perturb
        i = rng.integers(0, n)
        delta = rng.random() * 0.02 * a.mean()
        direction = 1.0 if rng.random() > 0.5 else -1.0

        a[i] = max(0.0, a[i] + direction * delta)
        a = _normalize(a, 1.0)
        s = _compute_obj(a)

        if s < best_score - 1e-12:
            best = a.copy()
            best_score = s
        else:
            a[i] = max(0.0, best[i])  # revert
            a = _normalize(a, 1.0)

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses alternating LP optimization (directly minimizing
    the cross-convolution peak) combined with softmax gradient descent and
    coordinate descent polishing. Explores multiple starting points.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 6.0)

    best_seq = None
    best_score = float('inf')

    # --- Gather starting sequences ---
    starts = []

    # Primary: use provided global solution
    global_seq = None
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 32 and np.all(np.isfinite(gs)):
            global_seq = gs
    except NameError:
        pass

    if global_seq is not None:
        # Use full length and truncated versions
        for L in [512, 256]:
            if L <= len(global_seq):
                starts.append(global_seq[:L].copy())
        if len(global_seq) != 512 and len(global_seq) != 256:
            starts.append(global_seq.copy())

    # Random starts
    for n in [256, 512]:
        starts.append(rng.random(n) * 0.4 + 0.1)

    # Structured starts (chirp, Gaussian, etc.)
    for n in [256, 512]:
        i = np.arange(n, dtype=np.float64)
        starts.append(0.3 + 0.7 * np.abs(np.cos(2 * np.pi * i * 7.0 / n)))
        center = n / 2.0
        starts.append(np.exp(-0.5 * ((i - center) / (n * 0.3)) ** 2))
        starts.append(np.abs(np.sin(2 * np.pi * i * 2.0 / n)) ** 0.5 + 0.1)

    # Deduplicate by length and keep best initial score per length
    starts.sort(key=lambda x: _compute_obj(x))

    # Keep at most 6 best starts to manage time
    seen_lens = set()
    filtered_starts = []
    for s in starts:
        n = len(s)
        if n in seen_lens:
            continue
        seen_lens.add(n)
        filtered_starts.append(s)
        if len(filtered_starts) >= 6:
            break
    starts = filtered_starts

    print(f"Total budget: {budget_s}s, using {len(starts)} starting sequences")
    print(f"Initial scores: {[f'{_compute_obj(s):.6f} (n={len(s)})' for s in starts]}")

    # --- Optimize each starting sequence ---
    for idx, a0 in enumerate(starts):
        if time.monotonic() > deadline - 60.0:
            break

        n = len(a0)
        init_score = _compute_obj(a0)
        print(f"\n[{idx+1}/{len(starts)}] n={n}, init_score={init_score:.6f}")

        if init_score >= best_score and best_score < float('inf'):
            # Skip if initial is already worse than best found
            # (unless it's the only option)
            if idx > 0 and best_score < init_score * 0.9:
                continue

        t0 = time.monotonic()

        # Phase 1: LP-based optimization (main workhorse)
        remaining = deadline - time.monotonic()
        lp_deadline = t_start + min(budget_s - 30, t0 + remaining * 0.6)
        seq1, score1 = _optimize_lp_based(a0, lp_deadline,
                                           max_lp_iters=5000,
                                           grad_every=30,
                                           grad_count=60)
        print(f"  After LP: {score1:.8f} ({time.monotonic()-t0:.1f}s)")

        # Phase 2: Gradient polish
        if time.monotonic() < deadline - 30.0:
            grad_deadline = t_start + min(budget_s - 15, t0 + remaining * 0.85)
            seq2, score2 = _optimize_gradient(seq1, grad_deadline, max_iters=30000)
            print(f"  After grad: {score2:.8f} ({time.monotonic()-t0:.1f}s)")

        # Phase 3: Coordinate descent polish
        if time.monotonic() < deadline - 10.0:
            coord_deadline = deadline - 5.0
            seq3, score3 = _polish_coordinate(seq2, coord_deadline)
            print(f"  After coord: {score3:.8f} ({time.monotonic()-t0:.1f}s)")

        # Update best
        final_seq = seq3
        final_score = score3
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"  ** New global best: {best_score:.8f} **")

    # --- Final refinement of the best sequence found ---
    if best_seq is not None and time.monotonic() < deadline - 10.0:
        print(f"\nFinal refinement (best={best_score:.8f})")
        for pass_i in range(4):
            if time.monotonic() > deadline - 5.0:
                break
            seq, score = _optimize_lp_based(best_seq, deadline - 5.0,
                                             max_lp_iters=2000,
                                             grad_every=20,
                                             grad_count=40)
            if score < best_score - 1e-10:
                best_score = score
                best_seq = seq.copy()
                print(f"  Refine pass {pass_i+1}: {best_score:.8f}")
            else:
                # Try gradient just in case
                if time.monotonic() < deadline - 5.0:
                    seq2, score2 = _optimize_gradient(best_seq, deadline - 5.0, max_iters=10000)
                    if score2 < best_score - 1e-10:
                        best_score = score2
                        best_seq = seq2.copy()
                        print(f"  Refine grad pass {pass_i+1}: {best_score:.8f}")
                        continue
                break

    # --- Return best ---
    if best_seq is not None:
        # Ensure all values are valid floats
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        result = [float(x) for x in best_seq]
        final_score = _compute_obj(np.array(result))
        print(f"\nFinal result: n={len(result)}, score={final_score:.8f}")
        try:
            eval_score = float(evaluate_sequence(result))
            print(f"evaluate_sequence confirms: {eval_score:.8f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return result

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
