"""

Uses multi-phase approach: (1) gradient descent on flattened objective, (2) gradient 
descent on soft-max of peak, (3) targeted local search with peak shaving.

import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the evaluation score for a sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def grad_softmax_conv(a, tau):
    """
    Compute soft-max of conv(a,a) and its gradient w.r.t. a.
    Returns (value, gradient, conv).
    """
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    value = c_max + tau * np.log(np.sum(exp_shifted))
    
    n = len(a)
    # Gradient: d/d a[m] of softmax(c, tau) = sum_k w[k] * (a[k-m] + a[m-k])
    # = sum_k w[k] * a[|k-m|] (with appropriate bounds)
    # More precisely: d c[k]/d a[m] = a[k-m] if 0<=k-m<n, plus a[m-k] if 0<=m-k<n
    # = a[k-m] (if valid) + a[m-k] (if valid and k≠m)
    # Actually: c[k] = sum_i a[i]*a[k-i], so d c[k]/d a[m] = a[k-m] (if 0<=k-m<n) + a[m-k] (if 0<=m-k<n and m≠k-m, i.e., 2m≠k)
    # Wait: d/d a[m] of sum_i a[i]*a[k-i] = a[k-m] (from i=m term) + a[m] * d/d a[m] of a[k-m]... 
    # No. c[k] = sum_i a[i] * a[k-i]. 
    # d c[k] / d a[m] = a[k-m] (when i=m) + sum_{i: k-i=m} a[i] * 1 = a[k-m] + a[k-m] (when i=k-m, if k-m ≠ m)
    # Wait, that's only if k-m ≠ m. If k-m = m (i.e., k=2m), then we get a[m] + a[m] = 2*a[m]... no.
    # Let me be more careful. c[k] = sum_i a[i] * a[k-i].
    # d c[k] / d a[m] = sum_i (delta_{i,m} * a[k-i] + a[i] * delta_{k-i,m})
    # = a[k-m] (if 0 <= k-m < n) + a[k-m] (if 0 <= k-m < n and k-m != m)
    # Wait no: the second term is sum_i a[i] * delta_{k-i,m} = a[k-m] (if 0 <= k-m < n)
    # So d c[k] / d a[m] = a[k-m] (if valid) + a[k-m] (if valid) = 2*a[k-m] (if valid)
    # Unless k-m = m, in which case it's just a[m] (one term, not two).
    # Actually: if k-m = m (k = 2m), then both terms give a[m], and since i=m and k-i=m are the same term,
    # we get a[m] (not 2*a[m]).
    # Hmm, let me think again. c[k] = sum_i a[i] * a[k-i].
    # This is a product of two sequences. The derivative w.r.t. a[m]:
    # - From the first factor (a[i]): contributes a[k-m] when i=m
    # - From the second factor (a[k-i]): contributes a[m] when k-i=m, i.e., i=k-m
    # So total: a[k-m] + a[k-m] = 2*a[k-m], unless m = k-m (i.e., k=2m), in which case it's a[m] + a[m] but these are the SAME term, so it's just 2*a[m]... 
    # Wait no. If k = 2m, then i=m gives a[m]*a[m] = a[m]^2. The derivative of a[m]^2 w.r.t. a[m] is 2*a[m]. So it IS 2*a[m].
    # So in all cases: d c[k]/d a[m] = 2*a[k-m] (if 0 <= k-m < n), else 0.
    # Wait, that's not right either. Let me use a concrete example.
    # a = (a0, a1, a2), n=3. c[2] = a0*a2 + a1*a1 + a2*a0 = a0*a2 + a1^2 + a2*a0 = 2*a0*a2 + a1^2.
    # d c[2]/d a[1] = 2*a1. And k-m = 2-1 = 1, a[1] = a1, 2*a[1] = 2*a1. ✓
    # d c[2]/d a[0] = 2*a2. And k-m = 2-0 = 2, a[2] = a2, 2*a[2] = 2*a2. ✓

    # d c[1]/d a[0] = 2*a1. k-m = 1-0 = 1, a[1] = a1, 2*a[1]. ✓

    # We can compute this via: flip a, convolve with w
    # cross_corr[m] = sum_j w[j+m] * a[j] for j = 0 to n-1, j+m < len(w)
    # = sum_j a[j] * w[j+m]
    # This is the cross-correlation of a and w, which can be computed as:
    # conv(a_flipped, w)[m + n - 1]
    
    a_flipped = a[::-1]
    corr = fft_convolve(a_flipped, w)
    gradient = 2.0 * corr[n-1:2*n-1]
    
    return value, gradient, c

def grad_sum_sq_conv(a):
    """
    Compute sum(c^2) and its gradient w.r.t. a.
    sum(c^2) = sum_k c[k]^2
    d/d a[m] sum(c^2) = 2 * sum_k c[k] * d c[k]/d a[m] = 2 * sum_k c[k] * 2*a[k-m] (if valid)
    = 4 * sum_k c[k] * a[k-m] = 4 * cross_corr(c, a)[m]
    """
    c = fft_convolve(a, a)
    value = np.sum(c ** 2)
    n = len(a)
    a_flipped = a[::-1]
    corr = fft_convolve(a_flipped, c)
    gradient = 4.0 * corr[n-1:2*n-1]
    return value, gradient, c

def optimize_phase_a(a_init, time_limit, rng):
    """
    Phase A: Minimize sum(c^2) to flatten the convolution.
    Uses Adam optimizer with log-parameterization.
    Returns best a and score.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    # Normalize
    a = a / np.sum(a)
    
    theta = np.log(a + 1e-10)  # log parameterization
    S = 1.0  # fixed sum
    
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    lr = 0.5
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        a = softmax_normalize(theta) * S
        value, grad_a, c = grad_sum_sq_conv(a)
        
        # Project gradient to handle the sum constraint
        mean_grad = np.mean(grad_a)
        grad_a_centered = grad_a - mean_grad
        
        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * grad_a_centered
        momentum2 = beta2 * momentum2 + (1 - beta2) * (grad_a_centered ** 2)
        m1_hat = momentum1 / (1 - beta1 ** step)
        m2_hat = momentum2 / (1 - beta2 ** step)
        
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        if step % 100 == 0:
            a_check = softmax_normalize(theta) * S
            score = compute_score(a_check)
            if score < best_score:
                best_score = score
                best_a = a_check.copy()
        
        # Adapt learning rate
        if step % 500 == 0:
            lr *= 0.8
    
    return best_a, best_score

def optimize_phase_b(a_init, time_limit, rng):
    """
    Phase B: Minimize soft-max of conv with annealing.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init.astype(np.float64).copy()
    a = a / np.sum(a) * S
    
    theta = np.log(a / S + 1e-12)  # parameterize a = softmax(theta) * S
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0
    lr = 1.5
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    tau_start = 50.0 / n
    tau_end = 0.1 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** min(1.0, elapsed * 1.2)
        
        a = softmax_normalize(theta) * S
        value, grad_a, c = grad_softmax_conv(a, tau)
        
        # Center the gradient (handle sum constraint)
        mean_grad = np.mean(grad_a)
        grad_a_centered = grad_a - mean_grad
        
        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * grad_a_centered
        momentum2 = beta2 * momentum2 + (1 - beta2) * (grad_a_centered ** 2)
        m1_hat = momentum1 / (1 - beta1 ** step)
        m2_hat = momentum2 / (1 - beta2 ** step)
        
        lr_eff = lr / (1.0 + step * 0.001)
        theta = theta - lr_eff * m1_hat / (np.sqrt(m2_hat) + eps)
        
        if step % 50 == 0:
            a_check = softmax_normalize(theta) * S
            score = compute_score(a_check)
            if score < best_score:
                best_score = score
                best_a = a_check.copy()
    
    return best_a, best_score

def local_search_peak_shave(a_init, time_limit, rng):
    """
    Targeted local search: identify peak contributors and adjust them.
    """
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    temp = 0.002
    cooling = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        peak_lag = int(np.argmax(c))
        peak_val = c[peak_lag]
        
        r = rng.random()
        candidate = current.copy()
        
        if r < 0.4:
            # Peak shaving: reduce contributors to peak
            # c[peak] = sum_i a[i]*a[peak-i]
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_lag - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            
            # Find top contributors
            n_top = max(3, n // 8)
            top_idx = np.argsort(contribs)[-n_top:]
            
            # Reduce them
            reductions = np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            candidate[top_idx] *= reductions
            
            # Compensate by increasing random other elements
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                total_reduction = 1.0 - np.prod(reductions) / len(reductions)  # approximate
                candidate[inc_idx] *= np.exp(0.03 * rng.random(n_inc))
                
        elif r < 0.6:
            # Targeted pair modification
            # Find the two largest contributors to the peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_lag - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            
            top_2 = np.argsort(contribs)[-2:]
            i1, i2 = top_2[0], top_2[1]
            
            # Reduce one, increase the other (or both reduce and increase elsewhere)
            delta = 0.05 * current[i1]
            candidate[i1] = max(0, current[i1] - delta)
            candidate[i2] = current[i2] + delta * 0.5
            
        elif r < 0.8:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 15))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.03, size=width))
            
        else:
            # Small global perturbation
            candidate *= np.exp(rng.normal(0, 0.008, size=n))
        
        candidate = np.clip(candidate, 1e-6, 1000.0)
        new_score = compute_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                temp = max(temp, 0.002)  # reheat slightly
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-10)):
            current = candidate
            current_score = new_score
        
        temp *= cooling
    
    return best, best_score

def make_chirp_init(n, rng):
    """Create a chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    alpha = 0.3 + 0.4 * rng.random()
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_random_init(n, rng):
    """Random initialization."""
    a = 0.5 + rng.random(n)
    return a

def make_triangular_init(n, rng):
    """Triangular wave initialization."""
    t = np.arange(n, dtype=np.float64)
    period = n // 2 + rng.integers(1, 5)
    a = np.abs(np.sin(np.pi * t / period))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_geometric_init(n, rng):
    """Geometric/exponential initialization."""
    ratio = 0.95 + 0.05 * rng.random()
    a = ratio ** np.arange(n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

"""
Multi-phase optimization for minimizing peak linear autocorrelation of non-negative sequences.
Combines gradient-based optimization (Adam with soft-max) and targeted local search.
"""

import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def soft_max_conv(a, tau):
    """Compute soft-max of conv(a,a) and gradient."""
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    w = e / np.sum(e)
    value = c_max + tau * np.log(np.sum(e))
    n = len(a)
    # gradient[m] = 2 * sum_k w[k] * a[k-m] (for valid k-m)
    # = 2 * cross_corr(a, w)[m]
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    grad = 2.0 * corr[n-1:2*n-1]
    return value, grad, c, w

def adam_step(theta, grad, m1, m2, step, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m1 = beta1 * m1 + (1 - beta1) * grad
    m2 = beta2 * m2 + (1 - beta2) * (grad * grad)
    m1_hat = m1 / (1 - beta1 ** step)
    m2_hat = m2 / (1 - beta2 ** step)
    theta_new = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
    return theta_new, m1, m2
