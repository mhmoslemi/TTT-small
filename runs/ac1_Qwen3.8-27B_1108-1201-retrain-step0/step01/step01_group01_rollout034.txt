def _compute_gradient(a, w, beta):
    """Compute gradient of log-sum-exp of convolution."""
    n = len(a)
    grad = np.zeros(n)
    # For each lag k in the convolution, add w[k] * (contribution to each a[j])
    # c_k = sum_j a[j] * a[k-j] (valid)
    # d c_k / d a[j] = a[k-j] + a[j-k] (valid)

        if wk < 1e-15:
            continue
        # a[k-j] is valid when 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(k, n-1)

                grad[idx2] += wk * a[idx2 - k]

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]

LSE-based projected gradient descent for minimizing the normalized autocorrelation

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute linear self-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    if fft_size is None:
        fft_size = 1
        while fft_size < size:
            fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)
    return c[:size]


def _lse_gradient(a, c, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*c_k)) w.r.t. a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    L = len(c)  # 2n-1
    fft_size = 1
    while fft_size < n + L - 1:
        fft_size *= 2

    # Stable weights
    c_max = np.max(c)
    exp_c = np.exp(beta * (c - c_max))
    Z = np.sum(exp_c)
    w = exp_c / Z  # normalized weights, sum to 1

    # FFT-based computation
    # grad[j] = 2 * sum_k w[k] * a[k-j] (valid k-j)
    # = 2 * P[j] where P[j] = sum_m a[m] * w[j+m]
    # P = convolution of a_reversed with w, shifted:
    # P[j] = conv[n-1+j] where conv = ifft(fft(a_rev) * fft(w))
    
    a_rev_p = np.zeros(fft_size)
    a_rev_p[:n] = a[::-1]
    w_p = np.zeros(fft_size)
    w_p[:L] = w

    fa_rev = np.fft.rfft(a_rev_p)
    fw = np.fft.rfft(w_p)
    conv = np.fft.irfft(fa_rev * fw, fft_size)

    return 2.0 * conv[n - 1:2 * n - 1]
