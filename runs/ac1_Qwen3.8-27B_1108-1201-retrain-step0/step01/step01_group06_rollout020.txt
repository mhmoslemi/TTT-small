import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def compute_score(a):
    """Compute the full score 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * fast_convolve(a).max() / (s * s)

def solve_lp_direction(f, M):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (unnormalized).
    """
    n = len(f)
    # Build constraint matrix: A_ub @ g <= b_ub
    # (f*g)[k] = sum_j f[k-j] * g[j] for valid indices
    # So A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            # A_ub[k, j_min:j_max+1] = f[k-j_min : k-j_max+1][::-1]
            A_ub[k, j_min:j_max+1] = f[k-j_min:k-j_max+1][::-1]
    
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

def mirror_descent_optimize(f0, time_limit, t_param=40.0, eta0=0.5):
    """
    Multiplicative mirror descent with soft-max smoothing.
    Returns (best_a, best_obj_normalized).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    eta = eta0
    patience = 0
    max_patience = 80
    current_t = t_param
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Soft-max weights
        scaled = current_t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient
        sg = np.zeros(n)
        thresh = w.max() * 0.003
        active_idx = np.where(w > thresh)[0]
        
        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end+1] += w[k] * 2.0 * f[k-j_end:k-j_start+1][::-1]
        
        # Multiplicative update
        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        f = f * np.exp(log_factor)
        f = np.maximum(f, 0.0)
        f_sum = f.sum()
        if f_sum < 1e-12:
            f = np.ones(n) / n
        else:
            f /= f_sum
        
        conv = fast_convolve(f)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-9:
            best_obj = current_obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            patience = 0
            eta *= 0.6
            if eta < 0.005:
                current_t *= 2.0
                eta = 0.3
                max_patience = 40
            else:
                current_t = np.clip(current_t * 1.2, 5.0, 200.0)
    
    return best_f, best_obj

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Gaussians
    for sf in [0.1, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - n*0.5) / (sf * n))**2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g
    
    # Ramp
    ramp = np.arange(1, n+1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Bimodal (edges)
    bi = np.zeros(n)
    bi[:max(1, n//4)] = 1.0
    bi[-max(1, n//4):] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Cosine
    cos_bump = 0.5 * (1 + np.cos(2*np.pi*x/n))
    cos_bump /= cos_bump.sum()
    inits['cosine'] = cos_bump
    
    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits['random1'] = r
    r2 = rng.dirichlet(np.ones(n) * 0.1)
    inits['random2'] = r2
    
    # Inverse ramp (concentrated at start)
    inv_ramp = np.arange(n, 0, -1, dtype=np.float64)
    inv_ramp /= inv_ramp.sum()
    inits['inv_ramp'] = inv_ramp
    
    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of LP-based
direction finding and multiplicative mirror descent. The LP provides
strong descent directions by maximizing volume under the linearized
convolution constraint, while mirror descent with soft-max peak
weighting provides smooth refinement. Multiple sequence lengths and
diverse initializations are explored.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_score(a):
    """Compute the full score."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * fast_convolve(a).max() / (s * s)


def build_constraint_matrix(f):
    """Build the (2n-1) x n constraint matrix for the LP."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max+1] = f[k - j_min : k - j_max + 1][::-1]
    return A


def solve_lp_direction(f, M):
    """
    Solve: max 1^T g  s.t.  A@f @ g <= M*1,  g >= 0.
    Returns (g_optimal, success_flag).
    """
    n = len(f)
    A_ub = build_constraint_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x, True
    return None, False
