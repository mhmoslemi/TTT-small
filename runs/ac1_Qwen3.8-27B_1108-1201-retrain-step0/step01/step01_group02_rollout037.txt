```python
    """One gradient descent step."""
    n = len(a)
    s = a.sum()
    # Compute conv at lag k_star
    # c[k_star] = sum_i a[i]*a[k_star-i] for valid i
    # Gradient: 4*n/(s^2) * (a[k_star-j] - c[k_star])
    
    # Compute c[k_star]
    # a[k_star-j] is valid when 0 <= k_star-j < n, i.e., k_star-n < j < k_star
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    
    # c[k_star] = sum_{j=lo}^{hi-1} a[j] * a[k_star-j]
    idx = np.arange(lo, hi)
    c_k = np.dot(a[idx], a[k_star - idx])
    
    # gradient[j] = 4*n/s^2 * (a[k_star-j] - c_k) for j in [lo, hi)
    # gradient[j] = -4*n*c_k/s^2 for j outside [lo, hi)
    
    grad = -4.0 * n * c_k / (s * s) * np.ones(n)
    grad[idx] += 4.0 * n * a[k_star - idx] / (s * s)
    
    return grad
```

```python
def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def fft_convolution(a):
    """Compute auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next fast Fourier transform size
    fsize = 1
    while fsize < size:
        fsize *= 2
    fa = np.fft.rfft(a, n=fsize)
    conv = np.fft.irfft(fa * fa, n=fsize)[:size]
    return conv
```

```python
import numpy as np
import time

def fft_conv(a):
    """Fast auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fft_conv(a)
    return 2.0 * n * c.max() / (s * s)

    Optimize a non-negative sequence to minimize score.
    Uses Adam optimizer with FFT-based gradient.
    """

    a = np.maximum(0, a0.astype(np.float64))
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s  # Normalize to sum=1
    
    best_a = a.copy()
    best_score = 2.0 * n  # Upper bound
    
    start = time.monotonic()
    lr = 0.1
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    stuck = 0
    
    for it in range(200000):
        if time.monotonic() - start > time_limit:
            break
        t += 1

        conv = fft_conv(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b  # sum=1
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stuck = 0
            if verbose and (it % 5000 == 0):
                print(f"    iter {it}: score={cur_score:.6f}")
        else:
            stuck += 1
        
        if stuck > 20000:
            break
        
        # Find top K lags
        K = min(k_top, 2 * n - 1)
        top_k = np.argpartition(conv, -K)[-K:]

        grad = np.zeros(n)
        for k in top_k:
            k = int(k)
            if k < n:
                grad[:k+1] += a[:k+1][::-1]
            else:
                si = k - n + 1
                L = n - si
                grad[si:] += a[:L][::-1]
        grad /= K
        # Multiply by 2 (derivative of a[j]*a[k-j] w.r.t. a[j] gives 2*a[k-j])
        grad *= 2.0
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        s_new = a_new.sum()
        if s_new < 1e-15:

        a_new /= s_new
        
        # Only accept if not much worse (simulated annealing style)
        # For simplicity, always accept (Adam should be stable)
        a = a_new
        
        # Decay learning rate
        lr = 0.1 * (0.999 ** (t // 1000))
    
    return best_a, best_score
```

```python
"""
Gradient-based optimization with Adam and FFT for minimizing peak auto-convolution
of non-negative sequences. Uses multiple lengths, initializations, and a two-phase
search (broad exploration + focused refinement).
"""

import numpy as np
import time


def fft_conv(a):
    """Fast auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def compute_score(a):
    """Compute the objective score."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fft_conv(a)
    return 2.0 * n * float(c.max()) / (s * s)


def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def init_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def init_smooth_random(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(2, n // 10), max(3, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return init_raised_cosine(n)
    return a / s


def init_ramp(n, alpha=0.5):
    """Asymmetric ramp: high at one end, low at the other."""
    t = np.arange(n, dtype=np.float64)
    return (1.0 - alpha) + alpha * (1.0 - t / (n - 1))


def optimize_adam(a0, time_limit=60.0, k_top=10, lr_init=0.05, seed=0, verbose=False):
    """
    Adam optimizer for minimizing peak auto-convolution.
    Works with sequences normalized to sum=1.
    """
    n = len(a0)
    a = np.maximum(0, a0.astype(np.float64))
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n

    start = time.monotonic()
    lr = lr_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stuck = 0
    best_stuck = best_score

    for it in range(500000):
        elapsed = time.monotonic() - start
        if elapsed > time_limit:
            break
        t += 1

        conv = fft_conv(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best_a = a.copy()
            stuck = 0
            if verbose and it % 10000 == 0:
                print(f"      iter {it}: score={cur_score:.6f}", flush=True)
        else:
            stuck += 1

        if stuck > 30000:
            break

        # Gradient from top K lags
        K = min(k_top, 2 * n - 1)
        if K < len(conv):
            top_k = np.argpartition(conv, -K)[-K:]
        else:
            top_k = np.arange(len(conv))

        grad = np.zeros(n)
        for k in top_k:
            k = int(k)
            if k < n:
                grad[:k + 1] += a[:k + 1][::-1]
            else:
                si = k - n + 1
                L = n - si
                grad[si:] += a[:L][::-1]
        grad *= 2.0 / K

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(0.0, a)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            continue
        a /= s

        # Learning rate schedule
        lr = lr_init * (0.9995 ** (t // 100))

    final_score = compute_score(best_a)
    if final_score < best_score:
        best_score = final_score
    return best_a, best_score


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return np.interp(new_x, old_x, a)
```

This just scales a! So for symmetric sequences, gradient descent doesn't change the shape, only the scale. This means we need to break symmetry.

For non-symmetric sequences, the step a[j] - lr*2*a[k*-j] does change the shape. The key is to find the right k* and lr.

- Where flip(a, k)[j] = a[k-j] if valid, else 0

```python
def optimize_mwu(a0, time_limit=60.0, lr_init=1.0, verbose=False):
    """Multiplicative Weights Update (exponential gradient)."""
    n = len(a0)
    a = np.maximum(1e-10, a0.astype(np.float64))  # Avoid log(0)
    a = a / a.sum()  # Normalize to sum=1

    best_a = a.copy()
    best_score = 2.0 * n
    start = time.monotonic()
    lr = lr_init
    stuck = 0

    for it in range(500000):
        if time.monotonic() - start > time_limit:
            break

        conv = fft_conv(a)
        k_star = int(conv.argmax())
        max_b = conv[k_star]
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 10000:
            break

        # Compute gradient: 2*a[k*-j] for valid j
        flip_a = np.zeros(n)
        if k_star < n:
            flip_a[:k_star + 1] = a[:k_star + 1][::-1]
        else:
            si = k_star - n + 1
            L = n - si
            flip_a[si:] = a[:L][::-1]

        # Exponential update
        a = a * np.exp(-lr * 2.0 * flip_a)
        a = np.maximum(1e-15, a)
        a = a / a.sum()

        # Decay lr
        lr = lr_init * (0.999 ** (it // 500))

    # Restore zeros (clip very small values)
    best_a[best_a < 1e-10] = 0.0
    best_a = best_a / best_a.sum()
    return best_a, best_score
```

"""
FFT-based optimization for minimizing peak auto-convolution of non-negative sequences.
Uses Multiplicative Weights Update (MWU) and projected gradient descent with
coarse-to-fine multi-scale search.
"""

import numpy as np
import time


def fft_conv(a):
    """Fast auto-convolution using FFT. Returns full convolution (length 2n-1)."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fft_conv(a)
    return 2.0 * n * float(c.max()) / (s * s)


def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def init_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0])
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return np.interp(new_x, old_x, a)


def optimize_mwu(a0, time_limit=60.0, lr_init=2.0, verbose=False):
    """
    Multiplicative Weights Update (exponential gradient) optimization.
    Minimizes peak auto-convolution for non-negative sequences.
    Works with sum(a)=1 normalization.
    """
    n = len(a0)
    a = np.maximum(1e-12, a0.astype(np.float64))
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n
    start = time.monotonic()
    lr = lr_init
    stuck = 0

    for it in range(1000000):
        if time.monotonic() - start > time_limit:
            break

        conv = fft_conv(a)
        k_star = int(conv.argmax())
        max_b = conv[k_star]
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stuck = 0
            if verbose and it % 20000 == 0:
                print(f"        MWU iter {it}: score={cur_score:.6f}", flush=True)
        else:
            stuck += 1

        if stuck > 50000:
            break

        # Build the flipped vector: flip_a[j] = a[k_star - j] if valid
        flip_a = np.zeros(n)
        if k_star < n:
            flip_a[:k_star + 1] = a[:k_star + 1][::-1]
        else:
            si = k_star - n + 1
            L = n - si
            flip_a[si:] = a[:L][::-1]

        # Exponential update: a[j] *= exp(-lr * flip_a[j])
        # Use log-domain for numerical stability
        # a_new = a * exp(-lr * flip_a) / sum(a * exp(-lr * flip_a))
        log_a = np.log(a)
        log_a_new = log_a - lr * flip_a
        # Subtract max for numerical stability
        log_a_new -= log_a_new.max()
        a = np.exp(log_a_new)
        a = a / a.sum()

        # Decay learning rate
        lr = lr_init * (0.9995 ** (it // 100))

    # Clean up: set very small values to 0
    best_a[best_a < 1e-12] = 0.0
    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s
    return best_a, best_score
