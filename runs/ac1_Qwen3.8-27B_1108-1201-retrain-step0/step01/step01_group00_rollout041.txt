    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:

    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

import numpy as np
import time
from scipy.special import logsumexp


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax_vals(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def soft_max_gradient_step(a, tau):
    """
    Compute score (soft-max approx) and gradient w.r.t. theta 
    (where a = softmax(theta) * S, S = sqrt(2n)).
    Returns (score, grad_theta).
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    sum_a = np.sum(a)  # Should be S
    c = fft_convolve(a, a)
    
    # Soft-max of c
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    log_sum = np.log(np.sum(np.exp(shifted)))
    val = c_max + tau * log_sum
    w = np.exp(shifted - log_sum)
    
    score = 2.0 * n * val / (sum_a * sum_a)
    
    # Gradient of val w.r.t. a:
    # d(val)/da[j] = 2 * sum_m a[m] * w[m+j] (where indices valid)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g_a = 2.0 * corr[n-1:2*n-1]
    
    # Gradient of score w.r.t. a:
    # score = 2n*val/sum_a^2
    # d(score)/da[j] = 2n*g_a[j]/sum_a^2 - 4n*val/(sum_a^3)
    grad_a = (2.0 * n / (sum_a * sum_a)) * g_a - (4.0 * n * val) / (sum_a ** 3)
    
    # Transform to theta gradient:

    # d(score)/dtheta[j] = sum_k grad_a[k] * a[j] * (delta[kj] - a[k]/S)
    # = a[j] * (grad_a[j] - sum_k grad_a[k]*a[k]/S)
    inner = np.dot(grad_a, a) / S
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta

"""
Hybrid optimization for minimizing the peak autocorrelation of non-negative sequences.
Uses gradient descent with annealed soft-max, scipy L-BFGS refinement,
targeted local search with peak-specific moves, and multi-length exploration.
Key insight: exploit non-symmetry to reduce the anti-diagonal autocorrelation peak.
"""

import numpy as np
import time
from scipy.special import logsumexp
from scipy.optimize import minimize as scipy_minimize


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax_vals(theta):
    """Stable softmax returning values summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s == 0:
        return np.ones(len(theta)) / len(theta)
    return e / s


def local_score(a):
    """Compute the actual evaluation score."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def soft_max_value_and_weights(x, tau):
    """Compute soft-max value and weight vector."""
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    log_sum = logsumexp(shifted)
    value = x_max + tau * log_sum
    weights = np.exp(shifted - log_sum)
    return value, weights


def compute_score_grad_theta(theta, tau):
    """
    Compute soft-max score and gradient w.r.t. theta.
    a = softmax(theta) * sqrt(2*len(theta))
    Returns (score, grad_theta).
    """
    n = len(theta)
    S = np.sqrt(2.0 * n)
    p = softmax_vals(theta)
    a = p * S
    sum_a = np.sum(a)

    c = fft_convolve(a, a)
    val, w = soft_max_value_and_weights(c, tau)
    score = 2.0 * n * val / (sum_a * sum_a)

    # Gradient of val w.r.t. a: g[j] = 2 * sum_m a[m]*w[m+j]
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g_a = 2.0 * corr[n - 1:2 * n - 1]

    # Gradient of score w.r.t. a
    grad_a = (2.0 * n / (sum_a * sum_a)) * g_a - (4.0 * n * val) / (sum_a ** 3)

    # Transform to theta gradient
    inner = np.dot(grad_a, a) / S
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def compute_score_grad_theta_topk(theta, tau, k):
    """
    Score based on sum of top-k autocorrelation values (smoothed).
    """
    n = len(theta)
    S = np.sqrt(2.0 * n)
    p = softmax_vals(theta)
    a = p * S
    sum_a = np.sum(a)

    c = fft_convolve(a, a)
    # Use soft-max on the top-k fraction
    val, w = soft_max_value_and_weights(c, tau)
    score = 2.0 * n * val / (sum_a * sum_a)

    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g_a = 2.0 * corr[n - 1:2 * n - 1]

    grad_a = (2.0 * n / (sum_a * sum_a)) * g_a - (4.0 * n * val) / (sum_a ** 3)
    inner = np.dot(grad_a, a) / S
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def gradient_optimize(n, a_init, time_limit, rng, tau_start=None, tau_end=None):
    """
    Adam gradient descent on theta with annealed temperature.
    """
    if tau_start is None:
        tau_start = 100.0 / n
    if tau_end is None:
        tau_end = 0.5 / n

    S = np.sqrt(2.0 * n)
    theta = np.log(a_init / np.sum(a_init) + 1e-15).astype(np.float64)

    m1 = np.zeros(n)
    m2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    lr = 1.5

    best_theta = theta.copy()
    best_score = local_score(softmax_vals(theta) * S)

    t0 = time.monotonic()
    no_improve = 0
    best_no_improve_score = best_score

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        # Use a slower annealing curve to stay in smooth regime longer
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)

        score, g_theta = compute_score_grad_theta(theta, tau)

        if not np.isfinite(score):
            theta += rng.normal(0, 0.05, size=n)
            continue

        step += 1
        m1 = beta1 * m1 + (1 - beta1) * g_theta
        m2 = beta2 * m2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = m1 / (1 - beta1 ** step)
        m2_hat = m2 / (1 - beta2 ** step)

        lr_t = lr / (1.0 + 0.5 * np.sqrt(step))
        theta = theta - lr_t * m1_hat / (np.sqrt(m2_hat) + eps)

        a_check = softmax_vals(theta) * S
        actual = local_score(a_check)

        if np.isfinite(actual) and actual < best_score:
            best_score = actual
            best_theta = theta.copy()
            no_improve = 0
            best_no_improve_score = actual
        else:
            no_improve += 1

        # Restart if stuck for too long
        if no_improve > 5000:
            # Try different perturbation strategies
            r = rng.random()
            if r < 0.5:
                theta += rng.normal(0, 0.05, size=n)
            elif r < 0.8:
                # Targeted: increase theta at positions that don't contribute to peak
                a_curr = softmax_vals(theta) * S
                conv = fft_convolve(a_curr, a_curr)
                peak_idx = int(np.argmax(conv))
                contribs = np.zeros(n)
                for i in range(n):
                    j = peak_idx - i
                    if 0 <= j < n:
                        contribs[i] = a_curr[i] * a_curr[j]
                low_contrib = np.argsort(contribs)[:n // 4]
                theta[low_contrib] += 0.1
                theta = theta - np.mean(theta)
            else:
                # Partial symmetrization disruption
                theta = 0.5 * theta + 0.5 * rng.normal(0, 0.1, size=n)

            no_improve = 0

        if step % 10000 == 0:
            print(f"        step={step}, score={actual:.6f}, best={best_score:.6f}, tau={tau:.4f}, t={elapsed:.0f}s")

    a_best = softmax_vals(best_theta) * S
    final_score = local_score(a_best)
    return a_best, final_score


def lbfgs_optimize(n, a_init, time_limit, tau):
    """
    Use scipy L-BFGS-B to optimize the soft-max objective.
    """
    S = np.sqrt(2.0 * n)
    theta0 = np.log(a_init / np.sum(a_init) + 1e-15).astype(np.float64)

    t0 = time.monotonic()
    best_theta = theta0.copy()
    best_score = local_score(softmax_vals(theta0) * S)

    def obj_func(theta_vec):
        if time.monotonic() - t0 > time_limit:
            return 1e10, np.zeros_like(theta_vec)
        a = softmax_vals(theta_vec) * S
        score, g = compute_score_grad_theta(theta_vec, tau)
        if not np.isfinite(score):
            return 1e10, np.zeros_like(theta_vec)
        return score, g

    # L-BFGS-B with custom objective
    result = scipy_minimize(
        obj_func, theta0, method='L-BFGS-B',
        options={'maxiter': 10000, 'maxfun': 5000,
                 'iprint': -1, 'gtol': 1e-6}
    )

    if result.success or np.isfinite(result.fun):
        a_result = softmax_vals(result.x) * S
        score_result = local_score(a_result)
        if score_result < best_score:
            best_score = score_result
            best_theta = result.x.copy()

    a_best = softmax_vals(best_theta) * S
    return a_best, local_score(a_best)


def targeted_local_search(a_init, time_limit, rng):
    """
    Local search with moves specifically designed to reduce the peak autocorrelation.
    """
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score

    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.99997
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fft_convolve(current, current)
        peak_val = np.max(conv)
        peak_idx = int(np.argmax(conv))

        # Get top 3 peak positions
        top_peaks = np.argsort(conv)[-3:]

        r = rng.random()
        candidate = current.copy()

        if r < 0.35:
            # Targeted reduction of peak contributors
            contribs = np.zeros(n)
            for pi in top_peaks:
                for i in range(n):
                    j = pi - i
                    if 0 <= j < n:
                        contribs[i] += current[i] * current[j]

            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            low_idx = np.argsort(contribs)[:n // 5]

            # Reduce high contributors
            red_factors = np.exp(rng.normal(-0.06, 0.02, size=len(top_idx)))
            candidate[top_idx] *= red_factors

            # Boost low contributors to maintain sum
            sum_change = np.sum(current[top_idx] * (1 - red_factors))
            if sum_change > 0 and len(low_idx) > 0:
                boost_total = sum_change * 0.9
                boost_per = boost_total / len(low_idx)
                candidate[low_idx] += boost_per * np.exp(rng.normal(0, 0.1, size=len(low_idx)))

        elif r < 0.50:
            # Anti-symmetric perturbation: increase a[i], decrease a[n-1-i]
            for _ in range(rng.integers(2, 6)):
                i = int(rng.integers(0, n))
                j = n - 1 - i
                if i != j:
                    amt = 0.03 * current[i] * rng.random()
                    candidate[i] += amt
                    candidate[j] = max(0, candidate[j] - amt)

        elif r < 0.60:
            # Block modification
            width = max(1, int(rng.integers(1, max(2, n // 15))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))

        elif r < 0.70:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 25))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, size=width))

        elif r < 0.80:
            # Pair swap from peak-contributing to non-contributing
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            hi = int(np.argmax(contribs))
            lo = int(np.argmin(contribs))
            amt = current[hi] * rng.uniform(0.02, 0.08)
            candidate[hi] -= amt
            candidate[lo] += amt

        elif r < 0.90:
            # Smooth + perturb
            if n > 4:
                kernel = np.ones(3) / 3.0
                smoothed = np.convolve(current, kernel, mode='same')
                candidate = 0.7 * smoothed + 0.3 * current
                candidate *= np.exp(rng.normal(0, 0.015, size=n))
            else:
                candidate = current * np.exp(rng.normal(0, 0.02, size=n))
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0, 0.012, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)

        if np.sum(candidate) < 0.01:
            continue

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif np.isfinite(new_score) and rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current = candidate
            current_score = new_score

        temp *= cooling

        if iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        local: iter={iter_count}, current={current_score:.6f}, best={best_score:.6f}, t={elapsed:.0f}s")

    return best, best_score


def make_init(n, rng, method='chirp'):
    """Create initial sequence with various structures."""
    if method == 'uniform':
        return np.ones(n)
    elif method == 'random':
        return np.abs(rng.standard_normal(n)) + 0.1
    elif method == 'chirp':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t * t / (2.0 * n))
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.05, None)
    elif method == 'multi_tone':
        t = np.arange(n, dtype=np.float64)
        a = np.ones(n)
        for k in [2, 3, 5, 7, 11]:
            a += (0.12 / k) * np.cos(2 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
        return np.clip(a, 0.05, None)
    elif method == 'ramp':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.3 * t / n
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.05, None)
    elif method == 'asymmetric_sine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.2 * np.sin(2 * np.pi * t / n) + 0.1 * t / n
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.05, None)
    elif method == 'bump':
        t = np.arange(n, dtype=np.float64)
        center = n * (0.3 + 0.4 * rng.random())
        width = n * (0.1 + 0.15 * rng.random())
        a = 1.0 + 0.3 * np.exp(-0.5 * ((t - center) / width) ** 2)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.05, None)
    else:
        return np.ones(n)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: gradient optimization + L-BFGS + targeted local search.
    Explores multiple sequence lengths and initialization strategies.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    print(f"\n=== Starting search, initial best={best_score:.6f} ===")
    print(f"Time budget: {budget_s:.0f}s, deadline in {deadline - t_global:.0f}s\n")

    # Phase 1: Gradient optimization for key lengths
    # Focus on n=256 (provided) and nearby lengths
    configs = [
        # (n, grad_time, lbfgs_time, local_time, inits)
        (256, 350, 120, 200, ['provided', 'chirp', 'multi_tone', 'asymmetric_sine']),
        (192, 200, 80, 120, ['chirp', 'multi_tone']),
        (320, 200, 80, 120, ['chirp', 'multi_tone']),
        (384, 150, 60, 100, ['chirp', 'random']),
        (128, 120, 60, 80, ['chirp', 'multi_tone']),
        (512, 150, 60, 100, ['chirp']),
    ]

    for n, t_grad, t_lbfgs, t_local, init_list in configs:
        if time.monotonic() > deadline - 90:
            break

        remaining = deadline - time.monotonic()
        if remaining < 120:
            break

        t_grad_actual = min(t_grad, remaining * 0.4)
        t_lbfgs_actual = min(t_lbfgs, remaining * 0.15)
        t_local_actual = min(t_local, remaining * 0.25)

        if t_grad_actual < 30:
            continue

        best_for_n = None
        best_score_n = np.inf

        for init_i, init_method in enumerate(init_list):
            if time.monotonic() > deadline - 60:
                break

            # Generate initial sequence
            if init_method == 'provided' and start_seq is not None:
                if len(start_seq) == n:
                    a0 = start_seq.copy()
                else:
                    x_old = np.linspace(0, 1, len(start_seq))
                    x_new = np.linspace(0, 1, n)
                    a0 = np.interp(x_new, x_old, start_seq)
            else:
                a0 = make_init(n, rng, init_method)

            a0 = np.clip(a0, 0.01, None)
            a0 = a0 / np.sum(a0) * np.sqrt(2.0 * n)

            # Gradient optimization
            t_g = min(t_grad_actual / (len(init_list) - init_i + 0.5), deadline - time.monotonic() - 60)
            if t_g < 15:
                continue

            print(f"  n={n}, init='{init_method}', grad_time={t_g:.0f}s")
            a_grad, score_grad = gradient_optimize(n, a0, t_g, rng)
            print(f"    grad score={score_grad:.6f}")

            if score_grad < best_score_n:
                best_score_n = score_grad
                best_for_n = a_grad.copy()

            # L-BFGS refinement
            if time.monotonic() < deadline - 60:
                t_l = min(t_lbfgs_actual, deadline - time.monotonic() - 60)
                if t_l > 10:
                    tau_lbfgs = 2.0 / n
                    a_lbfgs, score_lbfgs = lbfgs_optimize(n, a_grad, t_l, tau_lbfgs)
                    print(f"    lbfgs score={score_lbfgs:.6f}")
                    if score_lbfgs < best_score_n:
                        best_score_n = score_lbfgs
                        best_for_n = a_lbfgs.copy()

        # Local search on best for this n
        if best_for_n is not None and time.monotonic() < deadline - 60:
            t_polish = min(t_local_actual, deadline - time.monotonic() - 40)
            if t_polish > 20:
                print(f"  n={n}, local search for {t_polish:.0f}s")
                a_polished, score_polished = targeted_local_search(best_for_n, t_polish, rng)
                print(f"    polished score={score_polished:.6f}")
                if score_polished < best_score_n:
                    best_score_n = score_polished
                    best_for_n = a_polished.copy()

        if best_score_n < best_score:
            best_score = best_score_n
            best_seq = best_for_n.copy()
            print(f"  *** New global best: {best_score:.6f} (n={n}) ***\n")
        else:
            print(f"  n={n} best: {best_score_n:.6f} (no improvement)\n")

    # Phase 2: Final intensive polish on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_final = min(180.0, deadline - time.monotonic() - 5)
        if t_final > 15:
            print(f"=== Final polish: {t_final:.0f}s on n={len(best_seq)} ===")
            a_final, score_final = targeted_local_search(best_seq, t_final, rng)
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"  After final polish: {best_score:.6f}")

            # One more gradient pass on the polished result
            if time.monotonic() < deadline - 10:
                t_g2 = min(60.0, deadline - time.monotonic() - 10)
                if t_g2 > 10:
                    n_f = len(best_seq)
                    a_g2, score_g2 = gradient_optimize(n_f, best_seq, t_g2, rng,
                                                       tau_start=20.0 / n_f,
                                                       tau_end=0.3 / n_f)
                    if score_g2 < best_score:
                        best_score = score_g2
                        best_seq = a_g2.copy()
                        print(f"  After final gradient: {best_score:.6f}")

                    # Quick local polish
                    if time.monotonic() < deadline - 5:
                        t_lp = deadline - time.monotonic() - 5
                        if t_lp > 10:
                            a_lp, score_lp = targeted_local_search(best_seq, t_lp, rng)
                            if score_lp < best_score:
                                best_score = score_lp
                                best_seq = a_lp.copy()
                                print(f"  After final local: {best_score:.6f}")

    # Verify with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
