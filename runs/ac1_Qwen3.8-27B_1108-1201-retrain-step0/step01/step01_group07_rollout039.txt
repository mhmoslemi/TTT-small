We want to minimize max_k sum_i a[i]*a[i+k] / (sum a)^2.

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses corrected softmax-gradient (fixing spurious term in previous code),
top-k peak focusing, adaptive learning rate, multi-phase optimization,
and perturbation-based escape from local minima.

import numpy as np

def compute_obj_grad(a, tau, top_k=None):
    """Compute gradient of tau*log(sum(exp(c/tau))) w.r.t. a.
    Returns (gradient, convolution, softmax_weights)."""
    n = a.size
    c = np.convolve(a, a)
    if top_k is not None and top_k < c.size:
        c_work = c.copy()
        thresh = np.partition(c, -top_k)[top_k - 1]
        c_work[c_work < thresh] -= 5.0 * tau
    else:
        c_work = c
    c_shift = c_work - c_work.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    # Corrected gradient: 2 * correlate(q, a, 'valid')
    grad = 2.0 * np.correlate(q, a, 'valid')
    return grad, c, q


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    a = a.copy()
    n = a.size
    s = a.sum()
    if s <= 0:
        return np.full(n, S / n)
    if np.abs(s - S) < 1e-12:
        return np.maximum(a, 0.0)
    lo, hi = 0.0, a.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    diff = S - res.sum()
    if abs(diff) > 1e-10:
        idx = res.argmax()
        res[idx] += diff
    return res


def init_cosine(n, seed=0):
    t = np.linspace(0, 2 * np.pi, n, endpoint=False)

    a = project_simplex(a, S)
