def compute_soft_max_grad(a, tau):
    """Compute gradient of tau*log-sum-exp(conv(a,a)/tau) w.r.t. a.
    
    Uses the identity: dL/da[i] = (2/tau) * sum_{j=0}^{n-1} p[i+j] * a[j]
    where p = softmax(conv(a,a)/tau).
    """
    n = len(a)
    c = np.convolve(a, a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    p = exp_c / exp_c.sum()
    # grad[i] = (2/tau) * correlate(p, a, 'valid')[i]
    grad = (2.0 / tau) * np.correlate(p, a, 'valid')
    return grad, c


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    s = a.sum()
    if s <= S * (1 + 1e-12):
        return a + (S - s) / n
    lo, hi = 0.0, a.max()

    res = np.maximum(a - t, 0.0)
    diff = S - res.sum()
    if abs(diff) > 1e-12:
        mask = res > 0
        if mask.any():
            res[mask] += diff / mask.sum()
            res = np.maximum(res, 0.0)
        else:

    """Compute the actual score 2*n*max(conv)/sum^2."""

    if sum_a < 0.01:

    return 2.0 * n * max_b / (sum_a ** 2)

            contributions = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contributions[i] = best[i] * best[j]
            
            if contributions.sum() < 1e-12:
                continue
            
            # Reduce top contributors slightly
            top_idx = np.argsort(contributions)[-3:]
            for i in top_idx:
                red = min(delta / 3.0, best[i] * 0.05)
                new_a[i] -= red
                total_reduction += red
            
            # Add to low contributors
            bottom_idx = np.argsort(contributions)[:3]
            for i in bottom_idx:
                add = min(delta / 3.0, S * 0.01)
                new_a[i] += add
                total_addition += add
        
        new_a = np.maximum(new_a, 0.0)
        new_a = project_simplex(new_a, S)
        
        sc = compute_score(new_a)
        if sc < best_score - 1e-8:
            best_score = sc
            best = new_a.copy()
        else:
            delta *= 0.5
            if delta < 1e-10:
                break
    
    return best, best_score


def try_lp_direction(f, M, n_constraints=None):
    """Try to find an LP-based direction. Returns None if infeasible or too slow.
    
    Solves: max sum(g) s.t. (f*g)[k] <= M for (subset of) k, g >= 0
    """
    n = len(f)
    n_full = 2 * n - 1
    
    try:
        # Build constraint matrix A where A[k, j] = f[k-j] if 0 <= k-j <= n-1
        # Use only a subset of constraints for speed
        if n_constraints and n_constraints < n_full:
            # Use every m-th constraint
            step_c = max(1, n_full // n_constraints)
            rows = np.arange(0, n_full, step_c)
        else:
            rows = np.arange(n_full)
        
        n_rows = len(rows)
        A = np.zeros((n_rows, n))
        for col_idx, j in enumerate(range(n)):
            r_start = max(0, j - (rows[0] - j))  # approximate
            # A[r, j] = f[rows[r] - j] if 0 <= rows[r] - j <= n-1
            diffs = rows - j
            valid = (diffs >= 0) & (diffs < n)
            A[valid, j] = f[diffs[valid]]
        
        c_obj = -np.ones(n)
        b_ub = np.full(n_rows, M)
        
        t_start = time.monotonic()
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30})
        
        if time.monotonic() - t_start > 40:
            return None
        
        if res.success:
            return res.x
        return None
    except Exception:
        return None


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0.0)


def propose_candidate(budget_s=2400):
    """Main search function combining multiple strategies."""
    deadline = time.monotonic() + budget_s - 30.0
    best_seq = None
    best_score = np.inf
    
    # Get provided sequence if available
    prov = None
    try:
        prov = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if prov.size > 0 and np.isfinite(prov).all():
            prov = np.maximum(prov, 0.0)
            if prov.sum() < 0.01:
                prov = None
        else:
            prov = None
    except Exception:
        prov = None
    
    print(f"Start: budget={budget_s}s, provided={'yes' if prov is not None else 'no'} (len={len(prov) if prov is not None else 0})")
    
    # ============ Phase 1: Exploration (first 25% of time) ============
    p1_budget = (budget_s - 30) * 0.20
    p1_end = time.monotonic() + p1_budget
    
    n_options = [2500, 3500, 4500]
    if prov is not None:
        n_options.append(len(prov))
    n_options = sorted(set(n_options))
    
    init_list = []
    for n in n_options:
        init_list.append((n, 'chirp', 0))
        init_list.append((n, 'chirp', 1))
        init_list.append((n, 'flat', 42))
        init_list.append((n, 'random', 7))
    if prov is not None:
        init_list.insert(0, (len(prov), 'provided', 0))
    
    t_per = max(12.0, (p1_end - time.monotonic()) / max(1, len(init_list)))
    print(f"\nPhase 1: {len(init_list)} candidates, ~{t_per:.0f}s each")
    
    for i, (n, itype, seed) in enumerate(init_list):
        tb = p1_end - time.monotonic()
        if tb < 10:
            break
        tb = min(t_per, tb)
        
        if itype == 'provided' and prov is not None:
            a0 = resample_sequence(prov, n)
        elif itype == 'chirp':
            a0 = init_chirp(n, seed)
        elif itype == 'flat':
            a0 = init_flat_spectrum(n, seed)
        else:
            a0 = init_random_smooth(n, seed)
        
        a_r, sc = run_gd(a0, 1.0, 0.03, 1.0, tb, p1_end, 
                         lr_min=0.05, lr_max_mult=3.0, check_interval=30)
        print(f"  P1[{i+1}/{len(init_list)}] n={n} {itype}_{seed} -> score={sc:.6f}")
        
        if sc < best_score:
            best_score = sc
            best_seq = a_r.copy()
            print(f"    *** NEW BEST: {best_score:.6f} (n={n})")
    
    # ============ Phase 2: Focused optimization (next 50% of time) ============
    p2_budget = (budget_s - 30) * 0.50
    p2_end = time.monotonic() + p2_budget
    
    if best_seq is not None:
        n_best = best_seq.size
        print(f"\nPhase 2: polishing n={n_best} with {p2_budget:.0f}s")
        
        # Multiple rounds with decreasing tau
        rounds = [
            (0.8, 0.02, 1.0, 0.35),
            (0.4, 0.01, 0.5, 0.30),
            (0.2, 0.005, 0.3, 0.35),
        ]
        
        for ri, (tau_hi, tau_lo, lr, frac) in enumerate(rounds):
            rem = p2_end - time.monotonic()
            if rem < 20:
                break
            tb = rem * frac
            print(f"  Round {ri+1}: tau=[{tau_hi},{tau_lo}] lr={lr} t={tb:.0f}s")
            a_r, sc = run_gd(best_seq, tau_hi, tau_lo, lr, tb, p2_end,
                            lr_min=0.01, lr_max_mult=4.0, check_interval=15)
            if sc < best_score:
                best_score = sc
                best_seq = a_r.copy()
                print(f"  *** improved to {best_score:.6f}")
            else:
                print(f"  no improvement (score={sc:.6f})")
            
            # Try targeted refinement
            rem = p2_end - time.monotonic()
            if rem > 30:
                a_r, sc = targeted_refine(best_seq, rem * 0.2, p2_end)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_r.copy()
                    print(f"  *** targeted refine improved to {best_score:.6f}")
    
    # ============ Phase 3: LP-based direction + fine polishing ============
    p3_budget = (budget_s - 30) * 0.20
    p3_end = time.monotonic() + p3_budget
    
    if best_seq is not None:
        n_best = best_seq.size
        print(f"\nPhase 3: LP + fine polish, n={n_best}, {p3_end - time.monotonic():.0f}s")
        
        S = np.sqrt(2.0 * n_best)
        a_norm = project_simplex(best_seq, S)
        
        # Try LP direction
        c_val = np.convolve(a_norm, a_norm)
        M = c_val.max()
        
        # Try with slightly tighter M
        for m_frac in [0.95, 0.90, 0.85]:
            rem = p3_end - time.monotonic()
            if rem < 15:
                break
            M_try = c_val.max() * m_frac
            print(f"  LP attempt with M={M_try:.4f} ({m_frac*100}%)")
            g_dir = try_lp_direction(a_norm, M_try, n_constraints=max(500, n_best))
            if g_dir is not None:
                g_norm = project_simplex(g_dir, S)
                # Blend
                for t in [0.3, 0.5, 0.7]:
                    blend = (1 - t) * a_norm + t * g_norm
                    blend = project_simplex(blend, S)
                    sc = compute_score(blend)
                    if sc < best_score:
                        best_score = sc
                        best_seq = blend.copy()
                        a_norm = blend.copy()
                        c_val = np.convolve(a_norm, a_norm)
                        print(f"    t={t}: score={sc:.6f} *** NEW BEST")
        
        # Final fine polishing with very small tau
        rem = p3_end - time.monotonic()
        if rem > 20:
            a_r, sc = run_gd(best_seq, 0.1, 0.001, 0.2, rem, p3_end,
                            lr_min=0.005, lr_max_mult=2.0, check_interval=10)
            if sc < best_score:
                best_score = sc
                best_seq = a_r.copy()
                print(f"  *** final polish: {best_score:.6f}")
    
    # ============ Final: try again with different n if we have time ============
    rem = deadline - time.monotonic()
    if rem > 60 and best_seq is not None:
        # Try a slightly different length with the best as starting point
        for dn in [500, -500]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            n_try = best_seq.size + dn
            if n_try < 1000:
                continue
            a_try = resample_sequence(best_seq, n_try)
            a_r, sc = run_gd(a_try, 0.5, 0.01, 0.5, rem * 0.4, deadline,
                            lr_min=0.02, lr_max_mult=3.0, check_interval=20)
            if sc < best_score:
                best_score = sc
                best_seq = a_r.copy()
                print(f"\n  *** different n={n_try}: {best_score:.6f}")
    
    # Final evaluation
    if best_seq is None:
        best_seq = np.ones(2000) / 2000.0
    final_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"\n=== FINAL: score={final_score:.6f}, analytic={compute_score(best_seq):.6f}, n={len(best_seq)} ===")
    
    return [float(x) for x in best_seq]
