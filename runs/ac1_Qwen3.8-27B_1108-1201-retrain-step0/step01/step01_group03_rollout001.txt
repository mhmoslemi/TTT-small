def softmax_grad_step(a, lr, tau=30.0):
    """One soft-max gradient step minimizing peak autocorrelation."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize_a(a, S_t)
    
    c = np.convolve(a, a)
    # Soft-max weights
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    grad = np.zeros(n)
    for j in range(n):
        lo = max(0, j)
        hi = min(2 * n - 1, j + n - 1)
        # k ranges from j to j+n-1 (capped at 2n-2)
        # a[k-j] where k-j ranges from 0 to n-1
        # So grad[j] = 2 * sum_{k=j}^{min(2n-2, j+n-1)} w[k] * a[k-j]
        # Let i = k-j, then i ranges from 0 to min(n-1, 2n-1-j)
        # grad[j] = 2 * sum_{i=0}^{min(n-1, 2n-1-j)} w[j+i] * a[i]
        hi_i = min(n, 2 * n - j)
        grad[j] = 2.0 * np.dot(a[:hi_i], w[j:j + hi_i])
    
    # Project out mean (fixed-sum constraint)
    grad -= grad.mean()
    
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = normalize_a(a, S_t)
    return a

Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based refinement, QP-based refinement (CVXPY), soft-max gradient

import numpy as np
import time
from scipy.optimize import linprog
try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_a(a, target=None):
    """Normalize sequence to have target sum."""
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.full(n, target / n, dtype=np.float64)
    return a * (target / S)


def build_lp_matrix(f):
    """Build (2n-1) x n matrix A where (A@g)[k] = conv(f,g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = f[k - hi:k - lo + 1][::-1]
    return A


def solve_lp_max_sum(f, M):
    """LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    c_obj = -np.ones(n)
    A = build_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.status == 0 and res.x is not None:
        return res.x
    return None


def solve_qp_closest(f, M, S_target):
    """QP: min ||g - f||^2 s.t. sum(g) = S_target, conv(f,g) <= M, g >= 0."""
    if not HAS_CVXPY:
        return None
    n = len(f)
    g = cp.Variable(n)
    A = build_lp_matrix(f)
    constraints = [
        cp.sum(g) == S_target,
        A @ g <= M,
        g >= 0
    ]
    objective = cp.Minimize(cp.sum_squares(g - f))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS)
        if g.value is not None:
            return np.maximum(g.value, 0.0)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
            if g.value is not None:
                return np.maximum(g.value, 0.0)
        except Exception:
            pass
    return None

def softmax_grad_step(a, lr, tau=30.0):
    """One soft-max gradient step for peak autocorrelation minimization."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize_a(a, S_t)
    
    c = np.convolve(a, a)
    c_shifted = c - float(np.max(c))
    w = np.exp(tau * c_shifted)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    
    # Gradient computation
    grad = np.zeros(n, dtype=np.float64)
    for j in range(n):
        hi_i = min(n, 2 * n - j)
        grad[j] = 2.0 * np.dot(a[:hi_i], w[j:j + hi_i])
    
    # Project out mean
    grad -= float(np.mean(grad))
    
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = normalize_a(a, S_t)
    return a
