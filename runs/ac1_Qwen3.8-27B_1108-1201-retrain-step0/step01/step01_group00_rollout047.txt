```python
"""
Hybrid optimization for minimizing peak autocorrelation of non-negative step functions.
Combines LP-based Frank-Wolfe iterations, gradient descent with annealed soft-max,
and simulated annealing local search.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def build_A_matrix(f):
    """Build the constraint matrix A for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    """Solve the LP: max sum(g) s.t. A*g <= M*1, g >= 0.
    Returns (optimal_value, optimal_g) or (None, None) on failure.
    """
    n = len(f)
    A = build_A_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    
    if res.success:
        return -res.fun, res.x
    return None, None


def lp_optimize(f_init, time_limit, verbose=True):
    """LP-based Frank-Wolfe optimization."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    rate = 0.5
    iter_count = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(f, f))
        V, g = lp_step(f, M)
        
        if g is None or V < S * 0.5:
            break
        
        # Rescale to match sum
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g_rescaled = g * (S / g_sum)
        
        # Frank-Wolfe step
        f_new = (1 - rate) * f + rate * g_rescaled
        f_new = np.maximum(f_new, 0.0)
        
        new_score = compute_score(f_new)
        
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            stall_count = 0
            rate = min(0.8, rate * 1.01)  # Slightly increase rate on improvement
        else:
            stall_count += 1
            rate = max(0.05, rate * 0.98)  # Decrease rate on stall
        
        f = f_new
        iter_count += 1
        
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iter_count}, score={new_score:.6f}, best={best_score:.6f}, rate={rate:.3f}, t={elapsed:.0f}s")
    
    return best_f, best_score
```
