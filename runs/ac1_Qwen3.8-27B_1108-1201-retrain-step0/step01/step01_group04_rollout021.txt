        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        start = time.monotonic()
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                      method="highs", options={"disp": False, "time_limit": max(1.0, budget_s - 1.0)})
    except Exception:
        return p, best_score
    if res is None or not hasattr(res, "x") or res.x is None:
        return p, best_score
    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return p, best_score
    g = g / s
    for t in (1.0, 0.8, 0.6, 0.4, 0.25, 0.1):
        q = _project_simplex((1.0 - t) * p + t * g)
        sc = _internal_score(q)
        if np.isfinite(sc) and sc < best_score:
            return q, sc
    return p, best_score


def _local_optimize(p0, rng, time_limit, label="", k_top=30, use_lp=False, lp_nmax=800):
    p = _project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    if n == 0:
        return p, np.inf
    best_p = p.copy()
    best_score = _internal_score(best_p)
    current_score = best_score
    prev_p = p.copy()
    end = time.monotonic() + max(0.0, float(time_limit))
    T_factor = 0.15
    iter_no = 0
    since_improve = 0

    if use_lp and n <= lp_nmax:
        lp_budget = min(40.0, max(5.0, float(time_limit) * 0.12))
        p_lp, sc_lp = _lp_best_response(p, best_score, lp_budget)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            current_score = sc_lp
            best_p = p_lp.copy()
            p = p_lp.copy()
            prev_p = p.copy()

    while time.monotonic() < end:
        iter_no += 1
        since_improve += 1
        if since_improve > 200:
            T_factor = max(0.15, T_factor * 1.1)
        elif since_improve < 50:
            T_factor = min(0.3, T_factor * 0.995)

        if since_improve % 200 == 0:
            k_use = max(10, int(k_top * (1.0 - 0.5 * min(1.0, since_improve / 2000.0))))
        else:
            k_use = k_top

        g, c, cmax = _topk_gradient(p, k_use, T_factor)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)
        gabs = float(np.max(np.abs(g))) if g.size else 0.0

        if gabs > 1e-300:
            base_lr = (1.0 / float(n)) / gabs
            lrs = (0.5 * base_lr, 0.2 * base_lr, 0.08 * base_lr, 0.03 * base_lr,
                   0.01 * base_lr, 0.004 * base_lr, 0.001 * base_lr)
        else:
            lrs = (0.1, 0.03, 0.01)

        accepted = False
        for lr in lrs:
            lr = float(lr)
            if not np.isfinite(lr) or lr <= 0.0:
                continue
            v = p - lr * g
            pt = _project_simplex(v)
            sc = _internal_score(pt)
            if np.isfinite(sc):
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_p = pt.copy()
                    since_improve = 0
                    current_score = sc
                    prev_p = p
                    p = pt
                    accepted = True
                    break
                elif sc < current_score - 1e-14:
                    if sc < best_score + 0.05:
                        current_score = sc
                        prev_p = p
                        p = pt
                        accepted = True
                        break

        if not accepted:
            T_factor = max(0.05, T_factor * 0.7)
            if since_improve > 100:
                sigmas = (0.5 / n, 0.1 / n, 0.02 / n, 0.004 / n)
            else:
                sigmas = (0.05 / n, 0.01 / n, 0.002 / n)
            for sigma in sigmas:
                pt = _project_simplex(p + rng.normal(0.0, float(sigma), n))
                sc = _internal_score(pt)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_p = pt.copy()
                    current_score = sc
                    prev_p = p
                    p = pt
                    since_improve = 0
                    accepted = True
                    break

            if not accepted and n > 1:
                hi = int(np.argmax(g)) if g.size else 0
                lo = int(np.argmin(g)) if g.size else 0
                if hi != lo and g[hi] > g[lo] + 1e-15 and p[hi] > 1e-10:
                    for frac in (0.2, 0.05, 0.01):
                        delta = float(frac) * float(p[hi])
                        if delta <= 0.0:
                            continue
                        pt = p.copy()
                        pt[hi] -= delta
                        pt[lo] += delta
                        pt = _project_simplex(pt)
                        sc = _internal_score(pt)
                        if np.isfinite(sc) and sc < best_score:
                            best_score = sc
                            best_p = pt.copy()
                            current_score = sc
                            prev_p = p
                            p = pt
                            since_improve = 0
                            accepted = True
                            break

            if not accepted and since_improve > 400:
                jump = 0.3 + 0.2 * float(rng.random())
                p = _project_simplex((1.0 - jump) * best_p + jump * rng.dirichlet(np.ones(n)))
                current_score = _internal_score(p)
                prev_p = p
                T_factor = 0.2
                since_improve = 0

            if not accepted and since_improve > 800:
                p = _project_simplex(0.5 * best_p + 0.3 * p + 0.2 * rng.dirichlet(np.ones(n)))
                current_score = _internal_score(p)
                prev_p = p
                T_factor = 0.25
                since_improve = 0

        if use_lp and n <= lp_nmax and iter_no % 3000 == 0:
            remaining = end - time.monotonic()
            if remaining > 15.0:
                p_lp, sc_lp = _lp_best_response(p, best_score, min(30.0, remaining))
                if np.isfinite(sc_lp) and sc_lp < best_score:
                    best_score = sc_lp
                    current_score = sc_lp
                    best_p = p_lp.copy()
                    p = p.copy() if isinstance(p, np.ndarray) else p
                    p = p_lp.copy()
                    prev_p = p.copy()
                    since_improve = 0

        if iter_no % 5000 == 0:
            print("  %s iter=%d best=%.10f cur=%.10f T=%.4f" % (label, iter_no, best_score, current_score, T_factor), flush=True)

        if best_score < 1.495 and since_improve > 15000:
            break

    best_p = _project_simplex(best_p)
    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def _shape_search(n, rng, budget_s):
    basis = _cheb_basis(n, 14)
    dim = 14
    best_p = None
    best_score = np.inf

    end = time.monotonic() + max(0.0, float(budget_s))
    starts = []
    starts.append(np.zeros(dim, dtype=np.float64))
    for _ in range(15):
        starts.append(rng.uniform(-1.5, 1.5, dim).astype(np.float64))
    t = np.zeros(dim)
    for i in range(dim):
        t[i] = 0.7 * (1 if i % 2 == 0 else -1) * (1.0 - i / (dim + 2.0))
    starts.append(t)
    t = np.zeros(dim)
    t[0] = 1.2
    t[2] = -0.5
    t[4] = 0.3
    starts.append(t)
    t = np.zeros(dim)
    t[1] = 0.8
    t[3] = 0.4
    t[5] = -0.2
    starts.append(t)

    for x0 in starts:
        if time.monotonic() > end - 2.0:
            break
        p = _make_cheb(n, x0)
        sc = _internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p.copy()

    if time.monotonic() < end - 30.0 and best_score < np.inf:
        n_sample = 200
        for _ in range(n_sample):
            if time.monotonic() > end - 2.0:
                break
            theta = rng.uniform(-2.0, 2.0, dim)
            p = _make_cheb(n, theta)
            sc = _internal_score(p)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = p.copy()

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(60.0, float(budget_s) - 15.0)

    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 64 and float(parent_arr.sum()) > 0.0:
            parent = parent_arr
    except Exception:
        parent = None

    n_main = 4096
    if parent is not None and 64 <= parent.size <= 8192:
        n_main = int(parent.size)
    n_short = 700

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0 or not np.all(np.isfinite(p)):
            return
        p = _safe_normalize(p)
        if not np.isfinite(sc):
            sc = _internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("  [NEW BEST] score=%.10f n=%d time=%.0fs" % (sc, p.size, time.monotonic() - start_time), flush=True)

    # Phase 1: Shape search for good initial structure
    if deadline - time.monotonic() > 200.0:
        shape_budget = min(120.0, (deadline - time.monotonic()) * 0.1)
        shape_p, shape_sc = _shape_search(n_main, rng, shape_budget)
        try_record(shape_p, shape_sc)
        print("Phase 1 (shape): best=%.6f" % shape_sc, flush=True)
    else:
        shape_p = None
        shape_sc = np.inf

    # Phase 2: Generate diverse initials and quick-optimize
    inits = _generate_initials(n_main, rng, parent)
    if shape_p is not None and shape_p.size == n_main:
        inits.insert(0, shape_p)
    scored = []
    for p in inits:
        p = _safe_normalize(p)
        sc = _internal_score(p)
        if np.isfinite(sc):
            scored.append((sc, p))
    scored.sort(key=lambda x: x[0])
    for sc, p in scored[:5]:
        try_record(p, sc)

    # Select top candidates for local optimization
    top_candidates = []
    seen = set()
    for sc, p in scored:
        if len(top_candidates) >= 8:
            break
        key = tuple(np.round(p[:64] * 1000, 0))
        if key not in seen:
            seen.add(key)
            top_candidates.append((sc, p))

    # Phase 3: Local optimization on top candidates
    n_cands = len(top_candidates)
    if n_cands > 0 and deadline - time.monotonic() > 100.0:
        remaining = deadline - time.monotonic() - 120.0
        time_per = max(20.0, remaining / float(n_cands))
        for i, (sc, p) in enumerate(top_candidates):
            if time.monotonic() > deadline - 120.0:
                break
            limit = min(time_per, deadline - time.monotonic() - 120.0)
            if limit < 10.0:
                break
            print("Phase 3: optimizing candidate %d/%d (init=%.6f, limit=%.0fs)" % (i + 1, n_cands, sc, limit), flush=True)
            p_opt, sc_opt = _local_optimize(p, rng, limit, "c%d" % i, k_top=30, use_lp=False)
            try_record(p_opt, sc_opt)

    # Phase 4: LP-assisted optimization on short sequence
    if deadline - time.monotonic() > 300.0:
        remaining = deadline - time.monotonic() - 100.0
        lp_budget = min(remaining * 0.4, 200.0)
        short_inits = _generate_initials(n_short, rng, None)
        if best_p is not None:
            # Resample best_p to short length
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, best_p.size)
            x_new = np.linspace(0, 1, n_short)
            f_interp = interp1d(x_old, best_p, kind='cubic', fill_value='extrapolate')
            short_from_best = np.maximum(f_interp(x_new), 0.0)
            short_inits.insert(0, _safe_normalize(short_from_best))
        scored_short = []
        for p in short_inits:
            p = _safe_normalize(p)
            sc = _internal_score(p)
            if np.isfinite(sc):
                scored_short.append((sc, p))
        scored_short.sort(key=lambda x: x[0])
        for sc, p in scored_short[:4]:
            try_record(p, sc)
            if deadline - time.monotonic() > 60.0:
                limit = min(lp_budget / 4.0, deadline - time.monotonic() - 60.0)
                if limit < 10.0:
                    break
                p_opt, sc_opt = _local_optimize(p, rng, limit, "short", k_top=20, use_lp=True, lp_nmax=n_short)
                try_record(p_opt, sc_opt)

    # Phase 5: Intensive optimization on the overall best
    if best_p is not None and deadline - time.monotonic() > 100.0:
        limit = max(50.0, deadline - time.monotonic() - 30.0)
        print("Phase 5: intensive polish (limit=%.0fs, current best=%.8f)" % (limit, best_score), flush=True)
        p_opt, sc_opt = _local_optimize(best_p, rng, limit, "polish", k_top=40, use_lp=False)
        try_record(p_opt, sc_opt)

    # Phase 6: Final fine-tune with small k_top and low temperature
    if best_p is not None and deadline - time.monotonic() > 30.0:
        limit = max(5.0, deadline - time.monotonic() - 5.0)
        p_opt, sc_opt = _local_optimize(best_p, rng, limit, "final", k_top=10, use_lp=False)
        try_record(p_opt, sc_opt)

    if best_p is None:
        best_p = np.full(n_main, 1.0 / n_main, dtype=np.float64)

    # Final validation
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s <= 1e-12:
        best_p = np.full(n_main, 1.0 / n_main, dtype=np.float64)
    final_sc = _internal_score(best_p)
    print("FINAL: score=%.10f n=%d elapsed=%.0fs" % (final_sc, best_p.size, time.monotonic() - start_time), flush=True)
    return [float(x) for x in best_p]
