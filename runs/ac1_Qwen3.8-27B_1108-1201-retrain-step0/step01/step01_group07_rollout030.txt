Minimizes 2*n*max(conv(a,a))/sum(a)^2 using FFT-accelerated mirror descent
with Adam optimizer and tau annealing. Multiple initializations and lengths
are explored in a two-phase strategy.
"""

import numpy as np
import time


def next_power_of_2(n):
    return 1 << (n - 1).bit_length()


def fft_convolve(a, b):
    """FFT-based linear convolution of a and b."""
    n_out = len(a) + len(b) - 1
    N = next_power_of_2(n_out)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_out]


def fft_cross_corr(p, a):
    """
    FFT-based linear cross-correlation.
    Returns corr[k] = sum_i p[i+k] * a[i] for k = 0, 1, ...
    (matches np.correlate(p, a, 'full'))
    """
    n_out = len(p) + len(a) - 1
    N = next_power_of_2(n_out)
    fp = np.fft.rfft(p, N)
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fp * np.conj(fa), N)
    return result[:n_out]


def smooth_max(c, tau):
    """Compute tau * log(sum(exp(c/tau))) with numerical stability."""
    c_shift = c - c.max()
    lse = tau * np.log(np.sum(np.exp(c_shift / tau)))
    return lse


def compute_grad(c, a, tau, n, N_corr):
    """
    Compute the Euclidean gradient of tau*log(sum(exp(c/tau))) w.r.t. a.
    grad[j] = (2/tau) * sum_k p[k] * a[k-j]
    where p = softmax(c/tau).
    Returns the gradient of shape (n,).
    """
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    p = exp_c / exp_c.sum()
    
    # R[j] = sum_k p[k] * a[k-j] = cross-correlation of p and a at lag j
    # np.correlate(p, a, 'full')[j] = sum_i p[i+j] * a[i] = R[j]
    R = fft_cross_corr(p, a)[:n]
    
    grad = (2.0 / tau) * R
    return grad, p


def mirror_descent_step(a, grad, lr, beta1, beta2, m, v, t, eps=1e-8):
    """
    Adam + mirror descent step.
    a_new[j] = a[j] * exp(-lr * adam_step[j]) / Z
    """
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    
    m_hat = m / (1 - beta1 ** (t + 1))
    v_hat = v / (1 - beta2 ** (t + 1))
    
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    step = np.clip(step, -50, 50)  # prevent underflow/overflow
    
    a_new = a * np.exp(-step)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / a_new.sum()
    
    return a_new, m, v


def init_uniform(n):
    return np.ones(n, dtype=np.float64)


def init_hann(n):
    i = np.arange(n)
    return 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))


def init_sine_sq(n):
    i = np.arange(n)
    return np.sin(np.pi * (i + 0.5) / n) ** 2


def init_gaussian(n):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    return np.exp(-0.5 * ((i - center) / sigma) ** 2)


def init_flat_top(n, edge_frac=0.1):
    """Flat top with cosine tapers at edges."""
    i = np.arange(n, dtype=np.float64)
    edge = int(n * edge_frac)
    a = np.ones(n)
    # Taper the edges
    for j in range(edge):
        frac = (j + 1) / (edge + 1)
        a[j] = 0.5 * (1 - np.cos(np.pi * frac))
        a[n - 1 - j] = a[j]
    return a


def init_random(n, seed=42):
    rng = np.random.default_rng(seed)
    return rng.random(n) + 0.1


def optimize_sequence(a0, t_budget, tau_start=10.0, tau_end=0.01, lr_base=1.0, verbose=False):
    """
    Optimize a sequence on the simplex using mirror descent + Adam + tau annealing.
    """
    n = len(a0)
    S = np.sqrt(2.0 * n)
    
    # Initialize and normalize
    a = np.maximum(a0.astype(np.float64), 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a / a.sum() * S
    
    # FFT sizes
    conv_len = 2 * n - 1
    N_conv = next_power_of_2(conv_len)
    corr_len = conv_len + n - 1  # = 3n - 2
    N_corr = next_power_of_2(corr_len)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    t_step = 0
    
    # Track best
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(t_budget, 1e-6))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Compute convolution
        fa = np.fft.rfft(a, N_conv)
        c = np.fft.irfft(fa * fa, N_conv)[:conv_len]
        
        # Compute gradient
        grad, p = compute_grad(c, a, tau, n, N_corr)
        
        # Mirror descent + Adam step
        a, m, v = mirror_descent_step(a, grad, lr_base, beta1, beta2, m, v, t_step)
        a *= S  # rescale to have the right sum
        t_step += 1
        
        # Periodic evaluation
        if t_step % 100 == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if verbose:
                    print(f"      step={t_step}, tau={tau:.4f}, score={best_score:.6f}")
    
    # Final evaluation
    score = evaluate_sequence(a.tolist())
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score
