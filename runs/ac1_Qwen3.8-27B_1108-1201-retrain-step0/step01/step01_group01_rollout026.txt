def _compute_gradient(a, c, tau, n_lags=50):
    """
    Compute gradient of smooth max: (1/tau)*log(sum exp(tau*c_k)).
    Only uses top n_lags terms for efficiency.
    """
    n = len(a)
    # Compute softmax weights
    c_max = np.max(c)
    exponents = tau * (c - c_max)
    # Clip for numerical stability
    exponents = np.clip(exponents, -100, 100)
    weights = np.exp(exponents)
    total = np.sum(weights)
    if total < 1e-300:
        # All weights are essentially zero except max
        weights = np.zeros(len(c))
        weights[np.argmax(c)] = 1.0
    else:
        weights /= total
    
    # Take top n_lags
    if n_lags < len(c):
        top_idx = np.argpartition(weights, -n_lags)[-n_lags:]
        mask = np.zeros(len(c), dtype=bool)
        mask[top_idx] = True
        # Renormalize
        w_sum = weights[mask].sum()
        if w_sum < 1e-30:
            mask = np.zeros(len(c), dtype=bool)
            mask[np.argmax(c)] = True
            w_sum = 1.0

        weights_masked = weights

    if len(active_idx) == 0:

    grad = np.zeros(n)

"""
Nesterov-accelerated projected gradient with smooth max approximation to minimize
the normalized autocorrelation peak of a non-negative sequence. Uses structured
starting points (edge-heavy profiles, cosines, parabolas) with adaptive temperature
scheduling, Barzilai-Borwein step sizes, and multi-phase optimization.
"""

import numpy as np
import time


def _score_of(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * np.max(c) / (s * s))


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fft_convolve(a):
    """Full convolution using FFT."""
    n = len(a)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:need]


def _compute_gradient(a, c, tau, n_lags=100):
    """
    Gradient of smooth max (1/tau)*log(sum_k exp(tau*c_k)).
    Uses only top n_lags for efficiency.
    Returns gradient vector of length n.
    """
    n = len(a)
    c_max = np.max(c)
    exponents = tau * (c - c_max)
    exponents = np.clip(exponents, -80.0, 80.0)
    weights = np.exp(exponents)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(c))
        weights[np.argmax(c)] = 1.0
    else:
        weights /= w_sum

    # Take top n_lags by weight
    if n_lags < len(weights):
        top = np.argpartition(weights, -n_lags)[-n_lags:]
        sub_w = weights[top].sum()
        if sub_w > 1e-30:
            weights[top] = weights[top] / sub_w
            # Zero out the rest
            mask = np.zeros(len(weights), dtype=bool)
            mask[top] = True
            weights = np.where(mask, weights, 0.0)
        else:
            weights = np.zeros(len(c))
            weights[np.argmax(c)] = 1.0

    active = np.nonzero(weights > 1e-12)[0]
    if len(active) == 0:
        return np.zeros(n)

    grad = np.zeros(n)
    for k, w in zip(active, weights[active]):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(hi, lo - 1, -1)
            grad[lo:hi + 1] += 2.0 * w * a[k - idx]
    return grad


def _make_raised_cosine(n, freq=1):
    """a_i = 1 + cos(2*pi*freq*i/n), high at edges."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + np.cos(2.0 * np.pi * freq * i / n)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def _make_inverted_cosine(n, freq=1):
    """a_i = 1 - cos(2*pi*freq*i/n), low at edges."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.cos(2.0 * np.pi * freq * i / n)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def _make_parabolic(n, floor=0.05):
    """a_i = ((2*i/(n-1)) - 1)^2 + floor, U-shaped."""
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / (n - 1) - 1.0
    a = t * t + floor
    return a / np.sum(a)


def _make_gaussian(n, sigma_frac=0.4):
    """Gaussian centered, heavy in middle."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / (sigma_frac * n / 2.0)) ** 2)
    return a / np.sum(a)


def _make_two_level(n, ratio=1.5):
    """First half high, second half low."""
    a = np.ones(n)
    a[:n // 2] = ratio
    return a / np.sum(a)


def _make_scurve(n, sharpness=6.0):
    """Sigmoid-shaped profile."""
    i = np.arange(n, dtype=np.float64)
    t = sharpness * (2.0 * i / (n - 1) - 1.0)
    a = 1.0 / (1.0 + np.exp(-t))
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def _make_uniform(n):
    return np.ones(n) / n


def _single_optimize(a, deadline, tau_init=2.0, tau_max=200.0,
                     eta_init=0.1, n_lags_init=30, n_lags_max=300,
                     momentum_init=0.3, label=""):
    """
    Nesterov-accelerated projected gradient with smooth max.
    Uses BB step size and adaptive parameters.
    """
    n = len(a)
    a = _project_simplex(a.copy(), 1.0)

    best_a = a.copy()
    best_score = _score_of(best_a)

    y = a.copy()  # Nesterov y
    a_prev = a.copy()
    tau = tau_init
    eta = eta_init
    momentum = momentum_init
    n_lags = n_lags_init
    no_improve = 0
    step = 0
    bb_prev_dx = None
    bb_prev_dg = None

    while time.monotonic() < deadline:
        step += 1

        # Nesterov prediction
        if step > 1:
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step * step)
            alpha = (1.0 - t_curr) / (1.0 - t_curr + 1e-15)
            # Standard Nesterov: y = x + (x - x_prev) * (iter/(iter+3))
            ratio = (step - 1) / (step + 2.0)
            y = a + ratio * (a - a_prev)
        else:
            y = a.copy()

        # Compute convolution and score at y
        c = _fft_convolve(y)
        cur_score = 2.0 * n * np.max(c)

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature: increase over iterations
        tau = min(tau_max, tau_init * (1.0 + 0.0005 * step))

        # Compute gradient at y
        grad = _compute_gradient(y, c, tau, n_lags)

        # BB step size estimation
        if bb_prev_dx is not None:
            dx = y - a_prev
            dg = grad - bb_prev_dg
            gamma = np.dot(dx, dg)
            if abs(gamma) > 1e-20:
                eta_bb = np.dot(dx, dx) / gamma
                eta = 0.5 * eta + 0.5 * max(1e-10, min(eta_bb, 10.0 * eta))
        bb_prev_dx = y - a_prev
        bb_prev_dg = grad.copy()

        a_prev = a.copy()

        # Gradient step with momentum
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        a_new = y - eta * grad / gnorm

        # Project onto simplex
        a = _project_simplex(a_new, 1.0)

        # Adapt parameters
        if no_improve > 1000:
            eta *= 0.7
            momentum = min(0.95, momentum + 0.02)
            n_lags = min(n_lags_max, n_lags + 20)
            no_improve = 0
            if eta < 1e-10:
                break

        # Gradually increase n_lags
        if step % 500 == 0:
            n_lags = min(n_lags_max, int(n_lags * 1.2) + 5)

    return best_a, best_score

def _perturb(a, rng, scale=0.03):
    """Smooth random perturbation."""
    n = len(a)
    freqs = rng.integers(1, max(2, n // 8), size=min(8, n // 4))
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2.0 * np.pi * f * np.arange(n) / n + phase)
    max_n = np.max(np.abs(noise))
    if max_n > 1e-15:
        noise /= max_n
    a_new = a + scale * noise
    return _project_simplex(a_new, 1.0)

        if best_s < global_best_score:
            global_best_score = best_s
            global_best = best_a.copy()
        print(f"Phase 4 done: best={global_best_score:.6f}")

    # Final evaluation
    result = [float(x) for x in global_best]
    # Ensure all values are valid
    for i in range(len(result)):
        if result[i] < 0:
            result[i] = 0.0
        if result[i] > 1000:
            result[i] = 1000.0

    final_score = float(evaluate_sequence(result))
    print(f"FINAL SCORE: {final_score:.6f} (n={n})")
    return result
