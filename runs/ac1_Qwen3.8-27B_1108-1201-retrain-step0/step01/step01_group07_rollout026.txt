Minimizes 2*n*max(conv(a,a))/sum(a)^2 using FFT-accelerated projected
gradient descent with softmax smoothing, coordinate descent polishing,
and optional LP-based refinement. Multi-scale and multi-init strategy.
"""

import time
from scipy.optimize import linprog

def next_pow2(n):
    """Next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def fast_conv_same(a):
    """Compute np.convolve(a, a, 'full') using FFT."""
    n = len(a)
    conv_len = 2 * n - 1
    size = next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:conv_len]

def fast_conv_ab(a, b):
    """Compute np.convolve(a, b, 'full') using FFT."""
    n = len(a)
    m = len(b)
    conv_len = n + m - 1
    size = next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:conv_len]

def project_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S} using sorting-based method."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    s = a.sum()
    if s <= S + 1e-15:
        # Need to add (S-s) total, distribute uniformly
        return a + (S - s) / n
    # Sort in decreasing order
    idx = np.argsort(a)[::-1]
    sorted_a = a[idx]
    cumsum = np.cumsum(sorted_a)
    # Find largest m such that sorted_a[m-1] > (cumsum[m-1] - S) / m
    thresholds = (cumsum - S) / np.arange(1, n + 1)
    valid = sorted_a > thresholds
    if not valid.any():
        t = (s - S) / n
        return np.maximum(a - t, 0.0)
    m = np.where(valid)[0][-1] + 1
    t = (cumsum[m - 1] - S) / m
    result = np.maximum(a - t, 0.0)
    # Fix numerical issues
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        result[np.argmax(result)] += diff
    return result

def compute_grad_softmax(a, tau):
    """Compute gradient of tau*log(sum(exp(conv(a,a)/tau))) w.r.t. a."""
    n = len(a)
    c = fast_conv_same(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    Q = exp_c / exp_c.sum()
    
    # grad[j] = (Q*a)[j] + (Q*a)[2n-2-j]
    # Q has length 2n-1, a has length n
    # Full conv has length 3n-2
    conv_len = 3 * n - 2
    size = next_pow2(conv_len)
    fq = np.fft.rfft(Q, size)
    fa = np.fft.rfft(a, size)
    conv_qa = np.fft.irfft(fq * fa, size)[:conv_len]
    
    # grad[j] = conv_qa[j] + conv_qa[2n-2-j] for j=0,...,n-1
    # conv_qa[2n-2-j] for j=0,...,n-1 is conv_qa[2n-2], conv_qa[2n-3], ..., conv_qa[n-1]
    # = conv_qa[n-1:2n-1][::-1]
    grad = conv_qa[:n] + conv_qa[n-1:2*n-1][::-1]
    
    return grad, c, c_max

def init_uniform(n):
    return np.ones(n, dtype=np.float64)

def init_half_cosine(n):
    """Half-cosine (raised cosine) window - smooth bell shape."""
    t = np.linspace(0, np.pi, n)
    return 0.5 * (1 + np.cos(t))

def init_chirp(n, seed=0):
    t = np.arange(n, dtype=np.float64) - (n - 1) * 0.5
    a = 1.0 + 0.5 * np.cos(2 * np.pi * t * t / (2 * n))

    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

def init_gaussian(n, sigma_frac=0.3):

    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)

def run_pgd(a0, S, tau_hi, tau_lo, lr0, budget, verbose=False):
    """FFT-accelerated projected gradient descent with tau annealing."""
    n = a0.size
    a = project_simplex(a0.copy(), S)
    
    c = fast_conv_same(a)
    best_mc = c.max()
    best = a.copy()
    
    t0 = time.monotonic()
    lr = lr0
    step = 0
    patience = 0
    
    while time.monotonic() - t0 < budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(budget, 1e-6))
        tau = tau_hi * (tau_lo / tau_hi) ** frac
        
        grad, c, c_max = compute_grad_softmax(a, tau)
        
        # Backtracking line search (simplified)
        a_new = project_simplex(a - lr * grad, S)
        step += 1
        
        if step % 5 == 0:
            c_new = fast_conv_same(a_new)
            mc_new = c_new.max()
            
            if mc_new < best_mc - 1e-12:
                best_mc = mc_new
                best = a_new.copy()
                patience = 0
                # Increase learning rate slightly
                if lr < lr0 * 10:
                    lr *= 1.02
            else:
                patience += 1
                if patience > 50:
                    lr *= 0.7
                    patience = 0
            a = a_new
        else:
            a = a_new
    
    # Final evaluation
    c = fast_conv_same(best)
    mc = c.max()
    score = mc  # Since S = sqrt(2n), score = max_c

def coordinate_descent(a, S, n_sweeps, verbose=False):
    """Coordinate descent: for each position, try to optimize its value."""
    n = a.size
    a = a.copy()
    c = fast_conv_same(a)
    best_mc = c.max()
    best = a.copy()
    
    t0 = time.monotonic()
    sw = 0
    while sw < n_sweeps:
        sw += 1
        improved = False
        for j in range(n):
            # Try reducing a[j]
            old_val = a[j]
            if old_val <= 1e-10:
                continue
            
            # Try a few fractional changes
            for frac in [0.5, 0.7, 0.9, 1.1, 1.3]:
                new_val = old_val * frac
                if new_val < 0:
                    continue
                a[j] = new_val
                # Renormalize
                s = a.sum()
                a = a * (S / s)
                
                c_new = fast_conv_same(a)
                mc_new = c_new.max()
                if mc_new < best_mc - 1e-12:
                    best_mc = mc_new
                    best = a.copy()
                    improved = True
                    break
            a[j] = old_val
            # Re-normalize back
            s = a.sum()
            a = a * (S / s)
        
        if not improved and sw > 2:
            break
        if verbose and sw % 5 == 0:
            print(f"    CD sweep {sw}/{n_sweeps}, best_mc={best_mc:.6f}")
    
    return best, best_mc

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using FFT-accelerated projected
gradient descent with softmax smoothing, adaptive learning rate, tau
annealing, coordinate descent polishing, and LP-based refinement.
Multi-scale and multi-init strategy with restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog

def next_pow2(n):
    """Next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def fast_conv_same(a):
    """Compute np.convolve(a, a, 'full') using FFT."""
    n = len(a)
    conv_len = 2 * n - 1
    size = next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:conv_len]

def fast_conv_ab(a, b):
    """Compute np.convolve(a, b, 'full') using FFT."""
    n = len(a)
    m = len(b)
    conv_len = n + m - 1
    size = next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:conv_len]

def project_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S} using sorting-based method."""
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    n = a.size
    s = a.sum()
    if s <= S + 1e-15:
        return a + (S - s) / n
    idx = np.argsort(a)[::-1]
    sorted_a = a[idx]
    cumsum = np.cumsum(sorted_a)
    thresholds = (cumsum - S) / np.arange(1, n + 1)
    valid = sorted_a > thresholds
    if not valid.any():
        t = (s - S) / n
        return np.maximum(a - t, 0.0)
    m = int(np.where(valid)[0][-1]) + 1
    t = (cumsum[m - 1] - S) / m
    result = np.maximum(a - t, 0.0)
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        result[np.argmax(result)] += diff
    return result

def compute_grad_softmax(a, tau):
    """Compute gradient of tau*log(sum(exp(conv(a,a)/tau))) w.r.t. a using FFT."""
    n = len(a)
    c = fast_conv_same(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    Q = exp_c / exp_c.sum()
    conv_len = 3 * n - 2
    size = next_pow2(conv_len)
    fq = np.fft.rfft(Q, size)
    fa = np.fft.rfft(a, size)
    conv_qa = np.fft.irfft(fq * fa, size)[:conv_len]
    grad = conv_qa[:n] + conv_qa[n - 1:2 * n - 1][::-1]
    return grad, c

def init_uniform(n):
    return np.ones(n, dtype=np.float64)

def init_half_cosine(n):
    t = np.linspace(0, np.pi, n)
    return 0.5 * (1 + np.cos(t))

def init_chirp(n, seed=0):
    t = np.arange(n, dtype=np.float64) - (n - 1) * 0.5
    a = 1.0 + 0.5 * np.cos(np.pi * t * t / n)
    rng = np.random.default_rng(seed)
    a += 0.03 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

def init_random(n, seed=0):
    return np.random.default_rng(seed).exponential(1.0, size=n)

def init_gaussian(n, sigma_frac=0.35):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)

def init_bell(n, p=2.0):
    """Generalized bell: (1 + cos(pi*x))^p normalized."""
    t = np.linspace(0, 1, n)
    return (1 + np.cos(np.pi * t)) ** p

def run_pgd(a0, S, tau_hi, tau_lo, lr0, budget, verbose=False):
    """FFT-accelerated projected gradient descent with tau annealing and adaptive lr."""
    n = a0.size
    a = project_simplex(a0, S)
    c = fast_conv_same(a)
    best_mc = c.max()
    best = a.copy()

    step = 0

        tau = tau_hi * (tau_lo / tau_hi) ** (frac ** 0.7)

def lp_refine(a, S, max_iters, verbose=False):
    """LP-based refinement: find direction b maximizing sum(b) s.t. ||a*b||_inf <= M."""
    n = a.size
    M = fast_conv_same(a).max()
    f = a  # fixed
    best = a.copy()
    best_mc = M

            break
        
        # Build LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0
        # (f*b)[k] = sum_j f[k-j]*b[j] where f[idx]=0 if idx out of [0,n-1]
        # A has shape (2n-1, n): A[k,j] = f[k-j] if 0<=k-j<n else 0
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[k, j] = f[k - j]
        
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')
        
        if res.status != 0:
            break
        
        b = res.x
        b_sum = b.sum()
        if b_sum < 1e-10:
