def _fft_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def conv_full(p):
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def score_internal(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:

    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def smooth_max_grad(p, T_frac):
    """Compute gradient of smoothed max of self-convolution."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64), 0.0, 0.0
    c = conv_full(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.zeros(n), cmax, 0.0
    T = max(T_frac * cmax, 1e-15)
    diff = (c - cmax) / T
    e = np.exp(np.clip(diff, -50, 0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    if n == 1:

        tmp[:m] = G[:m]

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / n)
    return out / s

def make_shape(n, kind, params, rng):
    """Generate various parameterized shapes."""
    n = int(max(2, n))

        sig = params.get("sigma", 0.08)
        p = np.exp(-0.5 * ((x - mu1) / sig) ** 2) + np.exp(-0.5 * ((x - mu2) / sig) ** 2)
    elif kind == "power":
        a = params.get("a", 2.0)
        p = np.power(x + 1e-8, a)
    elif kind == "cup":
        a = params.get("a", 1.0)
        p = 1.0 + a * (2.0 * x - 1.0) ** 2
    elif kind == "cap":
        a = params.get("a", 0.5)
        p = np.maximum(1.0 - a * (2.0 * x - 1.0) ** 2, 0.01)
    elif kind == "cosine":
        p = 0.5 + 0.5 * np.cos(np.pi * x)
    elif kind == "sin_mod":
        f = params.get("freq", 3.0)
        a = params.get("amp", 0.3)
        p = np.maximum(1.0 + a * np.sin(2 * np.pi * f * x), 0.01)
    elif kind == "triangle":
        p = 1.0 - np.abs(2.0 * x - 1.0)
    elif kind == "exp_decay":
        rate = params.get("rate", 2.0)
        p = np.exp(-rate * x)
        p = np.maximum(p, 0.001)
    elif kind == "random_smooth":
        nterms = int(rng.integers(2, 6))
        p = np.ones(n)
        for _ in range(nterms):
            f = float(rng.uniform(1, 20))
            phase = float(rng.uniform(0, 2 * np.pi))
            amp = float(rng.uniform(-0.5, 0.5))
            p += amp * np.sin(2 * np.pi * f * x + phase)
        p = np.maximum(p, 0.01)
    elif kind == "cheb":
        deg = int(params.get("deg", 5))
        coeffs = [float(rng.uniform(-1, 1)) for _ in range(deg + 1)]
        z = 2 * x - 1
        p = np.zeros(n)
        T_nm = np.ones(n)
        T_n = z.copy()
        p += coeffs[0] * T_nm
        if deg >= 1:
            p += coeffs[1] * T_n
        for d in range(2, deg + 1):
            T_np = 2 * z * T_n - T_nm
            p += coeffs[d] * T_np
            T_nm, T_n = T_n, T_np
        p = np.maximum(p, 0.01)
    else:
        p = np.ones(n)
    return normalize(p)

                if time.monotonic() > deadline - final_reserve - 30:
                    break
                p_rand = make_shape(ln, "random_smooth", {}, rng)
                sc_rand = score_internal(p_rand)
                if np.isfinite(sc_rand) and sc_rand < best_score + 0.1:
                    limit = min(60.0, max(10.0, deadline - final_reserve - time.monotonic()))
                    if limit > 10:
                        p_opt, sc_opt = optimize_local(p_rand, rng, limit, ln <= 500)
                        record(p_opt, sc_opt)

    # Phase 4: Final polish on best
    if best_p is not None and time.monotonic() < deadline - 10:
        print("Phase 4: Final polish...", flush=True)
        limit = max(5.0, deadline - 5.0 - time.monotonic())
        p_fin, sc_fin = optimize_local(best_p, rng, limit, int(best_p.size) <= 500)
        record(p_fin, sc_fin)

    if best_p is None:
        n = int(parent.size) if parent is not None else 700
        best_p = make_shape(n, "uniform", {}, rng)
        best_score = score_internal(best_p)

    # Final validation
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s < 0.01:
        best_p = np.full(int(best_p.size), 1.0)
    final_score = score_internal(best_p)
    print("FINAL: score=%.10f n=%d elapsed=%.1fs" % (final_score, best_p.size, time.monotonic()-t0), flush=True)

    return [float(x) for x in best_p]
