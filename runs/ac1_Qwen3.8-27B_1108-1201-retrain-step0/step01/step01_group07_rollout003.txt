def compute_grad_topk(a, k=5):
    """Compute gradient using top-k peaks of convolution."""
    c = np.convolve(a, a)
    # Find top k indices
    top_idx = np.argsort(c)[-k:]
    # For each top index, compute the gradient direction
    grad = np.zeros_like(a)
    n = len(a)
    for ki in top_idx:
        ki = int(ki)
        # dc_ki/da_j = 2*a_{ki-j} (zero-padded)
        # So we add 2*a shifted by ki
        for j in range(n):
            idx = ki - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]
    return grad, c

def project_nonneg(a):
    """Project onto non-negative orthant."""
    return np.maximum(a, 0.0)

def init_chirp(n, seed=0):
    """Chirp-based initialization."""
    t = np.arange(n, dtype=np.float64)
    # Quadratic chirp
    a = 1.0 + 0.5 * np.cos(2 * np.pi * t * t / (2*n))
    rng = np.random.default_rng(seed)
    a += 0.1 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

def init_random(n, seed=0):
    """Random initialization."""
    return np.random.default_rng(seed).exponential(1.0, size=n)

def init_barker_like(n, seed=0):
    """Initialize with a pattern that has good autocorrelation properties."""
    rng = np.random.default_rng(seed)

    # Apply a window to reduce edge effects

    a = a * (0.5 + 0.5 * window)

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses multi-phase optimization: diverse exploration, softmax gradient descent
with adaptive temperature, momentum, and local search refinement.
"""

import numpy as np
import time

def compute_softmax_grad(a, tau):
    """
    Compute gradient of tau*log(sum(exp(c_k/tau))) where c=conv(a,a).
    Returns (gradient, convolution).
    """
    c = np.convolve(a, a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    q = exp_c / exp_c.sum()
    # grad_j = 2 * sum_k q_k * a_{k-j} = 2 * correlate(q, a, 'valid')
    grad = 2.0 * np.correlate(q, a, 'valid')
    return grad, c

def score_of(a):
    """Compute the score without calling evaluate_sequence."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s ** 2))

def normalize(a, target_sum=None):
    """Normalize to a target sum (default: sqrt(n))."""
    n = len(a)
    if target_sum is None:
        target_sum = np.sqrt(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * target_sum / n
    return a * (target_sum / s)

def init_gaussian(n, seed=0):
    """Gaussian window initialization."""
    t = np.arange(n, dtype=np.float64) - (n-1)/2.0
    sigma = n / 6.0
    a = np.exp(-t**2 / (2*sigma**2))
    # Full support, no zeros
    a = np.maximum(a, 0.01)
    rng = np.random.default_rng(seed)
    a += 0.02 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

def init_chirp(n, seed=0):
    """Quadratic chirp initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.3 * np.cos(2*np.pi*t**2/(2*n))
    rng = np.random.default_rng(seed)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

def init_window(n, window_type='hann', seed=0):
    """Window function initialization."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    if window_type == 'hann':
        a = 0.5 * (1 + np.cos(2*np.pi*t))
    elif window_type == 'hamming':
        a = 0.54 - 0.46 * np.cos(2*np.pi*t)
    elif window_type == 'blackman':
        a = 0.42 - 0.5*np.cos(2*np.pi*t) + 0.08*np.cos(4*np.pi*t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.001)
    rng = np.random.default_rng(seed)
    a += 0.01 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

def run_softmax_opt(a0, budget, tau_hi=1.0, tau_lo=0.005, lr_init=1.0, momentum=0.9, verbose=False):
    """
    Softmax gradient descent with momentum and adaptive learning rate.
    """
    n = a0.size
    a = a0.astype(np.float64).copy()
    a = normalize(a)
    
    vel = np.zeros(n)  # Momentum
    best_a = a.copy()
    best_score = score_of(a)
    best_mc = np.convolve(a, a).max()
    
    t0 = time.monotonic()
    lr = lr_init

    staleness = 0

        elif progress < 0.7:

        a_new = a + vel

        a = a_new

def peak_reduction(a, budget, verbose=False):
    """
    Direct peak reduction: identify the peak in the convolution and
    reduce the elements that contribute to it.
    """
    n = a.size
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = score_of(a)
    
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < budget:
        c = np.convolve(a, a)
        max_idx = c.argmax()
        max_val = c.max()
        
        # Find the top contributors to the peak
        # c[max_idx] = sum_i a_i * a[max_idx - i]
        # The top contributors are the i where a_i * a[max_idx-i] is largest
        products = np.zeros(n)
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                products[i] = a[i] * a[j]
        
        # Reduce the top contributors
        top_k = max(5, n // 50)
        top_idx = np.argsort(products)[-top_k:]
        
        # Reduce these elements proportionally
        reduction_factor = 0.01
        a[top_idx] *= (1 - reduction_factor)
        a = np.maximum(a, 0.0)
        a = normalize(a)
        
        step += 1
        
        if step % 5 == 0:
            sc = score_of(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

def init_from_provided_safe(prov, n_target):
    """Safely resample provided sequence to target length."""
    try:
        n_prov = len(prov)
        x_old = np.linspace(0.0, 1.0, n_prov)
        x_new = np.linspace(0.0, 1.0, n_target)
        a = np.interp(x_new, x_old, prov)
        return np.maximum(a, 0.0)
    except Exception:
        return np.ones(n_target)
