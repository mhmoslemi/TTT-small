    """Compute tau * log(sum(exp(vals/tau))) and corresponding weights."""

    P can be computed as: P[j] = (flip(w) * a)[len(w)-1-j] ... let me verify.
    (flip(w) * a)[k] = sum_i flip(w)[i]*a[k-i] = sum_i w[len(w)-1-i]*a[k-i]
    At k = len(w)-1-j: sum_i w[len(w)-1-i]*a[len(w)-1-j-i]
    Let m = len(w)-1-i, so i = len(w)-1-m: sum_m w[m]*a[j-m]
    That gives Q[j], not P[j].
    
    For P[j] = sum_m a[m]*w[j+m]:
    (w * a)[k] = sum_i w[i]*a[k-i]
    At k = j + n - 1: sum_i w[i]*a[j+n-1-i]
    Let m = j+n-1-i, i = j+n-1-m: sum_m w[j+n-1-m]*a[m]
    That's not P[j] either.
    
    OK, let me just use the most direct FFT approach:
    P[j] = sum_{m=0}^{n-1} a[m] * w[j+m]  (where w[j+m] = 0 if j+m >= len(w))
    This is a "sliding dot product" - can be computed via FFT as:
    P = ifft(fft(w) * conj(fft(a))) ... no, that gives correlation.
    
    Actually, let me define it differently. Let me just pad and use FFT:
    P[j] for j=0..n-1: this is the cross-correlation of w (length 2n-1) with a (length n)
    at lags j = 0, 1, ..., n-1.
    
    np.correlate(w, a, mode='full') gives:
    corr[k] = sum_m w[m + k - (n-1)] * a[m] ... the indexing is tricky.
    
    Let me just use: P[j] = sum_{m=0}^{n-1} a[m]*w[j+m]
    If I define b = w (length 2n-1) and a (length n), then:
    P[j] = sum_{m=0}^{n-1} a[m]*b[j+m] = (a reversed) convolved with b, at specific indices.
    
    (flip(a) * b)[k] = sum_i flip(a)[i] * b[k-i] = sum_i a[n-1-i]*b[k-i]
    Let m = n-1-i: sum_m a[m]*b[k-n+1+m]
    At k = n-1+j: sum_m a[m]*b[j+m] = P[j] ✓
    
    So P[j] = (flip(a) * b)[n-1+j] where b = w.
    
    Similarly: Q[j] = sum_{m=0}^{min(j,n-1)} a[m]*w[j-m]
    (a * b)[k] = sum_i a[i]*b[k-i]
    At k = j: sum_i a[i]*b[j-i] = sum_{i=0}^{min(j,n-1)} a[i]*b[j-i] = Q[j] ✓ (since b[j-i] = w[j-i] = 0 for j-i < 0 or j-i >= len(w), and j < n < len(w))
    
    So: g[j] = P[j] + Q[j] = (flip(a) * w)[n-1+j] + (a * w)[j]
    
    Both can be computed via FFT!
    """
    n = len(a)
    c = fft_convolve(a, a)
    value, w = logsumexp(c, tau)
    
    # P[j] = (flip(a) * w)[n-1+j] for j = 0..n-1
    flip_a = a[::-1]
    conv_pw = fft_convolve(flip_a, w)  # length n + len(w) - 1 = n + 2n-1 - 1 = 3n-2
    P = conv_pw[n-1:n-1+n]  # P[0..n-1]
    
    # Q[j] = (a * w)[j] for j = 0..n-1
    conv_aw = fft_convolve(a, w)  # length n + len(w) - 1 = 3n-2
    Q = conv_aw[0:n]  # Q[0..n-1]
    
    g = P + Q
    return value, g, c


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def optimize_gradient_v2(n, theta_init, time_limit, rng):
    """
    Improved gradient descent with:
    - Better learning rate schedule
    - More aggressive temperature annealing  
    - Periodic random kicks
    - Peak-targeted perturbations
    """
    S = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = np.inf
    tau_start = 50.0 / n
    tau_end = 0.1 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # More aggressive annealing: reach low tau earlier
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)
        
        a = softmax_normalize(theta) * S
        value, g_a, c = grad_smoothed_max_conv_fast(a, tau)
        
        # Gradient w.r.t. theta: a = softmax(theta) * S
        # da[dtheta] = a * (e_j/S - delta_ij) * S = a[e_j - delta_ij] (where e = softmax(theta))
        # = a * (a/S - delta_ij)  ... wait
        # a_i = S * exp(theta_i) / sum(exp(theta))
        # da_i/dtheta_j = S * exp(theta_i) * (delta_ij * sum(exp) - exp(theta_i)) / sum(exp)^2
        #               = a_i * (delta_ij - a_j/S)
        # So g_theta_j = sum_i g_a_i * da_i/dtheta_j = sum_i g_a_i * a_i * (delta_ij - a_j/S)
        #              = a_j * (sum_i g_a_i * a_i - sum_i g_a_i * a_i * a_j/S)
        #              = a_j * (inner - inner * a_j/S)  where inner = sum(g_a * a)
        # Wait let me redo: g_theta_j = sum_i g_a_i * a_i * (delta_ij - a_j/S)
        # = g_a_j * a_j - (a_j/S) * sum_i g_a_i * a_i
        # = a_j * (g_a_j - dot(g_a, a)/S)
        
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)
        
        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        
        # Learning rate: start high, decay, with periodic boosts
        base_lr = 3.0 / np.sqrt(step + 1)
        if step % 5000 == 0:
            base_lr *= 2.0  # periodic boost
        theta = theta - base_lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Compute actual score
        a_check = softmax_normalize(theta) * S
        score = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (S * S)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
        
        # Periodic random kick to escape local minima
        if step % 3000 == 0 and elapsed_frac < 0.8:
            kick_size = 0.5 * (1.0 - elapsed_frac)
            theta += rng.standard_normal(n) * kick_size
        
        if step % 10000 == 0:
            print(f"      step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}")
    
    return best_theta, best_score


def local_search_v2(a_init, time_limit, rng):
    """
    Improved local search with:
    - Peak-identifying perturbations
    - Adaptive step sizes
    - Symmetry exploitation
    """
    current = a_init.astype(np.float64).copy()
    current_score, current_conv = score_and_conv(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.001
    cooling = 0.9999
    same_count = 0
    max_same = 5000
    
    while time.monotonic() - t0 < time_limit:
        r = rng.random()
        candidate = current.copy()
        
        if r < 0.4:
            # Targeted peak reduction: find peak position, reduce contributing elements
            peak_idx = int(np.argmax(current_conv))
            # Elements contributing to peak: a[i]*a[peak_idx-i] for valid i
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            # Reduce top contributors
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            # Scale down top contributors
            scale = np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            candidate[top_idx] *= scale
            # Compensate by scaling up others to maintain sum
            rest = np.setdiff1d(np.arange(n), top_idx)
            if len(rest) > 0:
                sum_before = np.sum(candidate[top_idx])
                sum_after_target = sum_before * 0.5  # reduce by 50%
                candidate[top_idx] = candidate[top_idx] * scale
                adjustment = (sum_before - np.sum(candidate[top_idx])) / max(np.sum(candidate[rest]), 1e-10)
                candidate[rest] *= (1.0 + adjustment)
        elif r < 0.6:
            # Anti-peak: increase elements that are at the "edges" of the sequence
            # (to spread out the convolution)
            edge_size = max(2, n // 8)
            edge_idx = list(range(edge_size)) + list(range(n - edge_size, n))
            n_pick = min(4, len(edge_idx))
            pick = rng.choice(edge_idx, size=n_pick, replace=False)
            candidate[pick] *= np.exp(rng.normal(0.02, 0.01, size=n_pick))
        elif r < 0.75:
            # Block perturbation
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.85:
            # Symmetric perturbation
            k = int(rng.integers(1, n // 2))
            delta = rng.normal(0, 0.02)
            candidate[k] *= np.exp(delta)
            candidate[n - 1 - k] *= np.exp(delta)
            # Compensate elsewhere
            comp_idx = int(rng.integers(0, n))
            candidate[comp_idx] *= np.exp(-delta * current[k] / max(current[comp_idx], 0.01))
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.008, size=n))
        
        candidate = np.clip(candidate, 1e-6, 1000.0)
        new_score, new_conv = score_and_conv(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            current_conv = new_conv
            same_count = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score
            current_conv = new_conv
        else:
            same_count += 1
        
        # Restart if stuck
        if same_count > max_same:
            # Random restart around current best
            current = best * np.exp(rng.normal(0, 0.05, size=n))
            current_score, current_conv = score_and_conv(current)
            temperature = 0.002
            same_count = 0
        
        temperature *= cooling
    
    return best, best_score


def make_structured_init(n, rng, style='golden'):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if style == 'golden':
        # Golden ratio based frequency
        phi = (1 + np.sqrt(5)) / 2
        freq = 1.0 / phi
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
    elif style == 'chirp':
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (2 * n))
    elif style == 'multi':
        a = np.ones(n)
        n_tones = rng.integers(2, 6)
        for _ in range(n_tones):
            f = rng.uniform(0.5, n / 4)
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif style == 'bilinear':
        # Linear ramp up then down
        a = 1.0 + 0.3 * np.sin(np.pi * t / (n - 1))
    else:
        a = 1.0 + 0.2 * rng.standard_normal(n)
    
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for optimal non-negative step function coefficients.
    Phase 1: Quick evaluation of structured candidates
    Phase 2: Gradient optimization with annealed log-sum-exp
    Phase 3: Local search polish
    Returns best sequence found.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Evaluate provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s, _ = score_and_conv(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()
    
    # Quick check: uniform sequences
    for n_test in [128, 256, 512]:
        a_uniform = np.ones(n_test)
        s, _ = score_and_conv(a_uniform)
        print(f"Uniform n={n_test}: score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = a_uniform.copy()
    
    # Phase 2: Gradient optimization
    print("\n=== Phase 2: Gradient optimization ===")
    
    # Try multiple configurations
    configs = [
        # (n, init_type, time_fraction)
        (256, 'provided', 0.25),
        (256, 'chirp', 0.15),
        (256, 'multi', 0.15),
        (256, 'golden', 0.15),
        (512, 'chirp', 0.10),
        (512, 'multi', 0.10),
        (128, 'chirp', 0.05),
        (128, 'multi', 0.05),
    ]
    
    for n, init_type, time_frac in configs:
        if time.monotonic() > deadline - 60:
            break
        time_left = deadline - time.monotonic()
        t_per = min(time_frac * time_budget, time_left * 0.5)
        if t_per < 5:
            continue
        
        # Generate initial sequence
        if init_type == 'provided' and start_seq is not None:
            if len(start_seq) == n:
                a0 = start_seq.copy()
            else:
                x_old = np.linspace(0, 1, len(start_seq))
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, start_seq)
        else:
            a0 = make_structured_init(n, rng, style=init_type)
        
        a0 = np.clip(a0, 0.01, 1000.0)
        theta_init = np.log(a0 / np.sum(a0) + 1e-10)
        
        print(f"  Starting optimization: n={n}, init='{init_type}', time={t_per:.1f}s")
        t0 = time.monotonic()
        theta_best, grad_score = optimize_gradient_v2(n, theta_init, t_per, rng)
        elapsed = time.monotonic() - t0
        
        a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n)
        actual_score, _ = score_and_conv(a_best)
        print(f"    n={n}, init='{init_type}', score={actual_score:.6f}, time={elapsed:.1f}s")
        
        if actual_score < best_score:
            best_score = actual_score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 3: Local search polish
    print(f"\n=== Phase 3: Local search polish ===")
    if best_seq is not None:
        polish_time = min(300.0, max(20.0, deadline - time.monotonic()))
        print(f"  Polishing, score={best_score:.6f}, n={len(best_seq)}, time={polish_time:.1f}s")
        polished, polish_score = local_search_v2(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
    
    # Phase 4: If time remains, try more local search from different starts
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print(f"\n=== Phase 4: Additional refinement ===")
        remaining = deadline - time.monotonic()
        # Try perturbing the best sequence and re-optimizing
        n_best = len(best_seq)
        for attempt in range(3):
            if time.monotonic() > deadline - 15:
                break
            perturbed = best_seq * np.exp(rng.normal(0, 0.1, size=n_best))
            perturbed = np.clip(perturbed, 0.01, 1000.0)
            s, _ = score_and_conv(perturbed)
            print(f"  Perturbed attempt {attempt+1}: score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = perturbed.copy()
            # Quick local search
            t_ls = min(30.0, remaining / 2.0)
            if t_ls > 5:
                pol2, s2 = local_search_v2(perturbed, t_ls, rng)
                if s2 < best_score:
                    best_score = s2
                    best_seq = pol2.copy()
                    print(f"    After mini-polish: {best_score:.6f}")
    
    # Final validation
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
