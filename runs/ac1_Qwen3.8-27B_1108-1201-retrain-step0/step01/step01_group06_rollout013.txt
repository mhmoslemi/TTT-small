Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max mirror descent

import numpy as np
import time
import scipy.fft as fft

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft.rfft(a, fft_size)
    result = fft.irfft(A * A, fft_size)
    return result[:size]

def fast_cross_corr(w, a):
    """
    Compute cross-correlation: out[j] = sum_k w[k] * a[k-j]
    where a[m] = 0 for m outside [0, n-1].
    w has length 2n-1, a has length n.
    Returns array of length n.
    """
    n = len(a)
    L = 1
    while L < len(w) + n - 1:
        L *= 2
    W = fft.fft(w, L)
    A = fft.fft(a, L)
    R = fft.ifft(W * np.conj(A)).real
    return R[:n]

def smooth_objective_and_gradient(a, t):
    """
    Compute smooth approximation of 2n*max(conv(a,a)) and its gradient.
    a should be non-negative with sum 1.
    """
    n = len(a)
    c = fast_convolve(a)
    scaled = t * c
    max_c = c.max()
    shifted = scaled - t * max_c
    log_sum_exp = np.log(np.exp(shifted).sum() + 1e-300)
    obj = 2.0 * n * (max_c + log_sum_exp / t)
    
    # Weights
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # Gradient: 4n * cross_corr(w, a)
    corr = fast_cross_corr(w, a)
    grad = 4.0 * n * corr
    return obj, grad

def solve_direction_lp(f, M, n):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (unnormalized).

                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
