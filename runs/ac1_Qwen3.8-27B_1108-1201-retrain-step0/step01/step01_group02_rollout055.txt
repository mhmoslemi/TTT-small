Gradient-based optimization with FFT acceleration for minimizing peak auto-convolution.
Uses Adam optimizer, top-m peak gradients, and multi-length search.
"""

import numpy as np
import time

def score_sequence(a):
    """Compute the score for a sequence."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

    """Fast auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    size = len(conv)
    # Get top m positions

    else:
        top_idx = np.argsort(conv)[-top_m:][::-1]
    
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            indices = np.arange(lo, hi)
            grad[lo:hi] += a[k - indices]
    
    return grad

def adam_update(params, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step (in-place)."""
    m[:] = beta1 * m + (1 - beta1) * grad
    v[:] = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    params[:] -= lr * m_hat / (np.sqrt(v_hat) + eps)

        print(f"      LS done: best={best_score:.6f}, {n_accept} accepts, {time.monotonic()-start:.1f}s")

"""
FFT-accelerated gradient descent with Adam for minimizing peak auto-convolution.
Uses top-m peak subgradients, multi-length search, multimodal initializations,
and local search polishing. Much faster per-iteration than LP-based methods.
"""

import numpy as np
import time


def score_sequence(a):
    """Compute the evaluation score for sequence a."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def fft_convolve(a):
    """Fast auto-convolution using FFT. Returns full convolution of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient(a, conv, top_m=5):
    """Compute subgradient of max(conv) w.r.t. a using top-m peaks.
    
    For each peak position k: d(conv[k])/d(a[j]) = 2*a[k-j] for valid j.
    We sum over the top-m peaks for a smoothed gradient.
    """
    n = len(a)
    size = len(conv)
    if top_m >= size:
        top_idx = np.argsort(conv)[::-1]
    else:
        top_idx = np.argsort(conv)[-top_m:][::-1]
    
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            indices = np.arange(lo, hi)
            grad[lo:hi] += a[k - indices]
    
    return grad


def normalize_simplex(a):
    """Project to non-negative simplex (sum = 1)."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_double_bump(n, gap_frac=0.2, rng=None):
    """Two Gaussians separated by a gap."""
    t = np.arange(n, dtype=np.float64)
    center1 = n * (0.5 - gap_frac / 2.0)
    center2 = n * (0.5 + gap_frac / 2.0)
    sigma = n * 0.12
    a = np.exp(-0.5 * ((t - center1) / sigma) ** 2) + np.exp(-0.5 * ((t - center2) / sigma) ** 2)
    return a


def initial_triple_bump(n, rng=None):
    """Three Gaussians."""
    t = np.arange(n, dtype=np.float64)
    centers = [n * 0.25, n * 0.5, n * 0.75]
    sigma = n * 0.1
    a = np.zeros(n)
    for c in centers:
        a += np.exp(-0.5 * ((t - c) / sigma) ** 2)
    return a


def gradient_descent(a_init, time_limit, rng, verbose=True, top_m_start=10, top_m_end=3):
    """
    Adam-based gradient descent to minimize 2*n*max(conv(a,a)) subject to a>=0, sum(a)=1.
    Returns (best_a, best_score) where best_score = 2*n*max(conv) for sum-1 normalized a.
    """
    start_time = time.monotonic()
    a = normalize_simplex(a_init.copy().astype(np.float64))
    n = len(a)
    
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()
    
    if verbose:
        print(f"        GD start: score={best_score:.6f}, n={n}")
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    patience = 3000
    stagnant = 0
    step = 0
    max_steps = 500000
    
    while time.monotonic() - start_time < time_limit and step < max_steps:
        progress = min(1.0, step / 10000.0)
        top_m = max(top_m_end, int(top_m_start * (1 - progress) + top_m_end * progress))
        
        conv = fft_convolve(a)
        current_score = 2.0 * n * conv.max()
        
        if current_score < best_score - 1e-13:
            best_score = current_score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant >= patience:
            # Perturb and restart
            pert_scale = 0.05 + 0.1 * rng.random()
            a = best_a + pert_scale * rng.normal(0, 1, size=n)
            a = normalize_simplex(a)
            m[:] = 0
            v[:] = 0
            t = 0
            stagnant = 0
            patience = max(500, int(patience * 0.6))
            step += 1
            continue
        
        # Learning rate: start aggressive, decay
        lr = 0.04 * (1.0 + 0.01 * t) ** (-0.4)
        
        # Compute gradient
        grad = compute_gradient(a, conv, top_m=top_m)
        
        # Tiny regularization towards uniform to prevent collapse
        grad += 0.0003 * (a - 1.0 / n)
        
        # Adam update
        t += 1
        m[:] = 0.9 * m + 0.1 * grad
        v[:] = 0.999 * v + 0.001 * grad ** 2
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        a -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Project to simplex
        a = normalize_simplex(a)
        
        step += 1
        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"        Step {step:6d}: cur={current_score:.6f}, best={best_score:.6f}, lr={lr:.5f}, {elapsed:.1f}s")
    
    # Final evaluation
    final_score = 2.0 * n * fft_convolve(a).max()
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"        GD done: best={best_score:.6f} after {step} steps, {time.monotonic()-start_time:.1f}s")
    
    return best_a, best_score


def local_search(a_init, time_limit, rng, verbose=True):
    """Segment-based local search with geometric mutations."""
    start_time = time.monotonic()
    a = normalize_simplex(a_init.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()
    
    n_accept = 0
    noise_scale = 0.15
    
    while time.monotonic() - start_time < time_limit:
        # Choose mutation width (log-uniform)
        min_w = max(2, n // 100)
        max_w = max(min_w + 1, n // 3)
        width = int(np.exp(rng.uniform(np.log(min_w), np.log(max_w))))
        pos = int(rng.integers(0, n - width))
        
        candidate = a.copy()
        
        # Geometric noise on the segment
        candidate[pos:pos + width] *= np.exp(rng.normal(0, noise_scale, size=width))
        
        # Occasionally apply global smooth perturbation
        if rng.random() < 0.1:
            candidate *= np.exp(rng.normal(0, 0.05, size=n))
        
        # Occasionally enforce symmetry
        if rng.random() < 0.25:
            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = normalize_simplex(candidate)
        score = 2.0 * n * fft_convolve(candidate).max()
        
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            n_accept += 1
            noise_scale = min(0.3, noise_scale * 1.02)
        else:
            noise_scale = max(0.05, noise_scale * 0.999)
    
    if verbose:
        print(f"        LS done: best={best_score:.6f}, {n_accept} accepts, {time.monotonic()-start_time:.1f}s")
    
    return best_a, best_score


def propose_candidate(budget_s=2400):
    """
    Main search: FFT-accelerated gradient descent + local search across multiple
    lengths and initializations. Target: minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    try:
        start_seq = np.array(height_sequence_1, dtype=np.float64)
        start_n = len(start_seq)
        start_score = evaluate_sequence([float(x) for x in start_seq])
        print(f"Given starting sequence: n={start_n}, score={start_score:.6f}")
    except Exception:
        start_seq = None
        start_n = 0
        start_score = float('inf')
    
    rng = np.random.default_rng(42)
    total_start = time.monotonic()
    deadline = total_start + max(60.0, budget_s - 30.0)
    
    best_overall_score = start_score
    best_overall_seq = start_seq.copy() if start_seq is not None else None
    
    # Define search configurations: (n, init_func, time_fraction)
    # Prioritize n values near 500 and larger n for better theoretical bounds
    def make_configs():
        configs = []
        # Near the known best n=500
        for n in [400, 450, 500, 550, 600]:
            configs.append((n, 'raised_cosine', 0.06))
            configs.append((n, 'gaussian', 0.04))
        # Larger n for lower theoretical bound
        for n in [1000, 1500, 2000, 3000, 5000]:
            configs.append((n, 'raised_cosine', 0.05))
            configs.append((n, 'gaussian', 0.03))
        # Very large n
        configs.append((8000, 'raised_cosine', 0.04))
        configs.append((8000, 'gaussian', 0.03))
        configs.append((12000, 'raised_cosine', 0.03))
        # Multimodal
        configs.append((500, 'double_bump', 0.03))
        configs.append((1000, 'double_bump', 0.03))
        configs.append((1000, 'triple_bump', 0.02))
        configs.append((2000, 'double_bump', 0.02))
        # Blackman
        configs.append((1000, 'blackman', 0.03))
        configs.append((2000, 'blackman', 0.02))
        return configs
    
    configs = make_configs()
    
    # First pass: quick screening to find promising (n, init) combinations
    print("\n--- Phase 1: Quick screening ---")
    screen_results = []
    for n, init_type, frac in configs:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        # Generate initial
        if init_type == 'raised_cosine':
            a0 = initial_raised_cosine(n)
        elif init_type == 'gaussian':
            a0 = initial_gaussian(n, sigma_frac=0.3)
        elif init_type == 'blackman':
            a0 = initial_blackman(n)
        elif init_type == 'double_bump':
            a0 = initial_double_bump(n, rng=rng)
        elif init_type == 'triple_bump':
            a0 = initial_triple_bump(n, rng=rng)
        else:
            a0 = initial_raised_cosine(n)
        
        a0 = normalize_simplex(a0)
        init_score = 2.0 * n * fft_convolve(a0).max()
        screen_results.append((init_score, n, init_type, a0))
    
    # Sort by initial score and keep top candidates
    screen_results.sort(key=lambda x: x[0])
    print(f"  Top 10 screen results:")
    for i, (sc, n, it, _) in enumerate(screen_results[:10]):
        print(f"    {i+1}. n={n:5d}, init={it:15s}, score={sc:.6f}")
    
    # Second pass: detailed optimization on top candidates
    print("\n--- Phase 2: Detailed optimization ---")
    # Allocate time to top candidates
    top_candidates = screen_results[:12]
    n_candidates = len(top_candidates)
    time_per_candidate = (deadline - total_start) * 0.7 / n_candidates
    
    for i, (init_score, n, init_type, a0) in enumerate(top_candidates):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        t_alloc = min(time_per_candidate, remaining * 0.95)
        if t_alloc < 15:
            continue
        
        print(f"\n  [{i+1}/{n_candidates}] n={n}, init={init_type}, init_score={init_score:.6f}, t={t_alloc:.0f}s")
        
        if init_score < best_overall_score:
            best_overall_score = init_score
            best_overall_seq = a0.copy()
        
        # Gradient descent (70% of time)
        t_gd = t_alloc * 0.65
        a_gd, gd_score = gradient_descent(a0, t_gd, rng, verbose=(i < 4))
        
        if gd_score < best_overall_score:
            best_overall_score = gd_score
            best_overall_seq = a_gd.copy()
            print(f"    *** NEW BEST after GD: {gd_score:.6f} (n={n})")
        
        # Local search (30% of time)
        remaining = deadline - time.monotonic()
        if remaining > 10:
            t_ls = min(t_alloc * 0.35, remaining * 0.5)
            if t_ls > 8:
                a_ls, ls_score = local_search(a_gd, t_ls, rng, verbose=False)
                if ls_score < best_overall_score:
                    best_overall_score = ls_score
                    best_overall_seq = a_ls.copy()
                    print(f"    *** NEW BEST after LS: {ls_score:.6f} (n={n})")
        
        # If we found the best, also try GD on it from a slightly different start
        remaining = deadline - time.monotonic()
        if remaining > 20 and n == len(best_overall_seq):
            # Try a different perturbation
            a_pert = best_overall_seq + 0.1 * rng.normal(0, 1, size=n)
            a_pert = normalize_simplex(a_pert)
            t_gd2 = min(remaining * 0.4, 60.0)
            if t_gd2 > 10:
                a_gd2, gd2_score = gradient_descent(a_pert, t_gd2, rng, verbose=False)
                if gd2_score < best_overall_score:
                    best_overall_score = gd2_score
                    best_overall_seq = a_gd2.copy()
                    print(f"    *** NEW BEST after GD2: {gd2_score:.6f} (n={n})")
    
    # Phase 3: Final intensive polishing on the best sequence
    print("\n--- Phase 3: Final polishing ---")
    if best_overall_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_best = len(best_overall_seq)
            print(f"  Polishing n={n_best}, best_score={best_overall_score:.6f}, remaining={remaining:.0f}s")
            
            # Multiple GD rounds with different perturbations
            t_per_round = remaining / 4.0
            for round_i in range(3):
                remaining = deadline - time.monotonic()
                if remaining < 20:
                    break
                
                if round_i == 0:
                    a_round = best_overall_seq.copy()
                else:
                    a_round = best_overall_seq + 0.05 * rng.normal(0, 1, size=n_best)
                    a_round = normalize_simplex(a_round)
                
                a_pol, pol_score = gradient_descent(a_round, t_per_round * 0.7, rng, verbose=(round_i == 0))
                if pol_score < best_overall_score:
                    best_overall_score = pol_score
                    best_overall_seq = a_pol.copy()
                    print(f"    Round {round_i}: NEW BEST {pol_score:.6f}")
            
            # Final local search
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_ls, ls_score = local_search(best_overall_seq, remaining * 0.7, rng, verbose=True)
                if ls_score < best_overall_score:
                    best_overall_score = ls_score
                    best_overall_seq = a_ls.copy()
                    print(f"    Final LS: NEW BEST {ls_score:.6f}")
    
    # Fallback
    if best_overall_seq is None:
        best_overall_seq = initial_raised_cosine(500)
        best_overall_score = score_sequence(best_overall_seq)
    
    result = [float(x) for x in best_overall_seq]
    verify = evaluate_sequence(result)
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_overall_seq)}, score={verify:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")
    print(f"{'='*60}")
    
    return result
