    c = np.convolve(a, a)

    n = a.size

def compute_grad_sub(a, top_k=5):
    """Subgradient using top-k convolution coefficients for robustness."""
    c = np.convolve(a, a)
    c_sorted_idx = np.argpartition(c, -top_k)[-top_k:]
    q = np.zeros_like(c)
    top_vals = c[c_sorted_idx]
    if top_vals.max() > 0:
        weights = np.exp(top_vals / (top_vals.max() * 0.1 + 1e-12))
        weights = weights / weights.sum()

            q[idx] = weights[i]
    grad = 2.0 * np.correlate(q, a, 'valid')
    return grad, c
