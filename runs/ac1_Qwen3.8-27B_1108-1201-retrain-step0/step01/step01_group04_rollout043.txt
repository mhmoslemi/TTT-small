    if n<=0: return np.ones(0)

def _fft_len(n):
    return int(next_fast_len(max(1, n)))


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    if p.ndim != 1:
        p = p.ravel()
    if p.size == 0:
        return np.ones(1, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(p.size, 1.0 / p.size, dtype=np.float64)
    return p / s


def conv_sq(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F)[:2 * n - 1])
    c = np.where(np.isfinite(c), c, 0.0)
    return np.maximum(c, 0.0)


def internal_score(p):
    p = safe_normalize(p)
    n = int(p.size)
    if n == 0:
        return np.inf
    c = conv_sq(p)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m)


def hankel_vec(p, w):
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(3 * n - 2)
    fp = fft(p[::-1], L)
    fw = fft(w, L)
    r = np.real(ifft(fp * fw))
    s = r[n - 1:n - 1 + n]
    s = np.where(np.isfinite(s), s, 0.0)
    return np.maximum(s, 0.0)

def resample_seq(p_old, n):

def make_uniform(n):

    n = int(max(2, n))

def make_power(n, a, reverse=False):
    n = int(max(2, n))
    x = np.linspace(1e-5, 1.0, n)
    p = np.power(np.maximum(x, 1e-8), float(a))

def make_geometric(n, s, reverse=False):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-float(s) * x)
    if reverse:
        p = p[::-1]
    return safe_normalize(p)


def make_cup(n, a):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + float(a) * (2.0 * x - 1.0) ** 2
    return safe_normalize(p)


def make_cap(n, a):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 - float(a) * (2.0 * x - 1.0) ** 2
    return safe_normalize(p)


def make_beta(n, alpha, beta):
    n = int(max(2, n))
    x = np.linspace(1e-5, 1.0 - 1e-5, n)
    p = np.power(np.maximum(x, 1e-8), float(alpha)) * np.power(np.maximum(1.0 - x, 1e-8), float(beta))
    return safe_normalize(p)


def make_skew_gauss(n, a, mu, b):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    dx = x - float(mu)
    p = np.exp(-float(a) * (dx * dx) + float(b) * dx)
    return safe_normalize(p)


def make_two_gauss(n, mu1, sig1, w1, mu2, sig2, w2):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    a = float(w1) * np.exp(-((x - float(mu1)) ** 2) / (2.0 * max(1e-4, float(sig1)) ** 2))
    b = float(w2) * np.exp(-((x - float(mu2)) ** 2) / (2.0 * max(1e-4, float(sig2)) ** 2))
    return safe_normalize(a + b)


def make_cosine_asym(n, a, b, c):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + float(a) * np.cos(np.pi * x) + float(b) * np.cos(2.0 * np.pi * x) + float(c) * np.cos(3.0 * np.pi * x)
    return safe_normalize(p)


def make_random_smooth(n, rng, strength=1.0):
    n = int(max(2, n))

    return safe_normalize(p)

# ---------------- low-dimensional shape search ----------------

def shape_from_theta(theta):
    n = int(SHAPE_N)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    x = SHAPE_X
    if x is None or x.size != n:
        x = np.linspace(0.0, 1.0, n)

    D = int(SHAPE_DEG)
    theta = np.asarray(theta, dtype=np.float64)
    need = D + 2
    if theta.size < need:
        theta = np.concatenate((theta, np.zeros(need - int(theta.size), dtype=np.float64)))
    else:
        theta = theta[:need].copy()

    logp = np.zeros(n, dtype=np.float64)
    z = 2.0 * x - 1.0

    if D >= 1:
        T_nm = np.ones(n, dtype=np.float64)
        T_n = z.copy()
        logp += theta[0] * T_n
        for d in range(2, D + 1):
            T_np = 2.0 * z * T_n - T_nm
            logp += theta[d - 1] * T_np
            T_nm = T_n
            T_n = T_np

    eps = 1e-5
    logp += theta[D] * np.log(x + eps) + theta[D + 1] * np.log(1.0 - x + eps)

    if not np.all(np.isfinite(logp)):
        return np.full(n, 1.0 / n, dtype=np.float64)

    m = float(np.max(logp))
    if not np.isfinite(m):
        return np.full(n, 1.0 / n, dtype=np.float64)
    p = np.exp(logp - m)
    return safe_normalize(p)


def shape_objective(theta):
    try:
        p = shape_from_theta(theta)
        sc = internal_score(p)
        if not np.isfinite(sc):
            return 1e100
        return float(sc)
    except Exception:
        return 1e100


def run_shape_search(n, rng, budget_s):
    global SHAPE_N, SHAPE_X
    n = int(max(64, n))
    SHAPE_N = n
    SHAPE_X = np.linspace(0.0, 1.0, n)
    dim = int(SHAPE_DEG) + 2

    best_theta = None
    best_p = None
    best_score = np.inf

    def upd(theta_candidate):
        nonlocal best_theta, best_p, best_score
        try:
            sc = shape_objective(theta_candidate)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_theta = np.asarray(theta_candidate, dtype=np.float64).copy()
                best_p = shape_from_theta(best_theta)
        except Exception:
            pass

    starts = []
    starts.append(np.zeros(dim, dtype=np.float64))
    starts.append(rng.uniform(-1.2, 1.2, dim).astype(np.float64))
    starts.append(rng.uniform(-1.2, 1.2, dim).astype(np.float64))

    t = np.zeros(dim, dtype=np.float64)
    t[SHAPE_DEG] = 2.0
    t[SHAPE_DEG + 1] = 2.0
    starts.append(t)

    t = np.zeros(dim, dtype=np.float64)
    t[0] = 1.6
    t[1] = -0.7
    starts.append(t)

    t = np.zeros(dim, dtype=np.float64)
    for i in range(SHAPE_DEG):
        t[i] = 0.8 * (1 if i % 2 == 0 else -1)
    t[SHAPE_DEG] = 0.5
    starts.append(t)

    for x0 in starts:
        upd(x0)

    if _HAVE_DE and budget_s > 15.0:
        bounds = [(-2.5, 2.5) for _ in range(dim)]
        maxiter = max(8, min(35, int(budget_s / 4)))
        try:
            res = differential_evolution(
                shape_objective,
                bounds,
                maxiter=maxiter,
                popsize=12,
                tol=0.25,
                seed=int(rng.integers(0, 2**31 - 1)),
                polish=False,
                init="latinhypercube",
            )
            if res is not None and np.isfinite(getattr(res, "fun", np.inf)):
                upd(res.x)
                if getattr(res, "x", None) is not None:
                    starts.append(np.asarray(res.x, dtype=np.float64))
        except Exception:
            try:
                res = differential_evolution(
                    shape_objective,
                    bounds,
                    maxiter=max(6, maxiter // 2),
                    popsize=10,
                    seed=int(rng.integers(0, 2**31 - 1)),
                    polish=False,
                )
                if res is not None and np.isfinite(getattr(res, "fun", np.inf)):
                    upd(res.x)
                    if getattr(res, "x", None) is not None:
                        starts.append(np.asarray(res.x, dtype=np.float64))
            except Exception:
                pass

    if _HAVE_MIN:
        for x0 in starts:
            try:
                res = minimize(
                    shape_objective,
                    np.asarray(x0, dtype=np.float64),
                    method="Nelder-Mead",
                    options={"maxiter": 220, "xatol": 1e-4, "fatol": 1e-6},
                )
                if res is not None and np.isfinite(getattr(res, "fun", np.inf)):
                    upd(res.x)
            except Exception:
                pass

    if best_p is None:
        best_p = shape_from_theta(np.zeros(dim, dtype=np.float64))
        best_score = internal_score(best_p)
    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


# ---------------- L-BFGS objective in softmax coordinates ----------------

def p_from_y(y, idx, ref, n):
    n = int(max(1, n))
    y = np.asarray(y, dtype=np.float64)
    if y.size != max(0, n - 1):
        return np.full(n, 1.0 / n, dtype=np.float64)
    if not np.all(np.isfinite(y)):
        y = np.nan_to_num(y, nan=-50.0, posinf=50.0, neginf=-50.0)
    m = 0.0
    if y.size > 0:
        m = float(np.max(y))
        if m < 0.0:
            m = 0.0
    exps = np.exp(np.clip(y - m, -745.0, 709.0))
    ref_term = np.exp(-m)
    denom = float(ref_term + float(exps.sum()))
    p = np.zeros(n, dtype=np.float64)
    if not np.isfinite(denom) or denom <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    p[ref] = ref_term / denom
    if idx is not None and len(idx) == len(exps):
        p[idx] = exps / denom
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def _compute_obj_grad(y):
    y = np.asarray(y, dtype=np.float64)
    n = int(OBJ_N)
    idx = OBJ_IDX
    ref = int(OBJ_REF)
    T = float(OBJ_T)

    if n <= 0 or idx is None:
        return 1e100, np.zeros(0, dtype=np.float64), None, 0.0
    if y.size != n - 1:
        y = np.zeros(n - 1, dtype=np.float64)
    if not np.all(np.isfinite(y)):
        return 1e100, np.zeros(y.size, dtype=np.float64), None, 0.0

    p = p_from_y(y, idx, ref, n)
    c = conv_sq(p)
    if c.size == 0:
        return 1e100, np.zeros(y.size, dtype=np.float64), p, 0.0
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return 1e100, np.zeros(y.size, dtype=np.float64), p, 0.0

    if not np.isfinite(T) or T <= 0.0:
        T = 0.01 * cmax
        if T <= 0.0:
            T = 1e-12

    diff = (c - cmax) / T
    diff = np.clip(diff, -745.0, 709.0)
    e = np.exp(diff)
    den = float(e.sum())
    if not np.isfinite(den) or den <= 0.0:
        w = np.zeros_like(c, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / den

    surr = 2.0 * n * (cmax + T * np.log(den + 1e-300))

    s = hankel_vec(p, w)
    gp = 4.0 * n * s
    gp = np.where(np.isfinite(gp), gp, 0.0)
    dot = float(np.dot(p, gp))
    grad = p[idx] * (gp[idx] - dot)
    grad = np.where(np.isfinite(grad), grad, 0.0)

    return float(surr), grad, p, cmax


def _obj_value(y):
    global CACHE_Y, CACHE_F, CACHE_G
    y_arr = np.asarray(y, dtype=np.float64)
    if CACHE_Y is None or not np.array_equal(CACHE_Y, y_arr):
        f, g, _, _ = _compute_obj_grad(y_arr)
        CACHE_Y = y_arr.copy()
        CACHE_F = f
        CACHE_G = g
    return CACHE_F


def _obj_grad(y):
    global CACHE_Y, CACHE_F, CACHE_G
    y_arr = np.asarray(y, dtype=np.float64)
    if CACHE_Y is None or not np.array_equal(CACHE_Y, y_arr):
        f, g, _, _ = _compute_obj_grad(y_arr)
        CACHE_Y = y_arr.copy()
        CACHE_F = f
        CACHE_G = g
    return CACHE_G


def _lbfgs_callback(pk):
    return time.monotonic() > OBJ_DEADLINE


def lp_best_response(p, best_score, rng, budget_s):
    if not _HAVE_LP:
        return p, best_score
    p = safe_normalize(p)
    n = int(p.size)
    if n <= 0:
        return p, best_score
    budget_s = float(max(0.0, budget_s))
    if budget_s < 1.5:
        return p, best_score

    t0 = time.monotonic()
    if time.monotonic() - t0 > budget_s:
        return p, best_score

    c = conv_sq(p)
    if c.size == 0:
        return p, best_score
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return p, best_score

    f = p + 1e-10
    fs = float(f.sum())
    if not np.isfinite(fs) or fs <= 0.0:
        return p, best_score
    f = f / fs
    Mf = float(np.max(conv_sq(f)))
    M2 = max(M, Mf) * (1.0 + 1e-9) + 1e-15

    use_full = (n <= FULL_LP_NMAX)
    if use_full:
        active_rows = np.arange(2 * n - 1, dtype=np.int32)
    else:
        K = max(32, min(MAX_LP_ACTIVE, int(n // 18)))
        K = min(K, int(c.size))
        if K <= 0:
            return p, best_score
        active_rows = np.argpartition(c, -K)[-K:]
        active_rows = np.asarray(active_rows, dtype=np.int32)

    try:
        if use_full:
            i = np.arange(n, dtype=np.int32)
            j = np.arange(n, dtype=np.int32)
            rows = (i[:, None] + j[None, :]).ravel()
            cols = np.tile(j, n)
            data = np.tile(f.astype(np.float64, copy=False), n)
        else:
            row_parts = []
            col_parts = []
            data_parts = []
            for k in active_rows:
                k = int(k)
                j0 = max(0, k - (n - 1))
                j1 = min(n - 1, k)
                if j1 >= j0:
                    js = np.arange(j0, j1 + 1, dtype=np.int32)
                    row_parts.append(np.full(js.size, k, dtype=np.int32))
                    col_parts.append(js)
                    data_parts.append(f[j0:j1 + 1].astype(np.float64, copy=False))
            if not row_parts:
                return p, best_score
            rows = np.concatenate(row_parts)
            cols = np.concatenate(col_parts)
            data = np.concatenate(data_parts)

        A = csr_matrix((data, (rows, cols)), shape=(len(active_rows), n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(len(active_rows), M2, dtype=np.float64)

        opts = {"disp": False, "presolve": True}
        tl = max(0.5, min(25.0, (budget_s - (time.monotonic() - t0)) * 0.55))
        if tl > 0.2:
            opts["time_limit"] = tl

        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs", options=opts)
    except Exception:
        return p, best_score

    if res is None:
        return p, best_score
    g = getattr(res, "x", None)
    if g is None:
        return p, best_score
    g = np.asarray(g, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return p, best_score
    g = g / s

    p_best = p
    sc_best = float(best_score)
    for t in (1.0, 0.8, 0.6, 0.4, 0.25, 0.1):
        q = safe_normalize((1.0 - t) * p + t * g)
        try:
            sc = internal_score(q)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < sc_best:
            sc_best = float(sc)
            p_best = q
    return p_best, sc_best


def coordinate_polish(p, rng, time_limit):
    p = safe_normalize(p)
    n = int(p.size)
    best = p.copy()
    best_sc = internal_score(best)
    end = time.monotonic() + max(0.0, float(time_limit))
    no_improve = 0

    while time.monotonic() < end - 1.0 and no_improve < 120:
        c = conv_sq(best)
        if c.size == 0:
            break
        K = max(4, min(c.size, n // 40))
        if K < c.size:
            top = np.argpartition(c, -K)[-K:]
        else:
            top = np.arange(c.size)
        w = np.zeros_like(c, dtype=np.float64)
        w[top] = 1.0 / max(1, len(top))
        s = hankel_vec(best, w)
        v = 2.0 * s
        v = np.where(np.isfinite(v), v, 0.0)

        hi = np.argsort(v)[-8:]
        lo = np.argsort(v)[:8]
        improved = False

        for i in hi:
            for j in lo:
                if i == j:
                    continue
                for frac in (0.25, 0.05, 0.01):
                    delta = float(frac * best[i])
                    if delta <= 0.0:
                        continue
                    q = best.copy()
                    q[i] -= delta
                    q[j] += delta
                    q = safe_normalize(q)
                    sc = internal_score(q)
                    if np.isfinite(sc) and sc < best_sc:
                        best = q
                        best_sc = float(sc)
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            no_improve = 0
            continue

        if n > 1:
            if rng.random() < 0.7:
                i = int(hi[0])
                j = int(lo[0])
            else:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
            if i != j:
                for frac in (0.1, 0.02, 0.005):
                    delta = float(frac * best[i])
                    if delta <= 0.0:
                        continue
                    q = best.copy()
                    q[i] -= delta
                    q[j] += delta
                    q = safe_normalize(q)
                    sc = internal_score(q)
                    if np.isfinite(sc) and sc < best_sc:
                        best = q
                        best_sc = float(sc)
                        no_improve = 0
                        break

        if time.monotonic() >= end - 1.0:
            break

        if no_improve > 0:
            q = safe_normalize(np.maximum(best + rng.normal(0.0, 0.05 / n, n), 0.0))
            sc = internal_score(q)
            if np.isfinite(sc) and sc < best_sc:
                best = q
                best_sc = float(sc)
                no_improve = 0
            else:
                no_improve += 1

    return best, float(best_sc) if np.isfinite(best_sc) else np.inf


def lbfgs_polish(p0, time_limit, label="", allow_lp=True, rng=None, quick=False):
    global OBJ_N, OBJ_IDX, OBJ_REF, OBJ_T, OBJ_DEADLINE, CACHE_Y, CACHE_F, CACHE_G

    p = safe_normalize(p0)
    n = int(p.size)
    if n <= 1:
        return p, internal_score(p)
    if rng is None:
        rng = np.random.default_rng(12345)

    best_p = p.copy()
    best_score = internal_score(p)
    end = time.monotonic() + max(0.0, float(time_limit))

    if quick:
        factors = [0.30, 0.15, 0.075]
    else:
        factors = [0.30, 0.18, 0.09, 0.045, 0.022, 0.011, 0.005]

    no_improve = 0

    for stage, fac in enumerate(factors):
        if time.monotonic() >= end - 8.0:
            break
        if no_improve > 2:
            fac = max(fac, 0.09)

        c0 = conv_sq(p)
        cmax0 = float(np.max(c0)) if c0.size else 0.0
        T = float(fac * max(cmax0, 1e-12))
        if not np.isfinite(T) or T <= 0.0:
            T = 1e-9

        ref = int(np.argmax(p))
        idx = np.delete(np.arange(n), ref)
        y0 = np.log(np.maximum(p[idx] / max(p[ref], 1e-300), 1e-15))
        y0 = np.clip(y0, -50.0, 50.0)

        if quick:
            stage_budget = min(35.0, max(8.0, (end - time.monotonic()) * 0.55))
        else:
            stage_budget = min(130.0, max(15.0, (end - time.monotonic()) * 0.33))

        OBJ_N = n
        OBJ_IDX = idx
        OBJ_REF = ref
        OBJ_T = T
        OBJ_DEADLINE = time.monotonic() + stage_budget
        CACHE_Y = None
        CACHE_F = np.inf
        CACHE_G = None

        x_best = y0
        if _HAVE_MIN:
            bounds = [(-50.0, 50.0)] * (n - 1)
            try:
                res = minimize(
                    _obj_value,
                    y0,
                    jac=_obj_grad,
                    method="L-BFGS-B",
                    bounds=bounds,
                    callback=_lbfgs_callback,
                    options={"maxiter": 20000, "ftol": 1e-12, "gtol": 1e-8,
                             "maxfun": 1000000, "maxls": 80, "m": 20, "disp": False},
                )
                x = getattr(res, "x", None)
                if x is not None and np.all(np.isfinite(np.asarray(x))):
                    x_best = np.asarray(x, dtype=np.float64)
            except Exception:
                x_best = y0
        else:
            x_best = y0

        p_new = p_from_y(x_best, idx, ref, n)
        sc_new = internal_score(p_new)

        current = best_p
        current_sc = best_score
        if np.isfinite(sc_new):
            for blend_t in (1.0, 0.75, 0.5, 0.25):
                q = safe_normalize((1.0 - blend_t) * best_p + blend_t * p_new)
                sc = internal_score(q)
                if np.isfinite(sc) and sc < current_sc:
                    current = q
                    current_sc = float(sc)
            p = current
            if sc_new < best_score:
                best_score = float(sc_new)
                best_p = p_new.copy()
                no_improve = 0
            elif current_sc < best_score:
                best_score = current_sc
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            p = best_p
            no_improve += 1

        if allow_lp and (end - time.monotonic()) > 45.0 and (stage >= 1 or no_improve > 0):
            if n <= 1200:
                lp_budget = min(45.0, max(6.0, (end - time.monotonic()) * 0.12))
            else:
                lp_budget = min(25.0, max(4.0, (end - time.monotonic()) * 0.07))
            try:
                p_lp, sc_lp = lp_best_response(p, best_score, rng, lp_budget)
            except Exception:
                p_lp, sc_lp = p, best_score
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = float(sc_lp)
                best_p = p_lp.copy()
                p = p_lp.copy()
                no_improve = 0
                if label:
                    print(label + " LP improve %.10f" % best_score, flush=True)

        if no_improve >= 1 and (end - time.monotonic()) > 15.0:
            sig = 0.25 / n if no_improve == 1 else 0.55 / n
            q = safe_normalize(np.maximum(p + rng.normal(0.0, sig, n), 0.0))
            scq = internal_score(q)
            if np.isfinite(scq) and scq < best_score:
                best_score = float(scq)
                best_p = q.copy()
                p = q.copy()
                no_improve = 0
            else:
                p = safe_normalize(0.72 * best_p + 0.28 * q)

        if label and stage % 2 == 0:
            print(label + " stage %d best %.10f current %.10f" % (stage, best_score, internal_score(p)), flush=True)

        if best_score < 1.498 and no_improve >= 3:
            break

    if (end - time.monotonic()) > 15.0:
        pol_budget = min(60.0, max(5.0, (end - time.monotonic()) * 0.35))
        try:
            p_pol, sc_pol = coordinate_polish(p, rng, pol_budget)
        except Exception:
            p_pol, sc_pol = p, best_score
        if np.isfinite(sc_pol) and sc_pol < best_score:
            best_score = float(sc_pol)
            best_p = p_pol.copy()

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def _score_sort_key(item):
    return item[0]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    global GLOBAL_BEST_P, GLOBAL_BEST_SCORE

    t0 = time.monotonic()
    total_budget = min(float(budget_s), 2380.0)
    hard_deadline = t0 + max(10.0, total_budget - 30.0)

    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)

    GLOBAL_BEST_P = None
    GLOBAL_BEST_SCORE = np.inf

    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        s = float(arr.sum())
        if arr.ndim == 1 and arr.size >= 32 and s > 0.0:
            parent = arr / s
    except Exception:
        parent = None

    parent_n = int(parent.size) if parent is not None else 0

    if parent is not None:
        lengths = {parent_n, 1024, 2048, 4096}
    else:
        lengths = {1024, 2048, 4096}
    lengths = sorted([int(x) for x in lengths if int(x) >= 128 and int(x) <= 4096])

    record_candidate(parent)

    shape_p = None
    shape_score = np.inf
    if hard_deadline - time.monotonic() > 420.0:
        shape_n = parent_n if 96 <= parent_n <= 768 else 512
        shape_budget = min(95.0, max(20.0, (hard_deadline - time.monotonic()) * 0.11))
        try:
            shape_p, shape_score = run_shape_search(shape_n, rng, shape_budget)
            record_candidate(shape_p)
            print("shape search: %s" % shape_score, flush=True)
        except Exception:
            shape_p, shape_score = None, np.inf

    quick_pool = []
    quick_total = min(430.0, max(90.0, (hard_deadline - time.monotonic()) * 0.22))
    per_quick = max(25.0, quick_total / max(1, len(lengths)))
    quick_deadline = min(hard_deadline - 60.0, time.monotonic() + quick_total)

    for n in lengths:
        if time.monotonic() > quick_deadline - 25.0:
            break
        inits = []
        if parent is not None:
            inits.append(resample_seq(parent, n))
        if shape_p is not None:
            inits.append(resample_seq(shape_p, n))
        if GLOBAL_BEST_P is not None:
            inits.append(resample_seq(GLOBAL_BEST_P, n))
        try:
            inits.extend(generate_initials(n, rng, None, None))
        except Exception:
            pass

        scored = []
        for q in inits:
            q = safe_normalize(q)
            if q.size != n:
                continue
            try:
                sc = internal_score(q)
            except Exception:
                sc = np.inf
            if np.isfinite(sc):
                scored.append((sc, q))
        scored.sort(key=_score_sort_key)
        if not scored:
            continue

        sc0, q0 = scored[0]
        record_candidate(q0)
        budget = max(15.0, min(per_quick, quick_deadline - time.monotonic() - 5.0))
        if budget > 12.0:
            try:
                qp, scp = lbfgs_polish(q0, budget, "quick n%d:" % n,
                                       allow_lp=True, rng=rng, quick=True)
            except Exception:
                qp, scp = q0, sc0
            record_candidate(qp)
            quick_pool.append((scp, int(n), qp))
        else:
            quick_pool.append((sc0, int(n), q0))

        print("quick n=%d score=%s global=%s" % (n, np.min([s for s, _ in quick_pool[-1:]]) if quick_pool else np.inf, GLOBAL_BEST_SCORE), flush=True)

        if GLOBAL_BEST_P is not None and (shape_p is None or GLOBAL_BEST_SCORE < shape_score):
            shape_p = GLOBAL_BEST_P.copy()
            shape_score = GLOBAL_BEST_SCORE

    best_by_len = {}
    for sc, n, p in quick_pool:
        if p is None:
            continue
        if n not in best_by_len or sc < best_by_len[n][0]:
            best_by_len[n] = (sc, p)
    if GLOBAL_BEST_P is not None:
        n = int(GLOBAL_BEST_P.size)
        sc = float(GLOBAL_BEST_SCORE)
        if n not in best_by_len or sc < best_by_len[n][0]:
            best_by_len[n] = (sc, GLOBAL_BEST_P)

    items = [(sc, n, p) for n, (sc, p) in best_by_len.items()]
    items.sort(key=_score_sort_key)
    chosen = items[:3]
    if chosen:
        has_large = any(n >= 2048 for _, n, _ in chosen)
        if not has_large:
            for item in items:
                sc, n, p = item
                if n >= 2048 and sc <= chosen[0][0] + 0.025:
                    if len(chosen) < 3:
                        chosen.append(item)
                    else:
                        chosen[-1] = item
                    break

    long_time = max(0.0, min(1500.0, (hard_deadline - time.monotonic()) * 0.68))
    if len(chosen) > 0 and long_time > 70.0:
        per_long = long_time / float(len(chosen))
        for sc, n, p in chosen:
            if time.monotonic() > hard_deadline - 180.0:
                break
            budget = min(per_long, max(60.0, (hard_deadline - time.monotonic()) * 0.42))
            try:
                pp, sp = lbfgs_polish(p, budget, "long n%d:" % n,
                                      allow_lp=True, rng=rng, quick=False)
            except Exception:
                pp, sp = p, sc
            record_candidate(pp)
            print("long n=%d score=%s global=%s" % (n, sp, GLOBAL_BEST_SCORE), flush=True)

    # Upsample the current global best to 4096 if that may help.
    if GLOBAL_BEST_P is not None and int(GLOBAL_BEST_P.size) < 4096 and (hard_deadline - time.monotonic()) > 240.0:
        p_up = resample_seq(GLOBAL_BEST_P, 4096)
        record_candidate(p_up)
        if GLOBAL_BEST_SCORE < np.inf:
            budget = min(330.0, max(60.0, hard_deadline - time.monotonic() - 90.0))
            if budget > 50.0:
                try:
                    p_u, sc_u = lbfgs_polish(p_up, budget, "up n4096:",
                                             allow_lp=True, rng=rng, quick=False)
                except Exception:
                    p_u, sc_u = p_up, internal_score(p_up)
                record_candidate(p_u)
                print("upsample n=4096 score=%s global=%s" % (sc_u, GLOBAL_BEST_SCORE), flush=True)

    if GLOBAL_BEST_P is not None and (hard_deadline - time.monotonic()) > 45.0:
        final_budget = max(25.0, hard_deadline - time.monotonic() - 15.0)
        final_budget = min(final_budget, 420.0)
        try:
            pf, sf = lbfgs_polish(GLOBAL_BEST_P, final_budget, "final:",
                                  allow_lp=True, rng=rng, quick=False)
        except Exception:
            pf, sf = GLOBAL_BEST_P, GLOBAL_BEST_SCORE
        record_candidate(pf)

    if GLOBAL_BEST_P is not None and (time.monotonic() + 2.0 < hard_deadline):
        try:
            pc, sc_c = coordinate_polish(GLOBAL_BEST_P, rng, min(20.0, hard_deadline - time.monotonic() - 3.0))
        except Exception:
            pc, sc_c = GLOBAL_BEST_P, GLOBAL_BEST_SCORE
        record_candidate(pc)

    if GLOBAL_BEST_P is None:
        final_p = make_uniform(1024)
    else:
        final_p = safe_normalize(GLOBAL_BEST_P)
        final_p = np.maximum(final_p, 0.0)
        final_p = np.minimum(final_p, 1000.0)
        final_p = np.where(np.isfinite(final_p), final_p, 0.0)

    s = float(final_p.sum())
    if s <= 1e-12 or final_p.size == 0:
        final_p = make_uniform(1024)

    return [float(x) for x in final_p]
