Multi-method optimization for minimizing peak auto-convolution of non-negative sequences.
Combines parametric initialization, subgradient/multiplicative updates, LP-based refinement,

import numpy as np

from scipy.optimize import linprog
import time


def score_sequence(a):
    """Compute the score for a non-negative sequence."""
    n = len(a)
    if n == 0:
        return float('inf')
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    # Normalize
    a_norm = a / s
    conv = np.convolve(a_norm, a_norm)
    max_b = conv.max()
    return 2.0 * n * max_b


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def param_raised_cosine_power(n, p=1.0):
    """(cos(pi*t/n))^p type window, p > 0"""
    t = np.arange(n, dtype=np.float64)
    x = np.cos(np.pi * t / n)
    a = np.maximum(0.0, x) ** p
    return a

def param_sinc_power(n, p=1.0):
    """(sin(pi*(t+0.5)/n) / (pi*(t+0.5)/n))^p"""
    t = np.arange(n, dtype=np.float64) + 0.5
    x = np.sin(np.pi * t / n) / (np.pi * t / n)
    a = np.abs(x) ** p
    return a

    """Gaussian centered in the sequence."""
    t = np.arange(n, dtype=np.float64)

def build_conv_matrix(a):
    """Build sparse matrix for conv(a, x) as a linear operator on x."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=600.0, max_iters=500, verbose=True):
    """LP-based iterative refinement."""
    start_time = time.monotonic()

    stagnant = 0
    it = 0

        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max()

            best_a = a.copy()

def local_search_refine(a, time_limit_s=120.0, rng=None, verbose=True):

def momentum_subgradient_refine(a, time_limit_s=600.0, verbose=True):
    """Subgradient method with momentum (Nesterov-style)."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    MoSG start: score={best_score:.8f}, n={n}")
    
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    stagnant = 0
    
    for it in range(50000):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        max_val = conv.max()
        current_score = 2.0 * n * max_val
        
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 200 == 0 or it < 10):
                print(f"    MoSG iter {it:5d}: score={current_score:.8f}")
        else:
            stagnant += 1
        
        if stagnant > 500:
            if verbose:
                print(f"    MoSG stagnant at iter {it}, best={best_score:.8f}")
            break
        
        # Compute subgradient (smoothed)
        threshold = max_val * 0.95
        active = conv >= threshold
        active_indices = np.where(active)[0]
        
        g = np.zeros(n)
        for k in active_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        
        # Normalize gradient
        g_max = g.max()
        if g_max > 1e-15:
            g = g / g_max
        
        # Step size (decreasing)
        eta = 0.3 / (1.0 + 0.005 * it)
        
        # Momentum update
        momentum = mu * momentum - eta * g
        a_new = a + momentum
        a_new = normalize_nonneg(a_new)
        
        # Accept if not too bad
        new_score = score_sequence(a_new)
        if new_score < current_score * 1.05:  # Allow small increase for escaping
            a = a_new
        else:
            mu = max(0.5, mu * 0.95)  # Reduce momentum
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    MoSG done: best={best_score:.8f}, iters={it+1}, time={time.monotonic()-start_time:.1f}s")
    
    return best_a, best_score

Multi-method optimization for minimizing peak auto-convolution of non-negative sequences.
Combines parametric initialization, subgradient/multiplicative updates, frequency-domain
flattening (Gerchberg-Saxton style), LP-based refinement, and local search.

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    if n == 0:
        return float('inf')
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = np.convolve(a_norm, a_norm)
    return 2.0 * n * conv.max()


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


# Parametric families
def param_raised_cosine(n, p=1.0):
    t = np.arange(n, dtype=np.float64)
    x = np.cos(np.pi * t / n)
    return np.maximum(0.0, x) ** p

def param_sinc(n, p=1.0):
    t = np.arange(n, dtype=np.float64) + 0.5
    x = np.sin(np.pi * t / n) / (np.pi * t / n)
    return np.abs(x) ** p

def param_gaussian(n, width=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = width * n / 2.0
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)

def param_poly(n, order=2):
    if n == 1:
        return np.ones(1)
    t = np.arange(n, dtype=np.float64)
    x = 2.0 * t / (n - 1) - 1.0
    return np.maximum(0.0, 1.0 - x**2) ** order

def param_lorentzian(n, gamma=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / (gamma * n)
    return 1.0 / (1.0 + x**2)

# Subgradient computation via FFT
def compute_subgrad(a, active_mask):
    """Compute subgradient g[m] = sum_k active[k] * a[k-m] using FFT."""
    n = len(a)
    N = 1
    while N < 3 * n:
        N *= 2
    active_pad = np.zeros(N)
    active_pad[:len(active_mask)] = active_mask.astype(np.float64)
    A = np.fft.fft(a, N)
    C = np.fft.fft(active_pad)
    g = np.real(np.fft.ifft(C * np.conj(A)))[:n]
    return g

def freq_flatten_refine(a, time_limit_s=300.0, alpha=0.5, verbose=False):
    """Iteratively flatten the power spectrum while maintaining non-negativity."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    for it in range(10000):
        if time.monotonic() - start_time > time_limit_s:
            break
        
        # FFT with zero padding
        N = 2 * n
        A = np.fft.fft(a, N)
        P = np.abs(A)**2
        
        # Flatten: P_new = P^alpha (alpha < 1 flattens)
        P_max = P.max()
        P_norm = P / P_max
        P_new = P_norm ** alpha * P_max
        
        # Reconstruct with modified amplitude, keep phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        
        # Inverse FFT, take first n samples
        a_new = np.real(np.fft.ifft(A_new))[:n]
        a_new = np.maximum(0, a_new)
        a_new = normalize_nonneg(a_new)
        
        new_score = score_sequence(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
        
        a = a_new
    
    if verbose:
        print(f"    FreqFlatten: best={best_score:.8f}, iters={it+1}, t={time.monotonic()-start_time:.1f}s")
    
    return best_a, best_score


# LP-based refinement
def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=600.0, verbose=False):
