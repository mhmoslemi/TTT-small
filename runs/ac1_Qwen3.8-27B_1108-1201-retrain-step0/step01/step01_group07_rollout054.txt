Actually, I realize there might be an issue: the LP constraint is $(f*g)_k \leq M$, but we want to minimize the peak of $(a*a)_k$, not $(f*g)_k$. The LP is finding a $g$ such that the "cross-convolution" $f*g$ is bounded, which is a proxy for the goal.

$$\min_t \quad t$$
$$\text{s.t.} \quad ((1-\alpha) f + \alpha g) * ((1-\alpha) f + \alpha g))_k \leq t, \; \forall k$$

But this is quadratic (because of the product), not linear. So we'd need a different approach.

The "state of the art" LP is a linearization: fix $f$ and find the best $g$ such that $f*g$ is bounded. This works because if we move from $f$ towards $g$, the new auto-convolution is approximately $f*g$ for small steps.

Actually, for a step $a_{\text{new}} = (1-t)f + tg$, the auto-convolution is:
$(a_{\text{new}} * a_{\text{new}})_k = (1-t)^2 (f*f)_k + 2t(1-t)(f*g)_k + t^2 (g*g)_k$

The LP ensures $(f*g)_k \leq M$ for all $k$, which means the cross-term doesn't increase the peak. But it doesn't account for the $(g*g)_k$ term (which is $O(t^2)$).

    Compute gradient of tau * log(sum exp(c_k/tau)) w.r.t. a.

def project_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = a.size
    a = np.maximum(a, 0.0)
    s = a.sum()
    if abs(s - S) < 1e-12:
        return a
    if s == 0:
        return np.full(n, S/n)
    # Binary search for threshold

    for _ in range(60):

        t = np.maximum(a - mid, 0.0).sum()
        if t > S:
            lo = mid
        else:
            hi = mid

    res = np.maximum(a - t, 0.0)

    diff = S - res.sum()
    if diff != 0:

    return res

    t_step = 0
    
    t0 = time.monotonic()

        elapsed = time.monotonic() - t0
        f = min(1.0, elapsed / max(t_budget, 1e-6))
        tau = tau_start * (tau_end / tau_start) ** f

        lr = 0.5 * (1 + np.cos(np.pi * f)) + 0.01

        a = a_new

            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best = a.copy()
                if verbose:
                    print(f"      step {t_step}: score={sc:.6f}, tau={tau:.4f}")
    
    return best, best_score

Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Hybrid approach: soft-max gradient descent with clean gradient,
LP-based direction refinement (solving max sum(g) s.t. (f*g)<=M),
Adam optimizer, multi-resolution initialization, and time management.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_c(a):
    return np.convolve(a, a)


def score_of(a):
    n = len(a)
    c = np.convolve(a, a)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * np.max(c) / (s * s))


def grad_softmax(a, tau):
    n = len(a)
    c = np.convolve(a, a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    corr = np.correlate(a, q, 'full')
    grad = 2.0 * corr[2 * n - 2: 3 * n - 2]

def proj_simplex(a, S):
    n = a.size
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    if abs(s - S) < 1e-14:
        return a
    lo, hi = 0.0, a.max()
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) * 0.5
    res = np.maximum(a - mu, 0.0)
    res[np.argmax(res)] += S - res.sum()

def lp_refine_step(a, n_top=2000, verbose=False):
    """
    LP-based direction finding.
    Solves: max sum(g) s.t. (a*g)_k <= M for top lags, g >= 0.
    Returns updated sequence (same sum as a).
    """
    n = len(a)
    S = a.sum()
    c = np.convolve(a, a)
    M = c.max()
    
    if M < 1e-15:
        return a.copy()

    K = min(L, n_top)
    
    # Select top K lags
    top_idx = np.argsort(c)[::-1][:K]
    
    # Build sparse constraint matrix for selected lags
    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(top_idx):

        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(ii)
            cols.append(j)
            vals.append(a[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(K, n))
    b = np.full(K, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if res.status != 0:
        return a.copy()
    
    g = res.x
    g_sum = g.sum()
    if g_sum < 1e-12:
        return a.copy()
    
    # Rescale g to have same sum as a
    g = g / g_sum * S
    
    # Blend
    t = 0.3  # blending parameter
    a_new = (1 - t) * a + t * g
    a_new = proj_simplex(a_new, S)
    
    return a_new


def gd_adam(a0, tau_hi, tau_lo, t_budget, lr_base=0.1, verbose=False):
    """Adam gradient descent with soft-max and tau annealing."""
    n = a0.size
    S = a0.sum()
    a = a0 / S  # normalize to sum 1
    
    best = a.copy()
    best_sc = score_of(best)
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    step = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < t_budget:
        el = time.monotonic() - t0
        f = min(1.0, el / max(t_budget, 1e-6))
        tau = tau_hi * (tau_lo / tau_hi) ** f
        
        # Cosine LR
        lr = lr_base * (0.5 * (1 + np.cos(np.pi * f)) + 0.05)
        
        grad, c = grad_softmax(a, tau)
        step += 1
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** step)
        v_hat = v / (1 - b2 ** step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a_new, 1.0)
        a = a_new
        
        if step % 10 == 0:
            sc = score_of(a)
            if sc < best_sc:
                best_sc = sc
                best = a.copy()
                if verbose and step % 100 == 0:
                    print(f"        GD step {step}: score={sc:.6f}, tau={tau:.4f}")
    
    return best, best_sc


def init_from_provided(prov, n):
    """Resample provided sequence to length n."""
    if prov is None or prov.size == 0:
        return np.ones(n)
    x_old = np.linspace(0, 1, prov.size)
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, prov)
    return np.maximum(a, 0.0)


def init_chirp(n, seed=0):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.5 * np.cos(2 * np.pi * t * t / n)
    rng = np.random.default_rng(seed)
    a += 0.1 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_random(n, seed=0):
    return np.random.default_rng(seed).exponential(1.0, size=n)
