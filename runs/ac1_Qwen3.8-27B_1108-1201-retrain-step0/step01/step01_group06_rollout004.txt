def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(('uniform', np.ones(n) / n))
    
    # Gaussians
    for sf in [0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits.append((f'gauss_{sf}', g))
    
    # Edge concentration (U-shape)
    u = np.exp(0.5 * ((x - n * 0.5) / (0.2 * n)) ** 2)
    u /= u.sum()
    inits.append(('ushape', u))
    
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 6)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits.append(('bimodal', bi))
    
    # Triangular
    tri = 1.0 - np.abs(x - n * 0.5) / (n * 0.5)
    tri /= tri.sum()
    inits.append(('triangular', tri))
    
    # Cosine
    cos = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos /= cos.sum()
    inits.append(('cosine', cos))
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append((f'dirichlet_{alpha}', r))
    
    # "W-shape" (multiple peaks)
    for n_peaks in [3, 5]:
        w_shape = np.zeros(n)
        for i in range(n_peaks):
            center = i * n // (n_peaks - 1) if n_peaks > 1 else n // 2
            w_shape += np.exp(-0.5 * ((x - center) / (n * 0.08)) ** 2)
        w_shape /= w_shape.sum()
        inits.append((f'wshape_{n_peaks}', w_shape))
    
    # Parent-derived
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0).astype(np.float64)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits.append(('parent', pp))
        else:

            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits.append(('parent_interp', pp))
    
    return inits

            log_factor = -eta_try * sg_centered
            log_factor -= log_factor.max()
            log_factor = np.clip(log_factor, -50.0, 50.0)
            a_try = a * np.exp(log_factor)
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-12:
                eta_try *= 0.3
                continue
            a_try /= s_try
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()

            if obj_try < best_obj * (1.0 + 1e-4):
                a = a_try
                conv = conv_try
                current_obj = obj_try
                updated = True
                break
            else:
                eta_try *= 0.4

        if not updated:
            # Final attempt with very small step
            eta_try = max(eta * 0.05, 0.001)
            log_factor = -eta_try * sg_centered
            log_factor -= log_factor.max()
            log_factor = np.clip(log_factor, -10.0, 10.0)
            a = a * np.exp(log_factor)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:
                a = np.ones(n) / n
            else:
                a /= s
            conv = fast_convolve(a)
            current_obj = 2.0 * n * conv.max()

        iteration += 1

        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
            # Slight increase to exploit
            eta = min(eta * 1.01, eta_start)
        else:
            patience += 1
            if patience >= max_patience:
                patience = 0
                eta = max(eta * 0.5, eta_end)
                # Random restart if stuck for too long
                if restart_every > 0 and iteration % restart_every == 0:
                    noise = np.random.default_rng(iteration).standard_normal(n) * 0.1
                    a = a + noise * a.mean()
                    a = np.maximum(a, 1e-10)
                    a /= a.sum()
                    conv = fast_convolve(a)

        # Gradually decay eta
        if patience == 0 and iteration % 10 == 0:
            eta = max(eta * 0.98, eta_end)

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(('uniform', np.ones(n) / n))

    # Gaussians with various widths
    for sf in [0.12, 0.18, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits.append((f'gauss_{sf}', g))

    # Edge concentration (U-shape)
    u = np.exp(0.5 * ((x - n * 0.5) / (0.25 * n)) ** 2)
    u /= u.sum()
    inits.append(('ushape', u))

    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits.append(('bimodal', bi))

    # Triangular
    tri = 1.0 - np.abs(x - n * 0.5) / (n * 0.5 + 1)
    tri = np.maximum(tri, 0.01)
    tri /= tri.sum()
    inits.append(('triangular', tri))

    # Cosine bump
    cos = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos = np.maximum(cos, 0.01)
    cos /= cos.sum()
    inits.append(('cosine', cos))

    # Random Dirichlet
    for alpha in [0.3, 1.0, 3.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append((f'dirichlet_{alpha}', r))

    # Multi-peak
    for n_peaks in [3, 4]:
        w_shape = np.ones(n) * 0.01
        for i in range(n_peaks):
            center = i * (n - 1) / max(1, n_peaks - 1)
            w_shape += np.exp(-0.5 * ((x - center) / (n * 0.06)) ** 2)
        w_shape /= w_shape.sum()
        inits.append((f'multipk_{n_peaks}', w_shape))

    # Parent-derived
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0).astype(np.float64)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits.append(('parent', pp))
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits.append(('parent_interp', pp))

    return inits


def run_optimization(seed, budget_s):
    """Main optimization routine with three phases."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    best_sequence = None
    best_score = float('inf')

    # Try to access parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    # Phase 1: Quick scan over lengths and initializations
    print("Phase 1: Quick scan...")
    lengths = [64, 96, 128, 192, 256, 384, 512]

    all_configs = []
    for n in lengths:
        inits = generate_initializations(n, rng, parent)
        for name, a0 in inits:
            all_configs.append((n, name, a0))

    n_total = len(all_configs)
    phase1_time = min(500.0, (deadline - time.monotonic()) * 0.30)
    time_per_scan = max(1.5, phase1_time / max(1, n_total))

    scan_results = []
    for i, (n, name, a0) in enumerate(all_configs):
        if time.monotonic() > deadline - 150:
            break
        t_alloc = min(time_per_scan, deadline - time.monotonic() - 150)
        if t_alloc < 1.0:
            break

        a_opt, obj_opt = optimize_mirror_descent(
            a0, time_limit=t_alloc,
            t_start=40.0, t_end=10.0,
            eta_start=0.7, eta_end=0.03,
            entropy_reg=0.0)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt, obj_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan: n={n}, {name}, score={score:.6f}")

    # Phase 2: Deep optimization on best configs
    print(f"Phase 2: Deep optimization. Best so far: {best_score:.6f}")
    scan_results.sort(key=lambda x: x[0])

    top_configs = []
    seen = set()
    for score, n, name, a_opt, obj_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 12:
            top_configs.append((n, name, a_opt))
            seen.add(key)
        elif n not in {c[0] for c in top_configs} and len(top_configs) < 16:
            top_configs.append((n, name, a_opt))

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    remaining = deadline - time.monotonic() - 80
    if remaining < 30:
        top_configs = top_configs[:4]

    time_per_deep = max(8.0, remaining / max(1, len(top_configs)))

    # Multiple parameter settings for diversity
    param_sets = [
        (50.0, 8.0, 0.6, 0.02, 1.0),
        (100.0, 15.0, 0.4, 0.01, 0.5),
        (25.0, 5.0, 1.0, 0.05, 1.0),
    ]

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(time_per_deep * 1.2, deadline - time.monotonic() - 60)
        if t_alloc < 5:
            break

        for t_s, t_e, e_s, e_e, tf in param_sets:
            if time.monotonic() > deadline - 60:
                break
            t_single = (deadline - time.monotonic() - 60) / 3.0
            a_opt, obj_opt = optimize_mirror_descent(
                a0, time_limit=t_single,
                t_start=t_s, t_end=t_e,
                eta_start=e_s, eta_end=e_e,
                top_frac=tf)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep: n={n}, {name}, ts={t_s:.0f}, tf={tf}, score={score:.6f}")

    # Phase 3: Final polish on the best sequence
    if best_sequence is not None and time.monotonic() < deadline - 15:
        print(f"Phase 3: Polish. Current best: {best_score:.6f}")
        remaining = deadline - 15 - time.monotonic()

        polish_configs = [
            (60.0, 10.0, 0.3, 0.01, 1.0, 0.02),
            (120.0, 20.0, 0.2, 0.005, 0.4, 0.0),
            (30.0, 5.0, 0.5, 0.02, 1.0, 0.05),
            (80.0, 12.0, 0.25, 0.008, 0.6, 0.01),
        ]

        for t_s, t_e, e_s, e_e, tf, e_reg in polish_configs:
            if remaining < 8:
                break
            t_single = remaining / max(1, len(polish_configs))
            a_opt, obj_opt = optimize_mirror_descent(
                best_sequence, time_limit=min(t_single, remaining - 5),
                t_start=t_s, t_end=t_e,
                eta_start=e_s, eta_end=e_e,
                top_frac=tf, entropy_reg=e_reg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish: ts={t_s:.0f}, er={e_reg}, score={score:.6f}")
            remaining = deadline - 15 - time.monotonic()

    # Phase 4: Last-ditch random perturbations of best
    if best_sequence is not None and time.monotonic() < deadline - 5:
        remaining = deadline - 5 - time.monotonic()
        n = len(best_sequence)
        n_tries = max(1, int(remaining / 3.0))
        for _ in range(n_tries):
            if time.monotonic() > deadline - 5:
                break
            noise_scale = 0.05
            noise = rng.standard_normal(n) * noise_scale
            a_pert = best_sequence + noise * best_sequence.mean()
            a_pert = np.maximum(a_pert, 0.0)
            a_pert /= a_pert.sum()
            t_single = min(8.0, deadline - time.monotonic() - 5)
            if t_single < 1:
                break
            a_opt, obj_opt = optimize_mirror_descent(
                a_pert, time_limit=t_single,
                t_start=40.0, t_end=8.0,
                eta_start=0.5, eta_end=0.02)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Perturb: score={score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f}, length={len(best_sequence)}")
    return best_sequence


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses adaptive multiplicative mirror descent with
    temperature annealing, line search, and multi-start exploration to
    minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    result = run_optimization(seed=seed, budget_s=budget_s)
    return [float(x) for x in result]
