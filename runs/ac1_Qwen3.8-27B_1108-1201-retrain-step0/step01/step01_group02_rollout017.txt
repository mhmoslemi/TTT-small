"""
Gradient-based optimization for minimum peak auto-convolution of non-negative sequences.
Uses smoothed max of the auto-convolution as objective, with Adam optimizer in log-space.
Explores multiple sequence lengths and initializations including asymmetric starts.
"""

import numpy as np
from scipy.fft import fft, ifft
import time


def score_sequence(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def compute_conv(a):
    """Compute full auto-convolution using FFT for speed."""
    n = len(a)
    size = 2 * n  # power of 2 >= 2n-1
    A = fft(a, size)
    conv = np.real(ifft(A * A, size))[:2*n-1]
    return conv


def compute_grad_smoothed_max(a, conv, T):
    """
    Compute gradient of smoothed max of conv(a,a) with respect to a.
    Uses the formula: grad[j] = sum_k w[k] * a[k-j] (for valid k-j).
    Equivalent to a cross-correlation computed via FFT.
    """
    n = len(a)
    conv_len = 2 * n - 1
    # Softmax weights
    Tc = T * conv
    Tc -= Tc.max()  # numerical stability
    w = np.exp(Tc)
    w /= w.sum()
    # grad[j] = sum_m a[m] * w[j+m] for m=0..n-1, j+m < 2n-1
    # = conv(a, w_rev)[(2n-2)-j] where w_rev = w[::-1]
    # Computed via FFT:
    size = 1
    needed = n + conv_len - 1  # = 3n-2
    while size < needed:
        size *= 2
    A = fft(a, size)
    W = fft(w, size)
    # conv(a, w_rev) = ifft(A * W_rev) where W_rev = fft(w_rev)
    # But fft(w_rev) is not simply related to fft(w) in a useful way for real signals
    # Actually: ifft(A * fft(w)) = conv(a, w)
    # We want conv(a, w_rev) = ifft(A * fft(w_rev))
    # fft(w_rev) where w_rev = w[::-1]... this is the DFT of the reversed signal
    # For real w: fft(w_rev)[k] = conj(fft(w)[k]) * exp(-2*pi*i*k*(conv_len-1)/size)
    # This is getting complicated. Let me just use np.convolve for simplicity
    # or compute directly.
    
    # Direct computation via FFT:
    # conv(a, w) has length n + conv_len - 1 = 3n-2
    # conv(a, w)[k] = sum_i a[i] * w[k-i]
    # We want: grad[j] = sum_m a[m] * w[j+m]
    # Let's define: cross[j] = sum_m a[m] * w[j+m]
    # This is the cross-correlation of a and w at "negative lag" j
    # cross[j] = sum_m a[m] * w[m+j]
    # In terms of convolution: cross = conv(a, w[::-1])[::-1] ... let me verify
    # conv(a, w_rev)[k] = sum_i a[i] * w_rev[k-i] = sum_i a[i] * w[conv_len-1-k+i]
    # We want cross[j] = sum_m a[m] * w[j+m]
    # Set conv_len-1-k+m = j+m => conv_len-1-k = j => k = conv_len-1-j
    # So cross[j] = conv(a, w_rev)[conv_len-1-j]
    # Therefore: cross = conv(a, w_rev)[conv_len-1 : conv_len-1-n : -1]
    # i.e., cross = conv(a, w_rev)[n-1 : 2n-1][::-1]
    
    # For efficiency with FFT:
    w_rev = w[::-1]
    # conv(a, w_rev) via FFT
    conv_result = np.real(ifft(A * fft(w_rev, size), size))[:n + conv_len - 1]
    grad = conv_result[n-1:2*n-1][::-1]
    
    return grad, w


def adam_smoothed_max(a, max_iters=5000, T_init=5.0, T_final=50.0,
                       lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8,
                       time_limit=100.0, verbose=True):
    """
    Optimize a using Adam on the smoothed max of conv(a,a) in log-space.
    Works with sum(a) = 1 normalization.
    """
    n = len(a)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return a, score_sequence(a)
    a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_score = score_sequence(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    
    start_time = time.monotonic()
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit:
            break
        
        # Schedule T
        progress = min(1.0, it / max(1, max_iters * 0.5))
        T = T_init + (T_final - T_init) * progress
        
        conv = compute_conv(a)
        grad, w = compute_grad_smoothed_max(a, conv, T)
        
        # Scale gradient (the gradient of max(conv) w.r.t. a, with sum(a)=1 constraint)
        # In log-space: d log(a[j]) = -grad[j] (approximately, for the multiplicative update)
        # But we need to account for the constraint sum(a)=1.
        # Using the projection: the effective gradient in log-space is grad[j] - mean(grad)
        # (centering to enforce zero-sum constraint in log-space)
        grad_centered = grad - grad.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_centered
        v = beta2 * v + (1 - beta2) * grad_centered ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        # Update in log-space
        log_a = np.log(np.maximum(a, 1e-30))
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.exp(log_a)
        a_new = np.maximum(0, a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
        
        score = score_sequence(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose and (it % 100 == 0 or it < 5):
                print(f"    iter {it:5d}: T={T:.1f}, score={score:.6f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score


def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))
    return a


def init_gaussian(n, center_frac=0.5, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = center_frac * n
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    return a


def init_exponential(n, rate=2.0):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * t / n)
    return a


def init_sinusoidal(n, freq=1.0, phase=0.0):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * t / n + phase)
    return np.maximum(0, a)


def init_random_smooth(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return init_raised_cosine(n)
    return a / s


def init_comb(n, rng, teeth=3, jitter=0.1):
    """Initialize with a comb-like pattern."""
    a = np.zeros(n)
    positions = np.linspace(0, n, teeth + 2)[1:-1]
    for pos in positions:
        center = int(pos + rng.normal(0, jitter * n))
        width = max(1, int(n / (teeth * 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    return np.maximum(0, a)


def local_search(a, time_limit=60.0, rng=None):
    """Simple local search: perturb random segments multiplicatively."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a)
    a = np.maximum(0, a).copy()
    s = a.sum()
    if s < 1e-15:
        return a, score_sequence(a)
    a = a / s
    best_a = a.copy()
    best_score = score_sequence(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_limit:
        # Random segment perturbation
        width = int(rng.integers(max(2, n // 500), max(3, n // 50)))
        pos = int(rng.integers(0, max(1, n - width)))
        candidate = a.copy()
        # Multiplicative perturbation
        perturbation = np.exp(rng.normal(0, 0.15, size=width))
        candidate[pos:pos+width] *= perturbation
        candidate = np.maximum(0, candidate)
        s = candidate.sum()
        if s < 1e-15:
            continue
        candidate = candidate / s
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
    
    return best_a, best_score

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')

def compute_conv(a):
    n = len(a)
    if n <= 512:
        return np.convolve(a, a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    A = fft(a, size)
    conv = np.real(ifft(A * A, size))[:2 * n - 1]
    return conv


def compute_grad_smoothed(a, conv, T):
    n = len(a)
    conv_len = 2 * n - 1
    Tc = T * conv
    Tc -= Tc.max()
    w = np.exp(Tc)
    w /= w.sum()
    w_rev = w[::-1]
    needed = n + conv_len - 1
    size = 1
    while size < needed:
        size *= 2
    A = fft(a, size)
    Wr = fft(w_rev, size)
    full = np.real(ifft(A * Wr, size))[:needed]
    grad = full[n - 1:2 * n - 1][::-1]
    return grad, w

def lp_polish(a, time_limit=120.0, verbose=False):

    for n in [300, 500, 800, 1200, 2000, 3000, 5000]:
        configs.append((n, 'raised_cosine', None))
        configs.append((n, 'asym_gauss', None))
        configs.append((n, 'double_bump', None))
        configs.append((n, 'random', None))

    # Sort: prioritize smaller n first (faster per iteration), larger n later
    configs.sort(key=lambda x: x[0])

    # Time allocation
    for n, init_type, _ in configs:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        # Allocate time based on n: smaller n gets less time per config
        time_limit = min(remaining * 0.5, max(8.0, 60.0 * np.sqrt(n / 300.0)))
        if time_limit < 8:
            continue

        # Generate initialization
        if init_type == 'raised_cosine':
            a = init_raised_cosine(n)
        elif init_type == 'asym_gauss':
            a = init_asym_gaussian(n, rng.uniform(0.25, 0.75), rng.uniform(0.12, 0.30))
        elif init_type == 'double_bump':
            a = init_double_bump(n, rng)
        elif init_type == 'random':
            a = init_random_smooth(n, rng, n_bumps=int(rng.integers(2, 6)))
        else:
            a = init_raised_cosine(n)

        a = np.maximum(0.0, a)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        init_score = score_sequence(a)
        print(f"  n={n:5d}, init={init_type:12s}, init_score={init_score:.6f}, t_limit={time_limit:.0f}s")

        a_opt, score = adam_optimize(
            a, max_iters=50000, T_init=3.0, T_final=80.0,
            lr=0.08, time_limit=time_limit, verbose=False,
            jump_every=300, rng=rng)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  *** NEW BEST: score={score:.6f}, n={n}, init={init_type}")

    # Additional runs with different hyperparameters on best n
    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Run with different T schedule
            time_limit = min(remaining * 0.4, 300.0)
            print(f"\n=== Re-optimize best n={n_best} with T=[5,100], t={time_limit:.0f}s ===")
            a2, score2 = adam_optimize(
                best_seq + rng.normal(0, 0.01, n_best), max_iters=50000,
                T_init=5.0, T_final=100.0, lr=0.05,
                time_limit=time_limit, verbose=True, jump_every=200, rng=rng)
            if score2 < best_score:
                best_score = score2
                best_seq = a2.copy()
                print(f"  *** NEW BEST: score={score2:.6f}")

            remaining = deadline - time.monotonic()
            if remaining > 20:
                # Local search
                tl = min(remaining * 0.25, 120.0)
                print(f"\n=== Local search, t={tl:.0f}s ===")
                polished, score_ls = local_search(best_seq, time_limit=tl, rng=rng)
                if score_ls < best_score:
                    best_score = score_ls
                    best_seq = polished.copy()
                    print(f"  After LS: score={best_score:.6f}")

                remaining = deadline - time.monotonic()
                if remaining > 15 and n_best <= 3000:
                    # LP polish
                    tl_lp = min(remaining * 0.5, 180.0)
                    print(f"\n=== LP polish, t={tl_lp:.0f}s ===")
                    polished_lp, score_lp = lp_polish(best_seq, time_limit=tl_lp, verbose=True)
                    if score_lp < best_score:
                        best_score = score_lp
                        best_seq = polished_lp.copy()
                        print(f"  After LP: score={best_score:.6f}")

                # Final gradient pass
                remaining = deadline - time.monotonic()
                if remaining > 15:
                    tl_fin = min(remaining * 0.8, 200.0)
                    print(f"\n=== Final gradient pass, t={tl_fin:.0f}s ===")
                    a_fin, score_fin = adam_optimize(
                        best_seq, max_iters=50000, T_init=30.0, T_final=100.0,
                        lr=0.03, time_limit=tl_fin, verbose=True, rng=rng)
                    if score_fin < best_score:
                        best_score = score_fin
                        best_seq = a_fin.copy()
                        print(f"  After final: score={best_score:.6f}")

    if best_seq is None:
        best_seq = init_raised_cosine(500)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    print(f"\nFinal: score={best_score:.6f}, n={len(best_seq)}")
    verify = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified via evaluate_sequence: {verify:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")

    return [float(x) for x in best_seq]
