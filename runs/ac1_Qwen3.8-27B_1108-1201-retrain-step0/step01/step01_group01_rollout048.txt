def _fft_correlate(p, a):
    """Compute np.correlate(p, a, mode='valid') efficiently using FFT.
    p has length 2n-1, a has length n. Output has length n."""
    n_p = len(p)
    n_a = len(a)
    out_len = n_p - n_a + 1
    n_fft = 1
    while n_fft < n_p + n_a - 1:
        n_fft *= 2
    P = np.fft.rfft(p, n_fft)
    A = np.fft.rfft(a, n_fft)
    corr = np.fft.irfft(P * np.conj(A), n_fft)
    return corr[:out_len]

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimization of non-negative step functions to minimize normalized autocorrelation peak.
Uses log-sum-exp smoothed gradient with Adam optimizer, adaptive temperature,
multi-start strategy, basin hopping, and final polish phase.
Works with c = n*a normalization for better numerical scaling.
"""

import numpy as np
import time


def _fft_correlate(p, a):
    """Compute np.correlate(p, a, mode='valid') efficiently using FFT.
    p has length 2n-1, a has length n. Output has length n.
    """
    n_p = len(p)
    n_a = len(a)
    out_len = n_p - n_a + 1
    n_fft = 1
    while n_fft < n_p + n_a - 1:
        n_fft *= 2
    P = np.fft.rfft(p, n_fft)
    A = np.fft.rfft(a, n_fft)
    corr = np.fft.irfft(P * np.conj(A), n_fft)
    return corr[:out_len]


def _project_simplex(v, target):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score_c(c):
    """Score for c (sum=c_n) = max(conv(c,c)). Lower is better."""
    b = np.convolve(c, c)
    return float(np.max(b))


def _score_a(a):
    """Score for a (sum=1) = 2*n*max(conv(a,a)). Lower is better."""
    n = len(a)
    b = np.convolve(a, a)
    return float(2.0 * n * np.max(b))


def _adam_optimize(c, deadline, alpha_init=5.0, alpha_final=0.3,
                   lr=0.05, reheat_every=400, reheat_factor=2.5,
                   eval_every=50, tag=""):
    """
    Optimize c (sum(c)=n, c>=0) to minimize max(conv(c,c)).
    Uses log-sum-exp smoothed objective with Adam and adaptive temperature.
    Returns best c array.
    """
    n = len(c)
    c = _project_simplex(c.astype(np.float64), n).copy()

    best_c = c.copy()
    best_score = _score_c(best_c)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    alpha = alpha_init
    no_improve = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    while time.monotonic() < deadline:
        b = np.convolve(c, c)
        b_max = np.max(b)
        b_spread = b_max - np.min(b)
        # Adaptive temperature: fraction of the spread
        tau = alpha * b_spread if b_spread > 1e-12 else alpha / n

        # Log-sum-exp weights (soft max)
        shifted = (b - b_max) / tau
        shifted = np.clip(shifted, -80.0, 0.0)
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros_like(b)
            p[np.argmax(b)] = 1.0
        else:
            p /= p_sum

        # Gradient: (2/tau) * correlate(p, c, 'valid')
        grad = (2.0 / tau) * _fft_correlate(p, c)

        # Center gradient for sum constraint (tangent to simplex)
        grad -= np.mean(grad)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        c_new = c - lr * m_hat / (np.sqrt(v_hat) + eps)
        c_new = _project_simplex(c_new, n)

        if not np.all(np.isfinite(c_new)):
            c = best_c.copy()
            m[:] = 0
            v[:] = 0
            t = 0
            continue

        c = c_new

        # Periodic score evaluation
        if t % eval_every == 0:
            score = _score_c(c)
            if score < best_score:
                best_score = score
                best_c = c.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Temperature annealing / reheating
            if no_improve >= reheat_every:
                alpha = min(alpha_init, alpha * reheat_factor)
                no_improve = 0
                if tag:
                    print(f"    {tag}: reheat alpha={alpha:.2f}, score={best_score:.2f}")
            else:
                # Gradual cooling
                frac = no_improve / reheat_every
                alpha = alpha_final + (alpha - alpha_final) * (1.0 - frac)
                alpha = max(alpha, alpha_final)

    # Final check
    final_score = _score_c(c)
    if final_score < best_score:
        best_c = c.copy()

    return best_c


def _smooth_perturb(c, rng, scale=0.1):
    """Add smooth random perturbation to c and re-project."""
    n = len(c)
    i = np.arange(n, dtype=np.float64)
    n_modes = min(8, n // 2)
    freqs = rng.integers(1, n_modes + 1, size=min(5, n_modes))
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2.0 * np.pi)
        amp = rng.uniform(-1.0, 1.0)
        noise += amp * np.cos(2.0 * np.pi * f * i / n + phase)
    noise_std = np.std(noise)
    if noise_std > 1e-10:
        noise = scale * noise / noise_std * np.std(c)
    c_new = c + noise
    c_new = _project_simplex(c_new, n)
    return c_new


def _random_perturb(c, rng, scale=0.05):
    """Add random (non-smooth) perturbation."""
    n = len(c)
    noise = rng.normal(0, scale * np.std(c), size=n)
    c_new = c + noise
    c_new = _project_simplex(c_new, n)
    return c_new
