"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses FFT-based convolution, corrected smooth-max gradient (no spurious
correction term), heavy-ball momentum with adaptive LR, geometric tau
annealing, peak-focused refinement, and diverse initializations
(Blackman-Harris, Nuttall, Kaiser, chirp, random, provided).
Two-phase strategy: exploration then focused polishing.
"""

import numpy as np
import time


def fast_conv(a):
    """Full convolution a*a via FFT (length 2n-1)."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa)[:2 * n - 1]


def proj_simplex(a, S):
    """Project a onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    if s < S:
        a += (S - s) / n
        return a
    lo, hi = 0.0, float(a.max())
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    diff = S - res.sum()
    if abs(diff) > 1e-12:
        idx = int(res.argmax())
        res[idx] += diff
    return res


def smooth_max_grad(a, tau):
    """
    Returns (grad, c) where grad = gradient of tau*logsumexp(c/tau) wrt a,
    c = full convolution of a with itself.
    Gradient formula: grad[i] = 2 * sum_j q[i+j] * a[j]
    """
    c = fast_conv(a)
    c_max = c.max()
    # Numerically stable logsumexp
    exp_c = np.exp((c - c_max) / tau)
    q = exp_c / exp_c.sum()
    n = len(a)
    # q has length 2n-1, a has length n
    # correlate(q, a, 'valid')[i] = sum_{j=0}^{n-1} q[i+j]*a[j]
    grad = 2.0 * np.correlate(q, a, 'valid')
    return grad, c


def init_blackman_harris(n, seed=0):

         - 0.01168 * np.cos(6.0 * np.pi * x / d))

def mom_opt(a0, tau_hi, tau_lo, lr0, budget, momentum=0.85, verbose=False):

    Momentum projected gradient descent with tau annealing.
    """
    n = a0.size
    S = np.sqrt(2.0 * n)

    v = np.zeros(n)

    best = a.copy()
    best_c = fast_conv(best).max()

    if sc < best_score:

        best = a.copy()

def nesterov_opt(a0, tau_hi, tau_lo, lr0, budget, verbose=False):
    """
    Nesterov accelerated projected gradient descent with tau annealing.
    """
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = proj_simplex(a0.astype(np.float64) / max(a0.sum(), 1e-12) * S, S)
    a_prev = a.copy()
    t_val = 1.0

    best = a.copy()
    best_c = fast_conv(best).max()
    best_score = evaluate_sequence(best.tolist())

    t0 = time.monotonic()
    lr = lr0
    eval_every = max(10, n // 100)
    step = 0

    while time.monotonic() - t0 < budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(budget, 1e-6))
        tau = tau_hi * (tau_lo / tau_hi) ** (frac ** 0.5)

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        theta = (t_val - 1.0) / t_next
        y = a + theta * (a - a_prev)

        g, c = smooth_max_grad(y, tau)
        a_new = proj_simplex(y - (lr / t_next) * g, S)

        a_prev = a
        a = a_new
        t_val = t_next
        step += 1

        if step % eval_every == 0:
            mc = fast_conv(a).max()
            if mc < best_c - 1e-10:
                best_c = mc
                best = a.copy()
                lr = min(lr * 1.05, lr0 * 2.0)
            elif step % (eval_every * 10) == 0:
                lr = max(lr * 0.8, lr0 * 0.1)

    # Final
    mc = fast_conv(a).max()
    sc = evaluate_sequence(a.tolist())
    if sc < best_score:
        best_score = sc
        best = a.copy()

    if verbose:
        print(f"    [nesterov] steps={step}, best_mc={best_c:.6f}, score={best_score:.6f}")
    return best, best_score


def peak_refine(a, budget, verbose=False):
    """
    Direct subgradient steps targeting the top peaks in the convolution.
    """
    n = a.size
    S = np.sqrt(2.0 * n)
    a = proj_simplex(a / max(a.sum(), 1e-12) * S, S)

    best = a.copy()
    best_c = fast_conv(best).max()
    best_score = evaluate_sequence(best.tolist())

    t0 = time.monotonic()
    lr = 0.01
    step = 0

    while time.monotonic() - t0 < budget:
        c = fast_conv(a)
        # Find top K peaks
        K = min(5, len(c))
        top_idx = np.argsort(c)[-K:]
        
        # Subgradient of sum of top-K peak values
        # d c[k]/d a[i] = 2*a[k-i] for valid k-i
        sg = np.zeros(n)
        for k in top_idx:
            # a[k-i] for i in [0, n-1] where k-i in [0, n-1]
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    sg[i] += 2.0 * a[j]
        
        # Scale subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-10:
            a = proj_simplex(a - lr * sg / sg_norm, S)
        
        step += 1
        if step % 10 == 0:
            mc = fast_conv(a).max()
            if mc < best_c - 1e-8:
                best_c = mc
                best = a.copy()
                lr = min(lr * 1.1, 0.05)
            elif step % 50 == 0:
                lr = max(lr * 0.8, 0.001)

    # Final
    mc = fast_conv(a).max()
    sc = evaluate_sequence(a.tolist())
    if sc < best_score:
        best_score = sc
        best = a.copy()

    if verbose:
        print(f"    [peak_refine] steps={step}, best_mc={best_c:.6f}, score={best_score:.6f}")
    return best, best_score
