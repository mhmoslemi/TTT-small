    """Efficient projection onto {x >= 0, sum(x) = S}."""
    n = len(a)
    a = np.maximum(a, 0.0)

    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    t0 = time.monotonic()

        t += 1

        c = np.fft.irfft(fa * fa, N)[:2*n-1]

        exp_c = np.exp(c_shifted / tau)
        q = exp_c / exp_c.sum()

        fq = np.fft.rfft(q[::-1], N)
        conv = np.fft.irfft(fq * fa, N)
        grad = 2.0 * conv[n-1:2*n-1][::-1]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        a = a - lr_adam * m / (np.sqrt(v) + eps)
        
        # Project
        a = project_simplex_v2(a, S)

                best_a = a.copy()

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using FFT-accelerated Adam optimization
with aggressive tau annealing, multiple sequence lengths, and local search polishing.
Key improvements: FFT for O(n log n) convolution/gradient, Adam optimizer,
multi-length exploration, and gradient-based local search for final polishing.
"""

import numpy as np
import time


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-14:
        return np.full(n, S / n)
    if abs(s - S) < 1e-12:
        return a
    lo, hi = 0.0, a.max()
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    diff = S - res.sum()
    if diff > 1e-12:
        idx = np.where(res < res.max())[0]
        if len(idx) > 0:
            distribute = diff / len(idx)
            for i in idx:
                res[i] += distribute
        else:
            res[0] += diff
    elif diff < -1e-12:
        idx = res.argmax()
        res[idx] += diff
    return res


def fft_convolve_square(a, N):
    """Compute conv(a, a) using FFT. N must be >= 2*len(a)-1."""
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * len(a) - 1]


def compute_gradient_fft(a, q, N):
    """Compute gradient = 2 * correlate(q, a, 'valid') using FFT.
    q has length 2*n-1, a has length n."""
    n = len(a)
    fa = np.fft.rfft(a, N)
    fq = np.fft.rfft(q[::-1], N)
    conv = np.fft.irfft(fq * fa, N)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def init_chirp(n, seed=0):
    """Chirp-based non-negative initialization."""
    t = np.arange(n, dtype=np.float64) - (n - 1) * 0.5
    a = 1.0 + 0.8 * np.cos(2.0 * np.pi * t * t / (2.0 * n))
    rng = np.random.default_rng(seed)
    a += 0.03 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_random(n, seed=0):
    """Random initialization with exponential distribution."""
    return np.random.default_rng(seed).exponential(1.0, size=n)


def init_flat_with_noise(n, seed=0):
    """Near-uniform with structured noise."""
    rng = np.random.default_rng(seed)
    t = np.arange(n)
    a = 1.0 + 0.3 * np.sin(2.0 * np.pi * t / n * 7) + 0.2 * np.cos(2.0 * np.pi * t / n * 13)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_gaussian_window(n, seed=0):
    """Gaussian-windowed initialization."""
    t = np.arange(n) - n / 2.0
    sigma = n / 4.0
    a = np.exp(-t * t / (2.0 * sigma * sigma))
    rng = np.random.default_rng(seed)
    a += 0.02 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def optimize_adam(a0, budget_s, tau_start=1.0, tau_end=0.005, lr_base=0.5, verbose=False):
    """Adam-based optimization with soft-max smooth approximation and FFT."""
    n = len(a0)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    N = max(N, 1)

    S = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    a = project_simplex(a, S)

    best_a = a.copy()
    best_mc = fft_convolve_square(best_a, N).max()
    best_score = evaluate_sequence(best_a.tolist())

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_step = 0
    no_improve_count = 0
    stall_threshold = 200

    t0 = time.monotonic()

    while time.monotonic() - t0 < budget_s:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(budget_s, 1e-6))

        # Aggressive tau annealing with plateau
        if progress < 0.3:
            tau = tau_start * (0.2 / tau_start) ** (progress / 0.3)
        elif progress < 0.7:
            tau = 0.2 * (0.02 / 0.2) ** ((progress - 0.3) / 0.4)
        else:
            tau = 0.02 * (tau_end / 0.02) ** ((progress - 0.7) / 0.3)

        # Learning rate schedule
        lr = lr_base / (1.0 + 2.0 * progress)

        t_step += 1
        lr_t = lr * np.sqrt(1 - beta2 ** t_step) / (1 - beta1 ** t_step)

        # Compute convolution
        c = fft_convolve_square(a, N)

        # Softmax weights
        c_max = c.max()
        c_shifted = c - c_max
        exp_c = np.exp(c_shifted / tau)
        q = exp_c / exp_c.sum()

        # Gradient via FFT
        grad = compute_gradient_fft(a, q, N)

        # Adam update
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * grad * grad
        a = a - lr_t * m_adam / (np.sqrt(v_adam) + eps)

        # Project onto simplex
        a = project_simplex(a, S)

        # Periodic evaluation
        if t_step % 30 == 0 or (budget_s - (time.monotonic() - t0) < 20):
            mc = fft_convolve_square(a, N).max()
            if mc < best_mc - 1e-10:
                best_mc = mc
                best_a = a.copy()
                best_score = evaluate_sequence(best_a.tolist())
                no_improve_count = 0
                if verbose:
                    print(f"    step={t_step}, max_c={mc:.6f}, score={best_score:.6f}, tau={tau:.4f}")
            else:
                no_improve_count += 1
                # If stalled for too long, perturb and reset Adam
                if no_improve_count > stall_threshold:
                    rng = np.random.default_rng(t_step)
                    a = best_a + 0.01 * max(best_a.max(), 1.0) * rng.standard_normal(n)
                    a = project_simplex(a, S)
                    m_adam = np.zeros(n)
                    v_adam = np.zeros(n)
                    no_improve_count = 0

    return best_a, best_score


def local_search(a0, budget_s, verbose=False):
    """Gradient-based local search: move mass between indices to improve score."""
    n = len(a0)
    N = 1
    while N < 2 * n - 1:
        N *= 2

    S = np.sqrt(2.0 * n)
    a = a0.copy()
    best_a = a.copy()
    best_mc = fft_convolve_square(best_a, N).max()
    best_score = evaluate_sequence(best_a.tolist())

    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < budget_s:
        step += 1
        c = fft_convolve_square(a, N)
        mc = c.max()

        # Compute gradient with small tau
        tau = 0.05
        c_shifted = c - c.max()
        exp_c = np.exp(c_shifted / tau)
        q = exp_c / exp_c.sum()
        grad = compute_gradient_fft(a, q, N)

        # Find best increase index (most negative gradient) and decrease index (most positive)
        # But we need a[dec] > 0 to decrease
        pos_grad = grad.copy()
        pos_grad[a <= 1e-6] = np.inf  # Can't decrease zeros

        dec_idx = np.argmin(pos_grad)  # Wait, we want to DECREASE where grad is most POSITIVE
        # Actually: we want to decrease a[j] where grad[j] is most positive (reduces objective)
        # and increase a[i] where grad[i] is most negative
        dec_idx = np.argmax(grad)  # Most positive gradient -> decrease here
        inc_idx = np.argmin(grad)  # Most negative gradient -> increase here

        if dec_idx == inc_idx:
            # Try second best
            sorted_grad = np.argsort(grad)
            inc_idx = sorted_grad[0]
            dec_candidates = [i for i in range(n) if i != inc_idx and a[i] > 1e-6]
            if not dec_candidates:
                break
            dec_idx = max(dec_candidates, key=lambda i: grad[i])

        if dec_idx == inc_idx:
            continue

        # Try different amounts
        improved = False
        for frac in [0.01, 0.005, 0.002, 0.001, 0.02, 0.05]:
            amount = a[dec_idx] * frac
            if amount < 1e-10:
                continue
            a[dec_idx] -= amount
            a[inc_idx] += amount
            mc_new = fft_convolve_square(a, N).max()
            if mc_new < best_mc - 1e-10:
                best_mc = mc_new
                best_a = a.copy()
                best_score = evaluate_sequence(best_a.tolist())
                improved = True
                break
            else:
                a[dec_idx] += amount
                a[inc_idx] -= amount

        if not improved:
            # Try random pairs
            rng = np.random.default_rng(step)
            for _ in range(5):
                candidates = np.where(a > 1e-4)[0]
                if len(candidates) < 2:
                    break
                i, j = rng.choice(candidates, size=2, replace=False)
                amount = a[i] * 0.01
                if amount < 1e-10:
                    continue
                a[i] -= amount
                a[j] += amount
                mc_new = fft_convolve_square(a, N).max()
                if mc_new < best_mc - 1e-10:
                    best_mc = mc_new
                    best_a = a.copy()
                    best_score = evaluate_sequence(best_a.tolist())
                    improved = True
                    break
                else:
                    a[i] += amount
                    a[j] -= amount

    return best_a, best_score
