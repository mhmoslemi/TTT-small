```python
    n = len(a)
    S = a.sum()
    conv = np.convolve(a, a)
    max_conv = conv.max()
    
    # Find active lags
    threshold = max_conv * (1 - 1e-3)  # lags within 0.1% of max
    active_mask = conv >= threshold
    active_lags = np.where(active_mask)[0]
    if len(active_lags) > n_active:
        # Keep top n_active
        top_idx = np.argsort(-conv[active_mask])[:n_active]
        active_lags = active_lags[top_idx]
    
    # Compute subgradient (average over active lags)
    subgrad = np.zeros(n)
    for k in active_lags:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                subgrad[i] += 2.0 * a[j]
    subgrad /= len(active_lags)
    
    # Full gradient of f = max_conv / S^2 (ignoring the 2n factor for now)
    grad = subgrad / S**2 - 2.0 * max_conv / S**3 * np.ones(n)
    
    # Update
    a_new = a - alpha * grad
    a_new = np.maximum(0, a_new)
    a_new = a_new / a_new.sum()
    return a_new
```

```python
    n = len(a)
    ar = a[::-1]
    subgrad = np.zeros(n)
    for k, w in zip(active_lags, weights):
        s = n - 1 - k
        if s >= 0:
            subgrad[:n-s] += w * ar[s:]
        else:
            subgrad[-s:] += w * ar[:n+s]
    return 2.0 * subgrad
```

```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def compute_subgradient(a, active_lags, weights):
    n = len(a)
    ar = a[::-1]
    subgrad = np.zeros(n)
    for k, w in zip(active_lags, weights):
        s = n - 1 - k
        if s >= 0:
            subgrad[:n-s] += w * ar[s:]
        else:
            subgrad[-s:] += w * ar[:n+s]
    return 2.0 * subgrad

def subgradient_descent(a, time_limit=60.0, verbose=True):
    start_time = time.monotonic()
    n = len(a)
    a = normalize_nonneg(a.copy())
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"  SGD start: score={best_score:.6f}, n={n}")
    
    alpha = 1.0  # initial step size (in units of a itself)
    stagnant = 0
    
    for it in range(500000):
        if time.monotonic() - start_time > time_limit:
            break
        
        conv = np.convolve(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv  # S=1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 200 == 0 or it < 10):
                print(f"  SGD iter {it:5d}: score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 500:
            if verbose:
                print(f"  SGD stagnant at iter {it}, best={best_score:.6f}")
            break
        
        # Find active lags
        K = min(len(conv), max(10, n // 5))
        sorted_idx = np.argsort(-conv)
        active_lags = sorted_idx[:K]
        
        # Weights: emphasize lags closer to the max
        conv_active = conv[active_lags]
        # Use exponential weighting
        weights = np.exp(10.0 * (conv_active - max_conv) / max_conv)
        weights /= weights.sum()
        
        # Compute subgradient
        subgrad = compute_subgradient(a, active_lags, weights)
        
        # Gradient of score (S=1): grad = subgrad - 2*max_conv
        grad = subgrad - 2.0 * max_conv
        
        # Step
        a_new = a - alpha * grad
        a_new = np.maximum(0, a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            alpha *= 0.5
            continue
        a_new = a_new / s_new
        
        # Check if it improved (every 10 iterations for efficiency)
        if it % 10 == 0:
            new_score = score_sequence(a_new)
            if new_score > score * 1.01:
                # Rejected, try smaller step
                alpha *= 0.5
                a_new = a - alpha * grad
                a_new = np.maximum(0, a_new)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new /= s_new
                else:
                    continue
            a = a_new
        else:
            a = a_new
        
        # Decay step size
        alpha *= 0.9995
        if alpha < 0.01:
            alpha = 0.01
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"  SGD done: best={best_score:.6f}, iters={it+1}, time={time.monotonic()-start_time:.1f}s")
    return best_a, best_score
```

```python
def subgradient_descent(a, time_limit=60.0, verbose=True):
    start_time = time.monotonic()
    n = len(a)
    a = normalize_nonneg(a.copy())
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"  SGD start: score={best_score:.6f}, n={n}")
    
    alpha = 0.5  # step size
    momentum = 0.7  # momentum coefficient
    v = np.zeros(n)  # velocity
    stagnant = 0
    last_best_it = 0
    
    for it in range(200000):
        if time.monotonic() - start_time > time_limit:
            break
        
        conv = np.convolve(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv  # S=1
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            last_best_it = it
            if verbose and (it % 500 == 0 or it < 20):
                print(f"  SGD iter {it:5d}: score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 1000:
            if verbose:
                print(f"  SGD stagnant at iter {it}, best={best_score:.6f}, elapsed={time.monotonic()-start_time:.1f}s")
            break
        
        # Find active lags (top K)
        K = min(len(conv), max(10, n // 10))
        sorted_idx = np.argsort(-conv)
        active_lags = sorted_idx[:K]
        
        # Weights: exponential emphasis on peaks
        conv_active = conv[active_lags]
        # Temperature for weighting (lower = more peak-focused)
        temp = max(max_conv * 0.01, 1e-10)
        weights = np.exp(-(max_conv - conv_active) / temp)
        weights /= weights.sum()
        
        # Compute subgradient
        subgrad = compute_subgradient(a, active_lags, weights)

        grad = subgrad - 2.0 * max_conv

        v = momentum * v - alpha * grad
        a_new = a + v
        a_new = np.maximum(0, a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            alpha *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        if it % 50 == 0:
            check_score = score_sequence(a_new)
            if check_score > score * 1.1:
                # Rejected
                alpha *= 0.5
                v = np.zeros(n)
                continue
        
        a = a_new
        
        # Adapt alpha
        if stagnant > 200:
            alpha *= 1.1  # increase step size to escape
        else:
            alpha *= 0.999
        
        alpha = min(alpha, 2.0)  # cap step size
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
```

```python
    start_time = time.monotonic()
    n = len(a)
    a = normalize_nonneg(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"  SGD start: score={best_score:.6f}, n={n}")

    alpha = 0.5
    v = np.zeros(n)
    stagnant = 0
    K = min(2 * n - 1, max(15, n // 5))

    for it in range(500000):
        if time.monotonic() - start_time > time_limit:
            break

        conv = np.convolve(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 1000 == 0 or it < 20):
                print(f"  SGD iter {it:5d}: score={score:.6f}")
        else:
            stagnant += 1

        if stagnant > 2000:
            if verbose:
                print(f"  SGD stagnant at iter {it}, best={best_score:.6f}, t={time.monotonic()-start_time:.1f}s")
            break

        weights, _ = soft_max_weights(conv, temperature=30.0, K=K)
        active_lags = np.where(weights > 1e-12)[0]
        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(conv)])
            w_vec = np.array([1.0])
        else:
            w_vec = weights[active_lags]

        subgrad = compute_weighted_subgradient(a, active_lags, w_vec)
        grad = subgrad - 2.0 * max_conv

        v = gamma * v - alpha * grad
        a_new = a + v
        a_new = np.maximum(0, a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            alpha *= 0.5
            v[:] = 0.0
            continue
        a_new = a_new / s_new

        if it % 20 == 0:
            check_score = score_sequence(a_new)
            if check_score > score * 1.05:
                alpha *= 0.5
                v[:] = 0.0
                a_new = a - 0.5 * alpha * grad
                a_new = np.maximum(0, a_new)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new /= s_new
                else:
                    continue
        a = a_new

        if stagnant > 500:
            alpha *= 1.05
        else:
            alpha *= 0.9995
        alpha = np.clip(alpha, 0.01, 3.0)

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"  SGD done: best={best_score:.6f}, iters={it+1}, t={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search(a, time_limit=60.0, verbose=True, rng=None):
    if rng is None:
        rng = np.random.default_rng(123)
    start_time = time.monotonic()
    n = len(a)
    a = normalize_nonneg(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"  LS start: score={best_score:.6f}")

    accepts = 0
    T = 0.005

    while time.monotonic() - start_time < time_limit:
        width = int(rng.integers(max(2, n // 200), max(3, n // 4)))
        start = int(rng.integers(0, max(1, n - width)))
        end = min(n, start + width)

        candidate = a.copy()
        noise = rng.normal(0.0, 0.15, size=end - start)
        candidate[start:end] *= np.exp(noise)

        if rng.random() < 0.3:
            t = np.arange(n, dtype=np.float64) / max(1, n - 1)
            candidate *= (1.0 + 0.08 * rng.normal(0, 1) * (2.0 * t - 1.0))

        if rng.random() < 0.2:
            candidate2 = a.copy()
            width2 = int(rng.integers(max(2, n // 100), max(3, n // 10)))
            start2 = int(rng.integers(0, max(1, n - width2)))
            end2 = min(n, start2 + width2)
            noise2 = rng.normal(0.0, 0.25, size=end2 - start2)
            candidate2[start2:end2] *= np.exp(noise2)
            candidate = 0.5 * (candidate + candidate2)

        candidate = normalize_nonneg(candidate)
        sc = score_sequence(candidate)

        delta = sc - best_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a = candidate
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
                accepts += 1
                if verbose and accepts % 100 == 0:
                    print(f"  LS accept {accepts}: score={best_score:.6f}")

        T *= 0.9995

    if verbose:
        print(f"  LS done: best={best_score:.6f}, t={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def make_init(n, init_type, rng):
    t = np.arange(n, dtype=np.float64)
    if init_type == 'tilted_uniform':
        a = 1.0 + 0.6 * (2.0 * t / max(1, n - 1) - 1.0)
        return normalize_nonneg(np.maximum(0, a))
    elif init_type == 'sin1':
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * t / n)
        return normalize_nonneg(np.maximum(0, a))
    elif init_type == 'sin2':
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * t / n) + 0.2 * np.sin(2.0 * np.pi * t / n)
        return normalize_nonneg(np.maximum(0, a))
    elif init_type == 'random_fourier':
        a = np.ones(n)
        for k in range(1, 4):
            freq = rng.integers(1, max(2, n // 10))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.sin(2.0 * np.pi * freq * t / n + phase)
        return normalize_nonneg(np.maximum(0, a))
    elif init_type == 'two_bump':
        a = np.zeros(n)
        w = max(3, n // 6)
        a[:w] = np.exp(-0.5 * (np.arange(w) / (w * 0.4)) ** 2)
        a[-w:] = np.exp(-0.5 * (np.arange(w) / (w * 0.4)) ** 2)
        s = a.sum()
        if s < 1e-15:
            return normalize_nonneg(np.ones(n))
        return a / s
    elif init_type == 'asym_gauss':
        center = n * rng.uniform(0.2, 0.8)
        sigma = n * rng.uniform(0.2, 0.4)
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        a *= (1.0 + 0.3 * (2.0 * t / max(1, n - 1) - 1.0))
        return normalize_nonneg(np.maximum(0, a))
    elif init_type == 'raised_tilted':
        x = t / max(1, n - 1)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x)) * (1.0 + 0.5 * x)
        return normalize_nonneg(np.maximum(0, a))
    else:
        a = 1.0 + 0.5 * (2.0 * t / max(1, n - 1) - 1.0)
        return normalize_nonneg(np.maximum(0, a))


def propose_candidate(budget_s=2400):
    total_start = time.monotonic()
    deadline = total_start + max(60.0, budget_s - 30.0)

    best_score = float('inf')
    best_seq = None

    rng = np.random.default_rng(42)

    # Try using height_sequence_1 if available
    try:
        hs1 = np.array(height_sequence_1, dtype=np.float64)
        if len(hs1) > 10:
            hs1 = normalize_nonneg(hs1)
            sc1 = score_sequence(hs1)
            print(f"Starting from height_sequence_1: score={sc1:.6f}, n={len(hs1)}")
            if sc1 < best_score:
                best_score = sc1
                best_seq = hs1.copy()
    except Exception:
        pass

    configs = [
        (500, 'tilted_uniform'),
        (500, 'sin1'),
        (500, 'random_fourier'),
        (800, 'tilted_uniform'),
        (800, 'sin2'),
        (800, 'asym_gauss'),
        (1200, 'tilted_uniform'),
        (1200, 'random_fourier'),
        (1500, 'tilted_uniform'),
        (1500, 'two_bump'),
        (2000, 'tilted_uniform'),
        (2000, 'sin1'),
        (1000, 'raised_tilted'),
    ]

    for n, init_type in configs:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        time_limit = min(250.0, remaining * 0.4)
        if time_limit < 20:
            continue

        a = make_init(n, init_type, rng)
        init_score = score_sequence(a)
        print(f"\n=== n={n}, init={init_type}, T={time_limit:.0f}s, init_score={init_score:.6f} ===")

        a_ref, sc = subgradient_descent(a, time_limit=time_limit, verbose=True)

        if sc < best_score:
            best_score = sc
            best_seq = a_ref.copy()
            print(f"  *** NEW BEST: score={sc:.6f}, n={n}, elapsed={time.monotonic()-total_start:.0f}s")

    # Polish phase
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n=== Polish phase (remaining={remaining:.0f}s) ===")
            pol, sc = local_search(best_seq, time_limit=remaining * 0.3, verbose=True, rng=rng)
            if sc < best_score:
                best_score = sc
                best_seq = pol
                print(f"  After LS: {best_score:.6f}")

            remaining = deadline - time.monotonic()
            if remaining > 30:
                pol2, sc2 = subgradient_descent(best_seq, time_limit=remaining * 0.4, verbose=True)
                if sc2 < best_score:
                    best_score = sc2
                    best_seq = pol2
                    print(f"  After SGD2: {best_score:.6f}")

            remaining = deadline - time.monotonic()
            if remaining > 20:
                pol3, sc3 = local_search(best_seq, time_limit=remaining * 0.5, verbose=True, rng=rng)
                if sc3 < best_score:
                    best_score = sc3
                    best_seq = pol3
                    print(f"  After LS2: {best_score:.6f}")

            # Final SGD with smaller steps
            remaining = deadline - time.monotonic()
            if remaining > 20:
                pol4, sc4 = subgradient_descent(best_seq, time_limit=remaining * 0.6, verbose=True)
                if sc4 < best_score:
                    best_score = sc4
                    best_seq = pol4
                    print(f"  After final SGD: {best_score:.6f}")

    if best_seq is None:
        best_seq = make_init(500, 'tilted_uniform', rng)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    print(f"\nFinal result: score={best_score:.6f}, n={len(best_seq)}")
    verify = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified via evaluate_sequence: {verify:.6f}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")

    return [float(x) for x in best_seq]
```
