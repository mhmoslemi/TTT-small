import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Full linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)
    return result[:size]

def compute_objective(a):
    """Compute 2*n*max(conv) / sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lp_direction(a, M):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns normalized b (sum=1).
    """
    n = len(a)
    # Build constraint matrix
    # For each k in [0, 2n-2], the constraint is: sum_i A[k,i] * b[i] <= M
    # where A[k,i] = a[k-i] if 0 <= k-i < n, else 0
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for i in range(n):
            idx = k - i
            if 0 <= idx < n:
                A_ub[k, i] = a[idx]
    # Maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        b = res.x
        b_sum = b.sum()
        if b_sum > 1e-15:
            b /= b_sum
            return b
    return None

def lp_direction_sparse(a, M, top_k=200):
    """
    Sparse version: only use top-k constraints.
    """
    n = len(a)
    c = fast_convolve(a)
    # Get top constraints
    top_indices = np.argsort(c)[::-1][:top_k]
    
    A_ub = np.zeros((len(top_indices), n))
    for ik, k in enumerate(top_indices):
        for i in range(n):
            idx = k - i
            if 0 <= idx < n:
                A_ub[ik, i] = a[idx]
    
    c_obj = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        b = res.x
        b_sum = b.sum()
        if b_sum > 1e-15:
            b /= b_sum
            return b
    return None

    n = len(a)
    # Compute gradient
    grad = np.zeros(n)

        if 0 <= idx < n:
            if i == idx:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Spectral shaping (iterative power spectrum compression)
2. LP-based direction finding (min-max cross-convolution)
3. Gradient-based refinement (FISTA with adaptive steps)
Multiple sequence lengths and initializations are explored.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Full linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)[:size]
    return result


def compute_obj_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_step(a, alpha=0.5, fft_size=None):
    """
    One spectral shaping step: compress the power spectrum.
    a should be non-negative, sum=1.
    """
    n = len(a)
    if fft_size is None:
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    P = np.abs(A) ** 2
    P_safe = np.maximum(P, 1e-30)
    # Compress: P_new = P^alpha (alpha < 1 reduces dynamic range)
    P_new = P_safe ** alpha
    # Scale to match DC (preserve sum)
    P_new = P_new * (P[0] / max(P_new[0], 1e-30))
    # New spectrum: keep phase, change magnitude
    A_new = A * (np.sqrt(P_new / P_safe))
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new /= s
    return a_new


def spectral_optimize(a0, time_limit, alphas=None, check_interval=50):
    """
    Iterative spectral shaping with varying alpha.
    """
    if alphas is None:
        alphas = [0.7, 0.5, 0.3, 0.2, 0.15]
    
    a = a0.copy()
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = 0
    alpha_idx = 0
    count_no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Apply spectral step
        alpha = alphas[alpha_idx % len(alphas)]
        a_new = spectral_step(a, alpha=alpha, fft_size=fft_size)
        
        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            count_no_improve = 0
        else:
            # Try with blended step

                a_blend = np.maximum(a_blend, 0)
                sb = a_blend.sum()
                if sb > 1e-15:
                    a_blend /= sb
                    cb = fast_convolve(a_blend)
                    ob = 2.0 * n * cb.max()
                    if ob < best_obj - 1e-12:
                        best_obj = ob
                        best_a = a_blend.copy()
                        a = a_blend
                        count_no_improve = 0
                        break
        
        if count_no_improve == 0:
            count_no_improve = 0
        else:
            count_no_improve += 1
        
        step += 1
        if step % 20 == 0:

            if count_no_improve > 100:

                a = a + noise
                a = np.maximum(a, 0)
                a /= a.sum()
                count_no_improve = 0
                alpha_idx = 0
    
    return best_a, best_obj


def lp_minmax_direction(a, top_k=0):
    """
    Solve: min t s.t. (a*b)[k] <= t for all k, b >= 0, sum(b) = 1.
    Returns (b, t_opt). If top_k > 0, only use top_k largest constraints.
    """
    n = len(a)
    c = fast_convolve(a)
    
    if top_k > 0 and top_k < len(c):
        # Use only top constraints
        top_indices = np.argsort(c)[::-1][:top_k]
    else:
        top_indices = np.arange(len(c))
    
    m = len(top_indices)
    # A_ub: m x (n+1), where the last column is for t (coefficient -1)
    A_ub = np.zeros((m, n + 1))
    for ik, k in enumerate(top_indices):
        for i in range(n):
            idx = k - i
            if 0 <= idx < n:
                A_ub[ik, i] = a[idx]
        A_ub[ik, n] = -1.0
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        b = res.x[:n]
        t_opt = res.x[n]
        b_sum = b.sum()
        if b_sum > 1e-15:
            b /= b_sum
        return b, t_opt
    return None, None


def lp_optimize(a0, time_limit, top_k=0):

    LP-based optimization: repeatedly find min-max direction and move toward it.
    """
    a = a0.copy()
    n = len(a)

    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    c = fast_convolve(a)
    M = c.max()

    best_a = a.copy()

    step = 0
    count_no_improve = 0
    t_param = 0.3
    
    while time.monotonic() - t0 < time_limit:
        # Solve LP
        b, t_opt = lp_minmax_direction(a, top_k=top_k)
        
        if b is None:
            # LP failed, try gradient step
            break
        
        # Move toward b
        a_new = (1 - t_param) * a + t_param * b
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new
        
        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            count_no_improve = 0
            # Can increase step size
            t_param = min(0.8, t_param * 1.05)
        else:

            improved = False
            for t_try in [0.1, 0.2, 0.3, 0.5]:
                a_try = (1 - t_try) * a + t_try * b

                st = a_try.sum()
                if st < 1e-15:
                    continue
                a_try /= st
                ct = fast_convolve(a_try)
                ot = 2.0 * n * ct.max()
                if ot < best_obj - 1e-12:
                    best_obj = ot
                    best_a = a_try.copy()
                    a = a_try
                    count_no_improve = 0
                    t_param = t_try
                    improved = True
                    break
            if not improved:
                count_no_improve += 1
                t_param = max(0.05, t_param * 0.7)
                if count_no_improve > 30:
                    # Add noise and reset

                    noise = rng.random(n) * 0.05
                    a = a + noise
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    count_no_improve = 0
                    t_param = 0.3
        
        step += 1
        if step % 50 == 0:
            print(f"    LP step {step}: obj={best_obj:.6f}, t_param={t_param:.3f}")
    
    return best_a, best_obj


def fista_optimize(a0, time_limit, max_steps=5000):
    """
    FISTA-style optimization with the analytical gradient of max-conv.
    """
    a = a0.copy()
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    k = 1.0  # FISTA momentum parameter
    y = a.copy()
    step_size = 1.0 / (4.0 * n)  # Initial step size
    count_no_improve = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient at y
        c_y = fast_convolve(y)
        k_star = np.argmax(c_y)
        
        # grad[i] = 4*n * a[k*-i] if 0 <= k*-i < n
        grad = np.zeros(n)
        for i in range(n):
            idx = k_star - i
            if 0 >= 0 and idx < n:
                grad[i] = 4.0 * n * y[idx]
        
        # FISTA update
        new_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k * k))
        z = y - step_size * grad
        y_new = z + ((k - 1.0) / new_k) * (y - a)
        a_new = simplex_project(y_new)
        
        k = new_k
        y = y_new
        a = a_new
        
        # Periodic check
        if step % 10 == 0:
            c_a = fast_convolve(a)
            obj_a = 2.0 * n * c_a.max()
            if obj_a < best_obj - 1e-12:
                best_obj = obj_a
                best_a = a.copy()
                count_no_improve = 0
            else:
                count_no_improve += 1
                if count_no_improve > 50:
                    # Adaptive step size
                    step_size *= 0.8
                    if step_size < 1e-6:
                        step_size = 1.0 / (4.0 * n)
                        k = 1.0
                        y = a.copy()
                        count_no_improve = 0
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Spectral shaping (iterative power spectrum compression) for fast initial progress
2. LP-based min-max direction finding for principled improvements
3. Combined refinement with adaptive parameters
Tries multiple sequence lengths and diverse initializations including parent sequence.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Full linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)[:size]
    return result


def compute_obj_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def simplex_project(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = (a * b)_k for linear convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def spectral_step(a, alpha, fft_size):
    """One spectral shaping step: compress the power spectrum."""
    n = len(a)
    A = rfft(a, fft_size)
    P = np.abs(A) ** 2
    P_safe = np.maximum(P, 1e-30)
    P_new = P_safe ** alpha
    # Preserve DC to maintain sum
    if P_new[0] > 1e-30:
        P_new = P_new * (P[0] / P_new[0])
    A_new = A * (np.sqrt(P_new / P_safe))
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new /= s
    return a_new


def spectral_optimize(a0, time_limit, alphas=None):
    """Iterative spectral shaping with cycling alpha values."""
    if alphas is None:
        alphas = [0.7, 0.5, 0.3, 0.2, 0.15, 0.1]
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    step = 0
    alpha_idx = 0
    stuck = 0
    rng = np.random.default_rng(12345 + n)
    while time.monotonic() - t0 < time_limit:
        alpha = alphas[alpha_idx % len(alphas)]
        a_new = spectral_step(a, alpha, fft_size)
        improved = False
        for t_blend in [1.0, 0.7, 0.5, 0.3]:
            a_try = (1.0 - t_blend) * a + t_blend * a_new
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try /= st
            ct = fast_convolve(a_try)
            ot = 2.0 * n * ct.max()
            if ot < best_obj - 1e-13:
                best_obj = ot
                best_a = a_try.copy()
                a = a_try
                stuck = 0
                improved = True
                break
        if not improved:
            stuck += 1
        step += 1
        if step % 10 == 0:
            alpha_idx += 1
        if stuck > 80:
            noise = rng.random(n) * 0.05
            a = a + noise
            a = np.maximum(a, 0.0)
            a /= a.sum()
            stuck = 0
            alpha_idx = 0
    return best_a, best_obj


def lp_solve(a, use_top_k=0):
    """
    Solve: min t s.t. (a*b)[k] <= t for all k, b >= 0, sum(b) = 1.
    Returns (b_normalized, t_opt) or (None, None).
    """
    n = len(a)
    c_full = fast_convolve(a)
    if use_top_k > 0 and use_top_k < len(c_full):
        top_idx = np.argsort(c_full)[::-1][:use_top_k]
    else:
        top_idx = np.arange(len(c_full))
    m = len(top_idx)
    A_ub = np.zeros((m, n + 1))
    for ik, k in enumerate(top_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[ik, i] = a[k - i]
        A_ub[ik, n] = -1.0
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            b = res.x[:n]
            t_opt = res.x[n]
            bs = b.sum()
            if bs > 1e-15:
                b /= bs
            return b, t_opt
    except Exception:
        pass
    return None, None


def lp_optimize(a0, time_limit, top_k=0):
    """LP-based iterative optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    step = 0
    t_param = 0.4
    stuck = 0
    rng = np.random.default_rng(99999 + n)
    while time.monotonic() - t0 < time_limit:
        b, t_opt = lp_solve(a, top_k=top_k)
        if b is None:
            break
        improved = False
        for t_try in [t_param, min(0.8, t_param * 1.2), max(0.05, t_param * 0.6), 0.2, 0.4, 0.6]:
            a_try = (1.0 - t_try) * a + t_try * b
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try /= st
            ct = fast_convolve(a_try)
            ot = 2.0 * n * ct.max()
            if ot < best_obj - 1e-13:
                best_obj = ot
                best_a = a_try.copy()
                a = a_try
                stuck = 0
                improved = True
                t_param = min(0.8, t_try * 1.1)
                break
        if not improved:
            stuck += 1
            t_param = max(0.05, t_param * 0.8)
            if stuck > 40:
                noise = rng.random(n) * 0.03
                a = a + noise
                a = np.maximum(a, 0.0)
                a /= a.sum()
                stuck = 0
                t_param = 0.4
        step += 1
        if step % 100 == 0:
            print(f"      LP step {step}: obj={best_obj:.8f}")
    return best_a, best_obj


def gradient_optimize(a0, time_limit):
    """Gradient-based optimization with adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    step_size = 1.0 / (8.0 * n)
    stuck = 0
    momentum = 0.9
    v = np.zeros(n)
    while time.monotonic() - t0 < time_limit:
        c_a = fast_convolve(a)
        k_star = np.argmax(c_a)
        # Gradient of 2n*max(conv): grad[i] = 4n * a[k*-i] (when valid)
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        for i in range(i_start, i_end + 1):
            grad[i] = 4.0 * n * a[k_star - i]
        # Center gradient (tangent to simplex)
        grad -= grad.mean()
        v = momentum * v - step_size * grad
        a_new = simplex_project(a + v)
        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stuck = 0
        else:
            stuck += 1
            if stuck % 20 == 0:
                step_size *= 0.7
                if step_size < 1e-8:
                    step_size = 1.0 / (8.0 * n)
                    v = np.zeros(n)
                    stuck = 0
    return best_a, best_obj


def combined_optimize(a0, time_limit):
    """Alternate between spectral and LP optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    best_obj = float('inf')
    best_a = a.copy()
    t0 = time.monotonic()
    phase = 0
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        if phase % 3 == 0:
            # Spectral phase
            t_alloc = min(remaining * 0.15, 30.0)
            a_opt, obj_opt = spectral_optimize(a, time_limit=t_alloc)
            a = a_opt
            phase += 1
        elif phase % 3 == 1:
            # LP phase
            t_alloc = min(remaining * 0.3, 120.0)
            tk = 0 if n <= 256 else min(200, 2 * n)
            a_opt, obj_opt = lp_optimize(a, time_limit=t_alloc, top_k=tk)
            a = a_opt
            phase += 1
        else:
            # Gradient phase
            t_alloc = min(remaining * 0.1, 20.0)
            a_opt, obj_opt = gradient_optimize(a, time_limit=t_alloc)
            a = a_opt
            phase += 1
        c_cur = fast_convolve(a)
        obj_cur = 2.0 * n * c_cur.max()
        if obj_cur < best_obj:
            best_obj = obj_cur
            best_a = a.copy()
        if phase % 3 == 2:
            print(f"      Combined: obj={best_obj:.8f}, remaining={remaining:.0f}s")
    return best_a, best_obj


def get_initializations(n, seed, parent=None):
    """Generate diverse initializations for sequence of length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    bi = np.zeros(n)
    bi[:max(1, n // 4)] = 1.0
    bi[-max(1, n // 4):] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    cos_b = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_b /= cos_b.sum()
    inits['cosine'] = cos_b
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits['random1'] = r
    r2 = rng.dirichlet(np.ones(n) * 0.1)
    inits['random2'] = r2
    # Bump in center
    bump = np.exp(-((x - n * 0.3) / (0.1 * n)) ** 2) + np.exp(-((x - n * 0.7) / (0.1 * n)) ** 2)
    bump /= bump.sum()
    inits['double_bump'] = bump
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp
    return inits
