"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated mirror descent
with FFT-based exact subgradients and adaptive temperature annealing.
Significantly faster than naive approaches due to O(n log n) subgradient computation.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_objective(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_subgradient(a, conv, temperature):
    """
    Compute weighted subgradient using FFT cross-correlation trick.
    sg[i] = 2 * sum_j a[j] * w[i+j] where w is softmax weight over conv peaks.
    """
    n = len(a)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return np.zeros(n)
    
    # Soft-max weights
    scaled = temperature * (conv - max_conv) / max_conv
    scaled -= scaled.max()  # for numerical stability (it's already <= 0)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # FFT cross-correlation: sg[i] = 2 * sum_j a[j] * w[i+j]
    # = 2 * conv_ab[n-1:2n-1][::-1] where conv_ab = ifft(fft(a)*fft(w[::-1]))
    b = w[::-1]
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A_hat = np.fft.rfft(a, fft_size)
    B_hat = np.fft.rfft(b, fft_size)
    conv_ab = np.fft.irfft(A_hat * B_hat, fft_size)[:2 * n - 1]
    sg = 2.0 * conv_ab[n - 1:2 * n - 1][::-1]
    
    return sg


def mirror_descent(a0, time_limit, temperature_init=50.0, temperature_final=5.0,
                   eta_init=1.0, eta_min=0.01, decay_schedule='exp'):
    """
    Accelerated mirror descent on the probability simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a_sum
    
    conv = fast_convolve(a)
    best_obj = compute_objective(a)
    best_a = a.copy()
    
    t = time.monotonic()
    t0 = t
    total_time = time_limit
    
    # Parameters
    eta = eta_init
    temperature = temperature_init
    patience = 0
    max_patience = 50
    momentum = np.zeros(n)
    beta = 0.9
    
    iteration = 0
    while time.monotonic() - t0 < total_time:
        # Adaptive temperature: anneal from high to low
        progress = min(1.0, (time.monotonic() - t0) / total_time)
        if decay_schedule == 'exp':
            temperature = temperature_init * (temperature_final / temperature_init) ** progress
        else:
            temperature = temperature_init * (1 - progress) + temperature_final * progress
        
        # Compute subgradient
        sg = compute_subgradient(a, conv, temperature)
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * sg
        
        # Multiplicative update
        # Center the subgradient to preserve the mean (helps with convergence)
        sg_centered = momentum - momentum.mean()
        
        # Step size: use a heuristic based on the norm
        norm_sg = np.linalg.norm(sg_centered)
        if norm_sg < 1e-15:
            break
        step = eta / norm_sg  # normalized step
        
        log_factor = -step * sg_centered
        log_factor -= log_factor.max()
        a_new = a * np.exp(log_factor)
        a_new = np.maximum(a_new, 1e-30)  # avoid underflow
        a_sum = a_new.sum()
        if a_sum < 1e-12:
            a_new = np.ones(n) / n
        else:
            a_new /= a_sum
        
        # Evaluate
        conv = fast_convolve(a_new)
        current_obj = compute_objective(a_new)
        
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        iteration += 1
        
        # Adapt step size
        if patience >= max_patience:
            patience = 0
            eta *= 0.7
            if eta < eta_min:
                eta = eta_init * 0.1
                momentum *= 0.5
                max_patience = 30
            # Occasionally reheat temperature
            if iteration % 500 == 0:
                temperature = min(temperature_init, temperature * 1.5)
    
    return best_a, best_obj
