def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Key improvements over previous version:
1. Smooth gradient via soft-max approximation (replaces top-k subgradient)
2. FFT-based gradient computation (cross-correlation of softmax weights with a)
3. Adaptive beta schedule (smooth -> sharp) for better convergence
4. Multiple optimization methods: Adam, mirror descent, and exact subgradient polishing
5. Better diversity: multiple n values, structured initializations
6. Restart mechanism for escaping local minima
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.argmax(valid)) + 1
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_smooth_grad(a, w):
    """
    Gradient of soft-max objective w.r.t. a.
    g[i] = 2 * sum_j a[j] * w[i+j]
    Computed as 2 * conv(a_flipped, w)[n-1 : 2n-1]
    """
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    a_flip = a[::-1]
    Af = fft(a_flip, fsize)
    W = fft(w, fsize)
    conv = np.real(ifft(Af * W))[:size]
    return 2.0 * conv[n - 1:2 * n - 1]


def compute_exact_subgrad(a, c, n, eps_rel=1e-6):
    """
    Exact subgradient: average of 2*a[k-i] over active lags k where c[k] >= max_c - eps.
    """
    max_c = c.max()
    eps = eps_rel * max_c
    active = np.where(c >= max_c - eps)[0]
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * a[k - i]
    grad /= len(active)
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights for the soft-max approximation."""
    bc = beta * c
    bc_max = bc.max()
    w = np.exp(bc - bc_max)
    s = w.sum()
    if s < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w /= s
    return w


def normalize_to_sum(a, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        a = np.ones(len(a)) * S / len(a)
    else:
        a = a * (S / s)
    return a

                    lr_scale=1.0, restart_every=3000):

    a = normalize_to_sum(a_init, S)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = lr_scale * S / n

    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    no_improve = 0

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive beta
        progress = min(1.0, t / 4000.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        w = softmax_weights(c, beta)
        grad = compute_smooth_grad(a, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = base_lr * (0.5 ** (t / 3000.0))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new, S)

        # Restart mechanism
        if no_improve > restart_every and t % (restart_every // 2) == 0:
            noise_scale = 0.03 * S / np.sqrt(n)
            a_new = project_simplex(a_new + rng.normal(0.0, noise_scale, n), S)
            no_improve = 0

        a = a_new

    return best_a, best_score


def run_mirror_smooth(a_init, n, S, max_time, rng, eta_start=0.05,
                      beta_start=10.0, beta_end=500.0):
    """Mirror descent with smooth gradient."""
    a = normalize_to_sum(a_init, S)
    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    t = 0

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()

        progress = min(1.0, t / 3000.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        w = softmax_weights(c, beta)
        grad = compute_smooth_grad(a, w)

        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break
        grad = grad / gmax

        eta = eta_start / (1.0 + 0.003 * t)
        a_new = a * np.exp(-eta * grad)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.full(n, S / n)
        else:
            a = a_new * (S / s_new)

    return best_a, best_score


def run_exact_polish(a_init, n, S, max_time, rng, eps_rel=1e-4):
    """Polishing with exact subgradient at the active lags."""
    a = normalize_to_sum(a_init, S)
    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    t = 0
    lr = S / n * 0.5

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()

        grad = compute_exact_subgrad(a, c, n, eps_rel=eps_rel)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        a_new = a - lr * grad
        a_new = project_simplex(a_new, S)

        lr *= 0.9995  # Slow decay

        if t % 1000 == 0:
            noise = rng.normal(0.0, 0.01 * S / np.sqrt(n), n)
            a_new = project_simplex(a_new + noise, S)

        a = a_new

    return best_a, best_score
