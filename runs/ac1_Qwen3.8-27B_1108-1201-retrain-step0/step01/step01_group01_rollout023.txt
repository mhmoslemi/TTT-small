Multi-phase optimization using log-sum-exp smoothing and L-BFGS-B to minimize
the normalized autocorrelation peak of a non-negative sequence.

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

def _fast_convolve(x):
    n = len(x)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fx = rfft(x, fsize)
    return irfft(fx * fx, fsize)[:size]

def _score_from_b(b, n):
    return 2.0 * n * np.max(b)

def _normalize(a):
    s = np.sum(a)
    if s < 1e-15:
        return a
    return a / s

def _smoothed_obj_and_grad(x, tau):
    b = _fast_convolve(x)
    b_max = np.max(b)
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    log_sum = b_max + np.log(np.sum(exp_vals)) / tau
    w = exp_vals / np.sum(exp_vals)
    
    n = len(x)
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += w[k] * 2.0 * x[k - lo : hi + 1][::-1]
    return log_sum, grad

def _optimize_smoothed(a, tau, deadline, maxiter=200):
    """Optimize using L-BFGS-B on the smoothed objective."""
    n = len(a)
    
    def fun(x):
        val, _ = _smoothed_obj_and_grad(x, tau)
        return val
    
    def jac(x):
        _, g = _smoothed_obj_and_grad(x, tau)
        return g
    
    bounds = [(0, None)] * n
    
    x0 = a.copy()
    start_time = time.monotonic()
    
    result = minimize(
        fun, x0, jac=jac, method='L-BFGS-B',
        bounds=bounds, options={'maxiter': maxiter, 'maxfun': 5000,
                                 'ftol': 1e-12, 'gtol': 1e-8, 'factr': 1e10}
    )
    
    elapsed = time.monotonic() - start_time
