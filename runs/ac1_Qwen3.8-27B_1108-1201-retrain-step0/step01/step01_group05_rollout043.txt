def multiplicative_step(a, grad, eta, S, min_val, clip=6.0):
    g = grad - grad.mean()
    gmax = float(np.max(np.abs(g)))
    if not np.isfinite(gmax) or gmax < 1e-15:
        return None, False
    g = g / gmax

    delta = np.clip(delta, -clip, clip)
    cand = a * np.exp(delta)
    cand = np.maximum(cand, min_val)
    s = cand.sum()

        return None, False

```python
"""
Search for a non-negative sequence a minimizing
    2*n*max(a*a) / (sum(a))^2
where a*a is linear convolution.

The implementation uses:
  * FFT-based evaluation of the quadratic coefficients,
  * several analytically motivated non-negative starts (binomial / power-law /
    modulated profiles, plus the supplied previous construction if available),
  * a smoothed multiplicative-descent dynamics on the simplex,
  * adaptive beta (smoothing), adaptive step, and occasional gentle
    perturbations / blending to escape shallow local minima,
  * multiple continuation phases (warm-up, long run, optional second basin,
    final polish).
"""

import time
import numpy as np
from scipy.fft import fft, ifft


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_profile(p):
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    s = float(p.sum())
    if s <= 1e-300:
        return np.full(p.size, 1.0 / max(1, p.size), dtype=np.float64)
    return p / s


def prepare_sequence(arr, n, S, min_val=0.0):
    a = np.asarray(arr, dtype=np.float64).reshape(-1)
    if a.size != n:
        if a.size > n:
            a = a[:n]
        else:
            tmp = np.zeros(n, dtype=np.float64)
            if a.size > 0:
                tmp[:a.size] = a
            a = tmp
    a = np.clip(a, 0.0, 1000.0)
    a = np.where(np.isfinite(a), a, 0.0)
    if min_val > 0.0:
        a = np.maximum(a, min_val)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        a = np.full(n, S / float(n), dtype=np.float64)
    else:
        a = a * (S / s)
    return a


def convolve_self(a, size):
    n = len(a)
    A = fft(a, size)
    c = np.real(ifft(A * A))
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def stable_lse(c, score, beta):
    if beta <= 0.0 or not np.isfinite(score):
        return float(score)
    ex = (c - score) / beta
    ex = np.where(ex < -700.0, -700.0, ex)
    s = float(np.exp(ex).sum())
    if not np.isfinite(s) or s <= 0.0:
        return float(score)
    return float(score + beta * np.log(s))


def compute_c_score_grad(a, beta, main_size):
    n = len(a)
    if n == 1:
        c = np.array([a[0] * a[0]], dtype=np.float64)
        score = float(c[0])
        lse = score
        phi = np.array([1.0], dtype=np.float64)
        grad = np.array([2.0 * a[0]], dtype=np.float64)
        return c, score, lse, phi, grad

    A = fft(a, main_size)
    c = np.real(ifft(A * A))
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)

    score = float(c.max())
    if not np.isfinite(score):
        raise FloatingPointError("non-finite convolution score")

    ex = (c - score) / beta
    ex = np.where(ex < -700.0, -700.0, ex)
    w = np.exp(ex)
    wsum = float(w.sum())
    if not np.isfinite(wsum) or wsum <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        wsum = 1.0
    else:
        w = w / wsum

    lse = float(score + beta * np.log(wsum))

    p = fft(w[::-1], main_size)
    gfull = np.real(ifft(A * p))
    grad = 2.0 * gfull[2 * n - 2: n - 2: -1]
    grad = np.where(np.abs(grad) < 1e-300, 0.0, grad)
    return c, score, lse, w, grad


def multiplicative_step(a, grad, eta, S, min_val, clip=6.0):
    gm = float(grad.mean())
    if not np.isfinite(gm) or gm <= 1e-300:
        g = grad.copy()
    else:
        g = grad / gm - 1.0

    gmax = float(np.max(np.abs(g)))
    if not np.isfinite(gmax) or gmax < 1e-15:
        return None, False

    g = g / gmax
    delta = np.clip(-eta * g, -clip, clip)
    cand = a * np.exp(delta)
    cand = np.maximum(cand, min_val)

    s = float(cand.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None, False

    return cand * (S / s), True


def central_binomial(n):
    b = np.empty(n, dtype=np.float64)
    if n <= 0:
        return b
    b[0] = 1.0
    for i in range(n - 1):
        b[i + 1] = b[i] * (2.0 * i + 1.0) / (2.0 * i + 2.0)
    return b


def profile_power(n, alpha):
    i = np.arange(n, dtype=np.float64) + 1.0
    return i ** (-alpha)


def get_given_sequence(n):
    try:
        raw = height_sequence_1
    except Exception:
        return None
    try:
        arr = np.asarray(raw, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.size == n:
        return arr
    if arr.size > 1:
        old = np.linspace(0.0, 1.0, arr.size)
        new = np.linspace(0.0, 1.0, n)
        return np.interp(new, old, arr)
    return np.full(n, float(arr[0]), dtype=np.float64)


def make_starts(n, given):
    starts = []
    b_unit = normalize_profile(central_binomial(n))
    starts.append(("binom", b_unit))

    alphas = (0.35, 0.45, 0.55)
    for idx, alpha in enumerate(alphas):
        p = normalize_profile(profile_power(n, alpha))
        starts.append(("pow_%d" % idx, p))

    rev = b_unit[::-1]
    starts.append(("binom_rev", rev))
    starts.append(("mix_fw_bw", 0.5 * b_unit + 0.5 * rev))

    if given is not None:
        g_unit = normalize_profile(given)
        starts.append(("given", g_unit))
        starts.append(("given_binom", 0.70 * g_unit + 0.30 * b_unit))
        starts.append(("given_mix", 0.50 * g_unit + 0.25 * b_unit + 0.25 * rev))

    t = np.linspace(0.0, 1.0, n)
    wave_specs = ((1, 0.25), (2, 0.25), (3, 0.20))
    for idx, (freq, amp) in enumerate(wave_specs):
        wave = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
        wave = np.maximum(wave, 0.05)
        starts.append(("wave_%d" % idx, normalize_profile(b_unit * wave)))

    return starts


def run_smooth_descent(a_init, n, time_limit, rng, beta0=0.2, beta_min=0.02,
                       eta0=0.5, profile=None, max_gap=0.02, verbose=False):
    if time_limit <= 0.0:
        a = np.asarray(a_init, dtype=np.float64).reshape(-1)
        if a.size == n:
            S = np.sqrt(2.0 * float(n))
            return prepare_sequence(a, n, S), 1e300
        return a_init, 1e300

    S = np.sqrt(2.0 * float(n))
    min_val = 1e-12 * S / float(n)
    a = prepare_sequence(a_init, n, S, min_val)

    score_size = next_pow2(2 * n - 1)
    main_size = next_pow2(3 * n - 2)

    fallback_profile = normalize_profile(central_binomial(n))
    if profile is not None:
        profile = normalize_profile(profile)
    else:
        profile = fallback_profile

    c, score, lse, phi, grad = compute_c_score_grad(a, beta0, main_size)
    best_score = float(score)
    best_a = a.copy()

    start = time.monotonic()
    end = start + max(0.0, time_limit) - 1.0
    eta = max(0.02, float(eta0))
    beta = float(beta0)
    beta_tau = 300.0

    stall = 0
    last_best = start
    next_print = start + 20.0
    it = 0

    while time.monotonic() < end:
        now = time.monotonic()
        elapsed = max(0.0, now - start)

        try:
            c, score, lse, phi, grad = compute_c_score_grad(a, beta, main_size)
        except Exception:
            a = prepare_sequence(fallback_profile, n, S, min_val)
            continue

        if not np.isfinite(score):
            a = prepare_sequence(fallback_profile, n, S, min_val)
            continue

        if score < best_score - 1e-12:
            best_score = float(score)
            best_a = a.copy()
            last_best = now
            stall = 0
        else:
            stall += 1

        improved = False

        # One modest step, and one backup smaller step.
        for mult in (1.0, 0.5):
            cand, ok = multiplicative_step(a, grad, eta * mult, S, min_val)
            if not ok:
                break

            cand_c = convolve_self(cand, score_size)
            cand_score = float(cand_c.max())
            if not np.isfinite(cand_score):
                continue

            if cand_score < score - 1e-12:
                a = cand
                improved = True
                eta = min(2.0, eta * 1.08)
                break

            cand_lse = stable_lse(cand_c, cand_score, beta)
            if cand_lse < lse - 1e-10 and cand_score < best_score + max_gap:
                a = cand
                improved = True
                eta = min(2.0, eta * 1.03)
                break

        if improved:
            stall = max(0, stall - 1)
        else:
            eta = max(0.005, eta * 0.5)

        # Adaptive smoothing: if badly stuck, temporarily smooth more.
        scheduled_beta = max(beta_min, beta0 * np.exp(-elapsed / beta_tau))
        if stall > 150:
            beta = max(scheduled_beta, min(0.08, beta + 0.01))
        else:
            beta = max(scheduled_beta, beta * 0.999)

        # Escape/shake-up.
        if stall > 220 and (now - last_best) > 15.0:
            tried = False
            if rng.random() < 0.65:
                a_unit = a / S
                blend = 0.85 * a_unit + 0.15 * profile
                blend = prepare_sequence(blend, n, S, min_val)
                bc = float(convolve_self(blend, score_size).max())
                if np.isfinite(bc) and bc < best_score + 0.03:
                    a = blend
                    tried = True
                    stall = max(0, stall - 50)
            if not tried:
                a_unit = a / S
                noise = rng.normal(0.0, 1.0, n) * a_unit
                blend = a_unit + 0.015 * noise
                blend = prepare_sequence(blend, n, S, min_val)
                bc = float(convolve_self(blend, score_size).max())
                if np.isfinite(bc) and bc < best_score + 0.03:
                    a = blend
                    tried = True
                    stall = max(0, stall - 50)
            if not tried:
                pass

        it += 1
        if verbose and now >= next_print:
            print("smooth n=%d it=%d score=%.8f best=%.8f eta=%.4f beta=%.5f stall=%d"
                  % (n, it, score, best_score, eta, beta, stall))
            next_print = now + 20.0

    return best_a, best_score


def optimize_for_n(n, end_time, given, rng):
    start = time.monotonic()
    remaining = max(0.0, end_time - start)

    S = np.sqrt(2.0 * float(n))
    score_size = next_pow2(2 * n - 1)
    fallback = prepare_sequence(central_binomial(n), n, S)
    fallback_score = float(convolve_self(fallback, score_size).max())

    if remaining < 90.0:
        return fallback, fallback_score

    # Budgets tuned for ~2300s, scaled down if time is short.
    scale = min(1.0, remaining / 2300.0)
    warm_budget = max(20.0, 200.0 * scale)
    long_budget = max(250.0, 1150.0 * scale)
    sec_budget = max(0.0, 450.0 * scale)
    polish_budget = max(120.0, 500.0 * scale)

    allocated = warm_budget + long_budget + sec_budget + polish_budget
    if allocated > remaining - 20.0:
        factor = max(0.05, (remaining - 20.0) / max(1.0, allocated))
        warm_budget *= factor
        long_budget *= factor
        sec_budget *= factor
        polish_budget *= factor

    starts = make_starts(n, given)

    scored = []
    for name, prof in starts:
        a = prepare_sequence(prof, n, S)
        try:
            c = convolve_self(a, score_size)
            sc = float(c.max())
        except Exception:
            sc = 1e300
        if np.isfinite(sc):
            scored.append((sc, name, a))

    if not scored:
        return fallback, fallback_score

    scored.sort()
    keep = scored[:6]

    perturb_profile = normalize_profile(given) if given is not None else normalize_profile(central_binomial(n))

    results = []
    per_warm = warm_budget / max(1, len(keep))
    reserve_after_warm = long_budget + sec_budget + polish_budget + 30.0

    for sc, name, a in keep:
        now = time.monotonic()
        available = end_time - now - reserve_after_warm
        if available < 8.0:
            break
        lim = min(per_warm, 80.0, available)
        if lim < 8.0:
            break
        b, s = run_smooth_descent(
            a, n, lim, rng,
            beta0=0.2, beta_min=0.02,
            eta0=0.5, profile=perturb_profile,
            max_gap=0.03, verbose=False
        )
        results.append((float(s), b, "warm_" + name))
        print("warm %-18s score=%.8f" % (name, s))

    if not results:
        sc, name, a = scored[0]
        return a, float(sc)

    results.sort()
    best_score, best_a, best_name = results[0]

    # Long continuation from the best warm state.
    now = time.monotonic()
    long_lim = min(long_budget, end_time - now - sec_budget - polish_budget - 20.0)
    if long_lim > 30.0:
        a2, s2 = run_smooth_descent(
            best_a, n, long_lim, rng,
            beta0=0.05, beta_min=0.003,
            eta0=0.4, profile=perturb_profile,
            max_gap=0.008, verbose=True
        )
        if float(s2) < best_score:
            best_a, best_score = a2, float(s2)
        print("long best_score=%.8f" % best_score)

    # Optional secondary basin: if a second warm candidate was close, give it
    # some time too.
    used_secondary = False
    if len(results) >= 2:
        second_score, second_a, second_name = results[1]
        if second_score <= best_score + 0.010:
            now = time.monotonic()
            maximize_sec = max(60.0, 0.5 * polish_budget)
            sec_lim = min(sec_budget, end_time - now - maximize_sec - 10.0)
            if sec_lim > 40.0:
                a3, s3 = run_smooth_descent(
                    second_a, n, sec_lim, rng,
                    beta0=0.05, beta_min=0.003,
                    eta0=0.4, profile=perturb_profile,
                    max_gap=0.008, verbose=False
                )
                used_secondary = True
                if float(s3) < best_score:
                    best_a, best_score = a3, float(s3)
                print("secondary best_score=%.8f" % best_score)

    # Final polish: use whatever real time remains.
    now = time.monotonic()
    polish_lim = max(0.0, end_time - now - 8.0)
    if polish_lim > 30.0:
        a4, s4 = run_smooth_descent(
            best_a, n, polish_lim, rng,
            beta0=0.02, beta_min=0.001,
            eta0=0.25, profile=perturb_profile,
            max_gap=0.003, verbose=True
        )
        if float(s4) < best_score:
            best_a, best_score = a4, float(s4)
        print("polished best_score=%.8f" % best_score)

    best_a = np.asarray(best_a, dtype=np.float64).reshape(-1)
    if best_a.size != n or not np.all(np.isfinite(best_a)):
        return fallback, fallback_score

    return best_a, float(best_score)


def propose_candidate(seed=42, budget_s=2350):
    start = time.monotonic()
    remaining_budget = max(300.0, float(budget_s) - 60.0)
    hard_end = start + min(2350.0, remaining_budget)

    rng = np.random.default_rng(seed)
    n = 8192
    given = get_given_sequence(n)

    fallback = np.ones(n, dtype=np.float64)
    try:
        a, score = optimize_for_n(n, hard_end, given, rng)
        a = np.asarray(a, dtype=np.float64).reshape(-1)
        if a.size != n or not np.all(np.isfinite(a)):
            a = fallback
    except Exception:
        a = fallback

    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        a = np.ones(256, dtype=np.float64)

    return [float(x) for x in a]
```
