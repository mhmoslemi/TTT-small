    theta = (cumsum[rho-1] - S) / rho

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_full(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_smooth_grad(a, c, tau):
    """
    Compute exact gradient of tau*logsumexp(c/tau) w.r.t. a.
    Uses FFT-based cross-correlation for efficiency.
    grad_a[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    which equals 2 * convolve(p, a_reversed)[n-1 : 2n-1]
    """
    n = len(a)
    # Softmax weights
    c_scaled = c / tau
    c_max = c_scaled.max()
    p = np.exp(c_scaled - c_max)
    p_sum = p.sum()
    if p_sum <= 0:
        return np.zeros(n)
    p = p / p_sum

    # Cross-correlation via FFT convolution
    ar = a[::-1].copy()
    conv_len = (2 * n - 1) + n - 1  # = 3n - 2
    fsize = 1 << (conv_len - 1).bit_length()
    P = fft(p, fsize)
    AR = fft(ar, fsize)
    conv = np.real(ifft(P * AR))

    # Extract the right slice
    grad = 2.0 * conv[n - 1: 2 * n - 1]
    return grad

    if total <= 1e-15:
        return np.zeros(n)
    weights = top_vals / total
    grad = np.zeros(n)
    for w_val, k in zip(weights, top_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            segment = a[k - hi: k - lo + 1][::-1]
            grad[lo:hi + 1] += w_val * 2.0 * segment
    return grad


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cands = (cumsum - S) / idx
    valid = theta_cands <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0])) + 1
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clip to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def run_smooth_adam(a_init, n, max_time, tau_start, tau_end, lr_init,
                    anneal_frac=0.6, top_k_fallback=30, rng=None):
    """
    Adam optimizer with smoothed logsumexp gradient.
    Anneals tau from tau_start to tau_end over anneal_frac of the budget.
    Returns (best_a, best_maxc).
    """
    S = np.sqrt(2.0 * n)
    a = safe_normalize(a_init, n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    best_a = a.copy()
    best_val = np.inf
    start = time.monotonic()
    stall_count = 0
    stall_threshold = 500

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_full(a)
        cur = c.max()
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Anneal tau
        frac = min(1.0, float(t) / (max_time / anneal_frac) if max_time > 0 else 1.0)
        # Use iterations-based annealing normalized to max_time/anneal_frac
        # Better: use time-based
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / (max_time * anneal_frac))
        tau = tau_start * (tau_end / tau_start) ** frac

        # Compute gradient
        if tau > 0.05:
            grad = compute_smooth_grad(a, c, tau)
        else:
            grad = compute_topk_subgrad(a, c, top_k=top_k_fallback)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-12:
                grad = grad / gnorm

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

        # Slow learning rate decay
        lr = lr_init * max(0.1, 1.0 / (1.0 + 0.002 * t))

        # Restart with perturbation if stalled
        if stall_count > stall_threshold and rng is not None:
            a = project_simplex(a + rng.normal(0, 0.02 * S / n, n), S)
            stall_count = 0
            m = np.zeros(n)
            v = np.zeros(n)

    return best_a, best_val


def run_logparam_adam(a_init, n, max_time, tau_start, tau_end, lr_init,
                      anneal_frac=0.5, rng=None):
    """
    Adam in log-parameterization: a_j = S * exp(x_j) / sum(exp(x_k)).
    No projection needed - non-negativity is automatic.
    """
    S = np.sqrt(2.0 * n)
    a = safe_normalize(a_init, n)
    # Initialize x from a: x_j = log(a_j / S) (with floor for zeros)
    x = np.log(np.maximum(a / S, 1e-3))
    # Center x (gauge freedom)
    x = x - x.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    best_a = a.copy()
    best_val = a[0] * 0  # dummy
    c = fft_convolve_full(a)
    best_val = c.max()

    start = time.monotonic()

    while time.monotonic() - start < max_time:
        t += 1
        # Recover a from x
        x_shifted = x - x.max()  # numerical stability
        p = np.exp(x_shifted)
        p = p / p.sum()
        a = S * p

        c = fft_convolve_full(a)
        cur = c.max()
        if cur < best_val:
            best_val = cur
            best_a = a.copy()

        # Anneal tau
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / (max_time * anneal_frac))
        tau = tau_start * (tau_end / tau_start) ** frac

        # Compute grad wrt a
        if tau > 0.05:
            grad_a = compute_smooth_grad(a, c, tau)
        else:
            grad_a = compute_topk_subgrad(a, c, top_k=30)
            gn = np.linalg.norm(grad_a)
            if gn > 1e-12:
                grad_a = grad_a / gn

        # Transform to grad wrt x:
        # df/dx_m = a_m * (df/da_m - (1/S) * <grad_a, a>)
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot / S)

        # Adam update on x
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - x.mean()  # re-center for gauge

        lr = lr_init * max(0.1, 1.0 / (1.0 + 0.003 * t))

    return best_a, best_val


def make_spectral_init(n, rng):
    """
    Create an initialization with approximately flat power spectrum.
    Use random phases in frequency domain, take inverse DFT, take abs.
    """
    fsize = 1 << (n - 1).bit_length()
    # Random phases with conjugate symmetry
    phases = rng.uniform(0, 2 * np.pi, fsize)
    # Conjugate symmetry: X[k] = conj(X[fsize-k])
    X = np.exp(1j * phases)
    X[0] = 1.0  # DC
    if fsize % 2 == 0:
        X[fsize // 2] = 1.0
    # Enforce conjugate symmetry
    for k in range(1, fsize // 2 + 1):
        X[fsize - k] = np.conj(X[k])

    a = np.real(ifft(X, fsize))[:n]
    a = np.abs(a)
    a = np.maximum(a, 0.0)
    return a


def make_smooth_init(n, rng):
    """Smooth sinusoidal initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for freq in [2, 3, 5, 7]:
        amp = 0.15 / np.sqrt(freq)
        phase = rng.uniform(0, 2 * np.pi)
        a = a + amp * np.sin(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.1)
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(50.0, budget_s - 10.0)

    # Collect initializations
    inits = []

    # Given sequence (previous best)
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64)
        given = given.reshape(-1)
        if given.size >= 16 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given_8192", np.clip(given, 0, 1000)))
    except Exception:
        pass

    # Try to get height_sequence_1 for different lengths
    n_given = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64)
        given = given.reshape(-1)
        n_given = given.size
    except Exception:
        pass

    # Spectral inits
    inits.append(("spectral_8192", make_spectral_init(8192, rng)))
    inits.append(("spectral_16k", make_spectral_init(16384, rng)))

    # Smooth inits
    inits.append(("smooth_8192", make_smooth_init(8192, rng)))
    inits.append(("smooth_16k", make_smooth_init(16384, rng)))

    # Random with low-pass
    for n_try in [8192, 16384]:
        raw = rng.random(n_try)
        # Simple low-pass via repeated averaging
        for _ in range(3):
            raw = 0.5 * (raw + np.roll(raw, 1))
        inits.append(("lowpass_%d" % n_try, np.maximum(raw, 0.01)))

    print("Number of initializations: %d" % len(inits))
    for name, arr in inits:
        print("  %s: len=%d" % (name, len(arr)))

    global_best = np.inf
    global_a = None

    # Phase 1: Explore each initialization
    usable = max(10.0, deadline - time.monotonic())
    n_inits = len(inits)
    per_init = usable * 0.75 / n_inits  # 75% for exploration

    elapsed_total = 0.0
    for idx, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - usable * 0.25 - 5.0:
            print("Time to stop exploration at init %d" % idx)
            break

        n = len(arr)
        t0 = time.monotonic()
        left = min(per_init, deadline - time.monotonic() - usable * 0.25 - 5.0)
        if left < 3.0:
            continue

        seg = left / 3.0  # Split budget among 3 methods

        # Method 1: Smooth Adam with simplex projection
        a1, s1 = run_smooth_adam(arr, n, seg,
                                  tau_start=5.0, tau_end=0.02,
                                  lr_init=0.5 * np.sqrt(2.0 * n) / n,
                                  anneal_frac=0.7, top_k_fallback=20, rng=rng)
        if s1 < global_best:
            global_best = s1
            global_a = a1.copy()
            print("  %s [smooth_adam_1] maxc=%.6f (global=%.6f)" % (name, s1, global_best))

        if time.monotonic() > deadline - usable * 0.25 - 5.0:
            break

        # Method 2: Smooth Adam with different tau schedule
        a2, s2 = run_smooth_adam(a1, n, seg,
                                  tau_start=2.0, tau_end=0.01,
                                  lr_init=0.3 * np.sqrt(2.0 * n) / n,
                                  anneal_frac=0.5, top_k_fallback=50, rng=rng)
        if s2 < global_best:
            global_best = s2
            global_a = a2.copy()
            print("  %s [smooth_adam_2] maxc=%.6f (global=%.6f)" % (name, s2, global_best))

        if time.monotonic() > deadline - usable * 0.25 - 5.0:
            break

        # Method 3: Log-parameterization Adam
        a3, s3 = run_logparam_adam(a2, n, seg,
                                    tau_start=3.0, tau_end=0.01,
                                    lr_init=0.1,
                                    anneal_frac=0.6, rng=rng)
        if s3 < global_best:
            global_best = s3
            global_a = a3.copy()
            print("  %s [logparam] maxc=%.6f (global=%.6f)" % (name, s3, global_best))

        elapsed_total += time.monotonic() - t0

    print("\nAfter exploration phase: global_best_maxc=%.6f" % global_best)

    # Phase 2: Polish the best sequence
    left = deadline - time.monotonic()
    if global_a is not None and left > 5.0:
        n = len(global_a)
        print("\nStarting polish phase with %d property, budget=%.1fs" % (n, left))

        # Multiple polishing rounds with decreasing tau
        polish_rounds = 4
        per_round = left / polish_rounds

        a_cur = global_a.copy()
        for r in range(polish_rounds):
            if time.monotonic() > deadline - 3.0:
                break
            tau_s = 1.0 / (r + 1)
            tau_e = 0.02 / (r + 1)
            t0 = time.monotonic()
            a_round, s_round = run_smooth_adam(
                a_cur, n, per_round - (time.monotonic() - t0),
                tau_start=tau_s, tau_end=tau_e,
                lr_init=0.2 * np.sqrt(2.0 * n) / n / (r + 1),
                anneal_frac=0.8, top_k_fallback=50, rng=rng)
            if s_round < global_best:
                global_best = s_round
                global_a = a_round.copy()
            a_cur = a_round
            print("  Polish round %d: maxc=%.6f (global=%.6f)" % (r, s_round, global_best))

        # Final fine-tuning with top-k subgradient only
        if time.monotonic() < deadline - 5.0:
            t0 = time.monotonic()
            a_final, s_final = run_smooth_adam(
                a_cur, n, deadline - time.monotonic() - 5.0,
                tau_start=0.05, tau_end=0.005,
                lr_init=0.05 * np.sqrt(2.0 * n) / n,
                anneal_frac=0.9, top_k_fallback=100, rng=rng)
            if s_final < global_best:
                global_best = s_final
                global_a = a_final.copy()
            print("  Final fine-tune: maxc=%.6f (global=%.6f)" % (s_final, global_best))

    # Verify with authoritative evaluator
    if global_a is not None:
        final_seq = np.clip(global_a, 0.0, 1000.0)
        if final_seq.sum() < 0.01:
            final_seq = np.ones(256)
        ver = evaluate_sequence(final_seq.tolist())
        print("\nVerification score: %.6f" % ver)
        if not np.isfinite(ver):
            print("WARNING: Invalid sequence, using fallback")
            final_seq = np.ones(512)
            ver = evaluate_sequence(final_seq.tolist())
            print("Fallback score: %.6f" % ver)

    else:
        final_seq = np.ones(512)
        ver = evaluate_sequence(final_seq.tolist())

    return [float(x) for x in final_seq]
