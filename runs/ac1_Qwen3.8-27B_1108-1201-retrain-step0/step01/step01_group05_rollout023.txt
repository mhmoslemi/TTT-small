import numpy as np

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, n, top_indices, c, weight_mode="value"):
    """Weighted subgradient of max(c) using top-k active lags.
    weight_mode: 'value' (proportional to c[k]), 'excess' (proportional to max(0, c[k]-median)),
                 'uniform' (equal weights)
    """
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    if weight_mode == "value":
        weights = top_vals.copy()
    elif weight_mode == "excess":
        med = np.median(c)
        weights = np.maximum(top_vals - med, 0.0)
        if weights.sum() < 1e-15:
            weights = np.ones_like(top_vals)
    else:
        weights = np.ones_like(top_vals)
    total = weights.sum()
    if total <= 1e-15:
        return grad
    weights = weights / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def run_adam(a_init, n, S, max_time, rng, lr_init, top_k, weight_mode="value",
             cosine_decay=True, warm_restarts=3, restart_interval=300):
    """Adam optimizer with simplex projection, cosine decay, and warm restarts."""
    a = safe_normalize(a_init, n)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    t_start = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    total_planned = int(max_time / 0.3)  # estimate total iterations

    while time.monotonic() - start < max_time:

        restarts_done = 0
        while (t - t_start) >= restart_interval and restarts_done < warm_restarts:
            restarts_done += 1
            # Perturb and reset momentum
            noise_scale = 0.05 * S / n * (1.0 + restarts_done * 0.5)
            a = project_simplex(a + rng.normal(0.0, noise_scale, n), S)
            m *= 0.5
            v *= 0.5
            t_start = t

        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        grad = compute_subgradient(a, n, top_indices, c, weight_mode)
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-14:
            break
        grad = grad / gnorm

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Learning rate schedule
        if cosine_decay:
            progress = float(t) / max(1, total_planned)
            lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * min(progress, 1.0)))
            lr = max(lr, lr_init * 0.01)
        else:
            lr = lr_init * (0.5 ** (float(t) / 500.0))

        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

    return best_local, best_score

        # Blend: new_mag = alpha * flat + (1-alpha) * current
        alpha = 0.3
        new_mag = alpha * mag_avg + (1.0 - alpha) * mag
        # Inverse FFT
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new * (S / s)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < best_score:
            best_score = c_new.max()
            best_local = a_new.copy()
            a = a_new
        else:
            break

    return best_local, best_score


def interpolation_search(a1, a2, n, S, n_trials=20):
    """Try intermediate points between two good solutions."""
    best_a = a1.copy()
    c = fft_convolve_self(a1)
    best_score = c.max()

    for i in range(1, n_trials):
        t = i / float(n_trials)
        a = (1.0 - t) * a1 + t * a2
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * (S / s)
        c = fft_convolve_self(a)
        if c.max() < best_score:
            best_score = c.max()
            best_a = a.copy()

    return best_a, best_score


def collect_initializations(rng, n_target=None):
    """Generate diverse initial sequences."""
    inits = []

    # Try to use the provided sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64)
        given = given.reshape(-1)
        if given.size >= 16 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0, 1000)))
            n_target = given.size
    except Exception:
        pass

    if n_target is None:
        n_target = 4096

    n = n_target
    # Random
    inits.append(("rand", rng.random(n)))

    # Smooth sinusoidal
    tt = np.linspace(0.0, 1.0, n)
    smooth = 1.0 + 0.3 * np.sin(2.0 * np.pi * 3.0 * tt) + 0.2 * np.cos(2.0 * np.pi * 7.0 * tt)
    inits.append(("smooth", np.maximum(smooth, 0.0)))

    # Bump (concentrated in middle)
    bump = np.exp(-0.5 * ((np.arange(n) - n/2) / (n/6.0))**2)
    inits.append(("bump", bump + 0.1))

    # Ramped
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(("ramp", ramp))

    # Two-peak
    two_peak = np.exp(-0.5 * ((np.arange(n) - n*0.2) / (n/10.0))**2) + \
               np.exp(-0.5 * ((np.arange(n) - n*0.8) / (n/10.0))**2)
    inits.append(("twopk", two_peak + 0.05))

    return inits


def propose_candidate(seed=42, budget_s=2380, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0  # Leave 20s for final checks

    print("Starting optimization at t=0, budget=%.0fs" % budget_s)
    t0 = time.monotonic()

    inits = collect_initializations(rng)
    global_best = np.inf
    global_a = None
    all_good = []  # Store multiple good solutions for interpolation

    # ===== Phase 1: Exploration =====
    n_inits = len(inits)
    phase1_time = 0.45 * (deadline - time.monotonic())
    phase1_deadline = time.monotonic() + phase1_time
    per_init = phase1_time / max(1, n_inits)

    print("Phase 1: Exploration (%.0f init, %.1fs each)" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > phase1_deadline:
            break
        n = len(arr)
        if n < 16:
            continue
        S = np.sqrt(2.0 * float(n))
        seg_time = min(per_init * 0.6, 120.0)

        a_adam, s_adam = run_adam(arr, n, S, seg_time, rng,
                                  lr_init=0.8 * S / n, top_k=50,
                                  weight_mode="value", cosine_decay=True,
                                  warm_restarts=2, restart_interval=200)
        if s_adam < global_best:
            global_best, global_a = s_adam, a_adam.copy()
            all_good.append(a_adam.copy())
            print("  [%s] adam: %.6f (global=%.6f) t=%.0fs" %
                  (name, s_adam, global_best, time.monotonic() - t0))

        # Mirror descent on the best from Adam
        if time.monotonic() < phase1_deadline and per_init - seg_time > 5:
            seg2 = min(per_init - seg_time - 2, 60.0)
            a_mir, s_mir = run_mirror(a_adam, n, S, max(1.0, seg2), rng,
                                      eta_init=0.4, top_k=50)
            if s_mir < global_best:
                global_best, global_a = s_mir, a_mir.copy()
                all_good.append(a_mir.copy())
                print("  [%s] mirror: %.6f (global=%.6f) t=%.0fs" %
                      (name, s_mir, global_best, time.monotonic() - t0))

    print("Phase 1 complete. Best: %.6f at t=%.0fs" % (global_best, time.monotonic() - t0))

    if global_a is None:
        global_a = np.ones(256)

    n = len(global_a)
    S = np.sqrt(2.0 * float(n))

    # ===== Phase 2: Exploitation =====
    phase2_deadline = time.monotonic() + 0.35 * (deadline - time.monotonic())
    print("Phase 2: Exploitation (target deadline t=%.0fs)" %
          (phase2_deadline - t0))

    # Run with larger top_k and more careful learning
    a_curr = global_a.copy()
    best_p2 = global_best

    # Multiple rounds with different top_k
    top_ks = [80, 60, 100]
    round_time = (phase2_deadline - time.monotonic()) / len(top_ks)
    for tk in top_ks:
        if time.monotonic() > phase2_deadline - 5:
            break
        a_curr, s_curr = run_adam(a_curr, n, S, max(1.0, round_time - 2), rng,
                                  lr_init=0.3 * S / n, top_k=tk,
                                  weight_mode="excess", cosine_decay=True,
                                  warm_restarts=2, restart_interval=250)
        if s_curr < best_p2:
            best_p2 = s_curr
            global_best, global_a = s_curr, a_curr.copy()
            all_good.append(a_curr.copy())
            print("  Phase2 tk=%d: %.6f (global=%.6f) t=%.0fs" %
                  (tk, s_curr, global_best, time.monotonic() - t0))

    # ===== Phase 3: Polish =====
    phase3_deadline = deadline - 5.0
    print("Phase 3: Polish (target deadline t=%.0fs)" % (phase3_deadline - t0))

    # Coordinate descent
    if time.monotonic() < phase3_deadline - 10:
        cd_time = min(60.0, (phase3_deadline - time.monotonic()) * 0.4)
        a_cd, s_cd = coordinate_descent_polish(global_a, n, S, cd_time,
                                               top_k_lags=20, n_candidates=5)
        if s_cd < global_best:
            global_best, global_a = s_cd, a_cd.copy()
            print("  CD: %.6f (global=%.6f) t=%.0fs" %
                  (s_cd, global_best, time.monotonic() - t0))

    # Spectral shaping
    if time.monotonic() < phase3_deadline - 5:
        a_ss, s_ss = spectral_shaping(global_a, n, S, n_iters=15)
        if s_ss < global_best:
            global_best, global_a = s_ss, a_ss.copy()
            print("  Spectral: %.6f (global=%.6f) t=%.0fs" %
                  (s_ss, global_best, time.monotonic() - t0))
        # Try spectral shaping on other good solutions too
        for ga in all_good[-3:]:
            if len(ga) != n or time.monotonic() > phase3_deadline - 3:
                continue
            a_ss2, s_ss2 = spectral_shaping(ga, n, S, n_iters=10)
            if s_ss2 < global_best:
                global_best, global_a = s_ss2, a_ss2.copy()
                print("  Spectral2: %.6f (global=%.6f)" % (s_ss2, global_best))

    # Interpolation between best solutions
    if len(all_good) >= 2 and time.monotonic() < phase3_deadline - 3:
        for i in range(min(3, len(all_good) - 1)):
            a1 = all_good[i]
            a2 = all_good[-1]
            if len(a1) != n or len(a2) != n:
                continue
            a_interp, s_interp = interpolation_search(a1, a2, n, S, n_trials=15)
            if s_interp < global_best:
                global_best, global_a = s_interp, a_interp.copy()
                print("  Interp: %.6f (global=%.6f)" % (s_interp, global_best))

    # Final Adam polish with very small learning rate
    if time.monotonic() < phase3_deadline - 2:
        final_time = max(1.0, phase3_deadline - time.monotonic() - 2)
        a_fin, s_fin = run_adam(global_a, n, S, final_time, rng,
                                lr_init=0.1 * S / n, top_k=100,
                                weight_mode="excess", cosine_decay=True,
                                warm_restarts=0, restart_interval=999999)
        if s_fin < global_best:
            global_best, global_a = s_fin, a_fin.copy()
            print("  FinalAdam: %.6f (global=%.6f)" % (s_fin, global_best))

    # ===== Phase 4: Verify =====
    print("Phase 4: Verification")
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    verify_score = evaluate_sequence(global_a.tolist())
    print("Verify score: %.6f (internal estimate: %.6f)" % (verify_score, global_best))

    if not np.isfinite(verify_score):
        print("WARNING: Invalid sequence! Falling back.")
        global_a = np.ones(min(n, 1024), dtype=np.float64)
        verify_score = evaluate_sequence(global_a.tolist())

    print("Final answer: score=%.6f, length=%d, t=%.0fs" %
          (verify_score, len(global_a), time.monotonic() - t0))

    return [float(x) for x in global_a]
