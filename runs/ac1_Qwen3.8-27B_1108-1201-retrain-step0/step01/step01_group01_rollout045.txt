def _score(a):
    """Compute the objective for a non-negative sequence."""
    n = len(a)
    if np.sum(a) < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (np.sum(a) ** 2)

    """
