    """Full self-convolution of 1-D vector p, length 2n-1."""
    p = np.asarray(p, dtype=np.float64)

def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (cssum - 1.0) / idx
    k = int(np.count_nonzero(cond))
    if k <= 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    theta = (cssum[k - 1] - 1.0) / float(k)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s

            b_ub=np.full(2 * n - 1, M_f, dtype=np.float64),
            bounds=(0, None),
            method='highs',
            options={'disp': False},
        )
    except Exception:
        return p
    if res is None or res.status != 0 or res.x is None:
        return p
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0)
    gs = float(g.sum())
    if gs < 1e-12:
        return p
    return g / gs


# ---------------------------------------------------------------------------
# Safe normalize
# ---------------------------------------------------------------------------

def _safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s < 1e-12:
        return np.full(max(1, int(p.size)), 1.0 / max(1, int(p.size)), dtype=np.float64)
    return p / s


# ---------------------------------------------------------------------------
# Initialize u vector for a given p (warm start)
# ---------------------------------------------------------------------------

def _init_u(p, alpha):
    """Initialize the weight vector u for Newton method."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    L = _fft_len_grad(n)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]
    cmax = float(c.max()) if c.size else 1.0
    if not np.isfinite(cmax) or cmax < 1e-15:
        cmax = 1.0
    T = cmax / alpha
    if T < 1e-12:
        T = 1.0 / alpha
    diff = (c - cmax) / T
    e = np.exp(np.minimum(diff, 50.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    Fw = fft(w[::-1], L)
    u = np.real(ifft(Fp * Fw))
    if n == 1:
        u = u[:1]
    else:
        u = u[2 * n - 2:n - 2:-1]
    u = np.asarray(u, dtype=np.float64)
    if u.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, u.size)
        tmp[:m] = np.maximum(u[:m], 0)
        u = tmp
    u = np.maximum(u, 0.0)
    us = float(u.sum())
    if us < 1e-300:
        u = np.ones(n, dtype=np.float64)
    else:
        u = u / us
    return u


# ---------------------------------------------------------------------------
# Difference set construction
# ---------------------------------------------------------------------------

def _make_diffset_sequence(k, v, n):
    """
    Create a length-n nonnegative sequence inspired by a (v,k,lambda) difference set.
    Places k ones in a cyclic pattern of period v, extended/truncated to length n.
    Returns normalized sequence.
    """
    # Use quadratic residues mod v for prime v, or Singer construction
    # For simplicity, use the Singer difference set when possible
    # Singer: v = q^2 + q + 1, k = q+1, for prime power q
    # We'll use a simple cyclic construction: D = {0, 1, 3} mod 7 (Fano) or general quadratic residue
    
    # For general use, create a "spread" pattern
    seq = np.zeros(n, dtype=np.float64)
    if v <= 0 or k <= 0:
        return None
    
    # Compute the difference set positions (cyclic)
    # Use quadratic residues for prime v
    positions = set()
    if v >= 7 and v % 2 == 1:
        # Check if v is prime (simple trial division)
        is_prime = True
        for d in range(2, int(np.sqrt(v)) + 1):
            if v % d == 0:
                is_prime = False
                break
        if is_prime:
            for x in range(1, v):
                if (x * x) % v == 1 or (x * x) % v == -1 % v:
                    positions.add(x % v)
            # Actually, quadratic residues are {x^2 mod v : x = 1,...,v-1}
            positions = set()
            for x in range(1, v):
                positions.add((x * x) % v)
    
    if len(positions) < 2:
        # Fallback: use a simple pattern
        step = v // max(1, k)
        for i in range(k):
            positions.add((i * step) % v)
        positions.discard(0)
        if len(positions) < 2:
            positions = set((1, min(2, v - 1)))
    
    pos_list = sorted(positions)
    # Tile the pattern to fill length n
    for i in range(n):
        if (i % v) in positions:
            seq[i] = 1.0
    
    s = float(seq.sum())
    if s < 1:
        return None
    return seq / s

def _make_fourier_p(n, rng):
    """
    Construct p[j] = 1 + sum_m a_m cos(2*pi*m*j/n + phi_m), floored at 0.
    The Fourier coefficients are chosen to approximately equalize |P(k)|^2.
    """
    x = np.arange(n, dtype=np.float64)
    p = np.ones(n, dtype=np.float64)
    
    # Add a few cosine terms with random phases
    n_modes = min(20, n // 4)
    for m in range(1, n_modes + 1):
        a = float(rng.uniform(0.02, 0.15))
        phi = float(rng.uniform(0, 2 * np.pi))
        p += a * np.cos(2.0 * np.pi * m * x / n + phi)
    
    p = np.maximum(p, 0.01)
    return p / p.sum()


def _make_beta_p(n, rng, alpha_range=(0.1, 3.0)):
    """p[j] = (j+1)^a * (n-j)^b, random a,b."""
    a = float(rng.uniform(*alpha_range))
    b = float(rng.uniform(*alpha_range))
    x = np.arange(1, n + 1, dtype=np.float64)
    p = np.power(x, a) * np.power(n + 1 - x, b)
    p = np.maximum(p, 0)
    s = float(p.sum())
    if s < 1e-12:
        return np.ones(n, dtype=np.float64) / n
    return p / s


def _make_power_p(n, rng, exponent_range=(-2.0, 2.0)):
    """p[j] = (j/n + eps)^a for random a."""
    a = float(rng.uniform(*exponent_range))
    x = np.linspace(1e-5, 1.0, n)
    p = np.power(x, a)
    p = np.maximum(p, 0)
    s = float(p.sum())
    if s < 1e-12:
        return np.ones(n, dtype=np.float64) / n
    return p / s


def _make_cup_p(n, rng, a_range=(0.1, 2.0)):
    """p[j] = 1 + a*(2j/n - 1)^2 (U-shape)."""
    a = float(rng.uniform(*a_range))
    x = np.linspace(0, 1, n)
    p = 1.0 + a * (2.0 * x - 1.0) ** 2
    p = np.maximum(p, 0)
    s = float(p.sum())
    if s < 1e-12:
        return np.ones(n, dtype=np.float64) / n
    return p / s
