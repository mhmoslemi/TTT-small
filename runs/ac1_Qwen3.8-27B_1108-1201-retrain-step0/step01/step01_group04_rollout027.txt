def optimize_softmax(p0, rng, time_limit, label=""):

    current_score = best_score
    
    end = time.monotonic() + max(0.0, time_limit)
    T = 0.15  # Initial temperature
    lr = 0.5
    iter_no = 0
    stall = 0
    
    while time.monotonic() < end:
        iter_no += 1
        
        # Compute gradient
        dscore_dp = compute_dscore_dp(p, T)
        dt = softmax_gradient_chain(t, dscore_dp)
        
        if not np.all(np.isfinite(dt)):
            dt = np.zeros(n)
        
        # Adaptive learning rate
        gnorm = np.linalg.norm(dt)
        if gnorm > 1e-300:
            step = lr / (gnorm * np.sqrt(n))
        else:
            step = 0.01
        
        # Try several step sizes
        accepted = False
        for factor in (1.0, 0.5, 0.25, 0.1, 0.05):
            t_new = t - step * factor * dt
            p_new = softmax_params_to_p(t_new)
            sc_new = internal_score(p_new)
            
            if np.isfinite(sc_new) and sc_new < current_score:
                t = t_new
                p = p_new
                current_score = sc_new
                if sc_new < best_score:
                    best_score = sc_new
                    best_p = p_new.copy()
                    stall = 0
                else:
                    stall += 1
                accepted = True
                break
        
        if not accepted:
            stall += 1
            T = max(0.01, T * 0.95)
            lr = max(0.01, lr * 0.9)
            
            if stall > 100:
                # Random restart in parameter space
                t = t + rng.normal(0, 0.1, n)
                p = softmax_params_to_p(t)
                current_score = internal_score(p)
                T = 0.15
                lr = 0.5
                stall = 0

            T = max(0.02, T * 0.9)
        
        if iter_no % 500 == 0:
            print(f"{label} iter={iter_no} best={best_score:.10f} T={T:.4f}", flush=True)

def icm_step(p, rng, gamma_max=1.0, n_iters=200):
    """
    Iterative Convolution Minimization (pressure-based).
    Reduces indices that contribute to the peak of the convolution.
    """
    n = len(p)
    p = p.copy().astype(np.float64)
    s = p.sum()
    if s < 1e-30:
        return p
    p = p / s
    
    best_p = p.copy()
    best_score = internal_score(p)
    
    gamma = gamma_max
    for it in range(n_iters):
        c = conv_full(p)
        cmax = float(np.max(c))
        if cmax < 1e-30:
            break
        
        # Soft threshold: weight lags near the peak
        threshold = 0.95 * cmax
        w = np.zeros(2*n-1)
        mask = c > threshold
        w[mask] = 1.0
        # Also add a soft contribution for lags slightly below threshold
        soft_mask = (c > 0.85 * cmax) & (~mask)
        w[soft_mask] = 0.3
        
        if w.sum() == 0:
            break
        
        # Compute pressure on each index
        # pressure_i = sum_k w_k * p_{k-i} (valid indices)
        # Same computation as in gradient: pressure = conv(w, p_rev)[n-1:2n-1]
        L = get_fft_len_grad(n)
        Fw = fft(w, L)
        Fp_rev = fft(p[::-1], L)
        h = np.real(ifft(Fw * Fp_rev))
        pressure = h[n-1:2*n-1]
        if pressure.size != n:
            pressure = np.zeros(n)
        
        pressure = np.maximum(pressure, 0)
        
        # Update: reduce high-pressure indices
        p_new = p / (1.0 + gamma * pressure / max(pressure.max(), 1e-10))
        p_new = np.maximum(p_new, 0)
        s_new = p_new.sum()
        if s_new < 1e-10:
            break
        p_new = p_new / s_new
        
        sc_new = internal_score(p_new)
        if np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_p = p_new.copy()
            p = p_new
            # Slowly decrease gamma
            gamma = max(0.01, gamma * 0.99)
        else:
            gamma = max(0.001, gamma * 0.5)
            if gamma < 0.005:
                break
    
    return best_p, best_score

def pair_move_optimal(p, i, j, n_search=30):
    """
    Find optimal delta for pair move: p_i -= delta, p_j += delta.
    Use golden section or grid search.
    """
    n = len(p)
    base = p.copy()
    best_delta = 0.0

    inits = []

    # Uniform
    inits.append(np.full(n, 1.0 / n))

    # Sinusoidal modulations
    x = np.linspace(0, 1, n)
    for freq in (1, 2, 3, 5, 7):
        for amp in (0.3, 0.6, 0.8):
            for phase in (0, np.pi / 3, 2 * np.pi / 3):
                p = 1.0 + amp * np.cos(2 * np.pi * freq * x + phase)
                p = np.maximum(p, 0.01)
                p = p / p.sum()
                inits.append(p)

    # Two-hump
    for w1 in (0.3, 0.4, 0.5):
        for mu1 in (0.2, 0.3):
            mu2 = 1.0 - mu1
            p = np.exp(-((x - mu1) ** 2) / (2 * 0.04)) + w1 * np.exp(-((x - mu2) ** 2) / (2 * 0.04))
            p = p / p.sum()
            inits.append(p)

    # Power law
    for a in (0.5, 1.0, 2.0, -0.5, -1.0):
        p = np.power(x + 0.01, a)
        p = np.maximum(p, 0)
        s = p.sum()
        if s > 0:
            inits.append(p / s)

    # Random smooth
    for _ in range(8):
        f1 = int(rng.integers(1, max(2, n // 100)))
        f2 = int(rng.integers(1, max(2, n // 50)))
        p = 1.0
        p += 0.5 * np.sin(2 * np.pi * f1 * x + rng.uniform(0, 2 * np.pi))
        p += 0.3 * np.sin(2 * np.pi * f2 * x + rng.uniform(0, 2 * np.pi))
        p = np.maximum(p, 0.01)
        p = p / p.sum()
        inits.append(p)

    try:
        inits.append(spectral_init(n, rng))
    except Exception:
        pass

    return inits

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for nonnegative step function minimizing 2n*max(conv(p,p))/sum(p)^2.
    Uses CG on softmax params + ICM + pair moves.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(60.0, float(budget_s) - 30.0)

    # Get parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and parent_arr.sum() > 0:
            parent = parent_arr
    except Exception:
        pass

    n = 700
    if parent is not None:
        n = int(parent.size)
    if n < 64:
        n = 700
    if n > 2000:
        n = 2000

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0 or not np.all(np.isfinite(p)):
            return
        p = np.maximum(p, 0)
        s = p.sum()
        if s < 1e-20:
            return
        p = p / s
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print(f"  NEW BEST: {best_score:.10f} (n={n}) at t={time.monotonic()-start_time:.1f}s", flush=True)

    # Phase 1: Generate and score initials
    print(f"Phase 1: Generating initials (n={n})...", flush=True)
    inits = []
    if parent is not None:
        p_par = np.maximum(parent, 0).astype(np.float64)
        sp = p_par.sum()
        if sp > 0:
            p_par = p_par / sp
            inits.append(p_par)

    other_inits = make_initials(n, rng)
    inits.extend(other_inits)

    scored = []
    for p in inits:
        p = np.maximum(p, 0).astype(np.float64)
        s = p.sum()
        if s < 1e-20:
            continue
        p = p / s
        sc = internal_score(p)
        if np.isfinite(sc):
            scored.append((sc, p))

    scored.sort(key=lambda x: x[0])
    for sc, p in scored[:5]:
        try_record(p, sc)

    if not scored:
        best_p = np.full(n, 1.0 / n)
        best_score = internal_score(best_p)
        return [float(x) for x in best_p]

    # Phase 2: Optimize top candidates with CG
    print(f"Phase 2: CG optimization...", flush=True)
    top_k = min(4, len(scored))

    # Allocate time: give more time to better starts
    total_opts = sum(i + 1 for i in range(top_k))
    remaining_budget = deadline - time.monotonic() - 120.0  # reserve 120s for final
    if remaining_budget < 60:
        remaining_budget = max(60.0, deadline - time.monotonic() - 30.0)

    for idx in range(top_k):
        if time.monotonic() > deadline - 60:
            break
        sc, p = scored[idx]
        weight = (top_k - idx + 1) / total_opts
        time_alloc = max(30.0, remaining_budget * weight * 0.7)
        label = f"CG{idx}:"

        print(f"  Starting {label} score={sc:.10f} time={time_alloc:.0f}s at t={time.monotonic()-start_time:.1f}s", flush=True)

        p_opt, sc_opt = cg_optimize(p, rng, time_alloc, label)
        try_record(p_opt, sc_opt)

        # ICM refinement
        if time.monotonic() < deadline - 45:
            icm_time = min(60.0, (deadline - time.monotonic() - 30.0) * 0.15)
            if icm_time > 5:
                p_icm, sc_icm = icm_refine(p_opt, rng, n_iters=200)
                try_record(p_icm, sc_icm)

                # Pair moves after ICM
                if time.monotonic() < deadline - 30:
                    p_pair, sc_pair = pair_moves(p_icm, rng, max_moves=100)
                    try_record(p_pair, sc_pair)

    # Phase 3: Focused optimization on the best
    print(f"Phase 3: Focused optimization on best (score={best_score:.10f})...", flush=True)
    if best_p is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 30:
            # Multiple passes of CG + ICM + pair moves
            for pass_no in range(3):
                if time.monotonic() > deadline - 15:
                    break
                t_alloc = max(15.0, remaining * 0.35)
                p_cg, sc_cg = cg_optimize(best_p, rng, min(t_alloc, remaining - 10), f"FOC{pass_no}:")
                try_record(p_cg, sc_cg)

                if time.monotonic() > deadline - 15:
                    break
                p_icm, sc_icm = icm_refine(best_p, rng, n_iters=150)
                try_record(p_icm, sc_icm)

                if time.monotonic() > deadline - 15:
                    break
                p_pair, sc_pair = pair_moves(best_p, rng, max_moves=150)
                try_record(p_pair, sc_pair)

                remaining = deadline - time.monotonic() - 10.0

    # Phase 4: Final micro-pair-moves
    if best_p is not None and time.monotonic() < deadline - 5:
        p_final, sc_final = pair_moves(best_p, rng, max_moves=300)
        try_record(p_final, sc_final)

    # Ensure valid output
    if best_p is None:
        best_p = np.full(n, 1.0 / n)
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = best_p.sum()
    if s < 1e-20:
        best_p = np.full(n, 1.0 / n)
    else:
        best_p = np.minimum(best_p, 1000.0)

    elapsed = time.monotonic() - start_time
    final_score = internal_score(best_p)
    print(f"DONE: n={len(best_p)} score={final_score:.10f} elapsed={elapsed:.1f}s", flush=True)

    return [float(x) for x in best_p]
