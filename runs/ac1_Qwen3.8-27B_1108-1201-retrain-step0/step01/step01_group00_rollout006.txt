Multi-phase optimization for minimizing the peak of self-convolution of
non-negative step functions. Uses LP-based block coordinate descent as the
main engine, combined with greedy peak reduction and Nelder-Mead polishing.

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)

def greedy_peak_reduction(a_init, max_iters=5000, verbose=False):
    """
    Greedy peak reduction: identify peak, reduce contributors, compensate elsewhere.
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = np.sum(a)
    best_a = a.copy()
    best_score = local_score_fast(a)
    
    for it in range(max_iters):
        c = fft_convolve(a, a)
        peak_k = int(np.argmax(c))
        peak_val = c[peak_k]
        
        # Find top contributors to the peak
        # c[peak_k] = sum_m a[m]*a[peak_k - m] for valid m
        contribs = np.zeros(n)
        for m in range(n):
            j = peak_k - m
            if 0 <= j < n:
                contribs[m] = a[m] * a[j]
        
        total_contrib = np.sum(contribs)
        if total_contrib < 1e-10:
            break
        
        # Top contributors (fraction of total)
        top_frac = 0.1  # reduce top 10% of contributors
        n_top = max(2, int(n * top_frac))
        top_idx = np.argsort(contribs)[-n_top:]
        
        # Reduce top contributors
        reduction = 0.05 + 0.05 * (it / max_iters)  # increasing reduction
        a[top_idx] *= (1 - reduction)
        
        # Compensate: increase elements that contribute least to ALL peaks
        # Compute "contribution to max" for each element
        # For element i, its contribution to c[k] is a[i]*a[k-i] for valid k
        # We want to increase elements that are low contributors to the current peak
        # and also to other high peaks
        
        # Simple heuristic: increase elements not in top_idx that are currently small
        other_idx = np.setdiff1d(np.arange(n), top_idx)
        if len(other_idx) > 0:
            # Normalize to maintain sum
            deficit = np.sum(a[top_idx] * reduction)  # approximate
            # Actually, let's just normalize
            S_new = np.sum(a)
            if S_new > 0:
                a = a * (S / S_new)
        
        a = np.clip(a, 1e-6, None)
        
        score = local_score_fast(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def nelder_mead_polish(a_init, time_limit, rng):
    """
    Nelder-Mead polish on a perturbation of the best sequence.
    We perturb with a small number of free parameters.
    """
    n = len(a_init)
    S = np.sum(a_init)
    
    # Parameterize: a = a_init * exp(theta * v) where v is a fixed random vector
    # and theta is the free parameter. This is a 1D optimization.
    # Or use multiple directions.
    
    n_params = min(20, n)
    v_dirs = rng.standard_normal((n_params, n))
    v_dirs /= np.linalg.norm(v_dirs, axis=1, keepdims=True)
    
    def objective(thetas):
        a = a_init.copy()
        for i in range(len(thetas)):
            a = a * np.exp(thetas[i] * v_dirs[i])
        a = np.clip(a, 0.0, None)
        S_a = np.sum(a)
        if S_a < 0.01:
            return 1e6
        a = a * (S / S_a)
        return local_score_fast(a)
    
    x0 = np.zeros(n_params)
    t0 = time.monotonic()
    result = minimize(objective, x0, method='Nelder-Mead',
                      options={'maxiter': 5000, 'xatol': 1e-6, 'fatol': 1e-7,
                               'adaptive': True})
    
    best_a = a_init.copy()
    best_score = local_score_fast(a_init)
    
    if result.fun < best_score:
        a = a_init.copy()
        for i in range(len(result.x)):
            a = a * np.exp(result.x[i] * v_dirs[i])
        a = np.clip(a, 0.0, None)
        a = a * (S / np.sum(a))
        best_a = a
        best_score = result.fun
    
    return best_a, best_score


def random_walk_polish(a_init, time_limit, rng):
    """
    Random walk with momentum, accepting moves that improve the score.
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = np.sum(a)
    best_a = a.copy()
    best_score = local_score_fast(a)
    
    t0 = time.monotonic()
    step_size = 0.03
    momentum = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        # Generate perturbation
        noise = rng.standard_normal(n)
        momentum = 0.9 * momentum + 0.1 * noise
        candidate = a + step_size * momentum
        candidate = np.clip(candidate, 1e-6, None)
        candidate = candidate * (S / np.sum(candidate))
        
        new_score = local_score_fast(candidate)
        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
            a = candidate
            step_size *= 1.01  # slight increase on success
        else:
            a = candidate if rng.random() < 0.1 else a  # occasional acceptance
            step_size = max(0.001, step_size * 0.995)
    
    return best_a, best_score

"""
LP-based block coordinate descent with adaptive step sizes,
combined with multi-scale optimization and fine-grained local search
to minimize the peak of self-convolution of non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score_fast(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_lp_matrix(a):
    """Build the constraint matrix for the LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # (a*b)_k = sum_j a[k-j] * b[j] for 0 <= k-j < n
        # j ranges from max(0, k-n+1) to min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp(a, M):
    """Solve: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, 1000.0)] * n  # upper bound to prevent issues
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except:
        pass
    return a.copy()


def lp_bcd(a_init, max_iters=500, time_limit=None, verbose=False):
    """LP-based Block Coordinate Descent with adaptive step size."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = np.sum(a)
    best_a = a.copy()
    best_score = local_score_fast(a)
    
    t0 = time.monotonic()
    step_size = 0.5
    plateau_count = 0
    
    for it in range(max_iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        c = fft_convolve(a, a)
        M = np.max(c)
        
        b = solve_lp(a, M)
        S_b = np.sum(b)
        if S_b < 1e-10:
            break
        b = b * (S / S_b)
        
        # Update with adaptive step
        a_new = (1 - step_size) * a + step_size * b
        a_new = np.clip(a_new, 1e-8, None)
        a_new = a_new * (S / np.sum(a_new))
        
        score = local_score_fast(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            plateau_count = 0
            step_size = min(1.0, step_size * 1.05)
        else:
            plateau_count += 1
            if plateau_count > 50:
                step_size *= 0.9
                plateau_count = 0
        
        a = a_new
        
        if verbose and it % 50 == 0:
            print(f"    LP-BCD iter {it}: score={score:.6f}, best={best_score:.6f}, step={step_size:.3f}")
    
    return best_a, best_score


def random_walk(a_init, time_limit, rng, step_size=0.01):
    """Random walk with momentum."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = np.sum(a)
    best_a = a.copy()
    best_score = local_score_fast(a)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        noise = rng.standard_normal(n)
        momentum = 0.85 * momentum + 0.15 * noise
        candidate = a + step_size * momentum
        candidate = np.clip(candidate, 1e-6, None)
        candidate = candidate * (S / np.sum(candidate))
        
        new_score = local_score_fast(candidate)
        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
            a = candidate
        elif rng.random() < 0.05:
            a = candidate  # occasional random jump
    
    return best_a, best_score


def targeted_reduction(a_init, time_limit, rng):
    """Targeted peak reduction."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = np.sum(a)
    best_a = a.copy()
    best_score = local_score_fast(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        peak_k = int(np.argmax(c))
        
        # Find contributors
        contribs = np.zeros(n)
        for m in range(min(n, peak_k + 1)):
            j = peak_k - m
            if 0 <= j < n:
                contribs[m] = a[m] * a[j]
        
        top_k = max(3, n // 8)
        top_idx = np.argsort(contribs)[-top_k:]
        
        # Reduce top contributors
        reduction = rng.uniform(0.02, 0.08)
        a[top_idx] *= (1 - reduction)
        
        # Increase random other elements
        other = np.setdiff1d(np.arange(n), top_idx)
        if len(other) > 0:
            n_inc = min(10, len(other))
            inc_idx = rng.choice(other, size=n_inc, replace=False)
            a[inc_idx] *= (1 + reduction * 2)
        
        a = np.clip(a, 1e-6, None)
        a = a * (S / np.sum(a))
        
        score = local_score_fast(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score
