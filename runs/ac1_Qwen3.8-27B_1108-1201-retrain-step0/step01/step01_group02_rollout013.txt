```python
def score_sequence(a):
    n = len(a)

    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:

    return 2.0 * n * max_b / (sum_a ** 2)

    # and the coefficients are a[0], a[1], ..., a[n-1]
    # So A[j+m, j] = a[m] for m = 0, ..., n-1
    # This means A[k, j] = a[k-j] where k ranges from j to j+n-1

    data = np.tile(a, n)

def solve_lp_direction(a, M, timeout=30.0):

    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
```

Let me think about what the LP is actually doing. At each step:
1. We have sequence a with convolution peak M.
2. We find g that maximizes sum(g) subject to conv(a,g) ≤ M and g ≥ 0.
3. We blend: a_new = (1-α)a + αg.

The key property is that conv(a_new, a_new) should have a lower peak than conv(a,a). This is because:
conv(a_new, a_new) = (1-α)^2 conv(a,a) + 2α(1-α) conv(a,g) + α^2 conv(g,g)

The first term has peak (1-α)^2 M.
The second term has peak 2α(1-α) M (since conv(a,g) ≤ M by constraint).
The third term has peak α^2 * max(conv(g,g)).

If max(conv(g,g)) is not too large, the total peak is bounded by M * ((1-α)^2 + 2α(1-α) + α^2 * max_conv_g/M).

For this to be < M, we need max_conv_g/M < 1, which means g has a flatter convolution than a. This is the key: the LP finds a g whose convolution with a is bounded by M, and if g itself has a lower peak-to-sum ratio, the blend improves things.

The LP maximizes sum(g), which tends to spread out g, making its convolution flatter.

```python
"""
Multi-phase optimization for minimizing peak auto-convolution of non-negative sequences.
Combines LP-based refinement, peak-pushing heuristics, coordinate descent, and local search.
"""
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    return a / s


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_direction(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=600.0, verbose=False, seed=42):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")
    
    stagnant = 0
    it = 0
    max_iters = 2000
    
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        M = conv.max()
        sum_a = a.sum()
        score = 2.0 * n * M / (sum_a ** 2)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 60:
            break
        
        remaining = time_limit_s - elapsed
        if remaining < 15.0:
            break
        
        timeout = max(2.0, min(45.0, remaining * 0.2))
        
        # Try the standard LP direction
        g = solve_lp_direction(a, M, timeout=timeout)
        if g is None or g.sum() <= a.sum() * (1 + 1e-10):
            if g is None:
                break
            # Try with slightly different M
            g = solve_lp_direction(a, M * 1.01, timeout=timeout)
            if g is None or g.sum() <= a.sum() * (1 + 1e-10):
                break
        
        g = g / g.sum()
        
        # Adaptive step size via line search
        best_new_score = score
        best_alpha = 0
        for alpha in [0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1 - alpha) * a + alpha * g
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s < 0.01:
                continue
            a_new = a_new / s
            new_score = score_sequence(a_new)
            if new_score < best_new_score:
                best_new_score = new_score
                best_alpha = alpha
        
        if best_alpha == 0:
            # No improvement, try a smaller random perturbation
            a_perturbed = a + rng.normal(0, 0.01, n)
            a_perturbed = normalize_nonneg(a_perturbed)
            pert_score = score_sequence(a_perturbed)
            if pert_score < best_score:
                a = a_perturbed
            else:
                break
        else:
            a = best_a if best_new_score < best_score else ((1 - best_alpha) * a + best_alpha * g)
            a = normalize_nonneg(a)
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def peak_push_refine(a, time_limit_s=120.0, verbose=False):
    """Targeted reduction of the convolution peak."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    Peak push start: score={best_score:.6f}")
    
    it = 0
    max_iters = 500
    for it in range(max_iters):
        if time.monotonic() - start_time > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        k_star = conv.argmax()
        M = conv.max()
        sum_a = a.sum()
        score = 2.0 * n * M / (sum_a ** 2)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        # Find which pairs (i, k_star - i) contribute most to the peak
        # conv[k_star] = sum_i a[i] * a[k_star - i]
        contributions = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        
        # Identify the top contributors
        top_k = max(5, n // 50)
        top_idx = np.argsort(-contributions)[:top_k]
        
        # Reduce the larger element of each pair
        a_new = a.copy()
        for i in top_idx:
            j = k_star - i
            if 0 <= j < n:
                # Reduce the larger of a[i], a[j]
                if a[i] >= a[j]:
                    a_new[i] *= 0.95
                else:
                    a_new[j] *= 0.95
        
        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)
        
        if new_score < score:
            a = a_new
        else:
            # Try a different strategy: increase non-contributing positions
            a_new2 = a.copy()
            low_contrib = np.argsort(contributions)[:top_k]
            a_new2[low_contrib] *= 1.05
            a_new2 = normalize_nonneg(a_new2)
            new_score2 = score_sequence(a_new2)
            if new_score2 < score:
                a = a_new2
            else:
                break
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    Peak push done: best={best_score:.6f} after {it+1} iters")
    return best_a, best_score


def local_search_refine(a, time_limit_s=120.0, seed=42, verbose=False):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    LS start: score={best_score:.6f}")
    
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        # Random width perturbation
        width = max(2, int(rng.integers(max(2, n // 300), max(3, n // 6))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.05, size=stop - start))
        
        # Sometimes try symmetry
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 100 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")
    
    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_asymmetric(n):
    """Asymmetric initialization - concentrated on one side."""
    t = np.arange(n, dtype=np.float64)
    # Skewed raised cosine
    x = t / (n - 1) if n > 1 else 0
    return np.maximum(0, (1 - x) * (1 + np.cos(np.pi * x * 1.5)))


def initial_bimodal(n):
    """Two bumps at different heights."""
    t = np.arange(n, dtype=np.float64)
    center1 = n * 0.3
    center2 = n * 0.7
    sigma = n * 0.15
    a = np.exp(-0.5 * ((t - center1) / sigma) ** 2) + 0.7 * np.exp(-0.5 * ((t - center2) / sigma) ** 2)
    return a


def upsample_sequence(a, new_n):
    """Upsample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n)
    return np.interp(new_x, old_x, a)
```

```python
"""
Multi-phase optimization for minimizing peak auto-convolution of non-negative sequences.
Combines LP-based refinement with adaptive line search, peak-pushing heuristics,
coordinate-aware local search, and multi-length multi-start strategy.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    return a / s


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_direction(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=600.0, verbose=False, seed=42):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")

    stagnant = 0
    it = 0
    max_iters = 3000

    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break

        conv = np.convolve(a, a)
        M = conv.max()
        sum_a = a.sum()
        score = 2.0 * n * M / (sum_a ** 2)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 80:
            break

        remaining = time_limit_s - elapsed
        if remaining < 15.0:
            break

        timeout = max(2.0, min(40.0, remaining * 0.15))

        # Try standard LP direction
        g = solve_lp_direction(a, M, timeout=timeout)
        if g is None or g.sum() <= a.sum() * (1 + 1e-10):
            # Try with slightly relaxed constraint
            g2 = solve_lp_direction(a, M * 1.005, timeout=timeout)
            if g2 is not None and g2.sum() > a.sum() * (1 + 1e-10):
                g = g2
            else:
                # Last resort: try tighter
                g3 = solve_lp_direction(a, M * 0.995, timeout=timeout)
                if g3 is not None and g3.sum() > a.sum() * (1 + 1e-10):
                    g = g3
                else:
                    break

        g = g / g.sum()

        # Adaptive step size via line search
        best_new_score = score
        best_a_new = None
        for alpha in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03]:
            a_new = (1 - alpha) * a + alpha * g
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s < 0.01:
                continue
            a_new = a_new / s
            new_score = score_sequence(a_new)
            if new_score < best_new_score:
                best_new_score = new_score
                best_a_new = a_new

        if best_a_new is None:
            # No improvement from LP direction, try small random perturbation
            a_pert = a + rng.normal(0, 0.005, n)
            a_pert = normalize_nonneg(a_pert)
            pert_score = score_sequence(a_pert)
            if pert_score < best_score - 1e-10:
                a = a_pert
            else:
                break
        else:
            a = best_a_new.copy()

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def peak_push_refine(a, time_limit_s=120.0, verbose=False):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"    Peak push start: score={best_score:.6f}")

    it = 0
    max_iters = 2000
    for it in range(max_iters):
        if time.monotonic() - start_time > time_limit_s:
            break

        conv = np.convolve(a, a)
        k_star = int(conv.argmax())
        M = conv.max()
        sum_a = a.sum()
        score = 2.0 * n * M / (sum_a ** 2)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Compute contributions to the peak: conv[k_star] = sum_i a[i]*a[k_star-i]
        # Vectorize: for i in range(n), j = k_star - i, valid if 0 <= j < n
        i_valid = np.arange(n)
        j_vals = k_star - i_valid
        mask = (j_vals >= 0) & (j_vals < n)
        contributions = np.zeros(n)
        contributions[i_valid[mask]] = a[i_valid[mask]] * a[j_vals[mask]]

        # Identify top contributors and reduce them
        top_k = max(3, min(20, n // 40))
        top_idx = np.argsort(-contributions)[:top_k]

        a_new = a.copy()
        for i in top_idx:
            j = k_star - i
            if 0 <= j < n:
                if a[i] >= a[j]:
                    a_new[i] *= 0.92
                else:
                    a_new[j] *= 0.92

        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)

        if new_score < score - 1e-12:
            a = a_new
        else:
            # Try boosting low-contributing positions
            a_new2 = a.copy()
            low_idx = np.argsort(contributions)[:top_k]
            a_new2[low_idx] *= 1.08
            a_new2 = normalize_nonneg(a_new2)
            new_score2 = score_sequence(a_new2)
            if new_score2 < score - 1e-12:
                a = a_new2
            else:
                # Try a mixed strategy
                a_new3 = a.copy()
                a_new3[top_idx[:top_k//2]] *= 0.95
                a_new3[low_idx[:top_k//2]] *= 1.05
                a_new3 = normalize_nonneg(a_new3)
                new_score3 = score_sequence(a_new3)
                if new_score3 < score - 1e-12:
                    a = a_new3
                else:
                    break

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    Peak push done: best={best_score:.6f} after {it+1} iters")
    return best_a, best_score


def local_search_refine(a, time_limit_s=120.0, seed=42, verbose=False):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"    LS start: score={best_score:.6f}")

    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        # Random block perturbation
        width = max(2, int(rng.integers(max(2, n // 200), max(3, n // 4))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)

        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.04, size=stop - start))

        # Occasionally try anti-symmetric modification
        if rng.random() < 0.15:
            half = n // 2
            delta = rng.normal(0, 0.02, half)
            candidate[:half] += delta
            candidate[half:] -= delta[:n - half] if n - half == half else delta[:n-half]

        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)

        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 200 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_asymmetric(n):
    t = np.arange(n, dtype=np.float64)
    x = t / max(1, n - 1)
    return np.maximum(0, (1 - x) ** 0.7 * (1 + 0.5 * np.cos(np.pi * x)))


def initial_bimodal(n):
    t = np.arange(n, dtype=np.float64)
    c1 = n * 0.25
    c2 = n * 0.65
    sig = n * 0.12
    a = np.exp(-0.5 * ((t - c1) / sig) ** 2) + 0.6 * np.exp(-0.5 * ((t - c2) / sig) ** 2)
    return a


def initial_skewed_gaussian(n):
    t = np.arange(n, dtype=np.float64)
    x = t / max(1, n - 1)
    # Skewed Gaussian-like
    center = 0.4
    width = 0.15
    a = np.exp(-0.5 * ((x - center) / width) ** 2) * np.maximum(0, 1 + 0.5 * (x - center) / width)
    return a


def upsample_sequence(a, new_n):
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n)
    return np.interp(new_x, old_x, a)


def propose_candidate(seed=42, budget_s=2400):

    Phase 2: Peak-pushing and local search polishing on best candidates.
    Phase 3: Final LP and local search refinement.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20.0

    best_score = float('inf')

    prev_seq = None
    try:
        hs1 = globals().get('height_sequence_1', None)

            prev_seq = np.array(hs1, dtype=np.float64)
            if len(prev_seq) > 10 and np.all(np.isfinite(prev_seq)) and np.all(prev_seq >= 0):
                prev_score = score_sequence(prev_seq)
                print(f"Previous sequence: n={len(prev_seq)}, score={prev_score:.6f}")
                if prev_score < best_score:
                    best_score = prev_score
                    best_seq = prev_seq.copy()
    except Exception:
        prev_seq = None

    # Phase 1: Multi-start LP refinement
    phase1_fraction = 0.55
    phase1_time = phase1_fraction * budget_s
    phase1_deadline = total_start + phase1_time

    configs = []

    if prev_seq is not None:
        prev_n = len(prev_seq)
        configs.append((prev_n, 'prev', 0.18))

        a_refined, score = lp_refine(a, time_limit_s=time_limit, verbose=True, seed=seed + n)
```
