def _fft_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def conv_full(a):
    """Full self-convolution of a, length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(n)
    F = fft(a, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def score_of(a):
    """Compute the evaluation score for a normalized sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    c = conv_full(a)
    if c.size == 0:
        return np.inf
    cmax = float(c.max())
    if not np.isfinite(cmax):
        return np.inf
    return float(2.0 * n * cmax / (s * s))


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out


def make_inv_sqrt(n):
    """1/sqrt(k+1) sequence normalized to sum 1."""
    k = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(k + 1.0)
    return a / a.sum()


def make_inv_sqrt_windowed(n, taper_frac=0.1):
    """1/sqrt(k+1) with gentle endpoint tapering."""
    k = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(k + 1.0)
    if taper_frac > 0:
        t = taper_frac * n
        for i in range(int(t)):
            a[i] *= (i + 1) / (t + 1)
        for i in range(int(t)):
            a[n - 1 - i] *= (i + 1) / (t + 1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.full(n, 1.0 / n)
    return a / s


def make_beta_like(n, alpha=0.5, beta=0.5):
    """Beta-distribution-like shape: x^alpha * (1-x)^beta."""
    x = np.linspace(1e-6, 1.0 - 1e-6, n)
    a = np.power(x, alpha) * np.power(1.0 - x, beta)
    s = a.sum()
    if s < 1e-12:
        return np.full(n, 1.0 / n)
    return a / s


def make_power_law(n, exponent=0.5):

    k = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(k, -exponent)
