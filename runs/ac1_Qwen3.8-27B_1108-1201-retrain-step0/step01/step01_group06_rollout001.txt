def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def compute_exact_obj(a):
    """Compute exact 2n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return 1e10
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def smooth_objective(x, t):
    """
    Smooth objective: 2n * logsumexp(t*conv(a))/t where a = softmax(x).
    Returns (value, gradient).
    """
    n = len(x)
    a = softmax(x)
    conv = fast_convolve(a)

    tconv = t * conv
    tconv_max = tconv.max()
    exp_tconv = np.exp(tconv - tconv_max)
    log_sum_exp = tconv_max + np.log(exp_tconv.sum())
    smooth_max = log_sum_exp / t
    obj = 2.0 * n * smooth_max

    # Weights for each lag
    w = exp_tconv / exp_tconv.sum()

    # Gradient of smooth_max w.r.t. a
    # d smooth_max / d a[j] = (1/t) * sum_k w[k] * 2*a[k-j] (for valid k-j)
    grad_a = np.zeros(n)
    thresh = w.max() * 1e-6
    active = np.where(w > thresh)[0]
    for k_idx in active:
        k = int(k_idx)
        wk = w[k]
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            grad_a[j_lo:j_hi + 1] += 2.0 * wk * a[k - j_hi:k - j_lo + 1][::-1]
    grad_a *= 2.0 * n / t

    # Chain rule through softmax: a_i = exp(x_i)/S
    # d obj / d x_j = sum_i (d obj/d a_i) * a_i * (delta_ij - a_j)
    #               = a_j * (sum_i grad_a_i * a_i) - a_j^2 * (sum_i grad_a_i)
    ga = a * (grad_a @ a - a * grad_a.sum())

    return obj, ga

def proj_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_optimize(a0, time_limit, eta0=0.1):
    """
    Projected subgradient descent with exact peak.
    a0: initial point on simplex
    Returns optimized a.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    patience = 0
    max_patience = 60
    t0 = time.monotonic()
    avg_step = 0.0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_conv = conv.max()

        if max_conv < 1e-15:
            break

        # Find peaks (all lags within 1% of max)
        peak_thresh = max_conv * 0.99
        peak_indices = np.where(conv >= peak_thresh)[0]

        # Compute subgradient: average gradient over active peaks
        g = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                g[j_lo:j_hi + 1] += 2.0 * a[k - j_hi:k - j_lo + 1][::-1]
        g /= len(peak_indices)
        g *= 2.0 * n  # Scale by objective factor

        # Center the gradient (for simplex projection)
        g_centered = g - g.mean()

        # Adaptive step size
        g_norm = np.sqrt((g_centered ** 2).sum())
        if g_norm < 1e-15:
            break
        step = eta / g_norm
        avg_step = 0.9 * avg_step + 0.1 * step

        # Update
        a_new = proj_simplex(a - step * g_centered)

        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new

        if patience >= max_patience:
            patience = 0
            eta *= 0.5
            if eta < 0.001:
                eta = eta0
                max_patience = 30
            max_patience = min(max_patience + 10, 100)

    scan_time_total = min(600.0, (deadline - time.monotonic()) * 0.3)
    time_per_config = max(5.0, scan_time_total / max(1, len(scan_configs)))

    scan_results = []
    for i, (n, name, a0) in enumerate(scan_configs):
        if time.monotonic() > deadline - 120:
            break
        t_alloc = min(time_per_config, deadline - time.monotonic() - 120)
        if t_alloc < 3:
            break

        a_opt, obj_opt = optimize_from_init(a0, n, t_alloc, rng)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan [{i}]: n={n}, {name}, score={score:.6f}")

    print(f"Phase 1 done. Best so far: {best_score:.6f}")

    # Phase 2: Deep optimization on best configs
    scan_results.sort(key=lambda x: x[0])

    # Take top diverse configs
    deep_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            deep_configs.append((n, name, a_opt))
            seen.add(key)
        if len(deep_configs) >= 8:
            break

    # Also add the current best
    if best_sequence is not None:
        deep_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    # Deep optimization budget
    remaining = deadline - time.monotonic() - 30
    time_per_deep = max(20.0, remaining / max(1, len(deep_configs)))

    print(f"Phase 2: Deep optimization ({len(deep_configs)} configs)")
    for i, (n, name, a0) in enumerate(deep_configs):
        if time.monotonic() > deadline - 30:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 30)
        if t_alloc < 10:
            break

        a_opt, obj_opt = optimize_from_init(a0, n, t_alloc, rng)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep [{i}]: n={n}, {name}, score={score:.6f}")

    # Phase 3: Final polish on the best sequence
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10
        n = len(best_sequence)
        print(f"Phase 3: Final polish (n={n}, remaining={remaining:.0f}s)")

        # Multiple polish attempts with different strategies
        a_current = best_sequence.copy()
        for attempt in range(5):
            if remaining < 5:
                break
            t_single = remaining / (5 - attempt)
            if t_single < 3:
                break

            # Alternate between L-BFGS and subgradient
            if attempt % 2 == 0:
                x0 = np.log(a_current + 1e-15)
                for temp in [100.0, 250.0]:
                    if remaining < 3:
                        break
                    x_opt = l_bfgs_optimize(x0, temp, time_limit=t_single / 2 - 1, maxiter=5000)
                    a_opt = softmax(x_opt)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        a_current = a_opt.copy()
                        print(f"  Polish L-BFGS t={temp}: score={score:.6f}")
                    x0 = x_opt
                    remaining = deadline - time.monotonic() - 10
            else:
                a_opt, _ = subgradient_optimize(a_current, time_limit=t_single - 2, eta0=0.01)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    a_current = a_opt.copy()
                    print(f"  Polish SubGrad: score={score:.6f}")

            remaining = deadline - time.monotonic() - 10

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL: n={len(best_sequence)}, score={best_score:.6f}")
    return best_sequence


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Main entry point. Uses hybrid L-BFGS + subgradient optimization
    with log-sum-exp smooth surrogate and softmax parameterization.
    """
    result = run_search(seed=seed, budget_s=budget_s)
    return [float(x) for x in result]
