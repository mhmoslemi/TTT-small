```python
"""
Log-sum-exp smoothed optimization with L-BFGS and temperature annealing
for minimizing the normalized autocorrelation peak of non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    return exp_x / np.sum(exp_x)

def _true_score(a):
    """Compute the true objective: 2*n*max(convolve(a,a)) for sum(a)=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)

def _smoothed_obj_and_grad(x, T, n):
    """
    Compute the log-sum-exp smoothed objective and its gradient w.r.t. x.
    a = softmax(x), b = convolve(a,a), obj = 2n*T*log(sum(exp(b/T)))
    """
    a = _softmax(x)
    b = np.convolve(a, a)
    b_max = np.max(b)
    scaled = (b - b_max) / T
    scaled = np.clip(scaled, -50, 50)  # numerical stability
    exp_scaled = np.exp(scaled)
    S = np.sum(exp_scaled)
    obj = 2.0 * n * (b_max + T * np.log(S))
    
    p = exp_scaled / S
    # Gradient of b_k w.r.t. a_j: 2*a_{k-j} if 0<=k-j<n, else 0
    # grad_a_j = 2n * sum_k p_k * 2*a_{k-j} * I(0<=k-j<n)
    #           = 4n * sum_m a[m] * p[j+m]  (where 0<=j+m<=2n-2)
    #           = 4n * correlate(p, a, 'valid')[j]
    grad_a = 4.0 * n * np.correlate(p, a, mode='valid')
    # Convert to x-gradient via softmax chain rule
    grad_x = a * (grad_a - np.dot(grad_a, a))
    
    return obj, grad_x

def _make_starts(n, parent, rng):
    """Generate diverse starting points."""
    starts = []
    
    # 1. Parent (previous best)
    if parent is not None and len(parent) == n:
        p = np.clip(parent, 0, None).astype(np.float64)
        S = np.sum(p)
        if S > 1e-10:
            starts.append(p / S)
    
    # 2. Anti-symmetric two-level
    for eps in [0.3, 0.5, 0.7]:
        a = np.ones(n)
        a[:n//2] = 1.0 + eps
        a[n//2:] = 1.0 - eps
        starts.append(a / np.sum(a))
    
    # 3. Smooth cosine anti-symmetric
    i = np.arange(n, dtype=np.float64)
    for amp in [0.5, 0.8]:
        a = 1.0 + amp * np.cos(np.pi * i / (n - 1))
        a = np.maximum(a, 1e-6)
        starts.append(a / np.sum(a))
    
    # 4. Sigmoid transition
    mid = n / 2.0
    width = n / 8.0
    x = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((x - mid) / width))
    starts.append(a / np.sum(a))
    
    # 5. Linear ramp
    a = np.linspace(1.5, 0.5, n)
    starts.append(a / np.sum(a))
    
    # 6. Random
    a = rng.random(n) + 0.1
    starts.append(a / np.sum(a))
    
    return starts

def _anneal_lbfgs(a_init, deadline, n, T_values=None, verbose=True):
    """
    Optimize using log-sum-exp + L-BFGS with temperature annealing.
    Returns (best_a, best_score).
    """
    if T_values is None:
        T_values = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    
    best_a = a_init.copy()
    best_score = _true_score(best_a)
    x = np.log(a_init + 1e-30)
    
    for idx, T in enumerate(T_values):
        if time.monotonic() >= deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break
        
        # Determine max iterations based on remaining time
        # Each iteration takes roughly 5-20ms for n=6121
        # L-BFGS might do 3-10 function evals per iteration
        est_time_per_iter = 0.05  # conservative estimate
        max_iter = max(20, int(remaining * 0.8 / (len(T_values) - idx) / est_time_per_iter))
        max_iter = min(max_iter, 1000)
        
        try:
            result = minimize(
                _smoothed_obj_and_grad, x, method='L-BFGS-B',
                jac=True, args=(T, n),
                options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': max_iter * 5}
            )
            x = result.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error at T={T}: {e}")
            continue
        
        a = _softmax(x)
        score = _true_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose:
                print(f"    T={T:.6f}: score={score:.6f} (iter={result.nit}, status={result.message})")
        elif verbose:
            print(f"    T={T:.6f}: score={score:.6f} (no improvement, best={best_score:.6f})")
    
    return best_a, best_score

def _subgradient_polish(a_init, deadline, n, M=10, verbose=True):
    """
    Final polish using direct subgradient on the top-M peaks.
    """
    best_a = a_init.copy()
    best_score = _true_score(best_a)
    a = a_init.copy()
    
    eta = 0.05 / np.sqrt(n)
    no_improve = 0
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        score = 2.0 * n * b_max
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            eta *= 0.9
            no_improve = 0
            if eta < 1e-12:
                break
        
        # Get top M peaks
        k_idx = np.argpartition(b, -M)[-M:]
        b_sel = b[k_idx]
        wts = np.exp(-10.0 * (b_max - b_sel) / max(b_max, 1e-15))
        wts /= np.sum(wts)
        
        # Compute gradient
        grad = np.zeros(n)
        for ki in range(M):
            if wts[ki] < 1e-8:
                continue
            k = int(k_idx[ki])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                grad[idx] += wts[ki] * 2.0 * a[k - idx]
        
        # Center for sum constraint
        grad -= np.mean(grad)
        
        # Update
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        a = a - eta * grad / grad_norm
        
        # Project onto simplex
        a = _project_simplex(a, 1.0)
    
    return best_a, best_score

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
```
