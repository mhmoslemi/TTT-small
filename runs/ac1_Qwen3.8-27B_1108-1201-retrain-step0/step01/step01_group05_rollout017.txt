import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        return np.full(n, S/n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_self_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size-1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_subgrad_topk(a, c, k_indices):
    """Subgradient using top-k lags, weighted by their values."""
    n = len(a)
    grad = np.zeros(n)
    vals = c[k_indices]
    w = vals / vals.sum() if vals.sum() > 0 else np.ones(len(k_indices))/len(k_indices)
    for wi, ki in zip(w, k_indices):
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            grad[lo:hi+1] += wi * 2.0 * a[ki-hi:ki-lo+1][::-1]
    return grad

def run_optimization(a_init, n, S, time_limit, rng, top_k_start=1, top_k_end=64, lr_schedule='cosine'):

    a = np.clip(a_init, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_c = fft_self_conv(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    start = time.monotonic()
    last_improve = start
    no_improve_time = 0
    
    while time.monotonic() - start < time_limit:
        t += 1
        c = fft_self_conv(a)
        cur_max = c.max()
        
        if cur_max < best_c:
            best_c = cur_max
            best_a = a.copy()
            last_improve = time.monotonic()
        
        # Adaptive top-k: start small, grow over time
        frac = min(1.0, t / max(1, int(time_limit / 0.1)))  # grows over first 10% of time
        top_k = max(top_k_start, int(top_k_end * frac))
        top_k = min(top_k, len(c))

        grad = compute_subgrad_topk(a, c, top_indices)

        if gnorm < 1e-15:

        grad = grad / gnorm

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        mhat = m / (1 - beta1**t)
        vhat = v / (1 - beta2**t)
        a = project_simplex(a - lr * mhat / (np.sqrt(vhat) + eps), S)

import time

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):

    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho

def fft_self_conv(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1

    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgrad(a, c, indices):
    """Weighted subgradient of max(c) using specified lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    if len(indices) == 0:
        return grad
    vals = c[indices]
    total = vals.sum()
    if total <= 1e-15:
        return grad
    weights = vals / total
    for w_val, k in zip(weights, indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            segment = a[lo:hi + 1]

            # For j in [lo, hi], k-j is in [k-hi, k-lo] which is in [lo, hi]

    max_c = c.max()
    threshold = max_c * (1.0 - hysteresis_margin)
    active_set = set(np.where(c >= threshold)[0].tolist())

    start = time.monotonic()
    last_improve = start
    stuck_counter = 0
    lr_mult = 1.0

    while time.monotonic() - start < time_limit:
        t += 1
        c = fft_self_conv(a)
        cur_max = c.max()

        if cur_max < best_max * (1.0 - 1e-10):
            best_max = cur_max
            best_a = a.copy()
            last_improve = time.monotonic()
            stuck_counter = 0

        # Stuck detection
        if time.monotonic() - last_improve > 20:
            stuck_counter += 1
            if stuck_counter > 2:
                lr_mult *= 0.7
                stuck_counter = 0
                # Occasional noise kick
                if rng.random() < 0.5:
                    noise = rng.normal(0, 0.005 * S / n, n)
                    a = project_simplex(a + noise, S)

        # Adaptive top-k: interpolate from start to end
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1.0, time_limit * 0.6))
        cur_top_k = int(top_k_start + (top_k_end - top_k_start) * frac)
        cur_top_k = max(top_k_start, min(cur_top_k, 2 * n - 1))

        # Update hysteresis set
        max_c = c.max()
        threshold = max_c * (1.0 - hysteresis_margin)
        newly_active = set(np.where(c >= threshold)[0].tolist())
        # Shrink: remove lags that are well below max
        shrink_threshold = max_c * (1.0 - hysteresis_margin * 3)
        to_remove = set(k for k in active_set if c[k] < shrink_threshold)
        active_set = (active_set - to_remove) | newly_active
        # Cap the active set size
        if len(active_set) > cur_top_k * 3:
            active_arr = np.array(sorted(active_set))
            active_vals = c[active_arr]
            top_in_active = active_arr[np.argsort(active_vals)[::-1][:cur_top_k * 2]]
            active_set = set(top_in_active.tolist())

        # Use top of active set for subgradient
        if len(active_set) == 0:
            indices = np.argsort(c)[::-1][:cur_top_k]
        else:
            active_arr = np.array(sorted(active_set))
            active_vals = c[active_arr]
            sorted_idx = np.argsort(active_vals)[::-1][:cur_top_k]
            indices = active_arr[sorted_idx]

        grad = compute_subgrad(a, c, indices)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Adam update with decaying LR
        lr = base_lr * lr_mult * (1.0 / (1.0 + 0.02 * t))
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

        # Periodic noise for exploration
        if t % noise_interval == 0 and elapsed < time_limit * 0.5:
            noise = rng.normal(0, 0.01 * S / n, n)
            a = project_simplex(a + noise, S)

    return best_a, best_max


def run_mirror_descent(a_init, n, S, time_limit, rng,
                       eta_start=1.0, top_k=10):
    """Mirror descent (multiplicative weights) which naturally preserves non-negativity."""
    a = normalize_to_sum(a_init, S)
    c = fft_self_conv(a)
    best_max = c.max()
    best_a = a.copy()
    t = 0
    start = time.monotonic()
    last_improve = start
    eta = eta_start

    while time.monotonic() - start < time_limit:
        t += 1
        c = fft_self_conv(a)
        cur_max = c.max()

        if cur_max < best_max * (1.0 - 1e-10):
            best_max = cur_max
            best_a = a.copy()
            last_improve = time.monotonic()
            eta = max(eta * 1.05, 0.01)  # increase step on improvement
        else:
            if time.monotonic() - last_improve > 15:
                eta *= 0.8

        top_indices = np.argsort(c)[::-1][:top_k]
        grad = compute_subgrad(a, c, top_indices)
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.full(n, S / n)
        else:
            a = a_new * (S / s)

    return best_a, best_max


def polish_sequence(a, n, S, time_limit, rng):
    """Fine polishing with very small steps and large top-k."""
    a = normalize_to_sum(a, S)
    c = fft_self_conv(a)
    best_max = c.max()
    best_a = a.copy()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.95, 0.999, 1e-10
    base_lr = 0.3 * S / n
    top_k = min(200, 2 * n - 1)

    start = time.monotonic()
    last_improve = start
    lr_decay = 1.0

    while time.monotonic() - start < time_limit:
        t += 1
        c = fft_self_conv(a)
        cur_max = c.max()

        if cur_max < best_max * (1.0 - 1e-12):
            best_max = cur_max
            best_a = a.copy()
            last_improve = time.monotonic()

        if time.monotonic() - last_improve > 30:
            lr_decay *= 0.5
            if lr_decay < 0.01:
                break

        top_indices = np.argsort(c)[::-1][:top_k]
        grad = compute_subgrad(a, c, top_indices)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        lr = base_lr * lr_decay / (1.0 + 0.01 * t)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

    return best_a, best_max


def get_initializations(rng, n):
    """Generate diverse initializations of length n."""
    S = np.sqrt(2.0 * float(n))
    inits = []

    # Try to use the given sequence if available
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", normalize_to_sum(given, S)))
        elif given.size > 0:
            # Resample to length n
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, given.size)
            x_new = np.linspace(0, 1, n)
            f = interp1d(x_old, np.clip(given, 0, 1000), fill_value=0, bounds_error=False)
            resampled = f(x_new)
            inits.append(("given_resampled", normalize_to_sum(resampled, S)))
    except Exception:
        pass

    # Edge-heavy (U-shape)
    inits.append(("edge_heavy", normalize_to_sum(make_edge_heavy(n, rng, edge_frac=0.3, smoothness=2.0), S)))
    inits.append(("edge_heavy_sharp", normalize_to_sum(make_edge_heavy(n, rng, edge_frac=0.4, smoothness=4.0), S)))

    # Inverted cosine (high at edges)
    inits.append(("inv_cos", normalize_to_sum(make_cosine_window(n, invert=True) + 0.01, S)))

    # Two bump
    inits.append(("two_bump", normalize_to_sum(make_two_bump(n, rng), S)))

    # Smooth random
    smooth_len = 64
    base = rng.random(smooth_len)
    base = np.convolve(base, np.ones(8) / 8, mode='same')
    from scipy.interpolate import interp1d
    x_old = np.linspace(0, 1, smooth_len)
    x_new = np.linspace(0, 1, n)
    f = interp1d(x_old, base, fill_value=0, bounds_error=False)
    smooth = f(x_new)
    inits.append(("smooth_rand", normalize_to_sum(smooth, S)))

    # Pure random
    inits.append(("random", normalize_to_sum(rng.random(n), S)))

    # Ramp (high at start, low at end) - asymmetric
    ramp = np.linspace(1, 0.1, n)
    inits.append(("ramp", normalize_to_sum(ramp, S)))

    # V-shape (low in middle)
    v_shape = np.abs(np.linspace(-1, 1, n)) + 0.1
    inits.append(("v_shape", normalize_to_sum(v_shape, S)))

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 10.0)

    global_best = np.inf
    global_a = None
    global_n = 0

    # Try multiple lengths
    lengths_to_try = [4096, 8192]

    # First pass: quick evaluation of each initialization for each length
    # Second pass: optimize the best ones

    # Pre-allocate structure
    candidates = []  # (n, init_name, init_array, quick_score)

    for n in lengths_to_try:
        S = np.sqrt(2.0 * float(n))
        inits = get_initializations(rng, n)
        for name, arr in inits:
            if time.monotonic() > deadline - 60:
                break
            c = fft_self_conv(arr)
            score = c.max()
            candidates.append((n, name, arr.copy(), score))
            print("n=%d init=%s quick_score=%.6f" % (n, name, score))

    # Sort by quick score
    candidates.sort(key=lambda x: x[3])

    # Allocate time for optimization
    # Give more time to better candidates
    total_opt_time = min(budget_s - 20, deadline - time.monotonic() - 15)
    n_candidates = min(12, len(candidates))

    # Time allocation: more for top candidates
    weights = []
    for i in range(n_candidates):
        weights.append(max(1.0, 5.0 - 0.4 * i))
    total_w = sum(weights)
    times = [total_opt_time * w / total_w for w in weights]

    for i in range(n_candidates):
        if time.monotonic() > deadline - 15:
            break
        n, name, arr, quick_score = candidates[i]
        S = np.sqrt(2.0 * float(n))
        alloc = min(times[i], deadline - time.monotonic() - 15)
        if alloc < 5:
            break

        print("\n--- Optimizing n=%d init=%s (budget=%.0fs) ---" % (n, name, alloc))

        # Run Adam with hysteresis
        t_adam = alloc * 0.6
        a_adam, score_adam = run_adam_hysteresis(
            arr, n, S, t_adam, rng,
            top_k_start=1, top_k_end=80,
            base_lr=1.5 * S / n,
            hysteresis_margin=0.15,
            noise_interval=2000
        )
        print("  adam: score=%.6f" % score_adam)
        if score_adam < global_best:
            global_best = score_adam
            global_a = a_adam.copy()
            global_n = n

        # Run mirror descent on the adam result
        if time.monotonic() < deadline - 15:
            t_mir = min(alloc * 0.25, deadline - time.monotonic() - 15)
            if t_mir > 3:
                a_mir, score_mir = run_mirror_descent(
                    a_adam, n, S, t_mir, rng,
                    eta_start=0.8, top_k=10
                )
                print("  mirror: score=%.6f" % score_mir)
                if score_mir < global_best:
                    global_best = score_mir
                    global_a = a_mir.copy()
                    global_n = n

        # Quick verify
        if score_adam < global_best + 0.01:
            ver = evaluate_sequence(a_adam.tolist())
            if np.isfinite(ver):
                print("  verify: %.6f" % ver)
                if ver < global_best:
                    global_best = ver
                    global_a = a_adam.copy()
                    global_n = n

    # Polish phase
    if global_a is not None:
        left = deadline - time.monotonic()
        print("\n--- Polish phase (budget=%.0fs) ---" % left)
        if left > 10:
            n = global_n
            S = np.sqrt(2.0 * float(n))
            a_pol, score_pol = polish_sequence(global_a, n, S, left * 0.7, rng)
            print("  polish: score=%.6f" % score_pol)
            if score_pol < global_best:
                global_best = score_pol
                global_a = a_pol.copy()

            # Final mirror descent polish
            left2 = deadline - time.monotonic()
            if left2 > 5:
                a_mir2, score_mir2 = run_mirror_descent(
                    global_a, n, S, left2, rng,
                    eta_start=0.3, top_k=50
                )
                print("  final mirror: score=%.6f" % score_mir2)
                if score_mir2 < global_best:
                    global_best = score_mir2
                    global_a = a_mir2.copy()

    # Final verification
    if global_a is not None:
        final_ver = evaluate_sequence(global_a.tolist())
        if np.isfinite(final_ver):
            print("\nFinal verified: %.6f" % final_ver)
            global_best = final_ver
        else:
            print("\nWARNING: final verification failed, using previous best")
    else:
        print("\nWARNING: no valid sequence found, using fallback")
        global_a = np.ones(256, dtype=np.float64)

    # Ensure validity
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("\nBest score found: %.6f, length=%d" % (global_best, len(global_a)))
    return [float(x) for x in global_a]
