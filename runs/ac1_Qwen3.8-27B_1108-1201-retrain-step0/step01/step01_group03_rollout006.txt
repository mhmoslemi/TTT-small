def _autocorr_fft(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    N = 2 * n - 1
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)

def _softmax_gradient(a, tau):
    """Compute gradient of soft-max of autocorrelation w.r.t. a, using FFT."""
    n = len(a)
    c = _autocorr_fft(a)
    c_max = c.max()

    w = np.exp(tau * (c - c_max))

    # This is 2 * cross_correlation(w, a)[j]

def _spectral_init(n, seed):

    rng = np.random.default_rng(seed)
    N = 2 * n - 1
    # Random power spectrum with some structure

    # Low-pass filter to ensure smoothness

    power = mask * (1.0 + 0.5 * rng.random(N // 2 + 1))
    power[0] = 0  # Remove DC for zero-mean
    # Random phases
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    # Construct spectrum
    A = np.zeros(N // 2 + 1, dtype=complex)
    A = np.sqrt(power) * np.exp(1j * phases)
    # Make it Hermitian-symmetric for real output
    if N % 2 == 1:
        # N is odd (2n-1 is odd), so N//2+1 is the Nyquist
        full = np.zeros(N, dtype=complex)
        full[:N // 2 + 1] = A
        for k in range(1, N // 2):
            full[N - k] = np.conj(A[k])
    else:
        full = np.zeros(N, dtype=complex)
        full[:N // 2 + 1] = A
        for k in range(1, N // 2 + 1):
            full[N - k] = np.conj(A[k])

    a_padded = np.fft.irfft(full, N)
    a = np.abs(a_padded[:n])  # Take abs for non-negativity
    return a

    # --- Phase 0: Try previous best if available ---
    prev = None
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size >= 8 and np.all(np.isfinite(prev)):
            prev = np.maximum(prev, 0.0)
            print(f"Using previous best: n={len(prev)}, score={_objective(prev):.6f}")
            _try_and_update(prev, _objective(prev))
    except (NameError, Exception) as e:
        print(f"No previous sequence: {e}")

    # --- Phase 1: Quick gradient optimization from multiple starts ---
    print(f"\nPhase 1: Gradient optimization, deadline in {deadline - time.monotonic():.0f}s")

    starting_points = []

    # From previous best
    if prev is not None and prev.size > 0:
        for L in [min(512, len(prev)), min(256, len(prev)), min(1024, len(prev))]:
            if L >= 16:
                starting_points.append(prev[:L].copy())

    # Spectral initializations
    for n in [256, 512]:
        for s in [0, 1, 2]:
            starting_points.append(_spectral_init(n, seed + s))

    # Structured initializations
    for n in [256, 512]:
        for freq in [1, 2, 3, 5]:
            starting_points.append(_structured_init(n, freq))

    # Random starts
    for n in [256, 512]:
        starting_points.append(rng.random(n) * 0.5 + 0.25)

    # Sort by length (shorter first for quick exploration)
    starting_points.sort(key=lambda x: len(x))

    # Run gradient optimization on each start with a time slice
    n_starts = len(starting_points)
    for idx, a0 in enumerate(starting_points):
        if time.monotonic() > deadline - 60.0:
            break
        # Allocate time: more time for shorter sequences (faster iterations)
        n = len(a0)
        time_per_start = min(60.0, max(15.0, 300.0 / n_starts * (512.0 / n)))
        t_end = time.monotonic() + time_per_start

        seq, score = _optimize_gradient(a0, t_end)
        _try_and_update(seq, score)

        # Also try hybrid method
        if time.monotonic() < deadline - 30.0:
            t_end2 = time.monotonic() + min(30.0, time_per_start * 0.5)
            seq2, score2 = _optimize_hybrid(a0, t_end2)
            _try_and_update(seq2, score2)

    # --- Phase 2: Deep optimization on best candidates ---
    print(f"\nPhase 2: Deep refinement, deadline in {deadline - time.monotonic():.0f}s")

    if best_seq is not None:
        # Extended gradient optimization on the best
        t_end = deadline - 30.0
        seq, score = _optimize_gradient(best_seq, t_end,
                                        tau_start=10.0, tau_end=80.0,
                                        lr_start=0.08, lr_end=0.002,
                                        check_every=50)
        _try_and_update(seq, score)

        # Hybrid method on best
        if time.monotonic() < deadline - 20.0:
            t_end2 = deadline - 20.0
            seq2, score2 = _optimize_hybrid(best_seq, t_end2)
            _try_and_update(seq2, score2)

    # --- Phase 3: LP finishing ---
    print(f"\nPhase 3: LP finishing, deadline in {deadline - time.monotonic():.0f}s")

    if best_seq is not None and len(best_seq) <= 768:
        t_end = deadline - 5.0
        seq, score = _lp_refine(best_seq, t_end)
        _try_and_update(seq, score)
    else:
        # Try a shorter version
        if best_seq is not None:
            n_short = min(512, len(best_seq))
            a_short = best_seq[:n_short]
            t_end = deadline - 5.0
            seq, score = _lp_refine(a_short, t_end)
            _try_and_update(seq, score)

    # --- Phase 4: Final polish ---
    print(f"\nPhase 4: Final polish, deadline in {deadline - time.monotonic():.0f}s")

    if best_seq is not None and time.monotonic() < deadline - 10.0:
        # One more round of gradient descent with very low lr
        t_end = deadline - 5.0
        seq, score = _optimize_gradient(best_seq, t_end,
                                        tau_start=50.0, tau_end=100.0,
                                        lr_start=0.03, lr_end=0.001,
                                        check_every=20)
        _try_and_update(seq, score)

        # LP on the polished result
        if len(best_seq) <= 512 and time.monotonic() < deadline - 3.0:
            seq2, score2 = _lp_refine(best_seq, deadline - 1.0, max_iters=10)
            _try_and_update(seq2, score2)

    # --- Final output ---
    if best_seq is not None:
        final_score = _objective(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
        # Validate with evaluate_sequence
        try:
            valid = [float(x) for x in best_seq]
            eval_score = float(evaluate_sequence(valid))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
            if eval_score < best_score:
                best_score = eval_score
        except Exception as e:
            print(f"Validation note: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Fallback: using uniform sequence of length 256.")
    return [1.0] * 256
