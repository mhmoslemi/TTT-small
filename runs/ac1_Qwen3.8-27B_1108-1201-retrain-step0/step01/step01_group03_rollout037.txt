```python
def build_lp_matrix(a):
    """Build the LP constraint matrix for conv(a,g) <= M."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        k_start = j
        k_end = min(2*n-1, j+n)
        A[k_start:k_end, j] = a[:k_end-k_start]
    return A


def lp_refine(a, deadline, max_iters=200):
    """LP-based refinement."""
    n = len(a)
    if n > 2048:
        return a, compute_obj(a)
    
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)
    
    A = build_lp_matrix(a)  # will be rebuilt if a changes significantly
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break
        
        c = np.convolve(a, a)
        M = float(c.max())
        
        c_obj = -np.ones(n)
        b = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break
        
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1.0-t)*a + t*g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            s = compute_obj(a_new)
            if s < best_score:
                a = a_new
                best = a.copy()
                best_score = s
                improved = True
                # Rebuild LP matrix since a changed
                A = build_lp_matrix(a)
                break
        
        if not improved:
            # Try a small gradient step
            grad = soft_gradient(a, 4.0)
            grad -= grad.mean()

            a = np.maximum(a, 0.0)
            a = normalize(a, S_t)
            A = build_lp_matrix(a)
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
```

```python
"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: for non-symmetric sequences, max(conv) < sum(a^2), enabling
objective values below 2. Uses soft-max gradient descent with annealed peak
focusing, LP-based refinement, stagnation-aware restarts, and multi-start
exploration with diverse structured initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (S * S)


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def soft_gradient(a, p):
    """Gradient direction of sum_k c[k]^p where c=convolve(a,a)."""
    n = len(a)
    c = np.convolve(a, a)
    c = np.maximum(c, 0.0)
    if p <= 1.0:
        w = np.ones(2 * n - 1)
    else:
        w = c ** (p - 1)
    # grad[i] ∝ sum_m c[i+m]^(p-1) * a[m] = correlate(w, a, 'valid')[i]
    grad = np.correlate(w, a, mode='valid')
    return grad


def logsumexp_gradient(a, tau):
    """Gradient of (1/tau)*log(sum exp(tau*c[k])) - smooth max approximation."""
    n = len(a)
    c = np.convolve(a, a)
    c = np.maximum(c, 0.0)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[c.argmax()] = 1.0
    else:
        w /= w_sum
    grad = np.correlate(w, a, mode='valid')
    return grad


def optimize_soft(a0, deadline, max_iters=8000, use_lse=False):
    """Soft-max / log-sum-exp gradient descent with annealing and restarts."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)

    score = compute_obj(a)
    best = a.copy()
    best_score = score
    stagnant = 0
    rng_local = np.random.default_rng(12345 + n)

    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break

        frac = it / max_iters
        if use_lse:
            tau = 5.0 + 25.0 * frac
            grad = logsumexp_gradient(a, tau)
        else:
            p = 2.0 + 12.0 * (frac ** 0.7)
            grad = soft_gradient(a, p)

        grad -= grad.mean()
        lr = 0.06 * (1.0 - 0.85 * frac) ** 1.5
        lr = max(lr, 0.002)

        a = a - lr * grad
        a = np.maximum(a, 0.0)
        a = normalize(a, S_t)

        if (it + 1) % 40 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
                stagnant = 0
            else:
                stagnant += 1

            if stagnant >= 4:
                noise_scale = 0.05 * stagnant
                noise = rng_local.random(n) * noise_scale
                a = best + noise
                a = np.maximum(a, 0.0)
                a = normalize(a, S_t)
                stagnant = 0

    return best, best_score


def build_lp_matrix(a):
    """Build constraint matrix A where A[k,j]*g[j] sums to conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_refine(a, deadline, max_iters=150):
    """LP-based refinement: find g maximizing sum(g) s.t. conv(a,g)<=M."""
    n = len(a)
    if n > 2048:
        return a, compute_obj(a)

    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)
    A = build_lp_matrix(a)

    for it in range(max_iters):
        if time.monotonic() > deadline - 8.0:
            break

        c = np.convolve(a, a)
        M = float(c.max())

        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break

        improved = False
        for t in [0.4, 0.25, 0.15, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            s = compute_obj(a_new)
            if s < best_score:
                a = a_new
                best = a.copy()
                best_score = s
                improved = True
                A = build_lp_matrix(a)
                break

        if not improved:
            grad = soft_gradient(a, 4.0)
            grad -= grad.mean()
            a = a - 0.015 * grad
            a = np.maximum(a, 0.0)
            a = normalize(a, S_t)
            A = build_lp_matrix(a)
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s

    return best, best_score


def run_optimization_pipeline(a0, deadline, use_lse=False):
    """Run full pipeline: gradient descent then LP refinement."""
    n = len(a0)
    t_grad_end = deadline - max(30.0, 0.15 * (deadline - time.monotonic()))
    seq, score = optimize_soft(a0, t_grad_end, use_lse=use_lse)
    if n <= 1536 and time.monotonic() < deadline - 10.0:
        seq, score = lp_refine(seq, deadline)
    return seq, score
```
