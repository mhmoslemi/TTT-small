Combined with local search polishing and multi-length exploration.

import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def lse_objective(a, beta):
    conv = np.convolve(a, a)
    m = conv.max()
    return m + np.log(np.sum(np.exp(beta * (conv - m)))) / beta

def lse_gradient(a, beta):
    conv = np.convolve(a, a)
    m = conv.max()
    exp_b = np.exp(beta * (conv - m))
    p = exp_b / exp_b.sum()
    n = len(a)
    # Compute g_j = 2 * sum_k p_k * a_{k-j} using FFT
    N = 1
    while N < 2 * (2 * n - 1):
        N *= 2
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(P * np.conj(A), N)

def project_simplex(a, S=1.0):
    """Project a onto {x >= 0, sum(x) = S}"""
    n = len(a)
    lo = a.min() - 1
    hi = a.max() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(0, a - mid))
        if s > S:
            lo = mid
        else:
            hi = mid

"""

of non-negative sequences. Combines multiplicative updates, projected gradient
fallback, multi-length exploration, and local search polishing.

import numpy as np
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def auto_conv(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(A * A, N)
    return conv[:2 * n - 1]


def lse_objective(a, beta):
    conv = auto_conv(a)
    m = conv.max()
    return m + np.log(np.sum(np.exp(beta * (conv - m)))) / beta


def lse_gradient(a, beta):
    conv = auto_conv(a)
    m = conv.max()
    exp_b = np.exp(beta * (conv - m))
    p = exp_b / exp_b.sum()
    n = len(a)
    N = 1
    while N < 2 * (2 * n - 1):
        N *= 2
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(P * np.conj(A), N)
    return 2.0 * corr[:n]


def project_simplex(a, S=1.0):

    lo = a.min() - 1.0
    hi = a.max() + 1.0

        s = np.sum(np.maximum(0.0, a - mid))
        if s > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.maximum(0.0, a - t)
