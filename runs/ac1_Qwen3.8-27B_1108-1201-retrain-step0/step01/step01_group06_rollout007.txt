Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact LP-guided descent.
Solves an LP at each step to find the optimal mixing direction.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def build_A(f):
    """Build (2n-1) x n constraint matrix. Column j has f in rows j..j+n-1."""
    n = len(f)
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def solve_lp(f, M, method='highs'):
    """Solve max sum(g) s.t. A@g <= M, g >= 0. Returns (g, optimal_value)."""
    n = len(f)
    A = build_A(f)
    c = -np.ones(n)
    b = np.full(2*n - 1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method=method)
    if res.success:
        return res.x, -res.fun
    return None, 0.0

def lp_descent(f0, time_limit=60.0, verbose=False):
    """
    LP-guided descent. f0 should be non-negative with positive sum.
    Returns (best_f, best_M_normalized) where M_normalized = 2n*max_conv/sum^2.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-10:
        f = np.ones(n) / n
        f_sum = 1.0
    f = f / f_sum  # normalize to sum = 1
    
    conv = fast_convolve(f)
    M = conv.max()
    best_M = M
    best_f = f.copy()
    best_obj = 2.0 * n * M  # since sum = 1
    
    t0 = time.monotonic()
    stuck_count = 0
    step_size = 0.5
    eps_factor = 1.0  # multiplier for M in the LP constraint
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        # Solve LP
        g, lp_val = solve_lp(f, M * eps_factor)
        
        if g is None:
            # LP failed, try with larger M
            eps_factor *= 1.1
            if eps_factor > 2.0:
                break
            continue
        
        if lp_val <= 1.0 + 1e-9:

            stuck_count += 1

                eps_factor = 1.01
                step_size = 0.3

                conv = fast_convolve(f)

                step_size = 0.5

        g_sum = g.sum()
        if g_sum < 1e-10:

        h = g / g_sum

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact LP-guided descent with
line search, subgradient fallback, and multi-start strategy.
The LP finds the optimal mixing direction at each step, guaranteeing monotone
descent of the max-convolution when mixed with a sufficiently small step.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_A(f):
    """Build (2n-1) x n constraint matrix where A[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, upper_bound=None):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k, 0 <= g[j] <= upper_bound.
    Returns (g, optimal_sum) or (None, 0).
    """
    n = len(f)
    A = build_A(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    if upper_bound is not None:
        bounds = [(0, upper_bound)] * n
    else:
        bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, 0.0


def compute_subgradient(f, conv, active_frac=0.05):
    """
    Compute a (sub)gradient of max(conv) w.r.t. f using the active set.
    Returns gradient vector of length n.
    """
    n = len(f)
    max_val = conv.max()
    if max_val < 1e-15:
        return np.zeros(n)
    
    # Get active indices (top active_frac of positions)
    threshold = max_val * (1 - active_frac)
    active = np.where(conv >= threshold)[0]
    
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        # grad[j] += 2 * f[k-j] for valid j
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # For j in [j_start, j_end], f[k-j] = f[k-j_start : k-j_start+1][::-1]
            # = f[k-j_end : k-j_start+1][::-1]
            grad[j_start:j_end + 1] += 2.0 * f[k - j_end:k - j_start + 1][::-1]
    
    grad /= len(active)
    return grad


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def lp_descent(f0, time_limit, verbose=False, seed=0):
    """
    LP-guided descent with line search and subgradient fallback.
    f0: non-negative array, will be normalized to sum=1.
    Returns (best_f, best_objective) where objective = 2n*max_conv (sum=1 normalized).
    """
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = float(conv.max())
    best_M = M
    best_f = f.copy()
    best_obj = 2.0 * n * M

    t0 = time.monotonic()
    stuck_count = 0
    lp_copies = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        # --- Try LP step ---
        success = False
        M_for_lp = M * (1.0 + 0.005 * stuck_count) if stuck_count > 0 else M

        g, lp_val = solve_lp(f, M_for_lp, upper_bound=3.0 * max(1.0/n, f.max() * 3))

        if g is not None and lp_val > 1.0 + 1e-8:
            g_sum = g.sum()
            if g_sum > 1e-10:
                h = g / g_sum
                # Line search
                best_t = 0.0
                best_M_try = M
                for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
                    f_try = (1 - t) * f + t * h
                    f_try = np.maximum(f_try, 0.0)
                    st = f_try.sum()
                    if st < 1e-10:
                        continue
                    f_try = f_try / st
                    M_try = float(fast_convolve(f_try).max())
                    if M_try < best_M_try - 1e-14:
                        best_M_try = M_try
                        best_t = t

                if best_t > 0:
                    f = (1 - best_t) * f + best_t * h
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()
                    conv = fast_convolve(f)
                    M = float(conv.max())
                    success = True

        if success:
            stuck_count = 0
            lp_copies += 1
            if M < best_M:
                best_M = M
                best_f = f.copy()
                best_obj = 2.0 * n * M
                if verbose and lp_copies % 50 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LP step {lp_copies}: M={M:.8f}, obj={best_obj:.8f}, t={elapsed:.1f}s")
        else:
            stuck_count += 1

            if stuck_count <= 3:
                # Try subgradient step
                sg = compute_subgradient(f, conv, active_frac=0.1)
                eta = 0.5 / (np.linalg.norm(sg) + 1e-10)
                f_new = project_simplex(f - eta * sg)
                conv_new = fast_convolve(f_new)
                M_new = float(conv_new.max())
                if M_new < M:
                    f = f_new
                    conv = conv_new
                    M = M_new
                    if M < best_M:
                        best_M = M
                        best_f = f.copy()
                        best_obj = 2.0 * n * M
                    stuck_count = 0
                    lp_copies += 1
                else:
                    # Try smaller step
                    for eta_frac in [0.3, 0.1, 0.05]:
                        f_try = project_simplex(f - eta * eta_frac * sg)
                        M_try = float(fast_convolve(f_try).max())
                        if M_try < M:
                            f = f_try
                            M = M_try
                            if M < best_M:
                                best_M = M
                                best_f = f.copy()
                                best_obj = 2.0 * n * M
                            stuck_count = 0
                            break
            elif stuck_count <= 8:
                # Multiplicative update with subgradient
                sg = compute_subgradient(f, conv, active_frac=0.2)
                sg_centered = sg - sg.mean()
                log_factor = -0.3 * sg_centered
                log_factor -= log_factor.max()
                f_new = f * np.exp(log_factor)
                f_new = np.maximum(f_new, 0.0)
                f_new = f_new / f_new.sum()
                M_new = float(fast_convolve(f_new).max())
                if M_new < M:
                    f = f_new
                    M = M_new
                    if M < best_M:
                        best_M = M
                        best_f = f.copy()
                        best_obj = 2.0 * n * M
                    stuck_count = 0
                    lp_copies += 1
            elif stuck_count == 9:
                # Perturb f
                noise = rng.uniform(0, 0.02, n)
                f = f + noise
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                conv = fast_convolve(f)
                M = float(conv.max())
                stuck_count = 0
            else:
                # Try a different perturbation strategy
                if stuck_count % 10 == 0:
                    # Random restart with blend
                    frac = min(0.5, 0.1 * (stuck_count // 10 + 1))
                    f_rand = rng.dirichlet(np.ones(n) * 0.3)
                    f = (1 - frac) * f + frac * f_rand
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()
                    conv = fast_convolve(f)
                    M = float(conv.max())
                else:
                    # Small multiplicative random perturbation
                    multipliers = rng.uniform(0.8, 1.2, n)
                    f = f * multipliers
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()
                    conv = fast_convolve(f)
                    M = float(conv.max())
                    stuck_count = 0

                if stuck_count > 50:
                    # Give up on this initialization
                    break

    return best_f, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussians
    for sf in [0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Bimodal
    bi = np.zeros(n)
        w = max(1, n // 6)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Cosine
    cos_bump = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_bump /= cos_bump.sum()
    inits['cosine'] = cos_bump

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Random sparse-ish
    r = rng.random(n)
    r = r ** 3  # make sparse
    r = np.maximum(r, 0.01)
    r /= r.sum()
    inits['random_sparse'] = r

    # Parent-based
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000.0)
        if p.size >= n:
            pp = p[:n].copy()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            pp /= pp.sum()
            inits['parent'] = pp
            # Also a perturbed version
            pp2 = pp * rng.uniform(0.7, 1.3, n)
            pp2 = np.maximum(pp2, 0)
            if pp2.sum() > 1e-10:
                pp2 /= pp2.sum()
                inits['parent_perturb'] = pp2

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Uses LP-guided descent with multiple initializations
    and sequence lengths to minimize 2n*max(conv)/(sum)^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Loaded parent sequence: length={p.size}, sum={p.sum():.2f}")
    except Exception:
        pass

    best_sequence = None
    best_score = float('inf')

    def try_sequence(seq):
        nonlocal best_sequence, best_score
        if seq is None:
            return
        s = np.maximum(seq, 0.0)
        if s.sum() < 0.01:
            return
        score = float(evaluate_sequence(s.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = s.copy()
            print(f"  NEW BEST: score={score:.6f}, n={len(s)}")

    # Phase 1: Quick scan
    print("Phase 1: Quick scan over lengths and initializations...")
    lengths = [64, 96, 128, 160, 192, 256]
    
    # If parent exists, prioritize its length
    if parent is not None:
        plen = len(parent)
        if plen not in lengths:
            lengths.insert(0, plen)
        # Also try nearby lengths
        for dl in [-32, 32, -64, 64]:
            candidate_len = plen + dl
            if 32 <= candidate_len <= 512 and candidate_len not in lengths:
                lengths.append(candidate_len)
        lengths = sorted(set(lengths))[:8]

    scan_configs = []
    for n in lengths:
        inits = generate_initializations(n, rng, parent)
        for name, a0 in inits.items():
            scan_configs.append((n, name, a0))

    n_total = len(scan_configs)
    phase1_time = min(400.0, (deadline - time.monotonic()) * 0.25)
    time_per_scan = max(5.0, phase1_time / max(1, n_total))
    print(f"  {n_total} configs, ~{time_per_scan:.1f}s each, total ~{phase1_time:.0f}s")

    scan_results = []
    for i, (n, name, a0) in enumerate(scan_configs):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        t_alloc = min(time_per_scan, remaining - 60)
        if t_alloc < 3:
            break

        a_opt, obj_norm = lp_descent(a0, time_limit=t_alloc, seed=seed + i)
        try_sequence(a_opt)
        scan_results.append((best_score if a_opt is not None else float('inf'), n, name, a_opt, obj_norm))

    # Sort by the LP objective (which correlates with the true score)
    scan_results.sort(key=lambda x: x[4])

    # Phase 2: Deep optimization on best candidates
    print(f"\nPhase 2: Deep optimization. Best so far: {best_score:.6f}")
    top_configs = []
    seen = set()
    for sc, n, name, a_opt, obj_norm in scan_results:
        key = (n, name)
        if key not in seen and a_opt is not None:
            top_configs.append((n, name, a_opt))
            seen.add(key)
        if len(top_configs) >= 8:
            break

    # Always include current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    remaining = deadline - time.monotonic() - 30
    time_per_deep = max(20.0, remaining / max(1, len(top_configs)))

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 30:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 30)
        if t_alloc < 10:
            break
        print(f"  Deep: n={n}, {name}, alloc={t_alloc:.0f}s")
        a_opt, obj_norm = lp_descent(a0, time_limit=t_alloc, seed=seed + 1000 + i, verbose=True)
        try_sequence(a_opt)

    # Phase 3: Final polish
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print(f"\nPhase 3: Final polish. Best: {best_score:.6f}")
        remaining = deadline - 20 - time.monotonic()
        n = len(best_sequence)
        
        # Multiple LP descent runs with different seeds
        n_runs = min(3, max(1, int(remaining // 120)))
        time_per_run = remaining / n_runs
        for r in range(n_runs):
            if time.monotonic() > deadline - 15:
                break
            t_alloc = min(time_per_run, deadline - 15 - time.monotonic())
            if t_alloc < 10:
                break
            # Slight perturbation for diversity
            f_start = best_sequence * rng.uniform(0.9, 1.1, n)
            f_start = np.maximum(f_start, 0)
            f_start /= f_start.sum()
            a_opt, obj_norm = lp_descent(f_start, time_limit=t_alloc, seed=seed + 5000 + r, verbose=True)
            try_sequence(a_opt)

        # Also try a pure subgradient approach on the best
        if remaining > 100 and time.monotonic() < deadline - 20:
            print("  Trying multiplicative descent on best...")
            f = best_sequence.copy()
            f = f / f.sum()
            n = len(f)
            conv = fast_convolve(f)
            M = float(conv.max())
            t_sg = deadline - 20 - time.monotonic()
            eta = 0.1
            t0 = time.monotonic()
            while time.monotonic() - t0 < t_sg:
                sg = compute_subgradient(f, conv, active_frac=0.1)
                sg_c = sg - sg.mean()
                log_f = -eta * sg_c
                log_f -= log_f.max()
                f = f * np.exp(log_f)
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                conv = fast_convolve(f)
                M_new = float(conv.max())
                if M_new < M:
                    M = M_new
                    if 2 * n * M < best_score:
                        best_score = 2 * n * M
                        best_sequence = f.copy()
                        print(f"    SG: obj={best_score:.6f}")
                else:
                    eta *= 0.8
                    if eta < 0.001:
                        break

    # Final evaluation
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # One final check
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    print(f"{'='*50}")

    return [float(x) for x in best_sequence]
