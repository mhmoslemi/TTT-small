def adam_step(state, grad, lr, t):

    v_hat = v / (1 - beta2**(t+1))

def project_simplex(a, S):

    if s == S:
        return a

        return a + (S - s) / n

    for _ in range(60):

        else:

def next_power_of_2(x):
    """Return the next power of 2 >= x."""
    x = max(1, int(x))
    x -= 1
    x |= x >> 1
    x |= x >> 2
    x |= x >> 4
    x |= x >> 8
    x |= x >> 16
    x |= x >> 32
    return x + 1


def compute_softmax_grad(a, tau):
    """
    Compute the soft-max surrogate and its gradient.
    
    Surrogate: L(a) = tau * log(sum_k exp(c_k / tau)) + max(c)
    where c = conv(a, a).
    
    Gradient: dL/da[m] = 2 * sum_{j=0}^{n-1} a[j] * q[m+j]
    where q[k] = exp(c[k]/tau) / sum exp(c[k']/tau).
    """
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    q = exp_c / exp_c.sum()
    
    # Compute gradient using FFT
    N = next_power_of_2(3 * n)
    A = np.fft.rfft(a, N)
    Q = np.fft.rfft(q, N)
    grad = 2.0 * np.fft.irfft(np.conj(A) * Q)[:n]
    
    return q, c, grad


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s <= S:
        return a + (S - s) / n
    
    lo = 0.0
    hi = a.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.maximum(a - mid, 0.0)) > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    result = np.maximum(a - t, 0.0)
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        idx = result.argmax()
        result[idx] += diff
    return result


def adam_update(m, v, grad, lr, t):
    """Adam optimizer step. Returns updated m, v and the step."""
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** (t + 1))
    v_hat = v / (1 - beta2 ** (t + 1))
    step = -lr * m_hat / (np.sqrt(v_hat) + eps)
    return m, v, step


def optimize_sequence(a0, S, tau_start, tau_end, lr, time_budget, 
                       verbose=True, verbose_interval=100):
    """
    Optimize a sequence using Adam with soft-max surrogate and cosine tau annealing.
    
    Parameters:
    - a0: initial sequence
    - S: target sum (normalization)
    - tau_start: initial soft-max temperature
    - tau_end: final soft-max temperature
    - lr: learning rate
    - time_budget: seconds to spend
    - verbose: whether to print progress
    - verbose_interval: steps between prints
    
    Returns: (best_sequence, best_max_conv)
    """
    n = len(a0)
    a = a0.copy()
    a = a / max(a.sum(), 1e-12) * S
    a = project_simplex(a, S)
    
    best_a = a.copy()
    best_mc = np.convolve(best_a, best_a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_budget:
        # Cosine annealing of tau
        progress = min(1.0, (time.monotonic() - t0) / time_budget)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        q, c, grad = compute_softmax_grad(a, tau)
        m, v, step = adam_update(m, v, grad, lr, t)
        a_new = project_simplex(a + step, S)
        a = a_new
        t += 1
        
        if verbose and t % verbose_interval == 0:
            mc = np.convolve(a, a).max()
            if mc < best_mc:
                best_mc = mc
                best_a = a.copy()
            print(f"    step={t}, tau={tau:.5f}, max_c={mc:.4f}, best={best_mc:.4f}")
    
    # Final check
    mc = np.convolve(a, a).max()
    if mc < best_mc:
        best_mc = mc
        best_a = a.copy()
    
    return best_a, best_mc


def init_chirp(n, seed=0):
    """Chirp-based initialization."""
    t = np.arange(n, dtype=np.float64) - (n - 1) * 0.5
    a = 1.0 + 0.8 * np.cos(np.pi * t * t / (n * 0.5))
    rng = np.random.default_rng(seed)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def init_raised_cosine(n):
    """Raised cosine window initialization."""
    t = np.linspace(0, 2 * np.pi, n)
    return 0.5 * (1 + np.cos(t)) + 0.01


def init_gaussian(n, width_frac=0.4):
    """Gaussian window centered."""
    t = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = n * width_frac / 2.0
    return np.exp(-t**2 / (2 * sigma**2)) + 0.01

    """Inverse of cosine - high at edges, low at center."""
