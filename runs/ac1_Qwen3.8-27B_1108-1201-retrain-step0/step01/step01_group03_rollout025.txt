        progress = min(it / max_iters, 1.0)
        # Exponential annealing of tau
        tau = tau_start * (tau_end / tau_start) ** progress

        a_look = a + mu * v

        grad -= grad.mean()  # project to maintain sum

        v = mu * v - lr * grad
        a = a + v

            score = compute_obj(a)
            if score < best_score:
                best_score = score

def coordinate_descent(a, deadline, max_sweeps=100):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = a.copy()
    S_target = a.sum()
    best_score = compute_obj(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline - 3:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline - 3:
                break
            # Try adjusting a[j]
            old_val = a[j]
            # The objective depends on a[j] through:
            # b[k] for k = j (a[j]*a[0]), k = j+1 (a[j]*a[1] + ...), etc.
            # More precisely, b[k] contains a[j]*a[k-j] and a[k-j]*a[j] terms.
            # b[k] = sum_i a[i]*a[k-i]. Terms involving a[j]:
            #   i=j: a[j]*a[k-j]
            #   k-i=j: a[k-j]*a[j]
            # So b[k] = (rest) + 2*a[j]*a[k-j] (for k != 2j)
            # or b[k] = (rest) + a[j]^2 (for k = 2j)
            
            # For a given a[j] = x:
            # b[k] = base[k] + 2*x*a[k-j] (for valid k-j, k != 2j)
            # b[2j] = base[2j] + x^2 (if 2j < 2n-1)
            # where base[k] = b[k] - 2*a[j]*a[k-j] (or - a[j]^2 for k=2j)
            
            # We want to minimize max(b[k]) over k.
            # This is a 1D optimization problem in x >= 0.
            
            # Compute base convolution (without a[j])
            # b[k] without a[j]: remove terms involving a[j]
            b_full = np.convolve(a, a)
            
            # For efficiency, precompute the "base" 
            # b_base[k] = b_full[k] - contribution of a[j]
            # contribution of a[j] to b[k]:
            #   if 0 <= k-j < n: 2*a[j]*a[k-j] (or a[j]^2 if k=2j)
            
            aj = a[j]
            b_base = b_full.copy()
            for k in range(2*n-1):
                kj = k - j
                if 0 <= kj < n:
                    if k == 2*j:
                        b_base[k] -= aj * aj
                    else:
                        b_base[k] -= 2 * aj * a[kj]
            
            # Now b(x)[k] = b_base[k] + 2*x*a[k-j] (for valid k-j) or b_base[k] + x^2 (for k=2j)
            # We want to find x >= 0 that minimizes max_k b(x)[k].
            
            # The max is a convex function of x (max of affine/quadratic functions, all convex).
            # We can use bisection or ternary search.
            
            # Actually, for a fixed j, let's parameterize:
            # b(x)[k] = b_base[k] + coeff_k * x + quad_k * x^2
            # where quad_k = 1 if k=2j else 0, and coeff_k = 2*a[k-j] (if valid) else 0.
            
            # We want min_x max_k (b_base[k] + coeff_k * x + quad_k * x^2)
            # subject to x >= 0, and also sum constraint: x = S_target - sum(a excluding a[j])
            # Wait, we need to maintain sum(a) = S_target. So x is determined: x = S_target - sum(a without a[j]).
            # That means we can't freely choose x!
            
            # Hmm, coordinate descent with a sum constraint is tricky. Let me instead
            # optimize a pair of elements: increase a[j] by δ and decrease a[l] by δ.
            # This maintains the sum.
            
            # This gets complicated. Let me just use a simpler approach:
            # Try a few candidate values for a[j] (scaling up and down) and pick the best.
            
            candidates = [0.0, aj * 0.5, aj * 0.8, aj * 1.2, aj * 1.5, aj * 2.0]
            best_local = aj
            best_local_score = compute_obj(a)
            
            for x in candidates:
                a[j] = x
                a = normalize_seq(a, S_target)
                s = compute_obj(a)
                if s < best_local_score:
                    best_local_score = s
                    best_local = x
                a[j] = aj  # restore
            
            a[j] = best_local
            a = normalize_seq(a, S_target)
            if best_local_score < best_score:
                best_score = best_local_score
                improved = True
        
        if not improved:
            break
    
    return a, best_score


def spectral_init(n, rng):
    """Generate a sequence with near-flat power spectrum."""
    # Start with random phases in frequency domain
    L = 2 * n - 1
    A = np.zeros(L, dtype=complex)
    A[0] = 1.0  # DC
    for k in range(1, L):
        A[k] = np.exp(2j * np.pi * rng.random())
    
    # Inverse DFT
    a = np.real(np.fft.ifft(A))[:n]
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    return a

By the minimax theorem (Sion's theorem), this equals:

Where p is a probability distribution over the 2n-1 convolution positions.

The inner minimization is: min_a [2n * sum_k p[k] * (a*a)[k] / (sum(a))^2]

Let R_p = sum_k p[k] * (a*a)[k] = a^T (sum_k p[k] * C_k) a where C_k is the matrix such that (a*a)[k] = a^T C_k a.

Actually, (a*a)[k] = sum_i a[i] * a[k-i] = a^T C_k a where C_k is a matrix with (C_k)[i, k-i] = 1/2 and (C_k)[k-i, i] = 1/2 (symmetric form).

So the inner problem is: min_a a^T Q_p a / (sum(a))^2 where Q_p = sum_k p[k] * C_k.

This is a Rayleigh quotient problem! The minimum of a^T Q a / (1^T a)^2 over a >= 0.

Without the non-negativity constraint, this is related to the minimum generalized eigenvalue, but with the sum constraint it's different.

Actually, min a^T Q a / (1^T a)^2 over a (no constraint) can be solved by setting a proportional to the eigenvector of Q corresponding to its smallest eigenvalue... but that's for a^T Q a / a^T a, not a^T Q a / (1^T a)^2.

For a^T Q a / (1^T a)^2, we can use the substitution: let b = a / (1^T a), so 1^T b = 1. Then we minimize b^T Q b subject to 1^T b = 1.

This is a quadratic program with a linear constraint. The solution is:
b^* = Q^{-1} 1 / (1^T Q^{-1} 1) (if Q is positive definite).

But with the non-negativity constraint b >= 0, it becomes a QP which can be solved efficiently.

So the overall algorithm would be:
1. Initialize p (probability over convolution positions).
2. Solve the inner QP: min a^T Q_p a s.t. sum(a) = 1, a >= 0.
3. Get a^*.
4. Update p: set p[k] proportional to (a^* * a^*)[k] (put more weight on the peaks).
   Or use a multiplicative weights update: p[k] *= exp(gamma * (a^*a^*)[k] / max_m (a^*a^*)[m]).
5. Repeat.

This is a **Frank-Wolfe / cutting plane** type approach, or more specifically, it's related to the **multiplicative weights algorithm** for min-max problems.

This could be more effective than plain gradient descent because:
- The inner QP gives an exact (or near-exact) solution for the current p.
- The multiplicative weights update on p is principled.

"""
Multi-strategy optimizer for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Combines LSE-smoothed gradient descent with Nesterov momentum,
multiplicative weights min-max iterations, and hard-max polishing.

import numpy as np
import time


def compute_obj(a):
    """Compute 2*n*max(convolve(a,a))/(sum(a))^2."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = float(b.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (S * S))


def normalize_seq(a, S_target):
    """Normalize sequence to have sum = S_target."""
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (S_target / S)
