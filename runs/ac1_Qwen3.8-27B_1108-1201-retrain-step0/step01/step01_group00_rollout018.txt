    target_mag = np.ones(N)
    
    # Start with random non-negative sequence
    a = rng.random(n)
    
    for _ in range(n_iter):
        # FFT
        A = np.fft.rfft(a, N)
        # Replace magnitude with target, keep phase
        phase = np.angle(A)
        A_new = target_mag[:N//2+1] * np.exp(1j * phase)
        # Inverse FFT
        a = np.fft.irfft(A_new, N)[:n]
        # Project to non-negative
        a = np.clip(a, 0, None)
    
    return a / np.sum(a) if np.sum(a) > 0 else np.ones(n) / n

This is the Gerchberg-Saxton algorithm adapted for our problem. It alternates between:
1. Matching the target magnitude spectrum (flat)
2. Enforcing non-negativity in the time domain

This should converge to a sequence that's "as flat-spectrum as possible" while being non-negative.

Let me combine all these ideas into a final algorithm:

1. **Initialization**: Use iterative flat-spectrum projection to get good starting points for various n.
2. **LP iterations**: Use the LP-based peak reduction to iteratively improve.
3. **Polish**: Use local search (coordinate descent or simulated annealing) for final refinement.
4. **Multi-length search**: Try n = 128, 256, 384, 512, 768.

Let me also think about the LP more carefully. The constraint is:
`conv(a, g)[k] <= M` for all k.

`conv(a, g)[k] = sum_i a[i] * g[k-i]` (where the sum is over valid i).

This can be written as: for each k, the dot product of a (reversed and shifted) with g is <= M.

Specifically, `conv(a, g)[k] = dot(a_rev_shifted[k], g)` where `a_rev_shifted[k]` is a vector of length n with `(a_rev_shifted[k])[i] = a[k-i]` if `0 <= k-i < n`, else 0.

So the constraint matrix A has rows `a_rev_shifted[k]` for k = 0, ..., 2n-2.

This matrix is banded (each row has at most n non-zero entries, and they're in a consecutive block).

For the LP, I'll use `scipy.optimize.linprog` with method='highs' (or 'highs-ipm' for larger problems).

"""
Multi-phase optimization for minimizing the peak of linear convolution
of non-negative sequences. Uses: (1) iterative flat-spectrum projection
for initialization, (2) LP-based peak reduction iterations, (3) local
search polish. Targets score < 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score efficiently."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def iterative_flat_spectrum(n, rng, n_iter=200, target='flat'):
    """
    Gerchberg-Saxton style iterative projection to find a non-negative
    sequence with flat magnitude spectrum.
    """
    N = 1
    while N < 2 * n - 1:
        N *= 2
    
    # Target magnitude spectrum
    if target == 'flat':
        target_mag = np.ones(N // 2 + 1)
    elif target == 'raised_cosine':
        k = np.arange(N // 2 + 1)
        target_mag = 0.5 + 0.5 * np.cos(np.pi * k / (N // 2))
        target_mag[0] = 1.0
    
    # Start with random non-negative sequence
    a = rng.random(n) + 0.1
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_score = local_score(a)
    
    for it in range(n_iter):
        # FFT
        A = rfft(a, N)
        # Replace magnitude with target, keep phase
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        # Inverse FFT
        a = irfft(A_new, N)[:n]
        # Project to non-negative
        a = np.clip(a, 0, None)
        if np.sum(a) < 1e-10:
            break
        a = a / np.sum(a)
        
        if it % 50 == 0:
            s = local_score(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
    
    return best_a


def lp_peak_reduce(a, M=None, max_iter_lp=50):
    """
    Single LP-based peak reduction step.
    Given current sequence a, find g >= 0 such that conv(a,g) <= M everywhere,
    maximizing sum(g).
    Returns the new g (normalized to same sum as a).
    """
    n = len(a)
    sum_a = np.sum(a)
    
    if M is None:
        conv_aa = fft_convolve(a, a)
        M = np.max(conv_aa)
    
    # Build constraint matrix: A_lp @ g <= b
    # Row k: conv(a, g)[k] = sum_i a[i] * g[k-i] <= M
    # For each k in 0..2n-2, the coefficients of g[0]..g[n-1] are:
    #   coef[j] = a[k-j] if 0 <= k-j < n, else 0
    
    num_constraints = 2 * n - 1
    A_lp = np.zeros((num_constraints, n))
    b_lp = np.full(num_constraints, M)
    
    for k in range(num_constraints):
        # g[j] contributes a[k-j] to conv[k] if 0 <= k-j < n
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_lp[k, j] = a[idx]
    
    # Objective: maximize sum(g), i.e., minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_lp, b_ub=b_lp, bounds=bounds, method='highs')
    
    if result.status != 0:
        return a.copy()  # Return unchanged if LP fails
    
    g = result.x
    sum_g = np.sum(g)
    if sum_g < 1e-10:
        return a.copy()
    
    # Normalize to same sum as a
    g = g * (sum_a / sum_g)
    return g


def lp_iteration(a, t=0.1, M=None):
    """
    One LP-based iteration: compute g via LP, blend with current a.
    """
    n = len(a)
    sum_a = np.sum(a)
    
    if M is None:
        conv_aa = fft_convolve(a, a)
        M = np.max(conv_aa)
    
    g = lp_peak_reduce(a, M)
    
    a_new = (1 - t) * a + t * g
    a_new = np.clip(a_new, 0, None)
    
    # Renormalize
    sum_new = np.sum(a_new)
    if sum_new > 0:
        a_new = a_new * (sum_a / sum_new)
    
    return a_new

def simulated_annealing_polish(a_init, time_limit, rng):
    """
    Simulated annealing with targeted peak reduction.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    temperature = 0.001
    cooling = 0.9999
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.4:
            # Targeted: reduce peak contributors
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            # Find which pairs contribute most to the peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(2, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            # Boost others
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.05, size=width))
        elif r < 0.8:
            # Pair swap (redistribute mass)
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = rng.normal(0, 0.02) * np.mean(current)
                candidate[i] += delta
                candidate[j] -= delta
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0, 0.01, size=n))
        
        candidate = np.clip(candidate, 0, None)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif np.isfinite(new_score) and rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best, best_score

"""
LP-based iterative peak reduction with flat-spectrum initialization
for minimizing the peak-to-average ratio of linear convolution of
non-negative step functions.

Algorithm:
1. Initialize via Gerchberg-Saxton (iterative flat-spectrum projection)
2. Iteratively solve LP to find direction that reduces convolution peak
3. Polish with simulated annealing
4. Final LP refinement
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/(sum(a))^2 efficiently."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def iterative_flat_spectrum(n, rng, n_iter=300):
    """
    Gerchberg-Saxton style iterative projection to find a non-negative
    sequence with approximately flat magnitude spectrum.
    """
    N = 1
    while N < 2 * n - 1:
        N *= 2

    target_mag = np.ones(N // 2 + 1)

    a = rng.random(n) + 0.1
    a = a / np.sum(a)

    best_a = a.copy()
    best_score = local_score(a)

    for it in range(n_iter):
        A = rfft(a, N)
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        a = irfft(A_new, N)[:n]
        a = np.clip(a, 0, None)
        s_a = np.sum(a)
        if s_a < 1e-10:
            break
        a = a / s_a

        if it % 100 == 0 or it == n_iter - 1:
            s = local_score(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()

    return best_a


def build_lp_matrix(n, a):
    """
    Build the constraint matrix for the LP.
    Row k: coefficients of g[0..n-1] in conv(a,g)[k].
    A[k,j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    num_constraints = 2 * n - 1
    A_lp = np.zeros((num_constraints, n))
    a_rev = a[::-1]
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A_lp[k, j_start:j_end + 1] = a_rev[j_start:j_end + 1]
    return A_lp


def lp_peak_reduce(a, M, A_lp=None):
    """
    Solve LP: maximize sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns g normalized to same sum as a.
    """
    n = len(a)
    sum_a = np.sum(a)

    if A_lp is None:
        A_lp = build_lp_matrix(n, a)

    c_obj = -np.ones(n)
    b_lp = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_lp, b_ub=b_lp, bounds=bounds,
                         method='highs', options={'presolve': True})
    except Exception:
        return None

    if result.status != 0 or result.x is None:
        return None

    g = result.x
    sum_g = np.sum(g)
    if sum_g < 1e-10:
        return None

    g = g * (sum_a / sum_g)
    return g


def lp_iteration_sequence(a, time_limit, rng, verbose=True):
    """
    Run LP-based peak reduction iterations on sequence a (normalized to sum=1).
    Returns best sequence and score found.
    """
    n = len(a)
    a_work = a.astype(np.float64).copy()
    a_work = a_work / np.sum(a_work)

    current_score = local_score(a_work)
    best = a_work.copy()
    best_score = current_score

    t0 = time.monotonic()
    iter_count = 0
    A_lp = build_lp_matrix(n, a_work)

    while time.monotonic() - t0 < time_limit:
        conv_aa = fft_convolve(a_work, a_work)
        M = np.max(conv_aa)

        # Adaptive step size: start larger, decrease over time
        frac = (time.monotonic() - t0) / time_limit
        t_step = 0.3 * (1.0 - frac * 0.8)
        t_step = max(0.03, t_step)

        g = lp_peak_reduce(a_work, M, A_lp)
        if g is None:
            # Fallback: simple perturbation
            g = a_work * np.exp(rng.normal(0, 0.05, n))
            g = np.clip(g, 0, None)
            s_g = np.sum(g)
            if s_g > 0:
                g = g / s_g
            else:
                break

        a_new = (1 - t_step) * a_work + t_step * g
        a_new = np.clip(a_new, 0, None)
        sum_new = np.sum(a_new)
        if sum_new < 1e-10:
            break
        a_new = a_new / sum_new

        new_score = local_score(a_new)
        iter_count += 1

        if new_score < current_score:
            a_work = a_new
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = a_work.copy()
                if verbose and iter_count % 50 == 0:
                    print(f"      iter {iter_count}: new best {best_score:.6f}, t={time.monotonic()-t0:.1f}s")
        elif new_score < current_score + 0.005 and rng.random() < 0.2:
            # Occasional acceptance of slightly worse to escape local min
            a_work = a_new
            current_score = new_score
        else:
            # Try smaller step
            a_new2 = (1 - t_step * 0.3) * a_work + t_step * 0.3 * g
            a_new2 = np.clip(a_new2, 0, None)
            s2 = np.sum(a_new2)
            if s2 > 0:
                a_new2 = a_new2 / s2
                s2_score = local_score(a_new2)
                if s2_score < current_score:
                    a_work = a_new2
                    current_score = s2_score
                    if current_score < best_score:
                        best_score = current_score
                        best = a_work.copy()

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_score:.6f}, time={time.monotonic()-t0:.1f}s")

    return best, best_score


def simulated_annealing_polish(a_init, time_limit, rng):
    """
    Simulated annealing with targeted peak reduction perturbations.
    """
    current = a_init.astype(np.float64).copy()
    sum_c = np.sum(current)
    if sum_c > 0:
        current = current / sum_c
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()

    temperature = 0.0005
    cooling = 0.99995

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.35:
            # Targeted: reduce peak contributors
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 6)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.06, 0.025, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(12, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.06, 0.025, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.06, size=width))
        elif r < 0.7:
            # Multiple small blocks
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 12))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.85:
            # Pair redistribution
            n_pairs = max(2, n // 16)
            for _ in range(n_pairs):
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    delta = rng.normal(0, 0.03) * np.mean(current)
                    candidate[i] = max(0, candidate[i] + delta)
                    candidate[j] = max(0, candidate[j] - delta)
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0, 0.015, size=n))

        candidate = np.clip(candidate, 0, None)
        sum_cand = np.sum(candidate)
        if sum_cand < 1e-10:
            continue
        candidate = candidate / sum_cand

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif np.isfinite(new_score):
            delta = new_score - current_score
            if delta < temperature * 10:
                accept_prob = np.exp(-delta / max(temperature, 1e-15))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score

        temperature *= cooling

    return best, best_score


def make_freq_init(n, rng, n_iter=200):
    """
    Create initial sequence using frequency domain design.
    Tries to make |FFT(a)| as flat as possible while keeping a >= 0.
    """
    return iterative_flat_spectrum(n, rng, n_iter=n_iter)


def make_chirp_init(n, rng):
    """Create a chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    alpha = 0.3 + 0.3 * rng.random()
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None) / np.sum(np.clip(a, 0.01, None))


def make_barker_like_init(n, rng):
    """
    Create a sequence inspired by Barker/Legendre sequence structure,
    mapped to non-negative values.
    """
    # Start with +/- 1 sequence (random for now)
    s = rng.choice([-1, 1], size=n)
    # Map to non-negative: a_i = (s_i + 1) * 0.5 + noise
    a = (s + 1) * 0.5 + 0.3
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None) / np.sum(np.clip(a, 0.01, None))
