def compute_score(a):
    """Compute the score for a given sequence."""
    n = len(a)
    sum_val = np.sum(a)
    conv = np.convolve(a, a, mode="full")
    max_val = np.max(conv)
    score = (2 * n * max_val) / (sum_val ** 2)
    if score == np.inf or np.isnan(score):
        return np.inf
    return score

def project_sequence(a, S):
    """Project a sequence a to the simplex (non-negative) with the normalized sum S."""
    a = np.asarray(a, dtype=np.float64)
    a_sum = np.sum(a)
    if a_sum > 0:
        a = a / a_sum * S
    else:
        # If zero, find a non-zero candidate
        a = np.full_like(a, 1.0) * S

    a = np.maximum(a, np.zeros_like(a)).astype(np.float64)
