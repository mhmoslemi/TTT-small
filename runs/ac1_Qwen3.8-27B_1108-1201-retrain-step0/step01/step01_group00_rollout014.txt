```python
def lp_alterating(f, M, tol=1e-6, max_iter=100):
    """
    Given f (fixed), find g that maximizes sum(g) subject to:
    - g >= 0
    - (f * g)[k] <= M for all k
    Then normalize g to have same sum as f.
    """
    n = len(f)
    # Build constraint matrix A such that A @ g <= M
    # (f * g)[k] = sum_j f[j] * g[k-j] for valid j
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[j]
    
    # Maximize sum(g) subject to A @ g <= M, g >= 0
    # linprog minimizes, so minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n)
    
    if res.success:
        g = res.x
        return g
    return None
```

```python
import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def build_constraint_matrix(f):
    """Build the LP constraint matrix A where A[k,j] = f[j] if 0<=k-j<len(f)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f[j]
    return A


def lp_step(f, M):
    """
    Given fixed f and peak bound M, find g >= 0 maximizing sum(g)
    subject to (f * g)[k] <= M for all k.
    """
    n = len(f)
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None

def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def grad_soft_max_conv(a, tau):
    """Compute soft-max value and gradient of soft-max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    value = c_max + tau * np.log(sum_exp)
    # Gradient: g[j] = 2 * (a[::-1] * w)[j + n - 1]
    corr = fft_convolve(a[::-1], w)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return value, g, c

    m2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 1.0
    t0 = time.monotonic()
    step = 0
    best_a = a0.copy()
    best_score = local_score(a0)
    tau_start = 50.0 / n
    tau_end = 0.5 / n
    no_improve = 0

        elapsed_frac = min(1.0, (time.monotonic() - t0 + 1e-9) / max(time_limit, 1e-9))
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac

        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)
        inner = np.dot(a, g_a) / S
        g_theta = a * (g_a - inner)

        step += 1
        m1 = beta1 * m1 + (1 - beta1) * g_theta
        m2 = beta2 * m2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = m1 / (1 - beta1 ** step)
        m2_hat = m2 / (1 - beta2 ** step)
        step_lr = lr / np.sqrt(1 + 0.01 * step)
        theta = theta - step_lr * m1_hat / (np.sqrt(m2_hat) + eps)

        a_check = softmax_normalize(theta) * S
        score = local_score(a_check)

            best_a = a_check.copy()

                theta = 0.7 * theta + 0.3 * theta_best
                no_improve = 0

def peak_shave(a, time_limit, max_iters=5000):
    """
    Iterative peak shaving: identify peak contributors and reduce them,
    redistributing weight to less-contributing positions.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fft_convolve(a, a)
        # Find top peaks
        top_k = min(3, len(c))
        peak_indices = np.argsort(c)[-top_k:]

        # For each peak, find top contributors
        for pk in peak_indices:
            contribs = np.zeros(n)
            for i in range(n):
                j = pk - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]

            total_contrib = contribs.sum()
            if total_contrib < 1e-10:
                continue

            # Reduce top contributors, increase low contributors
            top_count = max(2, n // 32)
            bottom_count = max(2, n // 32)
            top_idx = np.where(contribs > np.percentile(contribs, 90))[0]
            bot_idx = np.where(contribs < np.percentile(contribs, 20))[0]
            top_idx = top_idx[:top_count]
            bot_idx = bot_idx[:bottom_count]

            if len(top_idx) == 0 or len(bot_idx) == 0:
                continue

            # Compute transfer amount
            reduce_frac = 0.03 + 0.02 * rng.random()
            reduce_amount = np.sum(a[top_idx]) * reduce_frac
            if reduce_amount < 1e-6:
                continue

            a_new = a.copy()
            a_new[top_idx] -= reduce_amount / len(top_idx)
            a_new[bot_idx] += reduce_amount / len(bot_idx)
            a_new = np.clip(a_new, 0.0, 1000.0)

            new_score = local_score(a_new)
            if new_score < best_score - 1e-10:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new.copy()
                break  # Recompute peaks
            elif new_score < local_score(a):
                a = a_new.copy()

        # Occasionally try a random perturbation
        if it % 50 == 0:
            a_pert = a * np.exp(rng.normal(0, 0.01, n))
            a_pert = np.clip(a_pert, 0, 1000)
            if local_score(a_pert) < local_score(a):
                a = a_pert

    return best_a, best_score


def simulated_annealing(a_init, time_limit):
    """Simulated annealing with diverse move types."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    temp = 0.005
    cooling = 0.9998
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.30:
            # Peaked reduction move
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            threshold = np.percentile(contribs, 85)
            high = np.where(contribs > threshold)[0]
            low = np.where(contribs < np.percentile(contribs, 30))[0]
            if len(high) > 0 and len(low) > 0:
                amt = np.sum(a[high]) * 0.05
                candidate[high] -= amt / len(high)
                candidate[low] += amt / len(low)
        elif r < 0.50:
            # Block scaling
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, width))
        elif r < 0.65:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, width))
        elif r < 0.80:
            # Transfer between random positions
            idx1 = int(rng.integers(0, n))
            idx2 = int(rng.integers(0, n))
            if idx1 != idx2:
                amt = a[idx1] * 0.1
                candidate[idx1] -= amt
                candidate[idx2] += amt
        else:
            # Global small perturbation
            candidate = a * np.exp(rng.normal(0, 0.008, n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        if candidate.sum() < 0.01:
            continue

        new_score = local_score(candidate)
        cur_score = local_score(a)

        if new_score < cur_score:
            a = candidate
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
        elif rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            a = candidate

        temp *= cooling

    return best_a, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):

def make_multi_tone_init(n, rng):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    n_tones = min(8, n // 16)
    for k in range(1, n_tones + 1):
        freq = k * rng.uniform(1, 3)
        phase = rng.uniform(0, 2 * np.pi)
        amp = 0.2 / k
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_linear_taper_init(n, rng, gamma=0.5):
    """Linear taper: high in middle, low at edges (or vice versa)."""
    t = np.linspace(-1, 1, n)
    a = 1.0 + gamma * (1 - t * t) * rng.uniform(0.5, 1.5)
    if rng.random() > 0.5:
        a = 1.0 + gamma * t * t * rng.uniform(0.5, 1.5)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

    # Phase 1: LP-based alternating optimization
    print("\n=== Phase 1: LP Alternating ===")
    lp_lengths = [128, 256, 384, 512, 768, 1024]
    for n in lp_lengths:
        if time.monotonic() > deadline - 120:
            break
        time_left = deadline - time.monotonic()
        n_inits = 3
        time_per_init = min(60.0, time_left / (n_inits + 1))

        for init_i in range(n_inits):
            if time.monotonic() > deadline - 60:
                break

            # Generate different initializations
            r = rng.random()
            if r < 0.33:
                a0 = make_random_init(n, rng, spread=0.3)
                init_name = 'random'
            elif r < 0.66:
                a0 = make_chirp_init(n, 0.3 + 0.3 * rng.random(), rng)
                init_name = 'chirp'
            else:
                a0 = make_multi_tone_init(n, rng)
                init_name = 'multi_tone'

            t0 = time.monotonic()
            a_best, score = lp_alternating(a0, time_per_init)
            elapsed = time.monotonic() - t0
            print(f"  LP n={n} init={init_i}({init_name}): score={score:.6f}, time={elapsed:.1f}s")

            if score < best_score:
                best_score = score
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Phase 2: Gradient optimization on best and new inits
    print(f"\n=== Phase 2: Gradient Optimization (current best: {best_score:.6f}) ===")
    grad_configs = []
    if best_seq is not None:
        grad_configs.append(('best', best_seq))
    for n in [256, 512, 1024]:
        grad_configs.append((f'n{n}_chirp', make_chirp_init(n, 0.5, rng)))
        grad_configs.append((f'n{n}_random', make_random_init(n, rng, 0.4)))

    for name, a0 in grad_configs:
        if time.monotonic() > deadline - 180:
            break
        time_left = deadline - time.monotonic()
        t_per = min(80.0, time_left / max(1, len(grad_configs)))
        n = len(a0)
        t0 = time.monotonic()
        a_best, score = gradient_optimize(n, a0, t_per)
        elapsed = time.monotonic() - t0
        print(f"  Grad {name}: score={score:.6f}, time={elapsed:.1f}s")
        if score < best_score:
            best_score = score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 3: Peak shaving
    print(f"\n=== Phase 3: Peak Shaving (current best: {best_score:.6f}) ===")
    if best_seq is not None:
        polish_time = min(120.0, max(10.0, (deadline - time.monotonic()) / 2.0))
        t0 = time.monotonic()
        a_best, score = peak_shave(best_seq, polish_time)
        elapsed = time.monotonic() - t0
        print(f"  Peak shave: score={score:.6f}, time={elapsed:.1f}s")
        if score < best_score:
            best_score = score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 4: Simulated annealing
    print(f"\n=== Phase 4: Simulated Annealing (current best: {best_score:.6f}) ===")
    if best_seq is not None:
        sa_time = min(180.0, max(10.0, deadline - time.monotonic()))
        t0 = time.monotonic()
        a_best, score = simulated_annealing(best_seq, sa_time)
        elapsed = time.monotonic() - t0
        print(f"  SA polish: score={score:.6f}, time={elapsed:.1f}s")
        if score < best_score:
            best_score = score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 5: Try different lengths with best structure
    if time.monotonic() < deadline - 30:
        print(f"\n=== Phase 5: Length exploration (current best: {best_score:.6f}) ===")
        if best_seq is not None:
            for new_n in [384, 640, 1280]:
                if time.monotonic() > deadline - 20:
                    break
                a_resampled = resample_sequence(best_seq, new_n)
                t_per = min(40.0, (deadline - time.monotonic()) / 2.0)
                a_best, score = lp_alternating(a_resampled, t_per)
                if score > 1.6:
                    a_best, score = gradient_optimize(new_n, a_resampled, t_per)
                print(f"  Resample to n={new_n}: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_seq = a_best.copy()
                    print(f"    *** New best: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
