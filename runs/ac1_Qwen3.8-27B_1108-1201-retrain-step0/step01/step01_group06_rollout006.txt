import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def objective_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / np.arange(1, n+1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def subgradient_max(a, conv, top_k=5):
    """Compute subgradient of max(conv) w.r.t. a, using top_k peaks."""
    n = len(a)
    # Find top k values of conv
    if top_k >= len(conv):
        top_idx = np.arange(len(conv))
    else:
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
    
    # Get weights (proportional to values, or uniform)
    top_vals = conv[top_idx]
    if top_vals.max() < 1e-15:
        return np.zeros(n)
    
    # Use weights proportional to values
    w = top_vals / top_vals.sum()
    
    # Compute subgradient
    sg = np.zeros(n)
    for i, k in enumerate(top_idx):
        k = int(k)
        wi = w[i]
        # b[k] = sum_j a[j] * a[k-j] for valid j
        # db[k]/da[j] = a[k-j] if 0 <= k-j < n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            sg[j_start:j_end+1] += wi * a[k - j_end : k - j_start + 1][::-1]
    
    return sg

def pgd_optimize(a0, time_limit, init_lr=0.1, top_k=5):
    """Projected gradient descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a /= S
    
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    lr = init_lr
    t0 = time.monotonic()
    stagnation = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Compute subgradient
        sg = subgradient_max(a, conv, top_k=top_k)
        
        # Center the gradient (for the sum=1 constraint)
        sg_centered = sg - sg.mean()
        
        # Step
        a_new = a - lr * sg_centered
        a_new = project_simplex(a_new)
        
        # Check if improvement
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < cur_obj:
            a = a_new
            lr = min(lr * 1.05, 1.0)  # slightly increase
        else:
            # Backtrack
            lr *= 0.7
            if lr < 1e-6:
                # Perturb
                a = a + 0.01 * np.random.randn(n)
                a = project_simplex(a)
                lr = 0.1
        
        if stagnation > 100:
            # Restart with perturbation
            a = a + 0.05 * np.random.randn(n)
            a = project_simplex(a)
            lr = 0.1
            stagnation = 0
    
    return best_a, best_obj

import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    j = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cssv - 1.0) / j > 0
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0)

def compute_grad(a, conv, top_k=10, sharpness=50.0):
    """Compute a sharp subgradient of max(conv) w.r.t. a."""
    n = len(a)
    m = conv.max()
    if m < 1e-15:
        return np.zeros(n)
    
    # Sharp soft-max weights
    x = sharpness * conv / m
    x -= x.max()
    w = np.exp(x)
    w_sum = w.sum()
    if w_sum < 1e-15:
        # Fall back to just the argmax
        idx = np.argmax(conv)
        k = idx
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        sg = np.zeros(n)
        sg[j_start:j_end+1] = a[k - j_end : k - j_start + 1][::-1]
        return sg
    w /= w_sum
    
    # Only process significant weights
    thresh = w.max() * 0.001
    active = np.where(w > thresh)[0]
    
    sg = np.zeros(n)
    for k in active:
        k = int(k)
        wi = w[k]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            sg[j_start:j_end+1] += wi * a[k - j_end : k - j_start + 1][::-1]
    
    return sg

def optimize_sequence(a0, time_limit, sharpness=50.0, top_k=10):
    """Projected gradient descent on the simplex with sharp subgradient."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a /= S
    
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    lr = 0.5
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        sg = compute_grad(a, conv, top_k=top_k, sharpness=sharpness)
        sg_centered = sg - sg.mean()
        
        # Gradient step with projection
        a_new = a - lr * sg_centered
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        new_obj = 2.0 * n * conv_new.max()
        
        if new_obj <= cur_obj * (1 + 1e-10):
            a = a_new
            # Increase LR slightly if we're making progress

            lr = 0.5
            no_improve = 0
            # Increase sharpness to focus on the peak
            sharpness = min(sharpness * 1.5, 500.0)
    
    return best_a, best_obj
