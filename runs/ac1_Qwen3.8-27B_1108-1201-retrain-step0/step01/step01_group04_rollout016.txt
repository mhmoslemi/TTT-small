Target: minimize 2*n*max(p*p)/(sum(p))^2 for p >= 0
"""

import time
import numpy as np

try:
    from scipy.fft import fft, ifft, next_fast_len
except Exception:
    from numpy.fft import fft, ifft

    def next_fast_len(n):
        n = int(max(1, n))
        m = 1
        while m < n:
            m <<= 1
        return m

try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _HAVE_LP = True
except Exception:
    _HAVE_LP = False


def _fft_len(n):
    return int(next_fast_len(max(1, n)))


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def conv_sq(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(max(1, 2 * n - 1))
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def gradient_smoothed_max(p, T_factor):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64), np.empty(0, dtype=np.float64), 0.0, 0.0

    L = _fft_len(max(1, 2 * n - 1))
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]

    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.zeros(n, dtype=np.float64), c.astype(np.float64, copy=False), 1.0, 0.0

    T = max(float(T_factor) * cmax, 1e-300)
    if not np.isfinite(T) or T <= 0.0:
        T = 1e-300

    diff = (c - cmax) / T
    e = np.exp(diff)
    raw = float(e.sum())
    if not np.isfinite(raw) or raw <= 0.0:
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        raw = 1.0

    w = e / raw
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))

    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2:n - 2:-1]

    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp

    G = np.maximum(G, 0.0)
    g = 2.0 * G
    surr = float(cmax + T * np.log(raw))
    return g, c.astype(np.float64, copy=False), T, surr


def peak_shave_step(p, c, top_k=5, strength=0.1):
    """
    Explicitly reduce contributions to the top-k peak lags of the convolution.
    For each peak lag k*, find pairs (j, k*-j) with large product p_j*p_{k*-j},
    and reduce the larger element.
    """
    n = int(p.size)
    if n == 0:
        return p

    cmax = float(np.max(c))
    if cmax <= 0:
        return p

    # Find top-k lags
    top_lags = np.argsort(c)[::-1][:top_k]

    q = p.copy()
    removed = np.zeros(n, dtype=np.float64)

    for k_star in top_lags:
        k_star = int(k_star)
        if k_star < 0 or k_star >= 2 * n - 1:
            continue
        # Find pairs contributing to C_{k_star}
        # C_{k_star} = sum_j p_j * p_{k_star - j} for valid j
        j_min = max(0, k_star - (n - 1))
        j_max = min(n - 1, k_star)
        if j_min > j_max:
            continue

        js = np.arange(j_min, j_max + 1)
        partners = k_star - js
        products = q[js] * q[partners]
        total = float(np.sum(products))
        if total <= 0:
            continue

        # Identify hot pairs (contributing > 2*average)
        avg_contrib = total / len(js)
        hot_mask = products > 2.0 * avg_contrib
        if not np.any(hot_mask):
            continue

        hot_js = js[hot_mask]
        hot_partners = partners[hot_mask]
        hot_prods = products[hot_mask]

        # For each hot pair, reduce the larger element
        for j, kj, prod in zip(hot_js, hot_partners, hot_prods):
            if j == kj:
                # Self-pair: reduce p_j by a small fraction
                reduction = strength * 0.5 * q[j]
                q[j] -= reduction
                removed[j] += reduction
            else:
                # Reduce the larger of the two
                if q[j] >= q[kj]:
                    reduction = strength * 0.3 * q[j]
                    q[j] -= reduction
                    removed[j] += reduction
                else:
                    reduction = strength * 0.3 * q[kj]
                    q[kj] -= reduction
                    removed[kj] += reduction

    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s <= 1e-12:
        return p
    return q / s


def spectral_equalize(p, iterations=3, strength=0.05):
    """
    Reduce the sidelobes of the periodogram by modifying p in the time domain.
    If |P(f)|^2 is too peaked at non-DC frequencies, adjust p to flatten it.
    """
    n = int(p.size)
    if n < 4:
        return p

    q = p.copy()
    L = _fft_len(n)

    for _ in range(iterations):
        P = fft(q, L)
        P2 = np.abs(P) ** 2

        # Exclude DC
        P2[0] = 0.0
        # Also exclude the Nyquist frequency if n is even
        if n % 2 == 0 and L > n:
            P2[n // 2] = 0.0

        P2_max = float(np.max(P2))
        if P2_max <= 0:
            break

        # Compute the "excess" signal: how much each time-domain sample
        # contributes to the large sidelobes
        # Gradient of P2_max w.r.t. q:
        # d|P_k|^2/dq_j = 2*Re(conj(P_k) * exp(-2*pi*i*k*j/L))
        k_peak = int(np.argmax(P2))
        phase = P[k_peak]
        grad = 2.0 * np.real(np.conjugate(phase) * np.exp(-2j * np.pi * k_peak * np.arange(n) / L))
        grad = grad[:n]

        # Also add contribution from the second peak
        P2_sorted = P2.copy()
        P2_sorted[k_peak] = 0
        k_peak2 = int(np.argmax(P2_sorted))
        phase2 = P[k_peak2]
        grad2 = 2.0 * np.real(np.conjugate(phase2) * np.exp(-2j * np.pi * k_peak2 * np.arange(n) / L))
        grad = grad[:n] + grad2[:n]

        gnorm = float(np.max(np.abs(grad)))
        if gnorm <= 1e-300:
            break

        step = strength * (1.0 / n) / gnorm
        q = q - step * grad
        q = project_simplex(q)

    return q


def make_uniform(n):
    return np.full(max(1, n), 1.0 / max(1, n), dtype=np.float64)


def make_fourier_shape(n, coeffs, rng):
    """Create a shape from random Fourier coefficients."""
    x = np.linspace(0.0, 1.0, n)
    logp = np.zeros(n, dtype=np.float64)
    for k, (a, b, ph) in enumerate(coeffs):
        freq = (k + 1)
        logp += a * np.cos(2 * np.pi * freq * x + ph) + b * np.sin(2 * np.pi * freq * x + ph)
    m = float(np.max(logp))
    logp = logp - m
    p = np.exp(logp)
    return safe_normalize(p)


def make_polynomial_shape(n, degree, rng):
    """Random polynomial shape."""
    x = np.linspace(0.0, 1.0, n)
    coeffs = rng.uniform(-1, 1, degree + 1)
    logp = np.zeros(n, dtype=np.float64)
    for d in range(degree + 1):
        logp += coeffs[d] * x ** d
    m = float(np.max(logp))
    p = np.exp(logp - m)
    return safe_normalize(p)


def generate_diverse_initials(n, rng, parent=None, count=30):
    """Generate diverse initial candidates."""
    out = []

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.size == n:
            out.append(safe_normalize(p))

    out.append(make_uniform(n))

    # Geometric
    for s in (0.3, 0.7, 1.4, 2.8, 5.6, 11.2):
        x = np.linspace(0.0, 1.0, n)
        p = np.exp(-s * x)
        out.append(safe_normalize(p))
        out.append(safe_normalize(p[::-1]))

    # Power
    for a in (-1.5, -0.8, -0.3, 0.3, 0.8, 1.5, 3.0):
        x = np.linspace(1e-6, 1.0, n)
        p = x ** a
        out.append(safe_normalize(p))

    # Beta-like
    for alpha, beta in ((0.2, 0.2), (0.5, 0.5), (1.0, 1.0), (0.3, 1.5), (1.5, 0.3), (2.0, 2.0), (0.5, 3.0), (3.0, 0.5)):
        x = np.linspace(1e-6, 1 - 1e-6, n)
        p = x ** alpha * (1 - x) ** beta
        out.append(safe_normalize(p))

    # Cosine taper (like a Hanning window)
    x = np.linspace(0.0, 1.0, n)
    for gamma in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0):
        p = np.cos(np.pi * x * 0.5) ** gamma
        out.append(safe_normalize(p))

    # Random Fourier shapes
    for _ in range(count // 3):
        ncoeffs = rng.integers(2, 8)
        coeffs = []
        for k in range(ncoeffs):
            a = rng.uniform(0.3, 2.0) * (0.5 / (k + 1))
            b = rng.uniform(0.3, 2.0) * (0.5 / (k + 1))
            ph = rng.uniform(0, 2 * np.pi)
            coeffs.append((a, b, ph))
        out.append(make_fourier_shape(n, coeffs, rng))

    # Random polynomial
    for _ in range(count // 4):
        deg = int(rng.integers(2, 8))
        out.append(make_polynomial_shape(n, deg, rng))

    # Random smooth (low-frequency sine combos)
    for _ in range(count // 3):
        x = np.linspace(0.0, 1.0, n)
        p = 1.0
        for _ in range(int(rng.integers(2, 5))):
            f = float(rng.integers(1, max(2, n // 50)))
            amp = float(rng.uniform(0.1, 0.8))
            ph = float(rng.uniform(0, 2 * np.pi))
            p += amp * np.sin(2 * np.pi * f * x + ph)
        p = np.maximum(p, 0.01)
        out.append(safe_normalize(p))

    # Multi-peak shapes
    for _ in range(count // 5):
        x = np.linspace(0.0, 1.0, n)
        p = np.zeros(n, dtype=np.float64)
        npeaks = int(rng.integers(2, 6))
        for _ in range(npeaks):
            mu = float(rng.uniform(0, 1))
            sig = float(rng.uniform(0.02, 0.15))
            w = float(rng.uniform(0.3, 1.5))
            p += w * np.exp(-((x - mu) ** 2) / (2 * sig ** 2))
        out.append(safe_normalize(p))

    # SPLIT shape: high at edges, low in middle (to reduce center convolution)
    x = np.linspace(0.0, 1.0, n)
    for a in (1.0, 2.0, 4.0, 8.0):
        p = 1.0 + a * (2 * x - 1.0) ** 4
        out.append(safe_normalize(p))

    # "Anti-triangle" shapes designed to flatten the convolution
    for a in (0.5, 1.0, 2.0, 3.0):
        p = 1.0 + a * np.cos(np.pi * x) ** 2
        out.append(safe_normalize(p))
        p2 = 1.0 + a * np.sin(np.pi * x) ** 2
        out.append(safe_normalize(p2))

    return out


def lp_best_response(p, budget_s):
    """LP best-response: maximize sum(g) s.t. p*g <= max(p*p), g >= 0."""
    if not _HAVE_LP:
        return p

    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 0 or n > 800:
        return p

    if budget_s <= 1.0:
        return p

    c = conv_sq(p)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0:
        return p

    f = p + 1e-14
    fs = float(f.sum())
    if fs <= 0:
        return p
    f = f / fs

    try:
        mid = np.arange(n, dtype=np.int32)
        rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel().astype(np.int32)
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(f, n).astype(np.float64)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs",
                      options={"disp": False, "time_limit": max(5.0, budget_s - 1.0)})

        if res is None:
            return p
        g = getattr(res, "x", None)
        if g is None:
            return p
        g = np.asarray(g, dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        s = float(g.sum())
        if s <= 1e-12:
            return p
        return g / s
    except Exception:
        return p


def optimize_local(p0, rng, time_limit, label=""):
    """
    Combined local optimizer:
    - Smoothed-max gradient descent
    - Explicit peak shaving for top lags
    - Spectral equalization
    - LP best-response (if n small enough)
    - Random restarts when stuck
    """
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    if n == 0:
        return p, np.inf

    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score
    prev_p = p.copy()

    end = time.monotonic() + max(0.0, float(time_limit))
    iter_no = 0
    since_improve = 0
    T_factor = 0.05

    # Try LP first for short sequences
    if n <= 800 and time_limit > 30:
        lp_budget = min(40.0, time_limit * 0.2)
        g = lp_best_response(p, lp_budget)
        if g is not None and g.size == n:
            for t in (1.0, 0.7, 0.4, 0.2, 0.1, 0.05):
                q = project_simplex((1 - t) * p + t * g)
                sc = internal_score(q)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    current_score = sc
                    best_p = q.copy()
                    p = q.copy()
                    prev_p = q.copy()
                    since_improve = 0

    while time.monotonic() < end:
        iter_no += 1

        # Anneal temperature
        T_factor = max(0.002, 0.05 * (0.998 ** iter_no))
        if since_improve > 200:
            T_factor = max(0.001, T_factor * 0.8)
        if since_improve > 800:
            T_factor = 0.0005

        # Compute gradient
        g, c_cur, T, surr = gradient_smoothed_max(p, T_factor)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)

        accepted = False

        # Gradient step with adaptive learning rate
        pmax = float(np.max(p)) if n > 0 else 0.0
        p_scale = max(1.0 / n, 0.1 * pmax, 1e-12)
        gabs = float(np.max(np.abs(g))) if g.size else 0.0

        if gabs > 1e-300:
            base_lr = p_scale / gabs
            lrs = (0.3 * base_lr, 0.1 * base_lr, 0.03 * base_lr, 0.006 * base_lr, 0.001 * base_lr)
        else:
            lrs = (0.5, 0.1, 0.03, 0.01, 0.003)

        momentum = 0.3 if iter_no > 50 else 0.0

        for lr in lrs:
            lr = min(max(float(lr), 1e-10), 10.0)
            v = p - lr * g
            if momentum > 0:
                v = v + momentum * (p - prev_p)
            pt = project_simplex(v)
            sc = internal_score(pt)

            if np.isfinite(sc) and sc < current_score - 1e-14:
                if sc < best_score:
                    best_score = sc
                    best_p = pt.copy()
                    since_improve = 0
                else:
                    since_improve += 1
                current_score = sc
                prev_p = p
                p = pt
                accepted = True
                break

        if not accepted:
            since_improve += 1

            # Peak shaving step
            if since_improve > 50 and np.all(np.isfinite(c_cur)):
                strength = 0.05 if since_improve < 300 else 0.02
                pt = peak_shave_step(p, c_cur, top_k=3, strength=strength)
                sc = internal_score(pt)
                if np.isfinite(sc) and sc < current_score - 1e-14:
                    if sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        since_improve = 0
                    else:
                        since_improve += 1
                    current_score = sc
                    prev_p = p
                    p = pt
                    accepted = True

        if not accepted:
            # Spectral equalization
            if since_improve > 100 and n >= 16:
                pt = spectral_equalize(p, iterations=2, strength=0.03)
                sc = internal_score(pt)
                if np.isfinite(sc) and sc < current_score - 1e-14:
                    if sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        since_improve = 0
                    else:
                        since_improve += 1
                    current_score = sc
                    prev_p = p
                    p = pt
                    accepted = True

        if not accepted:
            # Random perturbation
            if since_improve > 200:
                sigma = 0.5 / np.sqrt(n) if since_improve < 600 else 1.0 / np.sqrt(n)
                pt = project_simplex(p + rng.normal(0, sigma, n))
                sc = internal_score(pt)
                if np.isfinite(sc) and sc < current_score - 1e-14:
                    if sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        since_improve = 0
                    else:
                        since_improve += 1
                    current_score = sc
                    prev_p = p
                    p = pt
                    accepted = True

        if not accepted:
            # Coordinate-pair move: transfer mass from high-grad to low-grad
            if n > 1 and np.all(np.isfinite(g)):
                hi = int(np.argmax(g))
                lo = int(np.argmin(g))
                if hi != lo and g[hi] > g[lo] + 1e-15:
                    for frac in (0.2, 0.05, 0.01, 0.003):
                        delta = frac * p[hi]
                        if delta <= 0:
                            continue
                        pt = p.copy()
                        pt[hi] -= delta
                        pt[lo] += delta
                        pt = project_simplex(pt)
                        sc = internal_score(pt)
                        if np.isfinite(sc) and sc < current_score - 1e-14:
                            if sc < best_score:
                                best_score = sc
                                best_p = pt.copy()
                                since_improve = 0
                            else:
                                since_improve += 1
                            current_score = sc
                            prev_p = p
                            p = pt
                            accepted = True
                            break

        if not accepted and since_improve > 500:
            # Blend with best + noise
            blend = 0.6 + 0.3 * np.exp(-since_improve / 200.0)
            p = project_simplex(blend * best_p + (1 - blend) * p + rng.normal(0, 0.3 / np.sqrt(n), n))
            current_score = internal_score(p)
            prev_p = p
            since_improve = 0

        # Periodic LP for short sequences
        if n <= 800 and iter_no % 2000 == 0 and (end - time.monotonic()) > 20:
            g = lp_best_response(p, min(30.0, end - time.monotonic() - 5))
            if g is not None and g.size == n:
                for t in (0.5, 0.3, 0.15, 0.05):
                    pt = project_simplex((1 - t) * p + t * g)
                    sc = internal_score(pt)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        current_score = sc
                        best_p = pt.copy()
                        p = pt.copy()
                        prev_p = pt.copy()
                        since_improve = 0
                        break

        if iter_no % 3000 == 0:
            print("  %s iter=%d best=%.9f cur=%.9f" % (label, iter_no, best_score, current_score), flush=True)

        if best_score < 1.495 and since_improve > 8000:
            break

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def upsample(p, target_n):
    """Upsample a sequence to target_n using linear interpolation."""
    n = len(p)
    if n == target_n:
        return p.copy()
    if n == 1:
        return np.full(target_n, p[0], dtype=np.float64)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, target_n)
    result = np.interp(x_new, x_old, p)
    return safe_normalize(result)


def downsample(p, target_n):
    """Downsample a sequence to target_n by averaging blocks."""
    n = len(p)
    if n == target_n:
        return p.copy()
    if n < target_n:
        return upsample(p, target_n)
    n_blocks = n // target_n
    blocks = p[:n_blocks * target_n].reshape(target_n, n_blocks)
    result = blocks.mean(axis=1)
    return safe_normalize(result)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for nonnegative step function minimizing evaluate_sequence.
    Multi-phase: global search -> local optimization at multiple lengths -> final polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 15.0)

    # Try to use parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and float(parent_arr.sum()) > 0:
            parent = parent_arr / parent_arr.sum()
            print("Parent loaded: n=%d, score=%.9f" % (len(parent), internal_score(parent)), flush=True)
    except Exception:
        parent = None

    best_p_global = None
    best_score_global = np.inf

    def try_record(p, sc):
        nonlocal best_p_global, best_score_global
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0 or not np.all(np.isfinite(p)):
            return
        p = safe_normalize(p)
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score_global:
            best_score_global = float(sc)
            best_p_global = p.copy()
            print("*** New global best: %.9f (n=%d) ***" % (sc, len(p)), flush=True)

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    lengths.extend([128, 256, 512, 700, 1024, 2048, 4096])
    # Deduplicate and sort
    lengths = sorted(set(l for l in lengths if l >= 64))

    # Phase 1: Quick global search at a medium length
    med_n = 512
    print("Phase 1: Global search at n=%d" % med_n, flush=True)
    phase1_budget = min(80.0, (deadline - start_time) * 0.1)

    parent_med = None
    if parent is not None:
        if len(parent) == med_n:
            parent_med = parent
        else:
            parent_med = upsample(parent, med_n) if len(parent) < med_n else downsample(parent, med_n)

    inits_med = generate_diverse_initials(med_n, rng, parent_med, count=40)
    scored_med = []
    for p in inits_med:
        sc = internal_score(p)
        if np.isfinite(sc):
            scored_med.append((sc, p))
    scored_med.sort(key=lambda x: x[0])

    for sc, p in scored_med[:8]:
        try_record(p, sc)

    # Phase 2: Local optimization at multiple lengths
    print("Phase 2: Local optimization", flush=True)

    # Allocate time based on remaining budget
    elapsed = time.monotonic() - start_time
    remaining = max(10.0, deadline - time.monotonic())
    final_reserve = max(30.0, remaining * 0.1)
    opt_budget = max(20.0, remaining - final_reserve)

    # Prioritize: parent length first, then shorter lengths (for LP), then longer
    classed = []
    for L in lengths:
        if L <= 800:
            classed.append((0, L))  # LP-eligible, prioritize
        elif L <= 2048:
            classed.append((1, L))
        else:
            classed.append((2, L))
    classed.sort()

    # Deduplicate
    seen = set()
    unique_classed = []
    for cls, L in classed:
        if L not in seen:
            seen.add(L)
            unique_classed.append((cls, L))

    n_jobs = len(unique_classed)
    time_per_job = max(15.0, opt_budget / max(1, n_jobs))

    job_list = []
    for cls, L in unique_classed:
        # Get best initialization for this length
        best_init = None
        best_init_score = np.inf

        # Use parent if available
        if parent is not None:
            if len(parent) == L:
                q = parent.copy()
            elif len(parent) > L:
                q = downsample(parent, L)
            else:
                q = upsample(parent, L)
            sc = internal_score(q)
            if sc < best_init_score:
                best_init_score = sc
                best_init = q

        # Use medium-length results upsampled/downsampled
        if scored_med:
            sc_med, p_med = scored_med[0]
            if L == med_n:
                q = p_med.copy()
            elif L > med_n:
                q = upsample(p_med, L)
            else:
                q = downsample(p_med, L)
            sc = internal_score(q)
            if sc < best_init_score:
                best_init_score = sc
                best_init = q

        # Generate a few fresh inits
        fresh = generate_diverse_initials(L, rng, parent, count=12)
        for q in fresh[:6]:
            sc = internal_score(q)
            if sc < best_init_score:
                best_init_score = sc
                best_init = q

        if best_init is None:
            best_init = make_uniform(L)

        allow_lp = (L <= 800)
        job_list.append((best_init_score, best_init, allow_lp, L))

    job_list.sort(key=lambda x: x[0])

    # Run optimization on top jobs
    for idx, (sc_init, p_init, allow_lp, L) in enumerate(job_list):
        if time.monotonic() > deadline - final_reserve:
            break

        remaining_job = max(5.0, deadline - final_reserve - time.monotonic())
        # Give more time to shorter (LP-eligible) and parent-length sequences
        if allow_lp:
            t_alloc = min(max(30.0, time_per_job * 1.5), remaining_job * 0.8)
        elif L == (len(parent) if parent is not None else -1):
            t_alloc = min(max(30.0, time_per_job * 1.3), remaining_job * 0.8)
        else:
            t_alloc = min(max(15.0, time_per_job * 0.7), remaining_job * 0.8)

        t_alloc = max(10.0, t_alloc)
        print("  Job %d: n=%d, init_score=%.9f, time=%.0fs, LP=%s" % (idx, L, sc_init, t_alloc, allow_lp), flush=True)

        p_best, sc_best = optimize_local(p_init, rng, t_alloc, label="n=%d" % L)
        try_record(p_best, sc_best)

    # Phase 3: Final polish on the best found
    print("Phase 3: Final polish", flush=True)
    if best_p_global is not None and time.monotonic() < deadline - 10:
        polish_time = max(5.0, deadline - time.monotonic() - 5.0)
        print("  Polishing n=%d with %.0fs, current best=%.9f" % (len(best_p_global), polish_time, best_score_global), flush=True)
        p_final, sc_final = optimize_local(best_p_global, rng, polish_time, label="POLISH")
        try_record(p_final, sc_final)

    # Fallback
    if best_p_global is None:
        if parent is not None:
            best_p_global = parent.copy()
            best_score_global = internal_score(best_p_global)
        else:
            best_p_global = make_uniform(256)
            best_score_global = internal_score(best_p_global)

    # Also try LP on the final best if it's short enough
    if best_p_global is not None and len(best_p_global) <= 800 and time.monotonic() < deadline - 5:
        g = lp_best_response(best_p_global, min(20.0, deadline - time.monotonic() - 3))
        if g is not None and len(g) == len(best_p_global):
            for t in (0.3, 0.15, 0.07, 0.03):
                q = project_simplex((1 - t) * best_p_global + t * g)
                sc = internal_score(q)
                if np.isfinite(sc) and sc < best_score_global:
                    best_p_global = q
                    best_score_global = sc
                    print("  LP polish improved to: %.9f" % sc, flush=True)
                    break

    # Final sanity check and format
    best_p_global = np.asarray(best_p_global, dtype=np.float64)
    best_p_global = np.where(np.isfinite(best_p_global), best_p_global, 0.0)
    best_p_global = np.maximum(best_p_global, 0.0)
    best_p_global = np.minimum(best_p_global, 1000.0)
    s = float(best_p_global.sum())
    if s <= 1e-12:
        best_p_global = make_uniform(max(16, len(best_p_global)))

    final_score = internal_score(best_p_global)
    print("FINAL: n=%d, score=%.10f" % (len(best_p_global), final_score), flush=True)

    return [float(x) for x in best_p_global]
