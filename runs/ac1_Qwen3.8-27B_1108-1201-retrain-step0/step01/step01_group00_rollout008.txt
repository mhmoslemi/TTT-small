```python
def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n, non_neg=True)
    
    # Constraints: conv(f, g)[k] <= M for all k
    constraints = []
    # conv(f, g)[k] = sum_{i} f[i] * g[k-i] for valid i
    # For efficiency, we can express this as a matrix times g
    # Let A be a (2n-1) x n matrix where A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Then conv(f,g) = A @ g
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        for k in range(j, j+n):
            if k < 2*n-1:
                A[k, j] = f[k-j]
    
    constraints.append(A @ g <= M)
    
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve()
    
    if g.value is not None:
        return np.maximum(0, g.value)
    return None
```

"""
LP-based iterative peak reduction for minimizing the peak-to-sum ratio
of the self-convolution of non-negative sequences.
Uses the approach: given f, solve LP to find g maximizing sum(g) subject
to conv(f,g) <= M and g >= 0, then blend f and g.
Combined with gradient descent and local search for polishing.
"""

import numpy as np
import time
import cvxpy as cp


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score for a sequence."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def solve_lp_for_g(f, M):
    """
    Solve LP: maximize sum(g) subject to A@f_conv_g <= M, g >= 0.
    Where (f * g)[k] = sum_i f[k-i] * g[i] (linear convolution).
    """
    n = len(f)
    g = cp.Variable(n, non_neg=True)
    
    # Build constraint matrix A where (A@g)[k] = (f*g)[k]
    # (f*g)[k] = sum_{j} f[k-j] * g[j] for valid indices
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # g[j] contributes to (f*g)[k] for k = j, j+1, ..., j+n-1
        # with coefficient f[k-j] = f[0], f[1], ..., f[n-1]
        for i in range(n):
            k = j + i
            A[k, j] = f[i]
    
    constraints = [A @ g <= M]
    # Add upper bound to ensure boundedness
    constraints.append(g <= M / (np.min(f[f > 0]) + 1e-10))
    
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    if g.value is not None:
        return np.maximum(0, g.value)
    return None


def lp_iterations(f_init, max_iters=200, tol=1e-7, time_limit=60.0):
    """
    LP-based iterative peak reduction.
    f_init: initial sequence (will be normalized to sum = sqrt(2n))
    """
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = np.sqrt(2.0 * n)
    f = f / np.sum(f) * S
    
    t0 = time.monotonic()
    best_f = f.copy()
    best_peak = np.max(fft_convolve(f, f))
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_f = fft_convolve(f, f)
        M = np.max(conv_f)
        
        g = solve_lp_for_g(f, M)
        if g is None or np.sum(g) < 1e-10:
            break
        
        # Rescale g to have same sum as f
        g = g / np.sum(g) * S
        
        # Line search for step size
        best_t = 0.0
        for t in [0.8, 0.5, 0.3, 0.2, 0.1, 0.05, 0.01]:
            f_new = (1-t)*f + t*g
            peak_new = np.max(fft_convolve(f_new, f_new))
            if peak_new < M:
                best_t = t
                break
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.005, 0.001, 0.0005]:
                f_new = (1-t)*f + t*g
                peak_new = np.max(fft_convolve(f_new, f_new))
                if peak_new < M:
                    best_t = t
                    break
        
        if best_t == 0.0:
            break
        
        f = (1-best_t)*f + best_t*g
        # Ensure non-negativity
        f = np.maximum(0, f)
        # Renormalize
        f = f / np.sum(f) * S
        
        peak = np.max(fft_convolve(f, f))
        if peak < best_peak:
            best_peak = peak
            best_f = f.copy()
        
        if M - peak < tol * M:
            break
    
    return best_f, best_peak

            best_val = current_val
            best_local_peak = np.inf
            
            # The peak as a function of f[i] is:
            # max_k conv[k] where conv[k] includes terms with f[i]
            # conv[k] = old_conv[k] + f[i] * (a[k-i] + a[i] if k-i != i, else extra)
            # Actually, let's just compute it directly for candidates
            
            candidates = [
                current_val * 0.5,
                current_val * 0.7,
                current_val * 0.85,
                current_val * 1.0,
                current_val * 1.15,
                current_val * 1.3,
                current_val * 2.0,
                0.0,
                current_val * 0.1,
            ]
            candidates = [max(0, c) for c in candidates]
            
            for c in candidates:
                f[i] = c
                # Quick score computation
                conv = fft_convolve(f, f)
                peak = np.max(conv)
                if peak < best_local_peak:
                    best_local_peak = peak
                    best_val = c
            
            f[i] = best_val
            if abs(best_val - current_val) > 1e-8:
                improved = True
        
        f = np.maximum(0, f)
        if np.sum(f) > 1e-10:
            f = f / np.sum(f) * S
        
        peak = np.max(fft_convolve(f, f))
        if peak < best_peak:
            best_peak = peak
            best_f = f.copy()
        
        if not improved:
            break
    
    return best_f, best_peak


def make_chirp_init(n, rng):
    """Create a chirp-like initial sequence (non-negative)."""
    t = np.arange(n, dtype=np.float64)
    # Quadratic phase -> take absolute value or rectify
    phase = 2.0 * np.pi * t * t / (2.0 * n)
    a = 1.0 + 0.5 * np.cos(phase)
    a = np.maximum(0.01, a)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(0.01, a)


def make_random_init(n, rng):
    """Random initial sequence."""
    return np.abs(rng.standard_normal(n)) + 0.1


def make_multitone_init(n, rng):
    """Multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in [1, 2, 3, 5, 8]:
        a += 0.3 * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2*np.pi))
    return np.maximum(0.01, a)
