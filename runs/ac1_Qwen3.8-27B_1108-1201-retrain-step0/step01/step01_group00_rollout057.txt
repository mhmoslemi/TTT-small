Actually, let me reconsider the LP approach. It's elegant and might be more effective:

At each iteration:
1. Compute M = max(conv(f, f))
2. Solve LP: maximize sum(g) subject to g ≥ 0 and (f*g)[k] ≤ M for all k
   where (f*g) is the cross-convolution
3. The LP constraint (f*g)[k] = sum_j f[j]*g[k-j] ≤ M is linear in g
4. This gives us a direction g that we can move toward
5. Update: f ← (1-t)*f + t*g, then renormalize

The LP has n variables and 2n-1 constraints (plus n non-negativity constraints). For n=256, this is a 256-variable LP with ~767 constraints. This is very solvable with scipy's linprog or cvxpy.

The advantage of this approach is that it's guaranteed to not increase the peak (in a certain sense), and it always moves in a direction that increases the sum (which helps with the normalization).

Wait, but we also need to be careful: after the update, the new self-convolution might have a higher peak than M. The LP constraint is on the cross-convolution f*g, not on the new self-convolution.

For small t, the new self-convolution is:
((1-t)f + tg) * ((1-t)f + tg)
= (1-t)² f*f + 2t(1-t) f*g + t² g*g

At the point where f*f = M (the peak), this becomes:
(1-t)² M + 2t(1-t) (f*g)[peak] + t² (g*g)[peak]
≤ (1-t)² M + 2t(1-t) M + t² * max(g*g)
= M((1-t)² + 2t(1-t)) + t² * max(g*g)
= M(1-t) + t² * max(g*g)

For small t, this is ≈ M - tM + t² max(g*g), which is < M for sufficiently small t.

At other points where f*f < M, the cross term could push it up, but the LP constraint ensures (f*g)[k] ≤ M for all k, so:
((1-t)f + tg) * ((1-t)f + tg)[k]
= (1-t)² (f*f)[k] + 2t(1-t)(f*g)[k] + t²(g*g)[k]

import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    """Compute the evaluation score."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (sum_a ** 2)

    """Build the 2n-1 x n constraint matrix A where A[k,i] = f[k-i] if valid."""
    # A has shape (2n-1, n)
    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    # i ranges over 0..n-1, k ranges over 0..2n-2
    # k-i is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k

        i_end = min(n-1, k)
        # A[k, i] = f[k-i] for i in [i_start, i_end]
        # k-i ranges from k-i_end to k-i_start
        idx = k - np.arange(i_start, i_end+1)
        A[k, i_start:i_end+1] = f[idx]
    return A

LP-based fictitious play algorithm for minimizing the peak of autocorrelation
of non-negative step functions. Uses iterative LP solves to find best-response
directions, combined with gradient polish for fine-tuning.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score without external call."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (sum_a ** 2)

    """Build constraint matrix: A[k,i] = f[k-i] if 0<=k-i<n else 0."""
    # Efficient construction using the banded structure
    A = np.zeros((2 * n - 1, n))

        # So A[i:(i+n), i] = f[0:n] ... wait no
        # A[k, i] = f[k-i] if 0 <= k-i < n
        # So for fixed i, A[k, i] = f[k-i] where k ranges from i to i+n-1
        # and f[k-i] = f[0], f[1], ..., f[n-1]
        A[i:i+n, i] = f
    return A

    return None

LP-based fictitious play with FFT gradient polish for minimizing the peak
of autocorrelation of non-negative step functions. Uses iterative LP solves
for best-response directions, combined with projected subgradient polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (sum_a ** 2)

    A = np.zeros((2 * n - 1, n))

        A[i:i + n, i] = f
    return A


def lp_solve_fast(f, M, n):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    A = build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_iterate(f, n, time_limit=100.0, verbose=True):
    """Run LP-based fictitious play iterations to reduce peak."""
    f = np.asarray(f, dtype=np.float64).copy()
    S = np.sum(f)
    if S < 1e-12:
        f = np.ones(n)
        S = 1.0
    f = f / S

    best_f = f.copy()
    best_peak = np.max(fft_convolve(f, f))
    t0 = time.monotonic()
    t = 0.05
    stagnation = 0

    if verbose:
        print(f"  LP start: n={n}, initial_peak={best_peak:.8f}")

    while time.monotonic() - t0 < time_limit:
        conv_f = fft_convolve(f, f)
        M = np.max(conv_f)

        g = lp_solve_fast(f, M, n)
        if g is None:
            break

        g_sum = np.sum(g)
        if g_sum <= 1.0 + 1e-9:
            break

        g = g / g_sum

        # Backtracking line search
        accepted = False
        t_try = t
        for _ in range(25):
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.clip(f_new, 0.0, None)
            f_new_sum = np.sum(f_new)
            if f_new_sum < 1e-12:
                t_try *= 0.5
                continue
            f_new = f_new / f_new_sum
            peak_new = np.max(fft_convolve(f_new, f_new))
            if peak_new <= M * (1.0 + 1e-9):
                f = f_new
                accepted = True
                if peak_new < best_peak:
                    best_peak = peak_new
                    best_f = f.copy()
                    stagnation = 0
                else:
                    stagnation += 1
                break
            t_try *= 0.5

        if not accepted:
            break

        t = min(0.4, t * 1.01)

        if stagnation > 200:
            # Try a random perturbation to escape
            f_pert = f * np.exp(np.random.standard_normal(n) * 0.01)
            f_pert = np.clip(f_pert, 1e-6, None)
            f_pert = f_pert / np.sum(f_pert)
            peak_pert = np.max(fft_convolve(f_pert, f_pert))
            if peak_pert < best_peak * 1.05:
                f = f_pert
                stagnation = 0
            else:
                break

        if verbose and (int((time.monotonic() - t0) * 2) % 10 == 0):
            elapsed = time.monotonic() - t0
            print(f"    LP: peak={best_peak:.8f}, t={t:.4f}, t_elapsed={elapsed:.1f}s")

    return best_f, best_peak


def gradient_polish(f, n, time_limit=60.0):
    """Polish using projected subgradient with soft-max smoothing."""
    f = np.asarray(f, dtype=np.float64).copy()
    S = np.sum(f)
    f = f / S

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    best_f = f.copy()
    best_peak = np.max(fft_convolve(f, f))
    t0 = time.monotonic()
    lr = 0.5
    momentum = np.zeros(n)
    beta = 0.85
    tau = 0.05

    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        conv_max = np.max(conv)
        w = np.exp((conv - conv_max) / tau)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum

        # Gradient via FFT cross-correlation
        fw = np.fft.rfft(f, N)
        wF = np.fft.rfft(w, N)
        grad = 2.0 * np.fft.irfft(np.conj(fw) * wF, N)[:n]

        # Momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        m_hat = momentum / (1.0 - beta ** (iter_count + 1))

        # Step
        f_new = f - lr * m_hat
        f_new = np.clip(f_new, 0.0, None)
        f_new_sum = np.sum(f_new)
        if f_new_sum < 1e-12:
            f_new = f
        else:
            f_new = f_new / f_new_sum

        peak_new = np.max(fft_convolve(f_new, f_new))
        if peak_new < best_peak:
            best_peak = peak_new
            best_f = f_new.copy()

        f = f_new
        iter_count += 1
        lr *= 0.999
        tau *= 0.998

    return best_f, best_peak


def resample_sequence(a, n_target):
    """Resample a sequence to a target length using linear interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    x_old = np.linspace(0, 1, n_src)
    x_new = np.linspace(0, 1, n_target)
    return np.interp(x_new, x_old, a)


def make_initialization(n, rng, init_type='chirp'):
    """Create an initial sequence."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'chirp':
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.5))
        a = 1.0 + alpha * np.sin(2.0 * np.pi * t / n * rng.integers(2, 8))
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for k in [1, 2, 3, 5]:
            amp = 0.2 / k
            a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    elif init_type == 'asymmetric':
        # Create a non-symmetric sequence to break the center peak
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * t / (2 * n))
        a += 0.1 * np.cos(2.0 * np.pi * 3 * t / n)
    else:
        a = 1.0 + 0.2 * rng.standard_normal(n)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: LP-based fictitious play + gradient polish.
    Tries multiple sequence lengths and initializations.
    """
    t_global = time.monotonic()
    deadline = t_global + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = compute_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Define search configurations: (n, time_lp, time_polish, num_restarts)
    configs = [
        (256, 350, 80, 3),
        (512, 350, 80, 2),
        (768, 200, 60, 2),
        (1024, 200, 60, 1),
    ]

    init_types = ['chirp', 'multi_tone', 'asymmetric', 'random']

    for n, t_lp, t_poly, n_restarts in configs:
        if time.monotonic() > deadline - 60:
            break

        time_left = deadline - time.monotonic()
        t_lp_actual = min(t_lp, time_left * 0.7)
        t_poly_actual = min(t_poly, time_left * 0.2)

        print(f"\n=== n={n}, LP_time={t_lp_actual:.0f}s, polish_time={t_poly_actual:.0f}s ===")

        for restart in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break

            # Choose initialization
            if restart == 0 and start_seq is not None:
                a0 = resample_sequence(start_seq, n)
                init_name = 'provided'
            else:
                init_type = init_types[restart % len(init_types)]
                a0 = make_initialization(n, rng, init_type)
                init_name = init_type

            a0 = np.clip(a0, 0.01, None)
            a0 = a0 / np.sum(a0)

            # Phase 1: LP iterations
            t0 = time.monotonic()
            a_lp, peak_lp = lp_iterate(a0, n, time_limit=t_lp_actual, verbose=(restart == 0))
            elapsed_lp = time.monotonic() - t0
            score_lp = 2.0 * n * peak_lp  # since sum=1, score = 2n*peak/1

            print(f"  [{init_name}] LP done: peak={peak_lp:.8f}, score={score_lp:.6f}, time={elapsed_lp:.1f}s")

            # Phase 2: Gradient polish
            if time.monotonic() < deadline - 20:
                t0 = time.monotonic()
                a_pol, peak_pol = gradient_polish(a_lp, n, time_limit=t_poly_actual)
                elapsed_pol = time.monotonic() - t0
                score_pol = 2.0 * n * peak_pol

                print(f"  [{init_name}] Polish done: peak={peak_pol:.8f}, score={score_pol:.6f}, time={elapsed_pol:.1f}s")

                a_best_local = a_pol
                score_best_local = score_pol
            else:
                a_best_local = a_lp
                score_best_local = score_lp

            if score_best_local < best_score:
                best_score = score_best_local
                best_seq = a_best_local.copy()
                print(f"    *** NEW BEST: score={best_score:.6f} (n={n}, init={init_name}) ***")

    # Final verification and formatting
    if best_seq is not None:
        # Scale the sequence so that evaluate_sequence gives the right answer
        # The score formula is: 2n * max(conv(a,a)) / sum(a)^2
        # This is homogeneous, so scaling doesn't matter for the score
        # But we need sum > 0.01 for validity
        final_sum = np.sum(best_seq)
        if final_sum < 1.0:
            best_seq = best_seq * (10.0 / final_sum)
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: n={len(final_list)}, score={final_score:.6f} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]

    # Absolute fallback: uniform sequence of length 256
    return [1.0] * 256
