def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()

    if sum_a < 0.01:

def spectral_flatten(p, n_iters=1000, verbose=False):
    """
    Split constraint method to flatten the auto-convolution.
    Alternates between:
    - Frequency domain: flatten |DFT|
    - Time domain: enforce non-negativity and zero-padding

    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)

    u = np.sort(v)[::-1]

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    for it in range(n_iters):
        p_pad = np.zeros(L)
        p_pad[:n] = p
        P = np.fft.fft(p_pad)
        mag = np.abs(P)
        phase = np.angle(P)
        mean_mag = np.mean(mag)
        
        # Compress magnitude toward flat
        mag_ratio = mag / (mean_mag + 1e-30)
        mag_new = mean_mag * np.power(mag_ratio, gamma)
        
        P_new = mag_new * np.exp(1j * phase)
        p_pad_new = np.fft.ifft(P_new).real
        p = p_pad_new[:n]
        p = project_simplex(p)
        
        sc = score_sequence(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
        
        if verbose and (it % 500 == 0 or it < 3):
            print(f"    Spectral iter {it}: score={sc:.6f}, best={best_score:.6f}")
    
    return best_p, best_score

def lp_refine(a, time_limit_s=300.0, verbose=True):
    """LP-based refinement (similar to previous state of the art)."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    it = 0
    max_iters = 200
    
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 50 == 0 or it < 3):
                print(f"    LP iter {it}: score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 30:
            break
        
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.2))
        
        # Build constraint matrix
        n_rows = 2 * n - 1
        # A @ g <= M for all rows
        # A is the convolution matrix
        col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
        row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
        data = np.tile(a, n).astype(np.float64)
        A = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))
        
        c = -np.ones(n)
        b = np.full(n_rows, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': timeout})
            if res.success and res.x is not None:
                g = np.maximum(0.0, res.x)
                sum_g = g.sum()
                if sum_g > 1.0 + 1e-10:
                    g = g / sum_g
                    alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.2 * it)))
                    a_new = (1.0 - alpha) * a + alpha * g
                    a_new = normalize_nonneg(a_new)
                    a = a_new
                else:
                    break
            else:
                break
        except Exception:
            break
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

def local_search(a, time_limit_s=60.0, rng=None, verbose=False):
    """Random perturbation local search."""
    if rng is None:
        rng = np.random.default_rng(42)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(rng.integers(max(1, n // 100), max(2, n // 10))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.05, size=stop - start))
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
    
    if verbose:
        print(f"    LS done: best={best_score:.6f}")
    return best_a, best_score
