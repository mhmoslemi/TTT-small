    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]

import numpy as np
import time

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    # Simplex projection algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n, dtype=np.float64)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Compute the score 2*n*max(conv)/sum^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def _smooth_grad_step(a, tau):
    """
    Compute the gradient of the log-sum-exp smoothed max(conv(a,a)).
    Returns (smoothed_obj, gradient).
    """
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1
    
    # Log-sum-exp for numerical stability
    c_scaled = c / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    w = exp_c / np.sum(exp_c)
    
    smoothed_obj = tau * (c_max + np.log(np.sum(exp_c)))
    
    # Gradient: g[m] = 2 * sum_{j=0}^{n-1} a[j] * w[m+j]
    # Use FFT-based convolution: g = 2 * convolve(a[::-1], w)[n-1:2n-1]
    conv_g = np.convolve(a[::-1], w)
    g = 2.0 * conv_g[n - 1:2 * n - 1]
    
    return smoothed_obj, g

def _optimize(a_init, deadline, tau_init=0.1, tau_min=1e-4, name=""):
    """
    Optimize using projected gradient descent on the smoothed objective.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    tau = tau_init
    step = 1.0 / n  # initial step size
    iteration = 0
    no_improve = 0
    stagnation_limit = 100
    
    t0 = time.monotonic()
    
    while time.monotonic() < deadline:
        _, g = _smooth_grad_step(a, tau)
        
        # Center gradient (for sum constraint)
        g = g - np.mean(g)
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Descent direction
        direction = -g / g_norm
        
        # Try a few step sizes
        best_alpha = step
        best_trial_score = _score(a)
        for _ in range(10):
            alpha = best_alpha
            a_trial = _project_simplex(a + alpha * direction, 1.0)
            s_trial = _score(a_trial)
            if s_trial < best_trial_score:
                best_trial_score = s_trial
                best_alpha = alpha
            else:
                best_alpha *= 0.5
                if best_alpha < 1e-15:
                    break
        
        a = _project_simplex(a + best_alpha * direction, 1.0)
        cur_score = _score(a)
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        iteration += 1
        
        # Adapt step size and temperature
        if no_improve > stagnation_limit:
            step *= 0.8
            tau *= 0.7
            no_improve = 0
            if step < 1e-12 and tau < tau_min:
                break
        
        if iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {name} iter {iteration}: score={best_score:.6f}, step={step:.2e}, tau={tau:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score

        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n, dtype=np.float64)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score(a):
    """Compute 2*n*max(convolve(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def _smooth_grad(a, tau):
    """
    Compute gradient of tau * logsumexp(conv(a,a)/tau) w.r.t. a.
    Uses FFT-based convolution for efficiency.
    """
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1

    c_s = c / tau
    c_max = np.max(c_s)
    exp_c = np.exp(c_s - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp

    # Gradient: g[m] = 2 * sum_{j=0}^{n-1} a[j] * w[m+j]
    # Computed via: g = 2 * convolve(a_reversed, w)[n-1 : 2n-1]
    conv_g = np.convolve(a[::-1], w)  # length 3n-2
    g = 2.0 * conv_g[n - 1:2 * n - 1]

    return g


def _optimize_phase(a_init, deadline, tau, step_init, momentum, name=""):
    """
    Nesterov-style accelerated projected gradient descent on smoothed objective.
    Returns (best_a, best_score).
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    else:
        a = np.ones(n, dtype=np.float64) / n

    best_a = a.copy()
    best_score = _score(best_a)
    step = step_init
    v = np.zeros(n)  # velocity for momentum
    no_improve = 0
    iter_count = 0
    t_start = time.monotonic()

    while time.monotonic() < deadline:
        # Nesterov lookahead (using velocity)
        y = a + momentum * v

        # Evaluate gradient at projected y
        y_proj = _project_simplex(y, 1.0)
        g = _smooth_grad(y_proj, tau)
        g = g - np.mean(g)  # center for sum=1 constraint

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break

        # Backtracking line search from y_proj
        alpha = step
        a_new = y_proj
        s_new = _score(a_new)
        s_base = _score(a)

        for _ in range(12):
            a_trial = _project_simplex(y_proj - alpha * g, 1.0)
            s_trial = _score(a_trial)
            if s_trial < s_new - 1e-14:
                a_new = a_trial
                s_new = s_trial
                break
            alpha *= 0.5

        # If we couldn't improve, try from current a
        if s_new >= s_base - 1e-14:
            a_trial2 = _project_simplex(a - alpha * g, 1.0)
            s_trial2 = _score(a_trial2)
            if s_trial2 < s_base - 1e-14:
                a_new = a_trial2
                s_new = s_trial2

        # Update velocity and position
        v = a_new - a
        a = a_new
        cur_score = s_new

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            # Slowly increase step if making progress
            step = min(step * 1.02, 2.0 / n)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                step *= 0.92
                if step < 1e-6:
                    step = 1e-6

        iter_count += 1
        if iter_count % 5000 == 0:
            elapsed = time.monotonic() - t_start
            rate = iter_count / max(elapsed, 0.001)
            print(f"    {name} iter={iter_count} score={best_score:.8f} "
                  f"step={step:.2e} rate={rate:.0f}it/s")

    return best_a, best_score


def _make_starts(n, rng):
    """Generate diverse starting points."""
    starts = []

    # Uniform
    starts.append(np.ones(n, dtype=np.float64) / n)

    # Linear ramp
    a = np.arange(1, n + 1, dtype=np.float64)
    starts.append(a / np.sum(a))

    # Cosine bell
    i = np.arange(n, dtype=np.float64)
    a = 0.5 + 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.01)
    starts.append(a / np.sum(a))

    # Gaussian center
    center = (n - 1) / 2.0
    a = np.exp(-((i - center) ** 2) / (2 * (n / 5) ** 2))
    starts.append(a / np.sum(a))

    # U-shape (high edges, low center)
    a = np.ones(n, dtype=np.float64)
    q = n // 3
    a[q:2 * q] = 0.2
    starts.append(a / np.sum(a))

    # Sawtooth
    a = np.abs(np.sin(np.pi * i / n))
    a = np.maximum(a, 0.01)
    starts.append(a / np.sum(a))

    # Random
    a = rng.random(n)
    starts.append(a / np.sum(a))

    return starts


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize 2*n*max(convolve(a,a))/sum(a)^2 for non-negative sequence a of length 6121.
    Uses convex optimization (projected gradient on log-sum-exp smoothed max).
    """
    rng = np.random.default_rng(seed)
    total_time = max(1.0, float(budget_s) - 15.0)
    deadline = time.monotonic() + total_time

    n = 6121

    # Gather starting points
    starts = []
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.size == n:
            p = np.clip(parent, 0.0, None)
            S = np.sum(p)
            if S > 0.01:
                starts.append(p / S)
                print(f"Using parent sequence, sum={S:.4f}, initial score={_score(p/S):.6f}")
    except NameError:
        pass

    if not starts:
        starts.append(np.ones(n, dtype=np.float64) / n)

    starts.extend(_make_starts(n, rng))

    # Global best
    global_best = None
    global_score = np.inf

    # Phase 1: Multi-start with moderate tau (50% of time)
    t_remaining = deadline - time.monotonic()
    phase1_end = time.monotonic() + 0.5 * t_remaining
    tau1 = 0.02
    step1 = 1.0 / n
    mom1 = 0.85
    per_start = (phase1_end - time.monotonic()) / len(starts)

    print(f"\n=== Phase 1: Multi-start (tau={tau1}, {len(starts)} starts) ===")
    for idx, s in enumerate(starts):
        if time.monotonic() >= phase1_end:
            break
        s_deadline = min(time.monotonic() + per_start, phase1_end)
        ba, bs = _optimize_phase(s, s_deadline, tau1, step1, mom1, name=f"MS{idx}")
        if bs < global_score:
            global_score = bs
            global_best = ba.copy()
            print(f"  Start {idx}: NEW BEST score={global_score:.8f}")
        else:
            print(f"  Start {idx}: score={bs:.8f} (best remains {global_score:.8f})")

    # Phase 2: Polish with smaller tau (30% of time)
    t_remaining = deadline - time.monotonic()
    phase2_end = time.monotonic() + 0.35 * t_remaining
    tau2 = 0.002
    step2 = 0.3 / n
    mom2 = 0.9
    print(f"\n=== Phase 2: Polish (tau={tau2}) ===")
    ba, bs = _optimize_phase(global_best, phase2_end, tau2, step2, mom2, name="P2")
    if bs < global_score:
        global_score = bs
        global_best = ba.copy()
        print(f"  Phase 2: NEW BEST score={global_score:.8f}")
    else:
        print(f"  Phase 2: score={bs:.8f} (no improvement)")

    # Phase 3: Fine polish (remaining time)
    tau3 = 0.0002
    step3 = 0.08 / n
    mom3 = 0.95
    print(f"\n=== Phase 3: Fine polish (tau={tau3}) ===")
    ba, bs = _optimize_phase(global_best, deadline, tau3, step3, mom3, name="P3")
    if bs < global_score:
        global_score = bs
        global_best = ba.copy()
        print(f"  Phase 3: NEW BEST score={global_score:.8f}")
    else:
        print(f"  Phase 3: score={bs:.8f} (no improvement)")

    # Final check and return
    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"\nFINAL SCORE: {final_score:.8f}")
    print(f"Best internal score: {global_score:.8f}")
    print(f"Sequence stats: mean={np.mean(global_best):.6f}, "
          f"min={np.min(global_best):.6f}, max={np.max(global_best):.6f}")
    return result
