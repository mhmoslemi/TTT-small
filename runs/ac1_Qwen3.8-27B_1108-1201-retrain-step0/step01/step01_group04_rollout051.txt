Uses FFT-accelerated best-response iterations with mixing, combined with gradient polish.
The key innovation over gradient-only approaches is the MW best response, which
runs in O(n log n) per iteration and naturally handles the max constraint structure.

import time

except Exception:
    from numpy.fft import fft, ifft

    def next_fast_len(n):
        n = int(max(1, n))
        m = 1
        while m < n:
            m <<= 1
        return m


def _fft_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def conv_full(a):
    """Full linear convolution of a with itself (autocorrelation), length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0)
    L = _fft_len(n)
    F = fft(a, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def conv_cross(a, b):
    """Full linear convolution of a and b, length 2*min(len(a),len(b))-1."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = min(a.size, b.size)
    if n == 0:
        return np.empty(0)
    L = int(next_fast_len(max(1, a.size + b.size - 1)))
    Fa = fft(a, L)
    Fb = fft(b, L)
    c = np.real(ifft(Fa * Fb))
    return c[:a.size + b.size - 1]


def norm1(p):
    """Normalize to sum=1, non-negative."""
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-300:
        return np.full(p.size, 1.0 / max(1, p.size))
    return p / s


def score(p):
    """Compute 2n*max(conv(p,p))/(sum p)^2."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    s = float(p.sum())
    if s < 1e-300:
        return float('inf')
    c = conv_full(p)
    cm = float(np.max(c)) if c.size else 0.0
    if cm < 0:
        cm = 0.0
    return float(2.0 * n * cm / (s * s))


def mw_best_response(p, g_init, n_mw_iters, eta_init=0.5):

    Multiplicative weights approximation of:
        max sum(g) s.t. (p*g)_k <= M for all k, g >= 0
    Returns normalized g (sum=1).
    """
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    c_p = conv_full(p)
    M = float(np.max(c_p)) if c_p.size else 1.0
    if M < 1e-300:
        M = 1e-300

        g = norm1(g_init)
    else:
        g = np.full(n, 1.0 / n)

        pg = conv_cross(p, g)
        v = pg / M  # violations, ideally <= 1

        v_pad = np.concatenate([[0.0], v[:n-1]])  # length n

        v_with_zero = np.concatenate([[0.0], v])  # length 2n
        prefix = np.cumsum(v_with_zero)  # length 2n

        log_g -= np.max(log_g)
        g = np.exp(log_g)
        g = np.maximum(g, 1e-300)
        g = g / g.sum()

    return g

    # This gives h[m] = sum_j p[j] * w[m-j] for valid j
    # We want grad_j = 2 * sum_k w[k] * (p[k-j] if valid + p[j-k] if valid)
    # = 2 * (h[j] + h[2*(n-1) - j]) where h is computed as above with proper indices

    # Actually the cleanest way:
    # grad_j = 2 * sum_k w_k * p_{(k-j) mod stuff}
    # Let me just do it with the FFT properly.

    # Define: for each j, grad_j/2 = sum_k w_k * A_{k,j} where A_{k,j} = p_j if j<=k<=j+n-1
    # This is (A^T w)_j which we computed in the MW section!
    # A^T w)_j = p_j * sum_{k=max(0,j)}^{min(2n-2, j+n-1)} w_k
    # Wait no, that was for the specific structure where A_{k,j} = p_j * [j<=k<=j+n-1]
    # Let me re-derive: (p*g)_k = sum_j p_j * g_{k-j} (with g_i=0 outside [0,n-1])
    # So A_{k,j} = p_j if 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
    # (A^T w)_j = sum_k A_{k,j} w_k = p_j * sum_{k: j<=k<=j+n-1} w_k
    # (with k clamped to [0, 2n-2])

    # So grad_j = 2 * p_j * sum_{k=max(0,j)}^{min(2n-2, j+n-1)} w_k

    w_pad = np.concatenate([[0.0], w])  # length 2n
    prefix = np.cumsum(w_pad)
    window_sum = prefix[n:n+n] - prefix[0:n]
    grad = 2.0 * p * window_sum

    surr = cmax + T * np.log(s)
    return grad, c, T, surr


def project_simplex(v):
    """Project onto {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def make_shape(n, kind, params, rng=None):
    """Generate a shape of length n."""
    x = np.linspace(0.0, 1.0, n)
    if kind == 'uniform':
        p = np.ones(n)
    elif kind == 'cup':
        a = params[0] if params else 2.0
        p = 1.0 + a * (2.0 * x - 1.0) ** 2
    elif kind == 'cap':
        a = params[0] if params else 0.5
        p = 1.0 - a * (2.0 * x - 1.0) ** 2
    elif kind == 'power':
        a = params[0] if params else 1.0
        p = np.power(np.maximum(x, 1e-10), a) + np.power(np.maximum(1.0 - x, 1e-10), a)
    elif kind == 'geometric':
        s = params[0] if params else 2.0
        p = np.concatenate([np.exp(-s * x[:n//2]), np.exp(-s * (1.0 - x[n//2:]))])
        # Make it symmetric-ish: high in middle
        p = np.exp(-s * (2.0 * x - 1.0) ** 2)
    elif kind == 'two_peaks':
        mu1 = params[0] if params else 0.3
        mu2 = params[1] if len(params) > 1 else 0.7
        sig = params[2] if len(params) > 2 else 0.05
        p = np.exp(-((x - mu1) ** 2) / (2 * sig ** 2)) + np.exp(-((x - mu2) ** 2) / (2 * sig ** 2))
    elif kind == 'chebyshev_combo':
        if rng is None:
            rng = np.random.default_rng(0)
        z = 2.0 * x - 1.0
        logp = np.zeros(n)
        D = 8
        T_nm = np.ones(n)
        T_n = z.copy()
        for d in range(1, D + 1):
            coeff = float(rng.uniform(-1.5, 1.5))
            if d == 1:
                logp += coeff * T_n
            elif d == 2:
                T_np = 2.0 * z * T_n - T_nm
                logp += coeff * T_np
                T_nm = T_n
                T_n = T_np
            else:
                T_np = 2.0 * z * T_n - T_nm
                logp += coeff * T_np
                T_nm = T_n
                T_n = T_np
        logp += 1.5 * np.log(x + 1e-4) + 1.5 * np.log(1.0 - x + 1e-4)
        m = np.max(logp)
        p = np.exp(logp - m)
    elif kind == 'random_smooth':
        if rng is None:
            rng = np.random.default_rng(0)
        f1 = int(rng.integers(1, max(2, n // 100)))
        f2 = int(rng.integers(1, max(2, n // 50)))
        f3 = int(rng.integers(1, max(2, n // 20)))
        p1 = float(rng.uniform(0, 2 * np.pi))
        p2 = float(rng.uniform(0, 2 * np.pi))
        p3 = float(rng.uniform(0, 2 * np.pi))
        p = 1.0
        p += 0.5 * np.sin(2 * np.pi * f1 * x + p1)
        p += 0.3 * np.sin(2 * np.pi * f2 * x + p2)
        p += 0.2 * np.sin(2 * np.pi * f3 * x + p3)
        p = np.maximum(p, 0.01)
    else:
        p = np.ones(n)

    return norm1(np.maximum(p, 0.0))


def mw_optimize(p0, budget_s, n_mw_per_step=20, eta_init=0.5, verbose=False):
    """
    Main MW-based optimization: iterate p <- mix(p, best_response(p)).
    """
    p0 = np.asarray(p0, dtype=np.float64)
    n = int(p0.size)
    p = norm1(p0)
    g = np.full(n, 1.0 / n)  # warm start for best response

    best_p = p.copy()
    best_score = score(p)
    current_score = best_score

    end = time.monotonic() + max(0.0, float(budget_s))
    outer = 0
    no_improve = 0

    while time.monotonic() < end - 2.0:
        outer += 1

        # Compute best response
        n_mw = max(5, min(n_mw_per_step, int(n_mw_per_step * (1.0 + 0.5 * np.exp(-outer / 50.0)))))
        eta = eta_init / (1.0 + 0.02 * outer)
        g = mw_best_response(p, g, n_mw, eta)

        # Mix with decreasing step
        tau = min(1.0, 2.5 / (outer + 2.0))
        if outer > 200:
            tau = min(tau, 1.0 / (np.log(outer + 1) + 1.0))

        p_new = (1.0 - tau) * p + tau * g
        p_new = project_simplex(p_new)

        sc_new = score(p_new)

        if np.isfinite(sc_new) and sc_new < best_score - 1e-12:
            best_score = sc_new
            best_p = p_new.copy()
            no_improve = 0
            if verbose and outer % 50 == 0:
                print("  MW iter %d: score=%.8f (improved)" % (outer, best_score))
        elif np.isfinite(sc_new) and sc_new < current_score + 0.1 * best_score:
            no_improve += 1
        else:
            no_improve += 1

        p = p_new
        current_score = sc_new

        # Periodic perturbation to escape local optima
        if no_improve > 300 and n > 4:
            sigma = max(1.0 / n, 0.001)
            p = norm1(p + np.random.default_rng(outer * 7919).normal(0, sigma, n))
            current_score = score(p)
            no_improve = 0

        if no_improve > 2000:
            break

        # Check time for final gradient polish
        if time.monotonic() > end - 30.0:
            break

    return best_p, best_score


def gradient_polish(p0, budget_s, verbose=False):
    """Polish using smoothed-max gradient descent on the simplex."""
    p0 = np.asarray(p0, dtype=np.float64)
    n = int(p0.size)
    p = norm1(p0)

    best_p = p.copy()
    best_score = score(p)

    end = time.monotonic() + max(0.0, float(budget_s))
    T_factor = 0.05
    iters = 0
    no_improve = 0

    while time.monotonic() < end - 1.0:
        iters += 1
        T_factor = max(0.001, 0.05 * (0.995 ** iters))

        grad, c, T, surr = gradient_smoothed_max(p, T_factor)

        if not np.all(np.isfinite(grad)):
            break

        gmax = float(np.max(np.abs(grad))) if grad.size else 0.0
        if gmax < 1e-15:
            break

        base_lr = max(1.0 / n, 0.05 * float(np.max(p))) / (gmax + 1e-15)
        lrs = (0.5 * base_lr, 0.1 * base_lr, 0.02 * base_lr, 0.005 * base_lr, 0.001 * base_lr)

        accepted = False
        for lr in lrs:
            p_new = project_simplex(p - lr * grad)
            sc = score(p_new)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_p = p_new.copy()
                p = p_new
                no_improve = 0
                accepted = True
                if verbose and iters % 20 == 0:
                    print("  Grad iter %d: score=%.8f" % (iters, best_score))
                break
            elif np.isfinite(sc) and sc < score(p) + 1e-12:
                p = p_new
                no_improve += 1
                accepted = True
                break

        if not accepted:
            no_improve += 1
            # Small random perturbation
            if no_improve > 100:
                sigma = 0.05 / n
                p = norm1(p + np.random.default_rng(iters * 104729).normal(0, sigma, n))
                no_improve = 0

        if no_improve > 500:
            break

    return best_p, best_score


def generate_initials(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        parent = np.asarray(parent, dtype=np.float64)
        if parent.size == n:
            inits.append(norm1(parent))
        elif parent.size >= 16:
            # Resample parent to length n
            idx = np.linspace(0, parent.size - 1, n).astype(int)
            inits.append(norm1(parent[idx]))

    shapes = [
        ('uniform', []),
        ('cup', [0.5]), ('cup', [1.0]), ('cup', [2.0]), ('cup', [4.0]),
        ('cap', [0.3]), ('cap', [0.6]), ('cap', [0.9]),
        ('power', [0.5]), ('power', [1.0]), ('power', [2.0]), ('power', [4.0]),
        ('geometric', [1.0]), ('geometric', [2.0]), ('geometric', [4.0]), ('geometric', [8.0]),
        ('two_peaks', [0.25, 0.75, 0.04]),
        ('two_peaks', [0.2, 0.8, 0.06]),
        ('two_peaks', [0.15, 0.85, 0.03]),
        ('two_peaks', [0.35, 0.65, 0.08]),
    ]

    for kind, params in shapes:
        inits.append(make_shape(n, kind, params, rng))

    for _ in range(8):
        inits.append(make_shape(n, 'chebyshev_combo', [], rng))
    for _ in range(8):
        inits.append(make_shape(n, 'random_smooth', [], rng))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a nonnegative step function minimizing 2n*max(autocorr)/(sum)^2.
    Uses MW best-response iteration + gradient polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 15.0)

    best_p = None
    best_score = float('inf')

    # Try to use parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and float(parent_arr.sum()) > 0.0:
            parent = parent_arr
            print("Using parent of length %d" % parent_arr.size, flush=True)
    except Exception:
        pass

    # Determine sizes to try
    sizes = []
    if parent is not None:
        pn = int(parent.size)
        sizes.append(pn)
        sizes.append(max(256, pn // 2))
        sizes.append(min(32768, pn * 2))
    sizes.extend([512, 1024, 2048, 4096, 8192, 16384])
    # Deduplicate and sort
    sizes = sorted(set(s for s in sizes if s >= 128 and s <= 32768))

    # Phase 1: Quick scan to find best size and shape
    print("Phase 1: Quick scan", flush=True)
    size_best = []  # (score, n, p)

    for n in sizes:
        if time.monotonic() > deadline - 120.0:
            break
        inits = generate_initials(n, rng, parent)
        # Quick score all inits
        scored = []
        for p in inits:
            sc = score(p)
            if np.isfinite(sc):
                scored.append((sc, p))
        scored.sort(key=lambda x: x[0])
        if scored:
            size_best.append((scored[0][0], n, scored[0][1]))
            print("  n=%d best init score=%.6f" % (n, scored[0][0]), flush=True)

    size_best.sort(key=lambda x: x[0])
    # Pick top sizes for deeper optimization
    top_sizes = []
    for sc, n, p in size_best[:5]:
        if n not in top_sizes:
            top_sizes.append(n)
    top_sizes = top_sizes[:4]

    print("Top sizes for deep search: %s" % top_sizes, flush=True)

    # Phase 2: Deep MW optimization on best sizes
    print("Phase 2: MW optimization", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 60.0:
        time_per_size = max(30.0, (remaining - 90.0) / max(1, len(top_sizes)))

        for n in top_sizes:
            if time.monotonic() > deadline - 90.0:
                break
            budget = min(time_per_size, deadline - time.monotonic() - 90.0)
            if budget < 20.0:
                break

            inits = generate_initials(n, rng, parent)
            scored = []
            for p in inits:
                sc = score(p)
                if np.isfinite(sc):
                    scored.append((sc, p))
            scored.sort(key=lambda x: x[0])

            # Run MW on top 3 inits
            time_per_init = max(10.0, budget / 3.0)
            for sc0, p0 in scored[:3]:
                if time.monotonic() > deadline - 90.0:
                    break
                init_budget = min(time_per_init, deadline - time.monotonic() - 90.0)
                if init_budget < 10.0:
                    break

                p_opt, sc_opt = mw_optimize(p0, init_budget, n_mw_per_step=15, eta_init=0.4)
                if np.isfinite(sc_opt) and sc_opt < best_score:
                    best_score = sc_opt
                    best_p = p_opt.copy()
                    print("  MW n=%d: score=%.8f" % (n, best_score), flush=True)

    # Phase 3: Gradient polish on the best
    if best_p is not None and time.monotonic() < deadline - 20.0:
        print("Phase 3: Gradient polish", flush=True)
        polish_budget = min(120.0, deadline - time.monotonic() - 15.0)
        p_opt, sc_opt = gradient_polish(best_p, polish_budget, verbose=True)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_p = p_opt.copy()
            print("  After polish: score=%.8f" % best_score, flush=True)

    # Phase 4: Another MW round on the polished result
    if best_p is not None and time.monotonic() < deadline - 15.0:
        mw_budget = min(60.0, deadline - time.monotonic() - 10.0)
        if mw_budget > 10.0:
            p_opt, sc_opt = mw_optimize(best_p, mw_budget, n_mw_per_step=25, eta_init=0.3)
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_p = p_opt.copy()
                print("  After 2nd MW: score=%.8f" % best_score, flush=True)

    # Phase 5: Final gradient polish
    if best_p is not None and time.monotonic() < deadline - 5.0:
        polish_budget = max(5.0, deadline - time.monotonic() - 5.0)
        p_opt, sc_opt = gradient_polish(best_p, polish_budget)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_p = p_opt.copy()
            print("  Final polish: score=%.8f" % best_score, flush=True)

    if best_p is None:
        best_p = np.full(1024, 1.0 / 1024)
        best_score = score(best_p)

    # Output
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s < 0.01:
        best_p = np.full(max(128, int(best_p.size)), 1.0)
        s = float(best_p.sum())

    print("FINAL: n=%d score=%.8f sum=%.4f" % (best_p.size, best_score, s), flush=True)
    return [float(x) for x in best_p]
