"""
Multi-phase optimization for minimizing the peak of linear autocorrelation
of non-negative step functions. Uses FFT-based gradient descent with
log-sum-exp annealing, multiple sequence lengths and initializations,
followed by targeted peak-shaving local search.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def local_score(a):
    """Compute the score for a sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def lse_max(b, T):
    """Log-sum-exp approximation of max. Returns (value, weights)."""
    b_max = np.max(b)
    shifted = T * (b - b_max)
    # For numerical stability
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_shifted)
    log_sum_exp = b_max + max_shifted / T + np.log(sum_exp) / T
    weights = exp_shifted / sum_exp
    return log_sum_exp, weights

def gradient_lse_conv(a, T):
    """
    Compute value and gradient of LSE_T(conv(a,a)) w.r.t. a.
    Returns (value, grad).
    """
    n = len(a)
    c = fft_convolve(a, a)
    value, w = lse_max(c, T)
    # Gradient: grad[j] = 2 * sum_l a[l] * w[j+l] for l=0..n-1
    # = 2 * conv(a[::-1], w)[j + n - 1]
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad = 2.0 * cross[n-1:2*n-1]
    return value, grad, c

def make_u_shape(n, alpha, rng):
    """U-shaped initialization."""
    t = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2*t - 1)**2
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_inverted_u(n, alpha, rng):
    """Inverted U-shaped (hill) initialization."""
    t = np.linspace(0, 1, n)
    a = 1.0 + alpha * np.sin(np.pi * t)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_random(n, rng):
    """Random initialization."""
    return np.clip(1.0 + 0.3 * rng.standard_normal(n), 0.01, None)

def make_cosh(n, alpha, rng):
    """Cosh-shaped (U-like) initialization."""
    t = np.linspace(0, 1, n)
    a = np.cosh(alpha * (2*t - 1))
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_power(n, p, rng):
    """Power-law shaped initialization."""
    t = np.linspace(0, 1, n)
    # p > 0: U-shaped, p < 0: inverted U
    a = 1.0 + p * (2*t - 1)**2
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def gradient_optimize(a0, time_limit, rng):
    """
    Adam gradient descent on theta = log(a) with LSE annealing.
    """
    n = len(a0)
    S = np.sqrt(2.0 * n)  # normalization target
    theta = np.log(a0 / np.sum(a0) * S).astype(np.float64)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    lr = 1.0
    
    best_score = np.inf
    best_theta = theta.copy()
    
    t0 = time.monotonic()
    tau_start = 1.0  # initial T for LSE (small = smooth)
    tau_end = 50.0 * n  # final T (large = sharp, approximates max)
    
    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        # Anneal T from smooth to sharp
        frac = min(1.0, elapsed * 0.8)  # finish annealing at 80% of time
        T = tau_start * (tau_end / tau_start) ** frac
        
        # Compute a from theta
        theta_max = np.max(theta)
        a_unnorm = np.exp(theta - theta_max)
        a = a_unnorm / np.sum(a_unnorm) * S
        
        # Compute gradient
        value, grad_a, c = gradient_lse_conv(a, T)
        
        # Chain rule: d(theta)/d(a) for a = softmax(theta) * S
        # Actually, a = exp(theta - max) / sum(exp(theta - max)) * S
        # da_j/dtheta_k = a_j * (delta_jk - a_k / S)
        # So grad_theta_k = sum_j grad_a_j * a_j * (delta_jk - a_k/S)
        #                = a_k * (grad_a_k - sum_j grad_a_j * a_j / S)
        inner = np.dot(a, grad_a) / S
        grad_theta = a * (grad_a - inner)
        
        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta**2
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        
        # Adaptive learning rate
        lr_eff = lr / np.sqrt(1 + step * 0.01)
        theta = theta - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate actual score periodically
        if step % 50 == 0 or step < 10:
            score = local_score(a)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_theta = theta.copy()
        
        if step % 500 == 0:
            score_now = local_score(a)
            print(f"      step={step}, T={T:.1f}, score={score_now:.6f}, best={best_score:.6f}")
    
    # Final evaluation
    a_best = np.exp(best_theta - np.max(best_theta))
    a_best = a_best / np.sum(a_best) * S
    final_score = local_score(a_best)
    
    return a_best, final_score

def local_search(a_init, time_limit, rng):
    """
    Targeted peak-shaving local search.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    current_score = local_score(a)
    best = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    temp = 0.001
    cooling = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.4:
            # Targeted peak reduction
            c = fft_convolve(a, a)
            peak = np.max(c)
            mean_c = np.sum(c) / len(c)
            # Find peak lags
            peak_threshold = peak - 0.01 * (peak - mean_c)
            peak_lags = np.where(c >= peak_threshold)[0]
            
            # For each peak lag, find top contributors
            for p in peak_lags[:3]:  # top 3 peak lags
                # Contributions: a[i] * a[p-i] for valid i
                i_min = max(0, p - n + 1)
                i_max = min(n - 1, p)
                i_range = range(i_min, i_max + 1)
                contribs = np.array([a[i] * a[p - i] for i in i_range])
                idx = np.arange(i_min, i_max + 1)
                
                # Reduce top contributors
                top_k = max(2, len(idx) // 8)
                top_positions = idx[np.argsort(contribs)[-top_k:]]
                candidate[top_positions] *= np.exp(rng.normal(-0.03, 0.01, top_k))
                
                # Increase some non-contributors
                others = np.setdiff1d(np.arange(n), top_positions)
                if len(others) > 0:
                    n_inc = min(5, len(others))
                    inc_pos = rng.choice(others, n_inc, replace=False)
                    candidate[inc_pos] *= np.exp(rng.normal(0.03, 0.01, n_inc))
        
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0.0, 0.02, width))
        
        elif r < 0.8:
            # Pair swap (equalize two values)
            i, j = rng.choice(n, 2, replace=False)
            diff = a[i] - a[j]
            transfer = 0.02 * abs(diff) * rng.random()
            if diff > 0:
                candidate[i] -= transfer
                candidate[j] += transfer
            else:
                candidate[i] += transfer
                candidate[j] -= transfer
        
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.005, n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        elif np.exp(-(new_score - current_score) / max(temp, 1e-10)) > rng.random():
            a = candidate
            current_score = new_score
        
        temp *= cooling
    
    return best, best_score
