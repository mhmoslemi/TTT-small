    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def a_from_theta(theta, S):
    """Convert log-parameters to sequence: a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def compute_soft_weights(c, T):
    """Compute soft-max weights proportional to exp(T*c/c_max)."""
    c_max = c.max()
    if c_max < 1e-15:
        return np.ones(len(c)) / len(c)
    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        return np.ones(len(c)) / len(c)
    return exp_scaled / total


def compute_grad_a(w, a):
    """Compute grad_a[m] = 2 * sum_j w[m+j] * a[j] for m=0..n-1 using FFT."""
    n = len(a)
    a_rev = a[::-1]
    # grad[m] = 2 * conv(w, a_rev)[m + n - 1]
    full_conv = fft_convolve(w, a_rev)
    # full_conv has length (2n-1) + n - 1 = 3n-2
    # We need full_conv[n-1 : 2n-1]
    end_idx = min(2 * n - 1, len(full_conv))
    grad = np.zeros(n)
    grad[:] = 2.0 * full_conv[n - 1: end_idx][:n]
    return grad


def softmax_objective_and_grad(theta, n, T, S):
    """Objective (soft-max of self-conv) and gradient w.r.t. theta."""
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()

    # Soft-max: c_max + (1/T)*log(sum_k exp(T*(c_k - c_max)))
    shifted = T * (c - c_max)
    # max of shifted is 0, so log(sum exp(shifted)) = log(1 + sum exp(shifted[shifted<0]))
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T

    # Weights for gradient
    w = compute_soft_weights(c, T)
    grad_a = compute_grad_a(w, a)

    # Chain rule: dL/dtheta[j] = grad_a[j]*a[j] - (a[j]/S)*sum_i grad_a[i]*a[i]
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot

    return obj, grad_theta


def true_objective(a):
    """Compute the true max of self-convolution (for final evaluation)."""
    c = fft_convolve_self(a)
    return c.max()


def run_adam_theta(theta_init, n, T, S, time_budget, lr=0.05, top_k=0):
    """Adam optimizer on log-parameters with soft-max objective."""
    theta = theta_init.copy().astype(np.float64)
    n_par = len(theta)
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    start = time.monotonic()
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    best_theta = theta.copy()

    while time.monotonic() - start < time_budget:
        t += 1
        a = a_from_theta(theta, S)
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()
            best_theta = theta.copy()

        # Compute gradient
        if T < 1e6:
            obj, grad = softmax_objective_and_grad(theta, n, T, S)
        else:
            # Use hard top-k subgradient
            w = np.zeros(len(c))
            idx = np.argsort(c)[::-1][:max(1, top_k)]
            w[idx] = 1.0 / len(idx)
            grad_a = compute_grad_a(w, a)
            dot = np.dot(grad_a, a)
            grad = grad_a * a - (a / S) * dot

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm * gnorm  # normalize for adaptive lr

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adaptive learning rate
        lr_t = lr * (0.5 ** (t / 500.0))
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)

    return best_theta, best_a, best_obj


def run_adam_with_annealing(theta_init, n, S, time_budget):
    """Adam with temperature annealing: T goes from 5 to 200 over time."""
    theta = theta_init.copy().astype(np.float64)
    n_par = len(theta)
    total_steps = int(time_budget / 0.002)  # estimate
    start = time.monotonic()

    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    best_theta = theta.copy()

    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    while time.monotonic() - start < time_budget:
        t += 1
        # Temperature schedule: exponential increase
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = 5.0 * np.exp(4.0 * progress)  # from 5 to ~110

        a = a_from_theta(theta, S)
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()
            best_theta = theta.copy()

        if T < 1e6:
            obj, grad = softmax_objective_and_grad(theta, n, T, S)
        else:
            w = np.zeros(len(c))
            idx = np.argsort(c)[::-1][:8]
            w[idx] = 1.0 / len(idx)
            grad_a = compute_grad_a(w, a)
            dot = np.dot(grad_a, a)
            grad = grad_a * a - (a / S) * dot

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.05 * (0.7 ** (progress * 3.0))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_theta, best_a, best_obj


def coordinate_descent_polish(a, n, S, time_budget, max_sweeps=50):
    """Coordinate descent to polish the sequence."""
    a = a.copy().astype(np.float64)
    n_par = len(a)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n_par):
            if time.monotonic() - start > time_budget:
                break
            # Try to reduce a[i] (if it contributes to the max)
            # Quick check: does decreasing a[i] help?
            original = a[i]
            if original <= 1e-10:
                continue
            # Binary search for optimal a[i]
            lo, hi = 0.0, original * 1.5
            best_local = original
            best_local_obj = best_obj
            for _ in range(12):
                mid = (lo + hi) / 2.0
                a[i] = mid
                # Renormalize
                s = a.sum()
                if s < 1e-10:
                    a[i] = original
                    break
                a_norm = a * (S / s)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_local_obj:
                    best_local_obj = obj_check
                    best_local = mid
                if obj_check < best_obj * 1.01:
                    lo = mid
                else:
                    hi = mid
            a[i] = best_local
            if best_local < original - 1e-8 or best_local > original + 1e-8:
                improved = True
            # Renormalize
            s = a.sum()
            if s < 1e-10:
                a = np.full(n_par, S / n_par)
            else:
                a = a * (S / s)
            c = fft_convolve_self(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()

        if not improved:
            break

    return best_a, best_obj


def solve_lp_direction(f, M, n, top_k_constraints=50):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0.
    Returns the optimal g (normalized to sum=1) and the LP objective value."""
    from scipy.optimize import linprog

    c = fft_convolve_self(f)
    # Get top-k lags
    if top_k_constraints < len(c):
        active_lags = np.argsort(c)[::-1][:top_k_constraints]
    else:
        active_lags = np.arange(len(c))

    # Build constraint matrix A: A[k,j] = f[k-j] if 0<=k-j<n
    A_rows = []
    for k in active_lags:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            row[j] = f[k - j]
            row[k - j] += f[j]  # Wait, this double-counts. Let me redo.
        A_rows.append(row)
    # Actually, (f*g)[k] = sum_j f[k-j]*g[j] where 0<=k-j<n
    # So A[k,j] = f[k-j] if 0<=k-j<n, else 0
    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[idx, j] = f[k - j]

    # Maximize sum(g) = -minimize -sum(g)
    c_obj = -np.ones(n)
    b = np.full(len(active_lags), M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                         method='highs', options={'time_limit': 10, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return None, None


def lbfgs_fun(theta, n, T, S, state_dict):
    """Objective function for L-BFGS-B. Top-level, no closures."""
    elapsed = time.monotonic() - state_dict['start']
    if elapsed > state_dict['time_budget']:
        return state_dict['best_obj'], np.zeros(len(theta))
    obj, grad = softmax_objective_and_grad(theta, n, T, S)
    # Also track true max
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    true_max = c.max()
    if true_max < state_dict['best_obj']:
        state_dict['best_obj'] = true_max
        state_dict['best_a'] = a.copy()
    return obj, grad


def run_lbfgs(theta_init, n, T, S, time_budget, max_iter=500):
    """Run L-BFGS-B on the soft-max objective with log-parameters."""
    state_dict = {
        'n': n, 'T': T, 'S': S,
        'start': time.monotonic(), 'time_budget': time_budget,
        'best_obj': np.inf, 'best_a': None
    }
    # Initialize best
    a0 = a_from_theta(theta_init, S)
    c0 = fft_convolve_self(a0)
    state_dict['best_obj'] = c0.max()
    state_dict['best_a'] = a0.copy()

    try:
        result = minimize(
            lbfgs_fun, theta_init,
            args=(n, T, S, state_dict),
            jac=True, method='L-BFGS-B',
            options={'maxiter': max_iter, 'maxfun': 2000, 'maxls': 50,
                     'gtol': 1e-8}
        )
    except Exception as e:
        print("L-BFGS failed: %s" % str(e))

    return state_dict['best_a'], state_dict['best_obj']


def get_initialization(name, n, rng, S):
    """Generate various initializations."""
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "gaussian":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n / 2) / (n / 6)) ** 2)
        return a * (S / a.sum())
    elif name == "cosine":
        i = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * i / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "anti_peak":
        i = np.arange(n)
        a = 1.0 - 0.4 * np.cos(2 * np.pi * (n - 1) * i / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "raised":
        i = np.arange(n)
        t = i / (n - 1)
        a = np.sin(np.pi * t) ** 2
        return a * (S / a.sum())
    elif name == "two_humps":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n * 0.25) / (n / 10)) ** 2) + \
            np.exp(-0.5 * ((i - n * 0.75) / (n / 10)) ** 2)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    global_best = np.inf
    global_a = None

    # Try to load the given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given sequence: n=%d, obj=%.6f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given sequence: %s" % str(e))

    # Phase 1: Exploration with various lengths and initializations
    # Allocate 40% of time to exploration
    explore_budget = 0.35 * (deadline - time.monotonic())

    lengths = [2048, 4096, 8192]
    if given_a is not None:
        lengths.insert(0, len(given_a))

    init_names = ["uniform", "random", "gaussian", "cosine", "anti_peak", "two_humps"]
    attempts_per_length = max(1, int(explore_budget / (len(lengths) * 8.0)))

    for n in lengths:
        if time.monotonic() > deadline - 60.0:
            break
        S = np.sqrt(2.0 * float(n))
        per_init_time = min(5.0, explore_budget / (len(lengths) * min(len(init_names), attempts_per_length)))

        for init_name in init_names:
            if time.monotonic() > deadline - 60.0:
                break
            if per_init_time < 1.0:
                break

            a_init = get_initialization(init_name, n, rng, S)

            # Convert to log-parameters
            a_init = np.maximum(a_init, 1e-10)
            theta_init = np.log(a_init / a_init.sum())

            # Run Adam with annealing
            t0 = time.monotonic()
            theta_best, a_best, obj_best = run_adam_with_annealing(
                theta_init, n, S, per_init_time
            )
            elapsed = time.monotonic() - t0

            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print("Exploration: n=%d init=%s obj=%.6f (t=%.1fs)" %
                      (n, init_name, obj_best, elapsed))

    # Phase 2: Exploitation from the best found (and from given)
    # Allocate 45% of time to exploitation
    exploit_budget = 0.45 * (deadline - time.monotonic())
    if global_a is not None and exploit_budget > 5.0:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a = global_a.copy()
        a = np.maximum(a, 1e-10)
        a = a * (S / a.sum())
        theta = np.log(a / a.sum())

        # Try L-BFGS at various temperatures
        temps = [10, 30, 80, 200]
        per_temp_time = exploit_budget / (len(temps) + 2)

        for T in temps:
            if time.monotonic() > deadline - 30.0:
                break
            t0 = time.monotonic()
            a_best, obj_best = run_lbfgs(theta, n, T, S, per_temp_time, max_iter=300)
            elapsed = time.monotonic() - t0
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print("L-BFGS T=%d: obj=%.6f (t=%.1fs)" % (T, obj_best, elapsed))
            # Update theta from best
            a_best = np.maximum(a_best, 1e-10)
            a_best = a_best * (S / a_best.sum())
            theta = np.log(a_best / a_best.sum())

        # Continue with Adam at high temperature
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 5.0:
            t0 = time.monotonic()
            theta_best, a_best, obj_best = run_adam_theta(
                theta, n, 500.0, S, remaining, lr=0.02, top_k=8
            )
            elapsed = time.monotonic() - t0
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print("High-T Adam: obj=%.6f (t=%.1fs)" % (obj_best, elapsed))

    # Phase 3: Polish with coordinate descent
    polish_budget = deadline - time.monotonic()
    if global_a is not None and polish_budget > 3.0:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a = global_a.copy()
        a = a * (S / a.sum())
        t0 = time.monotonic()
        a_best, obj_best = coordinate_descent_polish(a, n, S, polish_budget * 0.6)
        elapsed = time.monotonic() - t0
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print("Coord descent: obj=%.6f (t=%.1fs)" % (obj_best, elapsed))

        # Final Adam polish with very high T
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 3.0 and global_a is not None:
            a = global_a.copy()
            a = np.maximum(a, 1e-10)
            a = a * (S / a.sum())
            theta = np.log(a / a.sum())
            t0 = time.monotonic()
            theta_best, a_best, obj_best = run_adam_theta(
                theta, n, 2000.0, S, remaining, lr=0.01, top_k=16
            )
            elapsed = time.monotonic() - t0
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print("Final polish: obj=%.6f (t=%.1fs)" % (obj_best, elapsed))

    # Phase 4: Try LP-based direction if time permits
    if global_a is not None:
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 10.0:
            n = len(global_a)
            S = np.sqrt(2.0 * float(n))
            a = global_a.copy()
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            M = c.max()

            g, lp_val = solve_lp_direction(a, M, n, top_k_constraints=30)
            if g is not None:
                # Blend
                for t_blend in [0.3, 0.5, 0.7]:
                    a_new = (1 - t_blend) * a + t_blend * g * S
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 0.01:
                        a_new = a_new * (S / s_new)
                        c_new = fft_convolve_self(a_new)
                        obj_new = c_new.max()
                        if obj_new < global_best:
                            global_best = obj_new
                            global_a = a_new.copy()
                            print("LP blend t=%.1f: obj=%.6f" % (t_blend, obj_new))
                            a = a_new

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    # Verify with evaluate_sequence
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("FINAL: obj=%.6f (verified=%s)" % (global_best,
          "%.6f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        # Fallback
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.6f" % ver)

    return final_a
