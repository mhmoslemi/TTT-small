    return a * (target / S)


def build_lp_matrix(a):
    """Build the (2n-1) x n matrix A where A[m,k] = a[m-k] if valid."""
    n = len(a)
    n_rows = 2 * n - 1
    A = np.zeros((n_rows, n))
    for m in range(n_rows):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        A[m, lo:hi + 1] = a[m - hi:m - lo + 1][::-1]
    return A


def solve_lp_min_peak(a, S_target):
    """Solve LP: min M' s.t. conv(a,g)_m <= M', sum(g)=S_target, g>=0.
    Returns (g, M_prime) or (None, None) on failure."""
    n = len(a)
    n_rows = 2 * n - 1
    
    # Objective: minimize M' (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: conv(a,g)_m - M' <= 0
    A_base = build_lp_matrix(a)
    A_ub = np.hstack([A_base, -np.ones((n_rows, 1))])
    b_ub = np.zeros(n_rows)
    
    # Equality: sum(g) = S_target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S_target])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_max_sum(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g)_m <= M, g>=0.
    Returns g or None."""
    n = len(a)
    n_rows = 2 * n - 1
    
    c_obj = -np.ones(n)
    A_ub = build_lp_matrix(a)
    b_ub = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def softmax_grad(a, tau):
    """Compute soft-max weighted gradient of peak autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    
    # grad[j] = 2 * sum_k w[j+k] * a[k] for valid k
    grad = np.zeros(n)
    # Use convolution: grad = 2 * convolve(w[:n] reversed, a) ... 
    # Actually: grad[j] = 2 * sum_{k=0}^{n-1} w[j+k] * a[k] (when j+k < 2n-1)
    # This is a cross-correlation of w and a
    # grad[j] = 2 * (w * a)[j] where * is cross-correlation
    # cross_correlation(w, a)[j] = sum_k w[j+k] * a[k]
    # = convolve(w, a_reversed)[j + n - 1] ... let me just do it directly
    # For efficiency, use the fact that this is a convolution
    # grad[j]/2 = sum_k a[k] * w[j+k]
    # Let's compute conv(a, w) which gives c[i] = sum_j a[j] * w[i-j]
    # We want sum_k a[k] * w[j+k] = sum_k a[k] * w[(j+k)]
    # This is cross-correlation: (a ⋆ w)[j] = sum_k a[k] * w[k+j]
    # = conv(a, flip(w))[j + len(w) - 1]
    
    # Simpler: just use the convolution directly
    # sum_k a[k] * w[j+k] for k in [0, n-1] and j+k in [0, 2n-2]
    # This equals (flip_convolve)(a, w)[j] where we define appropriately
    # Let's just use np.correlate
    # np.correlate(a, w)[j] = sum_k a[k] * w[k-j] ... not quite
    
    # Direct approach using convolution:
    # We want: grad[j] = 2 * sum_{k=0}^{n-1} a[k] * w[j+k]
    # Let u = w (length 2n-1), v = a (length n)
    # conv(v, u)[j] = sum_k v[k] * u[j-k] = sum_k a[k] * w[j-k]
    # That's not what we want. We want w[j+k] not w[j-k].
    # conv(v, flip(u))[j + 2n - 2] = sum_k v[k] * flip(u)[j+2n-2-k]
    #   = sum_k a[k] * u[2n-2-(j+2n-2-k)] = sum_k a[k] * u[k-j]
    # Still not right.
    
    # Let's just do: result[j] = sum_k a[k] * w[j+k]
    # This is the cross-correlation of a and w, evaluated at lag j
    # np.correlate(w, a, mode='full')[j + n - 1] = sum_k w[k] * a[k - j + n - 1 - (n-1)]
    # Hmm, let me just use a different approach.
    
    # result[j] = sum_k a[k] * w[j+k] for k in [0,n-1], j in [0,n-1]
    # = sum_k a[k] * w[j+k]
    # If I define w_ext = zeros(2n-1), then this is a valid sum.
    # This is equivalent to: take the convolution of a[::-1] with w, and extract the right part.
    # conv(a[::-1], w)[j + n - 1] = sum_k a[::-1][k] * w[j+n-1-k]
    #   = sum_k a[n-1-k] * w[j+n-1-k]
    # Let l = n-1-k: = sum_l a[l] * w[j+l]  (as k goes 0 to n-1, l goes n-1 to 0)
    #   = sum_{l=0}^{n-1} a[l] * w[j+l] ✓
    
    conv_result = np.convolve(a[::-1], w)
    # We want indices j + n - 1 for j = 0, ..., n-1
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    
    return grad


def gradient_step(a, lr, tau, vel, momentum):
    """One Nesterov-momentum gradient step on peak autocorrelation."""
    grad = softmax_grad(a, tau)
    # Remove mean to maintain sum constraint
    grad -= grad.mean()
    
    vel_new = momentum * vel - lr * grad
    a_new = a + vel_new
    a_new = np.maximum(a_new, 0.0)
    return a_new, vel_new


def optimize_one(a0, deadline, max_grad=3000, max_lp=300):
    """Optimize a single starting sequence. Returns (best_seq, best_score)."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    
    # Phase 1: Nesterov gradient descent
    vel = np.zeros(n)
    momentum = 0.9
    for it in range(max_grad):
        if time.monotonic() > deadline - 90.0:
            break
        progress = it / max_grad
        lr = 0.12 * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, 0.002)
        tau = 20.0 + 60.0 * progress
        
        a, vel = gradient_step(a, lr, tau, vel, momentum)
        a = normalize_seq(a, S)
        
        if (it + 1) % 300 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    # Phase 2: LP-based refinement
    if time.monotonic() < deadline - 20.0:
        no_improve_count = 0
        S_target = float(a.sum())
        
        for it in range(max_lp):
            if time.monotonic() > deadline - 5.0:
                break
            if no_improve_count > 30:
                break
            
            # Solve LP: min M' s.t. conv(a,g) <= M', sum(g) = S, g >= 0
            g, M_prime = solve_lp_min_peak(a, S_target)
            if g is None:
                # Fallback to max-sum LP
                c = np.convolve(a, a)
                M = float(c.max())
                g = solve_lp_max_sum(a, M)
                if g is None:
                    break
                g = normalize_seq(g, S_target)
            
            # Line search on blend factor
            improved = False
            for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
                a_new = (1.0 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize_seq(a_new, S_target)
                s_new = compute_obj(a_new)
                if s_new < best_score * (1.0 - 1e-9):
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    improved = True
                    break
            
            if improved:
                no_improve_count = 0
            else:
                no_improve_count += 1
                # Escape via gradient step
                if no_improve_count % 5 == 0:
                    c = np.convolve(a, a)
                    tau_esc = 50.0 + 20.0 * (no_improve_count / 10.0)
                    a, vel = gradient_step(a, 0.03, tau_esc, vel, 0.9)
                    a = normalize_seq(a, S)
                    s = compute_obj(a)
                    if s < best_score:
                        best = a.copy()
                        best_score = s
    
    return best, best_score


def generate_starts(rng, n_list):
    """Generate diverse starting sequences."""
    starts = []
    
    # Previous best
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size >= 64 and np.all(np.isfinite(prev)):
            starts.append(prev.copy())
            for L in n_list:
                if L <= len(prev):
                    starts.append(prev[:L].copy())
                elif L > len(prev):
                    # Pad with the tail repeated
                    padding_needed = L - len(prev)
                    pad = prev[-min(padding_needed, len(prev)):]
                    starts.append(np.concatenate([prev, pad[:padding_needed]]))
    except (NameError, UnboundLocalError):
        pass
    
    for n in n_list:
        i = np.arange(n, dtype=np.float64)
        
        # Multi-frequency cosines
        for freq_list in [[1, 3], [2, 5], [1, 4, 7], [3, 8], [1, 2, 4, 7]]:
            sig = np.zeros(n)
            for f in freq_list:
                sig += np.cos(2 * np.pi * f * i / n)
            sig = np.abs(sig) + 0.1
            starts.append(sig)
        
        # Chirp
        t = np.linspace(0, 1, n)
        freq_chirp = 2 + 8 * t
        phase = 2 * np.pi * np.cumsum(freq_chirp) / n
        sig = np.abs(np.cos(phase)) + 0.05
        starts.append(sig)
        
        # Gaussian window * carrier
        center = n / 2.0
        for width in [0.2, 0.35, 0.5]:
            window = np.exp(-0.5 * ((i - center) / (n * width)) ** 2)
            for freq in [2, 3, 5]:
                sig = window * (0.5 + 0.5 * np.cos(2 * np.pi * freq * i / n)) + 0.03
                starts.append(sig)
        
        # Random with structure
        sig = 0.3 + 0.4 * rng.random(n)
        sig += 0.2 * np.cos(2 * np.pi * 3.0 * i / n)
        sig = np.maximum(sig, 0.01)
        starts.append(sig)
        
        # Pure random
        starts.append(0.25 + 0.5 * rng.random(n))
        
        # Tooth-like (periodic with varying duty cycle)
        period = max(2, n // 20)
        for duty in [0.3, 0.5, 0.7]:
            pattern = (np.arange(period) < int(duty * period)).astype(float)
            sig = np.tile(pattern, n // period + 1)[:n] + 0.05
            starts.append(sig)
    
    return starts


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Explores multiple starting sequences of various lengths
    using LP-based peak reduction, Nesterov gradient, and line search blending.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    n_list = [512, 768, 1024, 1536]
    
    print(f"Budget: {budget_s}s, deadline in {deadline - t_start:.0f}s")
    
    # Generate starts
    starts = generate_starts(rng, n_list)
    
    # Sort by length (shorter first for quick results)
    starts.sort(key=len)
    
    print(f"Generated {len(starts)} starting sequences")
    
    # Deduplicate (by first few values to avoid redundant work)
    unique_starts = []
    seen = set()
    for s in starts:
        key = tuple(s[:10].round(3))
        if key not in seen:
            seen.add(key)
            unique_starts.append(s)
    starts = unique_starts
    print(f"After dedup: {len(starts)} starts")
    
    # Phase A: Quick pass on shorter sequences to get a good initial bound
    quick_deadline = t_start + 200.0
    for idx, a0 in enumerate(starts):
        if time.monotonic() > quick_deadline:
            break
        if len(a0) > 1024:
            continue
        t0 = time.monotonic()
        seq, score = optimize_one(a0, quick_deadline, max_grad=1500, max_lp=100)
        elapsed = time.monotonic() - t0
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Quick {idx}: n={len(seq)}, score={best_score:.6f} ({elapsed:.1f}s)")
    
    print(f"After quick phase: best={best_score:.6f}")
    
    # Phase B: Deeper optimization on more promising starts
    # Focus on starts that are similar to the current best (if any) and on longer sequences
    deep_deadline = deadline - 60.0
    for idx, a0 in enumerate(starts):
        if time.monotonic() > deep_deadline:
            break
        t0 = time.monotonic()
        seq, score = optimize_one(a0, deep_deadline, max_grad=3000, max_lp=300)
        elapsed = time.monotonic() - t0
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Deep {idx}: n={len(seq)}, score={best_score:.6f} ({elapsed:.1f}s)")
    
    print(f"After deep phase: best={best_score:.6f}")
    
    # Phase C: Final refinement of the best sequence
    if best_seq is not None and time.monotonic() < deadline - 10.0:
        print(f"Final refinement (score={best_score:.6f})...")
        for pass_i in range(5):
            if time.monotonic() > deadline - 5.0:
                break
            seq, score = optimize_one(best_seq, deadline - 5.0,
                                      max_grad=1000, max_lp=200)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Refine {pass_i}: score={best_score:.6f}")
            else:
                break
    
    # Phase D: Try to improve by adjusting length
    if best_seq is not None and time.monotonic() < deadline - 30.0:
        best_n = len(best_seq)
        for trial_n in [best_n + 128, best_n - 128, best_n + 256, best_n * 2]:
            if trial_n < 64 or trial_n > 4096:
                continue
            if time.monotonic() > deadline - 10.0:
                break
            # Resample the best sequence to new length
            old_i = np.linspace(0, 1, best_n)
            new_i = np.linspace(0, 1, trial_n)
            a_resampled = np.interp(new_i, old_i, best_seq)
            a_resampled = np.maximum(a_resampled, 0.0)
            
            seq, score = optimize_one(a_resampled, deadline - 5.0,
                                      max_grad=1000, max_lp=150)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Length change to {trial_n}: score={best_score:.6f}")
    
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"FINAL: n={len(best_seq)}, score={final_score:.6f}")
        return [float(x) for x in best_seq]
    
    # Fallback
    print("Falling back to uniform 512.")
    return [1.0] * 512
