"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Uses soft-max smoothing of the max, L-BFGS optimization in softmax
parameterization (a = S*softmax(b)), temperature annealing, and
multiple initializations. Final polish with exact max subgradient.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_from_b(b, S):
    """Compute a = S * softmax(b)."""
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted)
    a = S * exp_b / exp_b.sum()
    return a


def softmax_obj_and_grad(b, t):
    """Objective and gradient for soft-max of conv(a,a).
    
    a = S * softmax(b), S = sqrt(2n).
    Returns (objective, gradient w.r.t. b).
    """
    n = len(b)
    S = np.sqrt(2.0 * n)
    a = softmax_from_b(b, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    # Numerically stable soft-max
    tc = t * (c - c_max)
    exp_tc = np.exp(tc)
    Z = exp_tc.sum()
    obj = c_max + np.log(Z) / t
    w = exp_tc / Z  # soft-max weights
    
    # Gradient of F w.r.t. a:
    # grad_a[m] = 2 * sum_k w[k] * a[k-m] (with a[j]=0 for invalid j)
    # = 2 * cross_correlation(w, a)[m]
    # Using FFT: cross_corr(w,a)[m] = real(ifft(conj(fft(a,N)) * fft(w,N)))[m]
    N = 1 << (3 * n - 2 - 1).bit_length()
    A_fft = fft(a, N)
    W_fft = fft(w, N)
    grad_a = 2.0 * np.real(ifft(W_fft * np.conj(A.fft if hasattr(A, 'fft') else A_fft)))[:n]
    
    # Gradient w.r.t. b:
    # db[j] = a[j] * (grad_a[j] - (1/S) * sum_i grad_a[i]*a[i])
    ga_dot_a = np.dot(grad_a, a)
    grad_b = a * (grad_a - (ga_dot_a / S))

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Uses soft-max smoothing + L-BFGS in softmax parameterization,
temperature annealing, multiple initializations, and final polish.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_from_b(b, S):
    """a = S * softmax(b)."""
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted)
    return S * exp_b / exp_b.sum()


def obj_grad_soft(b, t):
    """Soft-max objective and gradient w.r.t. b.
    
    a = S*softmax(b), S = sqrt(2n).
    F = (1/t)*log(sum_k exp(t*c[k])) where c = conv(a,a).
    """
    n = len(b)
    S = np.sqrt(2.0 * n)
    a = softmax_from_b(b, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    tc = t * (c - c_max)
    exp_tc = np.exp(tc)
    Z = exp_tc.sum()
    obj = c_max + np.log(Z) / t
    w = exp_tc / Z
    
    # grad_a[m] = 2 * sum_k w[k] * a[k-m] (a[j]=0 for invalid j)
    # = 2 * (w cross- corr a)[m]
    N = 1 << (3 * n - 3).bit_length()
    W_N = fft(w, N)
    A_N = fft(a, N)
    grad_a = 2.0 * np.real(ifft(np.conj(A_N) * W_N))[:n]
    
    # grad_b[j] = a[j] * (grad_a[j] - dot(grad_a, a)/S)
    ga_a = np.dot(grad_a, a)
    grad_b = a * (grad_a - ga_a / S)
    
    return obj, grad_b


def run_anneal(b_init, n, max_time, rng, t_steps=None):
    """L-BFGS with temperature annealing."""
    if t_steps is None:
        t_steps = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1000.0]
    
    S = np.sqrt(2.0 * n)
    b = b_init.copy()
    best_obj = np.inf
    best_b = b.copy()
    start = time.monotonic()
    
    for t in t_steps:
        if time.monotonic() - start > max_time - 2.0:
            break
        
        def fun(b_in):
            return obj_grad_soft(b_in, t)
        
        # Limit time per stage
        stage_time = max(1.0, (max_time - (time.monotonic() - start)) / 
                        max(1, len(t_steps) - t_steps.index(t)))
        max_iter = min(300, max(50, int(stage_time * 1000 / 0.5)))
        
        result = minimize(
            fun, b,
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-6, 'maxls': 30}
        )
        
        b = result.x
        a = softmax_from_b(b, S)
        c = fft_convolve_self(a)
        cur_obj = c.max()  # Since S^2 = 2n, objective = max_c

def polish_exact(a_init, max_time, rng):
    """Polish using exact max with projected subgradient."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    
    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    lr = 0.1 * S / n
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        cur_obj = cur  # normalized
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Top-k subgradient
        k = max(3, n // 512)
        top_idx = np.argsort(c)[::-1][:k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-15:
            break
        weights = top_vals / total
        
        grad = np.zeros(n)
        for w_val, lag in zip(weights, top_idx):
            lag = int(lag)
            # grad[m] += 2 * w_val * a[lag - m] for valid m
            lo = max(0, lag - n + 1)
            hi = min(n - 1, lag)
            if lo <= hi:
                grad[lo:hi+1] += 2.0 * w_val * a[lag - hi:lag - lo + 1][::-1]
        
        # Projected gradient step on {a >= 0, sum = S}
        a_new = a - lr * grad
        # Project onto simplex
        a_new = project_simplex(a_new, S)
        
        a = a_new
        lr *= 0.999
    
    return best_a, best_obj


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def compute_obj(a):
    """Compute 2*n*max_c/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s ** 2)
