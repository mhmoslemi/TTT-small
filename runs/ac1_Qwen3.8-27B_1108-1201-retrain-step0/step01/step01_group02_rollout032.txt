```python
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def self_convolve(a):
    """Compute a * a using FFT."""
    n = len(a)
    size = 1
    while size < 2*n-1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]
```

Actually, I want to think more carefully about the subgradient method. The issue is that the max function makes the subgradient discontinuous. When the peak "jumps" from one position to another, the subgradient changes abruptly.

To handle this, I can use a "smooth approximation" of the max:
max_j b_j ≈ (1/beta) * log(sum_j exp(beta * b_j))  (log-sum-exp)

The gradient of this smooth approximation is:
grad_k = (1/beta) * sum_j (exp(beta*b_j) / sum_l exp(beta*b_l)) * 2*a_{j-k}
= sum_j softmax(beta*b)_j * 2*a_{j-k}

This is a weighted average of the "peak subgradients" with weights given by the softmax. As beta → ∞, this approaches the true subgradient at the argmax.

Using a moderate beta (e.g., beta = 50 or 100) gives a smooth gradient that's close to the true subgradient but better-behaved for optimization.

This is essentially the "entropic regularization" or "Fisher information" approach.

```python
def compute_subgrad(a, conv, num_peaks=10):
    """Compute subgradient using top peak positions."""
    n = len(a)
    max_val = conv.max()
    # Get top peaks
    top_indices = np.argsort(conv)[-num_peaks:][::-1]
    
    sg = np.zeros(n)
    for j in top_indices:
        # Add contribution from peak at position j
        # Contribution to sg_k is a_{j-k} for valid k
        lo = max(0, j - (n-1))
        hi = min(n-1, j)
        # a_{j-k} for k=lo,...,hi is a[j-hi : j-lo+1] reversed
        sg[lo:hi+1] += 2.0 * a[j-hi:j-lo+1][::-1]
    
    return sg
```

The LP solution gives the "best" direction to move. The difference g - a (where g is the LP solution normalized) is the descent direction.

```python
"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines FFT-accelerated subgradient descent, LP-based refinement,
and multi-length search with diverse initializations.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def self_convolve(a):
    """Compute a * a (full linear convolution) using FFT for efficiency."""
    n = len(a)
    if n <= 200:
        return np.convolve(a, a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.real(np.fft.irfft(fa * fa, size)[:2 * n - 1])


def fft_convolve(a, b):
    """Compute a * b (full linear convolution) using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.real(np.fft.irfft(fa * fb, size)[:n])


def score_seq(a):
    """Compute the score for a sequence."""
    n = len(a)
    conv = self_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    """Normalize to sum = 1 and non-negative."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def project_simplex(z):
    """Project z onto {x >= 0, sum(x) = 1} using binary search."""
    n = len(z)
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(z - mid, 0.0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    return np.maximum(z - mid, 0.0)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def subgrad_descent_fft(a, time_limit=300.0, verbose=False, beta_peaks=10):
    """
    Nesterov accelerated projected subgradient descent with FFT convolution.
    """
    start = time.monotonic()
    a = np.maximum(0, a.astype(np.float64)).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s  # normalize to sum = 1
    
    best_a = a.copy()
    best_score = score_seq(a)
    
    if verbose:
        print(f"      SG start: score={best_score:.6f}, n={n}")
    
    # Nesterov parameters
    a_prev = a.copy()
    t_k = 1.0
    base_step = 0.1  # will be scaled
    
    it = 0
    stagnant = 0
    while time.monotonic() - start < time_limit:
        t_k = 1.0 + np.sqrt(1.0 + 4.0 * it)
        t_next = 1.0 + np.sqrt(1.0 + 4.0 * (it + 1))
        
        # Nesterov momentum
        y = a + (t_k - 1.0) / t_next * (a - a_prev)
        y = np.maximum(0, y)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        # Compute convolution and subgradient
        conv = self_convolve(y)
        max_val = conv.max()
        
        # Find peak positions
        if beta_peaks >= conv.size:
            peak_indices = np.arange(conv.size)
        else:
            peak_indices = np.argsort(conv)[-beta_peaks:][::-1]
        
        # Compute subgradient
        sg = np.zeros(n)
        for j in peak_indices:
            lo = max(0, j - (n - 1))
            hi = min(n - 1, j)
            sg[lo:hi + 1] += 2.0 * y[j - hi:j - lo + 1][::-1]
        
        # Normalize subgradient for step size
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        
        # Adaptive step size
        alpha = base_step / (t_k * sg_norm * np.sqrt(n))
        
        a_new = y - alpha * sg
        a_new = project_simplex(a_new)
        
        a_prev = a.copy()
        a = a_new
        it += 1
        
        # Check score periodically
        if it % 20 == 0 or it < 5:
            sc = score_seq(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stagnant = 0
                if verbose and (it % 200 == 0 or it < 10):
                    print(f"      SG iter {it:5d}: score={sc:.6f}")
            else:
                stagnant += 1
        
        if stagnant > 50:
            # Try larger step or restart
            base_step *= 1.5
            stagnant = 0
            if base_step > 10.0:
                break
        
        # Decay step size slowly
        if it > 1000:
            base_step = max(0.01, base_step * 0.999)
    
    # Final check
    final_score = score_seq(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"      SG done: best={best_score:.6f} after {it} iters, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def build_conv_matrix_structure(n):
    """Build the sparsity pattern for the convolution matrix."""
    n_rows = 2 * n - 1
    # For each column k (variable b_k), it appears in rows k, k+1, ..., k+n-1
    # with values a_0, a_1, ..., a_{n-1}
    # Actually, (a*b)_j = sum_k a_{j-k} * b_k for valid j-k
    # So column k of the matrix has a_{j-k} in row j, for j = k, k+1, ..., k+n-1
    col_idx = []
    row_idx = []
    for k in range(n):
        for j in range(k, k + n):
            row_idx.append(j)
            col_idx.append(k)
    return np.array(row_idx, dtype=np.int32), np.array(col_idx, dtype=np.int32)


def lp_refine(a, time_limit_s=60.0, max_iters=100, verbose=False):
    """LP-based refinement step."""
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)
    
    if verbose:
        print(f"      LP start: score={best_score:.6f}, n={n}")
    
    # Precompute sparsity pattern
    row_idx, col_idx = build_conv_matrix_structure(n)
    n_rows = 2 * n - 1
    
    stagnant = 0
    it = 0
    while it < max_iters and time.monotonic() - start_time < time_limit_s:
        conv = self_convolve(a)
        M = conv.max()
        score = 2.0 * n * M  # since sum(a) = 1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose:
                print(f"      LP iter {it:3d}: score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 20:
            break
        
        # Build LP: max sum(b) s.t. (a*b)_j <= M, b >= 0
        # A_ub * b <= b_ub, minimize -sum(b)
        data = np.tile(a, n).copy()
        # Wait, the data depends on a which changes each iteration
        A = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))
        
        c = -np.ones(n)
        b_ub = np.full(n_rows, M)
        bounds = [(0.0, None)] * n
        
        remaining = time_limit_s - (time.monotonic() - start_time)
        if remaining < 2.0:
            break
        
        timeout = max(1.0, min(10.0, remaining * 0.3))
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': timeout})
            if res.success and res.x is not None:
                g = np.maximum(0.0, res.x)
                sum_g = g.sum()
                if sum_g > 1.0 + 1e-10:
                    g = g / sum_g
                    # Line search
                    for alpha in [0.8, 0.6, 0.4, 0.2, 0.1]:
                        a_new = (1.0 - alpha) * a + alpha * g
                        a_new = normalize(a_new)
                        if score_seq(a_new) <= score + 1e-12:
                            a = a_new
                            break
                    else:
                        break
                else:
                    break
            else:
                break
        except Exception:
            break
        
        it += 1
    
    if verbose:
        print(f"      LP done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search(a, time_limit_s=60.0, rng=None, verbose=False):
    """Random local search with smooth perturbations."""
    if rng is None:
        rng = np.random.default_rng(12345)
    start = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)
    
    if verbose:
        print(f"      LS start: score={best_score:.6f}")
    
    accept_count = 0
    while time.monotonic() - start < time_limit_s:
        # Random smooth perturbation
        sigma = min(0.1, 0.5 * (1.0 - (time.monotonic() - start) / time_limit_s))
        candidate = a * np.exp(rng.normal(0.0, sigma, size=n))
        candidate = normalize(candidate)
```

```python
"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines FFT-accelerated subgradient descent, LP-based refinement,
multi-length search, and random-restart local search.
Key improvements over previous approach:
- FFT-based convolution for large n (O(n log n) vs O(n^2))
- Nesterov-accelerated subgradient with adaptive step size
- More aggressive LP refinement with tighter per-iteration time limits
- Random restarts to escape local minima
- Better time allocation across search phases

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def self_convolve(a):
    """Compute a * a (full linear convolution) efficiently."""
    n = len(a)
    if n <= 300:
        return np.convolve(a, a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    return result


def score_seq(a):
    """Compute the score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    conv = self_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    """Normalize to non-negative with sum = 1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def project_simplex(z):
    """Project z onto {x >= 0, sum(x) = 1}."""
    n = len(z)
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(z - mid, 0.0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    return np.maximum(z - mid, 0.0)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2.0 * np.pi * x) + 0.08 * np.cos(4.0 * np.pi * x))


def initial_flat_top(n, frac=0.6):
    """Flat-top window: ones in the middle, tapers at edges."""
    a = np.zeros(n)
    mid = n // 2
    half_width = int(n * frac / 2)
    a[mid - half_width:mid + half_width] = 1.0
    # Smooth tapers
    taper = int(n * 0.15)
    if taper > 0:
        t = np.arange(taper)
        left_taper = 0.5 * (1 + np.cos(np.pi * t / taper))
        a[mid - half_width - taper:mid - half_width] = left_taper
        a[mid + half_width:mid + half_width + taper] = left_taper
    return np.maximum(0, a)


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def subgrad_descent(a, time_limit=100.0, verbose=False, seed=0):
    """
    Nesterov-accelerated projected subgradient descent with FFT convolution.
    Minimizes 2*n*max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    start = time.monotonic()
    a = np.maximum(0, a.astype(np.float64)).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = score_seq(a)
    if verbose:
        print(f"      SG start: score={best_score:.6f}, n={n}")

    a_prev = a.copy()
    t_k = 1.0
    base_step = 0.5
    it = 0
    stagnant = 0
    num_peak_sel = max(5, min(20, n // 20))

    while time.monotonic() - start < time_limit:
        t_k = 1.0 + np.sqrt(1.0 + 4.0 * it)
        t_next = 1.0 + np.sqrt(1.0 + 4.0 * (it + 1))

        # Nesterov momentum step
        y = a + ((t_k - 1.0) / t_next) * (a - a_prev)
        y = np.maximum(0, y)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a.copy()

        # Compute convolution and subgradient
        conv = self_convolve(y)
        max_val = conv.max()

        # Select peak positions (top few for smoothing)
        if n <= 100:
            peak_indices = np.argsort(conv)[-num_peak_sel:][::-1]
        else:
            # Use a threshold-based approach for speed
            thresh = max_val - 0.02 * max_val
            peak_mask = conv >= thresh
            peak_indices = np.where(peak_mask)[0]
            if len(peak_indices) > 50:
                peak_indices = np.argsort(conv)[-50:][::-1]

        # Compute subgradient: sg_k = 2 * sum_{j in P} y_{j-k}
        sg = np.zeros(n)
        for j in peak_indices:
            lo = max(0, j - (n - 1))
            hi = min(n - 1, j)
            if hi >= lo:
                sg[lo:hi + 1] += 2.0 * y[lo:j + 1 - (hi - lo) + 1 - (hi - lo) + 1][::-1]
                # Simplified: y[j-k] for k in [lo,hi] is y[j-hi : j-lo+1] reversed
                sg[lo:hi + 1] = 0  # reset (will recompute below)
        
        # Recompute subgradient correctly
        sg = np.zeros(n)
        for j in peak_indices:
            lo = max(0, j - (n - 1))
            hi = min(n - 1, j)
            if hi >= lo:
                # y[j-k] for k=lo,...,hi gives indices y[j-hi], y[j-hi+1], ..., y[j-lo]
                # which is y[j-hi : j-lo+1] in order (not reversed)
                # Wait: when k=lo, j-k = j-lo. When k=hi, j-k = j-hi.
                # So sg[lo] += 2*y[j-lo], sg[lo+1] += 2*y[j-lo-1], ..., sg[hi] += 2*y[j-hi]
                # This is y[j-hi : j-lo+1] reversed
                sg[lo:hi + 1] += 2.0 * y[j - hi:j - lo + 1][::-1]

        # Step size
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        alpha = base_step / (t_k * sg_norm) * np.sqrt(n)

        a_new = y - alpha * sg
        a_new = project_simplex(a_new)

        a_prev = a.copy()
        a = a_new
        it += 1

        # Evaluate score
        if it % 10 == 0 or it < 3:
            sc = score_seq(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
                stagnant = 0
                if verbose and (it % 100 == 0 or it < 5):
                    print(f"      SG iter {it:5d}: score={sc:.6f}")
            else:
                stagnant += 1

        # Adapt step size
        if stagnant > 30:
            if base_step < 5.0:
                base_step *= 1.3
            else:
                # Try a random restart
                rng_local = np.random.default_rng(it)
                perturb = initial_raised_cosine(n) * 0.1
                a = normalize(a + perturb * rng_local.standard_normal(0))
                a_prev = a.copy()
                base_step = 0.5
            stagnant = 0

        # Slow decay
        if it % 2000 == 0:
            base_step = max(0.05, base_step * 0.9)

    final_score = score_seq(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"      SG done: best={best_score:.6f} after {it} iters, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def lp_refine(a, time_limit_s=120.0, max_iters=200, verbose=False):
    """LP-based iterative refinement."""
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)

    if verbose:
        print(f"      LP start: score={best_score:.6f}, n={n}")

    # Precompute sparsity pattern
    n_rows = 2 * n - 1
    col_idx_list = []
    row_idx_list = []
    for k in range(n):
        for j in range(k, min(k + n, n_rows)):
            row_idx_list.append(j)
            col_idx_list.append(k)
    row_idx = np.array(row_idx_list, dtype=np.int32)
    col_idx = np.array(col_idx_list, dtype=np.int32)

    stagnant = 0
    it = 0
    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    while it < max_iters:
        elapsed = time.monotonic() - start_time
        remaining = time_limit_s - elapsed
        if remaining < 3.0:
            break

        conv = self_convolve(a)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 10 == 0 or it < 3):
                print(f"      LP iter {it:3d}: score={score:.6f}")
        else:
            stagnant += 1

        if stagnant > 30:
            if verbose:
                print(f"      LP stagnant at iter {it}")
            break

        # Build LP
        data = np.tile(a, n).astype(np.float64)
        A = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))
        b_ub = np.full(n_rows, M, dtype=np.float64)

        timeout = max(1.0, min(15.0, remaining * 0.25))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': timeout})
            if res.success and res.x is not None:
                g = np.maximum(0.0, res.x)
                sum_g = g.sum()
                if sum_g > 1.0 + 1e-10:
                    g = g / sum_g
                    # Backtracking line search
                    improved = False
                    for alpha in [0.9, 0.7, 0.5, 0.3, 0.15, 0.05]:
                        a_new = (1.0 - alpha) * a + alpha * g
                        a_new = normalize(a_new)
                        new_score = score_seq(a_new)
                        if new_score <= score + 1e-13:
                            a = a_new
                            improved = True
                            break
                    if not improved:
                        break
                else:
                    if verbose:
                        print(f"      LP: no improvement direction found")
                    break
            else:
                break
        except Exception:
            break

        it += 1

    if verbose:
        print(f"      LP done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search(a, time_limit_s=60.0, rng=None, verbose=False):
    """Random local search with smooth multiplicative perturbations."""
    if rng is None:
        rng = np.random.default_rng(999)
    start = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)

    if verbose:
        print(f"      LS start: score={best_score:.6f}")

    accept_count = 0
    while time.monotonic() - start < time_limit_s:
        frac_elapsed = (time.monotonic() - start) / time_limit_s
        sigma = 0.15 * (1.0 - frac_elapsed) + 0.02

        # Multiplicative smooth perturbation
        noise = rng.normal(0.0, sigma, size=n)
        # Smooth the noise slightly
        if n > 10:
            kernel_size = max(1, n // 100)
            if kernel_size > 1:
                kernel = np.ones(kernel_size) / kernel_size
                noise = np.convolve(noise, kernel, mode='same')
        candidate = a * np.exp(noise)
        candidate = normalize(candidate)

        score = score_seq(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 30 == 0:
                print(f"      LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"      LS done: best={best_score:.6f}, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def random_restart_improve(a_best, score_best, n, rng, time_limit=60.0, verbose=False):
    """Try random perturbations of the best solution and optimize briefly."""
    start = time.monotonic()
    best_a = a_best.copy()
    best_score = score_best

    for _ in range(10):
        remaining = time_limit - (time.monotonic() - start)
        if remaining < 5.0:
            break

        # Create a perturbed version
        frac = (time.monotonic() - start) / time_limit
        sigma = 0.1 * (1.0 - frac) + 0.02
        perturbation = initial_raised_cosine(n) * rng.standard_normal(0)
        # Actually use a random smooth perturbation
        t = np.arange(n, dtype=np.float64)
        n_modes = 3
        pert = np.zeros(n)
        for m in range(1, n_modes + 1):
            pert += rng.normal(0, 1) * np.cos(2 * np.pi * m * t / n + rng.uniform(0, 2 * np.pi))
        candidate = normalize(a_best + sigma * pert)

        # Brief subgradient optimization
        brief_time = min(5.0, remaining * 0.3)
        a_opt, score_opt = subgrad_descent(candidate, time_limit=brief_time, verbose=False)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"      Restart improved: {score_opt:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-strategy search for optimal non-negative sequence minimizing
    2*n*max(conv(a,a))/sum(a)^2.

    Strategy:
    Phase 1: Quick exploration with diverse initializations and lengths
    Phase 2: Intensive LP refinement on top candidates
    Phase 3: Alternating subgradient + LP with random restarts
    Phase 4: Final polishing
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(60.0, budget_s - 25.0)

    best_score = float('inf')
    best_seq = None

    # Load parent if available
    parent = None
    if 'height_sequence_1' in globals():
        try:
            parent = np.array(height_sequence_1, dtype=np.float64)
            parent = np.maximum(0, parent)
            if parent.sum() < 0.01:
                parent = None
        except Exception:
            parent = None

    def try_candidate(a, time_alloc, label=""):
        nonlocal best_score, best_seq
        a = normalize(a)
        sc = score_seq(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            if verbose_global:
                print(f"  [{label}] Init: {sc:.6f}")

        # Subgradient descent
        sg_time = time_alloc * 0.35
        a_sg, sg_score = subgrad_descent(a, time_limit=sg_time, verbose=verbose_global)
        if sg_score < best_score:
            best_score = sg_score
            best_seq = a_sg.copy()
            if verbose_global:
                print(f"  [{label}] SG: {sg_score:.6f}")

        # LP refinement
        lp_time = time_alloc * 0.45
        a_lp, lp_score = lp_refine(a_sg if sg_score < sc else a, time_limit_s=lp_time,
                                   verbose=verbose_global)
        if lp_score < best_score:
            best_score = lp_score
            best_seq = a_lp.copy()
            if verbose_global:
                print(f"  [{label}] LP: {lp_score:.6f}")

        # Local search
        ls_time = time_alloc * 0.2
        a_ls, ls_score = local_search(
            a_lp if lp_score < best_score else a_sg,
            time_limit_s=ls_time, rng=rng)
        if ls_score < best_score:
            best_score = ls_score
            best_seq = a_ls.copy()
            if verbose_global:
                print(f"  [{label}] LS: {ls_score:.6f}")

    verbose_global = True

    # ============ Phase 1: Exploration ============
    print("=" * 60)
    print("PHASE 1: Exploration")
    print("=" * 60)

    configs = []
    if parent is not None:
        configs.append((len(parent), 'parent', 0.10))

    configs += [
        (500, 'raised_cosine', 0.08),
        (500, 'gaussian', 0.05),
        (500, 'flat_top', 0.04),
        (700, 'raised_cosine', 0.06),
        (800, 'gaussian', 0.05),
        (1000, 'raised_cosine', 0.07),
        (1000, 'random', 0.04),
        (1200, 'raised_cosine', 0.06),
        (1500, 'raised_cosine', 0.06),
        (2000, 'raised_cosine', 0.06),
        (2000, 'random', 0.04),
        (3000, 'raised_cosine', 0.05),
    ]

    for n, init_type, time_frac in configs:
        remaining = deadline - time.monotonic()
        if remaining < 20.0:
            break
        time_alloc = min(time_frac * budget_s, remaining * 0.6)
        if time_alloc < 10.0:
            continue

        print(f"\n  --- n={n}, init={init_type}, alloc={time_alloc:.0f}s ---")

        if init_type == 'parent' and parent is not None:
            if len(parent) == n:
                a = parent.copy()
            else:
                x_old = np.linspace(0, 1, len(parent))
                x_new = np.linspace(0, 1, n)
                a = np.interp(x_new, x_old, parent)
        elif init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif init_type == 'gaussian':
            a = initial_gaussian(n, sigma_frac=0.25)
        elif init_type == 'flat_top':
            a = initial_flat_top(n, frac=0.7)
        elif init_type == 'random':
            a = random_smooth_init(n, rng, n_bumps=int(rng.integers(2, 6)))
        elif init_type == 'blackman':
            a = initial_blackman(n)
        else:
            a = initial_raised_cosine(n)

        try_candidate(a, time_alloc, f"n={n}/{init_type}")

    # ============ Phase 2: Intensive LP on best ============
    print("\n" + "=" * 60)
    print("PHASE 2: Intensive LP Refinement")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            lp_time = min(180.0, remaining * 0.5)
            print(f"\n  LP intensive: time={lp_time:.0f}s")
            a_opt, score_opt = lp_refine(best_seq, time_limit_s=lp_time,
                                         max_iters=300, verbose=True)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"  *** New best: {best_score:.6f}")

    # ============ Phase 3: Random restarts + optimize ============
    print("\n" + "=" * 60)
    print("PHASE 3: Random Restarts")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            n_best = len(best_seq)
            restart_time = min(120.0, remaining * 0.4)
            print(f"\n  Random restarts on n={n_best}: time={restart_time:.0f}s")
            a_opt, score_opt = random_restart_improve(
                best_seq, best_score, n_best, rng, time_limit=restart_time, verbose=True)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"  *** New best after restarts: {best_score:.6f}")

    # ============ Phase 4: Final polishing ============
    print("\n" + "=" * 60)
    print("PHASE 4: Final Polishing")
    print("=" * 60)

    if best_seq is not None:
        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 20.0:
            t = min(60.0, remaining * 0.4)
            a_opt, score_opt = lp_refine(best_seq, time_limit_s=t, verbose=True)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"  *** After LP polish: {best_score:.6f}")

        # Subgradient polish
        remaining = deadline - time.monotonic()
        if remaining > 15.0:
            t = min(40.0, remaining * 0.4)
            a_opt, score_opt = subgrad_descent(best_seq, time_limit=t, verbose=True)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"  *** After SG polish: {best_score:.6f}")

        # Local search
        remaining = deadline - time.monotonic()
        if remaining > 10.0:
            t = min(30.0, remaining * 0.5)
            a_opt, score_opt = local_search(best_seq, time_limit_s=t, rng=rng, verbose=True)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"  *** After LS polish: {best_score:.6f}")

        # One more LP
        remaining = deadline - time.monotonic()
        if remaining > 15.0:
            t = min(30.0, remaining * 0.7)
            a_opt, score_opt = lp_refine(best_seq, time_limit_s=t, verbose=True)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"  *** Final LP: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_seq(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Final verification
    print(f"\n{'=' * 60}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified via evaluate_sequence: {verify_score:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
```
