import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score_fast(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def solve_lp_for_direction(f, M):
    """
    Solve: maximize sum(g) subject to:
      g >= 0
      (f * g)_k <= M for all k
    
    Returns the optimal g (before rescaling).
    """
    n = len(f)
    # (f * g)_k = sum_i f_i * g_{k-i} (with valid indices)
    # This is a linear function of g for each k
    # Build constraint matrix A_ub such that A_ub @ g <= M * ones(2n-1)
    
    # For each lag k (0 to 2n-2):
    # (f*g)_k = sum_{i: 0<=i<n, 0<=k-i<n} f_i * g_{k-i}
    # Let j = k-i, so i = k-j
    # (f*g)_k = sum_{j: 0<=j<n, 0<=k-j<n} f_{k-j} * g_j
    
    # We need to build a (2n-1) x n matrix A where A[k,j] = f_{k-j} if 0<=k-j<n else 0
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return None

"""
LP-based iterative optimization for minimizing the peak of linear autocorrelation
of non-negative step functions. Combines LP direction-finding with gradient steps
and local search polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score_fast(a):
    """Compute the score without calling evaluate_sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def build_conv_matrix(f):
    """Build the matrix A where (A @ g)[k] = (f*g)[k]."""
    n = len(f)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max+1] = f[k - j_max : k - j_min - 1 : -1]
    return A

def solve_lp_direction(f, M):
    """
    Solve: maximize sum(g) subject to g >= 0 and (f*g)_k <= M for all k.
    Returns optimal g or None if infeasible.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_gradient_optimize(a_init, time_limit, rng, verbose=True):
    """
    Main optimization loop combining LP directions and gradient steps.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.01, None)
    S = np.sum(a)
    
    best_a = a.copy()
    best_score = local_score_fast(a)
    
    t0 = time.monotonic()
    iteration = 0
    blend_t = 0.3
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        M = np.max(c)
        peak_idx = int(np.argmax(c))
        current_score = 2.0 * n * M / (S ** 2)
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Try LP direction
        g = solve_lp_direction(a, M)
        if g is not None and np.sum(g) > 1e-10:
            g_rescaled = g * S / np.sum(g)
            # Blend
            a_new = (1 - blend_t) * a + blend_t * g_rescaled
            a_new = np.clip(a_new, 0.01, None)
            
            new_score = local_score_fast(a_new)
            if new_score < current_score:
                a = a_new
                iteration += 1
            else:
                # Try smaller blend
                a_new2 = (1 - blend_t/2) * a + (blend_t/2) * g_rescaled
                a_new2 = np.clip(a_new2, 0.01, None)
                new_score2 = local_score_fast(a_new2)
                if new_score2 < current_score:
                    a = a_new2
                    iteration += 1
                else:
                    # Gradient step
                    a = gradient_step(a, peak_idx, S, 0.01, rng)
                    no_improve += 1
        else:
            # LP degenerate, use gradient step
            a = gradient_step(a, peak_idx, S, 0.01, rng)
        
        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={iteration}, score={local_score_fast(a):.6f}, best={best_score:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_score

def gradient_step(a, peak_idx, S, lr, rng):
    """
    One gradient descent step on the score, with non-negativity projection.
    """
    n = len(a)
    # Gradient: [∂score/∂a_j] = 2n/S^2 * 2*a[peak_idx - j] * 1[0<=peak_idx-j<n] - 4n*c_peak/S^3
    c_peak = np.dot(a, a)  # Wait, this is only correct if peak_idx = n-1 (symmetric case)
    # Actually, c_peak = (a*a)[peak_idx]
    # Let me compute it properly
    c = fft_convolve(a, a)
    c_peak = c[peak_idx]
    
    # Gradient component j: 2n/S^2 * 2 * a[peak_idx - j] (if valid) - 4n*c_peak/S^3
    grad = -4.0 * n * c_peak / (S ** 3) * np.ones(n)
    for j in range(n):
        k = peak_idx - j
        if 0 <= k < n:
            grad[j] += 4.0 * n * a[k] / (S ** 2)
    
    a_new = a - lr * grad
    a_new = np.clip(a_new, 0.01, None)
    return a_new

- $g_j \geq 0$ for all $j$
- $\sum g_j = S$
- $(f*g)_k - M' \leq 0$ for all $k$ (i.e., $A[k,:] @ g - M' \leq 0$)

In matrix form:
- Objective: $[0, 0, \ldots, 0, 1]$ (the last variable is $M'$)
- A_ub: $[A, -\mathbf{1}]$ (each row is $[A[k,:], -1]$)
- b_ub: $\mathbf{0}$ (zeros vector of length 2n-1)
- A_eq: $[\mathbf{1}, 0]$ (the sum constraint)
- b_eq: $[S]$
- bounds: $g_j \geq 0$ for $j=0,...,n-1$, $M'$ unbounded (or $M' \geq 0$)

After solving, we can use $g^*$ as the new direction: $f \leftarrow (1-t)f + t \cdot g^*$.

      g >= 0
      sum(g) = S
      (f*g)_k <= M' for all k

    A_conv = build_conv_matrix(f)  # (2n-1) x n

    # Objective: minimize M'
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_conv @ g - M' <= 0
    A_ub = np.hstack([A_conv, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)

    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0

        M_opt = res.x[-1]
        g_opt = res.x[:n]
        return M_opt, g_opt
    else:
        return np.inf, None

"""
LP-based iterative optimization for minimizing the peak of linear convolution
of non-negative step functions. Uses a parametric LP to find the minimum achievable
peak of (f*g) for a fixed f, combined with gradient steps and local search polishing.
Key insight: asymmetric sequences can achieve scores below 2 by making the
convolution flatter than the symmetric case allows.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score_fast(a):
    """Fast local computation of the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_conv_matrix(f):
    """Build matrix A where (A @ g)[k] = (f * g)[k] for linear convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            A[k, idx] = f[k - idx]
    return A


def solve_lp_min_peak(f, S):
    """
    Minimize M' subject to:
      g >= 0, sum(g) = S, (f*g)_k <= M' for all k = 0,...,2n-2.
    Returns (M_opt, g_opt) or (inf, None).
    """
    n = len(f)
    A_conv = build_conv_matrix(f)
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0

    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    bounds = [(0, None)] * n_vars

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return float(res.x[-1]), res.x[:n]
    return np.inf, None


def gradient_step(a, peak_idx, S, c_peak, lr):
    """One gradient descent step on the score at a known peak location."""
    n = len(a)
    grad = np.full(n, -4.0 * n * c_peak / (S ** 3))
    for j in range(n):
        k = peak_idx - j
        if 0 <= k < n:
            grad[j] += 4.0 * n * a[k] / (S ** 2)
    a_new = a - lr * grad
    return np.clip(a_new, 0.01, None)


def lp_optimize(a_init, time_limit, rng, verbose=True):
    """
    Main LP-based optimization loop.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.01, None)
    S = np.sqrt(2.0 * n)
    a = a * S / np.sum(a)

    best_a = a.copy()
    best_score = local_score_fast(a)
    t0 = time.monotonic()
    iteration = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        c = fft_convolve(a, a)
        M = float(np.max(c))
        peak_idx = int(np.argmax(c))
        current_score = 2.0 * n * M / (S ** 2)

        if current_score < best_score - 1e-10:
            best_score = current_score
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if verbose and iteration % 100 == 0 and best_score < np.inf:
            print(f"      iter={iteration}, score={current_score:.6f}, best={best_score:.6f}, M*={M:.4f}, t={elapsed:.1f}s")

        # Solve LP for minimum peak direction
        M_star, g = solve_lp_min_peak(a, S)

        if g is not None and M_star < M - 1e-6:
            # LP found improvement
            blend = 0.4 if stagnation < 10 else 0.2
            a_new = (1.0 - blend) * a + blend * g
            a_new = np.clip(a_new, 0.01, None)
            new_score = local_score_fast(a_new)
            if new_score < current_score:
                a = a_new
            else:
                # Try smaller blend
                a_new2 = (1.0 - blend * 0.3) * a + (blend * 0.3) * g
                a_new2 = np.clip(a_new2, 0.01, None)
                new_score2 = local_score_fast(a_new2)
                if new_score2 < current_score:
                    a = a_new2
                else:
                    a = gradient_step(a, peak_idx, S, c[peak_idx], 0.05 / n)
                    stagnation += 1
        else:
            # LP degenerate or no improvement - use gradient + perturbation
            lr = 0.1 / (1.0 + 0.1 * stagnation)
            a = gradient_step(a, peak_idx, S, c[peak_idx], lr)
            # Add small random perturbation to escape plateaus
            a += 0.005 * rng.standard_normal(n)
            a = np.clip(a, 0.01, None)
            # Renormalize
            a = a * S / np.sum(a)

        iteration += 1

    return best_a, best_score


def gradient_optimize(a_init, time_limit, rng):
    """Pure gradient-based optimization with adaptive step size."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.01, None)
    S = np.sqrt(2.0 * n)
    a = a * S / np.sum(a)

    best_a = a.copy()
    best_score = local_score_fast(a)
    t0 = time.monotonic()
    iteration = 0
    lr = 1.0 / n
    momentum = np.zeros(n)
    beta = 0.9

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        M = float(np.max(c))
        peak_idx = int(np.argmax(c))
        current_score = 2.0 * n * M / (S ** 2)

        if current_score < best_score - 1e-10:
            best_score = current_score
            best_a = a.copy()

        # Compute gradient
        grad = np.full(n, -4.0 * n * c[peak_idx] / (S ** 3))
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                grad[j] += 4.0 * n * a[k] / (S ** 2)

        momentum = beta * momentum + (1 - beta) * grad
        a_new = a - lr * momentum
        a_new = np.clip(a_new, 0.01, None)
        a_new = a_new * S / np.sum(a_new)
        a = a_new

        # Adaptive learning rate
        if iteration % 100 == 0:
            lr = max(0.01 / n, lr * 0.95)

        iteration += 1

    return best_a, best_score


def local_search_polish(a_init, time_limit, rng):
    """Targeted local search focusing on peak reduction."""
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    current_score = local_score_fast(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9995

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.4:
            # Target peak contributors
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                k = peak_idx - i
                if 0 <= k < n:
                    contribs[i] = current[i] * current[k]
            total_contrib = np.sum(contribs)
            if total_contrib > 0:
                # Reduce top contributors
                top_k = max(3, n // 12)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
                # Increase bottom contributors
                bottom_idx = np.argsort(contribs)[:top_k]
                candidate[bottom_idx] *= np.exp(rng.normal(0.05, 0.02, size=len(bottom_idx)))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        elif r < 0.75:
            # Shift mass from one region to another
            half = n // 2
            amt = rng.uniform(0.02, 0.08) * current.sum()
            src = rng.integers(0, half)
            dst = rng.integers(half, n)
            candidate[src] = max(0.01, candidate[src] - amt)
            candidate[dst] += amt
        elif r < 0.9:
            # Asymmetric perturbation (break symmetry)
            mask = rng.random(n) > 0.5
            candidate[mask] *= np.exp(rng.normal(0.0, 0.02, size=np.sum(mask)))
            candidate[~mask] *= np.exp(rng.normal(0.0, 0.02, size=np.sum(~mask)))
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.clip(candidate, 0.01, 1000.0)
        new_score = local_score_fast(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 1e-8)

    return best, best_score


def make_init_ramp(n, rng):
    """Linear ramp initialization (asymmetric)."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.5 * t / n
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_exp(n, rng, decay=1.0):
    """Exponential decay initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-decay * t / n)
    a /= np.mean(a)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_asym_random(n, rng):
    """Random asymmetric initialization."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    # Add a linear trend to break symmetry
    t = np.arange(n, dtype=np.float64)
    a += 0.2 * (t / n - 0.5)
    return np.clip(a, 0.01, None)


def make_init_sine(n, rng):
    """Sine-based asymmetric initialization."""
    t = np.arange(n, dtype=np.float64)
    phase = rng.uniform(0, 2 * np.pi)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * t / n + phase)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
