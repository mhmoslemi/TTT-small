"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based spectral flattening (POCS), smooth gradient descent on sum-of-squares,
peak subgradient minimization, and LP-based refinement.
Multiple starting points and lengths are explored.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def _fft_length(n):
    """Get next power of 2 >= 2n-1."""
    target = 2 * n - 1
    length = 1
    while length < target:
        length *= 2
    return length


def _convolve_fft(a):
    """Compute full convolution using FFT."""
    n = len(a)
    N = _fft_length(n)
    A = np.fft.rfft(a, N)
    C = A * A
    c = np.fft.irfft(C, N)[:2*n-1]
    return c


def _pocs_step(a, S, n_flat_ratio=0.95):
    """
    One POCS (Projection onto Convex Sets) step.
    Projects onto: (1) sum(a) = S, (2) a >= 0, (3) flat power spectrum.
    """
    n = len(a)
    N = _fft_length(n)
    
    # Frequency domain: flatten the spectrum
    A = np.fft.rfft(a, N)
    # Target magnitude: S/sqrt(N) for DC, and uniform for others
    # Actually, we want |A(f)|^2 to sum to S^2 (Parseval), and be as flat as possible
    # DC: |A(0)| = S
    # Others: |A(f)| = S * sqrt(1/(2N-1)) approximately (so that sum |A(f)|^2 = S^2)
    # But we have N/2+1 frequency bins (for rfft), and the total energy is S^2
    # For flatness: |A(f)|^2 = S^2 / (N/2 + 1) for all f (approximately)
    # But we need |A(0)| = S, so let's be more careful
    
    # Actually, let's just make all non-DC magnitudes equal:
    # sum_f |A(f)|^2 = S^2 (Parseval for the padded sequence)
    # |A(0)|^2 + sum_{f=1}^{N/2} |A(f)|^2 = S^2
    # We want |A(f)| = const for f >= 1
    # So: S^2 + (N/2) * |A(f)|^2 = S^2 => |A(f)| = 0? That's wrong.
    # 
    # Wait, Parseval says sum_x a[x]^2 = (1/N) sum_f |A(f)|^2.
    # And sum_f |A(f)|^2 = N * sum_x a[x]^2.
    # 
    # We're constraining sum_x a[x] = S, not sum_x a[x]^2.
    # |A(0)| = sum_x a[x] = S.
    # So |A(0)|^2 = S^2.
    # 
    # The remaining energy: sum_{f>=1} |A(f)|^2 = N*sum_x a[x]^2 - S^2.
    # 
    # For flatness, we want |A(f)| = const for f >= 1.
    # Let's set |A(f)| = sqrt((N*sum_x a[x]^2 - S^2) / (N/2)) for f >= 1.
    # But sum_x a[x]^2 depends on a, which we're trying to find...
    
    # Simplification: Just set all non-DC magnitudes to a fixed value.
    # We'll use the current a to estimate the target.
    a_energy = float(np.sum(a * a))
    target_non_dc = np.sqrt(max(N * a_energy - S * S, 0) / (N // 2))
    
    A_new = A.copy()
    A_new[0] = S  # Keep DC
    # For non-DC, set magnitude to target_non_dc, keep phase

        else:
            A_new[f] = target_non_dc * (1 + 1j) / np.sqrt(2)
    
    # Time domain
    a_new = np.fft.irfft(A_new, N)[:n]
    
    # Project onto non-negativity
    a_new = np.maximum(a_new, 0)
    
    # Normalize
    a_new = _normalize(a_new, S)
    
    return a_new


def _grad_sum_of_squares(a):
    """
    Compute gradient of sum_k c[k]^2 where c = convolve(a, a).
    grad[i] = 4 * sum_j a[j] * c[i+j] = 4 * correlate(c, a, 'valid')[i]
    """
    n = len(a)
    c = _convolve_fft(a)
    # grad[i] = sum_{j=0}^{n-1} a[j] * c[i+j]
    # This is correlate(c, a, 'valid')
    grad = 4.0 * np.correlate(c, a, mode='valid')
    return grad


def _grad_variance(a, S):
    """Compute gradient of sum_k (c[k] - c_mean)^2."""
    n = len(a)
    c = _convolve_fft(a)
    c_mean = S * S / (2 * n - 1)
    d = c - c_mean
    grad = 4.0 * np.correlate(d, a, mode='valid')
    return grad


def _grad_peak_soft(a, tau=50.0):
    """Compute gradient using soft-max (smooth approximation of peak)."""
    n = len(a)
    c = _convolve_fft(a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    # grad[i] = 2 * sum_j a[j] * w[i+j]
    # This is 2 * correlate(w, a, 'valid')
    # Wait, let me recheck.
    # d/d a[i] of sum_k w[k] * c[k] = sum_k w[k] * dc[k]/da[i] = sum_k w[k] * 2*a[k-i]
    # = 2 * sum_j a[j] * w[i+j] (where j = k-i, so k = i+j)
    # = 2 * correlate(w, a, 'valid')[i]
    grad = 2.0 * np.correlate(w, a, mode='valid')

def _optimize_peak(a0, deadline, max_iters=3000):

def _lp_solve(f, M):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M (elementwise), g >= 0."""
    n = len(f)
    if n > 1024:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine(a0, deadline, max_iters=50):
    """Phase 4: LP-based refinement."""
    n = len(a0)
    if n > 1024:
        return a0, _compute_obj(a0)
    
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t)
    
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        c = _convolve_fft(a)
        M = float(c.max())
        
        g = _lp_solve(a, M)
        if g is None:
            break
        
        improved = False
        for t in [0.8, 0.5, 0.3, 0.1]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, S_t)
            s_new = _compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break
        
        if not improved:
            break
    
    return best, best_score

def _full_pipeline(a0, deadline):
    """Run the full optimization pipeline on a single sequence."""
    n = len(a0)
    
    # Phase 1: POCS (fast, good for initialization)
    if time.monotonic() < deadline - 30.0:

    else:

        s1 = _compute_obj(a0)
    
    # Phase 2: Smooth gradient (sum of squares)
    if time.monotonic() < deadline - 20.0:

Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Strategy: FFT-based spectral analysis, smooth gradient descent on sum-of-squares,
soft-peak minimization, POCS spectral flattening, and LP refinement.
Explores multiple lengths and diverse initializations within the time budget.

import numpy as np

def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _obj_fft(a):
    """Faster objective using FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def _convolve_fft(a):
    """Full convolution via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    return c


def _grad_sos(a):
    """Gradient of sum_k c[k]^2 where c = convolve(a,a)."""
    n = len(a)
    c = _convolve_fft(a)
    grad = 4.0 * np.correlate(c, a, mode='valid')
    grad -= grad.mean()
    return grad


def _grad_soft_peak(a, tau):
    """Soft peak gradient using softmax weighting."""
    n = len(a)
    c = _convolve_fft(a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    grad = 2.0 * np.correlate(w, a, mode='valid')
    grad -= grad.mean()
    return grad

    return grad

        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:

            a_new = np.maximum(a_new, 0.0)
