```python
import numpy as np
import time
from scipy.optimize import linprog

def solve_direction_lp(f, M, n):
    """
    Solve the LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    f: current sequence (normalized to sum=1)
    M: peak value to enforce
    n: length of sequence
    Returns: optimal g (normalized to sum=1)
    """
    # c = -1 (we minimize -sum(g))
    c = -np.ones(n)
    
    # A_ub: (2n-1) x n matrix
    # A_ub[k, i] = f[i] if 0 <= k-i <= n-1, else 0
    # This is banded, so let's construct it efficiently
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A_ub[k, lo:hi+1] = f[lo:hi+1]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        s = np.sum(g)
        if s > 1e-10:
            return g / s
        else:
            return np.ones(n) / n
    else:
        return np.ones(n) / n
```

```python
def build_convolution_matrix(f, n):
    """Build the matrix A where A @ g = convolve(f, g) (first n elements... no, all 2n-1 elements)."""
    # Actually, A @ g should give the full convolution f*g
    # (f*g)[k] = sum_i f[i] * g[k-i] for valid i
    # In matrix form: (A @ g)[k] = sum_i A[k,i] * g[i]
    # So A[k,i] = f[i] if 0 <= k-i <= n-1, else 0
    
    # Column i: A[k,i] = f[i] for k in [i, i+n-1] (clipped to [0, 2n-2])
    rows = []
    cols = []
    vals = []
    for i in range(n):
        k_start = i
        k_end = min(i + n - 1, 2*n - 2)
        if k_start <= k_end:
            rows.extend(range(k_start, k_end + 1))
            cols.extend([i] * (k_end - k_start + 1))
            vals.extend([f[i]] * (k_end - k_start + 1))
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```

```python
def solve_lp_with_row_generation(f, M, n, K=20, max_rounds=50):
    """
    Solve: max sum(g) s.t. (f*g)_k <= M for active k, g >= 0.
    Uses row generation to add violated constraints.
    """
    b = np.convolve(f, f)
    # Initial active set: top K peaks
    active_k = np.argsort(b)[-K:]
    
    for round in range(max_rounds):
        # Solve LP with current active set
        A = np.zeros((len(active_k), n))
        for row, k in enumerate(active_k):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[row, lo:hi+1] = f[lo:hi+1]
        
        c = -np.ones(n)
        b_ub = np.full(len(active_k), M)
        bounds = [(0, None)] * n
        
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not result.success:
            break
        
        g = result.x
        s = np.sum(g)
        if s < 1e-10:
            break
        
        # Check for violated constraints
        fg = np.convolve(f, g)
        violations = np.where(fg > M * (1 + 1e-8))[0]
        violations = np.setdiff1d(violations, active_k)
        
        if len(violations) == 0:
            break
        
        # Add most violated constraints
        violation_amounts = fg[violations] - M
        worst = violations[np.argsort(violation_amounts)[-K:]]
        active_k = np.setdiff1d(active_k, active_k)  # reset
        active_k = np.union1d(active_k, worst)
        # Keep at most 2*K constraints
        if len(active_k) > 2 * K:
            # Remove least violated
            current_violations = fg[active_k]
            active_k = active_k[np.argsort(current_violations)[-2*K:]]
    
    if s > 1e-10:
        return g / s
    return np.ones(n) / n
```

`a_new[i] = a[i] * exp(-eta * grad[i]) / Z`

```python
Multi-phase optimization for minimizing normalized autocorrelation peak.
Uses LogSumExp-smoothed gradient with multiplicative updates, LP-based
direction finding, and local polishing to find near-optimal non-negative
sequences with minimal peak-to-average autocorrelation ratio.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.signal import fftconvolve

def _score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    b = fftconvolve(a, a)
    return 2.0 * n * np.max(b) / (s * s)

def _norm1(a):
    """Normalize a to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s

def _logsumexp_grad(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*b_k)) where b = conv(a,a).
    Returns (value, gradient).
    """
    n = len(a)
    b = fftconvolve(a, a)
    # Stable logsumexp
    b_max = np.max(b)
    z = tau * (b - b_max)
    # Clip for numerical stability
    z = np.clip(z, -500, 500)
    exp_z = np.exp(z)
    p = exp_z / np.sum(exp_z)
    value = b_max + np.log(np.sum(exp_z)) / tau
    # Gradient: 2 * correlate(p, a, mode='full')[:n]
    grad = 2.0 * np.correlate(p, a, mode='full')[:n]
    return value, grad
```
