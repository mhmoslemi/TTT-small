    if abs(s - S) < 1e-12:
        return a
    if s == 0:
        return np.full(n, S/n)
    # Binary search for threshold
    lo, hi = 0.0, a.max()
    for _ in range(60):

        if np.sum(np.maximum(a - mid, 0.0)) > S:

        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    res[res.argmax()] += S - res.sum()
    return np.maximum(res, 0.0)
