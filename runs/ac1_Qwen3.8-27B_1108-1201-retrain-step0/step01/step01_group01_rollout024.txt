    b = convolve(a, a), so db[k]/da[i] = 2*a[k-i] (when valid).
    grad[i] = 2 * sum_j a[j] * w[i+j] where w = softmax(b/tau)
    Computed via FFT for efficiency.
    """
    # Compute weights
    b_max = np.max(b)
    w = np.exp((b - b_max) / tau)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # This is cross-correlation of a and w
    # Using FFT: zero-pad both to size >= n + (2n-1) - 1 = 3n-2
    N = 1
    while N < 3 * n:
        N *= 2
    
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    
    # cross-correlation: (a ⋆ w)[i] = sum_j a[j] * w[i+j]
    # = ifft(conj(F(a)) * F(w)) ... let me verify
    # Actually: FFT of cross-correlation = conj(F(a)) * F(w)
    # But we need to be careful about the definition.
    # sum_j a[j] * w[i+j] is the "toeplitz" product.
    # Let's define: c[i] = sum_j a[j] * w[i+j] for i = 0, ..., n-1
    # This is the same as the (n-1+i)-th element of convolve(a_rev, w)
    # where a_rev = a[::-1]
    # convolve(a_rev, w)[k] = sum_j a_rev[j] * w[k-j] = sum_j a[n-1-j] * w[k-j]
    # Let m = n-1-j: = sum_m a[m] * w[k-n+1+m]
    # So convolve(a_rev, w)[n-1+i] = sum_m a[m] * w[i+m] = c[i] ✓
    
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(fa_rev * fw, N)
    
    grad = 2.0 * conv[n - 1: 2 * n - 1]
    return grad


def _score(a):
    """Compute the objective value."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _nesterov_smooth_descent(a, deadline, rng, tau_start=None, tau_end=None,
                               lr=None, momentum_init=0.85, momentum_end=0.95,
                               max_iter=50000):
    """
    Nesterov accelerated smoothed gradient descent on the simplex.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    
    if tau_start is None:
        tau_start = 0.005 / n  # relative to typical b values ~ 1/n
    if tau_end is None:
        tau_end = 0.0001 / n
    
    if lr is None:
        lr = 0.02
    
    best = a.copy()
    best_score = _score(best)
    
    v = a.copy()  # previous iterate for momentum
    t = 0
    no_improve = 0
    
    while time.monotonic() < deadline and t < max_iter:
        t += 1
        progress = min(1.0, t / max_iter)
        
        # Adaptive temperature
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Adaptive momentum
        momentum = momentum_init + (momentum_end - momentum_init) * progress
        
        # Nesterov lookahead
        y = a + momentum * (a - v)
        v_old = a.copy()
        
        # Ensure y is on simplex for computation
        y = _project_simplex(y, 1.0)
        
        # Compute convolution
        b = np.convolve(y, y)
        
        # Compute smooth gradient
        grad = _smooth_gradient(b, y, tau, n)
        
        # Center gradient (for sum constraint)
        grad -= np.mean(grad)
        
        # Update
        a = a - lr * grad
        
        # Project onto simplex
        a = _project_simplex(a, 1.0)
        
        v = v_old
        
        # Track best
        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Early stop if converged
        if no_improve > 5000 and tau < tau_end * 10:
            break
    
    return best, best_score


def _perturb(a, rng, scale=0.05):
    """Add smooth random perturbation."""
    n = len(a)
    # Low-frequency perturbation
    freqs = rng.integers(1, max(2, n // 8), size=min(3, n // 4))
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    max_noise = np.max(np.abs(noise))
    if max_noise > 1e-10:
        noise /= max_noise
    a_new = a + scale * noise
    return _project_simplex(a_new, 1.0)


def _make_anti(n, eps):
    """Anti-symmetric two-level."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    return a


def _make_cosine(n, eps):
    """Cosine anti-symmetric."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 1e-10)
    a /= np.sum(a)
    return a

    """Numerically stable softmax."""

    ex = np.exp(x_shift)
    s = np.sum(ex)
    if s < 1e-300:
        return np.full(len(x), 1.0 / len(x))
    return ex / s


def _fft_convolve(a):
    """Compute convolve(a, a) using FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)
    return b[:2 * n - 1]


def _fft_cross_corr(a, w):
    """
    Compute W_a[i] = sum_j a[j] * w[i+j] for i=0,...,n-1
    where w has length 2n-1.
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N *= 2
    a_rev = a[::-1]
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(fa * fw, N)
    return conv[n - 1: 2 * n - 1]


def _score_from_b(b, n):
    """Compute score from convolution."""
    return 2.0 * n * np.max(b)


def _adam_smooth_descent(x, deadline, rng, tau_start, tau_end, lr=0.01,
                          beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam optimizer in log-space with adaptive log-sum-exp temperature.
    x is the log-parameter, a = softmax(x).
    Minimizes 2n * soft_max(conv(a,a), tau).
    """
    n = len(x)
    x = np.asarray(x, dtype=np.float64).copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    a = _softmax(x)
    b = _fft_convolve(a)
    best_score = _score_from_b(b, n)
    best_x = x.copy()
    
    no_improve = 0
    
    while time.monotonic() < deadline:
        t += 1
        progress = min(1.0, t / 50000)
        
        # Adaptive temperature
        tau = tau_start * (tau_end / tau_start) ** progress
        
        a = _softmax(x)
        b = _fft_convolve(a)
        
        # Softmax weights over b
        b_max = np.max(b)
        w = np.exp((b - b_max) / tau)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum
        
        # Compute W_a[i] = sum_j a[j] * w[i+j]
        W_a = _fft_cross_corr(a, w)
        
        # W_b = dot(w, b)
        W_b = np.dot(w, b)
        
        # Gradient in x-space: grad_x = 4n * a * (W_a - W_b)
        grad_x = 4.0 * n * a * (W_a - W_b)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * grad_x ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Track best (using true max, not smooth)
        if t % 10 == 0:  # Check every 10 steps to save time
            new_a = _softmax(x)
            new_b = _fft_convolve(new_a)
            new_score = _score_from_b(new_b, n)
            if new_score < best_score:
                best_score = new_score
                best_x = x.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve > 5000 and progress > 0.8:
            break
    
    return best_x, best_score

    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 4: Ultra-fine polish ({remaining:.1f}s) ===")
    if remaining > 15:
        t4 = min(time.monotonic() + remaining, deadline)
        best_x, best_s = _adam_smooth_descent(
            global_best_x, t4, rng,
            tau_start=0.001 * b_scale, tau_end=0.0001 * b_scale,
            lr=0.02, check_interval=1
        )
        if best_s < global_best_score:
            global_best_score = best_s
            global_best_x = best_x.copy()
            print(f"  Polish: NEW BEST! Score: {global_best_score:.8f}")

    # Final: verify with direct computation
    a_final = _softmax(global_best_x)
    a_final = np.maximum(a_final, 0.0)
    a_final /= np.sum(a_final)

    # Double-check score with direct convolution
    b_direct = np.convolve(a_final, a_final)
    direct_score = 2.0 * n * np.max(b_direct)
    print(f"Direct score check: {direct_score:.8f}")

    result = [float(x) for x in a_final]
    final_score = evaluate_sequence(result)
    print(f"FINAL SCORE: {final_score:.8f}")
    return result
