    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    idx = res.argmax()
    res[idx] += S - res.sum()
    return res

def compute_smooth_max_grad(a, tau):
    """Compute gradient and value of tau*log(sum(exp(conv(a,a)/tau)))."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    q = exp_c / exp_c.sum()
    value = tau * np.log(exp_c.sum()) + c_max  # = tau * log_sum_exp(c/tau)
    
    # Gradient: grad[j] = 2 * sum_m a[m] * q[m+j]
    ra = a[::-1]
    conv_ra_q = fft_convolve(ra, q)
    grad = 2.0 * conv_ra_q[n-1:2*n-1]
    
    return grad, c, value

def adam_step(params, grad, lr, beta1, beta2, eps, step):
    """Adam update step."""
    m = params.get('m', np.zeros_like(grad))
    v = params.get('v', np.zeros_like(grad))
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**step)
    v_hat = v / (1 - beta2**step)
    params['m'] = m
    params['v'] = v
    return m_hat / (np.sqrt(v_hat) + eps)

def run_grad_phase(a0, tau_hi, tau_lo, t_budget):

        v_hat = v_cache / (1 - beta2**step)
        update = m_hat / (np.sqrt(v_hat) + eps)

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses a hybrid approach: LP-based outer iterations (guaranteed monotone improvement)
followed by Adam gradient descent with log-sum-exp smoothing and FFT-based
convolution for fast gradient computation. Exploits symmetry and uses
multiple initializations including Hamming windows and edge-concentrated profiles.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve(a, b):
    """Full linear convolution using FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:size]


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    s = a.sum()
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, a.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    idx = res.argmax()
    res[idx] += S - res.sum()
    return res


def symmetrize(a):
    """Enforce a[i] = a[n-1-i]."""
    return 0.5 * (a + a[::-1])


def project_sym_simplex(a, S):
    """Project onto {a >= 0, sum(a) = S, a symmetric}."""
    a = symmetrize(a)
    a = project_simplex(a, S)
    a = symmetrize(a)
    a = project_simplex(a, S)
    return a


def compute_smooth_grad(a, tau):
    """Compute gradient and convolution for smoothed max objective."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    q = exp_c / exp_c.sum()
    # grad[j] = 2 * sum_m a[m] * q[m+j]
    ra = a[::-1]
    conv_ra_q = fft_convolve(ra, q)
    grad = 2.0 * conv_ra_q[n - 1:2 * n - 1]
    return grad, c

def init_edge(n, alpha=2.0):
    """Edge-concentrated initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-alpha * ((t - center) / center) ** 2)
    # Boost edges
    a += np.exp(-alpha * ((t - 0) / center) ** 2) * 0.5
    a += np.exp(-alpha * ((t - (n - 1)) / center) ** 2) * 0.5

        js = np.arange(j_start, j_end + 1)
        rows_list.extend([k] * len(js))
        cols_list.extend(js.tolist())
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)

        vals = a[idxs]
        A_sp = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        time_limit = max(0.5, t_budget - (time.monotonic() - t0))
        try:
            result = linprog(c_obj, A_ub=A_sp, b_ub=b_ub, bounds=bounds,
                             method='highs', options={'time_limit': time_limit})
        except Exception:
            break

        if not result.success:
            if it > 5:
                break
            continue

        b = result.x
        b_sum = b.sum()
        if b_sum < 1e-10:
            break

        # Normalize b to have sum S
        b_norm = b * S / b_sum
        if use_symmetry:
            b_norm = symmetrize(b_norm)
            b_norm = project_sym_simplex(b_norm, S)
        else:
            b_norm = project_simplex(b_norm, S)

        # Backtracking line search
        t_best = 0.0
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1 - t) * a + t * b_norm
            a_new = np.maximum(a_new, 0)
            mc = fft_convolve(a_new, a_new).max()
            if mc <= M * 1.0001:
                t_best = t
                break

        if t_best == 0.0:
            break

        a = (1 - t_best) * a + t_best * b_norm
        if use_symmetry:
            a = project_sym_simplex(a, S)
        else:
            a = project_simplex(a, S)

        mc_new = fft_convolve(a, a).max()
        if mc_new < best_mc - 1e-12:
            best_mc = mc_new
            best_a = a.copy()

        if it % 10 == 0:
            sc = evaluate_sequence(best_a.tolist())
            print(f"    LP[{it}]: M={M:.4f}, mc={mc_new:.4f}, t={t_best:.2f}, "
                  f"score={sc:.6f}, time={time.monotonic()-t0:.0f}s")

        if M - mc_new < 1e-10 * M:
            break

    sc = evaluate_sequence(best_a.tolist())
    elapsed = time.monotonic() - t0
    print(f"    LP done: {it} iters, mc={best_mc:.6f}, score={sc:.6f}, time={elapsed:.0f}s")
    return best_a, sc


def run_grad_phase(a0, tau_hi, tau_lo, t_budget, lr0=1.0, use_symmetry=True):
    """Adam gradient descent with log-sum-exp smoothing and tau annealing."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = (a0 / max(a0.sum(), 1e-12) * S).copy()
    if use_symmetry:
        a = project_sym_simplex(a, S)
    else:
        a = project_simplex(a, S)

    best_a = a.copy()
    best_mc = fft_convolve(a, a).max()
    best_sc = evaluate_sequence(a.tolist())

    t0 = time.monotonic()
    step = 0
    lr = lr0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_cache = np.zeros(n)
    v_cache = np.zeros(n)
    skip = max(1, n // 3000)  # Evaluate every 'skip' steps

    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        f = min(1.0, elapsed / max(t_budget, 1e-6))
        tau = tau_hi * (tau_lo / tau_hi) ** (f ** 0.7)

        grad, c = compute_smooth_grad(a, tau)
        if use_symmetry:
            grad = symmetrize(grad)

        m_cache = beta1 * m_cache + (1 - beta1) * grad
        v_cache = beta2 * v_cache + (1 - beta2) * grad ** 2
        step += 1
        m_hat = m_cache / (1 - beta1 ** step)
        v_hat = v_cache / (1 - beta2 ** step)
        update = m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - lr * update
        if use_symmetry:
            a_new = project_sym_simplex(a_new, S)
        else:
            a_new = project_simplex(a_new, S)

        a = a_new

        if step % skip == 0:
            mc = fft_convolve(a, a).max()
            if mc < best_mc - 1e-12:
                best_mc = mc
                best_a = a.copy()
                lr = min(lr * 1.02, 10.0)
            else:
                lr = max(lr * 0.5, 0.001)

    sc = evaluate_sequence(best_a.tolist())
    if sc < best_sc:
        best_sc = sc
    return best_a, best_sc


def run_adaptive_grad(a0, t_budget, use_symmetry=True):
    """Multi-round gradient with adaptive tau schedule."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = (a0 / max(a0.sum(), 1e-12) * S).copy()
    if use_symmetry:
        a = project_sym_simplex(a, S)
    else:
        a = project_simplex(a, S)

    best_a = a.copy()
    best_mc = fft_convolve(a, a).max()
    best_sc = evaluate_sequence(a.tolist())

    t0 = time.monotonic()
    step = 0
    lr = 0.5
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m_cache = np.zeros(n)
    v_cache = np.zeros(n)

    # Adaptive tau: start high, decrease based on progress
    tau = 1.0
    tau_min = 0.001
    last_improve = 0.0
    improve_count = 0
    skip = max(1, n // 5000)

    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        f = min(1.0, elapsed / max(t_budget, 1e-6))

        # Adaptive tau: decrease when stuck
        if float(step) - last_improve > 200 * skip:
            tau = max(tau * 0.5, tau_min)
            last_improve = float(step)

        grad, c = compute_smooth_grad(a, tau)
        if use_symmetry:
            grad = symmetrize(grad)

        m_cache = beta1 * m_cache + (1 - beta1) * grad
        v_cache = beta2 * v_cache + (1 - beta2) * grad ** 2
        step += 1
        m_hat = m_cache / (1 - beta1 ** step)
        v_hat = v_cache / (1 - beta2 ** step)
        update = m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - lr * update
        if use_symmetry:
            a_new = project_sym_simplex(a_new, S)
        else:
            a_new = project_simplex(a_new, S)
        a = a_new

        if step % skip == 0:
            mc = fft_convolve(a, a).max()
            if mc < best_mc - 1e-12:
                best_mc = mc
                best_a = a.copy()
                improve_count += 1
                last_improve = float(step)
                lr = min(lr * 1.01, 5.0)
                if tau > tau_min and improve_count > 5:
                    tau = max(tau * 0.8, tau_min)
                    improve_count = 0
            elif step % (skip * 20) == 0:
                lr = max(lr * 0.7, 0.001)

    sc = evaluate_sequence(best_a.tolist())
    if sc < best_sc:
        best_sc = sc
    return best_a, best_sc


def propose_candidate():
    """Main search: LP-based initialization followed by gradient polishing."""
    total_budget = 2385.0
    t_start = time.monotonic()
    remaining_fn = lambda: total_budget - (time.monotonic() - t_start)

    # Get provided sequence
    prov = None
    try:
        prov = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if prov.size > 0 and np.isfinite(prov).all():
            prov = np.maximum(prov, 0.0)
            if prov.sum() < 0.01:
                prov = None
    except (NameError, Exception):
        prov = None

    best_a = None
    best_sc = np.inf

    def update_best(a, sc):
        nonlocal best_a, best_sc
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"  *** NEW BEST: {best_sc:.6f} (n={len(a)})")

    # ---- Phase 1: LP-based optimization on provided/best sequence ----
    if prov is not None:
        n_prov = prov.size
        print(f"\n=== Phase 1: LP on provided sequence (n={n_prov}) ===")
        a0 = (prov / prov.sum() * np.sqrt(2.0 * n_prov)).copy()
        a0 = project_sym_simplex(a0, np.sqrt(2.0 * n_prov))
        lp_budget = min(600.0, remaining_fn() * 0.4)
        a_lp, sc_lp = run_lp_phase(a0, lp_budget, max_iters=200, use_symmetry=True)
        update_best(a_lp, sc_lp)
    else:
        # No provided sequence, try several initializations
        print(f"\n=== Phase 1: LP on multiple initializations ===")
        for n in [2500, 3500]:
            if remaining_fn() < 60:
                break
            for init_name, init_fn in [('hamming', init_hamming), ('hann', init_hann),
                                        ('edge', init_edge)]:
                if remaining_fn() < 60:
                    break
                a0 = init_fn(n)
                s = np.sqrt(2.0 * n)
                a0 = (a0 / a0.sum() * s).copy()
                a0 = project_sym_simplex(a0, s)
                lp_budget = min(200.0, remaining_fn() * 0.15)
                if lp_budget < 20:
                    continue
                print(f"  Trying n={n}, init={init_name}, budget={lp_budget:.0f}s")
                a_lp, sc_lp = run_lp_phase(a0, lp_budget, max_iters=80, use_symmetry=True)
                update_best(a_lp, sc_lp)

    if best_a is None:
        # Fallback
        n = 3000
        best_a = init_hamming(n)
        s = np.sqrt(2.0 * n)
        best_a = (best_a / best_a.sum() * s).copy()
        best_a = project_sym_simplex(best_a, s)
        best_sc = evaluate_sequence(best_a.tolist())
        print(f"  Fallback: score={best_sc:.6f}")

    # ---- Phase 2: Gradient-based polishing ----
    rem = remaining_fn()
    if rem > 30:
        n_best = best_a.size
        print(f"\n=== Phase 2: Gradient polishing (n={n_best}, budget={rem:.0f}s) ===")

        # Round 1: moderate tau
        tau_budget = rem * 0.4
        if tau_budget > 20:
            a_r, sc_r = run_adaptive_grad(best_a, tau_budget, use_symmetry=True)
            update_best(a_r, sc_r)

        # Round 2: lower tau for refinement
        rem = remaining_fn()
        if rem > 30:
            tau_budget = rem * 0.4
            a_r, sc_r = run_grad_phase(best_a, 0.3, 0.01, tau_budget, lr0=0.3,
                                        use_symmetry=True)
            update_best(a_r, sc_r)

        # Round 3: very low tau
        rem = remaining_fn()
        if rem > 30:
            tau_budget = rem * 0.5
            a_r, sc_r = run_grad_phase(best_a, 0.05, 0.001, tau_budget, lr0=0.1,
                                        use_symmetry=True)
            update_best(a_r, sc_r)

    # ---- Phase 3: Final LP polish ----
    rem = remaining_fn()
    if rem > 60:
        print(f"\n=== Phase 3: Final LP polish (budget={rem:.0f}s) ===")
        n_best = best_a.size
        s = np.sqrt(2.0 * n_best)
        a0 = (best_a / best_a.sum() * s).copy()
        a0 = project_sym_simplex(a0, s)
        lp_budget = rem * 0.7
        a_lp, sc_lp = run_lp_phase(a0, lp_budget, max_iters=100, use_symmetry=True)
        update_best(a_lp, sc_lp)

        # Final gradient after LP
        rem = remaining_fn()
        if rem > 20:
            a_r, sc_r = run_grad_phase(best_a, 0.1, 0.001, rem * 0.8, lr0=0.1,
                                        use_symmetry=True)
            update_best(a_r, sc_r)

    # ---- Final validation ----
    final_sc = evaluate_sequence(best_a.tolist())
    if final_sc != best_sc:
        print(f"  Note: final eval gives {final_sc:.6f} vs tracked {best_sc:.6f}")
        best_sc = final_sc

    print(f"\n=== FINAL: score={best_sc:.8f}, n={len(best_a)}, "
          f"time={time.monotonic()-t_start:.0f}s ===")
    return [float(x) for x in best_a]
