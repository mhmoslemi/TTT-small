def fast_convolve(a):

    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)

def convolve_ab(a, b):
    """Linear convolution of a and b using FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]

def build_constraint_matrix(a):
    """Build the (2n-1) x n matrix A where A[k,j] = a_j if j is valid for k, else 0."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end+1] = a[j_start:j_end+1]
    return A

def solve_lp(a, M):
    """Solve: maximize sum(g) s.t. A@g <= M, g >= 0.
    Returns the optimal g."""
    n = len(a)
    A = build_constraint_matrix(a)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b, method='highs', options={'presolve': True})
    if res.success:
        return res.x
    else:
        return None

"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses a linear program at each step to find the best direction g that is
"compatible" with the current sequence a (i.e., conv(a,g) <= M where M is
the current peak). Combines with multiplicative weights for diversity and
polishing. Tries multiple sequence lengths and initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b using FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]


def build_constraint_matrix(a):
    """
    Build the (2n-1) x n matrix A where A[k,i] = a[k-i] if 0<=k-i<=n-1, else 0.
    This satisfies: conv(a,g)_k = sum_i A[k,i] * g_i
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # A[k, i] = a[k-i] for i in [i_start, i_end]
        # k-i ranges from k-i_end (at i=i_end) to k-i_start (at i=i_start)
        # So A[k, i_start:i_end+1] = a[k-i_start : k-i_end-1 : -1]
        # = a[k-i_end+1 : k-i_start+1][::-1]
        A[k, i_start:i_end + 1] = a[k - i_end + 1:k - i_start + 1][::-1]
    return A


def solve_lp_for_g(a, M):
    """
    Solve: maximize sum(g) subject to A@g <= M*ones, g >= 0.
    Returns optimal g or None if infeasible.
    """
    n = len(a)
    A = build_constraint_matrix(a)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_optimize(a0, time_limit, t_init=0.5, max_iter=5000):
    """
    LP-based iterative optimization. Minimize max(conv(a,a)) subject to
    sum(a)=1, a>=0. Returns (best_a, best_max_conv).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-15:
        a = np.ones(n) / n
    else:
        a /= a_sum

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t = t_init
    no_improve_count = 0
    max_no_improve = 50
    t0 = time.monotonic()

    # Pre-build A (will be updated as a changes)
    A = build_constraint_matrix(a)

    for iteration in range(max_iter):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        M = conv.max()
        if M < 1e-15:
            break

        # Rebuild A if needed (we rebuild every iteration since a changes)
        A = build_constraint_matrix(a)

        # Solve LP
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, method='highs',
                          options={'presolve': True})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break

        g_norm = g / g_sum

        # Check if g is essentially the same as a (convergence)
        if np.max(np.abs(g_norm - a)) < 1e-8:
            break

        # Try the update
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new /= a_new_sum

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            a = a_new
            conv = conv_new
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve_count = 0
            # Increase step size slightly
            t = min(0.9, t * 1.05)
        else:
            no_improve_count += 1
            t *= 0.7
            if no_improve_count >= max_no_improve:
                # Try pure g
                conv_g = fast_convolve(g_norm)
                if conv_g.max() < best_obj - 1e-12:
                    a = g_norm.copy()
                    conv = conv_g
                    best_obj = conv.max()
                    best_a = a.copy()
                    t = 0.1
                    no_improve_count = 0
                else:
                    # Add small random perturbation to escape
                    perturb = np.random.default_rng(iteration).dirichlet(np.ones(n) * 0.5)
                    a = 0.9 * a + 0.1 * perturb
                    conv = fast_convolve(a)
                    t = 0.3
                    no_improve_count = 0

    return best_a, best_obj


def mw_optimize(a0, time_limit, t_param=50.0, eta0=0.5):
    """
    Multiplicative weights (mirror descent) optimization as a complement
    to the LP approach. Good for escaping local minima.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-15:
        a = np.ones(n) / n
    else:
        a /= a_sum

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    eta = eta0
    patience = 0
    max_patience = 60
    current_t = t_param
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Soft-max weights over convolution
        scaled = current_t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Compute weighted subgradient
        sg = np.zeros(n)
        thresh = w.max() * 0.005
        active_idx = np.where(w > thresh)[0]

        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]

        # Multiplicative update
        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-15:
            a = np.ones(n) / n
        else:
            a /= a_sum

        conv = fast_convolve(a)
        current_obj = conv.max()

        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            patience = 0
            eta *= 0.65
            if eta < 0.003:
                current_t *= 2.0
                eta = 0.3
                max_patience = 30
            else:
                current_t = np.clip(current_t * 1.3, 5.0, 300.0)

    return best_a, best_obj

def generate_initializations(n, parent, rng):
    """Generate diverse initializations for a given length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # From parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0).copy()
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits.append(pp)
        elif parent.size > 0:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size),
                           np.maximum(parent, 0))
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits.append(pp)

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 0.2))

    # Gaussian centered
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - n * 0.5) / (n * 0.2)) ** 2)
    g /= g.sum()
    inits.append(g)

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 6)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    if bi.sum() > 0:
        bi /= bi.sum()
        inits.append(bi)

    # Cosine
    cos_bump = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_bump /= cos_bump.sum()
    inits.append(cos_bump)

    return inits
