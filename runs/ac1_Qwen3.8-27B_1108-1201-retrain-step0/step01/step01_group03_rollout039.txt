```python
def optimize(a0, deadline):
    n = len(a0)
    a = clip(a0, 0, 1000)
    S_target = sqrt(2*n)
    a = normalize(a, S_target)
    
    best = a.copy()
    best_score = compute_obj(a)
    
    # Soft-peak gradient descent with line search
    k = 10  # number of top elements for soft peak
    tau = 1.0  # temperature
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution
        c = convolve(a, a)
        
        # Compute soft-peak gradient
        # Focus on top-k elements
        top_k_idx = argsort(c)[-k:]
        weights = softmax(c[top_k_idx] * tau)
        
        # Gradient of sum of weighted top-k convolution values
        grad = zeros(n)
        for m in top_k_idx:
            w = weights[...]  # weight for this m
            for j in valid_range(m, n):
                grad[j] += 2 * w * a[m-j]
        
        # Project gradient onto sum-zero subspace (for fixed-sum constraint)
        grad -= grad.mean()
        
        # Line search
        alpha = 1.0
        current_obj = compute_obj(a)
        while alpha > 1e-6:
            a_trial = a - alpha * grad
            a_trial = max(a_trial, 0)
            a_trial = normalize(a_trial, S_target)
            trial_obj = compute_obj(a_trial)
            if trial_obj < current_obj:
                break
            alpha *= 0.5
        
        a = a_trial
        score = compute_obj(a)
        if score < best_score:
            best_score = score
            best = a.copy()
    
    return best, best_score
```

```python
def proximal_step(a, M, n):
    """One proximal step: reduce elements that contribute most to peak."""
    c = convolve(a, a)
    S = sum(a)
    
    # Compute gradient of max(c) (using soft top)
    sorted_idx = argsort(c)[::-1]  # descending
    k = min(5, 2*n-1)  # top 5
    top_vals = c[sorted_idx[:k]]
    weights = softmax(top_vals / M)  # normalized weights
    
    grad = zeros(n)
    for i in range(k):
        m = sorted_idx[i]
        w = weights[i]
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            grad[j] += 2 * w * a[m - j]
    
    # Normalize gradient
    grad_norm = norm(grad)
    if grad_norm < 1e-10:
        return a
    
    # Step in negative gradient direction
    lr = 0.1
    a_new = a - lr * grad / grad_norm
    
    # Project: non-negativity and fixed sum
    a_new = max(a_new, 0)
    a_new = a_new * S / sum(a_new)
    
    return a_new
```

```python
"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses multi-start soft-peak gradient descent with backtracking line search,
diverse structured initializations, LP-based refinement, and exploration
of multiple sequence lengths. Includes periodic perturbation for escaping
local minima and adaptive learning rate scheduling.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_to(a, target):
    """Normalize sequence to have the given sum."""
    S = a.sum()
    if S < 1e-12:
        return np.full_like(a, target / len(a))
    return a * (target / S)


def grad_soft_peak(a, c, k, tau):
    """
    Compute gradient of soft-peak objective: sum_{m in top-k} w_m * c[m].
    c = convolve(a, a). Returns gradient vector of length n.
    """
    n = len(a)
    num_c = len(c)
    k = min(k, num_c)

    top_idx = np.argpartition(c, -k)[-k:]
    c_top = c[top_idx].astype(np.float64)
    c_max = c_top.max()
    exp_vals = np.exp(tau * (c_top - c_max))
    w = exp_vals / exp_vals.sum()

    grad = np.zeros(n, dtype=np.float64)
    for i in range(len(top_idx)):
        m = int(top_idx[i])
        wi = w[i]
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        grad[lo:hi + 1] += 2.0 * wi * a[m - hi:m - lo + 1][::-1]

    return grad


def grad_hard_peak(a, c, n):
    """Compute gradient of hard peak (single argmax)."""
    m_star = int(c.argmax())
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    grad[lo:hi + 1] = 2.0 * a[m_star - hi:m_star - lo + 1][::-1]
    return grad


def project_fixed_sum_nonneg(a, target_sum):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / S)


def optimize_gradient(a0, deadline, max_iters=50000):

    Returns (best_sequence, best_score).

    c = np.convolve(a, a)
    score = 2.0 * n * c.max() / (S_t * S_t)
    best = a.copy()
    best_score = score

    lr = 0.12
    patience_counter = 0
    patience_limit = 3000

    for it in range(max_iters):
        if time.monotonic() > deadline:

def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    if n > 1024:
        return None
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_refine_sequence(a, deadline, max_iters=80):
    """LP-based refinement for a sequence."""
    n = len(a)
    if n > 1024:
        return a.copy(), compute_obj(a)

    S_t = np.sqrt(2.0 * n)
    a = normalize_to(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)

    for _it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = np.convolve(a, a)
        M = float(c.max())

        g = solve_lp(a, M, n)
        if g is None or g.sum() < 1e-12:
            break

        improved = False
        for t in [0.85, 0.65, 0.45, 0.25, 0.1]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to(a_new, S_t)
            s_new = compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            # Small gradient nudge to escape
            c = np.convolve(a, a)
            grad = grad_hard_peak(a, c, n)
            grad -= grad.mean()
            a_nudge = a - 0.01 * grad
            a_nudge = project_fixed_sum_nonneg(a_nudge, S_t)
            s_nudge = compute_obj(a_nudge)
            if s_nudge < best_score:
                a = a_nudge
                best = a.copy()
                best_score = s_nudge
            else:
                break

            print(f"eval error: {e}")
        return final_list

    print("Fallback: uniform n=256")
    return [1.0] * 256
```
