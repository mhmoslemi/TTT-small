def _fft_convolve(a, b):
    """FFT-based full convolution of two 1D arrays."""
    n_out = len(a) + len(b) - 1
    N = 1
    while N < n_out:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

def _compute_b(a):

def _cross_corr(w, a):
    """Compute g_j = sum_m w[m+j] * a[m] for j=0..n-1 via FFT."""
    n = len(a)
    N = 1
    while N < len(w) + n - 1:
        N <<= 1
    W = np.fft.rfft(w, N)

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)

        # Get top M peaks with softmax weighting
        if M >= len(b):
            k_idx = np.arange(len(b))
            b_sel = b
        else:
            k_idx = np.argpartition(b, -M)[-M:]
            b_sel = b[k_idx]

        # Softmax weights over selected peaks
        b_sel_shifted = b_sel - np.max(b_sel)
        w_raw = np.exp(10.0 * b_sel_shifted)
        w = w_raw / np.sum(w_raw)

        # Compute weighted gradient
        grad = np.zeros(n)
        for ki in range(len(k_idx)):
            wt = w[ki]
            if wt < 1e-10:
                continue
            k = k_idx[ki]
            # grad_j += wt * 2 * a[k-j] if 0 <= k-j < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                grad[idx] += wt * 2.0 * a[k - idx]

        # Center gradient (for sum constraint)
        grad -= np.mean(grad)

        # Momentum
        vel = momentum_rate * vel + (1.0 - momentum_rate) * grad
        a = a - eta * vel

        # Project
        a = _project_simplex(a, 1.0)

    return best_a, best_score


def _local_search(a_init, deadline, rng, n_trials_per_iter=20, step_scale=0.02):
    """
    Targeted local search: try to reduce the peak by transferring mass
    from high-contribution indices to low-contribution ones.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S

    best_a = a.copy()
    best_score = _score(a)
    temp = 0.01
    min_temp = 1e-6

    iter_count = 0
    while time.monotonic() < deadline:
        b = _compute_b(a)
        k_peak = np.argmax(b)

        # Find indices with largest contribution to b[k_peak]
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        if lo > hi:
            break
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[k_peak - idx]
        total_contrib = np.sum(contrib)
        if total_contrib < 1e-15:
            break

        # Try several random transfers
        improved = False
        for _ in range(n_trials_per_iter):
            # Pick source from high-contribution indices
            src_weights = contrib / total_contrib
            i_src = idx[rng.choice(len(idx), p=src_weights)]

            # Pick destination randomly (prefer low values to reduce other peaks)
            i_dst = rng.integers(0, n)
            if i_dst == i_src:
                continue

            # Transfer amount
            delta = step_scale * a[i_src] * (1 if rng.random() < 0.7 else -1)
            if a[i_src] + delta < 0 or a[i_dst] - delta < 0:
                continue

            a_new = a.copy()
            a_new[i_src] += delta
            a_new[i_dst] -= delta

            # Quick score (only check if peak might be reduced)
            b_new = _compute_b(a_new)
            s_new = 2.0 * n * np.max(b_new) / (np.sum(a_new) ** 2)

            if s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break
            # Simulated annealing accept
            elif rng.random() < np.exp(-(s_new - best_score) / temp):
                a = a_new.copy()

        if not improved:
            a = best_a.copy()
            temp *= 0.95
            if temp < min_temp:
                break

        iter_count += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimum normalized autocorrelation peak sequence.
    Phase 1: Smooth optimization (log-sum-exp) with diverse starts.
    Phase 2: Direct subgradient with momentum.
    Phase 3: Targeted local search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size

    # Global best tracking
    global_best = None
    global_best_score = np.inf

    def _update_best(a, score):
        nonlocal global_best, global_best_score
        if score < global_best_score:
            global_best = a.copy()
            global_best_score = score
            print(f"  [NEW BEST] Score: {score:.8f} (n={n})")

    # Normalize parent
    a_parent = parent / np.sum(parent) if np.sum(parent) > 1e-10 else np.ones(n) / n
    s_parent = _score(a_parent)
    _update_best(a_parent, s_parent)

    # Create diverse starting points
    starts = [a_parent]

    # Two-level anti-symmetric
    for eps in [0.3, 0.5, 0.7, 0.9]:
        starts.append(_make_two_level(n, eps))

    # Smooth cosine profiles at various frequencies
    for freq in [1, 2, 3, 5]:
        for eps in [0.5, 0.8]:
            starts.append(_make_smooth_profile(n, freq, eps))

    # Linear ramp
    for eps in [0.5, 0.8]:
        starts.append(_make_linear_ramp(n, eps))

    # Random smooth (low-pass filtered random)
    for _ in range(2):
        r = rng.random(n)
        # Simple moving average as low-pass
        k = max(1, n // 100)
        r_smooth = np.convolve(r, np.ones(k) / k, mode='same')
        r_smooth = np.maximum(r_smooth, 0.01)
        starts.append(r_smooth / np.sum(r_smooth))

    print(f"Phase 1: Smooth optimization with {len(starts)} starting points")
    print(f"  Parent score: {s_parent:.6f}")

    # Phase 1: Smooth optimization (log-sum-exp + momentum in z-space)
    # Allocate time: 50% of budget
    phase1_deadline = min(time.monotonic() + 0.5 * (float(budget_s) - 10.0), deadline)

    # Give parent more time, others less
    n_starts = len(starts)
    time_per_start = (phase1_deadline - time.monotonic()) / n_starts

    for si, a_start in enumerate(starts):
        if time.monotonic() >= phase1_deadline:
            break
        # Parent gets double time
        if si == 0:
            t_alloc = min(2 * time_per_start, phase1_deadline - time.monotonic())
        else:
            t_alloc = min(time_per_start, phase1_deadline - time.monotonic())
        if t_alloc < 5:
            break

        start_dl = min(time.monotonic() + t_alloc, phase1_deadline)
        # Use shorter beta schedule for less time
        if t_alloc > 100:
            beta_sched = [1000, 5000, 15000, 30000]
        elif t_alloc > 30:
            beta_sched = [5000, 20000]
        else:
            beta_sched = [10000]

        best_a, best_s = _smooth_optimize(a_start, start_dl, rng, beta_schedule=beta_sched)
        _update_best(best_a, best_s)
        if si % 3 == 0:
            print(f"  Start {si+1}/{n_starts} done, best={global_best_score:.6f}, t={time.monotonic():.0f}s")

    # Phase 2: Direct subgradient optimization
    print(f"\nPhase 2: Subgradient optimization, current best={global_best_score:.6f}")
    phase2_deadline = min(time.monotonic() + 0.3 * (float(budget_s) - 10.0), deadline)

    if global_best is not None and phase2_deadline - time.monotonic() > 30:
        # Run subgradient from best
        best_a2, best_s2 = _subgradient_optimize(
            global_best, phase2_deadline, rng, M=15,
            eta_init=0.2 / np.sqrt(n), momentum_rate=0.6
        )
        _update_best(best_a2, best_s2)

        # Also try from a perturbed version
        if time.monotonic() < deadline:
            a_pert = global_best + 0.05 * rng.standard_normal(n)
            a_pert = _project_simplex(a_pert, 1.0)
            remaining = deadline - time.monotonic()
            if remaining > 60:
                best_a2b, best_s2b = _subgradient_optimize(
                    a_pert, min(time.monotonic() + remaining * 0.5, deadline), rng,
                    M=8, eta_init=0.1 / np.sqrt(n), momentum_rate=0.7
                )
                _update_best(best_a2b, best_s2b)

    # Phase 3: Local search
    print(f"\nPhase 3: Local search, current best={global_best_score:.6f}")
    phase3_deadline = deadline

    if global_best is not None and phase3_deadline - time.monotonic() > 20:
        # Multiple local search attempts with different step scales
        for step_scale in [0.05, 0.02, 0.01]:
            if time.monotonic() >= phase3_deadline:
                break
            t_alloc = min(60.0, phase3_deadline - time.monotonic())
            best_a3, best_s3 = _local_search(
                global_best, min(time.monotonic() + t_alloc, phase3_deadline),
                rng, n_trials_per_iter=30, step_scale=step_scale
            )
            _update_best(best_a3, best_s3)

    # Final: one more subgradient polish
    remaining = deadline - time.monotonic()
    if global_best is not None and remaining > 30:
        print(f"\nFinal polish with {remaining:.0f}s remaining")
        best_af, best_sf = _subgradient_optimize(
            global_best, deadline, rng, M=20,
            eta_init=0.05 / np.sqrt(n), momentum_rate=0.5
        )
        _update_best(best_af, best_sf)

    # Return result
    if global_best is None:
        result = [float(x) for x in a_parent]
    else:
        result = [float(x) for x in global_best]

    # Ensure validity
    for i in range(len(result)):
        if result[i] < 0:
            result[i] = 0.0
    if sum(result) < 0.01:
        result = [1.0 / n] * n

    final_score = float(evaluate_sequence(result))
    print(f"\nFINAL SCORE: {final_score:.6f}")
    print(f"Sequence length: {len(result)}")
    return result
