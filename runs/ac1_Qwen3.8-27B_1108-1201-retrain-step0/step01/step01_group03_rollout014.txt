```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Variance-based optimization (smooth surrogate via FFT)
2. Soft-max peak reduction
3. Hard subgradient refinement
4. Local perturbations to escape local minima
All gradient computations use FFT for O(n log n) efficiency.
"""

import numpy as np
import time

def _fft_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    N = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    import math
    Np = 1 << (N - 1).bit_length()
    F = np.fft.rfft(a, Np)
    c = np.fft.irfft(F * np.conj(F), Np)
    return c[:N]

def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = _fft_conv(a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def _normalize(a, target):
    """Normalize sequence to have a specific sum."""
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)

def _var_gradient(a, Np):
    """Compute gradient of sum(c^2) where c = conv(a,a). O(N log N)."""
    n = len(a)
    F = np.fft.rfft(a, Np)
    # d/da sum(c^2) = (4/Np) * Re(ifft(|F|^2 * F))
    grad_freq = np.abs(F)**2 * F
    grad = (4.0 / Np) * np.fft.irfft(grad_freq, Np)[:n]
    return grad

def _softmax_gradient(a, c, tau, Np):
    """Compute gradient of soft-max objective using FFT-based shortcut."""
    n = len(a)
    # softmax weights
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    # grad[j] = 2 * sum_m w[m] * a[m-j] (for valid m-j)
    # This is 2 * (w * a) circular-ish, but for linear it's:
    # grad = 2 * convolve(w_padded, a_reversed) - but that's O(n^2)
    # Alternative: use FFT
    # Actually w has length 2n-1, a has length n
    # grad[j] = 2 * sum_m w[m] * a[m-j] for 0<=m-j<n
    # = 2 * sum_{k=0}^{n-1} w[j+k] * a[k] (where k = m-j)
    # This is a cross-correlation-like operation
    # Use FFT: pad w and a to size >= 3n
    Nc = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, Nc)
    A = np.fft.rfft(a, Nc)
    # cross-correlation
    corr = np.fft.irfft(W * np.conj(A), Nc)[:2*n-1]
    grad = 2.0 * corr[:n]
    return grad

def _hard_gradient(a, c):
    """Compute subgradient at the peak of c."""
    n = len(a)
    m_star = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m_star - j]
    return grad

def _optimize_var(a0, deadline, max_iters=5000):
    """Phase 1: Minimize variance of autocorrelation via projected gradient with momentum."""
    n = len(a0)
    S_t = np.sqrt(2.0 * n)
    Np = 1 << (2 * n).bit_length()
    
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a, S_t)
    
    best = a.copy()
    best_score = _compute_obj(a)
    
    v = np.zeros(n)  # momentum
    beta = 0.9
    lr = 0.5
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break
        
        # Compute gradient of variance
        grad = _var_gradient(a, Np)
        
        # Project out mean (maintain fixed sum)
        grad -= grad.mean()
        
        # Momentum
        v = beta * v + (1 - beta) * grad
        v -= v.mean()  # re-project
        
        # Adaptive learning rate
        step = lr / (1.0 + it * 0.01)
        
        a_new = a - step * v
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S_t)
        
        # Check objective periodically
        if (it + 1) % 200 == 0:
            s = _compute_obj(a_new)
            if s < best_score:
                best_score = s
                best = a_new.copy()
        
        a = a_new
    
    return best, best_score

def _optimize_softmax(a0, deadline, max_iters=3000):
    """Phase 2: Minimize soft-max of autocorrelation."""
    n = len(a0)
    S_t = np.sqrt(2.0 * n)
    Np = 1 << (2 * n).bit_length()
    
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a, S_t)
    
    best = a.copy()
    best_score = _compute_obj(a)
    
    v = np.zeros(n)
    beta = 0.9
    tau_start = 5.0
    tau_end = 50.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break
        
        # Anneal temperature
        progress = it / max_iters
        tau = tau_start + (tau_end - tau_start) * progress
        lr = 0.3 / (1.0 + it * 0.005)
        
        c = _fft_conv(a)
        grad = _softmax_gradient(a, c, tau, Np)
        grad -= grad.mean()
        
        v = beta * v + (1 - beta) * grad
        v -= v.mean()
        
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S_t)
        
        if (it + 1) % 200 == 0:
            s = _compute_obj(a_new)
            if s < best_score:
                best_score = s
                best = a_new.copy()
        
        a = a_new
    
    return best, best_score

def _optimize_hard(a0, deadline, max_iters=2000):
    """Phase 3: Minimize peak using hard subgradient + perturbations."""
    n = len(a0)
    S_t = np.sqrt(2.0 * n)
    Np = 1 << (2 * n).bit_length()
    
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a, S_t)
    
    best = a.copy()
    best_score = _compute_obj(a)
    
    v = np.zeros(n)
    beta = 0.95
    lr = 0.1
    rng = np.random.default_rng(12345)
    
    stagnant = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break
        
        # Decay learning rate
        lr = 0.1 / (1.0 + it * 0.002)
        
        c = _fft_conv(a)
        grad = _hard_gradient(a, c)
        grad -= grad.mean()
        
        v = beta * v + (1 - beta) * grad
        v -= v.mean()
        
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S_t)
        
        s = _compute_obj(a_new)
        
        if s < best_score:
            best_score = s
            best = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Periodic perturbation to escape local minima
        if stagnant > 200:
            perturbation_scale = 0.05 * (1 + stagnant / 500.0)
            pert = rng.normal(0, perturbation_scale, n)
            a = a + pert
            a = np.maximum(a, 0.0)
            a = _normalize(a, S_t)
            v = np.zeros(n)
            stagnant = 0
        
        a = a_new
    
    return best, best_score

def _local_search(a0, deadline, rng, max_perturbs=500):
    """Local search: try random perturbations and keep improvements."""
    n = len(a0)
    S_t = np.sqrt(2.0 * n)
    
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0).copy()
    a = _normalize(a, S_t)
    best = a.copy()
    best_score = _compute_obj(a)
    
    scale = 0.1
    
    for it in range(max_perturbs):
        if time.monotonic() > deadline - 30.0:
            break
        
        # Random perturbation
        idx = rng.random(n) < 0.3  # perturb 30% of elements
        delta = rng.normal(0, scale, n) * idx
        a_new = a + delta
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S_t)
        
        s = _compute_obj(a_new)
        if s < best_score:
            best_score = s
            best = a_new.copy()
            a = a_new
            # Slowly decrease scale
            scale *= 0.995
        else:
            # Sometimes accept worse solutions (simulated annealing)
            if rng.random() < np.exp(-10 * (s - best_score)):
                a = a_new
    
    return best, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- FFT-based variance gradient (smooth, O(n log n) per step)
- Spectral flattening (Gerchberg-Saxton style)
- Adaptive hard/soft subgradient for peak reduction
- Simulated annealing local search
Multiple sequence lengths explored (128-2048) with exhaustive time management.
"""

import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_conv(a):
    """Compute linear autocorrelation conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 2 * n - 1
    Np = _next_pow2(N)
    F = np.fft.rfft(a, Np)
    c = np.fft.irfft(F * np.conj(F), Np)
    return c[:N]


def _compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = _fft_conv(a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target):
    S = a.sum()
    if S < 1e-14:
        return a
    return a * (target / S)


def _var_gradient(a, Np):
    """
    Gradient of sum(conv(a,a)^2) computed via FFT. O(N log N).
    For real a: d/da[j] sum_m c[m]^2 = (4/Np) * ifft(|F|^2 * F)[j]
    """
    n = len(a)
    F = np.fft.rfft(a, Np)
    grad = (4.0 / Np) * np.fft.irfft(np.abs(F) ** 2 * F, Np)[:n]
    return grad


def _spectral_flatten_step(a, t, Np):
    """
    One Gerchberg-Saxton style step: blend |F| toward uniform magnitude.
    Returns new non-negative sequence (unnormalized).
    """
    n = len(a)
    F = np.fft.rfft(a, Np)
    mag = np.abs(F)
    target_mag = np.sqrt(np.sum(mag ** 2) / len(mag))
    new_mag = (1.0 - t) * mag + t * target_mag
    phase = np.angle(F)
    F_new = new_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(F_new, Np)[:n]
    return np.maximum(a_new, 0.0)


def _hard_subgrad(a, c):
    """Subgradient at the argmax of c. O(n)."""
    n = len(a)
    m_star = int(c.argmax())
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    # grad[j] = 2*a[m_star - j] for j in [lo, hi]
    # a[m_star - j] for j in [lo, hi] is a[hi]...a[lo] reversed = a[m_star-hi]...a[m_star-lo]
    indices = m_star - np.arange(lo, hi + 1)
    return 2.0 * a[indices]


def _soft_max_grad(a, c, tau):
    """Gradient of (1/tau)*log(sum(exp(tau*c))) w.r.t. a. Uses FFT cross-correlation."""
    n = len(a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    # grad[j] = 2 * sum_m w[m] * a[m-j] for valid m-j in [0,n-1]
    # = 2 * (w starred a)[j] where * denotes appropriate correlation
    # Use FFT: pad both to size >= (2n-1) + n = 3n-1
    Nc = _next_pow2(3 * n)
    W = np.fft.rfft(w, Nc)
    A = np.fft.rfft(a, Nc)
    # Cross-correlation: sum_m w[m]*a[m-j] = ifft(W * conj(A))[j + (2n-1-n)]
    # Actually: ifft(W * conj(A))[k] = sum_m w[m] * a[(m-k) mod Nc]
    # We want sum_m w[m] * a[m-j], so k=j works: sum_m w[m]*a[m-j]
    corr = np.fft.irfft(W * np.conj(A), Nc)
    # corr[j] = sum_m w[m] * a[(m-j) mod Nc]
    # For j in [0, n-1] and m in [0, 2n-2], m-j is in [-n+1, 2n-2]
    # When m-j >= 0: corr[j] includes a[m-j] which is what we want
    # When m-j < 0: a[m-j mod Nc] = a[m-j+Nc] which is 0 (a is only nonzero in [0,n-1])
    # So corr[j] = sum_{m=j}^{2n-2} w[m]*a[m-j] = sum_{k=0}^{n-1} w[j+k]*a[k] (where k=m-j)
    # This is exactly what we want!
    grad = 2.0 * corr[:n]
    return grad


def _optimize_var(a0, deadline, max_iters=10000):
    """Minimize variance of autocorrelation with momentum (L-BFGS-like)."""
    n = len(a0)
    S_t = np.sqrt(2.0 * n)
    Np = _next_pow2(2 * n - 1)

    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a = np.random.default_rng(0).random(n) * 0.5 + 0.25
    a = _normalize(a, S_t)

    best = a.copy()
    best_score = _compute_obj(a)

    v = np.zeros(n)
    beta = 0.9
    base_lr = 0.3

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        lr = base_lr / (1.0 + 0.002 * it)
        grad = _var_gradient(a, Np)
        grad -= grad.mean()

        v = beta * v + (1.0 - beta) * grad
        v -= v.mean()

        a = a - lr * v
        a = np.maximum(a, 0.0)
        a = _normalize(a, S_t)

        if (it + 1) % 300 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score


def _optimize_spectral(a0, deadline, max_iters=5000):
    """Spectral flattening + variance gradient hybrid."""
    n = len(a0)
    S_t = np.sqrt(2.0 * n)
    Np = _next_pow2(2 * n - 1)

    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a = np.random.default_rng(1).random(n) * 0.5 + 0.25
    a = _normalize(a, S_t)

    best = a.copy()
    best_score = _compute_obj(a)

    v = np.zeros(n)
    beta = 0.85
    base_lr = 0.2
    flatten_t = 0.03

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        lr = base_lr / (1.0 + 0.003 * it)

        # Variance gradient
        grad_var = _var_gradient(a, Np)
        grad_var -= grad_var.mean()

        # Spectral flattening direction
        a_spec = _spectral_flatten_step(a, flatten_t, Np)
        a_spec = _normalize(a_spec, S_t)
        grad_spec = a_spec - a
        grad_spec -= grad_spec.mean()

        # Combine
        combined = grad_var + 0.3 * grad_spec
        combined -= combined.mean()

        v = beta * v + (1.0 - beta) * combined
        v -= v.mean()

        a = a - lr * v
        a = np.maximum(a, 0.0)
        a = _normalize(a, S_t)

        if (it + 1) % 300 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score


def _optimize_adaptive(a0, deadline, max_iters=8000, seed=42):
    """
    Adaptive optimization: switches between soft-max and hard subgradient
    based on peak dominance. Includes periodic perturbations.
    """
    n = len(a0)
    S_t = np.sqrt(2.0 * n)
    Np = _next_pow2(2 * n - 1)
    rng = np.random.default_rng(seed)

    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a = rng.random(n) * 0.5 + 0.25
    a = _normalize(a, S_t)

    best = a.copy()
    best_score = _compute_obj(a)

    v = np.zeros(n)
    beta = 0.93
    base_lr = 0.15
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        lr = base_lr / (1.0 + 0.001 * it)
        c = _fft_conv(a)

        # Check peak dominance
        sorted_c = np.sort(c)[::-1]
        dominance = sorted_c[0] / max(sorted_c[1], 1e-10)

        if dominance > 1.5:
            # One peak dominates: use hard subgradient
            grad = _hard_subgrad(a, c)
        else:
            # Multiple peaks: use soft-max
            tau = 10.0 + 20.0 * (it / max_iters)
            grad = _soft_max_grad(a, c, tau)

        grad -= grad.mean()

        v = beta * v + (1.0 - beta) * grad
        v -= v.mean()

        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S_t)

        s = _compute_obj(a_new)
        if s < best_score:
            best_score = s
            best = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Periodic perturbation to escape local minima
        if stagnant > 200:
            scale = 0.03 + 0.01 * min(stagnant // 200, 5)
            a = a + rng.normal(0, scale, n)
            a = np.maximum(a, 0.0)
            a = _normalize(a, S_t)
            v = np.zeros(n)
            stagnant = 0

        a = a_new

    return best, best_score


def _local_search(a0, deadline, rng, max_iters=3000):
    """Local search with simulated annealing and adaptive step size."""
    n = len(a0)
    S_t = np.sqrt(2.0 * n)

    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0).copy()
    a = _normalize(a, S_t)
    best = a.copy()
    best_score = _compute_obj(a)

    scale = 0.08
    T = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        T *= 0.9995
        scale = max(0.005, 0.08 * (1.0 - 0.5 * it / max_iters))

        mask = rng.random(n) < 0.15
        delta = rng.normal(0, scale, n) * mask
        a_new = a + delta
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S_t)

        s = _compute_obj(a_new)
        if s < best_score:
            best_score = s
            best = a_new.copy()
            a = a_new
        elif T > 0.01 and rng.random() < np.exp(-8.0 * (s - best_score) / T):
            a = a_new

    return best, best_score


def _multi_scale_optimize(a0, deadline, rng):
    """
    Multi-scale approach: optimize at a smaller scale, then upscale and continue.
    This helps escape local minima.
    """
    n = len(a0)
    if n < 512 or time.monotonic() > deadline - 60.0:
        return a0, _compute_obj(a0)

    # Downsample by averaging pairs
    n_small = n // 4
    a_small = a0[:n_small * 4].reshape(n_small, 4).mean(axis=1)

    # Optimize at small scale
    t_small_deadline = time.monotonic() + min(30.0, (deadline - time.monotonic()) * 0.3)
    a_small_opt, s_small = _optimize_var(a_small, t_small_deadline, max_iters=2000)
    a_small_opt, s_small = _optimize_adaptive(a_small_opt, t_small_deadline, max_iters=1500, seed=77)

    # Upscale: replicate each element 4 times
    a_up = np.repeat(a_small_opt, 4)[:n]

    # Blend with original
    a_blend = 0.5 * a0 + 0.5 * a_up
    a_blend = np.maximum(a_blend, 0.0)
    a_blend = _normalize(a_blend, np.sqrt(2.0 * n))

    return a_blend, _compute_obj(a_blend)
```
