def peak_subgrad(a, top_frac=0.1):
    """Uniform-weight subgradient on top-k peaks of conv(a,a)."""
    c = np.convolve(a, a)
    k = max(1, int(top_frac * len(c)))
    top_k = np.partition(c, -k)[-k:]
    thresh = top_k.min()
    active = c >= thresh
    q = active.astype(np.float64) / active.sum()
    return 2.0 * np.correlate(q, a, 'valid')

def coordinate_descent(a, max_passes=50, eps=1e-13):
    """
    Lazy coordinate descent reducing sum(a^2) (helps minimize c[0] and c[2n-2]).
    Preserves non-negativity and sum.
    """
    n = a.size
    S = a.sum()
    a = a.astype(np.float64).copy()
    improved = 0
    for _ in range(max_passes):
        si2 = a.sum()
        i = int(np.argmax(si2))
        if i == 0 or a[i - 1] < eps:
            break
        target = np.sqrt(a[i] * a[i - 1])
        a[i] = target
        a[i - 1] = target
        improved += 1
    # Re-project to ensure exact sum
    return project_simplex(a, S)


def peak_target_move(a, n_moves=200, move_size=0.3):
    """
    Targeted moves: find peak lag, identify contributing coordinate,
    move mass to low-impact coordinate.
    """
    n = a.size
    S = a.sum()
    a = a.astype(np.float64).copy()
    best_c = float(np.convolve(a, a).max())
    rng = np.random.default_rng(99)

    for _ in range(n_moves):
        c = np.convolve(a, a)
        k_peak = int(c.argmax())
        # Sensitivity: dc[k_peak]/da[i] = 2*a[k_peak - i] for valid i
        j = k_peak  # a[k_peak - i] = a[j] when i=0, = a[k_peak-i] in general
        # The coordinate i that maximizes a[k_peak-i] (for valid k_peak-i in [0,n-1])
        i_start = max(0, k_peak - (n - 1))
        i_end = min(n - 1, k_peak)
        if i_end < i_start:
            continue
        sens = a[k_peak - i_start: k_peak - i_start + (i_end - i_start + 1)]
        i_src = i_start + int(np.argmax(sens))
        # Find destination: coordinate with minimum sensitivity to all top peaks
        top_peaks = np.argsort(c)[-min(5, n):][::-1]
        dest_sens = np.zeros(n)
        for kp in top_peaks:
            s_start = max(0, kp - (n - 1))
            s_end = min(n - 1, kp)
            if s_end >= s_start:
                dest_sens[s_start: s_end + 1] += 2.0 * a[kp - s_start: kp - s_start + (s_end - s_start + 1)]
        # Pick destination with lowest sensitivity and not the source
        dest_sens[i_src] = np.inf
        i_dst = int(np.argmin(dest_sens))
        if a[i_dst] < 0 and a[i_src] < 0:
            continue
        # Move mass
        delta = move_size * min(a[i_src], S / n * 0.1)
        if delta < 1e-10:
            continue
        a[i_src] -= delta
        a[i_dst] += delta
        a = np.maximum(a, 0.0)
        a = project_simplex(a, S)

        c_new = float(np.convolve(a, a).max())
        if c_new > best_c * 1.02:
            # Undo
            a[i_dst] -= delta
            a[i_src] += delta
            a = project_simplex(np.maximum(a, 0.0), S)
        else:
            best_c = c_new

    return a

        print(f"\nPhase 3: Peak-targeted + coord descent, t={rem:.0f}s")
        n = len(best_seq)
        S = np.sqrt(2.0 * n)
        a = best_seq.copy()
        a = project_simplex(a, S)

        # Peak-targeted moves in chunks
        chunk_time = min(30.0, rem * 0.4)
        t_chunk_start = time.monotonic()
        n_moves = int(chunk_time * 5000 / n)  # estimate moves based on n
        n_moves = max(100, min(2000, n_moves))

        a = peak_target_move(a, n_moves=n_moves, move_size=0.1)
        a = project_simplex(a, S)
        sc = evaluate_sequence(a.tolist())
        print(f"  After peak moves: score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

        # Coordinate descent passes
        rem = deadline - time.monotonic()
        if rem > 5:
            a = coordinate_descent(best_seq, max_passes=100)
            sc = evaluate_sequence(a.tolist())
            print(f"  After coord descent: score={sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()

    # Phase 4: Final polish with very small learning rate
    rem = deadline - time.monotonic()
    if rem > 20:
        print(f"\nPhase 4: Final polish, t={rem:.0f}s")
        a_r, c_max, steps = nesterov_opt(
            best_seq, 0.01, 0.003, 0.05, 0.9, rem * 0.8,
            use_sharp=True, sharp_frac=0.3)
        sc = evaluate_sequence(a_r.tolist())
        print(f"  Final: steps={steps}, score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = a_r.copy()

    # Final coordinate descent
    rem = deadline - time.monotonic()
    if rem > 5:
        a = coordinate_descent(best_seq, max_passes=200)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  Final coord descent: {best_score:.6f}")

    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_seq)} ===")
    return [float(x) for x in best_seq]
