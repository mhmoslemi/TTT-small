import numpy as np
import time
from scipy.optimize import linprog

def _convolve_fft(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next fast FFT size
    fsize = 1
    while fsize < size:
        fsize *= 2
    fa = np.fft.fft(a, fsize)
    fc = np.fft.ifft(fa * fa).real[:size]
    return fc

def _objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    c = _convolve_fft(a)
    return 2.0 * n * c.max() / (S * S)

def _project_onto_simplex(a, S):
    """Project a onto {x >= 0, sum(x) = S}."""
    # This is the Euclidean projection onto the scaled simplex
    # We want to find the closest point to a in {x >= 0, sum(x) = S}
    # This is equivalent to projecting a/S onto the standard simplex and scaling back
    v = a / S if S > 0 else a
    # Project v onto {x >= 0, sum(x) = 1}
    # Using the algorithm from the paper "Euclidean projection onto the simplex"
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w * S

def _grad_peak(a):
    """Gradient of max(conv(a,a)) with respect to a."""
    n = len(a)
    c = _convolve_fft(a)
    k_star = c.argmax()
    grad = np.zeros(n)
    for j in range(n):
        # dc[k_star]/da[j] = a[k_star - j] + a[j - k_star] (if indices valid)
        val = 0.0
        idx1 = k_star - j
        if 0 <= idx1 < n:
            val += a[idx1]
        idx2 = j - k_star
        if 0 <= idx2 < n:
            val += a[idx2]
        grad[j] = val
    return grad

def _grad_topk(a, k=10):
    """Gradient of sum of top-k elements of conv(a,a)."""
    n = len(a)
    c = _convolve_fft(a)
    # Get top-k indices
    top_idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for k_ in top_idx:
        for j in range(n):
            val = 0.0
            idx1 = k_ - j
            if 0 <= idx1 < n:
                val += a[idx1]
            idx2 = j - k_
            if 0 <= idx2 < n:
                val += a[idx2]
            grad[j] += val
    return grad

def _lp_solve(f, M):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M (elementwise), g >= 0."""
    n = len(f)
    if n > 2048:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix A of shape (2n-1, n)
    # A[m, j] = f[m-j] if 0 <= m-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-based gradient descent with simplex projection and LP-based refinement.
Explores multiple sequence lengths and starting points within the time budget.

def _fft_convolve(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fc = np.fft.irfft(fa * fa, fsize)[:size]
    return fc


def _obj_value(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _grad_peak(a):
    """Gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    k = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    grad[lo:hi+1] = 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def _grad_topk(a, k_top=5):
    """Gradient of sum of top-k_top elements of conv(a,a)."""
    n = len(a)
    c = _fft_convolve(a)
    top_idx = np.argsort(c)[::-1][:k_top]
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def _project_simplex(a, S):
    """Project a onto {x >= 0, sum(x) = S} (Euclidean projection)."""
    n = len(a)
    if S <= 0:
        return np.zeros(n)
    v = a / S
    # Project v onto standard simplex {x >= 0, sum(x) = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.full(n, 1.0 / n) * S
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w * S


def _lp_solve(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    if n > 2500:
        return None
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            rows.append(m)
            cols.append(j)
            vals.append(f[m - j])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _optimize_grad(a, deadline, max_iters=5000, lr_init=0.1):
    """Projected gradient descent minimizing peak autocorrelation."""
    n = len(a)
    S = float(a.sum())
    best = a.copy()
    best_score = _obj_value(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Adaptive learning rate
        progress = it / max_iters
        lr = lr_init * (1.0 + np.cos(np.pi * progress)) * 0.5
        lr = max(lr, 0.001)
        
        grad = _grad_peak(a)
        # Project gradient onto sum=0 subspace (to maintain sum constraint)
        grad -= grad.mean()
        
        a_new = a - lr * grad
        a_new = _project_simplex(a_new, S)
        
        # Check improvement every few iterations
        if (it + 1) % 50 == 0:
            s = _obj_value(a_new)
            if s < best_score:
                best_score = s
                best = a_new.copy()
        
        a = a_new
    
    return best, best_score


def _optimize_lp(a, deadline, max_lp_iters=200):
    """LP-based refinement."""
    n = len(a)
    S = float(a.sum())
    best = a.copy()
    best_score = _obj_value(a)
    
    c = _fft_convolve(a)
    M = float(c.max())
    
    for it in range(max_lp_iters):
        if time.monotonic() > deadline:
            break
        
        g = _lp_solve(a, M)
        if g is None or g.sum() <= 0:
            break
        
        # Blend
        improved = False
        for t in [0.5, 0.3, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = _project_simplex(a_new, S)
            s_new = _obj_value(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break
        
        if not improved:
            # Try gradient step to escape
            grad = _grad_peak(a)
            grad -= grad.mean()
            a = _project_simplex(a - 0.01 * grad, S)
            s = _obj_value(a)
            if s < best_score:
                best = a.copy()
                best_score = s
        
        c = _fft_convolve(a)
        M = float(c.max())
    
    return best, best_score

def _spectral_optimize(a0, deadline, max_iters=10000):
    """Spectral method: alternate between flattening |FFT(a)|^2 and projecting to a>=0."""
    n = len(a0)
    a = a0.copy()
    S = a.sum()
    fsize = 1 << (2*n - 2).bit_length()
    
    best = a.copy()
    best_score = _obj_value(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Frequency domain: flatten |A|^2
        fa = np.fft.rfft(a, fsize)
        power = np.abs(fa)**2
        # Water-filling: reduce the peak power while maintaining total power
        # Total power = sum(|fa|^2) = fsize * sum(a^2) (by Parseval)
        # We want to make power as uniform as possible
        # Simple approach: power_new = power * (1 - alpha * (power - mean(power)) / max(power))
        mean_p = power.mean()
        max_p = power.max()
        if max_p > 0:
            alpha = 0.1
            power_new = power - alpha * np.maximum(power - mean_p, 0)

            if s < best_score:

                best = a.copy()
    
    return best, best_score
