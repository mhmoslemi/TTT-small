            grad[j] = 2.0 * np.dot(a[:k_end], w[j:j + k_end])
    return grad / tau


def grad_l2(a):
    """Gradient of ||conv(a,a)||_2 (L2 norm of autocorrelation)."""
    n = len(a)
    c = np.convolve(a, a)
    norm = np.linalg.norm(c)
    if norm < 1e-12:
        return np.zeros(n)
    grad = np.zeros(n)
    for j in range(n):
        k_end = min(n, 2 * n - 1 - j)
        if k_end > 0:
            grad[j] = 2.0 * np.dot(a[:k_end], c[j:j + k_end])
    return grad / norm


def lp_solve_max_sum(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    n = len(f)
    if n > 2048:
        return None
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def run_gradient_phase(a0, deadline, n_iters=20000, mode='inf'):
    """
    Gradient descent phase. mode='inf' uses smoothed L-infinity,
    mode='l2' uses L2 norm gradient, mode='combined' switches from L2 to Linf.
    Uses Nesterov momentum and normalized gradient steps.
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S_t = np.sqrt(2.0 * n)
    a = normalize_to_sum(a, S_t)

    best = a.copy()
    best_score = compute_obj(a)

    # Momentum
    v = np.zeros(n)
    beta = 0.85

    # Step size (in normalized a-space, after norm=1 gradient)
    base_step = 0.08 * np.sqrt(2.0 * n)

    tau_curr = 5.0
    tau_max = 500.0

    stagnant_count = 0
    last_improve = 0

    for it in range(n_iters):
        if time.monotonic() > deadline - 20.0:
            break

        frac = it / max(n_iters - 1, 1)

        # Choose which gradient to use
        if mode == 'combined':
            if frac < 0.3:
                g = grad_l2(a)
            else:
                tau_curr = 5.0 + 495.0 * ((frac - 0.3) / 0.7) ** 1.5
                g = smooth_grad_inf(a, tau_curr)
        elif mode == 'l2':
            g = grad_l2(a)
        else:
            tau_curr = 5.0 + (tau_max - 5.0) * frac ** 1.5
            g = smooth_grad_inf(a, tau_curr)

        # Center the gradient (preserve mean)
        g -= g.mean()

        # Normalize gradient direction
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g_dir = g / g_norm

        # Adaptive step: smaller when close to convergence
        step_scale = 1.0 - 0.5 * frac
        step = base_step * step_scale

        # Nesterov momentum
        v = beta * v + step * g_dir
        a_new = a - v * step_scale
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to_sum(a_new, S_t)

        a = a_new

        # Check improvement periodically
        if (it + 1) % 50 == 0:
            s = compute_obj(a)
            if s < best_score:
                if it - last_improve > 2000:
                    beta = min(beta + 0.02, 0.95)
                best_score = s
                best = a.copy()
                last_improve = it
                stagnant_count = 0
            else:
                stagnant_count += 1
                # If stagnant for a while, try perturbation
                if stagnant_count > 40 and (it + 1) % 200 == 0:
                    rng_p = np.random.default_rng(it * 31 + n)
                    a_pert = a + 0.08 * rng_p.standard_normal(n)
                    a_pert = np.maximum(a_pert, 0.0)
                    a_pert = normalize_to_sum(a_pert, S_t)
                    s_pert = compute_obj(a_pert)
                    if s_pert < best_score:
                        best_score = s_pert
                        best = a_pert.copy()
                        a = a_pert.copy()
                        v = np.zeros(n)
                        stagnant_count = 0
                        last_improve = it
                    elif s_pert < compute_obj(a):
                        a = a_pert.copy()
                        v = np.zeros(n)

    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score


def run_lp_phase(a0, deadline, max_iters=50):
    """LP-based refinement phase."""
    n = len(a0)
    if n > 1024:
        return a0.copy(), compute_obj(a0)

    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    S_t = np.sqrt(2.0 * n)
    a = normalize_to_sum(a, S_t)
    best = a.copy()
    best_score = compute_obj(a)

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        c = np.convolve(best, best)
        M = float(c.max())

        g = lp_solve_max_sum(best, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        if g.sum() < 1e-10:
            break

        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1.0 - t) * best + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to_sum(a_new, S_t)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-8:
                best = a_new.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            # Try a small gradient nudge to escape
            tau = 50.0
            grad = smooth_grad_inf(best, tau)
            grad -= grad.mean()
            gn = np.linalg.norm(grad)
            if gn > 1e-12:
                a_new = best - 0.02 * np.sqrt(2.0 * n) * grad / gn
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize_to_sum(a_new, S_t)
                s_new = compute_obj(a_new)
                if s_new < best_score - 1e-8:
                    best = a_new.copy()
                    best_score = s_new
                else:
                    break

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Explores multiple starting sequences using
    combined L2/Linf gradient descent with momentum, adaptive temperature,
    periodic perturbations, and LP-based refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    # Collect starting sequences
    starts = []

    # Try to use previous best as a strong start
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            prev_n = len(gs)
            starts.append(('prior', gs.copy(), 'combined'))
            # Also try truncated/extended versions
            for L in [max(64, prev_n // 2), min(1024, prev_n * 2)]:
                if L >= 32 and L <= 2048:
                    if L <= prev_n:
                        starts.append(('prior_trunc', gs[:L].copy(), 'combined'))
                    else:
                        # Pad with small values
                        pad = np.ones(L - prev_n) * 0.1
                        extended = np.concatenate([gs, pad])
                        starts.append(('prior_pad', extended, 'combined'))
    except (NameError, Exception):
        pass

    # Diverse random starts
    for n in [128, 256, 384, 512, 640, 768]:
        a = rng.random(n) ** 2  # Biased towards small values
        starts.append(('rand', a, 'combined'))

    # Structured starts
    for n in [256, 512]:
        i = np.arange(n, dtype=np.float64)
        # Chirp
        starts.append(('chirp', 0.3 + 0.7 * (0.5 + 0.5 * np.cos(2 * np.pi * i * 7.0 / n)), 'combined'))
        # Raised sinc-like
        x = (i - n / 2) / (n / 4)
        starts.append(('sinc', np.abs(np.sinc(x * 0.5)) + 0.05, 'inf'))
        # Two-bump
        starts.append(('bump', np.exp(-((i - n * 0.3) / (n * 0.12)) ** 2) +
                                             np.exp(-((i - n * 0.7) / (n * 0.12)) ** 2), 'combined'))
        # Periodic
        starts.append(('periodic', 0.2 + 0.8 * (0.5 + 0.5 * np.sin(2 * np.pi * i * 11.0 / n)) ** 2, 'inf'))

    # L2-focused starts (for the L2 phase to smooth things out first)
    for n in [256, 512]:
        starts.append(('l2_rand', rng.random(n) * 0.3 + 0.1, 'l2'))

    # Sort: prioritized starts first (previous best), then shorter, then longer
    starts.sort(key=lambda x: (0 if x[0] == 'prior' else 1, len(x[1])))

    print(f"Total starts: {len(starts)}, time budget: {budget_s}s")
    print(f"Sequence lengths: {[len(s[1]) for s in starts]}")

    # Track total time spent
    for idx, (name, a0, mode) in enumerate(starts):
        if time.monotonic() > deadline - 90.0:
            break

        t0 = time.monotonic()
        n = len(a0)

        # Time allocation: give more time to promising starts
        remaining = deadline - time.monotonic()
        n_left = len(starts) - idx
        if name == 'prior':
            time_alloc = min(remaining * 0.35, 600.0)
        elif n <= 384:
            time_alloc = min(remaining / max(n_left, 1) * 1.2, 180.0)
        else:
            time_alloc = min(remaining / max(n_left, 1) * 1.0, 240.0)
        time_alloc = max(time_alloc, 30.0)
        sub_deadline = time.monotonic() + time_alloc

        # Run gradient phase
        if mode in ('combined', 'inf', 'l2'):
            seq, score = run_gradient_phase(a0, sub_deadline, n_iters=15000, mode=mode)
        else:
            seq, score = run_gradient_phase(a0, sub_deadline, n_iters=15000, mode='combined')

        # Run LP phase if time allows
        if n <= 1024 and time.monotonic() < sub_deadline - 3.0 and score < best_score + 0.05:
            lp_deadline = min(sub_deadline, deadline - 30.0)
            seq, score = run_lp_phase(seq, lp_deadline, max_iters=25)

        elapsed = time.monotonic() - t0
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  [{idx+1:2d}/{len(starts)}] {name:10s} n={n:4d} score={score:.6f} "
                  f"({elapsed:.0f}s) *** NEW BEST ***")
        else:
            print(f"  [{idx+1:2d}/{len(starts)}] {name:10s} n={n:4d} score={score:.6f} "
                  f"({elapsed:.0f}s) best={best_score:.6f}")

    # ===== Final refinement phase =====
    if best_seq is not None and time.monotonic() < deadline - 30.0:
        n = len(best_seq)
        print(f"\n=== Final refinement (n={n}, score={best_score:.6f}) ===")

        # Alternate between gradient polish and LP
        for pass_i in range(20):
            if time.monotonic() > deadline - 15.0:
                break

            pass_deadline = min(deadline - 10.0, time.monotonic() + 90.0)

            if pass_i % 3 != 2 and n <= 1024:
                seq, score = run_lp_phase(best_seq, pass_deadline, max_iters=10)
            else:
                seq, score = run_gradient_phase(best_seq, pass_deadline,
                                                 n_iters=5000, mode='inf')

            if score < best_score - 1e-9:
                best_score = score
                best_seq = seq.copy()
                print(f"  Pass {pass_i}: {score:.6f}")
            else:
                # Try perturbation-based escape
                rng_r = np.random.default_rng(pass_i * 104729 + n)
                for attempt in range(3):
                    scale = 0.03 * (attempt + 1)
                    perturbed = best_seq + scale * rng_r.standard_normal(n)
                    perturbed = np.maximum(perturbed, 0.0)
                    perturbed = normalize_to_sum(perturbed)
                    s_p = compute_obj(perturbed)
                    if s_p < best_score:
                        best_score = s_p
                        best_seq = perturbed.copy()
                        print(f"  Pass {pass_i}: perturbation -> {best_score:.6f}")
                        break
                else:
                    # Try L2 smoothing of the best
                    l2_seq, s_l2 = run_gradient_phase(best_seq,
                                                       min(deadline - 10.0, time.monotonic() + 30.0),
                                                       n_iters=3000, mode='l2')
                    if s_l2 < best_score:
                        best_score = s_l2
                        best_seq = l2_seq.copy()
                        print(f"  Pass {pass_i}: L2 polish -> {best_score:.6f}")
                    else:
                        print(f"  Pass {pass_i}: no improvement, stopping refinement")
                        break

    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, local_score={final_score:.6f}, "
              f"tracked_best={best_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Fallback: uniform n=256")
    return [1.0] * 256
