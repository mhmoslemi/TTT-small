Multi-start smoothed subgradient descent with FFT-accelerated gradients
to minimize peak autocorrelation of non-negative sequences.
Uses temperature annealing, backtracking line search, and diverse
anti-symmetric/start profiles to escape local minima.
"""

import numpy as np
import time

def _smooth_max(b, T):
    """Compute T * log(sum(exp(b/T))) as a smooth approximation to max(b)."""
    b_shifted = b - np.max(b)
    return np.max(b) + T * np.log(np.sum(np.exp(b_shifted / T)))

def _grad_smooth_max(a, T):
    """
    Compute gradient of T*log(sum(exp(b_k/T))) w.r.t. a,
    where b = convolve(a, a).
    Returns gradient vector of length n.
    """
    n = len(a)
    b = np.convolve(a, a)  # length 2n-1
    b_shifted = b - np.max(b)
    p = np.exp(b_shifted / T)
    p /= np.sum(p)
    # Gradient: g_j = 2*T * sum_k p_k * a_{k-j} for valid k-j
    # = 2*T * convolve(a, p)[n-1 : 2n-1]
    conv = np.convolve(a, p)
    grad = 2.0 * T * conv[n-1 : 2*n-1]
    return grad

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n+1) > (cssv - target))[0])
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) / sum(a)^2."""
    n = len(a)
    b = np.convolve(a, a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    return 2.0 * n * np.max(b) / (s * s)

            best_a = a.copy()

"""
Multi-start smoothed subgradient descent with FFT-accelerated gradients and
temperature annealing to minimize peak autocorrelation of non-negative sequences.
Uses diverse anti-palindromic and smooth profiles as starting points, adaptive
step sizes with backtracking, and a three-phase (explore/exploit/polish) strategy.
"""

import numpy as np
import time


def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) / sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (s * s)


def _compute_score_normalized(a):
    """Compute 2*n*max(convolve(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_smoothed_max(a, T):
    """
    Compute gradient of T*log(sum(exp(b_k/T))) w.r.t. a,
    where b = convolve(a, a). Uses convolution for efficiency.
    """
    n = len(a)
    b = np.convolve(a, a)  # length 2n-1
    b_max = np.max(b)
    p = np.exp((b - b_max) / T)
    p_sum = np.sum(p)
    p /= p_sum
    # grad_j = 2*T * sum_k p_k * a_{k-j} for valid (k-j) in [0, n-1]
    # This equals 2*T * convolve(a, p)[n-1 : 2n-1]
    conv = np.convolve(a, p)
    grad = 2.0 * T * conv[n - 1:2 * n - 1]
    return grad


def _smoothed_subgradient(a_init, deadline, rng, T_init=0.5, T_min=0.005,
                          cool_every=20, cool_rate=0.97, eta_init=None,
                          min_eta=1e-10, reheat_thresh=500, jump_scale=0.08):
    """
    Smoothed subgradient descent with adaptive step size.
    """
    n = len(a_init)
    if n == 0:
        return a_init, np.inf
    if eta_init is None:
        eta_init = 0.05 / np.sqrt(n)

    a = np.asarray(a_init, dtype=np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _compute_score_normalized(a)
    T = T_init
    eta = eta_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        # Compute current score
        b = np.convolve(a, a)
        cur_score = 2.0 * n * np.max(b)

        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Reheat / jump when stuck
        if no_improve >= reheat_thresh:
            if eta > min_eta * 10:
                eta *= 0.5
            else:
                # Jump: perturb and restart
                noise = rng.normal(0, jump_scale / np.sqrt(n), n)
                a = _project_simplex(a + noise, 1.0)
                T = max(T * 2, T_init * 0.3)
                eta = eta_init * 0.5
                no_improve = 0
                if eta < min_eta:
                    break

        # Compute gradient
        grad = _grad_smoothed_max(a, T)
        grad -= np.mean(grad)  # center for sum constraint

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        direction = -grad / grad_norm

        # Adaptive step: try a few shrinks
        improved = False
        for shrink in range(4):
            step = eta * (0.5 ** shrink)
            a_new = _project_simplex(a + step * direction, 1.0)
            b_new = np.convolve(a_new, a_new)
            score_new = 2.0 * n * np.max(b_new)
            if score_new < cur_score * (1.0 + 1e-10):
                a = a_new
                cur_score = score_new
                improved = True
                break

        if not improved:
            eta *= 0.7

        # Cool temperature periodically
        it += 1
        if it % cool_every == 0:
            T = max(T * cool_rate, T_min)

    return best_a, best_score


def _make_anti_palindromic_smooth(n, eps, rng=None):
    """
    Create an anti-palindromic smooth sequence: a_i + a_{n-1-i} = 1 (after normalization).
    Uses a smooth envelope to control the profile.
    """
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    # Base: 0.5 + 0.5*eps*cos(pi * t) gives anti-palindromic structure
    a = 0.5 + 0.5 * eps * np.cos(np.pi * t)
    # Add some higher frequency modulation
    if rng is not None:
        for freq in rng.choice([2, 3, 4, 5], size=2, replace=False):
            phase = rng.uniform(0, np.pi)
            amp = rng.uniform(0.05, 0.15)
            a += amp * np.cos(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_anti_palindromic_sharp(n, frac, rng=None):
    """
    Anti-palindromic with sharp transition: first frac is high, rest is low.
    a_i + a_{n-1-i} = 1 (before normalization).
    """
    a = np.ones(n)
    mid = n // 2
    width = int(frac * n)
    a[:mid - width // 2:mid + width // 2] = 1.5
    a[mid - width // 2:mid + width // 2 + 1] = 0.0
    if mid + width // 2 < n:
        a[mid + width // 2:] = 0.5
    # Make it properly anti-palindromic: a_i + a_{n-1-i} = constant
    const = 1.0
    for i in range(n // 2):
        avg = (a[i] + a[n - 1 - i]) / 2.0
        a[i] = avg + (const - 2 * avg) / 2
        a[n - 1 - i] = const - a[i]
    if n % 2 == 1:
        a[n // 2] = const / 2
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_cosine_profile(n, alpha, beta):
    """Smooth cosine-based profile."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = 1.0 + alpha * np.cos(2 * np.pi * t) + beta * np.cos(4 * np.pi * t)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_two_level_anti(n, eps):
    """Two-level anti-palindromic: first half = 1+eps/2, second half = 1-eps/2."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps / 2
    a[n // 2:] = 1.0 - eps / 2
    s = np.sum(a)
    return a / s


def _perturb(a, rng, scale=0.1):
    """Add smooth random perturbation."""
    n = len(a)
    freqs = rng.integers(1, max(2, n // 3), size=min(8, n // 2))
    noise = np.zeros(n)
    t = np.arange(n, dtype=np.float64) / n

        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.5, 1.0)

    noise /= (np.max(np.abs(noise)) + 1e-15)
    a_new = _project_simplex(a + scale * noise, 1.0)
    return a_new

        print(f"\n=== Phase 4: Coordinate refinement ({remaining:.0f}s) ===")
        a = global_best.copy()
        eta_coord = 0.02 / np.sqrt(n)
        no_improve_cd = 0

        while time.monotonic() < deadline:
            b = np.convolve(a, a)
            b_max = np.max(b)
            cur_score = 2.0 * n * b_max

            if cur_score < global_best_score - 1e-10:
                global_best_score = cur_score
                global_best = a.copy()
                no_improve_cd = 0
            else:
                no_improve_cd += 1
                if no_improve_cd > 500:
                    break

            # Find peak location and reduce contribution
            k_peak = np.argmax(b)
            # Gradient at peak
            grad = np.zeros(n)
            lo = max(0, k_peak - n + 1)
            hi = min(n - 1, k_peak)
            idx = np.arange(lo, hi + 1)
            grad[idx] = 2.0 * a[k_peak - idx]
            grad -= np.mean(grad)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            a_new = _project_simplex(a - eta_coord * grad / grad_norm, 1.0)
            b_new = np.convolve(a_new, a_new)
            if np.max(b_new) < b_max:
                a = a_new
            else:
                eta_coord *= 0.5
                if eta_coord < 1e-14:
                    break

        print(f"  After coord refinement: Score={global_best_score:.6f}")

    # Final evaluation
    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    elapsed = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {final_score:.6f} (elapsed: {elapsed:.1f}s)")
    return result
