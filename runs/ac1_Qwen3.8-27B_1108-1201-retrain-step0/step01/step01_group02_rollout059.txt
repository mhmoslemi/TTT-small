"""
Gradient-based optimization with smooth max for minimizing peak auto-convolution.
Uses multiple initializations (U-shape, raised cosine, provided sequence) with
adaptive temperature softmax gradients and backtracking line search.
"""

import numpy as np
import time

def score_sequence(a):
    """Compute the evaluation score."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def init_u_shape(n):
    """U-shaped initialization: high at edges, low at center."""
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = (t - 0.5) ** 2
    return normalize(a)

def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return normalize(0.5 * (1.0 + np.cos(2.0 * np.pi * t / n)))

def init_bimodal(n, rng, k_bumps=2):
    """Bimodal or multi-modal initialization."""
    a = np.zeros(n)
    for _ in range(k_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n//20), max(2, n//4)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    return normalize(a)

def init_from_sequence(seq, n_target=None):
    """Resize a sequence to target length."""
    seq = np.array(seq, dtype=np.float64)
    if n_target is None:
        return normalize(seq)
    # Interpolate
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, n_target)
    a = np.interp(x_new, x_old, seq)
    return normalize(a)

def gradient_descent_smooth(a, max_time=120.0, T_init=0.05, T_decay=0.999, T_min=0.001, lr_init=0.1, verbose=False):
    """Gradient descent with smooth max approximation."""
    start = time.monotonic()
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = score_sequence(a)
    T = T_init
    lr = lr_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        conv = np.convolve(a, a)
        max_b = conv.max()
        current_score = 2.0 * n * max_b  # since sum=1
        
        if current_score < best_score - 1e-10:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            # Restart with lower temperature
            T = max(T_min, T * 0.5)
            lr = lr_init
            no_improve = 0
        
        # Compute softmax weights
        conv_shifted = conv - max_b
        w = np.exp(conv_shifted / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient using strided matrix trick
        W = np.lib.stride_tricks.as_strided(w, shape=(n, n), strides=(w.itemsize, w.itemsize))
        g = 2.0 * (W @ a)
        
        # Project gradient onto tangent space of simplex
        # (remove the component along the all-ones direction)
        g = g - g.mean()
        
        # Take step
        a_new = a - lr * g
        a_new = np.maximum(0.0, a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.1
            T = max(T_min, T * 0.9)
            continue
        a_new = a_new / s_new
        
        new_score = 2.0 * n * np.convolve(a_new, a_new).max()
        
        if new_score < current_score + 1e-10:
            a = a_new
            # Decay step size slightly
            lr = max(0.001, lr * 0.999)
        else:
            # Backtracking
            success = False
            for _ in range(8):
                lr *= 0.5
                a_new = a - lr * g
                a_new = np.maximum(0.0, a_new)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    break
                a_new = a_new / s_new
                new_score = 2.0 * n * np.convolve(a_new, a_new).max()
                if new_score < current_score + 1e-10:
                    a = a_new
                    success = True
                    break
            if not success:
                T = max(T_min, T * 0.9)
        
        # Decay temperature
        T = max(T_min, T * T_decay)
        it += 1
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    GD iter {it}: score={score_sequence(a):.6f}, T={T:.5f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")
    
    # Final check
    final_score = score_sequence(best_a)
    if final_score < best_score:
        best_score = final_score
    
    return best_a, best_score

def peak_hunt_refine(a, max_time=60.0, verbose=False):

            best_score = current_score
            best_a = a.copy()

            valid_i = np.arange(k_star - n + 1, n)
        
        # Randomly pick a contributing element and try to reduce it
        idx = int(np.random.randint(0, len(valid_i)))
        j = int(valid_i[idx])

        # Try reducing a[j] (or a[partner] if different)
        reduction = np.random.uniform(0.05, 0.3)
        a_new = a.copy()
        a_new[j] *= (1.0 - reduction)
        a_new = normalize(a_new)
        new_score = score_sequence(a_new)
        
        if new_score < current_score:
            a = a_new
        else:
            # Try a random perturbation
            j2 = int(np.random.randint(0, n))
            a_new = a.copy()
            a_new[j2] *= np.random.uniform(0.5, 1.5)
            a_new = normalize(a_new)
            new_score = score_sequence(a_new)
            if new_score < current_score:
                a = a_new
        it += 1
    
    return best_a, best_score
