```
def _conv_poly(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2*n-1
    if n <= 64:
       c = np.convolve(a,a,mode='full')
    else:
       nfft = _next_pow2(L)
       fa = np.fft.rfft(a, n=nfft)
       c = np.fft.irfft(fa*fa, n=nfft)[:L]
    c = np.asarray(c, dtype=np.float64)
    c[c < 0.0] = 0.0
    S = a.sum()
    csum = c.sum()
    if csum > 0.0 and S > 0.0:
       c *= (S*S)/csum
    return c
```

```
def _smooth_gradient(c, a, beta_dim):
    n = a.size; L = 2*n-1
    cmax = float(c.max())
    cmin = float(c.min())
    scale = cmax - cmin
    if scale <= 1e-30:
       p = np.full(L, 1.0/L, dtype=np.float64)
    else:
       z = (c - cmax) * (beta_dim / scale)
       p = np.exp(z)
       s = p.sum()
       if not np.isfinite(s) or s <= 0.0:
          p = np.zeros(L); p[int(np.argmax(c))] = 1.0
       else:
          p /= s
    a_rev = a[::-1]
    need = n + L - 1
    nfft = _next_pow2(need)
    fa = np.fft.rfft(a_rev, n=nfft)
    fp = np.fft.rfft(p, n=nfft)
    g = np.fft.irfft(fa*fp, n=nfft)
    grad = 2.0 * g[n-1:n-1+n]
    return grad
```

```
def _adam_step(a, grad, state, lr):
    m = state['m']; v=state['v']; t=state['t']+1
    b1=0.9; b2=0.999; eps=1e-8
    m = b1*m + (1-b1)*grad
    v = b2*v + (1-b2)*(grad*grad)
    m_hat = m/(1-b1**t)
    v_hat = v/(1-b2**t)
    delta = lr*m_hat/(np.sqrt(v_hat)+eps)
    a_new = a - delta
    state['m']=m; state['v']=v; state['t']=t
    return a_new
```

```python
"""
Search for a non-negative step-function height sequence minimizing

    2*n*max(np.convolve(a, a)) / (sum(a))^2

using a multi-start local search.  The main components are:

1. FFT-based convolution and objective evaluation.
2. Smoothed-max / active-set subgradient steps with Adam-like scalars.
3. Periodic linear-programming moves:
   - min t  s.t. A(g) <= t, sum(g)=1, g>=0
   - max sum(g)  s.t. A(g) <= current peak, 0<=g<=10
   These are inspired by the previous SOTA LP iteration.
4. Several structured and random initializations, including the supplied
   height_sequence_1 global when available.
5. Time-budgeted exploration and final polishing.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _next_pow2(m: int) -> int:
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).copy()
    s = float(a.sum())
    if s <= 1e-300:
        return np.full(a.shape, target / max(1, a.size), dtype=np.float64)
    a *= target / s
    return a


def _conv_poly(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1

    if n <= 64:
        c = np.convolve(a, a, mode="full")
    else:
        nfft = _next_pow2(L)
        fa = np.fft.rfft(a, n=nfft)
        c = np.fft.irfft(fa * fa, n=nfft)[:L]

    c = np.asarray(c, dtype=np.float64)
    c[c < 0.0] = 0.0

    s = float(a.sum())
    csum = float(c.sum())
    if csum > 0.0 and s > 0.0:
        c *= (s * s) / csum
    return c


def _score(a, c=None):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    if c is None:
        c = _conv_poly(a)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    return float(2.0 * a.size * float(c.max()) / (s * s))


def _gradient_from_density(p, a):
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = a.size
    L = p.size
    need = n + L - 1
    nfft = _next_pow2(need)

    ar = a[::-1]
    fa = np.fft.rfft(ar, n=nfft)
    fp = np.fft.rfft(p, n=nfft)
    conv = np.fft.irfft(fa * fp, n=nfft)

    # grad_i = 2 * sum_j a_j * p_{i+j}
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad


def _smooth_gradient(c, a, beta_dim=20.0):
    c = np.asarray(c, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    L = 2 * a.size - 1

    cmax = float(c.max())
    cmin = float(c.min())
    scale = cmax - cmin

    if scale <= 1e-30:
        p = np.full(L, 1.0 / L, dtype=np.float64)
    else:
        z = (c - cmax) * (float(beta_dim) / scale)
        p = np.exp(z)
        ps = float(p.sum())
        if not np.isfinite(ps) or ps <= 0.0:
            p = np.zeros(L, dtype=np.float64)
            p[int(np.argmax(c))] = 1.0
        else:
            p /= ps

    return _gradient_from_density(p, a)


def _top_gradient(c, a, K=3, beta_dim=25.0):
    c = np.asarray(c, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    L = 2 * a.size - 1

    K = int(K)
    K = max(1, min(K, L))

    if K == L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(-c, K - 1)[:K]

    if K == 1:
        p = np.zeros(L, dtype=np.float64)
        p[int(idx[0])] = 1.0
        return _gradient_from_density(p, a)

    cp = c[idx]
    cmin = float(cp.min())
    cmax = float(cp.max())
    scale = cmax - cmin

    if beta_dim > 0.0 and scale > 1e-30:
        z = (cp - cmax) * (float(beta_dim) / scale)
        w = np.exp(z)
        ws = float(w.sum())
        if not np.isfinite(ws) or ws <= 0.0:
            w = np.ones(K, dtype=np.float64)
        else:
            w /= ws
    else:
        w = np.ones(K, dtype=np.float64)

    p = np.zeros(L, dtype=np.float64)
    p[idx] = w
    p /= float(p.sum())
    return _gradient_from_density(p, a)


def _adam_step(a, grad, state, lr):
    m = state["m"]
    v = state["v"]
    t = int(state["t"]) + 1

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8

    m = b1 * m + (1.0 - b1) * grad
    v = b2 * v + (1.0 - b2) * (grad * grad)

    mh = m / (1.0 - (b1 ** t))
    vh = v / (1.0 - (b2 ** t))

    a_new = a - lr * mh / (np.sqrt(vh) + eps)

    state["m"] = m
    state["v"] = v
    state["t"] = t
    return a_new


def _build_conv_matrix(a, include_t=False):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    extra = 1 if include_t else 0
    A = np.zeros((L, n + extra), dtype=np.float64)

    if include_t:
        A[:, 0] = -1.0

    base = extra
    for m in range(L):
        lo = 0 if m < n - 1 else m - n + 1
        hi = n - 1 if m >= n - 1 else m
        if hi >= lo:
            seg = a[m - hi:m - lo + 1][::-1]
            A[m, base + lo:base + hi + 1] = seg

    return A


def _lp_min_peak(a, max_time=10.0, nmax=600):
    if linprog is None:
        return None

    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n > int(nmax):
        return None

    try:
        A = _build_conv_matrix(a, include_t=True)
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, 1:] = 1.0

        c_obj = np.zeros(n + 1, dtype=np.float64)
        c_obj[0] = 1.0

        bounds = [(0.0, None)] * (n + 1)

        options = {}
        if max_time is not None and max_time > 0.0:
            options["time_limit"] = float(max_time)

        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=np.zeros(2 * n - 1, dtype=np.float64),
            A_eq=A_eq,
            b_eq=np.array([1.0], dtype=np.float64),
            bounds=bounds,
            method="highs",
            options=options,
        )

        if res.status == 0:
            g = np.asarray(res.x[1:], dtype=np.float64)
            g = np.maximum(g, 0.0)
            s = float(g.sum())
            if s > 1e-12:
                return g / s
    except Exception:
        pass

    return None


def _lp_sota(a, M, max_time=10.0, nmax=600):

    a = np.asarray(a, dtype=np.float64)
    n = a.size

    try:

        options = {}

        res = linprog(
            c_obj,

            g = np.maximum(g, 0.0)

    except Exception:
        pass

    return None


def _resample(arr, n):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.ones(int(n), dtype=np.float64)
    if arr.size == int(n):
        return arr.copy()

    x = np.linspace(0.0, 1.0, arr.size, dtype=np.float64)
    y = np.linspace(0.0, 1.0, int(n), dtype=np.float64)
    return np.interp(y, x, arr).astype(np.float64)


def _make_starts(n, rng, parent=None):
    n = int(n)
    starts = []
    i = np.arange(n, dtype=np.float64)

    if n > 1:
        center = 2.0 * i / float(n - 1) - 1.0
    else:
        center = np.zeros(1, dtype=np.float64)

    ac = np.abs(center)

    if parent is not None:
        starts.append(_resample(parent, n))

    starts.append(np.ones(n, dtype=np.float64))

    for alpha in (0.5, 1.0, 2.0):
        for beta in (0.5, 1.5, 4.0):
            starts.append(1.0 + beta * np.power(ac + 1e-8, alpha))

    for alpha in (0.3, 0.7, 1.2):
        x = i + 0.5
        y = float(n) + 0.5 - x
        starts.append(np.power(x, -alpha) + np.power(y, -alpha))

    if n <= 1024:
        freqs = (1, 3, 7, 17, 41, 97)
    else:
        freqs = (1, 3, 7, 17)

    for freq in freqs:
        for amp in (0.4, 0.7, 0.95):
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            env = 1.0 + 0.8 * ac
            a = env * (1.0 + amp * np.cos(2.0 * np.pi * float(freq) * i / float(n) + phase))
            a = np.maximum(a, 0.0) + 1e-5
            starts.append(a)

    nrand = 3 if n <= 1024 else 2
    for _ in range(nrand):
        f1 = float(rng.uniform(1.0, 20.0))
        f2 = float(rng.uniform(1.0, 50.0))
        p1 = float(rng.uniform(0.0, 2.0 * np.pi))
        p2 = float(rng.uniform(0.0, 2.0 * np.pi))
        a = (
            np.abs(np.sin(2.0 * np.pi * f1 * i / float(n) + p1))
            + np.abs(np.sin(2.0 * np.pi * f2 * i / float(n) + p2))
            + 0.2
        )
        starts.append(a)

    out = []
    for a in starts:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size != n:
            continue
        a = np.maximum(a, 0.0)
        if not np.all(np.isfinite(a)):
            continue
        if float(a.sum()) <= 1e-12:
            continue
        out.append(_normalize(a, 1.0))

    return out

            c = _conv_poly(a)
            score = float(2.0 * n * float(c.max()))
            state = {
                "m": np.zeros(n, dtype=np.float64),

                "t": 0,

            continue

        a_new = _adam_step(a, grad, state, lr)
        a_new = _normalize(np.maximum(a_new, 0.0), 1.0)
        c = _conv_poly(a_new)
        score = float(2.0 * n * float(c.max()))

        if not np.isfinite(score):
            a = best_a.copy()
            c = _conv_poly(a)

                    g = _lp_min_peak(a, max_time=max_time, nmax=lp_nmax)

                    best_candidate = None
                    best_candidate_c = None

def propose_candidate(seed=42, budget_s=2400, **kwargs):

    try:

    except Exception:
        b = 2400.0

    for key in ("time", "budget", "time_limit"):
        if key in kwargs:
            try:
                b = min(b, float(kwargs[key]))
            except Exception:
                pass

    b = max(180.0, b)
    deadline = t0 + max(0.0, b - 15.0)

    rng = np.random.default_rng(seed)

    parent = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size >= 4:
            raw = np.where(np.isfinite(raw), raw, 0.0)
            raw = np.maximum(raw, 0.0)
            if float(raw.sum()) > 0.0:
                parent = raw
    except Exception:
        parent = None

    best_a = None
    best_score = float("inf")

    if parent is not None:
        p = _normalize(np.maximum(parent, 0.0), 1.0)
        sc = _score(p, _conv_poly(p))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = p.copy()

        if parent.size > 1024:
            p2 = _normalize(np.maximum(_resample(parent, 1024), 0.0), 1.0)
            sc2 = _score(p2, _conv_poly(p2))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_a = p2.copy()

    print(f"initial_best={best_score if best_score != float('inf') else 'inf'} parent_len={0 if parent is None else int(parent.size)}")

    queue = []
    seen = []

    if parent is not None:
        plen = int(parent.size)
        if 64 <= plen <= 1536:
            L = plen
            a0 = _normalize(np.maximum(parent, 0.0), 1.0)
        elif plen > 1536:
            L = 1024
            a0 = _normalize(np.maximum(_resample(parent, L), 0.0), 1.0)
        else:
            L = 512
            a0 = _normalize(np.maximum(_resample(parent, L), 0.0), 1.0)

        queue.append((L, a0, "parent"))
        seen.append(L)

        for L in (512, 1024, 256):
            if L not in seen:
                a0 = _normalize(np.maximum(_resample(parent, L), 0.0), 1.0)
                queue.append((L, a0, f"parent{L}"))
                seen.append(L)

    gen_lengths = [512, 1024, 256]
    if b > 1200:
        gen_lengths.append(1500)
        gen_lengths.append(2048)
    gen_lengths.append(128)

    for L in gen_lengths:
        if L in seen or L < 32:
            continue
        try:
            starts = _make_starts(L, rng, parent)
            scored = []
            for idx, a in enumerate(starts):
                sc = _score(a, _conv_poly(a))
                if np.isfinite(sc):
                    scored.append((sc, idx, a))
            scored.sort()

            take = 1 if L >= 1500 else 2
            for k in range(min(take, len(scored))):
                queue.append((L, scored[k][2], f"gen{L}"))
        except Exception as e:
            print(f"start failure L={L}: {e}")
        seen.append(L)

    qi = 0

    if qi < len(queue):
        L, a0, tag = queue[qi]
        qi += 1

        long_budget = min(1200.0, max(240.0, 0.45 * max(100.0, b - 300.0)))
        end = min(deadline - 300.0, t0 + long_budget)
        if end > time.monotonic() + 60.0:
            seq, sc = _optimize_one(a0, end, rng, tag=tag, use_lp=True, lp_every=90)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = seq
            print(f"after_first {tag} best={best_score:.6f}")

    if best_a is not None and b > 1500 and time.monotonic() < deadline - 300.0:
        nb = int(best_a.size)
        for amp in (0.025, 0.05, 0.08, 0.14):
            noise = amp * float(best_a.mean()) * rng.standard_normal(nb)
            a0 = np.maximum(best_a + noise, 0.0)
            a0 = _normalize(a0, 1.0)
            queue.append((nb, a0, f"pert{int(1000*amp)}"))

    while qi < len(queue) and time.monotonic() < deadline - 200.0:
        L, a0, tag = queue[qi]
        qi += 1

        if tag.startswith("parent"):
            alloc = 100.0
        elif tag.startswith("pert"):
            alloc = 110.0
        elif L >= 1500:
            alloc = 90.0
        elif L >= 1000:
            alloc = 150.0
        else:
            alloc = 110.0

        end = min(deadline - 180.0, time.monotonic() + alloc)
        if end > time.monotonic() + 45.0:
            seq, sc = _optimize_one(a0, end, rng, tag=tag, use_lp=(L <= 600), lp_every=80)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = seq
            print(f"after_{tag} n={L} best={best_score:.6f}")

    if best_a is not None and time.monotonic() < deadline - 80.0:
        final_end = deadline - 8.0
        seq, sc = _optimize_one(
            best_a,
            final_end,
            rng,
            tag="final",
            use_lp=(int(best_a.size) <= 600),
            lp_every=55,
        )
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = seq

    if best_a is None or best_a.size < 1 or not np.isfinite(best_score):
        fb = _normalize(np.ones(256, dtype=np.float64), 1.0)
        print("fallback uniform")
        return [float(x) for x in fb]

    best_a = np.maximum(np.asarray(best_a, dtype=np.float64).ravel(), 0.0)
    if float(best_a.sum()) < 0.01:
        best_a = _normalize(best_a, 1.0)

    try:
        ev = evaluate_sequence([float(x) for x in best_a])
        print(f"final_len={int(best_a.size)} final_custom={best_score:.6f} evaluator={ev}")
    except Exception:
        print(f"final_len={int(best_a.size)} final_custom={best_score:.6f}")

    return [float(x) for x in best_a]
```
