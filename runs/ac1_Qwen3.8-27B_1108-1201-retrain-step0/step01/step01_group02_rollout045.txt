"""
Soft-max gradient descent with LP polishing to minimize peak auto-convolution
of non-negative sequences. Uses FFT-accelerated gradient computation for
efficiency, combined with multiple initializations and sequence lengths.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def compute_grad_fft(a, p):
    """Compute grad[j] = sum_m p[j+m] * a[m] using FFT."""
    n = len(a)
    a_rev = a[::-1]
    # Full convolution of p (length 2n-1) and a_rev (length n) has length 3n-2
    size = 1
    needed = (2 * n - 1) + n - 1
    while size < needed:
        size *= 2
    A = np.fft.rfft(a_rev, size)
    P = np.fft.rfft(p, size)
    corr = np.fft.irfft(A * P, size)
    return corr[n - 1:n - 1 + n]


def softmax_gradient_step(a, beta):
    """Perform one softmax gradient descent step."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    exp_c = np.exp(beta * (c - max_c))
    p = exp_c / exp_c.sum()
    grad = compute_grad_fft(a, p) * beta
    return c, p, grad


def softmax_optimize(a, time_limit, verbose=True, beta_start=5.0, beta_end=100.0,
                     lr_start=0.05, lr_end=0.001, min_iters=100):
    """Optimize using softmax gradient descent with adaptive parameters."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_score = score_sequence(a)
    best_a = a.copy()
    
    # We'll do multiple "phases" with increasing beta
    phase_beta = beta_start
    phase = 0
    total_iters = 0
    
    while time.monotonic() - start_time < time_limit:
        phase = int((phase_beta - beta_start) / (beta_end - beta_start + 1e-10) * 10)
        # Learning rate schedule: decay within each phase, reset at new phase
        # Use a fixed number of iterations per phase
        iters_this_phase = 0
        phase_start_score = best_score
        
        for it in range(max(min_iters, 500)):
            elapsed = time.monotonic() - start_time
            if elapsed > time_limit:
                break
            
            t = it / max(1, min_iters)  # progress within phase [0, 1]
            lr = lr_start * (lr_end / lr_start) ** t  # geometric decay
            
            c, p, grad = softmax_gradient_step(a, phase_beta)
            
            # Center the gradient to preserve sum
            grad_centered = grad - grad.mean()
            
            # Update
            a_new = a - lr * grad_centered
            a_new = np.maximum(0.0, a_new)
            s = a_new.sum()
            if s < 1e-15:
                # Reset
                a_new = a.copy()
                it += 100  # skip ahead
                continue
            a_new = a_new / s
            
            # Check score
            new_score = score_sequence(a_new)
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
            
            a = a_new
            iters_this_phase += 1
            total_iters += 1
            
            if verbose and total_iters % 500 == 0:
                print(f"  SM iter {total_iters}: beta={phase_beta:.1f}, score={best_score:.6f}")
        
        # Increase beta for next phase
        phase_beta = min(beta_end, phase_beta * 1.5)
        if phase_beta >= beta_end:
            break
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"  SM done: best={best_score:.6f}, iters={total_iters}, {time.monotonic()-start_time:.1f}s")
    
    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=120.0, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    stagnant = 0
    it = 0
    max_iters = 500
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break

        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 30:

        remaining = time_limit_s - elapsed

        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g
        alpha = min(0.4, max(0.03, 1.0 / (1.0 + 0.2 * it)))

def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s

"""
Soft-max gradient descent with FFT acceleration and LP polishing to minimize
peak auto-convolution of non-negative sequences. The key innovation is using
a smooth surrogate (soft-max) of the max-convolution function, computing its
gradient efficiently via FFT-based cross-correlation, and combining this fast
gradient method with the precise LP direction-finding for final polishing.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def fft_autoconv(a):
    """Full auto-convolution via FFT for speed on large sequences."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def compute_grad_fft(a, p):
    """Compute grad[j] = sum_m p[j+m] * a[m] using FFT cross-correlation."""
    n = len(a)
    a_rev = a[::-1]
    needed = (2 * n - 1) + n - 1
    size = 1
    while size < needed:
        size *= 2
    A = np.fft.rfft(a_rev, size)
    P = np.fft.rfft(p, size)
    corr = np.fft.irfft(A * P, size)
    return corr[n - 1:n - 1 + n]


def softmax_gradient_step(a, beta, use_fft=True):
    """Perform one softmax gradient computation. Returns (conv, softmax_weights, gradient)."""
    n = len(a)
    if use_fft and n > 300:
        c = fft_autoconv(a)
    else:
        c = np.convolve(a, a)
    max_c = c.max()
    exp_c = np.exp(beta * (c - max_c))
    p = exp_c / exp_c.sum()
    grad = compute_grad_fft(a, p) * beta
    return c, p, grad


def softmax_optimize(a, time_limit, verbose=True, beta_start=5.0, beta_end=200.0,
                     lr_start=0.08, lr_end=0.002, iters_per_phase=200):

    start_time = time.monotonic()

    n = len(a)
    best_score = score_sequence(a)
    best_a = a.copy()
    total_iters = 0

    phase_beta = beta_start
    use_fft = n > 300

    while time.monotonic() - start_time < time_limit:
        phase_start_time = time.monotonic()
        phase_best = best_score
        no_improve = 0

        for it in range(iters_per_phase):
            elapsed = time.monotonic() - start_time
            if elapsed > time_limit:

            t = it / max(1, iters_per_phase)
            lr = lr_start * (lr_end / lr_start) ** t

            c, p, grad = softmax_gradient_step(a, phase_beta, use_fft)

            # Center gradient to preserve sum constraint
            grad_centered = grad - grad.mean()

            # Backtracking line search
            step_size = lr
            a_new = a - step_size * grad_centered
            a_new = np.maximum(0.0, a_new)
            s = a_new.sum()
            if s < 1e-15:

                a_new = a - step_size * grad_centered
                a_new = np.maximum(0.0, a_new)

                if s < 1e-15:
                    no_improve += 1
                    continue
            a_new = a_new / s

            new_score = score_sequence(a_new)
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                no_improve = 0
                a = a_new
            elif new_score < score_sequence(a) + 0.01 * (score_sequence(a) - best_score):
                # Accept if close to best (exploration)
                a = a_new
                no_improve += 1
            else:

                a_new = a - step_size * grad_centered
                a_new = np.maximum(0.0, a_new)
                s = a_new.sum()
                if s > 1e-15:

                            no_improve += 1
                    else:
                        no_improve += 1
                else:
                    no_improve += 1

            total_iters += 1
            if verbose and total_iters % 1000 == 0:
                print(f"    SM iter {total_iters}: beta={phase_beta:.1f}, best={best_score:.6f}")

        # Increase beta for next phase
        if no_improve > iters_per_phase * 0.8:
            phase_beta = min(beta_end, phase_beta * 2.0)
            lr_start = max(0.005, lr_start * 0.7)
        else:
            phase_beta = min(beta_end, phase_beta * 1.3)
            lr_start = max(0.005, lr_start * 0.9)

        if phase_beta >= beta_end:
            break

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    SM done: best={best_score:.6f}, iters={total_iters}, {time.monotonic()-start_time:.1f}s")

    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=120.0, verbose=True):
    """LP-based refinement: find best direction at each step."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    stagnant = 0
    it = 0
    max_iters = 300
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break

        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 25:
            break

        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(40.0, remaining * 0.2))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g
        alpha = min(0.5, max(0.05, 1.2 / (1.0 + 0.3 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-14:
            a = a_new
        else:
            alpha2 = alpha * 0.2
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it + 1} iters, {time.monotonic() - start_time:.1f}s")

    return best_a, best_score


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_hann(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * t / (n - 1)))


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def local_search_refine(a, time_limit_s=120.0, rng=None, verbose=True):
    """Local search with random perturbations."""
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(rng.integers(max(2, n // 200), max(3, n // 8))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.08, size=stop - start))
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = normalize_nonneg(candidate)
        sc = score_sequence(candidate)
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 200 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic() - start_time:.1f}s")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    Combines FFT-accelerated softmax gradient descent (fast, many iterations)
    with LP direction-finding (precise, fewer iterations) and local search.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 25.0)

    best_score = float('inf')
    best_seq = None
    best_n = 0

    # Load provided starting sequence
    try:
        provided = np.array(height_sequence_1, dtype=np.float64)
        provided = normalize_nonneg(provided)
        provided_score = score_sequence(provided)
        print(f"Provided sequence: n={len(provided)}, score={provided_score:.6f}")
        best_score = provided_score
        best_seq = provided.copy()
        best_n = len(provided)
    except (NameError, UnboundLocalError):
        pass

    # Search configurations: (n, init_type, time_fraction)
    configs = []
    if best_seq is not None:
        configs.append((best_n, 'provided', 0.15))
    configs.extend([
        (500, 'raised_cosine', 0.08),
        (1000, 'raised_cosine', 0.10),
        (2000, 'raised_cosine', 0.12),
        (4000, 'raised_cosine', 0.10),
        (500, 'gaussian', 0.05),
        (1000, 'gaussian', 0.07),
        (2000, 'gaussian', 0.08),
        (500, 'blackman', 0.04),
        (1000, 'hann', 0.06),
        (2000, 'hann', 0.07),
        (1000, 'random', 0.05),
        (2000, 'random', 0.06),
    ])

    for n, init_type, time_frac in configs:
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break
        time_limit = min(time_frac * budget_s, remaining * 0.92)
        if time_limit < 12.0:
            continue

        print(f"\n=== n={n}, init={init_type}, time={time_limit:.0f}s ===")

        if init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif init_type == 'gaussian':
            a = initial_gaussian(n, sigma_frac=0.25)
        elif init_type == 'blackman':
            a = initial_blackman(n)
        elif init_type == 'hann':
            a = initial_hann(n)
        elif init_type == 'random':
            a = random_smooth_init(n, rng, n_bumps=int(rng.integers(2, 6)))
        elif init_type == 'provided':
            a = best_seq.copy()
        else:
            a = initial_raised_cosine(n)

        a = normalize_nonneg(a)
        init_score = score_sequence(a)
        print(f"  Init score: {init_score:.6f}")

        # Phase 1: Softmax gradient descent
        sm_time = time_limit * 0.65
        a_refined, sm_score = softmax_optimize(a, time_limit_s=sm_time, verbose=True)

        # Phase 2: LP refinement
        lp_time = time_limit - sm_time - 5.0
        if lp_time > 8.0:
            a_refined, lp_score = lp_refine(a_refined, time_limit_s=lp_time, verbose=True)
        else:
            lp_score = sm_score

        if lp_score < best_score:
            best_score = lp_score
            best_seq = a_refined.copy()
            best_n = n
            print(f"  *** NEW BEST: score={lp_score:.6f}, n={n}, elapsed={time.monotonic() - total_start:.1f}s")

    # Final polishing phase
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40.0:
            print(f"\n=== Final polishing (remaining={remaining:.0f}s) ===")

            # Local search
            ls_time = remaining * 0.2
            polished, ls_score = local_search_refine(best_seq, time_limit_s=ls_time, rng=rng, verbose=True)
            if ls_score < best_score:
                best_score = ls_score
                best_seq = polished.copy()

            # High-beta softmax on polished
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20.0:
                polished2, sm2_score = softmax_optimize(best_seq, time_limit_s=remaining2 * 0.4,
                                                         verbose=True, beta_start=30.0, beta_end=500.0,
                                                         lr_start=0.03, lr_end=0.001, iters_per_phase=150)
                if sm2_score < best_score:
                    best_score = sm2_score
                    best_seq = polished2.copy()

            # Final LP on best
            remaining3 = deadline - time.monotonic()
            if remaining3 > 15.0:
                polished3, lp3_score = lp_refine(best_seq, time_limit_s=remaining3 * 0.7, verbose=True)
                if lp3_score < best_score:
                    best_score = lp3_score
                    best_seq = polished3.copy()

            # One more local search
            remaining4 = deadline - time.monotonic()
            if remaining4 > 10.0:
                polished4, ls4_score = local_search_refine(best_seq, time_limit_s=remaining4 * 0.5,
                                                           rng=rng, verbose=True)
                if ls4_score < best_score:
                    best_score = ls4_score
                    best_seq = polished4.copy()

    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)

    print(f"\nFinal result: score={best_score:.6f}, n={len(best_seq)}")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified score via evaluate_sequence: {verify_score:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")

    return [float(x) for x in best_seq]
