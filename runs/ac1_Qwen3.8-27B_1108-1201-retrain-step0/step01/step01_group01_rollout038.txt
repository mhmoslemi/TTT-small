    #                        ^this sum is over j where i+j < 2n-1

    # which is equivalent to a cross-correlation.

    # Define w_ext = w (length 2n-1), pad a to length 2n-1 with zeros.
    # Then sum_j w[i+j]*a[j] = (cumulative cross-correlation)
    
    # I'll use FFT:
    # The correlation corr[i] = sum_j w[j+i]*a[j] (j=0..n-1, w zero-padded beyond 2n-2)
    # This is the cross-correlation of w and a at lag i.
    #

    # = ifft(conjugate(FFT(x)) * FFT(y))[k] (with zero-padding for linear)

Objective: minimize f(a) = (1/t) * log(sum_k exp(t * b[k](a))) subject to a >= 0, sum(a) = 1.
(b[k](a) is the k-th element of convolve(a,a).)

The gradient is: grad = 2 * corr(w, a) where w = softmax(t * b) (computed with the b at the current iterate).

FISTA:

- x^{k+1} = Proj_simplex(y^{k+1} - (1/L) * grad_f(y^{k+1}))
- t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2

Where L is the Lipschitz constant of grad_f. We can estimate it or use backtracking.

For the Lipschitz constant: the Hessian of f is:
H = (1/t) * sum_k w_k * (H_k + t * g_k * g_k^T)

where H_k is the Hessian of b[k] and g_k = grad b[k].

The norm of H_k is bounded by 2 (since b[k] is a quadratic form with eigenvalues at most 2 * max(a)). So ||H|| ≤ (1/t) * sum_k w_k * (2 + t*||g_k||^2) = 2 + sum_k w_k * ||g_k||^2.

||g_k||^2 = 4 * sum_i a[k-i]^2 ≤ 4 * max(a) * sum_i a[k-i] ≤ 4 * max(a)^2 * n (roughly).

"""
Smoothed FISTA + momentum subgradient + simulated annealing to minimize
the normalized autocorrelation peak of a non-negative sequence.
Uses FFT-based gradient computation for the log-sum-exp smoothed objective,
adaptive temperature annealing, and structured perturbations for escape.
"""

import numpy as np
import time


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score(a):
    """Compute 2*n*max(convolve(a,a)) / sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    a_norm = a / S
    b = np.convolve(a_norm, a_norm)
    return 2.0 * n * np.max(b)


def _compute_gradient_fft(a, w):
    """
    Compute gradient of sum_k w[k]*b[k] wrt a, where b = convolve(a,a).
    Result: grad[i] = 2 * sum_j w[i+j] * a[j] (cross-correlation of w and a at lag i).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N *= 2
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(np.concatenate([a, np.zeros(N - n)]), N)
    corr = np.fft.irfft(np.conj(W) * A, N)
    return 2.0 * corr[:n]


def _smooth_grad(a, t_temp):
    """
    Compute gradient of (1/t)*log(sum_k exp(t*b[k])) using softmax weights.
    Returns (smooth_value, gradient, weights).
    """
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    expo = t_temp * (b - b_max)
    expo -= np.max(expo)  # numerical stability
    w = np.exp(expo)
    w_sum = np.sum(w)
    w /= w_sum
    smooth_val = b_max + np.log(w_sum) / t_temp
    grad = _compute_gradient_fft(a, w)
    return smooth_val, grad, w


def _fista_optimize(a_init, deadline, t_init=20.0, t_max=200.0, t_grow=1.002,
                    L_init=2.0, max_iter=None):
    """
    FISTA (accelerated projected gradient) on the simplex with smoothed max objective.
    Anneals the softmax temperature from t_init to t_max.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S

    n = len(a)
    best = a.copy()
    best_score = _score(best)

    y = a.copy()
    x_prev = a.copy()
    t_mom = 1.0
    L = L_init
    t_temp = t_init
    iter_count = 0

    while time.monotonic() < deadline:
        if max_iter is not None and iter_count >= max_iter:
            break
        iter_count += 1

        # Compute gradient at y
        smooth_val, grad_y, w = _smooth_grad(y, t_temp)

        # Center gradient (remove mean component for sum constraint)
        grad_y = grad_y - np.mean(grad_y)

        # Gradient step
        x_new = y - grad_y / L

        # Project onto simplex
        x_new = _project_simplex(x_new, 1.0)

        # Evaluate actual score
        cur_score = _score(x_new)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = x_new.copy()

        # FISTA momentum update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_next
        y = x_new + beta * (x_new - x_prev)
        y = _project_simplex(y, 1.0)

        x_prev = x_new.copy()
        t_mom = t_next

        # Anneal temperature
        t_temp = min(t_temp * t_grow, t_max)

        # Adaptive L: increase if we're making progress, decrease if stuck
        if cur_score > best_score + 1e-6 and iter_count % 100 == 0:
            L *= 0.9  # reduce step size
        elif iter_count % 500 == 0 and L < 50.0:
            L *= 1.05

    return best, best_score, iter_count


def _momentum_subgrad(a_init, deadline, eta_init=0.02, top_m=20,
                      momentum_init=0.9, momentum_decay=0.999):
    """
    Momentum subgradient descent on the simplex.
    Uses top-M peak weighting for the subgradient.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S

    n = len(a)
    best = a.copy()
    best_score = _score(best)

    momentum = np.zeros(n)
    mu = momentum_init
    eta = eta_init
    no_improve = 0

    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            eta *= 0.8
            no_improve = 0
            if eta < 1e-10:
                break

        # Top-M peaks with softmax weighting
        if top_m >= len(b):
            k_idx = np.arange(len(b))
            wts = np.ones(len(b)) / len(b)
        else:
            k_idx = np.argpartition(b, -top_m)[-top_m:]
            b_sel = b[k_idx]
            wts = np.exp(-10.0 * (b_max - b_sel) / max(b_max, 1e-15))
            wts /= np.sum(wts)

        # Compute weighted gradient using FFT
        # Build density: for each selected k, add w_k to position k
        # Then grad[i] = 2 * sum_k w_k * a[k-i] = 2 * cross_corr(weighted_w, a)
        w_full = np.zeros(len(b))
        for ki in range(len(k_idx)):
            if wts[ki] > 1e-10:
                w_full[k_idx[ki]] = wts[ki]

        grad = _compute_gradient_fft(a, w_full)
        grad -= np.mean(grad)

        # Momentum update
        momentum = mu * momentum - eta * grad
        a = a + momentum
        a = _project_simplex(a, 1.0)

        mu *= momentum_decay
        if mu < 0.5:
            mu = 0.5

    return best, best_score


def _simulated_annealing(a_init, deadline, T_init=0.01, T_min=0.0001,
                         cool_rate=0.9995, pert_scale=0.03):
    """
    Simulated annealing with structured (smooth) perturbations.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S

    n = len(a)
    best = a.copy()
    best_score = _score(best)

    T = T_init
    rng = np.random.default_rng(12345)
    accept_count = 0
    total_count = 0

    while time.monotonic() < deadline and T > T_min:
        # Generate smooth perturbation
        n_pert = rng.integers(5, max(6, n // 10))
        freqs = rng.integers(1, max(2, n // 5), size=min(3, n // 2))
        noise = np.zeros(n)
        for f in freqs:
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.normal(0, 1)
            noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
        noise /= (np.max(np.abs(noise)) + 1e-15)

        a_trial = a + pert_scale * noise
        a_trial = _project_simplex(a_trial, 1.0)

        trial_score = _score(a_trial)
        total_count += 1

        if trial_score < best_score - 1e-12:
            best_score = trial_score
            best = a_trial.copy()

        if trial_score <= _score(a) or rng.random() < np.exp((-trial_score + _score(a)) / T):
            a = a_trial
            accept_count += 1

        T *= cool_rate

    return best, best_score


def _make_bathtub(n, sharpness=2.0):
    """Create a bathtub-shaped sequence: high at ends, low in middle."""
    i = np.arange(n, dtype=np.float64)
    center = n / 2.0
    x = (i - center) / center  # ranges from -1 to 1
    a = (1.0 + sharpness * x * x)
    a = np.maximum(a, 0.01)
    S = np.sum(a)
    return a / S


def _make_two_level_asym(n, eps=0.5):
    """Two-level anti-symmetric: first half higher, second half lower."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _make_cosine_taper(n, eps=0.3):
    """Cosine tapered: high in middle, low at edges (opposite of bathtub)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.01)
    S = np.sum(a)
    return a / S


def _make_quadratic_bathtub(n, eps=0.5):
    """Quadratic bathtub: a[i] proportional to ((i - n/2)/(n/2))^2 + 1."""
    i = np.arange(n, dtype=np.float64)
    x = (i - (n - 1) / 2.0) / ((n - 1) / 2.0)
    a = 1.0 + eps * x * x
    S = np.sum(a)
    return a / S


def _perturb_antisymmetric(a, rng, scale=0.02):
    """Perturb with anti-symmetric smooth noise (preserves approximate anti-symmetry)."""
    n = len(a)
    noise = np.zeros(n)
    freqs = rng.integers(1, max(2, n // 8), size=3)
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.normal(0, 1)
        base = np.sin(2 * np.pi * f * np.arange(n) / n + phase)
        # Make anti-symmetric: use odd function
        noise += amp * base
    # Enforce anti-symmetry
    noise_anti = noise - noise[::-1]
    noise_anti *= 0.5
    a_new = a + scale * noise_anti
    return _project_simplex(a_new, 1.0)
