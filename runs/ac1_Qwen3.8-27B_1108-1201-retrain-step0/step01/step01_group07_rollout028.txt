import numpy as np
import time
from scipy.optimize import minimize

def compute_obj_and_grad(x, tau, eps_reg=1e-10):
    """
    Compute objective and gradient in log-domain.
    a = exp(x), objective = 2n * F(a) / S^2 + eps_reg * mean(x)^2
    where F(a) = tau * logsumexp(conv(a,a) / tau)
    """
    n = len(x)
    a = np.exp(x)
    S = np.sum(a)
    c = np.convolve(a, a)
    c_max = c.max()
    logsumexp_c = tau * np.log(np.sum(np.exp((c - c_max) / tau))) + c_max
    F = logsumexp_c
    
    obj = 2.0 * n * F / (S * S) + eps_reg * np.mean(x) ** 2
    
    # Gradient of F w.r.t. a
    q = np.exp((c - c_max) / tau)
    q /= q.sum()
    dF_da = np.correlate(q, a, 'valid')  # length n
    
    # Gradient of objective w.r.t. a
    dObj_da = 2.0 * n * dF_da / (S * S) - 4.0 * n * F / (S ** 3)
    
    # Chain rule: dObj/dx_i = dObj/da_i * a_i
    grad = dObj_da * a + 2.0 * eps_reg * np.mean(x)
    
    return obj, grad

def obj_and_grad(x, tau, eps=1e-8):
    n = len(x)
    m = x.max()
    a = np.exp(x - m)
    S = a.sum()
    c = np.convolve(a, a)
    c_max = c.max()
    # F = tau * logsumexp(c/tau)
    logsumexp_c = c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau)))
    F = logsumexp_c
    
    obj_val = 2.0 * n * F / (S * S) + eps * (x.sum() / n) ** 2
    
    # q = softmax(c / tau)
    q = np.exp((c - c_max) / tau)
    q /= q.sum()
    
    # dF/da = correlate(q, a, 'valid') — this gives length n
    dF_da = np.correlate(q, a, 'valid')
    
    # dg/da = 2n * dF_da / S^2 - 4n * F / S^3
    dg_da = 2.0 * n * dF_da / (S * S) - 4.0 * n * F / (S ** 3)
    
    # dg/dx = dg_da * a + 2*eps * mean(x)
    grad = dg_da * a + 2.0 * eps * (x.sum() / n)
    
    return obj_val, grad

import numpy as np

def fft_convolve(a):

    n = len(a)

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj_grad(x, tau, eps_reg=1e-12):
    """
    Compute objective and gradient in log-domain.
    a = exp(x - max(x)) for stability. Objective is scale-invariant.
    g(x) = 2n * F(a) / S^2 + eps_reg * mean(x)^2
    F(a) = tau * logsumexp(conv(a,a) / tau)
    """
    n = len(x)
    m = x.max()
    a = np.exp(x - m)
    S = a.sum()

    c = fft_convolve(a)
    c_max = c.max()

    final_score = evaluate_sequence(final_a.tolist())
    print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_a)} ===")

    return [float(v) for v in final_a]
