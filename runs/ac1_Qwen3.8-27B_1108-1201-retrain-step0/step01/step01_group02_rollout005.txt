"""
Gradient descent with soft-max smoothing to minimize peak auto-convolution
of non-negative sequences. Uses FFT-based gradient computation, beta scheduling,
projected gradient descent with non-negativity constraint, and local search polishing.
"""

import numpy as np
import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def compute_softmax_gradient(a, beta):
    """Compute gradient of logsumexp(beta * conv(a,a)) with respect to a.
    Returns array of same shape as a."""
    n = len(a)
    conv = np.convolve(a, a)  # length 2n-1
    # Softmax weights
    shifted = beta * conv - np.max(beta * conv)
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)  # length 2n-1
    
    # grad[i] = 2 * sum_k a[k] * p[i+k] for i in [0, n-1]
    # Use FFT for efficiency
    # We need: x = convolve(a[::-1], p_padded, 'full'), then grad = 2*x[n-1:n-1+n]
    
    # FFT size
    # a has length n, p has length 2n-1
    # result length = n + (2n-1) - 1 = 3n-2
    N = 1
    while N < 3 * n - 1:
        N *= 2
    
    A = np.fft.rfft(a[::-1], N)
    P = np.fft.rfft(p, N)
    x = np.fft.irfft(A * P, N)
    
    grad = 2.0 * x[n-1:2*n-1]
    return grad, conv, p

def gradient_descent_step(a, beta, lr):
    """One step of projected gradient descent minimizing logsumexp(beta*conv(a,a))."""
    n = len(a)
    # Normalize
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s * np.sqrt(2.0 * n)
    
    grad, conv, p = compute_softmax_gradient(a, beta)
    
    # Gradient step
    a_new = a - lr * grad
    
    # Project to non-negative
    a_new = np.maximum(0.0, a_new)
    
    # Renormalize
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
    a_new = a_new / s_new * np.sqrt(2.0 * n)
    
    return a_new

def soft_max_score(a, beta):
    """Compute soft approximation of score: 2*n*logsumexp(beta*conv)/sum^2"""
    n = len(a)
    conv = np.convolve(a, a)
    shifted = beta * conv - np.max(beta * conv)
    soft_max = np.max(beta * conv) + np.log(np.sum(np.exp(shifted))) / beta
    sum_a = a.sum()
    return 2.0 * n * soft_max / (sum_a ** 2)

def gd_optimize(a, time_limit_s=300.0, verbose=True):
    """Optimize using gradient descent with soft-max smoothing."""
    start_time = time.monotonic()
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s * np.sqrt(2.0 * n)
    
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    GD start: score={best_score:.6f}, n={n}")
    
    # Beta schedule: start small, increase
    beta_start = 1.0
    beta_end = 1000.0
    n_total_steps = 500
    
    # Learning rate schedule
    lr_start = 0.5
    lr_end = 0.01
    
    stagnant = 0
    for step in range(n_total_steps):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        # Progress fraction [0, 1]
        frac = step / n_total_steps
        
        # Beta increases (sharper approximation)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Learning rate decreases
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Take gradient step
        a_new = gradient_descent_step(a, beta, lr)
        
        # Evaluate
        new_score = score_sequence(a_new)
        
        if new_score < best_score - 1e-10:
            best_score = new_score
            best_a = a_new.copy()
            stagnant = 0
            if verbose and (step % 50 == 0 or step < 5):
                print(f"    GD step {step:4d}: score={new_score:.6f}, beta={beta:.1f}")
        else:
            a = a_new  # Still accept (helps with beta increase)
            stagnant += 1
        
        if stagnant > 100:
            if verbose:
                print(f"    GD stagnant at step {step}, best={best_score:.6f}")
            break
    
    return best_a, best_score

def adam_optimize(a, time_limit_s=300.0, verbose=True):

def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))

def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)

def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))

def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s
