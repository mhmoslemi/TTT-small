import numpy as np
import time

def fast_autocorrelation(a):
    """Compute np.convolve(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def objective_normalized(a):
    """Compute 2*n*max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    c = fast_autocorrelation(a)
    return 2.0 * n * c.max()

def optimize_adam(a0, time_limit, lr=0.01, top_k=10):
    """Adam-style optimization on the probability simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a_sum
    
    # Adam moments
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    
    best_a = a.copy()
    best_obj = objective_normalized(a)
    patience = 0
    max_patience = 200
    current_lr = lr
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        c = fast_autocorrelation(a)
        c_max = c.max()
        if c_max < 1e-15:
            break
        
        # Compute subgradient using top-k peaks
        # Soft assignment to top-k peaks
        top_idx = np.argsort(c)[-top_k:]
        weights = np.zeros(len(c))
        for idx in top_idx:
            weights[idx] = c[idx] / c_max
        weights /= weights.sum()
        
        # Subgradient: sg[i] = sum_k weights[k] * a[k-i] (for valid k-i)
        sg = np.zeros(n)
        active = np.where(weights > 1e-6)[0]
        for k_int in active:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end+1] += weights[k] * a[k - j_end:k - j_start + 1][::-1]
        
        # Scale gradient
        sg *= 2.0 * n
        
        # Adam update
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to simplex (non-negative, sum=1)
        # Simple projection: clip and normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = a.copy()
        else:
            a_new /= s
        
        a = a_new
        obj = objective_normalized(a)
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            # Reduce learning rate

            patience = 0
            if current_lr < 1e-5:

        per_reduce = total_reduce / len(top_contrib_idx)
        
        for j in top_contrib_idx:
            reduce_amount = min(a[j], per_reduce)
            a[j] -= reduce_amount
            total_reduce -= reduce_amount
        
        # Redistribute to random non-peak elements
        if total_reduce > 1e-12:
            # Find elements not in the top contributions
            non_peak = np.setdiff1d(np.arange(n), top_contrib_idx)
            if len(non_peak) > 0:
                a[np.random.choice(non_peak, size=1, replace=False)] += total_reduce
        
        # Clip and normalize
        a = np.maximum(a, 0)
        a /= a.sum()
    
    return a

- Note that `sum_j a[j]*a'[k-j]` is the cross-correlation of `a` and `a'` at lag k

- So the constraint becomes: `2 * (a ⊛ a')[k] <= t + c[k]` where `⊛` is cross-correlation and `c = conv(a,a)`
- We want to minimize t subject to these constraints and `sum(a') = 1`, `a' >= 0`

- Variables: `a'[0], ..., a'[n-1], t`

def optimize_mirror(a0, time_limit, initial_t=30.0, eta0=0.5):
    """Mirror descent with adaptive temperature."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_autocorr(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    eta = eta0
    t_param = initial_t
    patience = 0
    max_patience = 100
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        c_max = conv.max()
        if c_max < 1e-15:
            break
        
        # Softmax weights over convolution
        scaled = t_param * conv / c_max
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Subgradient
        sg = np.correlate(w, a, mode='valid')
        sg *= 2.0 * n
        
        # Center (remove mean to maintain sum constraint approximately)
        sg_centered = sg - sg.mean()
        
        # Multiplicative update
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        
        conv = fast_autocorr(a)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            patience = 0
            eta *= 0.7
            t_param *= 1.5
            if eta < 0.01:
                break
    
    return best_a, best_obj

def lp_refine(a, max_iters=50):
    """Refine using LP linearization of the bilinear constraints."""
    n = len(a)
    a = a.copy()
    
    for it in range(max_iters):
        c = fast_autocorr(a)
        c_max = c.max()
        current_obj = 2.0 * n * c_max
        
        # Formulate LP:
        # Minimize t
        # Subject to: 2 * sum_j a[k-j] * a'[j] <= t + c[k] for all k
        #             sum_j a'[j] = 1
        #             a'[j] >= 0
        #
        # Variables: [a'(0), ..., a'(n-1), t] (n+1 variables)
        # 
        # Inequality constraints: for each k (0 to 2n-2):
        #   sum_j 2*a[k-j]*a'[j] - t <= c[k]
        #   Row: [2*a[k-j] for valid j, 0, ..., 0, -1]
        #
        # Equality: sum a'[j] = 1
        #   Row: [1, 1, ..., 1, 0]
        
        # Build constraint matrix
        num_constraints = 2 * n - 1
        A = np.zeros((num_constraints, n + 1))
        b = c.copy()
        
        for k in range(num_constraints):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A[k, j] = 2.0 * a[k - j]
            A[k, n] = -1.0  # coefficient of t
        
        # This is O(n^2) to build, which is fine for n up to a few thousand
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [1.0]
        
        bounds = [(0, None)] * n + [(None, None)]
        
        result = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, 
                         bounds=bounds, method='highs')
        
        if result.success:
            a_new = result.x[:n]
            a_new = np.maximum(a_new, 0)
            a_new /= a_new.sum()
            t_lp = result.x[n]
            
            # Check if the LP solution actually improves
            c_new = fast_autocorr(a_new)
            c_new_max = c_new.max()
            obj_new = 2.0 * n * c_new_max
            
            if obj_new < current_obj - 1e-10:
                a = a_new
            else:
                # LP solution doesn't actually improve (linearization gap)
                # Try a line search
                best_alpha = 0
                best_line_obj = current_obj
                for alpha in [0.1, 0.3, 0.5, 0.7, 0.9]:
                    a_trial = (1-alpha)*a + alpha*a_new
                    a_trial = np.maximum(a_trial, 0)
                    a_trial /= a_trial.sum()
                    obj_trial = 2.0 * n * fast_autocorr(a_trial).max()
                    if obj_trial < best_line_obj:
                        best_line_obj = obj_trial
                        best_alpha = alpha
                if best_alpha > 0:
                    a = (1-best_alpha)*a + best_alpha*a_new
                    a = np.maximum(a, 0)
                    a /= a.sum()
                else:
                    break
        else:
            break
    
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative mirror descent with adaptive softmax peak weighting
2. LP linearization for local refinement
3. Simulated annealing perturbations for escaping local minima
4. Multiple n values and diverse asymmetric initializations
5. FFT-based cross-correlation for fast subgradient computation

import numpy as np
import time
from scipy.optimize import linprog


def fast_autocorr(a):
    """Linear autocorrelation (np.convolve(a,a)) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective assuming sum(a)=1: 2n * max(autocorr)."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def cross_corr_fft(w, a):
    """Compute sg[i] = sum_k w[k]*a[k-i] using FFT. w has length 2n-1, a has length n."""
    n = len(a)
    m = len(w)
    size = m + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(W) * A, fft_size)
    return corr[:n]


def compute_subgradient(a, conv):
    """Compute subgradient of 2n*max(conv) at a using softmax weighting over peaks."""
    n = len(a)
    c_max = conv.max()
    if c_max < 1e-15:
        return np.zeros(n)
    return cross_corr_fft(conv, a) * 2.0 * n / c_max


def optimize_mirror(a0, time_limit, initial_t=30.0, eta0=0.5, top_k=None):
    """Mirror descent (exponential weights) with adaptive softmax temperature."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_autocorr(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    t_param = initial_t
    patience = 0
    max_patience = 120

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        c_max = conv.max()
        if c_max < 1e-15:
            break

        # Softmax weights over convolution peaks
        scaled = t_param * conv / c_max
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Subgradient via FFT cross-correlation
        sg = cross_corr_fft(w, a)
        sg *= 2.0 * n

        # Center to maintain sum constraint
        sg_centered = sg - sg.mean()

        # Multiplicative (mirror) update
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s

        conv = fast_autocorr(a)
        current_obj = 2.0 * n * conv.max()

        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            patience = 0
            eta *= 0.65
            t_param *= 1.4

def lp_refine(a, max_iters=50):
    """Refine using LP linearization of bilinear autocorrelation constraints."""
    n = len(a)
    if n > 2048:
        return a
    a = a.copy()

    for it in range(max_iters):
        c = fast_autocorr(a)
        c_max = c.max()
        current_obj = 2.0 * n * c_max

        # Build LP: min t s.t. 2*a[k-j]*a'[j] summed <= t + c[k], sum(a')=1, a'>=0
        num_k = 2 * n - 1
        A_ub = np.zeros((num_k, n + 1))
        b_ub = c.copy()

        for k in range(num_k):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                A_ub[k, j_start:j_end + 1] = 2.0 * a[k - j_end:k - j_start + 1][::-1]
            A_ub[k, n] = -1.0

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(-1.0, 10.0)]

        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
        except Exception:
            break

        if not result.success:
            break

        a_new = np.maximum(result.x[:n], 0.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            break
        a_new /= s_new

        # Check actual improvement
        c_new = fast_autocorr(a_new)
        obj_new = 2.0 * n * c_new.max()

        if obj_new < current_obj - 1e-10:
            a = a_new
        else:
            # Line search
            best_alpha = 0.0
            best_line_obj = current_obj
            for alpha in [0.1, 0.2, 0.4, 0.6, 0.8]:
                a_trial = (1 - alpha) * a + alpha * a_new
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-12:
                    continue
                a_trial /= s_t
                obj_trial = 2.0 * n * fast_autocorr(a_trial).max()
                if obj_trial < best_line_obj - 1e-10:
                    best_line_obj = obj_trial
                    best_alpha = alpha
            if best_alpha > 0:
                a = (1 - best_alpha) * a + best_alpha * a_new
                a = np.maximum(a, 0.0)
                a /= a.sum()
            else:
                break

    return a

        inits.append((f'sine_{phase}', sine))

    # Random Dirichlet with varying concentration
    for alpha_base in [0.3, 1.0, 3.0]:
        alpha = np.ones(n) * alpha_base + rng.random(n) * 0.5
        r = rng.dirichlet(alpha)
        inits.append((f'rand_{alpha_base}', r))

    # "Triangle" with asymmetry
    tri = np.minimum(x, n - 1 - x * 0.7) + 0.1
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        tri /= tri.sum()
        inits.append(('tri_asym', tri))

    # Cosine window (asymmetric by shifting)
    cos_w = np.maximum(np.cos(np.pi * (x - n * 0.3) / n), 0)
    if cos_w.sum() > 0:
        cos_w /= cos_w.sum()
        inits.append(('cos_shift', cos_w))

    # Parent-derived
    if parent is not None:
        if parent.size == n:
            pp = np.maximum(parent, 0).astype(np.float64)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits.append(('parent', pp))
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits.append(('parent_interp', pp))

    return inits


def propose_candidate(seed=42, budget_s=2350, **kwargs):
    """
    Main entry point. Combines mirror descent, LP refinement, and
    simulated annealing to minimize the autocorrelation peak objective.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20  # safety margin

    # Get parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best_seq, best_score
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"  [t={elapsed:.0f}s] New best: n={len(a)}, score={score:.6f}")
        return score

    # ============ PHASE 1: Quick Scan ============
    print("Phase 1: Quick scan of configurations...")
    configs = []
    lengths = [64, 128, 256, 512, 1024, 2048]

    for n in lengths:
        inits = generate_initializations(n, rng, parent)
        for name, a0 in inits:
            configs.append((n, name, a0))

    # Shuffle to ensure diversity in time allocation
    rng2 = np.random.default_rng(seed + 1)
    order = rng2.permutation(len(configs))
    configs = [configs[i] for i in order]

    phase1_deadline = min(start_time + 500, deadline - 1200)
    time_per_scan = max(5.0, 500.0 / len(configs))

    scan_results = []
    for n, name, a0 in configs:
        if time.monotonic() > phase1_deadline:
            break
        t_alloc = min(time_per_scan, phase1_deadline - time.monotonic())
        if t_alloc < 3:
            break
        try:
            a_opt, _ = optimize_mirror(a0, time_limit=t_alloc, initial_t=30.0, eta0=0.5)
            score = try_update(a_opt)
            scan_results.append((score, n, name, a_opt))
        except Exception:
            pass

    print(f"  Phase 1 done. Best: {best_score:.6f}. Results: {len(scan_results)}")

    # ============ PHASE 2: Deep Optimization ============
    print("\nPhase 2: Deep optimization on top candidates...")
    scan_results.sort(key=lambda x: x[0])

    # Select top configurations with diversity
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 20:
            top_configs.append((n, name, a_opt))
            seen.add(key)

    # Ensure all lengths are represented
    seen_lengths = set(x[0] for x in top_configs)
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths and len(top_configs) < 25:
            top_configs.append((n, name, a_opt))
            seen_lengths.add(n)

    phase2_deadline = min(start_time + 1500, deadline - 400)
    remaining = max(60, phase2_deadline - time.monotonic())
    time_per_deep = remaining / max(1, len(top_configs))

    for n, name, a0 in top_configs:
        if time.monotonic() > phase2_deadline:
            break
        t_rem = phase2_deadline - time.monotonic()
        if t_rem < 10:
            break

        # Try multiple parameter settings
        settings = [(30.0, 0.5), (60.0, 0.4), (120.0, 0.3), (200.0, 0.25)]
        t_each = max(5.0, t_rem / len(settings))

        for t_p, eta_p in settings:
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = min(t_each, phase2_deadline - time.monotonic())
            if t_alloc < 3:
                break
            try:
                a_opt, _ = optimize_mirror(a0, time_limit=t_alloc, initial_t=t_p, eta0=eta_p)
                try_update(a_opt)
            except Exception:
                pass

    print(f"  Phase 2 done. Best: {best_score:.6f}")

    # ============ PHASE 3: Simulated Annealing Refinement ============
    print("\nPhase 3: Simulated annealing on best candidates...")
    if best_seq is not None:
        phase3_deadline = min(start_time + 1900, deadline - 200)

        # Take top few from scan + current best
        candidates = [(n, a_opt) for _, n, _, a_opt in top_configs[:8]]
        candidates.append((len(best_seq), best_seq.copy()))

        t_rem = max(30, phase3_deadline - time.monotonic())
        t_each = t_rem / max(1, len(candidates))

        for n, a0 in candidates:
            if time.monotonic() > phase3_deadline:
                break
            t_alloc = min(t_each, phase3_deadline - time.monotonic())
            if t_alloc < 10:
                break
            try:
                a_opt, _ = optimize_with_perturb(a0, time_limit=t_alloc, rng=rng,
                                                 initial_t=50.0, eta0=0.4)
                try_update(a_opt)
            except Exception:
                pass

    # ============ PHASE 4: LP Refinement ============
    print("\nPhase 4: LP refinement...")
    if best_seq is not None:
        n_best = len(best_seq)
        if n_best <= 2048:
            t_lp = min(200.0, deadline - time.monotonic() - 100)
            if t_lp > 10:
                a_lp = lp_refine(best_seq, max_iters=50)
                score_lp = try_update(a_lp)

                # Mirror descent after LP
                if time.monotonic() < deadline - 50:
                    t_alloc = max(10, deadline - time.monotonic() - 50)
                    a_opt, _ = optimize_mirror(a_lp, time_limit=t_alloc, initial_t=80.0, eta0=0.3)
                    try_update(a_opt)

    # ============ PHASE 5: Final Polish ============
    print("\nPhase 5: Final polish...")
    if best_seq is not None and time.monotonic() < deadline - 30:
        t_rem = deadline - time.monotonic() - 30
        # Extended mirror descent with high temperature
        for t_p in [100.0, 200.0, 400.0]:
            if time.monotonic() > deadline - 30:
                break
            t_alloc = max(5, t_rem / 3)
            a_opt, _ = optimize_mirror(best_seq, time_limit=t_alloc, initial_t=t_p, eta0=0.3)
            try_update(a_opt)
            t_rem = deadline - time.monotonic() - 30

        # One more LP pass
        if len(best_seq) <= 1024 and time.monotonic() < deadline - 30:
            a_lp = lp_refine(best_seq, max_iters=30)
            try_update(a_lp)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    total_time = time.monotonic() - start_time
    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}, time={total_time:.0f}s")
    return [float(x) for x in best_seq]
