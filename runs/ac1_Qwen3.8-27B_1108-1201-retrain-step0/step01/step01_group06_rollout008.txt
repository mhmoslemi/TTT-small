Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Adam-based projected gradient descent on smooth max of auto-convolution
2. LP-based descent direction refinement
3. Multi-start with progressive temperature annealing

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_gradient(a, conv, tau):
    """
    Compute gradient of soft_max(conv, tau) w.r.t. a.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Soft-max weights
    scaled = tau * conv
    max_val = scaled.max()
    exp_scaled = np.exp(scaled - max_val)
    w = exp_scaled / exp_scaled.sum()
    
    # grad[j] = 2 * sum_{k=0}^{n-1} w[j+k] * a[k]
    # Using FFT-based cross-correlation
    size = 3 * n - 2  # need this much for linear conv of length (2n-1) and n
    fft_size = 1 << (size - 1).bit_length()
    
    w_ext = np.zeros(fft_size)
    w_ext[:2*n-1] = w
    a_flip = np.zeros(fft_size)
    a_flip[:n] = a[::-1]
    
    conv_result = np.fft.irfft(
        np.fft.rfft(w_ext, fft_size) * np.fft.rfft(a_flip, fft_size), 
        fft_size
    )
    grad = 2.0 * conv_result[n-1:2*n-1]
    return grad

def hard_max_subgradient(a, conv):
    """
    Compute a subgradient of max(conv) w.r.t. a.
    Uses uniform weighting over the argmax set.
    """
    n = len(a)
    max_val = conv.max()
    # Find all positions within tolerance of max
    threshold = max_val - 1e-10 * max_val
    active = conv >= threshold
    w = active.astype(np.float64)
    w /= w.sum()
    
    # Same computation as smooth gradient but with hard weights
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    
    w_ext = np.zeros(fft_size)
    w_ext[:2*n-1] = w
    a_flip = np.zeros(fft_size)
    a_flip[:n] = a[::-1]
    
    conv_result = np.fft.irfft(
        np.fft.rfft(w_ext, fft_size) * np.fft.rfft(a_flip, fft_size), 
        fft_size
    )
    return 2.0 * conv_result[n-1:2*n-1]

def lp_direction(a, conv, M):
    """
    Solve LP: max sum(b) s.t. (a*b)[m] <= M for all m, b >= 0.
    Returns the optimal b (or None if infeasible/too slow).
    """
    n = len(a)
    
    # (a*b)[m] = sum_j a[j] * b[m-j] for valid j, m-j
    # This is a linear function of b.
    # Constraint: sum_j a[j] * b[m-j] <= M for each m
    # Coefficient of b[k] in constraint m: a[m-k] (when 0 <= m-k < n)
    
    # A_ub has shape (2n-1, n)
    # A_ub[m, k] = a[m-k] if 0 <= m-k < n, else 0
    
    A_ub = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k_start = max(0, m - n + 1)
        k_end = min(n - 1, m)
        for k in range(k_start, k_end + 1):
            A_ub[m, k] = a[m - k]
    
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=np.full(2*n-1, M), 
                        bounds=[(0, None)]*n, method='highs')
        if result.success:
            return result.x
        else:
            return None
    except Exception:
        return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam-based projected gradient descent
on smooth max of auto-convolution, with progressive temperature annealing, Nesterov
momentum, LP-based refinement directions, and multi-start across diverse lengths (64-512)
and initializations. Combines smooth and hard gradient phases for convergence.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def simplex_project(v, target=1.0):
    """Project vector onto {x >= 0, sum(x) = target}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rho = idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_smooth(a, conv, tau):
    """
    Compute gradient of soft_max(conv(a,a), tau) w.r.t. a.
    grad[j] = 2 * sum_{k=0}^{n-1} w[j+k] * a[k]
    where w[m] = exp(tau*conv[m]) / sum(exp(tau*conv))
    """
    n = len(a)
    scaled = tau * conv
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()

    # Cross-correlation via FFT: c[j] = sum_{k=0}^{n-1} w[j+k] * a[k]
    total_size = (2 * n - 1) + n - 1  # = 3n - 2
    fft_size = 1 << (total_size - 1).bit_length()

    w_ext = np.zeros(fft_size)
    w_ext[:2 * n - 1] = w
    a_flip_ext = np.zeros(fft_size)
    a_flip_ext[:n] = a[::-1]

    conv_result = np.fft.irfft(
        np.fft.rfft(w_ext, fft_size) * np.fft.rfft(a_flip_ext, fft_size),
        fft_size
    )
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad

    a_flip_ext[:n] = a[::-1]

    conv_result = np.fft.irfft(
        np.fft.rfft(w_ext, fft_size) * np.fft.rfft(a_flip_ext, fft_size),
        fft_size
    )
    return 2.0 * conv_result[n - 1:2 * n - 1]


def lp_direction(a, M):
    """
    Solve LP: max sum(b) s.t. (a*b)[m] <= M for all m, b >= 0.
    Returns optimal b or None.
    """
    n = len(a)
    if n > 300:
        return None  # Too slow for large n

    A_ub = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k_start = max(0, m - n + 1)
        k_end = min(n - 1, m)
        if k_start <= k_end:
            A_ub[m, k_start:k_end + 1] = a[m - k_end:m - k_start + 1][::-1]

    c_vec = -np.ones(n)
    try:
        result = linprog(c_vec, A_ub=A_ub, b_ub=np.full(2 * n - 1, M),
                         bounds=[(0, None)] * n, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def adam_optimize(a0, time_limit, tau_start=30.0, tau_end=300.0,
                  lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                  use_nesterov=True, use_lp=False, lp_every=200,
                  perturb_threshold=150):
    """
    Adam-based projected gradient descent with Nesterov momentum
    on smooth max of conv(a,a). Progressive tau annealing.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    m_ = np.zeros(n)
    v_ = np.zeros(n)
    v_pos = a.copy()  # Nesterov lookahead position
    t = 0
    stagnant = 0
    max_stagnant = perturb_threshold
    t0 = time.monotonic()
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1.0))

        # Progressive temperature: start smooth, end sharp
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)

        # Compute gradient (use Nesterov lookahead position if enabled)
        if use_nesterov and t > 1:
            a_eval = simplex_project(v_pos, target=1.0)
        else:
            a_eval = a

        conv_eval = fast_convolve(a_eval)

        if tau < 200:
            grad = compute_grad_smooth(a_eval, conv_eval, tau)
        else:
            grad = compute_grad_hard(a_eval, conv_eval)

        # Normalize gradient for stability
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            grad = grad / grad_norm

        # Adam update
        m_ = beta1 * m_ + (1.0 - beta1) * grad
        v_ = beta2 * v_ + (1.0 - beta2) * grad ** 2
        m_hat = m_ / (1.0 - beta1 ** t)
        v_hat = v_ / (1.0 - beta2 ** t)

        # Learning rate with warmup and decay
        warmup = min(1.0, t / 50.0)
        lr_t = lr * warmup / (1.0 + 0.0005 * t)

        if use_nesterov:
            # Nesterov: update lookahead, compute grad there, then step
            v_pos = v_pos - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = simplex_project(v_pos, target=1.0)
        else:
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = simplex_project(a, target=1.0)

        # Optional LP refinement (sparingly)
        if use_lp and t % lp_every == 0 and n <= 256:
            conv_now = fast_convolve(a)
            M_now = conv_now.max()
            b_opt = lp_direction(a, M_now)
            if b_opt is not None:
                b_sum = b_opt.sum()
                if b_sum > 1e-10:
                    b_opt = b_opt / b_sum
                    step = 0.15
                    a = (1.0 - step) * a + step * b_opt
                    a = simplex_project(a, target=1.0)
                    if use_nesterov:
                        v_pos = a.copy()

        # Evaluate
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Stagnation handling: random perturbation
        if stagnant >= max_stagnant:
            noise_scale = 0.02 * (1.0 + stagnant / max_stagnant)
            noise = rng.normal(0, noise_scale, n)
            a_pert = a + noise
            a_pert = simplex_project(a_pert, target=1.0)
            conv_pert = fast_convolve(a_pert)
            if 2.0 * n * conv_pert.max() < best_obj:
                a = a_pert
                conv = conv_pert
                if use_nesterov:
                    v_pos = a.copy()
            stagnant = 0
            max_stagnant = max(80, int(max_stagnant * 0.8))

    return best_a, best_obj


def mirror_descent_optimize(a0, time_limit, tau_start=30.0, tau_end=300.0,
                            eta_start=0.8, eta_min=0.01):
    """
    Multiplicative mirror descent (exponential weights) with progressive
    temperature annealing. Similar to previous SOTA but with improved schedule.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    eta = eta_start
    stagnant = 0
    t = 0
    t0 = time.monotonic()
    rng = np.random.default_rng(99999 + n)

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1.0))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)

        # Soft-max weights
        scaled = tau * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(2 * n - 1)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Weighted subgradient: sg[j] = sum_m w[m] * a[m-j]
        total_size = 3 * n - 2
        fft_size = 1 << (total_size - 1).bit_length()
        w_ext = np.zeros(fft_size)
        w_ext[:2 * n - 1] = w
        a_flip_ext = np.zeros(fft_size)
        a_flip_ext[:n] = a[::-1]
        conv_result = np.fft.irfft(
            np.fft.rfft(w_ext, fft_size) * np.fft.rfft(a_flip_ext, fft_size),
            fft_size
        )
        sg = 2.0 * conv_result[n - 1:2 * n - 1]

        # Multiplicative update with centering
        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.ones(n) / n
        else:
            a /= a_sum

        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant >= 80:
            stagnant = 0
            eta *= 0.7
            if eta < eta_min:
                # Perturb
                noise = rng.normal(0, 0.01, n)
                a = simplex_project(a + noise, target=1.0)
                conv = fast_convolve(a)
                eta = 0.3

    return best_a, best_obj


def coordinate_polish(a, time_limit, n_trials=200):
    """
    Simple coordinate descent polish: try adjusting pairs of coordinates
    to reduce the max convolution.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return a
    a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(77777 + n)
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        # Pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try transferring mass from i to j (and vice versa)
        for sign in [1, -1]:
            for frac in [0.01, 0.03, 0.05, 0.1]:
                delta = sign * frac * a[i]
                if a[i] - delta < 0 or a[j] + delta < 0:
                    continue
                a_copy = a.copy()
                a_copy[i] -= delta
                a_copy[j] += delta
                conv_c = fast_convolve(a_copy)
                obj_c = 2.0 * n * conv_c.max()
                if obj_c < best_obj - 1e-12:
                    a = a_copy
                    conv = conv_c
                    best_obj = obj_c
                    best_a = a.copy()
                    stagnant = 0
                    break
            else:
                continue
            break
        else:
            stagnant += 1
            if stagnant > n_trials * 3:
                break

    return best_a


def run_optimization(seed, budget_s):
    """Main optimization orchestration."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 25.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    print(f"Parent: {'yes (len=' + str(parent.size) + ')' if parent is not None else 'no'}")

    # Build configs
    configs = []
    lengths = [64, 96, 128, 192, 256, 384, 512]

    for n in lengths:
        x = np.arange(n, dtype=np.float64)

        # Uniform
        configs.append((n, 'uniform', np.ones(n) / n))

        # Gaussians
        for sf in [0.1, 0.2, 0.3, 0.4]:
            g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
            g /= g.sum()
            configs.append((n, f'gauss_{sf}', g))

        # Bimodal (edges)
        bi = np.zeros(n)
        bi[:max(1, n // 5)] = 1.0
        bi[-max(1, n // 5):] = 1.0
        bi /= bi.sum()
        configs.append((n, 'bimodal', bi))

        # Smooth cosine bump
        cos_b = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        cos_b = np.maximum(cos_b, 1e-6)
        cos_b /= cos_b.sum()
        configs.append((n, 'cosine', cos_b))

        # Power law (heavier center)
        pl = np.exp(-0.5 * ((x - n * 0.5) / (n * 0.15)) ** 4)
        pl /= pl.sum()
        configs.append((n, 'super_gauss', pl))

        # Random
        r = rng.dirichlet(np.ones(n) * 0.5)
        configs.append((n, 'random', r))

        # From parent
        if parent is not None:
            if parent.size == n:
                configs.append((n, 'parent', parent.copy()))
            elif parent.size >= n:
                pp = parent[:n].copy()
                pp = np.maximum(pp, 0)
                pp_s = pp.sum()
                if pp_s > 1e-10:
                    configs.append((n, 'parent_trunc', pp / pp_s))
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0)
                pp_s = pp.sum()
                if pp_s > 1e-10:
                    configs.append((n, 'parent_interp', pp / pp_s))

    print(f"Total configs: {len(configs)}")

    # Phase 1: Quick scan
    print("Phase 1: Quick scan...")
    scan_budget = min(350.0, (deadline - time.monotonic()) * 0.25)
    time_per = max(2.0, scan_budget / max(1, len(configs)))

    scan_results = []
    for i, (n, name, a0) in enumerate(configs):
        if time.monotonic() > deadline - 150:
            break
        t_alloc = min(time_per, deadline - time.monotonic() - 150)
        if t_alloc < 1.5:
            break

        # Alternate between Adam and mirror descent
        if i % 2 == 0:
            a_opt, obj_opt = adam_optimize(a0, time_limit=t_alloc,
                                           tau_start=20, tau_end=200, lr=0.01)
        else:
            a_opt, obj_opt = mirror_descent_optimize(a0, time_limit=t_alloc,
                                                     tau_start=20, tau_end=200,
                                                     eta_start=0.6)

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan: n={n}, {name}, score={score:.6f}")

    # Phase 2: Deep optimization
    print(f"\nPhase 2: Deep optimization. Best: {best_score:.6f}")
    scan_results.sort(key=lambda x: x[0])

    top_configs = []
    seen_n = set()
    for score, n, name, a_opt in scan_results:
        if len(top_configs) < 15:
            top_configs.append((n, name, a_opt))
            seen_n.add(n)
        elif n not in seen_n and len(top_configs) < 20:
            top_configs.append((n, name, a_opt))
            seen_n.add(n)

    # Ensure best is included
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    # Deduplicate (keep best per length)
    best_per_n = {}
    for n, name, a_opt in top_configs:
        if n not in best_per_n:
            best_per_n[n] = (name, a_opt)
    top_configs = [(n, v[0], v[1]) for n, v in best_per_n.items()]

    remaining = deadline - time.monotonic() - 80
    n_deep = len(top_configs)
    time_per_deep = max(8.0, remaining / max(1, n_deep * 2))

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 80:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 80)
        if t_alloc < 5:
            break

        # Adam with Nesterov
        t_single = t_alloc / 3.0
        a_opt, obj_opt = adam_optimize(a0, time_limit=t_single,
                                       tau_start=20, tau_end=400, lr=0.008,
                                       use_nesterov=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep Adam: n={n}, {name}, score={score:.6f}")

        # Mirror descent
        if time.monotonic() < deadline - 80:
            t_single = min(t_alloc / 3.0, deadline - time.monotonic() - 80)
            a_opt, obj_opt = mirror_descent_optimize(a0, time_limit=t_single,
                                                     tau_start=30, tau_end=300,
                                                     eta_start=0.5)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep MD: n={n}, {name}, score={score:.6f}")

        # LP-assisted (only for small n)
        if n <= 192 and time.monotonic() < deadline - 80:
            t_single = min(t_alloc / 3.0, deadline - time.monotonic() - 80)
            a_opt, obj_opt = adam_optimize(a0, time_limit=t_single,
                                           tau_start=30, tau_end=300, lr=0.008,
                                           use_nesterov=True,
                                           use_lp=True, lp_every=100)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep LP: n={n}, {name}, score={score:.6f}")

    # Phase 3: Final polish on best
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30
        if remaining > 20:
            print(f"\nPhase 3: Polish. Remaining: {remaining:.0f}s")
            n_best = len(best_sequence)

            # Deep Adam with high tau
            t1 = remaining * 0.35
            a_opt, obj_opt = adam_optimize(best_sequence, time_limit=max(5, t1),
                                           tau_start=100, tau_end=1000, lr=0.003,
                                           use_nesterov=True,
                                           use_lp=(n_best <= 192), lp_every=50)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish1: score={score:.6f}")

            # Mirror descent polish
            remaining = deadline - time.monotonic() - 30
            if remaining > 10:
                t2 = remaining * 0.35
                a_opt, obj_opt = mirror_descent_optimize(best_sequence,
                                                         time_limit=max(5, t2),
                                                         tau_start=80, tau_end=500,
                                                         eta_start=0.2)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish2: score={score:.6f}")

            # Coordinate polish
            remaining = deadline - time.monotonic() - 30
            if remaining > 10:
                a_opt = coordinate_polish(best_sequence, time_limit=max(5, remaining * 0.3))
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  CoordPolish: score={score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return best_sequence


def propose_candidate(seed=42, budget_s=2380, **kwargs):
    """
    Main entry point. Searches for the best sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2 using Adam projected gradient descent
    with Nesterov momentum, progressive temperature annealing,
    LP refinement, and multi-start across diverse lengths and initializations.
    """
    result = run_optimization(seed=seed, budget_s=budget_s)
    return [float(x) for x in result]
