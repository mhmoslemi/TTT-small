    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)

    # g[j] = sum_k w[k] * dc[k]/da[j]
    # dc[k]/da[j] = a[k-j] + a[j-k] (approximately, ignoring boundary effects)
    # g[j] ≈ 2 * sum_k w[k] * a[k-j]
    # = 2 * (cross_correlation(a, w))[j]
    # = 2 * (a * flip(w))[j + len(w) - 1] ... 
    # Using the formulation: conv(a[::-1], w)[n-1:2n-1]
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return value, g, c, w

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

    Nesterov-accelerated gradient descent on log-parameterization.

    mu = 0.9  # momentum

    best_score = np.inf

    tau = 5.0 / n  # initial temperature

        elapsed_frac = elapsed / time_limit

        # Start high, decrease over time
        # But also adapt based on the spread of top peaks
        base_tau = (10.0 / n) * (0.05 / (10.0 / n)) ** elapsed_frac
        base_tau = max(base_tau, 0.1 / n)

        theta_old = theta.copy()
        theta = theta - mu * v  # look-ahead
        a = softmax_normalize(theta) * S
        
        value, g_a, c, w = smooth_max_and_grad(a, base_tau)
        
        # Project gradient to tangent space of simplex
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        grad_norm = np.linalg.norm(g_theta)
        if grad_norm < 1e-12:
            break
        lr = 0.1 / (grad_norm + 1e-8)
        lr = min(lr, 5.0)  # cap learning rate
        
        v = mu * v + lr * g_theta
        theta = theta_old - v
        
        # Compute actual score
        a_check = softmax_normalize(theta) * S
        score = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (S * S)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
        
        step += 1
        if step % 500 == 0:
            print(f"    step={step}, score={score:.6f}, best={best_score:.6f}, tau={base_tau:.2e}")

Let me formalize:

- Current a gives conv(a,a) = c
- First-order correction: a + da, where conv(a+da, a+da) ≈ conv(a,a) + 2*conv(da, a) = u
- So: conv(da, a) = (u - c) / 2
- This is a linear system: M * da = (u-c)/2 where M is the convolution operator with a
- Solve for da, update a = max(a + da, 0), normalize

def newton_flat_convolution(a_init, n_iter=100):
    """Newton iteration to make conv(a,a) as flat as possible."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / np.sum(a)  # normalize to sum=1
    
    for it in range(n_iter):
        c = fft_convolve(a, a)
        target = np.ones(2*n-1) / (2*n-1)
        residual = target - c
        # Solve conv(da, a) = residual / 2
        # This is a Toeplitz system
        # da * a (linear convolution) = residual/2
        # In frequency domain: FFT(da) * FFT(a) = FFT(residual/2)
        # FFT(da) = FFT(residual/2) / FFT(a)
        size = 1
        while size < 2*n-1:
            size <<= 1
        fa = np.fft.rfft(a, size)
        fr = np.fft.rfft(residual / 2, size)
        # Guard against division by zero
        fa_safe = np.where(np.abs(fa) > 1e-10, fa, 1e-10)
        f_da = fr / fa_safe
        da = np.fft.irfft(f_da, size)[:n]
        # Damping
        alpha = 0.5
        a_new = a + alpha * da
        a_new = np.maximum(a_new, 0.0)
        if np.sum(a_new) < 1e-10:
            break
        a_new = a_new / np.sum(a_new)
        
        if np.max(np.abs(a_new - a)) < 1e-10:
            break
        a = a_new
    
    return a

Hybrid optimization for minimizing peak of linear self-convolution

L-BFGS-B refinement with adaptive log-sum-exp smoothing, symmetry
exploitation, and targeted local search polish.

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]

    """Fast score computation (matches evaluate_sequence formula)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

"""
Hybrid optimization for minimizing peak of linear self-convolution of non-negative
step functions. Combines Newton flat-convolution initialization, L-BFGS-B refinement
with adaptive log-sum-exp smoothing, symmetry exploitation, multi-scale strategy,
and targeted local search polish.

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def local_score(a):
    """Fast score computation matching evaluate_sequence formula."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def newton_flat(a_init, n_iter=50, damping=0.4):
    """Newton iteration to make conv(a,a) as flat as possible. Returns a with sum=1."""
    n = len(a_init)
    a = np.array(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-6)
    a = a / np.sum(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    for it in range(n_iter):
        c = fft_convolve(a, a)
        target = np.ones(2 * n - 1) / (2 * n - 1)
        residual = target - c
        fa = np.fft.rfft(a, size)
        fr = np.fft.rfft(residual / 2.0, size)
        fa_abs = np.abs(fa)
        fa_safe = np.where(fa_abs > 1e-10, fa, np.where(fa_abs > 0, fa / fa_abs * 1e-10, 1e-10 + 0j))
        f_da = fr / fa_safe
        da = np.fft.irfft(f_da, size)[:n]
        a_new = a + damping * da
        a_new = np.maximum(a_new, 1e-6)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        if np.max(np.abs(a_new - a)) < 1e-13:
            break
        a = a_new
    return a


def newton_weighted(a_init, n_iter=50, damping=0.3, peak_weight=2.0):
    """Newton iteration targeting a shape that emphasizes peak reduction."""
    n = len(a_init)

        size <<= 1

        c = fft_convolve(a, a)
        c_max = np.max(c)
        # Weight: want to reduce the peak more aggressively
        inv_c = 1.0 / (c + 1e-10)
        target = inv_c / np.sum(inv_c)
        # Blend with uniform
        uniform = np.ones(2 * n - 1) / (2 * n - 1)
        target = peak_weight * target + (1 - peak_weight + 1) * uniform
        target = target / np.sum(target)
        residual = target - c
        fa = np.fft.rfft(a, size)
        fr = np.fft.rfft(residual / 2.0, size)
        fa_abs = np.abs(fa)
        fa_safe = np.where(fa_abs > 1e-10, fa, np.where(fa_abs > 0, fa / fa_abs * 1e-10, 1e-10 + 0j))
        f_da = fr / fa_safe
        da = np.fft.irfft(f_da, size)[:n]

        a_new = np.maximum(a_new, 1e-6)
        s = np.sum(a_new)

            break
        a_new = a_new / s
        if np.max(np.abs(a_new - a)) < 1e-13:

def lgbfsb_refine(a_init, n, time_limit, rng):

    L-BFGS-B optimization with adaptive log-sum-exp temperature on log-parameterization.
    Returns (best_theta, best_score).
    """

    a = np.maximum(a_init, 1e-6)
    a = a / np.sum(a) * S
    theta = np.log(a / S + 1e-10)

    best_a = a.copy()
    best_score = local_score(a)

    batch_count = 0

        elapsed = time.monotonic() - t0

        tau_start = 15.0 / n
        tau_end = 0.15 / n
        # Use a smooth schedule with a "floor" at the end
        if elapsed_frac < 0.7:
            tau = tau_start * (tau_end / tau_start) ** (elapsed_frac / 0.7)
        else:
            frac2 = (elapsed_frac - 0.7) / 0.3
            tau = tau_end * (0.3 / 1.0) ** frac2

        def scipy_obj(tx):
            ax = softmax_normalize(tx) * S
            c = fft_convolve(ax, ax)
            c_max = np.max(c)
            shifted = (c - c_max) / tau
            exp_s = np.exp(shifted)
            sum_exp = np.sum(exp_s)
            return c_max + tau * np.log(sum_exp)

        def scipy_grad(tx):
            ax = softmax_normalize(tx) * S
            c = fft_convolve(ax, ax)
            c_max = np.max(c)

            rev_a = ax[::-1]
            corr = fft_convolve(rev_a, w)
            g_a = 2.0 * corr[n - 1:2 * n - 1]
            inner = np.dot(ax, g_a)
            return ax * (g_a - inner) / S

        # Run L-BFGS-B for a batch
        max_iter = 80 if n <= 512 else 40
        result = minimize(
            scipy_obj, theta, jac=scipy_grad,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-9, 'maxls': 40}
        )

        theta = np.array(result.x, dtype=np.float64)
        a_new = softmax_normalize(theta) * S
        score = local_score(a_new)

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            best_theta = theta.copy()

                best_score = score_sym
                best_a = a_sym.copy()
                theta = np.log(a_sym / S + 1e-10)
            elif score_sym < score:
                theta = np.log(a_sym / S + 1e-10)
                a_new = a_sym

        if batch_count % 10 == 0:
            print(f"    batch={batch_count}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={elapsed:.0f}s")

def local_search_polish(a_init, time_limit, rng):
    """
    Simulated annealing local search with targeted peak reduction.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    current = a_init.astype(np.float64).copy()
    current = current / np.sum(current) * S
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.99995
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.4:
            # Targeted: reduce peak contributors
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            # Compute contributions
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.75:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.85:
            # Partial symmetrization
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.008, size=n))
        elif r < 0.95:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.008, size=n))
        else:
            # Edge emphasis
            t = np.linspace(0, 1, n)
            edge_mask = 1.0 + 0.1 * np.cos(np.pi * t)
            candidate = current * edge_mask

        candidate = np.maximum(candidate, 1e-4)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature *= cooling
        iter_count += 1

        if iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LS it={iter_count}, current={current_score:.6f}, best={best_score:.6f}, t={elapsed:.0f}s")

def make_raised_cosine(n, rng):

        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Phase 1: Multi-start with Newton + L-BFGS-B
    print("\n=== Phase 1: Multi-start optimization ===")
    configs = [
        (256, ['newton', 'newton_w', 'chirp', 'raised', 'provided']),
        (384, ['newton', 'chirp', 'raised']),
        (512, ['newton', 'raised']),
    ]

    for n, init_types in configs:
        if time.monotonic() > deadline - 180:
            break
        time_left = deadline - time.monotonic()
        time_per_init = min(120.0, time_left / (len(init_types) + 0.5))

        for init_type in init_types:
            if time.monotonic() > deadline - 60:
                break

            if init_type == 'provided' and start_seq is not None:
                if len(start_seq) == n:
                    a0 = start_seq.copy()
                else:
                    x_old = np.linspace(0, 1, len(start_seq))
                    x_new = np.linspace(0, 1, n)
                    a0 = np.interp(x_new, x_old, start_seq)
            elif init_type == 'newton':
                a_uniform = np.ones(n)
                a0 = newton_flat(a_uniform, n_iter=40, damping=0.4)
            elif init_type == 'newton_w':
                a_uniform = np.ones(n)
                a0 = newton_weighted(a_uniform, n_iter=40, damping=0.3, peak_weight=1.5)
            elif init_type == 'chirp':
                a0 = make_chirp(n, rng)
            elif init_type == 'raised':
                a0 = make_raised_cosine(n, rng)
            else:
                a0 = np.ones(n)

            a0 = np.maximum(a0, 0.01)

            # Try L-BFGS-B
            t0 = time.monotonic()
            theta_best, grad_score = lgbfsb_refine(a0, n, time_per_init, rng)
            elapsed = time.monotonic() - t0

            a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n)
            # Symmetrize
            a_best = (a_best + a_best[::-1]) / 2.0
            a_best = np.maximum(a_best, 0.01)
            actual_score = local_score(a_best)
            print(f"  n={n}, init='{init_type}', score={actual_score:.6f}, t={elapsed:.0f}s")

            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

        print(f"  Best after n={n}: {best_score:.6f}")

    # Phase 2: Extended refinement on best candidate
    print(f"\n=== Phase 2: Extended refinement (score={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        polish_time = min(600.0, max(60.0, deadline - time.monotonic() - 200))
        print(f"  n={n_best}, time={polish_time:.0f}s")

        theta_init = np.log(best_seq / np.sum(best_seq) + 1e-10)
        theta_best2, score2 = lgbfsb_refine(best_seq, n_best, polish_time, rng)
        a_best2 = softmax_normalize(theta_best2) * np.sqrt(2.0 * n_best)
        a_best2 = (a_best2 + a_best2[::-1]) / 2.0
        a_best2 = np.maximum(a_best2, 0.01)
        actual_score2 = local_score(a_best2)
        print(f"  After refinement: {actual_score2:.6f}")

        if actual_score2 < best_score:
            best_score = actual_score2
            best_seq = a_best2.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 3: Direct gradient polish + local search
    print(f"\n=== Phase 3: Polish (score={best_score:.6f}) ===")
    if best_seq is not None:
        # Direct gradient polish
        if time.monotonic() < deadline - 120:
            grad_time = min(150.0, deadline - time.monotonic() - 120)
            print(f"  Direct gradient polish, time={grad_time:.0f}s")
            polished_g, score_g = polish_direct(best_seq, grad_time, rng)
            print(f"  After gradient polish: {score_g:.6f}")
            if score_g < best_score:
                best_score = score_g
                best_seq = polished_g.copy()
                print(f"    *** New best: {best_score:.6f} ***")

        # Local search
        if time.monotonic() < deadline - 60:
            ls_time = min(180.0, deadline - time.monotonic() - 30)
            print(f"  Local search, time={ls_time:.0f}s")
            polished_ls, score_ls = local_search_polish(best_seq, ls_time, rng)
            print(f"  After local search: {score_ls:.6f}")
            if score_ls < best_score:
                best_score = score_ls
                best_seq = polished_ls.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Final: verify with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
