def compute_grad(a, tau):
    """Compute gradient of tau*log(sum(exp(conv(a,a)/tau))) w.r.t. a via FFT."""
    n = a.size
    c = np.convolve(a, a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    A = np.fft.fft(a, N)
    Q = np.fft.fft(q, N)

    # T2[j] = sum_m a[m] * q[j-m] (convolution, with q=0 outside [0,2n-2])

    T2 = np.fft.irfft(np.fft.rfft(q, N) * np.fft.rfft(a, N), N)[:n]
    
    grad = T1 + T2
    return grad, c, q

    n = a.size
