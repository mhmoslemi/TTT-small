```python
import time
from scipy.optimize import linprog, minimize_scalar, minimize


def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (target / S)


def _soft_max_value(c, tau):
    """Compute soft-max of array c with parameter tau."""
    c_max = c.max()
    if tau < 1e-10:
        return c_max, np.full_like(c, 1.0 / len(c))
    shifted = tau * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    val = c_max + np.log(exp_shifted.sum()) / tau
    return val, w


def _grad_soft_max(a, tau):
    """Gradient of soft-max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    _, w = _soft_max_value(c, tau)
    # grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j in [0,n-1])
    # This equals 2 * (a circular-correlation w)[j] with zero-padding
    L = 2 * n - 1
    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    corr = np.fft.irfft(A * np.conj(W), L)
    grad = 2.0 * corr[:n]
    return grad
```

```python
"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-momentum gradient on soft-max peak, coordinate descent,
L-BFGS refinement, and LP-based blending. Explores multiple lengths and starts.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar


def _compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (target / S)


def _soft_max_and_weights(c, tau):
    c_max = c.max()
    shifted = tau * (c - c_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_s = np.exp(shifted)
    s = exp_s.sum()
    w = exp_s / s
    val = c_max + np.log(s) / tau
    return val, w


def _smooth_obj_grad(a, tau):
    """Returns (objective_value, gradient) of 2n*softmax(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    sm, w = _soft_max_and_weights(c, tau)
    S = float(a.sum())
    if S < 0.01:
        return 1e10, np.zeros(n)
    # grad of sm w.r.t. a: 2 * (a circular-cross-corr w)
    L = 2 * n - 1
    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    corr = np.fft.irfft(A * np.conj(W), L)
    grad_sm = 2.0 * corr[:n]
    # d/d a of [2n*sm/S^2]
    obj = 2.0 * n * sm / (S * S)
    grad = (2.0 * n / (S * S)) * grad_sm - (4.0 * n * sm / (S ** 3)) * np.ones(n)
    return obj, grad


def _nesterov_optimize(a_init, deadline, tau_list=None, max_total_iters=10000):
    """Nesterov momentum gradient on soft-max objective with non-negativity projection."""
    a = np.maximum(a_init.copy(), 0.0)
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t)

    best = a.copy()
    best_score = _compute_obj(a)

    if tau_list is None:
        tau_list = [5, 10, 20, 50, 100, 200, 400]

    per_phase = max_total_iters // len(tau_list)
    momentum = 0.85

    for tau in tau_list:
        if time.monotonic() > deadline - 20.0:
            break
        v = np.zeros(n)
        prev_a = a.copy()
        lr = 0.5 / (tau + 1.0)  # initial learning rate scaled by tau

        for it in range(per_phase):
            if time.monotonic() > deadline - 15.0:
                break

            # Nesterov look-ahead
            y = a + momentum * (a - prev_a)
            y = np.maximum(y, 0.0)

            obj_y, grad_y = _smooth_obj_grad(y, tau)

            # Gradient step with adaptive lr
            a_new = y - lr * grad_y
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, S_t)

            # Simple backtracking: if obj increased significantly, reduce lr
            obj_new, _ = _smooth_obj_grad(a_new, tau)
            tries = 0
            while obj_new > obj_y * 1.01 + 1e-6 and tries < 10:
                lr *= 0.7
                a_new = y - lr * grad_y
                a_new = np.maximum(a_new, 0.0)
                a_new = _normalize(a_new, S_t)
                obj_new, _ = _smooth_obj_grad(a_new, tau)
                tries += 1

            prev_a = a
            a = a_new

            # Periodic exact evaluation
            if (it + 1) % 30 == 0:
                s = _compute_obj(a)
                if s < best_score:
                    best_score = s
                    best = a.copy()

        # Decay momentum slightly between phases
        momentum = min(0.95, momentum * 1.02)

    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    return best, best_score


def _coord_descent_step(a, i, c, S):
    """Find optimal value for coordinate i given all others fixed.
    Returns new value of a[i]."""
    n = len(a)
    old = a[i]
    if old < 1e-15:
        # Try increasing from 0
        # c[k] changes: for k where 0<=k-i<n, c[k] += 2*t*a[k-i]
        # for k=2i, additional t^2
        # S changes: S + t
        pass

    # Build the "influence" vector: how c[k] changes per unit t
    # delta_c[k] = 2*t*a[k-i] for k != 2i (valid k)
    # delta_c[2i] = t^2 (replaces the 2*t*a[i] term since k-i=i)
    # Actually: delta_c[k] = 2*(t-old)*a[k-i] for k!=2i, valid 0<=k-i<n
    #            delta_c[2i] = (t-old)^2 + 2*(t-old)*old = t^2 - old^2

    # For optimization, define c_new[k] as a function of t:
    # c_new[k] = c[k] - 2*old*a[k-i] + 2*t*a[k-i] for k != 2i, 0<=k-i<n
    # c_new[2i] = c[2i] - old^2 + t^2
    # (when k-i is valid and k != 2i)
    # S_new = S - old + t

    lo_val = max(0, i)  # k-i >= 0 means k >= i... wait
    # k-i in [0, n-1] means k in [i, i+n-1], but k is in [0, 2n-2]
    # So valid k range: max(0, 0) to min(2n-2, i+n-1)... 
    # Actually k ranges over [0, 2n-2], and we need 0 <= k-i < n, so i <= k <= i+n-1
    # Also need k in [0, 2n-2], so k in [max(0,i), min(2n-2, i+n-1)]

    k_lo = max(0, i)
    k_hi = min(2 * n - 2, i + n - 1)

    # The peak of c_new is max over k of c_new[k]
    # For k not in [k_lo, k_hi] and k != 2i: c_new[k] = c[k] (unchanged)
    # For k in [k_lo, k_hi], k != 2i: c_new[k] = c[k] + 2*(t-old)*a[k-i]
    # For k = 2i (if 2i in [0, 2n-2]): c_new[2i] = c[2i] + t^2 - old^2

    base_max = 0.0
    # Max of unchanged bins
    mask_unchanged = np.ones(2 * n - 1, dtype=bool)
    mask_unchanged[k_lo:k_hi + 1] = False
    if 2 * i <= 2 * n - 2:
        mask_unchanged[2 * i] = False
    if mask_unchanged.any():
        base_max = c[mask_unchanged].max()

    # For changed bins, c_new[k] = c[k] + 2*(t-old)*a[k-i] (linear in t)
    # or c_new[2i] = c[2i] + t^2 - old^2 (quadratic in t)
    # We want to minimize: max(base_max, max_changed(t)) / (S - old + t)^2 * 2n

    # Since this is 1D, use golden section or minimize_scalar
    # Upper bound for t: if t is too large, c[2i] ~ t^2 and S ~ t, so obj ~ 2n*t
    # A reasonable upper bound: t_max = max(3*old, 3*S/n)
    t_upper = max(3.0 * old + 1e-6, 3.0 * S / n)

    def _f1d(t_val):
        if t_val < 0:
            t_val = 0.0
        S_new = S - old + t_val
        if S_new < 0.01:
            return 1e10
        peak = base_max
        # Changed linear bins
        for k in range(k_lo, k_hi + 1):
            if k == 2 * i:
                continue
            val = c[k] + 2.0 * (t_val - old) * a[k - i]
            if val > peak:
                peak = val
        # Quadratic bin
        if 2 * i <= 2 * n - 2 and 2 * i >= k_lo and 2 * i <= k_hi:
            val = c[2 * i] + t_val * t_val - old * old
            if val > peak:
                peak = val
        return 2.0 * n * peak / (S_new * S_new)

    res = minimize_scalar(_f1d, bounds=(0.0, t_upper), method='bounded',
                          options={'xatol': 1e-8})
    return res.x


def _coord_descent_pass(a, c, S, order, max_coords=None):
    """One pass of coordinate descent over given order."""
    n = len(a)
    updated = 0
    for i in order:
        if max_coords is not None and updated >= max_coords:
            break
        new_val = _coord_descent_step(a, i, c, S)
        old_val = a[i]
        if abs(new_val - old_val) > 1e-10:
            a[i] = max(0.0, new_val)
            # Update c and S incrementally
            S = S - old_val + a[i]
            # Update convolution incrementally
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            for k in range(k_lo, k_hi + 1):
                if k == 2 * i:
                    c[k] += a[i] * a[i] - old_val * old_val
                else:
                    c[k] += 2.0 * (a[i] - old_val) * a[k - i]
            updated += 1
    return a, c, S


def _coord_descent_optimize(a_init, deadline, n_passes=20, seed=0):
    """Coordinate descent optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.copy(), 0.0)
    n = len(a)
    S = float(a.sum())
    if S < 1e-15:
        return a_init, _compute_obj(a_init)
    c = np.convolve(a, a)

    best = a.copy()
    best_score = _compute_obj(a)

    for p in range(n_passes):
        if time.monotonic() > deadline - 10.0:
            break
        order = rng.permutation(n)
        a, c, S = _coord_descent_pass(a, c, S, order)
        s = _compute_obj(a)
        if s < best_score:
            best_score = s
            best = a.copy()

    return best, best_score


def _lp_solve(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    n = len(f)
    if n > 2048:
        return None
    c_obj = -np.ones(n)
    # Build constraint matrix using the structure
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # Row k has A[k,j] = f[k-j] if 0<=k-j<n
        k_lo = max(0, j)
        k_hi = min(2 * n - 2, j + n - 1)
        A[k_lo:k_hi + 1, j] = f[k_lo - j:k_hi - j + 1]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine(a_init, deadline, max_iters=50):
    """LP-based refinement: iteratively solve LP and blend."""
    a = np.maximum(a_init.copy(), 0.0)
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t)

    best = a.copy()
    best_score = _compute_obj(a)

    c = np.convolve(a, a)
    M = float(c.max())

    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break

        g = _lp_solve(a, M)
        if g is None:
            break

        # Try blending
        improved = False
        for t in [0.7, 0.5, 0.3, 0.15, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, S_t)
            s_new = _compute_obj(a_new)
            if s_new < best_score - 1e-8:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            # Slightly tighten M to encourage more change
            M *= 0.995
            c = np.convolve(a, a)
            M = max(M, float(c.max()) * 0.99)
            if it > 0 and M >= float(np.convolve(a, a).max()) * 0.999:
                break
        else:
            c = np.convolve(a, a)
            M = float(c.max())

    return best, best_score


def _generate_starts(rng, lengths):
    """Generate diverse starting sequences."""
    starts = []
    for n in lengths:
        i = np.arange(n, dtype=np.float64)
        # Random
        starts.append(rng.random(n) * 0.5 + 0.25)
        # Cosine-based (different frequencies)
        for freq in [1, 3, 5, 7]:
            starts.append(0.5 + 0.5 * np.cos(2.0 * np.pi * freq * i / n))
        # Raised cosine
        starts.append(0.5 + 0.5 * np.cos(2.0 * np.pi * i / n))
        # Gaussian bump
        center = n / 2.0
        sigma = n * 0.2
        starts.append(np.exp(-0.5 * ((i - center) / sigma) ** 2))
        # Two bumps
        starts.append(np.exp(-0.5 * ((i - n * 0.25) / (n * 0.15)) ** 2) +
                      np.exp(-0.5 * ((i - n * 0.75) / (n * 0.15)) ** 2))
        # Chirp
        starts.append(np.abs(np.sin(2.0 * np.pi * i * i / (n * 3.0))) + 0.1)
        # Alternating smoothed
        starts.append(0.5 + 0.3 * np.cos(2.0 * np.pi * i / 2.0))
    return starts


def _full_optimize(a_init, deadline, rng, use_lp=True):
    """Full optimization pipeline for one starting sequence."""
    n = len(a_init)
    best = a_init.copy()
    best_score = _compute_obj(a_init)

    # Phase 1: Nesterov gradient
    if time.monotonic() < deadline - 60.0:
        a1, s1 = _nesterov_optimize(
            a_init, deadline - max(30, (deadline - time.monotonic()) // 4),
            tau_list=[5, 10, 30, 80, 150, 300]
        )
        if s1 < best_score:
            best_score, best = s1, a1

    # Phase 2: Coordinate descent
    if time.monotonic() < deadline - 30.0:
        remaining = deadline - time.monotonic()
        n_passes = min(30, max(5, int(remaining / 5.0)))
        a2, s2 = _coord_descent_optimize(best, deadline - 10.0, n_passes=n_passes,
                                         seed=rng.integers(10000))
        if s2 < best_score:
            best_score, best = s2, a2

    # Phase 3: LP refinement

        a3, s3 = _lp_refine(best, deadline - 5.0, max_iters=40)
        if s3 < best_score:
            best_score, best = s3, a3

    # Phase 4: Final coord descent polish

                                         seed=rng.integers(10000))
        if s4 < best_score:

    # Try to use existing best as a starting point
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            gs = np.maximum(gs, 0.0)
            if gs.sum() > 0.01:
                s0 = _compute_obj(gs)
                print(f"Initial from height_sequence_1: n={len(gs)}, score={s0:.6f}")
                if s0 < best_score:
                    best_score = s0
                    best_seq = gs.copy()
    except NameError:
        pass

    # Lengths to try (shorter first for quick wins, longer for better asymptotics)
    lengths = [256, 384, 512, 640, 768]

    # Generate starts, prioritize those near the best known length
    all_starts = _generate_starts(rng, lengths)

    # If we have height_sequence_1, add truncated/extended versions as starts
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8:
            for L in [256, 384, 512, 640, 768]:
                if L <= len(gs):
                    all_starts.append(np.maximum(gs[:L].copy(), 0.0))
                elif L > len(gs):
                    # Pad with small random values
                    pad_len = L - len(gs)
                    padded = np.concatenate([gs, rng.random(pad_len) * 0.1])
                    all_starts.append(padded)
    except NameError:
        pass

    # Sort starts: prioritize lengths near 512 (proven to work), then others
    def _start_priority(s):
        return abs(len(s) - 512)
    all_starts.sort(key=_start_priority)

    print(f"Total starting points: {len(all_starts)}, budget={budget_s}s")

    # Time allocation: give more time to promising starts
    # First pass: quick evaluation of all starts to find good ones
    quick_results = []
    for a0 in all_starts:
        if time.monotonic() > deadline - 60.0:
            break
        s = _compute_obj(a0)
        quick_results.append((s, a0))
    quick_results.sort(key=lambda x: x[0])

    print("Top 5 starting points by initial score:")
    for s, a0 in quick_results[:5]:
        print(f"  n={len(a0)}, score={s:.6f}")

    # Second pass: full optimization on the best starting points
    # Allocate time based on remaining budget
    num_full = min(len(quick_results), max(3, int((deadline - time.monotonic()) / 120.0)))
    print(f"Will do full optimization on {num_full} starting points")

    for idx in range(num_full):
        if time.monotonic() > deadline - 30.0:
            break
        s0, a0 = quick_results[idx]
        print(f"  Optimizing start {idx+1}/{num_full}: n={len(a0)}, init_score={s0:.6f}")
        t0 = time.monotonic()

        use_lp = len(a0) <= 1024
        seq, score = _full_optimize(a0, deadline, rng, use_lp=use_lp)

        elapsed = time.monotonic() - t0
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"    NEW BEST: score={best_score:.6f}, n={len(seq)}, time={elapsed:.1f}s")
        else:
            print(f"    score={score:.6f}, n={len(seq)}, time={elapsed:.1f}s (no improvement)")

    # Final refinement of the overall best
    if best_seq is not None and time.monotonic() < deadline - 20.0:
        print(f"\nFinal refinement of best (score={best_score:.6f}, n={len(best_seq)})...")
        for pass_i in range(4):
            if time.monotonic() > deadline - 10.0:
                break
            seq, score = _full_optimize(best_seq, deadline, rng, use_lp=len(best_seq) <= 1024)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Refine pass {pass_i+1}: score={best_score:.6f}")
            else:
                print(f"  Refine pass {pass_i+1}: no improvement (score={score:.6f})")
                break

    # Final validation
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
```
