def cross_sum(p, a):
    """
    Compute h[i] = sum_m p[m]*a[m-i] for i=0,...,n-1
    where p has length 2n-1 and a has length n.
    Using FFT.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    P = scipy.fft.rfft(p, fft_size)
    Ar = scipy.fft.rfft(a[::-1], fft_size)  # reverse(a)
    conv_pa = scipy.fft.irfft(P * Ar, fft_size)[:size]
    
    return conv_pa[n-1:2*n-1]  # indices n-1 to 2n-2

import numpy as np
import time
import scipy.fft
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    A = scipy.fft.rfft(a, fft_size)
    return scipy.fft.irfft(A*A, fft_size)[:size]

def compute_lse_and_gradient(a, beta):
    """
    Returns (lse_value, gradient_wrt_z) where a=softmax(z).
    gradient_wrt_z[i] = 2*a[i]*(cross[i] - mean_val)
    where cross[i] = sum_m p[m]*a[m-i], mean_val = sum_m p[m]*conv[m]
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # p[m] proportional to exp(beta * conv[m])
    scaled = beta * conv
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    
    lse_val = conv.max() + np.log(exp_s.sum()) / beta
    
    # cross[i] = sum_m p[m]*a[m-i]
    # = (p * reverse(a))[n-1+i] for i=0..n-1
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    P = scipy.fft.rfft(p, fft_size)
    Ar = scipy.fft.rfft(a[::-1], fft_size)
    cross_full = scipy.fft.irfft(P * Ar, fft_size)[:size]
    cross = cross_full[n-1:2*n-1]
    
    mean_val = np.dot(p, conv)
    
    grad_z = 2.0 * a * (cross - mean_val)
    
    return lse_val, grad_z, conv

def adam_step(z, grad, m, v, t, lr=0.01, b1=0.9, b2=0.999, eps=1e-8):
    """Adam update step."""
    m = b1*m + (1-b1)*grad
    v = b2*v + (1-b2)*grad*grad
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def lse_adam_optimize(a0, time_limit, beta_start=5.0, beta_end=100.0, lr=0.01):
    """
    Optimize using Adam on softmax parameterization with LSE continuation.
    a0: initial non-negative sequence
    Returns (best_a, best_lse_normalized_obj)
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initialize z from a
    z = np.log(a + 1e-30)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    elapsed = 0
    total_time = time_limit
    beta = beta_start
    
    while elapsed < total_time:
        # Continuation: increase beta
        progress = min(1.0, elapsed / (total_time * 0.7))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        a = softmax(z)
        lse_val, grad, conv = compute_lse_and_gradient(a, beta)
        
        # Update Adam
        t += 1
        z, m, v = adam_step(z, grad, m, v, t, lr=lr)
        
        # Evaluate actual objective
        a_new = softmax(z)
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
        
        elapsed = time.monotonic() - t0
    
    return best_a, best_obj

def lp_frank_wolfe(a, M):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Returns the optimal b (normalized so sum=1) or None.
    """
    n = len(a)
    num_constraints = 2*n - 1
    
    # Build constraint matrix A_ub: A_ub @ b <= M
    # A_ub[k, j] = a[k-j] if k-j in [0, n-1], else 0
    A_ub = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[k - j]
    
    b_ub = np.full(num_constraints, M)
    c = -np.ones(n)  # minimize -sum(b)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        b = res.x
        s = b.sum()
        if s > 1e-10:
            return b / s
    return None

def frank_wolfe_optimize(a0, time_limit, max_lp_iter=50):
    """
    Frank-Wolfe with LP oracles.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    gamma = 1.0  # step size parameter
    
    for it in range(max_lp_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Solpone LP
        v = lp_frank_wolfe(a, M)
        if v is None:
            break
        
        # Frank-Wolfe step: a <- (1 - step)*a + step*v
        step = 2.0 / (it + 2)  # Standard FW step size
        a_new = (1 - step) * a + step * v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for frac in [0.5, 0.25, 0.1]:
                a_try = (1 - step*frac) * a + step*frac * v
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                obj_try = 2.0 * n * fast_convolve(a_try).max()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break
    
    return best_a, best_obj

Actually, for a sequence a with flat |DFT(a)|, the auto-convolution is flat. But the inverse DFT of a flat-magnitude spectrum is generally complex and oscillatory. Taking the absolute value destroys the flatness property. So this initialization is not as good as I hoped.

- Start from a random non-negative sequence
- Iteratively: compute conv, find the max lag, reduce the "responsibility" of elements that contribute to that lag

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-method approach:
(1) Log-sum-exp smooth approximation with Adam on softmax parameterization
(2) Multiplicative mirror descent with analytical LSE gradient
(3) LP-based Frank-Wolfe (conditional gradient) steps
(4) Multi-scale continuation, diverse initializations, and periodic restarts
"""

import numpy as np
import time
import scipy.fft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = scipy.fft.rfft(a, fft_size)
    return scipy.fft.irfft(A * A, fft_size)[:size]


def compute_grad_a(a, beta):
    """
    Compute gradient of LSE objective w.r.t. a.
    LSE(a) = (1/beta)*log(sum_m exp(beta*conv[m]))
    d(LSE)/da[i] = 2 * sum_m p[m]*a[m-i] where p ~ exp(beta*conv)
    Returns (grad_a, lse_value, conv, p)
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    lse_val = conv.max() + np.log(exp_s.sum()) / beta

    # cross[i] = sum_m p[m]*a[m-i] via FFT
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = scipy.fft.rfft(p, fft_size)
    Ar = scipy.fft.rfft(a[::-1], fft_size)
    cross_full = scipy.fft.irfft(P * Ar, fft_size)[:size]
    cross = cross_full[n - 1:2 * n - 1]

    grad_a = 2.0 * cross
    return grad_a, lse_val, conv, p


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def lse_adam_optimize(a0, time_limit, beta_start=5.0, beta_end=150.0, lr=0.05, seed=0):
    """
    Optimize using Adam on softmax parameterization with LSE continuation.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    z = np.log(a + 1e-30) + 0.01 * rng.standard_normal(n)
    z = z - z.max()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    npat = 0
    max_pat = 100

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / (time_limit * 0.6))
        beta = beta_start * (beta_end / beta_start) ** progress

        a = softmax_fn(z)
        conv = fast_convolve(a)
        scaled = beta * conv
        scaled -= scaled.max()
        exp_s = np.exp(scaled)
        p = exp_s / exp_s.sum()

        # cross[i] = sum_m p[m]*a[m-i]
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        P = scipy.fft.rfft(p, fft_size)
        Ar = scipy.fft.rfft(a[::-1], fft_size)
        cross_full = scipy.fft.irfft(P * Ar, fft_size)[:size]
        cross = cross_full[n - 1:2 * n - 1]

        # grad_a[i] = 2*cross[i], grad_z[i] = a[i]*(grad_a[i] - dot(a, grad_a))
        grad_a = 2.0 * cross
        mean_g = np.dot(a, grad_a)
        grad_z = a * (grad_a - mean_g)

        t += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * grad_z * grad_z
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_fn(z)
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            npat = 0
        else:
            npat += 1

        if npat >= max_pat:
            lr *= 0.6
            if lr < 0.001:
                break
            npat = 0
            z = z + 0.01 * rng.standard_normal(n)

    return best_a, best_obj


def mirror_descent_optimize(a0, time_limit, beta=30.0, eta0=0.4, seed=0):
    """
    Multiplicative mirror descent using LSE gradient with continuation.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    eta = eta0
    patience = 0
    t0 = time.monotonic()
    cur_beta = beta
    npat_iter = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        # Continuation: sharpen over time
        progress = min(1.0, elapsed / (time_limit * 0.5))
        cur_beta = 10.0 * (50.0) ** progress

        grad_a, lse_val, conv, p = compute_grad_a(a, cur_beta)

        # Mirror descent update: a_new[i] = a[i]*exp(-eta*grad_centered[i]) / Z
        grad_centered = grad_a - np.dot(a, grad_a)
        log_factor = -eta * grad_centered
        log_factor -= log_factor.max()
        a_new = a * np.exp(log_factor)
        a_new = np.maximum(a_new, 1e-30)
        s2 = a_new.sum()
        if s2 < 1e-15:
            break
        a_new /= s2

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            npat_iter = 0
        else:
            patience += 1
            npat_iter += 1

        if patience >= 60:
            eta *= 0.7
            if eta < 0.005:
                # Perturb and restart eta
                a_best = best_a.copy()
                a_best += 0.02 * np.abs(rng.standard_normal(n))
                a_best = np.maximum(a_best, 0)
                a_best /= a_best.sum()
                a = a_best
                eta = 0.3
                patience = 0
                npat_iter = 0
            else:
                patience = 0

    return best_a, best_obj


def lp_frank_wolfe_optimize(a0, time_limit, max_iter=200, seed=0):
    """
    Frank-Wolfe with LP oracle: at each step solve LP for best vertex, step toward it.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        M = conv.max() * (1.0 + 1e-10)

        # Build LP constraint matrix
        num_con = 2 * n - 1
        A_ub = np.zeros((num_con, n))
        for k in range(n - 1):
            # conv[k] = sum_{j=0}^{k} a[j]*b[k-j]
            for j in range(k + 1):
                A_ub[k, j] = a[k - j]
            # conv[2n-2-k] = sum_{j=n-1-k}^{n-1} a[j]*b[2n-2-k-j]
            for j in range(n - 1 - k, n):
                A_ub[2 * n - 2 - k, j] = a[2 * n - 2 - k - j]
        # Middle: conv[n-1] = sum_j a[j]*b[n-1-j]
        for j in range(n):
            A_ub[n - 1, j] = a[n - 1 - j]

        b_ub = np.full(num_con, M)
        c_vec = -np.ones(n)
        bounds = [(0.0, None)] * n

        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        if not res.success:
            break

        v = res.x
        sv = v.sum()
        if sv < 1e-12:
            break
        v = v / sv

        # FW step with backtracking
        step = 2.0 / (it + 2)
        improved = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_try = (1 - step * frac) * a + step * frac * v
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try /= s_try
            obj_try = 2.0 * n * fast_convolve(a_try).max()
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved and it > 20:
            # Perturb to escape
            a_pert = a + 0.05 * np.abs(rng.standard_normal(n))
            a_pert = np.maximum(a_pert, 0)
            a_pert /= a_pert.sum()
            obj_pert = 2.0 * n * fast_convolve(a_pert).max()
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
            a = a_pert

    return best_a, best_obj


def combine_optimize(a0, time_limit, seed=0):
    """
    Sequentially apply multiple optimization methods for maximum effect.
    """
    n = len(a0)
    best_a = a0.copy()
    best_obj = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()

    # Phase A: Adam (40% of time)
    tA = time_limit * 0.4
    if time.monotonic() - t0 < tA - 1:
        a_out, obj_out = lse_adam_optimize(best_a, tA - (time.monotonic() - t0), seed=seed)
        if obj_out < best_obj:
            best_obj = obj_out
            best_a = a_out

    # Phase B: Mirror descent (30% of time)
    tB = time_limit * 0.3
    rem = time_limit - (time.monotonic() - t0)
    t_alloc = min(tB, rem)
    if t_alloc > 5:
        a_out, obj_out = mirror_descent_optimize(best_a, t_alloc, seed=seed + 1)
        if obj_out < best_obj:
            best_obj = obj_out
            best_a = a_out

    # Phase C: LP-FW (30% of time)
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10 and n <= 512:
        a_out, obj_out = lp_frank_wolfe_optimize(best_a, rem, seed=seed + 2)
        if obj_out < best_obj:
            best_obj = obj_out
            best_a = a_out

    # Phase D: Final Adam polish
    rem = time_limit - (time.monotonic() - t0)
    if rem > 5:
        a_out, obj_out = lse_adam_optimize(best_a, rem, beta_start=50.0, beta_end=200.0, lr=0.02, seed=seed + 3)
        if obj_out < best_obj:
            best_obj = obj_out
            best_a = a_out

    return best_a, best_obj


def generate_initializations(n, rng, parent):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussians
    for sf in [0.1, 0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g

    # Cosine bump
    cos_b = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_b /= cos_b.sum()
    inits['cosine'] = cos_b

    # Bimodal (edges)
    bi = np.zeros(n)
    w = max(1, n // 6)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Random with structure (sum of random gaussians)
    struct = np.zeros(n)
    for _ in range(3):
        center = rng.uniform(0, n)
        width = rng.uniform(n * 0.05, n * 0.3)
        amp = rng.uniform(0.5, 1.0)
        struct += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    struct = np.maximum(struct, 0)
    struct /= struct.sum()
    inits['structured'] = struct

    # From parent
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            pp /= pp.sum()
            inits['parent'] = pp

    return inits
