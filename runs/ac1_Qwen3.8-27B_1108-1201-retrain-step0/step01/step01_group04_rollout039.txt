"""
Step-function search for low max self-convolution using LP best-response dynamics
and smoothed-max gradient descent.

The score is 2*n*max(conv(p,p))/sum(p)^2, scale-invariant so we work on the simplex.
Key innovations over the previous approach:
  * LP best-response dynamics (Nash-equilibrium iteration) as the primary optimizer
  * Focal-lag gradient: concentrate descent on the top few bottleneck lags
  * Nesterov-momentum projected gradient for fast local convergence
  * Multi-length sweep with time allocation based on LP tractability
  * Quadratic-affine shape parameterization matched to the continuous optimum
"""

import numpy as np

try:
    from scipy.fft import fft, ifft, next_fast_len
except Exception:
    from numpy.fft import fft, ifft

    def next_fast_len(n):
        n = int(max(1, n))
        m = 1
        while m < n:
            m <<= 1
        return m

try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _HAVE_LP = True
except Exception:
    _HAVE_LP = False

def safe_normalize(p):
    """Normalize to the probability simplex {p>=0, sum(p)=1}."""
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-15:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def conv_full(p):
    """Full linear convolution of p with itself via FFT."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    """Compute 2*n*max(conv(p,p))/sum(p)^2."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    c = conv_full(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def focal_gradient(p, n_top=5):
    """
    Compute gradient of the max of conv(p,p) using a focal (top-k softmax) approach.
    Returns (gradient, c_vector, T_used, score).
    """
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0), np.empty(0), 1.0, np.inf

    c = conv_full(p)
    cmax = float(np.max(c)) if c.size else 0.0
    s = float(p.sum())
    if not np.isfinite(cmax) or cmax <= 0 or not np.isfinite(s) or s <= 0:
        return np.zeros(n), c, 1.0, np.inf

    score = float(2.0 * n * cmax / (s * s))

    # Adaptive temperature: focus on top lags
    sorted_c = np.sort(c)[::-1]
    T = 0.05 * cmax
    # If the top lag is much larger than others, use smaller T (sharper focus)
    if sorted_c.size > n_top:
        ratio = sorted_c[n_top - 1] / max(cmax, 1e-30)
        if ratio < 0.5:
            T = 0.02 * cmax
        elif ratio < 0.8:
            T = 0.08 * cmax
    T = max(T, 1e-10)

    # Softmax weights
    diff = (c - cmax) / T
    e = np.exp(diff)
    total = float(e.sum())
    if not np.isfinite(total) or total <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / total

    # Gradient: grad_j = 2 * sum_k w_k * p[k-j] (convolution of w_rev with p)
    L = next_fast_len(3 * n - 2)
    Fp = fft(p, L)
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))

    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2:n - 2:-1]

    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp

    G = np.maximum(G, 0.0)
    g = 2.0 * G
    return g, c, T, score


def smooth_surrogate_score(p, T_factor=0.03):
    """Smoothed max score using log-sum-exp."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    c = conv_full(p)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-15 or c.size == 0:
        return np.inf
    cmax = float(np.max(c))
    if cmax <= 0:
        return 0.0
    T = max(T_factor * cmax, 1e-10)
    diff = (c - cmax) / T
    e = np.exp(diff)
    total = float(e.sum())
    if not np.isfinite(total) or total <= 0:
        return float(2.0 * n * cmax / (s * s))
    surr = cmax + T * np.log(total)
    return float(2.0 * n * surr / (s * s))


def lp_best_response(p, budget_s=30.0):
    """
    Solve LP: maximize sum(g) s.t. A@g <= 1, g >= 0.
    Returns normalized g and the optimal max(f*g) value.
    """
    if not _HAVE_LP:
        return None, np.inf
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 0 or n > 1500:
        return None, np.inf
    if budget_s <= 1.0:
        return None, np.inf

    # Build constraint matrix A: A[k, j] = p[k-j] if 0<=k-j<n else 0
    # Using sparse structure: for each row k, the non-zero entries are at columns
    # j where 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(k, n-1)
    rows_list = []
    cols_list = []
    data_list = []

    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows_list.append(k)
            cols_list.append(j)
            data_list.append(p[k - j])

    A = csr_matrix((data_list, (rows_list, cols_list)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.ones(2 * n - 1, dtype=np.float64)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                      method='highs', options={'disp': False, 'time_limit': max(1.0, budget_s)})
    except Exception:
        return None, np.inf

    if res is None or res.status != 0:
        return None, np.inf

    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return None, np.inf
    g = g / s
    # The optimal max(f*g) = 1/res.fun (since we maximized sum with constraint <= 1)
    opt_max = 1.0 / max(-res.fun, 1e-30)
    return g, opt_max


def lp_iteration(p, alpha, budget_s):
    """One LP best-response iteration: mix p with its best response."""
    g, _ = lp_best_response(p, budget_s)
    if g is None:
        return p
    q = project_simplex((1.0 - alpha) * p + alpha * g)
    return q


def lp_dynamics(p, rng, time_limit, n_lp_calls=50, alpha_start=1.0, alpha_min=0.1):
    """
    Run LP best-response dynamics (fictitious play).
    Geometrically decreasing alpha.
    """
    p = safe_normalize(p)
    n = int(p.size)
    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score

    end = time.monotonic() + max(0.0, time_limit)
    per_call_time = max(1.0, time_limit / max(1, n_lp_calls))

    for i in range(n_lp_calls):
        if time.monotonic() > end - per_call_time:
            break
        alpha = alpha_start * (alpha_min / alpha_start) ** (float(i) / max(1, n_lp_calls - 1))
        alpha = max(alpha, alpha_min)
        g, opt_max = lp_best_response(p, per_call_time)
        if g is None:
            continue
        t_mix = min(1.0, alpha)
        q = project_simplex((1.0 - t_mix) * p + t_mix * g)
        sc = internal_score(q)
        if np.isfinite(sc) and sc < current_score:
            p = q
            current_score = sc
            if sc < best_score:
                best_score = sc
                best_p = q.copy()
        else:
            # Try smaller step
            for t_small in (0.3, 0.1, 0.05):
                q2 = project_simplex((1.0 - t_small) * p + t_small * g)
                sc2 = internal_score(q2)
                if np.isfinite(sc2) and sc2 < current_score:
                    p = q2
                    current_score = sc2
                    if sc2 < best_score:
                        best_score = sc2
                        best_p = q2.copy()
                    break

        if i % 10 == 0:
            remaining = end - time.monotonic()
            print("  LP iter %d/%d score=%.8f remaining=%.0fs" % (i, n_lp_calls, best_score, remaining), flush=True)

    return best_p, best_score


def gradient_descent(p, rng, time_limit, n_top=5, use_nesterov=True):
    """
    Projected gradient descent with focal-lag gradients and optional Nesterov momentum.
    """
    p = safe_normalize(p)
    n = int(p.size)
    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score

    end = time.monotonic() + max(0.0, time_limit)
    iter_no = 0
    since_improve = 0
    momentum_v = np.zeros(n, dtype=np.float64)
    mu = 0.6  # Nesterov momentum

    T_factor = 0.05

    while time.monotonic() < end:
        iter_no += 1

        # Adaptive temperature: sharpen focus as we converge
        T_factor = max(0.01, 0.05 * (0.995 ** iter_no))
        since_improve_adj = T_factor * (1.0 + 0.1 * (since_improve / 100.0))

        g, c, T, score_g = focal_gradient(p, n_top=n_top)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)

        # Adaptive step size
        p_mean = 1.0 / n
        g_max = float(np.max(np.abs(g))) if g.size else 0.0
        if g_max > 1e-300:
            base_lr = 8.0 * p_mean / g_max
        else:
            base_lr = p_mean

        lrs = (base_lr, 0.4 * base_lr, 0.15 * base_lr, 0.05 * base_lr, 0.01 * base_lr)

        accepted = False
        for lr in lrs:
            lr = float(max(1e-9, min(lr, 2.0)))
            if use_nesterov:
                # Nesterov: look-ahead step
                v_new = mu * momentum_v - lr * g
                p_lookahead = project_simplex(p + v_new)
                # Evaluate at lookahead
                c_t, _, score_t, _ = focal_gradient(p_lookahead, n_top=n_top)
                # Accept if surrogate improves
                surr_cur = smooth_surrogate_score(p, T_factor)
                surr_new = smooth_surrogate_score(p_lookahead, T_factor)

                if np.isfinite(surr_new) and surr_new < surr_cur - 1e-14:
                    momentum_v = v_new
                    p = p_lookahead
                    sc = internal_score(p)
                    if np.isfinite(sc):
                        current_score = sc
                        if sc < best_score:
                            best_score = sc
                            best_p = p.copy()
                            since_improve = 0
                        else:
                            since_improve += 1
                    accepted = True
                    break
                else:
                    # Try at current point (non-Nesterov)
                    p_new = project_simplex(p - lr * g)
                    c_t, _, score_t, _ = focal_gradient(p_new, n_top=n_top)
                    score_t = internal_score(p_new)
                    if np.isfinite(score_t) and score_t < current_score:
                        momentum_v = -lr * g
                        p = p_new
                        current_score = score_t
                        if score_t < best_score:
                            best_score = score_t
                            best_p = p.copy()
                            since_improve = 0
                        else:
                            since_improve += 1
                        accepted = True
                        break
            else:
                p_new = project_simplex(p - lr * g)
                score_t = internal_score(p_new)
                if np.isfinite(score_t) and score_t < current_score - 1e-14:
                    p = p_new
                    current_score = score_t
                    if score_t < best_score:
                        best_score = score_t
                        best_p = p.copy()
                        since_improve = 0
                    else:
                        since_improve += 1
                    accepted = True
                    break

        if not accepted:
            since_improve += 1
            # Random perturbation to escape
            if since_improve > 100:
                sig = 0.1 / n
                p = project_simplex(p + rng.normal(0, sig, n))
                current_score = internal_score(p)
            if since_improve > 300:
                p = project_simplex(0.6 * best_p + 0.4 * p + rng.normal(0, 0.2 / n, n))
                current_score = internal_score(p)
                since_improve = 0
                T_factor = 0.05
                momentum_v = np.zeros(n, dtype=np.float64)

        if iter_no % 500 == 0:
            remaining = end - time.monotonic()
            print("  GD iter %d best=%.8f cur=%.8f rem=%.0fs" % (iter_no, best_score, current_score, remaining), flush=True)

    return best_p, best_score


def make_linear(n, slope=1.0, offset=0.0):
    """Linear profile: p[j] = max(offset + slope * j/(n-1), 0)."""
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = offset + slope * x
    p = np.maximum(p, 1e-6)
    return safe_normalize(p)


def make_quadratic(n, a, b, c):
    """Quadratic: p[j] = max(a + b*x + c*x^2, 0) where x = j/(n-1)."""
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = a + b * x + c * x * x
    p = np.maximum(p, 1e-6)
    return safe_normalize(p)


def make_afine_family(n, a, b, c):
    """Affine with optional quadratic: p[j] = max(a + b*x + c*(x-0.5)^2, 0)."""
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = a + b * x + c * (x - 0.5) ** 2
    p = np.maximum(p, 1e-6)
    return safe_normalize(p)


def shape_search(n, rng, budget_s):
    """
    Search over a low-dimensional shape family to find a good initialization.
    Uses random sampling + local refinement in the shape parameter space.
    """
    if budget_s < 10:
        return make_linear(n, 1.0, 0.2), internal_score(make_linear(n, 1.0, 0.2))

    best_p = None
    best_score = np.inf
    end = time.monotonic() + budget_s

    # Parametric family: p[j] = a + b*(j/(n-1)) + c*((j/(n-1))-0.5)^2
    # Constrained: a + b*x + c*(x-0.5)^2 > 0 for x in [0,1]
    # a ranges [0.1, 2], b ranges [-3, 3], c ranges [-4, 4]

    n_trials = int(min(5000, budget_s * 15))

    for i in range(n_trials):
        if time.monotonic() > end - 2:
            break

        a = float(rng.uniform(0.2, 2.0))
        b = float(rng.uniform(-3.0, 3.0))
        c = float(rng.uniform(-4.0, 4.0))

        p = make_afine_family(n, a, b, c)
        sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p.copy()

        # Also try pure linear
        if i % 10 == 0:
            slope = float(rng.uniform(-1.0, 3.0))
            offset = float(rng.uniform(0.1, 2.0))
            p2 = make_linear(n, slope, offset)
            sc2 = internal_score(p2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_p = p2.copy()

    # Local refinement of best shape
    if best_p is not None and _HAVE_MINIMIZE and time.monotonic() < end - 8:
        def obj(params):
            a, b, c = params
            p = make_afine_family(n, a, b, c)
            return internal_score(p)

        # Get best params by inverting (roughly)
        # Actually, just do a random local search around the best
        x = np.linspace(0, 1, n)
        # Fit the best_p to the parametric form
        # p[j] ~ a + b*x + c*(x-0.5)^2
        # Least squares
        A_mat = np.column_stack([np.ones(n), x, (x - 0.5) ** 2])
        coeffs, _, _, _ = np.linalg.lstsq(A_mat, best_p, rcond=None)
        a0, b0, c0 = [float(v) for v in coeffs]

        try:
            res = minimize(obj, [a0, b0, c0], method='Nelder-Mead',
                           options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10})
            if res is not None and np.isfinite(res.fun) and res.fun < best_score:
                a_opt, b_opt, c_opt = [float(v) for v in res.x]
                best_p = make_afine_family(n, a_opt, b_opt, c_opt)
                best_score = float(res.fun)
        except Exception:
            pass

    if best_p is None:
        best_p = make_linear(n, 1.0, 0.2)
        best_score = internal_score(best_p)

    return best_p, best_score
