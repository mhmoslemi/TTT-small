import time

def compute_grad_power(a, p):
    """Compute gradient of (sum(b^p))^(1/p) where b = conv(a,a)."""
    n = len(a)
    b = np.convolve(a, a)
    Sp = np.sum(b ** p) ** (1.0 / p)
    if Sp < 1e-30:
        return np.zeros(n), b
    w = (b ** (p - 1)) / (Sp ** (p - 1))
    # grad_a = 2 * correlate(w, a, 'full')[n-1:2n-1]
    grad = 2.0 * np.correlate(w, a, 'full')[n-1:2n-1]
    return grad, b

def adam_step(a, grad, m, v, t, lr, beta1, beta2, eps):
    """Adam update step."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v

def normalize(a, S):
    """Normalize to sum S, clip to non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return a
    return a * (S / s)

def optimize(a0, t_budget, p_schedule, lr, verbose=False):
    """Optimize using Adam with power-smoothed objective."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = normalize(a0.copy(), S)
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < t_budget:
        t += 1
        # Get current p from schedule
        frac = (time.monotonic() - t0) / t_budget
        p = p_schedule(frac)
        
        grad, b = compute_grad_power(a, p)
        a, m, v = adam_step(a, grad, m, v, t, lr, 0.9, 0.999, 1e-8)
        a = normalize(a, S)
        
        # Periodic evaluation
        if t % 50 == 0 or (time.monotonic() - t0 > t_budget - 5):
            mc = b.max()  # Note: b is from before the update, so this is approximate
            # Actually let's recompute
            b = np.convolve(a, a)
            mc = b.max()
            score = 2.0 * n * mc / S**2
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if verbose:
                    print(f"  t={t}, p={p:.1f}, score={score:.6f}, max_b={mc:.4f}")
    
    # Final evaluation
    b = np.convolve(best_a, best_a)
    final_score = evaluate_sequence(best_a.tolist())
