def _compute_score_normalized(f):
    """Compute 2*n*max(convolve(f,f)) for f with sum=1."""
    n = len(f)

    return 2.0 * n * np.max(b)

def _autocorrelation(f):
    """Compute autocorrelation (linear convolution)."""
    n = len(f)
    L = 1
    while L < 2*n - 1:
        L *= 2
    F = np.fft.rfft(f, L)
    return np.fft.irfft(F * F, L)[:2*n-1]

    """Compute gradient of smooth max of autocorrelation w.r.t. f."""
    n = len(f)

    return np.maximum(v - theta, 0.0)

"""
Smoothed L-BFGS optimization of non-negative step function to minimize
normalized autocorrelation peak. Uses log-sum-exp smoothing with adaptive
temperature, softmax parameterization for constraint handling, and FFT-based
gradient computation. Multiple restarts with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import minimize


def _autocorr_fft(f):
    """Compute linear autocorrelation using FFT."""
    n = len(f)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    F = np.fft.rfft(f, L)
    ac = np.fft.irfft(F * F, L)
    return ac[:2 * n - 1]


def _score_normalized(f):
    """Compute 2*n*max(autocorr) for f with sum approximately 1."""
    n = len(f)
    b = _autocorr_fft(f)
    return 2.0 * n * float(np.max(b))


def _logsumexp(x, tau=1.0):
    """Stable log-sum-exp of tau*x."""
    tx = tau * x
    m = np.max(tx)
    if m > 700:  # overflow protection
        return m / tau + np.log(len(x)) / tau
    return (m + np.log(np.sum(np.exp(tx - m)))) / tau


def _smooth_objective_and_grad(f, tau):
    """
    Compute smooth objective (log-sum-exp of autocorrelation) and its
    gradient w.r.t. f.
    
    Returns (value, gradient) where value approximates max(autocorr(f)).
    """
    n = len(f)
    b = _autocorr_fft(f)
    b_max = np.max(b)
    
    # Softmax weights
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    # Smooth max value
    value = b_max + np.log(np.sum(exp_shifted)) / tau
    
    # Gradient: grad[i] = 2*tau*sum_j w[i+j]*f[j]
    # Use FFT: grad[i] = conv[2n-2-i] where conv = ifft(fft(w_rev)*fft(f))
    w_rev = w[::-1]
    L = 1
    while L < 3 * n - 2:
        L *= 2
    Wr = np.fft.rfft(w_rev, L)
    F = np.fft.rfft(f, L)
    conv = np.fft.irfft(Wr * F, L)
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * tau * conv[idx]
    
    return value, grad


def _softmax(g):
    """Stable softmax."""
    g_max = np.max(g)
    exp_g = np.exp(g - g_max)
    return exp_g / np.sum(exp_g)


def _obj_and_grad_g(g, tau):
    """
    Objective and gradient w.r.t. g (log-parameterization).
    f = softmax(g), minimize 2n * smooth_max(autocorr(f)).
    """
    n = len(g)
    f = _softmax(g)
    value, grad_f = _smooth_objective_and_grad(f, tau)
    
    # Chain rule: dJ/dg = (diag(f) - f f^T) @ grad_f
    # = f * (grad_f - sum(f * grad_f))
    weighted_mean = np.dot(f, grad_f)
    grad_g = f * (grad_f - weighted_mean)
    
    return 2.0 * n * value, grad_g * 2.0 * n

def _autocorr_fft_fast(f, L):
    """Compute linear autocorrelation using precomputed FFT size."""
    n = len(f)
    F = np.fft.rfft(f, L)
    ac = np.fft.irfft(F * F, L)
    return ac[:2 * n - 1]


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _lbfgs_refine(f, deadline, tau=10.0, maxiter=500):
    """
    Refine f using L-BFGS on the log-parameterized smooth objective.
    f should have sum=1.
    """
    n = len(f)
    # Convert to log-parameters (add small offset to avoid log(0))
    eps = 1e-15
    g = np.log(f + eps)
    
    # Optimize
    result = minimize(
        _obj_and_grad_g,
        g,
        args=(tau,),
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': maxiter, 'maxfun': maxiter, 'ftol': 1e-12, 'gtol': 1e-8}
    )
    
    f_opt = _softmax(result.x)
    score = _score_normalized(f_opt)
    return f_opt, score


def _make_random_start(n, rng):
    """Random positive sequence, normalized."""
    a = rng.random(n) + 0.01
    return a / np.sum(a)


def _make_two_level(n, eps, rng):
    """Two-level anti-symmetric sequence."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_cosine(n, freq, amp, phase, rng):
    """Cosine-modulated sequence."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_multifreq(n, n_freqs, amp, rng):
    """Multi-frequency cosine sequence."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_freqs):
        freq = rng.integers(1, n // 2)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S
