"""
Smooth gradient descent (FISTA) with FFT-based gradient computation
to minimize the normalized autocorrelation peak of a non-negative sequence.
Exploits convexity of the problem for guaranteed convergence to global optimum.
"""

import numpy as np
import time

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _next_pow2(n):
    return 1 << (n - 1).bit_length()

def _smooth_gradient_descent(a_init, mu, deadline, N, use_fista=True, 
                               max_iterations=500000, restart_every=100):
    """
    Minimize f_mu(a) = b_max + mu*log(sum(exp((b-b_max)/mu)))
    subject to a >= 0, sum(a) = 1, using (FISTA) gradient descent.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    
    z = a.copy()
    x_prev = a.copy()
    t = 1.0
    best = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    
    # Pre-compute FFT of a (will update each iteration)
    step = mu / 4.0  # initial step size (conservative)
    no_improve = 0
    
    for it in range(max_iterations):
        if time.monotonic() >= deadline:
            break
        
        # Compute b = convolve(z, z) using FFT
        fz = np.fft.fft(z, N)
        b = np.fft.ifft(fz * fz).real[:2*n-1]
        b = np.maximum(b, 0.0)  # numerical safety
        
        # Compute softmax weights
        b_max = np.max(b)
        p = np.exp((b - b_max) / mu)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            p = np.zeros(2*n-1)
            p[np.argmax(b)] = 1.0
        else:
            p /= p_sum
        
        # Compute gradient: grad[i] = 2 * sum_k p[k] * z[k-i]
        fp = np.fft.fft(p, N)
        grad = 2.0 * np.fft.ifft(fp * np.conj(fz)).real[:n]
        
        # Center gradient for sum(a) = 1 constraint
        grad -= np.mean(grad)
        
        # Backtracking line search
        f_current = b_max + mu * np.log(p_sum)  # smooth objective at z
        local_step = step
        x_new = z
        for bt in range(20):
            x_trial = z - local_step * grad
            x_trial = _project_simplex(x_trial, 1.0)
            # Compute f at x_trial
            fx = np.fft.fft(x_trial, N)
            bx = np.fft.ifft(fx * fx).real[:2*n-1]
            bx = np.maximum(bx, 0.0)
            bx_max = np.max(bx)
            px = np.exp((bx - bx_max) / mu)
            px_sum = np.sum(px)
            f_trial = bx_max + mu * np.log(px_sum) if px_sum > 1e-300 else bx_max
            
            if f_trial <= f_current - 0.01 * local_step * np.sum(grad * (x_trial - z)):
                x_new = x_trial
                break
            local_step *= 0.5
        else:
            x_new = _project_simplex(z - local_step * grad, 1.0)
        
        # FISTA momentum update
        if use_fista:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            z_new = x_new + (t - 1.0) / t_new * (x_new - x_prev)
            z_new = _project_simplex(z_new, 1.0)
            x_prev = z  # this is x_k in FISTA notation
            z = z_new
            t = t_new
        else:
            x_prev = z
            z = x_new
        
        # Track best (using true objective, not smooth)
        f_true = 2.0 * n * np.max(np.convolve(z, z))
        if f_true < best_score - 1e-12:

            no_improve = 0

            no_improve += 1

            # Restart FISTA momentum
            z = best.copy()
            x_prev = z.copy()
            t = 1.0
            no_improve = 0
            step = local_step
    
    return best, best_score

    a[half:] = 1.0 - eps

    S = np.sum(a)
    if S < 1e-10:

"""
Smooth gradient descent (FISTA) with FFT-based gradient computation to minimize
the normalized autocorrelation peak of a non-negative sequence. Exploits the
convexity of the problem (max of PSD quadratic forms over a simplex) for
guaranteed global convergence. Uses adaptive mu schedule and multiple
starting points for robustness.
"""

import numpy as np
import time


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target} using sorted-based method."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _next_pow2(n):
    """Next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _smooth_fista(a_init, mu, deadline, N, use_fista=True,
                  max_iterations=300000, restart_every=100,
                  verbose=False):
    """
    Minimize f_mu(a) = b_max + mu*log(sum(exp((b-b_max)/mu)))
    subject to a >= 0, sum(a) = 1, using FISTA with backtracking.
    Returns (best_a, best_true_score) where best_true_score = 2*n*max(conv(a,a)).
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n

    z = a.copy()
    x_prev = a.copy()
    t = 1.0

    b = np.fft.ifft(np.fft.fft(z, N) * np.fft.fft(z, N)).real[:2*n-1]
    b = np.maximum(b, 0.0)
    best_score = 2.0 * n * np.max(b)
    best = a.copy()

    step = mu / 2.0
    no_improve = 0
    consecutive_no_improve_large = 0

    for it in range(max_iterations):
        if time.monotonic() >= deadline:
            break

        # Compute b = conv(z,z) via FFT
        fz = np.fft.fft(z, N)
        b = np.fft.ifft(fz * fz).real[:2*n-1]
        b = np.maximum(b, 0.0)
        b_max = np.max(b)

        # Early exit if smooth objective is essentially at the true max
        p = np.exp((b - b_max) / mu)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            p = np.zeros(2*n-1)
            p[np.argmax(b)] = 1.0
            p_sum = 1.0
        else:
            p /= p_sum

        f_current = b_max + mu * np.log(p_sum)

        # Compute gradient: grad[i] = 2 * sum_k p[k] * z[k-i]
        fp = np.fft.fft(p, N)
        grad = 2.0 * np.fft.ifft(fp * np.conj(fz)).real[:n]

        # Center for sum constraint
        grad_mean = np.mean(grad)
        grad -= grad_mean

        # Backtracking line search
        local_step = step
        x_new = z
        improved = False
        for bt in range(25):
            x_trial = z - local_step * grad
            x_trial = _project_simplex(x_trial, 1.0)
            # Compute f at x_trial
            fx = np.fft.fft(x_trial, N)
            bx = np.fft.ifft(fx * fx).real[:2*n-1]
            bx = np.maximum(bx, 0.0)
            bx_max = np.max(bx)
            px = np.exp((bx - bx_max) / mu)
            px_sum = np.sum(px)
            if px_sum > 1e-300:
                f_trial = bx_max + mu * np.log(px_sum)
            else:
                f_trial = bx_max

            # Armijo condition
            directional = local_step * np.sum(grad * (x_trial - z))
            if f_trial <= f_current - 0.01 * directional:
                x_new = x_trial
                improved = True
                break
            local_step *= 0.5
        if not improved:
            x_new = _project_simplex(z - local_step * grad, 1.0)

        # FISTA momentum update
        if use_fista:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            z_new = x_new + (t - 1.0) / t_new * (x_new - x_prev)
            z_new = _project_simplex(z_new, 1.0)
            x_prev = z.copy()
            z = z_new
            t = t_new
        else:
            x_prev = z.copy()
            z = x_new

        # Track best using true objective
        fz2 = np.fft.fft(z, N)
        b_z = np.fft.ifft(fz2 * fz2).real[:2*n-1]
        b_z = np.maximum(b_z, 0.0)
        cur_score = 2.0 * n * np.max(b_z)

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = z.copy()
            no_improve = 0
            consecutive_no_improve_large = 0
            # Slightly increase step
            step = min(local_step * 1.1, 0.5)
        else:
            no_improve += 1
            consecutive_no_improve_large += 1

        if no_improve >= restart_every:
            z = best.copy()
            x_prev = z.copy()
            t = 1.0
            no_improve = 0
            step = local_step * 0.8

        if verbose and it % 10000 == 0:
            print(f"    iter {it}: score={best_score:.8f}, step={step:.6f}, "
                  f"b_max={b_max:.8f}")

        # Converged if no improvement for a very long time
        if consecutive_no_improve_large > 5000:
            break

    return best, best_score


def _make_two_level(n, eps):
    """Two-level anti-symmetric sequence."""
    a = np.ones(n)
    half = n // 2
    a[:half] = 1.0 + eps
    a[half:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_smooth_profile(n, eps, mode='cos'):
    """Smooth anti-symmetric profile."""
    i = np.arange(n, dtype=np.float64)
    if mode == 'cos':
        a = 1.0 + eps * np.cos(2.0 * np.pi * i / n)
    elif mode == 'lin':
        a = 1.0 + eps * (2.0 * i / max(1, n - 1) - 1.0)
    elif mode == 'sin':
        a = 1.0 + eps * np.sin(2.0 * np.pi * i / n)
    else:
        a = 1.0 + eps * np.cos(np.pi * i / max(1, n - 1))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_four_level(n, eps):
    """Four-level anti-symmetric sequence."""
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2*q] = 1.0 + eps/3.0
    a[2*q:3*q] = 1.0 - eps/3.0
    a[3*q:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _perturb_smooth(a, rng, scale=0.05):
    """Add smooth random noise via low-frequency Fourier components."""
    n = len(a)
    freqs = rng.integers(1, max(2, n//4), size=min(5, n//2))
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2*np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2*np.pi*f*np.arange(n)/n + phase)
    max_abs = np.max(np.abs(noise))
    if max_abs > 1e-15:
        noise /= max_abs
    a_new = a + scale * noise
    return _project_simplex(a_new, 1.0)
