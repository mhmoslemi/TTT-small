import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def compute_convolution_peak(a):
    """Compute max of a*a (full convolution) for normalized a."""
    b = np.convolve(a, a)
    return np.max(b)

def solve_lp(f, M):
    """
    Given f (sum=1, f>=0) and M = max(f*f), solve:
    maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g.
    """
    n = len(f)
    # Build constraint matrix A (sparse, banded)
    # A[k, j] = f[k-j] if 0 <= k-j < n else 0
    # k ranges from 0 to 2n-2, j ranges from 0 to n-1
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        # j must satisfy 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            idx = k - j  # 0 <= idx < n
            rows.append(k)
            cols.append(j)
            vals.append(f[idx])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A @ g <= M
    b_vec = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    return res.x

"""
LP-based iterative method to minimize normalized autocorrelation peak.
Uses constraint LP: max sum(g) s.t. (f*g)[k] <= peak(f*f), g >= 0.
Combined with subgradient polishing for final refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _score_unit(a):
    """Score = 2n * max(conv(a,a)) for unit-sum non-neg a."""
    n = len(a)
    return 2.0 * n * np.max(np.convolve(a, a))

def _build_A(f, k_indices=None):
    """Build sparse matrix where (A@g)[i] = (f*g)[k_indices[i]]."""
    n = len(f)
    if k_indices is None:
        k_indices = np.arange(2*n-1)
    
    rows_list = []
    cols_list = []
    data_list = []
    
    for row_idx, k in enumerate(k_indices):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        cnt = j_hi - j_lo + 1
        if cnt <= 0:
            continue
        rows_list.append(np.full(cnt, row_idx, dtype=np.int32))
        cols_list.append(np.arange(j_lo, j_hi + 1, dtype=np.int32))
        idx_lo = k - j_hi
        idx_hi = k - j_lo
        data_list.append(f[idx_lo:idx_hi+1].astype(np.float64).copy())
    
    if not rows_list:
        return csr_matrix((len(k_indices), n))
    
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    data = np.concatenate(data_list)
    return csr_matrix((data, (rows, cols)), shape=(len(k_indices), n))

def _solve_lp(f, M, k_indices, time_limit=20):
    """Solve: max sum(g) s.t. (f*g)[k_i] <= M for i, g >= 0."""
    n = len(f)
    A = _build_A(f, k_indices)
    c = -np.ones(n)
    b_vec = np.full(len(k_indices), M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True,
                           'dual_feasibility_tolerance': 1e-9,
                           'primal_feasibility_tolerance': 1e-9})
    if res.status != 0:
        return None
    return res.x

def _subgradient_step(f, b, top_k, eta, temperature=3.0):
    """One subgradient step with top-k peak selection."""
    n = len(f)
    b_max = np.max(b)
    
    if top_k >= len(b):
        k_idx = np.arange(len(b))
        weights = np.ones(len(k_idx)) / len(k_idx)
    else:
        k_idx = np.argpartition(b, -top_k)[-top_k:]
        b_sel = b[k_idx]
        weights = np.exp(-temperature * (b_max - b_sel) / max(b_max, 1e-15))
        weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for wi, ki in zip(weights, k_idx):
        if wi < 1e-10:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        idx_lo = ki - j_hi
        idx_hi = ki - j_lo
        grad[j_lo:j_hi+1] += 2.0 * wi * f[idx_lo:idx_hi+1]
    
    grad -= np.mean(grad)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return f.copy()
    
    f_new = f - eta * grad / gnorm
    f_new = np.maximum(f_new, 0)
    f_sum = np.sum(f_new)
    if f_sum < 1e-10:
        return f.copy()
    return f_new / f_sum

            no_improve = 0

            best_f = avg_f.copy()
            no_improve = 0

            eta = max(eta * 0.5, 1e-8)

def _make_anti_symmetric(n, eps=0.5):
    """Create anti-symmetric two-level sequence."""
    a = np.ones(n)
    a[:n//2] = 1.0 + eps
    a[n//2:] = 1.0 - eps
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def _make_smooth_anti(n, eps=0.5):
    """Smooth anti-symmetric: 1 + eps*cos(pi*i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
