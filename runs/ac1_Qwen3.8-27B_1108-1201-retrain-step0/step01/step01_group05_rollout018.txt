"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- Work with sequences normalized to sum = sqrt(2n), reducing objective to max of self-convolution.
- Primary: FISTA with log-sum-exp (smooth approximation to max) using FFT-accelerated gradients.
- Secondary: Top-k subgradient method with Nesterov acceleration for polishing.
- Tertiary: Mirror descent (multiplicative weights) for diverse exploration.
- Multiple initializations: given sequence, random, sinusoidal, and structured.
- Adaptive parameters: beta (smoothness), learning rate, top-k.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_gradient_logsumexp(a, c, beta):
    """
    Compute gradient of (1/beta)*log(sum exp(beta*c[k])) with respect to a.
    Uses FFT-based cross-correlation for O(n log n) computation.
    """
    n = len(a)
    c_max = c.max()
    # Softmax weights (numerically stable)
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total  # probability distribution over lags
    
    # grad[m] = 2 * sum_k w[k] * a[k-m] (valid range)
    # = 2 * real(ifft(fft(w, fsize) * conj(fft(a, fsize))))[:n]
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_gradient_toppk(a, c, top_k):
    """Gradient using top-k lags (exact subgradient for max)."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    
    fsize = 1 << (len(c) + n - 2).bit_length()
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista_logsumexp(a_init, n, S, max_time, rng, beta_init=1.0, beta_max=200.0):
    """
    FISTA with log-sum-exp approximation to max convolution.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()  # FISTA momentum variable
    t = 1.0
    start = time.monotonic()
    
    best_score = np.inf
    best_local = a.copy()
    
    # Initial step size
    c = fft_convolve_self(a)
    grad = fft_gradient_logsumexp(a, c, beta_init)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        step = 1.0 / (0.01 * n)
    else:
        # Use a heuristic step size
        step = 1.0 / (beta_init * 2.0 * s + 1.0)
        step = min(step, S / n * 10.0)
    
    c = fft_convolve_self(y)
    cur = c.max()
    if cur < best_score:
        best_score = cur
        best_local = y.copy()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta
        beta = min(beta_max, beta_init * (1.0 + 0.001 * it))
        
        # Compute gradient at y
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            
        grad = fft_gradient_logsumexp(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        
        # Check improvement (every few iterations to save time)
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            cur_x = c_x.max()
            if cur_x < best_score:
                best_score = cur_x
                best_local = x_new.copy()
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)  # Ensure non-negativity (approximate)
        a = x_new
        t = t_new
        
        # Adaptive step size (geometric decrease)
        if it % 200 == 0:
            step = max(step * 0.95, S / n * 0.001)
    
    return best_local, best_score


def run_toppk_nesterov(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=64):
    """Top-k subgradient with Nesterov acceleration."""
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    step = 2.0 / (n + 1.0)  # Initial step
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 200)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        grad = fft_gradient_toppk(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Normalize gradient
        grad = grad / gnorm
        
        x_new = project_simplex(y - step * gnorm * grad, S)
        
        # Check

            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- Normalize to sum = sqrt(2n), reducing objective to max of self-convolution.
- Primary: FISTA with adaptive log-sum-exp (smooth approximation to max),
  using FFT-accelerated gradient computation in O(n log n) per iteration.
- Secondary: Nesterov-accelerated top-k subgradient for precise polishing.
- Tertiary: Mirror descent (multiplicative weights) for diverse exploration.
- Multiple initializations: given sequence, random, sinusoidal, structured.
- Adaptive beta: start smooth, sharpen towards exact max.
- Coarse-to-fine: explore broadly, then polish the best candidate.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_lse(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a.
    Uses FFT cross-correlation: grad[m] = 2 * sum_k w[k] * a[k-m].
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_toppk(a, c, top_k):
    """Gradient using top-k active lags, computed via FFT cross-correlation."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total

    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]

    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

    s = a.sum()

        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    step = 1.0 / (beta_init * 2.0)
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.002 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = fft_grad_lse(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = project_simplex(y - step * grad, S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        step = 1.0 / (beta * 2.0 * (1.0 + 0.0005 * it))
        step = max(step, 1e-6)

        if it % 3000 == 0 and rng.random() < 0.15:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=128):
    """Nesterov-accelerated top-k subgradient method with FFT gradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 150)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = fft_grad_toppk(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.005 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 4000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def run_mirror(a_init, n, S, max_time, rng, eta_init=0.3, top_k=10):
    """Mirror descent (multiplicative weights) with FFT-based subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        eta = eta_init / (1.0 + 0.002 * it)
        grad = fft_grad_toppk(a, c, top_k)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_new * (S / s_new)

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 12.0)

    global_best = np.inf
    global_a = None

    # --- Collect initializations ---
    inits = []

    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0, 1000)))
    except Exception:
        pass

    if inits:
        main_len = len(inits[0][1])
    else:
        main_len = 8192

    # Add structured inits at the main length
    n = main_len
    tt = np.linspace(0, 1, n, endpoint=False)

    inits.append(("rand_main", rng.random(n) + 0.05))
    inits.append(("sin3_7", 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) +
                  0.2 * np.cos(2 * np.pi * 7 * tt)))
    inits.append(("bump", np.exp(-0.5 * ((tt - 0.5) * 6.0) ** 2) + 0.05))
    inits.append(("lin_ramp", tt + 0.1 + 0.5 * np.sin(2 * np.pi * 11 * tt)))

    # Also try a different length for diversity
    n2 = main_len // 2 if main_len > 4096 else main_len * 2
    if n2 != main_len and 1024 <= n2 <= 32768:
        tt2 = np.linspace(0, 1, n2, endpoint=False)
        inits.append(("rand_alt", rng.random(n2) + 0.05))
        inits.append(("sin_alt", 1.0 + 0.4 * np.sin(2 * np.pi * 5 * tt2)))

    n_inits = len(inits)
    usable = max(1.0, deadline - time.monotonic())
    per_init = usable * 0.75 / n_inits

    print("=== Phase 1: Exploration (%d inits, %.1f sec each) ===" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 60:
            break
        n = len(arr)
        if n < 64:
            continue
        S = np.sqrt(2.0 * float(n))
        t0 = time.monotonic()

        # FISTA with log-sum-exp
        seg_fista = per_init * 0.55
        a_fista, score_fista = run_fista(arr, n, S, seg_fista, rng,
                                          beta_init=3.0, beta_max=200.0)
        a_norm = safe_normalize(a_fista, n)
        c = fft_convolve_self(a_norm)
        score = c.max()
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            print("  FISTA [%s]: %.6f" % (name, global_best))

        # Nesterov top-k
        rem = per_init - (time.monotonic() - t0)
        if rem > 2:
            a_pk, score_pk = run_nesterov_toppk(a_fista, n, S, rem, rng,
                                                 top_k_init=3, top_k_max=64)
            a_norm2 = safe_normalize(a_pk, n)
            c2 = fft_convolve_self(a_norm2)
            if c2.max() < global_best:
                global_best = c2.max()
                global_a = a_norm2.copy()
                print("  Nesterov [%s]: %.6f" % (name, global_best))

    print("=== After exploration: global_best=%.6f ===" % global_best)

    # --- Phase 2: Polish the best ---
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        remaining = deadline - time.monotonic()

        # High-beta FISTA polish
        seg = max(1.0, remaining * 0.25)
        a_p1, s1 = run_fista(global_a, n, S, seg, rng,
                              beta_init=50.0, beta_max=500.0)
        a_p1 = safe_normalize(a_p1, n)
        c1 = fft_convolve_self(a_p1)
        if c1.max() < global_best:
            global_best = c1.max()
            global_a = a_p1.copy()
            print("  Polish FISTA high-beta: %.6f" % global_best)

        # Top-k Nesterov polish with high k
        remaining2 = deadline - time.monotonic()
        seg2 = max(1.0, remaining2 * 0.25)
        a_p2, s2 = run_nesterov_toppk(global_a, n, S, seg2, rng,
                                       top_k_init=20, top_k_max=256)
        a_p2 = safe_normalize(a_p2, n)
        c2 = fft_convolve_self(a_p2)
        if c2.max() < global_best:
            global_best = c2.max()
            global_a = a_p2.copy()
            print("  Polish Nesterov high-k: %.6f" % global_best)

        # Mirror descent polish
        remaining3 = deadline - time.monotonic()
        seg3 = max(1.0, remaining3 * 0.2)
        a_p3, s3 = run_mirror(global_a, n, S, seg3, rng,
                               eta_init=0.05, top_k=30)
        a_p3 = safe_normalize(a_p3, n)
        c3 = fft_convolve_self(a_p3)
        if c3.max() < global_best:
            global_best = c3.max()
            global_a = a_p3.copy()
            print("  Polish Mirror: %.6f" % global_best)

        # Final fine polish
        remaining4 = deadline - time.monotonic()
        if remaining4 > 3:
            a_p4, s4 = run_nesterov_toppk(global_a, n, S, remaining4, rng,
                                           top_k_init=64, top_k_max=512)
            a_p4 = safe_normalize(a_p4, n)
            c4 = fft_convolve_self(a_p4)
            if c4.max() < global_best:
                global_best = c4.max()
                global_a = a_p4.copy()
                print("  Final polish: %.6f" % global_best)

    # --- Phase 3: Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.6f" %
          (len(global_a), global_best))
    return [float(x) for x in global_a]
