"""
Multi-phase optimization for minimizing autocorrelation peak of non-negative sequences.
Combines gradient-based optimization (softmax parameterization, Adam, annealed soft-max)
with diverse initializations, restart mechanisms, and targeted peak-reduction local search.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def soft_max_value_and_weights(c, tau):
    """Soft-max value and weights."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    return value, weights

def grad_soft_max_conv(a, tau):
    """Compute value and gradient of soft_max(conv(a,a), tau) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_value_and_weights(c, tau)
    # g_a[j] = 2 * sum_m a[m] * w[m+j]
    # = 2 * convolve(a[::-1], w)[j + n - 1]
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g_a = 2.0 * corr[n-1 : 2*n-1]
    return value, g_a, c

def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def local_score(a):
    """Compute the score directly."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def optimize_with_restarts(n, theta_init, time_limit, rng, lr_base=3.0, tau_start=50.0, tau_end=0.2):
    """
    Gradient optimization with restart mechanism.
    tau_start and tau_end are relative to n (will be divided by n).
    """
    S = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = np.inf
    no_improve_steps = 0
    restart_threshold = 3000

    
    tau_start_abs = tau_start / n
    tau_end_abs = tau_end / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Use a more aggressive cooling: reach low tau faster
        tau = tau_start_abs * (tau_end_abs / tau_start_abs) ** (elapsed_frac ** 0.7)
        
        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)

        m2_hat = momentum2 / (1.0 - beta2 ** step)

def peak_reduction_search(a_init, time_limit, rng):
    """
    Targeted local search that specifically tries to reduce the peak of the autocorrelation.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Find the peak location
        c = fft_convolve(current, current)
        peak_val = np.max(c)
        peak_idx = int(np.argmax(c))
        
        # Find all lags that are close to the peak
        threshold = peak_val * 0.95
        hot_lags = np.where(c >= threshold)[0]
        
        # For each hot lag, find the main contributors
        # and try to reduce them
        candidate = current.copy()
        improved = False
        
        for lag in hot_lags[:5]:  # Process top 5 hot lags
            # Contributors to C[lag]: a[i]*a[lag-i] for valid i
            contribs = np.zeros(n)
            for i in range(n):
                j = lag - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            
            # Find the top contributors
            top_k = max(1, n // 32)
            top_idx = np.argsort(contribs)[-top_k:]
            
            # Reduce them slightly
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            improved = True
        
        if not improved:
            # Random perturbation
            candidate = current * np.exp(rng.normal(0, 0.02, size=n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)
        
        # Accept
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(current_score * 0.001, 1e-10)):
            current = candidate
            current_score = new_score
    
    return best, best_score

def make_linear_ramp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 + alpha * t
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_quadratic_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 + alpha * (t - 0.5) ** 2
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_gaussian_init(n, center, width, rng):
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 + np.exp(-(t - center)**2 / (2 * width**2))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_lowpass_random_init(n, rng, smoothness=0.5):
    """Random sequence with low-frequency components (smooth)."""
    # Generate random in frequency domain, keep only low frequencies
    spectrum = np.zeros(n)
    n_keep = max(1, int(n * smoothness))
    spectrum[:n_keep] = rng.standard_normal(n_keep)
    spectrum[n//2 - n_keep:n//2] = rng.standard_normal(min(n_keep, n))
    a = np.fft.irfft(spectrum, n)
    a = a - np.min(a) + 0.1  # Make non-negative
    a += 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

"""
Multi-phase optimization for minimizing the autocorrelation peak of non-negative
sequences. Uses combined soft-max + L2 objective, diverse initializations,
adaptive restarts, and targeted peak-reduction local search.
Key improvement: the L2(C) term acts as a "flattening" regularizer that guides
optimization toward flat autocorrelation, complementing the soft-max peak target.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def soft_max_value_and_weights(c, tau):
    """Compute soft-max value and corresponding weights."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    return value, weights


def grad_for_objective(a, tau, beta_l2):
    """
    Compute gradient of combined objective: soft_max(C, tau) + beta_l2 * sum(C^2)/(2n-1)
    Returns (objective_value, grad_wrt_a, C).
    """
    n = len(a)
    c = fft_convolve(a, a)
    value_sm, w = soft_max_value_and_weights(c, tau)
    
    # Soft-max gradient: g_sm[j] = 2 * sum_m a[m] * w[m+j]
    rev_a = a[::-1]
    corr_sm = fft_convolve(rev_a, w)
    g_sm = 2.0 * corr_sm[n - 1: 2 * n - 1]
    
    # L2 gradient: g_l2[j] = 4 * sum_m a[m] * C[m+j] / (2n-1)
    corr_l2 = fft_convolve(rev_a, c)
    g_l2 = 4.0 * corr_l2[n - 1: 2 * n - 1] / (2 * n - 1)
    
    value = value_sm + beta_l2 * np.sum(c * c) / (2 * n - 1)
    g_a = g_sm + beta_l2 * g_l2
    
    return value, g_a, c


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Compute the evaluation score directly (same formula as evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def optimize_gradient(n, theta_init, time_limit, rng, lr_base=3.0, 
                       tau_start_mult=40.0, tau_end_mult=0.15,
                       beta_l2_start=1.0, beta_l2_end=0.1,
                       restart_interval=2000, restart_sigma=0.8):
    """
    Gradient optimization with combined soft-max + L2 objective,
    Adam optimizer, annealed temperature, and adaptive restarts.
    """
    S = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = np.inf
    no_improve = 0
    plateau_best = best_score
    
    tau_start = tau_start_mult / n
    tau_end = tau_end_mult / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Aggressive cooling: reach low tau in first 60% of budget
        cool_frac = min(1.0, elapsed_frac / 0.6)
        tau = tau_start * (tau_end / tau_start) ** (cool_frac ** 0.8)
        # L2 weight: start high (flattening), decrease over time
        beta_l2 = beta_l2_start + (beta_l2_end - beta_l2_start) * elapsed_frac
        
        a = softmax_normalize(theta) * S
        value, g_a, c = grad_for_objective(a, tau, beta_l2)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)
        
        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        
        # Learning rate with warmup
        if step < 100:
            lr = lr_base * step / 100.0
        else:
            lr = lr_base / np.sqrt(step - 100 + 1)
        
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Evaluate actual score periodically (not every step for speed)
        if step % 5 == 0 or step < 50:
            a_check = softmax_normalize(theta) * S
            score = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (S * S)
            
            if np.isfinite(score) and score < best_score - 1e-12:
                best_score = score
                best_theta = theta.copy()
                no_improve = 0
            else:
                no_improve += 5
        
        # Restart mechanism
        if no_improve > restart_interval:
            theta = best_theta + rng.normal(0, restart_sigma, size=n)
            momentum1 = np.zeros(n)
            momentum2 = np.zeros(n)
            step = 0
            no_improve = 0
        
        if step % 5000 == 0:
            a_check = softmax_normalize(theta) * S
            score = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (S * S)
            print(f"        step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, beta_l2={beta_l2:.3f}")
    
    return best_theta, best_score


def peak_reduction_search(a_init, time_limit, rng):
    """
    Targeted local search that specifically attacks the peak of the autocorrelation.
    Uses multiple strategies: direct peak reduction, block moves, and symmetrization.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.002 * current_score
    cooling = 0.9998
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.40:
            # Targeted: reduce peak contributors
            c = fft_convolve(current, current)
            peak_val = np.max(c)
            # Find top 3 hot lags
            sorted_lags = np.argsort(c)[::-1]
            for lag in sorted_lags[:3]:
                if c[lag] < peak_val * 0.9:
                    break
                contribs = np.zeros(n)
                for i in range(n):
                    j = lag - i
                    if 0 <= j < n:
                        contribs[i] = current[i] * current[j]
                top_k = max(2, min(n // 16, 32))
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
                # Compensate by increasing low-contribution positions
                other = np.setdiff1d(np.arange(n), top_idx)
                if len(other) > 0:
                    n_inc = min(10, len(other))
                    inc_idx = rng.choice(other, size=n_inc, replace=False)
                    candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.01, size=n_inc))
                    
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
            
        elif r < 0.65:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
                
        elif r < 0.75:
            # Partial symmetrization + noise
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.015, size=n))
            
        elif r < 0.85:
            # Oscillatory perturbation (helps flatten autocorrelation)
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            t = np.arange(n)
            perturbation = np.exp(rng.normal(0, 0.02)) * (1 + 0.1 * np.cos(2 * np.pi * freq * t / n + phase))
            candidate = current * perturbation
            
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.015, size=n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if iter_count % 500 == 0:
                    print(f"        polish iter={iter_count}, best={best_score:.6f}")
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best, best_score


def make_ramp_init(n, alpha, direction, rng):
    """Linear ramp initialization."""
    t = np.arange(n, dtype=np.float64) / n
    if direction == 1:
        a = 1.0 + alpha * t
    else:
        a = 1.0 - alpha * t
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_parabolic_init(n, alpha, rng):
    """Parabolic (U or inverted-U) initialization."""
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 + alpha * (t - 0.5) ** 2 * 4
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bump_init(n, center, width, amplitude, rng):
    """Gaussian bump initialization."""
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 + amplitude * np.exp(-(t - center) ** 2 / (2 * width ** 2))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_sinusoidal_init(n, freq, amp, rng):
    """Sinusoidal modulation initialization."""
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 + amp * np.sin(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_lowpass_init(n, rng, n_freqs=5):
    """Low-frequency random initialization (smooth)."""
    spectrum = np.zeros(n)
    for k in range(1, min(n_freqs, n // 2)):
        spectrum[k] = rng.standard_normal()
        spectrum[n - k] = spectrum[k]
    a = np.fft.irfft(spectrum, n)
    a = a - np.min(a) + 0.5
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_psd_match_init(n, rng):
    """
    Initialize by matching a target power spectrum that encourages flat autocorrelation.
    Target: flat power spectrum (white noise in frequency domain) gives flat circular autocorrelation.
    """
    # Generate random phases with flat magnitude
    n_half = n // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, size=n_half)
    magnitudes = np.ones(n_half)
    # Add some structure to avoid being perfectly random
    for k in range(1, min(10, n_half)):
        magnitudes[k] = 1.0 + 0.3 * rng.random()
        if n - k >= 0 and n - k < n_half:
            magnitudes[n - k] = magnitudes[k]
    # Inverse FFT to get time domain
    spectrum = magnitudes * np.exp(1j * phases)
    a = np.fft.irfft(spectrum, n)
    a = np.abs(a) + 0.1
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search:
    Phase 1: Gradient optimization with diverse initializations and multiple lengths.
    Phase 2: Targeted peak-reduction local search on best candidate.
    Phase 3: Additional exploration and final polish if time permits.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 25.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Evaluate provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()
    
    # Search configurations: (length, list of init types with parameters)
    # Focus on lengths around 256-768 where optimization is tractable
    length_init_configs = [
        (256, ['ramp+', 'ramp-', 'para', 'bump', 'sine', 'lowpass', 'psd', 'random']),
        (320, ['ramp+', 'para', 'bump', 'sine', 'lowpass']),
        (384, ['ramp+', 'ramp-', 'para', 'bump', 'sine']),
        (448, ['ramp+', 'para', 'bump', 'lowpass']),
        (512, ['ramp+', 'para', 'bump', 'sine']),
        (640, ['ramp+', 'para', 'bump']),
        (768, ['ramp+', 'para']),
        (1024, ['ramp+', 'bump']),
    ]
    
    # Time allocation: 55% to gradient, 30% to polish, 15% to exploration
    grad_time_total = time_budget * 0.55
    polish_time_total = time_budget * 0.30
    explore_time_total = time_budget * 0.15
    
    total_inits = sum(len(inits) for _, inits in length_init_configs)
    time_per_init = grad_time_total / total_inits
    
    print(f"=== Phase 1: Gradient optimization ({total_inits} configs, ~{time_per_init:.1f}s each) ===")
    
    for n, init_types in length_init_configs:
        if time.monotonic() > deadline - 60:
            break
        
        for init_type in init_types:
            if time.monotonic() > deadline - 30:
                break
            
            # Generate initial sequence
            if init_type == 'ramp+':
                a0 = make_ramp_init(n, 0.3 + 0.7 * rng.random(), 1, rng)
            elif init_type == 'ramp-':
                a0 = make_ramp_init(n, 0.3 + 0.7 * rng.random(), -1, rng)
            elif init_type == 'para':
                a0 = make_parabolic_init(n, 0.3 + 0.7 * rng.random(), rng)
            elif init_type == 'bump':
                center = 0.15 + 0.7 * rng.random()
                width = 0.08 + 0.15 * rng.random()
                amp = 0.3 + 0.5 * rng.random()
                a0 = make_bump_init(n, center, width, amp, rng)
            elif init_type == 'sine':
                freq = 1 + int(rng.integers(0, 5))
                amp = 0.2 + 0.4 * rng.random()
                a0 = make_sinusoidal_init(n, freq, amp, rng)
            elif init_type == 'lowpass':
                n_freqs = 3 + int(rng.integers(0, 8))
                a0 = make_lowpass_init(n, rng, n_freqs)
            elif init_type == 'psd':
                a0 = make_psd_match_init(n, rng)
            elif init_type == 'random':
                a0 = 1.0 + 0.3 * rng.standard_normal(n)
                a0 = np.clip(a0, 0.01, None)
            else:
                a0 = np.ones(n)
            
            a0 = np.clip(a0, 0.01, 1000.0)
            theta_init = np.log(a0 / np.sum(a0) + 1e-10)
            
            # Hyperparameters with some randomization
            lr_base = 2.0 + 2.0 * rng.random()
            tau_start_mult = 25.0 + 35.0 * rng.random()
            tau_end_mult = 0.1 + 0.4 * rng.random()
            beta_l2_start = 0.5 + 1.5 * rng.random()
            
            t0 = time.monotonic()
            time_left_for_init = min(time_per_init, deadline - time.monotonic() - 10)
            
            if time_left_for_init < 5:
                continue
                
            theta_best, grad_score = optimize_gradient(
                n, theta_init, time_left_for_init, rng,
                lr_base=lr_base,
                tau_start_mult=tau_start_mult,
                tau_end_mult=tau_end_mult,
                beta_l2_start=beta_l2_start,
                beta_l2_end=0.05,
                restart_interval=1500,
                restart_sigma=0.6 + 0.4 * rng.random()
            )
            elapsed = time.monotonic() - t0
            
            a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n)
            actual_score = local_score(a_best)
            print(f"  n={n}, init='{init_type}', score={actual_score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")
            
            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} (n={n}, {init_type}) ***")
    
    # Phase 2: Peak reduction local search on best candidate
    print(f"\n=== Phase 2: Peak reduction search ===")
    if best_seq is not None:
        polish_time = min(polish_time_total, max(30.0, deadline - time.monotonic() - 30))
        print(f"  Starting, score={best_score:.6f}, time={polish_time:.1f}s")
        polished, polish_score = peak_reduction_search(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
    
    # Phase 3: Try to improve with a second gradient pass on the best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 15
        n_best = len(best_seq)
        theta_init2 = np.log(best_seq / np.sum(best_seq) + 1e-10)
        
        if remaining > 30:
            print(f"\n=== Phase 3: Second gradient pass (n={n_best}, t={remaining:.1f}s) ===")
            theta_best2, _ = optimize_gradient(
                n_best, theta_init2, remaining * 0.5, rng,
                lr_base=1.5, tau_start_mult=15.0, tau_end_mult=0.1,
                beta_l2_start=0.3, beta_l2_end=0.05,
                restart_interval=1000, restart_sigma=0.3
            )
            a_best2 = softmax_normalize(theta_best2) * np.sqrt(2.0 * n_best)
            score2 = local_score(a_best2)
            print(f"  Score after 2nd pass: {score2:.6f}")
            if score2 < best_score:
                best_score = score2
                best_seq = a_best2.copy()
                print(f"    *** New best: {best_score:.6f} ***")
            
            # Final micro-polish
            if time.monotonic() < deadline - 10 and best_seq is not None:
                micro_time = min(20.0, deadline - time.monotonic() - 10)
                if micro_time > 5:
                    polished2, score_p2 = peak_reduction_search(best_seq, micro_time, rng)
                    if score_p2 < best_score:
                        best_score = score_p2
                        best_seq = polished2.copy()
                        print(f"    After micro-polish: {best_score:.6f}")
    
    # Final validation
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
