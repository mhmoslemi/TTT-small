def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_smoothed_grad(a, t):
    n = len(a)
    conv = fast_convolve(a)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return np.zeros(n), conv, 0.0
    
    scaled = t * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    thresh = w.max() * 1e-6
    active_idx = np.where(w > thresh)[0]
    
    for k_int in active_idx:
        k = int(k_int)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]
    
    return grad, conv, max_conv

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam-adapted multiplicative
mirror descent with soft-max peak smoothing, exponential temperature annealing,
log-space perturbations for local optima escape, and multi-start over sequence
lengths 64-4096 with diverse structured initializations.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_smoothed_grad(a, t):
    """
    Compute the soft-max smoothed subgradient of max(conv(a,a)) w.r.t. a.
    Returns (gradient_vector, convolution_array, max_conv_value).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return np.zeros(n), conv, 0.0

    scaled = t * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    thresh = w.max() * 1e-7
    active_idx = np.where(w > thresh)[0]

    for k_int in active_idx:
        k = int(k_int)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]

    return grad, conv, max_conv


def optimize_sequence(a0, time_limit, t_start=50.0, t_end=500.0,
                      lr0=0.08, beta1=0.9, beta2=0.999,
                      perturb_every=120, perturb_sigma0=0.12,
                      seed_offset=0):
    """
    Adam-adapted multiplicative mirror descent on the probability simplex.
    Uses soft-max smoothed gradient with exponential temperature annealing,
    linear LR decay, and log-space Gaussian perturbations.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    m = np.zeros(n)
    v = np.zeros(n)
    eps = 1e-8

    t0 = time.monotonic()
    deadline = t0 + time_limit

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    patience = 0
    rng = np.random.default_rng(seed_offset + int(t0 * 100000) % 1000000)

    # Estimate total iterations for annealing schedule
    # Roughly 100-200 iterations per second for n<=1024
    est_iters_per_sec = max(50, 200000 // max(n, 64))
    total_est = max(100, int(time_limit * est_iters_per_sec))

    while time.monotonic() < deadline:
        it += 1
        frac = min(1.0, it / total_est)

        # Exponential temperature annealing
        t = t_start * (t_end / t_start) ** frac
        # Linear learning rate decay
        lr = lr0 * (1.0 - 0.98 * frac)
        if lr < 1e-5:
            lr = 1e-5

        grad, conv, max_conv = compute_smoothed_grad(a, t)
        obj = 2.0 * n * max_conv

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = m_hat / (np.sqrt(v_hat) + eps)
        step_centered = step - step.mean()
        a = a * np.exp(-lr * step_centered)
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        # Periodic small perturbation
        if it % perturb_every == 0:
            sigma = perturb_sigma0 * (1.0 - 0.85 * frac)
            if sigma > 0.003:
                noise = rng.normal(0.0, sigma, n)
                a = a * np.exp(noise)
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum > 1e-15:
                    a /= a_sum

        # Patience-based large perturbation (escape local optima)
        if patience > 100:
            patience = 0
            sigma = 0.25 * (1.0 - 0.4 * frac)
            if sigma > 0.01:
                noise = rng.normal(0.0, sigma, n)
                a = a * np.exp(noise)
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum > 1e-15:
                    a /= a_sum
                else:
                    a = best_a.copy()

    return best_a, best_obj


def make_initializations(n, rng, parent=None):
    """Generate diverse structured initializations for sequence length n."""
    x = np.arange(n, dtype=np.float64)
    results = {}

    # Flat uniform
    results['flat'] = np.ones(n) / n

    # Hann window (raised cosine)
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    hann = np.maximum(hann, 1e-10)
    results['hann'] = hann / hann.sum()

    # Gaussian bumps with different widths
    for sf in [0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        results[f'g{int(sf * 100)}'] = g / g.sum()

    # Bimodal (concentrated at edges)
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    results['bi'] = bi / bi.sum()

    # Periodic bumps
    period = max(2, n // 6)
    bumps = np.zeros(n)
    for p in range(0, n, period):
        bumps[p:min(p + period // 2, n)] = 1.0
    if bumps.sum() > 0:
        results['bumps'] = bumps / bumps.sum()

    # Random Dirichlet (concentrated)
    results['rand_c'] = rng.dirichlet(np.ones(n) * 0.3)
    # Random Dirichlet (spread)
    results['rand_s'] = rng.dirichlet(np.ones(n) * 3.0)

    # Chirp magnitude
    phase = 2.0 * np.pi * 0.25 * x * x / max(1, n)
    chirp_mag = np.abs(np.cos(phase)) + 0.1
    results['chirp'] = chirp_mag / chirp_mag.sum()

    # Symsym (symmetric cosine)
    sym = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n + 1)))
    sym = np.maximum(sym, 0.0)
    if sym.sum() > 1e-10:
        results['sym'] = sym / sym.sum()

    # Parent-derived
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0)
            if pp.sum() > 1e-10:
                results['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                results['parent'] = pp / pp.sum()

    return results


def run_pipeline(seed, budget_s):
    """Main optimization pipeline: scan, deep optimize, polish."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Loaded parent sequence: n={p.size}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass

    # === Phase 1: Quick Scan ===
    print("=== Phase 1: Quick Scan ===")
    lengths = [64, 128, 256, 512, 1024, 2048, 4096]

    configs = []
    for n in lengths:
        inits = make_initializations(n, rng, parent)
        for name, a0 in inits.items():
            configs.append((n, name, a0))

    n_cfg = len(configs)
    p1_budget = min(350.0, (deadline - time.monotonic()) * 0.3)
    t_each = max(4.0, p1_budget / max(1, n_cfg))

    print(f"  {n_cfg} configs, ~{t_each:.1f}s each, budget={p1_budget:.0f}s")

    results = []
    for i, (n, name, a0) in enumerate(configs):
        if time.monotonic() > deadline - 200:
            break
        ta = min(t_each, deadline - time.monotonic() - 200)
        if ta < 3:
            break

        a_opt, obj = optimize_sequence(a0, ta,
                                       t_start=40.0, t_end=300.0,
                                       lr0=0.08, seed_offset=i)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc):
            results.append((sc, n, name, a_opt))
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  n={n:4d} {name:10s}: {sc:.6f}")

    print(f"  Phase 1 best: {best_score:.6f}")

    # === Phase 2: Deep Optimization ===
    results.sort(key=lambda x: x[0])

    # Select top configs with length diversity
    tops = []
    seen_n = set()
    for sc, n, name, a_opt in results:
        if len(tops) < 12:
            tops.append((n, name, a_opt))
            seen_n.add(n)
        elif n not in seen_n and len(tops) < 16:
            tops.append((n, name, a_opt))
            seen_n.add(n)

    if best_seq is not None:
        tops.append((len(best_seq), 'best', best_seq.copy()))

    rem = deadline - time.monotonic() - 100
    t_each2 = max(15.0, rem / max(1, len(tops)))

    print(f"=== Phase 2: Deep Optimization ({len(tops)} configs, ~{t_each2:.0f}s each) ===")

    param_sets = [
        (50.0, 400.0, 0.06),
        (100.0, 800.0, 0.04),
        (30.0, 200.0, 0.10),
    ]

    for i, (n, name, a0) in enumerate(tops):
        if time.monotonic() > deadline - 100:
            break
        ta = min(t_each2, deadline - time.monotonic() - 100)
        if ta < 10:
            break

        for (ts, te, lr) in param_sets:
            if time.monotonic() > deadline - 100:
                break
            t_single = ta / 3.0
            a_opt, obj = optimize_sequence(a0, t_single,
                                           t_start=ts, t_end=te,
                                           lr0=lr, seed_offset=1000 + i * 7)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  Deep n={n:4d} {name:10s} t=[{ts:.0f},{te:.0f}]: {sc:.6f}")

    # === Phase 3: Final Polish ===
    if best_seq is not None and time.monotonic() < deadline - 50:
        rem = deadline - 50 - time.monotonic()
        n = len(best_seq)
        print(f"=== Phase 3: Final Polish (n={n}, {rem:.0f}s) ===")

        polish_sets = [
            (200.0, 2000.0, 0.02),
            (500.0, 4000.0, 0.01),
            (100.0, 1000.0, 0.03),
            (1000.0, 8000.0, 0.005),
        ]

        for i, (ts, te, lr) in enumerate(polish_sets):
            if rem < 8:
                break
            t_single = rem / max(1, len(polish_sets) - i)
            a_opt, obj = optimize_sequence(best_seq, t_single,
                                           t_start=ts, t_end=te,
                                           lr0=lr,
                                           perturb_every=200,
                                           perturb_sigma0=0.05,
                                           seed_offset=5000 + i)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  Polish t=[{ts:.0f},{te:.0f}] lr={lr}: {sc:.6f}")
            rem = deadline - 50 - time.monotonic()

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return best_seq


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Minimizes 2n*max(conv(a,a))/(sum(a))^2 using
    Adam-adapted multiplicative mirror descent with soft-max smoothing,
    exponential temperature annealing, and log-space perturbations.
    Multi-start over n=64..4096 with diverse structured initializations.
    Three-phase strategy: scan, deep optimize, polish.
    """
    result = run_pipeline(seed=seed, budget_s=budget_s)
    return [float(x) for x in result]
