Actually, let me reconsider the multiplicative update approach. The idea is:

The multiplicative update: f_new[i] = f[i] * (μ / r[i]) where μ = mean(r over support)

The multiplicative update naturally preserves non-negativity (since f_i >= 0 and μ/r_i > 0).

        noise += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    noise_max = np.max(np.abs(noise))
    if noise_max > 1e-15:
        noise /= noise_max
    f_new = f + scale * noise
    return _project_simplex(f_new)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-strategy search for the best non-negative sequence minimizing
    2*n*max(self-convolution)/sum^2. Uses FISTA with entropic smoothing,
    IPF multiplicative updates, and diverse initialization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    n = 6121

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(n)
    parent = np.clip(parent, 0.0, None)
    if parent.size != n:
        parent = np.ones(n)
    S_parent = np.sum(parent)
    if S_parent > 1e-10:
        parent = parent / S_parent
    else:
        parent = np.ones(n) / n

    print(f"Parent score: {_compute_score(parent):.6f}")

    # Global best tracking
    global_best = parent.copy()
    global_best_score = _compute_score(parent)

    # Generate diverse starting points
    starts = [
        ("parent", parent.copy()),
        ("two_level_0.5", _make_two_level(n, 0.5)),
        ("two_level_0.8", _make_two_level(n, 0.8)),
        ("smooth_0.7", _make_smooth(n, 0.7)),
        ("kaiser", _make_kaiser(n, 5.0)),
        ("four_level", _make_four_level(n, 0.6)),
        ("osc_3", _make_oscillating(n, 3, 0.5)),
        ("osc_7", _make_oscillating(n, 7, 0.4)),
    ]

    # Time allocation: parent gets more time
    total_time = deadline - time.monotonic()
    n_starts = len(starts)
    # Parent gets 40%, rest split 60%
    time_parent = 0.4 * total_time
    time_rest = 0.6 * total_time / (n_starts - 1)

    print(f"Total time budget: {total_time:.1f}s, {n_starts} starts")

    for i, (name, start_f) in enumerate(starts):
        if time.monotonic() >= deadline:
            break
        if i == 0:
            t_alloc = min(time_parent, deadline - time.monotonic())
        else:
            t_alloc = min(time_rest, deadline - time.monotonic())
        if t_alloc < 5:
            t_alloc = max(5, deadline - time.monotonic())

        start_time = time.monotonic()
        best_f, best_s = _combined_optimize(start_f, start_time + t_alloc, seed=seed + i * 100)
        elapsed = time.monotonic() - start_time

        if best_s < global_best_score:
            global_best = best_f.copy()
            global_best_score = best_s
            print(f"  [{name}] NEW BEST: {global_best_score:.6f} (took {elapsed:.1f}s)")
        else:
            print(f"  [{name}] score={best_s:.6f} (no improvement, best={global_best_score:.6f})")

    # Phase 2: Intensive refinement of the global best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\nPhase 2: Refining best (score={global_best_score:.6f}) with {remaining:.1f}s")

        # Multiple refinement passes with different parameters
        refine_budget = remaining * 0.7
        per_refine = refine_budget / 4

        for r in range(4):
            if time.monotonic() >= deadline:
                break
            if r == 0:
                # Fine FISTA with very low temperature
                best_f, best_s = _fista_optimize(
                    global_best, time.monotonic() + per_refine,
                    tau_init=0.01, tau_end=0.001,
                    method='entropic', eta_init=0.5,
                    decay=0.9999, reheat_patience=5000, seed=seed + 1000
                )
            elif r == 1:
                # IPF with different tau
                best_f, best_s = _ipf_optimize(
                    global_best, time.monotonic() + per_refine,
                    tau=0.005, damping=0.2, seed=seed + 1100
                )
            elif r == 2:
                # Fine FISTA with top-k
                best_f, best_s = _fista_optimize(
                    global_best, time.monotonic() + per_refine,
                    tau_init=0.005, tau_end=0.001,
                    method='topk', eta_init=0.3,
                    decay=0.9999, reheat_patience=5000, seed=seed + 1200
                )
            else:
                # IPF with very small tau
                best_f, best_s = _ipf_optimize(
                    global_best, time.monotonic() + per_refine,
                    tau=0.003, damping=0.1, seed=seed + 1300
                )

            if best_s < global_best_score:
                global_best = best_f.copy()
                global_best_score = best_s
                print(f"  Refine {r+1}: NEW BEST: {global_best_score:.6f}")
            else:
                print(f"  Refine {r+1}: score={best_s:.6f} (no improvement)")

    # Phase 3: Diversification - perturb best and re-optimize
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\nPhase 3: Diversification with {remaining:.1f}s remaining")
        n_div = max(2, int(remaining / 80))
        per_div = remaining * 0.8 / n_div

        for d in range(n_div):
            if time.monotonic() >= deadline:
                break
            # Different perturbation scales
            scale = 0.03 + 0.03 * d
            perturbed = _perturb(global_best, rng, scale=scale)
            best_f, best_s = _combined_optimize(
                perturbed, time.monotonic() + per_div, seed=seed + 2000 + d
            )
            if best_s < global_best_score:
                global_best = best_f.copy()
                global_best_score = best_s
                print(f"  Div {d+1}: NEW BEST: {global_best_score:.6f}")
            else:
                print(f"  Div {d+1}: score={best_s:.6f}")

    # Final verification
    result = [float(max(0.0, min(1000.0, x))) for x in global_best]
    final_score = evaluate_sequence(result)
    print(f"\nFINAL SCORE: {final_score:.6f}")
    print(f"Target: 1.5027, Achieved: {final_score:.6f}")

    return result
