"""
Improved LP-based iterative refinement with adaptive step sizing, multi-length search,
and enhanced local search to minimize peak auto-convolution of non-negative sequences.
Key improvements: adaptive alpha via line search, tighter LP bounds, larger sequence lengths,
multi-scale initialization, and coordinate descent polishing.

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_exponential(n, rate=0.5):
    t = np.arange(n, dtype=np.float64)
    return np.exp(-rate * t / n)


def initial_triangle(n):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(t - center) / center)


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_parts = []
    for j in range(n):
        row_parts.append(np.arange(j, j + n, dtype=np.int32))
    row_idx = np.concatenate(row_parts)
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp(a, conv_matrix, M, time_limit=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=conv_matrix, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def find_best_alpha(a, g, base_score, n_alphas=8):
    """Find best blending alpha via line search."""
    alphas = np.linspace(0.05, 1.0, n_alphas)
    best_alpha = 0.0
    best_score = base_score
    for alpha in alphas:
        candidate = (1.0 - alpha) * a + alpha * g
        candidate = normalize_nonneg(candidate)
        s = score_sequence(candidate)
        if s < best_score:
            best_score = s
            best_alpha = alpha
    return best_alpha, best_score


def lp_refine(a, time_limit_s=300.0, aggressive=True, verbose=True):
    """LP-based iterative refinement with adaptive step size."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LP start: score={best_score:.8f}, n={n}")

    conv_matrix = build_conv_matrix(np.ones(n))
    # Actually we need to rebuild with current a each time since the matrix depends on a
    # Wait - the matrix A in the LP is conv_matrix where A[j,i] = a_i for the appropriate lag
    # We need to rebuild when a changes. Let's just rebuild each time for correctness.

    stagnant = 0
    it = 0
    max_iters = 2000
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break

        conv = np.convolve(a, a)
        current_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 10):
                print(f"    LP iter {it:4d}: score={current_score:.8f}")
        else:
            stagnant += 1

        if stagnant > (60 if aggressive else 100):
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.8f}")
            break

        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 10.0:
            break

        # Rebuild conv matrix with current a
        conv_matrix = build_conv_matrix(a)

        timeout = max(2.0, min(30.0, remaining * 0.1))
        g = solve_lp(a, conv_matrix, M, time_limit=timeout)
        if g is None:
            if verbose:
                print(f"    LP solve failed at iter {it}")
            break

        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-8:
            if verbose:
                print(f"    LP no improvement (sum_g={sum_g:.10f}) at iter {it}")
            stagnant += 10
            continue

        g = g / sum_g

        # Adaptive step size via line search
        alpha, new_score = find_best_alpha(a, g, current_score, n_alphas=10)
        if alpha > 0 and new_score < current_score:
            a = (1.0 - alpha) * a + alpha * g
            a = normalize_nonneg(a)
            stagnant = 0
        else:
            # Try smaller steps
            for small_alpha in [0.01, 0.02, 0.03]:
                candidate = (1.0 - small_alpha) * a + small_alpha * g
                candidate = normalize_nonneg(candidate)
                s = score_sequence(candidate)
                if s < current_score:
                    a = candidate
                    stagnant = 0
                    break
            else:
                stagnant += 5

        # Periodically try a more aggressive jump
        if it > 0 and it % 50 == 0 and aggressive:
            # Try the pure LP solution
            g_pure = normalize_nonneg(g)
            s_pure = score_sequence(g_pure)
            if s_pure < best_score:
                a = g_pure.copy()
                best_a = a.copy()
                best_score = s_pure
                stagnant = 0
                if verbose:
                    print(f"    LP jump at iter {it}: score={s_pure:.8f}")

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    LP done: best={best_score:.8f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def coordinate_descent_polish(a, time_limit_s=120.0, verbose=True):

def block_local_search(a, time_limit_s=120.0, rng=None, verbose=True):
    """Block-based local search with smooth perturbations."""
    if rng is None:

    if verbose:
        print(f"    BL search start: score={best_score:.8f}")

    noise_scale = 0.1
    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(rng.integers(n // 300, n // 10)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)

        candidate = a.copy()

        candidate[start:stop] *= perturbation

        candidate = np.maximum(0, candidate)

    if verbose:

    return best_a, best_score


def multi_scale_refine(a_small, target_n, time_limit_s=120.0, verbose=True):
    """Upsample a small optimized sequence to a larger one and refine."""
    from scipy.interpolate import interp1d
    n_small = len(a_small)
    t_small = np.arange(n_small, dtype=np.float64)
    t_large = np.linspace(0, n_small - 1, target_n)
    f = interp1d(t_small, a_small, kind='cubic', fill_value=0, bounds_error=False)
    a_large = f(t_large)
    a_large = np.maximum(0, a_large)
    a_large = normalize_nonneg(a_large)
    if verbose:
        print(f"    Multi-scale: {n_small} -> {target_n}, init score={score_sequence(a_large):.8f}")
    return lp_refine(a_large, time_limit_s=time_limit_s, aggressive=True, verbose=verbose)
