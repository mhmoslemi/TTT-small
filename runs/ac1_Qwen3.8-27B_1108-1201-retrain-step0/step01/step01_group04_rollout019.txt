    # Beta shapes
    for a in [0.2, 0.5, 1, 2, 4]:
        for b in [0.2, 0.5, 1, 2, 4]:
            cands.append(make_beta_shape(n, a, b))
    # Skewed Gaussians
    for mu in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sig in [0.1, 0.15, 0.2, 0.3]:
            for skew in [-2, -1, 0, 1, 2]:
                cands.append(make_skew_shape(n, mu, sig, skew))
    # Multi-peak
    cands.append(make_multi_peak(n, [(0.25, 0.1, 1.0), (0.75, 0.1, 1.0)]))
    cands.append(make_multi_peak(n, [(0.2, 0.08, 1.5), (0.8, 0.08, 1.0)]))
    cands.append(make_multi_peak(n, [(0.15, 0.06, 2.0), (0.5, 0.15, 1.0), (0.85, 0.06, 2.0)]))
    cands.append(make_multi_peak(n, [(0.3, 0.12, 1.0), (0.7, 0.08, 1.5)]))
    # Random smooth
    for _ in range(20):
        f1 = rng.integers(1, 5)
        f2 = rng.integers(1, 10)
        f3 = rng.integers(1, 20)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        p3 = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 1, n)
        p = 1.0 + 0.5 * rng.uniform(0.2, 1.0) * np.sin(2 * np.pi * f1 * x + p1)
        p += 0.3 * rng.uniform(0.2, 1.0) * np.sin(2 * np.pi * f2 * x + p2)
        p += 0.15 * rng.uniform(0.2, 1.0) * np.sin(2 * np.pi * f3 * x + p3)
        cands.append(normalize_nonneg(np.maximum(p, 0.01)))
    return cands

def optimize_local(p0, rng, time_limit, label=""):
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    if n == 0:
        return p, np.inf

    best_p = p.copy()
    best_score = compute_score(best_p)
    current_score = best_score
    prev_p = p.copy()

    end = time.monotonic() + max(0.0, float(time_limit))
    T_frac = 0.3
    iter_no = 0
    since_improve = 0
    stuck_count = 0

    while time.monotonic() < end:
        iter_no += 1
        T_frac = max(0.02, 0.3 * (0.9995 ** iter_no))
        if since_improve > 200:
            T_frac = max(0.01, T_frac * 0.8)

        G, c_cur, cmax, surr = smooth_max_and_grad(p, T_frac)
        if not np.all(np.isfinite(G)):
            G = np.zeros(n)

        gmax = float(np.max(np.abs(G))) if G.size > 0 else 0.0
        p_scale = max(1.0 / n, 0.05 * float(p.max()), 1e-12)

        if gmax > 1e-300:
            base_lr = p_scale / gmax
            lrs = (0.5 * base_lr, 0.15 * base_lr, 0.05 * base_lr,
                   0.01 * base_lr, 0.003 * base_lr, 0.001 * base_lr)
        else:
            lrs = (0.01, 0.003, 0.001)

        accepted = False

        # Gradient step with line search
        for lr in lrs:
            lr = min(max(lr, 1e-10), 5.0)
            v = p - lr * G
            pt = project_simplex(v)
            sc = compute_score(pt)
            # Also compute surrogate for acceptance
            _, _, _, surr_t = smooth_max_and_grad(pt, T_frac)
            if np.isfinite(sc) and np.isfinite(surr_t):
                if surr_t < surr - 1e-10 or sc < best_score - 1e-12:
                    if sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        since_improve = 0
                        stuck_count = 0
                    else:
                        since_improve += 1
                    current_score = sc
                    prev_p = p
                    p = pt
                    accepted = True
                    break

        if not accepted:
            # Peak-pushing: directly target the argmax of convolution
            k_star = int(np.argmax(c_cur))
            # Gradient of c[k_star] w.r.t. p[j] = 2*p[k_star - j] for valid
            peak_grad = np.zeros(n)
            for j in range(n):
                m = k_star - j
                if 0 <= m < n:
                    peak_grad[j] = 2.0 * p[m]
                else:
                    # Also check the "wrap-around" contribution
                    m2 = k_star - j + n
                    if 0 <= m2 < 2 * n - 1:
                        pass  # No extra term for linear conv

            pgmax = float(peak_grad.max()) if n > 0 else 0.0
            if pgmax > 1e-300:
                p_lr = p_scale / pgmax * 0.3
                pt = project_simplex(p - p_lr * peak_grad)
                sc = compute_score(pt)
                if np.isfinite(sc) and sc < current_score:
                    if sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        since_improve = 0
                        stuck_count = 0
                    else:
                        since_improve += 1
                    current_score = sc
                    prev_p = p
                    p = pt
                    accepted = True

        if not accepted:
            # Random perturbations
            sigmas = (0.5 / n, 0.1 / n, 0.02 / n, 0.005 / n)
            for sigma in sigmas:
                pt = project_simplex(p + rng.normal(0.0, sigma, n))
                sc = compute_score(pt)
                if np.isfinite(sc) and sc < current_score:
                    if sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        since_improve = 0
                        stuck_count = 0
                    else:
                        since_improve += 1
                    current_score = sc
                    prev_p = p
                    p = pt
                    accepted = True
                    break

        if not accepted:
            since_improve += 1
            stuck_count += 1
            if stuck_count > 100:
                # Bigger restart
                alpha = 0.3 + 0.5 * rng.random()
                p = project_simplex(alpha * best_p + (1 - alpha) * p +
                                    rng.normal(0, 0.5 / n, n))
                current_score = compute_score(p)
                prev_p = p.copy()
                stuck_count = 0
                T_frac = 0.3
            elif stuck_count > 30:
                p = project_simplex(0.8 * best_p + 0.2 * rng.random(n))
                current_score = compute_score(p)
                prev_p = p.copy()

        if iter_no % 500 == 0:
            print(f"{label} iter={iter_no} best={best_score:.10f} cur={current_score:.10f} T={T_frac:.4f}", flush=True)

        if best_score < 1.5030 and since_improve > 5000:
            break

    best_p = project_simplex(best_p)
    best_score = compute_score(best_p)
    return best_p, best_score

def optimize_with_peak_equalization(p0, rng, time_limit, label=""):
    """Optimization that specifically tries to equalize the top entries of the convolution."""
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    if n == 0:
        return p, np.inf

    best_p = p.copy()
    best_score = compute_score(best_p)
    end = time.monotonic() + max(0.0, float(time_limit))

    T_frac = 0.15
    iter_no = 0
    since_improve = 0

    while time.monotonic() < end:
        iter_no += 1
        T_frac = max(0.01, 0.15 * (0.9998 ** iter_no))

        c = conv_self(p)
        cmax = float(c.max())
        s = float(p.sum())
        current_score = 2.0 * n * cmax / (s * s)

        # Find top K entries
        K = min(5, 2 * n - 1)
        top_indices = np.argsort(c)[::-1][:K]
        top_vals = c[top_indices]

        # Compute weighted gradient focusing on top entries
        weights = np.zeros(2 * n - 1)
        w_max = top_vals[0]
        for i, idx in enumerate(top_indices):
            weights[idx] = np.exp(-0.5 * (top_vals[0] - top_vals[i]) / max(w_max * 0.05, 1e-10))
        weights /= weights.sum()

        # Gradient using these weights
        L2 = _fft_grad_len(n)
        Fw = fft(weights, L2)
        Fpflip = fft(p[::-1], L2)
        h = np.real(ifft(Fw * Fpflip))
        G = 2.0 * h[n - 1:2 * n - 1]
        G = np.maximum(G, 0.0)

        gmax = float(G.max()) if G.size > 0 else 0.0
        if gmax < 1e-300:
            break

        p_scale = max(1.0 / n, 0.03 * float(p.max()), 1e-12)
        base_lr = p_scale / gmax
        lrs = (0.4 * base_lr, 0.1 * base_lr, 0.03 * base_lr, 0.005 * base_lr)

        accepted = False
        for lr in lrs:
            pt = project_simplex(p - lr * G)
            sc = compute_score(pt)
            if np.isfinite(sc) and sc < current_score:
                if sc < best_score:
                    best_score = sc
                    best_p = pt.copy()
                    since_improve = 0
                else:
                    since_improve += 1
                p = pt
                accepted = True
                break

        if not accepted:
            since_improve += 1
            if since_improve > 50:
                p = project_simplex(0.7 * best_p + 0.3 * p + rng.normal(0, 0.1 / n, n))
                since_improve = 0

        if iter_no % 300 == 0:
            print(f"{label} iter={iter_no} best={best_score:.10f}", flush=True)

        if best_score < 1.5030 and since_improve > 3000:
            break

    best_p = project_simplex(best_p)
    best_score = compute_score(best_p)
    return best_p, best_score

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 15.0)

    # Try to use parent as starting point
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and float(parent_arr.sum()) > 0.0:
            parent = parent_arr
    except Exception:
        parent = None

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None or p.size == 0:
            return
        p = normalize_nonneg(p)
        if not np.isfinite(sc):
            sc = compute_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print(f"  NEW BEST: score={sc:.10f} n={p.size} at t={time.monotonic()-start_time:.0f}s", flush=True)

    # Try multiple lengths
    lengths = []
    if parent is not None and parent.size >= 100:
        lengths.append(parent.size)
    lengths.extend([2048, 4096])
    # Remove duplicates, keep unique
    seen = set()
    unique_lengths = []
    for ln in lengths:
        if ln not in seen and 64 <= ln <= 16384:
            seen.add(ln)
            unique_lengths.append(ln)

    print(f"Trying lengths: {unique_lengths}", flush=True)

    # Phase 1: Global shape search
    for n in unique_lengths:
        if time.monotonic() > deadline - 300:
            break
        print(f"\n=== Global search at n={n} ===", flush=True)
        cands = generate_candidates(n, rng)

        # Add parent if appropriate length
        if parent is not None and parent.size == n:
            cands.insert(0, normalize_nonneg(parent))

        # Score all candidates
        scored = []
        for cand in cands:
            sc = compute_score(cand)
            if np.isfinite(sc):
                scored.append((sc, cand))
        scored.sort(key=lambda x: x[0])

        # Record top candidates
        for sc, p in scored[:3]:
            try_record(p, sc)

        print(f"  Top-3 scores at n={n}: {[f'{s:.6f}' for s, _ in scored[:3]]}", flush=True)

        # Phase 2: Local optimization on top candidates
        if time.monotonic() > deadline - 200:
            break

        n_optimizing = min(4, len(scored))
        remaining = deadline - time.monotonic() - 60
        time_per = max(30.0, remaining / n_optimizing)

        for i in range(n_optimizing):
            if time.monotonic() > deadline - 60:
                break
            sc_init, p_init = scored[i]
            label = f"n={n}_init{i}"
            print(f"  Optimizing {label} (init score={sc_init:.6f}, time={time_per:.0f}s)", flush=True)

            limit = min(time_per, deadline - time.monotonic() - 60)
            if limit < 10:
                break

            # Alternate between two optimization strategies
            half = limit / 2.0
            p1, sc1 = optimize_local(p_init, rng, half, label + "A:")
            try_record(p1, sc1)
            p2, sc2 = optimize_with_peak_equalization(p1, rng, half, label + "B:")
            try_record(p2, sc2)

    # Phase 3: Final polish on global best
    if best_p is not None and time.monotonic() < deadline - 30:
        limit = deadline - time.monotonic() - 20
        print(f"\n=== Final polish (best={best_score:.10f}, time={limit:.0f}s) ===", flush=True)
        p_polished, sc_polished = optimize_local(best_p, rng, limit, "FINAL:")
        try_record(p_polished, sc_polished)

    # Phase 4: If we have time, try one more round on the best
    if best_p is not None and time.monotonic() < deadline - 20:
        limit = deadline - time.monotonic() - 15
        if limit > 20:
            p_extra, sc_extra = optimize_with_peak_equalization(best_p, rng, limit, "EXTRA:")
            try_record(p_extra, sc_extra)

    if best_p is None:
        # Fallback: uniform
        n_fallback = 4096
        best_p = np.full(n_fallback, 1.0 / n_fallback)
        best_score = compute_score(best_p)

    # Final validation
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s < 0.01:
        best_p = best_p * (0.02 / max(s, 1e-300))

    print(f"\n=== FINAL: score={compute_score(best_p):.10f} n={best_p.size} ===", flush=True)
    return [float(x) for x in best_p]
