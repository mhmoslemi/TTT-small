```
def random_smooth_init(n, rng, style=None):
   if style is None: style=rng.integers(0,3)
   t=np.linspace(0,1,n)
   if style==0:
      order=min(16, max(4, n//30))
      phi=np.zeros(n)
      for m in range(1,order+1):
         amp=rng.normal(0, 0.8/(m**0.5))
         phase=rng.uniform(0,2*np.pi)
         phi += amp*np.cos(2*np.pi*m*np.arange(n)/n + phase)
      a=np.exp(phi-phi.max())
   elif style==1:
      # boundary power laws with random exponents
      a1=t**rng.uniform(0.3,4.0)
      a2=(1-t)**rng.uniform(0.3,4.0)
      a=a1+a2+rng.uniform(0,0.1,size=n)
   else:
      # sum of gaussian bumps
      a=np.zeros(n); nb=int(rng.integers(2,6))
      i=np.arange(n)
      for _ in range(nb):
         c=float(rng.uniform(0,n)); w=float(rng.uniform(n/20,n/4))
         a += np.exp(-0.5*((i-c)/w)**2)
      a += rng.uniform(0,0.2,size=n)
   a=np.maximum(a,0)
   return normalize_nonneg(a)
```

```python
"""
Heuristic optimizer for nonnegative sequences minimizing
    2*n*max(conv(a,a))/sum(a)^2.

The method combines:
- LP-based Frank-Wolfe-style directions (maximize sum g subject to a*g <= M),
- smooth max-convolution optimization in a softmax/log parameterization,
- variance-flattening pre-optimization,
- multi-length restarts and local multiplicative polishing.

The code starts from height_sequence_1 when available, but also searches
perturbed and longer initializations to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog, Bounds
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft, next_fast_len


GLOBAL_N = 0
GLOBAL_REF = 0
GLOBAL_DEADLINE = 1e300


def normalize_nonneg(a):
    """Return a nonnegative finite sequence normalized to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def conv_square(a):
    """
    Linear convolution a*a via real FFT.
    Input is assumed nonnegative; small negative roundoff is clipped.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    L = next_fast_len(2 * n - 1)
    spec = rfft(a, L)
    c = irfft(spec * spec, L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def score_sequence(a):
    """
    Score compatible with evaluate_sequence, except for final clipping.
    Internal sequences are nonnegative and finite.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = int(a.size)
    if s <= 1e-15 or n == 0:
        return float("inf")
    c = conv_square(a)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * float(np.max(c)) / (s * s))


def adjoint_a(p, a):
    """
    Given p_k weights on convolution coefficients c_k = (a*a)_k,
    return d/d a_i [ sum_k p_k c_k ].

    This equals 2 * sum_k p_k a_{k-i}.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    p = np.asarray(p, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([2.0 * p[0] * a[0]], dtype=np.float64)

    L = next_fast_len(3 * n - 2)
    pr = p[::-1]
    spec_a = rfft(a, L)
    spec_pr = rfft(pr, L)
    x = irfft(spec_a * spec_pr, L)
    return 2.0 * x[n - 1:2 * n - 1][::-1]


def phi_to_a(phi, ref, n):
    """
    Convert relative log-coefficients phi (length n-1) to a probability vector.
    The reference coordinate ref is fixed to log-value 0.
    """
    n = int(n)
    ref = int(ref)
    phi = np.asarray(phi, dtype=np.float64).ravel()
    if ref < 0:
        ref = 0
    if ref >= n:
        ref = n - 1

    theta = np.empty(n, dtype=np.float64)
    if n == 1:
        theta[0] = 0.0
    else:
        if ref == 0:
            theta[0] = 0.0
            theta[1:] = phi
        elif ref == n - 1:
            theta[:-1] = phi
            theta[-1] = 0.0
        else:
            theta[:ref] = phi[:ref]
            theta[ref] = 0.0
            theta[ref + 1:] = phi[ref:]

    m = float(np.max(theta))
    if not np.isfinite(m):
        theta = np.zeros(n, dtype=np.float64)
    else:
        theta = theta - m

    e = np.exp(theta)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return e / s


def phi_from_a(a, ref):
    """
    Inverse of phi_to_a for a reference coordinate ref.
    Returns log(a_i / a_ref) for all i != ref.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    ref = int(ref)
    if ref < 0:
        ref = 0
    if ref >= n:
        ref = n - 1

    floor = 1e-300
    a_safe = np.maximum(np.where(np.isfinite(a), a, 0.0), floor)
    denom = float(a_safe[ref])
    if denom <= 0.0 or not np.isfinite(denom):
        denom = floor

    ratios = a_safe / denom
    phi_all = np.log(np.maximum(ratios, 1e-300))

    if n == 1:
        return np.empty(0, dtype=np.float64)
    if ref == 0:
        return phi_all[1:].copy()
    if ref == n - 1:
        return phi_all[:-1].copy()

    out = np.empty(n - 1, dtype=np.float64)
    out[:ref] = phi_all[:ref]
    out[ref:] = phi_all[ref + 1:]
    return out


def _variance_value(phi):
    """Sum of squared deviations of convolution coefficients from their mean."""
    if time.monotonic() > GLOBAL_DEADLINE:
        raise RuntimeError("time limit exceeded")
    n = GLOBAL_N
    ref = GLOBAL_REF
    a = phi_to_a(phi, ref, n)
    c = conv_square(a)
    avg = 1.0 / float(2 * n - 1)
    d = c - avg
    return float(np.dot(d, d))


def _variance_grad(phi):
    """Gradient of variance objective with respect to relative logs."""
    if time.monotonic() > GLOBAL_DEADLINE:
        raise RuntimeError("time limit exceeded")
    n = GLOBAL_N
    ref = GLOBAL_REF
    a = phi_to_a(phi, ref, n)
    c = conv_square(a)
    avg = 1.0 / float(2 * n - 1)
    d = c - avg

    # Gradient w.r.t. a is 4 * sum_k d_k a_{k-i}; adjoint_a supplies the factor 2,
    # so pass p = 2*d.
    ga = adjoint_a(2.0 * d, a)
    fda = float(np.dot(a, ga))
    gtheta = a * (ga - fda)

    if n == 1:
        return np.empty(0, dtype=np.float64)
    if ref == 0:
        return gtheta[1:].copy()
    if ref == n - 1:
        return gtheta[:-1].copy()

    out = np.empty(n - 1, dtype=np.float64)
    out[:ref] = gtheta[:ref]
    out[ref:] = gtheta[ref + 1:]
    return out


def _lsoft_value(phi, beta):
    """
    Smooth maximum of convolution coefficients:
    (1/beta)*log(sum_k exp(beta c_k)).
    """
    if time.monotonic() > GLOBAL_DEADLINE:
        raise RuntimeError("time limit exceeded")
    n = GLOBAL_N
    ref = GLOBAL_REF
    a = phi_to_a(phi, ref, n)
    c = conv_square(a)
    m = float(np.max(c))
    if beta <= 0.0:
        return m
    xc = beta * (c - m)
    exps = np.exp(np.clip(xc, -745.0, 0.0))
    z = float(np.sum(exps))
    if z <= 0.0:
        return m
    return float(m + np.log(z / float(2 * n - 1)) / beta)


def _lsoft_grad(phi, beta):
    """Gradient of smooth maximum with respect to relative logs."""
    if time.monotonic() > GLOBAL_DEADLINE:
        raise RuntimeError("time limit exceeded")
    n = GLOBAL_N
    ref = GLOBAL_REF
    a = phi_to_a(phi, ref, n)
    c = conv_square(a)
    m = float(np.max(c))
    xc = beta * (c - m)
    exps = np.exp(np.clip(xc, -745.0, 0.0))
    z = float(np.sum(exps))

    if z <= 0.0:
        p = np.full(c.size, 1.0 / float(c.size), dtype=np.float64)
    else:
        p = exps / z

    ga = adjoint_a(p, a)
    fda = float(np.dot(a, ga))
    gtheta = a * (ga - fda)

    if n == 1:
        return np.empty(0, dtype=np.float64)
    if ref == 0:
        return gtheta[1:].copy()
    if ref == n - 1:
        return gtheta[:-1].copy()

    out = np.empty(n - 1, dtype=np.float64)
    out[:ref] = gtheta[:ref]
    out[ref:] = gtheta[ref + 1:]
    return out


def run_var_lbfgs(phi, maxiter):
    """Run L-BFGS-B on the variance-flattening objective."""
    maxiter = int(max(1, maxiter))
    try:
        res = minimize(
            _variance_value,
            phi,
            jac=_variance_grad,
            method="L-BFGS-B",
            options={
                "maxiter": maxiter,
                "maxfun": maxiter * 50,
                "ftol": 1e-12,
                "gtol": 1e-6,
                "maxls": 20,
            },
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.asarray(res.x, dtype=np.float64)
    except Exception as e:
        print("variance LBFGS error:", e)
    return phi


def run_lbfgs(phi, beta, maxiter):
    """Run L-BFGS-B on a softened max-convolution objective."""
    maxiter = int(max(1, maxiter))
    beta = float(beta)
    try:
        res = minimize(
            _lsoft_value,
            phi,
            args=(beta,),
            jac=_lsoft_grad,
            method="L-BFGS-B",
            options={
                "maxiter": maxiter,
                "maxfun": maxiter * 50,
                "ftol": 1e-12,
                "gtol": 1e-7,
                "maxls": 20,
            },
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.asarray(res.x, dtype=np.float64)
    except Exception as e:
        print("softmax LBFGS error:", e)
    return phi


def lp_polish(a0, end_time, rng=None):
    """
    LP-based refinement.

    For current a (sum 1), solve:
        maximize sum g
        subject to (a*g)_k <= M, g >= 0,
    where M is the current peak of a*a, possibly multiplied by a small factor.
    The returned g, renormalized sum 1, is a search direction.
    """
    if rng is None:
        rng = np.random.default_rng(0)

    start = time.monotonic()
    a = normalize_nonng(a0)
    n = int(a.size)
    if n < 2 or end_time - start < 15.0:
        return a, score_sequence(a)

    best_a = a.copy()
    best_score = score_sequence(best_a)

    try:
        col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
        row_idx = col_idx + np.tile(np.arange(n, dtype=np.int32), n)
    except Exception as e:
        print("LP index construction failed:", e)
        return best_a, best_score

    c_obj = -np.ones(n, dtype=np.float64)
    scale = float(np.sqrt(2.0 * n))

    max_iters = 80 if n <= 600 else 45 if n <= 1500 else 18
    stagnant = 0

    for it in range(max_iters):
        rem = end_time - time.monotonic()
        if rem <= 12.0:
            break

        try:
            a_s = a * scale
            c = conv_square(a_s)
            M = float(np.max(c))
            if not np.isfinite(M) or M <= 0.0:
                break

            # Slightly relax constraints after stagnation to allow bigger moves.
            mf = 1.0 + 0.05 * min(stagnant, 4)
            b_ub = np.full(2 * n - 1, M * mf * (1.0 + 1e-12), dtype=np.float64)
            A = csr_matrix(
                (np.tile(a_s, n), (row_idx, col_idx)),
                shape=(2 * n - 1, n),
            )
            timeout = max(2.0, min(18.0, 0.35 * rem))
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=b_ub,
                bounds=Bounds(0.0, np.inf),
                method="highs",
                options={"presolve": True, "time_limit": timeout},
            )
        except Exception as e:
            print("LP call failed at iter", it, ":", e)
            break

        if res is None or res.x is None or not np.all(np.isfinite(res.x)):
            stagnant += 1
            if stagnant > 5:
                break
            continue

        g = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
        sum_g = float(g.sum())
        if sum_g <= scale * (1.0 + 1e-7):
            stagnant += 1
            if stagnant > 5:
                break
            continue

        h = g / sum_g
        sc_current = score_sequence(a)

        best_cand = a.copy()
        best_cand_sc = sc_current

        alphas = np.array([0.05, 0.1, 0.2, 0.35, 0.55, 0.8], dtype=np.float64)
        for alpha in alphas:
            if time.monotonic() > end_time - 5.0:
                break
            cand = (1.0 - float(alpha)) * a + float(alpha) * h
            cand = normalize_nonng(cand)
            sc = score_sequence(cand)
            if sc < best_cand_sc:
                best_cand_sc = sc
                best_cand = cand

        if best_cand_sc < sc_current * (1.0 - 1e-10):
            a = best_cand.copy()
            if best_cand_sc < best_score:
                best_score = best_cand_sc
                best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 5:
                break

    return best_a, best_score


def local_polish(a0, end_time, rng=None):
    """
    Multiplicative local search / annealing in a-space.
    Keeps best sequence, and uses simulated annealing locally when useful.
    """
    if rng is None:
        rng = np.random.default_rng(12345)

    start = time.monotonic()
    a = normalize_nonng(a0)
    n = int(a.size)
    if n < 2:
        return a, score_sequence(a)

    cur = a.copy()
    cur_sc = score_sequence(cur)
    best = cur.copy()
    best_sc = cur_sc
    budget = max(1.0, end_time - start)

    while time.monotonic() < end_time - 3.0:
        rem = max(0.0, end_time - time.monotonic())
        rem_frac = max(0.0, rem / budget)
        sigma = 0.05 + 0.25 * rem_frac

        cand = cur.copy()
        r = float(rng.random())

        if r < 0.45:
            cand *= np.exp(rng.normal(0.0, 0.5 * sigma, size=n))
        elif r < 0.85:
            max_blocks = max(4, min(60, n // 12))
            nblocks = int(rng.integers(3, max(4, max_blocks)))
            for _ in range(nblocks):
                w = int(rng.integers(1, max(2, n // 8)))
                s = int(rng.integers(0, max(1, n - w)))
                if s + w <= n:
                    cand[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
                else:
                    cand *= np.exp(rng.normal(0.0, 0.25 * sigma, size=n))
                    break
        else:
            exponent = float(rng.uniform(0.65, 1.35))
            cand = np.power(np.maximum(cand, 1e-300), exponent)

        if float(rng.random()) < 0.15:
            # Slightly flatten/sharpen all coordinates.
            cand *= np.exp(rng.normal(0.0, 0.05, size=n))

        cand = normalize_nonng(cand)
        sc = score_sequence(cand)

        if sc < best_sc:
            best_sc = sc
            best = cand.copy()
            cur = cand.copy()
            cur_sc = sc
        elif sc < cur_sc:
            cur = cand.copy()
            cur_sc = sc
        else:
            temp = 1e-5 + 0.004 * rem_frac
            if sc - cur_sc < 100.0 * temp and float(rng.random()) < np.exp(-(sc - cur_sc) / temp):
                cur = cand.copy()
                cur_sc = sc

    return best, best_sc


def random_smooth_init(n, rng, style=None):
    """
    Random smooth nonnegative initializations.
    Styles emphasize different asymmetries.
    """
    n = int(n)
    if n <= 1:
        return np.ones(n, dtype=np.float64) / float(max(1, n))
    if rng is None:
        rng = np.random.default_rng(0)
    if style is None:
        style = int(rng.integers(0, 3))
    else:
        style = int(style)

    t = np.linspace(0.0, 1.0, n, dtype=np.float64)

    if style % 3 == 0:
        # Smooth random cosine/log-spectrum sequence.
        order = min(18, max(4, n // 25))
        phi = np.zeros(n, dtype=np.float64)
        i = np.arange(n, dtype=np.float64)
        for m in range(1, order + 1):
            amp = float(rng.normal(0.0, 0.9 / np.sqrt(float(m))))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            phi += amp * np.cos(2.0 * np.pi * float(m) * i / float(n) + phase)
        a = np.exp(phi - float(np.max(phi)))
    elif style % 3 == 1:
        # U-shaped or one-sided boundary emphasis.
        alpha = float(rng.uniform(0.25, 4.5))
        beta = float(rng.uniform(0.25, 4.5))
        a = np.power(t + 1e-12, alpha) + np.power((1.0 - t) + 1e-12, beta)
        a = a + float(rng.uniform(0.0, 0.25)) * np.ones(n, dtype=np.float64)
    else:
        # Sum of Gaussian bumps plus small noise.
        a = np.zeros(n, dtype=np.float64)
        i = np.arange(n, dtype=np.float64)
        nb = int(rng.integers(2, 6))
        for _ in range(nb):
            center = float(rng.uniform(0.0, float(n)))
            width = float(rng.uniform(max(1.0, n / 25.0), max(2.0, n / 3.0)))
            a += np.exp(-0.5 * ((i - center) / width) ** 2)
        a = a + float(rng.uniform(0.0, 0.2)) * np.ones(n, dtype=np.float64)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    if float(a.sum()) <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return a / a.sum()


def mix_sequences(a, b, eps):
    """Mix two sequences, interpolating if lengths differ."""
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    if a.size == 0:
        return normalize_nonng(b)
    if b.size == 0:
        return normalize_nonng(a)
    if a.size != b.size:
        if b.size == 1:
            bb = np.full(a.size, float(b[0]))
        else:
            bb = np.interp(
                np.linspace(0.0, 1.0, a.size, dtype=np.float64),
                np.linspace(0.0, 1.0, b.size, dtype=np.float64),
                b,
            )
        b = bb
    return normalize_nonng((1.0 - float(eps)) * a + float(eps) * b)


def make_long_from_prior(a_prior, n_target, rng=None):
    """
    Create a longer initialization from a shorter known-good sequence by
    interpolation plus small random smooth noise.
    """
    a_prior = normalize_nonng(a_prior)
    n0 = int(a_prior.size)
    n_target = int(n_target)
    if n_target <= n0:
        return normalize_nonng(a_prior[:n_target])

    t_old = np.linspace(0.0, 1.0, n0, dtype=np.float64)
    t_new = np.linspace(0.0, 1.0, n_target, dtype=np.float64)
    interp = np.interp(t_new, t_old, a_prior).astype(np.float64)
    interp *= float(n0) / float(n_target)

    if rng is not None:
        noise = np.exp(rng.normal(0.0, 0.25, size=n_target).astype(np.float64))
        nz = float(noise.sum())
        if nz > 0.0:
            noise = noise / nz
        a = 0.65 * interp + 0.35 * noise
    else:
        a = interp

    floor = 1e-10 / float(n_target)
    a = np.maximum(a, floor)
    return normalize_nonng(a)


def optimize_sequence(a0, end_time, label, rng=None):
    """
    Refine one candidate sequence until end_time is reached.
    """
    if rng is None:
        rng = np.random.default_rng(0)

    start = time.monotonic()
    a = normalize_nonng(a0)
    n = int(a.size)
    if n < 2:
        return a, score_sequence(a)

    global GLOBAL_N, GLOBAL_REF, GLOBAL_DEADLINE

    GLOBAL_N = n
    best_a = a.copy()
    best_score = score_sequence(best_a)
    ref = int(np.argmax(best_a))
    GLOBAL_REF = ref
    phi = phi_from_a(best_a, ref)
    GLOBAL_DEADLINE = end_time

    print(f"[{label}] n={n} start_score={best_score:.6f}")

    # 1) LP polish when moderately sized.
    rem = end_time - time.monotonic()
    if n <= 1200 and rem > 120.0:
        if n <= 600:
            lp_frac = 0.30
            lp_cap = 240.0
        else:
            lp_frac = 0.16
            lp_cap = 120.0
        lp_end = min(end_time, time.monotonic() + min(max(30.0, rem * lp_frac), lp_cap))
        a_lp, sc_lp = lp_polish(best_a, lp_end, rng)
        if sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp.copy()

        ref = int(np.argmax(best_a))
        GLOBAL_REF = ref
        phi = phi_from_a(best_a, ref)

    # 2) Variance flattening pre-stage.
    rem = end_time - time.monotonic()
    if rem > 60.0:
        base_iter = 70 if n <= 800 else 35 if n <= 2000 else 20
        if rem < 120.0:
            base_iter = max(10, base_iter // 2)
        phi_new = run_var_lbfgs(phi, base_iter)
        if np.all(np.isfinite(phi_new)):
            phi = phi_new
            a = phi_to_a(phi, GLOBAL_REF, n)
            sc = score_sequence(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        ref = int(np.argmax(best_a))
        GLOBAL_REF = ref
        phi = phi_from_a(best_a, ref)

    # 3) Progressive smooth max reduction.
    factors = np.array([0.5, 2.0, 10.0, 40.0, 160.0, 400.0], dtype=np.float64)
    cycle = 0

    while time.monotonic() < end_time - 25.0 and cycle <= 6:
        # For later cycles, perturb the starting point slightly.
        state = best_a.copy()
        if cycle > 0:
            state = state * np.exp(rng.normal(0.0, 0.04 + 0.02 * cycle, size=n).astype(np.float64))
            state = np.maximum(state, 1e-12 / float(n))
            state = normalize_nonng(state)

        ref = int(np.argmax(state))
        GLOBAL_REF = ref
        phi = phi_from_a(state, ref)

        for factor in factors:
            rem = end_time - time.monotonic()
            if rem < 18.0:
                break

            a_cur = phi_to_a(phi, GLOBAL_REF, n)
            c = conv_square(a_cur)
            max_c = float(np.max(c))
            if max_c <= 1e-300 or not np.isfinite(max_c):
                break

            beta = float(factor) / max_c
            if beta < 20.0:
                beta = 20.0
            if beta > 2.0e6:
                beta = 2.0e6

            if n <= 700:
                base_iter = 90 if cycle == 0 else 50
            elif n <= 1500:
                base_iter = 45 if cycle == 0 else 25
            else:
                base_iter = 25 if cycle == 0 else 15

            if rem < 70.0:
                base_iter = max(8, int(base_iter * 0.5))

            phi_new = run_lbfgs(phi, beta, base_iter)
            if not np.all(np.isfinite(phi_new)):
                break

            phi = phi_new
            a_new = phi_to_a(phi, GLOBAL_REF, n)
            sc_new = score_sequence(a_new)

            if sc_new < best_score - 1e-13:
                best_score = sc_new
                best_a = a_new.copy()
                ref = int(np.argmax(best_a))
                GLOBAL_REF = ref
                phi = phi_from_a(best_a, ref)
                print(
                    f"[{label}] cycle={cycle} factor={float(factor):.1f} "
                    f"score={best_score:.6f} elapsed={time.monotonic()-start:.1f}s"
                )
            else:
                # When the search is not improving on harder objectives,
                # reheat around the best point before trying the next factor.
                if float(factor) >= 10.0:
                    state = best_a * np.exp(rng.normal(0.0, 0.07, size=n).astype(np.float64))
                    state = np.maximum(state, 1e-12 / float(n))
                    state = normalize_nonng(state)
                    ref = int(np.argmax(state))
                    GLOBAL_REF = ref
                    phi = phi_from_a(state, ref)

        cycle += 1

    # 4) A second short LP pass if there is meaningful time and size permits.
    if n <= 1200:
        rem = end_time - time.monotonic()
        if rem > 90.0:
            lp_cap = 180.0 if n <= 600 else 90.0
            lp_end = min(end_time, time.monotonic() + min(rem * 0.35, lp_cap))
            a_lp, sc_lp = lp_polish(best_a, lp_end, rng)
            if sc_lp < best_score:
                best_score = sc_lp
                best_a = a_lp.copy()

    # 5) Final local multiplicative polish.
    rem = end_time - time.monotonic()
    if rem > 50.0:
        pol_end = min(end_time, time.monotonic() + min(rem * 0.45, 180.0))
        a_pol, sc_pol = local_polish(best_a, pol_end, rng)
        if sc_pol < best_score:
            best_score = sc_pol
            best_a = a_pol.copy()

    print(f"[{label}] final_score={best_score:.6f} elapsed={time.monotonic()-start:.1f}s")
    return best_a, best_score


def propose_candidate():
    """
    Search for a nonnegative height sequence with low evaluate_sequence score.
    Returns a Python list of floats.
    """
    t0 = time.monotonic()
    HORIZON = 2320.0
    hard_deadline = t0 + HORIZON
    rng = np.random.default_rng(20250528)

    best_a = None
    best_score = float("inf")

    # Load previous construction if available.
    prior = None
    try:
        prior = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        prior = None

    if prior is not None:
        prior = np.where(np.isfinite(prior), prior, 0.0)
        prior = np.maximum(prior, 0.0)
        if prior.size < 50 or float(prior.sum()) <= 0.0:
            prior = None

    if prior is not None:
        a_base = normalize_nonng(prior)
        sc_base = score_sequence(a_base)
        if sc_base < best_score:
            best_a = a_base.copy()
            best_score = sc_base

        n0 = int(a_base.size)

        # Primary refinement of the known sequence.
        rem = hard_deadline - time.monotonic()
        budget = min(0.55 * HORIZON, rem - 900.0)
        if budget > 60.0:
            end = time.monotonic() + budget
            a_opt, sc_opt = optimize_sequence(a_base, end, "prior", rng)
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()

        # Perturbed restart around the current best.
        rem = hard_deadline - time.monotonic()
        budget = min(0.12 * HORIZON, rem - 560.0)
        if budget > 45.0:
            a_rand = random_smooth_init(n0, rng, 1)
            base = best_a if best_a is not None else a_base
            a_cand = mix_sequences(base, a_rand, 0.25)
            end = time.monotonic() + budget
            a_opt, sc_opt = optimize_sequence(a_cand, end, "perturb", rng)
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()

        # Longer construction derived from the best known shape.
        n_long1 = max(1000, 2 * n0)
        if 1000 <= n_long1 <= 5000:
            rem = hard_deadline - time.monotonic()
            budget = min(0.14 * HORIZON, rem - 360.0)
            if budget > 60.0:
                base = best_a if best_a is not None else a_base
                a_long = make_long_from_prior(base, n_long1, rng)
                end = time.monotonic() + budget
                a_opt, sc_opt = optimize_sequence(a_long, end, "long1", rng)
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_a = a_opt.copy()

        n_long2 = max(2500, 4 * n0)
        if 2000 < n_long2 <= 5000 and n_long2 > n_long1:
            rem = hard_deadline - time.monotonic()
            budget = min(0.08 * HORIZON, rem - 220.0)
            if budget > 50.0:
                base = best_a if best_a is not None else a_base
                a_long = make_long_from_prior(base, n_long2, rng)
                end = time.monotonic() + budget
                a_opt, sc_opt = optimize_sequence(a_long, end, "long2", rng)
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_a = a_opt.copy()

    else:
        # No prior: random multi-length search.
        configs = [
            (500, 0.35, 0),
            (1000, 0.30, 1),
            (2000, 0.25, 2),
        ]
        for n_init, frac, style in configs:
            rem = hard_deadline - time.monotonic()
            budget = min(float(frac) * HORIZON, rem - 300.0)
            if budget > 60.0:
                a_rand = random_smooth_init(n_init, rng, style)
                end = time.monotonic() + budget
                a_opt, sc_opt = optimize_sequence(a_rand, end, f"rand{n_init}", rng)
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_a = a_opt.copy()

    # If everything failed, fall back safely.
    if best_a is None:
        if prior is not None:
            best_a = normalize_nonng(prior)
        else:
            best_a = random_smooth_init(500, rng, 0)
        best_score = score_sequence(best_a)

    # Final polish on the global best.
    rem = hard_deadline - time.monotonic()
    if rem > 120.0:
        end = time.monotonic() + min(rem - 25.0, 0.20 * HORIZON)
        a_opt, sc_opt = optimize_sequence(best_a, end, "final", rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()

    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.where(np.isfinite(best_a), best_a, 0.0)
    best_a = np.maximum(best_a, 0.0)
    s = float(best_a.sum())
    if s <= 0.01 or best_a.size == 0:
        best_a = np.full(500, 0.002, dtype=np.float64)

    # Optional light scaling: keep values comfortably below the evaluator cap.
    mx = float(np.max(best_a))
    if not np.isfinite(mx) or mx <= 0.0:
        best_a = np.full(500, 0.002, dtype=np.float64)
    else:
        # Do not scale if already tiny; scale only to avoid cap issues.
        if mx > 100.0:
            best_a = best_a * (100.0 / mx)
        best_a = np.minimum(best_a, 1000.0)

    out = [float(x) for x in best_a]
    internal_score = score_sequence(best_a)
    print(f"internal best_score: {internal_score:.6f}, length={len(out)}")
    try:
        verified = evaluate_sequence(out)
        print(f"verified score via evaluate_sequence: {float(verified):.6f}")
    except Exception as e:
        print("verification failed:", e)

    return out
```
