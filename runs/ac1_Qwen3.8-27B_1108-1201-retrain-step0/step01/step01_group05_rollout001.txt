"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Key improvements over previous approach:
1. Log-sum-exp smooth approximation with adaptive temperature (annealing)
   for much smoother gradients than hard top-k
2. Better Adam schedule with warmup and cosine decay
3. Multiple diverse initializations including window functions and parametric shapes
4. Extended polishing phase with very fine steps
5. Try the given length (8192) primarily, with some exploration at other lengths
6. Periodic restarts when progress stalls to escape local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_softmax(a, c, top_k, temperature):
    """Compute smooth subgradient using softmax over top-k lags with temperature."""
    n = len(a)
    if top_k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    
    top_vals = c[top_idx].copy()
    max_val = top_vals.max()
    if max_val <= 1e-15:
        return np.zeros(n)
    
    # Softmax weighting: w[k] = exp((c[k] - max)/T) / sum
    weights = np.exp((top_vals - max_val) / max(temperature, 1e-12))
    total = weights.sum()
    if total <= 1e-15:
        # Fallback: uniform over top indices
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights = weights / total
    
    grad = np.zeros(n, dtype=np.float64)
    for i in range(len(top_idx)):
        k = int(top_idx[i])
        w_val = weights[i]
        if w_val <= 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * w_val * a[k - hi:k - lo + 1][::-1]
    return grad


def run_adam_lsexp(a_init, n, S, max_time, rng, lr_init, top_k_init, top_k_final,
                   T_init, T_min, warmup_frac=0.05, use_restart=True, restart_patience=5000):
    """Adam with log-sum-exp adaptive temperature."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    total_iters = max(1, int(max_time / 0.005))  # estimate ~5ms per iter
    warmup_iters = int(total_iters * warmup_frac)
    
    no_improve_count = 0
    last_best_score = best_score
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Adaptive temperature: start high, decay to low
        progress = min(1.0, float(t) / total_iters)
        # Cosine annealing for temperature
        T_cur = T_min + 0.5 * (T_init - T_min) * (1.0 + np.cos(np.pi * progress))
        
        # Adaptive top_k: start with more lags, narrow down
        top_k_cur = int(top_k_final + (top_k_init - top_k_final) * (1.0 - progress))
        top_k_cur = max(10, top_k_cur)
        
        grad = compute_grad_softmax(a, c, top_k_cur, T_cur)
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-14:
            break
        grad = grad / gnorm
        
        # Adam update with learning rate schedule
        if t < warmup_iters:
            lr = lr_init * (float(t) / max(1, warmup_iters))
        else:
            # Cosine decay after warmup
            post = (t - warmup_iters) / max(1, total_iters - warmup_iters)
            lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * post))
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)
        
        # Periodic restart to escape local minima
        if use_restart and no_improve_count > restart_patience and rng.random() < 0.01:
            noise_scale = 0.02 * S / n
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            no_improve_count = 0
        
        # Early exit if converged
        if t > 1000 and (last_best_score - best_score) < 1e-8:
            break
        last_best_score = best_score
    
    return best_local, best_score


def run_mirror_lsexp(a_init, n, S, max_time, rng, eta_init, top_k_init, top_k_final,
                     T_init, T_min):
    """Mirror descent (multiplicative weights) with log-sum-exp gradient."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    total_iters = max(1, int(max_time / 0.005))
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()
        
        progress = min(1.0, float(t) / total_iters)
        T_cur = T_min + 0.5 * (T_init - T_min) * (1.0 + np.cos(np.pi * progress))
        top_k_cur = max(10, int(top_k_final + (top_k_init - top_k_final) * (1.0 - progress)))
        
        grad = compute_grad_softmax(a, c, top_k_cur, T_cur)
        gmax = np.abs(grad).max()
        if gmax <= 1e-12:
            break
        
        # Mirror descent update: a_new = a * exp(-eta * grad), then renormalize
        eta = eta_init * max(0.01, 1.0 / (1.0 + 0.002 * t))
        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new <= 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / s_new)
    
    return best_local, best_score

        per_init_time = min(30.0, remaining / (len(other_inits) + 1))
        
        for name, arr in other_inits:
            if time.monotonic() > explore_deadline - 5.0:
                break
            seg = max(5.0, per_init_time)
            
            a, s = run_adam_lsexp(arr, n, S, seg, rng,
                                   lr_init=0.3 * S / n,
                                   top_k_init=80, top_k_final=15,
                                   T_init=0.1 * S * S / n, T_min=0.001 * S * S / n,
                                   use_restart=True, restart_patience=2000)
            update_global(a, s)
            
            # Quick mirror descent follow-up
            if time.monotonic() < explore_deadline - 5.0:
                seg2 = max(3.0, per_init_time * 0.5)
                a2, s2 = run_mirror_lsexp(a, n, S, seg2, rng,
                                            eta_init=0.2,
                                            top_k_init=40, top_k_final=10,
                                            T_init=0.05 * S * S / n, T_min=0.001 * S * S / n)
                update_global(a2, s2)
    
    # Try alternative length
    if time.monotonic() < explore_deadline - 10.0 and n_alt != n_primary:
        n = n_alt
        S = np.sqrt(2.0 * float(n))
        inits_alt = generate_initializations(n, rng)
        per_init = max(5.0, (explore_deadline - time.monotonic()) / (len(inits_alt) + 1))
        
        for name, arr in inits_alt[:5]:  # Limit to 5 inits for alt length
            if time.monotonic() > explore_deadline - 5.0:
                break
            a, s = run_adam_lsexp(arr, n, S, min(per_init, 30.0), rng,
                                   lr_init=0.3 * S / n,
                                   top_k_init=80, top_k_final=15,
                                   T_init=0.1 * S * S / n, T_min=0.001 * S * S / n,
                                   use_restart=True, restart_patience=2000)
            update_global(a, s)
    
    print("After exploration: best=%.8f" % global_best)
    
    # ============ PHASE 2: POLISH (remaining time) ============
    left = deadline - time.monotonic()
    if left > 5.0 and global_a is not None:
        print("Polishing for %.1f seconds..." % left)
        global_a = polish_sequence(global_a, left, rng)
    
    # Final verification
    if global_a is not None:
        global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n_primary, dtype=np.float64)
    
    # Verify with authoritative function
    if global_a is not None and global_a.sum() > 0.01:
        ver = evaluate_sequence([float(x) for x in global_a])
        print("Final verification: %.8f" % ver if np.isfinite(ver) else "Final verification: inf")
        if np.isfinite(ver):
            global_best = ver
    
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)
    
    return [float(x) for x in global_a]
