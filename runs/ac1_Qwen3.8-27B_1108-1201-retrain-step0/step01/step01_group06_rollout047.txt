def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_obj(a):
    """Compute 2*n*max(conv)/(sum)^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def conv_with_change(conv, a, j, x, v):
