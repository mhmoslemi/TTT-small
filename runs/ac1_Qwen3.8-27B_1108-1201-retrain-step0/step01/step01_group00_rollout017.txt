def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def local_score(a):
    """Score assuming sum(a) = 1 (normalized)."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

def pnorm_obj_grad(theta, p):
    """Objective and gradient for p-norm of convolution."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    # p-norm
    cp = c ** (p - 1)  # for gradient
    sum_cp = np.sum(cp * c)  # sum of c^p
    if sum_cp < 1e-30:
        pnorm = 0.0
        score = 0.0
        return score, np.zeros(n)
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    
    # Gradient
    # d(score)/d(a[i]) = 2*n * (sum c^p)^(1/p - 1) * sum_k c[k]^(p-1) * dc[k]/da[i]
    # dc[k]/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    # So d(score)/d(a[i]) = 4*n * (sum c^p)^(1/p - 1) * sum_{m=0}^{n-1} a[m] * c[i+m]^(p-1)
    
    factor = 4.0 * n * (sum_cp ** (1.0/p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)  # length n + (2n-1) - 1 = 3n-2
    # cross[n-1+i] = sum_m a[m] * cp[i+m] = sum_m a[m] * c[i+m]^(p-1)
    grad_a = factor * cross[n-1:2*n-1]
    
    # Chain rule through softmax
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta

def soft_max_obj_grad(theta, tau):
    """Objective and gradient using soft-max of convolution."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    
    score = 2.0 * n * soft_max_val
    
    # d(soft_max)/d(a[i]) = sum_k w[k] * dc[k]/da[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * cross_corr(a, w)[i]
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n-1:2*n-1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta

def polish_peak_reduction(a, time_limit, rng):
    """Targeted peak reduction polish."""
    current = a.copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        peak_val = np.max(c)
        
        # Find top contributing pairs for the peak
        peak_idx = int(np.argmax(c))
        contribs = np.zeros(n)
        for i in range(min(n, peak_idx + 1)):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = current[i] * current[j]
        
        top_k = max(3, n // 8)
        top_idx = np.argsort(contribs)[-top_k:]
        
        candidate = current.copy()
        # Reduce top contributors
        candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.01, size=len(top_idx)))
        # Increase others slightly
        other = np.setdiff1d(np.arange(n), top_idx)
        n_inc = min(10, len(other))
        if n_inc > 0:
            inc_idx = rng.choice(other, size=n_inc, replace=False)
            candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.01, size=n_inc))
        
        candidate = np.maximum(candidate, 1e-10)
        candidate = candidate / np.sum(candidate)

            current = current / np.sum(current)

    return best, best_score

"""
Multi-phase optimization for minimum peak autocorrelation non-negative sequences.
Uses L-BFGS with p-norm objective (annealed), soft-max gradient refinement,
and targeted peak-reduction local search. Explores multiple sequence lengths
starting from provided sequence and diverse random initializations.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def pnorm_obj_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p.
    Returns (score, grad_theta).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def soft_max_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Bell-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def polish_local_search(a_init, time_limit, rng):
    """
    Simulated annealing with targeted peak reduction.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998

    iters_per_sec = 0
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.30:
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.012, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(12, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.012, size=n_inc))
        elif r < 0.50:
            # Smoothed peak reduction: reduce based on full convolution profile
            c = fft_convolve(current, current)
            c_avg = np.mean(c)
            # Where c is high, pull down the contributing a values
            pull = np.zeros(n)
            for k in range(len(c)):
                weight = max(0, c[k] - c_avg)
                for i in range(min(n, k + 1)):
                    j = k - i
                    if 0 <= j < n:
                        pull[i] += weight * current[j]
            if np.max(pull) > 1e-10:
                pull = pull / np.max(pull)
                candidate *= np.exp(-0.02 * pull)
        elif r < 0.65:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        elif r < 0.80:
            # Symmetrize partially
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.015, size=n))
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.015, size=n))

        candidate = np.maximum(candidate, 1e-10)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 1e-8)

        if time.monotonic() - last_print > 5.0:
            last_print = time.monotonic()
            print(f"    polish: score={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")

    return best, best_score


def run_lbgfs_pnorm(theta, n, p_schedule, maxiter_per_p, deadline):
    """Run L-BFGS with annealed p-norm. Returns best theta and score."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))

    for p in p_schedule:
        if time.monotonic() > deadline - 15:
            break

        result = minimize(
            pnorm_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(p,),
            options={'maxiter': maxiter_per_p, 'ftol': 1e-14, 'gtol': 1e-12, 'maxfun': maxiter_per_p * 5}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)

        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        else:
            # Keep previous theta if this p didn't help
            pass
        theta = best_theta

    return best_theta, best_score


def run_lbgfs_softmax(theta, n, tau_schedule, maxiter_per_tau, deadline):
    """Run L-BFGS with annealed soft-max temperature."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))

    for tau in tau_schedule:
        if time.monotonic() > deadline - 15:
            break

        result = minimize(
            soft_max_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(tau,),
            options={'maxiter': maxiter_per_tau, 'ftol': 1e-14, 'gtol': 1e-12, 'maxfun': maxiter_per_tau * 5}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)

        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        theta = best_theta

    return best_theta, best_score
