import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def soft_max_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    
    score = 2.0 * n * soft_max_val
    
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def soft_max_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).
    """

    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)

    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

            theta = theta_new

    # Soft-max refinement with adaptive tau
    tau = min(max_tau, best_score / (2.0 * n) * 0.1)
    improvements_in_row = 0

    while time.monotonic() < deadline - 15:
        result = minimize(
            soft_max_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(tau,),
            options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-13, 'maxfun': 8000}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)

        if s < best_score - 1e-10:
            best_score = s
            best_theta = theta_new.copy()
            theta = theta_new
            improvements_in_row += 1
            tau *= 0.6  # Cool down on improvement
        else:
            improvements_in_row = 0
            if improvements_in_row == 0 and tau <= min_tau * 1.05:
                break
            tau = max(tau * 0.85, min_tau)

        tau = max(tau, min_tau)

    return best_theta, best_score


def make_structured_init(n, rng, base_seq=None):
    """
    Create structured initialization based on the structure of a known good sequence.
    Uses window * oscillation model.
    """
    # Window shape
    t = np.arange(n) / max(n - 1, 1)
    # Try different window shapes
    w_type = rng.integers(0, 4)
    if w_type == 0:
        w = np.sin(np.pi * t) ** 2
    elif w_type == 1:
        w = 0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.5))
    elif w_type == 2:
        w = np.exp(-2 * (t - 0.4) ** 2)  # Slightly off-center
    else:
        w = np.ones(n)

    # Oscillation
    num_freqs = rng.integers(1, 4)
    osc = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        osc += amp * np.cos(2 * np.pi * freq * t + phase)

    a = w * osc
    a = np.clip(a, 1e-4, None)

    # Add noise
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 1e-6, None)

    if base_seq is not None and rng.random() < 0.4:
        # Interpolate with base
        x_old = np.linspace(0, 1, len(base_seq))
        x_new = np.linspace(0, 1, n)
        a_base = np.interp(x_new, x_old, base_seq)
        alpha = rng.uniform(0.3, 0.7)
        a = alpha * a + (1 - alpha) * a_base
        a = np.clip(a, 1e-6, None)

    a /= np.sum(a)
    return a


def propose_candidate(seed=42, **kwargs):
    """
    Multi-phase optimization for minimum peak autocorrelation.
    Phases:
    1. Adaptive L-BFGS refinement of starting sequence
    2. Multi-length exploration with structured initialization
    3. Peak-directed local search
    4. Final L-BFGS polish
    5. Mirror descent refinement
    """
    t_global = time.monotonic()
    deadline = t_global + 2350.0  # 50s safety margin
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-8, 1000.0)
        s_sum = np.sum(start_seq)
        if s_sum > 0.01:
            start_seq = start_seq / s_sum
            s = local_score(start_seq)
            print(f"Starting sequence: n={len(start_seq)}, score={s:.6f}")
            best_score = s
            best_seq = start_seq.copy()

    # Phase 1: Adaptive L-BFGS on starting sequence
    print(f"\n=== Phase 1: Adaptive L-BFGS (starting score={best_score:.6f}) ===")
    if best_seq is not None:
        n = len(best_seq)
        theta = np.log(best_seq + 1e-30)
        phase1_deadline = min(t_global + 500, deadline - 800)
        theta_best, s = adaptive_lbgfs(theta, n, phase1_deadline, max_tau=0.003, min_tau=0.00005,
                                       p_warmup=[4, 8, 16, 32])
        if s < best_score:
            best_score = s
            best_seq = softmax(theta_best).copy()
            print(f"Phase 1 result: {best_score:.8f} *** IMPROVED ***")
        else:
            print(f"Phase 1 result: {s:.8f} (no improvement)")
        print(f"  Time used: {time.monotonic() - t_global:.0f}s")

    # Phase 2: Multi-length exploration
    print(f"\n=== Phase 2: Multi-length exploration ===")
    # Focus on lengths near the current best, plus some diverse ones
    if best_seq is not None:
        n_best = len(best_seq)
        lengths_to_try = sorted(set([
            n_best,  # try again with different init
            max(64, n_best - 32),
            n_best + 32,
            n_best + 64,
            max(64, n_best - 64),
            n_best + 96,
            n_best + 128,
        ]))
    else:
        lengths_to_try = [128, 192, 256, 320, 384]

    # Remove duplicates and sort by proximity to best length
    if best_seq is not None:
        lengths_to_try = [l for l in lengths_to_try if l >= 64]
        lengths_to_try = sorted(set(lengths_to_try), key=lambda l: abs(l - len(best_seq)))
    else:
        lengths_to_try = sorted(set(lengths_to_try))

    for n in lengths_to_try:
        if time.monotonic() > deadline - 400:
            break

        # Time allocation: more time for lengths close to best
        dist = abs(n - (len(best_seq) if best_seq is not None else 256))
        time_alloc = max(40.0, min(180.0, 200.0 * (1.0 - dist / 300.0)))
        phase2_deadline = min(time.monotonic() + time_alloc, deadline - 400)

        # Generate diverse initializations
        num_inits = 3 if time_alloc > 100 else 2

        for init_i in range(num_inits):
            if time.monotonic() > phase2_deadline:
                break

            if init_i == 0 and best_seq is not None:
                # Interpolate from best
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, best_seq)
                a0 = np.maximum(a0, 1e-8)
                a0 /= np.sum(a0)
                init_name = "interp"
            else:
                a0 = make_structured_init(n, rng, base_seq=best_seq)
                init_name = f"struct_{init_i}"

            theta = np.log(a0 + 1e-30)

            # Quick p-norm warmup
            for p in [8, 16]:
                if time.monotonic() > phase2_deadline:
                    break
                result = minimize(
                    pnorm_obj_grad, theta.copy(),
                    method='L-BFGS-B',
                    jac=True, args=(p,),
                    options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-10, 'maxfun': 1500}
                )
                theta = result.x

            # Soft-max refinement
            theta_best, s = adaptive_lbgfs(theta, n, phase2_deadline,
                                           max_tau=0.002, min_tau=0.0001,
                                           p_warmup=[32])

            if s < best_score:
                best_score = s
                best_seq = softmax(theta_best).copy()
                print(f"  n={n}, '{init_name}': score={s:.8f} *** NEW BEST ***")
            else:
                print(f"  n={n}, '{init_name}': score={s:.8f}")

    # Phase 3: Peak-directed local search
    print(f"\n=== Phase 3: Peak-directed search (score={best_score:.8f}) ===")
    if best_seq is not None:
        search_time = min(500.0, max(100.0, deadline - time.monotonic() - 200))
        print(f"  Search time: {search_time:.0f}s")
        polished, p_score = peak_directed_search(best_seq, search_time, rng)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"Phase 3 result: {best_score:.8f} *** NEW BEST ***")
        else:
            print(f"Phase 3 result: {p_score:.8f} (no improvement)")

    # Phase 4: Mirror descent refinement
    print(f"\n=== Phase 4: Mirror descent refinement (score={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_mirror = best_seq.copy()
        a_mirror /= np.sum(a_mirror)
        md_deadline = min(time.monotonic() + 200, deadline - 80)
        tau_md = 0.001
        lr = 0.02
        it = 0

        while time.monotonic() < md_deadline:
            candidate = mirror_descent_step(a_mirror, tau_md, lr)
            s, _ = local_score_and_conv(candidate)

            if s < best_score:
                best_score = s
                best_seq = candidate.copy()
                a_mirror = candidate.copy()
                lr *= 0.95
                if it % 50 == 0:
                    print(f"    MD iter {it}: score={s:.8f}, tau={tau_md:.6f}, lr={lr:.5f}")
            else:
                if s < local_score(a_mirror):
                    a_mirror = candidate.copy()
                lr *= 0.9
                tau_md *= 0.95

            lr = max(lr, 0.001)
            tau_md = max(tau_md, 0.0001)
            it += 1

    # Phase 5: Final L-BFGS polish with very small tau
    print(f"\n=== Phase 5: Final L-BFGS polish (score={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n = len(best_seq)
        theta = np.log(best_seq + 1e-30)

        for tau in [0.0003, 0.00015, 0.00008, 0.00004]:
            if time.monotonic() > deadline - 15:
                break
            result = minimize(
                soft_max_obj_grad, theta.copy(),
                method='L-BFGS-B',
                jac=True, args=(tau,),
                options={'maxiter': 600, 'ftol': 1e-16, 'gtol': 1e-14, 'maxfun': 12000}
            )
            a_new = softmax(result.x)
            s = local_score(a_new)
            if s < best_score:
                best_score = s
                best_seq = a_new.copy()
                theta = result.x
                print(f"  tau={tau:.5f}: score={s:.8f} *** NEW BEST ***")
            else:
                print(f"  tau={tau:.5f}: score={s:.8f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.8f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
