import time
from scipy.optimize import linprog, minimize

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (np.sum(a) ** 2)

    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def pnorm_obj_grad(theta, p):
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))

    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta

def solve_lp(f, M):
    """
    Solve: maximize sum(b_j) subject to (f*b)(k) <= M for all k, b_j >= 0.
    """
    n = len(f)
    # Build constraint matrix A such that A @ b <= M * ones(2n-1)
    # (f*b)(k) = sum_i f(i) * b(k-i) for valid i, k-i
    # For each k, the constraint involves b(j) where j = k-i, i.e., b(k-i)
    # So (f*b)(k) = sum_{j: 0<=j<n, 0<=k-j<n} f(k-j) * b(j)
    
    # A is (2n-1) x n matrix
    # A[k, j] = f(k-j) if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    
    # linprog minimizes, so we negate the objective
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return None

from scipy.optimize import linprog, minimize

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Score = 2n * max(conv(a,a)) / sum(a)^2"""
    n = len(a)
    c = fft_convolve(a, a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    return 2.0 * n * np.max(c) / (s * s)

def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def pnorm_obj_grad(theta, p):
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    """Build the (2n-1) x n matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # A[k, j] = f[k-j] for k = j, j+1, ..., j+n-1
        rows = np.arange(j, j+n)
        cols = np.zeros(n, dtype=int) + j
        A[rows, j] = f
    return A

def solve_lp_for_b(f, M, n):
    """Solve max sum(b) s.t. A@b <= M*ones, b >= 0."""
    A = build_toeplitz_A(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')

    return None

    LP-based iterative optimization.

            break

        M = np.max(c)

    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score = 2.0 * n * soft_max_val
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def build_conv_matrix(f, n):
    """
    Build (2n-1) x n matrix A where A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    Used for the LP constraint (f*b)(k) <= M.
    """
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A[rows, j] = f
    return A


def solve_lp_b(f, M, n):
    """
    Solve: maximize sum(b_j) subject to (f*b)(k) <= M for all k, b >= 0.
    Returns optimal b or None.
    """
    A = build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_iteration(a_init, deadline, verbose=True):
    """
    LP-based iterative optimization.
    Repeatedly: normalize, compute M, solve LP, rescale, blend.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_a = a.copy()
    best_score = local_score(a)
    t = 0.3
    t_min = 0.05
    it = 0

    while time.monotonic() < deadline:
        s = np.sum(a)
        if s < 1e-10:
            break
        a_norm = a / s * np.sqrt(2.0 * n)

        c = fft_convolve(a_norm, a_norm)
        M = np.max(c)
        if M < 1e-15:
            break

        b = solve_lp_b(a_norm, M, n)
        if b is None:
            break

        b_sum = np.sum(b)
        if b_sum < 1e-10:
            break
        b = b / b_sum * np.sqrt(2.0 * n)

        # Line search on blend parameter
        best_t = t
        best_blend = None
        for trial_t in [t * 0.5, t, t * 1.5, 0.1, 0.5]:
            blend = (1.0 - trial_t) * a_norm + trial_t * b
            blend = np.maximum(blend, 0.0)
            bs = np.sum(blend)
            if bs < 1e-10:
                continue
            sc = local_score(blend)
            if best_blend is None or sc < local_score(best_blend):
                best_blend = blend
                best_t = trial_t

        if best_blend is None:
            # Reduce t and retry
            t = max(t * 0.5, t_min)
            continue

        it += 1
        new_score = local_score(best_blend)

        if new_score < best_score:
            best_score = new_score
            best_a = best_blend.copy()
            # Keep or slightly increase t
            if verbose and it % 50 == 0:
                print(f"    LP iter {it}: score={best_score:.6f}, t={best_t:.3f}")
        else:
            # Score didn't improve, reduce t
            t = max(t * 0.7, t_min)
            if t == t_min and it > 200:
                if verbose:
                    print(f"    LP converged at iter {it}: score={best_score:.6f}")
                break

        a = best_blend

    return best_a, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Bell-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def make_window_init(n, rng, window_type='hann'):
    """Window function initialization."""
    t = np.arange(n, dtype=np.float64)
    if window_type == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
    elif window_type == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t / (n - 1)) + 0.08 * np.cos(4 * np.pi * t / (n - 1))
    elif window_type == 'raised_cosine':
        a = np.cos(np.pi * t / (n - 1)) ** 2
    else:
        a = np.ones(n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p, deadline):
    """Run L-BFGS with annealed p-norm."""
    best_theta = theta.copy()
    best_score = local_score_norm(softmax(theta))

    for p in p_schedule:
        if time.monotonic() > deadline - 15:
            break
        try:
            result = minimize(
                pnorm_obj_grad, theta.copy(),
                method='L-BFGS-B', jac=True, args=(p,),
                options={'maxiter': maxiter_per_p, 'ftol': 1e-14,
                         'gtol': 1e-12, 'maxfun': maxiter_per_p * 5}
            )
            theta_new = result.x
            s = local_score_norm(softmax(theta_new))
            if s < best_score:
                best_score = s
                best_theta = theta_new.copy()
        except Exception:
            pass
        theta = best_theta

    return best_theta, best_score


def run_lbgfs_softmax(theta, tau_schedule, maxiter_per_tau, deadline):
    """Run L-BFGS with annealed soft-max temperature."""
    best_theta = theta.copy()
    best_score = local_score_norm(softmax(theta))

    for tau in tau_schedule:
        if time.monotonic() > deadline - 15:
            break
        try:
            result = minimize(
                soft_max_obj_grad, theta.copy(),
                method='L-BFGS-B', jac=True, args=(tau,),
                options={'maxiter': maxiter_per_tau, 'ftol': 1e-14,
                         'gtol': 1e-12, 'maxfun': maxiter_per_tau * 5}
            )
            theta_new = result.x
            s = local_score_norm(softmax(theta_new))
            if s < best_score:
                best_score = s
                best_theta = theta_new.copy()
        except Exception:
            pass
        theta = best_theta

    return best_theta, best_score


def polish_local_search(a_init, time_limit, rng):
    """
    Simulated annealing with targeted peak reduction.
    """
    current = a_init.astype(np.float64).copy()
    current = np.maximum(current, 1e-10)
    current = current / np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.003
    cooling = 0.9999

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.25:
            # Targeted peak reduction
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(15, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.40:
            # Multi-peak reduction
            c = fft_convolve(current, current)
            c_thresh = 0.9 * np.max(c)
            peak_lags = np.where(c > c_thresh)[0]
            pull = np.zeros(n)
            for k in peak_lags:
                for i in range(min(n, int(k) + 1)):
                    j = int(k) - i
                    if 0 <= j < n:
                        pull[i] += current[j]
            if np.max(pull) > 1e-10:
                pull = pull / np.max(pull)
                candidate *= np.exp(-0.025 * pull)
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))
        elif r < 0.70:
            # Partial symmetrization
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.02, size=n))
        elif r < 0.85:
            # Smooth global
            candidate = current * np.exp(rng.normal(0.0, 0.02, size=n))
        else:
            # Edge reduction (edges contribute to conv[0] and conv[2n-2])
            edge_width = max(2, n // 16)
            candidate[:edge_width] *= np.exp(rng.normal(-0.02, 0.01, size=edge_width))
            candidate[-edge_width:] *= np.exp(rng.normal(-0.02, 0.01, size=edge_width))
            mid_start = (n - edge_width * 2) // 2
            mid_len = edge_width * 2
            if mid_len > 0:
                candidate[mid_start:mid_start + mid_len] *= np.exp(
                    rng.normal(0.02, 0.01, size=mid_len))

        candidate = np.maximum(candidate, 1e-10)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 1e-9)

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: LP iteration, L-BFGS p-norm/softmax, local search.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 40.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        best_score = s
        best_seq = start_seq.copy()

    # Phase 1: LP iteration on provided sequence
    print("\n=== Phase 1: LP Iteration ===")
    if best_seq is not None:
        lp_deadline = min(t_global + 600, deadline - 600)
        print(f"  LP deadline: {lp_deadline - t_global:.0f}s from start")
        a_lp, s_lp = lp_iteration(best_seq, lp_deadline, verbose=True)
        if s_lp < best_score:
            best_score = s_lp
            best_seq = a_lp.copy()
            print(f"  LP improved: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  LP best: {s_lp:.6f} (no improvement from {best_score:.6f})")

    # Phase 2: L-BFGS p-norm on multiple lengths and inits
    print(f"\n=== Phase 2: L-BFGS p-norm optimization ===")
    lengths = [256, 320, 384, 512]
    if best_seq is not None:
        n_best = len(best_seq)
        if n_best not in lengths:
            lengths.insert(0, n_best)

    for n in lengths:
        if time.monotonic() > deadline - 400:
            break

        time_remaining = deadline - time.monotonic()
        time_alloc = min(200.0, max(60.0, time_remaining * 0.25))
        n_deadline = time.monotonic() + time_alloc

        inits = []
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.maximum(a_interp, 1e-6)
            a_interp = a_interp / np.sum(a_interp)
            inits.append(('best_interp', a_interp))

        inits.append(('chirp', make_chirp_init(n, 0.3 + 0.3 * rng.random(), rng)))
        inits.append(('random', make_random_init(n, rng, spread=0.2 + 0.3 * rng.random())))
        inits.append(('hann', make_window_init(n, rng, 'hann')))
        inits.append(('blackman', make_window_init(n, rng, 'blackman')))

        p_schedule = [4, 8, 16, 32, 64]

        for init_name, a0 in inits:
            if time.monotonic() > n_deadline:
                break
            theta = np.log(a0 + 1e-30)
            theta_best, s = run_lbgfs_pnorm(theta, p_schedule, 150, n_deadline)
            if s < best_score:
                a_best = softmax(theta_best)
                best_score = s
                best_seq = a_best.copy()
                print(f"  n={n}, init='{init_name}': score={s:.6f} *** NEW BEST ***")
            else:
                print(f"  n={n}, init='{init_name}': score={s:.6f}")

    # Phase 3: LP iteration on best found so far
    print(f"\n=== Phase 3: LP refinement on best (score={best_score:.6f}) ===")
    if best_seq is not None:
        lp2_deadline = min(time.monotonic() + 400, deadline - 200)
        a_lp2, s_lp2 = lp_iteration(best_seq, lp2_deadline, verbose=True)
        if s_lp2 < best_score:
            best_score = s_lp2
            best_seq = a_lp2.copy()
            print(f"  LP2 improved: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  LP2: {s_lp2:.6f} (no improvement)")

    # Phase 4: Soft-max L-BFGS refinement
    print(f"\n=== Phase 4: Soft-max refinement ===")
    if best_seq is not None:
        n = len(best_seq)
        a_norm = best_seq / np.sum(best_seq)
        theta = np.log(a_norm + 1e-30)
        tau_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005]
        sm_deadline = min(time.monotonic() + 200, deadline - 150)
        theta_best, s = run_lbgfs_softmax(theta, tau_schedule, 200, sm_deadline)
        if s < best_score:
            best_score = s
            best_seq = softmax(theta_best).copy()
            print(f"  Soft-max improved: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Soft-max: {s:.6f} (no improvement)")

    # Phase 5: Final LP polish
    print(f"\n=== Phase 5: Final LP polish ===")
    if best_seq is not None:
        lp3_deadline = min(time.monotonic() + 200, deadline - 100)
        a_lp3, s_lp3 = lp_iteration(best_seq, lp3_deadline, verbose=False)
        if s_lp3 < best_score:
            best_score = s_lp3
            best_seq = a_lp3.copy()
            print(f"  Final LP: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Final LP: {s_lp3:.6f} (no improvement)")

    # Phase 6: Local search polish
    print(f"\n=== Phase 6: Local search (score={best_score:.6f}) ===")
    if best_seq is not None:
        polish_time = max(60.0, deadline - time.monotonic() - 20)
        print(f"  Polish time: {polish_time:.0f}s")
        polished, p_score = polish_local_search(best_seq, polish_time, rng)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  After polish: {p_score:.6f} (no improvement)")

    # Final: one more LP pass if time
    if time.monotonic() < deadline - 30 and best_seq is not None:
        a_final, s_final = lp_iteration(best_seq, deadline - 5, verbose=False)
        if s_final < best_score:
            best_score = s_final
            best_seq = a_final.copy()
            print(f"\n  Final LP pass: {best_score:.6f} *** NEW BEST ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
