"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a multi-phase strategy:
1. Exploration: Try multiple sequence lengths with fast Nesterov+soft-max optimization.
2. Exploitation: Alternating gradient descent with LP-based directional kicks.
3. Polish: Coordinate descent with local search.
Key innovations: adaptive temperature schedule, LP direction filtering, 
parallel coordinate descent, and multi-scale initializations.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def compute_conv_grad(c, a, top_k=16):
    """Compute weighted gradient of max(conv) using top-k lags.
    Returns gradient w.r.t. a (length n)."""
    n = len(a)
    c_max = c.max()
    # Find top-k lags
    if top_k >= len(c):
        active = np.arange(len(c))
    else:
        active = np.argsort(c)[::-1][:top_k]
    # Weight: normalize active values
    weights = c[active]
    w_sum = weights.sum()
    if w_sum < 1e-30:
        w = np.ones(len(c), dtype=np.float64) / len(c)
    else:
        w = np.zeros(len(c), dtype=np.float64)
        w[active] = weights / w_sum

    # grad[i] = 2 * sum_k w[k] * a[k-i] * 1_{0<=k-i<n}
    # = 2 * conv(w, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    full = fft_convolve(w, a_rev)
    end_idx = min(2 * n - 1, len(full))
    grad = np.zeros(n, dtype=np.float64)
    grad[:] = 2.0 * full[n - 1: end_idx][:n]
    return grad


def compute_soft_grad(c, a, T):
    """Compute gradient using soft-max weights."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-15:
        w = np.ones(len(c), dtype=np.float64) / len(c)
    else:
        scaled = T * (c - c_max) / c_max
        scaled = scaled - scaled.max()
        exp_s = np.exp(scaled)
        w = exp_s / exp_s.sum()
    a_rev = a[::-1]
    full = fft_convolve(w, a_rev)
    end_idx = min(2 * n - 1, len(full))
    grad = np.zeros(n, dtype=np.float64)
    grad[:] = 2.0 * full[n - 1: end_idx][:n]
    return grad


def nesterov_optimize(a_init, n, S, time_budget, initial_T=10.0, final_T=500.0,
                      top_k=8, lr_init=0.8, momentum=0.9):
    """Nesterov-accelerated optimization with adaptive temperature.
    Works directly in the 'a' space with projection onto {a>=0, sum(a)=S}."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())

    y = a.copy()  # Nesterov's y
    v = np.zeros(n, dtype=np.float64)  # velocity

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    step = 0

    while time.monotonic() - start < time_budget:
        step += 1
        # Temperature schedule (exponential)
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = initial_T * np.exp(np.log(final_T / initial_T) * progress)

        # Nesterov lookahead
        y = a + momentum * v

        # Compute convolution at y
        c_y = fft_convolve_self(y)
        c_y_max = c_y.max()

        if c_y_max < best_obj:
            best_obj = c_y_max
            best_a = y.copy()

        # Compute gradient (soft if T is moderate, top-k if T is large)
        if T < 100:
            grad = compute_soft_grad(c_y, y, T)
        else:
            grad = compute_conv_grad(c_y, y, top_k)

        # Project gradient onto sum=0 subspace
        grad -= grad.mean()

        # Adaptive step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        lr = lr_init * (1.0 / (1.0 + 0.01 * step)) * (S / gnorm) * 0.1
        lr = min(lr, S * 0.1)  # cap step size

        # Update
        v = momentum * v - lr * grad / gnorm
        y_new = a + v

        # Project onto {a >= 0, sum = S}
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-10:
            y_new = np.full(n, S / n)
        else:
            y_new = y_new * (S / s)

        a = y_new

        # Periodic check of true objective
        if step % 50 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() < best_obj:
                best_obj = c_check.max()
                best_a = a.copy()

    return best_a, best_obj


def solve_lp_direction(f, M, n, top_k_constraints=100, time_limit=5.0):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for top-k lags, g >= 0."""
    c = fft_convolve_self(f)
    if top_k_constraints < len(c):
        active_lags = np.argsort(c)[::-1][:top_k_constraints]
    else:
        active_lags = np.arange(len(c))

    # Build constraint matrix: (f*g)[k] = sum_j f[k-j]*g[j] for 0<=k-j<n
    m_con = len(active_lags)
    A = np.zeros((m_con, n), dtype=np.float64)
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[idx, j] = f[k - j]

    c_obj = -np.ones(n)
    b = np.full(m_con, M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return None, None


def coordinate_descent(a, n, S, time_budget, candidates_per_coord=8):
    """Coordinate descent: for each coordinate, try a few candidate values."""
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    n_sweeps = 0

    while time.monotonic() - start < time_budget:
        n_sweeps += 1
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            original = a[i]
            # Generate candidate values
            if original < 1e-10:
                candidates = [0.0, S / (n * 4), S / (n * 2)]
            else:
                candidates = [0.0, original * 0.3, original * 0.6,
                              original * 0.8, original * 1.0,
                              original * 1.2, original * 1.5, original * 2.0]

            best_local = original
            best_local_obj = best_obj

            # Precompute base convolution values (without a[i]'s contribution)
            # conv[k] - contribution of a[i]
            ci = int(i)
            relevant_ks = []
            relevant_vals = []
            for k in range(ci, min(ci + n, 2 * n - 1)):
                if k == 2 * ci:
                    # quadratic term
                    base_val = c[k] - original * original
                    relevant_ks.append(k)
                    relevant_vals.append(('quad', base_val))
                else:
                    ki = k - ci
                    if 0 <= ki < n:
                        base_val = c[k] - 2.0 * original * a[ki]
                        relevant_ks.append(k)
                        relevant_vals.append(('lin', base_val, a[ki]))

            # Max of unchanged values
            unchanged_mask = np.ones(2 * n - 1, dtype=bool)
            for k in relevant_ks:
                unchanged_mask[k] = False
            unchanged_max = c[unchanged_mask].max() if unchanged_mask.any() else 0.0

            for x in candidates:
                # Compute new max
                new_max = unchanged_max
                for item in relevant_vals:
                    if item[0] == 'quad':
                        val = item[1] + x * x
                    else:
                        val = item[1] + 2.0 * x * item[2]
                    if val > new_max:
                        new_max = val
                if new_max < best_local_obj:
                    best_local_obj = new_max
                    best_local = x

            if best_local != original:
                a[ci] = best_local
                # Renormalize
                s = a.sum()
                if s > 1e-10:
                    a = a * (S / s)
                else:
                    a = np.full(n, S / n)
                improved = True
                # Update convolution
                c = fft_convolve_self(a)
                if c.max() < best_obj:
                    best_obj = c.max()
                    best_a = a.copy()

        if not improved:
            break
    return best_a, best_obj, best_obj


def get_initialization(name, n, rng, S):
    """Generate various initializations."""
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "gaussian":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n / 2) / (n / 5)) ** 2)
        return a * (S / a.sum())
    elif name == "cosine_mod":
        i = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * 7 * i / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "anti_sym":
        i = np.arange(n)
        half = n // 2
        a = np.zeros(n)
        a[:half] = 1.0 + 0.3 * np.random.random(half)
        a[half:] = 0.1 + 0.1 * np.random.random(n - half)
        return a * (S / a.sum())
    elif name == "spike":
        a = np.ones(n) * 0.1
        n_spikes = max(1, n // 16)
        positions = rng.choice(n, size=n_spikes, replace=False)
        a[positions] = 1.0
        return a * (S / a.sum())
    elif name == "two_scale":
        i = np.arange(n)
        block = n // 8
        a = np.zeros(n)
        for b in range(8):
            a[b*block:(b+1)*block] = 0.5 + 0.5 * np.cos(2 * np.pi * b / 8)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)

            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
            print("  Final gradient: obj=%.6f" % obj_best)

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("FINAL: obj=%.6f (verified=%.6f)" % (global_best, ver if np.isfinite(ver) else float('inf')))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.6f" % ver)

    return final_a
