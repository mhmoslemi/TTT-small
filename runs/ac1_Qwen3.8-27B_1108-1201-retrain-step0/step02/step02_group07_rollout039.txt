import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_from_conv(conv, n):
    return 2.0 * n * conv.max()

def compute_subgradient(conv, a, temp=1.0):
    """Compute smoothed subgradient of max(conv) w.r.t. a."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    # Softmax weighting over conv values
    scaled = conv / (max_c * temp)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # Subgradient: for each j, sg[j] = sum_k w[k] * a[k-j] (valid range)
    sg = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-8:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            # a[je:js-1:-1] gives a[k-je], ..., a[k-js] = a[js], ..., a[je]
            # Wait, I need a[k-j] for j in [js, je]
            # a[k-j] for j=js..je is a[k-js]..a[k-je] in reverse
