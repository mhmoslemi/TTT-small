def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def soft_max_and_weights(x, tau):
    """Compute soft-max value and normalized weights."""
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    return x_max + tau * np.log(s), e / s


def grad_soft_max_conv(a, tau):
    """Compute soft-max(conv(a,a)) and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_and_weights(c, tau)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = (2.0 / tau) * corr[n - 1:2 * n - 1]
    return value, g, c


def fft_phase_init(n, rng, n_iter=80, N_override=None):
    """Gerchberg-Saxton style initialization: flatten magnitude spectrum."""
    if N_override is None:
        N = 2 * n - 1
        N_fft = 1
        while N_fft < N:
            N_fft <<= 1
    else:
        N_fft = N_override

    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a

    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)

def multi_tone_init(n, rng, n_tones=4):

    a = np.ones(n)

def random_init(n, rng, spread=0.4):

def upsample(a, new_n, rng):
    """Upsample a sequence to a longer length via interpolation + noise."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.02 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)

    m2 = np.zeros(n)

def local_search(a_init, time_limit, rng, top_k=8):
    """
    Simulated annealing with convolution-peak-directed perturbations.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.004
    cooling = 0.9996
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        iter_count += 1

        if r < 0.35:
            # Peak-directed reduction
            conv = fft_convolve(current, current)
            n_top = min(top_k, len(conv))
            top_lags = np.argsort(conv)[-n_top:]
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            n_reduce = max(2, n // 10)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.06, 0.02, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.06, 0.02, size=len(bot_idx)))
        elif r < 0.5:
            # Adjacent swap-like: reduce one block, increase adjacent
            width = max(1, int(rng.integers(1, max(2, n // 20))))
            start = int(rng.integers(0, n - 2 * width))
            candidate[start:start + width] *= np.exp(rng.normal(-0.08, 0.02, size=width))
            candidate[start + width:start + 2 * width] *= np.exp(rng.normal(0.08, 0.02, size=width))
        elif r < 0.65:
            # Multi-block perturbation
            n_blocks = int(rng.integers(2, 5))
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 15))))
                bs = int(rng.integers(0, n - bw))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0.0, 0.05, size=bw))
        elif r < 0.8:
            # Partial symmetrization
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.15 * (sym - current)
            candidate += 0.01 * rng.standard_normal(n)
        elif r < 0.9:
            # Sparsification attempt: reduce a random subset significantly
            n_sparse = max(1, n // 5)
            sparse_idx = rng.choice(n, size=n_sparse, replace=False)
            candidate[sparse_idx] *= np.exp(rng.normal(-0.3, 0.1, size=n_sparse))
            dense_idx = np.setdiff1d(np.arange(n), sparse_idx)
            if len(dense_idx) > 0:
                n_inc = min(5, len(dense_idx))
                inc_idx = rng.choice(dense_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.3, 0.1, size=n_inc))
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0.0, 0.015, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current
        else:
            candidate = candidate / s * np.sum(current)

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 10:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score

        temp *= cooling

    return best, best_score
