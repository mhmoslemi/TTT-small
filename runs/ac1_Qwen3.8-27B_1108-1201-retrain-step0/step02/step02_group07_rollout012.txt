```python
def lp_iteration(a, M, method='highs'):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0"""
    n = len(a)
    # Build the convolution matrix
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1
    # Size: (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx <= n-1:
                A[k, j] = a[idx]
    
    # LP: max 1^T g s.t. A g <= M, g >= 0
    # scipy linprog minimizes, so: min -1^T g s.t. A g <= M, g >= 0
    from scipy.optimize import linprog
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method=method)
    if res.success:
        return res.x
    return None
```

```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

    A = np.fft.rfft(a, fft_size)
    G = np.fft.rfft(g, fft_size)
    result = np.fft.irfft(A * G, fft_size)
    return result[:size]

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        valid_k = range(j, j+n)
        A[valid_k, j] = a
    return A

def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_iteration(a, max_iter=1000, time_limit=100.0, blend_rate=None):
    """LP-based iterative optimizer. Normalizes a to sum=1."""
    a = np.maximum(a, 0.0).copy()

    best_a = a.copy()

            elapsed = time.monotonic() - t0

            break

        g = lp_direction(a, M)
        if g is None:
            break

        if g_sum < 1e-10:

        g /= g_sum  # Normalize g to sum=1
        
        # Line search for optimal blend
        # Try several blend factors
        best_t = 0.0
        best_blend_obj = best_obj

def compute_subgradient(conv, a, t):
    """Smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = t * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    thresh = w.max() * 0.002
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je+1] += w[k] * a[k-je:k-js+1][::-1]
    return sg

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_descent(a0, time_limit, t_init=50.0, eta_init=0.3):
    """Multiplicative mirror descent on the simplex."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    max_pat = 50
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient(conv, a, t)
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            eta *= 0.65
            t *= 1.3
            if eta < 0.004:
                elapsed = time.monotonic() - t0
                frac = min(1.0, elapsed / max(time_limit, 1.0))
                eta = eta_init * (1.0 - 0.5 * frac)
                t = t_init * (1.0 + 3.0 * frac)
    return best_a, best_obj

def peak_reduction_local_search(a, time_limit, rng):
    """Local search that specifically targets peak reduction."""
    a = np.maximum(a, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        peak_pos = np.where(conv >= 0.9 * max_c)[0]
        
        # Pick a peak position
        k = int(rng.choice(peak_pos))
        
        # Find contributing pairs
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        
        # Find the pair with maximum contribution
        best_delta = 0
        best_i = -1
        best_j = -1
        for i in range(js, je + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > best_delta:
                best_delta = prod
                best_i = i
                best_j = j
        
        if best_i < 0 or best_delta < 1e-15:
            continue
        
        # Choose where to move the mass
        # Prefer positions that don't contribute to peaks
        candidates = []
        for _ in range(20):
            c = int(rng.integers(n))
            if c == best_i or c == best_j:
                continue
            # Check contribution to peaks
            contrib = 0
            for pk in peak_pos[:5]:
                d1 = pk - c
                d2 = pk - (n-1) + c  # reversed
                if 0 <= d1 < n:
                    contrib += a[d1]
                if 0 <= d2 < n:
                    contrib += a[d2]
            candidates.append((contrib, c))
        
        if not candidates:
            continue
        
        candidates.sort()
        target = candidates[0][1]
        
        # Transfer mass
        delta = a[best_i] * rng.uniform(0.02, 0.1)
        a[best_i] -= delta
        a[target] += delta
        if a[best_i] < 0:
            a[best_i] += delta
            a[target] -= delta
            continue
        
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        
        if new_obj < cur_obj - 1e-12:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Revert
            a[best_i] += delta
            a[target] -= delta
            no_improve += 1
            if no_improve > 500:
                break
    
    return best_a, best_obj
```

```python
"""
LP-based iterative optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: Uses a linear programming oracle at each step to find the

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_iteration(a, time_limit, max_iter=100000, restart_patience=200):
    """LP-based iterative optimizer with line search and restarts."""
    a = np.maximum(a, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    stagnation = 0

    for it in range(max_iter):
        if it % 500 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print("      LP iter %d: obj=%.7f, elapsed=%.1fs" % (it, best_obj, elapsed))
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max()
        g = lp_direction(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search: try decreasing blend factors
        best_t = 0.0
        best_blend_obj = best_obj
        for t in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
            a_blend = (1.0 - t) * a + t * g
            conv_blend = fast_convolve(a_blend)
            obj_blend = 2.0 * n * conv_blend.max()
            if obj_blend < best_blend_obj:
                best_blend_obj = obj_blend
                best_t = t

        if best_t < 0.01:
            stagnation += 1
            if stagnation > restart_patience:
                # Add random perturbation to escape
                noise = np.random.default_rng(it).dirichlet(np.ones(n) * 0.5)
                a = 0.9 * a + 0.1 * noise
                conv = fast_convolve(a)
                best_obj = 2.0 * n * conv.max()
                if best_obj < 2.0 * n * best_a.max() if hasattr(best_a, 'max') else best_obj:
                    pass
                best_obj = 2.0 * n * conv.max()
                stagnation = 0
            continue

        a = (1.0 - best_t) * a + best_t * g
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

    return best_a, best_obj


def compute_subgradient(conv, a, t):
    """Smoothed subgradient of max(conv)."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = t * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    thresh = w.max() * 0.002
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    return sg


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mirror_descent(a0, time_limit, t_init=50.0, eta_init=0.3):
    """Mirror descent on simplex."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    max_pat = 60
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient(conv, a, t)
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            eta *= 0.6
            t *= 1.4
            if eta < 0.003:
                elapsed = time.monotonic() - t0
                frac = min(1.0, elapsed / max(time_limit, 1.0))

    return best_a, best_obj

def peak_local_search(a, time_limit, rng):
    """Local search targeting peak reduction via mass transfer."""
    a = np.maximum(a, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        stall += 1
        if stall > 3000:
            break

        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_pos = np.where(conv >= 0.85 * max_c)[0]
        k = int(rng.choice(peak_pos))

        js = max(0, k - n + 1)
        je = min(n - 1, k)

        # Find the contributing pair with max product
        best_prod = 0.0
        best_i = -1
        for i in range(js, je + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > best_prod:
                best_prod = prod
                best_i = i
        if best_i < 0 or best_prod < 1e-15:
            continue
        best_j = k - best_i

        # Choose target: position with low contribution to peaks
        target = int(rng.integers(n))
        for _ in range(10):
            cand = int(rng.integers(n))
            if cand == best_i or cand == best_j:
                continue
            contrib = 0.0
            for pk in peak_pos[:6]:
                d = pk - cand
                if 0 <= d < n:
                    contrib += a[d]
            if contrib < a.mean() * 1.5:
                target = cand
                break

        if target == best_i or target == best_j:
            continue

        delta = a[best_i] * rng.uniform(0.01, 0.08)
        if delta < 1e-10:
            continue
        a[best_i] -= delta
        a[target] += delta
        if a[best_i] < 0:
            a[best_i] += delta
            a[target] -= delta
            continue

        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        if new_obj < cur_obj - 1e-12:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                stall = 0
        else:
            a[best_i] += delta
            a[target] -= delta

    return best_a, best_obj

def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0).copy()
            pp /= pp.sum()
            inits['parent'] = pp
        else:
            x_new = np.linspace(0, 1, n)
            x_old = np.linspace(0, 1, len(parent))
            pp = np.interp(x_new, x_old, np.maximum(parent, 0.0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp

    # Structured: half-cosine (concentrated in middle)
    cos_half = np.sin(np.pi * x / (2 * n)) ** 2
    if cos_half.sum() > 1e-10:
        cos_half /= cos_half.sum()
        inits['half_cos'] = cos_half

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Bimodal
    bi = np.zeros(n)
    w = max(2, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random Dirichlet
    for alpha in [0.3, 1.0, 3.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['rand_a%s' % str(alpha)] = r

    # Gaussian-like
    g = np.exp(-0.5 * ((x - 0.4 * n) / (0.2 * n)) ** 2)
    g = np.maximum(g, 0)
    if g.sum() > 1e-10:
        g /= g.sum()
        inits['gauss'] = g

    return inits

    if best_sequence is not None and time.monotonic() < deadline - 60:
        print("Phase 3: Peak local search...")
        phase3_budget = min(350.0, deadline - time.monotonic() - 60)
        n_ls = 5
        ls_per = phase3_budget / n_ls

        for i in range(n_ls):
            if time.monotonic() > deadline - 60:
                break
            a_start = best_sequence.copy()
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
            rng_ls = np.random.default_rng(seed + 10000 + i * 777)
            a_opt, _ = peak_local_search(a_start, time_limit=ls_per, rng=rng_ls)
            record(a_opt)

        # Also try LP from the local search results
        if time.monotonic() < deadline - 60:
            rem = deadline - time.monotonic() - 60
            if rem > 10:
                a_opt, _ = lp_iteration(best_sequence, time_limit=rem * 0.6, max_iter=20000)
                record(a_opt)
                if time.monotonic() < deadline - 60:
                    rem2 = deadline - time.monotonic() - 60
                    if rem2 > 5:
                        a_opt2, _ = mirror_descent(a_opt, time_limit=rem2, t_init=300, eta_init=0.03)
                        record(a_opt2)

        print("  Phase 3 done. Best: %.6f" % best_score)

    # Phase 4: Final micro-optimization
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("Phase 4: Final micro-optimization...")
        remaining = deadline - time.monotonic() - 5
        a_best = best_sequence.copy()
        a_best = np.maximum(a_best, 0.0)
        a_best /= a_best.sum()

        if remaining > 20:
            # One more round of LP
            a_opt, _ = lp_iteration(a_best, time_limit=remaining * 0.4, max_iter=100000)
            record(a_opt)

        if time.monotonic() < deadline - 5:
            rem = deadline - time.monotonic() - 5
            if rem > 5:
                a_opt, _ = pgd_optimize(a_best, time_limit=rem, t_init=500, lr_init=0.005)
                record(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
```
