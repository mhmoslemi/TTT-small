```python
"""
Optimization for minimum peak auto-convolution of non-negative sequences.
Uses the key insight that np.convolve(a,a) (auto-convolution) can have its
maximum away from lag 0 for non-negative sequences, enabling scores below 2.
Employs L-BFGS with annealed p-norm objective, multiple initializations,
multiple sequence lengths, and local search polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def softmax_fn(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    c = fft_convolve(a, a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    return 2.0 * n * np.max(c) / (s * s)


def pnorm_obj_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p / (sum softmax)^2.
    Uses p-norm as smooth approximation to max.
    """
    a = softmax_fn(theta)
    n = len(a)
    c = fft_convolve(a, a)

    c_pow = c ** p
    c_sum = np.sum(c_pow)
    if c_sum < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = c_sum ** (1.0 / p)
    S = np.sum(a)
    score = 2.0 * n * pnorm / (S * S)

    # dP/dc_k = c_k^(p-1) / c_sum^(1 - 1/p)
    w = (c ** (p - 1)) / (c_sum ** (1.0 - 1.0 / p))

    # grad_a = d<w, conv(a,a)>/da = conv(w, a_reversed)[n-1 : 2n-1]
    a_rev = a[::-1]
    size = 1
    while size < len(w) + n:
        size <<= 1
    F_w = np.fft.rfft(w, size)
    F_ar = np.fft.rfft(a_rev, size)
    conv_result = np.fft.irfft(F_w * F_ar, size)
    grad_P = conv_result[n - 1:2 * n - 1]

    # d score / d a_j = 2n * (grad_P_j * S - 2 * pnorm) / S^3
    dscore_da = 2.0 * n * (grad_P * S - 2.0 * pnorm) / (S * S * S)

    # Convert to grad_theta via softmax Jacobian
    inner = np.dot(dscore_da, a)
    grad_theta = a * (dscore_da - inner)

    return score, grad_theta


def local_search(a_init, time_limit, rng, step_scale=0.01):
    """
    Simulated annealing local search with targeted convolution flattening.
    Works on normalized sequence (sum=1).
    """
    current = a_init.astype(np.float64).copy()
    current /= np.sum(current)
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.001
    cooling = 0.9997
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.25:
            # Targeted: reduce contribution to peak lag
            c = fft_convolve(current, current)
            peak_k = int(np.argmax(c))
            # Find indices that contribute most to c[peak_k]
            lo = max(0, peak_k - n + 1)
            hi = min(n - 1, peak_k)
            contribs = np.zeros(n)
            for i in range(lo, hi + 1):
                j = peak_k - i
                if 0 <= j < n:
                    contribs[i] += current[i] * current[j]
            # Reduce top contributors, increase others
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.01, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.01, size=n_inc))
        elif r < 0.45:
            # Flatten: push values toward the "average" convolution level
            c = fft_convolve(current, current)
            c_mean = np.sum(c) / len(c)
            # For each index, compute how much it contributes to "excess" convolution
            pull = np.zeros(n)
            for k in range(len(c)):
                excess = max(0.0, c[k] - c_mean)
                if excess > 0:
                    lo = max(0, k - n + 1)
                    hi = min(n - 1, k)
                    for i in range(lo, hi + 1):
                        j = k - i
                        if 0 <= j < n:
                            pull[i] += excess * current[j]
            if np.max(pull) > 1e-12:
                pull = pull / np.max(pull)
                candidate *= np.exp(-0.015 * pull)
        elif r < 0.60:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.75:
            # Periodic structure injection
            period = int(rng.integers(2, max(3, n // 10)))
            phase = rng.random() * period
            t = np.arange(n)
            mod = np.exp(-0.5 * ((t + phase) % period - period / 2) ** 2 / max(1, period * 0.3))
            candidate *= (1.0 - 0.1) + 0.1 * mod / np.max(mod)
        elif r < 0.85:
            # Symmetrize partially
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.01, size=n))
        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0.0, step_scale, size=n))

        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)
        new_score = compute_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 1e-10)

        if time.monotonic() - last_print > 10.0:
            last_print = time.monotonic()
            print(f"    local: score={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")

    return best, best_score


def make_periodic_init(n, period, rng, spread=0.2):
    """Create a periodic initialization."""
    t = np.arange(n, dtype=np.float64)
    base = 1.0 + 0.3 * np.cos(2.0 * np.pi * t / period)
    base += spread * rng.standard_normal(n)
    base = np.clip(base, 0.01, None)
    return base / np.sum(base)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.1 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_binary_init(n, density, rng):
    """Binary-like initialization with smoothing."""
    a = (rng.random(n) < density).astype(float)
    # Add small smoothing
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p, deadline):
    """Run L-BFGS with annealed p-norm. Returns best theta and score."""
    best_theta = theta.copy()
    best_score = compute_score(softmax_fn(theta))

    for p in p_schedule:
        if time.monotonic() > deadline - 10:
            break

        result = minimize(
            pnorm_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(p,),
            options={'maxiter': maxiter_per_p, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': maxiter_per_p * 10}
        )
        theta_new = result.x
        a_new = softmax_fn(theta_new)
        s = compute_score(a_new)

        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        theta = best_theta

    return best_theta, best_score
```
