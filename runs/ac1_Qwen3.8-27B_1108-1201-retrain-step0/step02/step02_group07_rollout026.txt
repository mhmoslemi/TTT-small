"""
LP-based best response iteration combined with L-BFGS-B refinement and
informed local search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Exploits U-shaped (bimodal) sequence structure to achieve objectives below 2.
Key innovation: uses linear programming to find optimal "partner" directions,
combined with proper gradient-based optimization via FFT-accelerated cross-correlation.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_value(a):
    """Objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(a, p):
    """Compute gradient of sum_k p[k]*conv_k w.r.t. a using FFT.
    p has length 2n-1. Returns vector of length n.
    grad[i] = 2 * sum_j a[j] * p[i+j] (with bounds)
    """
    n = len(a)
    qlen = 2 * n - 1
    q = p[::-1]
    conv_len = n + qlen - 1
    L = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, L)
    Q = np.fft.rfft(q, L)
    conv = np.fft.irfft(A * Q, L)[:conv_len]
    # grad[i] = 2 * conv[2n-2-i] for i=0..n-1
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv[idx]
    return grad


def smoothed_obj_grad(a, t):
    """Compute smoothed objective and its gradient.
    f(a) = 2n/t * log(sum_k exp(t*conv_k))
    grad = 2n * sum_k p_k * d(conv_k)/da where p = softmax(t*conv)
    """
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return 2.0 * n * max_c, np.zeros(n)
    scaled = t * (conv - max_c)
    w = np.exp(scaled)
    w_sum = w.sum()
    p = w / w_sum
    # Objective: 2n * max_c + (2n/t) * log(w_sum) ... actually:
    # logsumexp(t*conv) = t*max_c + log(w_sum)
    # f = 2n/t * logsumexp(t*conv) = 2n*max_c + 2n/t*log(w_sum)
    obj = 2.0 * n * max_c + 2.0 * n * np.log(w_sum) / t
    grad = compute_gradient_fft(a, p)
    return obj, grad


def lp_solve(a, M):
    """Solve LP: max sum(b) s.t. conv_a(b) <= M componentwise, b >= 0.
    Returns b or None."""
    n = len(a)
    num_c = 2 * n - 1
    A_ub = np.zeros((num_c, n))
    for k in range(num_c):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            A_ub[k, i_lo:i_hi + 1] = a[k - i_hi:k - i_lo + 1]
    b_ub = np.full(num_c, M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0.0)
    return None


def lp_iterate(a, time_limit, max_iters=1000):
    """LP-based best response iteration."""
    a = normalize(a)
    n = len(a)
    t0 = time.monotonic()
    t = 0.6
    best_obj = obj_value(a)
    best_a = a.copy()
    stall = 0
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_convolve(a)
        M = conv.max()
        b = lp_solve(a, M)
        if b is None:
            break
        b_sum = b.sum()
        if b_sum < 1e-12:
            break
        b = b / b_sum
        a_new = (1.0 - t) * a + t * b
        a_new = normalize(a_new)
        new_obj = obj_value(a_new)
        cur_obj = obj_value(a)
        if new_obj < best_obj - 1e-10:
            best_obj = new_obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        if new_obj < cur_obj:
            a = a_new
            t = max(0.3, t * 0.995)
        else:
            t = max(0.15, t * 0.85)
        if stall > 200:
            t = 0.6
            stall = 0
    return best_a, best_obj


def lbfgs_optimize(a0, time_limit, t_start=10.0, t_end=0.5):
    """L-BFGS-B optimization on smoothed objective with decreasing t."""
    a = normalize(a0)
    n = len(a)
    t0 = time.monotonic()
    best_obj = obj_value(a)
    best_a = a.copy()
    t = t_start
    n_restarts = 5
    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        t = t_start * (t_end / t_start) ** (r / max(1, n_restarts - 1))

        def fun_and_grad(x):
            x_pos = np.maximum(x, 0.0)
            s = x_pos.sum()
            if s < 1e-15:
                return 0.0, np.zeros(len(x))
            x_norm = x_pos / s
            obj, grad = smoothed_obj_grad(x_norm, t)
            return obj, grad

        x0 = best_a.copy()
        res = minimize(fun_and_grad, x0, method='L-BFGS-B',
                       options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8,
                                'maxls': 50})
        a_new = normalize(res.x)
        new_obj = obj_value(a_new)
        if new_obj < best_obj - 1e-10:
            best_obj = new_obj
            best_a = a_new.copy()
    return best_a, best_obj


def subgradient_polish(a0, time_limit, t_init=50.0, eta_init=0.3):
    """Mirror descent / multiplicative weight update."""
    a = normalize(a0)
    n = len(a)
    t0 = time.monotonic()
    t = t_init
    eta = eta_init
    best_obj = obj_value(a)
    best_a = a.copy()
    patience = 0
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        sg = compute_gradient_fft(a, w)
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        cur_obj = obj_value(a)
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= 30:
            patience = 0
            eta *= 0.6
            t *= 1.4
            if eta < 0.005:
                break
    return best_a, best_obj


def informed_anneal(a0, time_limit, rng):
    """Informed simulated annealing: target peak-reducing transfers."""
    a = normalize(a0)
    n = len(a)
    t0 = time.monotonic()
    cur_obj = obj_value(a)
    best_obj = cur_obj
    best_a = a.copy()
    conv = fast_convolve(a)
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.02 * (1.0 - frac) ** 2 + 0.0005
        max_c = conv.max()
        peak_thresh = max_c * 0.7
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        # Target: position with low contribution to peaks
        target_j = int(rng.integers(n))
        for _ in range(3):
            candidate = int(rng.integers(n))
            contrib = 0.0
            for pk in peak_positions[:5]:
                d = pk - candidate
                if 0 <= d < n:
                    contrib += a[d]
            if contrib < a.mean() * 1.5:
                target_j = candidate
                break
        if target_j == top_j:
            continue
        delta = a[top_j] * rng.uniform(0.01, 0.05)
        if delta < 1e-10:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n
    # U-shaped with varying width
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.4]:
        u = np.exp(-0.5 * (x / (sigma_frac * n)) ** 2) + np.exp(-0.5 * ((x - (n - 1)) / (sigma_frac * n)) ** 2)
        u = np.maximum(u, 1e-10)
        u /= u.sum()
        inits['u_%s' % str(sigma_frac)] = u
    # Center-heavy
    for sigma_frac in [0.15, 0.25, 0.35]:
        c = np.exp(-0.5 * ((x - (n - 1) / 2.0) / (sigma_frac * n / 2.0)) ** 2)
        c /= c.sum()
        inits['center_%s' % str(sigma_frac)] = c
    # Bimodal (two bumps)
    for pos in [0.2, 0.3]:
        b = np.exp(-0.5 * ((x - pos * n) / (0.08 * n)) ** 2) + np.exp(-0.5 * ((x - (1 - pos) * n) / (0.08 * n)) ** 2)
        b = np.maximum(b, 1e-10)
        b /= b.sum()
        inits['bimodal_%s' % str(pos)] = b
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['rand_%s' % str(alpha)] = r
    # Sine-shaped
    s = 0.5 * (1.0 + np.cos(np.pi * x / (n - 1)))
    s = np.maximum(s, 1e-10)
    s /= s.sum()
    inits['sine'] = s
    # Parent
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            if p.size == n:
                inits['parent'] = normalize(p)
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
                inits['parent_interp'] = normalize(pp)
    return inits
