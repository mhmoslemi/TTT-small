def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_value(a):
    """Objective assuming sum(a) = 1: 2*n*max(conv(a,a))"""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(conv, a, t):
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = t * conv[:2*n-1] / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:

        w[np.argmax(conv[:2*n-1])] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    for ki in range(len(w)):
        if w[ki] < 1e-10:
            continue
        k = ki
        js = max(0, k - n + 1)
        je = min(n - 1, k)

Optimized step function search combining mirror descent, peak-targeting coordinate
descent, and informed annealing with multi-length exploration and adaptive restarts.
Focuses on flattening the self-convolution to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_value(a):
    """Objective assuming sum(a) = 1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(conv, a, t, thresh_frac=0.003):
    """Smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = t * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    thresh = w.max() * thresh_frac
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * 2.0 * a[k - je:k - js + 1][::-1]
    return sg


def mirror_descent(a0, time_limit, t_init=50.0, eta_init=0.35):
    """Multiplicative mirror descent on the simplex with adaptive params."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    max_pat = 50
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient(conv, a, t)
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            eta *= 0.6
            t *= 1.4
            if eta < 0.003:
                break
    return best_a, best_obj


def peak_targeting_cd(a0, time_limit, rng, allow_uhill=False):
    """Coordinate descent targeting convolution peaks with optional stochasticity."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    stall_count = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.85
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])

        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total = products.sum()
        if total < 1e-15:
            continue

        top_local = int(np.argmax(products))
        top_j = js + top_local

        # Score positions by their contribution to peaks
        contrib = np.zeros(n)
        for pk in peak_positions[:12]:
            ps = max(0, pk - n + 1)
            pe = min(n - 1, pk)
            if ps <= pe:
                contrib[ps:pe + 1] += a[pk - pe:pk - ps + 1][::-1]

        scores = contrib + a * 0.05
        target_options = np.where(scores < np.median(scores))[0]
        if len(target_options) == 0:
            target_options = np.arange(n)
        target_j = int(rng.choice(target_options))

        if target_j == top_j:
            continue

        delta = a[top_j] * rng.uniform(0.03, 0.12)
        if delta < 1e-12:
            continue

        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue

        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()

        if new_obj < cur_obj:
            cur_obj = new_obj
            conv = new_conv
            stall_count = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if allow_uhill and rng.random() < 0.1:
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            # Perturb to escape local minimum
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j:
                    d = a[i] * 0.05
                    a[i] -= d
                    a[j] += d
                    if a[i] >= 0:
                        conv = fast_convolve(a)
                        cur_obj = 2.0 * n * conv.max()
                        break

    return best_a, best_obj


def informed_anneal(a0, time_limit, rng):
    """Informed simulated annealing targeting peak reduction."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1.0))
        T = 0.025 * (1.0 - frac) ** 2.5 + 0.0008

        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.8
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])

        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total = products.sum()
        if total < 1e-15:
            continue
        probs = products / total
        top_j = js + int(rng.choice(je - js + 1, p=probs))

        target_j = int(rng.integers(n))
        if target_j == top_j:
            continue
        delta = a[top_j] * rng.uniform(0.02, 0.1)
        if delta < 1e-12:
            continue

        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue

        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj

        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            a[top_j] += delta
            a[target_j] -= delta

    return best_a, best_obj


def pgd_optimize(a0, time_limit, t_init=30.0, lr_init=0.06):
    """Projected gradient descent with softmax smoothing."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    lr = lr_init
    patience = 0
    max_pat = 40
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient(conv, a, t)
        a_new = a - lr * sg
        a_new = project_simplex(a_new)
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            lr *= 0.65
            t *= 1.3
            if lr < 0.002:
                break
    return best_a, best_obj


def get_parent():
    """Access parent sequence from global variable."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def safe_evaluate(a):
    """Safely evaluate a sequence via the external evaluate_sequence function."""
    try:
        a_clean = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a_clean = np.minimum(a_clean, 1000.0)
        if a_clean.sum() < 0.01:
            return float('inf')
        if not np.all(np.isfinite(a_clean)):
            return float('inf')
        score = float(evaluate_sequence(a_clean.tolist()))
        return score
    except Exception:
        return float('inf')


def generate_initializations(n, rng, parent):
    """Generate diverse initializations for a given length."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n

    for sf in [0.12, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % sf] = g

    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    for freq in [1, 2]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        s /= s.sum()
        inits['sin_%d' % freq] = s

    for alpha in [0.3, 1.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['rand_%s' % alpha] = r

    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0).copy()
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(
                np.linspace(0, 1, n),
                np.linspace(0, 1, parent.size),
                np.maximum(parent, 0)
            )
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp

    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search: multi-length exploration with mirror descent, peak-targeting CD,
    PGD, and informed annealing. Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)

    parent = get_parent()
    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d" % (score, len(a)))

    # Phase 1: Multi-length quick scan
    print("Phase 1: Multi-length exploration...")
    lengths = [128, 256, 512, 1024, 2048, 4096]
    if parent is not None:
        pn = len(parent)
        lengths = sorted(set([pn] + lengths))

    results = []
    phase1_budget = 350.0

    for n in lengths:
        if time.monotonic() > deadline - 1600:
            break
        inits = generate_initializations(n, rng, parent)
        inits_list = list(inits.items())
        time_per_init = max(2.5, 60.0 / max(1, len(inits_list)))

        for name, a0 in inits_list:
            if time.monotonic() > deadline - 1600:
                break
            t_alloc = min(time_per_init, deadline - time.monotonic() - 1600)
            if t_alloc < 1.5:
                break
            a_opt, _ = mirror_descent(a0, time_limit=t_alloc, t_init=40.0, eta_init=0.4)
            score = safe_evaluate(a_opt)
            if np.isfinite(score):
                results.append((score, n, name, a_opt))
            record(a_opt)

    print("  Phase 1 done. Best: %.6f" % best_score)
    results.sort(key=lambda x: x[0])

    # Phase 2: Deep multi-method optimization
    print("Phase 2: Deep optimization...")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in results:
        key = (n, name)
        if key not in seen and len(top_configs) < 12:
            top_configs.append((n, name, a_opt))
            seen.add(key)
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'current_best', best_sequence.copy()))

    phase2_budget = min(1300.0, (deadline - time.monotonic()) * 0.55)
    n_deep = len(top_configs)
    time_per_deep = max(10.0, phase2_budget / max(1, n_deep))

    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 700:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 700)
        if t_alloc < 8.0:
            break
        print("  [%d/%d] n=%d, %s, budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))

        a_cur = a0.copy()
        n_sub = 5
        sub_time = t_alloc / n_sub

        for si in range(n_sub):
            if time.monotonic() > deadline - 700:
                break
            if si % 3 == 0:
                a_cur, _ = mirror_descent(
                    a_cur, time_limit=sub_time,
                    t_init=40 + si * 60, eta_init=0.3 / (1 + si * 0.3)
                )
            elif si % 3 == 1:
                rl = np.random.default_rng(seed + idx * 200 + si)
                a_cur, _ = peak_targeting_cd(
                    a_cur, time_limit=sub_time, rng=rl, allow_uhill=True
                )
            else:
                a_cur, _ = pgd_optimize(
                    a_cur, time_limit=sub_time,
                    t_init=30 + si * 40, lr_init=0.05 / (1 + si * 0.2)
                )
            record(a_cur)

    print("  Phase 2 done. Best: %.6f" % best_score)

    # Phase 3: Annealing and peak-targeting refinement
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("Phase 3: Annealing refinement...")
        phase3_budget = min(500.0, deadline - time.monotonic() - 100)
        n_refine = 6
        refine_per = phase3_budget / n_refine

        for i in range(n_refine):
            if time.monotonic() > deadline - 100:
                break
            a_start = best_sequence.copy()
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
            # Add small perturbation for diversity
            if i > 0:
                a_start += 0.01 * rng.standard_normal(len(a_start))
                a_start = np.maximum(a_start, 0.0)
                a_start /= a_start.sum()

            rl = np.random.default_rng(seed + 20000 + i * 13)
            if i % 3 == 0:
                a_opt, _ = informed_anneal(a_start, time_limit=refine_per, rng=rl)
            elif i % 3 == 1:
                a_opt, _ = peak_targeting_cd(a_start, time_limit=refine_per, rng=rl, allow_uhill=True)
            else:
                a_opt, _ = mirror_descent(a_start, time_limit=refine_per, t_init=100 + i * 50, eta_init=0.08)
            record(a_opt)

        print("  Phase 3 done. Best: %.6f" % best_score)

    # Phase 4: Final fine-grained polish
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_best = best_sequence.copy()
            a_best = np.maximum(a_best, 0.0)
            a_best /= a_best.sum()

            third = remaining / 3.0
            a_opt, _ = mirror_descent(a_best, time_limit=third, t_init=300.0, eta_init=0.04)
            record(a_opt)

            if time.monotonic() < deadline - 5:
                rem2 = deadline - time.monotonic() - 5
                rl = np.random.default_rng(seed + 99999)
                a_opt2, _ = peak_targeting_cd(a_best, time_limit=rem2, rng=rl, allow_uhill=False)
                record(a_opt2)

            if time.monotonic() < deadline - 5:
                rem3 = deadline - time.monotonic() - 5
                if rem3 > 3:
                    rl2 = np.random.default_rng(seed + 88888)
                    a_opt3, _ = informed_anneal(best_sequence, time_limit=rem3, rng=rl2)
                    record(a_opt3)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))
    return [float(x) for x in best_sequence]
