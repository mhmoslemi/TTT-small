import numpy as np
import time

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def ramp_topk_gradient(a, c, k, ramp_width=10):
    """Gradient with ramp-weighted top-k subgradient (smoother than hard top-k)."""
    n = len(a)
    indices = np.argsort(c)[::-1][:k + ramp_width]
    w_full = np.zeros(len(c))
    for i, idx in enumerate(indices):
        if i < k:
            w_full[idx] = 1.0
        else:
            w_full[idx] = max(0.0, 1.0 - (i - k + 1) / ramp_width)

    if total < 1e-30:
        return np.zeros(n)
    w_full /= total
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def quantize_to_levels(a, k):
    """Quantize a to k distinct levels using k-means-like iteration."""
    a = np.asarray(a, dtype=np.float64)
    if k <= 1:
        return np.full_like(a, a.mean())
    # Initialize cluster centers
    percentiles = np.linspace(5, 95, k)
    centers = np.percentile(a, percentiles)
    for _ in range(20):
        dists = np.abs(a[:, None] - centers[None, :])
        labels = np.argmin(dists, axis=1)
        new_centers = centers.copy()
        for j in range(k):
            mask = labels == j
            if mask.any():
                new_centers[j] = a[mask].mean()
        if np.allclose(centers, new_centers, atol=1e-10):
            break
        centers = new_centers
    centers.sort()
    dists = np.abs(a[:, None] - centers[None, :])
    labels = np.argmin(dists, axis=1)
    return centers[labels]

    inits.append(("rand", rng.random(n) + 0.1))

    # 3. Frequency-domain init
    try:
        fd_init = freq_domain_init(n, rng, n_trials=30)
        inits.append(("freq", fd_init))
    except Exception:
        pass

    # 4. Sinusoidal structure
    inits.append(("sin", np.abs(1.0 + 0.3 * np.sin(2 * np.pi * 5 * tt) +
                                0.2 * np.cos(2 * np.pi * 13 * tt)) + 0.05))

    # 5. Two-level (binary) random
    binary = (rng.random(n) > 0.5).astype(float)
    inits.append(("binary", binary + 0.01))

    # 6. Three-level random
    three_level = rng.choice([0.2, 1.0, 2.0], size=n)
    inits.append(("three_lvl", three_level))

    # 7. Gaussian envelope
    inits.append(("gauss", np.exp(-0.5 * ((tt - 0.3) * 4) ** 2) +
                  np.exp(-0.5 * ((tt - 0.7) * 4) ** 2) + 0.05))

    # 8. Ramp
    inits.append(("ramp", np.concatenate([np.linspace(0.1, 2.0, n // 2),
                                          np.linspace(2.0, 0.1, n - n // 2)])))

    print("=== Phase 1: Quick exploration (%d inits) ===" % len(inits))

    # Quick screen: run short FISTA on each init to pick the best
    screen_time = min(20.0, (deadline - time.monotonic()) * 0.08)
    per_screen = screen_time / max(1, len(inits))

    best_screen_score = np.inf
    best_screen_a = None
    best_screen_name = ""

    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break
        arr = np.clip(np.asarray(arr, dtype=np.float64), 0, 1000.0)
        sn = arr.sum()
        if sn < 1e-10:
            continue
        arr_norm = arr * (S / sn)

        # Quick FISTA
        a_s, score_s = fista_adaptive(arr_norm, n, S, per_screen,
                                       beta_start=5.0, beta_end=100.0,
                                       quantize_every=0, noise_every=0)
        c_s = fft_convolve_self(safe_normalize(a_s, n))
        final_score = c_s.max()
        if final_score < best_screen_score:
            best_screen_score = final_score
            best_screen_a = safe_normalize(a_s, n).copy()
            best_screen_name = name
            print("  [%s] score=%.6f" % (name, final_score))

    if best_screen_a is not None:
        global_a = best_screen_a.copy()
        global_best = best_screen_score
    else:
        global_a = np.ones(n) * S / n
        global_best = n  # constant sequence gives max(C) = n*(S/n)^2 = S^2/n

    # Also try given sequence directly if available
    if given_arr is not None:
        ga = safe_normalize(given_arr, len(given_arr))
        cg = fft_convolve_self(ga)
        if cg.max() < global_best:
            global_best = cg.max()
            global_a = ga.copy()
            n = len(global_a)
            S = np.sqrt(2.0 * float(n))

    print("After screening: best=%.6f (from %s), n=%d" % (global_best, best_screen_name, n))

    # --- Coarse-to-fine: try optimizing at coarser resolution ---
    if time.monotonic() < deadline - 400:
        n_coarse = max(256, n // 8)
        S_coarse = np.sqrt(2.0 * float(n_coarse))
        # Downsample global_a to coarse
        idx_c = np.linspace(0, n - 1, n_coarse)
        coarse_init = np.interp(idx_c, np.arange(n), global_a)

        a_c, score_c = fista_adaptive(coarse_init, n_coarse, S_coarse, 60.0,
                                       beta_start=5.0, beta_end=200.0,
                                       quantize_every=2000, quantize_levels=3)
        # Upsample
        a_up = coarse_to_fine(safe_normalize(a_c, n_coarse), n, S)
        a_up = safe_normalize(a_up, n)
        c_up = fft_convolve_self(a_up)
        if c_up.max() < global_best:
            global_best = c_up.max()
            global_a = a_up.copy()
            print("  Coarse-to-fine improved: %.6f" % global_best)

    # --- Phase 2: Main FISTA optimization ---
    print("=== Phase 2: Main optimization (n=%d) ===" % n)
    remaining = deadline - time.monotonic()
    main_time = max(10.0, remaining * 0.6)

    a_main, score_main = fista_adaptive(global_a, n, S, main_time,
                                         beta_start=5.0, beta_end=500.0,
                                         quantize_every=5000, quantize_levels=4,
                                         noise_every=10000, noise_scale=0.003,
                                         anneal_every=8000, anneal_factor=0.4)
    a_main = safe_normalize(a_main, n)
    c_main = fft_convolve_self(a_main)
    if c_main.max() < global_best:
        global_best = c_main.max()
        global_a = a_main.copy()
        print("  Main FISTA: %.6f" % global_best)

    # --- Phase 3: Second FISTA pass with different parameters ---
    if time.monotonic() < deadline - 100:
        remaining2 = deadline - time.monotonic()
        time3 = max(10.0, remaining2 * 0.4)
        a_3, score_3 = fista_adaptive(global_a, n, S, time3,
                                       beta_start=50.0, beta_end=1000.0,
                                       quantize_every=8000, quantize_levels=5,
                                       noise_every=0,
                                       ramp_topk_start=int(time3 * 100))
        a_3 = safe_normalize(a_3, n)
        c_3 = fft_convolve_self(a_3)
        if c_3.max() < global_best:
            global_best = c_3.max()
            global_a = a_3.copy()
            print("  Second pass: %.6f" % global_best)

    # --- Phase 4: Fine polishing with ramp top-k ---
    if time.monotonic() < deadline - 30:
        remaining4 = deadline - time.monotonic()
        time4 = max(5.0, remaining4 * 0.7)
        a_4, score_4 = fista_adaptive(global_a, n, S, time4,
                                       beta_start=200.0, beta_end=2000.0,
                                       quantize_every=0, noise_every=0,
                                       ramp_topk_start=0, ramp_k=20)
        a_4 = safe_normalize(a_4, n)
        c_4 = fft_convolve_self(a_4)
        if c_4.max() < global_best:
            global_best = c_4.max()
            global_a = a_4.copy()
            print("  Polish: %.6f" % global_best)

    # --- Phase 5: Try a different n for diversity ---
    if time.monotonic() < deadline - 60:
        n_alt = n * 2 if n <= 8192 else n // 2
        if n_alt >= 1024 and n_alt <= 32768:
            S_alt = np.sqrt(2.0 * float(n_alt))
            # Interpolate global_a to new length
            idx_alt = np.linspace(0, n_alt - 1, n_alt)
            init_alt = np.interp(idx_alt, np.arange(len(global_a)), global_a)
            time_alt = min(120.0, (deadline - time.monotonic()) * 0.3)
            a_alt, score_alt = fista_adaptive(init_alt, n_alt, S_alt, time_alt,
                                               beta_start=10.0, beta_end=300.0,
                                               quantize_every=4000, quantize_levels=3)
            a_alt = safe_normalize(a_alt, n_alt)
            # Evaluate using the actual formula: 2*n*max(C)/S^2
            c_alt = fft_convolve_self(a_alt)
            sum_alt = a_alt.sum()
            obj_alt = 2.0 * n_alt * c_alt.max() / (sum_alt ** 2)
            # Compare with global_best using same formula
            sum_g = global_a.sum()
            obj_g = 2.0 * len(global_a) * fft_convolve_self(global_a).max() / (sum_g ** 2)
            if obj_alt < obj_g:
                global_best = c_alt.max()
                global_a = a_alt.copy()
                print("  Alt n=%d: obj=%.6f" % (n_alt, obj_alt))

    # --- Final verification ---
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    try:
        ver = evaluate_sequence(final_list)
        print("Verification score: %.6f" % ver)
        if not np.isfinite(ver):
            print("  Rejected by evaluator, using fallback")
            global_a = np.ones(1024, dtype=np.float64)
            final_list = [float(x) for x in global_a]
            ver2 = evaluate_sequence(final_list)
            print("  Fallback score: %.6f" % ver2)
    except Exception as e:
        print("  Evaluation error: %s" % str(e))
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, objective~%.6f" % (len(global_a), global_best))
    return final_list
