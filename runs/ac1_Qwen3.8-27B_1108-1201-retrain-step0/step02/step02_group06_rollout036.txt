```python
def solve_lp(f, M):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g.
    """
    n = len(f)
    # (f*g)[k] = sum_j f[k-j]*g[j] for valid indices
    # We need: for each k in 0..2n-2, sum_{j} A[k,j] * g[j] <= M
    # where A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    
    # Build constraint matrix A (2n-1 rows, n columns)
    # A[k,j] = f[k-j] if 0 <= k-j < n else 0
    
    # For efficiency, use the FFT structure
    # Actually for n up to ~2000, building the matrix explicitly is fine
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # g[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # conv[k] gets contribution f[k-j]*g[j]
        for k in range(j, min(2*n-1, j+n)):
            A[k, j] = f[k-j]
    
    # Objective: minimize -sum(g) (i.e., maximize sum(g))
    c = -np.ones(n)
    
    # Constraints: A @ g <= M * ones
    b = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    return res.x
```

```python
"""
LP-based alternating minimization combined with subgradient descent
for minimizing peak linear autocorrelation of non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build the constraint matrix A for the LP: (f*g)[k] = sum_j A[k,j]*g[j]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in conv[k] for k = j, j+1, ..., j+n-1
        # conv[k] += f[k-j] * g[j]
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = f[:end_k - start_k]
    return A


def solve_lp_max_sum(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)  # minimize -sum(g)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x


def lp_alternating(f_init, time_limit, t_mix=0.5, verbose=False):
    """LP-based alternating minimization."""
    f = f_init.copy().astype(np.float64)
    f = np.clip(f, 0, None)
    s = np.sum(f)
    if s < 1e-10:
        return f, np.inf
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    prev_score = best_score
    stagnation = 0
    
    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_f = fft_convolve(f, f)
        M = np.max(conv_f)
        
        g = solve_lp_max_sum(f, M)
        if g is None:
            break
        
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        # Rescale
        g = g / g_sum
        
        # Adaptive mixing
        f_new = (1 - t_mix) * f + t_mix * g
        f_new = np.clip(f_new, 0, None)
        fs = np.sum(f_new)
        if fs < 1e-10:
            break
        f_new = f_new / fs
        
        new_score = score_fast(f_new)
        f = f_new
        
        if new_score < best_score:

            stagnation = 0

def subgradient_descent(f_init, time_limit, verbose=False):
    """Subgradient descent on the simplex for min max_k conv[k]."""
    f = f_init.copy().astype(np.float64)
    f = np.clip(f, 0, None)
    s = np.sum(f)
    if s < 1e-10:
        return f, np.inf
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    step = 0
    lr = 1.0 / n
    
    # Use smoothed max for better convergence
    tau = 1.0 / n

        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = (1.0 / n) * (0.01) ** (elapsed_frac ** 0.5)
        lr = (1.0 / n) * (1.0 / (1.0 + step * 0.001))
        
        conv = fft_convolve(f, f)
        max_c = np.max(conv)
        
        # Smoothed max gradient
        shifted = (conv - max_c) / tau
        exp_vals = np.exp(shifted - np.max(shifted))  # numerical stability
        w = exp_vals / np.sum(exp_vals)
        
        # grad[i] = 2 * sum_j w[j] * f[j] where j+i = k... 
        # Actually: d(conv[k])/da[i] = 2*f[k-i] (if valid)
        # So d(smoothed_max)/da[i] = 2 * sum_k w[k] * f[k-i] (if valid)
        # = 2 * cross_correlation(w, f)[i]
        
        # cross_correlation(w, f)[i] = sum_k w[k] * f[k-i] = (w * reverse_f)[i + n - 1] ... 
        # Let me just compute it directly
        # grad[i] = 2 * sum_{k: 0<=k-i<n} w[k] * f[k-i]
        # Let j = k-i: grad[i] = 2 * sum_{j: 0<=j<n} w[i+j] * f[j] (where w[i+j] = 0 if i+j >= 2n-1)
        # This is a cross-correlation!
        
        # Use FFT for speed
        size = 1
        while size < 4 * n:
            size <<= 1
        fw = np.fft.rfft(w, size)
        ff = np.fft.rfft(f, size)
        # cross_corr[i] = sum_j f[j] * w[i+j]
        # = sum_j f[j] * w[i+j]
        # This is the cross-correlation of f and w
        # = IFFT(conj(FFT(f)) * FFT(w)) ... 
        # Actually: cross_corr[i] = sum_j f[j] * w[i+j]
        # Let's think of it as: (f * w_flipped)[i + n - 1] where w_flipped[k] = w[2n-2-k]
        # Hmm, let me just use the direct FFT approach
        # sum_j f[j] * w[i+j] = sum_j f[j] * w[i+j]
        # = (f ⋆ w)[i] where ⋆ denotes cross-correlation
        # In FFT: cross_corr = IFFT(conj(FFT(f)) * FFT(w))
        # But we need to be careful about the definition
        
        # Actually: let's define h[i] = sum_j f[j] * w[i+j] for i = 0, ..., n-1
        # This is the same as: h = convolution of f with reversed w, shifted
        # h[i] = sum_j f[j] * w[i+j]
        # Let v[k] = w[n-1-k] (reversed w)... no that doesn't work directly.
        
        # Simpler: just use the fact that this is a correlation
        # h[i] = sum_j f[j] * w[i+j] = (f ⋆ w)[i] (cross-correlation)
        # In FFT: FFT(f ⋆ w) = conj(FFT(f)) * FFT(w)
        # So h = IFFT(conj(FFT(f)) * FFT(w))
        
        corr = np.fft.irfft(np.conj(ff) * fw, size)[:n]
        grad = 2.0 * corr
        
        # Step
        f_new = project_simplex(f - lr * grad)
        f = f_new
        step += 1
        
        if step % 100 == 0:
            sc = score_fast(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
            if verbose and step % 1000 == 0:
                print(f"      SGD step {step}: score={sc:.6f}, best={best_score:.6f}")
    
    return best_f, best_score
```

```python
"""
LP-based alternating minimization with subgradient polish for minimizing
peak linear autocorrelation of non-negative sequences. The core idea:
fix f, solve LP to find g maximizing sum(g) subject to f*g <= peak(f*f),
then rescale and iterate. Combined with smoothed-gradient descent for polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = f[:end_k - start_k]
    return A


def solve_lp_max_sum(f, M):
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.status != 0:
            return None
        return res.x
    except Exception:
        return None


def lp_alternating(f_init, time_limit, t_mix=0.5, m_scale=1.0, verbose=False):
    f = f_init.copy().astype(np.float64)
    f = np.clip(f, 0, None)
    s = np.sum(f)
    if s < 1e-10:
        return f, np.inf
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    stagnation = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if it > 5000:
            break

        conv_f = fft_convolve(f, f)
        M = np.max(conv_f) * m_scale

        g = solve_lp_max_sum(f, M)
        if g is None:
            break

        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break

        g = g / g_sum

        f_new = (1.0 - t_mix) * f + t_mix * g
        f_new = np.clip(f_new, 0, None)
        fs = np.sum(f_new)
        if fs < 1e-10:
            break
        f_new = f_new / fs

        new_score = score_fast(f_new)
        f = f_new
        it += 1

        if new_score < best_score:
            best_score = new_score
            best_f = f.copy()
            stagnation = 0
        else:
            stagnation += 1

        if verbose and it % 20 == 0:
            print(f"        LP iter {it}: score={new_score:.6f}, best={best_score:.6f}, M={M:.4e}")

        if stagnation > 200:
            break

    return best_f, best_score


def smoothed_max_grad(f, tau):
    n = len(f)
    conv = fft_convolve(f, f)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    shifted = shifted - np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    value = max_c + tau * np.log(np.sum(exp_vals) * np.exp(np.max(shifted) - shifted[0] + shifted[0]) / np.exp(shifted[0]))
    value = max_c + tau * (np.log(np.sum(exp_vals)) + np.max(shifted))

    size = 1
    while size < 4 * n:
        size <<= 1
    fw = np.fft.rfft(w, size)
    ff = np.fft.rfft(f, size)
    corr = np.fft.irfft(np.conj(ff) * fw, size)[:n]
    grad = 2.0 * corr
    return value, grad, conv


def gradient_polish(f_init, time_limit, verbose=False):
    f = f_init.copy().astype(np.float64)
    f = np.clip(f, 0, None)
    s = np.sum(f)
    if s < 1e-10:
        return f, np.inf
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    step = 0

    tau_start = 5.0 / n
    tau_end = 0.05 / n
    lr_start = 2.0 / n

    f_prev = f.copy()
    t_mom = 1.0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)
        lr = lr_start / (1.0 + 0.01 * step)

        t_prev = t_mom
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        beta = (t_prev - 1.0) / t_mom

        y = f + beta * (f - f_prev)
        y = project_simplex(y)

        value, grad, conv = smoothed_max_grad(y, tau)

        candidate = y - lr * grad
        candidate = project_simplex(candidate)

        f_prev = f
        f = candidate
        step += 1

        if step % 50 == 0:
            sc = score_fast(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
            if verbose and step % 500 == 0:
                print(f"        GD step {step}: score={sc:.6f}, best={best_score:.6f}, tau={tau:.2e}")

    return best_f, best_score


def coordinate_polish(f_init, time_limit, rng, verbose=False):

                candidate[add_idx] += total_removed / len(add_idx)
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate /= cs

        new_score = score_fast(candidate)
        if new_score < best_score:
            best_score = new_score
            best_f = candidate.copy()

        else:

        if verbose and patience % 500 == 0:

        t = np.linspace(0, 1, n)
        a = 1.0 - 0.7 * (2 * t - 1) ** 2
    elif init_type == 'sinc_clipped':
        t = np.linspace(-4, 4, n)
        a = np.abs(np.sinc(t / np.pi))
        a = np.clip(a, 0.01, None)
    elif init_type == 'cosine':
        t = np.linspace(0, np.pi, n)
        a = 0.3 + 0.7 * np.cos(t) ** 2
    elif init_type == 'previous':
        if ref_seq is not None:
            x_old = np.linspace(0, 1, len(ref_seq))
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, ref_seq)
        else:
            a = np.ones(n)
    elif init_type == 'step':
        a = np.ones(n)
        a[n // 3:2 * n // 3] *= 1.5
    else:
        a = np.ones(n)

    a = a + 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_global = time.monotonic()
    time_budget = max(30.0, float(budget_s) - 60.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.001, 1000.0)
        s0 = score_fast(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s0:.6f}")
        if s0 < best_score:
            best_score = s0
            best_seq = start_seq / np.sum(start_seq)

    init_types = ['uniform', 'gaussian', 'quadratic', 'sinc_clipped',
                  'cosine', 'random', 'step', 'previous']

    # Phase 1: LP-based alternating (50% of time)
    print("\n=== Phase 1: LP alternating minimization ===")
    phase1_time = time_budget * 0.50
    phase1_deadline = t_global + phase1_time

    lengths = [256, 384, 512, 768, 1024]
    configs = []
    for n in lengths:
        for itype in init_types[:6]:
            configs.append((n, itype, 0.5))
    for n in [512, 768]:
        configs.append((n, 'previous', 0.5))
        configs.append((n, 'gaussian', 0.7))
        configs.append((n, 'quadratic', 0.6))
        configs.append((n, 'uniform', 0.4))

    rng.shuffle(configs)
    configs.sort(key=lambda x: (0 if x[0] in [384, 512, 768] else 1, rng.random()))

    t_per_config = min(25.0, phase1_time / max(1, len(configs)))
    print(f"  Configs: {len(configs)}, time each: {t_per_config:.1f}s")

    for n, itype, t_mix in configs:
        if time.monotonic() > phase1_deadline - 20:
            break
        if time.monotonic() > deadline - 120:
            break

        t_alloc = min(t_per_config, phase1_deadline - time.monotonic() - 5)
        if t_alloc < 4:
            break

        a0 = make_init(n, itype, rng, ref_seq=best_seq)

        f_best, sc = lp_alternating(a0, t_alloc, t_mix=t_mix)
        f_scaled = f_best * np.sqrt(2.0 * n)
        actual_sc = score_fast(f_scaled)

        print(f"  n={n}, init='{itype}', t_mix={t_mix}: score={actual_sc:.6f}")

        if actual_sc < best_score:
            best_score = actual_sc
            best_seq = f_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 2: Gradient polish (25%)
    print(f"\n=== Phase 2: Gradient polish ===")
    print(f"  Best before: {best_score:.6f}")
    phase2_time = time_budget * 0.25
    phase2_deadline = t_global + phase1_time + phase2_time

    if best_seq is not None and time.monotonic() < phase2_deadline - 20:
        n_best = len(best_seq)
        t_alloc = min(phase2_time * 0.5, phase2_deadline - time.monotonic() - 10)
        if t_alloc > 5:
            f_best, sc = gradient_polish(best_seq, t_alloc)
            f_scaled = f_best * np.sqrt(2.0 * n_best)
            actual_sc = score_fast(f_scaled)
            print(f"  After GD polish: score={actual_sc:.6f}")
            if actual_sc < best_score:
                best_score = actual_sc
                best_seq = f_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

        # LP polish on gradient result
        if time.monotonic() < phase2_deadline - 10:
            t_alloc2 = min(phase2_time * 0.4, phase2_deadline - time.monotonic() - 5)
            if t_alloc2 > 5:
                f_best2, sc2 = lp_alternating(best_seq, t_alloc2, t_mix=0.3)
                f_scaled2 = f_best2 * np.sqrt(2.0 * n_best)
                actual_sc2 = score_fast(f_scaled2)
                print(f"  After LP polish: score={actual_sc2:.6f}")
                if actual_sc2 < best_score:
                    best_score = actual_sc2
                    best_seq = f_best2.copy()
                    print(f"    *** New best: {best_score:.6f} ***")

    # Phase 3: Coordinate descent + exploration (25%)
    print(f"\n=== Phase 3: Coordinate descent + exploration ===")
    phase3_deadline = deadline

    if best_seq is not None and time.monotonic() < phase3_deadline - 30:
        t_rem = phase3_deadline - time.monotonic() - 30

        # Coordinate descent on best
        t_cd = min(t_rem * 0.4, 60.0)
        if t_cd > 5:
            f_best, sc = coordinate_polish(best_seq, t_cd, rng)
            f_scaled = f_best * np.sqrt(2.0 * len(f_best))
            actual_sc = score_fast(f_scaled)
            print(f"  After CD: score={actual_sc:.6f}")
            if actual_sc < best_score:
                best_score = actual_sc
                best_seq = f_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

        # More LP runs with perturbations
        if time.monotonic() < phase3_deadline - 30:
            t_rem2 = phase3_deadline - time.monotonic() - 30
            n_trials = max(2, int(t_rem2 // 20))
            t_per = t_rem2 / n_trials

            for trial in range(n_trials):
                if time.monotonic() > phase3_deadline - 30:
                    break
                t_alloc = min(t_per, phase3_deadline - time.monotonic() - 10)
                if t_alloc < 5:
                    break

                a_pert = best_seq * (1.0 + 0.15 * rng.standard_normal(len(best_seq)))
                a_pert = np.clip(a_pert, 0.001, None)
                a_pert = a_pert / np.sum(a_pert)

                f_best, sc = lp_alternating(a_pert, t_alloc, t_mix=0.5)
                f_scaled = f_best * np.sqrt(2.0 * len(f_best))
                actual_sc = score_fast(f_scaled)
                print(f"  trial {trial}: score={actual_sc:.6f}")
                if actual_sc < best_score:
                    best_score = actual_sc
                    best_seq = f_best.copy()
                    print(f"    *** New best: {best_score:.6f} ***")

        # Final gradient polish
        if time.monotonic() < phase3_deadline - 20:
            t_final = min(30.0, phase3_deadline - time.monotonic() - 20)
            if t_final > 5:
                f_best, sc = gradient_polish(best_seq, t_final)
                f_scaled = f_best * np.sqrt(2.0 * len(f_best))
                actual_sc = score_fast(f_scaled)
                print(f"  Final GD: score={actual_sc:.6f}")
                if actual_sc < best_score:
                    best_score = actual_sc
                    best_seq = f_best.copy()
                    print(f"    *** New best: {best_score:.6f} ***")

    # Also try a few new lengths near the end
    if time.monotonic() < phase3_deadline - 40 and best_seq is not None:
        for n_try in [448, 640, 896]:
            if time.monotonic() > phase3_deadline - 40:
                break
            t_alloc = min(40.0, phase3_deadline - time.monotonic() - 30)
            if t_alloc < 10:
                break
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a0 = np.interp(x_new, x_old, best_seq * (1 + 0.05 * rng.standard_normal(len(best_seq))))
            a0 = np.clip(a0, 0.001, None)
            a0 = a0 / np.sum(a0)
            f_best, sc = lp_alternating(a0, t_alloc, t_mix=0.5)
            f_scaled = f_best * np.sqrt(2.0 * n_try)
            actual_sc = score_fast(f_scaled)
            print(f"  n_try={n_try}: score={actual_sc:.6f}")
            if actual_sc < best_score:
                best_score = actual_sc
                best_seq = f_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        n_final = len(best_seq)
        f_final = best_seq * np.sqrt(2.0 * n_final)
        f_final = np.clip(f_final, 0.0, 1000.0)
        final_list = [float(x) for x in f_final]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
