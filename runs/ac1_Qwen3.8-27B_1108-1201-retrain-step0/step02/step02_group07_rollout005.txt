def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

    c = np.zeros(2 * n - 1)
    d = np.zeros(2 * n - 1)

            else:
                c[k] += 2.0 * a[i]

For two peaks k1 and k2:

This gives: c1*delta + d1*delta^2 - c2*delta - d2*delta^2 = conv[k2] - conv[k1]
(c1-c2)*delta + (d1-d2)*delta^2 = conv[k2] - conv[k1]

If d1 != d2: delta^2 * (d1-d2) + delta*(c1-c2) - (conv[k2]-conv[k1]) = 0
This is a quadratic in delta, easily solvable.

If d1 == d2: delta = (conv[k2]-conv[k1]) / (c1-c2) (if c1 != c2)

                    else:
                        c1, d1 = 0.0, 0.0

                    if i <= k2 <= i + n - 1:
                        c2 = 2.0 * a[k2 - i]
                        d2 = 1.0 if k2 == 2 * i else 0.0
                    else:
                        c2, d2 = 0.0, 0.0

                    A_coeff = d1 - d2
                    B_coeff = c1 - c2
                    C_coeff = conv[k2] - conv[k1]

                    if abs(A_coeff) > 1e-15:
                        disc = B_coeff * B_coeff - 4.0 * A_coeff * C_coeff
                        if disc >= 0:
                            sq = np.sqrt(disc)
                            for delta in [(-B_coeff + sq) / (2.0 * A_coeff),
                                          (-B_coeff - sq) / (2.0 * A_coeff)]:
                                x = ai + delta
                                if 0.0 <= x <= ai * 10 + 5.0:
                                    candidates.append(x)
                    elif abs(B_coeff) > 1e-15:
                        delta = -C_coeff / B_coeff
                        x = ai + delta
                        if 0.0 <= x <= ai * 10 + 5.0:
                            candidates.append(x)

            # Add grid points
            if ai > 1e-10:
                candidates.append(ai * 0.5)
                candidates.append(ai * 0.8)
                candidates.append(ai * 1.25)
                candidates.append(ai * 1.5)
            candidates.append(cur_sum / n)

            # Evaluate candidates
            cur_obj = 2.0 * n * conv.max() / (cur_sum * cur_sum)
            best_x = ai
            best_local = cur_obj

            # Precompute outer max
            outer_max = 0.0
            if i > 0:
                outer_max = conv[:i].max()
            outer_end = i + n
            if outer_end < 2 * n - 1:
                om = conv[outer_end:].max()
                if om > outer_max:
                    outer_max = om

            for x in candidates:
                x = max(0.0, x)
                if x == ai:
                    continue
                delta = x - ai
                new_sum = S_base + x
                if new_sum < 1e-10:
                    continue
                new_inner = conv[i:i + n] + c_seg * delta + d_seg * delta * delta
                new_max = max(outer_max, new_inner.max())
                obj = 2.0 * n * new_max / (new_sum * new_sum)
                if obj < best_local - 1e-13:
                    best_local = obj
                    best_x = x

            if best_x != ai and best_local < best_score:
                delta = best_x - ai
                conv[i:i + n] += c_seg * delta + d_seg * delta * delta
                conv = np.maximum(conv, 0.0)
                a[i] = best_x
                cur_sum += delta
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved:
            stagnation += 1
            if stagnation >= 5:
                break
        else:
            stagnation = 0

    return normalize(best_a), best_score


def coord_descent_annealed(a, time_limit=30.0, max_passes=100, top_k=12, rng=None):
    """Coordinate descent with occasional worse moves (annealing)."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = normalize(a).copy()
    conv = fast_convolve(a)
    conv = np.maximum(conv, 0.0)
    t0 = time.monotonic()
    cur_sum = a.sum()
    cur_score = 2.0 * n * conv.max() / (cur_sum * cur_sum)
    best_score = cur_score
    best_a = a.copy()
    T = 0.01

    for p in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        frac = (time.monotonic() - t0) / max(time_limit, 1.0)
        T = 0.01 * (1.0 - frac) ** 1.5 + 0.0005
        improved = False
        cur_sum = a.sum()

        for i in range(n):
            ai = a[i]
            S_base = cur_sum - ai
            c_seg = 2.0 * a
            d_seg = np.zeros(n)
            d_seg[i] = 1.0

            conv_max = conv.max()
            if conv_max < 1e-30:
                break
            thresh = conv_max * 0.75
            top_idx = np.where(conv >= thresh)[0]
            if len(top_idx) == 0:
                top_idx = np.array([np.argmax(conv)])
            top_idx = top_idx[:top_k]

            candidates = [0.0, ai]
            for ii in range(len(top_idx)):
                for jj in range(ii + 1, len(top_idx)):
                    k1 = int(top_idx[ii])
                    k2 = int(top_idx[jj])
                    if i <= k1 <= i + n - 1:
                        c1 = 2.0 * a[k1 - i]
                        d1 = 1.0 if k1 == 2 * i else 0.0
                    else:
                        c1, d1 = 0.0, 0.0
                    if i <= k2 <= i + n - 1:
                        c2 = 2.0 * a[k2 - i]
                        d2 = 1.0 if k2 == 2 * i else 0.0
                    else:
                        c2, d2 = 0.0, 0.0
                    A_coeff = d1 - d2
                    B_coeff = c1 - c2
                    C_coeff = conv[k2] - conv[k1]
                    if abs(A_coeff) > 1e-15:
                        disc = B_coeff * B_coeff - 4.0 * A_coeff * C_coeff
                        if disc >= 0:
                            sq = np.sqrt(disc)
                            for delta in [(-B_coeff + sq) / (2.0 * A_coeff),
                                          (-B_coeff - sq) / (2.0 * A_coeff)]:
                                x = ai + delta
                                if 0.0 <= x <= ai * 10 + 5.0:
                                    candidates.append(x)
                    elif abs(B_coeff) > 1e-15:
                        delta = -C_coeff / B_coeff
                        x = ai + delta
                        if 0.0 <= x <= ai * 10 + 5.0:
                            candidates.append(x)
            if ai > 1e-10:
                candidates.append(ai * 0.5)
                candidates.append(ai * 1.5)
            candidates.append(cur_sum / n)

            cur_obj = 2.0 * n * conv.max() / (cur_sum * cur_sum)
            best_x = ai
            best_local = cur_obj

            outer_max = 0.0
            if i > 0:
                outer_max = conv[:i].max()
            outer_end = i + n
            if outer_end < 2 * n - 1:
                om = conv[outer_end:].max()
                if om > outer_max:
                    outer_max = om

            for x in candidates:
                x = max(0.0, x)
                if x == ai:
                    continue
                delta = x - ai
                new_sum = S_base + x
                if new_sum < 1e-10:
                    continue
                new_inner = conv[i:i + n] + c_seg * delta + d_seg * delta * delta
                new_max = max(outer_max, new_inner.max())
                obj = 2.0 * n * new_max / (new_sum * new_sum)
                if obj < best_local - 1e-13:
                    best_local = obj
                    best_x = x

            if best_x != ai:
                delta = best_x - ai
                delta_obj = best_local - cur_score
                accept = False
                if delta_obj < 0:
                    accept = True
                else:
                    if rng.random() < np.exp(-delta_obj / max(T, 1e-15)):
                        accept = True
                if accept:
                    conv[i:i + n] += c_seg * delta + d_seg * delta * delta
                    conv = np.maximum(conv, 0.0)
                    a[i] = best_x
                    cur_sum += delta
                    cur_score = best_local
                    improved = True
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        if not improved:
            break

    return normalize(best_a), best_score


def generate_init(n, rng, parent=None, init_type=0):
    """Generate a diverse initialization."""
    x = np.arange(n, dtype=np.float64)
    if init_type == 0:
        a = np.ones(n)
    elif init_type == 1:
        a = np.exp(-0.5 * ((x - n * 0.5) / (0.2 * n)) ** 2)
    elif init_type == 2:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    elif init_type == 3:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * 2 * x / n))
    elif init_type == 4:
        bi = np.zeros(n)
        w = max(1, n // 5)
        bi[:w] = 1.0
        bi[-w:] = 1.0
        a = bi
    elif init_type == 5:
        a = rng.dirichlet(np.ones(n) * 0.5)
    elif init_type == 6:
        a = rng.dirichlet(np.ones(n) * 2.0)
    elif init_type == 7:
        a = np.abs(np.sin(2.0 * np.pi * 3.0 * x / n)) + 0.1
    elif init_type == 8:
        a = np.abs(np.cos(np.pi * x / n)) + 0.1
    else:
        a = rng.random(n) + 0.01

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    if parent is not None:
        pw = 0.3
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0)
            pp /= pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size),
                           np.maximum(parent, 0.0))
            pp = np.maximum(pp, 0.0)
            ss = pp.sum()
            if ss > 1e-15:
                pp /= ss
            else:
                pp = np.ones(n) / n
        a = (1.0 - pw) * a + pw * pp
        a = normalize(a)

    return a


def structured_mutation(a, rng, strength=0.05):
    """Apply a structured mutation to the sequence."""
    n = len(a)
    a = a.copy()
    x = np.arange(n, dtype=np.float64)
    mut_type = rng.integers(0, 4)
    if mut_type == 0:
        freq = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        a += strength * np.cos(2.0 * np.pi * freq * x / n + phase)
    elif mut_type == 1:
        a += strength * rng.normal(0, 1, n)
    elif mut_type == 2:
        block = n // 4
        start = rng.integers(0, n - block)
        a[start:start + block] *= rng.uniform(0.3, 2.0)
        rest_sum = a.sum() - a[start:start + block].sum()
        if rest_sum > 1e-10:
            a[start:start + block] *= rest_sum / max(a[start:start + block].sum(), 1e-15)
    else:
        idx = rng.choice(n, size=n // 4, replace=False)
        a[idx] *= rng.uniform(0.2, 3.0, len(idx))
    return normalize(a)


def propose_candidate(seed=42, budget_s=2400):
    """
    Main entry point. Uses exact coordinate descent with peak-crossing,
    multi-length exploration, and structured mutations to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)

    parent = get_parent()

    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        if not np.all(np.isfinite(a)):
            return
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d, time=%.0fs" % (score, len(a), time.monotonic() - t_start))

    # Phase 0: Polish parent
    if parent is not None and time.monotonic() < deadline - 60:
        print("Phase 0: Polishing parent (n=%d)..." % len(parent))
        p_budget = min(120.0, (deadline - time.monotonic()) * 0.08)
        a_parent = normalize(parent)
        a_opt, obj = coord_descent(a_parent, time_limit=p_budget, max_passes=100, top_k=15)
        record(a_opt)
        print("  Parent polished: score=%.6f" % (2.0 * len(a_opt) * fast_convolve(a_opt).max() / (a_opt.sum()**2)))

    # Phase 1: Multi-length exploration
    print("Phase 1: Multi-length exploration...")
    lengths = [128, 192, 256, 320, 384, 512, 768, 1024]
    if len(np.ones(1)) < 1:
        pass

    candidates_pool = []
    phase1_budget = min(600.0, (deadline - time.monotonic()) * 0.25)
    per_length = phase1_budget / len(lengths)
    n_inits_per_len = 3

    for n in lengths:
        if time.monotonic() > deadline - 1200:
            break
        alloc = min(per_length, deadline - time.monotonic() - 1200)
        if alloc < 10:
            break
        per_init = alloc / n_inits_per_len
        for init_i in range(n_inits_per_len):
            if time.monotonic() > deadline - 1200:
                break
            t_alloc = min(per_init, deadline - time.monotonic() - 1200)
            if t_alloc < 3:
                break
            a0 = generate_init(n, rng, parent, init_type=init_i % 9)
            a_opt, obj = coord_descent(a0, time_limit=t_alloc, max_passes=80, top_k=10)
            score = safe_evaluate(a_opt)
            if np.isfinite(score):
                candidates_pool.append((score, n, a_opt))
            record(a_opt)

    print("  Phase 1 done. Best: %.6f" % best_score)
    candidates_pool.sort(key=lambda x: x[0])

    # Phase 2: Deep optimization on top candidates
    print("Phase 2: Deep optimization...")
    top_n = min(8, len(candidates_pool))
    phase2_budget = min(1000.0, (deadline - time.monotonic()) * 0.40)
    per_cand = phase2_budget / max(1, top_n)

    for ci in range(top_n):
        if time.monotonic() > deadline - 500:
            break
        score, n, a0 = candidates_pool[ci]
        alloc = min(per_cand, deadline - time.monotonic() - 500)
        if alloc < 10:
            break
        print("  [%d/%d] n=%d, score=%.6f, budget=%.0fs" % (ci + 1, top_n, n, score, alloc))

        # Multiple CD passes with different parameters
        sub = alloc / 4.0
        a_cur = a0.copy()
        a_cur, _ = coord_descent(a_cur, time_limit=sub, max_passes=150, top_k=15)
        record(a_cur)
        a_cur, _ = coord_descent(a_cur, time_limit=sub, max_passes=150, top_k=20)
        record(a_cur)

        # Annealed CD
        a_cur, _ = coord_descent_annealed(a_cur, time_limit=sub, max_passes=80, top_k=12, rng=rng)
        record(a_cur)

        # Mutation + CD
        a_mut = structured_mutation(a_cur, rng, strength=0.03)
        a_mut, _ = coord_descent(a_mut, time_limit=sub, max_passes=100, top_k=15)
        record(a_mut)

    print("  Phase 2 done. Best: %.6f" % best_score)

    # Phase 3: Mutation + CD loop from best
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("Phase 3: Mutation + CD loop...")
        phase3_budget = min(500.0, deadline - time.monotonic() - 100)
        n_iters = max(3, int(phase3_budget / 40))
        per_iter = phase3_budget / n_iters

        for it in range(n_iters):
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(per_iter, deadline - time.monotonic() - 100)
            if t_alloc < 5:
                break
            strength = 0.02 * (1.0 + 0.5 * it / n_iters)
            a_mut = structured_mutation(best_sequence, rng, strength=strength)
            a_opt, _ = coord_descent(a_mut, time_limit=t_alloc, max_passes=100, top_k=15)
            record(a_opt)

        print("  Phase 3 done. Best: %.6f" % best_score)

    # Phase 4: Final polish
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_best = normalize(best_sequence)
            a_opt, _ = coord_descent(a_best, time_limit=remaining, max_passes=500, top_k=20)
            record(a_opt)
            print("  Phase 4 done. Best: %.6f" % best_score)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))

    # Final verification
    final_score = safe_evaluate(best_sequence)
    if np.isfinite(final_score):
        print("Verified score: %.6f" % final_score)

    return [float(x) for x in best_sequence]
