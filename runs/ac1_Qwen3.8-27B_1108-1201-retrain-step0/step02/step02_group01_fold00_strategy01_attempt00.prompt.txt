def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def a_from_theta(theta, S):
    """Convert log-parameters to sequence: a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def compute_soft_weights(c, T):
    """Compute soft-max weights proportional to exp(T*c/c_max)."""
    c_max = c.max()
    if c_max < 1e-15:
        return np.ones(len(c)) / len(c)
    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        return np.ones(len(c)) / len(c)
    return exp_scaled / total


def compute_grad_a(w, a):
    """Compute grad_a[m] = 2 * sum_j w[m+j] * a[j] for m=0..n-1 using FFT."""
    n = len(a)
    a_rev = a[::-1]
    # grad[m] = 2 * conv(w, a_rev)[m + n - 1]
    full_conv = fft_convolve(w, a_rev)
    # full_conv has length (2n-1) + n - 1 = 3n-2
    # We need full_conv[n-1 : 2n-1]
    end_idx = min(2 * n - 1, len(full_conv))
    grad = np.zeros(n)
    grad[:] = 2.0 * full_conv[n - 1: end_idx][:n]
    return grad


def softmax_objective_and_grad(theta, n, T, S):
    """Objective (soft-max of self-conv) and gradient w.r.t. theta."""
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()

    # Soft-max: c_max + (1/T)*log(sum_k exp(T*(c_k - c_max)))
    shifted = T * (c - c_max)
    # max of shifted is 0, so log(sum exp(shifted)) = log(1 + sum exp(shifted[shifted<0]))
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T

    # Weights for gradient
    w = compute_soft_weights(c, T)
    grad_a = compute_grad_a(w, a)

    # Chain rule: dL/dtheta[j] = grad_a[j]*a[j] - (a[j]/S)*sum_i grad_a[i]*a[i]
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot

    return obj, grad_theta


def true_objective(a):
    """Compute the true max of self-convolution (for final evaluation)."""
    c = fft_convolve_self(a)

        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)

def coordinate_descent_polish(a, n, S, time_budget, max_sweeps=50):
    """Coordinate descent to polish the sequence."""
    a = a.copy().astype(np.float64)
    n_par = len(a)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n_par):
            if time.monotonic() - start > time_budget:
                break
            # Try to reduce a[i] (if it contributes to the max)
            # Quick check: does decreasing a[i] help?
            original = a[i]
            if original <= 1e-10:
                continue
            # Binary search for optimal a[i]
            lo, hi = 0.0, original * 1.5
            best_local = original
            best_local_obj = best_obj
            for _ in range(12):
                mid = (lo + hi) / 2.0
                a[i] = mid
                # Renormalize
                s = a.sum()
                if s < 1e-10:
                    a[i] = original
                    break
                a_norm = a * (S / s)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_local_obj:
                    best_local_obj = obj_check
                    best_local = mid
                if obj_check < best_obj * 1.01:
                    lo = mid
                else:
                    hi = mid
            a[i] = best_local
            if best_local < original - 1e-8 or best_local > original + 1e-8:
                improved = True
            # Renormalize
            s = a.sum()
            if s < 1e-10:
                a = np.full(n_par, S / n_par)
            else:
                a = a * (S / s)
            c = fft_convolve_self(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()

        if not improved:
            break

    return best_a, best_obj


def solve_lp_direction(f, M, n, top_k_constraints=50):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0.
    Returns the optimal g (normalized to sum=1) and the LP objective value."""
    from scipy.optimize import linprog

    c = fft_convolve_self(f)
    # Get top-k lags
    if top_k_constraints < len(c):
        active_lags = np.argsort(c)[::-1][:top_k_constraints]
    else:
        active_lags = np.arange(len(c))

    # Build constraint matrix A: A[k,j] = f[k-j] if 0<=k-j<n
    A_rows = []
    for k in active_lags:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            row[j] = f[k - j]
            row[k - j] += f[j]  # Wait, this double-counts. Let me redo.
        A_rows.append(row)
    # Actually, (f*g)[k] = sum_j f[k-j]*g[j] where 0<=k-j<n
    # So A[k,j] = f[k-j] if 0<=k-j<n, else 0
    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[idx, j] = f[k - j]

    # Maximize sum(g) = -minimize -sum(g)
    c_obj = -np.ones(n)
    b = np.full(len(active_lags), M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                         method='highs', options={'time_limit': 10, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return None, None


def lbfgs_fun(theta, n, T, S, state_dict):
    """Objective function for L-BFGS-B. Top-level, no closures."""
    elapsed = time.monotonic() - state_dict['start']
    if elapsed > state_dict['time_budget']:
        return state_dict['best_obj'], np.zeros(len(theta))
    obj, grad = softmax_objective_and_grad(theta, n, T, S)
    # Also track true max
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    true_max = c.max()
    if true_max < state_dict['best_obj']:
        state_dict['best_obj'] = true_max
        state_dict['best_a'] = a.copy()
    return obj, grad


def run_lbfgs(theta_init, n, T, S, time_budget, max_iter=500):
    """Run L-BFGS-B on the soft-max objective with log-parameters."""
    state_dict = {
        'n': n, 'T': T, 'S': S,
        'start': time.monotonic(), 'time_budget': time_budget,
        'best_obj': np.inf, 'best_a': None
    }
    # Initialize best
    a0 = a_from_theta(theta_init, S)
    c0 = fft_convolve_self(a0)
    state_dict['best_obj'] = c0.max()
    state_dict['best_a'] = a0.copy()

    try:
        result = minimize(
            lbfgs_fun, theta_init,
            args=(n, T, S, state_dict),
            jac=True, method='L-BFGS-B',
            options={'maxiter': max_iter, 'maxfun': 2000, 'maxls': 50,
                     'gtol': 1e-8}
        )
    except Exception as e:
        print("L-BFGS failed: %s" % str(e))

    return state_dict['best_a'], state_dict['best_obj']


def get_initialization(name, n, rng, S):
    """Generate various initializations."""
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "gaussian":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n / 2) / (n / 6)) ** 2)
        return a * (S / a.sum())
    elif name == "cosine":
        i = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * i / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "anti_peak":
        i = np.arange(n)
        a = 1.0 - 0.4 * np.cos(2 * np.pi * (n - 1) * i / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "raised":
        i = np.arange(n)
        t = i / (n - 1)
        a = np.sin(np.pi * t) ** 2
        return a * (S / a.sum())
    elif name == "two_humps":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n * 0.25) / (n / 10)) ** 2) + \
            np.exp(-0.5 * ((i - n * 0.75) / (n / 10)) ** 2)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)
