"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.
Combines segment-based structure search, FISTA with log-sum-exp,
power-sum gradient descent, and Nesterov-accelerated top-k polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_from_weights(a, w):
    """
    Compute gradient of sum_k w[k]*c[k] w.r.t. a.
    grad[i] = 2 * sum_k w[k] * a[k-i] (with bounds checking)
    = 2 * (w * a)[i] via cross-correlation
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_lse(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    return fft_grad_from_weights(a, w)


def fft_grad_powersum(a, c, p):
    """Gradient of (sum c[k]^p)^(1/p) w.r.t. a (up to a constant factor)."""
    n = len(a)
    c_safe = np.maximum(c, 1e-30)
    # w[k] ∝ c[k]^(p-1)
    # For numerical stability, normalize by max
    c_max = c_safe.max()
    if c_max < 1e-30:
        return np.zeros(n)
    w = (c_safe / c_max) ** (p - 1)
    w_total = w.sum()
    if w_total < 1e-30:
        return np.zeros(n)
    w = w / w_total
    return fft_grad_from_weights(a, w)


def fft_grad_toppk(a, c, top_k):
    """Gradient using top-k active lags."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    return fft_grad_from_weights(a, w_full)


def run_fista(a_init, n, S, max_time, rng, beta_init=2.0, beta_max=300.0, 
              use_powersum=False, p_init=10.0, p_max=10000.0):
    """FISTA with adaptive log-sum-exp or power-sum smoothing."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        if use_powersum:
            p = min(p_max, p_init * (1.0 + 0.005 * it))
        else:
            p = min(beta_max, beta_init * (1.0 + 0.002 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        if use_powersum:
            grad = fft_grad_powersum(y, c, p)
        else:
            grad = fft_grad_lse(y, c, p)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (p * 1.0)
        step = max(step, 1e-8)
        x_new = project_simplex(y - step * grad, S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 3000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score

def run_mirror(a_init, n, S, max_time, rng, eta_init=0.2, top_k=10):
    """Mirror descent (multiplicative weights)."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        eta = eta_init / (1.0 + 0.001 * it)
        grad = fft_grad_toppk(a, c, top_k)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_new * (S / s_new)

    return best_local, best_score


def seg_convolution_max(heights, sizes, n):
    """
    Compute max of self-convolution for a piecewise-constant sequence.
    heights: array of segment heights (k elements)
    sizes: array of segment sizes (k elements, summing to n)
    Returns the max value of the full self-convolution.
    """
    k = len(heights)
    # Build the full sequence
    a = np.zeros(n)
    pos = 0
    for i in range(k):
        a[pos:pos+sizes[i]] = heights[i]
        pos += sizes[i]
    
    c = fft_convolve_self(a)
    return c.max(), a


def optimize_segments(k, n, max_time, rng):
    """
    Optimize segment heights for a piecewise-constant sequence with k segments.
    Uses Nelder-Mead on the height parameters.
    """
    S = np.sqrt(2.0 * n)
    seg_size = n // k
    remainder = n - seg_size * k
    sizes = np.full(k, seg_size, dtype=int)
    for i in range(remainder):
        sizes[i] += 1
    
    def objective(h):
        h = np.maximum(h, 0.0)
        total = h.sum()
        if total < 1e-10:
            return 1e10
        # Scale to have sum = S
        h_scaled = h * (S / total)
        max_conv, _ = seg_convolution_max(h_scaled, sizes, n)
        return max_conv
    
    # Initial guess: random positive heights
    h0 = rng.random(k) + 0.1
    # Also try structured inits
    results = []
    
    for init_type in ['random', 'ramp', 'bimodal', 'alternating']:
        if init_type == 'random':
            h0 = rng.random(k) + 0.1
        elif init_type == 'ramp':
            h0 = np.linspace(0.1, 2.0, k)
        elif init_type == 'bimodal':
            mid = k // 2
            h0 = np.ones(k) * 0.5
            h0[:mid] = 1.5
            h0[mid:] = 0.3
        elif init_type == 'alternating':
            h0 = np.ones(k) * 0.5
            h0[::2] = 1.5
            h0[1::2] = 0.3
        
        try:
            res = minimize(objective, h0, method='Nelder-Mead',
                          options={'maxiter': 500, 'xatol': 1e-6, 'fatol': 1e-8})
            if res.success or res.fun < 1e10:
                results.append((res.fun, res.x))
        except Exception:
            pass
    
    if not results:
        return None, None
    
    results.sort(key=lambda x: x[0])
    return results[0]


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lbfgs_polish_unconstrained(a_init, n, S, max_time, rng, beta=100.0):
    """Polish using L-BFGS on softmax-parameterized objective."""
    a0 = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    a0 = np.maximum(a0, 1e-10)
    a0 = a0 / a0.sum() * S
    
    # v = log(a) as initial parameterization
    v0 = np.log(a0)
    
    start = time.monotonic()
    best_a = a0.copy()
    best_score = np.inf
    it_count = [0]
    
    def softmax_scale(v):
        # Numerically stable softmax scaled to sum S
        v_shift = v - v.max()
        exp_v = np.exp(v_shift)
        return exp_v / exp_v.sum() * S
    
    def objective(v):
        a = softmax_scale(v)
        c = fft_convolve_self(a)
        c_max = c.max()
        return c_max + np.log(np.exp(beta * (c - c_max)).sum()) / beta
    
    def gradient(v):
        a = softmax_scale(v)
        c = fft_convolve_self(a)
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        w = exp_bc / exp_bc.sum()
        grad_c = fft_grad_from_weights(a, w)
        # Chain rule: d/dv = d/da * da/dv
        # da/dv[i] = a[i] * (delta_{ij} - a[j]/sum(a)) = a[i]*(delta_ij - a[j]/S)
        # So grad_v[i] = sum_j grad_c[j] * a[j]*(delta_ij - a[i]/S)
        # = a[i] * grad_c[i] - a[i] * (sum_j grad_c[j] * a[j] / S)
        # Wait, that's not right. Let me redo.
        # a = softmax(v) * S
        # da/dv[i] = a[i] * (delta_ij - a[j]/S) for each component i
        # Actually: a_i = S * exp(v_i) / sum_k exp(v_k)
        # da_i/dv_j = a_i * (delta_ij - a_j/S)
        # So grad_v_j = sum_i grad_c_i * da_i/dv_j = sum_i grad_c_i * a_i * (delta_ij - a_j/S)
        # = a_j * grad_c_j - a_j * (sum_i grad_c_i * a_i/S)
        # = a_j * (grad_c_j - sum_i grad_c_i * a_i / S)
        dot = np.dot(grad_c, a) / S
        grad_v = a * (grad_c - dot)
        return grad_v
    
    def cb(vk):
        it_count[0] += 1
        if time.monotonic() - start > max_time:
            return True
        if it_count[0] % 50 == 0:

            score = c.max()
            if score < best_score:
                best_score = score
                best_a[:] = a
        return False

                      callback=cb, options={'maxiter': 5000, 'maxfun': 20000,
                                             'ftol': 1e-14, 'gtol': 1e-10})
    except Exception as e:
        print("    L-BFGS error: %s" % str(e))
    
    return best_a, best_score

Minimize 2*n*max_conv / sum^2 over non-negative sequences.
Combines segment-based structure search, FISTA with log-sum-exp smoothing,
L-BFGS polish with softmax parameterization, Nesterov top-k subgradient,
and mirror descent. Uses FFT-accelerated convolution for O(n log n) gradients.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]
