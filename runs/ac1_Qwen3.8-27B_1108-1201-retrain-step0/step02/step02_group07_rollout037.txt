"""
LP-based iterative optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the convexification trick: at each step, fix a and solve an LP to find
g that satisfies the same peak constraint but has larger sum. Blend and repeat.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_normalized(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def solve_lp(a, M, solve_time_limit=30.0):
    """
    Solve: max sum(g) s.t. g >= 0, conv(a,g) <= M for all k, g[i] <= 1.
    Returns the optimal g (rescaled to sum=1) or None.
    """
    n = len(a)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix A: A[k, m] = a[m] if 0 <= k-m < n
    # A has shape (2n-1, n)
    # c <= A @ g where c = M * ones(2n-1)
    # linprog minimizes c^T x s.t. A_ub @ x <= b_ub, A_eq @ x = b_eq, bounds
    # So we minimize -sum(g) s.t. A @ g <= M, g >= 0, g <= 1
    
    # Build A matrix (sparse would be better but n=256 is small)
    A_mat = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        A_mat[k, m_start:m_end+1] = a[m_start:m_end+1]
    
    c_obj = -np.ones(n)  # minimize -sum = maximize sum
    b_ub = np.full(num_constraints, M)
    bounds = [(0, 1)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            s = g.sum()
            if s > 1e-12:
                g = g / s
            return g
        else:
            return None
    except Exception:
        return None

def lp_iteration(a0, M_init, num_iters=200, t_start=0.5, t_decay=0.97, time_limit=300.0):
    """
    Iterative LP method.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    M = float(fast_convolve(a).max())
    best_M = M
    best_a = a.copy()
    t = t_start
    t0 = time.monotonic()
    
    for it in range(num_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        g = solve_lp(a, M)
        if g is None:
            break
        
        # Blend
        a = (1 - t) * a + t * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            break
        a = a / s
        
        M = float(fast_convolve(a).max())
        if M < best_M:
            best_M = M
            best_a = a.copy()
        
        t *= t_decay
        if t < 0.001:
            break
    
    return best_a, best_M

def mirror_descent(a0, time_limit, t_init=50.0, eta_init=0.3):
    """Multiplicative mirror descent on the simplex."""
    a = np.maximum(a0, 0.0).copy()

        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)

        if s < 1e-12:

        else:

        conv = fast_convolve(a)

        if cur_obj < best_obj - 1e-9:

            patience = 0

def compute_subgradient(conv, a, t, thresh_frac=0.002):
    """Smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = t * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    thresh = w.max() * thresh_frac
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            # sg[j] += w[k] * a[k-j] for j in [js, je]
            # Note: conv[k] = sum_j a[j]*a[k-j], d(conv[k])/da[j] = a[k-j]
            # (assuming k-j is valid, which it is by construction of js, je)
            sg[js:je+1] += w[k] * a[k-je:k-js+1][::-1]
    return sg

def project_simplex(v):
    """Project vector onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def pgd_optimize(a0, time_limit, t_init=30.0, lr_init=0.08):
    """Projected gradient descent."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    lr = lr_init
    patience = 0
    max_pat = 40
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient(conv, a, t)
        a_new = a - lr * sg
        a_new = project_simplex(a_new)
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            lr *= 0.7
            t *= 1.2
            if lr < 0.002:
                lr = lr_init * 0.5
                t = t_init * 2.0
    return best_a, best_obj

def anneal_informed(a0, time_limit, rng):
    """Informed simulated annealing."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.015 * (1.0 - frac) ** 2 + 0.0005
        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.75
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je+1]
        seg_b = a[k-je:k-js+1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        target_j = int(rng.integers(n))
        delta = a[top_j] * rng.uniform(0.01, 0.06)
        if delta < 1e-9:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
    return best_a, best_obj

def get_parent():
    """Try to access parent sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:

    return None

def safe_evaluate(a):
    """Safely evaluate a sequence."""
    try:
        a_clean = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a_clean = np.minimum(a_clean, 1000.0)
        if a_clean.sum() < 0.01:
            a_clean = a_clean * (1.0 / max(a_clean.sum(), 1e-15))
        if not np.all(np.isfinite(a_clean)):
            return float('inf')
        score = float(evaluate_sequence(a_clean.tolist()))
        return score
    except Exception:
        return float('inf')

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian centered
    g = np.exp(-0.5 * ((x - n * 0.5) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss_center'] = g
    
    # Gaussian at edge
    g = np.exp(-0.5 * (x / (0.15 * n)) ** 2)
    g /= g.sum()
    inits['gauss_edge'] = g
    
    # Sine windows
    for freq in [1, 2, 3]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        s = np.maximum(s, 0)
        if s.sum() > 1e-10:
            s /= s.sum()
            inits['sin_%d' % freq] = s
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Bimodal (edges)
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    if bi.sum() > 0:
        bi /= bi.sum()
        inits['bimodal'] = bi
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['rand_%s' % alpha] = r
    
    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0).copy()
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            # Interpolate
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp
    
    return inits

"""
LP-based iterative optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses convexification trick: fix current a, solve LP for g maximizing sum(g)
subject to conv(a,g) <= M and g>=0. Blend and iterate. Combined with mirror
descent polishing and simulated annealing for final refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_normalized(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_lp_matrix(a):
    """Build the (2n-1) x n matrix A where conv(a,g)[k] = sum_i A[k,i]*g[i]."""
    n = len(a)
    num_constraints = 2 * n - 1
    A_mat = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        A_mat[k, i_start:i_end + 1] = a[k - i_end:k - i_start + 1][::-1]
    return A_mat


def solve_lp(a, M):
    """
    Solve: max sum(g) s.t. g >= 0, conv(a,g)[k] <= M for all k.
    Returns g normalized to sum=1, or None.
    """
    n = len(a)
    num_constraints = 2 * n - 1
    A_mat = build_lp_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0.0, 1.0)] * n
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            s = g.sum()
            if s > 1e-12:
                return g / s
            return None
        else:
            return None
    except Exception:
        return None


def lp_iteration(a0, time_limit, t_start=0.5, t_decay=0.98, max_iters=500):
    """
    Iterative LP method: alternate between LP solve and blending.
    Returns (best_a, best_M) where best_M = max(conv(a,a)).
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    M = float(fast_convolve(a).max())
    best_M = M
    best_a = a.copy()
    t = t_start
    t0 = time.monotonic()
    prev_lp_time = 0.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        g = solve_lp(a, M)
        if g is None:
            break

        # Blend
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            break
        a = a / s

        M = float(fast_convolve(a).max())
        if M < best_M:
            best_M = M
            best_a = a.copy()

        t *= t_decay
        if t < 0.005:
            break

    return best_a, best_M


def compute_subgradient(conv, a, t, thresh_frac=0.002):
    """Smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = t * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    thresh = w.max() * thresh_frac
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    return sg


def part_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.7

        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]

        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))

        delta = a[top_j] * rng.uniform(0.008, 0.05)

        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()

        if delta_obj < 0:
            cur_obj = new_obj

                best_a = a.copy()

    time_per_deep = max(8.0, phase2_budget / max(1, n_deep))

    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 700:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 700)
        if t_alloc < 6.0:
            break
        print("  [%d/%d] n=%d, %s, budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))

        a_cur = np.maximum(a0, 0.0).copy()
        a_cur /= a_cur.sum()

        # Step A: LP iteration (primary)
        M_init = float(fast_convolve(a_cur).max())
        lp_time = min(t_alloc * 0.45, 350.0)
        a_lp, M_lp = lp_iteration(a_cur, time_limit=lp_time, t_start=0.4, t_decay=0.985)
        record(a_lp)
        a_cur = a_lp

        # Step B: Mirror descent polish
        if time.monotonic() < deadline - 700:
            rem = min(t_alloc * 0.2, deadline - time.monotonic() - 700)
            if rem > 3:
                a_md, _ = mirror_descent(a_cur, time_limit=rem, t_init=100.0, eta_init=0.15)
                record(a_md)
                if obj_normalized(a_md) < obj_normalized(a_cur):
                    a_cur = a_md

        # Step C: Second LP round (tighter)
        if time.monotonic() < deadline - 700:
            rem = min(t_alloc * 0.2, deadline - time.monotonic() - 700)
            if rem > 5:
                M_cur = float(fast_convolve(a_cur).max())
                a_lp2, M_lp2 = lp_iteration(a_cur, time_limit=rem, t_start=0.25, t_decay=0.99)
                record(a_lp2)
                if M_lp2 < M_cur:
                    a_cur = a_lp2

        # Step D: PGD fine-tune
        if time.monotonic() < deadline - 700:
            rem = min(t_alloc * 0.15, deadline - time.monotonic() - 700)
            if rem > 3:
                a_pgd, _ = pgd_optimize(a_cur, time_limit=rem, t_init=80.0, lr_init=0.04)
                record(a_pgd)

    print("  Phase 2 complete. Best: %.6f" % best_score)

    # Phase 3: Annealing on best
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("Phase 3: Annealing...")
        phase3_budget = min(450.0, deadline - time.monotonic() - 50)
        n_anneal = 4
        anneal_per = phase3_budget / n_anneal

        for i in range(n_anneal):
            if time.monotonic() > deadline - 50:
                break
            a_start = np.maximum(best_sequence, 0.0).copy()
            a_start /= a_start.sum()
            rng_a = np.random.default_rng(seed + 7000 + i * 137)
            a_opt, _ = anneal_informed(a_start, time_limit=anneal_per, rng=rng_a)
            record(a_opt)

        print("  Annealing complete. Best: %.6f" % best_score)

    # Phase 4: Final LP + polish
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 10
        a_best = np.maximum(best_sequence, 0.0).copy()
        a_best /= a_best.sum()
        M_best = float(fast_convolve(a_best).max())

        # Final LP
        if remaining > 40:
            t_lp = remaining * 0.45
            a_final, M_final = lp_iteration(a_best, time_limit=t_lp,
                                            t_start=0.15, t_decay=0.99, max_iters=300)
            record(a_final)
            if M_final < M_best:
                a_best = a_final
                M_best = M_final

        # Mirror descent
        if time.monotonic() < deadline - 5:
            rem = deadline - time.monotonic() - 5
            if rem > 5:
                a_md, _ = mirror_descent(a_best, time_limit=rem, t_init=400.0, eta_init=0.04)
                record(a_md)

        # PGD
        if time.monotonic() < deadline - 3:
            rem = deadline - time.monotonic() - 3
            if rem > 3:
                a_pgd, _ = pgd_optimize(a_best, time_limit=rem, t_init=300.0, lr_init=0.015)
                record(a_pgd)

        # Final anneal
        if time.monotonic() < deadline - 2:
            rem = deadline - time.monotonic() - 2
            if rem > 5:
                rng_f = np.random.default_rng(seed + 99999)
                a_an, _ = anneal_informed(a_best, time_limit=rem, rng=rng_f)
                record(a_an)

    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
