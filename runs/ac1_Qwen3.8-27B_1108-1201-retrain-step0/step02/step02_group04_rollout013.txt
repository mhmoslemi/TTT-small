import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score_and_grad(a):
    """Compute score, max index, and gradient."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, 0, None
    c = fft_convolve(a, a)
    M = np.max(c)
    k_star = np.argmax(c)
    score = 2.0 * n * M / (S * S)
    
    # Gradient of f(a) = 2n * M / S^2
    # df/da[j] = 2n/S * (dM/da[j] - 2M/S)
    # dM/da[j] = (j in I_{k*} ? a[k*-j] + a[j] : 0)
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    dM = np.zeros(n)
    dM[j_min:j_max+1] = a[j_min:j_max+1] + a[k_star - j_max:k_star - j_min + 1]
    
    grad = (2.0 * n / S) * (dM - 2.0 * M / S)
    return score, k_star, grad

def fft_convolve(a, b):
    n = len(a)
    m = len(b)

def compute_soft_max_gradient(c, T, n):
    """Compute gradient of soft-max of c with respect to a."""
    # Shift for numerical stability
    c_shifted = c - np.max(c)
    exp_c = np.exp(T * c_shifted)
    weights = exp_c / np.sum(exp_c)

        j_max = min(n - 1, k)
