def adam_step(params, grads, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m[:] = beta1 * m + (1 - beta1) * grads
    v[:] = beta2 * v + (1 - beta2) * (grads ** 2)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    params[:] = params - lr * m_hat / (np.sqrt(v_hat) + eps)

"""
Adam-based optimization with top-k smooth max objective for minimizing
peak autocorrelation of non-negative sequences. Combines gradient-based
optimization with peak-targeting local search.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = e.sum()
    if s == 0:
        return np.ones(len(theta)) / len(theta)
    return e / s

def local_score(a):
    """Score: 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

def topk_smooth_max_obj_grad(theta, k, tau):
    """
    Objective and gradient for 2n * topk_smooth_max(conv(softmax(theta), softmax(theta))).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    # Get top-k values and indices
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    
    # Log-sum-exp on top-k
    m = np.max(top_vals)
    shifted = (top_vals - m) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    smooth_max_val = m + tau * np.log(sum_exp)
    w_top = exp_shifted / sum_exp
    
    score = 2.0 * n * smooth_max_val
    
    # Weight vector w (zero except at top-k positions)
    w = np.zeros_like(c)
    w[top_idx] = w_top
    
    # Gradient of score w.r.t. a
    a_rev = a[::-1]
    conv_w = fft_convolve(w, a_rev)
    grad_a = 4.0 * n * conv_w[n-1:2*n-1]
    
    # Transform to theta space
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta

def make_chirp_init(n, alpha, gamma, rng):
    """Windowed chirp initialization."""
    t = np.arange(n, dtype=np.float64)
    window = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1))) if n > 1 else np.array([1.0])
    window = np.maximum(window, 0.01)
    a = window * (1.0 + alpha * np.cos(2 * np.pi * gamma * t * t / n))
    a = np.maximum(a, 0.01)
    a += 0.02 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)

def make_smooth_random_init(n, rng, n_freq=5):
    """Smooth random initialization using random Fourier features."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_freq):
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)

def make_hann_init(n, rng):
    """Hann window initialization with perturbation."""
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1))) if n > 1 else np.array([1.0])
    a = np.maximum(a, 0.01)
    a *= np.exp(rng.normal(0, 0.1, n))
    a = np.maximum(a, 0.01)
    return a / np.sum(a)

def run_adam(theta, n, k_schedule, tau_schedule, max_iters, lr_init, lr_final, deadline, rng):
    """
    Run Adam optimizer with scheduled k and tau parameters.
    Returns best theta and score.
    """
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    
    total_iters = max_iters
    
    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        
        # Determine current k and tau based on progress
        progress = it / total_iters
        k_idx = min(int(progress * len(k_schedule)), len(k_schedule) - 1)
        tau_idx = min(int(progress * len(tau_schedule)), len(tau_schedule) - 1)
        k = k_schedule[k_idx]
        tau = tau_schedule[tau_idx]
        
        score, grad = topk_smooth_max_obj_grad(theta, k, tau)
        
        # Adam update
        t += 1
        lr = lr_init * (lr_final / lr_init) ** (it / total_iters)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Check score
        a = softmax(theta)
        s = local_score(a)
        if s < best_score:
            best_score = s
            best_theta = theta.copy()
        
        if it % 500 == 0:
            print(f"      Adam iter {it}: score={score:.6f}, best={best_score:.6f}, lr={lr:.6f}")
    
    return best_theta, best_score

def peak_targeting_search(a_init, time_limit, rng):
    """
    Local search that specifically targets the worst peaks in the convolution.
    Works in a-space with simplex projection.
    """
    current = a_init.astype(np.float64).copy()
    current = current / np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        c_max = np.max(c)
        peak_indices = np.where(c > c_max * 0.95)[0]  # Near-max peaks
        
        # For each peak, find contributing elements
        move_applied = False
        for pk in peak_indices:
            # Find which pairs (i, j) with i+j=pk contribute to c[pk]
            contribs = np.zeros(n)
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            for i in range(lo, hi + 1):
                j = pk - i
                if 0 <= j < n:
                    contribs[i] += current[i] * current[j]
            
            # Reduce top contributors, increase non-contributors
            top_k = max(2, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            
            # Non-contributors (elements not involved in this peak)
            non_contrib = np.setdiff1d(np.arange(n), 
                                       np.union1d(np.arange(lo, hi+1), 
                                                  np.arange(max(0, pk-hi), min(n, pk-lo+1))))
            
            candidate = current.copy()
            # Reduce contributors
            reduce_factor = np.exp(rng.uniform(0.02, 0.08, size=len(top_idx)))
            candidate[top_idx] /= reduce_factor
            # Increase non-contributors
            if len(non_contrib) > 0:
                n_inc = min(10, len(non_contrib))
                inc_idx = rng.choice(non_contrib, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.uniform(0.02, 0.08, size=n_inc))
            
            candidate = np.maximum(candidate, 1e-12)
            candidate = candidate / np.sum(candidate)
            new_score = local_score(candidate)
            
            if new_score < current_score:
                current = candidate
                current_score = new_score
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
                move_applied = True
                break
        
        if not move_applied:
            # Try random perturbation
            candidate = current * np.exp(rng.normal(0, 0.02, n))
            candidate = np.maximum(candidate, 1e-12)
            candidate = candidate / np.sum(candidate)
            new_score = local_score(candidate)
            if new_score < current_score:
                current = candidate
                current_score = new_score
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
            else:
                # Try larger perturbation
                candidate = current * np.exp(rng.normal(0, 0.05, n))
                candidate = np.maximum(candidate, 1e-12)
                candidate = candidate / np.sum(candidate)
                new_score = local_score(candidate)
                if new_score < current_score:
                    current = candidate
                    current_score = new_score
                    if current_score < best_score:
                        best_score = current_score
                        best = current.copy()
    
    return best, best_score

Actually, I want to add one more important idea: **the "water-filling" approach**.

The idea is:
1. Compute the current convolution c
2. Identify the "excess" at each peak: excess[k] = max(0, c[k] - target) where target is the desired peak level
3. Find a direction d that reduces the excess the most
4. The optimal direction is the one that "fills in" the valleys while reducing the peaks

More concretely:
- We want to find d (a perturbation to a) that minimizes max(c + d*c_{linear}) 
- Linearizing: c_new ≈ c + 2 * conv(a, d) (first-order)
- We want to minimize max_k (c[k] + 2 * conv(a, d)[k])
- This is equivalent to: find d such that c[k] + 2 * (a*d)[k] <= T for all k, for the smallest T
- This is a linear feasibility problem!

- Take the top M peaks
- For each, the constraint is: sum_j d[j] * 2*a[k-j] <= T - c[k]
- Find d that satisfies all M constraints with minimum T

"""
Adam-based optimization with top-k smooth max objective for minimizing
peak autocorrelation of non-negative sequences. Uses projected gradient
in simplex space, multi-length exploration, and peak-targeting local search.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def local_score(a):
    """Score: 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def topk_smooth_max_obj_grad(theta, k, tau):
    """
    Objective and gradient for 2n * topk_smooth_max(conv(softmax(theta), softmax(theta))).
    Returns (score, grad_theta).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    # Get top-k values and indices
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]

    # Log-sum-exp on top-k
    m = np.max(top_vals)
    shifted = (top_vals - m) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    smooth_max_val = m + tau * np.log(sum_exp)
    w_top = exp_shifted / sum_exp

    score = 2.0 * n * smooth_max_val

    # Weight vector w (nonzero only at top-k positions)
    w = np.zeros_like(c)
    w[top_idx] = w_top

    # Gradient: 4n * (w * a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_w = fft_convolve(w, a_rev)
    grad_a = 4.0 * n * conv_w[n - 1:2 * n - 1]

    # Transform to theta space
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def simplex_projection(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    cv = sv + np.arange(1, n + 1)
    p = np.nonzero(sv * n + 1.0 - cv > 0)[0][-1]
    mu = (sv[p] + 1.0 / (p + 1.0) - 1.0) / (p + 1.0)
    return np.maximum(v - mu, 0.0)


def make_chirp_init(n, alpha, gamma, rng):
    """Windowed chirp initialization."""
    t = np.arange(n, dtype=np.float64)
    if n > 1:
        window = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
    else:
        window = np.array([1.0])
    window = np.maximum(window, 0.02)
    a = window * (1.0 + alpha * np.cos(2 * np.pi * gamma * t * t / n))
    a = np.maximum(a, 0.01)
    a += 0.02 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def make_smooth_random_init(n, rng, n_freq=4):
    """Smooth random init via random Fourier features."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_freq):
        freq = rng.uniform(0.3, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.6)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def make_hann_init(n, rng):
    """Hann window with perturbation."""
    t = np.arange(n, dtype=np.float64)
    if n > 1:
        a = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
    else:
        a = np.array([1.0])
    a = np.maximum(a, 0.02)
    a *= np.exp(rng.normal(0, 0.15, n))
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def run_adam_opt(theta, n, k_schedule, tau_schedule, max_iters,
                 lr_init, lr_final, deadline, rng):
    """
    Run Adam optimizer with scheduled top-k and tau.
    Returns (best_theta, best_score).
    """
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    no_improve_count = 0
    best_at_last_restart = best_score

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = it / max(1, max_iters)
        k_idx = min(int(progress * len(k_schedule)), len(k_schedule) - 1)
        tau_idx = min(int(progress * len(tau_schedule)), len(tau_schedule) - 1)
        k = max(1, k_schedule[k_idx])
        tau = tau_schedule[tau_idx]

        score, grad = topk_smooth_max_obj_grad(theta, k, tau)

        # Adam update with cosine-annealed LR
        t_step += 1
        lr = 0.5 * lr_init * (1 + np.cos(np.pi * it / max(1, max_iters)))
        lr = max(lr, lr_final)

        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * (grad ** 2)
        m_hat = m_buf / (1 - beta1 ** t_step)
        v_hat = v_buf / (1 - beta2 ** t_step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = softmax(theta)
        s = local_score(a)

        if s < best_score:
            best_score = s
            best_theta = theta.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Restart if stuck for too long
        if no_improve_count > 800 and no_improve_count % 400 == 0:
            theta = best_theta + rng.normal(0, 0.1, n)
            m_buf = np.zeros(n)
            v_buf = np.zeros(n)
            no_improve_count = 0

        if it % 1000 == 0 and it > 0:
            elapsed = time.monotonic() - (deadline - max_iters * 0.05)
            print(f"      Adam iter {it}/{max_iters}: score={score:.6f}, best={best_score:.6f}, lr={lr:.5f}, k={k}, tau={tau:.5f}")

    return best_theta, best_score


def run_simplex_gradient(a_init, n, k, tau, max_iters, lr_init, lr_final, deadline, rng):
    """
    Projected gradient descent in a-space (simplex constraint).
    Uses top-k smooth max subgradient.
    Returns (best_a, best_score).
    """
    current = a_init.copy()
    current = np.maximum(current, 1e-12)
    current = current / np.sum(current)
    best = current.copy()
    best_score = local_score(current)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        a = current
        c = fft_convolve(a, a)

        # Top-k smooth max weights
        top_idx = np.argsort(c)[::-1][:k]
        top_vals = c[top_idx]
        m = np.max(top_vals)
        shifted = (top_vals - m) / tau
        exp_shifted = np.exp(shifted)
        w_top = exp_shifted / np.sum(exp_shifted)

        w = np.zeros_like(c)
        w[top_idx] = w_top
        a_rev = a[::-1]
        conv_w = fft_convolve(w, a_rev)
        grad_a = 4.0 * n * conv_w[n - 1:2 * n - 1]

        # Adam on a-space
        t_step += 1
        lr = 0.5 * lr_init * (1 + np.cos(np.pi * it / max(1, max_iters)))
        lr = max(lr, lr_final)

        m_buf = 0.9 * m_buf + 0.1 * grad_a
        v_buf = 0.999 * v_buf + 0.001 * (grad_a ** 2)
        m_hat = m_buf / (1 - 0.9 ** t_step)
        v_hat = v_buf / (1 - 0.999 ** t_step)

        candidate = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        candidate = simplex_projection(candidate)

        s = local_score(candidate)
        if s < best_score:
            best_score = s
            best = candidate.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500 and no_improve % 250 == 0:
            candidate = best * np.exp(rng.normal(0, 0.1, n))
            candidate = simplex_projection(candidate)
            s = local_score(candidate)
            if s < best_score:
                best = candidate.copy()
                best_score = s
            current = best.copy()
            m_buf = np.zeros(n)
            v_buf = np.zeros(n)
            no_improve = 0
        else:
            current = candidate

    return best, best_score


def peak_targeting_search(a_init, time_limit, rng):
    """
    Local search targeting the worst peaks in the convolution.
    Works in a-space with simplex projection.
    """
    current = a_init.astype(np.float64).copy()
    current = np.maximum(current, 1e-12)
    current = current / np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        c_max = np.max(c)
        # Near-max peaks (within 1% of max)
        threshold = c_max * 0.98
        peak_indices = np.where(c >= threshold)[0]

        move_success = False

        for pk in peak_indices[:5]:  # Try top 5 peaks
            # Find contributing indices
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            contribs = np.zeros(n)
            for i in range(lo, hi + 1):
                j = pk - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
                    contribs[j] = max(contribs[j], current[i] * current[j])

            # Reduce top contributors
            top_k = max(2, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]

            # Increase non-contributors
            involved = set(range(lo, hi + 1))
            non_contrib = np.array([i for i in range(n) if i not in involved])

            candidate = current.copy()
            red_scale = rng.uniform(0.92, 0.98)
            candidate[top_idx] *= red_scale

            if len(non_contrib) > 0:
                n_inc = min(15, len(non_contrib))
                inc_idx = rng.choice(non_contrib, size=n_inc, replace=False)
                inc_scale = rng.uniform(1.02, 1.12, size=n_inc)
                candidate[inc_idx] *= inc_scale

            candidate = np.maximum(candidate, 1e-12)
            candidate = candidate / np.sum(candidate)
            new_score = local_score(candidate)

            if new_score < current_score:
                current = candidate
                current_score = new_score
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
                stall_count = 0
                move_success = True
                break

        if not move_success:
            stall_count += 1
            # Try various random perturbations
            strategies = [
                0.01, 0.03, 0.05, 0.1
            ]
            for scale in strategies:
                candidate = current * np.exp(rng.normal(0, scale, n))
                candidate = np.maximum(candidate, 1e-12)
                candidate = candidate / np.sum(candidate)
                new_score = local_score(candidate)
                if new_score < current_score:
                    current = candidate
                    current_score = new_score
                    if current_score < best_score:
                        best_score = current_score
                        best = current.copy()
                    stall_count = 0
                    break

            # If still stuck, try block modification
            if stall_count > 20:
                block_size = max(2, n // 16)
                start = int(rng.integers(0, n - block_size))
                candidate = current.copy()
                candidate[start:start + block_size] *= np.exp(rng.normal(-0.05, 0.03, block_size))
                # Increase elsewhere
                other = np.setdiff1d(np.arange(n), np.arange(start, start + block_size))
                if len(other) > 0:
                    n_inc = min(8, len(other))
                    inc_idx = rng.choice(other, size=n_inc, replace=False)
                    candidate[inc_idx] *= np.exp(rng.uniform(0.03, 0.08, n_inc))
                candidate = np.maximum(candidate, 1e-12)
                candidate = candidate / np.sum(candidate)
                new_score = local_score(candidate)
                if new_score < current_score:
                    current = candidate
                    current_score = new_score
                    if current_score < best_score:
                        best_score = current_score
                        best = current.copy()
                    stall_count = 0

    return best, best_score

            a_pert = a_pert / np.sum(a_pert)
            s = local_score(a_pert)
            if s < best_score:
                # Quick local search from this point
                t_rem = deadline - time.monotonic() - 10
                if t_rem > 10:
                    polished, ps = peak_targeting_search(a_pert, min(30.0, t_rem), rng)
                    if ps < best_score:
                        best_score = ps
                        best_seq = polished.copy()
                        print(f"  Trial {trial}: {ps:.6f} *** NEW BEST ***")

    # Final verification and return
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
