Minimize 2*n*max_conv / sum^2 over non-negative sequences.
Uses FISTA with adaptive top-k subgradient and multiple initialization strategies.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_topk_grad(a, c, top_k):
    """Gradient of the top-k sum of convolution, via FFT cross-correlation."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    
    w_full = np.zeros(len(c))
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def fft_logsumexp_grad(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c))."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def run_fista_topk(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=128):
    """FISTA with top-k subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        grad = fft_topk_grad(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        anorm = np.linalg.norm(a)
        step = 1.0 / (2.0 * max(anorm, 1e-6) * (1.0 + 0.001 * it))
        
        x_new = project_simplex(y - step * grad, S)
        
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score

Algorithm: Multi-phase optimization with FISTA (log-sum-exp + top-k),
peak shaving, and multiple initializations.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.astype(np.float64)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n

        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    """Subgradient using top-k active lags."""
    n = len(a)
    c_len = len(c)
    top_indices = np.argsort(c)[::-1][:top_k]
    weights_full = np.zeros(c_len)

    W = fft(weights_full, fsize)
    A = fft(a, fsize)

                best_a = x_new.copy()

        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 6000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_a, best_score


def peak_shave(a, n, S, max_time, rng):
    """Coordinate descent: reduce values contributing to the peak lag."""
    a = np.clip(a, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
        
        # Find the max lag
        k_star = int(np.argmax(c))
        
        # Gradient of b[k_star] w.r.t. a[j] = 2 * a[k_star - j] for valid j
        # a[k_star - j] is valid when 0 <= k_star - j <= n-1
        # i.e., k_star - (n-1) <= j <= k_star
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        
        if lo > hi:
            break
        
        # The gradient at a[lo:hi+1] is 2 * a reversed and shifted
        # grad[j] = 2 * a[k_star - j] for j in [lo, hi]
        grad_slice = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]
        
        # Reduce a at these positions proportionally
        total_grad = grad_slice.sum()
        if total_grad < 1e-15:
            break
        
        # Step: reduce a[j] by alpha * grad[j] / total_grad * some_factor
        alpha = 0.1  # fraction of mass to redistribute
        reduction = alpha * S * grad_slice / total_grad
        
        a_new = a.copy()
        a_new[lo:hi+1] = np.maximum(a_new[lo:hi+1] - reduction, 0.0)
        
        # Renormalize
        s_new = a_new.sum()
        if s_new < 0.01:
            a_new = a.copy()
        else:
            # Add the removed mass to positions that don't contribute to the peak
            removed = S - s_new
            # Add to positions far from the peak
            mask = np.ones(n, dtype=bool)
            mask[lo:hi+1] = False
            if mask.any():
                # Distribute to non-peak positions weighted by current value
                weights = a_new[mask].copy()
                if weights.sum() < 1e-15:
                    a_new[mask] += removed / mask.sum()
                else:
                    a_new[mask] += removed * weights / weights.sum()
            a_new = a_new * (S / a_new.sum())
        
        a = a_new
    
    return best_a, best_score


def make_gauss_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.linspace(0, 1, n)
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    return a + 0.01  # small floor to avoid zeros


def make_cosine_init(n):
    """Raised cosine."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    return a + 0.01


def make_w_init(n, n_bumps=3):
    """Multi-lobe W-shape."""
    x = np.linspace(0, 1, n)
    a = np.abs(np.sin(np.pi * n_bumps * x))
    return a + 0.01


def make_random_smooth_init(n, rng, n_frequencies=5):
    """Random smooth signal."""
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for _ in range(n_frequencies):
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        a += amp * np.sin(2 * np.pi * freq * x + phase)
    return np.maximum(a, 0.01)
