    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=400.0,
                  step_scale=0.5, restart_every=2500, restart_noise=0.008):
    """FISTA with LSE smoothing, adaptive beta, and restarts."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    since_improve = 0
    prev_best = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: start low, increase gradually
        progress = min(1.0, it / (max_time * 50.0))
        beta = beta_init + (beta_max - beta_init) * progress * progress

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            since_improve = 0
        else:
            since_improve += 1

        # Restart if stuck
        if since_improve > restart_every and rng.random() < 0.3:
            noise_scale = restart_noise * S / np.sqrt(n)
            y = project_simplex(best_local + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            since_improve = 0

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = step_scale / (beta * 2.0)
        x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    return best_local, best_score


def run_exact_subgrad_nesterov(a_init, n, S, max_time, rng,
                                k_init=3, k_max=200, alpha=2.0):
    """Nesterov-accelerated method with exact/soft subgradients."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    since_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Gradually increase k and alpha
        k = min(k_max, k_init + it // 200)
        alpha = min(10.0, alpha_init_val + 0.01 * it)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            since_improve = 0
        else:
            since_improve += 1

        if since_improve > 3000 and rng.random() < 0.2:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(best_local + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            since_improve = 0

        if k <= 1:
            grad = single_max_subgradient(y, c)
        else:
            grad = smooth_weighted_subgradient(y, c, k, alpha)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Decreasing step size
        step = 2.0 * S / (np.sqrt(2.0 * it) * gnorm)
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_mom + 1  # use t = it+1 style

    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, eta_init=0.1, k=10):
    """Mirror descent (multiplicative weights) for diversity."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()

        eta = eta_init / (1.0 + 0.005 * it)
        if k <= 1:
            grad = single_max_subgradient(a, c)
        else:
            grad = topk_subgradient(a, c, k)

        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_new * (S / s_new)

    return best_local, best_score


def make_ramp_init(n, rng, style=0):
    """Create asymmetric ramp-like initializations."""
    t = np.linspace(0, 1, n, endpoint=False)
    if style == 0:
        # Linear ramp up
        return t + 0.1
    elif style == 1:
        # Linear ramp down
        return 1.0 - t + 0.1
    elif style == 2:
        # Quadratic ramp
        return t * t + 0.1
    elif style == 3:
        # Exponential ramp
        return np.exp(2.0 * t) + 0.05
    elif style == 4:
        # Ramp with sinusoidal modulation
        return t + 0.2 * np.sin(2 * np.pi * 5 * t) + 0.05
    elif style == 5:
        # Two-segment: high then low
        return np.where(t < 0.5, 1.5 - 2.0 * t, 0.5 + 2.0 * (t - 0.5)) + 0.05
    elif style == 6:
        # Smooth step (sigmoid-like)
        return 1.0 / (1.0 + np.exp(-8.0 * (t - 0.4))) + 0.05
    else:
        return rng.random(n) + 0.05


def run_segment_optimization(n, S, max_time, rng):
    """Optimize piecewise-constant sequences with varying number of segments."""
    best_a = None
    best_score = np.inf

    n_seg_options = [4, 6, 8, 12, 16, 24]
    time_per_seg = max_time / len(n_seg_options)

    for n_seg in n_seg_options:
        if time.monotonic() > deadline_global:
            break
        t_seg_start = time.monotonic()

        # Try multiple random segment boundary configurations
        n_configs = 3
        for cfg in range(n_configs):
            # Random segment boundaries
            boundaries = np.sort(rng.choice(np.arange(1, n), n_seg - 1, replace=False))
            boundaries = np.concatenate([[0], boundaries, [n]])
            lengths = np.diff(boundaries).astype(int)

            # Initialize heights
            h = rng.random(n_seg) + 0.1

            # Optimize heights using projected gradient
            for it in range(500):
                a = np.repeat(h, lengths)
                s = a.sum()
                if s < 1e-15:
                    break
                a_n = a * (S / s)
                c = fft_convolve_self(a_n)
                cur = c.max()
                if cur < best_score:
                    best_score = cur
                    best_a = a_n.copy()

                # Gradient w.r.t. h
                c_max = c.max()
                beta_h = 20.0
                exp_bc = np.exp(beta_h * (c - c_max))
                w = exp_bc / exp_bc.sum()
                fsize = 1 << (len(w) + n - 2).bit_length()
                W = fft(w, fsize)
                A = fft(a_n, fsize)
                grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

                # Average gradient within each segment
                grad_h = np.zeros(n_seg)
                for j in range(n_seg):
                    start_idx = int(boundaries[j])
                    end_idx = int(boundaries[j + 1])
                    grad_h[j] = grad_a[start_idx:end_idx].mean()

                # Learning rate
                lr = 0.5 / (beta_h * np.max(np.abs(grad_h)) + 1e-10)
                h = h - lr * grad_h
                h = np.maximum(h, 0.0)
                # Normalize sum constraint
                h_sum = (h * lengths).sum()
                if h_sum > 1e-15:
                    h = h * (S / h_sum)

            if time.monotonic() - t_seg_start > time_per_seg * 1.5:
                break

    return best_a, best_score


def try_multiple_lengths(a_best, n_best, score_best, max_time, rng):
    """Try downsampling/upsampling the best sequence to other lengths."""
    candidates = [(a_best, n_best, score_best)]

    # Try length n_best // 2
    if n_best > 2048:
        n_half = n_best // 2
        # Downsample by averaging pairs
        n_use = n_best - (n_best % 2)
        a_down = a_best[:n_use].reshape(-1, 2).mean(axis=1)
        s = a_down.sum()
        S_half = np.sqrt(2.0 * n_half)
        a_down = a_down * (S_half / s)
        score_down = normalized_score(a_down)
        candidates.append((a_down, n_half, score_down))

    # Try length n_best * 2 (upsample by interpolation)
    if n_best < 16384:
        n_dbl = n_best * 2
        from numpy import interp
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_dbl)
        a_up = interp(x_new, x_old, a_best)
        s = a_up.sum()
        S_dbl = np.sqrt(2.0 * n_dbl)
        a_up = a_up * (S_dbl / s)
        score_up = normalized_score(a_up)
        candidates.append((a_up, n_dbl, score_up))

    return candidates


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv)/sum^2.
    Uses FISTA-LSE, exact subgradient Nesterov, mirror descent,
    and segment-based optimization with multiple initializations.
    """
    global deadline_global
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline_global = time.monotonic() + max(0.0, budget_s - 10.0)

    # --- Determine main length and gather initializations ---
    main_len = 8192
    given_seq = None
    try:
        g = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if g.size >= 64:
            g = np.clip(g, 0, 1000)
            if g.sum() > 0.01:
                given_seq = g
                main_len = len(g)
    except Exception:
        pass

    S = np.sqrt(2.0 * main_len)
    inits = []

    if given_seq is not None:
        inits.append(("given", given_seq))
        # Reversal
        inits.append(("given_rev", given_seq[::-1].copy()))

    # Asymmetric ramp initializations
    for style in range(7):
        inits.append(("ramp%d" % style, make_ramp_init(main_len, rng, style)))

    # Random
    inits.append(("rand", rng.random(main_len) + 0.05))
    # Sinusoidal
    t = np.linspace(0, 1, main_len, endpoint=False)
    inits.append(("sin5", 1.0 + 0.4 * np.sin(2 * np.pi * 5 * t)))

    global_best = np.inf
    global_a = None

    def update_best(a, name):
        nonlocal global_best, global_a
        sc = normalized_score(a)
        if sc < global_best:
            global_best = sc
            global_a = a.copy()
            print("  New best [%s]: %.6f (len=%d)" % (name, sc, len(a)))
            return True
        return False

    # --- Phase 0: Quick segment optimization for coarse structure ---
    print("=== Phase 0: Segment optimization ===")
    seg_time = 60.0
    seg_a, seg_score = run_segment_optimization(main_len, S, seg_time, rng)
    if seg_a is not None:
        update_best(seg_a, "segment")

    # --- Phase 1: FISTA-LSE on top initializations ---
    print("=== Phase 1: FISTA-LSE exploration ===")
    # Allocate time: give more time to better initializations
    # First, quick evaluation of all inits
    init_scores = []
    for name, arr in inits:
        a_n = arr.copy()
        s = a_n.sum()
        a_n = a_n * (S / s)
        sc = normalized_score(a_n)
        init_scores.append((name, a_n, sc))
        if sc < global_best:
            global_best = sc
            global_a = a_n.copy()
            print("  Init [%s]: %.6f" % (name, sc))

    # Sort by initial score, best first
    init_scores.sort(key=lambda x: x[2])

    # Time allocation: top 3 get more time
    remaining = deadline_global - time.monotonic()
    n_top = min(3, len(init_scores))
    time_per_top = remaining * 0.5 / n_top
    time_per_rest = remaining * 0.2 / max(1, len(init_scores) - n_top)

    fista_results = []
    for i, (name, a_init, sc0) in enumerate(init_scores):
        if time.monotonic() > deadline_global - 60:
            break
        alloc = time_per_top if i < n_top else time_per_rest
        alloc = min(alloc, max(30.0, (deadline_global - time.monotonic()) * 0.6))

        a_fista, sc_fista = run_fista_lse(a_init, main_len, S, alloc * 0.7, rng,
                                           beta_init=8.0, beta_max=300.0,
                                           step_scale=0.5)
        if update_best(a_fista, "fista_" + name):
            pass
        fista_results.append((a_fista, sc_fista, name))

        # Mirror descent on the FISTA result
        rem = alloc * 0.3
        if rem > 5:
            a_md, sc_md = run_mirror_descent(a_fista, main_len, S, rem, rng,
                                             eta_init=0.05, k=10)
            update_best(a_md, "mirror_" + name)

    # --- Phase 2: Polish the best with exact subgradient ---
    print("=== Phase 2: Exact subgradient Nesterov polish ===")
    remaining = deadline_global - time.monotonic()
    if global_a is not None and remaining > 30:
        pol_time = remaining * 0.4
        a_pol, sc_pol = run_exact_subgrad_nesterov(global_a, main_len, S, pol_time, rng,
                                                    k_init=3, k_max=150, alpha=2.0)
        update_best(a_pol, "polish_nesterov")

    # --- Phase 3: High-beta FISTA polish ---
    remaining = deadline_global - time.monotonic()
    if global_a is not None and remaining > 20:
        pol_time2 = remaining * 0.5
        a_p2, sc_p2 = run_fista_lse(global_a, main_len, S, pol_time2, rng,
                                     beta_init=100.0, beta_max=600.0,
                                     step_scale=0.3)
        update_best(a_p2, "polish_highbeta")

    # --- Phase 4: Try different lengths ---
    remaining = deadline_global - time.monotonic()
    if global_a is not None and remaining > 60:
        print("=== Phase 4: Multi-length exploration ===")
        candidates = try_multiple_lengths(global_a, main_len, global_best, remaining, rng)
        for a_cand, n_cand, sc_cand in candidates:
            if n_cand == main_len:
                continue
            S_cand = np.sqrt(2.0 * n_cand)
            a_norm = np.clip(a_cand, 0, 1000)
            a_norm = a_norm * (S_cand / a_norm.sum())
            sc = normalized_score(a_norm)
            # Quick FISTA on this
            alloc = min(60.0, remaining * 0.3)
            a_res, sc_res = run_fista_lse(a_norm, n_cand, S_cand, alloc, rng,
                                           beta_init=10.0, beta_max=400.0)
            update_best(a_res, "len%d" % n_cand)

    # --- Phase 5: Final fine polish ---
    remaining = deadline_global - time.monotonic()
    if global_a is not None and remaining > 10:
        print("=== Phase 5: Final polish ===")
        # Very high beta, small steps
        a_final, sc_final = run_fista_lse(global_a, len(global_a),
                                           np.sqrt(2.0 * len(global_a)), remaining, rng,
                                           beta_init=200.0, beta_max=800.0,
                                           step_scale=0.2)
        update_best(a_final, "final_polish")

        # One more round of exact subgradient
        remaining2 = deadline_global - time.monotonic()
        if remaining2 > 10:
            a_final2, _ = run_exact_subgrad_nesterov(global_a, len(global_a),
                                                      np.sqrt(2.0 * len(global_a)),
                                                      remaining2, rng,
                                                      k_init=10, k_max=300, alpha=3.0)
            update_best(a_final2, "final_subgrad")

    # --- Verification ---
    print("=== Verification ===")
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
        S_fb = np.sqrt(2.0 * 256)
        global_a = global_a * (S_fb / global_a.sum())

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Final score via evaluate_sequence: %.6f" % ver)
    print("Sequence length: %d" % len(global_a))

    if not np.isfinite(ver):
        print("WARNING: evaluate_sequence rejected, using fallback")
        global_a = np.ones(256, dtype=np.float64)
        S_fb = np.sqrt(2.0 * 256)
        global_a = global_a * (S_fb / global_a.sum())
        final_list = [float(x) for x in global_a]

    return final_list
