import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute the evaluation score for a sequence (normalized to sum=1)."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    a_norm = a / S
    c = fft_convolve(a_norm, a_norm)
    return 2.0 * n * np.max(c)

def compute_score_full(a):
    """Compute score directly from raw sequence (like evaluate_sequence)."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]

    """Compute 2*n*max(conv(a,a))/sum(a)^2 for non-negative a."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def grad_soft_max(a, tau):
    """
    Given a (normalized to sum=1), compute soft-max(a*a, tau) and its gradient w.r.t. a.
    Returns (value, gradient, convolution).
    """
    n = len(a)
    c = fft_convolve(a, a)
    cm = np.max(c)
    sh = (c - cm) / tau
    e = np.exp(sh)
    s = e.sum()
    w = e / s
    val = cm + tau * np.log(s)
    # grad_i = 2 * sum_k w[k] * a[k-i] for valid indices
    # Computed via: ifft(conj(FFT(w)) * FFT(a))[:n]
    sz = 1
    while sz >= 2 * n - 1:
        sz <<= 1
    W = np.fft.rfft(w, sz)
    A = np.fft.rfft(a, sz)
    corr = np.fft.irfft(np.conj(W) * A, sz)
    grad = 2.0 * corr[:n]
    return val, grad, c

    """Gerchberg-Saxton style FFT phase initialization."""
