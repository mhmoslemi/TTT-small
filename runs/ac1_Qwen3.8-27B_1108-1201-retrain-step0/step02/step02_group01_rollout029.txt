from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]

    Actually, let me recompute.
    grad[j] = 2 * sum_{k: 0<=k-j<n} w[k] * a[k-j]
            = 2 * sum_{i=0}^{n-1} w[j+i] * a[i]

    Define w_ext as w padded, and compute the relevant part.

    # We want conv_result[n-1 : n-1+n] = conv_result[n-1 : 2n-1]
    end = min(2*n-1, len(conv_result))
    grad = 2.0 * conv_result[n-1:end][:n]

    w = np.zeros(len(c))
    w[idx] = 1.0 / top_k

    return M, grad, idx

    if s < 1e-300:

        # Decay learning rate

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Top-K subgradient with Barzilai-Borwein adaptive step size and Nesterov momentum
2. Exact coordinate descent with incremental convolution updates
3. Multi-start exploration across different lengths
4. Periodic perturbation to escape local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def compute_subgrad_topk(a, top_k):
    """Compute subgradient at top-k convolution peaks.
    Returns (max_val, gradient, top_k_indices)."""
    n = len(a)
    c = fft_convolve_self(a)
    M = c.max()
    
    if top_k >= len(c):
        idx = np.arange(len(c))
        k_actual = len(c)
    else:
        # Efficient partial argsort
        if top_k < 64:
            idx = np.argsort(c)[::-1][:top_k]
        else:
            # Use partition for larger k
            part_idx = np.argpartition(c, -top_k)[-top_k:]
            idx = part_idx[np.argsort(c[part_idx])[::-1]]
        k_actual = top_k
    
    w = np.zeros(len(c))
    w[idx] = 1.0 / k_actual
    
    # grad[j] = 2 * sum_{i=0}^{n-1} a[i] * w[j+i]
    # = 2 * (cross-correlation of a and w at lag j)
    # cross_corr(a, w)[j] = sum_i a[i]*w[i+j] = (rev(a) * w)[n-1+j]
    a_rev = a[::-1]
    conv_res = fft_convolve(a_rev, w)
    # conv_res has length n + (2n-1) - 1 = 3n-2
    # We need conv_res[n-1 : 2n-1]
    end = min(2 * n - 1, len(conv_res))
    grad = 2.0 * conv_res[n - 1:end][:n]
    
    return M, grad, idx


def normalize_nonneg(a, S):
    """Clip to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_subgradient_optimize(a_init, S, time_budget, top_k_start=16, top_k_end=256,
                              lr_init=0.05, momentum=0.85):
    """Top-K subgradient with BB step and momentum."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    # Nesterov momentum
    a_prev = a.copy()
    v = np.zeros(n)
    
    # BB tracking
    s_prev = np.zeros(n)
    y_prev = np.zeros(n)
    
    M_curr = fft_convolve_self(a).max()
    a_best = a.copy()
    M_best = M_curr
    
    start = time.monotonic()
    t = 0
    lr = lr_init
    
    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget
        
        # Anneal top_k
        top_k = max(top_k_start, int(top_k_end * progress + top_k_start * (1 - progress)))
        
        # Nesterov extrapolation
        if t > 1:
            y = a + momentum * (a - a_prev)
            y = normalize_nonneg(y, S)
        else:
            y = a
        
        # Compute subgradient at y
        M_y, grad_y, _ = compute_subgrad_topk(y, top_k)
        
        # Project gradient (remove mean to maintain sum)
        grad_proj = grad_y - grad_y.mean()
        
        # BB step size
        if t > 1:
            s_curr = a - a_prev
            y_curr = grad_proj - (grad_proj.mean())
            # Use BB formula with projection
            sy = np.dot(s_curr, y_curr)
            ss = np.dot(s_curr, s_curr)
            yy = np.dot(y_curr, y_curr)
            if ss > 1e-30 and yy > 1e-30:
                if sy > 1e-15:
                    lr = ss / sy
                else:
                    lr = lr  # keep previous
                lr = np.clip(lr, 1e-6, 1.0)
            else:
                lr = lr_init / (1 + 0.05 * t)
        else:
            lr = lr_init
        
        # Update
        a_new = a - lr * grad_proj
        a_new = normalize_nonneg(a_new, S)
        
        a_prev = a
        a = a_new
        
        # Evaluate
        M_curr = fft_convolve_self(a).max()
        if M_curr < M_best:
            M_best = M_curr
            a_best = a.copy()
        
        # Periodic perturbation every ~500 seconds equivalent
        if t % 5000 == 0:
            perturbation = 0.01 * a_best
            a = a_best + rng_perturb(perturbation, len(a))
            a = normalize_nonneg(a, S)
    
    return a_best, M_best


def rng_perturb(scale, n):
    """Generate a random perturbation of the given scale."""
    return np.random.standard_normal(n) * scale


def coordinate_descent(a_init, S, time_budget, max_sweeps=1000, binary_iters=30):
    """Exact coordinate descent with incremental convolution updates."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    b = fft_convolve_self(a)  # Current convolution
    M_best = b.max()
    a_best = a.copy()
    
    start = time.monotonic()
    sweep = 0
    
    while time.monotonic() - start < time_budget and sweep < max_sweeps:
        sweep += 1
        improved = False
        
        # Random order for diversity
        order = np.arange(n)
        np.random.shuffle(order)
        
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            
            a_j = a[j]
            if a_j < 1e-12:
                continue
            
            # Fingerprint: w[k] = a[k-j] for valid k
            # new_b[k] = b[k] + 2*delta*a[k-j] + delta^2*(1 if k==2j else 0)
            # We want to minimize max(new_b) over delta (with a_j + delta >= 0)
            
            # Compute the effect: delta * (2*a[k-j]) + delta^2 * I(k==2j)
            # The range of k affected: k in [j, j+n-1] (where a[k-j] is valid)
            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)
            
            # For binary search, we need to evaluate max(b[k] + 2*delta*a[k-j] + ...)
            # Let's parametrize: new_a_j = a_j + delta, delta in [-a_j, a_j] (conservative)
            # Actually delta in [-a_j, inf) but we'll bound it
            
            # Quick check: what's the current max in the affected region?
            affected = b[k_lo:k_hi+1]
            current_max_affected = affected.max()
            global_max = b.max()
            
            # Only bother if this coordinate affects the max region
            # (optimization: skip if unaffected)
            
            # Binary search for optimal new_a_j
            # Objective: max_k (b[k] + 2*(new_a_j - a_j)*a[k-j] + (new_a_j-a_j)^2*I(k==2j))
            # = max of:
            #   - max over k != 2j in [k_lo,k_hi]: b[k] + 2*(new_a_j-a_j)*a[k-j]
            #   - over k == 2j (if in range): b[2j] + 2*(new_a_j-a_j)*a[j] + (new_a_j-a_j)^2
            #   - max over k outside [k_lo,k_hi]: b[k] (unchanged)
            
            unchanged_max = b.max()
            # Actually the unchanged max is the max over k NOT in [k_lo, k_hi]
            if k_lo > 0 or k_hi < 2*n-2:
                mask = np.ones(2*n-1, dtype=bool)
                mask[k_lo:k_hi+1] = False
                if mask.any():
                    unchanged_max = b[mask].max()
                else:
                    unchanged_max = 0.0
            else:
                unchanged_max = 0.0
            
            # For the affected region, the change is linear in delta (plus quadratic at k=2j)
            # Let delta = new_a_j - a_j
            # affected_new[k] = b[k] + 2*delta*a[k-j] (+ delta^2 if k==2j)
            
            # We want to find delta that minimizes max(unchanged_max, max_affected_new)
            # This is a 1D convex optimization (max of affine functions + one quadratic)
            
            # Binary search on delta
            lo = -a_j  # new_a_j >= 0
            hi = a_j * 2  # reasonable upper bound
            
            def eval_obj(delta):
                # Compute max over affected k
                # a[k-j] for k in [k_lo, k_hi] is a[0], a[1], ..., a[n-1]
                # So affected_new = b[k_lo:k_hi+1] + 2*delta*a[0:n-1] (padded)
                # Length of affected: k_hi - k_lo + 1
                n_affected = k_hi - k_lo + 1
                # a values: a[0 : n_affected] (since k-j ranges from 0 to n_affected-1)
                a_slice = a[:n_affected]
                val = b[k_lo:k_hi+1] + 2.0 * delta * a_slice
                # Add quadratic term if 2j is in [k_lo, k_hi]
                if k_lo <= 2*j <= k_hi:
                    pos = 2*j - k_lo
                    val[pos] += delta * delta
                return max(unchanged_max, val.max())
            
            # Minimize eval_obj(delta) over delta in [lo, hi]
            # Use ternary search (function is convex)
            for _ in range(binary_iters):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if eval_obj(m1) < eval_obj(m2):
                    hi = m2
                else:
                    lo = m1
            
            optimal_delta = (lo + hi) / 2.0
            new_a_j = a_j + optimal_delta
            
            if abs(new_a_j - a_j) > 1e-10:
                # Update a and b incrementally
                delta = new_a_j - a_j
                a[j] = new_a_j
                # Update b
                a_slice = a[:k_hi - k_lo + 1]  # Note: a[j] is already updated
                # Actually we need the OLD a for the update
                # b[k] += 2*delta*old_a[k-j] + delta^2*I(k==2j)
                # old_a[k-j] for k in [k_lo, k_hi] is old_a[0 : n_affected]
                # But we already updated a[j], so old_a[j] = a_j, new a[j] = new_a_j
                # For the update formula, we use OLD values:
                # b[k] += 2*delta*a_old[k-j] + delta^2*I(k==2j)
                # a_old[k-j] = a[k-j] except a_old[j] = a_j (not new_a_j)
                # Since we're computing with the slice a[:n_affected], and a[j] is now new_a_j,
                # we need to be careful. Let's just recompute using old values.
                # Actually, simpler: just use the formula with old a.
                # a_old[k-j] for k in [j, j+n-1]:
                #   k-j = 0, 1, ..., n-1
                #   a_old[0], a_old[1], ..., a_old[n-1]
                # But a_old[j] = a_j (the old value), and the current a[j] = new_a_j
                # So we need to temporarily restore
                saved = a[j]
                a[j] = a_j  # restore old
                a_slice_old = a[:k_hi - k_lo + 1]
                a[j] = saved  # restore new
                
                b[k_lo:k_hi+1] += 2.0 * delta * a_slice_old
                if k_lo <= 2*j <= k_hi:
                    b[2*j] += delta * delta
                improved = True
            
            # Check improvement
            if sweep % 10 == 0 or improved:
                current_M = b.max()
                if current_M < M_best:
                    M_best = current_M
                    a_best = a.copy()
        
        if not improved:
            break
    
    return a_best, M_best


def make_initialization(init_name, n, S, rng):

        a = np.maximum(a, 1e-6)
    elif init_name == "linear_up":

    elif init_name == "linear_down":
        a = n - i

        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
        print("Fallback: obj=%.6f" % ver)
    
    return final_list


# Helper for perturbation (top-level to avoid closure)
_rng_pert = np.random.default_rng(12345)

def rng_perturb(scale, n):
    """Generate random perturbation."""
    return _rng_pert.standard_normal(n) * scale
