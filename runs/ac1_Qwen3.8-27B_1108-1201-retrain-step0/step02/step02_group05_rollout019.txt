import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    return irfft(rfft(a, size) * rfft(b, size), size)[:n+m-1]

def objective_normalized(a):
    """2n * max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

def full_objective(a):
    """2n * max(conv) / sum^2."""
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def solve_lp(f, M):
    """
    Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0
    """
    n = len(f)
    # Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n
    # Actually, conv(f,g)[k] = sum_j f[k-j]*g[j] for valid j
    # So C[k,j] = f[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency, build as a sparse-like structure
    # But for n=256, dense is fine
    k_max = 2 * n - 2  # max index of convolution
    C = np.zeros((k_max + 1, n))
    for k in range(k_max + 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            C[k, j] = f[k - j]
    
    # c vector for objective: maximize sum(g) = -minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: C @ g <= M
    b_ub = np.full(k_max + 1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        return np.ones(n) / n  # fallback
    
def lp_iteration(a, num_iters=50, alpha=0.3):
    """Run LP-based iterations."""
    n = len(a)
    a = a / np.sum(a)  # normalize
    current = a.copy()
    
    for it in range(num_iters):
        c = fft_convolve(current, current)
        M = np.max(c)
        
        g = solve_lp(current, M)
        g = g / np.sum(g)  # normalize
        
        # Interpolate

        a_new = np.maximum(a_new, 0)

            a_try = a_try / np.sum(a_try)
            if objective_normalized(a_try) < score_old:
                current = a_try
            else:
                break  # stuck
    
    return current

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def local_score(a):
    n = len(a)

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_normalized(a):
    """Score for sum(a)=1: 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def full_score(a):
    """Full score: 2n * max(conv) / sum^2."""
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def build_conv_matrix(f):
    """Build matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def lp_min_peak(g, n=None):
    """
    Given fixed g, find f minimizing max_k conv(f,g)[k] s.t. sum(f)=1, f>=0.
    Returns (f, t_optimal).
    """
    if n is None:
        n = len(g)
    # conv(f,g)[k] = sum_j f_j * g[k-j] for valid j
    # So C[k,j] = g[k-j] if 0<=k-j<n
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = g

    # Variables: [f_0, ..., f_{n-1}, t]
    # Minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t

    # Constraints: C @ f - t <= 0  =>  C @ f + [-1]*t <= 0
    A_ub = np.hstack([C, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(f) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success:
        f = res.x[:n]
        t = res.x[n]
        f = np.maximum(f, 0)
        f = f / np.sum(f)
        return f, t
    else:
        return np.ones(n) / n, np.inf


def qp_best_response(f, M, mu=0.0):
    """
    Solve: min g^T g + mu*||g-f||^2 s.t. conv(f,g) <= M, sum(g)=1, g>=0.
    """
    n = len(f)
    if not HAS_CVXPY:
        return np.ones(n) / n

    g = cp.Variable(n)
    C = build_conv_matrix(f)

    if mu > 0:
        P = (1.0 + mu) * np.eye(n)
        q = -2.0 * mu * f
    else:
        P = np.eye(n)
        q = np.zeros(n)

    obj = cp.Minimize(0.5 * cp.quad_form(g, P) + q @ g)
    constraints = [C @ g <= M, cp.sum(g) == 1, g >= 0]

    try:
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            g_val = g_val / np.sum(g_val)
            return g_val
    except Exception:
        pass

    # Fallback: scipy
    def qp_obj(gv):
        if mu > 0:
            return (1.0 + mu) * np.dot(gv, gv) - 2.0 * mu * np.dot(f, gv)
        return np.dot(gv, gv)

    def qp_grad(gv):
        if mu > 0:
            return 2.0 * (1.0 + mu) * gv - 2.0 * mu * f
        return 2.0 * gv

    cons = [{'type': 'eq', 'fun': lambda gv: np.sum(gv) - 1.0}]
    for k in range(2 * n - 1):
        row = C[k]
        cons.append({'type': 'ineq', 'fun': lambda gv, row=row: M - np.dot(row, gv)})

    bounds = [(0, None)] * n
    x0 = np.ones(n) / n
    res = minimize(qp_obj, x0, jac=qp_grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': 200})
    if res.success:
        g_val = np.maximum(res.x, 0)
        return g_val / np.sum(g_val)
    return x0

def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    cp_ = c ** (p - 1)
    sum_cp = np.sum(cp_ * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp_)
    grad_a = factor * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def fourth_moment_obj_grad(theta):

def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)

    a = np.clip(a, 0.01, None)

        print(f"  4th-moment L-BFGS: score={s_new:.6f} ({time.monotonic()-t3:.0f}s)")
        if s_new < best_score:
            best_score = s_new
            best_seq = a_new.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 4: L-BFGS with p-norm annealing on best
    print(f"\n=== Phase 4: p-norm L-BFGS annealing ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        n = len(best_seq)
        theta = np.log(best_seq + 1e-30)
        p_schedule = [4, 8, 16, 32, 64, 128, 256]
        theta_best, s_best = run_lbgfs(theta, pnorm_obj_grad, p_schedule,
                                       maxiter=300, deadline=deadline - 150)
        a_best = softmax(theta_best)
        print(f"  p-norm L-BFGS: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 5: Try other lengths with quick optimization
    print(f"\n=== Phase 5: Multi-length exploration ===")
    other_lengths = [128, 192, 256, 320, 384, 512]
    if best_seq is not None:
        other_lengths = [l for l in other_lengths if l != len(best_seq)]

    for n_try in other_lengths:
        if time.monotonic() > deadline - 150:
            break
        # Initialize from best (resampled) or fresh
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / np.sum(a_init)
        else:
            a_init = make_chirp_init(n_try, 0.3, rng)

        theta = np.log(a_init + 1e-30)
        # Quick p-norm optimization
        theta_best, s = run_lbgfs(theta, pnorm_obj_grad, [8, 32, 128],
                                  maxiter=150, deadline=deadline - 100)
        a_candidate = softmax(theta_best)
        print(f"  n={n_try}: score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = a_candidate.copy()
            print(f"  *** NEW BEST at n={n_try}: {best_score:.6f} ***")

    # Phase 6: Final local search polish
    print(f"\n=== Phase 6: Local search polish ===")
    if best_seq is not None:
        polish_time = min(300.0, max(30.0, deadline - time.monotonic() - 20))
        print(f"  Polish time budget: {polish_time:.0f}s, current best: {best_score:.6f}")
        polished, p_score = local_search_polish(best_seq, polish_time, rng)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  After polish: {p_score:.6f} (no improvement, keeping {best_score:.6f})")

    # Phase 7: One more round of LP on the best if time permits
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\n=== Phase 7: Final LP refinement ===")
        n = len(best_seq)
        current = best_seq.copy()
        current = current / np.sum(current)
        cur_score = score_normalized(current)
        fin_t = time.monotonic()
        fin_budget = min(60.0, deadline - time.monotonic() - 20)
        it = 0
        while time.monotonic() - fin_t < fin_budget:
            c = fft_convolve(current, current)
            M = np.max(c)
            g, _ = lp_min_peak(current)
            for alpha_try in [0.2, 0.1, 0.05]:
                a_new = (1.0 - alpha_try) * current + alpha_try * g
                a_new = np.maximum(a_new, 0)
                a_new = a_new / np.sum(a_new)
                s_new = score_normalized(a_new)
                if s_new < cur_score:
                    current = a_new
                    cur_score = s_new
                    break
            if cur_score < best_score:
                best_score = cur_score
                best_seq = current.copy()
            it += 1
        print(f"  Final LP: {it} iters, score={best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
