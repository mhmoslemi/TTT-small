def linear_convolve_fftn(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n + m - 1]

def score_of(a):

    if s < 0.01:
        return np.inf
    conv = linear_convolve_fftn(a, a)

And a[k-j] = 0 if k-j < 0 or k-j >= n.

This is the max of:

- One quadratic function
- A constant

- conv[2j] + 2*a[j]*delta + delta^2 <= t
