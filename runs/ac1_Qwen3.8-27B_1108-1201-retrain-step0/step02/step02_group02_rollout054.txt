```python
"""
Multi-phase optimization for minimizing the autocorrelation peak score of non-negative step functions.
Combines LP-based Frank-Wolfe, L-BFGS with adaptive softmax temperature, and subgradient methods.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

# --- Helper Functions ---

def fft_convolve(a, b=None):
    """Compute linear convolution of two 1-D arrays using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def softmax(x):
    """Stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)

def compute_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (S * S)

def convolve_and_peak(a):
    """Compute convolution and return (convolution, peak_value, peak_index)."""
    c = fft_convolve(a)
    k_star = int(np.argmax(c))
    return c, c[k_star], k_star

def softmax_score_theta(theta):
    """Score for theta parameterization (a = softmax(theta))."""
    a = softmax(theta)
    return compute_score(a)

def softmax_grad_theta(theta, tau=None):
    """
    Compute gradient of the (soft)max of conv(softmax(theta), softmax(theta)) w.r.t. theta.
    If tau is None, uses exact subgradient at the peak.
    If tau > 0, uses soft-max approximation.
    Returns (value, gradient).
    """
    n = len(theta)
    p = softmax(theta)
    c = fft_convolve(p, p)
    L = len(c)  # 2n-1
    
    if tau is not None and tau > 0:
        # Soft-max approximation
        c_scaled = c / tau
        c_max = np.max(c_scaled)
        exp_c = np.exp(c_scaled - c_max)
        sum_exp = np.sum(exp_c)
        w = exp_c / sum_exp
        value = tau * (c_max + np.log(sum_exp))
        
        # Cross-correlation: cross[j] = sum_k w[k] * p[k-j] for valid k-j
        p_rev = p[::-1]
        conv_wp = fft_convolve(w, p_rev)  # length L + n - 1 = 3n-2
        cross = conv_wp[n-1:2*n-1]  # length n
        
        c_w = np.dot(w, c)
        grad_theta = 2.0 * p * (cross - c_w)
    else:
        # Exact subgradient
        k_star = int(np.argmax(c))
        value = c[k_star]
        
        # d(conv[k_star])/dp[j] = 2*p[k_star-j] if 0<=k_star-j<n
        grad_p = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(k_star, n - 1)
        for j in range(j_min, j_max + 1):
            grad_p[j] = 2.0 * p[k_star - j]
        
        # d(conv[k_star])/dtheta[j] = p[j] * (grad_p[j] - dot(grad_p, p))
        inner = np.dot(grad_p, p)
        grad_theta = p * (grad_p - inner)
    
    return value, grad_theta

def lp_frank_wolfe(f, max_iters=100, time_limit=60.0):
    """
    LP-based Frank-Wolfe optimization.
    f: current normalized sequence (sum=1, non-negative)
    Returns improved sequence.
    """
    n = len(f)
    f_cur = f.astype(np.float64).copy()
    S = np.sum(f_cur)
    f_cur = f_cur / S  # normalize to sum=1
    
    c_cur, peak_cur, _ = convolve_and_peak(f_cur)
    best = f_cur.copy()
    best_score = peak_cur
    
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = peak_cur  # current peak
        
        # Build constraint matrix: A[k,j] = f_cur[k-j] if 0<=k-j<n
        # A is (2n-1) x n
        # Each row k: A[k, :] = f_cur[max(0,k-n+1):min(k+1,n)][::-1] padded
        
        # Build A efficiently
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_start <= j_end:
                A[k, j_start:j_end+1] = f_cur[k-j_end:k-j_start+1][::-1]
        
        # Solve LP: maximize sum(b) s.t. A @ b <= M, b >= 0
        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), 
                          bounds=[(0, None)]*n, method='highs',
                          options={'presolve': True, 'time_limit': 5.0})
        except Exception:
            break
        
        if not res.success:
            break
        
        b = res.x
        sum_b = np.sum(b)
        if sum_b < 1e-10:
            break
        
        # Rescale to sum=1
        g = b / sum_b
        
        # Line search for step size
        best_t = 0.0
        best_new_score = peak_cur
        for t in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            f_new = (1 - t) * f_cur + t * g
            f_new = np.clip(f_new, 0, None)
            S_new = np.sum(f_new)
            if S_new < 1e-10:
                continue
            f_new = f_new / S_new
            _, new_peak, _ = convolve_and_peak(f_new)
            if new_peak < best_new_score:
                best_new_score = new_peak
                best_t = t
        
        if best_t > 0:
            f_cur = (1 - best_t) * f_cur + best_t * g
            f_cur = np.clip(f_cur, 0, None)
            f_cur = f_cur / np.sum(f_cur)
            c_cur, peak_cur, _ = convolve_and_peak(f_cur)
            
            if peak_cur < best_score:
                best_score = peak_cur
                best = f_cur.copy()
        
        # Update M for next iteration
        # (peak_cur is already updated)
    
    return best, best_score

def lbfgs_optimize(theta_init, tau_start, tau_end, time_limit, n):
    """
    L-BFGS optimization with decreasing soft-max temperature.
    theta_init: initial theta (log-softmax parameter)
    tau_start, tau_end: temperature range
    time_limit: time in seconds
    Returns best theta and score.
    """
    t0 = time.monotonic()
    theta = theta_init.astype(np.float64).copy()
    best_theta = theta.copy()
    best_score = compute_score(softmax(theta))
    
    n_steps = 0
    stuck_count = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Exponential sweep of temperature
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        def obj_func(th):
            return compute_score(softmax(th))
        
        def obj_grad(th):
            if tau > 1e-6:
                _, g = softmax_grad_theta(th, tau=tau)
            else:
                _, g = softmax_grad_theta(th)
            return g
        
        # Run L-BFGS for a limited number of iterations
        try:
            res = minimize(obj_func, theta, jac=obj_grad, method='L-BFGS-B',
                          options={'maxiter': 50, 'gtol': 1e-6, 'maxls': 20})
            if res.success or res.fun < best_score:
                new_score = res.fun
                if new_score < best_score:
                    best_score = new_score
                    best_theta = res.x.copy()
                    stuck_count = 0
                else:
                    stuck_count += 1
                theta = res.x.copy()
        except Exception:
            stuck_count += 1
        
        n_steps += 1
        
        # If stuck, try a perturbation
        if stuck_count >= 3:
            theta = best_theta + 0.1 * np.random.standard_normal(n)
            stuck_count = 0
        
        if n_steps % 10 == 0:
            el = time.monotonic() - t0
            print(f"      L-BFGS: step={n_steps}, tau={tau:.4f}, best={best_score:.8f}, t={el:.1f}s")
    
    return best_theta, best_score

def subgradient_polish(a_init, time_limit):
    """
    Subgradient descent with backtracking line search on the exact objective.
    Operates directly on the sequence (not theta).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    a = a / S  # normalize to sum=1
    
    _, cur_peak, _ = convolve_and_peak(a)
    best = a.copy()
    best_peak = cur_peak
    
    t0 = time.monotonic()
    lr = 0.01
    
    while time.monotonic() - t0 < time_limit:
        c, peak, k_star = convolve_and_peak(a)
        
        # Subgradient
        grad = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(k_star, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] = 2.0 * a[k_star - j]
        
        # Projected gradient step (maintain sum=1, a>=0)
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0, None)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        
        _, new_peak, _ = convolve_and_peak(a_new)
        
        if new_peak < cur_peak:
            a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best = a.copy()
            # Increase learning rate slightly
            lr = min(lr * 1.05, 0.1)
        else:
            # Decrease learning rate
            lr *= 0.7
            if lr < 1e-8:
                break
    
    return best, best_peak
```

Actually, I think the key insight I'm missing is: **what makes a good sequence for this problem?**

Let me think about it from first principles. We want to minimize max_k conv(a,a)[k] subject to sum(a)=1, a>=0.

The convolution at lag k measures the "overlap" of a with its k-shifted version. We want all overlaps to be small.

This is like wanting a sequence that "spreads out" its mass so that no shift gives a large overlap.

One approach: think of it as a "discrete spectrum" problem. The convolution is related to the Fourier transform:
conv(a,a) = IFFT(|FFT(a)|^2)

So max(conv(a,a)) is the peak of the power spectrum (in the time domain). We want to minimize this peak.

By uncertainty principles, if a is concentrated in time (a few large values), its Fourier transform is spread out, and the peak of |FFT(a)|^2 is small. But then conv(a,a) (which is the autocorrelation) has a small peak... wait, no. conv(a,a) is in the "time" domain, and its peak corresponds to the maximum autocorrelation.

Actually, I think the right way to think about it is:
- conv(a,a)[0] = a[0]^2 (small if a[0] is small)
- conv(a,a)[n-1] = sum_i a[i]*a[n-1-i] (the "center" value)
- We want ALL of these to be small

The center value is the most "dangerous" because it has the most terms (n terms). So we particularly want to minimize the center value.

For the center: sum_i a[i]*a[n-1-i]. This is the dot product of a with its reverse. By Cauchy-Schwarz, this is at least (sum a)^2 / (2n-1) ... no, that's not right.

By the rearrangement inequality, for fixed sum, the dot product of a with its reverse is minimized when a is "antimonotone" relative to its reverse. That is, a is increasing and a_reversed is also increasing (or one is increasing and the other decreasing).

If a is increasing (a[0] < a[1] < ... < a[n-1]), then a_reversed is decreasing. The dot product of an increasing sequence with a decreasing sequence is minimized (by the rearrangement inequality). So an increasing sequence minimizes the center convolution!

But an increasing sequence might have large values at other lags. For example, conv(a,a)[n-2] = sum_i a[i]*a[n-2-i]. For an increasing sequence, this might still be large.
