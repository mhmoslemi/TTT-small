def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def blame_transfer(a0, time_limit, top_k=5, n_trials=100):
    """
    Blame-based mass transfer: move mass from high-blame to low-blame positions.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    t_start = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - t_start < time_limit:
        # Batch of trials
        for _ in range(n_trials):
            if time.monotonic() - t_start > time_limit:
                break
            
        # Compute blame
        t_smooth = 5.0
        blame = compute_blame(conv, a, top_k=top_k, t_smooth=t_smooth)
        
        # Find high-blame and low-blame positions
        # High blame: positions that contribute to peaks
        # Low blame: positions that don't
        mean_blame = blame.mean()
        high_mask = blame > mean_blame * 1.5
        low_mask = blame < mean_blame * 0.5
        
        high_idx = np.where(high_mask)[0]
        low_idx = np.where(low_mask)[0]
        
        if len(high_idx) == 0 or len(low_idx) == 0:
            # Fallback: use top and bottom quartile
            sorted_idx = np.argsort(blame)
            q = max(1, n // 4)
            low_idx = sorted_idx[:q]
            high_idx = sorted_idx[-q:]
        
        # Choose source (high blame, significant value) and target (low blame)
        high_vals = a[high_idx]
        if high_vals.max() < 1e-8:
            break
        
        # Weighted random selection
        source_weights = a[high_idx]
        source_weights /= source_weights.sum()
        src = high_idx[np.random.choice(len(high_idx), p=source_weights)]
        
        target_weights = np.ones(len(low_idx))
        target = low_idx[np.random.choice(len(low_idx))]
        
        if src == target:
            continue
        
        # Transfer amount
        frac = np.random.uniform(0.02, 0.08)
        delta = a[src] * frac
        if delta < 1e-10:
            continue
        
        a[src] -= delta
        a[target] += delta
        
        if a[src] < 0:
            a[src] += delta
            a[target] -= delta
            continue
        
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        
        if new_obj < cur_obj:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            # Reject (no annealing for speed)
            a[src] += delta
            a[target] -= delta
        
        iter_count += 1
        if iter_count % 500 == 0:
            print("    BT iter %d: obj=%.6f" % (iter_count, best_obj))
    
    return best_a, best_obj

def generate_inits(n, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Raised cosine (symmetric)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    rc /= rc.sum()
    inits['raised_cos'] = rc
    
    # Gaussian (centered)
    for sigma in [0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - n/2) / (sigma * n))**2)
        g /= g.sum()
        inits['gauss_%.1f' % sigma] = g
    
    # Asymmetric Gaussian (shifted)
    for shift in [0.3, 0.4, 0.6, 0.7]:
        g = np.exp(-0.5 * ((x - shift * n) / (0.3 * n))**2)
        g /= g.sum()
        inits['asym_gauss_%.1f' % shift] = g
    
    # Hamming window
    hw = np.sin(np.pi * (x + 0.5) / n) ** 2
    hw /= hw.sum()
    inits['hamming'] = hw
    
    # Bimodal
    for w_frac in [0.2, 0.3]:
        w = int(w_frac * n)
        b = np.zeros(n)
        b[:w] = 1.0
        b[-w:] = 1.0
        b /= b.sum()
        inits['bimodal_%.1f' % w_frac] = b
    
    # Ramp (asymmetric)
    ramp = np.linspace(0.1, 1.0, n)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Parent (if available)
    if parent is not None:
        p = np.maximum(parent, 0.0)
        if p.size >= n:
            p = p[:n]
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        p = np.maximum(p, 0)
        if p.sum() > 1e-10:
            p /= p.sum()
            inits['parent'] = p
    
    # Cosine with asymmetry
    for c in [0.3, 0.5, 0.7]:
        cos_a = 0.5 * (1 + np.cos(2 * np.pi * (x - c * n) / n))
        cos_a = np.maximum(cos_a, 0)
        if cos_a.sum() > 1e-10:
            cos_a /= cos_a.sum()
            inits['cos_asym_%.1f' % c] = cos_a
    
    return inits

    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_value(a):
    """Compute 2*n*max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def safe_evaluate(a):
    """Safely call the evaluation function."""
    try:
        a_clean = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a_clean = np.minimum(a_clean, 1000.0)
        s = a_clean.sum()
        if s < 0.01:
            a_clean = a_clean / max(s, 1e-15)
        if not np.all(np.isfinite(a_clean)):
            return float('inf')
        return float(evaluate_sequence(a_clean.tolist()))
    except Exception:
        return float('inf')


def softmax_subgradient(conv, a, t):
    """Compute softmax-weighted subgradient of max(conv)."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = t * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    # Only process significant weights
    thresh = max(1e-6, 0.5 / len(conv))
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    return sg


def mirror_descent_nesterov(a0, time_limit, t_init=10.0, t_final=500.0,
                            lr_init=0.08, momentum_init=0.9):
    """
    Nesterov-accelerated mirror descent in log-space.
    Uses adaptive temperature schedule from t_init to t_final.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    # Log-space parameterization
    log_a = np.log(np.maximum(a, 1e-30))
    y_log = log_a.copy()  # Nesterov lookahead in log-space

    lr = lr_init
    beta = momentum_init
    t_smooth = t_init

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    patience = 0
    max_pat = 40
    t_start = time.monotonic()
    step = 0

    while time.monotonic() - t_start < time_limit:
        elapsed = time.monotonic() - t_start
        frac = min(1.0, elapsed / max(time_limit, 1.0))

        # Adaptive temperature: exponential schedule
        t_smooth = t_init * (t_final / t_init) ** frac
        # Adaptive learning rate: decay over time
        lr = lr_init * (1.0 - 0.7 * frac)
        # Decay momentum
        beta = momentum_init * (1.0 - 0.3 * frac)

        # Compute subgradient at lookahead point
        y = np.exp(y_log - y_log.max())
        y = y / y.sum()
        conv_y = fast_convolve(y)
        sg = softmax_subgradient(conv_y, y, t_smooth)

        # Center the gradient (remove mean component for simplex)
        sg_centered = sg - np.dot(y, sg)

        # Nesterov update in log-space
        # da/dx = a*(e_i - a), so dx = a*(g - a@g)
        log_a_new = y_log - lr * sg_centered
        # Prevent overflow/underflow
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            a_new = y.copy()
            log_a_new = np.log(np.maximum(a_new, 1e-30))
        else:
            a_new /= a_sum
            log_a_new = np.log(np.maximum(a_new, 1e-30))

        # Nesterov extrapolation
        y_log_new = log_a_new + beta * (log_a_new - log_a)
        a = a_new
        log_a = log_a_new
        y_log = y_log_new

        # Evaluate
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_pat:
            patience = 0
            lr *= 0.6
            if lr < 0.002:
                # Reset: restart momentum
                y_log = log_a.copy()
                lr = lr_init * 0.4 * (1.0 - 0.5 * frac)
            # Increase temperature to escape
            t_smooth = min(t_smooth * 1.3, t_final)

        step += 1

    return best_a, best_obj


def blame_peak_shave(a0, time_limit, n_inner=200):
    """
    Deterministic blame-based peak shaving.
    At each step: identify worst peak, compute blame, transfer mass
    from highest-blame to lowest-blame position.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t_start = time.monotonic()
    total_iters = 0

    while time.monotonic() - t_start < time_limit:
        for _ in range(n_inner):
            if time.monotonic() - t_start > time_limit:
                break

            # Find the worst peak
            peak_idx = int(np.argmax(conv))
            max_val = conv[peak_idx]
            if max_val < 1e-15:
                break

            # Compute blame for this peak: blame[i] = a[peak_idx - i] for valid i
            js = max(0, peak_idx - n + 1)
            je = min(n - 1, peak_idx)
            blame = np.zeros(n)
            if js <= je:
                blame[js:je + 1] = a[peak_idx - je:peak_idx - js + 1][::-1]

            # Find source: high blame AND high value (most "responsible")
            blame_weighted = blame * a
            # Find target: low blame (barely contributes to peaks)
            # Use a broader set of peaks for target computation
            top_peaks = np.argsort(conv)[::-1][:5]
            low_blame = np.ones(n)
            for pk in top_peaks:
                pk = int(pk)
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                if js2 <= je2:
                    low_blame[js2:je2 + 1] += a[pk - je2:pk - js2 + 1][::-1]

            # Source: argmax of blame_weighted
            src = int(np.argmax(blame_weighted))
            # Target: argmin of low_blame (among positions with some mass)
            min_blame = low_blame.min()
            candidates = np.where(low_blame <= min_blame * 1.1 + 1e-12)[0]
            if len(candidates) == 0:
                candidates = np.array([int(np.argmin(low_blame))])
            tgt = int(candidates[np.argmin(a[candidates])]) if len(candidates) > 1 else int(candidates[0])

            if src == tgt:
                # Try next worst peak
                conv_alt = conv.copy()
                conv_alt[peak_idx] = 0
                if conv_alt.max() < 1e-15:
                    break
                peak_idx = int(np.argmax(conv_alt))
                js = max(0, peak_idx - n + 1)
                je = min(n - 1, peak_idx)
                blame = np.zeros(n)
                if js <= je:
                    blame[js:je + 1] = a[peak_idx - je:peak_idx - js + 1][::-1]
                blame_weighted = blame * a
                src = int(np.argmax(blame_weighted))
                if src == tgt:
                    continue

            # Transfer fraction of source mass
            frac = min(0.05, 0.01 + 0.03 * (max_val / conv.mean() if conv.mean() > 0 else 1))
            delta = a[src] * frac
            if delta < 1e-12:
                continue

            a[src] -= delta
            a[tgt] += delta

            if a[src] < 0:
                a[src] += delta
                a[tgt] -= delta
                continue

            new_conv = fast_convolve(a)
            new_obj = 2.0 * n * new_conv.max()

            if new_obj < cur_obj:
                cur_obj = new_obj
                conv = new_conv
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
            else:
                a[src] += delta
                a[tgt] -= delta

            total_iters += 1

        if total_iters % 1000 == 0 and total_iters > 0:
            print("      BPS iter %d: obj=%.6f" % (total_iters, best_obj))

    return best_a, best_obj


def stochastic_mirror_descent(a0, time_limit, t_init=5.0, lr_init=0.1):
    """
    Stochastic mirror descent: sample a single peak and use its
    subgradient. Cheaper per step, allows more steps in the time budget.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t_start = time.monotonic()
    step = 0
    patience = 0
    max_pat = 60

    while time.monotonic() - t_start < time_limit:
        elapsed = time.monotonic() - t_start
        frac = min(1.0, elapsed / max(time_limit, 1.0))
        lr = lr_init * (1.0 - 0.6 * frac)

        # Find top peaks and sample one
        sorted_idx = np.argsort(conv)[::-1]
        top_k = min(5, len(conv))
        top_vals = conv[sorted_idx[:top_k]]
        weights = top_vals / top_vals.sum()
        chosen = int(sorted_idx[np.random.choice(top_k, p=weights)])

        # Subgradient for single peak
        sg = np.zeros(n)
        js = max(0, chosen - n + 1)
        je = min(n - 1, chosen)
        if js <= je:
            sg[js:je + 1] = a[chosen - je:chosen - js + 1][::-1]

        # Center
        sg_c = sg - np.dot(a, sg)

        # Mirror descent step (multiplicative)
        log_factor = -lr * sg_c
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.ones(n) / n
        else:
            a /= a_sum

        # Evaluate (every few steps to save time)
        if step % 3 == 0:
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-10:
                best_obj = cur_obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

            if patience >= max_pat:
                patience = 0
                lr *= 0.7
                if lr < 0.005:
                    lr = lr_init * 0.3 * (1.0 - 0.5 * frac)

        step += 1

    return best_a, best_obj


def generate_inits(n, parent=None):
    """Generate diverse initializations with emphasis on asymmetric shapes."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform baseline
    inits['uniform'] = np.ones(n) / n

    # Raised cosine (symmetric - good baseline)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    rc /= rc.sum()
    inits['raised_cos'] = rc

    # Asymmetric raised cosine (shifted peak)
    for c in [0.3, 0.35, 0.4, 0.6, 0.65, 0.7]:
        rc_a = 0.5 * (1 + np.cos(2 * np.pi * (x - c * n) / n))
        rc_a = np.maximum(rc_a, 0.0)
        if rc_a.sum() > 1e-10:
            rc_a /= rc_a.sum()
            inits['rc_asym_%.2f' % c] = rc_a

    # Gaussian with varying centers and widths
    for center in [0.25, 0.35, 0.5, 0.65, 0.75]:
        for sigma in [0.2, 0.3, 0.4]:
            g = np.exp(-0.5 * ((x - center * n) / (sigma * n)) ** 2)
            if g.sum() > 1e-10:
                g /= g.sum()
                inits['g_%.2f_%.1f' % (center, sigma)] = g

    # Linear ramp (strongly asymmetric)
    ramp = np.linspace(0.05, 1.0, n)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Inverse ramp
    inv_ramp = np.linspace(1.0, 0.05, n)
    inv_ramp /= inv_ramp.sum()
    inits['inv_ramp'] = inv_ramp

    # Bimodal
    for w_frac in [0.15, 0.25, 0.35]:
        w = max(2, int(w_frac * n))
        b = np.zeros(n)
        b[:w] = 1.0
        b[-w:] = 1.0
        b /= b.sum()
        inits['bi_%.2f' % w_frac] = b

    # Exponential decay
    for rate in [0.5, 1.0, 2.0]:
        exp_d = np.exp(-rate * x / n)
        exp_d /= exp_d.sum()
        inits['exp_%.1f' % rate] = exp_d

    # Quadratic (parabola)
    q = 1.0 - ((x - n / 2) / (n / 2)) ** 2
    q = np.maximum(q, 0.0)
    if q.sum() > 1e-10:
        q /= q.sum()
        inits['quad'] = q

    # Asymmetric quadratic
    for c in [0.3, 0.4, 0.7, 0.8]:
        q_a = 1.0 - ((x - c * n) / (n / 2)) ** 2
        q_a = np.maximum(q_a, 0.0)
        if q_a.sum() > 1e-10:
            q_a /= q_a.sum()
            inits['quad_%.1f' % c] = q_a

    # Parent sequence
    if parent is not None:
        p = np.maximum(parent, 0.0)
        if p.size >= n:
            p_use = p[:n]
        else:
            p_use = np.interp(np.linspace(0, 1, n),
                              np.linspace(0, 1, p.size), p)
        p_use = np.maximum(p_use, 0.0)
        if p_use.sum() > 1e-10:
            p_use /= p_use.sum()
            inits['parent'] = p_use
        # Perturbed parent
        noise = np.random.default_rng(123).normal(0, 0.1, n)
        p_pert = p_use * (1 + noise)
        p_pert = np.maximum(p_pert, 0.0)
        if p_pert.sum() > 1e-10:
            p_pert /= p_pert.sum()
            inits['parent_pert'] = p_pert

    return inits


def get_parent():
    """Access the parent sequence from global variable."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point: multi-phase optimization using Nesterov mirror descent,
    blame-based peak shaving, and stochastic mirror descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)

    parent = get_parent()

    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        score = safe_evaluate(a_norm)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_norm.copy()
            print("  *** NEW BEST: score=%.6f, n=%d, t=%.0fs" %
                  (score, len(a), time.monotonic() - t_start))

    # ===== Phase 1: Broad exploration =====
    print("Phase 1: Broad exploration...")
    scan_sizes = [128, 192, 256, 320, 384, 512, 640, 768, 1024]
    scan_results = []
    phase1_budget = min(500.0, (deadline - time.monotonic()) * 0.22)

    for n in scan_sizes:
        if time.monotonic() > deadline - 1500:
            break
        inits = generate_inits(n, parent)
        n_inits = len(inits)
        time_per = max(1.5, phase1_budget / max(1, len(scan_sizes) * n_inits))

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 1500:
                break
            t_alloc = min(time_per, deadline - time.monotonic() - 1500)
            if t_alloc < 1.0:
                break
            # Quick stochastic mirror descent
            a_opt, _ = stochastic_mirror_descent(a0, time_limit=t_alloc,
                                                 t_init=5.0, lr_init=0.1)
            score = safe_evaluate(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt))
            record(a_opt)

    print("  Phase 1 done. Best: %.6f" % best_score)
    scan_results.sort(key=lambda x: x[0])

    # ===== Phase 2: Deep optimization =====
    print("Phase 2: Deep optimization...")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 15:
            top_configs.append((n, name, a_opt))
            seen.add(key)

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'overall_best', best_sequence.copy()))

    phase2_budget = min(1300.0, (deadline - time.monotonic()) * 0.48)
    n_deep = len(top_configs)
    time_per_deep = max(10.0, phase2_budget / max(1, n_deep))

    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 500:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 500)
        if t_alloc < 5.0:
            break
        print("  [%d/%d] n=%d %s budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))

        a_cur = a0.copy()

        # Multi-stage: coarse to fine
        # Stage 1: Coarse (low temperature, large steps)
        sub = t_alloc * 0.2
        a_cur, _ = mirror_descent_nesterov(a_cur, sub,
                                           t_init=5.0, t_final=30.0,
                                           lr_init=0.1, momentum_init=0.9)
        record(a_cur)

        # Stage 2: Medium
        if time.monotonic() < deadline - 500:
            sub2 = min(t_alloc * 0.2, deadline - time.monotonic() - 500)
            if sub2 > 2:
                a_cur, _ = blame_peak_shave(a_cur, sub2, n_inner=100)
                record(a_cur)

        # Stage 3: Fine (higher temperature, smaller steps)
        if time.monotonic() < deadline - 500:
            sub3 = min(t_alloc * 0.25, deadline - time.monotonic() - 500)
            if sub3 > 2:
                a_cur, _ = mirror_descent_nesterov(a_cur, sub3,
                                                   t_init=30.0, t_final=200.0,
                                                   lr_init=0.04, momentum_init=0.85)
                record(a_cur)

        # Stage 4: Peak shaving
        if time.monotonic() < deadline - 500:
            sub4 = min(t_alloc * 0.2, deadline - time.monotonic() - 500)
            if sub4 > 2:
                a_cur, _ = blame_peak_shave(a_cur, sub4, n_inner=200)
                record(a_cur)

        # Stage 5: Final fine mirror descent
        if time.monotonic() < deadline - 500:
            sub5 = min(t_alloc * 0.15, deadline - time.monotonic() - 500)
            if sub5 > 1:
                a_cur, _ = mirror_descent_nesterov(a_cur, sub5,
                                                   t_init=200.0, t_final=1000.0,
                                                   lr_init=0.015, momentum_init=0.8)
                record(a_cur)

    print("  Phase 2 done. Best: %.6f" % best_score)

    # ===== Phase 3: Intensive local refinement =====
    if best_sequence is not None and time.monotonic() < deadline - 40:
        print("Phase 3: Intensive refinement...")
        remaining = min(500.0, deadline - time.monotonic() - 40)
        n_rounds = 5
        per_round = remaining / n_rounds

        for r in range(n_rounds):
            if time.monotonic() > deadline - 40:
                break

            a_cur = best_sequence.copy()

            # Alternate methods each round
            if r % 2 == 0:
                # NSG with very high temperature (exact subgradient)
                a_cur, _ = mirror_descent_nesterov(a_cur, per_round * 0.5,
                                                   t_init=500.0, t_final=5000.0,
                                                   lr_init=0.008, momentum_init=0.7)
                record(a_cur)
                if time.monotonic() < deadline - 40:
                    a_cur, _ = blame_peak_shave(a_cur, per_round * 0.5, n_inner=300)
                    record(a_cur)
            else:
                # Stochastic MD for diversity, then NSG polish
                a_cur, _ = stochastic_mirror_descent(a_cur, per_round * 0.4,
                                                     t_init=3.0, lr_init=0.06)
                record(a_cur)
                if time.monotonic() < deadline - 40:
                    a_cur, _ = mirror_descent_nesterov(a_cur, per_round * 0.6,
                                                       t_init=100.0, t_final=500.0,
                                                       lr_init=0.01, momentum_init=0.85)
                    record(a_cur)

    print("  Phase 3 done. Best: %.6f" % best_score)

    # ===== Phase 4: Final polish =====
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("Phase 4: Final polish...")
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_cur = best_sequence.copy()
            # Very fine NSG
            a_cur, _ = mirror_descent_nesterov(a_cur, rem * 0.5,
                                               t_init=1000.0, t_final=10000.0,
                                               lr_init=0.004, momentum_init=0.7)
            record(a_cur)

            if time.monotonic() < deadline - 5:
                rem2 = deadline - time.monotonic() - 5
                if rem2 > 3:
                    a_cur, _ = blame_peak_shave(a_cur, rem2, n_inner=500)
                    record(a_cur)

            if time.monotonic() < deadline - 5:
                rem3 = deadline - time.monotonic() - 5
                if rem3 > 2:
                    a_cur, _ = mirror_descent_nesterov(a_cur, rem3,
                                                       t_init=5000.0, t_final=50000.0,
                                                       lr_init=0.002, momentum_init=0.6)
                    record(a_cur)

    # Fallback
    if best_sequence is None:
        print("WARNING: No valid solution found, using fallback.")
        best_sequence = np.ones(256)
        best_score = safe_evaluate(best_sequence)

    print("FINAL: score=%.6f, n=%d, time=%.0fs" %
          (best_score, len(best_sequence), time.monotonic() - t_start))
    return [float(x) for x in best_sequence]
