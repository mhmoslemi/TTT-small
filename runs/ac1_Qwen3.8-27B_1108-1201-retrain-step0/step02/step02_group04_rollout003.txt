Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses smoothed max (log-sum-exp) gradient descent + LP-based Frank-Wolfe refinement.
"""

import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import linprog
from scipy.signal import correlate
import time

def compute_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fftconvolve(a, a, mode='full')
    return 2.0 * n * np.max(conv) / (s**2)

def optimize_smoothed_grad(a_init, time_limit, rng, tau_init=0.005, verbose=True):
    """
    Smoothed max gradient descent with annealing of tau.
    a is normalized to sum=1.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    a = a / S
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    tau = tau_init
    lr = 0.05
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fftconvolve(a, a, mode='full')
        
        # Softmax weights (numerically stable)
        max_conv = np.max(conv)
        exp_conv = np.exp((conv - max_conv) / tau)
        w = exp_conv / np.sum(exp_conv)
        
        # Gradient: g[j] = 2 * correlate(w, a, 'valid')[j]
        g = 2.0 * correlate(w, a, mode='valid')
        
        # Gradient step with normalization
        a_new = a - lr * g
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a  # revert
        else:
            a_new = a_new / s
        
        new_score = compute_score(a_new)
        
        it += 1
        if new_score < best_score - 1e-10:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slightly increase lr
            lr = min(lr * 1.01, 0.5)
        else:
            a = a_new if new_score < compute_score(a) else a
            stall += 1
            lr *= 0.98
            
            if stall > 200:
                # Anneal tau (make it smaller, more peaked)
                tau = max(tau * 0.9, 1e-6)
                stall = 0
                # Perturb
                a = best_a + 0.005 * rng.standard_normal(n)
                a = np.clip(a, 0, None)
                a = a / np.sum(a)
                lr = 0.05
        
        if verbose and it % 5000 == 0:
            el = time.monotonic() - t0
            print(f"    SG iter={it}, tau={tau:.6f}, lr={lr:.4f}, score={best_score:.8f}, elapsed={el:.1f}s")
    
    return best_a, best_score


def optimize_lp_based(a_init, time_limit, n_iters=1000, verbose=True):
    """
    LP-based Frank-Wolfe optimization.
    a is normalized to sum=1.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    a = a / S
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_aa = fftconvolve(a, a, mode='full')
        M = np.max(conv_aa)
        
        # Build constraint matrix A: (a*b)[k] = sum_j A[k,j] * b[j] <= M
        # A[k,j] = a[j] if 0 <= k-j < n else 0
        # i.e., for each k, A[k, j] = a[j] for j in [max(0,k-n+1), min(k, n-1)]
        
        # Use a banded matrix structure implicitly
        # For linprog, we need the full matrix or use sparse
        from scipy.sparse import lil_matrix
        A = lil_matrix((2*n-1, n))
        for k in range(2*n-1):
            j_start = max(0, k - (n-1))
            j_end = min(n-1, k)
            A[k, j_start:j_end+1] = a[j_start:j_end+1]
        
        # Solve: maximize sum(b) s.t. A@b <= M, b >= 0
        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A.tocsr(), b_ub=np.full(2*n-1, M), 
                         bounds=[(0, None)]*n, method='highs')
        except Exception as e:
            print(f"    LP error: {e}")
            break
        
        if res.success:
            b = res.x
            S_b = np.sum(b)
            if S_b > 1e-10:
                b = b / S_b
                # Frank-Wolfe update
                gamma = 2.0 / (it + 2)  # diminishing step size
                a_new = (1 - gamma) * a + gamma * b
                a_new = np.clip(a_new, 0, None)
                s = np.sum(a_new)
                if s > 0:
                    a_new = a_new / s
                    new_score = compute_score(a_new)
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a_new.copy()
                    a = a_new
        
        if verbose and it % 50 == 0:
            el = time.monotonic() - t0
            print(f"    LP iter={it}, score={best_score:.8f}, elapsed={el:.1f}s")
    
    return best_a, best_score
