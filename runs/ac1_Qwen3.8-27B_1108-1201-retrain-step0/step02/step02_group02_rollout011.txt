    # = sum_m w'[m]*a[m] where w'[m] = w[j+m]
    # This is equivalent to: pad a to length 2n-1, flip it, convolve with w.
    # Actually, let's use the property:
    # sum_m w[j+m]*a[m] = (w_convolve_a_flipped)[j + n - 1]
    # where w_convolve_a_flipped = convolve(w, a[::-1])
    
    a_flip = a[::-1]
    # convolve(w, a_flip)[k] = sum_i w[i]*a_flip[k-i] = sum_i w[i]*a[n-1-k+i]
    # We want: sum_m w[j+m]*a[m]
    # Let's set i = j+m, m = i-j:
    # = sum_i w[i]*a[i-j] where i ranges over j to j+n-1
    # = sum_i w[i]*a[i-j]
    # Now, a[i-j] = a_flip[n-1-(i-j)] = a_flip[n-1-i+j]
    # = sum_i w[i]*a_flip[n-1-i+j]
    # Let p = n-1-i+j, i = n-1+j-p:
    # = sum_p w[n-1+j-p]*a_flip[p]
    # = (w * a_flip)[n-1+j]... 
    # Convolution: (w * a_flip)[k] = sum_p w[p]*a_flip[k-p]
    # We have sum_p w[n-1+j-p]*a_flip[p]
    # Let q = n-1+j-p, p = n-1+j-q:
    # = sum_q w[q]*a_flip[n-1+j-q]
    # = (w * a_flip)[n-1+j]... 
    # (w * a_flip)[k] = sum_q w[q]*a_flip[k-q]
    # So with k = n-1+j: (w * a_flip)[n-1+j] = sum_q w[q]*a_flip[n-1+j-q]
    # Yes! So sum_m w[j+m]*a[m] = (w * a_flip)[n-1+j]
    
    w_aflip = fft_convolve(w, a_flip)
    grad = 4.0 * n * tau * w_aflip[n-1:2*n-1]
    
    return value, grad

def optimize_single(n, a_init, time_limit, rng, verbose=True):
    """
    Optimize a single sequence of length n.
    Uses adaptive log-sum-exp smoothing with gradient descent in direct space.
    """
    S = 1.0  # Normalize sum to 1
    a = project_simplex(a_init.copy(), S)
    current_score = score(a)
    best_a = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    step = 0
    tau = 0.01  # Initial temperature
    tau_min = 1e-5
    lr = 0.01
    no_improve = 0
    patience = 500
    
    if verbose:
        print(f"    Start: n={n}, score={current_score:.6f}")
    
    while time.monotonic() - t0 < time_limit:
        if step % 100 == 0:
            a = project_simplex(a, S)
            s = score(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 100
            
            # Anneal temperature
            if step > 1000:
                tau = max(tau_min, tau * 0.99)
            
            # Escape mechanism
            if no_improve > patience:
                r = rng.random()
                if r < 0.3:
                    # Gaussian perturbation
                    a = project_simplex(best_a + 0.01 * rng.standard_normal(n), S)
                elif r < 0.6:
                    # Peak-directed: reduce peak contributors
                    b = compute_b(best_a)
                    k_star = int(np.argmax(b))
                    # Reduce a[j] for j that contribute to peak
                    contrib = np.zeros(n)
                    for j in range(n):
                        if 0 <= k_star - j < n:
                            contrib[j] = best_a[k_star - j]
                    top_k = np.argsort(contrib)[::-1][:max(4, n//16)]
                    a = best_a.copy()
                    a[top_k] *= 0.9
                    a = project_simplex(a, S)
                else:
                    # Random restart from best
                    a = project_simplex(best_a * np.exp(0.05 * rng.standard_normal(n)), S)
                no_improve = 0
                current_score = score(a)
            
            if verbose and step % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    step={step}, score={current_score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={elapsed:.0f}s")
        
        # Compute gradient
        a = project_simplex(a, S)
        value, grad = lse_score_and_grad_fast(a, tau)
        
        # Gradient step with line search
        # Projected gradient: we want to minimize value subject to a >= 0, sum(a) = S
        # Simple approach: take step in negative gradient, then project
        alpha = lr / (np.max(np.abs(grad)) + 1e-10)
        a_new = project_simplex(a - alpha * grad, S)
        
        # Check if it improved
        new_score = score(a_new)
        if new_score < current_score + 1e-12:
            a = a_new
            current_score = new_score
        else:
            # Try smaller step
            alpha *= 0.5
            a_new = project_simplex(a - alpha * grad, S)
            new_score = score(a_new)
            if new_score < current_score + 1e-12:
                a = a_new
                current_score = new_score
        
        # Adaptive learning rate
        if no_improve < patience // 2:
            lr = min(0.1, lr * 1.01)
        else:
            lr = max(0.001, lr * 0.95)
        
        step += 1
    
    # Final check
    a = project_simplex(a, S)
    s = score(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    
    return best_a, best_score

def local_polish(a_init, time_limit, rng):
    """
    Local polish using coordinate descent and small perturbations.
    """
    n = len(a_init)
    S = np.sum(a_init)
    a = a_init.copy()
    current_score = score(a / S)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Try a random perturbation
        r = rng.random()
        if r < 0.5:
            # Coordinate perturbation
            j = int(rng.integers(0, n))
            for frac in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                trial = a.copy()
                trial[j] = a[j] * frac
                S_t = np.sum(trial)
                if S_t < 1e-10:
                    continue
                s = score(trial / S_t)
                if s < current_score:
                    a = trial
                    current_score = s
                    if s < best_score:
                        best_score = s
                        best_a = a.copy()
                    break
        elif r < 0.8:
            # Block perturbation
            width = max(2, int(rng.integers(2, n//8 + 2)))
            start = int(rng.integers(0, n - width))
            for frac in [0.8, 0.9, 1.1, 1.2]:
                trial = a.copy()
                trial[start:start+width] *= frac
                S_t = np.sum(trial)
                if S_t < 1e-10:
                    continue
                s = score(trial / S_t)
                if s < current_score * 1.001:
                    a = trial
                    current_score = s
                    if s < best_score:
                        best_score = s
                        best_a = a.copy()
                    break
        else:
            # Global small perturbation
            trial = a * np.exp(rng.normal(0, 0.02, size=n))
            S_t = np.sum(trial)
            if S_t > 1e-10:
                s = score(trial / S_t)
                if s < current_score * 1.001:
                    a = trial
                    current_score = s
                    if s < best_score:
                        best_score = s
                        best_a = a.copy()
        
        it += 1
        if it % 5000 == 0:
            print(f"    Polish iter={it}, cur={current_score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score

**Alternative: Use the structure of the optimal solution**

For the minimax problem min max_k (a*a)[k] s.t. a >= 0, sum(a) = 1:

At the optimum, we expect:
1. Several constraints are active (multiple k's achieve the maximum)
2. The solution is likely to have a specific "shape"

One insight: if the optimal solution is symmetric (a[i] = a[n-1-i]), then the center of the convolution b[n-1] = sum a[i]^2 is the peak (by Cauchy-Schwarz, this is >= S^2/n = 1/n), and the score is at least 2. So the optimal solution MUST be asymmetric.

For an asymmetric solution, the peak is at some k* ≠ n-1. The value is b[k*] = sum_{i} a[i]*a[k*-i].

If k* < n-1 (peak is to the left of center), it means the sequence has "energy concentrated" on the left side. If k* > n-1, on the right.

Actually wait, for k* < n-1, the overlap region is smaller (fewer terms in the sum), so b[k*] = sum of fewer products. This could be smaller or larger than b[n-1] depending on the values.

Let me think about a specific structure. What if we use a "ramp" shape: a[i] = c*(i+1) for i = 0,...,n-1 (increasing)?

Then S = c*n*(n+1)/2.
b[k] = c^2 * sum_{i} (i+1)*(k-i+1) for valid i.

For k = n-1 (center): b[n-1] = c^2 * sum_{i=0}^{n-1} (i+1)*(n-i)
= c^2 * sum_{i=1}^{n} i*(n+1-i)
= c^2 * (n+1)*sum(i) - c^2*sum(i^2)
= c^2 * (n+1)*n(n+1)/2 - c^2 * n(n+1)(2n+1)/6
= c^2 * n(n+1)[(n+1)/2 - (2n+1)/6]
= c^2 * n(n+1)*(3n+3-2n-1)/6

S = c*n*(n+1)/2, so S^2 = c^2 * n^2*(n+1)^2/4.
