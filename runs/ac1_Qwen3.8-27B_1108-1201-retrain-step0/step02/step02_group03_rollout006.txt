combined with Adam gradient descent and targeted local search.

from scipy.sparse import csr_matrix
import time

def local_score(a):
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def lp_update(f, M):
    """Find g maximizing sum(g) s.t. g>=0, conv(f,g) <= M."""
    n = len(f)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M),

    return np.zeros(n)

"""
LP-guided iterative optimization with gradient polishing and local search
for minimizing peak autocorrelation of non-negative step functions.
Uses linear programming to find optimal update directions that respect
current peak bounds, combined with Adam gradient descent and targeted
peak-directed mutations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def local_score(a):
    """Compute the evaluation score for a sequence (fast version)."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def fft_convolve(a, b):
    """Compute linear convolution using FFT for speed."""
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def build_lp_matrix(n, f):
    """Build the constraint matrix for the LP: A[k,i] = f[k-i] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_solve(f, M):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g)<=M."""
    n = len(f)
    A = build_lp_matrix(n, f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_optimize(f, max_iters=150, blend_start=0.4, blend_end=0.1, time_limit=300):
    """LP-based iterative optimization with adaptive blending."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    prev_score = best_score
    blend = blend_start

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)

        g = lp_solve(f, M)
        if g is None:
            break
        Sum_g = np.sum(g)
        if Sum_g < 1e-10:
            break

        # Rescale to match sum
        g = g / Sum_g * S

        # Adaptive blending
        new_f = (1.0 - blend) * f + blend * g
        new_f = np.clip(new_f, 0.0, None)
        s_new = np.sum(new_f)
        if s_new < 1e-10:
            break
        new_f = new_f / s_new * S

        score = local_score(new_f)
        if score < best_score:
            best_score = score
            best_f = new_f.copy()

        # Adaptive blend rate
        if score < prev_score:
            blend = min(0.5, blend * 1.1)
        else:
            blend = max(0.05, blend * 0.7)

        prev_score = score
        f = new_f

        if it % 30 == 0:
            print(f"    LP iter {it}: score={score:.6f}, best={best_score:.6f}, blend={blend:.3f}")

    return best_f, best_score


def soft_max_value_and_grad(c, tau):
    """Compute soft-max of c and gradient weights."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    w = e / s
    value = c_max + tau * np.log(s)
    return value, w

        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr_base / (1.0 + 0.05 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 300 == 0:
            a_check = np.exp(theta - np.max(theta))
            a_check = a_check / np.sum(a_check) * S
            sc = local_score(a_check)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()
            if step % 600 == 0:
                print(f"    Grad step {step}: score={sc:.6f}, best={best_score:.6f}, tau={tau:.2e}")

    a_final = np.exp(best_theta - np.max(best_theta))
    a_final = a_final / np.sum(a_final) * S
    return a_final, local_score(a_final)


def local_search(f, time_limit=120, rng=None):
    """Simulated annealing with peak-directed and structural mutations."""
    if rng is None:
        rng = np.random.default_rng(999)
    f = f.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    temp = 0.002
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        candidate = f.copy()
        r = rng.random()

        if r < 0.35:
            # Peak-directed: reduce high contributors, boost low ones
            conv = np.convolve(f, f)
            # Find top few peak lags
            n_peaks = min(5, len(conv) // 4)
            top_lags = np.argsort(conv)[-n_peaks:]
            contrib = np.zeros(n)
            for lag in top_lags:
                i_lo = max(0, lag - n + 1)
                i_hi = min(n - 1, lag)
                for i in range(i_lo, i_hi + 1):
                    j = lag - i
                    if 0 <= j < n:
                        contrib[i] += f[i] * f[j]
            n_mod = max(2, n // 10)
            top_idx = np.argsort(contrib)[-n_mod:]
            bot_idx = np.argsort(contrib)[:n_mod]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, n_mod))
            candidate[bot_idx] *= np.exp(rng.normal(0.04, 0.015, n_mod))
        elif r < 0.5:
            # Adjacent block swap-like
            width = max(1, int(rng.integers(1, max(2, n // 20))))
            start = int(rng.integers(0, max(1, n - 2 * width)))
            candidate[start:start + width] *= np.exp(rng.normal(-0.06, 0.02, width))
            candidate[start + width:start + 2 * width] *= np.exp(rng.normal(0.06, 0.02, width))
        elif r < 0.65:
            # Multi-block
            n_blocks = int(rng.integers(2, 5))
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 12))))
                bs = int(rng.integers(0, n - bw))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0.0, 0.04, bw))
        elif r < 0.8:
            # Symmetrization tendency
            sym = 0.5 * (f + f[::-1])
            candidate = f + 0.1 * (sym - f)
            candidate += 0.005 * rng.standard_normal(n)
        elif r < 0.9:
            # Monotonic tendency
            mono = np.sort(f)[::-1]
            candidate = f + 0.05 * (mono - f)
            candidate += 0.003 * rng.standard_normal(n)
        else:
            candidate = f * np.exp(rng.normal(0.0, 0.01, n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s * S

        new_score = local_score(candidate)
        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 8:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    f = candidate
                    current_score = new_score
        temp *= 0.9995

    return best_f, best_score


def monotonically_decreasing_init(n, gamma=1.0, rng=None):
    """Create a monotonically decreasing sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + gamma * (1.0 - t / (n - 1))
    if rng is not None:
        a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def bimodal_init(n, center_frac=0.5, width=0.3, rng=None):
    """Create a bimodal (high at edges, low in middle) sequence."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 + 1.5 * np.exp(-((t - center_frac) / width) ** 2)
    if rng is not None:
        a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def edge_concentrated_init(n, power=2.0, rng=None):
    """Create a sequence concentrated at the edges."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    dist_from_center = np.abs(t - 0.5)
    a = 0.1 + (dist_from_center / 0.5) ** power
    if rng is not None:
        a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def fft_flat_init(n, n_iter=50, rng=None, N_fft=None):
    """Gerchberg-Saxton style: flatten the spectrum."""
    if N_fft is None:
        N_fft = 1 << (2 * n - 1).bit_length()
    a = np.ones(n) / n
    for _ in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.001, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def propose_candidate(budget_s=2400, **kwargs):
    """
    Multi-phase search: LP optimization, gradient descent, and local search
    to minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 40.0
    rng = np.random.default_rng(777)

    best_score = np.inf
    best_seq = None

    # Start from provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, None)
        s = local_score(start_seq)
        print(f"Provided: n={len(start_seq)}, score={s:.6f}")
        best_score = s
        best_seq = start_seq.copy()

    # Phase 1: LP optimization on provided sequence
    if start_seq is not None and time.monotonic() < deadline - 300:
        print(f"\n=== Phase 1: LP on provided seq (n={len(start_seq)}) ===")
        t_alloc = min(400.0, deadline - time.monotonic() - 300)
        lp_res, lp_score = lp_optimize(start_seq, max_iters=200, blend_start=0.3,
                                       blend_end=0.05, time_limit=t_alloc)
        if lp_score < best_score:
            best_score = lp_score
            best_seq = lp_res.copy()
        print(f"  LP result: {lp_score:.6f} (best={best_score:.6f})")

    # Phase 2: Try structured initializations at multiple lengths
    print(f"\n=== Phase 2: Structured inits + LP ===")
    lengths_to_try = [128, 192, 256, 320, 384]

    for n in lengths_to_try:
        if time.monotonic() > deadline - 200:
            break

        inits = []
        inits.append(('mono_dec', monotonically_decreasing_init(n, gamma=1.5, rng=rng)))
        inits.append(('mono_dec2', monotonically_decreasing_init(n, gamma=3.0, rng=rng)))
        inits.append(('bimodal', bimodal_init(n, center_frac=0.5, width=0.25, rng=rng)))
        inits.append(('edge', edge_concentrated_init(n, power=3.0, rng=rng)))
        inits.append(('fft', fft_flat_init(n, n_iter=30, rng=rng)))

        for name, init in inits:
            if time.monotonic() > deadline - 100:
                break
            s0 = local_score(init)
            t_alloc = min(80.0, deadline - time.monotonic() - 100)
            if t_alloc < 10:
                continue
            lp_res, lp_score = lp_optimize(init, max_iters=60, blend_start=0.3,
                                           blend_end=0.08, time_limit=t_alloc)
            improved = " ***" if lp_score < best_score else ""
            print(f"  n={n}, {name}: init={s0:.4f} -> LP={lp_score:.6f}{improved}")
            if lp_score < best_score:
                best_score = lp_score
                best_seq = lp_res.copy()

    # Phase 3: Gradient polishing
    if best_seq is not None:
        t_alloc = min(200.0, deadline - time.monotonic() - 100)
        if t_alloc > 20:
            print(f"\n=== Phase 3: Gradient polish (t={t_alloc:.0f}s) ===")
            grad_res, grad_score = gradient_optimize(best_seq, time_limit=t_alloc)
            if grad_score < best_score:
                best_score = grad_score
                best_seq = grad_res.copy()
            print(f"  Gradient: {grad_score:.6f} (best={best_score:.6f})")

    # Phase 4: Local search
    if best_seq is not None:
        t_alloc = min(180.0, deadline - time.monotonic() - 60)
        if t_alloc > 20:
            print(f"\n=== Phase 4: Local search (t={t_alloc:.0f}s) ===")
            ls_res, ls_score = local_search(best_seq, time_limit=t_alloc, rng=rng)
            if ls_score < best_score:
                best_score = ls_score
                best_seq = ls_res.copy()
            print(f"  Local search: {ls_score:.6f} (best={best_score:.6f})")

    # Phase 5: Second LP round on improved sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        t_alloc = min(120.0, deadline - time.monotonic() - 20)
        if t_alloc > 15:
            print(f"\n=== Phase 5: Final LP refinement (t={t_alloc:.0f}s) ===")
            lp_res, lp_score = lp_optimize(best_seq, max_iters=100, blend_start=0.2,
                                           blend_end=0.05, time_limit=t_alloc)
            if lp_score < best_score:
                best_score = lp_score
                best_seq = lp_res.copy()
            print(f"  Final LP: {lp_score:.6f} (best={best_score:.6f})")

    # Phase 6: Final local search polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_alloc = min(60.0, deadline - time.monotonic() - 5)
        if t_alloc > 10:
            ls_res, ls_score = local_search(best_seq, time_limit=t_alloc, rng=rng)
            if ls_score < best_score:
                best_score = ls_score
                best_seq = ls_res.copy()
            print(f"  Final polish: {ls_score:.6f} (best={best_score:.6f})")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Ensure no NaN or inf
        final_list = [x if (np.isfinite(x) and x >= 0) else 0.0 for x in final_list]
        if not final_list or np.sum(final_list) < 0.01:
            final_list = [1.0] * 256
        final_score = evaluate_sequence(final_list)
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        return final_list

    # Fallback
    return [1.0] * 256
