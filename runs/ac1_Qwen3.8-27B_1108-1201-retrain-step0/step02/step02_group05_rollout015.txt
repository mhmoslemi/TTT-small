def lp_direction(a, M):
    """Find the LP direction: max sum(b) s.t. conv(a,b)[k] <= M, b >= 0."""
    n = len(a)
    # A_ub: shape (2n-1, n)
    # A_ub[k, j] such that sum_j A_ub[k,j] * b[j] <= M for each k
    # conv(a,b)[k] = sum_i a[i] * b[k-i] for valid i
    # = sum_j a[k-j] * b[j] for valid j (0 <= j < n, 0 <= k-j < n)
    # So A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = a[i]
    
    c = -np.ones(n)  # minimize -sum(b) = max sum(b)
    result = linprog(c, A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    return result.x

"""
LP-guided optimization for minimum peak autocorrelation non-negative sequences.
Uses linear programming to find optimal descent directions, combined with
p-norm gradient refinement and local search polish.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score for sequence: 2n * max(conv(a,a)) / (sum(a))^2."""
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def norm_score(a):
    """Score assuming sum(a) = 1: 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def lp_direction(a, M):
    """
    Find b >= 0 maximizing sum(b) subject to conv(a,b)[k] <= M for all k.
    a is normalized to sum=1.
    """
    n = len(a)
    # Build constraint matrix: conv(a,b)[k] = sum_j A[k,j] * b[j] <= M
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((2*n - 1, n))
    for j in range(n):
        # b[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # with coefficient a[k-j] = a[0], a[1], ..., a[n-1]
        A_ub[j:j+n, j] = a
    
    c_obj = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(2*n - 1, M)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                     bounds=[(0, None)] * n, method='highs')
    if result.success:
        return result.x
    return None


def construct_A_ub(a):
    """Build the constraint matrix for the LP."""
    n = len(a)
    A_ub = np.zeros((2*n - 1, n))
    for j in range(n):
        A_ub[j:j+n, j] = a
    return A_ub


def pnorm_obj_grad(theta, p):
    """Objective and gradient for p-norm of convolution."""
    a = softmax(theta)
    n = len(a)
    c = fft_convolve(a, a)
    
    if p == np.inf:
        # Use soft approximation
        c_max = np.max(c)
        return 2.0 * n * c_max, None
    
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    
    factor = 4.0 * n * (sum_cp ** (1.0/p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n-1:2*n-1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def soft_max_obj_grad(theta, tau):
    """Objective and gradient for soft-max of convolution."""
    a = softmax(theta)
    n = len(a)
    c = fft_convolve(a, a)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    
    score = 2.0 * n * soft_max_val
    
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n-1:2*n-1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def lp_iteration(a, max_iters=50, t_init=0.1, verbose=False):
    """
    Main LP-based iteration. Each step:
    1. Compute peak M
    2. Solve LP for direction b
    3. Line search for step size
    4. Update
    """
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_score = norm_score(a)
    
    for it in range(max_iters):
        c = fft_convolve(a, a)
        M = np.max(c)
        
        b = lp_direction(a, M)
        if b is None or np.sum(b) < 1e-10:
            break
        
        b = b / np.sum(b)
        
        # Line search for step size t
        best_t = 0
        best_val = best_score
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_new = (1 - t) * a + t * b
            a_new = np.maximum(a_new, 0)
            s = norm_score(a_new)
            if s < best_val:
                best_val = s
                best_t = t
        
        if best_t == 0:
            # Try smaller steps
            for t in [0.01, 0.02, 0.03]:
                a_new = (1 - t) * a + t * b
                a_new = np.maximum(a_new, 0)
                s = norm_score(a_new)
                if s < best_val:
                    best_val = s
                    best_t = t
        
        if best_t == 0:
            break
        
        a = (1 - best_t) * a + best_t * b
        a = np.maximum(a, 0)
        a = a / np.sum(a)
        
        s = norm_score(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
        
        if verbose and it % 10 == 0:
            print(f"    LP iter {it}: score={best_score:.6f}, t={best_t:.3f}")
    
    return best_a, best_score


def polish_local_search(a_init, time_limit, rng):
    """
    Simulated annealing with targeted peak reduction.
    """
    current = a_init.astype(np.float64).copy()
    current = current / np.sum(current)
    current_score = norm_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.0005
    cooling = 0.9999
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.4:
            # Targeted peak reduction
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            # Identify top contributors to the peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(5, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.008, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(15, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.008, size=n_inc))
        elif r < 0.6:
            # Multi-peak reduction
            c = fft_convolve(current, current)
            c_sorted = np.sort(c)[::-1]
            threshold = c_sorted[min(5, len(c_sorted)-1)] * 0.95
            high_peaks = np.where(c > threshold)[0]
            for pk in high_peaks:
                contribs = np.zeros(n)
                for i in range(n):
                    j = pk - i
                    if 0 <= j < n:
                        contribs[i] = current[i] * current[j]
                top_k = max(3, n // 20)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.01, 0.005, size=len(top_idx)))
        elif r < 0.8:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        else:
            # Small global perturbation
            candidate *= np.exp(rng.normal(0.0, 0.01, size=n))
        
        candidate = np.maximum(candidate, 1e-10)
        candidate = candidate / np.sum(candidate)
        new_score = norm_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score
        
        temperature = max(temperature * cooling, 1e-10)
    
    return best, best_score

"""
LP-guided optimization for minimum peak autocorrelation non-negative sequences.
Uses linear programming to find optimal descent directions at each step,
combined with p-norm/soft-max gradient refinement and targeted local search.
The LP approach directly solves for the best direction that keeps the
autocorrelation peak bounded, providing a principled alternative to
pure gradient-based methods.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def norm_score(a):
    """Score assuming sum(a) = 1: 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def local_score(a):
    """Score for arbitrary non-negative sequence: 2n * max(conv) / (sum)^2."""
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def lp_direction(a, M):
    """
    Solve LP: max sum(b) subject to conv(a,b)[k] <= M for all k, b >= 0.
    Returns optimal b vector (unnormalized).
    """
    n = len(a)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = a
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                     bounds=[(0, None)] * n, method='highs')
    if result.success:
        return result.x
    return None


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    a = softmax(theta)
    n = len(a)
    c = fft_convolve(a, a)

    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def soft_max_obj_grad(theta, tau):
    """Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau)."""
    a = softmax(theta)
    n = len(a)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta

def polish_local_search(a_init, time_limit, rng):
    """
    Simulated annealing with targeted peak reduction and multi-peak handling.
    """
    current = a_init.astype(np.float64).copy()
    current = current / np.sum(current)
    current_score = norm_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.0003
    cooling = 0.99995
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.35:
            # Targeted single-peak reduction
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(5, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.015, 0.007, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(20, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.015, 0.007, size=n_inc))
        elif r < 0.55:
            # Multi-peak reduction
            c = fft_convolve(current, current)
            c_sorted = np.sort(c)[::-1]
            n_peaks = min(3, len(c_sorted))
            threshold = c_sorted[n_peaks - 1] * 0.98
            high_peaks = np.where(c > threshold)[0]
            for pk in high_peaks:
                contribs = np.zeros(n)
                for i in range(n):
                    j = pk - i
                    if 0 <= j < n:
                        contribs[i] = current[i] * current[j]
                top_k = max(3, n // 25)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.008, 0.005, size=len(top_idx)))
        elif r < 0.70:
            # Asymmetric shift: move weight from one side to the other
            shift = rng.integers(1, max(2, n // 20))
            if rng.random() < 0.5:
                candidate[shift:] = current[:-shift].copy()
                candidate[:shift] = current[-shift:].copy()
            else:
                candidate[:-shift] = current[shift:].copy()
                candidate[-shift:] = current[:shift].copy()
            candidate *= np.exp(rng.normal(0, 0.005, n))
        elif r < 0.85:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        else:
            # Small global
            candidate *= np.exp(rng.normal(0.0, 0.008, size=n))

        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)
        new_score = norm_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 1e-12)

        if time.monotonic() - last_print > 5.0:
            last_print = time.monotonic()
            print(f"    polish: current={current_score:.7f}, best={best_score:.7f}, "

    return best, best_score


def make_interpolated_seq(seq, n_new, rng):
    """Interpolate a sequence to a new length with perturbation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, n_new)
    a = np.interp(x_new, x_old, seq)
    a = a * np.exp(rng.normal(0, 0.1, n_new))
    a = np.maximum(a, 1e-6)
    return a / np.sum(a)

    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None
    best_n = 256

    # Load starting sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        s = norm_score(start_seq)
        print(f"Starting sequence: n={len(start_seq)}, score={s:.7f}")
        best_score = s
        best_seq = start_seq.copy()
        best_n = len(start_seq)

    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = norm_score(best_seq)
        best_n = 256
        print(f"Default uniform sequence: n=256, score={best_score:.7f}")

    # === Phase 1: LP optimization on primary length ===
    print(f"\n{'='*60}")
    print(f"Phase 1: LP optimization (n={best_n})")
    print(f"{'='*60}")
    lp_time1 = min(1000.0, (deadline - t_global) * 0.4)
    a_out, s_out = run_lp_phase(best_seq, lp_time1, verbose=True)
    if s_out < best_score:
        best_score = s_out
        best_seq = a_out.copy()
        print(f"  Phase 1 result: {best_score:.7f}")

    # === Phase 2: p-norm refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 2: p-norm L-BFGS refinement (n={len(best_seq)})")
    print(f"{'='*60}")
    pn_time = min(300.0, (deadline - t_global) * 0.15)
    a_out, s_out = run_pnorm_phase(best_seq, pn_time)
    if s_out < best_score:
        best_score = s_out
        best_seq = a_out.copy()
        print(f"  Phase 2 result: {best_score:.7f}")

    # === Phase 3: Soft-max refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Soft-max L-BFGS refinement (n={len(best_seq)})")
    print(f"{'='*60}")
    sm_time = min(250.0, (deadline - t_global) * 0.12)
    a_out, s_out = run_softmax_phase(best_seq, sm_time)
    if s_out < best_score:
        best_score = s_out
        best_seq = a_out.copy()
        print(f"  Phase 3 result: {best_score:.7f}")

    # === Phase 4: LP refinement on gradient-refined sequence ===
    print(f"\n{'='*60}")
    print(f"Phase 4: LP refinement (n={len(best_seq)})")
    print(f"{'='*60}")
    lp_time2 = min(400.0, (deadline - t_global) * 0.18)
    a_out, s_out = run_lp_phase(best_seq, lp_time2, verbose=True)
    if s_out < best_score:
        best_score = s_out
        best_seq = a_out.copy()
        print(f"  Phase 4 result: {best_score:.7f}")

    # === Phase 4b: Try alternative lengths ===
    if time.monotonic() < deadline - 300:
        print(f"\n{'='*60}")
        print(f"Phase 4b: Alternative length exploration")
        print(f"{'='*60}")
        alt_lengths = [192, 320, 384, 448]
        for n_alt in alt_lengths:
            if time.monotonic() > deadline - 200:
                break
            a_alt = make_interpolated_seq(best_seq, n_alt, rng)
            # Quick LP run
            alloc = min(60.0, max(20.0, (deadline - time.monotonic()) * 0.1))
            a_out, s_out = run_lp_phase(a_alt, alloc, verbose=False)
            if s_out < best_score:
                best_score = s_out
                best_seq = a_out.copy()
                best_n = n_alt
                print(f"  n={n_alt}: score={best_score:.7f} *** NEW BEST ***")
            else:
                print(f"  n={n_alt}: score={s_out:.7f} (best={best_score:.7f})")

    # === Phase 5: Final polish ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Local search polish (n={len(best_seq)})")
    print(f"{'='*60}")
    polish_time = max(30.0, deadline - time.monotonic() - 30)
    print(f"  Allocating {polish_time:.0f}s for polish")
    polished, p_score = polish_local_search(best_seq, polish_time, rng)
    if p_score < best_score:
        best_score = p_score
        best_seq = polished.copy()
        print(f"  After polish: {best_score:.7f} *** NEW BEST ***")
    else:
        print(f"  After polish: {p_score:.7f} (no improvement)")

    # === Final verification ===
    print(f"\n{'='*60}")
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(final_list)
    print(f"FINAL: evaluate_sequence score={final_score:.7f}, n={len(final_list)}")
    print(f"{'='*60}")

    return final_list
