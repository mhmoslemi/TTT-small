"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- L-BFGS-B on softmax parameterization with log-sum-exp continuation (increasing beta)
- Multiple initializations: given sequence, random, structured patterns
- Perturbation and restart to escape local minima
- Final polishing with Nesterov top-k subgradient
- FFT-accelerated gradient computation
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def smooth_objective_and_grad(z, beta, S):
    """
    Compute smoothed objective and gradient w.r.t. z (softmax params).
    a = S * softmax(z)
    Objective: (1/beta) * log(sum exp(beta * c_k)) where c = conv(a,a)
    """
    n = len(z)
    # Compute a from z (numerically stable softmax)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp
    
    # Compute convolution
    c = fft_convolve_self(a)
    
    # Log-sum-exp
    c_max = c.max()
    log_sum_exp = c_max + np.log(np.sum(np.exp(beta * (c - c_max))))
    obj = log_sum_exp / beta
    
    # Gradient w.r.t. a: grad_a = (2/beta) * sum_k softmax(beta*c)_k * d(c_k)/da
    # d(c_k)/da_j = a_{j-k} + a_{j+k} (cross-correlation)
    # This is: grad_a = (2/beta) * cross_corr(softmax_weights, a)
    
    w = np.exp(beta * (c - c_max))
    w = w / w.sum()
    
    # grad_a[j] = 2 * sum_k w[k] * (a[j-k] + a[j+k]) (with boundary)
    # = 2 * (w cross-correlated with a) evaluated at j
    # Using FFT: cross-correlation of w and a
    
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    # Gradient w.r.t. z: dz_k = sum_j (grad_a_j) * (da_j/dz_k)
    # da_j/dz_k = a_j * (delta_jk - a_j/S)
    # grad_z_k = grad_a_k * a_k - (sum_j grad_a_j * a_j) / S
    
    grad_z = grad_a * a - np.dot(grad_a, a) * a / S
    
    return obj, grad_z


def run_lbfgs_continuation(a_init, n, S, max_time, rng, 
                            beta_schedule=None):
    """
    Run L-BFGS-B with continuation on beta.
    a_init: initial sequence (will be converted to softmax params)
    """
    if beta_schedule is None:
        beta_schedule = [3, 5, 10, 20, 50, 100, 200, 400]
    
    # Convert to softmax params
    a = np.clip(a_init, 1e-10, 1000.0)
    a = a / a.sum() * S
    z = np.log(a / S)  # log of relative values
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    
    # Evaluate initial
    c = fft_convolve_self(a)
    best_score = c.max()
    
    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 10:
            break
        
        # Define objective and gradient for scipy
        def obj_grad(z):
            return smooth_objective_and_grad(z, beta, S)
        
        # Run L-BFGS-B
        t0 = time.monotonic()
        result = scipy_minimize(
            lambda z: smooth_objective_and_grad(z, beta, S)[0],
            z,
            jac=lambda z: smooth_objective_and_grad(z, beta, S)[1],
            method='L-BFGS-B',
            options={
                'maxiter': 200,
                'maxfun': 200,
                'ftol': 1e-12,
                'gtol': 1e-8,
                'maxls': 50
            }
        )
        elapsed = time.monotonic() - t0
        
        z = result.x
        a = S * np.exp(z - z.max()) / np.exp(z - z.max()).sum()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * S / s
        
        c = fft_convolve_self(a)
        score = c.max()
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print("    beta=%d: score=%.8f (L-BFGS %d iters, %.2fs)" % 
                  (beta, score, result.nit, elapsed))
        else:
            print("    beta=%d: score=%.8f (no improvement)" % (beta, score))
        
        # Check time
        if time.monotonic() - start > max_time - 30:
            break
        
        # At high beta, add perturbation to escape local minima
        if beta >= 100 and rng.random() < 0.3:
            z = z + rng.normal(0, 0.01, n)
    
    return best_a, best_score


def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=128):
    """Nesterov-accelerated top-k subgradient method with FFT gradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Top-k subgradient
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 5000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- L-BFGS-B on softmax parameterization with log-sum-exp continuation (increasing beta)
- Multiple initializations: given sequence, random, structured patterns
- Perturbation and restart to escape local minima
- Final polishing with Nesterov top-k subgradient
- FFT-accelerated gradient computation O(n log n) per evaluation
- Key improvement over FISTA: uses curvature info for superlinear convergence near optimum
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_a(a, w):
    """Compute grad_a[j] = 2 * sum_k w[k] * a[j-k] (cross-correlation via FFT)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """
    Compute smoothed objective and gradient w.r.t. z (unconstrained softmax params).
    a = S * softmax(z), objective = (1/beta)*log(sum exp(beta*c_k))
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_grad_a(a, w)

    # grad_z[k] = a[k] * (grad_a[k] - dot(grad_a, a)/S)
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=300):
    """
    Run L-BFGS-B with continuation on beta (smooth-to-sharp).
    Returns best (a, score) found.
    """
    if beta_schedule is None:
        beta_schedule = [3, 5, 10, 20, 50, 100, 200, 400, 800, 1500]

    # Convert to softmax params
    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        elapsed_beta = 0.0
        n_attempts = 1
        # At high beta, try multiple L-BFGS runs with slight perturbation
        if beta >= 200 and max_time > 60:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 5:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.02, n)
                z = z - z.mean()

            # Define objective function
            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            t0 = time.monotonic()
            try:
                result = scipy_minimize(
                    fun, z,
                    jac=jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_beta,
                        'maxfun': maxiter_per_beta * 2,
                        'ftol': 1e-14,
                        'gtol': 1e-9,
                        'maxls': 40,
                        'iprint': -1
                    }
                )
            except Exception as e:
                print("      L-BFGS error: %s" % str(e))
                break

            elapsed_beta += time.monotonic() - t0
            z = result.x
            # Re-center z for numerical stability
            z = z - z.mean()

            # Convert back to a
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

        if elapsed_beta > 0:
            print("    beta=%4d: score=%.8f (%d attempts, %.1fs)" %
                  (beta, best_score, n_attempts, elapsed_beta))

    return best_a, best_score


def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=256):
    """Nesterov-accelerated top-k subgradient with FFT gradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 80)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 5000 == 0 and rng.random() < 0.08:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

            if len(sorted_results) >= 2:
                n_xpoll = max(2, int(remaining / 50))
                xpoll_budget = remaining * 0.5 / n_xpoll

                for xp in range(n_xpoll):
                    if time.monotonic() > deadline - 60:
                        break

                    idx1 = xp % len(sorted_results)
                    idx2 = (xp + 1) % len(sorted_results)
                    if idx1 == idx2:
                        idx2 = (idx2 + 1) % len(sorted_results)

                    a1 = sorted_results[idx1][1][0]
                    a2 = sorted_results[idx2][1][0]

                    # Try several interpolation factors
                    for blend in [0.3, 0.5, 0.7]:
                        if time.monotonic() > deadline - 30:
                            break
                        a_blend = blend * a1 + (1 - blend) * a2
                        a_blend = safe_normalize(a_blend, n)

                        a_new, score = run_lbfgs_continuation(
                            a_blend, n, S, xpoll_budget / 3, rng,
                            beta_schedule=[20, 50, 100, 300, 800],
                            maxiter_per_beta=100
                        )
                        a_new = safe_normalize(a_new, n)
                        c_new = fft_convolve_self(a_new)
                        if c_new.max() < global_best:
                            global_best = c_new.max()
                            global_a = a_new.copy()
                            print("  XPoll %d (blend=%.1f): %.8f" % (xp, blend, global_best))

    print("\n=== After Phase 2: global_best=%.8f ===" % global_best)

    # --- Phase 3: Perturbation + L-BFGS restart ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("\n=== Phase 3: Perturbation + L-BFGS (%.1f sec) ===" % remaining)

            n_perturb = max(3, int(remaining / 45))
            perturb_budget = remaining * 0.5 / n_perturb

            for p in range(n_perturb):
                if time.monotonic() > deadline - 45:
                    break

                # Multiple noise scales
                noise_scale = 0.03 / (1.0 + 0.3 * p)
                a_pert = global_a + rng.normal(0, noise_scale * S / np.sqrt(n), n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = safe_normalize(a_pert, n)

                a_new, score = run_lbfgs_continuation(
                    a_pert, n, S, perturb_budget, rng,
                    beta_schedule=[20, 50, 100, 300, 800, 2000],
                    maxiter_per_beta=150
                )
                a_new = safe_normalize(a_new, n)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < global_best:
                    global_best = c_new.max()
                    global_a = a_new.copy()
                    print("  Perturb %d (scale=%.4f): %.8f" % (p, noise_scale, global_best))

    print("\n=== After Phase 3: global_best=%.8f ===" % global_best)

    # --- Phase 4: Nesterov + L-BFGS polishing ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        print("\n=== Phase 4: Final Polishing (%.1f sec) ===" % remaining)

        # Nesterov polish
        if remaining > 20:
            seg1 = min(remaining * 0.4, 120)
            a_polish, score = run_nesterov_toppk(
                global_a, n, S, seg1, rng,
                top_k_init=10, top_k_max=512
            )
            a_polish = safe_normalize(a_polish, n)
            c = fft_convolve_self(a_polish)
            if c.max() < global_best:
                global_best = c.max()
                global_a = a_polish.copy()
                print("  Nesterov: %.8f" % global_best)

        # Final high-beta L-BFGS
        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            a_final, score = run_lbfgs_continuation(
                global_a, n, S, remaining2, rng,
                beta_schedule=[200, 500, 1000, 2000, 5000, 10000],
                maxiter_per_beta=200
            )
            a_final = safe_normalize(a_final, n)
            c_final = fft_convolve_self(a_final)
            if c_final.max() < global_best:
                global_best = c_final.max()
                global_a = a_final.copy()
                print("  Final L-BFGS: %.8f" % global_best)

    # --- Phase 5: Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, score=%.8f" % (len(global_a), global_best))
    print("Time used: %.1fs" % (time.monotonic() - (deadline - budget_s + 15)))
    return [float(x) for x in global_a]
