"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses aggressive FISTA with adaptive log-sum-exp, top-k subgradient polishing,
and local refinement focused on peak lags.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_gradient(a, c, weights):
    """Compute gradient given weights on convolution lags."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total

def toppk_weights(c, top_k):
    """Compute top-k normalized weights."""
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights

    """FISTA with aggressive beta schedule and adaptive step size."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()

        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Aggressive beta: start high, increase rapidly
        beta = min(beta_max, beta_init + 5.0 * it)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Use softmax weights
        w = softmax_weights(c, beta)
        grad = compute_gradient(y, c, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta * 1.5)

top-k subgradient polishing (starting from k=1), backtracking line search,

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient given weights on convolution lags using FFT cross-correlation."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Compute normalized top-k weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights

            if c_v.max() < global_best:
                global_best = c_v.max()
                global_a = a_v.copy()
                print("  Variant %d: %.6f (obj %.6f)" % (v, global_best, 2*n*global_best/S**2))

    # === Phase 6: Final ultra-fine polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 5:
        print("Phase 6: Final polish (%.0f sec)" % remaining)
        # Very high beta, very low top-k
        a_f1, _ = run_toppk_refined(global_a, n, S, remaining * 0.4, rng,
                                     top_k_init=1, top_k_max=16, top_k_rate=500)
        a_f1 = safe_normalize(a_f1, n)
        c_f1 = fft_convolve_self(a_f1)
        if c_f1.max() < global_best:
            global_best = c_f1.max()
            global_a = a_f1.copy()
            print("  Polish1: %.6f (obj %.6f)" % (global_best, 2*n*global_best/S**2))

        remaining2 = deadline - time.monotonic()
        if remaining2 > 3:
            a_f2, _ = run_peak_focused(global_a, n, S, remaining2, rng)
            a_f2 = safe_normalize(a_f2, n)
            c_f2 = fft_convolve_self(a_f2)
            if c_f2.max() < global_best:
                global_best = c_f2.max()
                global_a = a_f2.copy()
                print("  Polish2: %.6f (obj %.6f)" % (global_best, 2*n*global_best/S**2))

    # === Final output ===
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    final_obj = 2.0 * n * global_best / (S * S)
    print("FINAL: length=%d, score=%.6f, objective=%.6f" % (len(final_list), global_best, final_obj))

    # Verify
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: %.6f" % ver)
    else:
        print("Verification failed, using fallback")
        final_list = [0.5] * 256

    return final_list
