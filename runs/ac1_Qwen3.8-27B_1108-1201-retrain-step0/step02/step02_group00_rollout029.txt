"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
- Phase 1 (Warm-up): Minimize ||f*f||_2^2 (smooth quartic) via accelerated gradient
  with simplex projection. Uses FFT-based gradient: 4*Re(ifft(|fft(f)|^2 * fft(f))).
- Phase 2 (Peak reduction): Switch to minimizing max(f*f) using adaptive top-k
  subgradient with Nesterov acceleration and backtracking.
- Phase 3 (Polish): Very high top-k with tiny steps for final refinement.
- Multiple diverse initializations (power-law, bimodal, chirp, filtered random, ramps).
- Try multiple sequence lengths (4096, 8192, 16384) and keep the best.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_toppk(a, c, top_k):
    """Gradient using top-k active lags, computed via FFT cross-correlation."""
    n = len(a)
    if top_k >= len(c):
        top_k = len(c)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total

    w_full = np.zeros(len(c))
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]

    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_variance(a):
    """Gradient of ||f*f||_2^2 w.r.t. f. Equals 4*Re(ifft(|fft(f)|^2 * fft(f)))."""
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    A = fft(a, fsize)
    grad = 4.0 * np.real(ifft(np.abs(A) ** 2 * A))[:n]
    return grad


def adaptive_topk(c, max_k=1024, min_k=3):
    """Choose top-k adaptively based on the flatness of the peak region."""
    c_max = c.max()
    if c_max < 1e-30:
        return min_k
    # Use lags within 5% of the max, capped at max_k
    threshold = c_max * 0.95
    n_active = int(np.sum(c >= threshold))
    # Also consider within 1%
    threshold_strict = c_max * 0.99
    n_strict = int(np.sum(c >= threshold_strict))
    k = max(min_k, min(max_k, n_active))
    return k


def run_variance_descent(a_init, n, S, max_time, rng):
    """Phase 1: Minimize ||f*f||_2^2 with Nesterov acceleration and backtracking."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_local = a.copy()
    step = 1.0 / (2.0 * n * S)
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        obj = np.sum(c ** 2)
        if obj < best_obj:
            best_obj = obj
            best_local = y.copy()

        grad = fft_grad_variance(y)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        lr = step
        x_new = project_simplex(y - lr * grad / gnorm, S)
        backtrack = 0
        while backtrack < 8:
            c_x = fft_convolve_self(x_new)
            obj_x = np.sum(c_x ** 2)
            if obj_x < obj * (1.0 - 1e-8 * lr * gnorm):
                break
            lr *= 0.5
            x_new = project_simplex(y - lr * grad / gnorm, S)
            backtrack += 1

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        step *= 0.999

    return best_local, best_obj


def run_toppk_descent(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=512):
    """Phase 2: Minimize max(f*f) with adaptive top-k subgradient + Nesterov."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    no_improve = 0
    patience = 3000

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive top-k
        top_k = adaptive_topk(c, max_k=top_k_max, min_k=top_k_init)
        # Gradually increase minimum k
        min_k = min(top_k_max, top_k_init + it // 200)
        top_k = max(top_k, min_k)

        grad = fft_grad_toppk(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size with backtracking
        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)

        # Backtracking
        c_x = fft_convolve_self(x_new)
        obj_x = c_x.max()
        backtrack = 0
        while obj_x > cur * 1.001 and backtrack < 5:
            step *= 0.5
            x_new = project_simplex(y - step * grad / gnorm, S)
            c_x = fft_convolve_self(x_new)
            obj_x = c_x.max()
            backtrack += 1

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Restart if stuck
        if no_improve > patience and rng.random() < 0.1:
            y = project_simplex(best_local + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            t = 1.0
            no_improve = 0
            step = 2.0 * S / (n * 10.0)

    return best_local, best_score


def run_polish(a_init, n, S, max_time, rng, top_k_init=128, top_k_max=2048):
    """Phase 3: Fine polishing with very high top-k and small steps."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Use a large fraction of the near-max lags
        c_max = c.max()
        threshold = c_max * (0.99 - 0.02 * min(1.0, it / 5000.0))
        n_active = int(np.sum(c >= threshold))
        top_k = max(64, min(top_k_max, n_active))

        grad = fft_grad_toppk(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Very small steps for polishing
        step = S / (n * (5.0 + 0.5 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)

        c_x = fft_convolve_self(x_new)
        if c_x.max() < cur:
            # Accept
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = x_new + ((t - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t = t_new
        else:
            # Reject, reduce step
            step *= 0.7
            t = 1.0

        # Occasional perturbation to escape flat regions
        if it % 2000 == 0 and rng.random() < 0.2:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def score_sequence(a):
    """Compute the objective value for a normalized sequence."""
    c = fft_convolve_self(a)
    return c.max()


def generate_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(("uniform", np.ones(n) * 1.0))

    # Random
    inits.append(("rand", rng.random(n) + 0.05))

    # Power law (increasing)
    for alpha in [0.3, 0.6, 1.0]:
        a = (tt + 0.01) ** alpha
        inits.append(("powlaw_%s" % alpha, a))

    # Power law (decreasing)
    inits.append(("powlaw_dec", (1.0 - tt + 0.01) ** 0.5))

    # Bimodal
    for sigma in [0.08, 0.12, 0.18]:
        a = np.exp(-((tt - 0.3) / sigma) ** 2) + np.exp(-((tt - 0.7) / sigma) ** 2) + 0.02
        inits.append(("bimodal_%s" % sigma, a))

    # Single bump (Gaussian)
    for sigma in [0.15, 0.25, 0.35]:
        a = np.exp(-((tt - 0.5) / sigma) ** 2) + 0.02
        inits.append(("bump_%s" % sigma, a))

    # Chirp (made non-negative)
    for freq in [3.0, 7.0, 13.0]:
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * tt + np.pi * freq * tt ** 2)
        a = np.maximum(a, 0.01)
        inits.append(("chirp_%s" % freq, a))

    # Sinusoidal
    for freq in [2, 5, 11]:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * tt) + 0.1 * np.cos(4 * np.pi * tt)
        a = np.maximum(a, 0.01)
        inits.append(("sin_%d" % freq, a))

    # Asymmetric (Skew)
    a = np.exp(-((tt - 0.35) / 0.2) ** 2) + 0.3 * np.exp(-((tt - 0.75) / 0.1) ** 2) + 0.02
    inits.append(("asym", a))

    # Low-pass filtered random
    a_rand = rng.random(n)
    A = fft(a_rand, n)
    cutoff = n // 8
    A[cutoff:] = 0
    A[:n // 2 + 1 - cutoff] = 0
    a_fp = np.real(ifft(A)) + 0.1
    a_fp = np.maximum(a_fp, 0.01)
    inits.append(("lpf_rand", a_fp))

    # Edge-suppressed (small at boundaries, large in middle)
    a = np.sin(np.pi * tt) ** 2 + 0.05
    inits.append(("edge_suppress", a))

    # Triangle
    a = 1.0 - 2.0 * np.abs(tt - 0.5)
    a = np.maximum(a, 0.01)
    inits.append(("triangle", a))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None

    # Check if we have a given starting sequence
    given_arr = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given_arr = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Determine lengths to try
    if given_arr is not None:
        lengths = [len(given_arr)]
    else:
        lengths = [8192]
    # Add alternative lengths
    for n_alt in [4096, 16384, 12288]:
        if n_alt not in lengths and 2048 <= n_alt <= 32768:
            lengths.append(n_alt)

    lengths.sort()  # Try smaller first (faster)

    print("Attempting lengths: %s" % lengths)
    total_n_inits = 0
    for n in lengths:
        total_n_inits += len(generate_initializations(n, rng))
    print("Total initializations to consider: %d" % total_n_inits)

    # Phase 1: Explore multiple initializations at potentially multiple lengths
    usable = max(10.0, deadline - time.monotonic())
    # Allocate 70% to exploration, 30% to polishing
    explore_budget = usable * 0.65
    polish_budget = usable * 0.35

    # Try each length, allocating time proportionally
    n_lengths = len(lengths)
    for length_idx, n in enumerate(lengths):
        time_left = deadline - time.monotonic()
        if time_left < 30:
            break

        # For the primary length (given or first), spend more time
        if length_idx == 0:
            time_for_n = explore_budget * 0.5
        else:
            time_for_n = explore_budget * 0.15

        time_for_n = min(time_for_n, time_left - 60)
        if time_for_n < 10:
            continue

        print("\n=== Exploring n=%d (budget: %.1fs) ===" % (n, time_for_n))
        S = np.sqrt(2.0 * float(n))

        inits = generate_initializations(n, rng)
        if given_arr is not None and n == len(given_arr):
            inits.insert(0, ("given", given_arr.copy()))

        n_inits = len(inits)
        per_init = time_for_n / (n_inits * 2.5)  # 2.5 phases per init

        for name, arr in inits:
            time_left = deadline - time.monotonic()
            if time_left < 20:
                break

            S = np.sqrt(2.0 * float(n))
            arr_norm = safe_normalize(arr, n)

            # Phase 1a: Variance descent (short warm-up)
            t_warm = min(per_init * 0.3, 15.0)
            if t_warm > 2.0:
                a_warm, _ = run_variance_descent(arr_norm, n, S, t_warm, rng)
                a_warm = safe_normalize(a_warm, n)
                sc = score_sequence(a_warm)
            else:
                a_warm = arr_norm
                sc = score_sequence(a_warm)

            # Phase 1b: Top-k descent (main optimization)
            time_left = deadline - time.monotonic()
            t_main = min(per_init * 1.2, time_left * 0.3)
            if t_main > 3.0:
                a_main, sc_main = run_toppk_descent(
                    a_warm, n, S, t_main, rng,
                    top_k_init=3, top_k_max=256
                )
                a_main = safe_normalize(a_main, n)
                sc_main = score_sequence(a_main)
                if sc_main < sc:
                    sc = sc_main
                    a_warm = a_main

            # Update global best
            if sc < global_best:
                global_best = sc
                global_a = a_warm.copy()
                print("  [%s n=%d] score=%.6f" % (name, n, global_best))

    print("\n=== After exploration: global_best=%.6f (n=%d) ===" %
          (global_best, len(global_a) if global_a is not None else 0))

    # Phase 2: Polish the best solution
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        remaining = deadline - time.monotonic()

        if remaining > 10:
            print("\n=== Polishing (budget: %.1fs) ===" % remaining)

            # Polish step 1: Medium top-k
            seg1 = remaining * 0.3
            a_p1, s1 = run_toppk_descent(
                global_a, n, S, seg1, rng,
                top_k_init=32, top_k_max=512
            )
            a_p1 = safe_normalize(a_p1, n)
            s1 = score_sequence(a_p1)
            if s1 < global_best:
                global_best = s1
                global_a = a_p1.copy()
            else:
                global_a = safe_normalize(global_a, n)
            print("  Polish 1: %.6f" % global_best)

            # Polish step 2: Fine polish with very high k
            remaining2 = deadline - time.monotonic()
            if remaining2 > 5:
                seg2 = remaining2 * 0.4
                a_p2, s2 = run_polish(
                    global_a, n, S, seg2, rng,
                    top_k_init=128, top_k_max=2048
                )
                a_p2 = safe_normalize(a_p2, n)
                s2 = score_sequence(a_p2)
                if s2 < global_best:
                    global_best = s2
                    global_a = a_p2.copy()
                print("  Polish 2: %.6f" % global_best)

            # Polish step 3: Ultra-fine (if time remains)
            remaining3 = deadline - time.monotonic()
            if remaining3 > 5:
                a_p3, s3 = run_polish(
                    global_a, n, S, remaining3, rng,
                    top_k_init=256, top_k_max=4096
                )
                a_p3 = safe_normalize(a_p3, n)
                s3 = score_sequence(a_p3)
                if s3 < global_best:
                    global_best = s3
                    global_a = a_p3.copy()
                print("  Polish 3: %.6f" % global_best)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64) * 1.0
        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            print("\nVerification: evaluate_sequence = %.6f" % ver if np.isfinite(ver)
                  else "\nVerification: inf (INVALID!)")
            if not np.isfinite(ver):
                print("  Falling back to uniform sequence")
                global_a = np.ones(256, dtype=np.float64)
                final_list = [float(x) for x in global_a]
                ver = evaluate_sequence(final_list)
                print("  Fallback score: %.6f" % ver)
        except Exception as e:
            print("Verification error: %s" % str(e))
            global_a = np.ones(256, dtype=np.float64)

    else:
        global_a = np.ones(256, dtype=np.float64)

    print("\nFinal: length=%d, internal_score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
