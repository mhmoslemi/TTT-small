def coordinate_descent(a, max_sweeps=50, verbose=False):
    """
    Coordinate descent: for each coordinate i, solve the 1D subproblem
    to find the optimal value of a[i].
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    
    if np.sum(a) < 0.01:
        a = np.ones(n)
    
    current_score = score_fast(a)
    
    for sweep in range(max_sweeps):
        max_change = 0.0
        for i in range(n):
            old = a[i]
            a[i] = 0.0
            S_b = np.sum(a)
            
            if S_b < 0.01:
                a[i] = old
                break
            
            bb = np.convolve(a, a)  # background convolution (length 2n-1)
            
            # Linear coefficients: L[k] = 2*a[k-i] if valid and k-i != i, else 0
            L = np.zeros(2 * n - 1)
            # k ranges from i to i+n-1 (where k-i is in [0, n-1])
            k_start = max(0, i)
            k_end = min(2 * n - 1, i + n)  # exclusive
            idx = np.arange(k_start, k_end)
            j_vals = idx - i  # these are in [0, n-1]
            mask = j_vals != i
            L[idx[mask]] = 2.0 * a[j_vals[mask]]
            
            # Quadratic position
            quad_k = 2 * i
            
            # Golden section search for optimal x
            x_lo = 0.0
            x_hi = max(3.0 * old if old > 0 else 0.0, 2.0 * S_b / n, 1e-6)
            # Make sure x_hi is large enough
            # f(x_hi) should be > current score for the search to be valid
            # Actually, let's just make x_hi large enough
            x_hi = max(x_hi, 10.0 * S_b / n)

            def f_of_x(x):
                S_total = S_b + x
                if S_total < 0.01:
                    return np.inf
                conv_val = bb + x * L
                conv_val[quad_k] += x * x
                m = np.max(conv_val)
                if m < 0:
                    m = 0.0
                return 2.0 * n * m / (S_total * S_total)
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = x_hi - gr * (x_hi - x_lo)
            c2 = x_lo + gr * (x_hi - x_lo)
            f1 = f_of_x(c1)
            f2 = f_of_x(c2)

                    c2 = c1
                    f2 = f1
                    c1 = x_hi - gr * (x_hi - x_lo)

                else:
                    x_lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = x_lo + gr * (x_hi - x_lo)
                    f2 = f_of_x(c2)

            best_f = f_of_x(best_x)
            
            # Compare with current
            current_f = f_of_x(old)
            if best_f < current_f:

                max_change = max(max_change, abs(best_x - old))
            else:
                a[i] = old

"""
Coordinate descent optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact 1D golden-section subproblem solving at each coordinate with random
ordering, multiple restarts, and different sequence lengths.
"""

import numpy as np
import time


def score_fast(a):
    """Fast score computation matching evaluate_sequence formula."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return float(2.0 * n * np.max(conv) / (S * S))


def coordinate_descent(a, max_sweeps=100, verbose=False, seed=0, order='random'):
    """
    Coordinate descent: for each coordinate i, solve the 1D subproblem
    via golden section search to find optimal a[i].
    """
    n = len(a)
    a = np.clip(a.astype(np.float64), 0.0, 1000.0).copy()
    
    if np.sum(a) < 0.01:
        a = np.ones(n)
    
    rng = np.random.default_rng(seed)
    current_score = score_fast(a)
    
    for sweep in range(max_sweeps):
        # Determine coordinate order
        if order == 'random':
            coord_order = rng.permutation(n)
        else:
            coord_order = np.arange(n)
        
        max_change = 0.0
        for i in coord_order:
            old = a[i]
            a[i] = 0.0
            S_b = np.sum(a)
            
            if S_b < 0.01:
                a[i] = old
                continue
            
            # Background convolution (with a[i] = 0)
            bb = np.convolve(a, a)
            
            # Linear coefficients L[k] = 2*a[k-i] for valid k-i in [0,n-1], k-i != i
            L = np.zeros(2 * n - 1)
            k_min = i
            k_max = min(2 * n - 1, i + n - 1)
            if k_max >= k_min:
                idx = np.arange(k_min, k_max + 1)
                j_vals = idx - i
                mask = j_vals != i
                if np.any(mask):
                    L[idx[mask]] = 2.0 * a[j_vals[mask]]
            
            quad_k = 2 * i
            
            # Golden section search for optimal x >= 0
            x_lo = 0.0
            # Upper bound: make it large enough
            if old > 0:
                x_hi = max(3.0 * old, 2.0 * S_b / n)
            else:
                x_hi = max(2.0 * S_b / n, 1e-6)
            x_hi = max(x_hi, 5.0 * S_b / n)
            
            def f_of_x(x):
                S_t = S_b + x
                if S_t < 0.01:
                    return np.inf
                conv_val = bb + x * L
                conv_val[quad_k] += x * x
                m = np.max(conv_val)
                if m < 0.0:
                    m = 0.0
                return 2.0 * n * m / (S_t * S_t)
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = x_hi - gr * (x_hi - x_lo)
            c2 = x_lo + gr * (x_hi - x_lo)
            f1 = f_of_x(c1)
            f2 = f_of_x(c2)
            
            for _ in range(35):
                if f1 < f2:
                    x_hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = x_hi - gr * (x_hi - x_lo)
                    f1 = f_of_x(c1)
                else:
                    x_lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = x_lo + gr * (x_hi - x_lo)
                    f2 = f_of_x(c2)
            
            best_x = (x_lo + x_hi) / 2.0
            best_f = f_of_x(best_x)
            
            # Also check the boundary x=0
            f0 = f_of_x(0.0)
            if f0 < best_f:
                best_x = 0.0
                best_f = f0
            
            current_f = f_of_x(old)
            if best_f < current_f:
                a[i] = best_x
                max_change = max(max_change, abs(best_x - old))
            else:
                a[i] = old
        
        if verbose and sweep % 10 == 0:
            s = score_fast(a)
            print(f"    CD sweep {sweep}: score={s:.8f}, max_change={max_change:.2e}")
        
        if max_change < 1e-14:
            break
    
    a = np.clip(a, 0.0, 1000.0)
    final_score = score_fast(a)
    return a, final_score
