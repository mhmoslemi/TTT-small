"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
1. FISTA with adaptive log-sum-exp for broad optimization
2. Active-set projected gradient for precise polishing
3. Barzilai-Borwein step sizes for faster convergence
4. Multiple length exploration (4096, 8192, 16384)
5. LP-based direction finding for final polish
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog, minimize


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT (length 2n-1)."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_lse(a, c, beta):
    """Gradient of log-sum-exp approximation to max(c) via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_gradient_toppk(a, c, top_k):
    """Subgradient using top-k lags."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    w_full = np.zeros(2 * n - 1)
    for k in top_indices:
        w_full[int(k)] = 1.0 / top_k  # uniform weight over top-k
    
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def bb_step(a_prev, a_curr, g_prev, g_curr):
    """Barzilai-Borwein step size."""
    s = a_curr - a_prev
    y = g_curr - g_prev
    ss = np.dot(s, s)
    sy = np.dot(s, y)
    yy = np.dot(y, y)
    if ss < 1e-30 or abs(sy) < 1e-30:
        return 1.0 / max(np.linalg.norm(g_curr), 1e-15)
    alpha_plus = ss / sy if sy > 0 else 1.0 / max(np.linalg.norm(g_curr), 1e-15)
    alpha_minus = sy / yy if yy > 0 else 1.0 / max(np.linalg.norm(g_curr), 1e-15)
    # Alternate between plus and minus
    return max(1e-10, min(alpha_plus, 1e6))

def run_toppk_polish(a_init, n, S, max_time, rng, k_min=5, k_max=200):

    a = np.clip(a_init, 0.0, 1000.0).copy()

def run_active_set(a_init, n, S, max_time, rng, active_frac=0.05):
    """Active set method: focus on the lags near the maximum."""
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s <= 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_local = a.copy()
        
        # Active set: lags within a fraction of the max
        threshold = c_max * (1.0 - active_frac)
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Weighted subgradient: weight each active lag by how close it is to max
        weights = np.zeros(2 * n - 1)
        for k in active:
            w = (c[k] - threshold) / (c_max - threshold + 1e-15)
            weights[int(k)] = w
        weights /= weights.sum()
        
        # Compute gradient of weighted sum
        fsize = 1 << (len(weights) + n - 2).bit_length()
        W = fft(weights, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: try to reduce the max by a fixed fraction
        target_reduction = 1e-6
        step = target_reduction / max(gnorm, 1e-15)
        step = min(step, S / n)
        
        a_new = project_simplex(a - step * grad, S)
        
        # Line search: check if we improved
        c_new = fft_convolve_self(a_new)
        if c_new.max() < c_max:
            a = a_new
        else:
            # Try smaller step
            a_new = project_simplex(a - 0.1 * step * grad, S)
            c_new2 = fft_convolve_self(a_new)
            if c_new2.max() < c_max:
                a = a_new
            else:
                break
        
        # Occasionally perturb
        if rng.random() < 0.01:
            a = project_simplex(a + rng.normal(0, 1e-4 * S / np.sqrt(n), n), S)
    
    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, eta_init=0.1):
    """Mirror descent (multiplicative weights) with FFT subgradient."""
    a = np.clip(a_init, 1e-10, 1000.0).copy()
    s = a.sum()
    if s <= 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        eta = eta_init / (1.0 + 0.01 * it)
        
        # Use top-10 lags for gradient
        top_k = 10
        top_indices = np.argsort(c)[::-1][:top_k]
        w_full = np.zeros(2 * n - 1)
        for k in top_indices:
            w_full[int(k)] = 1.0 / top_k
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new * (S / s_new)
        else:
            a = np.ones(n) * S / n
    
    return best_local, best_score
