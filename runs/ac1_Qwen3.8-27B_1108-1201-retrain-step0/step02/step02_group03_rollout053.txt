"""
LP-guided optimization for minimizing the peak of the self-convolution of
non-negative step functions. The core method iteratively solves a linear
program to find the optimal descent direction, combined with gradient-based
polish and peak-directed local search.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import next_fast_len


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = next_fast_len(n + m - 1)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(n):
        A[k, :k+1] = f[k::-1]
    for k in range(n, 2 * n - 1):
        idx = k - n + 1
        A[k, idx:] = f[n-1:idx-1:-1]
    return A


def lp_solve(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def lp_iterate(f, time_limit, rng):
    """Run LP-based optimization iterations on sequence f."""
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    f = f / S * np.sqrt(2.0 * len(f))  # normalize to sum = sqrt(2n)
    
    best_f = f.copy()
    best_score = local_score(f)
    n = len(f)
    t0 = time.monotonic()
    stagnation = 0
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = np.max(fft_convolve(f, f))
        g = lp_solve(f, M)
        if g is None or np.sum(g) < 1e-10:
            stagnation += 1
            if stagnation > 50:
                break
            continue
        stagnation = 0
        
        # Rescale g to match f's sum
        g = g / np.sum(g) * S
        
        # Line search
        t = 1.0
        found = False
        for _ in range(10):
            f_new = (1.0 - t) * f + t * g
            f_new = np.clip(f_new, 0.0, None)
            s_new = np.sum(f_new)
            if s_new < 1e-10:
                t *= 0.5
                continue
            f_new = f_new / s_new * S
            score_new = local_score(f_new)
            if score_new <= best_score:
                f = f_new
                best_f = f.copy()
                best_score = score_new
                found = True
                break
            t *= 0.5
        
        if not found:
            # Try a very small step
            t = 1e-4
            f_new = (1.0 - t) * f + t * g
            f_new = np.clip(f_new, 0.0, None)
            s_new = np.sum(f_new)
            if s_new > 1e-10:
                f_new = f_new / s_new * S
                f = f_new
    
    return best_f, best_score


def fft_phase_init(n, rng, n_iter=80):
    a = np.ones(n) / n
    N_fft = next_fast_len(2 * n)
    for _ in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def soft_max_value(c, tau):
    c_max = np.max(c)
    return c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau)))


def grad_soft_max_conv(a, tau):
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
