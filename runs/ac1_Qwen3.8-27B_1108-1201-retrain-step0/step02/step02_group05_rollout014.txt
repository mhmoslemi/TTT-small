```python
def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Using the algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    i = np.where(cond)[0][-1] if np.any(cond) else 0
    theta = cssv[i] / (i + 1)
    w = np.maximum(v - theta, 0)
    return w
```

```python
    grad_a = factor * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def soft_max_obj_grad(theta, tau):
    """Objective and gradient for soft-max of autocorrelation."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c = np.maximum(c, 0)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp + 1e-30)
    w = exp_shifted / sum_exp
    
    score = 2.0 * n * soft_max_val
    
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.where(cond)[0][-1]
    theta = cssv[i] / (i + 1)
    w = np.maximum(v - theta, 0)
    return w


def subgradient_descent(a, time_limit, n_peaks=3, lr_init=0.01, lr_end=0.0001, cooling=0.999):
    """Subgradient descent on the exact max autocorrelation."""
    n = len(a)
    current = a.copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    
    lr = lr_init
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        
        # Find top peaks
        peak_indices = np.argsort(c)[-n_peaks:]
        peak_val = c[peak_indices[0]]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in peak_indices:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                grad[j] += current[k - j]
        
        # Normalize
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm
        
        # Step
        candidate = current - lr * grad
        
        # Project to simplex
        candidate = project_simplex(candidate)
        
        new_score = local_score(candidate)
        
        if new_score < best_score:
            best_score = new_score
            best = candidate.copy()
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
        
        lr = max(lr * cooling, lr_end)
    
    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2 * np.pi * t * t / (2 * n))
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Bell-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def local_search(a_init, time_limit, rng, n_peaks=5):
    """
    Structure-aware local search with multiple mutation types.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.001
    cooling = 0.9999
    
    iters = 0
    last_print = t0
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        c = fft_convolve(current, current)
        
        if r < 0.25:
            # Peak-targeted: reduce top contributors to the worst peak
            peak_idx = int(np.argmax(c))
            # Compute contribution of each element to this peak
            contribs = np.zeros(n)
            j_start = max(0, peak_idx - n + 1)
            j_end = min(peak_idx, n - 1)
            for j in range(j_start, j_end + 1):
                contribs[j] = current[j] * current[peak_idx - j]
            
            top_k = max(5, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            # Reduce top contributors
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            # Increase some others
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(20, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.015, size=n_inc))
                
        elif r < 0.40:
            # Multi-peak reduction
            peak_indices = np.argsort(c)[-n_peaks:]
            # Build combined contribution
            combined = np.zeros(n)
            for k in peak_indices:
                j_start = max(0, k - n + 1)
                j_end = min(k, n - 1)
                for j in range(j_start, j_end + 1):
                    combined[j] += current[j] * current[k - j]
            
            # Reduce high-contribution elements
            threshold = np.percentile(combined[combined > 0], 70) if np.any(combined > 0) else 0
            high_idx = np.where(combined > threshold)[0]
            if len(high_idx) > 0:
                candidate[high_idx] *= np.exp(rng.normal(-0.04, 0.015, size=len(high_idx)))
            low_idx = np.where(combined <= threshold)[0]
            if len(low_idx) > 0:
                n_inc = min(20, len(low_idx))
                inc_idx = rng.choice(low_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
                
        elif r < 0.55:
            # Subgradient-inspired step
            peak_idx = int(np.argmax(c))
            grad = np.zeros(n)
            j_start = max(0, peak_idx - n + 1)
            j_end = min(peak_idx, n - 1)
            for j in range(j_start, j_end + 1):
                grad[j] = current[peak_idx - j]
            
            # Direction: reduce where gradient is high
            step = 0.01 * grad / (np.max(grad) + 1e-15)
            candidate = current - step
            candidate = project_simplex(candidate)
            
        elif r < 0.70:
            # Block perturbation
            width = max(2, int(rng.integers(2, max(3, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))
            
        elif r < 0.85:
            # Smooth global perturbation
            scale = rng.normal(0, 0.02) + rng.uniform(0, 0.01)
            candidate = current * np.exp(rng.normal(0.0, 0.02, size=n))
            # Add a smooth trend
            trend = np.linspace(-scale, scale, n)
            candidate = current * np.exp(trend)
            
        else:
            # Re-center: try to make autocorrelation more flat
            c_avg = np.mean(c)
            c_dev = c - c_avg
            # Elements that contribute to above-average lags should be reduced
            # This is approximate - use the peak lags
            hot_lags = np.where(c > c_avg * 1.5)[0]
            if len(hot_lags) > 0:
                pull = np.zeros(n)
                for k in hot_lags[:10]:  # Limit to top 10
                    j_start = max(0, k - n + 1)
                    j_end = min(k, n - 1)
                    for j in range(j_start, j_end + 1):
                        pull[j] += current[k - j] * (c[k] - c_avg)
                if np.max(pull) > 1e-12:
                    pull = pull / np.max(pull)
                    candidate = current * np.exp(-0.03 * pull)
        
        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score
        
        temperature = max(temperature * cooling, 1e-10)
        iters += 1
        
        if time.monotonic() - last_print > 10.0:
            last_print = time.monotonic()
            print(f"      LS: iter={iters}, score={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s, temp={temperature:.6f}")
    
    return best, best_score


def run_optimization(a0, time_limit, deadline):
    """
    Run full optimization pipeline on a starting sequence.
    Returns best sequence and score.
    """
    n = len(a0)
    best = a0.copy()

    theta = np.log(a0 + 1e-30)
    
    # Phase A: p-norm L-BFGS

    for p in p_schedule:
        if time.monotonic() > deadline:
            break

        maxiter = int(1000 * time_per_p / 60.0)

        try:

        except Exception as e:
            print(f"      L-BFGS p={p} failed: {e}")
            continue

        s = local_score(a_new)

        if s < best_score:
            best_score = s
            best = a_new.copy()
            theta = result.x.copy()
            print(f"      p={p}: {s:.6f} (improved, {elapsed:.1f}s)")
        else:
            print(f"      p={p}: {s:.6f} (no improvement, {elapsed:.1f}s)")
    
    # Phase B: Soft-max L-BFGS
    tau_schedule = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

            )
        except Exception as e:
            print(f"      SoftMax tau={tau} failed: {e}")
            continue
        
        a_new = softmax(result.x)
        s = local_score(a_new)
        elapsed = time.monotonic() - t_start
        
        if s < best_score:
            best_score = s
            best = a_new.copy()
            theta = result.x.copy()
            print(f"      tau={tau}: {s:.6f} (improved, {elapsed:.1f}s)")
        else:
            print(f"      tau={tau}: {s:.6f} (no improvement, {elapsed:.1f}s)")

            print(f"      Subgradient: {best_score:.6f} (improved)")
        else:
            print(f"      Subgradient: {sg_score:.6f} (no improvement)")
    
    return best, best_score
```

```python
"""
Multi-phase optimization for minimum peak autocorrelation non-negative sequences.
Combines aggressive p-norm/soft-max L-BFGS, subgradient descent on the exact max,
structure-aware local search with multi-peak targeting, and coordinate descent.
Explores multiple sequence lengths starting from provided sequence.

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def compute_autocorr(a):
    """Compute full autocorrelation."""
    return fft_convolve(a, a)


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||autocorr(softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c = np.maximum(c, 0.0)

    cp = np.maximum(c, 1e-30) ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def soft_max_obj_grad(theta, tau):
    """Objective and gradient for 2n * soft_max(autocorr, tau)."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c = np.maximum(c, 0.0)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp + 1e-30)
    w = exp_shifted / sum_exp

    score = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.where(cond)[0][-1]
    theta = cssv[i] / (i + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def peak_subgradient(a, peak_k):
    """Compute subgradient of autocorrelation at lag peak_k w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, peak_k - n + 1)
    j_end = min(peak_k, n - 1)
    for j in range(j_start, j_end + 1):
        grad[j] = a[peak_k - j]
    return grad


def subgradient_descent(a, time_limit, n_peaks=3, lr_init=0.008, lr_end=0.0005, cooling=0.9995):
    """Subgradient descent directly on the max autocorrelation."""
    n = len(a)
    current = a.copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score

    lr = lr_init
    t0 = time.monotonic()
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        c = compute_autocorr(current)

        # Find top peaks
        peak_indices = np.argsort(c)[-n_peaks:]

        # Compute combined subgradient
        grad = np.zeros(n)
        for k in peak_indices:
            grad += peak_subgradient(current, k)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm

        # Try a few step sizes (mini line search)
        best_candidate = current
        best_candidate_score = current_score
        for lr_try in [lr, lr * 0.5, lr * 0.25]:
            candidate = current - lr_try * grad
            candidate = project_simplex(candidate)
            s = local_score(candidate)
            if s < best_candidate_score:
                best_candidate_score = s
                best_candidate = candidate

        if best_candidate_score < best_score:
            best_score = best_candidate_score
            best = best_candidate.copy()

        if best_candidate_score < current_score:
            current = best_candidate
            current_score = best_candidate_score

        lr = max(lr * cooling, lr_end)

        if time.monotonic() - last_print > 5.0:
            last_print = time.monotonic()
            print(f"        SG: score={current_score:.6f}, best={best_score:.6f}, "
                  f"t={time.monotonic()-t0:.0f}s, lr={lr:.6f}")

    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * t * t / (2.0 * n))
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Bell-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def coordinate_descent(a, time_limit, n_steps=5):
    """
    Coordinate descent: for each coordinate, do a 1D minimization.
    """
    n = len(a)
    current = a.copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()

    step_budget = max(1, int(time_limit / (n * 0.001)))

    for outer in range(min(step_budget, 100)):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            # Try a few perturbations of coordinate j
            v0 = current[j]
            if v0 < 1e-10:
                continue

            best_v = v0
            best_s = current_score

            # Try increasing and decreasing
            for scale in [1.05, 1.02, 0.98, 0.95, 0.90, 1.10, 0.85]:
                trial = current.copy()
                trial[j] = v0 * scale
                trial = np.maximum(trial, 0.0)
                trial = trial / np.sum(trial)
                s = local_score(trial)
                if s < best_s:
                    best_s = s
                    best_v = v0 * scale

            if best_v != v0:
                current[j] = best_v
                # Renormalize
                current = np.maximum(current, 0.0)
                current = current / np.sum(current)
                current_score = local_score(current)
                improved = True

                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()

        if not improved:
            break

    return best, best_score


def local_search(a_init, time_limit, rng, n_peaks=5):
    """
    Structure-aware local search with multiple mutation types and reheating.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.0008
    cooling = 0.99995
    base_temp = 1e-11

    iters = 0
    last_print = t0
    no_improve_count = 0
    reheat_interval = 5000

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        c = compute_autocorr(current)

        if r < 0.20:
            # Peak-targeted: reduce top contributors to the worst peak
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            j_start = max(0, peak_idx - n + 1)
            j_end = min(peak_idx, n - 1)
            for j in range(j_start, j_end + 1):
                contribs[j] = current[j] * current[peak_idx - j]

            top_k = max(5, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(15, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.012, size=n_inc))

        elif r < 0.35:
            # Multi-peak reduction
            peak_indices = np.argsort(c)[-min(n_peaks, 5):]
            combined = np.zeros(n)
            for k in peak_indices:
                j_start = max(0, k - n + 1)
                j_end = min(k, n - 1)
                for j in range(j_start, j_end + 1):
                    combined[j] += current[j] * current[k - j]

            if np.max(combined) > 1e-15:
                threshold = np.percentile(combined[combined > 1e-15], 65)
                high_idx = np.where(combined > threshold)[0]
                if len(high_idx) > 0:
                    candidate[high_idx] *= np.exp(rng.normal(-0.035, 0.012, size=len(high_idx)))
                low_idx = np.where(combined <= threshold)[0]
                if len(low_idx) > 0:
                    n_inc = min(15, len(low_idx))
                    inc_idx = rng.choice(low_idx, size=n_inc, replace=False)
                    candidate[inc_idx] *= np.exp(rng.normal(0.035, 0.012, size=n_inc))

        elif r < 0.50:
            # Subgradient-inspired deterministic step
            peak_indices = np.argsort(c)[-min(n_peaks, 3):]
            grad = np.zeros(n)
            for k in peak_indices:
                grad += peak_subgradient(current, k)
            grad_max = np.max(grad)
            if grad_max > 1e-15:
                step = 0.008 * grad / grad_max
                candidate = current - step
                candidate = project_simplex(candidate)

        elif r < 0.60:
            # Balancing: try to flatten the autocorrelation
            c_mean = np.mean(c)
            c_std = np.std(c)
            if c_std > 1e-12:
                # Elements contributing to hot lags
                hot_lags = np.where(c > c_mean + 0.5 * c_std)[0][:8]
                pull = np.zeros(n)
                for k in hot_lags:
                    pull += peak_subgradient(current, k) * (c[k] - c_mean)
                pull_max = np.max(pull)
                if pull_max > 1e-15:
                    pull = pull / pull_max
                    candidate = current * np.exp(-0.02 * pull)

        elif r < 0.75:
            # Block perturbation
            width = max(2, int(rng.integers(2, max(3, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))

        elif r < 0.88:
            # Smooth global trend
            scale = rng.normal(0, 0.015)
            trend = np.linspace(-scale, scale, n)

            candidate = current * np.exp(trend + noise)

        else:
            # Pair swap: move mass from high-autocorr contributor to low
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            j_start = max(0, peak_idx - n + 1)
            j_end = min(peak_idx, n - 1)
            for j in range(j_start, j_end + 1):

            donors = np.argsort(contribs)[-3:]
            recipients = np.argsort(contribs)[:3]
            for d in donors:

                    if d != rec:
                        amt = current[d] * 0.03 * rng.random()
                        candidate[d] -= amt
                        candidate[rec] += amt

        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)

        if new_score < current_score:

            current_score = new_score
            no_improve_count = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, base_temp)):
            current = candidate
            current_score = new_score

        no_improve_count += 1
        if no_improve_count > reheat_interval:
            # Reheat to escape local minimum
            temperature = min(temperature * 10, 0.001)
            no_improve_count = 0
            # Also try a larger perturbation
            current = current * np.exp(rng.normal(0, 0.03, n))
            current = np.maximum(current, 1e-12)
            current = current / np.sum(current)
            current_score = local_score(current)

        temperature = max(temperature * cooling, base_temp)
        iters += 1

        if time.monotonic() - last_print > 8.0:
            last_print = time.monotonic()
            print(f"        LS: iter={iters}, score={current_score:.6f}, best={best_score:.6f}, "

def run_lbgfs_pipeline(a0, deadline):
    """
    Full L-BFGS pipeline: p-norm annealing then soft-max refinement.
    Returns (best_sequence, best_score, best_theta).
    """
    n = len(a0)
    best = a0.copy()
    best_score = local_score(a0)
    theta = np.log(a0 + 1e-30)

    # Phase A: p-norm annealing
    p_schedule = [2, 3, 4, 6, 8, 12, 16, 24, 32, 48, 64]
    for p in p_schedule:
        if time.monotonic() > deadline - 10:
            break

        maxiter = 300
        try:
            result = minimize(
                pnorm_obj_grad, theta,
                method='L-BFGS-B',
                jac=True, args=(p,),
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxfun': maxiter * 4, 'maxls': 50}
            )
        except Exception:
            continue

        a_new = softmax(result.x)
        s = local_score(a_new)

        if s < best_score:
            best_score = s
            best = a_new.copy()
            theta = result.x.copy()
            print(f"          p={p}: {s:.8f} ***")
        else:
            print(f"          p={p}: {s:.8f}")

    # Phase B: Soft-max refinement
    tau_schedule = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for tau in tau_schedule:
        if time.monotonic() > deadline - 5:
            break

        maxiter = 400
        try:
            result = minimize(
                soft_max_obj_grad, theta,
                method='L-BFGS-B',
                jac=True, args=(tau,),
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-15,
                         'maxfun': maxiter * 4, 'maxls': 50}
            )
        except Exception:
            continue

        a_new = softmax(result.x)
        s = local_score(a_new)

        if s < best_score:
            best_score = s
            best = a_new.copy()
            theta = result.x.copy()
            print(f"          tau={tau}: {s:.8f} ***")
        else:
            print(f"          tau={tau}: {s:.8f}")

    return best, best_score, theta

        a_interp = a_interp / np.sum(a_interp)

        # Also try a chirp init
        a_chirp = make_chirp_init(n, 0.3 + 0.3 * rng.random(), rng)

        # Try both initializations if time allows
        inits = [a_interp, a_chirp]
        n_inits = 2 if time_alloc > 50 else 1

        for i in range(n_inits):
            if time.monotonic() > phase_deadline - 5:
                break

            a0 = inits[i]
            sub_deadline = phase_deadline - 5
            res, sc, _ = run_lbgfs_pipeline(a0, sub_deadline)

            if sc < best_score:
                best_score = sc
                best_seq = res.copy()
                print(f"  n={n}, init={i}: {sc:.8f} *** NEW BEST ***")
            else:
                print(f"  n={n}, init={i}: {sc:.8f}")

    # === Phase 4: Intensive local search polish ===
    print(f"\n=== Phase 4: Intensive local search (best={best_score:.8f}) ===")
    if best_seq is not None:
        polish_time = max(60.0, deadline - time.monotonic() - 40)
        print(f"  Local search: {polish_time:.0f}s, n={len(best_seq)}")

        polished, p_score = local_search(best_seq, polish_time, rng, n_peaks=5)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After local search: {best_score:.8f} *** NEW BEST ***")
        else:
            print(f"  After local search: {p_score:.8f} (no improvement)")

        # Final subgradient pass
        if time.monotonic() < deadline - 20:
            sg_time = min(40.0, deadline - time.monotonic() - 10)
            sg_result, sg_score = subgradient_descent(
                best_seq, sg_time, n_peaks=5, lr_init=0.003, lr_end=0.0003, cooling=0.9997
            )
            if sg_score < best_score:
                best_score = sg_score
                best_seq = sg_result.copy()
                print(f"  Final subgradient: {best_score:.8f} *** NEW BEST ***")

        # Final short local search
        if time.monotonic() < deadline - 10:
            ls_time = min(30.0, deadline - time.monotonic() - 5)
            final_polished, fp_score = local_search(best_seq, ls_time, rng, n_peaks=5)
            if fp_score < best_score:
                best_score = fp_score
                best_seq = final_polished.copy()
                print(f"  Final local search: {best_score:.8f} *** NEW BEST ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.8f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
