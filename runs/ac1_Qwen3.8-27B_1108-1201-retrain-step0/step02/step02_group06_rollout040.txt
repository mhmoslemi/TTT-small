"""
Optimized search for non-negative step functions minimizing peak autocorrelation.
Uses LP-based mirror descent for global search and efficient coordinate descent
with incremental convolution updates for local polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, M, n):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)[k] <= M for all k."""
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b) → minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: (f*b)[k] <= M for k = 0, ..., 2n-2
    # (f*b)[k] = sum_{i: 0<=i<n, 0<=k-i<n} f[i] * b[k-i]
    # For a given k, b[j] appears when k-j is in [0, n-1], i.e., j in [k-n+1, k] ∩ [0, n-1]
    # So (f*b)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * b[j]
    
    # Build constraint matrix
    # A_ub[k, j] = f[k-j] if max(0,k-n+1) <= j <= min(n-1,k), else 0
    # This is a "convolution matrix"
    
    n_k = 2 * n - 1
    A_ub = np.zeros((n_k, n))
    for k in range(n_k):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A_ub[k, j] = f[k - j]
    
    b_ub = np.full(n_k, M)
    
    # Non-negativity: b >= 0 → bounds
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return -result.fun / n  # normalized direction (sum = 1)
        # Actually return the raw b
        # return result.x
    return None

"""
Optimized search for non-negative step functions minimizing peak convolution.
Uses LP-based mirror descent for global structure, efficient coordinate descent
with incremental convolution updates for local polishing, and multi-start
exploration across different sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def make_convolution_matrix_indices(n):
    """Precompute index ranges for each lag k in the convolution."""
    ranges = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        ranges.append((j_lo, j_hi))
    return ranges


def lp_solve_direction(f, M, n):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)[k] <= M for all k."""
    c = -np.ones(n)
    n_k = 2 * n - 1
    
    # Build constraint matrix efficiently
    A_ub = np.zeros((n_k, n))
    for k in range(n_k):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            rows = slice(j_lo, j_hi + 1)
            cols = range(j_lo, j_hi + 1)
            for j in cols:
                A_ub[k, j] = f[k - j]
    
    b_ub = np.full(n_k, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            b = result.x
            s = np.sum(b)
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    return None


def lp_global_search(f_init, n, n_iters=20, t=0.3, rng=None):
    """LP-based mirror descent: iterate to reduce peak."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    f = f_init.copy()
    f = np.clip(f, 0, None)
    s = np.sum(f)
    if s < 1e-10:
        return f_init
    f = f / s  # normalize to sum=1
    
    best_f = f.copy()
    best_peak = np.max(fft_convolve(f, f))
    
    for it in range(n_iters):
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        direction = lp_solve_direction(f, M * (1 + 1e-6), n)
        if direction is None:
            # Fallback: use negated gradient
            k_star = np.argmax(conv)
            grad = np.zeros(n)
            for j in range(n):
                kj = k_star - j
                if 0 <= kj < n:
                    grad[j] = 2 * f[kj]
            direction = project_simplex(f - 0.01 * grad)
        
        # Update
        f_new = (1 - t) * f + t * direction
        f_new = np.clip(f_new, 0, None)
        s_new = np.sum(f_new)
        if s_new < 1e-10:
            break
        f_new = f_new / s_new
        
        new_peak = np.max(fft_convolve(f_new, f_new))
        if new_peak < best_peak:
            best_peak = new_peak
            best_f = f_new.copy()
        f = f_new
        t = max(0.05, t * 0.85)
    
    return best_f

                conv = new_conv
                current_max_val = np.max(conv)
                k_star = int(np.argmax(conv))
                no_improve = 0
            else:
                no_improve += 1
                
        else:
            # Shape adjustment: nudge toward a monotonic/exponential shape
            # Compute target shape based on current "lopsidedness"
            t_arr = np.arange(n, dtype=np.float64) / max(n - 1, 1)
            # Exponential decay template
            alpha = rng.uniform(0.3, 2.0)
            target = np.exp(-alpha * t_arr)
            target = target / np.sum(target)
            
            # Blend current with target
            blend = rng.uniform(0.01, 0.05)
            candidate = (1 - blend) * a + blend * target
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            
            new_conv = fft_convolve(candidate, candidate)
            new_max = np.max(new_conv)
            
            if new_max < current_max_val:
                a = candidate
                conv = new_conv
                current_max_val = new_max
                k_star = int(np.argmax(conv))
                no_improve = 0
            else:
                no_improve += 1
        
        # Track best
        if current_max_val < best_max:
            best_max = current_max_val
            best_a = a.copy()
        
        # Periodic full recomputation to fix drift
        if step % 500 == 0:
            conv = fft_convolve(a, a)
            current_max_val = np.max(conv)
            k_star = int(np.argmax(conv))
        
        # Reheat if stuck
        if no_improve > max_no_improve:
            # Larger perturbation
            for _ in range(20):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                frac = rng.uniform(0.1, 0.3)
                delta = frac * a[i]
                if a[i] - delta > 0:
                    a[i] -= delta
                    a[j] += delta
            conv = fft_convolve(a, a)
            current_max_val = np.max(conv)
            k_star = int(np.argmax(conv))
            no_improve = 0
        
        if verbose and step % 5000 == 0:
            sc = 2.0 * n * best_max
            print(f"    CD step={step}, best_score={sc:.6f}, "
                  f"cur={2.0*n*current_max_val:.6f}, t={elapsed:.0f}s")
    
    return best_a, 2.0 * n * best_max


def adam_simplex(a_init, time_limit, rng, n):
    """Adam optimization on the simplex for max-conv reduction."""
    a = a_init.copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a_init, np.inf
    a = a / s
    
    best_a = a.copy()
    best_val = np.max(fft_convolve(a, a))
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.5
    step = 0
    t0 = time.monotonic()
    
    tau_start = 10.0 / n
    tau_end = 0.5 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.6)
        
        conv = fft_convolve(a, a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / tau
        # Clip for numerical stability
        shifted = np.clip(shifted, -50, 50)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        
        # Gradient: g[j] = 2 * sum_k w[k] * a[k-j] (if 0<=k-j<n)
        # This is a cross-correlation
        size = 2 * n
        fa = np.fft.rfft(a, size)
        fw = np.fft.rfft(w, size)
        corr = np.fft.irfft(np.conj(fa) * fw, size)[:n]
        grad = 2.0 * corr
        
        # Project gradient to simplex tangent space
        grad = grad - np.mean(grad)
        
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        lr_eff = lr / np.sqrt(step + 1)
        a = a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a)
        
        if step % 100 == 0:
            val = np.max(fft_convolve(a, a))
            if val < best_val:
                best_val = val
                best_a = a.copy()
        
        if step % 2000 == 0:
            sc = 2.0 * n * best_val
            print(f"      Adam step={step}, score={sc:.6f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, 2.0 * n * best_val


def spectral_init(n, rng, n_iter=50):
    """Spectral initialization via alternating projections."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    
    for it in range(n_iter):
        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        corr = np.fft.irfft(power, 2 * n)[:n]
        alpha = 1.0 / (n + 1.0)
        a = a - alpha * corr
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def ramp_init(n, rng, style='exp_decay'):
    """Create structured initializations."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if style == 'exp_decay':
        alpha = rng.uniform(0.5, 2.5)
        a = np.exp(-alpha * t)
    elif style == 'exp_rise':
        alpha = rng.uniform(0.5, 2.5)
        a = np.exp(alpha * t)
    elif style == 'power':
        p = rng.uniform(0.3, 3.0)
        a = (1.0 - t + 1e-6) ** p
    elif style == 'asym_exp':
        alpha1 = rng.uniform(0.3, 2.0)
        alpha2 = rng.uniform(0.3, 2.0)
        center = rng.uniform(0.2, 0.8)
        a = np.exp(-alpha1 * np.maximum(t - center, 0) - alpha2 * np.maximum(center - t, 0))
    elif style == 'linear_dec':
        a = 1.0 + 0.5 * (1.0 - t)
    else:
        a = np.exp(-1.0 * t)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search minimizing 2n*max(conv(a,a))/(sum a)^2 for non-negative a.
    Phase 1: LP-based global search + diverse inits.
    Phase 2: Aggressive coordinate descent with incremental updates.
    Phase 3: Multi-start exploration with remaining time.
    """
    t_global = time.monotonic()
    time_budget = max(30.0, float(budget_s) - 45.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Try to use provided sequence
    start_seq = None
    try:
        hs1 = globals().get('height_sequence_1', None)
        if hs1 is not None:
            start_seq = np.clip(np.asarray(hs1, dtype=np.float64), 0.0, 1000.0)
            s0 = score_fast(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s0:.6f}")
            if s0 < best_score and np.isfinite(s0):
                best_score = s0
                best_seq = start_seq.copy()
    except Exception as e:
        print(f"Warning: could not load provided sequence: {e}")
    
    # Phase 1: Generate candidates via various methods
    print("\n=== Phase 1: Global search ===")
    phase1_time = time_budget * 0.25
    phase1_deadline = t_global + phase1_time
    
    candidates = []  # list of (normalized_array, n)
    
    # Try provided sequence
    if start_seq is not None:
        a_norm = start_seq / np.sum(start_seq)
        candidates.append((a_norm, len(a_norm)))
    
    # LP-based search for n=512
    n_lp = 512
    a_start = ramp_init(n_lp, rng, 'exp_decay')
    a_start = a_start / np.sum(a_start)
    lp_result = lp_global_search(a_start, n_lp, n_iters=15, t=0.3, rng=rng)
    lp_score = 2.0 * n_lp * np.max(fft_convolve(lp_result, lp_result))
    print(f"  LP n={n_lp}: score={lp_score:.6f}")
    candidates.append((lp_result, n_lp))
    if lp_score < best_score:
        best_score = lp_score
        best_seq = lp_result * np.sqrt(2.0 * n_lp)
    
    # Try a few more inits for LP
    if time.monotonic() < phase1_deadline:
        for style in ['exp_rise', 'power', 'asym_exp']:
            if time.monotonic() > phase1_deadline:
                break
            a_s = ramp_init(n_lp, rng, style)
            a_s = a_s / np.sum(a_s)
            lp_r = lp_global_search(a_s, n_lp, n_iters=10, t=0.3, rng=rng)
            sc = 2.0 * n_lp * np.max(fft_convolve(lp_r, lp_r))
            print(f"  LP n={n_lp} style='{style}': score={sc:.6f}")
            candidates.append((lp_r, n_lp))
            if sc < best_score:
                best_score = sc
                best_seq = lp_r * np.sqrt(2.0 * n_lp)
    
    # Spectral init + Adam for a couple of lengths
    if time.monotonic() < phase1_deadline - 30:
        for n_try in [512, 1024]:
            if time.monotonic() > phase1_deadline - 30:
                break
            a_sp = spectral_init(n_try, rng, n_iter=40)
            t_alloc = min(60.0, phase1_deadline - time.monotonic() - 15)
            if t_alloc < 10:
                break
            a_best, ad_score = adam_simplex(a_sp, t_alloc, rng, n_try)
            actual_sc = score_fast(a_best * np.sqrt(2.0 * n_try))
            print(f"  Adam n={n_try}: score={actual_sc:.6f}")
            candidates.append((a_best, n_try))
            if actual_sc < best_score:
                best_score = actual_sc
                best_seq = a_best * np.sqrt(2.0 * n_try)
    
    print(f"\n  Best after Phase 1: {best_score:.6f}")
    
    # Phase 2: Coordinate descent polishing (main workhorse)
    print("\n=== Phase 2: Coordinate descent polishing ===")
    phase2_time = min(time_budget * 0.6, deadline - time.monotonic() - 120)
    if phase2_time < 30:
        phase2_time = deadline - time.monotonic() - 30
    
    # Polish the best candidate
    if best_seq is not None:
        n_best = len(best_seq)
        a_norm_best = best_seq / np.sum(best_seq)
        print(f"  Polishing n={n_best}, initial score={best_score:.6f}")
        a_polished, pol_score = coord_descent_optimize(
            a_norm_best, phase2_time * 0.7, rng, verbose=True
        )
        pol_scaled = a_polished * np.sqrt(2.0 * n_best)
        pol_actual = score_fast(pol_scaled)
        print(f"  After polish: score={pol_actual:.6f}")
        if pol_actual < best_score:
            best_score = pol_actual
            best_seq = pol_scaled.copy()
            print(f"    *** New best: {best_score:.6f} ***")
    
    # Also polish other top candidates if time remains
    remaining = deadline - time.monotonic() - 60
    if remaining > 60:
        candidates.sort(key=lambda x: 2.0 * x[1] * np.max(fft_convolve(x[0], x[0])))
        for a_cand, n_cand in candidates[1:4]:
            if remaining < 30:
                break
            t_alloc = remaining * 0.4
            print(f"  Polishing alt candidate n={n_cand}, "
                  f"init_score={2.0*n_cand*np.max(fft_convolve(a_cand, a_cand)):.6f}")
            a_p, sc_p = coord_descent_optimize(a_cand, t_alloc, rng, verbose=False)
            sc_p_actual = score_fast(a_p * np.sqrt(2.0 * n_cand))
            print(f"    result: {sc_p_actual:.6f}")
            if sc_p_actual < best_score:
                best_score = sc_p_actual
                best_seq = a_p * np.sqrt(2.0 * n_cand)
                print(f"    *** New best: {best_score:.6f} ***")
            remaining = deadline - time.monotonic() - 60
    
    # Phase 3: Final exploration with remaining time
    remaining = deadline - time.monotonic() - 45
    print(f"\n=== Phase 3: Final exploration (remaining {remaining:.0f}s) ===")
    
    if remaining > 30 and best_seq is not None:
        # Try a fresh start with a different length
        n_fresh = 768
        a_fresh = ramp_init(n_fresh, rng, 'asym_exp')
        a_fresh = a_fresh / np.sum(a_fresh)
        t_alloc = min(remaining * 0.5, 120)
        a_f, sc_f = coord_descent_optimize(a_fresh, t_alloc, rng, verbose=False)
        sc_f_actual = score_fast(a_f * np.sqrt(2.0 * n_fresh))
        print(f"  Fresh n={n_fresh}: score={sc_f_actual:.6f}")
        if sc_f_actual < best_score:
            best_score = sc_f_actual
            best_seq = a_f * np.sqrt(2.0 * n_fresh)
            print(f"    *** New best: {best_score:.6f} ***")
        
        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            # Another fresh start
            n_fresh2 = 640
            a_fresh2 = spectral_init(n_fresh2, rng, n_iter=60)
            t_alloc2 = min(remaining * 0.7, 120)
            a_f2, sc_f2 = coord_descent_optimize(a_fresh2, t_alloc2, rng, verbose=False)
            sc_f2_actual = score_fast(a_f2 * np.sqrt(2.0 * n_fresh2))
            print(f"  Fresh n={n_fresh2}: score={sc_f2_actual:.6f}")
            if sc_f2_actual < best_score:
                best_score = sc_f2_actual
                best_seq = a_f2 * np.sqrt(2.0 * n_fresh2)
                print(f"    *** New best: {best_score:.6f} ***")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    
    # Last resort fallback
    n_fall = 512
    a_fall = np.exp(-np.linspace(0, 1, n_fall))
    a_fall = a_fall / np.sum(a_fall)
    a_fall_scaled = a_fall * np.sqrt(2.0 * n_fall)
    return [float(max(0.0, min(1000.0, x))) for x in a_fall_scaled]
