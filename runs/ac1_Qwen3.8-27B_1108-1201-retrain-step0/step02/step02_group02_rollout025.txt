from scipy.optimize import linprog
import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def solve_lp_for_g(f, M):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns optimal g.
    """
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) -> minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_i f[i] * g[k-i] for valid i
    # For each k, the coefficient of g[j] in (f*g)[k] is f[k-j] if 0 <= k-j < n, else 0.
    
    A_ub = np.zeros((2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    for k in range(2*n-1):
        for j in range(n):
            # g[j] contributes to (f*g)[k] as f[k-j] if 0 <= k-j < n
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]
    
    # Bounds: g_j >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.status != 0:
        return None
    
    g = result.x
    # Clean up small negatives
    g = np.clip(g, 0, None)
    return g

LP-based SOTA optimization with soft-max gradient descent and peak-shave polish
for minimizing the step function evaluation score.
"""

import time

    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:

    fa = np.fft.rfft(a, size)

    a = np.clip(a, 0.0, 1000.0)

    if n == 0:

    S = np.sum(a)
    if S < 0.01:

def build_lp_matrix(f):
    """
    Build the constraint matrix A for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < len(f), else 0
    for k = 0, ..., 2n-2 and j = 0, ..., n-1
    """
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # g[j] contributes to (f*g)[k] as f[k-j] for k in [j, j+n-1]
        valid_k = slice(j, j+n)
        A[valid_k, j] = f
    return A

def solve_lp(f, M):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns optimal g or None if infeasible.
    """
    n = len(f)
    c_obj = -np.ones(n)
    A = build_lp_matrix(f)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.status != 0:
        return None

"""
LP-based SOTA optimization combined with soft-max gradient descent and peak-shave polish
for minimizing the step function evaluation score. Uses the cutting-plane / Frank-Wolfe
style LP iteration as the primary engine, with adaptive restarts and multi-phase polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def build_lp_matrix(f):
    """Build constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<len(f)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        valid_k_start = j
        valid_k_end = j + n
        A[valid_k_start:valid_k_end, j] = f
    return A


def solve_lp(f, M, verbose=False):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns optimal g or None.
    """
    n = len(f)
    c_obj = -np.ones(n)
    A = build_lp_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})

    if result.status != 0:
        return None

    g = np.clip(result.x, 0, None)
    return g


def sota_iterate(f, t_values=(0.5, 0.3, 0.2, 0.1, 0.05, 0.01)):
    """
    One SOTA iteration: compute M, solve LP, try different step sizes.
    Returns (best_f, best_score, M, g, score_f_only).
    """
    S = np.sum(f)
    c = fft_convolve(f, f)
    M = np.max(c)

    if M <= 0:
        return f.copy(), local_score(f), M, f.copy()

    g_raw = solve_lp(f, M)
    if g_raw is None or np.sum(g_raw) < 1e-10:
        return f.copy(), local_score(f), M, f.copy()

    # Rescale g to have the same sum as f
    g = g_raw * (S / np.sum(g_raw))

    best_f = f.copy()
    best_score = local_score(f)

    for t in t_values:
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        S_new = np.sum(f_new)
        if S_new < 0.01:
            continue
        score = local_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()

    return best_f, best_score, M, g


def peak_shave_step(f, rng, strength=0.03):
    """
    One peak-shave step: reduce top contributors to the peak, increase others.
    Returns modified f.
    """
    n = len(f)
    c = fft_convolve(f, f)
    peak_idx = int(np.argmax(c))

    # Compute contribution of each element to the peak
    contribs = np.zeros(n)
    for i in range(n):
        if 0 <= peak_idx - i < n:
            contribs[i] = f[i] * f[peak_idx - i]

    total_contrib = np.sum(contribs)
    if total_contrib < 1e-10:
        return f.copy()

    # Reduce top contributors proportionally, increase others
    top_k = max(2, n // 10)
    sorted_idx = np.argsort(contribs)[::-1]
    top_idx = sorted_idx[:top_k]
    bottom_idx = sorted_idx[top_k:]

    reduction = strength * rng.uniform(0.5, 1.5)
    mass_removed = 0.0
    for i in top_idx:
        red = f[i] * reduction
        f[i] -= red
        mass_removed += red

    if len(bottom_idx) > 0:
        # Distribute to bottom proportionally to their (small) values
        bottom_vals = f[bottom_idx].copy()
        total_bv = np.sum(bottom_vals)
        if total_bv < 1e-10:
            f[bottom_idx] += mass_removed / len(bottom_idx)
        else:
            f[bottom_idx] += mass_removed * bottom_vals / total_bv

    f = np.clip(f, 0, None)
    return f

    best = f.copy()
    best_score = local_score(f)

    iter_count = 0
    no_improve = 0
    patience = 200

    if max_iters is None:
        max_iters = 500000

    while time.monotonic() - t0 < time_limit and iter_count < max_iters:
        f, score, M, g = sota_iterate(f)

        if score < best_score:
            best_score = score
            best = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        iter_count += 1

        if iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            remaining = time_limit - elapsed
            iters_per_sec = iter_count / elapsed if elapsed > 0 else 0
            print(f"    LP iter={iter_count}, score={local_score(f):.6f}, "
                  f"best={best_score:.6f}, M={M:.4f}, {iters_per_sec:.0f} it/s, "
                  f"~{remaining / max(iters_per_sec,1):.0f}s remaining")

        if no_improve > patience:
            # Try a restart strategy
            r = rng.random()
            if r < 0.4:
                # Perturb best solution
                perturbation_scale = 0.1 * np.mean(best)
                f = best + perturbation_scale * rng.standard_normal(n)

            else:

                new_dir /= np.sum(new_dir)
                new_dir *= np.sum(best)
                f = 0.5 * best + 0.5 * new_dir
                f = np.clip(f, 0, None)

            no_improve = 0
            if local_score(f) > best_score * 1.1:
                # If restart made things much worse, go back to best
                f = best.copy()

    return best, best_score

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    step = 0

    tau = 2.0 / n  # initial temperature
    lr_base = 5.0

    t0 = time.monotonic()
    no_improve = 0
    patience = 1000

    while time.monotonic() - t0 < time_limit:
        a = np.exp(theta - np.max(theta))
        a = a / np.sum(a) * S

        # Compute soft-max gradient
        c = fft_convolve(a, a)
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        shifted = np.clip(shifted, -100, 100)
        w = np.exp(shifted)
        w = w / np.sum(w)

        # Gradient of soft-max value w.r.t. a
        # grad_a[j] = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]
        # This is a cross-correlation
        # Using direct computation (fast enough for n <= 512)
        grad_a = np.zeros(n)
        for j in range(n):
            valid = min(n, 2 * n - 1 - j)
            grad_a[j] = 2.0 * np.dot(a[:valid], w[j:j + valid])

        # Convert to theta-space gradient
        # a[j] = exp(theta[j]) * S / sum(exp(theta))
        # da[j]/dtheta[j'] = a[j] * (delta_{jj'} - a[j']/S * S) ... 
        # Actually: a = S * exp(theta) / sum(exp(theta))
        # da[j]/dtheta[k] = S * exp(theta[j]) / sum(exp) * (delta_{jk} - exp(theta[k])/sum(exp))
        # = a[j] * (delta_{jk} - a[k]/S)
        
        inner = np.dot(a, grad_a) / S
        grad_theta = a * (grad_a / S - inner)

        # Adam update
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * grad_theta
        v_buf = beta2 * v_buf + (1 - beta2) * grad_theta ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)

        lr = lr_base / np.sqrt(step + 1)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate
        a_new = np.exp(theta - np.max(theta))
        a_new = a_new / np.sum(a_new) * S
        score = local_score(a_new)

        if score < best_score:
            best_score = score
            best = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Anneal temperature
        tau *= 0.9995
        if tau < 0.01 / n:
            tau = 0.01 / n

        if no_improve > patience:
            # Restart
            r = rng.random()
            if r < 0.5:
                theta = np.log(best + 1e-10) + 0.1 * rng.standard_normal(n)
            else:
                f_perturbed = peak_shave_step(best.copy(), rng, 0.05)
                theta = np.log(f_perturbed + 1e-10)
            no_improve = 0
            tau = 2.0 / n

        if step % 500 == 0:
            print(f"    GD step={step}, score={local_score(a_new):.6f}, "
                  f"best={best_score:.6f}, tau={tau:.5f}")

    return best, best_score


def coordinate_polish(f_init, time_limit, rng):
    """
    Coordinate descent polish: try small adjustments to individual elements.
    """
    n = len(f_init)
    current = f_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score

    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        improved = False

        # Try random coordinates
        n_tries = max(5, n // 4)
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            j = int(rng.integers(0, n))
            base = current[j]
            if base < 1e-8:
                continue

            # Try several multiplicative factors
            for factor in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                trial = base * factor
                current[j] = trial
                new_score = local_score(current)
                if new_score < current_score:
                    current_score = new_score
                    improved = True
                    if current_score < best_score:
                        best_score = current_score
                        best = current.copy()
                else:
                    current[j] = base
                    break

        # Also try peak-shave step occasionally
        if rng.random() < 0.3:
            trial = peak_shave_step(current.copy(), rng, 0.02)
            new_score = local_score(trial)
            if new_score < current_score:
                current = trial
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best = current.copy()
                improved = True

        if not improved and iteration % 100 == 0:
            # Escalate: larger perturbations
            idx = rng.choice(n, size=max(1, n // 20), replace=False)
            current[idx] *= np.exp(0.1 * rng.standard_normal(len(idx)))
            current = np.clip(current, 0, None)

        if iteration % 200 == 0:
            print(f"    CD iter={iteration}, cur={current_score:.6f}, best={best_score:.6f}")

    return best, best_score


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def make_random_init(n, rng, method='broadband'):
    """Create a structured random initial sequence."""
    t = np.arange(n, dtype=np.float64)

    if method == 'broadband':
        a = np.ones(n)
        for k in range(1, 20):
            freq = k * (0.5 + rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.2 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'alternating':
        a = np.ones(n)
        a[::2] = 1.3
        a[1::2] = 0.7
        a += 0.1 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'tapered':
        # Tapered: high in middle, low at edges
        a = 1.0 + 0.5 * np.cos(np.pi * t / n)
        a += 0.1 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    else:
        return 1.0 + 0.2 * rng.standard_normal(n)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search combining LP-based SOTA iterations, gradient refinement,
    and local polish to minimize the evaluation score.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 40.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # ==========================================
    # PHASE 1: LP-based SOTA optimization (n=384)
    # ==========================================
    print(f"\n{'='*60}")
    print(f"PHASE 1: LP-based SOTA optimization (n=384)")
    print(f"{'='*60}")

    n_main = 384
    if start_seq is not None:
        if len(start_seq) == n_main:
            f0 = start_seq.copy()
        else:
            f0 = resample_sequence(start_seq, n_main)
    else:
        f0 = make_random_init(n_main, rng, 'broadband')

    t_phase1 = 0.55 * time_budget
    remaining_before_p1 = deadline - time.monotonic()
    t_phase1 = min(t_phase1, max(30.0, remaining_before_p1 - 100.0))
    print(f"  Time allocated: {t_phase1:.0f}s, starting score: {local_score(f0):.6f}")

    f_best, score1 = lp_optimize(f0, t_phase1, rng)
    print(f"  Phase 1 result: score={score1:.6f}")
    if score1 < best_score:
        best_score = score1
        best_seq = f_best.copy()
        print(f"  *** New best: {best_score:.6f} ***")

    # ==========================================
    # PHASE 2: Second LP run from different init
    # ==========================================
    print(f"\n{'='*60}")
    print(f"PHASE 2: LP from different initialization (n=384)")
    print(f"{'='*60}")

    t_phase2 = min(0.20 * time_budget, max(20.0, deadline - time.monotonic() - 100.0))
    if t_phase2 > 15:
        # Try alternating init
        f0_2 = make_random_init(n_main, rng, 'alternating')
        # Also try from best with perturbation
        f0_2b = best_seq + 0.05 * np.mean(best_seq) * rng.standard_normal(n_main)
        f0_2b = np.clip(f0_2b, 0.01, None)

        # Split time between the two
        t_half = t_phase2 / 2

        f_best2, score2 = lp_optimize(f0_2, t_half, rng)
        print(f"  Alt-init result: score={score2:.6f}")
        if score2 < best_score:
            best_score = score2
            best_seq = f_best2.copy()
            print(f"  *** New best: {best_score:.6f} ***")

        if time.monotonic() < deadline - 60:
            t_remaining = deadline - time.monotonic() - 60
            f_best2b, score2b = lp_optimize(f0_2b, t_remaining, rng)
            print(f"  Perturb-init result: score={score2b:.6f}")
            if score2b < best_score:
                best_score = score2b
                best_seq = f_best2b.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # ==========================================
    # PHASE 3: Gradient refinement
    # ==========================================
    print(f"\n{'='*60}")
    print(f"PHASE 3: Gradient refinement")
    print(f"{'='*60}")

    t_phase3 = min(0.10 * time_budget, max(15.0, deadline - time.monotonic() - 60.0))
    if t_phase3 > 10:
        print(f"  Time allocated: {t_phase3:.0f}s, starting score: {best_score:.6f}")
        f_gd, score_gd = gradient_refine(best_seq, t_phase3, rng)
        print(f"  Gradient result: score={score_gd:.6f}")
        if score_gd < best_score:
            best_score = score_gd
            best_seq = f_gd.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # ==========================================
    # PHASE 4: Coordinate polish
    # ==========================================
    print(f"\n{'='*60}")
    print(f"PHASE 4: Coordinate polish")
    print(f"{'='*60}")

    t_phase4 = min(0.08 * time_budget, max(10.0, deadline - time.monotonic() - 30.0))
    if t_phase4 > 8:
        print(f"  Time allocated: {t_phase4:.0f}s, starting score: {best_score:.6f}")
        f_cd, score_cd = coordinate_polish(best_seq, t_phase4, rng)
        print(f"  CD result: score={score_cd:.6f}")
        if score_cd < best_score:
            best_score = score_cd
            best_seq = f_cd.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # ==========================================
    # PHASE 5: Try other lengths
    # ==========================================
    print(f"\n{'='*60}")
    print(f"PHASE 5: Other lengths")
    print(f"{'='*60}")

    for n_try in [256, 512, 320]:
        if time.monotonic() > deadline - 40:
            break
        t_try = min(30.0, deadline - time.monotonic() - 30.0)
        if t_try < 10:
            break

        if best_seq is not None:
            f0_try = resample_sequence(best_seq, n_try)
        else:
            f0_try = make_random_init(n_try, rng, 'broadband')

        print(f"  Trying n={n_try}...")
        f_try, score_try = lp_optimize(f0_try, t_try, rng)
        print(f"  n={n_try}: score={score_try:.6f}")
        if score_try < best_score:
            best_score = score_try
            best_seq = f_try.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # ==========================================
    # FINAL: Quick polish with remaining time
    # ==========================================
    t_remaining = deadline - time.monotonic() - 10.0
    if best_seq is not None and t_remaining > 15:
        print(f"\n{'='*60}")
        print(f"FINAL POLISH (t={t_remaining:.0f}s)")
        print(f"{'='*60}")

        # Do a quick coordinate polish
        t_final = min(t_remaining, 20.0)
        f_final, score_final = coordinate_polish(best_seq, t_final, rng)
        if score_final < best_score:
            best_score = score_final
            best_seq = f_final.copy()
            print(f"  Final score: {best_score:.6f}")

        # One more short LP burst if time remains
        t_remaining2 = deadline - time.monotonic() - 10.0
        if t_remaining2 > 10:
            f_final2, score_final2 = lp_optimize(best_seq, t_remaining2, rng)
            if score_final2 < best_score:
                best_score = score_final2
                best_seq = f_final2.copy()
                print(f"  After final LP: {best_score:.6f}")

    # Verify with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'='*60}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
