import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, c, top_k):
    """Compute subgradient of max(c) using top-k lags."""
    n = len(a)
    # Sort lags by convolution value
    sorted_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[sorted_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    
    # For each top lag k, the gradient of c[k] w.r.t. a is 2 * flip(a) shifted
    # c[k] = sum_j a[j] * a[k-j], so dc[k]/da[m] = 2 * a[k-m] (if 0 <= k-m < n)
    # We can compute this via cross-correlation
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(sorted_idx):
        w_full[int(k)] = weights[i]
    
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def nesterov_optimize(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=128, 
                       use_backtrack=True):
    """Nesterov-accelerated subgradient method with optional backtracking."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        grad = compute_subgradient(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: use 2/(k*(1+t)) where k is Lipschitz constant estimate
        # For our problem, the Lipschitz constant of the gradient (of the smooth approximation)
        # is approximately 2 * max(a) * 2 (for the factor of 2 in gradient)
        # But we use the subgradient, so we use a different step size rule
        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        
        if use_backtrack:
            # Backtracking line search
            alpha = step
            f_y = cur  # This is the max, not the true function value at y
            # We'll use a simpler approach: just reduce step if we don't make progress
            x_new = project_simplex(y - alpha * grad / max(gnorm, 1e-15), S)
            
            # Check if we made progress (rough check)
            if it % 10 == 0:
                c_x = fft_convolve_self(x_new)
                if c_x.max() > cur * 1.01:
                    step *= 0.5
                    x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        else:
            x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)

            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
    
    return best_local, best_score

"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def grad_lse(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def grad_toppk_ramp(a, c, top_k, decay=0.97):
    """Gradient using top-k with geometric decay weighting (ramp)."""
    n = len(a)
    sorted_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[sorted_idx]
    ranks = np.arange(top_k, dtype=np.float64)
    weights = top_vals * (decay ** ranks)
    total = weights.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = weights / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(sorted_idx):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def grad_lp(a, c, p):
    """Gradient of Lp norm of c (smooth approximation to max for large p)."""
    n = len(a)
    c_clip = np.maximum(c, 1e-30)
    c_max = c_clip.max()
    # Use log-sum-exp trick for numerical stability
    # Lp = c_max * (sum (c/c_max)^p)^(1/p)
    ratio = c_clip / c_max
    log_sum = p * np.log(np.sum(ratio ** p) + 1e-300) / p
    # grad of Lp w.r.t. a[m] = (1/Lp) * (1/p) * sum p * c[k]^(p-1) * dc[k]/da[m]
    # = (1/Lp) * sum c[k]^(p-1) * 2*a[k-m] ... but we need to be careful
    # Actually d(Lp)/da[m] = (1/p) * Lp^(1-p) * sum_k p * c[k]^(p-1) * dc[k]/da[m]
    # = Lp^(1-p) * sum_k c[k]^(p-1) * 2*a[k-m]
    # To avoid overflow, use: c[k]^(p-1) = c_max^(p-1) * ratio[k]^(p-1)
    # Lp^(1-p) = c_max^(1-p) * (sum ratio^p)^(1/p)^(1-p)
    # So grad[m] = 2 * (sum ratio^p)^(1/p)^(1-p) * sum_k ratio[k]^(p-1) * a[k-m]
    
    ratio_p = ratio ** p
    sum_ratio_p = ratio_p.sum()
    Lp_factor = sum_ratio_p ** (1.0 / p)  # = (sum c^p)^(1/p) / c_max
    # grad[m] = 2 * Lp_factor^((1-p)) * ... hmm let me redo this
    
    # Simpler: just compute w[k] = c[k]^(p-1) / sum(c^p) * p
    # grad[m] = 2 * sum_k w[k] * a[k-m] / Lp ... 
    # Let me just use: grad of Lp = (sum c^p)^(1/p - 1) * p * sum c[k]^(p-1) * dc[k]/da[m] / p
    # = (sum c^p)^((1-p)/p) * sum c[k]^(p-1) * 2*a[k-m]
    
    # For numerical stability, factor out c_max:
    # c[k]^(p-1) = c_max^(p-1) * ratio[k]^(p-1)
    # sum c^p = c_max^p * sum ratio^p
    # (sum c^p)^((1-p)/p) = c_max * (sum ratio^p)^((1-p)/p)
    # So grad[m] = 2 * c_max * (sum ratio^p)^((1-p)/p) * c_max^(p-1) * sum ratio[k]^(p-1) * a[k-m]
    # = 2 * c_max^p * (sum ratio^p)^((1-p)/p) * sum ratio[k]^(p-1) * a[k-m]
    
    sum_rp = sum_ratio_p
    if sum_rp < 1e-300:
        return np.zeros(n)
    
    w = ratio ** (p - 1)
    w = w / w.sum()  # normalize for FFT
    
    coeff = 2.0 * c_max ** p * sum_rp ** ((1.0 - p) / p)
    
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, len(w) + n - 1 if (len(w) + n - 2).bit_length() == (len(w) + n - 2) else 1 << (len(w) + n - 2).bit_length())
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = coeff * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_lp_fista(a_init, n, S, max_time, rng, p_start=20, p_end=500):
    """FISTA minimizing Lp norm of self-convolution, with p increasing over time."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 0.5 / (2.0 * S)
    
    while time.monotonic() - start < max_time:
        it += 1
        # Gradually increase p
        frac = min(1.0, it / max(1, int(max_time * 2)))
        p = int(p_start + (p_end - p_start) * frac)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()
        
        grad = grad_lp(y, np.maximum(c, 1e-20), p)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        x_new = project_simplex(y - step * grad, S)
        
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        step = 0.5 / (2.0 * S * (1.0 + 0.001 * it))
        step = max(step, 1e-8)
        
        if it % 5000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score

def peak_shaving(a_init, n, S, max_time, rng, max_iter=50000):
    """Local peak-shaving: iteratively reduce the maximum peak of self-convolution."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time and it < max_iter:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Find top lags
        top_k = 10
        sorted_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[sorted_idx]
        
        # Compute subgradient (weighted combination)
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(sorted_idx):
            w_full[int(k)] = top_vals[i] / top_vals.sum()
        
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Find the element to reduce (highest gradient, positive value)
        gpos = np.maximum(grad, 0)
        if gpos.max() < 1e-15:
            break
        
        # Reduce top contributors and redistribute
        n_reduce = min(50, n // 10)
        reduce_idx = np.argsort(gpos)[::-1][:n_reduce]
        reduce_idx = reduce_idx[a[reduce_idx] > 1e-10]
        
        if len(reduce_idx) == 0:
            break
        
        # How much to reduce
        total_reduce = gpos[reduce_idx].sum() / (n_reduce + 1) * 0.1
        total_reduce = min(total_reduce, 0.3 * a[reduce_idx].min())
        
        # Find elements to increase (low gradient, positive or zero)
        grow_idx = np.argsort(gpos)[:n_reduce]
        grow_idx = grow_idx[~np.isin(grow_idx, reduce_idx)]
        
        if len(grow_idx) == 0:
            break
        
        a[reduce_idx] -= total_reduce
        a[reduce_idx] = np.maximum(a[reduce_idx], 0)
        freed = total_reduce * len(reduce_idx)
        a[grow_idx] += freed / len(grow_idx)
        
        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a * (S / s)
    
    return best_local, best_score


def coarse_to_fine(a_coarse, n_fine, S, max_time, rng):
    """Upsample coarse solution and polish at fine resolution."""
    n_coarse = len(a_coarse)
    # Upsample by repeating each element
    scale = n_fine // n_coarse
    if scale < 1:
        a = a_coarse[:n_fine].copy()
    else:
        a = np.repeat(a_coarse, scale)[:n_fine]
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n_fine) * S / n_fine
    else:
        a = a * (S / s)
    
    # Polish with Nesterov
    return run_nesterov_ramp(a, n_fine, S, max_time, rng, top_k_init=10, top_k_max=200, decay=0.96)


def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    for n_try in [4096, 8192, 16384]:
        if n_try not in n_candidates and 512 <= n_try <= 32768:
            n_candidates.append(n_try)
    
    n_candidates = sorted(set(n_candidates))
    print("Trying n values: %s" % n_candidates)
    
    # Allocate time
    n_vals = len(n_candidates)
    per_n = max(10.0, (deadline - time.monotonic()) * 0.7 / n_vals)
    
    # --- Main optimization loop over n values ---
    for n in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        
        S = np.sqrt(2.0 * float(n))
        segment_time = per_n
        
        # Get initializations for this n
        inits = []
        if given is not None:
            if len(given) >= n:
                inits.append(given[:n].copy())
            elif len(given) > 64:
                # Upsample
                from numpy import interp
                x_orig = np.arange(len(given))
                x_new = np.arange(n)
                inits.append(interp(x_new, x_orig, given))
        
        # Add structured initializations
        tt = np.linspace(0, 1, n, endpoint=False)
        inits.append(rng.random(n) + 0.05)
        inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt))
        inits.append(np.exp(-0.5 * ((tt - 0.5) * 5.0) ** 2) + 0.1)
        inits.append(0.5 + 0.5 * np.abs(np.sin(2 * np.pi * 13 * tt)))
        
        # Deduplicate and limit
        inits = inits[:5]
        
        print("=== n=%d, %d inits, %.1f sec total ===" % (n, len(inits), segment_time))
        
        best_n_score = np.inf
        best_n_a = None
        
        per_init = segment_time / len(inits)
        
        for init_idx, init in enumerate(inits):
            if time.monotonic() > deadline - 60:
                break
            
            n_init = len(init)
            S_init = np.sqrt(2.0 * float(n_init))
            init = safe_normalize(init, n_init)
            
            c0 = fft_convolve_self(init)
            s0 = c0.max()
            
            # Phase 1: Lp-FISTA
            t1 = per_init * 0.4
            a1, s1 = run_lp_fista(init, n_init, S_init, t1, rng, p_start=10, p_end=300)
            a1 = safe_normalize(a1, n_init)
            c1 = fft_convolve_self(a1)
            if c1.max() < best_n_score:
                best_n_score = c1.max()
                best_n_a = a1.copy()
            
            # Phase 2: Nesterov ramp
            t2 = per_init * 0.35
            a2, s2 = run_nesterov_ramp(a1, n_init, S_init, t2, rng, 
                                         top_k_init=5, top_k_max=128, decay=0.96)
            a2 = safe_normalize(a2, n_init)
            c2 = fft_convolve_self(a2)
            if c2.max() < best_n_score:
                best_n_score = c2.max()
                best_n_a = a2.copy()
            
            # Phase 3: Peak shaving
            t3 = per_init * 0.25
            a3, s3 = peak_shaving(a2, n_init, S_init, t3, rng)
            a3 = safe_normalize(a3, n_init)
            c3 = fft_convolve_self(a3)
            if c3.max() < best_n_score:
                best_n_score = c3.max()
                best_n_a = a3.copy()
            
            if best_n_score < global_best:
                global_best = best_n_score
                global_a = best_n_a.copy()
                print("  n=%d init %d: score=%.6f" % (n, init_idx, global_best))
        
        # Segment-based optimization
        if time.monotonic() < deadline - 90:
            for k in [3, 5, 9]:
                if time.monotonic() > deadline - 60:
                    break
                t_seg = min(30, (deadline - time.monotonic()) * 0.1)
                a_seg, s_seg = segment_optimize(n, k, S, t_seg, rng)
                a_seg = safe_normalize(a_seg, n)
                c_seg = fft_convolve_self(a_seg)
                if c_seg.max() < global_best:
                    global_best = c_seg.max()
                    global_a = a_seg.copy()
                    print("  Segment k=%d: score=%.6f" % (k, global_best))
    
    # --- Final polishing phase ---
    if global_a is not None:
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * float(n_final))
        remaining = deadline - time.monotonic()
        
        print("=== Final polishing: %.1f sec remaining ===" % remaining)
        
        # Aggressive Nesterov with high top-k
        seg = max(1.0, remaining * 0.3)
        a_p1, s_p1 = run_nesterov_ramp(global_a, n_final, S_final, seg, rng,
                                         top_k_init=50, top_k_max=512, decay=0.95)
        a_p1 = safe_normalize(a_p1, n_final)
        c_p1 = fft_convolve_self(a_p1)
        if c_p1.max() < global_best:
            global_best = c_p1.max()
            global_a = a_p1.copy()
            print("  Polish Nesterov: %.6f" % global_best)
        
        # Lp-FISTA with high p
        seg2 = max(1.0, (deadline - time.monotonic()) * 0.3)
        a_p2, s_p2 = run_lp_fista(global_a, n_final, S_final, seg2, rng, p_start=100, p_end=1000)
        a_p2 = safe_normalize(a_p2, n_final)
        c_p2 = fft_convolve_self(a_p2)
        if c_p2.max() < global_best:
            global_best = c_p2.max()
            global_a = a_p2.copy()
            print("  Polish Lp: %.6f" % global_best)
        
        # Peak shaving
        seg3 = max(1.0, (deadline - time.monotonic()) * 0.3)
        a_p3, s_p3 = peak_shaving(global_a, n_final, S_final, seg3, rng)
        a_p3 = safe_normalize(a_p3, n_final)
        c_p3 = fft_convolve_self(a_p3)
        if c_p3.max() < global_best:
            global_best = c_p3.max()
            global_a = a_p3.copy()
            print("  Polish PeakShave: %.6f" % global_best)
        
        # Final Nesterov with very small steps
        rem = deadline - time.monotonic()
        if rem > 2:
            a_p4, s_p4 = run_nesterov_ramp(global_a, n_final, S_final, rem, rng,
                                             top_k_init=100, top_k_max=512, decay=0.93)
            a_p4 = safe_normalize(a_p4, n_final)
            c_p4 = fft_convolve_self(a_p4)
            if c_p4.max() < global_best:
                global_best = c_p4.max()
                global_a = a_p4.copy()
                print("  Final Nesterov: %.6f" % global_best)
    
    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        # Clamp to valid range
        global_a = np.minimum(global_a, 1000.0)
    else:
        global_a = np.ones(256, dtype=np.float64)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification score: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
    
    if not np.isfinite(ver):
        global_a = np.ones(256, dtype=np.float64)
    
    print("Returning sequence of length %d, internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
