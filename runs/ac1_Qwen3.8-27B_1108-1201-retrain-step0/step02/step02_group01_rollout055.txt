```python
def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
Minimize 2*n*max(conv(a,a))/sum(a)^2 using focal subgradient method with
FISTA acceleration, simplex projection, and adaptive hot-set management.

Key differences from previous approach:
1. Direct subgradient on the max (not softmax approximation)
2. FISTA (Nesterov accelerated) for faster convergence
3. Adaptive hot-set (window of near-max lags) for efficiency
4. Simplex projection (exploits the sum=1 constraint geometrically)
5. Periodic softmax "recentering" to escape local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad_ffi_from_lags(hot_indices, weights, p):
    """Compute subgradient: g[i] = 2 * sum_k w_k * p[k-i] for i=0..n-1."""
    n = len(p)
    fsize = 1 << (3 * n - 2).bit_length()
    W = np.zeros(fsize)
    for k, w in zip(hot_indices, weights):
        W[k] = w
    P = np.zeros(fsize)
    P[:n] = p
    xcorr = np.real(ifft(fft(W) * np.conj(fft(P))))
    return 2.0 * xcorr[:n]


def run_focal_fista(p_init, n, time_budget, delta_frac=0.02, max_hot=200):
    """Focal FISTA optimization."""
    p = p_init.copy().astype(np.float64)
    p = np.maximum(p, 0.0)
    p = p / p.sum()
    
    c = fft_convolve_self(p)
    c_max = c.max()
    best_c_max = c_max
    best_p = p.copy()
    
    # FISTA variables
    q = p.copy()  # extrapolated point
    t_val = 1.0
    start = time.monotonic()
    
    # Initial step size
    # L is the Lipschitz constant; we estimate it adaptively
    eta = 1.0  # initial step size
    
    iteration = 0
    while time.monotonic() - start < time_budget:
        iteration += 1
        
        # Compute subgradient at q
        q_safe = np.maximum(q, 1e-30)
        q_safe = q_safe / q_safe.sum()
        
        c_q = fft_convolve_self(q_safe)
        c_q_max = c_q.max()
        
        # Identify hot lags
        threshold = c_q_max * (1.0 - delta_frac)
        hot_mask = c_q >= threshold
        hot_count = hot_mask.sum()
        
        if hot_count == 0:
            hot_count = 1
            hot_mask = np.zeros(len(c_q), dtype=bool)
            hot_mask[np.argmax(c_q)] = True
        
        if hot_count > max_hot:
            # Keep only the top max_hot
            hot_indices_all = np.where(hot_mask)[0]
            hot_values = c_q[hot_indices_all]
            top_idx = np.argsort(hot_values)[::-1][:max_hot]
            hot_mask = np.zeros(len(c_q), dtype=bool)
            hot_mask[hot_indices_all[top_idx]] = True
            hot_count = max_hot
        
        hot_indices = np.where(hot_mask)[0]
        # Uniform weights on hot lags
        weights = np.full(hot_count, 1.0 / hot_count)
        
        g = compute_subgrad_ffi_from_lags(hot_indices, weights, q_safe)
        
        # FISTA update
        t_old = t_val
        p_new = project_simplex(q_safe - eta * g)
        t_val = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        q = p_new + ((t_old - 1.0) / t_val) * (p_new - p)
        p = p_new
        
        # Ensure non-negativity and normalization
        p = np.maximum(p, 0.0)
        p_sum = p.sum()
        if p_sum < 1e-10:
            p = np.full(n, 1.0 / n)
        else:
            p = p / p_sum
        
        # Check objective
        c = fft_convolve_self(p)
        c_max = c.max()
        if c_max < best_c_max:
            best_c_max = c_max
            best_p = p.copy()
        
        # Adaptive step size: decrease if objective increased
        # (simple heuristic)
        if c_max > best_c_max * 1.1 and eta > 0.01:
            eta *= 0.8
        
    return best_p, best_c_max
```

```python
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a combination of:
1. Nesterov-accelerated gradient descent on softmax-smoothed objective
2. Direct stochastic subgradient with variance reduction
3. Focal subgradient with simplex projection
4. Coordinate descent polish with exact line search
5. Multi-start with diverse initializations

Key improvements over previous approach:
- Nesterov momentum (more stable than Adam for non-stationary problems)
- Geometric temperature schedule with epoch-based restarts
- Stochastic subgradient with importance sampling for efficiency
- Direct top-K subgradient (no smoothing) for final stages
- Diversity perturbations to escape local minima
- Aggressive coordinate descent with batch updates

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.full(n, 1.0 / n)
    rho = rho_candidates[-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_obj_and_grad(p, n, T):
    """Compute softmax-smoothed max of conv(p,p) and gradient w.r.t. p."""
    c = fft_convolve_self(p)
    c_max = c.max()
    if c_max < 1e-15:
        return c_max, np.zeros(n)
    shifted = T * (c - c_max)
    shifted = shifted - shifted.max()
    exp_shifted = np.exp(shifted)
    log_sum_exp = np.log(exp_shifted.sum())
    obj = c_max + log_sum_exp / T
    # Weights
    w = exp_shifted / exp_shifted.sum()
    # Gradient: grad[i] = 2 * sum_k w[k] * p[k-i]
    # Use FFT cross-correlation
    fsize = 1 << (3 * n - 2).bit_length()
    W = np.zeros(fsize)
    W[:len(w)] = w
    P = np.zeros(fsize)
    P[:n] = p
    xcorr = np.real(ifft(fft(W) * np.conj(fft(P))))
    grad = 2.0 * xcorr[:n]
    return obj, grad


def topk_subgrad(p, n, top_k, rng):
    """Compute subgradient using top-k lags (with random tie-breaking)."""
    c = fft_convolve_self(p)
    c_max = c.max()
    # Add small noise for randomization
    c_noisy = c + rng.random(len(c)) * 1e-10
    top_indices = np.argsort(c_noisy)[::-1][:top_k]
    # Weights proportional to c values on the top-k
    vals = c[top_indices]
    w = vals / vals.sum()
    # Compute gradient
    fsize = 1 << (3 * n - 2).bit_length()
    W = np.zeros(fsize)
    for idx, k in enumerate(top_indices):
        W[k] = w[idx]
    P = np.zeros(fsize)
    P[:n] = p
    xcorr = np.real(ifft(fft(W) * np.conj(fft(P))))
    grad = 2.0 * xcorr[:n]
    return c_max, grad


def stochastic_subgrad_step(p, n, sample_weights, rng):
    """Single stochastic subgradient step: pick one lag, compute its gradient."""
    k = rng.choice(2 * n - 1, p=sample_weights)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if hi >= lo:
        grad[lo:hi + 1] = 2.0 * p[hi:lo - 1:-1] if lo > 0 else 2.0 * p[k:lo - 1:-1][- (hi - lo + 1):]
    # Simpler: just compute directly
    grad = np.zeros(n)
    for i in range(lo, hi + 1):
        grad[i] = 2.0 * p[k - i]
    return grad


def run_nesterov_softmax(p_init, n, T_start, T_end, time_budget, eta_base=0.05):
    """Nesterov momentum + softmax smoothed gradient descent."""
    p = p_init.copy().astype(np.float64)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        p = np.full(n, 1.0 / n)
    else:
        p = p / s

    c = fft_convolve_self(p)
    c_max = c.max()
    best_c_max = c_max
    best_p = p.copy()

    p_old = p.copy()
    start = time.monotonic()
    total_time = time_budget
    progress = 0.0
    momentum = np.zeros(n)
    beta = 0.9
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        # Geometric temperature schedule
        T = T_start * (T_end / T_start) ** progress

        obj, grad = softmax_obj_and_grad(p, n, T)

        # Nesterov extrapolation
        p_extrap = p + beta * (p - p_old)
        p_extrap = np.maximum(p_extrap, 0.0)
        ps = p_extrap.sum()
        if ps < 1e-10:
            p_extrap = p.copy()
        else:
            p_extrap = p_extrap / ps

        # Recompute gradient at extrapolated point for better accuracy
        if T < 5000:
            obj_e, grad_e = softmax_obj_and_grad(p_extrap, n, T)
        else:
            grad_e = grad

        # Step size: inversely proportional to T
        eta = eta_base / max(1.0, np.sqrt(T))

        p_new = p_extrap - eta * grad_e
        p_new = project_simplex(p_new)

        p_old = p
        p = p_new

        # Check objective periodically
        if it % 50 == 1 or progress > 0.99:
            c = fft_convolve_self(p)
            c_max = c.max()
            if c_max < best_c_max:
                best_c_max = c_max
                best_p = p.copy()

    return best_p, best_c_max

    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        p = np.full(n, 1.0 / n)
    else:
        p = p / s

    c = fft_convolve_self(p)
    c_max = c.max()
    best_c_max = c_max
    best_p = p.copy()

    momentum = np.zeros(n)
    beta = 0.85
    eta = 0.01
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(123)

        c_max, grad = topk_subgrad(p, n, top_k, rng)

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        momentum = beta * momentum + (1.0 - beta) * grad
        direction = momentum

        p_new = project_simplex(p_new)

def coordinate_descent_polish(p, n, time_budget, max_sweeps=100):
    """Coordinate descent with exact 1D minimization."""
    p = p.copy().astype(np.float64)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        p = np.full(n, 1.0 / n)
    else:
        p = p / s

    c = fft_convolve_self(p)
    best_obj = c.max()
    best_p = p.copy()
    start = time.monotonic()
    rng = np.random.default_rng(456)

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        # Process coordinates in random order for this sweep
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            orig = p[i]
            if orig < 1e-12:
                # Try turning it on
                for trial in [1e-4, 1e-3, 1e-2, orig + 0.01]:
                    p[i] = trial
                    s_tmp = p.sum()
                    p_norm = p / s_tmp
                    c_tmp = fft_convolve_self(p_norm)
                    if c_tmp.max() < best_obj:
                        best_obj = c_tmp.max()
                        best_p = p_norm.copy()
                        p = p_norm
                        improved = True
                        break
                p[i] = 0.0
                continue

            # Binary search for optimal value of p[i]
            # The objective as a function of p[i] is convex (max of linear functions
            # in p[i] after normalizing), so we can use ternary search or binary search on derivative
            lo = 0.0
            hi = orig * 3.0
            best_local = orig
            best_local_obj = np.inf

            for _ in range(10):
                mid = (lo + hi) / 2.0
                p[i] = mid
                s_tmp = p.sum()
                if s_tmp < 1e-10:
                    p[i] = orig
                    break
                p_norm = p / s_tmp
                c_tmp = fft_convolve_self(p_norm)
                obj_tmp = c_tmp.max()
                if obj_tmp < best_local_obj:
                    best_local_obj = obj_tmp
                    best_local = mid
                # For max of linear functions, check which side to go
                # Simple heuristic: if mid < orig and obj improved, go lower
                if mid < orig and obj_tmp < best_obj:
                    hi = mid
                elif mid > orig and obj_tmp < best_obj:
                    lo = mid
                else:
                    # Check both sides
                    p[i] = max(0, mid * 0.5)
                    s_t2 = p.sum()
                    if s_t2 > 1e-10:
                        c_t2 = fft_convolve_self(p / s_t2)
                        if c_t2.max() < obj_tmp:
                            hi = mid
                        else:
                            lo = mid
                    p[i] = mid

            p[i] = best_local
            s_tmp = p.sum()
            if s_tmp > 1e-10:
                p = p / s_tmp
                c = fft_convolve_self(p)
                if c.max() < best_obj:
                    best_obj = c.max()
                    best_p = p.copy()
                    improved = True

        if not improved and sweep > 5:
            break

    return best_p, best_obj


def get_initializations(n, rng, S):
    """Generate diverse initializations."""
    inits = []
    names = []

    # Uniform
    a = np.full(n, S / n)
    inits.append(a)
    names.append("uniform")

    # Gaussian center
    i = np.arange(n)
    a = np.exp(-0.5 * ((i - n / 2) / (n / 8)) ** 2)
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    inits.append(a)
    names.append("gaussian")

    # Cosine modulation (various frequencies)
    for freq in [2, 3, 5, 13]:
        for alpha in [0.3, 0.6, 0.8]:
            a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * i / n)
            a = np.maximum(a, 0.01)
            a = a * (S / a.sum())
            inits.append(a)
            names.append("cos_f%d_a%.1f" % (freq, alpha))

    # Anti-peak (high at edges)
    a = 1.0 - 0.5 * np.cos(2.0 * np.pi * i / n)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    inits.append(a)
    names.append("anti_peak")

    # Two humps
    a = np.exp(-0.5 * ((i - n * 0.25) / (n / 12)) ** 2) + \
        np.exp(-0.5 * ((i - n * 0.75) / (n / 12)) ** 2)
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    inits.append(a)
    names.append("two_humps")

    # Random
    a = rng.random(n)
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    inits.append(a)
    names.append("random")

    # Blackman-like window
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    inits.append(a)
    names.append("blackman")

    return inits, names


def run_focal_ffi(p_init, n, time_budget, delta=0.01, max_ffi=100):
    """Focal subgradient with simplex projection (exact max, no smoothing)."""
    p = p_init.copy().astype(np.float64)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        p = np.full(n, 1.0 / n)
    else:
        p = p / s

    c = fft_convolve_self(p)
    c_max = c.max()
    best_c_max = c_max
    best_p = p.copy()

    p_old = p.copy()
    start = time.monotonic()
    it = 0
    eta = 0.02
    momentum = np.zeros(n)
    beta = 0.9
    rng = np.random.default_rng(789)

    while time.monotonic() - start < time_budget:
        it += 1

        # Compute convolution
        c = fft_convolve_self(p)
        c_max = c.max()
        if c_max < best_c_max:
            best_c_max = c_max
            best_p = p.copy()

        # Identify hot lags
        threshold = c_max * (1.0 - delta)
        hot_mask = c >= threshold
        hot_count = hot_mask.sum()

        if hot_count == 0:
            idx = np.argmax(c)
            hot_mask = np.zeros(len(c), dtype=bool)
            hot_mask[idx] = True
            hot_count = 1

        if hot_count > max_ffi:
            hot_indices_all = np.where(hot_mask)[0]
            vals = c[hot_indices_all]
            top_idx = np.argsort(vals)[::-1][:max_ffi]
            hot_mask = np.zeros(len(c), dtype=bool)
            hot_mask[hot_indices_all[top_idx]] = True
            hot_count = max_ffi

        # Add some random lags for diversity
        if it % 10 == 0 and hot_count < max_ffi:
            n_random = min(5, max_ffi - hot_count)
            random_lags = rng.choice(2 * n - 1, size=n_random, replace=False)
            for rl in random_lags:
                if not hot_mask[rl]:
                    hot_mask[rl] = True
                    hot_count += 1

        hot_indices = np.where(hot_mask)[0]
        weights = np.full(hot_count, 1.0 / hot_count)

        # Compute subgradient via FFT
        fsize = 1 << (3 * n - 2).bit_length()
        W = np.zeros(fsize)
        for idx, k in enumerate(hot_indices):
            W[k] = weights[idx]
        P = np.zeros(fsize)
        P[:n] = p
        xcorr = np.real(ifft(fft(W) * np.conj(fft(P))))
        grad = 2.0 * xcorr[:n]

        # Momentum + step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        momentum = beta * momentum + (1.0 - beta) * grad
        p_new = p - eta * momentum
        p_new = project_simplex(p_new)

        p_old = p
        p = p_new

        # Adaptive step size
        if it % 100 == 0:
            c_check = fft_convolve_ffi(p)
            if c_check.max() > best_c_max * 1.02:
                eta *= 0.7
            elif c_check.max() < best_c_max * 0.99:
                eta = min(eta * 1.1, 0.1)

    return best_p, best_c_max
```

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho_candidates) == 0:

    rho = rho_candidates[-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
