    Returns (b, t_opt) where t_opt is the optimal value.
    """
    n = len(a)
    # Variables: b_0, ..., b_{n-1}, t
    # Minimize t
    # Subject to: sum_i a[i]*b[k-i] - t <= 0 for k = 0, ..., 2n-2
    #             sum(b) = 1
    #             b >= 0, t >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= 0
    # For each k, the row is: [a[k-0] (if valid), a[k-1] (if valid), ..., a[k-(n-1)] (if valid), -1]
    # Specifically, for variable b[j], coefficient is a[k-j] if 0 <= k-j < n, else 0
    # For variable t, coefficient is -1
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for k in range(n_constraints):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a[idx]
        A_ub[k, n] = -1.0  # -t
    
    # Equality constraint: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=np.zeros(n_constraints),
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        b = result.x[:n]
        t_opt = result.x[n]
        return b, t_opt
    else:
        return None, np.inf

"""

step functions. Uses linear programming to find optimal descent directions,
combined with smoothed gradient descent warm-up and coordinate descent polish.
The key innovation is solving an LP at each iteration to find the best direction
b (sum=1, b>=0) minimizing max_k conv(a,b)[k], which guarantees first-order
descent toward the global minimum of max(conv(a,a)).

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def normalize_seq(a):
    """Normalize to sum=1 with non-negativity."""
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_max_gradient(a, tau):
    """
    Compute smoothed max of conv(a,a) (log-sum-exp) and gradient w.r.t. a.
    Returns (value, gradient).
    """
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    value = max_c + tau * np.log(exp_sum)
    w = exp_vals / exp_sum
    # Gradient: grad[m] = 2 * sum_j a[j] * w[m+j] for j=0..n-1
    # This is the cross-correlation of a and w at lags 0..n-1
    N = 1
    while N < 2 * n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    corr = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    grad = 2.0 * corr
    return value, grad


def build_lp_matrix(a):

    n = len(a)

    A = np.zeros((n_constraints, n + 1))

        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
        A[k, n] = -1.0
    return A

        beta = (t_prev - 1.0) / t_mom

        a_y = a + beta * (a - a_prev)
        a_y = project_simplex(a_y)

        value, grad = smoothed_max_gradient(a_y, tau)

        # Project gradient to simplex tangent
        grad_proj = grad - np.mean(grad)

        a_prev = a.copy()
        a = project_simplex(a_y - lr * grad_proj)
        step += 1

        if step % 100 == 0:
            actual = np.max(fft_convolve(a, a))
            if actual < best_val:
                best_val = actual
                best_a = a.copy()
            if verbose and step % 1000 == 0:
                print(f"      SGD step={step}, val={actual:.8f}, "
                      f"score={2*n*actual:.6f}, tau={tau:.2e}, "
                      f"t={time.monotonic()-t0:.1f}s")

        if step % 200 == 0:
            actual = np.max(fft_convolve(a, a))
            if actual < best_val:
                best_val = actual
                best_a = a.copy()

    return best_a, best_val


def optimize_lbfgs(a_init, time_limit, verbose=True):
    """
    L-BFGS-B optimization on softmax parameterization.
    """
    n = len(a_init)
    a0 = normalize_seq(a_init)
    theta0 = np.log(a0 + 1e-15)

    tau = 2.0 / n

    def objective_and_grad(theta):
        t = theta - np.max(theta)
        e = np.exp(t)
        a = e / np.sum(e)
        value, grad_a = smoothed_max_gradient(a, tau)
        # Chain rule through softmax
        inner = np.dot(a, grad_a)
        grad_theta = a * (grad_a - inner)
        return value, grad_theta

    t0 = time.monotonic()
    result = minimize(objective_and_grad, theta0, method='L-BFGS-B',
                      options={'maxiter': 500, 'maxfun': 5000,
                               'gtol': 1e-8, 'ftol': 1e-10})
    theta_opt = result.x
    t = theta_opt - np.max(theta_opt)
    a_opt = np.exp(t) / np.sum(np.exp(t))
    best_val = np.max(fft_convolve(a_opt, a_opt))
    if verbose:
        print(f"      L-BFGS: score={2*n*best_val:.6f}, "
              f"fun={result.fun:.8f}, time={time.monotonic()-t0:.1f}s")
    return a_opt, best_val


def spectral_init(n, rng, n_iter=40):
    """Spectral initialization for good starting point."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    for it in range(n_iter):
        size = 1
        while size < 2 * n:
            size <<= 1
        A = np.fft.rfft(a, size)
        power = np.abs(A) ** 2
        corr = np.fft.irfft(power, size)[:n]
        alpha = 1.0 / (n * 0.5 + 1.0)
        a = a - alpha * (corr - corr.mean())
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def ramp_init(n, rng, style='cosine'):
    """Create structured initializations."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if style == 'cosine':
        a = 0.8 + 0.4 * np.cos(np.pi * (t - 0.5))
    elif style == 'linear':
        a = 1.0 + 0.3 * (2 * t - 1)
    elif style == 'bell':
        a = 1.0 + 0.5 * np.exp(-4.0 * (t - 0.5) ** 2)
    elif style == 'plateau':
        a = 0.7 + 0.6 * np.clip(4.0 * (t - 0.25) * (0.75 - t), 0, 1)
    else:
        a = np.ones(n)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def coordinate_polish(a_init, time_limit, rng, verbose=True):
    """
    Coordinate descent / local search polish.
    Works on normalized (sum=1) sequence.
    """
    n = len(a_init)
    a = normalize_seq(a_init)
    current_val = np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_val = current_val
    t0 = time.monotonic()
    stagnation = 0
    max_stagnation = 200
    imp_count = 0

    while time.monotonic() - t0 < time_limit and stagnation < max_stagnation:
        r = rng.random()
        candidate = a.copy()

        if r < 0.35:
            # Single element scale
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.03))
            candidate[j] *= factor
        elif r < 0.6:
            # Pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.1)
            amt = a[i] * frac
            candidate[i] -= amt
            candidate[j] += amt
        elif r < 0.8:
            # Block perturbation
            width = max(2, int(rng.integers(2, max(3, n // 32))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.02, size=width))
        else:
            # Peak-targeted: reduce top contributors
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Find top contributors
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 16)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= (1.0 - rng.uniform(0.01, 0.04, size=len(top_idx)))
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_add = min(5, len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                total_removed = np.sum(a[top_idx] - candidate[top_idx])
                candidate[add_idx] += total_removed / n_add

        candidate = np.clip(candidate, 0, None)
        cs = np.sum(candidate)
        if cs < 1e-10:
            stagnation += 1
            continue
        candidate = candidate / cs

        new_val = np.max(fft_convolve(candidate, candidate))
        if new_val < current_val:
            a = candidate
            current_val = new_val
            stagnation = 0
            imp_count += 1
            if current_val < best_val:
                best_val = current_val
                best_a = a.copy()
        else:
            stagnation += 1

        if verbose and imp_count % 500 == 0 and imp_count > 0:
            print(f"        Polish: improvements={imp_count}, "
                  f"score={2*n*best_val:.6f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_val


def try_optimize_length(n, a_init, time_budget, rng, deadline, verbose=True):
    """
    Full optimization pipeline for a given sequence length.
    Returns (best_a_normalized, best_score).
    """
    a = normalize_seq(a_init)
    current_val = np.max(fft_convolve(a, a))
    best_val = current_val
    best_a = a.copy()
    t0 = time.monotonic()

    if verbose:
        print(f"    n={n}: initial score={2*n*current_val:.6f}")

    # Phase 1: Smoothed gradient warm-up
    grad_time = min(90.0, time_budget * 0.25)
    if time.monotonic() < deadline - 60 and grad_time > 10:
        a_grad, val_grad = optimize_smoothed_grad(a, grad_time, verbose=verbose)
        if val_grad < best_val:
            best_val = val_grad
            best_a = a_grad
            if verbose:
                print(f"      After SGD: score={2*n*best_val:.6f}")

    # Phase 2: LP-based optimization (main method)
    remaining = min(deadline - 120, t0 + time_budget) - time.monotonic()
    lp_time = max(0.0, min(remaining, time_budget * 0.55))
    if lp_time > 10:
        a_lp, val_lp = optimize_lp(best_a, lp_time, verbose=verbose)
        if val_lp < best_val:
            best_val = val_lp
            best_a = a_lp

    # Phase 3: Coordinate descent polish
    remaining = min(deadline - 30, t0 + time_budget) - time.monotonic()
    polish_time = max(0.0, min(remaining, time_budget * 0.2))
    if polish_time > 5:
        a_pol, val_pol = coordinate_polish(best_a, polish_time, rng, verbose=verbose)
        if val_pol < best_val:
            best_val = val_pol
            best_a = a_pol
            if verbose:
                print(f"      After polish: score={2*n*best_val:.6f}")

    final_score = 2.0 * n * best_val
    if verbose:
        print(f"    n={n}: final score={final_score:.6f}")
    return best_a, final_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-method search for optimal non-negative sequence minimizing
    2n * max(convolve(a,a)) / (sum a)^2.
    """
    t_global = time.monotonic()
    time_budget = max(30.0, float(budget_s) - 60.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.001, 1000.0)
        s0 = score_fast(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s0:.6f}")
        if s0 < best_score:
            best_score = s0
            best_seq = start_seq.copy()

    # Define search configurations: (n, time_fraction)
    # Focus on n=512 (matching previous best) and try nearby sizes
    configs = [
        (512, 0.40),
        (512, 0.25),   # Second run with different init
        (768, 0.15),
        (1024, 0.10),
        (384, 0.10),
    ]

    init_styles = ['cosine', 'bell', 'plateau', 'linear', 'spectral']

    for config_idx, (n, time_frac) in enumerate(configs):
        if time.monotonic() > deadline - 90:
            break

        time_alloc = time_budget * time_frac
        if time_alloc < 30:
            time_alloc = 30
        if time.monotonic() + time_alloc > deadline - 60:
            time_alloc = deadline - 60 - time.monotonic()
        if time_alloc < 20:
            break

        # Choose initialization
        if config_idx == 0 and start_seq is not None:
            # Use provided sequence (resized if needed)
            if len(start_seq) == n:
                a_init = start_seq.copy()
            else:
                x_old = np.linspace(0, 1, len(start_seq))
                x_new = np.linspace(0, 1, n)
                a_init = np.interp(x_new, x_old, start_seq)
        else:
            style = init_styles[config_idx % len(init_styles)]
            if style == 'spectral':
                a_init = spectral_init(n, rng, n_iter=30)
            else:
                a_init = ramp_init(n, rng, style)

        print(f"\n  Config {config_idx}: n={n}, time={time_alloc:.0f}s, "
              f"init={'provided' if config_idx == 0 and start_seq is not None else style}")

        a_norm, sc = try_optimize_length(n, a_init, time_alloc, rng, deadline, verbose=True)

        if sc < best_score:
            best_score = sc
            # Scale for final output
            best_seq = a_norm * np.sqrt(2.0 * n)
            print(f"  *** New best: n={n}, score={best_score:.6f} ***")

    # Final verification and return
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
