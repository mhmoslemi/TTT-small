Multi-strategy optimizer for minimum peak autocorrelation non-negative sequences.
Combines LP-based directional search, L-BFGS with top-k smoothing,
and randomized local search. Focuses on length 256.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

def build_conv_matrix(a, n):
    """Build matrix A such that A @ b = conv(a, b)."""
    n_cons = 2 * n - 1
    A = np.zeros((n_cons, n))
    for k in range(n_cons):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    return A

def solve_lp_direction(a, M):
    """
    Solve: max 1^T b s.t. A_a @ b <= M * 1, b >= 0
    where A_a is the convolution matrix for 'a'.
    """
    n = len(a)
    A = build_conv_matrix(a, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')

def topk_obj_grad(theta, k, n):
    """
    Objective and gradient for 2n * mean(top-k of conv(softmax(theta), softmax(theta))).
    """
    a = np.exp(theta - np.max(theta))
    a = a / np.sum(a)
    c = fft_convolve(a, a)
    
    # Top-k values
    idx = np.argsort(c)[-k:]
    c_top = c[idx]
    obj = 2.0 * n * np.mean(c_top)
    
    # Gradient: only the top-k entries contribute
    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    
    # Gradient of conv(a,a) w.r.t. a
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    
    # Convert to theta gradient (softmax)
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return obj, grad_theta

            break

            topk_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(k, n),
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12}

    a_best = a_best / np.sum(a_best)

"""
LP-guided optimization for minimum peak linear autocorrelation of non-negative sequences.
Primary method: LP-based directional search that minimizes max cross-correlation,
combined with L-BFGS soft-max polishing and randomized local search for escape.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def build_conv_matrix(a, n):
    """Build matrix A (2n-1 x n) such that A @ b = conv(a, b)."""
    n_cons = 2 * n - 1
    A = np.zeros((n_cons, n))
    for k in range(n_cons):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            i_vals = k - np.arange(j_start, j_end + 1)
            A[k, j_start:j_end + 1] = a[i_vals]
    return A


def solve_lp_minpeak(a, M_hint=None):
    """
    Solve: min t s.t. (a*b)[k] <= t for all k, sum(b) = 1, b >= 0.
    Returns (optimal_b, optimal_t).
    """
    n = len(a)
    A = build_conv_matrix(a, n)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0.0, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success and res.x is not None:
        b_opt = np.maximum(res.x[:n], 0)
        t_opt = res.x[-1]
        return b_opt, t_opt
    return None, None


def solve_lp_maxsum(a, M):
    """
    Solve: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Returns optimal b (unnormalized).
    """
    n = len(a)
    A = build_conv_matrix(a, n)

    c_obj = -np.ones(n + 1)

    # Actually just use n variables
    c_obj = -np.ones(n)

    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None


def lp_iteration_minpeak(a):
    """
    One LP iteration using min-peak formulation.
    Returns (new_a, new_score, t_used).
    """
    n = len(a)
    c_aa = fft_convolve(a, a)
    M = np.max(c_aa)

    b, t_opt = solve_lp_minpeak(a)
    if b is None or np.sum(b) < 1e-15:
        return a, local_score(a), 0.0

    b = b / np.sum(b)

    # Line search for mixing parameter
    best_t = 0.1
    best_score = local_score(a)
    new_a = a

    for t_try in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        candidate = (1.0 - t_try) * a + t_try * b
        candidate = np.maximum(candidate, 1e-15)
        candidate = candidate / np.sum(candidate)
        s = local_score(candidate)
        if s < best_score:
            best_score = s
            best_t = t_try
            new_a = candidate

    if best_t == 0.0:
        # Try very small t
        for t_try in [0.005, 0.01, 0.015]:
            candidate = (1.0 - t_try) * a + t_try * b
            candidate = np.maximum(candidate, 1e-15)
            candidate = candidate / np.sum(candidate)
            s = local_score(candidate)
            if s < best_score:
                best_score = s
                best_t = t_try
                new_a = candidate

    return new_a, best_score, best_t


def lp_phase(a_init, time_limit, deadline, verbose=True):
    """Run LP iterations until time limit."""
    t0 = time.monotonic()
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    rng = np.random.default_rng(7)

    stagnant = 0
    it = 0

    while time.monotonic() < deadline and time.monotonic() - t0 < time_limit:
        it += 1
        new_a, new_score, t_used = lp_iteration_minpeak(current)

        if new_score < current_score - 1e-12:
            current = new_a
            current_score = new_score
            stagnant = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if verbose and it % 20 == 1:
                    print(f"    LP iter {it}: score={best_score:.7f}, t={t_used:.3f}")
        else:
            stagnant += 1
            if stagnant > 15:
                # Restart with perturbation from best
                current = best * np.exp(rng.normal(0, 0.02, n))
                current = np.maximum(current, 1e-10)
                current = current / np.sum(current)
                current_score = local_score(current)
                stagnant = 0

        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: current={current_score:.7f}, best={best_score:.7f}, t={time.monotonic()-t0:.0f}s")

    return best, best_score


def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def softmax_max_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * logsumexp(conv(softmax(theta), softmax(theta)))/tau.
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def lbgfs_phase(a_init, time_limit, deadline):
    """L-BFGS with annealed soft-max temperature."""
    n = len(a_init)
    theta = np.log(a_init + 1e-30)
    best_theta = theta.copy()
    best_score = local_score(a_init)

    tau_schedule = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
    maxiter_per_tau = 300

    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break

        result = minimize(
            softmax_max_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(tau,),
            options={'maxiter': maxiter_per_tau, 'ftol': 1e-15, 'gtol': 1e-13,
                     'maxfun': maxiter_per_tau * 5}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)

        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
            print(f"    L-BFGS tau={tau}: score={s:.7f} *** improved ***")
        else:
            print(f"    L-BFGS tau={tau}: score={s:.7f}")

        theta = best_theta

    a_best = softmax(best_theta)
    return a_best, best_score


def topk_obj_grad(theta, k, n):
    """
    Objective and gradient for 2n * mean(top-k of conv(softmax(theta), softmax(theta))).
    """
    a = softmax(theta)
    c = fft_convolve(a, a)

    idx = np.argsort(c)[-k:]
    c_top = c[idx]
    obj = 2.0 * n * np.mean(c_top)

    w = np.zeros(len(c))
    w[idx] = 1.0 / k

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return obj, grad_theta


def lbgfs_topk_phase(a_init, time_limit, deadline):
    """L-BFGS with top-k mean objective."""
    n = len(a_init)
    theta = np.log(a_init + 1e-30)
    best_theta = theta.copy()
    best_score = local_score(a_init)

    k_schedule = [64, 32, 16, 8, 4, 2]
    maxiter_per_k = 200

    for k in k_schedule:
        if time.monotonic() > deadline:
            break

        result = minimize(
            topk_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(k, n),
            options={'maxiter': maxiter_per_k, 'ftol': 1e-15, 'gtol': 1e-13}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)

        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
            print(f"    TopK k={k}: score={s:.7f} *** improved ***")
        else:
            print(f"    TopK k={k}: score={s:.7f}")

        theta = best_theta

    a_best = softmax(best_theta)
    return a_best, best_score


def randomized_local_search(a_init, time_limit, deadline, seed=0):
    """
    Multi-strategy randomized local search.
    """
    rng = np.random.default_rng(seed)
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.0005
    it = 0

    while time.monotonic() < deadline and time.monotonic() - t0 < time_limit:
        it += 1
        candidate = current.copy()
        r = rng.random()

        if r < 0.20:
            # Peak-targeted: reduce contributors to peak
            c = fft_convolve(current, current)
            # Get top-3 peaks
            peak_indices = np.argsort(c)[-3:]
            for peak_idx in peak_indices:
                contribs = np.zeros(n)
                for i in range(min(n, peak_idx + 1)):
                    j = peak_idx - i
                    if 0 <= j < n:
                        contribs[i] = current[i] * current[j]
                top_k = max(5, n // 10)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=len(top_idx)))
                other = np.setdiff1d(np.arange(n), top_idx)
                if len(other) > 0:
                    n_inc = min(15, len(other))
                    inc_idx = rng.choice(other, size=n_inc, replace=False)
                    candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.35:
            # Convolution-profile-based: reduce where convolution is high
            c = fft_convolve(current, current)
            c_max = np.max(c)
            pull = np.zeros(n)
            for k in range(len(c)):
                weight = max(0.0, c[k] - 0.8 * c_max)
                if weight > 0:
                    for i in range(min(n, k + 1)):
                        j = k - i
                        if 0 <= j < n:
                            pull[i] += weight * current[j]
            if np.max(pull) > 1e-15:
                pull = pull / np.max(pull)
                candidate *= np.exp(-0.03 * pull)
                candidate /= candidate.sum()
        elif r < 0.50:
            # Block perturbation
            width = max(1, int(n * rng.uniform(0.02, 0.08)))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.60:
            # Symmetric perturbation
            half = n // 2
            pert = np.exp(rng.normal(0, 0.02, size=half))
            candidate[:half] *= pert
            candidate[half:] *= pert[::-1]
        elif r < 0.75:
            # Small global noise
            candidate *= np.exp(rng.normal(0, 0.015, size=n))
        elif r < 0.85:
            # Flatten: reduce high values, boost low
            mean_a = np.mean(current)
            above = candidate > mean_a
            if np.any(above):
                candidate[above] *= np.exp(rng.normal(-0.02, 0.01, size=np.sum(above)))
            below = candidate <= mean_a
            if np.any(below):
                candidate[below] *= np.exp(rng.normal(0.02, 0.01, size=np.sum(below)))
        else:
            # Restart-like: best + noise
            candidate = best * np.exp(rng.normal(0, 0.04, size=n))

        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score

        temperature = max(temperature * 0.99995, 1e-12)

        if it % 500 == 0:
            print(f"    RLS iter {it}: current={current_score:.7f}, best={best_score:.7f}, t={time.monotonic()-t0:.0f}s")

    return best, best_score
