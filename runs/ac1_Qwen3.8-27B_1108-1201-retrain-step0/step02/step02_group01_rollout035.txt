"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using mirror descent on the simplex with
adaptive active set, periodic restarts, and two-coordinate polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_subgradient(w, p):
    """Compute subgradient g[i] = 2 * sum_k w[k] * p[k-i] (valid k)."""
    n = len(p)
    p_rev = p[::-1]
    # conv(w, p_rev)[n-1+i] = sum_j w[j] * p[n-1-(n-1+i)+j] = sum_j w[j] * p[j-i]
    out_len = len(w) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    P = fft(p_rev, fsize)
    conv = np.real(ifft(W * P))
    g = 2.0 * conv[n-1 : 2*n-1]
    return g

def simplex_projection(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def true_objective_norm(p):
    """Compute max of self-convolution for normalized p (sum p = 1)."""
    c = fft_convolve_self(p)
    return c.max()

def run_mirror_descent(p_init, n, time_budget, top_k_start=5, top_k_end=50, 
                       eta_0=0.1, restart_interval=200, seed=42):
    """Mirror descent on the simplex with adaptive active set."""
    rng = np.random.default_rng(seed)
    p = p_init.copy()
    p = np.maximum(p, 0)
    p = p / p.sum()
    
    start = time.monotonic()
    c = fft_convolve_self(p)
    best_obj = c.max()
    best_p = p.copy()
    
    total_iters = int(time_budget / 0.005)  # estimate
    t = 0
    
    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget
        
        # Adaptive top-k
        k = int(top_k_start + (top_k_end - top_k_start) * progress)
        k = max(k, 3)
        
        # Compute convolution
        c = fft_convolve_self(p)
        cur_obj = c.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_p = p.copy()
        
        # Active set: top-k lags
        active = np.argsort(c)[::-1][:k]
        w = np.zeros(2*n-1)
        w[active] = 1.0 / k
        
        # Subgradient
        g = compute_subgradient(w, p)
        
        # Learning rate
        eta = eta_0 / (1 + t / 500)
        
        # Mirror descent update
        # p_new[i] = p[i] * exp(-eta * g[i]) / sum
        log_p = np.log(np.maximum(p, 1e-300))
        log_p_new = log_p - eta * g
        log_p_new = log_p_new - log_p_new.max()
        p_new = np.exp(log_p_new)
        p_new = p_new / p_new.sum()
        
        p = p_new
        
        # Periodic restart: mix with a random point
        if t % restart_interval == 0:
            mix = 0.05 * (t / restart_interval)  # increasing mix
            mix = min(mix, 0.3)
            if rng.random() < 0.1:  # only sometimes
                random_p = rng.dirichlet(np.ones(n))
                p = (1 - mix) * best_p + mix * random_p
                p = p / p.sum()
    
    return best_p, best_obj

def two_coordinate_polish(p, n, time_budget, max_sweeps=20, seed=42):
    """Two-coordinate descent: move mass between pairs of coordinates."""
    rng = np.random.default_rng(seed)
    p = p.copy()
    p = p / p.sum()
    
    start = time.monotonic()
    c = fft_convolve_self(p)
    best_obj = c.max()
    best_p = p.copy()
    
    # Find coordinates with significant mass
    significant = np.where(p > 1e-6)[0]
    if len(significant) < 2:
        return best_p, best_obj
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        # Random pairs from significant coordinates
        pairs_per_sweep = min(200, len(significant) * 5)
        improved = False
        
        for _ in range(pairs_per_sweep):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(len(significant))
            j = rng.integers(len(significant))
            if i == j:
                continue
            i = significant[i]
            j = significant[j]
            if p[j] < 1e-8:
                continue
            
            # Try moving fraction of p[j] to p[i]
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                delta = frac * p[j]
                p_trial = p.copy()
                p_trial[i] += delta
                p_trial[j] -= delta
                if p_trial[j] < 0:
                    continue
                c_trial = fft_convolve_self(p_trial)
                obj_trial = c_trial.max()
                if obj_trial < best_obj:
                    best_obj = obj_trial
                    best_p = p_trial.copy()
                    p = p_trial
                    improved = True
                    break  # Don't try larger fracs if this worked
            # Also try moving from i to j
            if p[i] > 1e-8:
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    delta = frac * p[i]
                    p_trial = p.copy()
                    p_trial[j] += delta
                    p_trial[i] -= delta
                    if p_trial[i] < 0:
                        continue
                    c_trial = fft_convolve_self(p_trial)
                    obj_trial = c_trial.max()
                    if obj_trial < best_obj:
                        best_obj = obj_trial
                        best_p = p_trial.copy()
                        p = p_trial
                        improved = True
                        break
        
        if not improved:
            break
    
    return best_p, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using momentum mirror descent on the probability
simplex with adaptive active set, simulated annealing restarts, and two-coordinate polish.
Key innovations: (1) momentum-accelerated mirror descent, (2) adaptive top-k active set
with periodic hard/soft subgradient switching, (3) simulated annealing for local
minima escape, (4) efficient two-coordinate pair polish at the end.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(w, p):
    """Compute subgradient g[i] = 2 * sum_k w[k] * p[k-i] (summing over valid k)."""
    n = len(p)
    p_rev = p[::-1]
    out_len = len(w) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    P = fft(p_rev, fsize)
    conv = np.real(ifft(W * P))
    g = 2.0 * conv[n - 1:2 * n - 1]
    return g


def simplex_projection(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    s = v.sum()
    if s <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def get_initialization(name, n, rng, S=1.0):
    """Generate various initializations, normalized to sum=S."""
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        a[a < 0.01] = 0.01
        return a * (S / a.sum())
    elif name == "cosine":
        i = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * 3 * i / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "anti_peak":
        i = np.arange(n)
        a = 1.0 - 0.5 * np.cos(2 * np.pi * (n - 1) * i / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "ramp":
        i = np.arange(n)
        a = (i + 1.0) / (n + 1.0)
        return a * (S / a.sum())
    elif name == "two_humps":
        i = np.arange(n)
        a = (np.exp(-0.5 * ((i - n * 0.25) / (n / 8.0)) ** 2) +
             np.exp(-0.5 * ((i - n * 0.75) / (n / 8.0)) ** 2))
        return a * (S / a.sum())
    elif name == "edge_conc":
        i = np.arange(n)
        a = (np.exp(-0.5 * (i / (n / 5.0)) ** 2) +
             np.exp(-0.5 * ((i - n + 1) / (n / 5.0)) ** 2))
        return a * (S / a.sum())
    elif name == "flat_top":
        i = np.arange(n)
        t = i / (n - 1)
        a = np.ones(n)
        mask = (t < 0.1) | (t > 0.9)
        a[mask] = 0.3
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def run_momentum_mirror_descent(p_init, n, time_budget, top_k_start=5, top_k_end=200,
                                eta_0=0.02, momentum=0.9, sa_temp=0.0, seed=42,
                                hard_soft_cycle=50, min_eta=0.0005):
    """
    Momentum mirror descent on the simplex with adaptive active set.
    Uses simulated annealing for occasional escape from local minima.
    """
    rng = np.random.default_rng(seed)
    p = p_init.copy()
    p = np.maximum(p, 1e-30)
    p = p / p.sum()
    p_prev = p.copy()
    q = p.copy()  # momentum/mailing point

    start = time.monotonic()
    c = fft_convolve_self(p)
    best_obj = c.max()
    best_p = p.copy()
    t = 0

    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Adaptive top-k: increase over time
        k = int(top_k_start + (top_k_end - top_k_start) * progress)
        k = max(k, 3)

        # Compute convolution at mailing point q
        c = fft_convolve_self(q)
        cur_obj = c.max()

        # Track best
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_p = p.copy()

        # Active set
        active = np.argsort(c)[::-1][:k]

        # Switch between hard and soft subgradient periodically
        use_soft = (t % (hard_soft_cycle * 2)) < hard_soft_cycle
        if use_soft:
            # Soft-max weights
            c_active = c[active]
            T_soft = 50.0 * (1.0 + progress)
            w_active = np.exp(T_soft * (c_active - c_active.max()))
            w_active = w_active / w_active.sum()
        else:
            w_active = np.full(k, 1.0 / k)

        w = np.zeros(2 * n - 1)
        w[active] = w_active

        # Subgradient at q
        g = compute_subgradient(w, q)

        # Learning rate: geometric decay with floor
        eta = max(min_eta, eta_0 * (0.999 ** t))

        # Mirror descent step from p using gradient at q
        log_p = np.log(np.maximum(p, 1e-300))
        log_p_new = log_p - eta * g
        log_p_new = log_p_new - log_p_new.max()
        p_new = np.exp(log_p_new)
        p_new = p_new / p_new.sum()

        # Enforce minimum mass to prevent coordinates from vanishing
        min_mass = 1e-10
        p_new = np.maximum(p_new, min_mass)
        p_new = p_new / p_new.sum()

        # Momentum update for mailing point
        p_prev = p.copy()
        p = p_new
        q = p + momentum * (p - p_prev)
        q = np.maximum(q, 1e-30)
        q = q / q.sum()

        # Simulated annealing: with decreasing probability, jump toward a random point
        sa_prob = sa_temp * (1.0 - progress)
        if sa_prob > 0 and rng.random() < sa_prob:
            mix = 0.1 + 0.2 * rng.random()
            if rng.random() < 0.5:
                random_p = rng.dirichlet(np.ones(n) * 0.5)
            else:
                random_p = rng.dirichlet(np.ones(n))
            p = (1 - mix) * best_p + mix * random_p
            p = np.maximum(p, 1e-10)
            p = p / p.sum()
            q = p.copy()
            p_prev = p.copy()

    return best_p, best_obj


def two_coordinate_polish(p, n, time_budget, max_sweeps=100, seed=42):
    """
    Two-coordinate descent: try moving mass between pairs of coordinates.
    Uses adaptive fraction search.
    """
    rng = np.random.default_rng(seed)
    p = p.copy()
    p = np.maximum(p, 0)
    p = p / p.sum()

    start = time.monotonic()
    c = fft_convolve_self(p)
    best_obj = c.max()
    best_p = p.copy()

    # Find coordinates with significant mass
    threshold = p.max() * 0.01
    significant = np.where(p > threshold)[0]
    if len(significant) < 2:
        return best_p, best_obj

    fractions = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        pairs_per_sweep = min(500, len(significant) * 10)
        improved = False

        for _ in range(pairs_per_sweep):
            if time.monotonic() - start > time_budget:
                break
            i = significant[rng.integers(len(significant))]
            j = significant[rng.integers(len(significant))]
            if i == j:
                continue
            if p[j] < 1e-10 or p[i] < 1e-10:
                continue

            # Try moving from j to i
            for frac in fractions:
                delta = frac * p[j]
                if delta > p[j]:
                    continue
                p_trial = p.copy()
                p_trial[i] += delta
                p_trial[j] -= delta
                c_trial = fft_convolve_self(p_trial)
                obj_trial = c_trial.max()
                if obj_trial < best_obj * 0.9999:
                    best_obj = obj_trial
                    best_p = p_trial.copy()
                    p = p_trial
                    improved = True
                    break

            # Try moving from i to j
            if p[i] > 1e-10:
                for frac in fractions:
                    delta = frac * p[i]
                    p_trial = p.copy()
                    p_trial[j] += delta
                    p_trial[i] -= delta
                    c_trial = fft_convolve_self(p_trial)
                    obj_trial = c_trial.max()
                    if obj_trial < best_obj * 0.9999:
                        best_obj = obj_trial
                        best_p = p_trial.copy()
                        p = p_trial
                        improved = True
                        break

        # Update significant set
        threshold = p.max() * 0.005
        significant = np.where(p > threshold)[0]
        if len(significant) < 2:
            break
        if not improved:
            break

    return best_p, best_obj

            eta_0=0.02, momentum=0.9, sa_temp=0.0,
            hard_soft_cycle=30, min_eta=0.001,
            seed=11111
        )
        elapsed = time.monotonic() - t0
        full_obj = 2 * n * obj_best
        if full_obj < global_best:
            global_best = full_obj
            global_p = p_best.copy()
            print("  MD-1: obj=%.8f (%.1fs)" % (full_obj, elapsed))

        # Run 2: Smaller eta, larger active set
        if time.monotonic() < deadline - 30:
            p = global_p.copy()
            remain2 = deadline - time.monotonic() - 30
            t0 = time.monotonic()
            p_best, obj_best = run_momentum_mirror_descent(
                p, n, remain2 * 0.5,
                top_k_start=20, top_k_end=400,
                eta_0=0.008, momentum=0.92, sa_temp=0.0,
                hard_soft_cycle=50, min_eta=0.0003,
                seed=22222
            )
            elapsed = time.monotonic() - t0
            full_obj = 2 * n * obj_best
            if full_obj < global_best:
                global_best = full_obj
                global_p = p_best.copy()
                print("  MD-2: obj=%.8f (%.1fs)" % (full_obj, elapsed))

        # Run 3: With simulated annealing to escape local minima
        if time.monotonic() < deadline - 30:
            p = global_p.copy()
            remain3 = deadline - time.monotonic() - 30
            t0 = time.monotonic()
            p_best, obj_best = run_momentum_mirror_descent(
                p, n, remain3 * 0.5,
                top_k_start=10, top_k_end=200,
                eta_0=0.015, momentum=0.9, sa_temp=0.08,
                hard_soft_cycle=40, min_eta=0.0005,
                seed=33333
            )
            elapsed = time.monotonic() - t0
            full_obj = 2 * n * obj_best
            if full_obj < global_best:
                global_best = full_obj
                global_p = p_best.copy()
                print("  MD-3 (SA): obj=%.8f (%.1fs)" % (full_obj, elapsed))

    # Phase 3: Polish
    remain = deadline - time.monotonic()
    polish_budget = 0.2 * remain

    print("Phase 3: Polish, budget=%.0fs" % polish_budget)

    if global_p is not None and polish_budget > 10:
        n = global_n

        # Two-coordinate polish
        t0 = time.monotonic()
        p_best, obj_best = two_coordinate_polish(
            global_p.copy(), n, polish_budget * 0.5,
            max_sweeps=80, seed=44444
        )
        elapsed = time.monotonic() - t0
        full_obj = 2 * n * obj_best
        if full_obj < global_best:
            global_best = full_obj
            global_p = p_best.copy()
            print("  2Coord: obj=%.8f (%.1fs)" % (full_obj, elapsed))

        # Single-coordinate fine tuning
        remain4 = deadline - time.monotonic() - 10
        if remain4 > 5:
            t0 = time.monotonic()
            p_best, obj_best = fine_tune_single_coord(
                global_p.copy(), n, remain4, seed=55555
            )
            elapsed = time.monotonic() - t0
            full_obj = 2 * n * obj_best
            if full_obj < global_best:
                global_best = full_obj
                global_p = p_best.copy()
                print("  1Coord: obj=%.8f (%.1fs)" % (full_obj, elapsed))

        # Another short MD round after polish
        remain5 = deadline - time.monotonic() - 10
        if remain5 > 10:
            p = global_p.copy()
            t0 = time.monotonic()
            p_best, obj_best = run_momentum_mirror_descent(
                p, n, remain5,
                top_k_start=50, top_k_end=500,
                eta_0=0.003, momentum=0.95, sa_temp=0.0,
                hard_soft_cycle=100, min_eta=0.0001,
                seed=66666
            )
            elapsed = time.monotonic() - t0
            full_obj = 2 * n * obj_best
            if full_obj < global_best:
                global_best = full_obj
                global_p = p_best.copy()
                print("  MD-final: obj=%.8f (%.1fs)" % (full_obj, elapsed))

    # Final: try a different n if we have time
    remain = deadline - time.monotonic()
    if remain > 100 and global_p is not None:
        # Try the best shape at a different n by resampling
        for n_try in [int(global_n * 1.25), int(global_n * 0.75)]:
            if time.monotonic() > deadline - 30:
                break
            n_try = max(1024, min(8192, n_try))
            # Resample
            x_old = np.linspace(0, 1, len(global_p))
            x_new = np.linspace(0, 1, n_try)
            p_resampled = np.interp(x_new, x_old, global_p)
            p_resampled = np.maximum(p_resampled, 1e-10)
            p_resampled = p_resampled / p_resampled.sum()

            t0 = time.monotonic()
            p_best, obj_best = run_momentum_mirror_descent(
                p_resampled, n_try, 60.0,
                top_k_start=10, top_k_end=100,
                eta_0=0.01, momentum=0.9, sa_temp=0.0,
                seed=77777
            )
            elapsed = time.monotonic() - t0
            full_obj = 2 * n_try * obj_best
            if full_obj < global_best:
                global_best = full_obj
                global_p = p_best.copy()
                global_n = n_try
                print("  Resample n=%d: obj=%.8f (%.1fs)" % (n_try, full_obj, elapsed))

    # Final verification
    if global_p is None:
        global_p = np.ones(256) / 256
        global_n = 256

    final_seq = global_p.tolist()

    # Verify with evaluate_sequence
    ver = evaluate_sequence(final_seq)
    print("FINAL: obj=%.8f (verified=%.6f), n=%d" % (global_best, ver, global_n))

    if not np.isfinite(ver):
        final_seq = [1.0] * 256
        ver = evaluate_sequence(final_seq)
        print("Fallback: obj=%.6f" % ver)

    return final_seq
