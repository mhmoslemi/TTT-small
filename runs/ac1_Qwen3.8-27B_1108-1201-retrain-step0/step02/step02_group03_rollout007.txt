"""
Optimized step-function sequence search minimizing 2*n*max(autocorr)/(sum)^2.
Uses multi-scale initialization, Adam gradient descent with softmax parameterization,
annealed soft-max, and peak-directed local search with time-budgeted phases.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def softmax_safe(theta):
    """Numerically stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones_like(theta) / len(theta)
    return e / s


def soft_max_value_weights(x, tau):
    """Compute smoothed max and weights."""
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    return x_max + tau * np.log(s), e / s


def grad_smoothed_max_conv(a, tau):
    """Gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    val, w = soft_max_value_weights(c, tau)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = (2.0 / tau) * corr[n - 1:2 * n - 1]
    return val, g, c


def init_fft_phase(n, rng, n_iter=100):
    """Gerchberg-Saxton initialization for flat spectrum."""
    a = np.ones(n) / n
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    for _ in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a /= s
    return a


def init_random(n, rng, spread=0.5):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def init_structured(n, rng):
    """Structured initialization with block patterns."""
    a = np.zeros(n)
    n_blocks = rng.integers(2, max(3, n // 64))
    for b in range(n_blocks):
        start = int(b * n / n_blocks)
        end = int((b + 1) * n / n_blocks)
        amp = 0.5 + 0.5 * rng.random()
        a[start:end] = amp
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def optimize_adam(n, a_init, time_limit, rng, verbose=False):
    """
    Adam gradient descent on log-parameterized non-negative sequence.
    Minimizes smooth max of auto-correlation with annealed temperature.
    """
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = np.clip(a, 1e-8, None)
    a = a / np.sum(a) * S

    # Use log parameterization: a_i = exp(theta_i)
    theta = np.log(a / S)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr0 = 3.0
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = compute_score(a)
    tau_start = 50.0 / n
    tau_end = 0.2 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)

        # Compute current a from theta
        a = np.exp(theta - np.max(theta))
        a = a / np.sum(a) * S

        # Gradient of smooth max
        _, g_a, _ = grad_smoothed_max_conv(a, tau)

        # Convert to theta gradient: a_i = exp(theta_i) * S / sum(exp(theta))
        # d/dtheta_j [sum of a_i] = 0 constraint handled by using
        # the fact that we're optimizing over the simplex
        # grad_theta_j = a_j * (g_j - sum(a_k * g_k)/S)
        inner = np.dot(a, g_a) / S
        g_theta = a * (g_a - inner)

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + step * 0.3)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # Keep theta bounded to prevent overflow
        theta = np.clip(theta, -20, 20)

        if step % 200 == 0:
            a_check = np.exp(theta - np.max(theta))
            a_check = a_check / np.sum(a_check) * S
            sc = compute_score(a_check)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()
            if verbose and step % 1000 == 0:
                print(f"          step={step}, score={sc:.6f}, best={best_score:.6f}, tau={tau:.2e}")

    a_final = np.exp(best_theta - np.max(best_theta))
    a_final = a_final / np.sum(a_final) * S
    final_score = compute_score(a_final)
    if final_score < best_score:
        best_score = final_score
    return a_final, best_score


def optimize_direct(n, a_init, time_limit, rng, verbose=False):
    """
    Direct optimization with multiplicative updates (like RLS but for autocorrelation).
    Uses subgradient of max autocorrelation.
    """
    a = np.clip(a_init.copy(), 1e-10, None)
    a = a / np.sum(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    lr = 0.1
    tau = 10.0 / n

    while time.monotonic() - t0 < time_limit:
        elapsed = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = (10.0 / n) * (0.1) ** (elapsed ** 0.4)
        lr = lr * (0.999 ** int((time.monotonic() - t0) * 10))
        lr = max(lr, 0.001)

        c = fft_convolve(a, a)
        val, w = soft_max_value_weights(c, tau)

        # Subgradient: contribution of each a_i to the peak convolutions
        a_rev = a[::-1]
        corr = fft_convolve(a_rev, w)
        g = (2.0 / tau) * corr[n - 1:2 * n - 1]

        # Normalized gradient (direction in simplex)
        inner = np.dot(a, g)
        g = g - inner * a / np.sum(a)

        # Multiplicative update
        a = a * np.exp(-lr * g)
        a = np.clip(a, 1e-15, None)
        a = a / np.sum(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose and int((time.monotonic() - t0) * 10) % 5 == 0:
            print(f"          direct: score={sc:.6f}, best={best_score:.6f}")

    return best_a, best_score


def local_search_peak(a_init, time_limit, rng, verbose=False):
    """
    Local search targeting the peak convolution lags.
    Reduces contributions to peak lags by redistributing mass.
    """
    current = np.clip(a_init.copy(), 0.0, None)
    s = np.sum(current)
    if s < 1e-10:
        return a_init.copy(), np.inf
    current = current / s
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9997
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        iteration += 1

        if r < 0.4:
            # Peak-directed: find top contributing pairs and reduce them
            conv = fft_convolve(current, current)
            n_lags = min(5, len(conv))
            top_lags = np.argsort(conv)[-n_lags:]
            # For each top lag, find which indices contribute most
            contrib = np.zeros(n)
            for lag in top_lags:
                lag = int(lag)
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    if j < n:
                        contrib[i] += current[i] * current[j]
            # Reduce top contributors, increase bottom
            n_mod = max(2, n // 12)
            top_idx = np.argsort(contrib)[-n_mod:]
            bot_idx = np.argsort(contrib)[:n_mod]
            reduce_amt = np.exp(rng.normal(-0.08, 0.03, size=len(top_idx)))
            increase_amt = np.exp(rng.normal(0.08, 0.03, size=len(bot_idx)))
            candidate[top_idx] *= reduce_amt
            candidate[bot_idx] *= increase_amt
        elif r < 0.55:
            # Adjacent block swap
            width = max(1, int(rng.integers(1, max(2, n // 25))))
            start = int(rng.integers(0, n - 2 * width - 1))
            amt = np.exp(rng.uniform(-0.15, 0.15))
            candidate[start:start + width] *= amt
            candidate[start + width:start + 2 * width] /= amt
        elif r < 0.7:
            # Gaussian perturbation on random indices
            n_pert = max(3, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            candidate[idx] *= np.exp(rng.normal(0, 0.06, size=n_pert))
        elif r < 0.85:
            # Symmetrize partially + noise
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.2 * (sym - current)
            candidate += 0.01 * rng.standard_normal(n)
        else:
            # Coarse block operation
            n_blocks = int(rng.integers(1, 4))
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 10))))
                bs = int(rng.integers(0, n - bw))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0, 0.1, size=bw))

        candidate = np.clip(candidate, 0.0, None)
        cs = np.sum(candidate)
        if cs < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / cs * s

        new_score = compute_score(candidate)
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 5:
                prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < prob:
                    current = candidate
                    current_score = new_score

        temp *= cooling
        if verbose and iteration % 5000 == 0:
            print(f"          LS iter={iteration}, score={current_score:.6f}, best={best_score:.6f}")

    return best, best_score


def upsample_sequence(a, new_n, rng):
    """Upsample via interpolation + small noise."""
    n = len(a)
    if new_n <= n:
        return a[:new_n].copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)

                best_seq = a_new.copy()
                print(f"  *** New best: {best_score:.6f} (n={target_n}) ***")
            else:
                print(f"  Upsample result: {sc_new:.6f} (no improvement)")

    # Phase 6: Final local search if time remains
    print("\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        t6 = min(60.0, max(5.0, deadline - time.monotonic() - 10))
        a_final, sc_final = local_search_peak(best_seq, t6, rng)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"  Final polish: {best_score:.6f}")

    # Return best sequence
    print(f"\n{'='*60}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0}")
    print(f"Time used: {time.monotonic() - t_global:.1f}s / {budget_s:.0f}s")

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Verify with evaluate_sequence
        try:
            verified = evaluate_sequence(final_list)
            print(f"Verified score: {verified:.6f}")
        except Exception as e:
            print(f"Verification error: {e}")
        return final_list

    # Fallback: uniform sequence
    return [1.0] * 256
