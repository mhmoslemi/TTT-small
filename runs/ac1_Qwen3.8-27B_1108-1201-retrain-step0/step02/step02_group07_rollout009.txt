```python
from scipy.optimize import linprog
import numpy as np

def lp_frank_wolfe_step(f, M, n):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)[k] <= M for all k."""
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Inequality constraints: (f*b)[k] <= M for k = 0, ..., 2n-2
    # (f*b)[k] = sum_j f[j] * b[k-j] where j ranges over valid indices
    # For each k, the coefficient of b[i] in (f*b)[k] is f[k-i] (if valid)
    
    # Build A_ub matrix: A_ub[i, j] * b[j] <= b_ub[i]
    # Row k: sum_j f[k-j] * b[j] <= M (where 0 <= k-j < n and 0 <= j < n)
    # i.e., for each j in 0..n-1, coefficient is f[k-j] if 0 <= k-j < n, else 0
    
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.full(n_constraints, M)
    
    for k in range(n_constraints):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]
    
    # Bounds: b[i] >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    else:
        return f.copy()
```

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def lp_direction(f, M, n):
    """Solve LP: max sum(b) s.t. b >= 0, b <= 1, (f*b)[k] <= M for all k."""
    c = -np.ones(n)
    
    # Convolution constraints
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.full(n_constraints, M)
    
    for k in range(n_constraints):
        # (f*b)[k] = sum_j f[k-j] * b[j] for valid j
        js = range(max(0, k - n + 1), min(n, k + 1))
        for j in js:
            A_ub[k, j] = f[k - j]
    
    # Bounds: 0 <= b[i] <= 1
    bounds = [(0, 1)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    else:
        # Fallback: return f itself

"""
LP-based Frank-Wolfe optimizer with log-sum-exp smoothing and annealing polish
for minimizing 2*n*max(autoconv)/(sum^2) of non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time

# Top-level helper functions

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def project_simplex(v):
    """Project vector onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(f, n):
    """Build the LP constraint matrix for cross-convolution constraints."""
    # (f*b)[k] = sum_j f[k-j] * b[j] for valid j
    # We need: (f*b)[k] <= M for k = 0, ..., 2n-2
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n))
    for k in range(n_constraints):
        # For each j in 0..n-1, coefficient of b[j] in (f*b)[k] is f[k-j] if 0<=k-j<n
        js = range(max(0, k - n + 1), min(n, k + 1))
        for j in js:
            A_ub[k, j] = f[k - j]
    return A_ub

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(b) s.t. b >= 0, b <= 1, (f*b)[k] <= M for all k."""
    c = -np.ones(n)
    A_ub = build_lp_matrix(f, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, 1)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        b = result.x
        s = b.sum()
        if s > 1e-15:
            return b / s
        else:
            return f.copy()
    else:
        return f.copy()

def lp_frank_wolfe(a0, time_limit, max_iter=500):
    """LP-based Frank-Wolfe optimization."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    A_ub_cache = None
    last_n = 0
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = conv.max()
        
        # Solve LP
        b = solve_lp_direction(a, M, n)
        
        # Check if b is different from a
        diff = np.abs(b - a).sum()
        if diff < 1e-10:
            break
        
        # Line search
        best_t = 0
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
            a_try = (1 - t) * a + t * b
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t == 0:
            # Try smaller steps
            for t in [0.01, 0.02, 0.03]:
                a_try = (1 - t) * a + t * b
                conv_try = fast_convolve(a_try)
                obj_try = conv_try.max()
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_t = t
        
        if best_t == 0:
            break
        
        a = (1 - best_t) * a + best_t * b
        a = np.maximum(a, 0)
        a /= a.sum()
        conv = fast_convolve(a)
        
        if conv.max() < best_obj:
            best_obj = conv.max()
            best_a = a.copy()
    
    return best_a, best_obj

def lse_smoothed_descent(a0, time_limit, t_init=10.0):
    """Log-sum-exp smoothed gradient descent on the simplex."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    lr = 0.05
    patience = 0
    max_pat = 30
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Compute smoothed gradient
        max_c = conv.max()
        if max_c < 1e-15:
            break
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient: grad_a[i] = 2 * sum_k w[k] * a[k-i]
        grad_a = np.zeros(n)
        for i in range(n):
            k_start = max(0, i)
            k_end = min(len(w), i + n)
            if k_start < k_end:
                m = np.arange(k_start, k_end) - i
                grad_a[i] = 2.0 * np.dot(w[k_start:k_end], a[m])
        
        # Projected gradient step
        a_new = a - lr * grad_a
        a_new = project_simplex(a_new)
        a = a_new
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_pat:
            patience = 0
            lr *= 0.6
            t *= 1.5
            if lr < 0.001:
                lr = 0.02
                t = t_init * 2.0
    
    return best_a, best_obj

def anneal_polish(a0, time_limit, rng):
    """Informed annealing for final polish."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.005 * (1.0 - frac) ** 2 + 0.0001
        
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # Find peak positions
        peak_thresh = max_c * 0.8
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        
        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        target_j = int(rng.integers(n))
        
        if target_j == top_j:
            continue
        
        delta = a[top_j] * rng.uniform(0.005, 0.03)
        if delta < 1e-12:
            continue
        
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
    
    return best_a, best_obj

"""
LP-based Frank-Wolfe optimizer with LSE smoothing, mirror descent, and annealing
for minimizing the auto-correlation peak-to-sum ratio of non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(b) s.t. b >= 0, b <= 1, (f*b)[k] <= M for all k=0..2n-2."""
    c = -np.ones(n)
    
    # Build constraint matrix
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n))
    for k in range(n_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A_ub[k, j] = f[k - j]
    
    b_ub = np.full(n_constraints, M)
    bounds = [(0.0, 1.0)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        b = result.x
        s = b.sum()
        if s > 1e-15:
            return b / s
    return f.copy()


def lp_optimize(a0, time_limit, max_iter=500):
    """LP-based Frank-Wolfe with backtracking line search."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    stall_count = 0
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = conv.max()
        b = solve_lp_direction(a, M, n)
        b = np.maximum(b, 0.0)
        s_b = b.sum()
        if s_b < 1e-15:
            break
        b /= s_b
        
        # Check if direction is useful
        diff = np.abs(b - a).sum()
        if diff < 1e-8:
            stall_count += 1
            if stall_count > 10:
                break
            continue
        stall_count = 0
        
        # Backtracking line search
        cur_peak = conv.max()
        a_new = None
        t = 1.0
        for _ in range(8):
            a_try = (1.0 - t) * a + t * b
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try /= s_try
            conv_try = fast_convolve(a_try)
            peak_try = conv_try.max()
            if peak_try < cur_peak - 1e-12:
                a_new = a_try
                break
            t *= 0.5
        
        if a_new is None:
            break
        
        a = a_new
        conv = fast_convolve(a)
        
        if conv.max() < best_obj:
            best_obj = conv.max()
            best_a = a.copy()
    
    return best_a, best_obj


def lse_descent(a0, time_limit, t_init=20.0):
    """Log-sum-exp smoothed gradient descent on the simplex."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t = t_init
    lr = 0.05
    patience = 0
    max_pat = 40
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # Compute softmax weights
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum
        
        # Compute gradient
        grad_a = np.zeros(n)
        for i in range(n):
            k_start = max(0, i)
            k_end = min(len(w), i + n)
            if k_start < k_end:
                m = np.arange(k_start, k_end) - i
                grad_a[i] = 2.0 * np.dot(w[k_start:k_end], a[m])
        
        # Projected gradient step
        a_new = project_simplex(a - lr * grad_a)
        a = a_new
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a /= s
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_pat:
            patience = 0
            lr *= 0.6
            t *= 1.5
            if lr < 0.002:
                lr = 0.03
