Uses L-BFGS with log-sum-exp smoothing, exact 1D coordinate descent for

Exploits scale invariance of the score to work in probability simplex.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score computation (scale-invariant version with sum=1)."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def lse_value_grad(a, tau):

    Compute log-sum-exp of convolution(a,a) with temperature tau,
    and its gradient with respect to a.

    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)

    lse_val = c_max + tau * np.log(sum_exp)

        elapsed = (time.monotonic() - t0) / time_limit
        # Exponential schedule
        tau = tau_start * (tau_end / tau_start) ** (elapsed / eta)
        if tau < tau_end:
            tau = tau_end

        # L-BFGS for this tau
        def obj(x):
            x_pos = np.maximum(x, 0.0)
            Sx = np.sum(x_pos)
            if Sx < 1e-10:
                return np.inf
            xn = x_pos / Sx  # normalize
            lse_v, _ = lse_value_grad(xn, tau)
            return 2.0 * n * lse_v  # approximate score (smoothed)

        def obj_grad(x):
            x_pos = np.maximum(x, 0.0)
            Sx = np.sum(x_pos)
            if Sx < 1e-10:
                return np.zeros_like(x)
            xn = x_pos / Sx
            lse_v, lse_g = lse_value_grad(xn, tau)
            # Chain rule: x = S * p, so dx = S * dp
            # d(score)/dp = 2n * lse_g (approximately, for fixed S)
            # But we're optimizing in x-space with p = x/sum(x)
            # d(score)/dx[i] = 2n * lse_g[i] / S - 2*score/S (by chain rule)
            score_val = 2.0 * n * lse_v
            grad_x = 2.0 * n * lse_g / Sx - 2.0 * score_val / Sx
            # Zero out where x < 0 (bound)
            grad_x[x < 0] = 0
            return grad_x

        # Try L-BFGS
        x_start = a0.copy()
        # Add small noise for diversity on restarts
        if elapsed > 0.1:
            x_start = x_start + 0.01 * rng.standard_normal(n)
            x_start = np.clip(x_start, 0.0, None)
            Sx = np.sum(x_start)
            if Sx > 0:
                x_start = x_start / Sx

        time_left = time_limit - (time.monotonic() - t0)
        max_iter = max(50, int(200 * time_left / time_limit))

        try:
            result = minimize(
                obj, x_start, jac=obj_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'maxfun': max_iter * 10,
                         'ftol': 1e-10, 'gtol': 1e-6}
            )
            a0 = np.maximum(result.x, 0.0)
            S0 = np.sum(a0)
            if S0 > 1e-10:
                a0 = a0 / S0
        except Exception:
            pass

        # Evaluate actual score
        c_now = fft_convolve(a0, a0)
        current_score = 2.0 * n * np.max(c_now)
        if current_score < best_score:
            best_score = current_score
            best_a = a0.copy()

        if tau <= tau_end * 1.01:
            break

    c_final = fft_convolve(best_a, best_a)
    final_score = 2.0 * n * np.max(c_final)
    return best_a, 2.0 * n * final_score / (np.sum(best_a) ** 2)


def coordinate_descent_1d(a, j, time_per_iter=None):
    """
    Solve the 1D subproblem for coordinate j:
    minimize max_k c[k](a[j]) = max_k (b_k + a[j]*d_k) subject to a[j] >= 0
    """
    n = len(a)
    # Compute d_k and b_k
    # c[k] = sum_i a[i]*a[k-i]
    # Separate the a[j] terms:
    # c[k] = a[j]*a[k-j] (if 0<=k-j<n) + a[k-j]*a[j] (if 0<=k-j<n and k-j != j)
    #        + other terms not involving a[j]
    # Actually: c[k] = sum_{i != j} a[i]*a[k-i] + a[j]*a[k-j] (if valid) + a[k-j]*a[j] (if valid and k-j != j)
    # More carefully:
    # c[k] = sum_i a[i]*a[k-i]
    # Terms involving a[j]: when i=j: a[j]*a[k-j] (if 0<=k-j<n)
    #                      when k-i=j, i.e., i=k-j: a[k-j]*a[j] (if 0<=k-j<n and k-j != j)
    # If k-j == j (i.e., k=2j): only one term a[j]*a[j] = a[j]^2
    # So: c[k] = base_k + a[j]*d_k where:
    #   d_k = a[k-j] (if 0<=k-j<n) + a[k-j] (if 0<=k-j<n and k-j != j)
    #         = a[k-j] + a[k-j] = 2*a[k-j] (if 0<=k-j<n and k-j != j)
    #         = a[j] if k=2j (but this uses a[j] itself, so it's a[j]^2)
    
    # Let me redo this. c[k] is quadratic in a[j] only when k=2j (giving a[j]^2).
    # For k != 2j: c[k] = base_k + a[j] * 2*a[k-j] (if 0<=k-j<n)
    #                = base_k + a[j] * a[k-j] (if k-j < j, since a[k-j] is fixed)
    # Wait, I'm confusing myself. Let me be very explicit.
    
    # c[k] = sum_{i=0}^{n-1} a[i] * a[k-i] * I[0 <= k-i < n]
    # 
    # Terms where i = j: a[j] * a[k-j] * I[0 <= k-j < n]
    # Terms where k-i = j, i.e., i = k-j: a[k-j] * a[j] * I[0 <= k-(k-j) < n] = a[k-j]*a[j] * I[0<=j<n]
    #   (this is valid always since 0<=j<n, but we need i=k-j to be in [0,n-1])
    #   So: a[k-j]*a[j] * I[0 <= k-j < n]
    #
    # If j != k-j (i.e., k != 2j): two separate terms, each = a[j]*a[k-j]*I[0<=k-j<n]
    #   Total a[j] contribution: 2*a[j]*a[k-j]*I[0<=k-j<n]
    # If j == k-j (i.e., k == 2j): one term a[j]*a[j] = a[j]^2
    #
    # So: c[k] = base_k + a[j]*d_k + (a[j]^2 if k==2j else 0)
    # where d_k = 2*a[k-j] if (0<=k-j<n and k != 2j), or a[j] if k==2j... 
    # Hmm, for k==2j: c[2j] = base_2j + a[j]^2, where base_2j = sum_{i!=j} a[i]*a[2j-i]*I[valid]
    # This is quadratic, not linear! So the 1D subproblem is not purely convex unless k==2j is handled.
    #
    # For k==2j: c[2j](x) = base_2j + x^2 (where x = a[j])
    # For k != 2j: c[k](x) = base_k + x * d_k (linear in x)
    #
    # The max of a quadratic and linear functions is still convex (the quadratic x^2 + base is convex,
    # and max of convex functions is convex). So we can still solve it as a 1D convex problem.
    
    # Compute base_k and d_k
    base = np.zeros(2 * n - 1)
    d = np.zeros(2 * n - 1)
    
    for k in range(2 * n - 1):
        b_k = 0.0
        d_k = 0.0
        for i in range(n):
            if i == j:
                # This term involves a[j]
                if 0 <= k - i < n:
                    if k == 2 * j:
                        # a[j]^2 term - handle separately
                        pass
                    else:
                        d_k += a[k - i]
            else:
                if 0 <= k - i < n:
                    b_k += a[i] * a[k - i]
        base[k] = b_k
        d[k] = d_k
    
    # For k == 2j: c[2j](x) = base[2j] + x^2
    # For k != 2j: c[k](x) = base[k] + d[k]*x
    
    def f(x):
        """max_k c[k](x)"""
        vals = base + d * x
        if 2 * j < 2 * n - 1:
            vals[2 * j] = base[2 * j] + x * x  # quadratic term
        return np.max(vals)
    
    if f(0.0) == np.inf:
        return 0.0
    
    # Find upper bound: where f(x) stops being useful
    # f(x) >= x^2 (from the k=2j term) if 2j is in range
    # So for large x, f(x) ~ x^2, and we want f(x) <= current_best
    current_max = f(a[j])
    upper = np.sqrt(current_max + 1.0) + 1.0
    
    # Use minimize_scalar on [0, upper]
    try:
        res = minimize_scalar(f, bounds=(0.0, upper), method='bounded',
                             options={'xatol': 1e-10, 'maxiter': 100})
        return max(0.0, res.x)
    except Exception:
        return a[j]


def coordinate_descent_polish(a_init, time_limit, rng):
    """
    Coordinate descent with exact 1D subproblem solving.
    """
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    S = np.sum(current)
    current = current / S  # normalize to sum=1
    current_score = 2.0 * n * np.max(fft_convolve(current, current))
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        epoch += 1
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = current[j]
            new_val = coordinate_descent_1d(current, j)
            current[j] = new_val
            
            # Check improvement
            c = fft_convolve(current, current)
            new_score = 2.0 * n * np.max(c)
            if new_score < current_score - 1e-12:
                current_score = new_score
                improved = True
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
            else:
                current[j] = old_val
        
        if epoch % 10 == 0:
            print(f"    CD epoch {epoch}, score={current_score:.8f}, best={best_score:.8f}, t={time.monotonic()-t0:.0f}s")
        
        if not improved and epoch > 20:
            # Try random perturbation to escape
            j = int(rng.integers(0, n))
            current[j] = max(0.0, current[j] + 0.05 * rng.standard_normal())
            S = np.sum(current)
            current = current / max(S, 1e-10)
            current_score = 2.0 * n * np.max(fft_convolve(current, current))
    
    # Final normalization
    S = np.sum(best)
    if S > 0:
        best = best / S
    final_score = 2.0 * len(best) * np.max(fft_convolve(best, best))
    return best, final_score


def peak_targeted_polish(a_init, time_limit, rng):
    """
    Targeted peak reduction: identify the peak lag and redistribute mass
    to reduce it while maintaining sum constraint.
    """
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    S = np.sum(current)
    current = current / S
    current_score = 2.0 * n * np.max(fft_convolve(current, current))
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]

        # Find elements contributing most to the peak
        # c[peak_idx] = sum_i current[i] * current[peak_idx - i] * I[valid]
        contribs = np.zeros(n)
        for i in range(n):
            jp = peak_idx - i
            if 0 <= jp < n:
                contribs[i] = current[i] * current[jp]

        strategy = rng.random()

        if strategy < 0.4:
            # Reduce top contributors, add to random bottom elements
            top_k = max(2, n // 16)
            top_idx = np.argsort(contribs)[-top_k:]
            bottom_idx = np.argsort(contribs)[:top_k]
            reduce_frac = rng.uniform(0.03, 0.08)
            total_red = 0.0
            for i in top_idx:
                red = current[i] * reduce_frac
                current[i] -= red
                total_red += red
            for i in bottom_idx:
                current[i] += total_red / len(bottom_idx)

        elif strategy < 0.7:
            # Swap: reduce a top contributor, increase a non-contributor
            top_k = max(2, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            j_red = top_idx[int(rng.integers(0, top_k))]
            # Pick j_inc that doesn't contribute to peak
            non_contrib = np.where(contribs < np.max(contribs) * 0.1)[0]
            if len(non_contrib) > 0:
                j_inc = non_contrib[int(rng.integers(0, len(non_contrib)))]
                amt = current[j_red] * rng.uniform(0.05, 0.15)
                current[j_red] -= amt
                current[j_inc] += amt

        else:
            # Global small perturbation
            current *= np.exp(rng.normal(0, 0.02, size=n))

        current = np.clip(current, 0.0, None)
        S = np.sum(current)
        if S < 1e-10:
            current = np.ones(n) / n
            S = 1.0
        current = current / S
        new_score = 2.0 * n * np.max(fft_convolve(current, current))

        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / (0.001 * (1.0 + 0.995 ** step))):
            current_score = new_score  # accept worse (simulated annealing)

        if step % 500 == 0:
            print(f"    PT step={step}, score={current_score:.8f}, best={best_score:.8f}, t={time.monotonic()-t0:.0f}s")

    S = np.sum(best)
    if S > 0:
        best = best / S
    return best, 2.0 * len(best) * np.max(fft_convolve(best, best))


def adam_optimize(a_init, time_limit, rng):
    """
    Adam-based optimization with log-sum-exp smoothing on the probability simplex.
    """
    n = len(a_init)
    a0 = a_init.astype(np.float64).copy()
    S = np.sum(a0)
    a0 = a0 / max(S, 1e-10)

    # Work in log-space (log-barrier for positivity)
    theta = np.log(a0 + 1e-15)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0

    best_a = a0.copy()
    best_score = 2.0 * n * np.max(fft_convolve(a0, a0))
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        # Temperature schedule
        tau = (3.0 / n) * (0.01 / n) ** (elapsed ** 0.8)
        tau = max(tau, 0.005 / n)

        # Compute softmax normalization
        t = theta - np.max(theta)
        e = np.exp(t)
        p = e / np.sum(e)  # probability vector, sum = 1

        # Compute LSE and gradient
        lse_v, lse_g = lse_value_grad(p, tau)

        # Convert gradient to theta-space
        # p = softmax(theta), dp/dtheta = p*(I - p*1^T)
        # d(lse)/dtheta = J^T * d(lse)/dp where J[i,j] = p[i]*(delta_ij - p[j])
        # d(lse)/dtheta[i] = p[i] * (lse_g[i] - sum_j p[j]*lse_g[j])
        inner = np.dot(p, lse_g)
        g_theta = p * (lse_g - inner)

        # Adam update
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g_theta
        v_buf = beta2 * v_buf + (1 - beta2) * g_theta ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)

        lr = 0.5 / np.sqrt(step + 1)
        if elapsed < 0.3:
            lr *= 1.5
        elif elapsed > 0.7:
            lr *= 0.3

        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic evaluation
        if step % 100 == 0:
            t_check = theta - np.max(theta)
            p_check = np.exp(t_check) / np.sum(np.exp(t_check))
            score = 2.0 * n * np.max(fft_convolve(p_check, p_check))
            if score < best_score:
                best_score = score
                best_a = p_check.copy()
                no_improve = 0
            else:
                no_improve += 100

            # Escape mechanism
            if no_improve > 5000:
                r = rng.random()
                if r < 0.5:
                    theta = np.log(best_a + 1e-15) + 0.1 * rng.standard_normal(n)
                else:
                    # Perturb peak contributors
                    c_cur = fft_convolve(best_a, best_a)
                    peak_idx = int(np.argmax(c_cur))
                    contribs = np.zeros(n)
                    for i in range(n):
                        jp = peak_idx - i
                        if 0 <= jp < n:
                            contribs[i] = best_a[i] * best_a[jp]
                    top_k = np.argsort(contribs)[-n // 8:]
                    theta[top_k] += 0.05 * rng.standard_normal(len(top_k))
                no_improve = 0

            if step % 2000 == 0:
                print(f"    Adam step={step}, score={score:.8f}, best={best_score:.8f}, tau={tau:.2e}")

    return best_a, 2.0 * n * np.max(fft_convolve(best_a, best_a))


def make_initialization(n, rng, method='structured'):
    """Create various initializations."""
    t = np.arange(n, dtype=np.float64)

    if method == 'structured':
        # Sum of incommensurate frequencies
        a = np.ones(n)
        n_freqs = min(20, n // 4)
        for k in range(1, n_freqs + 1):
            freq = k * (1.0 + 0.5 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.2 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'random':
        return np.clip(1.0 + 0.5 * rng.standard_normal(n), 0.01, None)

    elif method == 'bimodal':
        a = np.ones(n)
        center = n // 2
        width = n // 6
        a[center - width:center + width] = 0.3
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'cosine':
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t / n * rng.integers(2, 8))
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    else:
        return np.clip(1.0 + 0.3 * rng.standard_normal(n), 0.01, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimizing the evaluation score.
    Phase 1: L-BFGS optimization with multiple initializations
    Phase 2: Adam-based refinement
    Phase 3: Coordinate descent polish
    Phase 4: Peak-targeted polish
    Phase 5: Length exploration
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 20.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        S = np.sum(start_seq)
        if S > 0.01:
            s = 2.0 * len(start_seq) * np.max(fft_convolve(start_seq, start_seq)) / (S * S)
            print(f"Provided: n={len(start_seq)}, score={s:.8f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()

    # Primary lengths to focus on
    primary_lengths = [256, 320, 384, 448]

    # === PHASE 1: L-BFGS with multiple strategies ===
    print(f"\n{'='*60}")
    print(f"PHASE 1: L-BFGS optimization")
    print(f"{'='*60}")

    for n_try in primary_lengths:
        if time.monotonic() > deadline - 60:
            break

        # Get initialization
        if start_seq is not None:
            x_old = np.linspace(0, 1, len(start_seq))
            x_new = np.linspace(0, 1, n_try)
            a0 = np.interp(x_new, x_old, start_seq)
        else:
            a0 = make_initialization(n_try, rng, 'structured')
        S0 = np.sum(a0)
        a0 = a0 / max(S0, 1e-10)

        # Run L-BFGS
        t_alloc = min(120.0, max(20.0, (deadline - time.monotonic() - 60) * 0.3 / len(primary_lengths)))
        a_res, score_res = optimize_lbfgs(a0, t_alloc, rng)
        actual = evaluate_sequence([float(x) for x in a_res])
        print(f"  n={n_try}: L-BFGS score={actual:.8f} (local={score_res:.8f})")

        if actual < best_score:
            best_score = actual
            best_seq = a_res.copy()
            print(f"  *** New best: {best_score:.8f} at n={n_try} ***")

    # === PHASE 2: Adam refinement on best ===
    print(f"\n{'='*60}")
    print(f"PHASE 2: Adam refinement")
    print(f"{'='*60}")

    if best_seq is not None:
        t_phase2 = min(300.0, max(30.0, (deadline - time.monotonic() - 60) * 0.3))
        print(f"  Time allocated: {t_phase2:.0f}s, starting score: {best_score:.8f}")
        a0_adam = best_seq.copy()
        S = np.sum(a0_adam)
        a0_adam = a0_adam / max(S, 1e-10)
        a_res, score_res = adam_optimize(a0_adam, t_phase2, rng)
        actual = evaluate_sequence([float(x) for x in a_res])
        print(f"  Adam result: score={actual:.8f}")
        if actual < best_score:
            best_score = actual
            best_seq = a_res.copy()
            print(f"  *** New best: {best_score:.8f} ***")

    # === PHASE 3: Coordinate descent ===
    print(f"\n{'='*60}")
    print(f"PHASE 3: Coordinate descent polish")
    print(f"{'='*60}")

    if best_seq is not None:
        t_phase3 = min(200.0, max(20.0, (deadline - time.monotonic() - 30) * 0.3))
        print(f"  Time allocated: {t_phase3:.0f}s, starting score: {best_score:.8f}")
        cd_res, cd_score = coordinate_descent_polish(best_seq, t_phase3, rng)
        # Scale back to proper values for evaluation
        actual = evaluate_sequence([float(x) for x in cd_res])
        print(f"  CD result: score={actual:.8f} (local={cd_score:.8f})")
        if actual < best_score:
            best_score = actual
            best_seq = cd_res.copy()
            print(f"  *** New best: {best_score:.8f} ***")

    # === PHASE 4: Peak-targeted polish ===
    print(f"\n{'='*60}")
    print(f"PHASE 4: Peak-targeted polish")
    print(f"{'='*60}")

    if best_seq is not None:
        t_phase4 = min(200.0, max(20.0, (deadline - time.monotonic() - 30) * 0.3))
        print(f"  Time allocated: {t_phase4:.0f}s, starting score: {best_score:.8f}")
        pt_res, pt_score = peak_targeted_polish(best_seq, t_phase4, rng)
        actual = evaluate_sequence([float(x) for x in pt_res])
        print(f"  PT result: score={actual:.8f} (local={pt_score:.8f})")
        if actual < best_score:
            best_score = actual
            best_seq = pt_res.copy()
            print(f"  *** New best: {best_score:.8f} ***")

    # === PHASE 5: Try more initializations and lengths ===
    print(f"\n{'='*60}")
    print(f"PHASE 5: Extended exploration")
    print(f"{'='*60}")

    # Try different initializations on the best length
    n_best = len(best_seq) if best_seq is not None else 256
    methods = ['random', 'bimodal', 'cosine', 'structured']

    for method in methods:
        if time.monotonic() > deadline - 30:
            break
        t_try = min(60.0, max(10.0, (deadline - time.monotonic() - 20) / 4))
        if t_try < 5:
            break

        a0 = make_initialization(n_best, rng, method)
        S = np.sum(a0)
        a0 = a0 / max(S, 1e-10)

        # Quick L-BFGS
        a_res, score_res = optimize_lbfgs(a0, t_try, rng)
        actual = evaluate_sequence([float(x) for x in a_res])
        print(f"  n={n_best}, init={method}: score={actual:.8f}")
        if actual < best_score:
            best_score = actual
            best_seq = a_res.copy()
            print(f"  *** New best: {best_score:.8f} ***")

    # Try a few more lengths
    for n_extra in [192, 288, 480, 512]:
        if time.monotonic() > deadline - 20:
            break
        t_try = min(30.0, max(8.0, deadline - time.monotonic() - 15))
        if t_try < 5:
            break

        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_extra)
            a0 = np.interp(x_new, x_old, best_seq)
        else:
            a0 = make_initialization(n_extra, rng, 'structured')
        S = np.sum(a0)
        a0 = a0 / max(S, 1e-10)

        a_res, score_res = optimize_lbfgs(a0, t_try, rng)
        actual = evaluate_sequence([float(x) for x in a_res])
        print(f"  n={n_extra}: score={actual:.8f}")
        if actual < best_score:
            best_score = actual
            best_seq = a_res.copy()
            print(f"  *** New best: {best_score:.8f} ***")

    # === FINAL POLISH ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_final = deadline - time.monotonic() - 10
        if t_final > 10:
            print(f"\n{'='*60}")
            print(f"FINAL POLISH (t={t_final:.0f}s)")
            print(f"{'='*60}")
            # Quick CD
            cd_res, cd_score = coordinate_descent_polish(best_seq, t_final * 0.5, rng)
            actual = evaluate_sequence([float(x) for x in cd_res])
            if actual < best_score:
                best_score = actual
                best_seq = cd_res.copy()
                print(f"  After final CD: {best_score:.8f}")

            # Quick peak shave
            if time.monotonic() < deadline - 5:
                t_ps = deadline - time.monotonic() - 5
                pt_res, pt_score = peak_targeted_polish(best_seq, t_ps, rng)
                actual = evaluate_sequence([float(x) for x in pt_res])
                if actual < best_score:
                    best_score = actual
                    best_seq = pt_res.copy()
                    print(f"  After final PT: {best_score:.8f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"=== FINAL: score={final_eval:.8f}, n={len(final_list)} ===")
        print(f"{'='*60}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
