"""
LP-guided optimization for minimum peak autocorrelation non-negative sequences.
Combines p-norm L-BFGS initialization with LP-constrained mixing iterations
and targeted local search to minimize 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def pnorm_obj_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p.
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def solve_lp_for_g(f, M):
    """
    Solve LP: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g vector.
    """
    n = len(f)
    
    # Build constraint matrix A: A[k, i] = f[k-i] if 0 <= k-i < n else 0
    # Number of rows: 2n-1
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        # For row k: A[k, i] = f[k-i] if 0 <= k-i < n
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            A[k, i] = f[k - i]
    
    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        # Fallback: return f itself
        return f.copy()

def subgradient_step(f, step_size, momentum=0.9):
    """
    One subgradient step on the max objective with momentum.
    Returns updated f.
    """
    n = len(f)
    c = fft_convolve(f, f)
    peak_idx = int(np.argmax(c))
    
    # Subgradient of max(c) w.r.t. f
    # c[peak_idx] = sum_i f[i] * f[peak_idx - i]
    # dc/df[i] = f[peak_idx - i] (for valid i)
    grad = np.zeros(n)
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            grad[i] = f[j]
    
    # Also need to account for the (sum f)^2 in the denominator
    # But since we keep f normalized (sum=1), the denominator is constant.
    # So we just minimize max(conv(f,f)) subject to sum(f) = 1, f >= 0.
    
    # Projected gradient step
    f_new = f - step_size * grad
    # Project onto simplex: f_new >= 0, sum(f_new) = 1
    f_new = np.maximum(f_new, 0)
    f_new = f_new / np.sum(f_new)
    
    return f_new


def nesterov_subgradient(f, time_limit=60.0, max_iters=5000):
    """
    Nesterov accelerated subgradient method on max(conv(f,f)) with sum(f)=1 constraint.
    """
    n = len(f)
    best_f = f.copy()
    best_score = local_score(f)
    
    x = f.copy()  # current point
    y = f.copy()  # Nesterov lookahead
    
    t0 = time.monotonic()
    v = np.zeros(n)  # momentum
    lr = 0.01
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fft_convolve(y, y)
        peak_idx = int(np.argmax(c))
        
        # Subgradient at peak
        grad = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                grad[i] = y[j]
        
        # Momentum update
        v = 0.9 * v - lr * grad
        x = x + v
        
        # Project onto simplex
        x = np.maximum(x, 0)
        x = x / np.sum(x)
        
        # Nesterov lookahead
        alpha = 2.0 / (it + 2.0)
        y = (1 - alpha) * x + alpha * y
        
        s = local_score(x)
        if s < best_score:
            best_score = s
            best_f = x.copy()
        
        # Decay learning rate
        lr = 0.01 / (1 + it * 0.001)
    
    return best_f, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Bell-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)

"""
LP-guided optimization with p-norm L-BFGS initialization for minimum peak
autocorrelation of non-negative sequences. Uses linear programming to find
optimal mixing directions that reduce the peak convolution value.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def build_lp_matrix(n):
    """Build the constraint matrix for the LP. A[k, i] = 1 if 0 <= k-i < n else 0."""
    # Actually A[k, i] = f[k-i] which depends on f, so we can't pre-build it.
    # But we can pre-build the index structure.
    # For row k, the non-zero entries are at columns i where 0 <= k-i < n
    # i.e., max(0, k-n+1) <= i <= min(k, n-1)
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows), np.array(cols)


def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    # A[k, i] = f[k-i] if 0 <= k-i < n else 0
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            A[k, i] = f[k - i]
    
    c_obj = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return f.copy()

```python
"""
LP-guided optimization for minimum peak autocorrelation non-negative sequences.
Combines p-norm L-BFGS initialization with LP-constrained mixing directions
and targeted local search to minimize 2n*max(conv(a,a))/(sum(a))^2.
Key innovation: uses linear programming to find optimal mixing directions
that provably reduce the peak convolution value.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def build_lp_matrix(f, n, top_k=None):
    """Build LP constraint matrix. A[k,i] = f[k-i] if valid."""
    if top_k is not None:
        c = fft_convolve(f, f)
        all_lags = np.arange(2 * n - 1)
        top_lags = all_lags[np.argsort(c)[-top_k:]]
        num_rows = top_k
    else:
        top_lags = np.arange(2 * n - 1)
        num_rows = 2 * n - 1

    A = np.zeros((num_rows, n))
    for idx, k in enumerate(top_lags):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            A[idx, i] = f[k - i]
    return A


def solve_lp(f, M, n, top_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for selected k, g >= 0."""
    A = build_lp_matrix(f, n, top_k)
    num_rows = A.shape[0]

    c_obj = -np.ones(n)
    b_ub = np.full(num_rows, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return result.x
    except Exception:
        pass
    return f.copy()


def lp_iteration(f, time_limit=120.0, max_rounds=2000, rng=None, top_k=None):
    """
    LP-guided iteration: solve LP, mix with line search.
    Returns best f and score found.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    stagnation = 0

    for it in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        c = fft_convolve(f, f)
        M = np.max(c)
        g = solve_lp(f, M, n, top_k=top_k)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        improved = False
        for t in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            f_s = np.sum(f_new)
            if f_s < 1e-10:
                continue
            f_new = f_new / f_s
            s_new = local_score(f_new)
            if s_new < best_score - 1e-12:
                f = f_new
                best_f = f.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            stagnation += 1
            if stagnation > 15:
                # Perturb g and retry
                g_pert = g * np.exp(rng.normal(0, 0.05, n))
                g_pert = np.maximum(g_pert, 0)
                gp = np.sum(g_pert)
                if gp > 1e-10:
                    g_pert = g_pert / gp
                    f_new = 0.5 * f + 0.5 * g_pert
                    f_new = np.maximum(f_new, 0)
                    f_new = f_new / np.sum(f_new)
                    s_new = local_score(f_new)
                    if s_new < best_score:
                        f = f_new
                        best_f = f.copy()
                        best_score = s_new
                        stagnation = 0
                    else:
                        break
                else:
                    break
        else:
            stagnation = 0

        if (it + 1) % 500 == 0:
            print(f"    lp_iter: it={it+1}, score={best_score:.6f}, t={time.monotonic()-t0:.0f}s")

    return best_f, best_score


def nesterov_subgrad(f, time_limit=60.0):
    """Nesterov accelerated subgradient on max(conv(f,f)) with sum(f)=1."""
    n = len(f)
    best_f = f.copy()
    best_score = local_score(f)
    x = f.copy()
    y = f.copy()
    t0 = time.monotonic()
    lr = 0.1

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fft_convolve(y, y)
        peak_idx = int(np.argmax(c))

        grad = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                grad[i] = y[j]

        x = x - lr * grad
        x = np.maximum(x, 0.0)
        x = x / np.sum(x)

        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * x + alpha * y

        if it % 20 == 0:
            s = local_score(x)
            if s < best_score:
                best_score = s
                best_f = x.copy()

        lr = 0.1 / (1.0 + 0.003 * it)

    return best_f, best_score


def local_polish(f, time_limit, rng):
    """Targeted local search with coordinate descent and peak shaving."""
    n = len(f)
    current = f.copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))
        r = rng.random()

        if r < 0.30:
            # Reduce top contributors to the peak lag
            candidate = current.copy()
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(5, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(15, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.015, size=n_inc))
            candidate = np.maximum(candidate, 1e-10)
            candidate = candidate / np.sum(candidate)
            s = local_score(candidate)
            if s < current_score:
                current = candidate
                current_score = s
                if s < best_score:
                    best_score = s
                    best = current.copy()

        elif r < 0.60:
            # Coordinate descent on random coordinate
            i = int(rng.integers(0, n))
            j = peak_idx - i
            if 0 <= j < n:
                base = current[i]
                for factor in [0.3, 0.5, 0.7, 1.3, 1.6, 2.0, 3.0]:
                    candidate = current.copy()
                    candidate[i] = base * factor
                    candidate = np.maximum(candidate, 1e-10)
                    candidate = candidate / np.sum(candidate)
                    s = local_score(candidate)
                    if s < current_score:
                        current = candidate
                        current_score = s
                        if s < best_score:
                            best_score = s
                            best = current.copy()
                        break
            else:
                candidate = current.copy()
                candidate[i] *= 1.5
                candidate = candidate / np.sum(candidate)
                s = local_score(candidate)
                if s < current_score:
                    current = candidate
                    current_score = s
                    if s < best_score:
                        best_score = s
                        best = current.copy()

        elif r < 0.80:
            # Multi-peak reduction: reduce contributors to top-3 peaks
            candidate = current.copy()
            top_peaks = np.argsort(c)[-3:]
            all_contribs = np.zeros(n)
            for pk in top_peaks:
                for i in range(n):
                    j = pk - i
                    if 0 <= j < n:
                        all_contribs[i] += current[i] * current[j]
            top_k = max(8, n // 6)
            top_idx = np.argsort(all_contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.012, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(20, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.012, size=n_inc))
            candidate = np.maximum(candidate, 1e-10)
            candidate = candidate / np.sum(candidate)
            s = local_score(candidate)
            if s < current_score:
                current = candidate
                current_score = s
                if s < best_score:
                    best_score = s
                    best = current.copy()

        elif r < 0.92:
            # Block perturbation
            width = max(2, int(rng.integers(2, max(3, n // 4))))
            start = int(rng.integers(0, n - width))
            candidate = current.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))
            candidate = np.maximum(candidate, 1e-10)
            candidate = candidate / np.sum(candidate)
            s = local_score(candidate)
            if s < current_score:
                current = candidate
                current_score = s
                if s < best_score:
                    best_score = s
                    best = current.copy()
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0, 0.015, n))
            candidate = np.maximum(candidate, 1e-10)
            candidate = candidate / np.sum(candidate)
            s = local_score(candidate)
            if s < current_score:
                current = candidate
                current_score = s
                if s < best_score:
                    best_score = s
                    best = current.copy()

        if it % 300 == 0:
            print(f"    polish: it={it}, score={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")

    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)
