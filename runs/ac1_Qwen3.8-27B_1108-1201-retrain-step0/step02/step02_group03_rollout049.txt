"""
Multi-strategy optimization for minimizing peak self-convolution of non-negative sequences.
Uses gradient descent with annealed soft-max and direct peak targeting, combined with
peak-canceling mutations and local search. Starts from provided sequence and also explores
fresh initializations.

import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = max(1, (n + m - 2).bit_length())
    size = 1 << size
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def score_and_grad(a, tau):
    """Compute soft-max score and gradient."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    w = e / np.sum(e)
    value = c_max + tau * np.log(np.sum(e))
    # Gradient of soft-max of conv w.r.t. a
    # d(conv[k])/da[i] = a[k-i] (for 0 <= k-i < n)
    # grad[i] = sum_k w[k] * d(conv[k])/da[i] = sum_k w[k] * a[k-i]
    # This is a convolution of w with a, evaluated at the right positions
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    # corr has length 2n-1
    # We need corr[n-1 : 2n-1] which gives sum_k w[k] * a[k-i] for i = 0,...,n-1
    g = corr[n-1:2*n-1]
    return value, g, c

def direct_peak_grad(a):
    """Compute gradient of actual max of conv."""
    n = len(a)
    c = fft_convolve(a, a)
    k_star = np.argmax(c)
    # Gradient of conv[k_star] w.r.t. a[i] = a[k_star - i] for valid k_star - i
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] = a[j]
    return c[k_star], g, c, k_star

def optimize_from(a_init, time_budget, rng, tau_start=50.0, tau_end=0.05):
    """Adam gradient descent on softmax-parameterized sequence."""
    n = len(a_init)
    S = np.sum(a_init)
    a_norm = a_init / S
    theta = np.log(a_norm + 1e-15)
    
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    step = 0
    t0 = time.monotonic()
    best_score = local_score(a_init)
    best_theta = theta.copy()
    best_a = a_init.copy()
    patience = 0
    lr0 = 3.0
    
    while time.monotonic() - t0 < time_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_budget)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        
        a = softmax(theta) * S
        
        # Choose between soft-max and direct peak gradient
        if tau > np.max(fft_convolve(a, a)) * 0.1:
            value, g_a, c = score_and_grad(a, tau)
        else:
            value, g_a, c, k_star = direct_peak_grad(a)
        
        # Adjust gradient for the softmax parameterization
        # a = softmax(theta) * S
        # da/dtheta[i] = a[i] * (delta[i,j] - a[j]/S * S) ... 
        # Actually: da[i]/dtheta[j] = a[i] * (delta_ij - softmax_j)
        # So grad_theta[j] = sum_i grad_a[i] * da[i]/dtheta[j]
        # = sum_i g_a[i] * a[i] * (delta_ij - p[j]) where p = a/S
        # = a[j] * g_a[j] - S * p[j] * sum_i g_a[i] * p[i] * S / S
        # Let me redo this more carefully.
        # a[i] = S * softmax(theta)[i] = S * exp(theta[i]) / sum(exp(theta))
        # da[i]/dtheta[j] = a[i] * (delta_ij - softmax(theta)[j]) = a[i] * (delta_ij - a[j]/S)
        # grad_theta[j] = sum_i g_a[i] * da[i]/dtheta[j]
        # = sum_i g_a[i] * a[i] * delta_ij - sum_i g_a[i] * a[i] * a[j]/S
        # = g_a[j] * a[j] - (a[j]/S) * sum_i g_a[i] * a[i]
        
        inner = np.dot(g_a, a)
        g_theta = a * (g_a - inner / S)
        
        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Check score periodically
        if step % 200 == 0:
            a_check = softmax(theta) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_a = a_check.copy()
                best_theta = theta.copy()
                patience = 0
            else:
                patience += 1
            
            if step % 1000 == 0:
                print(f"      step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.4f}, t={elapsed:.1f}s")
        
        # Early stopping if stuck
        if patience > 20:
            break
    
    a_final = softmax(best_theta) * S
    # Ensure non-negative
    a_final = np.clip(a_final, 0.0, None)
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
    return a_final, best_score

def peak_cancel(a, n_peaks=5, strength=0.05, rng=None):
    """Reduce the top peaks by redistributing mass."""
    n = len(a)
    c = fft_convolve(a, a)
    S = np.sum(a)
    
    # Find top peak lags
    top_lags = np.argsort(c)[-n_peaks:]
    
    # Compute contribution of each element to the peaks
    contrib = np.zeros(n)
    for k in top_lags:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                contrib[i] += a[i] * a[j] / c[k]
    
    # Reduce high-contributing elements, increase low-contributing ones
    if rng is not None:
        a_new = a.copy()
        n_reduce = max(1, n // 10)
        n_incr = max(1, n // 10)
        
        high_idx = np.argsort(contrib)[-n_reduce:]
        low_idx = np.argsort(contrib)[:n_incr]
        
        a_new[high_idx] *= (1 - strength * rng.uniform(0.5, 1.5, size=len(high_idx)))
        a_new[low_idx] *= (1 + strength * rng.uniform(0.5, 1.5, size=len(low_idx)))
        
        a_new = np.clip(a_new, 0.0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            return a
        a_new = a_new / s_new * S
        return a_new
    else:
        # Deterministic version
        a_new = a.copy()
        n_reduce = max(1, n // 10)
        n_incr = max(1, n // 10)
        
        high_idx = np.argsort(contrib)[-n_reduce:]
        low_idx = np.argsort(contrib)[:n_incr]
        
        total_reduce = 0
        for i in high_idx:
            reduction = a_new[i] * strength
            a_new[i] -= reduction
            total_reduce += reduction
        
        a_new = np.clip(a_new, 0.0, None)
        
        # Distribute to low-contributing elements
        for i in low_idx:
            a_new[i] += total_reduce / len(low_idx)
        
        a_new = np.clip(a_new, 0.0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            return a
        return a_new / s_new * S

def local_search(a_init, time_budget, rng):
    """Simulated annealing with peak-directed perturbations."""
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9997
    
    while time.monotonic() - t0 < time_budget:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.3:
            # Peak-directed reduction
            c = fft_convolve(current, current)
            k_star = np.argmax(c)
            # Reduce elements that contribute to the peak
            contrib_mask = np.zeros(n, dtype=bool)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n and current[i] * current[j] > 0.1 * np.max(current) * np.max(current):
                    contrib_mask[i] = True
            
            if np.any(contrib_mask):
                n_reduce = max(1, int(np.sum(contrib_mask) * 0.5))
                indices = np.where(contrib_mask)[0]
                chosen = rng.choice(indices, size=min(n_reduce, len(indices)), replace=False)
                candidate[chosen] *= np.exp(rng.normal(-0.1, 0.03, size=len(chosen)))
                # Boost non-contributing
                non_contrib = np.where(~contrib_mask)[0]
                if len(non_contrib) > 0:
                    n_boost = max(1, min(5, len(non_contrib)))
                    boost_idx = rng.choice(non_contrib, size=n_boost, replace=False)
                    candidate[boost_idx] *= np.exp(rng.normal(0.1, 0.03, size=n_boost))
        
        elif r < 0.5:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.08, size=width))
        
        elif r < 0.7:
            # Two-block: reduce one, boost another
            w1 = max(1, int(rng.integers(1, max(2, n // 20))))
            w2 = max(1, int(rng.integers(1, max(2, n // 20))))
            s1 = int(rng.integers(0, n - w1))
            s2 = int(rng.integers(0, n - w2))
            candidate[s1:s1+w1] *= np.exp(rng.normal(-0.06, 0.02, size=w1))
            candidate[s2:s2+w2] *= np.exp(rng.normal(0.06, 0.02, size=w2))
        
        elif r < 0.85:
            # Small global noise
            candidate *= np.exp(rng.normal(0, 0.01, size=n))
        
        else:
            # Symmetrization attempt (partial)
            asym = current - current[::-1]
            candidate = current - 0.1 * asym + 0.005 * rng.standard_normal(n)
        
        candidate = np.clip(candidate, 0.0, None)
        S = np.sum(current)
        s_cand = np.sum(candidate)
        if s_cand < 1e-10:
            continue
        candidate = candidate / s_cand * S
        
        new_score = local_score(candidate)
        delta = new_score - current_score
        
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        
        temp *= cooling
    
    return best, best_score
