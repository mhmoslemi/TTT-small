Optimization of non-negative step functions to minimize peak linear autocorrelation.
Uses soft-max gradient descent with Nesterov momentum, simplex projection,
and multi-phase search with diverse initializations and length exploration.

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

    """Project onto the simplex {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v

Let me also think about the "peak shaving" approach more carefully. The idea is:

1. Find the lag k* where c is maximum.
2. The contribution to c_{k*} from position i is `a_i * a_{k*-i}`.
3. We want to reduce `c_{k*}` without too much increasing other lags.
4. A position i that has `a_{k*-i}` large (i.e., the partner position for the peak lag is also large) is a good candidate for reduction.
5. But reducing `a_i` also reduces c_k for other k where i contributes.

The "smart" peak shaving:
- Compute the "importance" of each position for the peak lag: `imp_i = a_i * a_{k*-i}`
- Compute the "collateral damage" of reducing position i: how much would other lags increase?
  - This is tricky to compute exactly, but we can approximate: reducing `a_i` by `da` reduces `c_k` by `da * a_{k-i}` for all k.
  - The lags that would be most affected are those where `a_{k-i}` is large.
  - Net effect on the max: we reduce the peak by `imp_i * da` but might increase other lags.

"""
Multi-strategy optimization for minimizing peak linear autocorrelation of non-negative
step functions. Uses soft-max gradient descent with Nesterov momentum on the non-negative
simplex, peak-shaving refinement, FFT-phase initialization, and multi-length search.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence (scale-invariant)."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    tau = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - tau, 0.0)


def compute_grad(a, w):
    """
    Compute gradient of sum_k w_k * c_k where c = conv(a, a) (linear).
    Returns grad[j] = 2 * sum_m a[m] * w[j+m] (with w out of range = 0).
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    a_fft = np.fft.rfft(a, N)
    w_fft = np.fft.rfft(w, N)
    x = np.fft.irfft(np.conj(a_fft) * w_fft, N)[:n]
    return 2.0 * x


def soft_max_descent(a_init, time_limit, T_start=None, T_end=None, lr=0.3, mu=0.85, verbose=False):
    """Soft-max gradient descent with heavy-ball momentum on the non-negative simplex."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = np.sum(a)
    a = a / s if s > 1e-10 else np.ones(n) / n

    if T_start is None:
        T_start = 2.0 / n
    if T_end is None:
        T_end = 0.05 / n

    v = np.zeros(n)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    it = 0
    patience = 0
    patience_limit = 2000

    while time.monotonic() - t0 < time_limit:
        remaining_frac = (time_limit - (time.monotonic() - t0)) / time_limit
        frac = 1.0 - remaining_frac
        T = T_start * (T_end / T_start) ** (frac ** 0.5)

        c = fft_convolve(a, a)
        c_max = np.max(c)
        w = np.exp((c - c_max) / T)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum

        g = compute_grad(a, w)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

            a = a_new

            patience = 0

def peak_shaving(a_init, time_limit, eta_init=0.3, verbose=False):
    """
    Iterative peak shaving: reduce positions that contribute most to the peak lag,
    while trying not to increase other lags too much.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = np.sum(a)
    a = a / s if s > 1e-10 else np.ones(n) / n

    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    it = 0
    eta = eta_init

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        c_max = np.max(c)
        # Find active peak lags (within 1% of max)
        threshold = c_max * 0.99
        active = c >= threshold
        n_active = np.sum(active)
        if n_active == 0:
            break

        # Compute contribution of each position to active peak lags
        # For position i, its contribution to lag k is a[i] * a[k-i]
        # Total contribution to active lags:
        contrib = np.zeros(n)
        for k in np.where(active)[0]:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                j = k - i
                contrib[i] += a[i] * a[j]

        # Also compute "collateral": how much does position i contribute to NON-peak lags?
        # (We want to reduce positions that contribute a lot to peak but little to others)
        # For simplicity, just use the peak contribution as the pressure

        # Reduction factor: reduce high-contribution positions
        max_contrib = np.max(contrib)
        if max_contrib < 1e-15:
            break
        pressure = contrib / max_contrib
        # Exponential reduction
        a_new = a * np.exp(-eta * pressure)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / np.sum(a_new)

        score = local_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            # Slowly reduce eta
            eta *= 0.999
        else:
            # Revert and reduce step
            eta *= 0.9

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    peak_shave iter={it}, score={local_score(a):.6f}, best={best_score:.6f}, eta={eta:.4f}, t={elapsed:.1f}s")
        it += 1

    return best_a, best_score


def fft_phase_init(n, rng, n_iter=120):
    """Gerchberg-Saxton style initialization with non-negativity."""
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1

    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-14] = 1e-14
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def water_filling_init(n, rng, n_iter=80):
    """Water-filling: iteratively flatten the convolution."""
    a = np.ones(n) / n
    for it in range(n_iter):
        c = fft_convolve(a, a)
        c_max = np.max(c)
        c_mean = np.mean(c)
        # Scale: positions contributing to peaks get reduced
        # Compute per-position peak contribution
        contrib = np.zeros(n)
        # Use weighted sum of all lags (weighted by excess over mean)
        excess = np.maximum(c - c_mean, 0)
        N_fft = 1
        while N_fft < 3 * n:
            N_fft <<= 1
        a_fft = np.fft.rfft(a, N_fft)
        e_fft = np.fft.rfft(excess, N_fft)
        contrib = np.fft.irfft(np.conj(a_fft) * e_fft, N_fft)[:n]
        max_c = np.max(contrib)
        if max_c < 1e-15:
            break
        a = a * np.exp(-0.3 * contrib / max_c)
        a = np.maximum(a, 1e-10)
        a = a / np.sum(a)
    return a

def local_search_gradient(a_init, time_limit, rng, step_size=0.02, verbose=False):

            amp = 0.3 / (k + 1)
            phase = rng.uniform(0, 2 * np.pi)
            a_mt += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a_mt = np.clip(a_mt, 0.01, None)
        inits['multitone'] = a_mt / np.sum(a_mt)

        # Random smooth
        a_rand = np.exp(-0.3 * rng.standard_normal(n) ** 2 + 0.1 * rng.standard_normal(n))
        inits['rand'] = a_rand / np.sum(a_rand)

        for name, a in inits.items():
            score = local_score(a)
            if time.monotonic() > deadline - 300:
                break
            print(f"  n={n:4d}, {name:10s}: score={score:.6f}", flush=True)
            candidates.append((score, n, a.copy(), name))

    # Sort by score
    candidates.sort(key=lambda x: x[0])
    top5 = [(f"{c[0]:.4f}", c[1], c[3]) for c in candidates[:5]]
    print(f"\n  Top 5: {top5}", flush=True)

    # Phase 2: Gradient optimization on top candidates
    print("\n=== Phase 2: Main optimization ===", flush=True)
    time_left = deadline - time.monotonic() - 200  # reserve for polish
    n_opt = min(6, len(candidates))
    # Allocate more time to better candidates
    total_weights = np.zeros(n_opt)
    for i in range(n_opt):
        # Weight inversely proportional to score (better candidates get more time)
        total_weights[i] = 1.0 / (candidates[i][0] + 0.1)
    total_weights = total_weights / np.sum(total_weights)

    for i in range(n_opt):
        if time.monotonic() > deadline - 100:
            break
        score0, n, a, name = candidates[i]
        t_alloc = max(20.0, time_left * total_weights[i] * 0.8)
        t_alloc = min(t_alloc, deadline - time.monotonic() - 60)
        if t_alloc < 10:
            break

        print(f"\n  [{i+1}/{n_opt}] n={n}, init={name}, score0={score0:.6f}, t_alloc={t_alloc:.0f}s", flush=True)

        # Run gradient refinement
        a_opt, s_opt = gradient_refine(a, t_alloc, rng)
        print(f"    After gradient: score={s_opt:.6f}", flush=True)

        # Peak shaving
        if t_alloc > 30:
            t_peak = min(30.0, (deadline - time.monotonic() - 60) * 0.3)
            if t_peak > 5:
                a_shaved, s_shaved = peak_shaving(a_opt, t_peak, eta_init=0.2)
                if s_shaved < s_opt:
                    print(f"    After peak shaving: score={s_shaved:.6f}", flush=True)
                    s_opt = s_shaved
                    a_opt = a_shaved

        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***", flush=True)

    # Phase 3: Coarse-to-fine upsampling
    print(f"\n=== Phase 3: Coarse-to-fine ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)
        for target_n in [512, 1024, 1536, 2048]:
            if time.monotonic() > deadline - 100:
                break
            if n_best >= target_n:
                continue
            # Upsample
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, target_n)
            a_up = np.interp(x_new, x_old, best_seq)
            a_up += 0.01 * rng.standard_normal(target_n)
            a_up = np.clip(a_up, 0.001, None)
            a_up = a_up / np.sum(a_up)

            t_alloc = min(120.0, max(20.0, (deadline - time.monotonic() - 80) * 0.3))
            if t_alloc < 15:
                break
            print(f"  Upsample to n={target_n}, t_alloc={t_alloc:.0f}s", flush=True)
            a_opt, s_opt = gradient_refine(a_up, t_alloc, rng)
            print(f"    -> score={s_opt:.6f}", flush=True)
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} (n={target_n}) ***", flush=True)

    # Phase 4: Final polish with local search
    print(f"\n=== Phase 4: Local search polish ===", flush=True)
    if best_seq is not None:
        polish_time = min(150.0, max(15.0, deadline - time.monotonic() - 30))
        print(f"  Polishing n={len(best_seq)}, score={best_score:.6f}, time={polish_time:.0f}s", flush=True)
        a_polished, s_polished = local_search_gradient(best_seq, polish_time, rng, step_size=0.03)
        if s_polished < best_score:
            best_score = s_polished
            best_seq = a_polished.copy()
            print(f"  After local search: {best_score:.6f}", flush=True)

        # One more gradient pass
        remaining = deadline - time.monotonic() - 20
        if remaining > 20:
            t_gr = min(60.0, remaining * 0.5)
            if t_gr > 10:
                a_final, s_final = soft_max_descent(best_seq, t_gr, T_start=0.1/len(best_seq), T_end=0.01/len(best_seq), lr=0.1, mu=0.95)
                if s_final < best_score:
                    best_score = s_final
                    best_seq = a_final.copy()
                    print(f"  After final gradient: {best_score:.6f}", flush=True)

    # Phase 5: If time remains, try alternative approaches
    remaining = deadline - time.monotonic() - 25
    if remaining > 40 and best_seq is not None:
        print(f"\n=== Phase 5: Alternative approaches ===", flush=True)
        n_best = len(best_seq)

        # Try a completely different initialization at the same length
        a_alt = fft_phase_init(n_best, rng, n_iter=200)
        t_alloc = min(60.0, remaining * 0.6)
        if t_alloc > 15:
            a_opt, s_opt = gradient_refine(a_alt, t_alloc, rng)
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"  Alternative init: {best_score:.6f}", flush=True)

        # Try anti-peak construction
        remaining2 = deadline - time.monotonic() - 20
        if remaining2 > 30:
            # Start from a "spiky" sequence and flatten
            n_try = min(512, max(256, n_best))
            a_spike = np.zeros(n_try)
            a_spike[n_try // 2] = 1.0
            # Spread it out
            kernel = np.exp(-0.5 * (np.arange(n_try) - n_try // 2) ** 2 / (n_try / 8) ** 2)
            a_flat = kernel / np.sum(kernel)
            t_alloc2 = max(10.0, remaining2 * 0.7)
            a_opt, s_opt = gradient_refine(a_flat, t_alloc2, rng)
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"  Anti-peak: {best_score:.6f}", flush=True)

    # Final verification
    print(f"\n{'=' * 50}", flush=True)
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===", flush=True)
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
