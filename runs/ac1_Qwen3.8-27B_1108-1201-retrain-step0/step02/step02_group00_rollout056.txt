"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Combined approach:
1. Spectral equalization: flatten the power spectrum to reduce convolution peaks.
2. FISTA with adaptive log-sum-exp: smooth convex-like optimization.
3. Mirror descent in log-space: different geometry for exploration.
4. Nesterov top-k subgradient: precise polishing of the actual max.
5. Multi-restart with perturbation: escape local minima.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def spectral_equalize_step(a, L=None, alpha=0.3):
    """One step of spectral equalization to flatten the power spectrum."""
    n = len(a)
    if L is None:
        L = 1 << (2 * n - 1).bit_length()
    a_padded = np.zeros(L)
    a_padded[:n] = a
    A = fft(a_padded)
    power = np.abs(A) ** 2
    dc_power = power[0]
    total_power = power.sum()
    excess = total_power - dc_power
    if excess < 1e-30:
        return a.copy()
    target = excess / (L - 1)
    # Build new power spectrum
    new_power = power.copy()
    new_power[0] = dc_power  # keep DC fixed
    # For non-DC bins, pull towards target
    mask = power[1:] > 1e-30
    new_power[1:][mask] = (1 - alpha) * power[1:][mask] + alpha * target
    # Ensure non-negative (should be by construction)
    new_power = np.maximum(new_power, 0)
    # Magnitude update, preserve phase
    A_new = A.copy()
    phase = np.angle(A)
    A_new[0] = A[0]  # keep DC
    A_new[1:] = np.sqrt(new_power[1:]) * np.exp(1j * phase[1:])
    # Enforce Hermitian symmetry for real output
    A_new[L // 2] = np.abs(A_new[L // 2])  # Nyquist (if L is even)
    # Actually, for real input, A must satisfy A[k] = conj(A[L-k])
    # Let's just take the real part of the inverse
    a_new = np.real(ifft(A_new))[:n]
    return a_new


def run_spectral_equalization(a_init, n, S, max_time, rng, alpha=0.3, L=None):
    """Iterative spectral equalization with projection."""
    if L is None:
        L = 1 << (2 * n - 1).bit_length()
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Spectral equalization step
        a_new = spectral_equalize_step(a, L, alpha)
        # Project to simplex
        a_new = project_simplex(a_new, S)
        
        # Evaluate
        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a_new.copy()
        
        a = a_new
        
        # Periodic perturbation
        if it % 100 == 0 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
    
    return best_local, best_score


def run_fista_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=500.0):
    """FISTA with log-sum-exp smoothing."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    fsize = 1 << (2 * n - 1).bit_length()
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(0.001 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        # Log-sum-exp gradient
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total
        
        # Gradient via FFT
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (beta * 2.0)
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return best_local, best_score


def run_logspace_mirror(a_init, n, S, max_time, rng, eta_init=0.1):
    """Mirror descent in log-space (a = softmax(b) * S)."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    
    # Initialize b in log-space
    a_clipped = np.maximum(a, 1e-10)
    b = np.log(a_clipped / a_clipped.sum())  # log of normalized
    
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    prev_best = np.inf
    stall = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Recover a from b
        b_shifted = b - b.max()
        a_raw = np.exp(b_shifted)
        a = a_raw / a_raw.sum() * S
        
        # Compute convolution and gradient
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 5000:
            break
        
        # Top-k subgradient in log-space
        top_k = min(32, max(3, it // 100))
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Gradient in log-space: grad_b = grad_a / a
        eta = eta_init / (1.0 + 0.001 * it)
        grad_b = grad_a / np.maximum(a, 1e-10)
        gmax = np.abs(grad_b).max()
        if gmax < 1e-14:
            break
        
        b = b - eta * grad_b / gmax
        
        # Stabilize
        if it % 500 == 0:
            b = b - b.mean()  # remove mean (invariance)
    
    return best_local, best_score

        top_vals = c[top_idx]
        weights = np.exp(-np.arange(top_k) * 0.1)  # decay

        total = weights.sum()
        if total < 1e-30:

Actually, let me reconsider. If |A[k]|^2 = P0 for k=0 and P1 for all k≠0, then:
c[m] = (1/L) * sum_k P_k * exp(2πikm/L)

For m=0: sum_{k≠0} exp(0) = L-1, so c[0] = (1/L)(P0 + P1(L-1))
For m≠0 (mod L): sum_{k≠0} exp(2πikm/L) = -1 (since sum_{all k} = 0), so c[m] = (1/L)(P0 - P1)

So if P0 > P1, we get c[0] > c[m] for m≠0. The peak is at m=0.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def spectral_equalize_step(a, L, alpha):
    """One step of spectral equalization to flatten the non-DC power spectrum."""
    n = len(a)
    a_padded = np.zeros(L)
    a_padded[:n] = a
    A = fft(a_padded)
    power = np.abs(A) ** 2
    dc_power = power[0]
    total_power = power.sum()
    excess = total_power - dc_power
    if excess < 1e-30:
        return a.copy()
    target = excess / (L - 1)
    # For non-DC bins, pull towards target
    mask = power[1:] > 1e-30
    new_power = power.copy()
    new_power[0] = dc_power
    new_power[1:][mask] = (1.0 - alpha) * power[1:][mask] + alpha * target
    new_power = np.maximum(new_power, 0.0)
    # Preserve phase, update magnitude
    phase = np.angle(A)
    A_new = A.copy()
    A_new[0] = A[0]
    nonzero = new_power[1:] > 0
    A_new[1:][nonzero] = np.sqrt(new_power[1:][nonzero]) * np.exp(1j * phase[1:][nonzero])
    A_new[1:][~nonzero] = 0
    # Ensure Hermitian symmetry
    for k in range(1, L // 2 + 1):
        A_new[L - k] = np.conj(A_new[k])
    if L % 2 == 0:
        A_new[L // 2] = np.abs(A_new[L // 2])
    a_new = np.real(ifft(A_new))[:n]
    return a_new


def run_spectral_eq(a_init, n, S, max_time, rng, alpha=0.25, L=None):
    """Iterative spectral equalization with simplex projection."""
    if L is None:
        L = 1 << (2 * n - 1).bit_length()
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    prev_best = best_score

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive alpha: start aggressive, then gentle
        cur_alpha = alpha * max(0.3, 1.0 - 0.0005 * it)
        a_new = spectral_equalize_step(a, L, cur_alpha)
        a_new = project_simplex(a_new, S)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a_new.copy()

        a = a_new

        # Periodic perturbation to escape local minima
        if it % 200 == 0 and rng.random() < 0.25:
            noise_scale = 0.02 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

        # Early termination if stuck
        if best_score < prev_best:
            prev_best = best_score
        elif it % 1000 == 0 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.03 * S / np.sqrt(n), n), S)

    return best_local, best_score


def run_fista_lse(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=1000.0):
    """FISTA with log-sum-exp smoothing and FFT-based gradients."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    fsize = 1 << (2 * n - 1).bit_length()
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Exponential beta growth
        beta = min(beta_max, beta_init * np.exp(0.0008 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Log-sum-exp gradient
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size with mild decay
        step = 1.0 / (beta * 2.0 * (1.0 + 0.0003 * it))
        step = max(step, 1e-8)
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Sparse perturbation
        if it % 5000 == 0 and rng.random() < 0.12:
            y = project_simplex(y + rng.normal(0, 0.008 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def run_logspace_mirror(a_init, n, S, max_time, rng, eta_init=0.08):
    """Mirror descent in log-space with top-k subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n

    a_clipped = np.maximum(a, 1e-10)
    b = np.log(a_clipped) - np.log(a_clipped.sum())

    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stall_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Recover a from log-space with softmax normalization
        b_shifted = b - b.max()
        a_raw = np.exp(b_shifted)
        a_sum_raw = a_raw.sum()
        if a_sum_raw < 1e-300:
            b = rng.normal(0, 1, n)
            continue
        a = a_raw / a_sum_raw * S

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 8000:
            break

        # Top-k subgradient
        top_k = min(64, max(3, 3 + it // 80))
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        # Exponentially decaying weights (focus on top)
        decay = np.exp(-np.arange(top_k, dtype=np.float64) * 0.15)
        weights = decay * top_vals
        total = weights.sum()
        if total < 1e-30:
            break
        weights = weights / total

        w_full = np.zeros(2 * n - 1)
        for i in range(top_k):
            w_full[int(top_idx[i])] = weights[i]

        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Chain rule: grad_b = grad_a / a
        eta = eta_init / (1.0 + 0.001 * it)
        grad_b = grad_a / np.maximum(a, 1e-10)
        gmax = np.abs(grad_b).max()
        if gmax < 1e-14:
            break

        b = b - eta * grad_b / gmax

        # Re-center for numerical stability
        if it % 300 == 0:
            b = b - b.mean()

        # Perturbation
        if it % 3000 == 0 and rng.random() < 0.15:
            b = b + rng.normal(0, 0.05, n)

    return best_local, best_score


def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=256):
    """Nesterov-accelerated top-k subgradient with adaptive step."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    fsize = 1 << (2 * n - 1).bit_length()
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 40)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Exponentially weighted top-k
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        decay = np.exp(-np.arange(top_k, dtype=np.float64) * 0.1)
        weights = decay * top_vals
        total = weights.sum()
        if total < 1e-30:
            break
        weights = weights / total

        w_full = np.zeros(2 * n - 1)
        for i in range(top_k):
            w_full[int(top_idx[i])] = weights[i]

        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step: O(1/sqrt(t)) schedule
        step = 2.0 * S / (n * np.sqrt(it + 1.0))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Perturbation
        if it % 6000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.006 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def make_piecewise_init(n, k, rng, S):
    """Create a piecewise-constant initialization with k segments."""
    # Random segment boundaries
    boundaries = np.sort(rng.choice(np.arange(1, n), size=k - 1, replace=False))
    boundaries = np.concatenate([[0], boundaries, [n]])
    lengths = np.diff(boundaries)

    # Random heights (U-shaped bias for diversity)
    if rng.random() < 0.3:
        # U-shaped
        t = np.linspace(0, 1, k)
        heights = 1.0 + 0.5 * (2 * t - 1) ** 2
    elif rng.random() < 0.5:
        # Random
        heights = rng.random(k) + 0.1
    else:
        # Cosine
        t = np.linspace(0, 1, k)
        heights = 1.0 + 0.4 * np.cos(np.pi * t)

    a = np.zeros(n)
    for i in range(k):
        a[boundaries[i]:boundaries[i + 1]] = heights[i]

    return project_simplex(a, S)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

        update_best(a_alt_opt)
        print("  Alt n=%d: obj=%.6f" % (n_alt, global_best_obj))

    # === Final verification ===
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * 1.0
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Final: length=%d, sum=%.4f" % (len(global_a), global_a.sum()))
    return [float(x) for x in global_a]
