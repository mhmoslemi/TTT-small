import numpy as np

def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def smooth_max_value_and_weights(conv, tau):
    """Compute log-sum-exp smoothed max and weights."""
    m = np.max(conv)
    shifted = (conv - m) / tau
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    w = exp_vals / total
    value = m + tau * np.log(total)
    return value, w


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)

    s = np.sum(e)
    if s < 1e-300:

    return e / s


def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    r = rho_candidates[-1]
    theta_val = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta_val, 0.0)

    a_best = softmax_normalize(theta) * S
    best_score = score_fast(a_best)

    tau_start = tau_mult * 5.0 / n
    tau_end = tau_mult * 0.01 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Adaptive temperature: start large, end small
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.6)

        a = softmax_normalize(theta) * S
        conv = fft_convolve(a, a)
        value, w = smooth_max_value_and_weights(conv, tau)

        # Gradient w.r.t. a
        grad_a = compute_grad_smooth_max(a, w, tau)

        # Gradient w.r.t. theta
        p = a / S
        inner = np.dot(p, grad_a)
        grad_theta = S * p * (grad_a - inner)

        step += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad_theta
        v_adam = beta2 * v_adam + (1 - beta2) * (grad_theta ** 2)
        m_hat = m_adam / (1.0 - beta1 ** step)
        v_hat = v_adam / (1.0 - beta2 ** step)
        lr_eff = lr / (1.0 + 0.05 * elapsed_frac)
        theta = theta - lr_eff * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic evaluation
        if step % 50 == 0:
            a_check = softmax_normalize(theta) * S
            sc = score_fast(a_check)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()
                a_best = a_check.copy()

        if step % 2000 == 0:
            a_check = softmax_normalize(theta) * S
            sc = score_fast(a_check)
            print(f"    Adam: step={step}, score={sc:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")

    return best_theta, a_best, best_score


def optimize_active_set(n, a_init, time_limit, rng, n_peaks=10):
    """
    Active-set optimization: focus gradient only on the top peaks.
    Uses projected gradient descent on the simplex.
    """
    a = a_init.copy()
    a = np.clip(a, 1e-10, None)
    a = a / np.sum(a)
    n = len(a)
    S = np.sqrt(2.0 * n)
    a_scaled = a * S

    best_a = a.copy()
    best_score = score_fast(a_scaled)
    t0 = time.monotonic()
    lr = 1.0 / (n + 1)
    momentum = np.zeros(n)
    mu = 0.9

    step = 0
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        lr_cur = lr * (1.0 + 0.5 * elapsed_frac)

        conv = fft_convolve(a_scaled, a_scaled)
        max_c = np.max(conv)

        # Find top peaks
        if len(conv) <= n_peaks:
            peaks = np.argsort(conv)[::-1]
        else:
            top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
            peaks = top_idx[np.argsort(conv[top_idx])[::-1]]

        # Create weights for active set
        peak_vals = conv[peaks]
        # Soft weights based on how close to max
        weights = np.exp((peak_vals - max_c) / (max_c * 0.1 + 1e-10))
        weights = weights / np.sum(weights)

        w_full = np.zeros(len(conv))
        w_full[peaks] = weights

        # Gradient
        grad_a = compute_grad_smooth_max(a_scaled, w_full, 1.0)

        # Momentum
        momentum = mu * momentum + (1 - mu) * grad_a

        # Projected step
        candidate = a - lr_cur * momentum / S
        candidate = project_simplex(candidate)

        # Evaluate
        a_test = candidate * S
        sc = score_fast(a_test)
        if sc <= score_fast(a_scaled) * (1 + 1e-8):
            a = candidate
            a_scaled = a_test
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        step += 1
        if step % 1000 == 0:
            print(f"    ActiveSet: step={step}, score={sc:.6f}, best={best_score:.6f}, "
                  f"t={time.monotonic()-t0:.1f}s")

    return best_a * S, best_score


def coordinate_descent(a_init, time_limit, rng, n_trials_per_iter=50):
    """
    Aggressive coordinate descent with simulated annealing.
    Works on normalized sequences (sum=1).
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    n = len(a)

    S = np.sqrt(2.0 * n)
    current_score = score_fast(a * S)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.05
    patience = 0
    max_patience = 5000

    iter_count = 0
    while time.monotonic() - t0 < time_limit and patience < max_patience:
        iter_count += 1
        for trial in range(n_trials_per_iter):
            r = rng.random()

            if r < 0.35:
                # Pair transfer
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                frac = rng.uniform(0.02, 0.1)
                amt = a[i] * frac
                candidate = a.copy()
                candidate[i] = max(0.0, a[i] - amt)
                candidate[j] = a[j] + amt
                cs = np.sum(candidate)
                if cs < 1e-10:
                    continue
                candidate = candidate / cs

            elif r < 0.55:
                # Single element scale
                j = rng.integers(n)
                factor = np.exp(rng.normal(0, 0.08))
                candidate = a.copy()
                candidate[j] *= factor
                candidate = np.clip(candidate, 0.0, None)
                cs = np.sum(candidate)
                if cs < 1e-10:
                    continue
                candidate = candidate / cs

            elif r < 0.75:
                # Block perturbation
                width = max(2, int(rng.integers(2, max(3, n // 16))))
                start = int(rng.integers(0, max(1, n - width)))
                candidate = a.copy()
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.05, size=width))
                candidate = np.clip(candidate, 0.0, None)
                cs = np.sum(candidate)
                if cs < 1e-10:
                    continue
                candidate = candidate / cs

            elif r < 0.9:
                # Peak-targeted reduction
                conv = fft_convolve(a, a)
                peak_idx = int(np.argmax(conv))
                # Find top contributors to this peak
                contribs = np.zeros(n)
                for i in range(n):
                    j = peak_idx - i
                    if 0 <= j < 2 * n - 1:
                        # contribution of a[i] to conv[peak_idx]
                        # a[i] * a[peak_idx - i] if valid
                        if 0 <= peak_idx - i < n:
                            contribs[i] = a[i] * a[peak_idx - i]
                        if 0 <= i < n and 0 <= peak_idx - i < n:
                            pass  # already counted
                # Also: a[peak_idx - i] contributes when i is the second index
                for j in range(n):
                    i = peak_idx - j
                    if 0 <= i < n:
                        contribs[j] += a[j] * a[i]
                top_k = max(3, n // 32)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate = a.copy()
                reduction = rng.uniform(0.05, 0.15)
                candidate[top_idx] *= (1.0 - reduction)
                # Redistribute
                total_removed = np.sum(a[top_idx] - candidate[top_idx])
                others = np.setdiff1d(np.arange(n), top_idx)
                if len(others) > 0:
                    n_add = min(10, len(others))
                    add_idx = rng.choice(others, size=n_add, replace=False)
                    candidate[add_idx] += total_removed / n_add
                candidate = np.clip(candidate, 0.0, None)
                cs = np.sum(candidate)
                if cs < 1e-10:
                    continue
                candidate = candidate / cs

            else:
                # Random restart from current (large perturbation)
                candidate = a.copy()
                frac = rng.uniform(0.1, 0.3)
                mask = rng.random(n) < frac
                candidate[mask] = rng.uniform(0.01, 2.0, size=np.sum(mask))
                candidate = np.clip(candidate, 0.0, None)
                cs = np.sum(candidate)
                if cs < 1e-10:
                    continue
                candidate = candidate / cs

            new_score = score_fast(candidate * S)

            # Simulated annealing acceptance
            if new_score <= current_score:
                a = candidate
                current_score = new_score
                patience = 0
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
            else:
                # Accept with probability exp(-delta / temperature)
                delta = new_score - current_score
                if temperature > 0.001 and rng.random() < np.exp(-delta / temperature):
                    a = candidate
                    current_score = new_score
                else:
                    patience += 1

        # Anneal temperature
        if iter_count % 10 == 0:
            temperature *= 0.95

    return best_a * S, best_score


def optimize_projected_nesterov(n, a_init, time_limit, rng):
    """
    Nesterov-accelerated projected gradient descent on the simplex.
    Directly minimizes max self-convolution.
    """
    a = a_init.copy()
    a = np.clip(a, 1e-10, None)
    a = a / np.sum(a)
    n = len(a)
    S = np.sqrt(2.0 * n)

    a_y = a.copy()
    a_prev = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_val = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    step = 0

    tau_start = 10.0 / n
    tau_end = 0.05 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)

        t_prev = t_mom
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        beta = (t_prev - 1.0) / t_mom

        # Nesterov extrapolation
        y = a_y + beta * (a_y - a_prev)
        y = project_simplex(y)

        # Compute smooth max gradient
        conv = fft_convolve(y, y)
        value, w = smooth_max_value_and_weights(conv, tau)
        grad = compute_grad_smooth_max(y, w, tau)
        grad = grad / np.sqrt(n)  # normalize

        # Learning rate with backtracking
        lr = 1.0 / np.sqrt(step + 1)
        candidate = y - lr * grad
        candidate = project_simplex(candidate)

        a_prev = a_y
        a_y = candidate
        step += 1

        # Evaluate
        actual_val = np.max(fft_convolve(a_y, a_y))
        if actual_val < best_val:
            best_val = actual_val
            best_a = a_y.copy()

        if step % 2000 == 0:
            sc = 2.0 * n * best_val
            print(f"    PGD: step={step}, score={sc:.6f}, t={time.monotonic()-t0:.1f}s")

    return best_a * S, 2.0 * n * best_val


def try_rescale(a, S):
    """Try to improve by rescaling (shouldn't change score but verify)."""
    a_new = a / np.sum(a) * S
    return score_fast(a_new)


def trim_sequence(a, min_val=0.001):
    """Trim leading/trailing elements below threshold."""
    a = a.copy()
    # Find first and last significant elements
    significant = np.where(a > min_val * np.max(a))[0]
    if len(significant) == 0:
        return a
    start = significant[0]
    end = significant[-1] + 1
    return a[start:end]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for optimal non-negative sequence minimizing
    2n * max(convolve(a,a)) / (sum a)^2.
    """
    t_global = time.monotonic()
    time_budget = max(30.0, float(budget_s) - 60.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            parent = np.asarray(height_sequence_1, dtype=np.float64)
            parent = np.clip(parent, 0.0, 1000.0)
            if len(parent) > 0 and np.sum(parent) > 0.01:
                s0 = score_fast(parent)
                if np.isfinite(s0):
                    start_seq = parent.copy()
                    best_score = s0
                    best_seq = parent.copy()
                    print(f"Starting from provided sequence: n={len(parent)}, score={s0:.6f}")
        except Exception as e:
            print(f"Error loading provided sequence: {e}")

    # Phase 1: Optimize from provided sequence (or good init) with multiple algorithms
    print("\n=== Phase 1: Primary optimization ===")
    phase1_budget = time_budget * 0.45

    if start_seq is not None:
        n_start = len(start_seq)
        a_norm = start_seq / np.sum(start_seq)

        # 1a: Adam on provided sequence
        t_alloc = phase1_budget * 0.4
        print(f"  Adam on provided sequence (n={n_start}), budget={t_alloc:.0f}s")
        theta_best, a_best, sc = optimize_adam(
            n_start, a_norm, t_alloc, rng, lr_init=0.3, tau_mult=1.0)
        print(f"  Adam result: score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

        # 1b: Active set on provided
        if time.monotonic() < deadline - 120:
            t_alloc = min(phase1_budget * 0.25, deadline - 120 - time.monotonic())
            if t_alloc > 10:
                print(f"  ActiveSet on provided (n={n_start}), budget={t_alloc:.0f}s")
                a_best, sc = optimize_active_set(
                    n_start, a_norm, t_alloc, rng, n_peaks=15)
                print(f"  ActiveSet result: score={sc:.6f}")
                if sc < best_score:
                    best_score = sc
                    best_seq = a_best.copy()
                    print(f"    *** New best: {best_score:.6f} ***")

    # Phase 2: Try different sequence lengths and initializations
    print(f"\n=== Phase 2: Length exploration ===")
    phase2_budget = time_budget * 0.30
    if time.monotonic() < deadline - 120:
        lengths_to_try = []
        # Focus on the length that worked before and nearby
        if start_seq is not None:
            n0 = len(start_seq)
            lengths_to_try = [n0, max(64, n0 // 2), min(4096, n0 * 2)]
        else:
            lengths_to_try = [256, 512, 1024, 2048]

        # Remove duplicates and sort
        lengths_to_try = sorted(set([l for l in lengths_to_try if 64 <= l <= 4096]))

        inits_to_try = ['uniform_noise', 'quadratic_dip', 'edges_high',
                        'sinusoidal', 'two_peaks', 'gaussian_mixture']

        t_per_config = max(15.0, phase2_budget / (len(lengths_to_try) * 2))

        config_count = 0
        for n in lengths_to_try:
            if time.monotonic() > deadline - 120:
                break
            for init_style in inits_to_try[:3]:  # Try 3 inits per length
                if time.monotonic() > deadline - 120:
                    break
                t_alloc = min(t_per_config, deadline - 120 - time.monotonic())
                if t_alloc < 10:
                    break

                a0 = ramp_init(n, rng, style=init_style)
                a0 = a0 / np.sum(a0)

                print(f"  n={n}, init='{init_style}', budget={t_alloc:.0f}s")
                theta_best, a_best, sc = optimize_adam(
                    n, a0, t_alloc, rng, lr_init=0.4, tau_mult=1.0)
                print(f"    score={sc:.6f}")

                if sc < best_score:
                    best_score = sc
                    best_seq = a_best.copy()
                    print(f"    *** New best: {best_score:.6f} ***")
                config_count += 1

    # Phase 3: Coordinate descent polish
    print(f"\n=== Phase 3: Coordinate descent polish ===")
    print(f"  Best before polish: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 60:
        polish_time = min(400.0, max(60.0, deadline - time.monotonic() - 60))
        print(f"  Polish budget: {polish_time:.0f}s")

        a_norm = best_seq / np.sum(best_seq)
        polished, polish_score = coordinate_descent(
            a_norm, polish_time, rng, n_trials_per_iter=100)
        print(f"  After CD polish: {polish_score:.6f}")

        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 4: Final targeted optimization with remaining time
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 30
        n_final = len(best_seq)
        a_norm = best_seq / np.sum(best_seq)

        print(f"\n=== Phase 4: Final targeted optimization (n={n_final}) ===")
        print(f"  Budget: {remaining:.0f}s")

        # Try Nesterov PGD
        if remaining > 30:
            t_alloc = min(remaining * 0.5, 200.0)
            a_best, sc = optimize_projected_nesterov(
                n_final, a_norm, t_alloc, rng)
            print(f"  Nesterov PGD: score={sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

        # One more CD round if time
        if time.monotonic() < deadline - 30:
            t_alloc = min(remaining * 0.3, deadline - time.monotonic() - 30)
            if t_alloc > 10:
                a_norm2 = best_seq / np.sum(best_seq)
                polished, sc = coordinate_descent(
                    a_norm2, t_alloc, rng, n_trials_per_iter=150)
                print(f"  Final CD: score={sc:.6f}")
                if sc < best_score:
                    best_score = sc
                    best_seq = polished.copy()
                    print(f"    *** New best: {best_score:.6f} ***")

    # Phase 5: Try spectral init with long optimization on best length
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 30
        n_final = len(best_seq)
        if remaining > 30:
            print(f"\n=== Phase 5: Spectral init + optimization (n={n_final}) ===")
            a0 = spectral_init_v2(n_final, rng, n_iter=60)
            a0 = a0 / np.sum(a0)
            t_alloc = min(remaining * 0.6, deadline - time.monotonic() - 30)
            theta_best, a_best, sc = optimize_adam(
                n_final, a0, t_alloc, rng, lr_init=0.3, tau_mult=0.8)
            print(f"  Spectral+Adam: score={sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Final: try trimming
    if best_seq is not None:
        trimmed = trim_sequence(best_seq, min_val=0.005)
        if len(trimmed) >= 10:
            trim_score = score_fast(trimmed)
            if trim_score < best_score:
                best_score = trim_score
                best_seq = trimmed.copy()
                print(f"\n  After trimming: score={best_score:.6f}")

    # Final verification
    print(f"\n{'='*60}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0}")
    print(f"Total time: {time.monotonic()-t_global:.1f}s")

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Verify with actual evaluator
        try:
            eval_score = evaluate_sequence(final_list)
            print(f"Verified score via evaluate_sequence: {eval_score:.6f}")
            if eval_score != best_score:
                print(f"  (Note: computed {best_score:.6f} vs evaluated {eval_score:.6f})")
                best_score = eval_score
        except Exception as e:
            print(f"  Verification error: {e}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
