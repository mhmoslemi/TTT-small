from scipy.fft import fft, ifft
from scipy.optimize import linprog

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def true_max_and_subgrad(a):

    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()

    tol = 1e-10 * max(1.0, c_max)

    subgrad = np.zeros(n)
    a_rev = a[::-1]  # a_rev[j] = a[n-1-j]

    # a[k-j] is valid if 0 <= k-j < n, i.e., j in [k-n+1, k] intersect [0, n-1]
    # a[j-k] is valid if 0 <= j-k < n, i.e., j in [k, k+n-1] intersect [0, n-1]
    # 
    # Using FFT: sum_{k in K} a[k-j] for fixed j = conv(K_indicator, a)[j + n - 1]... 
    # Actually, let me just compute it directly for small |K|, or use FFT for large |K|.
    
    if len(K) <= 100:

        a_padded = np.zeros(2*n-1)

            # a[k-j] for valid j: j in [max(0,k-n+1), min(k,n-1)]
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            subgrad[lo:hi+1] += a[k - np.arange(lo, hi + 1)]
            # a[j-k] for valid j: j in [max(0,k), min(k+n-1,n-1)]
            lo2 = max(0, k)

            subgrad[lo2:hi2+1] += a[np.arange(lo2, hi2 + 1) - k]
        subgrad /= len(K)

        K_ind = np.zeros(2*n-1)

        # This is the cross-correlation of K_ind and a, which equals conv(K_ind, flip(a))

        # Let m = k-j, so k = m+j: h[j] = sum_{m=0}^{n-1} K_ind[m+j] * a[m]
        # This is the cross-correlation: h = correlate(K_ind, a)
        from scipy.signal import correlate
        h = correlate(K_ind, a, mode='full')
        # h[j + (n-1)] in 'full' mode... actually scipy's correlate is defined as:
        # correlate(a, v)[k] = sum_n a[n+k] * conj(v[n])
        # So correlate(K_ind, a)[j] = sum_n K_ind[n+j] * a[n] = h[j] (what we want)
        # In 'full' mode, the output has length len(K_ind) + len(a) - 1 = 3n-2
        # And it's centered: output[j + (n-1)] corresponds to the "zero lag"

        # output[m] = sum_n a[n] * v[n+m-(len(v)-1)]...

        # Actually, for large |K|, I'll use FFT properly.
        # h[j] = sum_m a[m] * K_ind[m+j] for j = 0..n-1
        # This is the cross-correlation of a and K_ind, evaluated at lags 0 to n-1.

            subgrad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    subgrad /= len(K)
    return c_max, c, subgrad, K


def softmax_obj_and_grad(theta, n, T, S):
    """Soft-max objective and gradient w.r.t. theta."""
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T
    # Soft-max weights
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    # Gradient w.r.t. a
    n_c = len(c)
    grad_a = np.zeros(n)
    for k_idx in range(min(200, n_c)):
        if w[k_idx] < 1e-15:
            continue
        k = k_idx
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            grad_a[lo:hi + 1] += 2.0 * w[k_idx] * a[k - hi:k - lo + 1][::-1]
    # Chain rule
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    return obj, grad_theta


def softmax_obj_and_grad_fast(theta, n, T, S):
    """Faster version using only top weights."""
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T
    # Use FFT-based gradient
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    # Truncate small weights
    w_thresh = 1e-10
    w[w < w_thresh] = 0
    if w.sum() < 1e-15:
        return obj, np.zeros(n)
    w /= w.sum()
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # This is 2 * correlate(w, a) evaluated at appropriate lags
    # correlate(w, a)[j] = sum_k w[k] * a[k-j] for k-j in [0,n-1]
    # = sum_m a[m] * w[m+j]
    N = 1 << (3 * n - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    corr = np.real(ifft(np.conj(W) * A))
    # corr[j + (n-1)] should give sum_m a[m] * w[m+j]... let me check
    # Actually: ifft(conj(W)*A)[m] = sum_n w[n] * a[n-m] (cross-correlation)
    # I want h[j] = sum_n w[n] * a[n-j] = corr[j] (for j >= 0 in the "positive lag" part)
    # In the FFT output of length N, the "zero lag" of correlation is at index 0.
    # corr[0] = sum_n w[n]*a[n], corr[1] = sum_n w[n]*a[n-1], etc.
    # But for negative indices it wraps around.
    # For j in [0, N/2]: corr[j] = sum_n w[n]*a[n-j]
    # For j in [N/2, N-1]: corr[j] = sum_n w[n]*a[n-j+N] (wrapped)
    # I want h[j] = sum_n w[n]*a[n-j] for j = 0..n-1
    # Since w has support [0, 2n-2] and a has support [0, n-1]:
    # h[j] = sum_{n=j}^{2n-2} w[n] * a[n-j] (since a[n-j]=0 for n-j >= n or n-j < 0)
    # = sum_{m=0}^{n-1} w[m+j] * a[m]
    # This equals corr[j] for j in [0, n-1] (as long as N is large enough)
    grad_a = 2.0 * corr[:n]
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    return obj, grad_theta


def run_adam_anneal(theta_init, n, S, time_budget, T_start=5.0, T_end=2000.0, lr_start=0.08):
    """Adam with exponential temperature annealing."""
    theta = theta_init.copy().astype(np.float64)
    n_par = len(theta)
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    start = time.monotonic()
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * np.exp(np.log(T_end / T_start) * progress)

        a = a_from_theta(theta, S)
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()

        if T > 1e5:
            # Use true subgradient
            c_max, c_sg, subgrad, K = true_max_and_subgrad(a)
            dot = np.dot(subgrad, a)
            grad = subgrad * a - (a / S) * dot
        else:
            obj, grad = softmax_obj_and_grad_fast(theta, n, T, S)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (0.5 ** (progress * 2.0))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

    return theta, best_a, best_obj


def run_multiplicative(a_init, n, S, time_budget, eta0=0.5, n_kick_lp=0):
    """True max subgradient with multiplicative updates."""
    a = a_init.copy().astype(np.float64)
    a = a * (S / a.sum())
    n_par = len(a)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    # Adaptive tolerance for K set
    tol_factor = 1e-5
    step = 0

    while time.monotonic() - start < time_budget:
        step += 1
        c_max, c, subgrad, K = true_max_and_subgrad(a)

        if c_max < best_obj:
            best_obj = c_max
            best_a = a.copy()

        # Adaptive tolerance: widen K set as we converge
        tol = max(1e-8, tol_factor * c_max * (1.0 + step * 1e-6))
        K = np.where(c >= c_max - tol)[0]
        if len(K) > 500:
            K = np.argsort(c)[::-1][:500]

        # Recompute subgradient with adapted K
        subgrad = np.zeros(n_par)
        for k in K:
            lo = max(0, k - n_par + 1)
            hi = min(k, n_par - 1)
            if hi >= lo:
                subgrad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        subgrad /= len(K)

        # Multiplicative update with mean subtraction
        subgrad_centered = subgrad - np.mean(subgrad)
        # Normalize for stable step
        sg_max = np.max(np.abs(subgrad_centered))
        if sg_max < 1e-15:
            break
        subgrad_norm = subgrad_centered / sg_max

        # Learning rate decay
        eta = eta0 * (1.0 + step * 1e-4) ** (-0.5)
        a = a * np.exp(-eta * subgrad_norm)
        a = np.maximum(a, 1e-15)
        a = a * (S / a.sum())

    return best_a, best_obj


def coordinate_descent_polish(a, n, S, time_budget, max_sweeps=200):
    """Coordinate descent with multiplicative shrink on high-contribution coords."""
    a = a.copy().astype(np.float64)
    a = a * (S / a.sum())
    n_par = len(a)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        c_max, c, subgrad, K = true_max_and_subgrad(a)
        if c_max < best_obj:
            best_obj = c_max
            best_a = a.copy()

        # Adaptive tolerance
        tol = max(1e-8, 1e-5 * c_max)
        K = np.where(c >= c_max - tol)[0]
        if len(K) > 300:
            K = np.argsort(c)[::-1][:300]

        # Compute subgradient
        sg = np.zeros(n_par)
        for k in K:
            lo = max(0, k - n_par + 1)
            hi = min(k, n_par - 1)
            if hi >= lo:
                sg[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        sg /= len(K)

        # Shrink high-contribution coordinates
        sg_centered = sg - np.mean(sg)
        sg_pos = np.maximum(sg_centered, 0)
        sg_neg = np.maximum(-sg_centered, 0)

        # For positive sg (coordinat
        a_shrink = a * np.exp(-0.3 * sg_pos / (np.max(sg_pos) + 1e-15))
        # For negative sg (coordinat
        a_grow = a * np.exp(0.1 * sg_neg / (np.max(sg_neg) + 1e-15))
        # Combine: only apply shrink where sg>0, grow where sg<0
        a_new = a.copy()
        mask_pos = sg_centered > 0
        mask_neg = sg_centered < 0
        a_new[mask_pos] = a_shrink[mask_pos]
        a_new[mask_neg] = a_grow[mask_neg]

        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new * (S / s_new)

        c_new = fft_convolve_self(a_new)
        obj_new = c_new.max()
        if obj_new < best_obj:
            a = a_new
            best_obj = obj_new
            best_a = a.copy()
        else:
            # Try smaller step
            a_half = a * (1 - 0.15 * sg_pos / (np.max(sg_pos) + 1e-15))
            a_half += 0.05 * sg_neg / (np.max(sg_neg) + 1e-15) * a
            a_half = np.maximum(a_half, 1e-15)
            s_half = a_half.sum()
            if s_half > 1e-10:
                a_half = a_half * (S / s_half)
                c_half = fft_convolve_self(a_half)
                if c_half.max() < best_obj:
                    a = a_half
                    best_obj = c_half.max()
                    best_a = a.copy()

    return best_a, best_obj


def solve_lp_direction(f, M, n, top_k_constraints=30, time_limit=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0."""
    c = fft_convolve_self(f)
    if top_k_constraints < len(c):
        active_lags = np.argsort(c)[::-1][:top_k_constraints]
    else:
        active_lags = np.arange(len(c))

    # Build constraint matrix
    n_cons = len(active_lags)
    A = np.zeros((n_cons, n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            A[idx, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

    c_obj = -np.ones(n)
    b = np.full(n_cons, M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                         method='highs', options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return None, None


def run_lbfgs_smooth(theta_init, n, T, S, time_budget, max_iter=300):
    """L-BFGS-B on soft-max objective."""
    state_dict = {'start': time.monotonic(), 'time_budget': time_budget,
                  'best_obj': np.inf, 'best_a': None}
    a0 = a_from_theta(theta_init, S)
    c0 = fft_convolve_self(a0)
    state_dict['best_obj'] = c0.max()
    state_dict['best_a'] = a0.copy()

    def lbfgs_fun(theta):
        elapsed = time.monotonic() - state_dict['start']
        if elapsed > state_dict['time_budget']:
            return state_dict['best_obj'], np.zeros(len(theta))
        obj, grad = softmax_obj_and_grad_fast(theta, n, T, S)
        a = a_from_theta(theta, S)
        c = fft_convolve_self(a)
        true_max = c.max()
        if true_max < state_dict['best_obj']:
            state_dict['best_obj'] = true_max
            state_dict['best_a'] = a.copy()
        return obj, grad

    try:
        result = minimize(lbfgs_fun, theta_init.copy(), jac=True, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'maxfun': 1000, 'maxls': 50, 'gtol': 1e-9})
    except Exception:
        pass

    return state_dict['best_a'], state_dict['best_obj']


def get_initialization(name, n, rng, S):
    """Generate various initializations."""
    i = np.arange(n)
    if name == "uniform":
        a = np.ones(n)
    elif name == "decaying":
        a = np.exp(-i / (n * 0.3))
    elif name == "increasing":
        a = np.exp(i / (n * 0.3))
    elif name == "random":
        a = rng.random(n) + 0.01
    elif name == "gaussian":
        a = np.exp(-0.5 * ((i - n * 0.3) / (n / 5)) ** 2)
    elif name == "cosine":
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 2 * i / n)
        a = np.maximum(a, 0.01)
    elif name == "power_law":
        a = (i + 1) ** (-0.5)
    elif name == "two_humps":
        a = np.exp(-0.5 * ((i - n * 0.2) / (n / 8)) ** 2) + \
            np.exp(-0.5 * ((i - n * 0.6) / (n / 8)) ** 2)
    elif name == "edge_concentrated":
        dist = np.minimum(i, n - 1 - i).astype(float)
        a = np.exp(-dist / (n * 0.15))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-6)
    return a * (S / a.sum())

    ver = evaluate_sequence(final_a)
    print("FINAL: obj=%.6f (verified=%s)" % (global_best,
                                               "%.6f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.6f" % ver)

    return final_a
