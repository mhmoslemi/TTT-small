import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    n = len(v)

    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]

Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses LP-based iterative refinement (maximizing sum(b) subject to f*b <= M)
combined with smoothed-max projected gradient descent.
Explores multiple sequence lengths and initializations.

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def proj_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    max_c = np.max(c)
    return 2.0 * len(a) * max_c / (sum_a ** 2)


def build_lp_matrix(f):
    """Build the constraint matrix A for the LP. A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            A[k, j] = f[k - j]
    return A


def lp_solve(f, M):
    """Solve: max sum(b) s.t. A@b <= M*1, b >= 0. Returns b or None."""
    n = len(f)
    A = build_lp_matrix(f)
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

    a = a_init.astype(np.float64).copy()
    a = proj_simplex(a, 1.0)
    n = len(a)

    best_a = a.copy()

        M = np.max(c)

            break
        b = np.maximum(b, 0)
        b_sum = np.sum(b)
        if b_sum < 1e-12:
            break
        b = b / b_sum

        a_new = (1.0 - t_mix) * a + t_mix * b
        a_new = proj_simplex(a_new, 1.0)

        new_score = compute_score(a_new)

            best_a = a_new.copy()

            print(f"    LP iter={it}, M={M:.10f}, best_score={best_score:.8f}")

    return best_a, best_score


def optimize_smoothed_max(a_init, time_budget, rng, beta_start=1.0, beta_end=200.0):
    """Smoothed max (log-sum-exp) with projected gradient descent on simplex."""
    a = a_init.astype(np.float64).copy()
    a = proj_simplex(a, 1.0)
    n = len(a)

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    n_inner = 200000
    lr = 0.5

    for it in range(n_inner):
        if time.monotonic() - t0 > time_budget:
            break

        frac = it / max(n_inner - 1, 1)
        beta = beta_start + (beta_end - beta_start) * frac

        c = fft_convolve(a, a)

        # Softmax weights
        bc = beta * c
        bc -= np.max(bc)
        p = np.exp(bc)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            p = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            p = p / p_sum

        # Gradient: g[j] = 2 * sum_k p[k] * a[k-j]
        # Compute via cross-correlation using FFT
        L = 1
        while L < 2 * n:
            L <<= 1
        P = np.fft.rfft(p, L)
        A_fft = np.fft.rfft(a, L)
        g = 2.0 * np.fft.irfft(P * np.conj(A_fft), L)[:n]

        # Clip for stability
        g = np.clip(g, -100.0, 100.0)

        # Gradient step
        a_new = a - lr * g
        a_proj = proj_simplex(a_new, 1.0)

        new_score = compute_score(a_proj)
        if new_score < best_score:
            best_score = new_score
            best_a = a_proj.copy()

        a = a_proj

        if it % 5000 == 0:
            print(f"    smooth iter={it}, beta={beta:.1f}, best_score={best_score:.8f}")

    return best_a, best_score


def optimize_mirror_descent(a_init, time_budget, rng):
    """Mirror descent (multiplicative weights) on the simplex."""
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    n = len(a)

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    n_inner = 100000
    lr = 0.1
    beta = 50.0

    for it in range(n_inner):
        if time.monotonic() - t0 > time_budget:
            break

        # Anneal beta
        frac = it / max(n_inner - 1, 1)
        beta_t = 1.0 + frac * (beta - 1.0)

        c = fft_convolve(a, a)
        bc = beta_t * c
        bc -= np.max(bc)
        p = np.exp(bc)
        p /= np.sum(p)

        # Gradient: g[j] = 2 * sum_k p[k] * a[k-j]
        L = 1
        while L < 2 * n:
            L <<= 1
        P = np.fft.rfft(p, L)
        A_fft = np.fft.rfft(a, L)
        g = 2.0 * np.fft.irfft(P * np.conj(A_fft), L)[:n]
        g = np.clip(g, -100.0, 100.0)

        # Multiplicative update
        a_new = a * np.exp(-lr * g / a)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)

        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    mirror iter={it}, beta={beta_t:.1f}, best_score={best_score:.8f}")

    return best_a, best_score


def bell_init(n, width=1.0):
    """Create a bell-shaped initialization."""
    x = np.linspace(-1, 1, n)
    a = np.exp(-width * x ** 2)
    a = np.maximum(a, 1e-6)
    return a / np.sum(a)


def tri_init(n):
    """Create a triangular initialization."""
    a = np.arange(1, n + 1)
    a = np.concatenate([a, a[::-1][:max(0, n - len(a))]])[:n]
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Find non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP-based iteration, smoothed max gradient descent, and mirror descent
    across multiple sequence lengths and initializations.
    """
    t_global = time.monotonic()
    deadline = t_global + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def update_best(seq, score):
        nonlocal best_score, best_seq
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  >>> NEW BEST: score={score:.8f}, n={len(seq)}")

    # Define candidate lengths and time allocation
    configs = [
        (1024, 0.30),  # (n, fraction of time)
        (2048, 0.35),
        (4096, 0.25),
    ]

    total_time = max(0.0, deadline - time.monotonic())

    for n, frac in configs:
        time_left = deadline - time.monotonic()
        if time_left < 60:
            break
        t_budget = min(frac * total_time, time_left - 30)
        if t_budget < 30:
            continue

        print(f"\n=== n={n}, budget={t_budget:.0f}s ===")

        # Initialize with various starting points
        inits = [
            ("uniform", np.ones(n) / n),
            ("bell", bell_init(n, 1.0)),
            ("bell_wide", bell_init(n, 0.5)),
            ("tri", tri_init(n)),
        ]

        # Try LP from best init
        for name, init in inits[:2]:  # uniform and bell for LP (expensive)
            t_left = deadline - time.monotonic()
            if t_left < 20:
                break
            t_sub = min(t_budget * 0.35, t_left - 10)
            if t_sub < 10:
                continue
            print(f"  LP from {name}...")
            t0 = time.monotonic()
            result, score = optimize_lp(init, t_sub, rng, t_mix=0.3)
            elapsed = time.monotonic() - t0
            print(f"    LP {name}: score={score:.8f} ({elapsed:.0f}s)")
            update_best(result, score)

        # Try smoothed max gradient
        for name, init in inits:
            t_left = deadline - time.monotonic()
            if t_left < 20:
                break
            t_sub = min(t_budget * 0.15, t_left - 10)
            if t_sub < 10:
                continue
            print(f"  Smoothed max from {name}...")
            t0 = time.monotonic()
            result, score = optimize_smoothed_max(init, t_sub, rng)
            elapsed = time.monotonic() - t0
            print(f"    Smooth {name}: score={score:.8f} ({elapsed:.0f}s)")
            update_best(result, score)

        # Continue optimization from best found so far for this n
        t_left = deadline - time.monotonic()
        if t_left > 60 and best_seq is not None and len(best_seq) == n:
            t_sub = min(t_budget * 0.2, t_left - 20)
            if t_sub > 10:
                print(f"  Continuing smoothed max from best...")
                result, score = optimize_smoothed_max(best_seq, t_sub, rng,
                                                       beta_start=50.0, beta_end=500.0)
                print(f"    Continued: score={score:.8f}")
                update_best(result, score)

    # If we have a good result, try to polish with longer n
    t_left = deadline - time.monotonic()
    if t_left > 120:
        n_polish = 4096
        print(f"\n=== Polish with n={n_polish} ===")
        # Initialize from scaled version of best
        if best_seq is not None:
            # Interpolate best to new length
            from numpy import interp
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_polish)
            init_polish = interp(x_new, x_old, best_seq)
            init_polish = np.maximum(init_polish, 0)
            init_polish = init_polish / np.sum(init_polish)
        else:
            init_polish = bell_init(n_polish, 0.8)

        t_sub = min(t_left - 20, 300)
        result, score = optimize_smoothed_max(init_polish, t_sub, rng,
                                               beta_start=10.0, beta_end=300.0)
        print(f"  Polish smoothed: score={score:.8f}")
        update_best(result, score)

    # Final local search: try small perturbations
    t_left = deadline - time.monotonic()
    if t_left > 30 and best_seq is not None:
        print(f"\n=== Local search polish (time_left={t_left:.0f}s) ===")
        a = best_seq.copy()
        n = len(a)
        for trial in range(200):
            if time.monotonic() > deadline:
                break
            # Random perturbation
            pert = rng.standard_normal(n) * 0.01
            a_new = a + pert
            a_new = proj_simplex(a_new, 1.0)
            s = compute_score(a_new)
            if s < best_score:
                best_score = s
                best_seq = a_new.copy()
                a = a_new.copy()

    # Validate and return
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = 2.0

    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(final_list)
    print(f"\n=== FINAL: score={final_score:.8f}, n={len(final_list)} ===")

    return final_list
