def gradient_step(a, S, n, T, lr, momentum_state=None):
    """
    One gradient step using soft-max objective.
    Returns (new_a, new_momentum_state).
    """
    c = fft_convolve_self(a)
    c_max = c.max()
    
    # Soft-max weights
    shifted = T * (c - c_max)
    shifted = shifted - shifted.max()
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient of soft-max w.r.t. a: grad_a[m] = 2 * sum_j w[m+j] * a[j]
    # This is 2 * (w * a)[m] where * is cross-correlation
    # Actually: d/da[m] sum_k w[k] * conv(a,a)[k]
    # conv(a,a)[k] = sum_j a[j] * a[k-j]
    # d/d a[m] = sum_j w[k] * (a[k-j] * delta(j,m) + a[j] * delta(k-j,m))
    # = sum_j w[k] * (a[k-m] + a[m] * delta(k-j,m) ...) 
    # Let me be more careful.
    # conv(a,a)[k] = sum_{j=0}^{n-1} a[j] * a[k-j] (with a[i]=0 outside [0,n-1])
    # d/d a[m] conv(a,a)[k] = a[k-m] + (m==k-m ? a[m] : a[k-m]) 
    # Wait: d/d a[m] [sum_j a[j] a[k-j]]
    # = sum_j [delta(j,m) a[k-j] + a[j] delta(k-j,m)]
    # = a[k-m] + a[k-m] (if k-m is in [0,n-1] and m is in [0,n-1])
    # Actually the two terms are: a[k-m] from the first sum (j=m) and a[k-m] from the second sum (j=k-m)
    # But these are the same term! Unless m = k-m, in which case it's 2*a[k-m]... no wait.
    # d/d a[m] of a[j]*a[k-j] = delta(m,j)*a[k-j] + a[j]*delta(m,k-j)
    # Sum over j: a[k-m] + a[k-m] = 2*a[k-m] (when k-m is in valid range)
    # Wait no. When j=m: a[k-m]. When k-j=m, i.e., j=k-m: a[k-m].
    # These are the same value only when m = k-m, i.e., k=2m.
    # Actually no, they're both equal to a[k-m]. The first term (j=m) gives a[k-m].
    # The second term (j=k-m) gives a[k-m]. So total is 2*a[k-m].
    # But wait, if m = k-m (i.e., k = 2m), then both terms are a[m], and the derivative is 2*a[m].
    # Hmm, actually I think the correct answer is:
    # d/d a[m] conv(a,a)[k] = 2 * a[k-m] if k-m is in [0,n-1], else 0
    # (The factor of 2 comes from the two occurrences of a[m] in the product, 
    #  one as a[j] with j=m, and one as a[k-j] with k-j=m, i.e., j=k-m.)
    # But when m = k-m, i.e., k=2m, the product is a[m]*a[m], and the derivative is 2*a[m].
    # So the formula 2*a[k-m] works in all cases (including k=2m where it gives 2*a[m]).
    
    # So grad_a[m] = sum_k w[k] * 2 * a[k-m] (where k-m is in [0,n-1])
    # = 2 * sum_k w[k] * a[k-m] (with a[i]=0 outside [0,n-1])
    # This is 2 * cross-correlation of w and a, evaluated at lag m.
    # cross_correlation(w, a)[m] = sum_k w[k] * a[k-m]
    # Which is the same as convolution of w and reversed a:
    # conv(w, a_rev)[m + n - 1] where a_rev = a[::-1]
    
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    # We want indices n-1 to 2n-2 of the full convolution
    grad_a = 2.0 * full_conv[n-1:2*n-1]
    
    # Project gradient (remove mean component to maintain sum constraint approximately)
    grad_a = grad_a - np.mean(grad_a)
    
    # Apply step with momentum
    if momentum_state is None:
        v = np.zeros(n)
    else:
        v = momentum_state['v']
    
    beta = 0.9
    v = beta * v + (1 - beta) * grad_a
    a_new = a - lr * v
    
    return a_new, {'v': v}


def run_gradient_optimization(a, S, n, time_budget, initial_T=5, final_T=200, lr=0.1):
    """Run gradient optimization with annealing."""
    a = a.copy()
    a = np.maximum(a, 1e-15)
    a = normalize_to_sum(a, S)
    best_obj = compute_objective(a)
    best_a = a.copy()
    
    momentum = None
    start_time = time.monotonic()
    
    while time.monotonic() - start_time < time_budget:
        progress = (time.monotonic() - start_time) / time_budget
        T = initial_T * np.exp(np.log(final_T / initial_T) * progress)
        lr_t = lr * (1.0 - progress * 0.5)
        
        a_new, momentum = gradient_step(a, S, n, T, lr_t, momentum)
        a_new = np.maximum(a_new, 1e-15)
        a_new = normalize_to_sum(a_new, S)
        
        obj_new = compute_objective(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj


def get_difference_set_init(n, rng, S):
    """
    Create initialization based on a periodic difference-set-like structure.
    Uses a prime p close to n and quadratic residues mod p.
    """
    # Find a prime p close to n
    def is_prime(x):
        if x < 2:
            return False
        for i in range(2, int(np.sqrt(x)) + 1):
            if x % i == 0:
                return False
        return True
    
    p = n
    while not is_prime(p) and p > n // 2:
        p -= 1
    
    if p < 10:
        p = n
        while not is_prime(p) and p < 2 * n:
            p += 1
    
    # Quadratic residues mod p
    residues = set()
    for x in range(1, p):
        residues.add((x * x) % p)
    
    # Create sequence: a[i] = 1 if i mod p is a quadratic residue, else some small value
    a = np.zeros(n)
    for i in range(n):
        if (i % p) in residues:
            a[i] = 1.0
        else:
            a[i] = 0.1  # small background
    
    return normalize_to_sum(a, S)


def get_multi_scale_init(n, rng, S):
    """Create initialization using multi-scale structure."""
    a = np.ones(n)
    # Add several cosine components with random frequencies and phases
    num_components = rng.integers(3, 8)
    for _ in range(num_components):
        freq = rng.uniform(1, n / 4)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        a += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    
    a = np.maximum(a, 0.01)
    return normalize_to_sum(a, S)

"""
Hybrid LP + gradient optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous approach:
1. Proper LP formulation (SoTA) with epsilon-tightened peak constraints
2. Adaptive step size via multi-point line search
3. Interleaved LP and gradient steps to escape local minima
4. Multi-length exploration (4096, 8192)
5. Spectral shaping (Gerchberg-Saxton style) as additional diversification
6. Better initialization strategies
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fft_convolve_self(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def normalize_to_sum(a, S):
    """Scale a to have sum S, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def build_lp_matrix(f, n, lags=None):
    """Build constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    full_len = 2 * n - 1
    if lags is None:
        lags = np.arange(full_len)
    A = np.zeros((len(lags), n))
    for idx, k in enumerate(lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[idx, lo:hi+1] = f[k-hi:k-lo+1][::-1]
    return A


def solve_lp_direction(f, M, n, epsilon=0.02, top_k=None, time_limit=45):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= b[k] for all k, g >= 0.
    b[k] = M*(1-epsilon) for top-k peak lags, M for others.
    Returns (g, g_sum) or (None, None).
    """
    c = fft_convolve_self(f)
    full_len = 2 * n - 1
    
    if top_k is not None and top_k < full_len:
        active_lags = np.argsort(c)[::-1][:top_k]
    else:
        active_lags = np.arange(full_len)
    
    # Determine which active lags are "peaks"
    c_active = c[active_lags]
    M_eff = M * (1 - epsilon)
    peak_threshold = M * (1 - epsilon * 0.5)
    
    b_vec = np.full(len(active_lags), M)
    # Tighten constraints at peaks
    peak_mask = c_active >= peak_threshold
    b_vec[peak_mask] = M_eff
    
    A = build_lp_matrix(f, n, active_lags)
    c_obj = -np.ones(n)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_vec,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum
    except Exception as e:
        print("  LP error: %s" % str(e))
    
    return None, None


def lp_blend_iteration(a, S, n, epsilon=0.02, top_k=None, time_limit=45):
    """
    Single LP blending iteration.
    Returns (new_a, improved, obj_new).
    """
    c = fft_convolve_self(a)
    M = c.max()
    obj_cur = compute_objective(a)
    
    g, g_sum = solve_lp_direction(a, M, n, epsilon=epsilon,
                                  top_k=top_k, time_limit=time_limit)
    if g is None:
        return a, False, obj_cur
    
    g = normalize_to_sum(g, S)
    
    # Adaptive step size: try multiple values
    best_t = 0.0
    best_t_obj = obj_cur
    
    t_candidates = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]
    for t in t_candidates:
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 0.01:
            continue
        a_new = a_new * (S / s_new)
        obj_new = compute_objective(a_new)
        if obj_new < best_t_obj:
            best_t_obj = obj_new
            best_t = t
    
    if best_t < 0.005:
        return a, False, obj_cur
    
    a_new = (1 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_to_sum(a_new, S)
    
    return a_new, True, best_t_obj


def run_lp_optimization(a, S, n, time_budget, epsilon=0.02, top_k=None,
                        time_per_lp=40, max_iterations=100):
    """
    Run LP-based optimization loop.
    Returns (best_a, best_obj).
    """
    a = a.copy()
    a = normalize_to_sum(a, S)
    best_obj = compute_objective(a)
    best_a = a.copy()
    
    start_time = time.monotonic()
    iteration = 0
    stall_count = 0
    
    while time.monotonic() - start_time < time_budget and iteration < max_iterations:
        a_new, improved, obj_new = lp_blend_iteration(
            a, S, n, epsilon=epsilon, top_k=top_k, time_limit=time_per_lp)
        
        iteration += 1
        
        if improved:
            a = a_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
                stall_count = 0
                if iteration % 3 == 0:
                    print("    LP iter %d: obj=%.8f" % (iteration, best_obj))
            else:
                stall_count += 1
        else:
            stall_count += 1
        
        # If stalled, try reducing epsilon or changing top_k
        if stall_count >= 3:
            if epsilon > 0.005:
                epsilon = max(0.005, epsilon * 0.7)
            elif top_k is not None and top_k > 100:
                top_k = max(100, top_k // 2)
            elif top_k is None:
                top_k = 500
            stall_count = 0
            print("    Adjusting params: eps=%.4f, top_k=%s" % (epsilon, str(top_k)))
        
        if time.monotonic() - start_time > time_budget - time_per_lp:
            break
    
    print("  LP done: %d iterations, best_obj=%.8f" % (iteration, best_obj))
    return best_a, best_obj


def gradient_step_v2(a, S, n, T, lr, state=None):
    """
    One gradient step using soft-max of self-convolution.
    Uses log-parameterization implicitly by working in a-space with clipping.
    """
    c = fft_convolve_self(a)
    c_max = c.max()
    
    # Soft-max weights
    shifted = T * (c - c_max)
    shifted = shifted - shifted.max()
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w = exp_shifted / total
    
    # Gradient: grad_a[m] = 2 * sum_k w[k] * a[k-m] (with a=0 outside [0,n-1])
    # = 2 * cross_correlation(w, a)[m]
    # Compute via FFT: cross_correlation(w, a)[m] = conv(w, a_rev)[m + n - 1]
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    grad_a = 2.0 * full_conv[n-1:2*n-1]
    
    # Remove mean component (to approximately preserve sum)
    grad_a = grad_a - np.mean(grad_a)
    
    # Adam-style update
    if state is None:
        state = {'m': np.zeros(n), 'v': np.zeros(n), 't': 0}
    state['t'] += 1
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    state['m'] = beta1 * state['m'] + (1 - beta1) * grad_a
    state['v'] = beta2 * state['v'] + (1 - beta2) * (grad_a ** 2)
    m_hat = state['m'] / (1 - beta1 ** state['t'])
    v_hat = state['v'] / (1 - beta2 ** state['t'])
    
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    
    return a_new, state


def run_gradient_optimization(a, S, n, time_budget, T_start=10, T_end=500, lr=0.3):
    """Run gradient optimization with temperature annealing."""
    a = a.copy()
    a = np.maximum(a, 1e-12)
    a = normalize_to_sum(a, S)
    best_obj = compute_objective(a)
    best_a = a.copy()
    
    state = None
    start_time = time.monotonic()
    
    while time.monotonic() - start_time < time_budget:
        progress = min(1.0, (time.monotonic() - start_time) / time_budget)
        T = T_start * np.exp(np.log(max(T_end, T_start + 1)) * progress)
        lr_t = lr * (1.0 - 0.5 * progress)
        
        a_new, state = gradient_step_v2(a, S, n, T, lr_t, state)
        a_new = np.maximum(a_new, 1e-12)
        a_new = normalize_to_sum(a_new, S)
        
        obj_new = compute_objective(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj


def spectral_shaping(a, S, n, num_iters=20):
    """
    Gerchberg-Saxton style: iterate between making |DFT|^2 flat and clipping.
    """
    a = a.copy()
    a = normalize_to_sum(a, S)
    best_obj = compute_objective(a)
    best_a = a.copy()
    
    for _ in range(num_iters):
        # Compute DFT
        fsize = 1 << (2 * n - 1).bit_length()
        A = fft(a, fsize)
        
        # Target: flat magnitude -> |A|^2 = mean(|A|^2)
        # Practically: reduce the dynamic range of |A|
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress dynamic range
        mag_min = mag[mag > 1e-10].min() if np.any(mag > 1e-10) else 1e-10
        mag_max = mag.max()
        
        # Apply soft compression: new_mag = mag^alpha for alpha < 1
        alpha = 0.5
        new_mag = mag ** alpha
        # Renormalize to preserve energy
        energy_ratio = (mag ** 2).sum() / (new_mag ** 2).sum()
        new_mag = new_mag * np.sqrt(max(energy_ratio, 1e-30))
        
        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fsize))[:n]
        
        # Clip and normalize
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to_sum(a_new, S)
        
        obj_new = compute_objective(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj


def get_initialization(name, n, rng, S):
    """Generate various initializations."""
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return normalize_to_sum(a, S)
    elif name == "gaussian":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n / 2) / (n / 6)) ** 2)
        return normalize_to_sum(a, S)
    elif name == "cosine_multi":
        i = np.arange(n)
        a = np.ones(n)
        for k in range(1, 5):
            amp = rng.uniform(0.1, 0.4)
            freq = rng.uniform(1, n / 3)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * freq * i / n + phase)
        a = np.maximum(a, 0.01)
        return normalize_to_sum(a, S)
    elif name == "banded":
        # Sequence concentrated at edges
        i = np.arange(n)
        a = np.exp(-0.5 * (i / (n/4))**2) + np.exp(-0.5 * ((i-n+1) / (n/4))**2)
        return normalize_to_sum(a, S)
    elif name == "quasiperiodic":
        # Use golden ratio for quasi-periodic structure
        i = np.arange(n)
        tau = (1 + 5**0.5) / 2
        a = 0.5 + 0.4 * np.cos(2 * np.pi * tau * i / n)
        a = np.maximum(a, 0.01)
        return normalize_to_sum(a, S)
    else:
        return np.full(n, S / n)

        a_best, obj_best = run_lp_optimization(
            a, S, n, lp_time1, epsilon=0.03, top_k=None, time_per_lp=40)
        
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print("  After LP pass 1: obj=%.8f" % global_best)
        
        # Gradient polish
        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            grad_time = min(60.0, 0.04 * remaining)
            a_g, obj_g = run_gradient_optimization(
                global_a.copy(), S, n, grad_time, T_start=20, T_end=200, lr=0.2)
            if obj_g < global_best:
                global_best = obj_g
                global_a = a_g.copy()
                print("  After gradient: obj=%.8f" % global_best)
        
        # Second LP pass: tighter epsilon
        remaining = deadline - time.monotonic() - 30
        if remaining > 60:
            lp_time2 = min(300.0, 0.15 * remaining)
            a_best, obj_best = run_lp_optimization(
                global_a.copy(), S, n, lp_time2, epsilon=0.01, top_k=800,
                time_per_lp=35)
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print("  After LP pass 2: obj=%.8f" % global_best)
        
        # Spectral shaping attempt
        remaining = deadline - time.monotonic() - 30
        if remaining > 20:
            a_sp, obj_sp = spectral_shaping(global_a.copy(), S, n, num_iters=15)
            if obj_sp < global_best:
                global_best = obj_sp
                global_a = a_sp.copy()
                print("  After spectral shaping: obj=%.8f" % global_best)
        
        # Third LP pass: very tight
        remaining = deadline - time.monotonic() - 30
        if remaining > 60:
            lp_time3 = min(200.0, 0.1 * remaining)
            a_best, obj_best = run_lp_optimization(
                global_a.copy(), S, n, lp_time3, epsilon=0.005, top_k=500,
                time_per_lp=35)
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print("  After LP pass 3: obj=%.8f" % global_best)
    
    # === Phase 2: Try n=8192 ===
    remaining = deadline - time.monotonic() - 30
    if remaining > 90:
        n2 = 8192
        S2 = np.sqrt(2.0 * n2)
        
        print("Phase 2: Trying n=%d" % n2)
        
        # Initialize by upsample
        if global_a is not None:
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, len(global_a))
            x_new = np.linspace(0, 1, n2)
            a2 = np.interp(x_new, x_old, global_a)
        else:
            a2 = get_initialization("cosine_multi", n2, rng, S2)
        a2 = normalize_to_sum(a2, S2)
        
        # Quick gradient to get into basin
        grad_time = min(40.0, 0.15 * remaining)
        a2_best, obj2 = run_gradient_optimization(
            a2, S2, n2, grad_time, T_start=5, T_end=50, lr=0.5)
        
        print("  n=%d after gradient: obj=%.6f" % (n2, obj2))
        
        # LP optimization
        if obj2 < 1.55:
            lp_time = min(80.0, 0.2 * remaining)
            a2_best, obj2 = run_lp_optimization(
                a2_best, S2, n2, lp_time, epsilon=0.03, top_k=1000,
                time_per_lp=30)
            print("  n=%d after LP: obj=%.6f" % (n2, obj2))
        
        if obj2 < global_best:
            global_best = obj2
            global_a = a2_best.copy()
            global_n = n2
            print("  n=%d improved global: obj=%.8f" % (n2, global_best))
    
    # === Phase 3: Try alternative initializations at n=4096 ===
    remaining = deadline - time.monotonic() - 30
    if remaining > 60:
        n3 = 4096
        S3 = np.sqrt(2.0 * n3)
        
        for init_name in ["cosine_multi", "quasiperiodic", "banded"]:
            remaining = deadline - time.monotonic() - 30
            if remaining < 30:
                break
            
            a3 = get_initialization(init_name, n3, rng, S3)
            
            # Quick gradient
            grad_time = min(20.0, 0.1 * remaining)
            a3_best, obj3 = run_gradient_optimization(
                a3, S3, n3, grad_time, T_start=5, T_end=100, lr=0.5)
            
            # LP if promising
            remaining = deadline - time.monotonic() - 30
            if obj3 < 1.53 and remaining > 30:
                lp_time = min(40.0, 0.3 * remaining)
                a3_best, obj3 = run_lp_optimization(
                    a3_best, S3, n3, lp_time, epsilon=0.03, top_k=600,
                    time_per_lp=25)
            
            if obj3 < global_best:
                global_best = obj3
                global_a = a3_best.copy()
                global_n = n3
                print("  Init %s n=%d: obj=%.8f (new best!)" % (init_name, n3, global_best))
    
    # === Phase 4: Final aggressive polish ===
    if global_a is not None:
        remaining = deadline - time.monotonic() - 5
        n = global_n
        S = np.sqrt(2.0 * n)
        
        if remaining > 30:
            # Final LP with very tight epsilon
            a_final, obj_final = run_lp_optimization(
                global_a.copy(), S, n, min(remaining * 0.5, 120.0),
                epsilon=0.003, top_k=300, time_per_lp=30)
            if obj_final < global_best:
                global_best = obj_final
                global_a = a_final.copy()
                print("  Final LP: obj=%.8f" % global_best)
        
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            # Final gradient with very high T (targets the max directly)
            a_final, obj_final = run_gradient_optimization(
                global_a.copy(), S, n, remaining,
                T_start=200, T_end=2000, lr=0.05)
            if obj_final < global_best:
                global_best = obj_final
                global_a = a_final.copy()
                print("  Final gradient: obj=%.8f" % global_best)
    
    # === Fallback ===
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    # Verify with evaluate_sequence
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("FINAL: computed=%.8f, verified=%.8f, n=%d" % (global_best, ver, len(global_a)))
    
    if not np.isfinite(ver):
        final_a = [1.0] * 512
        ver = evaluate_sequence(final_a)
        print("Fallback used: obj=%.6f" % ver)
    
    return final_a
