```python
"""
LP-projected subgradient optimization for minimizing peak autocorrelation
of non-negative step functions. Uses iterative linear programming to find
the most "spread out" sequence under a peak constraint, combined with
gradient-based refinement and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import next_fast_len
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = next_fast_len(n + m - 1)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def build_lp_matrix(f, n):
    """
    Build the constraint matrix A for the LP.
    A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    Constraint: A @ g <= M (elementwise)
    """
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        # g[i] contributes to c[k] with coefficient f[k-i]
        # k-i must be in [0, n-1], so k in [i, i+n-1]
        k_start = i
        k_end = i + n  # exclusive
        A[k_start:k_end, i] = f[0:n]
    return A

def solve_lp(f, M, n):
    """
    Solve: max sum(g) s.t. A @ g <= M, g >= 0
    """
    A = build_lp_matrix(f, n)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

def lp_iterate(f, time_limit, rng, verbose=False):
    """
    Iterative LP-based optimization.
    """
    n = len(f)
    S = np.sum(f)
    f = f / S * np.sqrt(2.0 * n)  # normalize
    S_target = np.sqrt(2.0 * n)
    
    current = f.copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    iteration = 0
    step_size = 0.3
    
    # Precompute the LP matrix structure (it depends on f, so we rebuild each time)
    # Actually, the matrix changes as f changes, so we can't precompute.
    # But for the LP, we use the CURRENT f to build constraints.
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        c = fft_convolve(current, current)
        M = np.max(c)
        
        # Solve LP
        g = solve_lp(current, M, n)
        if g is None:
            # LP failed, try smaller M
            M *= 0.95
            g = solve_lp(current, M, n)
            if g is None:
                break
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        # Rescale g to have target sum
        g = g / sum_g * S_target
        
        # Convex combination
        current = (1 - step_size) * current + step_size * g
        current = np.clip(current, 0, 1000)
        
        # Renormalize
        S_cur = np.sum(current)
        if S_cur < 1e-10:
            current = f.copy()
            continue
        current = current / S_cur * S_target
        
        new_score = local_score(current)
        if new_score < best_score:
            best_score = new_score
            best = current.copy()
            if verbose and iteration % 10 == 0:
                print(f"  LP iter={iteration}, score={best_score:.6f}")
        
        # Adaptive step size
        if new_score > current_score * 1.01:
            step_size *= 0.9
        else:
            step_size = min(step_size * 1.01, 0.5)
    
    return best, best_score
```

```python
"""
LP-guided optimization for minimizing peak autocorrelation of non-negative
step functions. Combines linear programming projections with gradient-based
refinement and targeted peak-shaving mutations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import next_fast_len
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = next_fast_len(n + m - 1)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def make_lp_matrix(f):
    """Build constraint matrix for (f*g) <= M."""
    n = len(f)
    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        # For variable g[i], it appears in c[k] with coefficient f[k-i]
        # k-i must be in [0, n-1], so k in [i, i+n-1]
        A[i:i+n, i] = f
    return A

def solve_lp_peak(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = make_lp_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tol': 1e-9})
    if res.success:
        return res.x, res
    return None, res

def lp_optimize(f_init, time_limit, rng, step_size_init=0.3):
    """LP-iterative optimization."""
    n = len(f_init)
    S_target = np.sqrt(2.0 * n)
    current = f_init.copy()
    current = current / np.sum(current) * S_target
    
    best = current.copy()
    best_score = local_score(current)
    step_size = step_size_init
    
    t0 = time.monotonic()
    iteration = 0
    stuck_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        c = fft_convolve(current, current)
        M = np.max(c)
        
        # Solve LP
        g, res = solve_lp_peak(current, M)
        if g is None:
            # Try slightly larger M
            M *= 1.001
            g, res = solve_lp_peak(current, M)
            if g is None:
                break
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        # Rescale g
        g = g / sum_g * S_target
        
        # Convex combination with line search
        best_t = step_size
        best_trial_score = np.inf
        for t in [step_size * 0.5, step_size, step_size * 1.5]:
            trial = (1 - t) * current + t * g
            trial = np.clip(trial, 0, 1000)
            S_t = np.sum(trial)
            if S_t < 1e-10:
                continue
            trial = trial / S_t * S_target
            s = local_score(trial)
            if s < best_trial_score:
                best_trial_score = s
                best_t = t
        
        current = (1 - best_t) * current + best_t * g
        current = np.clip(current, 0, 1000)
        S_cur = np.sum(current)
        current = current / S_cur * S_target
        
        new_score = local_score(current)
        if new_score < best_score:
            best_score = new_score
            best = current.copy()
            stuck_count = 0
        else:
            stuck_count += 1
            step_size *= 0.95
            if stuck_count > 50:
                # Reset to best and increase step size
                current = best.copy()
                step_size = min(step_size * 2, 0.5)
                stuck_count = 0
        
        if iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iteration}, score={best_score:.6f}, M={M:.4f}, t={elapsed:.0f}s")
    
    return best, best_score

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def gradient_refine(a_init, time_limit, rng):
    """Softmax-parameterized gradient descent with adaptive temperature."""
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    theta = np.log(a_init / np.sum(a_init) + 1e-10)
    
    best = a_init.copy()
    best_score = local_score(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    
    t0 = time.monotonic()
    tau_start = 50.0 / n
    tau_end = 0.01 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        a = softmax_normalize(theta) * S_target
        c = fft_convolve(a, a)
        c_max = np.max(c)
        
        # Softmax gradient
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        w = exp_shifted / np.sum(exp_shifted)
        
        # Gradient of soft-max w.r.t. a
        # d(soft_max)/da[j] = 2 * sum_k w[k] * dc[k]/da[j] / len... 
        # Actually: soft_max = c_max + tau * log(sum exp((c-k_max)/tau))
        # d(soft_max)/da[j] = sum_k w[k] * dc[k]/da[j]
        # dc[k]/da[j] = a[k-j] + a[j-k] (with bounds)
        # = (a_reversed * w)[j + n - 1] ... let me use FFT
        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g_a = 2.0 * corr[n - 1:2 * n - 1]
        
        # Convert to theta-space
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S_target)
        
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g_theta
        v_buf = beta2 * v_buf + (1 - beta2) * g_theta ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)
        
        lr = 2.0 / np.sqrt(step + 1)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if step % 500 == 0:
            a_check = softmax_normalize(theta) * S_target
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best = a_check.copy()
    
    a_final = softmax_normalize(theta) * S_target
    score = local_score(a_final)
    if score < best_score:
        best_score = score
        best = a_final.copy()
    
    return best, best_score

def peak_shave(a_init, time_limit, rng):
    """Targeted peak-shaving with simulated annealing."""
    current = a_init.copy()
    S_target = np.sqrt(2.0 * len(current))
    current = current / np.sum(current) * S_target
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    
    t0 = time.monotonic()
    temp = 0.0005
    cooling = 0.9999
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))
        
        # Find top contributors to the peak
        contribs = np.zeros(n)
        for i in range(n):
            if 0 <= peak_idx - i < n:
                contribs[i] = current[i] * current[peak_idx - i]
        
        r = rng.random()
        if r < 0.4:
            # Reduce top contributors, increase bottom
            k = max(4, n // 16)
            top_idx = np.argsort(contribs)[::-1][:k]
            bottom_idx = np.argsort(contribs)[:k * 2]
            red = rng.uniform(0.03, 0.08)
            current[top_idx] *= (1 - red)
            current[bottom_idx] *= (1 + red * k / (2 * k))
        elif r < 0.6:
            # Pair swap
            k = max(4, n // 12)
            top_idx = np.argsort(contribs)[::-1][:k]
            j_red = top_idx[int(rng.integers(0, k))]
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red:
                amt = current[j_red] * rng.uniform(0.02, 0.06)
                current[j_red] = max(0, current[j_red] - amt)
                current[j_inc] += amt
        elif r < 0.8:
            # Block scaling
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, max(1, n - width)))
            factor = np.exp(rng.normal(0, 0.05))
            current[start:start + width] *= factor
        else:
            # Global small perturbation
            current = current * np.exp(rng.normal(0, 0.01, size=n))
        
        current = np.clip(current, 0, 1000)
        S_cur = np.sum(current)
        if S_cur < 0.01:
            current = best * np.exp(rng.normal(0, 0.1, size=n))
            current = np.clip(current, 0, 1000)
            current = current / np.sum(current) * S_target
            new_score = local_score(current)
            current_score = new_score
            temp *= cooling
            continue
        
        current = current / S_cur * S_target
        new_score = local_score(current)
        
        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current_score = new_score
        
        temp = max(temp * cooling, 1e-7)
    
    return best, best_score
```

- LP: max g0+g1+g2+g3 s.t. (f*g)[k] <= 4 for all k, g >= 0
- (f*g)[k] = sum of g's in the valid range for lag k
  - k=0: g0 <= 4
  - k=1: g0+g1 <= 4
  - k=2: g0+g1+g2 <= 4
  - k=3: g0+g1+g2+g3 <= 4
  - k=4: g1+g2+g3 <= 4
  - k=5: g2+g3 <= 4
  - k=6: g3 <= 4

The tightest constraint is k=3: g0+g1+g2+g3 <= 4. So max sum = 4, achieved by any g with sum=4. The LP doesn't give a unique solution!
