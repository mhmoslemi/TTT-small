```python
from scipy.optimize import linprog

def lp_direction(f, M):
    """Find g that maximizes sum(g) subject to g >= 0 and ||f*g||_inf <= M."""
    n = len(f)
    c = -np.ones(n)  # minimize -sum(g)
    
    # Constraints: (f*g)[k] <= M for all k = 0, ..., 2n-2
    # (f*g)[k] = sum_i f[i]*g[k-i] (valid i)
    # A_ub[row, col] represents constraint row: sum_j A_ub[row,j] * g[j] <= b_ub[row]
    
    # For each k, the constraint is sum_{i: 0<=i<n, 0<=k-i<n} f[i] * g[k-i] <= M
    # Let j = k-i, so i = k-j. The constraint is sum_{j: 0<=j<n, 0<=k-j<n} f[k-j] * g[j] <= M
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(k, n-1) + 1):
            A_ub[k, j] = f[k-j]
    
    b_ub = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.x
    return None
```

```python
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def max_conv_val(a):
    """Return max of conv(a,a)."""
    return np.max(fft_convolve(a, a))


def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build the convolution matrix A where (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_min:k - j_max + 1]
    return A


def lp_direction(a, max_time=5.0):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None if LP fails.
    """
    n = len(a)
    M = max_conv_val(a)
    if M < 1e-15:
        return None

    t0 = time.monotonic()
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': max_time})
    except Exception:
        return None

    if time.monotonic() - t0 > max_time * 1.5:
        return None

    if res.success:
        g = res.x
        s = np.sum(g)
        if s > 1e-10:
            return g / s
    return None


def smoothed_max_and_grad(a, tau):
    """Compute smoothed max of conv(a,a) and gradient w.r.t. a."""
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    value = max_c + tau * np.log(np.sum(exp_vals))

    # Gradient via FFT: grad ≈ 2 * cross_correlation(a, w)
    n = len(a)
    size = 2 * n
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    cross_corr = np.fft.irfft(np.conj(fa) * fw, size)[:n]
    grad = 2.0 * cross_corr

    return value, grad, conv


def adam_smoothed(a_init, time_limit, lr=1.0, tau_start=None, tau_end=None):
    """Adam optimization on smoothed max_conv, maintaining simplex."""
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    n = len(a)
    if tau_start is None:
        tau_start = 5.0 / n
    if tau_end is None:
        tau_end = 0.01 / n

    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t0 = time.monotonic()
    best_a = a.copy()
    best_val = max_conv_val(a)

    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / max(time_limit, 1e-10)
        tau = tau_start * (tau_end / tau_start) ** (elapsed ** 0.6)

        value, grad, conv = smoothed_max_and_grad(a, tau)

        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr_eff = lr / (1.0 + 0.01 * step ** 0.5)

        # Projected gradient step on simplex
        candidate = a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        candidate = project_simplex(candidate)

        a = candidate
        actual_val = max_conv_val(a)
        if actual_val < best_val:
            best_val = actual_val
            best_a = a.copy()

    return best_a, 2.0 * n * best_val


def coordinate_descent(a_init, time_limit, rng):
    """Coordinate descent / local search polish on the simplex."""
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    n = len(a)

    current_score = 2.0 * n * max_conv_val(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    patience = 0
    max_patience = 3000

    while time.monotonic() - t0 < time_limit and patience < max_patience:
        r = rng.random()

        if r < 0.35:
            # Pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.1)
            amt = a[i] * frac
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            new_score = 2.0 * n * max_conv_val(candidate)
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        elif r < 0.55:
            # Single element scale
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.05))
            candidate = a.copy()
            candidate[j] *= factor
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            new_score = 2.0 * n * max_conv_val(candidate)
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        elif r < 0.75:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.02, size=width))
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            new_score = 2.0 * n * max_conv_val(candidate)
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        else:
            # Peak-targeted: reduce elements at the peak location
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Elements contributing to conv[peak_idx]
            i_min = max(0, peak_idx - n + 1)
            i_max = min(peak_idx, n - 1)
            contrib = np.zeros(n)
            for i in range(i_min, i_max + 1):
                j = peak_idx - i
                contrib[i] = a[i] * a[j]
            top_k = max(3, n // 25)
            top_idx = np.argsort(contrib)[-top_k:]
            candidate = a.copy()
            candidate[top_idx] *= (1.0 - rng.uniform(0.005, 0.03, size=len(top_idx)))
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_add = min(5, len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                total_removed = np.sum(a[top_idx] - candidate[top_idx])
                candidate[add_idx] += total_removed / n_add
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            new_score = 2.0 * n * max_conv_val(candidate)
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

    return best_a, best_score


def fw_optimize(a_init, time_limit, use_lp=True, lp_max_time=3.0):
    """
    Frank-Wolfe optimization using LP direction finding.
    a_init should be on the simplex (sum=1, a>=0).
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    current_val = max_conv_val(a)
    best_a = a.copy()
    best_val = current_val
    t0 = time.monotonic()
    stagnation = 0
    max_stagnation = 200

    iteration = 0
    while time.monotonic() - t0 < time_limit and stagnation < max_stagnation:
        if use_lp:
            t_remaining = time_limit - (time.monotonic() - t0)
            if t_remaining < lp_max_time + 0.5:
                break
            g = lp_direction(a, max_time=lp_max_time)
            if g is None:
                # Fall back to uniform direction
                g = np.ones(n) / n
        else:
            g = np.ones(n) / n

        # Step size: use standard FW step size 2/(t+2) or diminishing
        iteration += 1
        t_fw = 2.0 / (iteration + 2.0)
        # Use a slightly aggressive step
        t_fw = min(1.0, t_fw * 1.5)

        candidate = (1.0 - t_fw) * a + t_fw * g
        candidate = np.clip(candidate, 0.0, None)
        cs = np.sum(candidate)
        if cs < 1e-10:
            break
        candidate = candidate / cs

        new_val = max_conv_val(candidate)
        if new_val < current_val:
            a = candidate
            current_val = new_val
            stagnation = 0
            if current_val < best_val:
                best_val = current_val
                best_a = a.copy()
        else:
            # Try smaller step
            t_small = t_fw * 0.5
            candidate2 = (1.0 - t_small) * a + t_small * g
            candidate2 = np.clip(candidate2, 0.0, None)
            cs2 = np.sum(candidate2)
            if cs2 > 1e-10:
                candidate2 = candidate2 / cs2
                new_val2 = max_conv_val(candidate2)
                if new_val2 < current_val:
                    a = candidate2
                    current_val = new_val2
                    stagnation = 0
                    if current_val < best_val:
                        best_val = current_val
                        best_a = a.copy()
                else:
                    stagnation += 1
            else:
                stagnation += 1

        if iteration % 50 == 0:
            print(f"        FW iter={iteration}, val={best_val:.8f}, "
                  f"score={2*n*best_val:.6f}, t={time.monotonic()-t0:.1f}s")

    return best_a, 2.0 * n * best_val


def spectral_init(n, rng, n_iter=50):
    """Spectral iterative initialization."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)

    for it in range(n_iter):
        conv = fft_convolve(a, a)
        # Reduce the peak
        peak = np.max(conv)
        alpha = (peak - np.mean(conv)) / (peak * (n + 1.0))
        a = a - alpha * conv[:n]
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s

    return a


def ramp_init(n, rng, style='linear'):
    """Create ramp-like initializations."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if style == 'linear':
        a = 1.0 + 0.3 * t
    elif style == 'quadratic':
        a = 1.0 + 0.4 * t ** 2
    elif style == 'sqrt':
        a = 1.0 + 0.5 * np.sqrt(t)
    elif style == 'sinusoidal':
        a = 1.0 + 0.3 * np.sin(np.pi * t)
    elif style == 'half':
        a = np.ones(n)
        a[n // 2:] *= 1.3
    elif style == 'triangle':
        a = 1.0 - 0.4 * np.abs(2 * t - 1)
    else:
        a = 1.0 + 0.2 * t
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP Frank-Wolfe + smoothed gradient + coordinate descent.
    """
    t_global = time.monotonic()
    time_budget = max(30.0, float(budget_s) - 60.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.001, 1000.0)
        s0 = score_fast(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s0:.6f}")
        if s0 < best_score:
            best_score = s0
            best_seq = start_seq.copy()

    def evaluate_and_track(a, label=""):
        nonlocal best_score, best_seq
        sc = score_fast(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** New best: {sc:.6f} ({label}) ***")
        return sc

    # Configuration: (n, n_fw_iters_budget, use_lp)
    configs = [
        (512, 0.35, True),
        (512, 0.25, True),
        (768, 0.25, True),
        (1024, 0.20, True),
    ]

    init_types = ['spectral', 'ramp_linear', 'ramp_sqrt', 'half', 'sinusoidal', 'triangle']

    print("\n=== Phase 1: LP Frank-Wolfe Optimization ===")
    total_p1_time = time_budget * 0.55

    for cfg_idx, (n, time_frac, use_lp) in enumerate(configs):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = time_budget * time_frac
        t_remain_for_phase = deadline - 180 - time.monotonic()
        t_alloc = min(t_alloc, max(10.0, t_remain_for_phase))

        n_inits = 2 if n <= 512 else 1
        t_per_init = t_alloc / (n_inits + 0.5)

        for init_i in range(n_inits):
            if time.monotonic() > deadline - 180:
                break

            if init_i == 0 and start_seq is not None and abs(n - len(start_seq)) < 100:
                a0 = start_seq.copy()
                init_label = "provided"
            else:
                itype = init_types[(cfg_idx + init_i) % len(init_types)]
                if itype == 'spectral':
                    a0 = spectral_init(n, rng, n_iter=40)
                else:
                    a0 = ramp_init(n, rng, itype)
                init_label = itype

            a0 = a0 / np.sum(a0)
            sc0 = 2 * n * max_conv_val(a0)
            print(f"  n={n}, init='{init_label}', initial score={sc0:.6f}")

            # FW optimization
            t_fw = min(t_per_init * 0.6, max(5.0, time_budget * 0.15))
            a_fw, score_fw = fw_optimize(a0, t_fw, use_lp=use_lp)
            print(f"    After FW: score={score_fw:.6f}")
            evaluate_and_track(a_fw, f"FW n={n} init={init_label}")

            # Smoothed gradient polish
            t_grad = min(t_per_init * 0.3, max(3.0, time_budget * 0.05))
            if time.monotonic() < deadline - 180 and t_grad > 2:
                a_grad, score_grad = adam_smoothed(a_fw, t_grad)
                print(f"    After grad: score={score_grad:.6f}")
                evaluate_and_track(a_grad, f"grad n={n} init={init_label}")

    # Phase 2: Coordinate descent polish on best
    print(f"\n=== Phase 2: Coordinate Descent Polish ===")
    print(f"  Best before polish: {best_score:.6f}")
    if best_seq is not None:
        polish_time = min(120.0, max(10.0, deadline - time.monotonic() - 60))
        print(f"  Polish budget: {polish_time:.0f}s")
        a_norm = best_seq / np.sum(best_seq)
        polished, sc_polished = coordinate_descent(a_norm, polish_time, rng)
        # Scale back
        polished_scaled = polished * np.sum(best_seq)
        sc_final = score_fast(polished_scaled)
        print(f"  After polish: {sc_final:.6f}")
        evaluate_and_track(polished_scaled, "polish")

    # Phase 3: Additional exploration with remaining time
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print(f"\n=== Phase 3: Final Exploration ===")
        remaining = deadline - time.monotonic() - 30
        n_trials = 3

        for trial in range(n_trials):
            if time.monotonic() > deadline - 30:
                break
            t_alloc = remaining / (n_trials - trial)
            n_try = [512, 640, 768, 1024][trial % 4]

            # Perturb the best sequence and re-optimize
            a_pert = best_seq.copy()
            a_pert += 0.05 * rng.standard_normal(len(a_pert))
            a_pert = np.clip(a_pert, 0.0, None)
            a_pert = a_pert / np.sum(a_pert)

            # Short FW + gradient
            t_sub = t_alloc * 0.5
            a_opt, score_opt = fw_optimize(a_pert, t_sub, use_lp=True, lp_max_time=2.0)
            t_sub2 = t_alloc * 0.3
            if time.monotonic() < deadline - 30:
                a_opt2, score_opt2 = adam_smoothed(a_opt, t_sub2)
            else:
                a_opt2, score_opt2 = a_opt, score_opt

            print(f"  Trial {trial}: n={n_try}, score={score_opt2:.6f}")
            evaluate_and_track(a_opt2, f"trial {trial}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
