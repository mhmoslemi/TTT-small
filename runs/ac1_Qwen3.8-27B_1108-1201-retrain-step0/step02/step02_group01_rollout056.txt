"""
Mirror descent (multiplicative weights) with soft subgradient for minimizing
the peak of the self-convolution of a non-negative sequence.

Key insights:
1. The subgradient of max_k conv(a,a)[k] at a is 2*a[k*-m] (reversed/shifted a)
2. Mirror descent with KL divergence gives multiplicative updates
3. Temperature annealing on the soft-max gives smooth -> sharp subgradients
4. Direct work on the simplex (no log-parameterization needed)
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def self_convolve(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_soft_subgradient(a, c, T):
    """Compute soft subgradient: g[m] = 2 * sum_k w[k] * a[k-m].
    
    Uses the identity: g/2 = cross_correlation(w, a) = conv(w, a[::-1])[n-1:n-1+n]
    """
    n = len(a)
    c_max = c.max()
    scaled = T * (c - c_max)
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    
    # g[m] = 2 * sum_k w[k] * a[k-m] (valid when 0<=k-m<n)
    # = 2 * (w cross-correlate a)[m]
    # = 2 * conv(w, a[::-1])[m + n - 1]
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    # full_conv has length (len(w) + n - 1) = (2n-1 + n - 1) = 3n-2
    # We need indices n-1 to 2n-2 (n values)
    end_idx = min(2 * n - 1, len(full_conv))
    g = 2.0 * np.zeros(n)
    g[:] = 2.0 * full_conv[n - 1: end_idx][:n]
    return g


def compute_hard_subgradient(a, c):
    """Compute hard subgradient at the argmax lag."""
    n = len(a)
    k_star = int(np.argmax(c))
    g = np.zeros(n)
    # g[m] = 2 * a[k_star - m] if 0 <= k_star - m < n
    valid = np.arange(n)
    idx = k_star - valid
    mask = (idx >= 0) & (idx < n)
    g[mask] = 2.0 * a[idx[mask]]
    return g


def mirror_descent_step(a, g, eta):
    """One step of mirror descent (multiplicative weights).
    
    a_new[j] = a[j] * exp(-eta * g[j]) / Z
    """
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - eta * g
    log_a_new = log_a_new - log_a_new.max()
    a_new = np.exp(log_a_new)
    a_new = a_new / a_new.sum()
    return a_new


def mirror_descent(a_init, time_budget, T_schedule=None, eta0=0.1, 
                   max_iter=None, verbose=False):
    """Run mirror descent with soft subgradient and temperature annealing."""
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    
    start = time.monotonic()
    c = self_convolve(a)
    best_obj = c.max() * 2 * n  # objective with sum(a)=1
    best_a = a.copy()
    
    if max_iter is None:
        max_iter = 10**9
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute current objective
        c = self_convolve(a)
        cur_obj = c.max() * 2 * n
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 10000 == 0:
                print("  iter %d: obj=%.8f" % (it, best_obj))
        
        # Temperature schedule
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        if T_schedule == 'linear':
            T = 1.0 + 200.0 * progress
        elif T_schedule == 'exp':
            T = 1.0 * np.exp(3.0 * progress)  # 1 to ~20
        elif T_schedule == 'fast_exp':
            T = 1.0 * np.exp(5.0 * progress)  # 1 to ~148
        else:
            T = 10.0
        
        # Compute subgradient
        if T < 500:
            g = compute_soft_subgradient(a, c, T)
        else:
            g = compute_hard_subgradient(a, c)
        
        # Step size (diminishing)
        eta = eta0 / np.sqrt(it + 1)
        
        # Mirror descent update
        a_new = mirror_descent_step(a, g, eta)
        
        # Check if update is valid
        if not np.all(np.isfinite(a_new)) or a_new.sum() < 1e-10:
            break
        
        a = a_new
    
    return best_a, best_obj


def mirror_descent_with_momentum(a_init, time_budget, eta0=0.05, 
                                  T0=5.0, T1=200.0, verbose=False):
    """Mirror descent with Nesterov-like momentum."""
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    
    # Momentum buffer (in log-space)
    log_a = np.log(np.maximum(a, 1e-300))
    momentum = np.zeros(n)
    
    start = time.monotonic()
    c = self_convolve(a)
    best_obj = c.max() * 2 * n
    best_a = a.copy()
    
    for it in range(10**9):
        if time.monotonic() - start > time_budget:
            break
        
        c = self_convolve(a)
        cur_obj = c.max() * 2 * n
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T0 * np.exp(np.log(T1 / T0) * progress)
        
        g = compute_soft_subgradient(a, c, T)
        
        eta = eta0 / np.sqrt(it + 1)
        
        # Update in log-space with momentum
        grad_log = -eta * g  # gradient of log-objective w.r.t. log_a
        momentum = 0.9 * momentum + 0.1 * grad_log
        log_a = log_a + momentum
        
        # Project to simplex in original space
        a = np.exp(log_a - log_a.max())
        a = a / a.sum()
        
        if not np.all(np.isfinite(a)):
            break
    
    return best_a, best_obj


def coordinate_descent_polish(a, time_budget, max_sweeps=100):
    """Coordinate descent polish: try to reduce each coordinate."""

    c = self_convolve(a)

    best_a = a.copy()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            orig = a[i]
            if orig < 1e-12:
                continue
            
            # Try reducing a[i] (this tends to help if a[i] contributes to the peak)
            # Binary search for the best value
            lo, hi = 0.0, orig
            best_val = orig
            best_local = best_obj
            c_check = self_convolve(a)

                if s < 1e-10:

"""
Mirror descent (multiplicative weights) with soft subgradient for minimizing
2*n*max(conv(a,a))/sum(a)^2.

Key insights:
1. The subgradient of max_k conv(a,a)[k] at a is 2*a[k*-m] (reversed/shifted a)
2. Mirror descent with KL divergence gives multiplicative updates that naturally
   maintain non-negativity
3. Temperature annealing on the soft-max goes from smooth exploration to sharp
   exploitation
4. Random restarts escape local minima
5. Direct work on the probability simplex (no log-parameterization)
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def self_convolve(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_soft_subgradient(a, c, T):
    """Compute soft subgradient: g[m] = 2 * sum_k w[k] * a[k-m].
    
    Uses FFT: g[m] = 2 * conv(w, a[::-1])[m + n - 1]
    """
    n = len(a)
    c_max = c.max()
    scaled = T * (c - c_max)
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()

    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    end_idx = min(2 * n - 1, len(full_conv))
    g = np.zeros(n)
    g[:] = 2.0 * full_conv[n - 1: end_idx][:n]
    return g

    n = len(a)

    idx = k_star - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g[mask] = 2.0 * a[idx[mask]]
    return g

    best_a = a.copy()
    last_improve = 0
    stagnation_counter = 0

    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        c = self_convolve(a)
        cur = c.max()
        if cur < best_obj - 1e-12:
            best_obj = cur
            best_a = a.copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T0 * np.exp(np.log(max(T1, T0 + 1e-6) / T0) * progress)

        if T < 500:
            g = compute_soft_subgradient(a, c, T)
        else:
            g = compute_hard_subgradient(a, c)

        eta = eta0 / np.sqrt(it + 1)

        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30 or not np.all(np.isfinite(a_new)):
            break
        a = a_new / a_new_sum

    return best_a, best_obj


def mirror_descent_with_restarts(a_init, time_budget, rng, num_phases=3):
    """Mirror descent with temperature annealing and random restarts."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = self_convolve(a).max()

    phase_time = time_budget / num_phases

    phases = [
        (1.0, 30.0, 0.8),
        (30.0, 200.0, 0.4),
        (200.0, 1000.0, 0.15),
        (1000.0, 5000.0, 0.05),
    ]

    for phase in range(min(num_phases, len(phases))):
        if time.monotonic() > time_budget - time.monotonic() + time_budget - phase_time * (num_phases - 1 - phase):
            pass
        T0, T1, eta0 = phases[phase]
        remaining = min(phase_time, time_budget - (time.monotonic() - (time_budget - 0)))

        # Use actual time tracking
        elapsed_so_far = 0
        for restart in range(3):
            phase_elapsed = 0.0
            t0 = time.monotonic()
            a_sub, obj_sub = mirror_descent_core(a, phase_time, T0, T1, eta0)
            phase_elapsed = time.monotonic() - t0

            if obj_sub < best_obj:
                best_obj = obj_sub
                best_a = a_sub.copy()
                a = a_sub.copy()
            elif obj_sub < best_obj + 1e-6:
                # Small improvement or stagnation - continue with current
                a = a_sub.copy()
            else:
                # No improvement - random restart
                if restart < 2:
                    # Perturb and continue
                    noise = rng.random(n) * 0.1
                    a_pert = a * (1.0 + 0.05 * (noise - 0.5))
                    a_pert = np.maximum(a_pert, 0.0)
                    a_pert /= a_pert.sum()
                    a = a_pert

    return best_a, best_obj


def coordinate_descent_polish(a, time_budget, n_inner_steps=8):
    """Coordinate descent: iteratively reduce the coordinate that most contributes to the peak."""
    a = a.copy().astype(np.float64)
    np.maximum(a, 0.0, out=a)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)

    start = time.monotonic()
    c = self_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        c = self_convolve(a)
        k_peak = int(np.argmax(c))

        # Contribution of each coordinate to the peak
        idx = k_peak - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]]

        # Try the top few contributors
        top_k = min(16, n)
        top_indices = np.argsort(contrib)[::-1][:top_k]

        for i in top_indices:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue

            # Binary search for best reduction
            lo, hi = 0.0, orig
            best_val = orig
            best_local = best_obj

            for _ in range(n_inner_steps):
                mid = (lo + hi) / 2.0
                a[i] = mid
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_norm = a / s
                c_new = self_convolve(a_norm)
                obj_new = c_new.max()
                if obj_new < best_local:
                    best_local = obj_new
                    best_val = mid
                    a = a_norm
                else:
                    a[i] = best_val
                    a = a / a.sum()
                    hi = mid
                    break
            else:
                a[i] = best_val
                a = a / a.sum()

            if best_local < best_obj:
                best_obj = best_local
                best_a = a.copy()
                c = self_convolve(a)
                k_peak = int(np.argmax(c))
                idx = k_peak - np.arange(n)
                mask = (idx >= 0) & (idx < n)
                contrib = np.zeros(n)
                contrib[mask] = a[idx[mask]]

    return best_a, best_obj


def get_initialization(name, n, rng):
    """Generate various initializations (normalized to sum=1)."""
    if name == "uniform":
        return np.ones(n) / n
    elif name == "random_smooth":
        x = rng.random(n)
        from scipy.ndimage import gaussian_filter1d
        x = gaussian_filter1d(x, sigma=n / 20.0)
        x = np.maximum(x, 1e-10)
        return x / x.sum()
    elif name == "cosine_low":
        i = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * i / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif name == "cosine_high":
        i = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 11 * i / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif name == "anti_sym":
        i = np.arange(n)
        a = 1.0 + 0.4 * (i / n - 0.5)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif name == "bipolar":
        i = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * i / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif name == "flat_center":
        i = np.arange(n)
        a = np.ones(n)
        mid = n // 2
        width = n // 4
        a[mid - width:mid + width] = 1.5
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function.
    
    Uses mirror descent (multiplicative weights) with soft subgradient,
    temperature annealing, random restarts, and coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    global_best = np.inf
    global_a = None

    # Load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 64 and given_raw.sum() > 0.01:
            given_a = given_raw.copy()
            n_given = len(given_raw)
            ver = evaluate_sequence(given_raw.tolist())
            print("Given sequence: n=%d, obj=%.6f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_raw.copy()
    except Exception as e:
        print("Could not load given sequence: %s" % str(e))

    # Determine primary length
    if given_a is not None:
        primary_n = len(given_a)
    else:
        primary_n = 4096

    # Phase 1: Optimization at primary length
    n = primary_n
    a_work = np.ones(n) / n
    if given_a is not None and len(given_a) == n:
        a_work = given_a / given_a.sum()

    # Try multiple initializations
    init_names = ["uniform", "random_smooth", "cosine_low", "cosine_high",
                  "anti_sym", "bipolar", "flat_center"]

    # Allocate time
    explore_time = 0.25 * (total_deadline - time.monotonic())
    main_time = 0.55 * (total_deadline - time.monotonic())
    polish_time = 0.20 * (total_deadline - time.monotonic())

    # Exploration: quick trials with different inits
    print("Phase 1: Exploration (t=%.1fs budget)" % explore_time)
    per_init_explore = explore_time / len(init_names)
    for init_name in init_names:
        if time.monotonic() > total_deadline - 60.0:
            break
        a_init = get_initialization(init_name, n, rng)
        t0 = time.monotonic()
        a_sub, obj_sub = mirror_descent_core(a_init, min(per_init_explore, 30.0),
                                             T0=1.0, T1=50.0, eta0=1.0)
        elapsed = time.monotonic() - t0
        obj_full = obj_sub * 2 * n  # objective with sum=1
        if obj_full < global_best:
            global_best = obj_full
            global_a = (a_sub * np.sqrt(2.0 * n)).copy()
            a_work = a_sub.copy()
            print("  Explore %s: obj=%.6f" % (init_name, obj_full))

    # Main optimization: long mirror descent on best so far
    print("Phase 2: Main optimization (t=%.1fs budget)" % main_time)
    a_main = a_work.copy()
    best_obj_main = self_convolve(a_main).max() * 2 * n
    best_a_main = a_main.copy()

    # Multi-phase with increasing temperature
    main_phases = [
        (1.0, 20.0, 1.5, 0.3),
        (20.0, 100.0, 0.8, 0.25),
        (100.0, 500.0, 0.4, 0.2),
        (500.0, 2000.0, 0.2, 0.2),
        (2000.0, 10000.0, 0.1, 0.15),
    ]

    for phase_idx, (T0, T1, eta0, frac) in enumerate(main_phases):
        if time.monotonic() > total_deadline - 30.0:
            break
        phase_budget = main_time * frac

        # Try multiple restarts within this phase
        num_restarts = 2
        per_restart = phase_budget / num_restarts

        for restart in range(num_restarts):
            if time.monotonic() > total_deadline - 30.0:
                break
            t0 = time.monotonic()
            if restart > 0:
                # Perturb
                noise = rng.random(n)
                a_pert = a_main * (1.0 + 0.1 * (noise - 0.5))
                a_pert = np.maximum(a_pert, 0.0)
                a_pert /= a_pert.sum()
                a_sub, obj_sub = mirror_descent_core(a_pert, per_restart, T0, T1, eta0)
            else:
                a_sub, obj_sub = mirror_descent_core(a_main, per_restart, T0, T1, eta0)

            elapsed = time.monotonic() - t0
            obj_full = obj_sub * 2 * n
            if obj_full < global_best:
                global_best = obj_full
                global_a = (a_sub * np.sqrt(2.0 * n)).copy()
                best_a_main = a_sub.copy()
                best_obj_main = obj_sub
                print("  Phase %d restart %d: obj=%.6f (T: %.0f->%.0f, t=%.1fs)" %
                      (phase_idx, restart, obj_full, T0, T1, elapsed))
            a_main = a_sub if obj_sub < best_obj_main else a_main

    # Phase 3: Polish with coordinate descent
    print("Phase 3: Coordinate descent polish (t=%.1fs budget)" % polish_time)
    if time.monotonic() < total_deadline - 10.0:
        a_cd, obj_cd = coordinate_descent_polish(best_a_main, polish_time, n_inner_steps=10)
        obj_cd_full = obj_cd * 2 * n
        if obj_cd_full < global_best:
            global_best = obj_cd_full
            global_a = (a_cd * np.sqrt(2.0 * n)).copy()
            print("  Coord descent: obj=%.6f" % obj_cd_full)
        else:
            # Try CD on the global best
            a_best_norm = global_a / global_a.sum() if global_a is not None else best_a_main
            a_cd2, obj_cd2 = coordinate_descent_polish(a_best_norm, polish_time * 0.5, n_inner_steps=10)
            obj_cd2_full = obj_cd2 * 2 * n
            if obj_cd2_full < global_best:
                global_best = obj_cd2_full
                global_a = (a_cd2 * np.sqrt(2.0 * n)).copy()
                print("  Coord descent (alt): obj=%.6f" % obj_cd2_full)

        # Final short mirror descent on the polished result
        remaining = total_deadline - time.monotonic() - 10.0
        if remaining > 5.0 and global_a is not None:
            a_final = global_a / global_a.sum()
            a_fin, obj_fin = mirror_descent_core(a_final, remaining,
                                                 T0=1000.0, T1=5000.0, eta0=0.05)
            obj_fin_full = obj_fin * 2 * n
            if obj_fin_full < global_best:
                global_best = obj_fin_full
                global_a = (a_fin * np.sqrt(2.0 * n)).copy()
                print("  Final MD: obj=%.6f" % obj_fin_full)

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("FINAL: computed=%.6f, verified=%.6f, n=%d" %
          (global_best, ver if np.isfinite(ver) else float('inf'), len(global_a)))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.6f" % ver)

    return final_a
