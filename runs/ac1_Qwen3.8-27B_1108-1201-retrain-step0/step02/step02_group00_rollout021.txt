import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_objective(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_c = c.max()
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * max_c / (s * s)

def run_fista_persistent(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=1000.0):

    - Adaptive step size with backtracking
    - Periodic restart with perturbations

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(0.003 * it))
        
        c = fft_convolve_self(y)

        if cur < best_score:

            best_local = y.copy()

        if total < 1e-300:

        w = exp_bc / total

        W = fft(w, fsize)

Minimize 2*n*max_conv/sum^2 over non-negative sequences using
FISTA with adaptive log-sum-exp smoothing, peak shaving,
and multi-scale optimization.

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_c = float(c.max())
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return 2.0 * n * max_c / (s * s)

def normalize_sum(a, S):
    """Rescale a so that sum(a) = S, maintaining non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def peak_shave(a_init, n, S, max_time, rng, top_k=5, step_scale=0.5):
    """
    Direct peak shaving: target the top-k peak lags with exact subgradient.
    """
    a = normalize_sum(a_init, S).copy()
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    conv_size = 2 * n - 1
    fsize_conv = 1 << (conv_size - 1).bit_length()
    fsize_grad = 1 << (conv_size + n - 2).bit_length()
    
    while time.monotonic() - start < max_time:
        A = fft(a, fsize_conv)
        c = np.real(ifft(A * A))[:conv_size]
        cur = float(c.max())
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Top-k peak lags
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-15:
            break
        
        # Weights proportional to peak values
        w = np.zeros(conv_size)
        for i, k in enumerate(top_idx):
            w[int(k)] = top_vals[i] / total
        
        # Gradient
        W = fft(w, fsize_grad)
        A2 = fft(a, fsize_grad)
        grad = 2.0 * np.real(ifft(W * np.conj(A2)))[:n]
        
        gmax = float(np.abs(grad).max())
        if gmax < 1e-14:
            break
        
        # Step size
        step = step_scale * S / (n * gmax)
        
        a_new = project_simplex(a - step * grad, S)
        
        # Check improvement
        A_new = fft(a_new, fsize_conv)
        c_new = np.real(ifft(A_new * A_new))[:conv_size]
        
        if c_new.max() < cur * (1 + 1e-10):
            a = a_new
        else:
            # Try smaller step
            a_new = project_simplex(a - 0.5 * step * grad, S)
            a = a_new
        
        # Occasional noise
        if rng.random() < 0.05:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
    
    return best_local, best_score
