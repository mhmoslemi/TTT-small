def fft_convolve(a, b):
    n = len(a)

def _fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:

    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    c = _fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

    e = np.exp(t)
    return e / np.sum(e)

    Since softmax gives sum=1, this is just 2n * pnorm."""

    a = _softmax(theta)
    c = _fft_convolve(a, a)
    cp = c ** (p - 1)

    cross = _fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)

def _coordinate_descent(a, time_limit, rng, use_block=True):
    """
    Fast greedy coordinate descent with optional block moves.
    Uses exact FFT convolution for each evaluation.
    """
    current = a.astype(np.float64).copy()
    n = len(current)
    current_score = _local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()

    deltas = np.array([-0.3, -0.15, -0.07, -0.03, 0.03, 0.07, 0.15, 0.3])
    # Also try larger moves occasionally
    big_deltas = np.array([-0.6, -0.4, 0.4, 0.6])

    sweep = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved_this_sweep = False
        order = rng.permutation(n)

        for i in order:
            ai = current[i]
            if ai < 1e-12:
                continue
            r = rng.random()
            if r < 0.1 and use_block:
                # Block move: perturb a small block around i
                half = int(rng.integers(2, max(3, n // 10)))
                lo = max(0, i - half)
                hi = min(n, i + half + 1)
                candidate = current.copy()
                candidate[lo:hi] *= np.exp(rng.normal(0, 0.08, hi - lo))
                candidate = np.maximum(candidate, 1e-12)
                s = _local_score(candidate)
                if s < current_score - 1e-12:
                    current = candidate
                    current_score = s
                    improved_this_sweep = True
                continue

            # Single coordinate: try deltas
            best_d = 0.0
            best_s = current_score
            trial_deltas = deltas if rng.random() > 0.05 else np.concatenate([deltas, big_deltas])
            for d in trial_deltas:
                new_val = ai * (1.0 + d)
                if new_val < 0:
                    continue
                candidate = current.copy()
                candidate[i] = new_val
                s = _local_score(candidate)
                if s < best_s - 1e-14:
                    best_s = s
                    best_d = d
            if best_d != 0.0:
                current[i] *= (1.0 + best_d)
                current_score = best_s
                improved_this_sweep = True

        if improved_this_sweep:
            no_improve_count = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            no_improve_count += 1
            if no_improve_count > 30:
                # Restart with perturbation
                current = best * np.exp(rng.normal(0, 0.05, n))
                current = np.maximum(current, 1e-10)
                current_score = _local_score(current)
                no_improve_count = 0

        if sweep % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    coord: sweep={sweep}, cur={current_score:.6f}, best={best_score:.6f}, t={elapsed:.0f}s")

    # Final exact check
    final_score = _local_score(best)
    if final_score < best_score:
        best_score = final_score
    return best, best_score

    """
    current = a.astype(np.float64).copy()
    n = len(current)
    current_score = _local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = _fft_convolve(current, current)
        c_max = np.max(c)
        peak_idx = int(np.argmax(c))

        # Compute contribution of each a[i] to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = current[j]

        # Reduce high contributors, increase low contributors
        mean_c = np.mean(c)
        excess = np.maximum(c - 0.9 * c_max, 0)
        # For all peaks, compute pull
        pull = np.zeros(n)
        for k in range(len(c)):
            weight = max(0.0, c[k] - 0.95 * c_max)
            if weight > 0:
                for i in range(min(n, k + 1)):
                    j = k - i
                    if 0 <= j < n:
                        pull[i] += weight * current[j]

        if np.max(pull) > 1e-14:
            pull = pull / np.max(pull)
            candidate = current * np.exp(-eta * pull)
            candidate = np.maximum(candidate, 1e-12)
            s = _local_score(candidate)
            if s < current_score - 1e-14:
                current = candidate
                current_score = s
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()

        # Also try small random perturbations
        if rng.random() < 0.3:
            candidate = current * np.exp(rng.normal(0, 0.01, n))
            candidate = np.maximum(candidate, 1e-12)
            s = _local_score(candidate)
            if s < current_score - 1e-14:
                current = candidate
                current_score = s
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    peak_shave: it={it}, score={current_score:.6f}, best={best_score:.6f}, t={elapsed:.0f}s")

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: L-BFGS p-norm, soft-max, coordinate descent, peak shaving.
    Explores diverse initializations and refines the best candidate.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 40.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        s = _local_score(start_seq)
        print(f"Starting sequence: n={len(start_seq)}, score={s:.6f}")
        best_score = s
        best_seq = start_seq.copy()

    n_main = 256

    # Interpolate to main length if needed
    if start_seq is not None and len(start_seq) != n_main:
        start_seq = _make_interp(n_main, start_seq, rng, 0.0)
        s = _local_score(start_seq)
        print(f"Interpolated to n={n_main}: score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # === Phase 1: L-BFGS p-norm on starting sequence ===
    print(f"\n=== Phase 1: L-BFGS p-norm refinement (score={best_score:.6f}) ===")
    if start_seq is not None and time.monotonic() < deadline - 800:
        theta = np.log(start_seq + 1e-30)
        p_schedule = [8, 16, 32, 64, 128, 256, 512]
        t1_dead = min(t_global + 180, deadline - 800)
        theta_best, s = _run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p=400, deadline=t1_dead)
        a_best = _softmax(theta_best)
        s = _local_score(a_best)
        if s < best_score:
            best_score = s
            best_seq = a_best.copy()
            print(f"  Phase 1 improved to: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Phase 1: {s:.6f} (no improvement)")

    # === Phase 2: Soft-max L-BFGS refinement ===
    if best_seq is not None and time.monotonic() < deadline - 700:
        print(f"\n=== Phase 2: Soft-max L-BFGS (score={best_score:.6f}) ===")
        theta = np.log(best_seq + 1e-30)
        tau_schedule = [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002]
        t2_dead = min(t_global + 350, deadline - 700)
        theta_best, s = _run_lbgfs_softmax_obj(theta, tau_schedule, maxiter_per_tau=300, deadline=t2_dead)
        a_best = _softmax(theta_best)
        s = _local_score(a_best)
        if s < best_score:
            best_score = s
            best_seq = a_best.copy()
            print(f"  Soft-max improved to: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Soft-max: {s:.6f} (no improvement)")

    # === Phase 3: Multi-start L-BFGS with diverse inits ===
    print(f"\n=== Phase 3: Multi-start L-BFGS (score={best_score:.6f}) ===")
    inits = []
    if best_seq is not None:
        inits.append(('best_n5', _make_interp(n_main, best_seq, rng, 0.05)))
        inits.append(('best_n10', _make_interp(n_main, best_seq, rng, 0.10)))
        inits.append(('best_n15', _make_interp(n_main, best_seq, rng, 0.15)))
    inits.append(('gauss_c', _make_gaussian_init(n_main, n_main * 0.5, n_main * 0.25, rng)))
    inits.append(('gauss_l', _make_gaussian_init(n_main, n_main * 0.35, n_main * 0.2, rng)))
    inits.append(('gauss_r', _make_gaussian_init(n_main, n_main * 0.65, n_main * 0.2, rng)))
    inits.append(('sine_f2', _make_sine_init(n_main, 2, 0.0, rng)))
    inits.append(('sine_f3', _make_sine_init(n_main, 3, np.pi / 3, rng)))
    inits.append(('lopsided04', _make_lopsided_init(n_main, 0.4, rng)))
    inits.append(('lopsided06', _make_lopsided_init(n_main, 0.6, rng)))
    inits.append(('chirp', _make_chirp_init(n_main, 0.5, rng)))
    inits.append(('rand1', np.maximum(1.0 + 0.4 * rng.standard_normal(n_main), 0.01)))
    inits.append(('rand2', np.maximum(1.0 + 0.7 * rng.standard_normal(n_main), 0.01)))

    # Allocate time per init
    remaining_for_phase3 = deadline - time.monotonic() - 900
    if remaining_for_phase3 > 60:
        time_per_init = max(15.0, remaining_for_phase3 / len(inits))
        p_sched_short = [16, 32, 64, 128]
        for name, a0 in inits:
            if time.monotonic() > deadline - 900:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / np.sum(a0)
            theta = np.log(a0 + 1e-30)
            t_init_dead = time.monotonic() + time_per_init
            theta_best, s = _run_lbgfs_pnorm(theta, p_sched_short, maxiter_per_p=200, deadline=t_init_dead)
            a_best = _softmax(theta_best)
            s = _local_score(a_best)
            if s < best_score:
                best_score = s
                best_seq = a_best.copy()
                print(f"  '{name}': {s:.6f} *** NEW BEST ***")
            else:
                print(f"  '{name}': {s:.6f}")

    # === Phase 4: Coordinate descent (main search phase) ===
    print(f"\n=== Phase 4: Coordinate descent (score={best_score:.6f}) ===")
    if best_seq is not None:
        t4_dead = deadline - 250
        t4_time = max(60.0, t4_dead - time.monotonic())
        print(f"  Time allocated: {t4_time:.0f}s")
        polished, p_score = _coordinate_descent(best_seq, t4_time, rng)
        p_score = _local_score(polished)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  Coord descent improved to: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Coord descent: {p_score:.6f}")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak shaving (score={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        t5_time = min(150.0, deadline - time.monotonic() - 100)
        if t5_time > 20:
            polished, p_score = _peak_shaving(best_seq, t5_time, rng, eta=0.03)
            p_score = _local_score(polished)
            if p_score < best_score:
                best_score = p_score
                best_seq = polished.copy()
                print(f"  Peak shaving improved to: {best_score:.6f} *** NEW BEST ***")
            else:
                print(f"  Peak shaving: {p_score:.6f}")

    # === Phase 6: Final L-BFGS + coord descent polish ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        # Quick L-BFGS
        theta = np.log(best_seq + 1e-30)
        t6_dead = min(time.monotonic() + 100, deadline - 50)
        theta_best, s = _run_lbgfs_pnorm(theta, [64, 128, 256], maxiter_per_p=200, deadline=t6_dead)
        a_best = _softmax(theta_best)
        s = _local_score(a_best)
        if s < best_score:
            best_score = s
            best_seq = a_best.copy()
            print(f"  Final L-BFGS: {best_score:.6f} *** NEW BEST ***")

        # Final coord descent
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic() - 10
            if remaining > 20:
                polished, p_score = _coordinate_descent(best_seq, remaining, rng)
                p_score = _local_score(polished)
                if p_score < best_score:
                    best_score = p_score
                    best_seq = polished.copy()
                    print(f"  Final coord: {best_score:.6f} *** NEW BEST ***")

    # Final verification and return
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
