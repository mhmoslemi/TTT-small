spectral initialization, adaptive softmax-Adam optimization, and

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def objective_value(a):
    """Compute the true objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def a_from_theta(theta, S):
    """Convert log-parameters to sequence: a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total

    # sum_k w[k] * a[k-m]: let j = k-m, so k = j+m: sum_j w[j+m] * a[j] = (w_shifted conv a)[?]
    # This is getting confusing. Let me just use the FFT approach from the previous code.
    
    n = len(a)
    a_rev = a[::-1]

    # where (w conv a_rev) has length (len(w) + n - 1) = (2n-1 + n - 1) = 3n-2
    out_len = len(w) + n - 1

    return grad

def softmax_grad(theta, n, T, S):
    """Compute softmax-smoothed objective gradient w.r.t. theta."""
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    
    if c_max < 1e-15:
        return c_max, np.zeros(n)
    
    # Softmax weights
    scaled = T * (c - c_max) / c_max  # normalize by c_max for better conditioning
    scaled = scaled - scaled.max()  # max is 0
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    
    # Gradient of smoothed max w.r.t. a
    grad_a = compute_grad_for_active(w, a)
    
    # Chain rule: dL/dtheta[j] = grad_a[j]*a[j] - (a[j]/S)*sum_i grad_a[i]*a[i]
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    
    obj = c_max + np.log(exp_scaled.sum()) / T
    return obj, grad_theta

def hard_topk_grad(theta, n, k, S):
    """Compute subgradient using top-k positions."""
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    
    grad_a = compute_grad_for_active(w, a)
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    
    return c_max, grad_theta

    theta = theta_init.copy().astype(np.float64)
    n_par = len(theta)
    
    m = np.zeros(n_par)
    v = np.zeros(n_par)

    best_theta = theta.copy()
    
    # Track progress for adaptive LR

    lr_base = 0.1

        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        a = a_from_theta(theta, S)
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj - 1e-10:
            best_obj = cur
            best_a = a.copy()
            best_theta = theta.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

            lr_base *= 0.9
            no_improve_count = 0

        if t % 500 == 0 and t > 0:

            noise = np.exp(np.random.normal(0, 0.1, n_par))
            noise = noise / noise.mean()  # preserve the "center"
            theta = theta + 0.3 * (np.log(noise) - np.log(noise).mean())

        gnorm = np.linalg.norm(grad)

            lr = lr_base * (1.0 - 0.5 * progress)

                break

            for factor in [0.5, 0.7, 0.85, 1.1, 1.3, 1.5]:
                a[i] = orig * factor

                if s < 1e-10:
                    a[i] = orig
                    break

                if obj_check < best_local:

                best_obj = c.max()
                best_a = a.copy()

            break

def spectral_init(n, rng, S):
    """Generate initialization from random phases (flat spectrum approach)."""
    phases = rng.uniform(0, 2 * np.pi, n)
    # Random magnitude with some structure
    magnitudes = 1.0 + 0.3 * rng.random(n)
    X = magnitudes * np.exp(1j * phases)
    # Ensure Hermitian symmetry for real result
    if n % 2 == 0:
        X[n//2] = np.abs(X[n//2])  # Nyquist
        X[n//2+1:] = np.conj(X[n//2-1::-1])
    else:
        X[(n//2+1):] = np.conj(X[n//2::-1])
    a = np.real(ifft(X))
    # Apply softplus-like transformation for non-negativity
    a = np.exp(a / (np.abs(a).max() + 1e-10))  # exponential to ensure positive
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    return a

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using multi-length search with
adaptive softmax-Adam optimization, multiplicative noise restarts,
deflation-based peak reduction, and coordinate descent polishing.

Key innovations over previous approach:
1. Multiplicative noise restarts (exponential family perturbations) for escaping local minima
2. Deflation step: explicitly reduce peak-contributing coordinates
3. Aggressive temperature annealing with randomization
4. Multi-length search (4096, 8192, 16384) for better exploitation of degrees of freedom
5. Reversal averaging as a symmetry exploitation heuristic
6. Better time management with proportional allocation

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def objective_value(a):
    """Compute the true objective value (unnormalized)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def a_from_theta(theta, S):
    """Convert log-parameters to sequence: a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def compute_grad_from_weights(w, a):
    """Compute grad[m] = 2 * sum_k w[k] * (a[k-m] + a[k+m]) using FFT.
    
    Uses the identity: grad[m] = 2 * (w * a_rev)[m + n - 1]
    where * denotes convolution and a_rev is the reversed sequence.
    """
    n = len(a)
    a_rev = a[::-1]
    out_len = len(w) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    Ar = fft(a_rev, fsize)
    conv_result = np.real(ifft(W * Ar))[:out_len]
    end_idx = min(2 * n - 1, out_len)
    grad = 2.0 * conv_result[n - 1:n]
    return grad


def softmax_objective_grad(theta, T, S):
    """Compute softmax-smoothed max of self-conv and gradient w.r.t. theta."""
    n = len(theta)
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    
    if c_max < 1e-15:
        return c_max, np.zeros(n)
    
    # Softmax weights with temperature
    scaled = T * (c - c_max) / (c_max + 1e-30)
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    
    grad_a = compute_grad_from_weights(w, a)
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    
    obj = c_max + np.log(exp_scaled.sum() + 1e-30) / (T + 1e-30)
    return obj, grad_theta


def topk_subgradient(theta, k, S):
    """Compute subgradient using top-k active positions."""
    n = len(theta)
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    
    idx = np.argsort(c)[::-1][:max(1, k)]
    w = np.zeros(len(c))
    w[idx] = 1.0 / len(idx)
    
    grad_a = compute_grad_from_weights(w, a)
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    
    return c_max, grad_theta


def deflation_step(a, c, S, strength=0.1):
    """Reduce peak-contributing coordinates and redistribute.
    
    For the position k* where c is maximized, the gradient w.r.t. a[i] is
    a[k*-i] + a[k*+i]. We reduce high-gradient coordinates and boost low ones.
    """
    n = len(a)
    k_star = int(np.argmax(c))
    
    # Compute contribution of each coordinate to c[k_star]
    contrib = np.zeros(n)
    for i in range(n):
        left = k_star - i
        right = k_star + i
        if 0 <= left < n:
            contrib[i] += a[left]
        if 0 <= right < n:
            contrib[i] += a[right]
    
    # Reduce high-contribution, boost low-contribution (multiplicatively)
    contrib_mean = contrib.mean()
    if contrib_mean < 1e-15:
        return a
    
    # Scale: high contributors get reduced, low contributors get boosted
    ratio = contrib / contrib_mean
    # Apply gentle deflation
    adjustment = 1.0 - strength * (ratio - 1.0)
    adjustment = np.maximum(adjustment, 0.1)  # Don't reduce by more than 90%
    
    a_new = a * adjustment
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new * (S / s)
    return a_new


def run_optimizer(theta_init, n, S, time_budget, T_start=5.0, T_end=500.0,
                  lr_base=0.08, use_deflation=True):
    """Main Adam optimizer with adaptive temperature and periodic restarts."""
    theta = theta_init.copy().astype(np.float64)

    v = np.zeros(n_par)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    start = time.monotonic()
    
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    best_theta = theta.copy()
    
    no_improve = 0
    patience = 200
    
    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Temperature schedule: exponential with randomization
        T = T_start * np.exp(np.log(T_end / T_start) * progress)
        T *= (0.7 + 0.6 * np.random.random())
        
        # Choose gradient method based on temperature
        if T < 80.0:
            _, grad = softmax_objective_grad(theta, T, S)
        else:
            k = max(4, int(2 + 6 * progress))
            _, grad = topk_subgradient(theta, k, S)
        
        # Evaluate true objective
        a = a_from_theta(theta, S)
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj - 1e-12:
            best_obj = cur
            best_a = a.copy()
            best_theta = theta.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic deflation step
        if use_deflation and t % 200 == 0:
            a_deflated = deflation_step(a, c, S, strength=0.05)
            a_deflated = np.maximum(a_deflated, 1e-10)
            a_deflated = a_deflated * (S / a_deflated.sum())
            c_def = fft_convolve_self(a_deflated)
            if c_def.max() < best_obj:
                best_obj = c_def.max()
                best_a = a_deflated.copy()
                theta = np.log(a_deflated / a_deflated.sum())
                m *= 0.5
                v *= 0.5
                no_improve = 0

        if no_improve > patience:
            noise_scale = 0.1 + 0.2 * progress
            log_noise = np.random.normal(0, noise_scale, n_par)
            log_noise -= log_noise.mean()
            theta = theta + log_noise
            m *= 0.3
            v *= 0.3
            no_improve = 0
            patience = max(50, patience - 20)
        
        # Adam update with adaptive LR
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr_base * (1.0 - 0.3 * progress)
            theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
    
    return best_theta, best_a, best_obj


def reversal_blend(a, S, alpha=0.5):
    """Blend a sequence with its reversal (exploits convolution symmetry)."""
    a_rev = a[::-1].copy()
    a_blend = (1 - alpha) * a + alpha * a_rev
    a_blend = np.maximum(a_blend, 1e-10)
    a_blend = a_blend * (S / a_blend.sum())
    return a_blend

def spectral_init(n, rng, S):
    """Flat-spectrum inspired initialization."""
    # Random phases with Hermitian symmetry
    mags = 1.0 + 0.5 * rng.random(n)
    phases = rng.uniform(0, 2 * np.pi, n)
    X = mags * np.exp(1j * phases)
    # Enforce Hermitian symmetry
    if n % 2 == 0:
        X[n // 2] = np.abs(X[n // 2])
        X[n // 2 + 1:] = np.conj(X[n // 2 - 1::-1])
    else:
        X[(n // 2) + 1:] = np.conj(X[n // 2::-1])
    a = np.real(ifft(X))
    # Softplus for non-negativity
    a = np.log1p(np.exp(a))
    a = np.maximum(a, 1e-8)
    a = a * (S / a.sum())
    return a


def power_iter_init(n, rng, S, iters=50):
    """Initialize via power iteration on the convolution operator.
    
    The idea: start random, repeatedly convolve and normalize.
    This converges to the dominant eigenvector of the convolution operator,
    which has interesting autocorrelation properties.
    """
    a = rng.random(n) + 0.5
    a = a * (S / a.sum())
    for _ in range(iters):
        c = fft_convolve_self(a)
        # Take the "middle" n values and use as new a
        mid = c[n // 2:n // 2 + n]
        mid = np.maximum(mid, 1e-10)
        a = mid * (S / mid.sum())
    return a
