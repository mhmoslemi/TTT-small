import numpy as np
from scipy.optimize import linprog
from scipy.optimize import minimize_scalar
import time

def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0, 1000)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (s ** 2)

def solve_lp(f, S):
    """
    Solve: min t s.t. (f*g)_k <= t for all k, sum(g) = S, g >= 0
    Returns optimal g and t.
    """
    n = len(f)
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: (f*g)_k - t <= 0 for k = 0..2n-2
    # A @ x <= b
    A = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(k+1, n)):
            A[k, i] = f[k-i]
        A[k, n] = -1.0  # -t
    b_ineq = np.zeros(2*n-1)
    
    # Equality: sum(g) = S
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: g_i >= 0, t free
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        g = res.x[:n]
        t = res.x[n]

The alternating approach works as follows:
1. Find g that minimizes peak(f*g)
2. Find f that minimizes peak(g*f) = peak(f*g) (same thing by commutativity)
3. Now f and g both minimize peak with respect to each other
4. If f = g, we're at a local minimum of peak(f*f)

But in general, f ≠ g after the alternating steps. The score of f (which uses f*f) might be worse than the score of g. So I should track both and keep the better one.

Actually, let me think about this differently. The score of a is 2n * max(a*a) / (sum a)^2. After step 1, g minimizes max(f*g), not max(g*g). After step 2, f minimizes max(g*f) = max(f*g). 

The key insight is: if we're at a fixed point where f = g = a*, then a* minimizes max(a* * a) over a (with constraints), which is a necessary condition for a* to minimize max(a*a).

But we might not reach a fixed point. In that case, we can:
1. Use the best of all f and g values seen (evaluate their f*f or g*g scores)
2. Use the average of f and g as a starting point for further optimization

"""
Optimizes non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses alternating LP minimization and coordinate descent.
"""
import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time

def fft_convolve(a, b=None):
    """Compute linear convolution of two 1-D arrays using FFT."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]

def compute_score(a):
    """Fast computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * np.max(c) / (s * s))

def build_conv_matrix(f):
    """Build matrix A where A[k,i] = f[k-i] if 0<=k-i<n else 0."""
    n = len(f)
    k = np.arange(2 * n - 1)[:, None]
    i = np.arange(n)[None, :]
    idx = k - i
    valid = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = f[idx[valid]]
    return A

def solve_lp(f, S):
    """
    Solve: minimize t subject to (f*g)_k <= t for all k, sum(g) = S, g >= 0.
    Returns optimal g.
    """
    n = len(f)
    A_conv = build_conv_matrix(f)
    A_ub = np.empty((2 * n - 1, n + 1))
    A_ub[:, :n] = A_conv
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        g = np.clip(res.x[:n], 0.0, None)
        gs = g.sum()
        if gs > 0:
            g = g * S / gs
        return g
    # Fallback: return f (no improvement)
    return f.copy()

def alternating_lp(a_init, max_iter=500, time_limit=600.0):
    """
    Alternating LP: alternate between finding g = argmin max(f*g) and 
    f = argmin max(g*f).
    """
    t0 = time.monotonic()
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 0.01:
        return a, np.inf
    a = a * (S / S)  # already normalized
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    
    print(f"  AltLP start: n={n}, score={best_score:.6f}")
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        g = solve_lp(a, S)
        a_new = solve_lp(g, S)
        
        # Evaluate both
        for cand in [a_new, g]:
            cand = np.clip(cand, 0.0, 1000.0)
            cs = cand.sum()
            if cs > 0.01:

                    if it % 20 == 0:

def coordinate_descent(a_init, max_sweeps=200, time_limit=600.0):
    """
    Coordinate descent: for each coordinate, find optimal value
    that minimizes max of the convolution.
    """
    t0 = time.monotonic()
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0).copy()
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return a, np.inf
    
    best_a = a.copy()
    best_score = compute_score(a)
    c = fft_convolve(a)
    current_max = np.max(c)
    
    print(f"  CoordDescent start: n={n}, score={best_score:.6f}, max_conv={current_max:.4f}")
    
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            # Find optimal x for a[i]
            old_val = a[i]
            # Compute base convolution (a[i] = 0)
            # Affected k range: [i, i+n-1]
            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)
            
            # Base c values (contribution from all terms except those involving a[i])
            # For k = 2i: base = c[2i] - old_val^2
            # For k in [i, i+n-1], k != 2i: base = c[k] - 2*old_val*a[k-i]
            # For other k: base = c[k]

            # where c_k(x) = base_k + 2*x*a[k-i] (for k != 2i, i <= k < i+n)
            #                          = base_k + x^2 (for k = 2i)
            #                          = base_k (for other k)

            # From c_2i(x) = base_2i + x^2 <= current_max → x <= sqrt(current_max - base_2i)
            # From c_k(x) = base_k + 2*x*a[k-i] <= current_max → x <= (current_max - base_k)/(2*a[k-i])
            
            base_2i = c[2*i] - old_val * old_val
            ub = current_max  # general upper bound
            if base_2i < current_max:
                ub = min(ub, np.sqrt(current_max - base_2i))
            for k in range(k_min, k_max + 1):
                if k == 2 * i:
                    continue
                coef = 2.0 * a[k - i]
                if coef > 1e-15:
                    base_k = c[k] - 2.0 * old_val * a[k - i]
                    if base_k < current_max:
                        ub = min(ub, (current_max - base_k) / coef)
                    else:
                        ub = 0.0
                        break
            
            if ub < 0:
                ub = 0.0
            ub = max(ub, old_val * 2.0)  # allow some room
            ub = min(ub, 1000.0)
            
            def obj(x):
                # Compute max of c_k(x)
                # Only need to check k in [k_min, k_max]
                # and the global max of unchanged c values
                unchanged_max = 0.0
                # For k outside [k_min, k_max], c_k is unchanged
                if k_min > 0:
                    unchanged_max = max(unchanged_max, np.max(c[:k_min]))
                if k_max < 2*n-2:
                    unchanged_max = max(unchanged_max, np.max(c[k_max+1:]))
                
                # For k in [k_min, k_max]:
                local_max = unchanged_max
                for k in range(k_min, k_max + 1):
                    if k == 2 * i:
                        val = base_2i + x * x
                    else:
                        val = c[k] - 2.0 * old_val * a[k-i] + 2.0 * x * a[k-i]
                    if val > local_max:
                        local_max = val
                return local_max
            
            # Minimize obj(x) for x in [0, ub]
            if ub > 0:
                res = minimize_scalar(obj, bounds=(0, ub), method='bounded', 
                                      options={'xatol': 1e-10})
                if res.success:
                    a[i] = max(0, res.x)
                    improved = True
            
            # Update c
            delta = a[i] - old_val
            if abs(delta) > 1e-15:
                c[2*i] += delta * (a[i] + old_val)  # x^2 - old^2 = (x-old)(x+old)
                for k in range(k_min, k_max + 1):
                    if k != 2 * i:
                        c[k] += 2.0 * delta * a[k - i]
            
            old_val = a[i]  # update for next iteration (shouldn't need but safety)

        current_max = np.max(c)
        sc = 2.0 * n * current_max / (S * S)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            if sweep % 10 == 0:
                print(f"    sweep {sweep}: score={best_score:.6f}")
        
        if not improved and sweep > 0:
            # Try to check if we really haven't improved
            pass
    
    return best_a, best_score

"""
Optimizes non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses alternating LP minimization with multiple initializations and coordinate descent refinement.
Key insight: the linear (not circular) autocorrelation can be made flatter than the 
triangular shape of uniform sequences, allowing scores below 2.0.

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b=None):
    """Compute linear convolution of two 1-D arrays using FFT."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]


def compute_score(a):
    """Fast computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * np.max(c) / (s * s))


def build_conv_matrix(f):
    """Build matrix A where A[k,i] = f[k-i] if 0<=k-i<n else 0."""
    n = len(f)
    k = np.arange(2 * n - 1)[:, None]
    i = np.arange(n)[None, :]
    idx = k - i
    valid = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = f[idx[valid]]
    return A


def solve_lp(f, S):
    """
    Solve: minimize t subject to (f*g)_k <= t for all k, sum(g) = S, g >= 0.
    Returns optimal g.
    """
    n = len(f)
    A_conv = build_conv_matrix(f)
    A_ub = np.empty((2 * n - 1, n + 1))
    A_ub[:, :n] = A_conv
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        g = np.clip(res.x[:n], 0.0, None)
        gs = g.sum()
        if gs > 1e-12:
            g = g * S / gs
        return g
    return f.copy()


def solve_lp_relaxed(f, S, slack=0.1):

    Relaxed LP: minimize t subject to (f*g)_k <= t, S*(1-slack) <= sum(g) <= S*(1+slack), g >= 0.
    This allows the sum to vary, giving more flexibility.
    """
    n = len(f)
    A_conv = build_conv_matrix(f)
    A_ub = np.empty((2 * n - 1, n + 1))
    A_ub[:, :n] = A_conv
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((2, n + 1))
    A_eq[0, :n] = 1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n] = 1.0  # sum(g) + t ... no this doesn't work
    # Let me use inequality for the sum bounds instead
    A_ub2 = np.vstack([
        A_ub,
        np.vstack([-np.ones(n), np.zeros(1)]).T,  # -sum(g) <= -S*(1-slack)
        np.vstack([np.ones(n), np.zeros(1)]).T,   # sum(g) <= S*(1+slack)
    ])
    b_ub2 = np.concatenate([b_ub, [-S * (1 - slack), S * (1 + slack)]])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A_ub2, b_ub=b_ub2, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        g = np.clip(res.x[:n], 0.0, None)
        gs = g.sum()
        if gs > 1e-12:
            g = g * S / gs
        return g
    return f.copy()

        for i in range(n):
            old_val = a[i]
            if old_val < 1e-10:
                continue
            
            others_mask = np.arange(n) != i
            others_sum = S - old_val
            if others_sum < 1e-10:
                continue
            
            # Try removing a[i] and redistributing
            a[i] = 0.0

    def try_candidate(a, label=""):
        nonlocal best_score, best_seq
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST ({label}): score={best_score:.6f}, n={len(a)}")
    
    # Try different sequence lengths
    for n in [256, 512]:
        if time.monotonic() > deadline - 30:
            break
        print(f"\n{'='*50}")
        print(f"  n = {n}")
        print(f"{'='*50}")
        
        # Generate multiple initializations
        inits = []
        
        # 1. Perturbed uniform
        for eps in [0.2, 0.5, 1.0]:
            a = np.ones(n) + eps * rng.standard_normal(n)
            a = np.clip(a, 0.01, None)
            inits.append((f'perturbed_{eps}', a))
        
        # 2. Cosine modulation (various frequencies)
        x = np.arange(n)
        for freq in [1, 2, 3, 5]:
            for amp in [0.3, 0.5, 0.7]:
                a = 1.0 + amp * np.cos(2 * np.pi * freq * x / n)
                a = np.clip(a, 0.01, None)
                inits.append((f'cos_f{freq}_a{amp}', a))
        
        # 3. Two-tone (alternating)
        a = np.ones(n)
        a[::2] = 1.5
        a[1::2] = 0.5
        inits.append(('two_tone', a.copy()))
        
        a = np.ones(n)
        a[::2] = 2.0
        a[1::2] = 0.5
        inits.append(('two_tone2', a.copy()))
        
        # 4. Triangular
        a = 1.0 - np.abs(x - n/2) / (n/2)
        a = np.clip(a, 0.01, None)
        inits.append(('triangular', a))
        
        # 5. Exponential decay from center
        a = np.exp(-np.abs(x - n/2) / (n/4))
        inits.append(('gaussian', a))
        
        # 6. Random
        for _ in range(3):
            a = rng.random(n) + 0.1
            inits.append(('random', a.copy()))
        
        # 7. Provided sequence if available
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            hs = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            if len(hs) == n and hs.sum() > 0.01:
                inits.append(('provided', hs.copy()))
        
        # For each initialization, run alternating LP
        num_inits = len(inits)
        time_per_init = max(30.0, (deadline - time.monotonic()) / num_inits)
        
        for init_name, a0 in inits:
            if time.monotonic() > deadline - 30:
                break
            
            # Quick check
            sc0 = compute_score(a0)
            try_candidate(a0, f"{init_name}_init")
            
            # Run alternating LP
            lp_time = min(time_per_init - 10, 200.0)
            if lp_time < 20:
                continue
            
            a_lp, score_lp = alternating_lp(a0, max_iter=200, time_limit=lp_time)
            try_candidate(a_lp, f"{init_name}_lp")
            
            # Try relaxed LP
            time_left = deadline - time.monotonic()
            if time_left > 60:
                lp_time2 = min(100.0, time_left * 0.3)
                a_lr, score_lr = alternating_lp(a0, max_iter=100, time_limit=lp_time2, use_relaxed=True)
                try_candidate(a_lr, f"{init_name}_lp_relaxed")
        
        # After LP phase, do stochastic CD on the best so far
        if best_seq is not None and time.monotonic() < deadline - 30:
            time_left = deadline - time.monotonic()
            a_cd, score_cd = cd_stochastic(best_seq, max_iters=300, time_limit=min(200.0, time_left * 0.5), seed=seed)
            try_candidate(a_cd, "stochastic_cd")
            
            # Also try simple CD
            if time.monotonic() < deadline - 30:
                time_left = deadline - time.monotonic()
                a_cd2, score_cd2 = cd_simple(best_seq, max_sweeps=50, time_limit=min(100.0, time_left * 0.3))
                try_candidate(a_cd2, "simple_cd")
    
    # If we have a best sequence, try one more round of LP on it
    if best_seq is not None and time.monotonic() < deadline - 30:
        time_left = deadline - time.monotonic()
        n = len(best_seq)
        a_final, score_final = alternating_lp(best_seq, max_iter=100, time_limit=min(200.0, time_left * 0.7))
        try_candidate(a_final, "final_lp")
    
    # Final answer
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = 2.0
    
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_list = [float(x) for x in best_seq]
    final_score = compute_score(best_seq)
    print(f"\n{'='*50}")
    print(f"  FINAL: score={final_score:.6f}, n={len(final_list)}")
    print(f"  Time used: {time.monotonic() - t_global:.1f}s")
    print(f"{'='*50}")
    
    return final_list
