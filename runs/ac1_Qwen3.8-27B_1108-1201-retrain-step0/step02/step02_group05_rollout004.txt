    """Solve the LP to find g that maximizes sum(g) subject to g>=0 and conv(f,g) <= M."""
    from scipy.optimize import linprog
    n = len(f)
    # Maximize sum(g) = minimize -sum(g)
    c = -np.ones(n)
    # Constraints: conv(f, g)[k] <= M for all k
    # conv(f,g)[k] = sum_i f[i] * g[k-i] for valid i
    # This is a linear function of g
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = f[i]
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:

    return None

"""
Optimization of non-negative sequences to minimize peak self-convolution.
Uses LP-based alternating optimization (core method), L-BFGS p-norm refinement,
and targeted local search. The key insight is that fixing one sequence in the
convolution makes it linear in the other, enabling exact LP solutions for
the "flattest" compatible partner sequence.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for sequence: 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def build_conv_matrix(f, n):
    """Build matrix A such that A @ g = conv(f, g)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for i in range(n):
            A[i + j, j] = f[i]
    return A


def lp_find_partner(f, M, n=None):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and conv(f, g) <= M.
    Returns optimal g (unnormalized) or None if infeasible.
    """
    if n is None:
        n = len(f)
    c = -np.ones(n)
    A = build_conv_matrix(f, n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None


def lp_alternating(a_init, max_iters=500, blend_rate=0.3, deadline=None, verbose=True):
    """
    LP-based alternating optimization.
    At each step: fix f, find g maximizing sum(g) s.t. conv(f,g) <= max(conv(f,f)).
    Blend and repeat.
    """
    f = a_init.astype(np.float64).copy()
    f = f / np.sum(f)
    n = len(f)
    current_score = local_score(f)
    best = f.copy()
    best_score = current_score
    t0 = time.monotonic()

    if verbose:
        print(f"    LP start: score={current_score:.6f}")

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        M = np.max(fft_convolve(f, f))
        g = lp_find_partner(f, M, n)
        if g is None:
            break

        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break

        g = g / g_sum  # normalize to sum=1
        f_new = (1.0 - blend_rate) * f + blend_rate * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / np.sum(f_new)

        new_score = local_score(f_new)

        if new_score < best_score:
            best_score = new_score
            best = f_new.copy()

        if new_score < current_score:
            f = f_new
            current_score = new_score
        else:
            # Try smaller blend rate
            f_new2 = (1.0 - blend_rate * 0.3) * f + blend_rate * 0.3 * g
            f_new2 = np.maximum(f_new2, 0.0)
            f_new2 = f_new2 / np.sum(f_new2)
            new_score2 = local_score(f_new2)
            if new_score2 < current_score:
                f = f_new2
                current_score = new_score2
            else:
                # No improvement, try to reduce peak directly
                # Use peak shaving: solve LP with slightly lower M
                M_shave = M * 0.999
                g_shave = lp_find_partner(f, M_shave, n)
                if g_shave is not None:
                    g_shave = g_shave / np.sum(g_shave)
                    f_new3 = (1.0 - blend_rate * 0.5) * f + blend_rate * 0.5 * g_shave
                    f_new3 = np.maximum(f_new3, 0.0)
                    f_new3 = f_new3 / np.sum(f_new3)
                    new_score3 = local_score(f_new3)
                    if new_score3 < current_score:
                        f = f_new3
                        current_score = new_score3
                    else:
                        # Stuck, break
                        if verbose and it > 20:
                            print(f"    LP stuck at iter {it}, score={current_score:.6f}")
                        break
                else:
                    if verbose and it > 20:
                        print(f"    LP infeasible at iter {it}, breaking")
                    break

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: score={current_score:.6f}, best={best_score:.6f}, t={elapsed:.0f}s")

        # Convergence check
        if it > 10 and abs(current_score - best_score) < 1e-10:
            break

    if verbose:
        print(f"    LP done: best_score={best_score:.6f}")

    return best, best_score


def pnorm_obj_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p / (sum)^2.
    Since softmax sums to 1, denominator is always 1.
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def soft_max_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def run_lbgfs(a0, p_schedule, maxiter_per_p, deadline, verbose=True):
    """Run L-BFGS with annealed p-norm on softmax parameterization."""
    n = len(a0)
    theta = np.log(a0 + 1e-30)
    best_score = local_score(a0)
    best_a = a0.copy()

    for p in p_schedule:
        if time.monotonic() > deadline - 10:
            break
        result = minimize(
            pnorm_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(p,),
            options={'maxiter': maxiter_per_p, 'ftol': 1e-15, 'gtol': 1e-12}
        )
        a_new = softmax(result.x)
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / np.sum(a_new)
        s = local_score(a_new)
        if s < best_score:
            best_score = s
            best_a = a_new.copy()
            theta = result.x.copy()
        if verbose:
            print(f"      p={p}: score={s:.6f}, best={best_score:.6f}")

    return best_a, best_score

def peak_targeted_local_search(a_init, time_limit, rng, verbose=True):
    """
    Local search with moves specifically targeted at reducing the convolution peak.
    Uses the structure: identify which pairs (i,j) contribute most to the peak,
    and reduce them.
    """
    current = a_init.astype(np.float64).copy()
    current = current / np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.001
    cooling = 0.9999
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        peak_val = np.max(c)
        peak_idx = int(np.argmax(c))

        # Find top contributing pairs to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = current[i] * current[j]

        r = rng.random()

        if r < 0.40:
            # Reduce top contributors to peak
            candidate = current.copy()
            top_k = max(3, min(n // 8, 20))
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.008, size=len(top_idx)))
            # Increase some other values to maintain "spread"
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.008, size=n_inc))
        elif r < 0.60:
            # Reduce based on full convolution profile (smoothed)
            candidate = current.copy()
            c_avg = np.mean(c)
            excess = np.maximum(c - c_avg, 0)
            # Compute pull for each element
            pull = np.zeros(n)
            for i in range(n):
                for k in range(min(n, peak_idx + 1)):
                    j = k - i
                    if 0 <= j < n:
                        pull[i] += excess[k] * current[j]
            pull = pull / (np.max(pull) + 1e-20)
            candidate *= np.exp(-0.015 * pull)
        elif r < 0.75:
            # Block move
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - width))
            candidate = current.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.90:
            # Partial symmetrization (breaks symmetry which helps!)
            candidate = current.copy()
            # Mix with anti-symmetric component
            anti = 0.5 * (current - current[::-1])
            candidate = current + 0.1 * rng.normal() * anti
        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 1e-10)

        if time.monotonic() - last_print > 5.0:
            last_print = time.monotonic()
            if verbose:
                print(f"      polish: score={current_score:.6f}, best={best_score:.6f}, "
                      f"t={time.monotonic()-t0:.0f}s")

    return best, best_score


def make_diverse_init(n, rng, mode='mixed'):
    """Generate diverse initializations."""
    if mode == 'asymmetric':
        # Strongly asymmetric
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-2.0 * t / n)
    elif mode == 'chirp':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * t * t / (2 * n))
    elif mode == 'bump':
        t = np.arange(n, dtype=np.float64)
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(0.1, 0.3) * n
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
    elif mode == 'sine':
        t = np.arange(n, dtype=np.float64)
        freq = rng.uniform(1, 3)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * freq * t / n + phase)
    else:  # mixed
        r = rng.random()
        if r < 0.25:
            a = np.exp(-2.0 * np.arange(n) / n)
        elif r < 0.5:
            t = np.arange(n, dtype=np.float64)
            a = 1.0 + 0.4 * np.cos(2 * np.pi * t * t / (2 * n))
        elif r < 0.75:
            t = np.arange(n, dtype=np.float64)
            center = rng.uniform(0.2, 0.8) * n
            width = rng.uniform(0.1, 0.3) * n
            a = np.exp(-0.5 * ((t - center) / width) ** 2)
        else:
            a = np.ones(n)
            a[:n//3] *= 1.5
            a[-n//3:] *= 0.7

    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)

            best_seq = a_refined.copy()
            print(f"  L-BFGS p-norm improved: {best_score:.6f} *** NEW BEST ***")

        # Soft-max refinement
        if time.monotonic() < phase2_deadline - 30:
            tau_sched = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
            a_sm, s = run_lbgfs_softmax_obj(best_seq, tau_sched, maxiter_per_tau=200,
                                            deadline=phase2_deadline, verbose=True)
            if s < best_score:
                best_score = s
                best_seq = a_sm.copy()
                print(f"  Soft-max improved: {best_score:.6f} *** NEW BEST ***")

    # ============================================================
    # Phase 3: LP refinement on best sequence (second pass with smaller blend)
    # ============================================================
    print(f"\n=== Phase 3: LP refinement with small blend (score={best_score:.6f}) ===")
    if best_seq is not None:
        n = len(best_seq)
        phase3_deadline = min(time.monotonic() + 200, deadline - 150)
        result, score = lp_alternating(best_seq, max_iters=300, blend_rate=0.1,
                                       deadline=phase3_deadline, verbose=True)
        if score < best_score:
            best_score = score
            best_seq = result.copy()
            print(f"  LP pass2 improved: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  LP pass2: no improvement (score={score:.6f})")

    # ============================================================
    # Phase 4: Peak-targeted local search
    # ============================================================
    print(f"\n=== Phase 4: Peak-targeted local search (score={best_score:.6f}) ===")
    if best_seq is not None:
        polish_time = min(200.0, max(30.0, deadline - time.monotonic() - 20))
        print(f"  Polish time: {polish_time:.0f}s")
        polished, p_score = peak_targeted_local_search(best_seq, polish_time, rng, verbose=True)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  After polish: {p_score:.6f} (no improvement)")

    # ============================================================
    # Phase 5: Final verification and return
    # ============================================================
    print(f"\n=== Final Verification ===")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
