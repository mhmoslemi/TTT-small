from scipy.sparse import lil_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    while time.monotonic() - t0 < time_limit:
        grad = compute_gradient_fft(a, conv, temp)
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            a_new = a - trial_lr * grad
            a_new = project_simplex(a_new)
            new_conv = fast_convolve(a_new)
            new_obj = 2.0 * n * new_conv.max()
            if new_obj < cur_obj:
                a = a_new
                cur_obj = new_obj
                conv = new_conv
                lr = trial_lr
                no_improve = 0
                break
            elif new_obj < best_obj:
                best_obj = new_obj
                best_a = a_new.copy()
        else:
            no_improve += 1
            lr *= 0.5
            temp *= 1.2
            if no_improve > 50:
                break
    
    # Final check
    if cur_obj < best_obj:
        best_obj = cur_obj
        best_a = a.copy()
    
    return best_a, best_obj

def local_search(a0, time_limit, rng, verbose=False):
    """Simple local search with random perturbations."""
    t0 = time.monotonic()
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    a = a / a.sum()
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        # Random perturbation: move mass from one position to another
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        frac = rng.uniform(0.01, 0.08)
        delta = a[i] * frac
        if delta < 1e-12:
            continue
        a[i] -= delta
        a[j] += delta
        if a[i] < 0:
            a[i] += delta
            a[j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        
        # Accept if better, or with small probability (annealing)
        frac_time = (time.monotonic() - t0) / time_limit
        T = 0.01 * (1 - frac_time) ** 2 + 0.0005
        if new_obj < cur_obj or rng.random() < np.exp(-(new_obj - cur_obj) / T):
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            a[i] += delta
            a[j] -= delta
    
    return best_a, best_obj

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian (various widths)
    for sig in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n / 2) / (sig * n)) ** 2)
        g = np.maximum(g, 0)
        inits['gauss_%d' % int(sig * 100)] = g / g.sum()
    
    # Sine modulated
    for freq in [1, 2, 4]:
        s = 0.5 * (1 + np.cos(2 * np.pi * freq * x / n))
        inits['sin_%d' % freq] = s / s.sum()
    
    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    # Anti-ramp (high at edges)
    ar = np.ones(n) - r / r.sum()
    ar = np.maximum(ar, 0.01)
    inits['anti_ramp'] = ar / ar.sum()
    
    # Bimodal
    bi = np.zeros(n)
    w = n // 5
    if w > 0:
        bi[:w] = 1.0
        bi[-w:] = 1.0
        inits['bimodal'] = bi / bi.sum()
    
    # Flat-top window (Kaiser-Bessel approximation)
    for beta in [3.0, 5.0, 8.0]:
        ft = np.zeros(n)
        for i in range(n):
            t = 2.0 * i / n - 1.0
            ft[i] = np.i0(beta * np.sqrt(max(0.0, 1.0 - t * t)))
        ft = np.maximum(ft, 1e-10)
        inits['flat_%d' % int(beta)] = ft / ft.sum()
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        inits['dirichlet_%d' % int(alpha)] = rng.dirichlet(np.ones(n) * alpha)
    
    # Parent
    if parent is not None:
        p = np.maximum(parent, 0).astype(np.float64)
        if p.size >= n:
            p = p[:n]
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            inits['parent'] = p / ps
    
    return inits
