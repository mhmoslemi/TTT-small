def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score(a):
    """Compute the evaluation score."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

        lambda th: objective_and_grad(th)[0],

            'maxiter': n_inner,
            'maxfun': n_inner * 3,

        if s_new < best_score:

    s_final = score(a_final)

            i_min = max(0, lag - n + 1)
            i_max = min(n - 1, lag)
            for i in range(i_min, i_max + 1):
                j = lag - i
                contrib[i] += weight * current[i] * current[j]

        n_reduce = max(3, n // 20)
        top_idx = np.argsort(contrib)[-n_reduce:]
        bot_idx = np.argsort(contrib)[:n_reduce]
        
        candidate = current.copy()
        candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=n_reduce))
        candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_reduce))
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s > 1e-10:

            current_score = new_score

                best = current.copy()

            if delta < temp * 5:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    current = candidate
                    current_score = new_score

def upsample(a, new_n, rng):
    """Upsample via interpolation + noise."""
    n = len(a)
    if new_n <= n:
        return a[:new_n].copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)

"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions. Key insight: score = 2n*max(conv(a,a))/sum(a)^2 is scale-invariant,
so we normalize sum(a)=1 and minimize 2n*max(conv(a,a)). Uses FFT-based gradient
descent with annealed soft-max surrogate, L-BFGS-B refinement, peak-directed
local search, and coarse-to-fine upsampling across multiple sequence lengths.

import numpy as np

def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score(a):
    """Compute the evaluation score: 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def softmax_params(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def soft_max_value_weights(x, tau):
    """Compute soft-max value and weights."""
    x_shifted = (x - np.max(x)) / tau
    e = np.exp(x_shifted)
    w = e / np.sum(e)
    val = np.max(x) + tau * np.log(np.sum(e))
    return val, w


def compute_grad_theta(theta, tau):
    """
    Compute gradient of 2n*softmax_tau(conv(a,a)) w.r.t. theta,
    where a = softmax(theta) (sum=1).
    Returns (value, grad_theta).
    """
    n = len(theta)
    a = softmax_params(theta)
    c = fft_convolve(a, a)
    val, w = soft_max_value_weights(c, tau)

    conv_w_arev = fft_convolve(w, a_rev)

    dot_a_ga = np.dot(a, grad_a)
    grad_theta = a * (grad_a - dot_a_ga)

    return 2.0 * n * target_val, grad_theta * 2.0 * n

        s = np.sum(a)

        else:

        elif r < 0.7:
            # Symmetry exploitation: blend with reversed
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.2 * (sym - current)
            candidate += 0.005 * rng.standard_normal(n)

        elif r < 0.85:
            # Multiplicative noise (small)
            candidate = current * np.exp(rng.normal(0.0, 0.02, size=n))
        else:
            # Targeted: find the single worst lag and reduce its contributors
            c = fft_convolve(current, current)
            worst_lag = int(np.argmax(c))
            i_min = max(0, worst_lag - n + 1)
            i_max = min(n - 1, worst_lag)
            idx = np.arange(i_min, i_max + 1)
            if len(idx) > 0:
                candidate = current.copy()
                contrib_at_worst = current[idx] * current[worst_lag - idx]
                worst_idx = idx[np.argsort(contrib_at_worst)[-max(3, len(idx) // 4):]]
                candidate[worst_idx] *= np.exp(-0.05 * rng.uniform(0.5, 1.5, size=len(worst_idx)))
                # Increase some low-contribution elements
                other_idx = np.setdiff1d(np.arange(n), worst_idx)
                n_inc = min(5, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(0.05 * rng.uniform(0.5, 1.5, size=n_inc))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / s * np.sum(current)

        new_score = score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if temp > min_temp and delta < temp * 10:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    current = candidate
                    current_score = new_score

        temp = max(temp * cooling, min_temp)

    return best, best_score


def upsample(a, new_n, rng, noise_level=0.01):
    """Upsample via interpolation + small noise."""
    n = len(a)
    if new_n <= n:
        return a[:new_n].copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += noise_level * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search for optimal non-negative step function minimizing
    2n*max(conv(a,a))/sum(a)^2.
    """
    t_global = time.monotonic()
    time_budget = float(budget_s) - 30.0
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence if available
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Phase 1: Generate and screen initializations
    print("\n=== Phase 1: Initialization ===")
    candidates = []

    lengths_to_try = [256, 384, 512, 768, 1024, 1536, 2048]

    for n in lengths_to_try:
        if time.monotonic() > deadline - 120:
            break

        inits = []

        # FFT phase init (2 variants)
        a1 = fft_phase_init(n, rng, n_iter=80)
        inits.append(('fft1', a1))

        a2 = fft_phase_init(n, rng, n_iter=150)
        inits.append(('fft2', a2))

        # Weighted FFT phase
        a3 = fft_phase_init_weighted(n, rng, n_iter=100, decay=0.2 + 0.3 * rng.random())
        inits.append(('fft_w', a3))

        # Structured
        a4 = structured_init(n, rng, n_tones=4 + rng.integers(0, 4))
        inits.append(('struct', a4))

        # Smooth random
        a5 = random_smooth_init(n, rng, smoothness=2.0 + 4.0 * rng.random())
        inits.append(('smooth', a5))

        for name, a in inits:
            s = score(a)
            candidates.append((s, n, a.copy(), name))
            if s < best_score:
                best_score = s
                best_seq = a.copy()

    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates")
    print(f"  Top 5: {[(f'{c[0]:.5f}', c[1], c[3]) for c in candidates[:5]]}")
    print(f"  Current best: {best_score:.6f}")

    # Phase 2: Gradient optimization on top candidates
    print("\n=== Phase 2: Gradient optimization ===")
    n_cand = min(10, len(candidates))
    time_left = deadline - time.monotonic() - 300
    time_per = max(5.0, time_left / max(1, n_cand))

    optimized_results = []

    for i in range(n_cand):
        if time.monotonic() > deadline - 150:
            break

        score0, n, a, name = candidates[i]

        # Allocate more time to smaller (faster) sequences
        t_alloc = time_per * min(1.5, 1024.0 / n)
        t_alloc = max(5.0, min(t_alloc, deadline - time.monotonic() - 100))

        print(f"  [{i+1}/{n_cand}] n={n}, init={name}, score0={score0:.5f}, t={t_alloc:.0f}s")

        # Convert to theta parameterization
        theta_init = np.log(a + 1e-15)

        a_opt, s_opt = optimize_adam_batch(theta_init, t_alloc, rng)
        print(f"    -> score={s_opt:.6f}")

        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}, init={name}) ***")

        optimized_results.append((s_opt, n, a_opt))

    optimized_results.sort(key=lambda x: x[0])

    # Phase 3: L-BFGS-B refinement on best candidates
    print("\n=== Phase 3: L-BFGS-B refinement ===")
    n_lbfgs = min(5, len(optimized_results))
    for i in range(n_lbfgs):
        if time.monotonic() > deadline - 100:
            break

        s0, n, a = optimized_results[i]
        t_alloc = min(120.0, deadline - time.monotonic() - 60)
        if t_alloc < 10:
            break

        print(f"  LBFGS on n={n}, score={s0:.6f}, t={t_alloc:.0f}s")
        a_ref, s_ref = lbfgs_refine(a, t_alloc, max_iter=min(300, int(t_alloc * 2)))
        print(f"    -> score={s_ref:.6f}")

        if s_ref < best_score:
            best_score = s_ref
            best_seq = a_ref.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")

        # Update optimized results
        optimized_results[i] = (s_ref, n, a_ref)

    optimized_results.sort(key=lambda x: x[0])

    # Phase 4: Coarse-to-fine upsample
    print("\n=== Phase 4: Coarse-to-fine upsampling ===")
    if optimized_results:
        best_short = optimized_results[0]
        target_lengths = sorted([l for l in [512, 1024, 2048, 4096]
                                  if l > len(best_short[2])])
        for target_n in target_lengths:
            if time.monotonic() > deadline - 80:
                break
            t_alloc = min(80.0, deadline - time.monotonic() - 50)
            if t_alloc < 15:
                break

            a_up = upsample(best_short[2], target_n, rng, noise_level=0.015)
            print(f"  Upsample to n={target_n}, t={t_alloc:.0f}s")

            theta_up = np.log(a_up + 1e-15)
            a_opt, s_opt = optimize_adam_batch(theta_up, t_alloc, rng)
            print(f"    -> score={s_opt:.6f}")

            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} (n={target_n}) ***")

            # Quick LBFGS on upsampled result
            if deadline - time.monotonic() > 40:
                a_ref, s_ref = lbfgs_refine(a_opt, min(30.0, deadline - time.monotonic() - 30), max_iter=100)
                if s_ref < best_score:
                    best_score = s_ref
                    best_seq = a_ref.copy()
                    print(f"    ** LBFGS improved: {best_score:.6f} ***")

    # Phase 5: Peak-directed local search polish
    print(f"\n=== Phase 5: Peak-directed local search ===")
    if best_seq is not None:
        polish_time = min(250.0, max(20.0, deadline - time.monotonic() - 40))
        n_polish = len(best_seq)
        print(f"  Polishing n={n_polish}, score={best_score:.6f}, time={polish_time:.0f}s")

        a_pol, s_pol = peak_directed_search(best_seq, polish_time, rng, n_top=6)
        if s_pol < best_score:
            best_score = s_pol
            best_seq = a_pol.copy()
            print(f"  After polish: {best_score:.6f}")
        else:
            print(f"  After polish: {s_pol:.6f} (no improvement)")

    # Phase 6: Additional restarts if time remains
    remaining = deadline - time.monotonic() - 30
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Phase 6: Additional restarts (remaining={remaining:.0f}s) ===")
        n_best = len(best_seq)
        n_restarts = min(4, int(remaining // 50))

        for r in range(n_restarts):
            if time.monotonic() > deadline - 35:
                break

            # Different perturbation strategies
            strategy = r % 3
            if strategy == 0:
                a_new = best_seq * np.exp(rng.normal(0, 0.15, size=n_best))
            elif strategy == 1:
                # Block-level perturbation
                a_new = best_seq.copy()
                n_blocks = n_best // 16
                for b in range(n_blocks):
                    start = b * 16
                    end = min(start + 16, n_best)
                    a_new[start:end] *= np.exp(rng.normal(0, 0.1))
            else:
                # Blend with a fresh FFT init
                a_fresh = fft_phase_init(n_best, rng, n_iter=50)
                a_new = 0.7 * best_seq + 0.3 * a_fresh
                a_new = a_new / np.sum(a_new) * np.sum(best_seq)

            a_new = np.clip(a_new, 0.01, None)
            a_new = a_new / np.sum(a_new) * np.sum(best_seq)

            t_alloc = min(40.0, (deadline - time.monotonic() - 25) / max(1, n_restarts - r))
            theta_new = np.log(a_new + 1e-15)
            a_opt, s_opt = optimize_adam_batch(theta_new, t_alloc, rng)
            print(f"  Restart {r+1} (strategy={strategy}): score={s_opt:.6f}")

            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} ***")

                # Quick polish
                if deadline - time.monotonic() > 30:
                    a_pol2, s_pol2 = peak_directed_search(best_seq,
                        min(25.0, deadline - time.monotonic() - 20), rng)
                    if s_pol2 < best_score:
                        best_score = s_pol2
                        best_seq = a_pol2.copy()

    # Final verification
    print(f"\n{'='*50}")
    elapsed = time.monotonic() - t_global
    print(f"Total time: {elapsed:.1f}s")

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = score(np.array(final_list))
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")

        # Double-check with evaluate_sequence
        actual_score = evaluate_sequence(final_list)
        print(f"=== VERIFIED with evaluate_sequence: {actual_score:.6f} ===")
        return final_list

    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
