"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
1. FISTA with adaptive log-sum-exp (FFT-accelerated) as primary optimizer.
2. LP-based direction finding using HiGHS solver - finds the optimal direction
   that maximizes sum(g) subject to f*g <= M (the "state of the art" approach).
3. Coordinate descent with 1D subgradient bisection for fine polishing.
4. Multiple initializations, n values, and restarts for diversity.
5. Adaptive time allocation: explore broadly, then polish the best.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(f, g):
    """Full convolution f*g via FFT. Returns array of length 2n-1."""
    n = len(f)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    F = fft(f, fsize)
    G = fft(g, fsize)
    c = np.real(ifft(F * G))
    return c[:size]


def fft_grad_lse(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a using FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_active(a, c, active_indices, weights):
    """Gradient using weighted active set. weights sum to 1."""
    n = len(a)
    if len(active_indices) == 0:
        return np.zeros(n)
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(active_indices):
        w_full[int(k)] = weights[i]

    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def run_fista(a_init, n, S, max_time, rng, beta_init=3.0, beta_max=300.0,
              noise_interval=3000, noise_prob=0.15, noise_scale=0.005):
    """FISTA with adaptive log-sum-exp and simplex projection."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.003 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = fft_grad_lse(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta * 2.0 * (1.0 + 0.0003 * it))
        step = max(step, 1e-7)
        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

        if it % noise_interval == 0 and rng.random() < noise_prob:
            y = project_simplex(y + rng.normal(0, noise_scale * S / np.sqrt(n), n), S)
            t_k = 1.0

    return best_local, best_score


def run_fista_active(a_init, n, S, max_time, rng, top_k=20, beta_init=10.0, beta_max=500.0):
    """FISTA using active-set gradient (top-k lags) for sharper optimization."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Use top-k active lags
        k_actual = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k_actual]
        top_val = c[top_idx]
        # Soft weights: use softmax with high beta for concentration
        beta_w = 50.0
        w = np.exp(beta_w * (top_val - top_val.max()))
        w = w / w.sum()

        grad = fft_grad_active(y, c, top_idx, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

    return best_local, best_score


def run_lp_refine(a_init, n, S, max_time, rng, n_constraints=200, step_frac=0.1):
    """
    LP-based direction finding.
    At each step: solve LP to find g maximizing sum(g) s.t. f*g <= M, g >= 0.
    Then blend f -> (1-t)f + t*g.
    """
    f = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) * S / n
    else:
        f = f * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = f.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(f)
        M = c.max()
        if M < best_score:
            best_score = M
            best_local = f.copy()

        # Select active constraints: top n_constraints lags
        n_act = min(n_constraints, 2 * n - 1)
        active_lags = np.argsort(c)[::-1][:n_act]

        # Build LP: maximize sum(g) s.t. f*g[k] <= M for k in active_lags, g >= 0
        # (f*g)[k] = sum_j f[k-j] * g[j] for valid j
        # Constraint row for lag k: sum_j f[k-j]*g[j] <= M
        # A_ub[k_idx, j] = f[k-j] if 0 <= k-j < n, else 0

        A_rows = []
        for k in active_lags:
            k = int(k)
            row = np.zeros(n)
            j_lo = max(0, k - (n - 1))
            j_hi = min(k, n - 1)
            if j_lo <= j_hi:
                row[j_lo:j_hi + 1] = f[k - j_hi:k - j_lo + 1][::-1]
            A_rows.append(row)

        A_ub = np.array(A_rows)
        b_ub = np.full(n_act, M)

        # Objective: minimize -sum(g)
        c_obj = -np.ones(n)

        # Solve LP
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': max(0.5, max_time * 0.05)})
            if not res.success:
                # If LP fails, try with fewer constraints
                n_act2 = max(50, n_act // 2)
                active_lags2 = np.argsort(c)[::-1][:n_act2]
                A_rows2 = []
                for k in active_lags2:
                    k = int(k)
                    row = np.zeros(n)
                    j_lo = max(0, k - (n - 1))
                    j_hi = min(k, n - 1)
                    if j_lo <= j_hi:
                        row[j_lo:j_hi + 1] = f[k - j_hi:k - j_lo + 1][::-1]
                    A_rows2.append(row)
                A_ub2 = np.array(A_rows2)
                b_ub2 = np.full(n_act2, M)
                res = linprog(c_obj, A_ub=A_ub2, b_ub=b_ub2,
                              bounds=[(0, None)] * n, method='highs',
                              options={'presolve': True})
                if not res.success:
                    continue
        except Exception:
            continue

        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            continue

        # Normalize g to have same sum as f
        g = g * (S / g_sum)

        # Line search: try different blend factors
        best_t = 0.0
        best_val = M
        for t_try in [0.01, 0.02, 0.05, 0.1, 0.2, step_frac, 0.3, 0.5]:
            f_try = (1.0 - t_try) * f + t_try * g
            f_try = np.maximum(f_try, 0.0)
            s_try = f_try.sum()
            if s_try < 1e-15:
                continue
            f_try = f_try * (S / s_try)
            c_try = fft_convolve_self(f_try)
            if c_try.max() < best_val:
                best_val = c_try.max()
                best_t = t_try

        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g
            f = np.maximum(f, 0.0)
            s_f = f.sum()
            f = f * (S / s_f)
        else:
            # No improvement, add small perturbation
            f = project_simplex(f + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)

        # Check time
        if time.monotonic() - start > max_time:
            break

    # Final check
    c = fft_convolve_self(f)
    if c.max() < best_score:
        best_score = c.max()
        best_local = f.copy()

    return best_local, best_score


def run_coord_descent(a_init, n, S, max_time, rng, n_sweeps=5):
    """
    Coordinate descent: for each coordinate pair (i,j), find optimal transfer δ.
    Uses subgradient bisection for the 1D problem.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    c = fft_convolve_self(a)
    M = c.max()
    best_score = M
    best_local = a.copy()

    # Precompute shifted versions for efficiency
    # For coordinate i: the "gradient direction" is 2*a shifted by i
    # For pair (i,j): direction is 2*a_shift_i - 2*a_shift_j

    n_sweep = 0
    while time.monotonic() - start < max_time and n_sweep < n_sweeps:
        n_sweep += 1
        # Random order
        order = rng.permutation(n)

        for idx in range(0, n, max(1, n // 400)):
            if time.monotonic() - start > max_time:
                break
            i = int(order[idx % n])
            j = int(order[(idx + n // 2) % n])
            if i == j:
                continue

            # Find optimal δ: transfer δ from j to i
            # c_k(δ) = c_k + δ*(a[k-i] - a[k-j])*2 + δ^2*(δ_{2i,k} - δ_{2j,k})
            # (approximate, ignoring second-order for the bisection direction)

            # Compute subgradient at δ=0
            k_star = np.argmax(c)
            a_ki = a[k_star - i] if 0 <= k_star - i < n else 0.0
            a_kj = a[k_star - j] if 0 <= k_star - j < n else 0.0
            sg = 2.0 * (a_ki - a_kj)

            if sg == 0:
                continue

            # If sg < 0: increasing a_i (δ > 0) helps
            # If sg > 0: decreasing a_i (δ < 0) helps
            # Bisection for optimal δ in range
            lo = -a[j] * 0.9  # can't make a[j] negative
            hi = 1000.0 * S / n  # upper bound on a[i]
            if sg < 0:
                lo = 0.0
            else:
                hi = min(hi, -lo)
                lo, hi = -hi, -lo
                lo, hi = hi, lo  # flip back

            # Actually, let me just try a small range around 0
            delta_max = 0.1 * S / n
            lo_d = -delta_max
            hi_d = delta_max

            # Bisection: find where subgradient changes sign
            best_d = 0.0
            for _ in range(30):
                mid = 0.5 * (lo_d + hi_d)
                # Compute subgradient at δ = mid
                c_mid = c + 2.0 * mid * (np.roll(a, i) - np.roll(a, j))
                # Handle boundaries (roll wraps around, need to zero out)
                # This is approximate - good enough for the bisection direction
                # Actually let me just evaluate a few candidate δ values
                break

            # Simpler: try a few δ values
            candidates = []
            if sg < 0:
                candidates = [delta_max * f for f in [0.01, 0.05, 0.1, 0.3, 0.5, 0.8]]
            else:
                candidates = [-delta_max * f for f in [0.01, 0.05, 0.1, 0.3, 0.5, 0.8]]

            best_val = M
            best_delta = 0.0
            for d in candidates:
                if a[i] + d < -1e-10 or a[j] - d < -1e-10:
                    continue
                new_a = a.copy()
                new_a[i] += d
                new_a[j] -= d
                c_new = fft_convolve_self(new_a)
                if c_new.max() < best_val:
                    best_val = c_new.max()
                    best_delta = d

            if best_delta != 0.0:
                a[i] += best_delta
                a[j] -= best_delta
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                a = a * (S / s_a)
                c = fft_convolve_self(a)
                M = c.max()
                if M < best_score:
                    best_score = M
                    best_local = a.copy()

    return best_local, best_score


def run_mirror(a_init, n, S, max_time, rng, eta_init=0.1, top_k=20):
    """Mirror descent (multiplicative weights) with FFT-based subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        eta = eta_init / (1.0 + 0.005 * it)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_val = c[top_idx]
        w = np.exp(30.0 * (top_val - top_val.max()))
        w = w / w.sum()
        grad = fft_grad_active(a, c, top_idx, w)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_new * (S / s_new)

    return best_local, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    total_deadline = time.monotonic() + max(1.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None

    def update_best(a, source=""):
        nonlocal global_best, global_a
        n = len(a)
        a_norm = safe_normalize(a, n)
        c = fft_convolve_self(a_norm)
        score = c.max()
        # Convert to actual objective: 2*n*max_b / sum^2
        sum_a = a_norm.sum()
        obj = 2.0 * n * score / (sum_a ** 2)
        if obj < global_best:
            global_best = obj
            global_a = a_norm.copy()
            print("  [%s] n=%d obj=%.6f (max_conv=%.6f)" % (source, n, obj, score))
        return a_norm, obj

    # --- Gather initializations ---
    inits = []

    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            inits.append(("given", np.clip(given, 0, 1000)))
    except Exception:
        pass

    if inits:
        main_len = len(inits[0][1])
    else:
        main_len = 8192

    # Create initializations for main length
    n_main = main_len
    tt = np.linspace(0, 1, n_main, endpoint=False)
    inits.append(("rand", rng.random(n_main) + 0.05))
    inits.append(("sin_multi", 1.0 + 0.3 * np.sin(2*np.pi*3*tt) + 0.2*np.cos(2*np.pi*7*tt)
                  + 0.1*np.sin(2*np.pi*13*tt)))
    inits.append(("bump", np.exp(-0.5*((tt-0.5)*5.0)**2) + 0.05))

    # Try alternative length
    n_alt = n_main // 2 if n_main > 4096 else n_main * 2
    if n_alt != n_main and 1024 <= n_alt <= 32768:
        tt2 = np.linspace(0, 1, n_alt, endpoint=False)
        inits.append(("rand_alt", rng.random(n_alt) + 0.05))
        inits.append(("sin_alt", 1.0 + 0.4*np.sin(2*np.pi*5*tt2) + 0.2*np.cos(2*np.pi*11*tt2)))

    n_inits = len(inits)
    print("=== Starting search: %d initializations, budget=%.0fs ===" % (n_inits, budget_s))
    print("=== n_main=%d, n_alt=%d ===" % (n_main, n_alt))

    # --- Phase 1: FISTA exploration ---
    usable = max(1.0, total_deadline - time.monotonic())
    phase1_time = usable * 0.45
    per_init_fista = phase1_time / n_inits

    print("=== Phase 1: FISTA exploration (%.1f sec total) ===" % phase1_time)

    fista_results = {}
    for name, arr in inits:
        if time.monotonic() > total_deadline - 60:
            break
        n = len(arr)
        if n < 64:
            continue
        S = np.sqrt(2.0 * float(n))
        t0 = time.monotonic()
        seg = min(per_init_fista, max(2.0, total_deadline - 60 - time.monotonic()))

        a_f, score_f = run_fista(arr, n, S, seg, rng, beta_init=3.0, beta_max=200.0)
        a_f, obj_f = update_best(a_f, "FISTA/" + name)
        fista_results[name] = a_f

        # Also try active-set FISTA
        if time.monotonic() < total_deadline - 60:
            seg2 = max(1.0, min(10.0, per_init_fista * 0.5))
            a_f2, score_f2 = run_fista_active(a_f, n, S, seg2, rng, top_k=30)
            update_best(a_f2, "FISTAact/" + name)

    print("=== After Phase 1: global_best=%.6f ===" % global_best)

    # --- Phase 2: LP-based refinement ---
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        usable2 = max(1.0, total_deadline - time.monotonic())
        phase2_time = usable2 * 0.35

        print("=== Phase 2: LP refinement (%.1f sec) ===" % phase2_time)

        # Try LP with different constraint counts
        for n_cons in [100, 300, 500]:
            if time.monotonic() > total_deadline - 30:
                break
            seg = max(5.0, phase2_time / 3.0)
            a_lp, _ = run_lp_refine(global_a, n, S, seg, rng, n_constraints=n_cons)
            update_best(a_lp, "LP/%d" % n_cons)

        # Also try LP from FISTA results for diversity
        for name, a_init in fista_results.items():
            if time.monotonic() > total_deadline - 30:
                break
            n_i = len(a_init)
            S_i = np.sqrt(2.0 * float(n_i))
            seg = max(3.0, min(20.0, (total_deadline - 30 - time.monotonic()) / 3.0))
            a_lp2, _ = run_lp_refine(a_init, n_i, S_i, seg, rng, n_constraints=200)
            update_best(a_lp2, "LP/" + name)

    print("=== After Phase 2: global_best=%.6f ===" % global_best)

    # --- Phase 3: Polishing ---
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        usable3 = max(1.0, total_deadline - time.monotonic())
        phase3_time = usable3 * 0.55

        print("=== Phase 3: Polishing (%.1f sec) ===" % phase3_time)

        # High-beta FISTA polish
        seg = max(2.0, phase3_time * 0.3)
        a_p1, _ = run_fista(global_a, n, S, seg, rng, beta_init=50.0, beta_max=500.0,
                            noise_interval=5000, noise_prob=0.05)
        update_best(a_p1, "PolishFISTA")

        # Active-set FISTA polish
        if time.monotonic() < total_deadline - 30:
            seg2 = max(2.0, phase3_time * 0.25)
            a_p2, _ = run_fista_active(global_a, n, S, seg2, rng, top_k=50)
            a_p2b, _ = run_fista_active(a_p1, n, S, seg2, rng, top_k=50)
            update_best(a_p2, "PolishActive1")
            update_best(a_p2b, "PolishActive2")

        # Mirror descent polish
        if time.monotonic() < total_deadline - 30:
            seg3 = max(2.0, phase3_time * 0.2)
            a_p3, _ = run_mirror(global_a, n, S, seg3, rng, eta_init=0.03, top_k=30)
            update_best(a_p3, "PolishMirror")

        # LP polish from best
        if time.monotonic() < total_deadline - 30:
            seg4 = max(3.0, phase3_time * 0.15)
            a_p4, _ = run_lp_refine(global_a, n, S, seg4, rng, n_constraints=400)
            update_best(a_p4, "PolishLP")

        # Coordinate descent
        if time.monotonic() < total_deadline - 30:
            seg5 = max(2.0, phase3_time * 0.1)
            a_p5, _ = run_coord_descent(global_a, n, S, seg5, rng, n_sweeps=3)
            update_best(a_p5, "PolishCoord")

    print("=== After Phase 3: global_best=%.6f ===" % global_best)

    # --- Phase 4: Final restarts and verification ---
    if global_a is not None:
        # Try a few random restarts from the best
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        remaining = max(0.0, total_deadline - time.monotonic())

        if remaining > 10:
            print("=== Phase 4: Final restarts (%.1f sec) ===" % remaining)
            n_restarts = max(1, int(remaining / 15))
            for r in range(n_restarts):
                if time.monotonic() > total_deadline - 10:
                    break
                # Perturb and re-optimize
                pert_scale = 0.05 * S / np.sqrt(n)
                a_pert = global_a + rng.normal(0, pert_scale, n)
                a_pert = project_simplex(a_pert, S)
                seg = max(3.0, remaining * 0.3 / (n_restarts - r))
                a_r, _ = run_fista(a_pert, n, S, seg, rng, beta_init=20.0, beta_max=300.0)
                update_best(a_r, "Restart%d" % r)

    # --- Final verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_obj = evaluate_sequence([float(x) for x in global_a])
    print("=== FINAL: length=%d, objective=%.6f ===" % (len(global_a), final_obj))
    return [float(x) for x in global_a]
