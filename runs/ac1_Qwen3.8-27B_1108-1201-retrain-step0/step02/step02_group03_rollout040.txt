    """Optimize a[j] given current convolution c and sum S."""
    n = len(a)
    x = a[j]
    S_0 = S - x
    
    # Compute c0 (conv with a[j] = 0)
    c0 = c.copy()
    # Subtract the contribution of a[j]
    # c[k] includes:
    #   a[j] * a[k-j] if 0 <= k-j < n (and k != 2j)
    #   a[k-j] * a[j] if 0 <= k-j < n (and k != 2j) -- same as above, counted in the sum
    #   a[j]^2 if k == 2j
    # Actually, c[k] = sum_i a[i]*a[k-i]
    # The terms with index j: a[j]*a[k-j] (when 0<=k-j<n)
    # When k-j = j (k=2j): a[j]*a[j] = a[j]^2 (only one term)
    # When k-j != j: a[j]*a[k-j] appears once as i=j, and a[k-j]*a[j] appears once as i=k-j
    #   But wait, these are the same term! i=j gives a[j]*a[k-j], and i=k-j gives a[k-j]*a[j].
    #   Since multiplication is commutative, these are equal. But they're different terms in the sum
    #   (different values of i), so both are present.
    #   So the total contribution when k != 2j is 2*a[j]*a[k-j].
    #   When k = 2j, the contribution is a[j]^2 (just one term, since i=j is the only one).
    
    for k in range(2*n - 1):
        kj = k - j
        if 0 <= kj < n:
            if k == 2*j:
                c0[k] -= x * x
            else:
                c0[k] -= 2 * x * a[kj]
    
    # Now for candidate x_new:
    # c_new[k] = c0[k] + delta_k(x_new)
    # where delta_k(x_new) = x_new^2 if k==2j, 2*x_new*a[k-j] if 0<=k-j<n and k!=2j
    
    # Precompute linear coefficients
    lin = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        kj = k - j
        if 0 <= kj < n and k != 2*j:
            lin[k] = 2.0 * a[kj]
    
    quad_k = 2*j  # the only quadratic term
    quad_val = 1.0  # coefficient of x^2

            S_new = S_0 + x_new

            s = 2.0 * n * np.max(c_new) / (S_new * S_new)

                best_x = x_new

                    best_x = x_new

"""
Coordinate descent with vectorized 1D subproblem solving for minimizing
peak autocorrelation of non-negative step functions. Uses exact per-coordinate
optimization via batched grid search, multiple restarts with geometric
perturbations, and adaptive upscaling to exploit larger sequence lengths.

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def compute_score_and_conv(a):
    """Compute score and convolution simultaneously."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf, None
    c = fft_convolve(a, a)
    s = 2.0 * n * np.max(c) / (S * S)
    return s, c


def optimize_coord(a, c, S, j, n_coarse=80, n_fine=30):
    """
    Optimize a single coordinate j using batched grid search.
    Returns the new value of a[j].
    """
    n = len(a)
    x = a[j]
    S_0 = S - x
    c_len = 2 * n - 1

    if S_0 < 1e-10:
        a[j] = 0.0
        return

    # Compute c0 = conv with a[j] set to 0 (vectorized)
    c0 = c.copy()
    k_end = min(j + n, c_len)
    k_indices = np.arange(j, k_end)
    kj_indices = k_indices - j
    c0[k_indices] -= 2.0 * x * a[kj_indices]
    quad_k = 2 * j
    if quad_k < c_len:
        c0[quad_k] += x * x  # correct over-subtraction at quadratic point

    # Linear coefficients for the search
    lin = np.zeros(c_len)
    lin[k_indices] = 2.0 * a[kj_indices]
    if quad_k < c_len:
        lin[quad_k] = 0.0

    # Determine search range
    a_max = np.max(a)
    if x < 1e-8:
        x_hi = max(a_max * 3.0, 1.0)
        x_grid = np.linspace(0.0, x_hi, n_coarse)
    else:
        x_lo = max(0.0, x * 0.02)
        x_hi = x * 8.0 + 0.5 * a_max
        if x_lo > 0:
            x_grid = np.geomspace(x_lo, x_hi, n_coarse)
            x_grid = np.concatenate([[0.0], x_grid])
        else:
            x_grid = np.linspace(0.0, x_hi, n_coarse)

    valid = S_batch > 1e-10
    if not np.any(valid):
        a[j] = 0.0
        return

    x_valid = x_grid[valid]
    S_valid = S_batch[valid]

    # c_batch[i, k] = c0[k] + x_valid[i] * lin[k] + x_valid[i]^2 * (1 if k==quad_k)
    c_batch = c0[None, :] + x_valid[:, None] * lin[None, :]

def coordinate_descent(a, time_limit, rng, n_coarse=80, n_fine=30, reset_period=200):
    """
    Run coordinate descent with periodic perturbations to escape local minima.
    Returns the best sequence found.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best = a.copy()
    S, c = None, None
    S = np.sum(a)
    c = fft_convolve(a, a)
    best_score = 2.0 * n * np.max(c) / (S * S)
    t0 = time.monotonic()
    n_cycles = 0
    noise_amp = 0.3

    while time.monotonic() - t0 < time_limit:
        # One full cycle of coordinate descent
        S = np.sum(a)
        for j in range(n):
            # Update c to reflect current a (incremental)
            # Actually, for correctness after each coordinate update, we need to update c
            # But this is expensive. Instead, recompute c at the start of each cycle.
            # For within-cycle updates, we track c incrementally.
            x_old = a[j]
            # Remove old contribution
            k_end = min(j + n, 2 * n - 1)
            k_indices = np.arange(j, k_end)
            kj_indices = k_indices - j
            c[k_indices] -= 2.0 * x_old * a[kj_indices]
            quad_k = 2 * j
            if quad_k < 2 * n - 1:
                c[quad_k] += x_old * x_old

            # Optimize coordinate j
            optimize_coord(a, c, S, j, n_coarse, n_fine)

            # Add new contribution
            x_new = a[j]
            if x_new > 0:
                c[k_indices] += 2.0 * x_new * a[kj_indices]
                if quad_k < 2 * n - 1:
                    c[quad_k] -= x_new * x_new

            S = S - x_old + x_new

        n_cycles += 1

        # Evaluate
        current_score = 2.0 * n * np.max(c) / (S * S) if S > 1e-10 else np.inf
        if current_score < best_score:
            best_score = current_score
            best = a.copy()

        # Periodic perturbation to escape local minima
        if n_cycles % reset_period == 0 and time.monotonic() - t0 < time_limit * 0.7:
            a_noisy = best * np.exp(rng.normal(0, noise_amp, size=n))
            a_noisy = np.clip(a_noisy, 0.0, None)
            s_noisy = np.sum(a_noisy)
            if s_noisy > 1e-10:
                a = a_noisy / s_noisy * S
            noise_amp *= 0.85

        # If stuck for a while, try a bigger perturbation
        if n_cycles % (reset_period * 5) == 0 and noise_amp < 0.05:
            a = best * np.exp(rng.normal(0, 0.5, size=n))
            s_noisy = np.sum(a)
            if s_noisy > 1e-10:
                a = a / s_noisy * S
            noise_amp = 0.3

    return best, best_score


def upsample_sequence(a, new_n, rng):
    """Upsample using linear interpolation + small noise."""
    n = len(a)
    if new_n <= n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.0, None)
    s = np.sum(a_interp)
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_interp / s * np.sum(a)

            a_opt, s_opt = coordinate_descent(best_seq, remaining, rng, n_coarse=100, n_fine=50, reset_period=300)
            print(f"  After polish: score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"  *** Final best: {best_score:.6f} ***")

    # Final verification
    print(f"\n{'='*50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        # Use the actual evaluated score for comparison
        if final_score < best_score:
            best_score = final_score
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
