Optimization of non-negative sequences to minimize peak autocorrelation.
Uses FFT-accelerated gradient descent with smoothed max objective,
coordinate descent, and local search.
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)

    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    soft_max_val = c_max + tau * np.log(sum_exp)
    score = 2.0 * n * soft_max_val

    # Gradient: grad_j = 4n * sum_i a_i * w_{j+i}
    # = 4n * cross_corr(w, a)[j]
    size = 2
    while size < 3 * n:
        size *= 2
    FW = np.fft.rfft(w, size)
    FA = np.fft.rfft(a, size)
    corr = np.fft.irfft(FW * np.conj(FA), size)[:n]
    grad = 4.0 * n * corr

    return score, grad


def grad_to_theta(grad_a, a):
    """Convert gradient w.r.t. a to gradient w.r.t. theta (softmax parametrization)."""
    inner = np.dot(grad_a, a)
    return a * (grad_a - inner)


def weighted_peaks_score_grad(a, top_k=5, beta=5.0):
    """
    Score and gradient using weighted top peaks.
    Weights are proportional to exp(beta * (c_k - mean)/std).
    """
    n = len(a)
    c = fft_convolve(a, a)
    score = 2.0 * n * np.max(c)

    c_mean = np.mean(c)
    c_std = np.std(c)
    if c_std < 1e-15:
        c_std = 1e-15
    weights = np.exp(beta * (c - c_mean) / c_std)
    weights = weights / np.sum(weights)

    # Gradient: grad_j = 4n * sum_i a_i * w_{j+i}
    size = 2
    while size < 3 * n:
        size *= 2
    FW = np.fft.rfft(weights, size)
    FA = np.fft.rfft(a, size)
    corr = np.fft.irfft(FW * np.conj(FA), size)[:n]
    grad = 4.0 * n * corr * (np.max(c) - c_mean)  # scale by peak excess

    return score, grad


def coordinate_optimize(a, j):
    """
    Optimize coordinate j keeping others fixed.
    Returns optimal value for a[j].
    """
    n = len(a)
    a_j = a[j]
    s = np.sum(a)

    # Precompute: b_k = a_{k-j} if 0 <= k-j < n else 0
    # d_k = 1 if k == 2j else 0
    b = np.zeros(2 * n - 1)
    for k in range(j, j + n):
        b[k] = a[k - j]
    d = np.zeros(2 * n - 1)
    d[2 * j] = 1.0

    c_current = fft_convolve(a, a)

    def obj(x):
        delta = x - a_j
        c_new = c_current + 2.0 * delta * b + delta * delta * d
        s_new = s + delta
        if s_new < 1e-15:
            return 1e10
        return 2.0 * n * np.max(c_new) / (s_new * s_new)

    # Search range: [0, max(3*a_j, 2/n)]
    upper = max(3.0 * a_j, 4.0 / n)
    upper = max(upper, 0.1 / n)

    result = minimize_scalar(obj, bounds=(0, upper), method='bounded',
                              options={'xatol': 1e-10})
    return result.x


def local_search_peak_reduction(a, rng, n_iters=100):
    """
    Local search that targets the dominant peaks in the convolution.
    """
    n = len(a)
    current = a.copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score

    for it in range(n_iters):
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))

        # Find which coordinates contribute most to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = current[i] * current[j]

        # Reduce top contributors, increase others slightly
        top_k = max(5, n // 8)
        top_idx = np.argsort(contribs)[-top_k:]
        other_idx = np.setdiff1d(np.arange(n), top_idx)

        candidate = current.copy()
        scale_down = np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
        candidate[top_idx] *= scale_down

        if len(other_idx) > 0:
            n_inc = min(10, len(other_idx))
            inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
            candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))

        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)
        new_score = compute_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()

    return best, best_score


def adam_soft_descent(theta, tau_schedule, n_iters_total, deadline):
    """
    Adam optimization on softmax parameterization with soft-max objective.
    """
    n = len(theta)
    best_theta = theta.copy()
    best_score = compute_score(softmax(theta))

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for tau in tau_schedule:
        # Distribute iterations across tau values
        remaining_taus = len(tau_schedule) - tau_schedule.index(tau)
        iters_for_tau = n_iters_total // remaining_taus

        for _ in range(iters_for_tau):
            if time.monotonic() > deadline:
                return best_theta, best_score

            t += 1
            a = softmax(theta)
            score, grad_a = soft_max_score_grad(a, tau)
            grad_theta = grad_to_theta(grad_a, a)

            m = beta1 * m + (1 - beta1) * grad_theta
            v = beta2 * v + (1 - beta2) * (grad_theta ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            lr = 0.01 * np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t)
            theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = softmax(theta)
            s_new = compute_score(a_new)
            if s_new < best_score:
                best_score = s_new
                best_theta = theta.copy()

    return best_theta, best_score


def projected_gradient_descent(a_init, n_iters, deadline, tau_start=0.05, tau_end=0.001):
    """
    Projected gradient descent on the non-negative simplex.
    Uses soft-max gradient with decreasing temperature.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / np.sum(a)
    best = a.copy()
    best_score = compute_score(a)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for i in range(n_iters):
        if time.monotonic() > deadline:
            break

        # Anneal temperature
        progress = i / max(1, n_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** progress

        t += 1
        score, grad = soft_max_score_grad(a, tau)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = 0.05 * np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-15:
            a_new = a / n * n
            s = np.sum(a_new)
        a_new = a_new / s

        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best = a_new.copy()

        a = a_new

    return best, best_score


def coordinate_descent_passes(a, n_passes, deadline):
    """Multiple passes of coordinate descent."""
    n = len(a)
    current = a.copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    rng = np.random.default_rng(12345)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break

        # Random or cyclic order
        order = rng.permutation(n) if p % 3 != 0 else np.arange(n)

        for j in order:
            if time.monotonic() > deadline:
                break
            new_val = coordinate_optimize(current, j)
            if new_val >= 0:
                current[j] = new_val
                current = np.maximum(current, 0)
                current = current / np.sum(current)

        new_score = compute_score(current)
        if new_score < best_score:
            best_score = new_score
            best = current.copy()
            if p % 20 == 0:
                print(f"      CD pass {p}: score={best_score:.8f}")

    return best, best_score


def diverse_init(n, rng, method='chirp'):
    """Generate diverse initializations."""
    if method == 'chirp':
        alpha = 0.3 + 0.5 * rng.random()
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.8))
        a = np.clip(a, 0.01, None)
        a += 0.05 * rng.standard_normal(n)
        a = np.clip(a, 0.005, None)
        return a / np.sum(a)
    elif method == 'random':
        a = 1.0 + 0.4 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
        return a / np.sum(a)
    elif method == 'bell':
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        width = n * (0.15 + 0.2 * rng.random())
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
        a += 0.03 * rng.standard_normal(n)
        a = np.clip(a, 0.005, None)
        return a / np.sum(a)
    elif method == 'flat_osc':
        # Flat with small oscillation
        t = np.arange(n, dtype=np.float64)
        freq = rng.integers(3, n // 4)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
        a = np.clip(a, 0.01, None)
        return a / np.sum(a)
    else:
        return np.ones(n) / n

    """
