def solve_lp(f, M, n):
    """Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all k"""
    # c = f * g, where c[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[i] * g[k-i]
    # We need c[k] <= M for k = 0, ..., 2n-2

    # A_ub[k, j] = f[j] if k-j is valid (0 <= k-j < n and 0 <= j < n)

    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = f[i]
    
    # Maximize sum(g) = minimize -sum(g)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None

For QCQPs, we can use:
1. Semidefinite relaxation (SDR)
2. Penalty methods
3. Alternating minimization

Actually, let me think about the SDR approach. The constraint (a*a)[k] <= t can be written as:
a^T T_k a <= t

where T_k is the (n x n) Toeplitz matrix with (T_k)_{i,j} = 1 if i-j = n-1-k (or some such relation depending on indexing).

Wait, let me be more careful. (a*a)[k] = sum_i a[i] * a[k-i] (with bounds). This is a quadratic form:
(a*a)[k] = a^T M_k a

where (M_k)_{i,j} = 1/2 if i + j = k (with appropriate bounds and the 1/2 factor to handle symmetry).

Actually, (a*a)[k] = sum_{i+j=k} a[i] * a[j] (with 0 <= i,j < n).

This equals a^T D_k a where D_k is the matrix with (D_k)_{i,j} = 1 if i+j=k, else 0. But this double-counts when i ≠ j. Actually no:
a^T D_k a = sum_{i,j} a[i] * D_k[i,j] * a[j] = sum_{i+j=k} a[i]*a[j]

That's exactly (a*a)[k]. Good.

So the problem is:
minimize t
subject to a^T D_k a <= t for k = 0, ..., 2n-2
sum(a) = 1
a_i >= 0

This is a QCQP with n^2 + 1 constraints (approximately). For n=256, that's about 65000 quadratic constraints. This is too many for most solvers.

However, we can use the SDR: introduce X = a*a^T (rank-1 PSD matrix) and replace a^T D_k a with Tr(D_k * X). The relaxation is:
minimize t
subject to Tr(D_k * X) <= t for all k
sum(a) = 1 (this becomes sum(X_ii) + 2*sum(X_ij) = 1... complicated)
X >= 0
rank(X) = 1 (dropped in relaxation)

This gives an SDP with n^2 variables and 2n-1 + 1 constraints. For n=256, that's 65536 variables. This is too large for most SDP solvers.

So SDR is not practical for n=256. Let me stick with the LP iteration approach.

Let me also consider another approach: **Projected Gradient Descent with a smooth approximation to the max**.

The function max_k (a*a)[k] can be approximated by the log-sum-exp:
max_k c_k ≈ (1/β) * log(sum_k exp(β * c_k))

for large β. The gradient of this is:
∂/∂a_i [(1/β) log(sum_k exp(β c_k))] = (1/sum_k exp(β c_k)) * sum_k exp(β c_k) * ∂c_k/∂a_i

And ∂c_k/∂a_i = sum_j D_k[i,j] * a[j] (the i-th row of D_k dotted with a).

This is computable for each i using FFT.

So I can do:
1. Start with some initial a (sum=1, a>=0)
2. Iterate: a := project_simplex(a - η * grad) where grad is the gradient of the log-sum-exp approximation
3. Increase β over time

The projection onto the simplex {a: sum(a)=1, a>=0} can be done in O(n log n) time.

This is essentially the approach in the original code (soft-max L-BFGS), but I can make it more efficient by using the simplex projection instead of the softmax reparameterization.

2. Use gradient-based refinement for the final polish

"""
LP-iteration based optimization for minimum peak autocorrelation non-negative sequences.
Uses the linear programming approach to find optimal update directions,
combined with gradient-based refinement and local search polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix A where (A @ g)[k] = (f*g)[k]."""
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A


def lp_solve(f, n):
    """
    Solve: maximize sum(g) s.t. (f*g)[k] <= 1 for all k, g >= 0.
    Returns (g, sum_g) or (None, 1.0) if infeasible.
    """
    M = np.max(fft_convolve(f, f))
    A = build_conv_matrix(f, n)
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, 
                  method='highs', options={'presolve': True})
    
    if not res.success:
        return None, 1.0
    
    g = res.x
    sum_g = np.sum(g)
    return g, sum_g


def lp_iteration(f, n, max_iters=5000, deadline=None, verbose=False):
    """
    LP-based iteration to minimize max(conv(f,f)) subject to sum(f)=1, f>=0.
    """
    f = np.maximum(f, 0).astype(np.float64)
    f = f / np.sum(f)
    
    best_f = f.copy()
    best_score = local_score(f)
    
    c = fft_convolve(f, f)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        g, sum_g = lp_solve(f, n)
        
        if g is None:
            break
        
        if sum_g <= 1.0 + 1e-8:
            # Can't improve much
            if verbose and it % 100 == 0:
                print(f"    LP iter {it}: sum_g={sum_g:.6f}, score={best_score:.6f}")
            break
        
        # Rescale g to sum = 1
        g = g / sum_g
        
        # Line search for step size
        best_t = 0
        best_new_score = best_score
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            s_new = f_new.sum()
            if s_new < 1e-10:
                continue
            f_new = f_new / s_new
            score = local_score(f_new)
            if score < best_new_score:
                best_new_score = score
                best_t = t
        
        if best_t == 0:
            # Try even smaller steps
            for t in [0.02, 0.01, 0.005]:
                f_new = (1 - t) * f + t * g
                f_new = np.maximum(f_new, 0)
                s_new = f_new.sum()
                if s_new < 1e-10:
                    continue
                f_new = f_new / s_new
                score = local_score(f_new)
                if score < best_new_score:
                    best_new_score = score
                    best_t = t
        
        if best_t == 0:
            break
        
        f = (1 - best_t) * f + best_t * g
        f = np.maximum(f, 0)
        f = f / np.sum(f)
        
        score = local_score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: t={best_t:.3f}, sum_g={sum_g:.4f}, score={score:.6f}, best={best_score:.6f}")
    
    if verbose:
        print(f"    LP done: best_score={best_score:.6f}")
    
    return best_f, best_score

        # v_flipped[m] = v[2n-2-m] = w[2n-2-m]
        # So (u conv v_flipped)[i+n-1] = sum_j a[j] * w[2n-2-(i+n-1-j)] = sum_j a[j] * w[n-1-i+j]
        # That's not quite what I want.
        
        # Let me try a different approach. Define:
        # u = a (length n)
        # v = reversed(w) = w[::-1] (length 2n-1)
        # (u * v)[k] = sum_j u[j] * v[k-j] = sum_j a[j] * w[2n-2-k+j]
        # We want grad_i/2 = sum_j a[j] * w[j+i]
        # Setting 2n-2-k+j = j+i => 2n-2-k = i => k = 2n-2-i
        # So (u * v)[2n-2-i] = sum_j a[j] * w[j+i]
        # Therefore: grad_i = 2 * (a * w_rev)[2n-2-i]
        # Or equivalently: grad = 2 * (a * w_rev)[2n-2-i] for i=0,...,n-1
        # Which means grad = 2 * (a * w_rev)[n-1 : 2n-1][::-1]
        # Or: grad = 2 * reverse((a * w_rev)[n-1 : 2n-1])
        
        # Let me verify for n=2:
        # a = [a0, a1], w = [w0, w1, w2], w_rev = [w2, w1, w0]
        # a * w_rev (convolution, length 2+3-1=4):
        # [a0*w2, a0*w1+a1*w2, a0*w0+a1*w1, a1*w0]
        # (a * w_rev)[n-1:2n-1] = (a * w_rev)[1:3] = [a0*w1+a1*w2, a0*w0+a1*w1]
        # reverse: [a0*w0+a1*w1, a0*w1+a1*w2]
        # grad = 2 * [a0*w0+a1*w1, a0*w1+a1*w2]
        # = [2*a0*w0+2*a1*w1, 2*a0*w1+2*a1*w2]
        # ✓ Matches!
        
        # Great! So the gradient is:
        # grad = 2 * reverse(fft_convolve(a, w[::-1])[n-1 : 2n-1])
        
        # Now I can compute this efficiently with FFT.
        
        w_rev = w[::-1]
        cross = fft_convolve(f, w_rev)
        grad = 2.0 * cross[n-1:2*n-1][::-1]
        
        # Step size: use a simple adaptive scheme
        eta = 1.0 / (beta + 1)
        
        # Gradient step
        f_new = f - eta * grad
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s
        
        new_score = local_score(f_new)
        
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
        
        f = f_new
        n_iters_done += 1

            print(f"    Grad iter {n_iters_done}: beta={beta:.1f}, score={new_score:.6f}, best={best_score:.6f}, t={elapsed:.0f}s")
    
    return best_f, best_score


def projected_subgradient_search(f, n, deadline, verbose=False):
    """
    More aggressive local search using projected subgradient with
    multiple step sizes and restarts.
    """
    f = np.maximum(f, 0).astype(np.float64)
    f = f / np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    it = 0
    while time.monotonic() < deadline:
        c = fft_convolve(f, f)
        c_max = np.max(c)
        
        # Subgradient: pick the lag(s) with maximum value
        peak_indices = np.where(c >= c_max - 1e-15)[0]
        
        # Compute subgradient for the active lags
        peak_idx = rng.choice(peak_indices)
        
        # grad_i = 2 * a_{peak_idx - i} (for valid peak_idx - i)
        grad = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                grad[i] = 2.0 * f[j]
        
        # Try multiple step sizes
        improved = False
        for eta in [0.01, 0.02, 0.05, 0.1, 0.2]:
            f_new = f - eta * grad
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            score = local_score(f_new)
            if score < best_score - 1e-12:
                best_score = score
                best_f = f_new.copy()
                f = f_new
                improved = True
                break
        
        if not improved:
            # Add random perturbation to escape
            f = f * (1 + 0.01 * rng.standard_normal(n))
            f = np.maximum(f, 0)
            f = f / np.sum(f)
        
        it += 1
    
    return best_f, best_score


def make_initial_sequence(n, rng, method='random'):
    """Create an initial sequence."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'random':
        a = 1.0 + 0.5 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
    elif method == 'bell':
        t = np.arange(n)
        center = (n-1) / 2
        width = n * 0.25
        a = np.exp(-0.5 * ((t - center) / width)**2)
        a = np.maximum(a, 0.01)
    elif method == 'chirp':
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    elif method == 'linear':
        a = np.linspace(0.5, 1.5, n)
    else:
        a = np.ones(n)
    
    return a / np.sum(a)
