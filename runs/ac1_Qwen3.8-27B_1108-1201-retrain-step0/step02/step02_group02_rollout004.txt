"""
Optimized peak-factor minimization for non-negative step functions.
Uses direct simplex-constrained gradient descent with combined soft-max + L2-flattening
objectives, entropy-based alternatives, peak-directed mutations, and multi-length exploration.
Key differences from previous: direct (non-softmax) parameterization, adaptive flattening

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT with padding to next power of 2."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def compute_conv_and_grads(a, tau, lambda_l2=0.0):
    """
    Compute convolution, soft-max value, and combined gradient.
    Returns (c, softmax_value, gradient, peak_indices).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    
    # Soft-max weights
    shifted = (c - c_max) / max(tau, 1e-10)
    exp_vals = np.exp(np.clip(shifted, -500, 500))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    sm_value = c_max + tau * np.log(sum_exp)
    
    # Gradient of soft-max w.r.t. a: g[i] = 2 * sum_m a[m] * (w[m+i] + w[i-m]) with bounds
    # Use FFT: grad = 2 * (convolve(a_reversed, w))[n-1:2n-1]
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g_sm = 2.0 * corr[n - 1:2 * n - 1]
    
    # L2 flattening gradient: d(||c||^2)/da[i] = 2 * sum_k c[k] * dc[k]/da[i]
    g_l2 = np.zeros(n)
    if lambda_l2 > 0:
        # sum_k c[k] * dc[k]/da[i] = (a_reversed * c)[n-1:2n-1] * 2 (by same structure)
        corr_c = fft_convolve(rev_a, c)
        g_l2 = 2.0 * corr_c[n - 1:2 * n - 1]
    
    gradient = g_sm + lambda_l2 * g_l2
    
    # Identify peak region (top values)
    threshold = c_max - 0.1 * (c_max - np.median(c))
    peak_indices = np.where(c >= threshold)[0]
    
    return c, sm_value, gradient, peak_indices


def project_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(len(a)) * S / len(a)
    else:
        a = a * (S / s)
    return a


def optimize_direct(a_init, S, time_limit, rng, lambda_l2_init=0.3, use_entropy=False):

    Direct gradient descent on the simplex with soft-max + L2 objectives.
    Uses adaptive step size, cooling, and restarts.
    """

    best_score = current_score
    
    tau = 5.0 / n  # initial temperature
    tau_min = 0.01 / n
    tau_decay = 0.9995
    lambda_l2 = lambda_l2_init

    no_improve = 0
    patience = 500
    step = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        
        # Adaptive lambda_l2: increase when multiple peaks detected
        c, sm_val, grad, peak_idx = compute_conv_and_grads(a, tau, lambda_l2)
        num_peaks = len(peak_idx)
        if num_peaks > 5:
            lambda_l2 = min(2.0, lambda_l2 * 1.02)
        else:
            lambda_l2 = max(0.05, lambda_l2 * 0.995)
        
        # Project gradient to tangent space of simplex (remove mean component)
        grad = grad - np.mean(grad)
        
        # Adaptive learning rate
        effective_lr = lr * (1.0 + 0.5 * np.exp(-step / 100.0))
        
        # Take step
        a_new = a - effective_lr * grad
        a_new = project_simplex(a_new, S)
        
        new_score = local_score(a_new)
        
        if new_score < current_score:
            a = a_new

        else:
            # Backtrack

                    a = a_new
                    current_score = local_score(a)
                    no_improve = 0

def optimize_entropy(a_init, S, time_limit, rng):
    """
    Optimize using negative entropy of normalized convolution as objective.
    Maximizes entropy = makes c as uniform as possible.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()

    current_score = local_score(a)

    lr = 0.03 / S
    no_improve = 0
    patience = 400
    step = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        c_pos = np.maximum(c, 1e-15)
        p = c_pos / np.sum(c_pos)
        
        # Negative entropy gradient
        # d(-sum p log p)/da[i] = -sum_k (1 + log(p[k]))/S_c * dc[k]/da[i]
        S_c = np.sum(c)
        weights = -(1.0 + np.log(p)) / S_c  # shape (2n-1,)
        
        # Gradient via FFT
        rev_a = a[::-1]
        corr = fft_convolve(rev_a, weights)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        c_max = np.max(c)
        shifted = (c - c_max) / tau
        exp_vals = np.exp(np.clip(shifted, -500, 500))

        corr_w = fft_convolve(rev_a, w)
        grad_sm = 2.0 * corr_w[n - 1:2 * n - 1]
        
        grad_total = grad_sm + 0.5 * grad
        grad_total = grad_total - np.mean(grad_total)
        
        a_new = a - lr * grad_total
        a_new = project_simplex(a_new, S)
        new_score = local_score(a_new)
        
        if new_score < current_score:
            a = a_new
            current_score = new_score
            no_improve = 0
            if new_score < best_score:

                best_a = a.copy()
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.8

            a = best_a + 0.01 * S * rng.standard_normal(n)
            a = project_simplex(a, S)
            current_score = local_score(a)
            no_improve = 0
            lr = 0.03 / S
        
        step += 1
        if step % 500 == 0:

    return best_a, best_score


def peak_shave_direct(a_init, S, time_limit, rng):
    """
    Aggressive peak shaving: iteratively identify and reduce convolution peaks
    by redistributing mass from peak contributors to non-contributors.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = project_simplex(a, S)
    current_score = local_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        c_max = np.max(c)
        c_mean = np.mean(c)
        
        # Find all significant peaks (above 1.05 * mean)
        peak_threshold = c_mean + 0.05 * (c_max - c_mean)
        peak_positions = np.where(c >= peak_threshold)[0]
        
        if len(peak_positions) == 0:
            break
        
        # For each peak, find top contributors
        for k_peak in peak_positions[:min(5, len(peak_positions))]:
            # Contribution of each a[i] to c[k_peak]
            contribs = np.zeros(n)
            for i in range(n):
                if 0 <= k_peak - i < n:
                    contribs[i] += a[i] * a[k_peak - i]
                if 0 <= i - k_peak < n:
                    contribs[i] += a[i] * a[i - k_peak]
            
            # Reduce top contributors, boost bottom
            top_k = max(2, n // 20)
            bot_k = max(4, n // 6)
            top_idx = np.argsort(contribs)[-top_k:]
            bot_idx = np.argsort(contribs)[:bot_k]
            
            # Ensure we don't overlap
            overlap = set(top_idx) & set(bot_idx)
            if overlap:
                bot_idx = np.array([x for x in bot_idx if x not in overlap])
                if len(bot_idx) == 0:
                    continue
            
            reduction = rng.uniform(0.01, 0.04)
            total_red = 0.0
            for idx in top_idx:
                red_amt = a[idx] * reduction
                a[idx] -= red_amt
                total_red += red_amt
            
            if total_red > 0 and len(bot_idx) > 0:
                a[bot_idx] += total_red / len(bot_idx)
            
            a = project_simplex(a, S)
        
        new_score = local_score(a)
        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # Revert and try smaller change
            a = best_a.copy()
            current_score = best_score
            # Try a single small perturbation
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if j != k:
                amt = a[j] * rng.uniform(0.005, 0.02)
                a[j] -= amt
                a[k] += amt
                a = project_simplex(a, S)
                s = local_score(a)
                if s < best_score:
                    best_score = s
                    best_a = a.copy()
                    current_score = s
        
        iteration += 1
        if iteration % 200 == 0:
            print(f"      PS iter={iteration}, score={current_score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score


def upsampling_sequence(a, n_target, rng):
    """Upsample a sequence to a target length using interpolation + refinement."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    x_old = np.linspace(0, 1, n_src)
    x_new = np.linspace(0, 1, n_target)
    a_up = np.interp(x_new, x_old, a)
    # Add small perturbation to break any periodicity
    a_up += 0.01 * np.std(a_up) * rng.standard_normal(n_target)
    return np.maximum(a_up, 0.0)


def make_random_init(n, rng, method='uniform'):
    """Generate diverse random initializations."""
    if method == 'uniform':
        return 1.0 + 0.3 * rng.standard_normal(n)
    elif method == 'bimodal':
        mid = n // 2
        a = np.ones(n)
        a[:mid//2] = 1.5
        a[mid - mid//4:mid + mid//4] = 0.3
        a[n - mid//2:] = 1.5
        a += 0.05 * rng.standard_normal(n)
        return np.maximum(a, 0.05)
    elif method == 'cosine_mod':
        t = np.arange(n) / n
        freq = rng.uniform(2, 8)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2*np.pi))
        a += 0.03 * rng.standard_normal(n)
        return np.maximum(a, 0.05)
    elif method == 'geometric':
        ratio = rng.uniform(0.95, 0.99)
        a = ratio ** np.arange(n)
        a = a / np.max(a)
        a = 0.5 + 0.5 * a
        a += 0.02 * rng.standard_normal(n)
        return np.maximum(a, 0.05)
    else:
        return np.maximum(1.0 + 0.2 * rng.standard_normal(n), 0.05)

        for method in methods:
            if time.monotonic() > deadline - 30:
                break
            t_alloc = min(t_per_method, deadline - time.monotonic() - 20)
            if t_alloc < 10:
                break
            
            a_init = make_random_init(n_primary, rng, method)
            a_init = project_simplex(a_init, S)
            print(f"  Trying '{method}' init ({t_alloc:.0f}s)...")
            
            a_res3, score3 = optimize_direct(a_init, S, t_alloc, rng, lambda_l2_init=0.8)
            score3_actual = local_score(a_res3)
            print(f"    Score: {score3_actual:.6f}")
            
            if score3_actual < best_score:
                best_score = score3_actual
                best_seq = a_res3.copy()
                print(f"  *** New best: {best_score:.6f} ***")
    
    # === PHASE 4: Try different lengths ===
    # Time: ~5%
    if time.monotonic() < deadline - 30:
        t_phase4 = min(0.05 * time_budget, deadline - time.monotonic() - 20)
        if t_phase4 > 15:
            print(f"\n{'='*60}")
            print(f"Phase 4: Length exploration ({t_phase4:.0f}s)")
            print(f"{'='*60}")
            
            for n_try in [n_primary + 64, n_primary - 64, 512]:
                if time.monotonic() > deadline - 25:
                    break
                if n_try < 32 or n_try > 1024:
                    continue
                t_alloc = min(t_phase4 / 3.0, deadline - time.monotonic() - 15)
                if t_alloc < 8:
                    break
                
                S_try = np.sqrt(2.0 * n_try)
                if best_seq is not None:
                    a_try = upsampling_sequence(best_seq, n_try, rng)
                else:
                    a_try = make_random_init(n_try, rng, 'uniform')
                a_try = project_simplex(a_try, S_try)
                
                print(f"  n={n_try}, starting: {local_score(a_try):.6f}")
                a_res4, score4 = optimize_direct(a_try, S_try, t_alloc, rng, lambda_l2_init=0.5)
                score4_actual = local_score(a_res4)
                print(f"    Score: {score4_actual:.6f}")
                
                if score4_actual < best_score:
                    best_score = score4_actual
                    best_seq = a_res4.copy()
                    print(f"  *** New best: {best_score:.6f} ***")
    
    # === PHASE 5: Final intensive polish ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_final = deadline - time.monotonic() - 10
        if t_final > 20:
            print(f"\n{'='*60}")
            print(f"Phase 5: Final polish ({t_final:.0f}s)")
            print(f"{'='*60}")
            
            S_final = np.sum(best_seq)
            # Alternate between gradient and peak shaving
            t_half = t_final / 2.0
            
            a_f1, s_f1 = optimize_direct(best_seq, S_final, t_half, rng, lambda_l2_init=0.3)
            s_f1_actual = local_score(a_f1)
            if s_f1_actual < best_score:
                best_score = s_f1_actual
                best_seq = a_f1.copy()
            
            if time.monotonic() < deadline - 10:
                S_final = np.sum(best_seq)
                a_f2, s_f2 = peak_shave_direct(best_seq, S_final, 
                                               min(t_half, deadline - time.monotonic() - 10), rng)
                s_f2_actual = local_score(a_f2)
                if s_f2_actual < best_score:
                    best_score = s_f2_actual
                    best_seq = a_f2.copy()
            
            print(f"  Final best: {best_score:.6f}")
    
    # === Verify and return ===
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'='*60}")
        
        # If evaluation differs from local (shouldn't happen, but safety check)
        if final_eval < best_score:
            best_score = final_eval
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
