2. Finding the subgradient at those points.
3. If the subgradient changes sign, the minimum is in between.
4. Use bisection on the subgradient.

- At a_i = 0: k* = argmax C_k, subgradient = D_{k*}. If D_{k*} >= 0, minimum is at 0.
- At a_i = max: k* = argmax (C_k + max*D_k), subgradient = D_{k*}. If D_{k*} <= 0, minimum is at max.

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact coordinate descent on the simplex with quadratic correction
2. LP-based direction finding via scipy linprog (HiGHS)
3. Multiple diverse starting points and sequence lengths
4. Adaptive blending and random perturbation
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def objective_normalized(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def safe_eval(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        return float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')


def _coord_step(a, i, slack):
    """One coordinate descent step: optimize a[i] with a[slack] as adjusting variable.
    Includes exact quadratic correction for the two self-interaction terms."""
    n = len(a)
    lo = -a[i]
    hi = a[slack]
    if hi - lo < 1e-15:
        return

    # Linear coefficient: L_k = 2*(a[k-i] - a[k-slack])
    L = np.zeros(2 * n - 1)
    ks = max(0, i)
    ke = min(2 * n - 2, i + n - 1)
    if ks <= ke:
        L[ks:ke + 1] += 2.0 * a[0:ke - ks + 1]
    ks2 = max(0, slack)
    ke2 = min(2 * n - 2, slack + n - 1)
    if ks2 <= ke2:
        L[ks2:ke2 + 1] -= 2.0 * a[0:ke2 - ks2 + 1]

    # Quadratic coefficient: Q_k (nonzero only at k=2i, k=2*slack, k=i+slack)
    Q = np.zeros(2 * n - 1)
    if 0 <= 2 * i < 2 * n - 1:
        Q[2 * i] += 1.0
    if 0 <= 2 * slack < 2 * n - 1:
        Q[2 * slack] += 1.0
    if 0 <= i + slack < 2 * n - 1:
        Q[i + slack] -= 2.0

    # Current convolution
    conv = fast_convolve(a)

    # Bisection using subgradient of max
    def calc_subgrad(delta):
        vals = conv + L * delta + Q * delta * delta
        k_star = np.argmax(vals)
        return L[k_star] + 2.0 * Q[k_star] * delta

    sg_lo = calc_subgrad(lo)
    if sg_lo >= -1e-12:
        best_delta = lo
    else:
        sg_hi = calc_subgrad(hi)
        if sg_hi <= 1e-12:
            best_delta = hi
        else:
            for _ in range(50):
                mid = (lo + hi) / 2.0
                sg_mid = calc_subgrad(mid)
                if sg_mid < 0:
                    lo = mid
                else:
                    hi = mid
            best_delta = (lo + hi) / 2.0

    a[i] += best_delta
    a[slack] -= best_delta
    if a[i] < 0:
        a[i] = 0
    if a[slack] < 0:
        a[slack] = 0


def coord_descent(a, max_sweeps=50, tol=1e-12, max_time=None, t0=None):
    """Exact coordinate descent on simplex with two-pass slack strategy."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        return np.ones(n) / n
    a /= s

    best_obj = objective_normalized(a)
    best_a = a.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if max_time is not None and t0 is not None:
            if time.monotonic() - t0 > max_time:
                break

        # Pass 1: optimize a_0..a_{n-2}, slack = a_{n-1}
        for i in range(n - 1):
            _coord_step(a, i, n - 1)

        # Pass 2: optimize a_1..a_{n-1}, slack = a_0
        for i in range(1, n):
            _coord_step(a, i, 0)

        # Cleanup
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            break
        a /= s

        cur_obj = objective_normalized(a)
        if cur_obj < best_obj - tol:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 5:
            break

    return best_a


def lp_direction(a):
    """Solve LP: min t s.t. (a*g)_k <= t, sum(g)=1, g>=0.
    Returns (g, t) where g sums to 1."""
    n = len(a)
    if n > 1024:
        return a.copy(), objective_normalized(a) / (2.0 * n)

    # Build constraint matrix A_ub: (2n-1) x (n+1)
    # Row k: [a_{k-i} for valid i, -1] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            A_ub[k, i_start:i_end + 1] = a[k - i_end:k - i_start + 1]
        A_ub[k, n] = -1.0

    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c = np.zeros(n + 1)
    c[n] = 1.0

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success:
        g = np.maximum(res.x[:n], 0.0)
        t = res.x[n]
        gs = g.sum()
        if gs < 1e-12:
            return a.copy(), objective_normalized(a) / (2.0 * n)
        g /= gs
        return g, t
    else:
        return a.copy(), objective_normalized(a) / (2.0 * n)


def lp_blend_step(a, alpha):
    """Apply one LP-based blending step. Returns (new_a, improved)."""
    g, t_lp = lp_direction(a)
    a_new = (1.0 - alpha) * a + alpha * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        return a, False
    a_new /= s

    obj_old = objective_normalized(a)
    obj_new = objective_normalized(a_new)
    improved = obj_new < obj_old
    return a_new, improved


def make_ramp(n, direction=1):
    """Create a ramp sequence (increasing or decreasing)."""
    if direction == 1:
        x = np.arange(1, n + 1, dtype=np.float64)
    else:
        x = np.arange(n, 0, -1, dtype=np.float64)
    x /= x.sum()
    return x


def make_quadratic(n, direction=1):
    """Create a quadratic ramp sequence."""
    if direction == 1:
        x = np.arange(1, n + 1, dtype=np.float64) ** 2
    else:
        x = (n + 1 - np.arange(1, n + 1, dtype=np.float64)) ** 2
    x /= x.sum()
    return x


def make_init(n, name, rng, parent=None):
    """Generate an initialization based on name."""
    if name == 'ramp_up':
        return make_ramp(n, 1)
    elif name == 'ramp_down':
        return make_ramp(n, -1)
    elif name == 'quad_up':
        return make_quadratic(n, 1)
    elif name == 'quad_down':
        return make_quadratic(n, -1)
    elif name == 'random':
        return rng.dirichlet(np.ones(n) * 0.3)
    elif name == 'random_smooth':
        x = rng.dirichlet(np.ones(n) * 2.0)
        # Smooth it
        k = max(1, n // 20)
        kernel = np.ones(k) / k
        x = np.convolve(x, kernel, mode='same')
        x = np.maximum(x, 0)
        x /= x.sum()
        return x
    elif name == 'parent' and parent is not None:
        if parent.size == n:
            p = parent.copy()
        else:
            p = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, parent.size), parent)
            p = np.maximum(p, 0)
        if p.sum() > 1e-10:
            p /= p.sum()
            return p
        return np.ones(n) / n
    else:
        return np.ones(n) / n

    # Phase 1: Exploration with diverse initializations
    print("Phase 1: Exploration")
    t_phase1_end = t_start + min(350.0, (deadline - t_start) * 0.2)

    candidates = []
    lengths = [128, 256, 512, 1024]
    init_names = ['ramp_up', 'ramp_down', 'quad_up', 'quad_down',
