"""
LP-based peak minimization for non-negative step functions.
Uses linear programming to find optimal directions that reduce the
autocorrelation peak, combined with gradient-based refinement and
local search polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def lp_direction(f, M):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g.
    """
    n = len(f)
    
    # Build constraint matrix
    # A[k, j] = f[j] if max(0, k-n+1) <= j <= min(k, n-1)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # Column j: non-zeros in rows j through j+n-1
        row_start = j
        row_end = min(j + n, 2*n-1)
        A[row_start:row_end, j] = f[j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        # Fallback: return f itself (always feasible)
        return f.copy()


def lp_iteration(f, n_iter, verbose=False):
    """
    Perform LP-based iterations to reduce the peak of f*f.
    f is normalized to have sum = sqrt(2n).
    """
    n = len(f)
    S_target = np.sqrt(2.0 * n)
    f = f / np.sum(f) * S_target  # normalize
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    for it in range(n_iter):
        M = np.max(fft_convolve(f, f))
        
        # Solve LP
        g = lp_direction(f, M)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum * S_target  # rescale to match target sum
        
        # Take a step toward g
        t = 0.1  # step size
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        
        # Check if it improves
        score_new = compute_score(f_new)
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
        f = f_new
        
        if verbose and it % 100 == 0:
            print(f"      LP iter {it}: score={score_new:.6f}, best={best_score:.6f}")
    
    return best_f, best_score

(f_t * f_t)[k] = (1-t)^2 * (f*f)[k] + 2t(1-t) * (f*g)[k] + t^2 * (g*g)[k]

We want this <= M for all k. Since (f*f)[k] <= M and (f*g)[k] <= M, for small t:
(f_t*f_t)[k] <= (1-t)^2*M + 2t(1-t)*M + t^2*(g*g)[k]
= M*(1-t)^2 + 2t(1-t)*M + t^2*(g*g)[k]
= M*(1-2t+t^2) + 2tM - 2t^2*M + t^2*(g*g)[k]
= M - 2tM + t^2*M + 2tM - 2t^2*M + t^2*(g*g)[k]
= M + t^2*((g*g)[k] - M)

So (f_t*f_t)[k] <= M + t^2 * max_k((g*g)[k] - M)

If max(g*g) <= M, then the peak doesn't increase at all! In that case, we can take t=1.

"""
LP-guided peak minimization for non-negative step functions.
Combines linear programming for direction finding with gradient-based
refinement and local search polishing to minimize the autocorrelation peak.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def build_lp_matrix(n, f):
    """Build the constraint matrix A for the LP."""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        row_start = j
        row_end = min(j + n, 2*n-1)
        A[row_start:row_end, j] = f[j]
    return A


def lp_direction(n, f, M):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g.
    """
    A = build_lp_matrix(n, f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return f.copy()


def lp_optimize(a_init, time_budget, rng, verbose=False):
    """
    LP-based optimization to reduce the peak of the autocorrelation.
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a = a_init / np.sum(a_init) * S_target
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    iter_count = 0
    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a, a)
        M = np.max(c)
        
        # Solve LP
        g = lp_direction(n, a, M)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum * S_target
        
        # Check if g is meaningfully different from a
        diff = np.sum((g - a)**2) / (np.sum(a**2) + 1e-10)
        
        # Take a step
        if diff < 1e-8:
            # Converged, try a different strategy
            a = a * np.exp(rng.normal(0, 0.05, size=n))
            a = np.clip(a, 0, None)
            a = a / np.sum(a) * S_target
            continue
        
        # Adaptive step size
        g_peak = np.max(fft_convolve(g, g))
        if g_peak <= M:
            t_step = 0.3
        else:
            t_step = min(0.3, 0.5 * M / (g_peak - M + 1e-10))
            t_step = max(t_step, 0.01)
        
        a_new = (1 - t_step) * a + t_step * g
        a_new = np.clip(a_new, 0, None)
        if np.sum(a_new) < 1e-10:
            a_new = a
        else:
            a_new = a_new / np.sum(a_new) * S_target
        
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
        iter_count += 1
        
        if verbose and iter_count % 50 == 0:
            print(f"    LP iter {iter_count}: score={score_new:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score


def gradient_refine(a_init, time_budget, rng):
    """
    Gradient-based refinement using log-sum-exp smoothing.
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a = a_init / np.sum(a_init) * S_target
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    no_improve = 0
    patience = 500
    
    while time.monotonic() - t0 < time_budget:
        step += 1
        c = fft_convolve(a, a)
        c_max = np.max(c)
        
        # Annealed temperature
        frac = (time.monotonic() - t0) / time_budget
        tau = max(0.005 / n, (1.0 / n) * (1.0 - frac) ** 2)
        
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted - np.max(shifted))
        w = exp_shifted / np.sum(exp_shifted)
        
        # Gradient
        a_flipped = a[::-1]
        corr = fft_convolve(w, a_flipped)
        grad = 2.0 * corr[n-1:2*n-1]
        
        # Adam
        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * grad ** 2
        m_hat = m_buf / (1 - beta1 ** step)
        v_hat = v_buf / (1 - beta2 ** step)
        
        lr = 3.0 / np.sqrt(step + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.clip(a_new, 0, None)
        if np.sum(a_new) > 1e-10:
            a_new = a_new / np.sum(a_new) * S_target
        else:
            a_new = a
        
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Escape mechanism
        if no_improve > patience:
            r = rng.random()
            if r < 0.5:
                a = a * np.exp(rng.normal(0, 0.1, size=n))
            else:
                # Targeted perturbation
                peak_idx = int(np.argmax(c))
                contribs = np.zeros(n)
                for i in range(n):
                    if 0 <= peak_idx - i < n:
                        contribs[i] = a[i] * a[peak_idx - i]
                top_k = np.argsort(contribs)[::-1][:n//8]
                a[top_k] *= np.exp(rng.normal(0, 0.1, size=len(top_k)))
            a = np.clip(a, 0, None)
            a = a / np.sum(a) * S_target
            no_improve = 0
    
    return best_a, best_score

    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a = a_init / np.sum(a_init) * S_target
    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.99995
    
    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))
        
        contribs = np.zeros(n)
        for i in range(n):
            if 0 <= peak_idx - i < n:
                contribs[i] = a[i] * a[peak_idx - i]
        
        a_new = a.copy()
        r = rng.random()
        
        if r < 0.35:
            # Reduce top contributors, increase bottom
            k_red = max(2, n // 20)
            k_inc = max(3, n // 10)
            top_idx = np.argsort(contribs)[::-1][:k_red]
            bot_idx = np.argsort(contribs)[:k_inc]
            amt = a[top_idx].sum() * rng.uniform(0.03, 0.1)
            a_new[top_idx] -= amt / k_red
            a_new[bot_idx] += amt / k_inc
        elif r < 0.55:
            # Single pair swap
            top5 = np.argsort(contribs)[::-1][:5]
            j_red = top5[int(rng.integers(0, 5))]
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red:
                amt = a[j_red] * rng.uniform(0.02, 0.08)
                a_new[j_red] -= amt
                a_new[j_inc] += amt
        elif r < 0.75:
            # Block operation
            width = max(2, int(rng.integers(2, max(3, n // 10))))
            start = int(rng.integers(0, max(1, n - width)))
            factor = np.exp(rng.normal(0, 0.04))
            a_new[start:start+width] *= factor
        else:
            # Global perturbation
            scale = rng.uniform(0.95, 1.05, size=n)
            a_new = a * scale
        
        a_new = np.clip(a_new, 0, None)
        if np.sum(a_new) < 1e-10:
            continue
        a_new = a_new / np.sum(a_new) * S_target
        
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            cur_score = score_new
        elif rng.random() < np.exp(-(score_new - cur_score) / max(temp, 1e-15)):
            cur_score = score_new

        temp *= cooling
    
    return best_a, best_score

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def lp_direction(f, M):
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:min(j+n, 2*n-1), j] = f[j]
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0,None)]*n, method='highs')

"""
LP-guided peak minimization for non-negative step functions.
Uses linear programming to find optimal redistribution directions that reduce
the autocorrelation peak, combined with gradient-based refinement and local
search polishing. Explores multiple sequence lengths with adaptive time allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score 2*n*max(conv)/sum^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def lp_direction(f, M):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g vector.
    """
    n = len(f)
    # Build constraint matrix: A[k,j] = f[j] if max(0,k-n+1) <= j <= min(k,n-1)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:min(j + n, 2 * n - 1), j] = f[j]
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return f.copy()


def lp_optimize(a_init, time_budget, rng, verbose=False):
    """
    LP-based iterative optimization to reduce the peak of autocorrelation.
    At each step, solve an LP to find the best direction, then take a step.
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a = a_init / np.sum(a_init) * S_target
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        M = np.max(fft_convolve(a, a))
        g = lp_direction(a, M)
        gs = np.sum(g)
        if gs < 1e-10:
            break
        g = g / gs * S_target

        # Check convergence
        diff = np.linalg.norm(g - a) / (np.linalg.norm(a) + 1e-10)
        if diff < 1e-6:
            stagnant += 1
            if stagnant > 20:
                # Perturb to escape
                a = np.clip(a * np.exp(rng.normal(0, 0.1, n)), 0, None)
                a = a / np.sum(a) * S_target
                stagnant = 0
            continue
        stagnant = 0

        g_peak = np.max(fft_convolve(g, g))
        if g_peak <= M * 1.01:
            t_step = 0.25

            t_step = max(0.03, min(0.25, 0.4 * M / (g_peak - M + 1e-10)))

        a = a_new
        it += 1
        if verbose and it % 50 == 0:
            print(f"      LP it={it}, score={s_new:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def gradient_refine(a_init, time_budget, rng):
    """
    Gradient-based refinement using annealed log-sum-exp of the convolution peak.
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a = a_init / np.sum(a_init) * S_target
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    no_imp = 0

    while time.monotonic() - t0 < time_budget:
        step += 1
        c = fft_convolve(a, a)
        c_max = np.max(c)
        frac = min(1.0, (time.monotonic() - t0) / time_budget)
        tau = max(0.003 / n, (0.8 / n) * (1.0 - frac) ** 2)
        shifted = (c - c_max) / tau
        exp_sh = np.exp(shifted - np.max(shifted))
        w = exp_sh / np.sum(exp_sh)

        # Gradient of weighted sum of convolution
        corr = fft_convolve(w, a[::-1])
        grad = 2.0 * corr[n - 1:2 * n - 1]

        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)
        lr = 4.0 / np.sqrt(step + 1)
        a_new = np.clip(a - lr * m_hat / (np.sqrt(v_hat) + eps), 0, None)
        if np.sum(a_new) < 1e-10:
            a_new = a
        else:
            a_new = a_new / np.sum(a_new) * S_target
        s_new = compute_score(a_new)
        if s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        a = a_new

        if no_imp > 300:
            r = rng.random()
            if r < 0.5:
                a = np.clip(a * np.exp(rng.normal(0, 0.12, n)), 0, None)
            else:
                peak_idx = int(np.argmax(c))
                contribs = np.zeros(n)
                for i in range(n):
                    if 0 <= peak_idx - i < n:
                        contribs[i] = a[i] * a[peak_idx - i]
                top_k = np.argsort(contribs)[::-1][:max(4, n // 8)]
                a[top_k] *= np.exp(rng.normal(0, 0.15, size=len(top_k)))
                a = np.clip(a, 0, None)
            a = a / np.sum(a) * S_target
            no_imp = 0

    return best_a, best_score


def local_polish(a_init, time_budget, rng):
    """
    Local search with peak-directed mass redistribution and simulated annealing.
    """
    n = len(a_init)
    S_t = np.sqrt(2.0 * n)
    a = a_init / np.sum(a_init) * S_t
    best_a = a.copy()
    best_s = compute_score(a)
    cur_s = best_s
    t0 = time.monotonic()
    temp = 0.004
    cool = 0.99998

    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a, a)
        pi = int(np.argmax(c))
        contribs = np.zeros(n)
        for i in range(n):
            if 0 <= pi - i < n:
                contribs[i] = a[i] * a[pi - i]

        an = a.copy()
        r = rng.random()
        if r < 0.30:
            kr = max(2, n // 25)
            ki = max(3, n // 12)
            ti = np.argsort(contribs)[::-1][:kr]
            bi = np.argsort(contribs)[:ki]
            amt = a[ti].sum() * rng.uniform(0.03, 0.12)
            an[ti] -= amt / kr
            an[bi] += amt / ki
        elif r < 0.50:
            t5 = np.argsort(contribs)[::-1][:max(3, n // 30)]
            jr = t5[int(rng.integers(0, len(t5)))]
            ji = int(rng.integers(0, n))
            if ji != jr:
                amt = a[jr] * rng.uniform(0.02, 0.1)
                an[jr] -= amt
                an[ji] += amt
        elif r < 0.70:
            w = max(2, int(rng.integers(2, max(3, n // 12))))
            s = int(rng.integers(0, max(1, n - w)))
            an[s:s + w] *= np.exp(rng.normal(0, 0.05))
        elif r < 0.85:
            # Multi-peak attack: find top 3 peaks and reduce contributors
            peaks = np.argsort(c)[::-1][:3]
            for pk in peaks:
                pcontribs = np.zeros(n)
                for i in range(n):
                    if 0 <= pk - i < n:
                        pcontribs[i] = a[i] * a[pk - i]
                top = np.argsort(pcontribs)[::-1][:max(2, n // 40)]
                bot = np.argsort(pcontribs)[:max(3, n // 20)]
                amt2 = a[top].sum() * 0.04
                an[top] -= amt2 / len(top)
                an[bot] += amt2 / len(bot)
        else:
            an = a * rng.uniform(0.97, 1.03, size=n)

        an = np.clip(an, 0, None)
        if np.sum(an) < 1e-10:
            continue
        an = an / np.sum(an) * S_t
        sn = compute_score(an)
        if sn < best_s:
            best_s = sn
            best_a = an.copy()
            cur_s = sn
        elif rng.random() < np.exp(-(sn - cur_s) / max(temp, 1e-15)):
            cur_s = sn
        a = an
        temp *= cool

    return best_a, best_s


def make_init(n, rng, method='broadband'):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if method == 'broadband':
        a = np.ones(n)
        for k in range(1, 15):
            freq = k * (1.0 + 0.6180339887 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.18 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.04 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'chirp':
        a = np.ones(n)
        for k in range(1, 10):
            fb = k * (1.0 + 0.5 * rng.random())
            cr = rng.uniform(-0.8, 0.8)
            ph = rng.uniform(0, 2 * np.pi)
            amp = 0.14 / np.sqrt(k)
            ft = fb + cr * t / n
            a += amp * np.cos(2 * np.pi * np.cumsum(ft) / n + ph)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'binary':
        a = np.ones(n)
        for _ in range(8):
            binary = rng.choice([0.4, 1.6], size=n)
            a += 0.08 * binary
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)


def hybrid_optimize(a_init, time_budget, rng, verbose=False):
    """
    Combined LP + gradient optimization with automatic switching.
    Uses LP for direction finding, switches to gradient when LP stagnates.
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a = a_init / np.sum(a_init) * S_target
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Phase A: LP-based
    t_lp = time_budget * 0.6
    a, best_score = _lp_phase(a, best_score, best_a, t0, t_lp, rng, verbose)
    best_a = a.copy()

    # Phase B: Gradient-based refinement
    t_grad = time_budget * 0.25
    remaining = max(0.0, time_budget - (time.monotonic() - t0))
    t_grad = min(t_grad, remaining * 0.6)
    if t_grad > 5:
        a, best_score = _grad_phase(a, best_score, t0, t_grad, rng)
        best_a = a.copy()

    # Phase C: Short LP restart from gradient result
    remaining = max(0.0, time_budget - (time.monotonic() - t0))
    t_lp2 = min(time_budget * 0.15, remaining * 0.5)
    if t_lp2 > 5:
        a, best_score = _lp_phase(a, best_score, best_a, time.monotonic(), t_lp2, rng, False)
        best_a = a.copy()

    return best_a, best_score


def _lp_phase(a, best_score, best_a, t0, time_budget, rng, verbose):
    """Run LP iterations until time runs out or convergence."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_budget:
        M = np.max(fft_convolve(a, a))
        g = lp_direction(a, M)
        gs = np.sum(g)
        if gs < 1e-10:
            break
        g = g / gs * S_target
        diff = np.linalg.norm(g - a) / (np.linalg.norm(a) + 1e-10)
        if diff < 1e-5:
            stagnant += 1
            if stagnant > 30:
                break
            a = np.clip(a * np.exp(rng.normal(0, 0.08, n)), 0, None)
            a = a / np.sum(a) * S_target
            continue
        stagnant = 0

        g_peak = np.max(fft_convolve(g, g))
        if g_peak <= M * 1.02:
            t_step = 0.2
        else:
            t_step = max(0.02, min(0.2, 0.3 * M / (g_peak - M + 1e-10)))

        a_new = np.clip((1 - t_step) * a + t_step * g, 0, None)
        if np.sum(a_new) < 1e-10:
            continue
        a_new = a_new / np.sum(a_new) * S_target
        s_new = compute_score(a_new)
        if s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
        a = a_new
        it += 1
        if verbose and it % 50 == 0:
            print(f"      LP it={it}, score={s_new:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")

    return a, best_score


def _grad_phase(a, best_score, t0, time_budget, rng):
    """Run gradient iterations until time runs out."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    no_imp = 0

    while time.monotonic() - t0 < time_budget:
        step += 1
        c = fft_convolve(a, a)
        c_max = np.max(c)
        frac = min(1.0, (time.monotonic() - t0) / time_budget)
        tau = max(0.002 / n, (0.5 / n) * (1.0 - frac) ** 2)
        shifted = (c - c_max) / tau
        exp_sh = np.exp(shifted - np.max(shifted))
        w = exp_sh / np.sum(exp_sh)
        corr = fft_convolve(w, a[::-1])
        grad = 2.0 * corr[n - 1:2 * n - 1]

        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)
        lr = 3.5 / np.sqrt(step + 1)
        a_new = np.clip(a - lr * m_hat / (np.sqrt(v_hat) + eps), 0, None)
        if np.sum(a_new) < 1e-10:
            a_new = a
        else:
            a_new = a_new / np.sum(a_new) * S_target
        s_new = compute_score(a_new)
        if s_new < best_score:
            best_score = s_new
            no_imp = 0
        else:
            no_imp += 1
        a = a_new

        if no_imp > 200:
            a = np.clip(a * np.exp(rng.normal(0, 0.1, n)), 0, None)
            a = a / np.sum(a) * S_target
            no_imp = 0

    return a, best_score
