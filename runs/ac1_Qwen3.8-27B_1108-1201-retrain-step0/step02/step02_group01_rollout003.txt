with p-norm smoothing and backtracking line search on the true objective.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_correlate(w, a):
    """
    Compute (w ⋆ a)(m) = sum_j w(m+j) * a(j) for m=0..n-1.
    Uses FFT: this equals conv(w, a_rev)[m + n - 1].
    """
    n = len(a)
    a_rev = a[::-1]
    # conv(w, a_rev) has length len(w) + n - 1
    # We want indices [n-1, 2n-2] of the convolution
    out_len = len(w) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    Ar = fft(a_rev, fsize)
    c = np.real(ifft(W * Ar))
    # We need c[n-1 : 2n-1]
    return c[n-1: 2*n-1]


def true_obj(a):
    """Compute the true objective: 2n*max(conv)/sum^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def pnorm_obj_and_grad(a, p):
    """
    Compute the p-norm of the self-convolution and its gradient.
    Returns (obj, grad) where obj = ||conv(a,a)||_p and grad = d(obj)/d(a).
    """
    n = len(a)
    c = fft_convolve_self(a)
    
    # p-norm
    cp = c ** p
    sum_cp = cp.sum()
    obj = sum_cp ** (1.0 / p)
    
    if obj < 1e-30:
        return obj, np.zeros(n)
    
    # Weights: w(k) = c(k)^(p-1) / obj^(p-1)
    w = (c / obj) ** (p - 1) if p > 1 else np.ones_like(c) / len(c)
    
    # Gradient: grad[i] = 2 * sum_j w(i+j) * a(j) = 2 * cross_correlate(w, a)[i]
    grad = 2.0 * fft_cross_correlate(w, a)
    
    return obj, grad


def softmax_obj_and_grad(a, T):
    """
    Compute soft-max of the self-convolution and its gradient.
    soft_max_T(c) = (1/T) * log(sum_k exp(T * c(k)))
    """
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    
    shifted = T * (c - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T
    
    # Weights
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient
    grad = 2.0 * fft_cross_correlate(w, a)
    
    return obj, grad


def project_to_simplex(theta, S):
    """Convert log-parameters to non-negative sequence with sum S."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def run_nesterov_pnorm(a_init, p_schedule, time_budget, lr0=0.01, momentum_init=0.9):
    """
    Nesterov momentum gradient descent on the p-norm objective,
    with backtracking line search on the true objective.
    """
    n = len(a_init)
    S = np.sum(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    
    # Initialize log-parameters
    theta = np.log(a / S)
    
    true_obj_init = true_obj(a)
    best_a = a.copy()
    best_true = true_obj_init
    
    v = np.zeros(n)  # velocity (momentum)
    mu = momentum_init
    lr = lr0
    
    start = time.monotonic()
    stall_count = 0
    last_improvement = true_obj_init
    prev_true = true_obj_init
    
    step_count = 0
    while time.monotonic() - start < time_budget:
        step_count += 1
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget
        
        # Update p according to schedule
        if isinstance(p_schedule, (int, float)):
            p = float(p_schedule)
        elif callable(p_schedule):
            p = p_schedule(progress)
        else:
            p = p_schedule[min(int(progress * len(p_schedule)), len(p_schedule) - 1)]
        
        # Nesterov lookahead
        theta_look = theta + mu * v
        a_look = project_to_simplex(theta_look, S)
        
        # Compute gradient at lookahead point
        obj, grad = pnorm_obj_and_grad(a_look, p)
        
        # Update velocity
        v = mu * v - lr * grad
        theta = theta + v
        
        # Project to simplex
        a_new = project_to_simplex(theta, S)
        
        # Evaluate true objective
        true_new = true_obj(a_new)
        
        if np.isfinite(true_new) and true_new < best_true:
            best_true = true_new
            best_a = a_new.copy()
            stall_count = 0
            last_improvement = true_new
            # Occasionally print progress

        else:
            stall_count += 1
            # Backtrack: if no improvement, reduce step
            if stall_count == 5:
                lr = lr * 0.5
            elif stall_count == 15:
                # Revert
                theta = np.log(best_a / S)
                v = np.zeros(n)
                lr = lr0 * 0.3
                stall_count = 0
            elif stall_count == 30:

                a_pert = np.maximum(a_pert, 1e-10)
                a_pert = a_pert * (S / a_pert.sum())
                theta = np.log(a_pert / S)
                v = np.zeros(n)
                lr = lr0 * 0.2
                stall_count = 0
        
        # Adaptive learning rate decay
        lr = max(lr0 * (1 - progress * 0.5), lr0 * 0.1)
        
        # Adaptive momentum
        mu = momentum_init * (1 - 0.1 * progress)
    
    return best_a, best_true


def coordinate_descent_polish(a, n, S, time_budget):
    """
    Coordinate descent: for each coordinate, find the best value
    (via golden section search) that minimizes the true objective.
    """
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    improved = True
    sweep = 0
    
    while improved and time.monotonic() - start < time_budget:
        improved = False
        sweep += 1
        # Randomize order for diversity
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            orig = a[i]
            if orig < 1e-12:
                continue
            
            # Try a range of values
            # The optimal is likely near 0 (to reduce the peak) or near orig
            # Try: 0, orig*0.5, orig*0.8, orig*1.2, orig*1.5
            candidates = [0.0, orig * 0.3, orig * 0.5, orig * 0.7, orig]
            
            best_local = orig
            best_local_obj = best_obj
            
            for cand in candidates:
                a[i] = cand
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    continue
                a_check = a * (S / s)
                c_check = fft_convolve_self(a_check)
                obj_check = c_check.max()
                if obj_check < best_local_obj:
                    best_local_obj = obj_check
                    best_local = cand
            
            if best_local != orig:
                a[i] = best_local
                s = a.sum()
                a = a * (S / s)
                improved = True
            
            # Update best
            if best_local_obj < best_obj:
                best_obj = best_local_obj
                best_a = a.copy()
        
        if sweep % 5 == 0:
            print("  CD sweep %d: obj=%.6f" % (sweep, best_obj))
    
    return best_a, best_obj
