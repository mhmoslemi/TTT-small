For the Frank-Wolfe approach on the simplex `{a >= 0, sum a = 1}`:
1. Compute subgradient `g` of the objective.
2. Find `s = argmin_{b in simplex} g · b = e_{j*}` where `j* = argmin_j g[j]`.
3. Update: `a ← (1-t)*a + t*s` for some step size `t`.

"""
Multi-phase optimization for minimum peak self-convolution non-negative sequences.
Uses Frank-Wolfe, projected subgradient, L-BFGS with soft-max, and local search.
Employs multi-scale strategy and diverse initializations including U-shaped priors.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def conv_self(a):
    """Compute a * a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s

def score_of(a):
    """Compute 2n * max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    c = conv_self(a)
    return 2.0 * n * np.max(c)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    csv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (csv - 1))[0][-1]
    theta = (csv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def soft_max_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).
    """
    n = len(theta)
    a = softmax(theta)
    c = conv_self(a)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    
    score = 2.0 * n * soft_max_val
    
    # Gradient: d(soft_max)/da = sum_k w_k * da c[k] / da
    # da c[k] / da[j] = 2 * a[k-j] (if valid)
    # So grad_a[j] = 2 * sum_k w_k * a[k-j]
    # This is 2 * (w * a_reversed) [n-1:2n-1]
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta

def kwtopk_obj_grad(theta, k):
    """
    Objective and gradient for 2n * kth_largest(conv(softmax(theta), softmax(theta))).
    Uses the identity: kth_largest(x) = min_t {t + (1/k) * sum_i max(0, x_i - t)}
    """
    n = len(theta)
    a = softmax(theta)
    c = conv_self(a)
    
    # Find the k-th largest value
    c_sorted = np.sort(c)[::-1]
    kth_val = c_sorted[min(k, len(c)) - 1]
    
    # Active set: elements >= kth_val
    active = c >= kth_val - 1e-12
    n_active = np.sum(active)
    if n_active == 0:
        return 2.0 * n * kth_val, np.zeros(n)
    
    # The k-th largest = average of top k elements (when there's a unique k-th)
    # More precisely, d(kth_largest)/dc[j] = 1/k if c[j] is in the top k
    w = np.zeros(len(c))
    top_k_idx = np.argsort(c)[::-1][:k]
    w[top_k_idx] = 1.0 / k
    
    score = 2.0 * n * np.sum(w * c)
    
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta

def make_u_init(n, alpha, rng):
    """U-shaped initialization: large at edges, small in middle."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 + alpha * ((t - center) / center) ** 2
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)

def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)

def make_cosine_u_init(n, alpha, rng):
    """Inverted cosine initialization (high at edges)."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t / (n - 1))
    a = np.clip(a, 0.001, None)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)

def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)

def projected_subgradient(a_init, time_limit, rng, verbose=True):
    """
    Projected subgradient method with adaptive step size.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    current = a_init.astype(np.float64).copy()
    current = project_simplex(current)
    current_score = score_of(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    eta = 0.05  # Initial step size
    decay = 0.9995
    no_improve = 0
    patience = 500
    
    it_count = 0
    last_print = t0
    
    while time.monotonic() - t0 < time_limit:
        c = conv_self(current)
        c_max = np.max(c)
        peak_idx = np.argmax(c)
        
        # Subgradient of max at the peak
        # grad[j] = 2 * a[peak_idx - j] (for valid indices)
        # Use soft-max for smoother gradient
        tau = eta * c_max * 0.1
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        w = exp_shifted / np.sum(exp_shifted)
        
        a_rev = current[::-1]
        cross = fft_convolve(a_rev, w)
        grad = 2.0 * cross[n - 1:2 * n - 1]
        
        # Take a step
        candidate = current - eta * grad
        candidate = project_simplex(candidate)
        candidate = np.maximum(candidate, 0)
        candidate = candidate / np.sum(candidate)
        
        new_score = score_of(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            no_improve += 1
        
        # Adapt step size
        if no_improve > patience:
            eta *= 0.9
            no_improve = 0
        else:
            eta = max(0.001, eta * decay)
        
        it_count += 1
        
        if verbose and time.monotonic() - last_print > 5.0:
            last_print = time.monotonic()
            print(f"    PSG: score={current_score:.6f}, best={best_score:.6f}, "
                  f"eta={eta:.6f}, it={it_count}, t={time.monotonic()-t0:.0f}s")
    
    return best, best_score

def frank wolfe_step(a, c, w):
    """One Frank-Wolfe step."""
    n = len(a)
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad = 2.0 * cross[n - 1:2 * n - 1]
    # Linear minimization over simplex: pick the coordinate with min gradient
    j_star = np.argmin(grad)
    # Return the vertex e_{j_star}
    return j_star

def frank_wolfe(a_init, time_limit, rng, verbose=True):
    """
    Frank-Wolfe (conditional gradient) method for min max(conv(a,a)) s.t. a in simplex.
    """
    current = a_init.astype(np.float64).copy()
    current = project_simplex(current)
    current_score = score_of(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    it_count = 0
    last_print = t0
    rate = 2.0 / 3.0  # FW step size rate
    
    while time.monotonic() - t0 < time_limit:
        c = conv_self(current)
        c_max = np.max(c)
        
        # Use soft-max weights for the active set
        tau = max(0.01 * c_max, 1e-6)
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        w = exp_shifted / np.sum(exp_shifted)
        
        # Gradient
        j_star = frank_wolfe_step(current, c, w)
        
        # Step size: t_k = 2/(k+2)
        t_step = rate / (it_count + 1)
        
        # Update: a <- (1-t)*a + t*e_{j_star}
        candidate = (1 - t_step) * current
        candidate[j_star] += t_step
        
        new_score = score_of(candidate)
        
        if new_score < best_score:
            best_score = new_score
            best = candidate.copy()
        
        current = candidate
        it_count += 1
        
        if verbose and time.monotonic() - last_print > 5.0:
            last_print = time.monotonic()
            print(f"    FW: score={current_score:.6f}, best={best_score:.6f}, "
                  f"it={it_count}, t={time.monotonic()-t0:.0f}s")
            current_score = new_score
    
    return best, best_score

def local_search_peak_reduction(a_init, time_limit, rng, verbose=True):
    """
    Local search that specifically targets peak reduction.
    At each step, identify the top peaks and reduce the elements contributing most to them.
    """
    current = a_init.astype(np.float64).copy()
    current_score = score_of(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    temperature = 0.003
    cooling = 0.9999
    last_print = t0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        c = conv_self(current)
        c_max = np.max(c)
        
        r = rng.random()
        candidate = current.copy()
        
        if r < 0.40:
            # Targeted peak reduction
            # Find top few peaks
            top_peaks = np.argsort(c)[::-1][:5]
            for pk in top_peaks:
                # Find elements contributing most to this peak
                contribs = np.zeros(n)
                lo = max(0, pk - n + 1)
                hi = min(n - 1, pk)
                for i in range(lo, hi + 1):
                    j = pk - i
                    if 0 <= j < n:
                        contribs[i] = current[i] * current[j]
                # Reduce top contributors
                top_k = max(2, n // 20)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
                # Increase some others to maintain sum
                other = np.setdiff1d(np.arange(n), top_idx)
                if len(other) > 0:
                    n_inc = min(10, len(other))
                    inc_idx = rng.choice(other, size=n_inc, replace=False)
                    candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.60:
            # Smoothed redistribution
            c_avg = np.mean(c)
            pull = np.zeros(n)
            # For each peak above average, pull down contributors
            for pk in np.where(c > c_avg * 1.1)[0]:
                weight = c[pk] - c_avg
                lo = max(0, pk - n + 1)
                hi = min(n - 1, pk)
                for i in range(lo, hi + 1):
                    j = pk - i
                    if 0 <= j < n:
                        pull[i] += weight * current[j]
            if np.max(pull) > 1e-10:
                pull = pull / np.max(pull)
                candidate *= np.exp(-0.03 * pull)
        elif r < 0.80:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))
        else:
            # Global perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.02, size=n))
        
        candidate = np.maximum(candidate, 1e-10)
        candidate = candidate / np.sum(candidate)
        new_score = score_of(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score
        
        temperature = max(temperature * cooling, 1e-8)
        it_count += 1
        
        if verbose and time.monotonic() - last_print > 5.0:
            last_print = time.monotonic()
            print(f"    LSPR: score={current_score:.6f}, best={best_score:.6f}, "
                  f"temp={temperature:.6f}, it={it_count}, t={time.monotonic()-t0:.0f}s")
    
    return best, best_score

def upsample(a, new_n):
    """Upsample sequence a to length new_n using linear interpolation."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-8)
    return a_new / np.sum(a_new)

def run_lbgfs(theta, n, objective_type='softmax', param_schedule=None, maxiter=300, deadline=float('inf')):
    """Run L-BFGS with given objective."""
    if objective_type == 'softmax':
        func = soft_max_obj_grad
    elif objective_type == 'kwtopk':
        func = kwtopk_obj_grad
    else:
        func = soft_max_obj_grad
    
    best_theta = theta.copy()
    best_score = score_of(softmax(theta))
    
    for param in param_schedule:
        if time.monotonic() > deadline - 10:
            break
        
        result = minimize(
            func, theta.copy(),
            method='L-BFGS-B',
            jac=True,
            args=(param,),
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': maxiter * 5}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = score_of(a_new)
        
        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        theta = best_theta
    
    return best_theta, best_score

    """

import numpy as np

from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def conv_self(a):
    """Compute a * a (self-convolution) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def score_of(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = conv_self(a)
    return 2.0 * n * np.max(c)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    csv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (csv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (csv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def soft_max_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).
    """
    n = len(theta)
    a = softmax(theta)
    c = conv_self(a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta

    """U-shaped initialization: large at edges, small in middle."""
    t = np.arange(n, dtype=np.float64)

    a = 1.0 + alpha * ((t - center) / center) ** 2
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)

            candidate = current * np.exp(rng.normal(0.0, 0.012, size=n))

        candidate = np.maximum(candidate, 1e-10)
        candidate = candidate / np.sum(candidate)
        new_score = score_of(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 1e-9)
        it_count += 1

        if verbose and time.monotonic() - last_print > 5.0:
            last_print = time.monotonic()
            print(f"    LSA: score={current_score:.6f}, best={best_score:.6f}, "
                  f"temp={temperature:.6f}, it={it_count}, t={time.monotonic()-t0:.0f}s")

    return best, best_score


def fine_polish(a_init, time_limit, rng, verbose=True):
    """
    Very fine-grained polish with tiny perturbations.
    """
    current = a_init.astype(np.float64).copy()
    current_score = score_of(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    last_print = t0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        c = conv_self(current)
        peak_idx = int(np.argmax(c))

        # Compute gradient at the peak
        grad = np.zeros(n)
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        for i in range(lo, hi + 1):
            j = peak_idx - i
            if 0 <= j < n:
                grad[i] += 2.0 * current[j]

        # Gradient step with small step size
        eta = 0.002
        candidate = current - eta * grad / (np.max(np.abs(grad)) + 1e-10)
        candidate = project_simplex(candidate)

        new_score = score_of(candidate)
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()

        # Also try random tiny perturbations
        if rng.random() < 0.5:
            idx = rng.integers(0, n)
            delta = rng.normal(0, 0.001) * current[idx]
            candidate2 = current.copy()
            candidate2[idx] += delta
            candidate2 = np.maximum(candidate2, 1e-10)
            candidate2 = candidate2 / np.sum(candidate2)
            s2 = score_of(candidate2)
            if s2 < current_score:
                current = candidate2
                current_score = s2
                if s2 < best_score:
                    best_score = s2
                    best = current.copy()

        it_count += 1
        if verbose and time.monotonic() - last_print > 5.0:
            last_print = time.monotonic()
            print(f"    POLISH: score={current_score:.6f}, best={best_score:.6f}, "
                  f"it={it_count}, t={time.monotonic()-t0:.0f}s")

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimum peak self-convolution non-negative sequences.

    Strategy:
    1. Multi-start L-BFGS with diverse initializations (U-shaped, chirp, random).
    2. Projected subgradient refinement.
    3. Multi-scale: optimize at smaller n, upsample, refine.
    4. Intensive advanced local search with targeted peak reduction.
    5. Fine-grained gradient polish.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        if np.sum(start_seq) > 0:
            start_seq = start_seq / np.sum(start_seq)
            s = score_of(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            best_score = s
            best_seq = start_seq.copy()

    # === Phase 1: Multi-start L-BFGS ===
    print("\n=== Phase 1: Multi-start L-BFGS ===")

    n_target = 256
    if start_seq is not None:
        n_target = len(start_seq)

    lengths_to_try = [256]
    if n_target != 256:
        lengths_to_try.insert(0, n_target)
    # Also try a couple other lengths
    if 192 not in lengths_to_try:
        lengths_to_try.append(192)
    if 320 not in lengths_to_try:
        lengths_to_try.append(320)

    for n in lengths_to_try:
        if time.monotonic() > deadline - 400:
            break

        time_alloc = min(250.0, max(80.0, (deadline - time.monotonic() - 400) / max(1, len(lengths_to_try))))
        n_deadline = time.monotonic() + time_alloc

        # Generate diverse initializations
        inits = []

        if start_seq is not None:
            a_interp = upsample(start_seq, n)
            inits.append(('parent', a_interp))

        if best_seq is not None and len(best_seq) != n:
            a_best_interp = upsample(best_seq, n)
            inits.append(('best_interp', a_best_interp))

        inits.append(('u1', make_u_init(n, 0.8 + 0.4 * rng.random(), rng)))
        inits.append(('u2', make_u_init(n, 2.0 + 1.5 * rng.random(), rng)))
        inits.append(('u3', make_u_init(n, 4.0 + 3.0 * rng.random(), rng)))
        inits.append(('chirp', make_chirp_init(n, 0.2 + 0.5 * rng.random(), rng)))
        inits.append(('cos_u', make_cosine_u_init(n, 0.3 + 0.7 * rng.random(), rng)))
        inits.append(('gauss', make_gaussian_init(n, n * (0.2 + 0.2 * rng.random()), rng)))
        inits.append(('rand', make_random_init(n, rng, spread=0.4)))

        # Perturbed versions of best
        if best_seq is not None:
            a_best_n = upsample(best_seq, n)
            for pert in [0.05, 0.1]:
                a_pert = a_best_n * np.exp(rng.normal(0, pert, n))
                a_pert = np.maximum(a_pert, 1e-6)
                a_pert = a_pert / np.sum(a_pert)
                inits.append((f'pert{pert:.2f}', a_pert))

        tau_schedule = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005]

        for init_name, a0 in inits:
            if time.monotonic() > n_deadline:
                break

            theta = np.log(a0 + 1e-30)
            maxiter = 300 if n <= 256 else 200

            # Run soft-max L-BFGS
            theta_best, s = run_lbgfs(theta, 'softmax', tau_schedule,
                                       maxiter=maxiter, deadline=n_deadline)
            if s < best_score:
                best_score = s
                best_seq = softmax(theta_best).copy()
                print(f"  n={n}, init='{init_name}': score={s:.6f} *** NEW BEST ***")
            else:
                print(f"  n={n}, init='{init_name}': score={s:.6f}")

            # Try k-top-k refinement from the best so far
            if time.monotonic() < n_deadline - 15:
                theta_k = np.log(softmax(theta_best) + 1e-30)
                for k in [5, 10, 20]:
                    if time.monotonic() > n_deadline - 10:
                        break
                    theta_k_best, s_k = run_lbgfs(theta_k, 'kwtopk', [k],
                                                   maxiter=150, deadline=n_deadline)
                    if s_k < best_score:
                        best_score = s_k
                        best_seq = softmax(theta_k_best).copy()
                        print(f"  n={n}, ktopk(k={k}): score={s_k:.6f} *** NEW BEST ***")
                        theta_k = np.log(softmax(theta_k_best) + 1e-30)

        print(f"  After n={n}: best_score={best_score:.6f}")

    # === Phase 2: Projected Subgradient ===
    print(f"\n=== Phase 2: Projected Subgradient (score={best_score:.6f}) ===")
    if best_seq is not None:
        psg_time = min(90.0, max(20.0, (deadline - time.monotonic() - 300) * 0.15))
        print(f"  PSG time: {psg_time:.0f}s")
        psg_best, psg_score = projected_subgradient(best_seq, psg_time, rng, verbose=True)
        if psg_score < best_score:
            best_score = psg_score
            best_seq = psg_best.copy()
            print(f"  After PSG: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  After PSG: {psg_score:.6f} (no improvement)")

    # === Phase 3: Multi-scale refinement ===
    print(f"\n=== Phase 3: Multi-scale refinement (score={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        n_cur = len(best_seq)
        n_small = max(32, n_cur // 2)
        a_small = best_seq[::2][:n_small]
        a_small = a_small / np.sum(a_small)

        theta_small = np.log(a_small + 1e-30)
        t_small = min(60.0, (deadline - time.monotonic() - 200) * 0.2)
        td = time.monotonic() + t_small
        theta_small_best, s_small = run_lbgfs(theta_small, 'softmax',
                                               [0.01, 0.005, 0.001], maxiter=300, deadline=td)

        a_up = upsample(softmax(theta_small_best), n_cur)
        s_up = score_of(a_up)
        if s_up < best_score:
            best_score = s_up
            best_seq = a_up.copy()
            print(f"  Multi-scale upsample: {best_score:.6f} *** NEW BEST ***")
        else:
            # Try blending
            blend = 0.5 * best_seq + 0.5 * a_up
            blend = blend / np.sum(blend)
            s_blend = score_of(blend)
            if s_blend < best_score:
                best_score = s_blend
                best_seq = blend.copy()
                print(f"  Multi-scale blend: {best_score:.6f} *** NEW BEST ***")
            else:
                print(f"  Multi-scale: up={s_up:.6f}, blend={s_blend:.6f} (no improvement)")

    # === Phase 4: Intensive Local Search ===
    print(f"\n=== Phase 4: Advanced Local Search (score={best_score:.6f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 60
        if remaining > 60:
            n_trials = 3
            t_per_trial = remaining / n_trials
            for trial in range(n_trials):
                if time.monotonic() > deadline - 60:
                    break
                t_this = min(t_per_trial, deadline - 60 - time.monotonic())
                if t_this < 10:
                    break

                if trial == 0:
                    a_start = best_seq.copy()
                elif trial == 1:
                    a_start = best_seq * np.exp(rng.normal(0, 0.03, len(best_seq)))
                    a_start = a_start / np.sum(a_start)
                else:
                    a_start = best_seq * np.exp(rng.normal(0, 0.08, len(best_seq)))
                    a_start = np.maximum(a_start, 1e-6)
                    a_start = a_start / np.sum(a_start)

                polished, p_score = local_search_adv(a_start, t_this, rng, verbose=True)
                if p_score < best_score:
                    best_score = p_score
                    best_seq = polished.copy()
                    print(f"  Trial {trial}: {best_score:.6f} *** NEW BEST ***")
                else:
                    print(f"  Trial {trial}: {p_score:.6f} (no improvement)")

    # === Phase 5: Fine Gradient Polish ===
    print(f"\n=== Phase 5: Fine Gradient Polish (score={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        polish_time = min(60.0, deadline - 20 - time.monotonic())
        if polish_time > 10:
            polished, p_score = fine_polish(best_seq, polish_time, rng, verbose=True)
            if p_score < best_score:
                best_score = p_score
                best_seq = polished.copy()
                print(f"  After polish: {best_score:.6f} *** NEW BEST ***")
            else:
                print(f"  After polish: {p_score:.6f} (no improvement)")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
