LP-guided hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based direction finding, mirror descent, and informed annealing.
Explores multiple sequence lengths with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_nonneg_sum1(v):
    """Project onto {x >= 0, sum(x) = 1} using simplex projection."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_subgradient(conv, a, temperature):
    """Compute smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = temperature * (conv - max_c) / max_c
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    # Only process positions with significant weight
    thresh = w.max() * 0.001
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    return sg

        log_f -= log_f.max()

        if s < 1e-12:

def build_constraint_matrix(f):
    """Build the constraint matrix A for the LP.
    A[k, j] = f[k-j] if max(0,k-n+1) <= j <= min(n-1,k), else 0
    """
    n = len(f)
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    return A


def lp_direction(f, M, n_max_constraints=None):

    Maximize sum(g) subject to (f*g)_k <= M for all k, g >= 0.
    Returns g or None if LP fails.
    """
    n = len(f)
    c = -np.ones(n)  # minimize -sum(g)
    
    # Build constraint matrix
    A = build_constraint_matrix(f)
    b = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a0, time_limit):

    a = np.maximum(a0, 0.0).copy()

    s = a.sum()
    if s < 1e-12:

        if M < 1e-15:

"""
LP-guided hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding (state-of-the-art), mirror descent,
and informed annealing. Explores multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_normalized(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_subgradient(conv, a, temperature):
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = temperature * (conv - max_c) / max_c
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    thresh = w.max() * 0.001
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    return sg


def mirror_descent(a0, time_limit, t_init=50.0, eta_init=0.3):
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    best_obj = obj_normalized(a)
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    max_pat = 40
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        sg = softmax_subgradient(conv, a, t)
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a = a / s
        conv = fast_convolve(a)
        cur_obj = obj_normalized(a)
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            eta *= 0.6
            t *= 1.4
            if eta < 0.002:
                elapsed = time.monotonic() - t0
                frac = min(1.0, elapsed / max(time_limit, 1.0))
                eta = max(eta_init * (1.0 - 0.5 * frac), 0.001)
                t = t_init * (1.0 + 4.0 * frac)
    return best_a, best_obj


def build_lp_matrix(f):
    """Build constraint matrix for LP: A[k,j] = f[k-j] if valid."""
    n = len(f)
    num_row = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(num_row):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        length = j_end - j_start + 1
        rows.extend([k] * length)
        cols.extend(range(j_start, j_end + 1))
        # f[k-j] for j from j_start to j_end
        # = f[k-j_end] to f[k-j_start] in reverse
        vals.extend(f[k - j_end:k - j_start + 1][::-1])
    return csr_matrix((vals, (rows, cols)), shape=(num_row, n))


def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A = build_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_optimize(a0, time_limit):
    """LP-guided optimization with line search."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    best_obj = obj_normalized(a)
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        g = lp_direction(a, M)
        if g is None or g.sum() < 1e-12:
            break
        g = g / g.sum()
        
        # Line search
        best_t = 0.0
        best_t_obj = best_obj  # use current best as reference
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-12:
                continue
            a_trial = a_trial / st
            obj_trial = obj_normalized(a_trial)
            if obj_trial < best_t_obj:
                best_t_obj = obj_trial
                best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            conv = fast_convolve(a)
            cur_obj = obj_normalized(a)
            if cur_obj < best_obj - 1e-10:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve >= 15:
            break
    
    return best_a, best_obj


def anneal_informed(a0, time_limit, rng):
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    a = a / a.sum()
    conv = fast_convolve(a)
    cur_obj = obj_normalized(a)
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.02 * (1.0 - frac) ** 2 + 0.0003
        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.7
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total = products.sum()
        if total < 1e-15:
            continue
        probs = products / total
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        target_j = int(rng.integers(n))
        delta = a[top_j] * rng.uniform(0.01, 0.05)
        if delta < 1e-9:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = obj_normalized(a)
        d_obj = new_obj - cur_obj
        if d_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-d_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta

def generate_inits(n, rng, parent):
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        inits['gauss_' + str(sf)] = g / g.sum()
    for freq in [1, 2, 3, 5]:
        for amp in [0.3, 0.6]:
            s = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
            s = np.maximum(s, 0.0)
            if s.sum() > 1e-10:
                inits['cos_%d_%d' % (freq, int(amp*10))] = s / s.sum()
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    for alpha in [0.5, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['rand_%d' % int(alpha*10)] = r
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0).copy()
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits['parent_interp'] = pp / pp.sum()
    return inits
