```python
import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / s**2

def solve_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0"""
    n = len(f)
    # Build constraint matrix A: A @ g <= M * ones
    # (f*g)_k = sum_j f[k-j] * g_j for valid j
    # A has shape (2n-1, n)
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
            A[k, j] = f[k - j]
    
    b = M * np.ones(2*n - 1)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x, -res.fun
    return None, 0

def optimize_lp(a_init, time_limit, rng):
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sqrt(2 * n)
    a = a / np.sum(a) * S  # normalize
    
    best_score = compute_score(a)
    best_a = a.copy()
    t0 = time.monotonic()
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = np.max(fft_convolve(a, a))
        
        g, g_sum = solve_lp(a, M)
        if g is None:
            break
        
        if g_sum <= np.sum(a) * (1 + 1e-10):
            # No improvement possible
            break
        
        # Update with small step
        g_norm = g / g_sum * S
        t = 0.1  # step size
        new_a = (1 - t) * a + t * g_norm
        new_a = np.clip(new_a, 0, None)
        
        new_score = compute_score(new_a)
        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
            a = new_a
        else:
            # Try smaller step
            t = 0.01
            new_a = (1 - t) * a + t * g_norm
            new_a = np.clip(new_a, 0, None)
            new_score = compute_score(new_a)
            if new_score < best_score:

                best_a = new_a.copy()
                a = new_a
            else:
                break
        
        if it % 10 == 0:
            print(f"  iter {it}, M={M:.6f}, score={best_score:.8f}")
    
    return best_a, best_score
```

```python
import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def score(a):
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = a.sum()
    if s < 0.01: return np.inf
    return 2*n*np.max(fft_convolve(a,a))/s**2

def lp_update(f, M, max_sum=None):
    """Solve max sum(g) s.t. f*g <= M (element-wise), g >= 0, sum(g) <= max_sum"""
    n = len(f)
    # Build A matrix where (A @ g)[k] = (f*g)[k]
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(k, n-1)
        for j in range(j_start, j_end+1):
            A[k, j] = f[k-j]
    
    b = M * np.ones(2*n-1)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    # Add sum constraint
    if max_sum is not None:
        A_eq = np.ones((1, n))
        b_eq = np.array([max_sum])
        res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    else:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return None

def optimize(a_init, time_limit, rng, step_sizes=None):
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sqrt(2*n)
    a = a / a.sum() * S
    
    best_score = score(a)
    best_a = a.copy()
    t0 = time.monotonic()
    
    if step_sizes is None:
        step_sizes = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fft_convolve(a, a)
        M = conv.max()
        
        g = lp_update(a, M, max_sum=S * 2)  # cap the sum to prevent huge solutions
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum <= a.sum() * (1 + 1e-9):

            if it > 5:
                break
            continue

        improved = False
        for t in step_sizes:
            new_a = (1-t)*a + t * g * S / g_sum
            new_a = np.clip(new_a, 0, None)
            if new_a.sum() < 0.01:

            s_new = score(new_a)
            if s_new < best_score:
                best_score = s_new
                best_a = new_a.copy()
                a = new_a.copy()
                improved = True
                break
        
        if not improved and it > 10:

            a = a_init.copy() / a_init.sum() * S
            a = np.clip(a + 0.1 * rng.standard_normal(n), 0, None)
            a = a / a.sum() * S
        
        if it % 20 == 0:
            print(f"  iter={it}, M={M:.6f}, best_score={best_score:.8f}")
    
    return best_a, best_score
```

```python
import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def compute_score(a):
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))

def build_conv_matrix(f):
    """Build matrix A where (A @ g)[k] = (f*g)[k]"""
    n = len(f)
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
            A[k, j] = f[k - j]
    return A

def solve_lp(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    b = M * np.ones(2*n - 1)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def optimize_sequence(a_init, time_limit, rng):
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = 1.0  # normalize to sum = 1
    a = a / np.sum(a) * S
    
    best_score = compute_score(a)
    best_a = a.copy()
    t0 = time.monotonic()
    
    step_sizes = [0.5, 0.3, 0.1, 0.05]
    it = 0
    stuck_count = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fft_convolve(a, a)
        M = np.max(conv)
        
        g = solve_lp(a, M)
        if g is None:
            break
        
        g_sum = np.sum(g)
        if g_sum <= S * (1 + 1e-9):
            stuck_count += 1
            if stuck_count > 20:
                break
            continue
        stuck_count = 0
        
        g_norm = g * S / g_sum
        
        improved = False
        for t in step_sizes:
            new_a = (1 - t) * a + t * g_norm
            new_a = np.clip(new_a, 0, None)
            if np.sum(new_a) < 0.001:
                continue
            new_score = compute_score(new_a)
            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()
                a = new_a.copy()
                improved = True
                break
        
        if not improved:
            stuck_count += 1
            if stuck_count > 30:
                # Random perturbation
                a = best_a + 0.01 * rng.standard_normal(n)
                a = np.clip(a, 0, None)
                a = a / np.sum(a)
                stuck_count = 0
        
        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  iter={it}, M={M:.8f}, best={best_score:.8f}, time={elapsed:.1f}s")
    
    return best_a, best_score
```

```python
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def compute_score(a):
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        js = range(max(0, k - n + 1), min(k, n - 1) + 1)
        A[k, list(js)] = f[k - np.array(list(js))]
    return A

def solve_lp(f, M, sum_cap):
    n = len(f)
    A = build_conv_matrix(f)
    b = M * np.ones(2*n - 1)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # Add sum constraint
    A_eq = np.ones((1, n))
    b_eq = np.array([sum_cap])
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
```
