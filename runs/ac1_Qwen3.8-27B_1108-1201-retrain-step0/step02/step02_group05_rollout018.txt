def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score: 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def pnorm_obj_grad(theta, p):

    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p.

"""
Multi-stage optimization for minimum peak autocorrelation non-negative sequences.
Uses aggressive p-norm L-BFGS (p up to 16384), exact coordinate descent with
vectorized 1D minimax, and multi-start perturbation. Improves on previous 1.5124
by more precise max-optimization and targeted local refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score: 2n * max(conv(a,a)). Assumes a is non-negative."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (np.sum(a) ** 2)


def pnorm_obj_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p / ||softmax||_1^2.
    Since softmax has sum 1, this is 2n * pnorm.
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    cp = c ** (p - 1)
    sum_cp_c = np.sum(cp * c)
    if sum_cp_c < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp_c ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = (sum_cp_c) ** (1.0 / p - 1.0)

    # T*(cp)[k] = 2 * sum_m cp[m+k] * a[m]
    # sum_m cp[m+k]*a[m] is the 'valid' cross-correlation of cp and a
    v = np.correlate(cp, a, mode='valid')
    grad_a = 4.0 * n * factor * v

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.5))
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Bell-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p, deadline):
    """Run L-BFGS with annealed p-norm. Returns best theta and score."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))

    for p in p_schedule:
        if time.monotonic() > deadline - 10:
            break

        try:
            result = minimize(
                pnorm_obj_grad, theta.copy(),
                method='L-BFGS-B',
                jac=True, args=(p,),
                options={'maxiter': maxiter_per_p, 'ftol': 1e-16,
                         'gtol': 1e-14, 'maxfun': maxiter_per_p * 10}
            )
            theta_new = result.x
            a_new = softmax(theta_new)
            s = local_score(a_new)

            if s < best_score:
                best_score = s
                best_theta = theta_new.copy()
            theta = best_theta
        except Exception:
            break

    return best_theta, best_score


def coordinate_descent(a_init, n_sweeps=100, time_limit=120.0, rng=None):
    """
    Coordinate descent with exact 1D minimax optimization.
    Exploits scale invariance of the score.
    Each step optimizes one coordinate exactly.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    c = fft_convolve(a, a)
    current_score = local_score(a)
    best_a = a.copy()
    best_score = current_score

    t0 = time.monotonic()
    if rng is None:
        rng = np.random.default_rng(12345)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        order = rng.permutation(n)

        improved = False
        for k in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[k]
            S = np.sum(a) - old_val  # sum of all other coordinates
            if S < 1e-30 and old_val < 1e-30:
                continue

            # Affected j range
            j_start = k
            j_end = min(2 * n - 2, k + n - 1)
            length_affected = j_end - j_start + 1

            if length_affected <= 0:
                continue

            # Build base and slope arrays for affected positions
            # c[j] = base[j] + slope[j]*x (linear) for most j
            # c[2k] = base[2k] + x^2 (quadratic)
            idx = np.arange(j_start, j_end + 1)
            m_idx = idx - k  # m = j - k, ranges 0 to n-1

            # base[j] = c[j] - current_contribution
            # contribution = 2*old_val*a[m] for linear, old_val^2 for quadratic
            base_arr = np.empty(length_affected)
            slope_arr = np.empty(length_affected)

            quad_pos = -1  # index in base_arr where quadratic is
            for ii, (j, m) in enumerate(zip(idx, m_idx)):
                if j == 2 * k:
                    base_arr[ii] = c[j] - old_val * old_val
                    slope_arr[ii] = 0.0
                    quad_pos = ii
                else:
                    base_arr[ii] = c[j] - 2.0 * old_val * a[m]
                    slope_arr[ii] = 2.0 * a[m]

            # Unaffected max
            if j_start > 0 or j_end < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[j_start:j_end + 1] = False
                if np.any(mask):
                    unaffected_max = np.max(c[mask])
                else:
                    unaffected_max = 0.0
            else:
                unaffected_max = 0.0

            if S <= 0:
                # Edge case: all mass is at k, can't do much
                continue

            # Define 1D score function (vectorized)
            x_upper = max(old_val * 3.0, np.max(a) * 2.0 + 1e-8)

            def score_1d(x):
                # max of affected: base + slope*x, plus quadratic at quad_pos
                lin_max = np.max(base_arr + slope_arr * x)
                if quad_pos >= 0:
                    quad_val = base_arr[quad_pos] + x * x
                else:
                    quad_val = -np.inf
                num = max(unaffected_max, lin_max, quad_val)
                den = (S + x) ** 2
                if den < 1e-30:
                    return 1e30
                return 2.0 * n * num / den

            # Minimize
            res = minimize_scalar(score_1d, bounds=(0.0, x_upper), method='bounded',
                                  options={'xatol': 1e-14})
            x_opt = max(0.0, res.x)

            if abs(x_opt - old_val) < 1e-15 * max(1.0, old_val):
                continue

            # Update
            delta = x_opt - old_val
            a[k] = x_opt

            # Update c incrementally
            for ii, (j, m) in enumerate(zip(idx, m_idx)):
                if j == 2 * k:
                    c[j] += delta * (x_opt + old_val)
                else:
                    c[j] += 2.0 * delta * a[m]

            new_score = 2.0 * n * np.max(c) / (np.sum(a) ** 2)
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                improved = True

        if not improved and sweep > 10:
            # No improvement in this sweep, try a few more
            pass

    # Normalize
    total = np.sum(best_a)
    if total > 0:
        best_a = best_a / total
    final_score = local_score(best_a)
    return best_a, final_score


def polish_with_sa(a_init, time_limit=120.0, rng=None):
    """
    Simulated annealing with targeted peak reduction.
    More aggressive than previous version.
    """
    if rng is None:
        rng = np.random.default_rng(67890)

    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score

    t0 = time.monotonic()
    temperature = 0.001
    cooling = 0.99995
    c = fft_convolve(current, current)

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.35:
            # Targeted peak reduction
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            # Find top contributors to the peak
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            for i in range(lo, hi + 1):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(15, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))
        elif r < 0.70:
            # Partial symmetrization
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.02, size=n))
        elif r < 0.85:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.02, size=n))
        else:
            # Two-point swap
            i, j = rng.choice(n, size=2, replace=False)
            amt = current[i] * rng.uniform(0.05, 0.3)
            candidate[i] -= amt
            candidate[j] += amt
            candidate = np.maximum(candidate, 1e-12)

        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 1e-10)

    return best, best_score
