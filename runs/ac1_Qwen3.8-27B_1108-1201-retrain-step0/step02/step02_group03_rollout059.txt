Uses: diverse initialization (monotone, GPS, random), gradient descent with

import time


def fft_convolve(a, b):

    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)

def local_score(a):
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))


def softmax_normalize(theta):

    e = np.exp(t)
    return e / np.sum(e)


def peak_grad(a):
    """Exact subgradient of max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    k = int(np.argmax(c))
    g = np.zeros(n)

    n = len(a)

    k_idx = np.argsort(c)[-K:]
    vals = c[k_idx]

    w = np.exp(vals / max(vals.max(), 1e-10) - 1.0)

def init_monotone(n, rng, direction=1, curvature=1.0):
    """Monotonically increasing/decreasing initialization."""
    t = np.arange(n, dtype=np.float64)
    if direction > 0:
        a = (t + 1) ** curvature
    else:
        a = (n - t) ** curvature
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 1e-6, None)
    return a / a.sum()


def init_gps(n, rng, n_iter=100):
    """Gerchberg-Saxton initialization."""
    N_fft = 1

        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = a.sum()
        if s < 1e-10:

"""
Peak-convolution minimization for non-negative step functions.
Key innovations over previous approach:
1. Monotone-biased initialization exploiting the fact that asymmetric sequences
   have off-center convolution peaks (breaking the symmetric lower bound of 2).
2. Exact peak subgradient (no soft-max approximation) for faster, more focused descent.
3. Structured local search using peak-blame vectors for directed mass redistribution.
4. Multi-scale optimization with aggressive monotone shape parameterization.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score: 2*n*max(conv)/sum^2."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))


def softmax_normalize(theta):
    """Stable softmax normalization."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def exact_peak_grad(a):
    """
    Compute exact subgradient of max_k conv(a,a)_k w.r.t. a.
    For conv_k = sum_i a_i * a_{k-i}, d(conv_k)/da_j = 2*a_{k-j} (valid indices).
    Returns: (convolution, gradient vector)
    """
    n = len(a)
    c = fft_convolve(a, a)
    k = int(np.argmax(c))
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k - idx]
    return c, g


def topk_smooth_grad(a, K=5, alpha=0.1):
    """
    Smoothed gradient using top-K peaks with exponential weighting.
    Provides smoother optimization landscape than exact peak.
    """
    n = len(a)
    c = fft_convolve(a, a)
    k_idx = np.argsort(c)[-K:]
    vals = c[k_idx]
    span = vals.max() - vals.min()
    if span < 1e-15:
        w = np.ones(K) / K
    else:
        normalized = (vals - vals.min()) / span
        w = np.exp(alpha * K * (normalized - 1.0))
        w = w / w.sum()
    g = np.zeros(n)
    for k, wi in zip(k_idx, w):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g[idx] += wi * 2.0 * a[k - idx]
    return c, g


def init_monotone_power(n, rng, exponent=1.0, direction=1, noise=0.02):
    """
    Monotone power-law initialization. Key insight: asymmetric sequences
    have convolution peaks away from center, allowing scores below 2.
    """
    t = np.arange(n, dtype=np.float64)
    if direction > 0:
        a = (t + 1.0) ** exponent
    else:
        a = (n - t + 1.0) ** exponent
    a += noise * rng.standard_normal(n)
    a = np.clip(a, 1e-8, None)
    return a / a.sum()


def init_gps(n, rng, n_iter=100):
    """Gerchberg-Saxton: flatten magnitude spectrum."""
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    a = np.ones(n) / n
    for _ in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def init_gps_monotone(n, rng, n_iter=120, bias_strength=0.4):
    """
    GPS with monotone bias: combines spectral flattening with asymmetry.
    This is the key initialization that breaks the symmetric bound.
    """
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    bias = 1.0 + bias_strength * t
    a = bias + 0.05 * rng.standard_normal(n)
    a = np.clip(a, 1e-6, None)
    a = a / a.sum()
    for i in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = a.sum()
        if s < 1e-10:
            a = bias / bias.sum()
        else:
            a = a / s
        # Gradually reduce bias in later iterations
        if i > n_iter // 2:
            fade = 1.0 - 0.3 * (i - n_iter // 2) / (n_iter // 2)
            a = a * (1.0 + 0.1 * fade * t)
            a = np.clip(a, 1e-8, None)
            a = a / a.sum()
    return a


def init_bimodal(n, rng, gap=0.3, width=0.2):
    """Bimodal initialization with a gap in the middle."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    center = 0.5
    sigma = width
    mode1 = np.exp(-((t - (center - gap)) ** 2) / (2 * sigma ** 2))
    mode2 = np.exp(-((t - (center + gap * 0.5)) ** 2) / (2 * sigma ** 2))
    a = 0.1 + 0.7 * mode1 + 0.9 * mode2
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 1e-6, None)
    return a / a.sum()

def optimize_gradient(n, a_init, time_budget, rng, K=3, lr_decay=0.5):
    """
    Adam gradient descent in softmax parameterization with exact peak subgradient.
    Normalizes to sum = sqrt(2n) for numerical convenience.
    """
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / a.sum() * S
    theta = np.log(a / S + 1e-12)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    step = 0
    best_score = local_score(a)
    best_theta = theta.copy()
    t0 = time.monotonic()
    
    # Progressive K: start with more peaks for smoothness, reduce for precision
    K_schedule = [max(5, K), K, K, 2, 1, 1]
    
    while time.monotonic() - t0 < time_budget:
        elapsed_frac = (time.monotonic() - t0) / time_budget
        # Determine current K
        frac_idx = min(int(elapsed_frac * len(K_schedule)), len(K_schedule) - 1)
        cur_K = K_schedule[frac_idx]
        
        a = softmax_normalize(theta) * S
        if cur_K <= 1:
            c, g_a = exact_peak_grad(a)
        else:
            c, g_a = topk_smooth_grad(a, K=cur_K, alpha=0.1)
        
        # Project gradient to theta-space (accounting for softmax constraint sum=1)
        inner = np.dot(g_a, a) / S
        g_theta = g_a - a * inner
        
        step += 1
        m = b1 * m + (1 - b1) * g_theta
        v = b2 * v + (1 - b2) * g_theta ** 2
        m_hat = m / (1.0 - b1 ** step)
        v_hat = v / (1.0 - b2 ** step)
        
        # Learning rate with decay
        lr = 3.0 / (1.0 + lr_decay * np.log1p(step)) ** 0.8
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if step % 500 == 0:
            a_check = softmax_normalize(theta) * S
            sc = local_score(a_check)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()
    
    a_final = softmax_normalize(best_theta) * S
    sc = local_score(a_final)
    if sc < best_score:
        best_score = sc
    return a_final, best_score


def optimize_direct(n, a_init, time_budget, rng):
    """
    Direct optimization in a-space with projection.
    Uses exact peak subgradient with adaptive step size.
    Good for final refinement after softmax optimization.
    """

def local_search_blame(a_init, time_budget, rng, K_peaks=5):
    """
    Structured local search using peak-blame vectors.
    Identifies which positions contribute most to top peaks and redirects mass.
    """

    score = local_score(a)
    best = a.copy()
    best_score = score
    n = len(a)

    temp = 0.003
    cooling = 0.9997
    iter_count = 0
    
    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a, a)
        top_k = np.argsort(c)[-K_peaks:]
        
        # Compute blame: contribution of each position to weighted top peaks
        blame = np.zeros(n)
        for k in top_k:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                blame[i] += a[i] * a[j] * c[k]
        
        blame = blame / (blame.max() + 1e-15)
        
        iter_count += 1
        r = rng.random()
        
        if r < 0.5:
            # Blame-directed: move mass from high-blame to low-blame
            n_move = max(2, n // 10)
            high_idx = np.argsort(blame)[-n_move:]
            low_idx = np.argsort(blame)[:n_move]
            
            # Compute how much to move
            move_frac = 0.05 + 0.05 * rng.random()
            amounts = a[high_idx] * move_frac * rng.uniform(0.5, 1.5, n_move)
            total_move = amounts.sum()
            
            candidate = a.copy()
            candidate[high_idx] -= amounts
            # Distribute to low-blame positions inversely proportional to blame
            weights = 1.0 / (blame[low_idx] + 0.05)
            weights /= weights.sum()
            candidate[low_idx] += total_move * weights
            
        elif r < 0.7:
            # Local smoothing: blur high-blame regions
            candidate = a.copy()
            n_smooth = max(3, n // 8)
            smooth_idx = np.argsort(blame)[-n_smooth:]
            for i in smooth_idx:
                if i > 0 and i < n - 1:
                    candidate[i] = 0.6 * a[i] + 0.2 * a[i - 1] + 0.2 * a[i + 1]
                elif i > 0:
                    candidate[i] = 0.7 * a[i] + 0.3 * a[i - 1]
                elif i < n - 1:
                    candidate[i] = 0.7 * a[i] + 0.3 * a[i + 1]
        
        elif r < 0.85:
            # Block swap: reduce a block, increase another
            bw = max(1, int(rng.integers(1, max(2, n // 15))))
            s1 = int(rng.integers(0, n - 2 * bw))
            s2 = int(rng.integers(0, n - bw))
            if s2 + bw <= s1 or s1 + bw <= s2:
                candidate = a.copy()
                candidate[s1:s1 + bw] *= np.exp(rng.normal(-0.08, 0.03))
                candidate[s2:s2 + bw] *= np.exp(rng.normal(0.08, 0.03))
            else:
                candidate = a * np.exp(rng.normal(0, 0.02, n))
        
        else:
            # Global noise
            candidate = a * np.exp(rng.normal(0, 0.03, n))
        
        candidate = np.clip(candidate, 0.0, None)
        s_cand = candidate.sum()
        if s_cand < 1e-10:
            candidate = a.copy()
        else:
            candidate = candidate / s_cand * S
        
        new_score = local_score(candidate)
        
        if new_score < score:
            a = candidate
            score = new_score
            if score < best_score:
                best_score = score
                best = a.copy()
        else:
            delta = new_score - score
            if delta < temp * 20:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    a = candidate
                    score = new_score
        
        temp *= cooling
        if temp < 1e-7:
            temp = 0.002  # Reheat
    
    return best, best_score


def level_set_refine(a_init, time_budget, rng):
    """
    Level-set method: try to equalize the top K peaks by solving
    a linear system for the optimal perturbation.
    """
    a = a_init.copy().astype(np.float64)
    S = a.sum()
    score = local_score(a)
    best = a.copy()
    best_score = score
    n = len(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a, a)
        top_k = np.argsort(c)[-5:]
        peak_val = c[top_k[0]]
        
        # Build Jacobian: for each top peak k, the gradient w.r.t. a
        # J has shape (K, n) where J[k, j] = 2*a[k-j] (valid)
        K = len(top_k)
        J = np.zeros((K, n))
        for mi, k in enumerate(top_k):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                J[mi, idx] = 2.0 * a[k - idx]
        
        # We want to find delta such that J @ delta reduces all top peaks equally
        # Minimize ||J @ delta + (c[top_k] - target)||^2 subject to sum(delta)=0
        # where target = mean(c[top_k]) (reducing the max by pulling others up)
        target = np.mean(c[top_k])
        residual = c[top_k] - target  # Positive for peaks above average
        
        # Solve least squares with constraint sum(delta) = 0
        # Use Lagrange: minimize ||J@delta - residual||^2 + lambda*(sum(delta))^2
        # Gradient: 2*J^T*(J@delta - residual) + 2*lambda*1 = 0
        # => (J^T J + lambda*I) @ delta = J^T @ residual
        lam = 0.1
        A_mat = J.T @ J + lam * np.eye(n)
        b_vec = J.T @ residual
        try:
            delta = np.linalg.solve(A_mat, b_vec)
        except np.linalg.LinAlgError:
            delta = np.linalg.lstsq(A_mat, b_vec, rcond=None)[0]
        
        # Step size
        step = min(0.01, 0.5 * S / (np.linalg.norm(delta) * np.max(np.abs(a)) + 1e-10))
        candidate = a - step * delta
        candidate = np.clip(candidate, 0.0, None)
        s_cand = candidate.sum()
        if s_cand < 1e-10:
            candidate = a.copy()
        else:
            candidate = candidate / s_cand * S
        
        new_score = local_score(candidate)
        if new_score < score:
            a = candidate
            score = new_score
            if score < best_score:
                best_score = score
                best = a.copy()
        else:
            # Small random perturbation to escape
            a = a * np.exp(rng.normal(0, 0.02, n))
            a = np.clip(a, 0, None)
            a = a / a.sum() * S
            score = local_score(a)
    
    return best, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: monotone-biased init, gradient optimization,
    blame-based local search, and level-set refinement.
    """
    t_global = time.monotonic()
    total_budget = min(float(budget_s), 2400.0)
    deadline = t_global + total_budget - 40  # Reserve for verification
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Check provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        sc = local_score(start)
        print(f"Provided seq: n={len(start)}, score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = start.copy()
    
    # Phase 1: Diverse initialization
    print("\n=== Phase 1: Initialization ===")
    candidates = []
    lengths = [128, 256, 384, 512, 768, 1024]
    
    for n in lengths:
        if time.monotonic() > deadline - 180:
            break
        # Monotone scan (key innovation)
        a_mono = init_scan_monotone(n, rng)
        candidates.append((local_score(a_mono), n, a_mono, 'mono_scan'))
        
        # GPS with monotone bias
        for bias in [0.2, 0.4, 0.6, 0.8]:
            a_gm = init_gps_monotone(n, rng, n_iter=80, bias_strength=bias)
            candidates.append((local_score(a_gm), n, a_gm, f'gps_mono_{bias}'))
        
        # Pure GPS
        a_gps = init_gps(n, rng, n_iter=100)
        candidates.append((local_score(a_gps), n, a_gps, 'gps'))
        
        # Bimodal
        a_bi = init_bimodal(n, rng)
        candidates.append((local_score(a_bi), n, a_bi, 'bimodal'))
        
        # Random
        a_rnd = 1.0 + 0.4 * rng.standard_normal(n)
        a_rnd = np.clip(a_rnd, 0.01, None)
        a_rnd = a_rnd / a_rnd.sum()
        candidates.append((local_score(a_rnd), n, a_rnd, 'random'))
    
    candidates.sort(key=lambda x: x[0])
    print(f"  Best init: {candidates[0][0]:.6f} (n={candidates[0][1]}, {candidates[0][3]})")
    print(f"  Top 5: {[(f'{c[0]:.4f}', c[1], c[3]) for c in candidates[:5]]}")
    
    # Phase 2: Gradient optimization
    print(f"\n=== Phase 2: Gradient optimization (time budget: {deadline - time.monotonic():.0f}s) ===")
    n_candidates = min(12, len(candidates))
    time_reserved_ls = 250  # Reserve for local search
    time_available = deadline - time.monotonic() - time_reserved_ls
    time_per = max(4.0, time_available / n_candidates)
    
    optimized = []
    for i in range(n_candidates):
        if time.monotonic() > deadline - time_reserved_ls - 30:
            break
        sc0, n, a, name = candidates[i]
        # Allocate more time for better initial scores
        quality_factor = max(0.5, 1.5 - i * 0.1)
        t_alloc = min(time_per * quality_factor, deadline - time.monotonic() - time_reserved_ls - 20)
        # Scale for length (larger n = slower per iteration)
        t_alloc *= min(1.0, 384.0 / n)
        t_alloc = max(3.0, t_alloc)
        
        if t_alloc < 3:
            break
        
        print(f"  [{i+1}/{n_candidates}] n={n}, {name}, init={sc0:.6f}, alloc={t_alloc:.1f}s", end="")
        a_opt, sc_opt = optimize_gradient(n, a, t_alloc, rng, K=3)
        print(f" -> {sc_opt:.6f}")
        
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"    *** NEW BEST: {best_score:.6f} (n={n}) ***")
        
        optimized.append((sc_opt, n, a_opt))
    
    # Phase 3: Multi-scale upsample
    print(f"\n=== Phase 3: Multi-scale ===")
    if optimized:
        optimized.sort(key=lambda x: x[0])
        for target_n in [512, 1024, 2048]:
            if time.monotonic() > deadline - 150:
                break
            base = optimized[0]
            if len(base[2]) >= target_n:
                continue
            x_old = np.linspace(0, 1, len(base[2]))
            x_new = np.linspace(0, 1, target_n)
            a_up = np.interp(x_new, x_old, base[2])
            a_up += 0.01 * rng.standard_normal(target_n)
            a_up = np.clip(a_up, 1e-6, None)
            a_up = a_up / a_up.sum()
            
            t_alloc = min(40.0, deadline - time.monotonic() - 120)
            if t_alloc < 5:
                break
            print(f"  Upsample to {target_n} (alloc={t_alloc:.1f}s)", end="")
            a_opt, sc_opt = optimize_gradient(target_n, a_up, t_alloc, rng, K=3)
            print(f" -> {sc_opt:.6f}")
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"    *** NEW BEST: {best_score:.6f} ***")
    
    # Phase 4: Local search + level-set refinement
    print(f"\n=== Phase 4: Refinement ===")
    if best_seq is not None:
        time_left = deadline - time.monotonic() - 30
        if time_left > 20:
            # Split time between blame search and level-set
            t_blame = time_left * 0.5
            t_level = time_left * 0.3
            t_direct = time_left * 0.2
            
            print(f"  Blame search ({t_blame:.0f}s)...", end="")
            a_ls, sc_ls = local_search_blame(best_seq, t_blame, rng, K_peaks=5)
            print(f" -> {sc_ls:.6f}")
            if sc_ls < best_score:
                best_score = sc_ls
                best_seq = a_ls.copy()
                print(f"    *** NEW BEST: {best_score:.6f} ***")
            
            if time.monotonic() < deadline - 40:
                print(f"  Level-set refine ({t_level:.0f}s)...", end="")
                a_lv, sc_lv = level_set_refine(best_seq, t_level, rng)
                print(f" -> {sc_lv:.6f}")
                if sc_lv < best_score:
                    best_score = sc_lv
                    best_seq = a_lv.copy()
                    print(f"    *** NEW BEST: {best_score:.6f} ***")
            
            if time.monotonic() < deadline - 30:
                print(f"  Direct optimize ({t_direct:.0f}s)...", end="")
