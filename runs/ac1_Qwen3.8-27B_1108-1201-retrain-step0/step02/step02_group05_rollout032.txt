def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def pnorm_obj_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p.
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c = np.maximum(c, 1e-30)

        sum_cp = np.sum(cp * c)
        if sum_cp < 1e-30:

    grad_a = factor * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)

"""
Multi-phase optimization for minimum peak autocorrelation non-negative sequences (n=256).
Uses L-BFGS with annealed p-norm, soft-max refinement, coordinate descent with 1D
convex subproblems, and random restarts. All helper functions are top-level.

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score = 2n * max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def norm_score(a):
    """Score = 2n * max(conv(a,a)) / (sum(a))^2, scale-invariant."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def pnorm_obj_grad(theta, p):
    """L-BFGS objective: 2n * ||conv(softmax(theta), softmax(theta))]_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c = np.maximum(c, 1e-30)
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def softmax_obj_grad(theta, tau):
    """L-BFGS objective: 2n * soft_max(conv(softmax(theta), softmax(theta)), tau)."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score = 2.0 * n * soft_max_val
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def make_init_spike(n, rng, n_spikes=8):

    a = np.ones(n) * 0.1
    spike_positions = rng.choice(n, size=n_spikes, replace=False)
    for pos in spike_positions:
        width = max(1, n // (30 * n_spikes))
        for offset in range(-width, width + 1):
            idx = (pos + offset) % n
            a[idx] += np.exp(-0.5 * (offset / max(1, width // 2)) ** 2)
    a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)

    a += 0.05 * rng.standard_normal(n)

            s = local_score(a_new)
            if s < best_score:

            theta = best_theta

    (plus the quadratic term at k=2j)

        rng = np.random.default_rng()

        order = rng.permutation(n)
        for j in order:

                    break

            # Plus: at k = 2j, the term is c_{2j} - old_val^2 + x^2

                    return np.inf

                #        but the actual change is -old^2 + x^2, so we need to adjust)

                        else:

                    else:

                return np.max(vals)

            result = minimize_scalar(g, bounds=[0, upper], method='bounded', options={'xatol': 1e-10})
            a[j] = max(0.0, result.x)
        
        pass_count += 1
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:

def coord_1d_solve(c, a, j, old_val, n):
    """Solve the 1D subproblem for coordinate j."""
    # Precompute affected lags
    # k is affected if 0 <= k-j < n
    k_affected = np.arange(max(0, j - (n-1)), min(2*n-2, j + n - 1) + 1)
    # Wait, k-j in [0, n-1] means k in [j, j+n-1], clipped to [0, 2n-2]
    k_start = max(0, j)
    k_end = min(2 * n - 2, j + n - 1)
    k_affected = np.arange(k_start, k_end + 1)
    idx_affected = k_affected - j  # indices into a
    
    # Linear part: for k in k_affected, k != 2j
    mask = k_affected != 2 * j
    if np.any(mask):
        slopes = 2.0 * a[idx_affected[mask]]
        intercepts = c[k_affected[mask]] - 2.0 * old_val * a[idx_affected[mask]]
    else:
        slopes = np.array([])
        intercepts = np.array([])
    
    # Quadratic part at k = 2j (if in range)
    has_quad = (2 * j >= 0) and (2 * j <= 2 * n - 2)
    quad_const = c[2 * j] - old_val * old_val if has_quad else 0
    
    # Unaffected max
    if len(k_affected) < 2 * n - 1:
        all_k = np.arange(2 * n - 1)
        mask_unaffected = np.ones(2 * n - 1, dtype=bool)
        mask_unaffected[k_affected] = False
        unaffected_max = np.max(c[mask_unaffected]) if np.any(mask_unaffected) else 0
    else:
        unaffected_max = 0
    
    def g(x):
        val = unaffected_max
        if len(slopes) > 0:
            val = max(val, np.max(slopes * x + intercepts))
        if has_quad:
            val = max(val, x * x + quad_const)
        return val
    
    # Find minimum of g on [0, upper]
    upper = min(1.0, 3.0 * old_val + 0.02)
    if upper < 1e-6:
        upper = 0.01
    
    result = minimize_scalar(g, bounds=[0, upper], method='bounded', options={'xatol': 1e-12})
    return max(0.0, result.x)

"""
Multi-phase optimization for minimum peak autocorrelation non-negative sequences.
Strategy: L-BFGS with annealed p-norm and soft-max, coordinate descent with 1D
convex subproblems, anti-symmetric initializations, and iterated refinement cycles.
"""

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score = 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def pnorm_obj_grad(theta, p):
    """L-BFGS: 2n * ||conv(softmax(theta), softmax(theta))||_p with gradient."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c = np.maximum(c, 1e-30)
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def softmax_obj_grad(theta, tau):
    """L-BFGS: 2n * soft_max(conv, tau) with gradient."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score = 2.0 * n * soft_max_val
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta

def run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p, deadline):
    """L-BFGS with annealed p-norm schedule."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))
    for p in p_schedule:
        if time.monotonic() > deadline - 5:
            break
        try:
            result = minimize(
                pnorm_obj_grad, theta.copy(),
                method='L-BFGS-B', jac=True, args=(p,),
                options={'maxiter': maxiter_per_p, 'ftol': 1e-15,
                         'gtol': 1e-12, 'maxfun': maxiter_per_p * 5}
            )
            a_new = softmax(result.x)
            s = local_score(a_new)
            if s < best_score:
                best_score = s
                best_theta = result.x.copy()
            theta = best_theta
        except Exception:
            break
    return best_theta, best_score


def run_lbgfs_softmax(theta, tau_schedule, maxiter_per_tau, deadline):
    """L-BFGS with annealed soft-max temperature."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))
    for tau in tau_schedule:
        if time.monotonic() > deadline - 5:
            break
        try:
            result = minimize(
                softmax_obj_grad, theta.copy(),
                method='L-BFGS-B', jac=True, args=(tau,),
                options={'maxiter': maxiter_per_tau, 'ftol': 1e-15,
                         'gtol': 1e-12, 'maxfun': maxiter_per_tau * 5}
            )
            a_new = softmax(result.x)
            s = local_score(a_new)
            if s < best_score:
                best_score = s
                best_theta = result.x.copy()
            theta = best_theta
        except Exception:
            break
    return best_theta, best_score


def coord_solve_1d(c, a, j, old_val, n):
    """Solve 1D subproblem for coordinate j given current convolution c."""
    # Affected lags: k where 0 <= k-j < n, i.e., k in [j, j+n-1] cap [0, 2n-2]
    k_start = max(0, j)
    k_end = min(2 * n - 2, j + n - 1)
    k_aff = np.arange(k_start, k_end + 1)
    idx_aff = k_aff - j  # indices into a

    # Separate the quadratic lag (k=2j) from linear lags
    quad_k = 2 * j
    mask_quad = (k_aff == quad_k)
    mask_lin = ~mask_quad

    if np.any(mask_lin):
        lin_a = a[idx_aff[mask_lin]]
        slopes = 2.0 * lin_a
        intercepts = c[k_aff[mask_lin]] - 2.0 * old_val * lin_a
    else:
        slopes = np.array([0.0])
        intercepts = np.array([0.0])

    has_quad = (0 <= quad_k <= 2 * n - 2)
    quad_c = c[quad_k] - old_val * old_val if has_quad else 0.0

    # Unaffected max
    total_lags = 2 * n - 1
    if len(k_aff) < total_lags:
        all_mask = np.ones(total_lags, dtype=bool)
        all_mask[k_aff] = False
        unaffected_max = np.max(c[all_mask])
    else:
        unaffected_max = 0.0

    # Precompute for fast evaluation
    def g_val(x):
        v = unaffected_max
        if len(slopes) > 1:
            v = max(v, np.dot(slopes, x) if len(slopes) == 1 else np.max(slopes * x + intercepts))
        elif len(slopes) == 1:
            v = max(v, slopes[0] * x + intercepts[0])
        if has_quad:
            v = max(v, x * x + quad_c)
        return v

    upper = min(1.0, max(0.01, 3.0 * old_val + 0.01))
    result = minimize_scalar(g_val, bounds=[0.0, upper], method='bounded',
                             options={'xatol': 1e-13})
    return max(0.0, float(result.x))


def coordinate_descent(a_init, time_budget, rng, verbose=False):
    """Cycle-and-repeat coordinate descent with 1D convex subproblems."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best = a.copy()
    best_score = local_score(best)
    t0 = time.monotonic()
    pass_count = 0
    stall_count = 0

    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a, a)
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_budget:
                break
            old_val = a[j]
            new_val = coord_solve_1d(c, a, j, old_val, n)
            if new_val != old_val:
                a[j] = new_val
                # Update c incrementally
                k_start = max(0, j)
                k_end = min(2 * n - 2, j + n - 1)
                for k in range(k_start, k_end + 1):
                    idx = k - j
                    if k == 2 * j:
                        c[k] = c[k] - old_val * old_val + new_val * new_val
                    else:
                        c[k] = c[k] + 2.0 * (new_val - old_val) * a[idx]
        pass_count += 1
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s
        cur_score = local_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        if verbose and pass_count % 50 == 0:
            print(f"    CD: pass={pass_count}, cur={cur_score:.8f}, best={best_score:.8f}, t={time.monotonic()-t0:.0f}s")

        # If stalled for too long, add small perturbation
        if stall_count > 30:
            a = a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 1e-8)
            a = a / np.sum(a)
            stall_count = 0

    return best, best_score


def active_set_refine(a_init, time_budget, rng):
    """Subgradient method using active set of peak lags."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / np.sum(a)
    best = a.copy()
    best_score = local_score(best)
    t0 = time.monotonic()
    step = 0.002

    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a, a)
        c_max = np.max(c)
        tol = c_max * 0.005 + 1e-10
        active = np.where(c >= c_max - tol)[0]

        # Subgradient: grad[j] = (2/|A|) * sum_{k in A} a[k-j] (if valid)
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                # a[k-j] for j in [j_lo, j_hi] = a[k-j_hi : k-j_lo+1] reversed
                idx_lo = k - j_hi
                idx_hi = k - j_lo
                grad[j_lo:j_hi + 1] += a[idx_lo:idx_hi + 1][::-1]
        grad *= 2.0 / len(active)

        # Project to sum=0 direction
        grad_eff = grad - np.mean(grad)

        # Backtracking
        alpha = step
        improved = False
        for _ in range(20):
            a_try = a - alpha * grad_eff
            a_try = np.maximum(a_try, 0.0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = fft_convolve(a_try, a_try)
            if np.max(c_try) < c_max - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            step *= 0.8
            if step < 1e-7:
                break
        else:
            step = min(step * 1.1, 0.01)

        cur_score = local_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimum peak autocorrelation non-negative sequence.
    Uses L-BFGS (p-norm + soft-max), coordinate descent, and active-set subgradient.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 50.0
    rng = np.random.default_rng(seed)
    n = 256

    best_score = np.inf
    best_seq = None

    # Load parent sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            parent = np.asarray(height_sequence_1, dtype=np.float64)
            parent = np.clip(parent, 0, 1000.0)
            if np.sum(parent) > 0.01 and len(parent) == n:
                parent = parent / np.sum(parent)
                s = local_score(parent)
                print(f"Parent sequence: n={n}, score={s:.6f}")
                best_score = min(best_score, s)
                best_seq = parent.copy()
        except Exception as e:
            print(f"Warning: could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = local_score(best_seq)
        print(f"Default uniform init: score={best_score:.6f}")

    # === Phase 1: L-BFGS with p-norm from diverse initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 1: L-BFGS p-norm optimization (target deadline: {deadline - t_global:.0f}s)")
    print(f"{'='*60}")

    phase1_deadline = min(deadline - 900, t_global + 700)
    p_schedule = [8, 16, 32, 64, 128, 256]

    inits = []
    if best_seq is not None:
        inits.append(('parent', best_seq.copy()))
    inits.append(('ramp1', make_init_ramp(n, rng)))
    inits.append(('ramp2', make_init_ramp(n, rng)))
    inits.append(('spike', make_init_spike(n, rng)))
    inits.append(('wave', make_init_wave(n, rng)))
    inits.append(('uniform', make_init_uniform_pert(n, rng, 0.4)))
    inits.append(('anti', make_init_anti(n, rng)))
    inits.append(('uniform_big', make_init_uniform_pert(n, rng, 1.0)))

    for name, a0 in inits:
        if time.monotonic() > phase1_deadline:
            break
        theta = np.log(np.maximum(a0, 1e-30))
        theta_best, s = run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p=250,
                                        deadline=phase1_deadline)
        a_result = softmax(theta_best)
        if s < best_score:
            best_score = s
            best_seq = a_result.copy()
            print(f"  [{name}] score={s:.6f} *** NEW BEST ***")
        else:
            print(f"  [{name}] score={s:.6f}")

    print(f"  Phase 1 best: {best_score:.6f}")

    # === Phase 2: Soft-max refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Soft-max L-BFGS refinement")
    print(f"{'='*60}")

    phase2_deadline = min(deadline - 500, time.monotonic() + 300)
    tau_schedule = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    theta = np.log(np.maximum(best_seq, 1e-30))
    theta_best, s = run_lbgfs_softmax(theta, tau_schedule, maxiter_per_tau=300,
                                      deadline=phase2_deadline)
    if s < best_score:
        best_score = s
        best_seq = softmax(theta_best).copy()
        print(f"  Soft-max: {s:.6f} *** NEW BEST ***")
    else:
        print(f"  Soft-max: {s:.6f}")

    # === Phase 3: Coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Coordinate descent polishing")
    print(f"{'='*60}")

    cd_time = max(200, min(500, deadline - time.monotonic() - 300))
    print(f"  Allocated time: {cd_time:.0f}s")
    best_seq, s = coordinate_descent(best_seq, time_budget=cd_time, rng=rng, verbose=True)
    if s < best_score:
        best_score = s
        print(f"  Coord descent: {s:.6f} *** NEW BEST ***")
    else:
        print(f"  Coord descent: {s:.6f}")

    # === Phase 4: Iterated refinement cycles ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Iterated refinement cycles")
    print(f"{'='*60}")

    for cycle in range(5):
        if time.monotonic() > deadline - 150:
            break

        # Soft-max step
        t_sm = min(120, (deadline - time.monotonic() - 100) * 0.4)
        if t_sm > 30:
            theta = np.log(np.maximum(best_seq, 1e-30))
            tau_sched = [0.0003, 0.0001, 0.00003]
            theta_best, s = run_lbgfs_softmax(theta, tau_sched, maxiter_per_tau=200,
                                              deadline=time.monotonic() + t_sm)
            if s < best_score:
                best_score = s
                best_seq = softmax(theta_best).copy()
                print(f"  Cycle {cycle} soft-max: {s:.6f} *** NEW BEST ***")

        # Active set step
        t_as = min(100, (deadline - time.monotonic() - 100) * 0.25)
        if t_as > 30:
            best_seq, s = active_set_refine(best_seq, time_budget=t_as, rng=rng)
            if s < best_score:
                best_score = s
                print(f"  Cycle {cycle} active-set: {s:.6f} *** NEW BEST ***")

        # Coordinate descent step
        t_cd = min(150, deadline - time.monotonic() - 80)
        if t_cd > 30:
            best_seq, s = coordinate_descent(best_seq, time_budget=t_cd, rng=rng)
            if s < best_score:
                best_score = s
                print(f"  Cycle {cycle} coord: {s:.6f} *** NEW BEST ***")

    # === Final verification ===
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(final_list)
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
    print(f"Time used: {time.monotonic() - t_global:.0f}s / {budget_s}s")
    print(f"{'='*60}")
    return final_list
