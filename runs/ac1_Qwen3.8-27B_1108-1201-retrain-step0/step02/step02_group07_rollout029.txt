import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(conv, a):
    """Compute subgradient of max(conv) w.r.t. a."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    # Use top-10% of convolution values
    thresh = max_c * 0.9
    active = np.where(conv >= thresh)[0]
    # Weight by excess over threshold
    weights = np.zeros(len(conv))
    weights[active] = (conv[active] - thresh * 0.5)
    weights /= weights.sum()
    
    sg = np.zeros(n)
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += weights[k] * a[k - je:k - js + 1][::-1]
    return sg

def basis_functions(n):
    """Generate a set of basis functions for low-dimensional parameterization."""
    x = np.arange(n, dtype=np.float64)
    bases = []
    
    # Uniform
    bases.append(np.ones(n))
    
    # Centered Gaussian
    g = np.exp(-0.5 * ((x - n*0.5) / (0.2*n))**2)
    bases.append(g)
    
    # Off-center Gaussians
    for center in [0.25, 0.75]:
        g = np.exp(-0.5 * ((x - n*center) / (0.15*n))**2)
        bases.append(g)
    
    # Sine waves
    for freq in [1, 2, 3]:
        s = np.maximum(0.5 + 0.5*np.cos(2*np.pi*freq*x/n), 0)
        bases.append(s)
    
    # Edge-heavy
    edge = ((x - n/2)**2) / (n**2)
    bases.append(edge)
    
    # V-shaped
    v = np.abs(x - n/2) / n
    bases.append(v)
    
    # Ramp
    bases.append(x.astype(float) / n)
    
    # Inverse ramp
    bases.append(1.0 - x.astype(float) / n)
    
    return np.array(bases)  # shape (m, n)

Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines low-dimensional basis parameterization, adaptive subgradient descent,
and structure-aware simulated annealing with multiple sequence lengths.

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_aware(conv, a, temp=1.0):
    """
    Compute a weighted subgradient of max(conv) using temperature-based
    softmax over the convolution values, with special handling for the
    structure of the convolution.
    """
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)

    scaled = conv / (temp * max_c)
    scaled -= scaled.max()
    w = np.exp(scaled)

    if w_sum < 1e-30:

        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    return sg

    step = eta_init / n**0.5
    
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient_aware(conv, a, temp)
        # Center the subgradient (remove mean for simplex compatibility)
        sg_c = sg - sg.mean()
        a_new = a - step * sg_c
        a_new = project_simplex(a_new)
        # Ensure we don't move too far
        if np.sum(np.abs(a_new - a)) > 0.3:
            a_new = 0.7 * a + 0.3 * a_new
            a_new = project_simplex(a_new)
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
            # Slightly increase step on improvement
            step = min(step * 1.02, eta_init)
        else:
            patience += 1
        
        if patience >= max_pat:
            patience = 0
            step *= 0.6
            temp *= 1.5
            if step < eta_init * 0.01:
                step = eta_init * 0.1
                temp = temp_init * 3.0
    
    return best_a, best_obj


def low_dim_optimize(n, time_limit, parent=None):
    """
    Optimize in a low-dimensional subspace defined by basis functions.
    Returns the best sequence found.
    """
    x = np.arange(n, dtype=np.float64)
    
    # Define basis functions
    bases = []
    bases.append(np.ones(n))  # Uniform
    bases.append(np.exp(-0.5 * ((x - 0.5*n) / (0.2*n))**2))  # Center Gaussian
    bases.append(np.exp(-0.5 * ((x - 0.25*n) / (0.12*n))**2))  # Left Gaussian
    bases.append(np.exp(-0.5 * ((x - 0.75*n) / (0.12*n))**2))  # Right Gaussian
    bases.append(np.maximum(0.5 + 0.5*np.cos(2*np.pi*x/n), 0))  # Cos 1
    bases.append(np.maximum(0.5 + 0.5*np.cos(4*np.pi*x/n), 0))  # Cos 2
    bases.append(np.maximum(0.5 - 0.5*np.cos(2*np.pi*x/n), 0))  # Sin-squared 1
    bases.append(((x - 0.5*n) / n)**2 + 0.1)  # Edge-heavy
    bases.append(np.abs(x - 0.5*n) / n + 0.1)  # V-shape
    bases.append(x.astype(float) / n)  # Ramp
    bases.append(1.0 - x.astype(float) / n)  # Inverse ramp
    bases.append(np.exp(-0.5 * ((x - 0.5*n) / (0.35*n))**2))  # Wide Gaussian
    bases.append(np.exp(-0.5 * ((x - 0.15*n) / (0.08*n))**2) + np.exp(-0.5 * ((x - 0.85*n) / (0.08*n))**2))
    
    B = np.array(bases)  # shape (m, n)
    m = len(bases)
    
    # If we have a parent, try to fit it
    if parent is not None and len(parent) == n:
        p = np.maximum(parent, 0).copy()
        p /= p.sum()
        # Project onto span of bases (least squares)
        coeffs, _, _, _ = np.linalg.lstsq(B, p, rcond=None)
        coeffs = np.maximum(coeffs, 0)
        if coeffs.sum() > 1e-10:
            coeffs /= coeffs.sum()
            x0 = coeffs
        else:
            x0 = np.ones(m) / m
    else:
        x0 = np.ones(m) / m
    
    def objective(coeffs):
        a = coeffs @ B
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return 1e10
        a = a / s
        conv = fast_convolve(a)
        return 2.0 * n * conv.max()
    
    # Multiple restarts
    best_a = None
    best_val = float('inf')
    
    n_restarts = min(15, max(5, int(time_limit / 2)))
    t_per_restart = time_limit / n_restarts
    
    for r in range(n_restarts):
        if r == 0:
            start = x0.copy()
        else:
            start = x0 + np.random.default_rng(42 + r).normal(0, 0.3, m)
            start = np.maximum(start, 0)
            start /= start.sum()
        
        t0 = time.monotonic()
        res = scipy_minimize(
            objective, start, method='Nelder-Mead',
            options={'maxiter': int(max(20, t_per_restart * 50)), 'xatol': 1e-8, 'fatol': 1e-8,
                     'adaptive': True}
        )
        if res.fun < best_val:
            best_val = res.fun
            a = np.maximum(res.x @ B, 0)
            s = a.sum()
            if s > 1e-10:
                a /= s
                best_a = a.copy()
        
        if time.monotonic() - t0 > t_per_restart * 2:
            break
    
    if best_a is not None:
        return best_a
    
    # Fallback: evaluate x0
    a = np.maximum(x0 @ B, 0)
    a /= a.sum()
    return a


def structure_anneal(a0, time_limit, rng, mode='mixed'):
    """
    Structure-aware simulated annealing that uses convolution structure
    to guide moves. Supports different move modes for diversity.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    a /= a.sum()
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1.0))
        T = 0.01 * (1.0 - frac)**2.5 + 0.00005
        
        # Choose move type
        r = rng.random()
        
        if r < 0.4:
            # Peak-targeted: move mass away from peak-contributing position
            max_c = conv.max()
            thresh = max_c * (0.7 + 0.3 * frac)
            peaks = np.where(conv >= thresh)[0]
            if len(peaks) == 0:
                peaks = np.array([np.argmax(conv)])
            k = int(rng.choice(peaks[:min(10, len(peaks))]))
            
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            seg_a = a[js:je+1]
            seg_b = a[k-je:k-js+1][::-1]
            products = seg_a * seg_b
            total = products.sum()
            if total < 1e-15:
                i = int(rng.integers(n))
            else:
                probs = products / total
                i = js + int(rng.choice(je - js + 1, p=probs))
            
            # Choose destination: prefer low-contribution positions
            j = int(rng.integers(n))
            attempts = 0
            while attempts < 3:
                cand = int(rng.integers(n))
                # Check contribution to top peaks
                contrib = 0.0
                top_peaks = peaks[:5]
                for pk in top_peaks:
                    d = pk - cand
                    if 0 <= d < n:
                        contrib += a[d]
                    d2 = cand - pk + 2*n - 2
                    if 0 <= d2 < n:
                        contrib += a[d2]
                    d3 = pk - cand - n + 1
                    if 0 <= d3 < n:
                        contrib += a[d3]
                    d4 = cand - pk - n + 1
                    if 0 <= d4 < n:
                        contrib += a[d4]
                if contrib < a.mean():
                    j = cand
                    break
                attempts += 1
            
            if i == j:
                continue
            delta = a[i] * rng.uniform(0.02, 0.08)
        
        elif r < 0.7:
            # Symmetric pair move: move mass from both ends of a peak to spread
            max_c = conv.max()
            peaks = np.where(conv >= max_c * 0.8)[0]
            if len(peaks) == 0:
                peaks = np.array([np.argmax(conv)])
            k = int(rng.choice(peaks[:5]))
            
            # Find top contributors to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je - js + 1 >= 2:
                i1 = js + int(rng.integers(je - js + 1))
                i2 = js + int(rng.integers(je - js + 1))
                if i1 == i2:
                    i2 = (i2 + 1) % n
                j1 = int(rng.integers(n))
                j2 = int(rng.integers(n))
                if j1 == i1 or j1 == i2 or j2 == i1 or j2 == i2 or j1 == j2:
                    continue
                delta1 = a[i1] * rng.uniform(0.01, 0.04)
                delta2 = a[i2] * rng.uniform(0.01, 0.04)
                a[i1] -= delta1
                a[i2] -= delta2
                a[j1] += delta1
                a[j2] += delta2
                if a[i1] < 0 or a[i2] < 0:
                    a[i1] += delta1
                    a[i2] += delta2
                    a[j1] -= delta1
                    a[j2] -= delta2
                    continue
            else:
                continue
        
        else:
            # Random pairwise
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            delta = a[i] * rng.uniform(0.005, 0.04)
        
        # Skip if delta not set (from symmetric move)
        if r >= 0.4 and r < 0.7:
            new_conv = fast_convolve(a)
            new_obj = 2.0 * n * new_conv.max()
            delta_obj = new_obj - cur_obj
            iters += 1
            if delta_obj < 0:
                cur_obj = new_obj
                conv = new_conv
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
            else:
                if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                    cur_obj = new_obj
                    conv = new_conv
                else:
                    # Revert
                    a[i1] += delta1
                    a[i2] += delta2
                    a[j1] -= delta1
                    a[j2] -= delta2
            continue
        
        if delta < 1e-12:
            continue
        
        a[i] -= delta
        a[j] += delta
        if a[i] < 0:
            a[i] += delta
            a[j] -= delta
            continue
        
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        iters += 1
        
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[i] += delta
                a[j] -= delta
    
    return best_a, best_obj


def coordinate_refine(a0, time_limit, rng):
    """
    Local refinement by trying small perturbations of individual coordinates
    and accepting improvements. More targeted than full annealing.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    a /= a.sum()
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    
    # Sort by value for efficiency
    order = np.argsort(a)[::-1]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for idx in order[:min(50, n)]:
            i = int(idx)
            if a[i] < 1e-8:
                continue
            # Try moving small amounts from i to random j
            for _ in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = a[i] * frac
                    if delta < 1e-10:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    new_conv = fast_convolve(a)
                    new_obj = 2.0 * n * new_conv.max()
                    if new_obj < cur_obj:
                        cur_obj = new_obj
                        conv = new_conv
                        improved = True
                        if cur_obj < best_obj:
                            best_obj = cur_obj
                            best_a = a.copy()
                    else:
                        a[i] += delta
                        a[j] -= delta
                    break
            if time.monotonic() - t0 > time_limit:
                break
        if not improved:
            break
    
    return best_a, best_obj


def get_parent():
    """Access parent sequence if available."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def safe_evaluate(a):
    """Safely evaluate a sequence."""
    try:
        a_clean = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a_clean = np.minimum(a_clean, 1000.0)
        if a_clean.sum() < 0.01:
            return float('inf')
        if not np.all(np.isfinite(a_clean)):
            return float('inf')
        score = float(evaluate_sequence(a_clean.tolist()))
        return score
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer:
    Phase 1: Low-dim parameterization for diverse starts (20% time)
    Phase 2: Subgradient descent on top candidates (30% time)
    Phase 3: Structure-aware annealing (30% time)
    Phase 4: Coordinate refinement + final polish (20% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    
    parent = get_parent()
    best_sequence = None
    best_score = float('inf')
    
    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d" % (score, len(a)))
        return score
    
    # Determine which n values to try
    n_values = [256]
    if parent is not None:
        if len(parent) not in n_values:
            n_values.insert(0, len(parent))
    for n_extra in [200, 300, 320, 384, 512]:
        if n_extra not in n_values:
            n_values.append(n_extra)
    
    # Sort: prefer parent length, then moderate sizes
    if parent is not None:
        n_values.sort(key=lambda x: abs(x - len(parent)))
    else:
        n_values.sort(key=lambda x: abs(x - 256))
    
    # Phase 1: Diverse initialization via low-dim optimization
    print("Phase 1: Low-dim optimization for diverse starts...")
    t1_start = time.monotonic()
    phase1_budget = min(400.0, (deadline - t_start) * 0.18)
    
    candidates = []
    
    # Start with parent if available
    if parent is not None:
        p = np.maximum(parent, 0).copy()
        p /= p.sum()
        candidates.append((len(p), 'parent', p))
        score = record(p)
        print("  Parent score: %.6f" % score)
    
    # Try low-dim optimization for each n
    time_per_n = phase1_budget / len(n_values)
    for n in n_values[:6]:
        if time.monotonic() > deadline - 1600:
            break
        t_alloc = min(time_per_n, deadline - time.monotonic() - 1600)
        if t_alloc < 2.0:
            break
        print("  Low-dim n=%d, budget=%.0fs" % (n, t_alloc))
        a_opt = low_dim_optimize(n, time_limit=t_alloc, parent=parent)
        if a_opt is not None:
            candidates.append((n, 'lowdim', a_opt))
            record(a_opt)
    
    # Also try some random inits with quick subgradient
    for n in n_values[:4]:
        if time.monotonic() > deadline - 1600:
            break
        for _ in range(2):
            a_rand = rng.dirichlet(np.ones(n) * 0.5)
            candidates.append((n, 'rand', a_rand))
    
    # Quick subgradient on random candidates
    for n, name, a0 in candidates:
        if name not in ('parent', 'lowdim') or True:
            if time.monotonic() > deadline - 1500:
                break
            t_quick = min(3.0, (deadline - time.monotonic() - 1500) / 4.0)
            if t_quick < 0.5:
                break
            a_opt, _ = subgradient_descent(a0, time_limit=t_quick, eta_init=0.3, temp_init=20.0)
            candidates.append((n, name + '_sg', a_opt))
            record(a_opt)
    
    # Sort candidates by objective
    def get_obj(a):
        a = np.maximum(a, 0)
        a /= a.sum()
        conv = fast_convolve(a)
        return 2.0 * len(a) * conv.max()
    
    candidates_scored = [(get_obj(c[2]), c[0], c[1], c[2]) for c in candidates]
    candidates_scored.sort(key=lambda x: x[0])
    print("  Phase 1 done. Best: %.6f" % best_score)
    
    # Phase 2: Deep subgradient optimization
    print("Phase 2: Deep subgradient optimization...")
    t2_start = time.monotonic()
    phase2_budget = min(800.0, (deadline - t2_start) * 0.35)
    
    top_n = min(10, len(candidates_scored))
    time_per = phase2_budget / max(1, top_n)
    
    for idx in range(top_n):
        if time.monotonic() > deadline - 800:
            break
        _, n, name, a0 = candidates_scored[idx]
        t_alloc = min(time_per, deadline - time.monotonic() - 800)
        if t_alloc < 3.0:
            break
        print("  [%d/%d] n=%d %s, budget=%.0fs" % (idx+1, top_n, n, name, t_alloc))
        
        a_cur = a0.copy()
        a_cur = np.maximum(a_cur, 0)
        a_cur /= a_cur.sum()
        
        # Multi-stage subgradient with varying parameters
        stages = [
            (0.4, 40.0, 0.35),
            (0.3, 60.0, 0.25),
            (0.2, 100.0, 0.20),
            (0.1, 150.0, 0.20),
        ]
        for si, (eta_i, temp_i, frac) in enumerate(stages):
            t_sub = t_alloc * frac
            if time.monotonic() > deadline - 800:
                break
            a_cur, _ = subgradient_descent(a_cur, time_limit=t_sub, eta_init=eta_i, temp_init=temp_i)
            record(a_cur)
        
    print("  Phase 2 done. Best: %.6f" % best_score)
    
    # Phase 3: Structure-aware annealing
    if best_sequence is not None and time.monotonic() < deadline - 200:
        print("Phase 3: Structure-aware annealing...")
        t3_start = time.monotonic()
        phase3_budget = min(600.0, deadline - t3_start - 200)
        
        # Multiple annealing runs from best with perturbations
        n_anneal = 5
        time_per_anneal = phase3_budget / n_anneal
        
        for i in range(n_anneal):
            if time.monotonic() > deadline - 200:
                break
            a_start = best_sequence.copy()
            if i == 0:
                pass  # Start from best directly
            else:
                # Add perturbation scaled by iteration
                scale = 0.01 * (i + 1)
                pert = rng.normal(0, scale, len(a_start))
                a_start = np.maximum(a_start + pert, 0)
                a_start /= a_start.sum()
            
            rng_a = np.random.default_rng(seed + 10000 + i * 7)
            t_alloc = time_per_anneal
            print("  Anneal [%d/%d] n=%d, budget=%.0fs" % (i+1, n_anneal, len(a_start), t_alloc))
            a_opt, _ = structure_anneal(a_start, time_limit=t_alloc, rng=rng_a)
            record(a_opt)
            
            # Quick coordinate refine after annealing
            if time.monotonic() < deadline - 200:
                rem = min(10.0, deadline - time.monotonic() - 200)
                if rem > 2:
                    a_opt, _ = coordinate_refine(a_opt, time_limit=rem, rng=rng_a)
                    record(a_opt)
        
        print("  Phase 3 done. Best: %.6f" % best_score)
    
    # Phase 4: Final polish
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 5
        
        a_best = best_sequence.copy()
        a_best = np.maximum(a_best, 0)
        a_best /= a_best.sum()
        
        # Very fine subgradient
        if remaining > 30:
            t_sg = remaining * 0.3
            a_opt, _ = subgradient_descent(a_best, time_limit=t_sg, eta_init=0.05, temp_init=200.0)
            record(a_opt)
        
        if time.monotonic() < deadline - 5:
            rem = deadline - time.monotonic() - 5
            if rem > 20:
                # Final anneal with very low temperature
                rng_f = np.random.default_rng(seed + 99999)
                a_opt, _ = structure_anneal(best_sequence, time_limit=rem * 0.5, rng=rng_f)
                record(a_opt)
            
            if time.monotonic() < deadline - 5:
                rem2 = deadline - time.monotonic() - 5
                if rem2 > 5:
                    a_opt, _ = coordinate_refine(best_sequence, time_limit=rem2, rng=rng_f)
                    record(a_opt)
        
        print("  Phase 4 done. Best: %.6f" % best_score)
    
    # Try a completely different n if time remains
    if best_sequence is not None and time.monotonic() < deadline - 60:
        print("Bonus: Trying different n...")
        cur_n = len(best_sequence)
        for n_try in [cur_n - 32, cur_n + 32, cur_n // 2 * 3]:
            if n_try <= 0 or n_try > 2048 or n_try == cur_n:
                continue
            if time.monotonic() > deadline - 30:
                break
            # Interpolate parent to new length
            p_interp = np.interp(
                np.linspace(0, 1, n_try),
                np.linspace(0, 1, cur_n),
                best_sequence
            )
            p_interp = np.maximum(p_interp, 0)
            p_interp /= p_interp.sum()
            
            t_alloc = min(30.0, deadline - time.monotonic() - 30)
            if t_alloc < 5:
                break
            a_opt, _ = subgradient_descent(p_interp, time_limit=t_alloc * 0.5, eta_init=0.2, temp_init=50.0)
            record(a_opt)
            if time.monotonic() < deadline - 30:
                rng_b = np.random.default_rng(seed + 77777)
                a_opt, _ = structure_anneal(a_opt, time_limit=min(15.0, deadline - time.monotonic() - 30), rng=rng_b)
                record(a_opt)
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = safe_evaluate(best_sequence)
    
    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))
    
    return [float(x) for x in best_sequence]
