Uses the linear programming approach from the SOTA, combined with adaptive step sizes,

from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Fast evaluation of the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def subgradient_max_conv(a):
    """Compute subgradient of max_k conv(a,a)_k with respect to a."""
    n = len(a)
    c = fft_convolve(a, a)
    k_star = int(np.argmax(c))
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= k_star - j < n:
            val += a[k_star - j]
        if 0 <= j - k_star < n:
            val += a[j - k_star]
        g[j] = 2.0 * val
    return g, k_star, c[k_star]

import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def build_lp_indices(n):
    """Precompute index structure for LP constraint matrix."""
    indices = []
    for k in range(2 * n - 1):

        # corr[k] = sum_i f[i] * w[k-i]
        # We need: for each j, sum_k w[k] * f[k-j] (with bounds)
        # This is (f * w)[j + n - 1] ... let me think more carefully
        # sum_k w[k] * f[k-j] where 0<=k<2n-1, 0<=k-j<n
        # = sum_k w[k] * f[k-j]
        # Let i = k-j, so k = i+j, and sum_i w[i+j] * f[i] for 0<=i<n, 0<=i+j<2n-1
        # This is (rev_f * w)[j] where rev_f = f[::-1]... 
        # Actually let's just use the direct formula
        # g[j] = 2 * sum_{k: 0<=k-j<n} w[k] * f[k-j] + 2 * sum_{k: 0<=j-k<n} w[k] * f[j-k]
        # The second term is the same as the first (just different index range)
        # So g[j] = 2 * sum_{i: max(0,j-(2n-2))<=i<=min(n-1,j+n-1)} w[i+j] * f[i] (with bounds on w)
        # More simply: g[j] = 2 * (w * f_reversed)[j + n - 1] where f_reversed is zero-padded... 
        
        # Actually, the gradient of c[k] = sum_i a[i]*a[k-i] w.r.t. a[j] is:
        # dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        # So d(sum_k w[k]*c[k])/da[j] = sum_k w[k] * (a[k-j]*I[0<=k-j<n] + a[j-k]*I[0<=j-k<n])
        # = sum_k w[k]*a[k-j]*I[0<=k-j<n] + sum_k w[k]*a[j-k]*I[0<=j-k<n]
        # Let i = k-j in first sum: sum_i w[i+j]*a[i]*I[0<=i<n] (and 0<=i+j<2n-1)
        # Let i = j-k in second sum: sum_i w[j-i]*a[i]*I[0<=i<n] (and 0<=j-i<2n-1)
        
        # First term: (w_padded * a)[j] where w_padded has length 2n-1
        # Second term: (w_padded_rev * a)[j] where w_padded_rev = w[::-1]
        
        # Using FFT:
        # First term at position j: this is cross-correlation
        # sum_i a[i] * w[i+j] = (a * w_rev)[j + n - 1]... 
        
        # Let me just compute it directly for correctness (n is small enough)
        g = np.zeros(n)
        # Efficient: use the fact that it's a correlation
        # g[j] = 2 * sum_{i} f[i] * (w[i+j] + w[j-i]) (with bounds)
        # Use FFT for the first part
        # sum_i f[i] * w[i+j] for valid i, i+j
        # This is convolution of f with w (reversed), evaluated at specific points
        
        # Simpler approach: just loop (n=384, so 384*384 = 147K ops, fast)
        for j in range(n):
            val = 0.0
            # First part: sum_k w[k] * f[k-j] for 0<=k-j<n
            k_lo = max(0, j)  # k-j >= 0 => k >= j
            k_hi = min(2*n-2, j + n - 1)  # k-j <= n-1
            if k_lo <= k_hi:
                val += np.dot(w[k_lo:k_hi+1], f[0:k_hi-k_lo+1])
            # Second part: sum_k w[k] * f[j-k] for 0<=j-k<n
            k_lo2 = max(0, j - (n - 1))  # j-k >= 0 => k <= j
            k_hi2 = min(2*n-2, j)  # j-k < n => k > j-n
            if k_lo2 <= k_hi2:
                val += np.dot(w[k_lo2:k_hi2+1], f[j-k_hi2:j-k_lo2:-1])
            g[j] = 2.0 * val

        # Normalize and add a component that encourages increasing the sum
        # (since score = 2n*max_c/S^2, increasing S helps)
        # But we're working at fixed normalization, so the gradient should handle it
        
        grad_norm = np.linalg.norm(g)
        if grad_norm < 1e-10:
            break
        
        # Adam update
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g / grad_norm
        v_buf = beta2 * v_buf + (1 - beta2) * (g / grad_norm) ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)

        # Adaptive learning rate
        lr = lr_schedule_start / (1.0 + 0.01 * step)
        if elapsed_frac > 0.7:
            lr *= 0.3

        new_f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        new_f = np.clip(new_f, 0.0, None)

        # Renormalize
        S_new = np.sum(new_f)
        if S_new < 1e-10:
            no_improve += 1
            continue
        new_f = new_f * (np.sqrt(2.0 * n) / S_new)

        new_score = local_score(new_f)
        if new_score < best_score:
            best_score = new_score
            best = new_f.copy()
            f = new_f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                # Diversification
                strategy = rng.random()
                if strategy < 0.5:
                    f = best + 0.03 * np.sqrt(2.0 * n) * rng.standard_normal(n)
                    f = np.clip(f, 0.0, None)
                else:
                    # Peak targeting
                    c_best = fft_convolve(best, best)
                    peak_idx = int(np.argmax(c_best))
                    f = best.copy()
                    # Reduce top contributors
                    contribs = np.zeros(n)
                    for i in range(n):
                        if 0 <= peak_idx - i < n:
                            contribs[i] = best[i] * best[peak_idx - i]
                    top_k = np.argsort(contribs)[::-1][:max(4, n // 12)]
                    f[top_k] *= 0.8
                    bottom_k = np.argsort(contribs)[:max(4, n // 12)]
                    f[bottom_k] *= 1.2
                    f = np.clip(f, 0.0, None)
                S_f = np.sum(f)
                if S_f > 1e-10:
                    f = f * (np.sqrt(2.0 * n) / S_f)
                no_improve = 0

        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD step={step}, score={local_score(f):.6f}, best={best_score:.6f}, lr={lr:.4f}, elapsed={elapsed:.0f}s")

    return best, best_score


def make_structured_init(n, rng, method='broadband'):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if method == 'broadband':
        a = np.ones(n)
        for k in range(1, 12):
            freq = k * (1.0 + 0.5 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.12 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'two_hump':
        # Two-hump: high at edges, low in middle
        center = n / 2.0
        sigma = n / 6.0
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2) + 0.5
        a = a / np.max(a)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimizing the evaluation score.
    Phase 1: LP optimization on primary sequence
    Phase 2: Subgradient polish
    Phase 3: Explore other lengths
    Phase 4: Final LP polish
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Use provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start_seq) > 0:
            s = local_score(start_seq)
            print(f"Provided: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()

    # === PHASE 1: LP Optimization ===
    print(f"\n{'='*60}")
    print(f"PHASE 1: LP Optimization")
    print(f"{'='*60}")

    if best_seq is not None:
        n = len(best_seq)
        t1 = min(0.45 * time_budget, deadline - time.monotonic() - 60)
        if t1 > 30:
            print(f"  n={n}, time={t1:.0f}s")
            result, score = lp_optimize(best_seq, t1, rng)
            if score < best_score:
                best_score = score
                best_seq = result.copy()
                print(f"  After LP: {best_score:.6f}")
            else:
                print(f"  LP: no improvement (still {best_score:.6f})")

    # Also try LP from a fresh start with different init
    if time.monotonic() < deadline - 120:
        n_try = 384
        a0 = make_structured_init(n_try, rng, 'broadband')
        t1b = min(0.1 * time_budget, deadline - time.monotonic() - 60)
        if t1b > 20:
            print(f"\n  Fresh LP start: n={n_try}, time={t1b:.0f}s")
            result, score = lp_optimize(a0, t1b, rng, verbose=False)
            if score < best_score:
                best_score = score
                best_seq = result.copy()
                print(f"  *** Fresh LP: {best_score:.6f} ***")
            else:
                print(f"  Fresh LP: {score:.6f} (not better)")

    # === PHASE 2: Subgradient Polish ===
    if best_seq is not None:
        print(f"\n{'='*60}")
        print(f"PHASE 2: Subgradient Polish")
        print(f"{'='*60}")
        n = len(best_seq)
        t2 = min(0.20 * time_budget, deadline - time.monotonic() - 60)
        if t2 > 20:
            print(f"  n={n}, time={t2:.0f}s, starting score={best_score:.6f}")
            result, score = subgradient_polish(best_seq, t2, rng)
            if score < best_score:
                best_score = score
                best_seq = result.copy()
                print(f"  After SGD: {best_score:.6f}")
            else:
                print(f"  SGD: no improvement (still {best_score:.6f})")

    # === PHASE 3: Explore other lengths ===
    print(f"\n{'='*60}")
    print(f"PHASE 3: Length Exploration")
    print(f"{'='*60}")

    for n_try in [320, 256, 448, 512]:
        if time.monotonic() > deadline - 60:
            break
        t3 = min(25.0, deadline - time.monotonic() - 40)
        if t3 < 15:
            break

        if best_seq is not None and len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a0 = np.interp(x_new, x_old, best_seq)
        elif best_seq is not None:
            a0 = best_seq.copy()
        else:
            a0 = make_structured_init(n_try, rng, 'broadband')

        print(f"  Trying n={n_try}, time={t3:.0f}s...")
        # Quick subgradient optimization
        result, score = subgradient_polish(a0, t3, rng, verbose=False)
        if score < best_score:
            best_score = score
            best_seq = result.copy()
            print(f"  *** New best at n={n_try}: {best_score:.6f} ***")
        else:
            print(f"  n={n_try}: score={score:.6f}")

    # === PHASE 4: Final LP Polish ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n{'='*60}")
        t4 = deadline - time.monotonic() - 15
        if t4 > 30:
            print(f"PHASE 4: Final LP Polish (t={t4:.0f}s)")
            print(f"{'='*60}")
            n = len(best_seq)
            print(f"  Starting score: {best_score:.6f}, n={n}")
            result, score = lp_optimize(best_seq, t4, rng)
            if score < best_score:
                best_score = score
                best_seq = result.copy()
                print(f"  After final LP: {best_score:.6f}")
            else:
                print(f"  Final LP: no improvement (still {best_score:.6f})")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'='*60}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
