"""
Multi-phase optimization for minimizing peak self-convolution of non-negative sequences.
Combines Adam gradient descent, LP-based directional search, and peak-targeted local search.
"""

import numpy as np
import time
from scipy.optimize import linprog

# Helper functions (top-level, no closures)

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)

def smooth_max_loss(a, tau):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    log_sum = max_c + tau * np.log(np.sum(exp_vals))
    w = exp_vals / np.sum(exp_vals)
    # Gradient: d/d a_j of smooth_max = 2 * sum_k w[k] * (conv contribution of a_j at k)
    # conv[k] = sum_i a[i] * a[k-i]
    # d conv[k] / d a[j] = a[k-j] + a[j] * (1 if j != k-j else 1) ... 
    # Actually: d/d a[j] sum_i a[i]*a[k-i] = a[k-j] + a[j] (if j != k-j) or 2*a[j] (if j = k-j)

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2
over non-negative sequences a. Uses Adam gradient descent with softmax
parameterization, LP-based directional search, and peak-targeted local search.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def score_fast(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    return 2.0 * n * np.max(fft_convolve(a, a)) / (S*S)

def smooth_max_and_grad(a, tau):
    """
    Compute smooth approximation to max(conv(a,a)) and its gradient w.r.t. a.
    Returns (value, grad, conv).
    """
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = np.clip((conv - max_c) / tau, -50, 0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    value = max_c + tau * np.log(s)
    w = exp_vals / s
    # Gradient: 2 * cross_correlation(a, w)[j] = 2 * sum_m a[m]*w[m+j]
    size = 2 * n
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(fa) * fw, size)[:n]
    grad = 2.0 * corr
    return value, grad, conv

def softmax(theta):
    """Numerically stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def adam_optimize(n, a_init, time_limit, seed):
    """
    Adam optimization on softmax parameterization.
    Minimizes smooth max of conv(a,a) where a = softmax(theta) * sqrt(2n).
    """
    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * n)
    a0 = np.clip(a_init, 1e-10, None)
    a0 = a0 / np.sum(a0)
    theta = np.log(a0 + 1e-15)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    # Learning rate schedule
    lr0 = 1.0
    lr_min = 0.01
    
    # Temperature schedule
    tau_start = 50.0 / n
    tau_end = 0.5 / n
    
    best_theta = theta.copy()
    best_score = np.inf
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** (elapsed ** 0.5)
        lr = lr0 * (lr_min / lr0) ** (elapsed ** 0.8)
        
        a = softmax(theta) * S
        value, g_a, conv = smooth_max_and_grad(a, tau)
        
        # Gradient w.r.t. theta
        p = a / S  # = softmax(theta)
        inner = np.dot(p, g_a)
        g_theta = S * p * (g_a - inner)
        
        step += 1
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * (g_theta * g_theta)
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic evaluation
        if step % 100 == 0:
            a_eval = softmax(theta) * S
            sc = score_fast(a_eval)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()

            sc = score_fast(a_eval)
            print(f"    Adam step={step}, score={sc:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.2e}, lr={lr:.4f}, t={time.monotonic()-t0:.0f}s")
    
    return softmax(best_theta) * S, best_score

def lp_direction_search(a_current, time_limit, max_iters=50):
    """
    LP-based directional search.
    Repeatedly solves LP to find direction g that maximizes sum(g)
    subject to conv(a_current, g) <= M, then updates a_current.
    """
    n = len(a_current)
    a = a_current.copy()
    S = np.sum(a)
    a = a / S * np.sqrt(2.0 * n)  # Normalize to sum = sqrt(2n)
    
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = np.max(fft_convolve(a, a))
        
        # Build LP: maximize sum(g) s.t. g >= 0, conv(a, g)[k] <= M
        # conv(a, g)[k] = sum_i a[i] * g[k-i] <= M
        # Constraint matrix: A_ub[k, i] = a[i] if 0 <= k-i <= n-1
        A_rows = []
        for k in range(2*n - 1):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                row = np.zeros(n)
                row[i_start:i_end+1] = a[i_start:i_end+1]
                A_rows.append(row)
            else:
                # This shouldn't happen for valid k
                A_rows.append(np.zeros(n))
        A_ub = np.array(A_rows)
        b_ub = np.full(2*n - 1, M)
        
        # Solve LP
        try:
            result = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub,
                           bounds=[(0, None)]*n, method='highs')
            if not result.success:
                break
            g = result.x
        except Exception:
            break
        
        if np.sum(g) <= 0:
            break
        
        # Rescale g to have same sum as a
        target_sum = np.sqrt(2.0 * n)
        g = g / np.sum(g) * target_sum
        
        # Try different step sizes
        best_t = 0.1
        best_sc = np.inf
        for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
            candidate = (1 - t) * a + t * g
            candidate = np.clip(candidate, 0, None)
            if np.sum(candidate) < 1e-10:

            candidate = candidate / np.sum(candidate) * target_sum
            sc
