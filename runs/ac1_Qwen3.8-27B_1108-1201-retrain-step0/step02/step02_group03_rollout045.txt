```python
            current_score = 2.0 * n * np.max(c) / (S ** 2)
            no_improve_streak = 0
        
        if sweep_count % 100 == 0:
            print(f"    sweep={sweep_count}, score={current_score:.8f}, best={best_score:.8f}")
    
    return best_a, best_score

def optimize_coordinate(a, c, S, i):
    """
    Optimize coordinate i while keeping others fixed.
    Uses golden section search.
    Returns (a_new, c_new, S_new).
    """
    n = len(a)
    old_val = a[i]
    
    # Determine affected lags
    # b_k changes for k where k-i is in [0, n-1]
    # k ranges from i to i+n-1, intersected with [0, 2n-2]
    k_min = max(0, i)
    k_max = min(2*n-2, i+n-1)
    
    # For each affected k, the change is 2*d*a_{k-i} (for k != 2i)
    # or d*(d+2*old_val) (for k = 2i)
    
    # Precompute the coefficients
    k_range = np.arange(k_min, k_max+1)
    j_range = k_range - i  # j = k - i, the paired index
    # j_range is in [0, n-1] by construction
    
    # Coefficients: 2*a_j for j != i, and for j == i it's special
    coeffs = 2.0 * a[j_range]  # This is 2*a_{k-i}
    # For k = 2i, the coefficient should be different (quadratic term)
    # b_{2i} changes by d*(d + 2*old_val) = d^2 + 2*d*old_val
    # While the linear formula gives 2*d*a_i = 2*d*old_val
    # So the quadratic term d^2 is extra
    
    # Base values for affected lags
    base = c[k_range].copy()
    # Subtract the old contribution: 2*old_val*a_j for j != i, old_val^2 for j == i
    old_contrib = 2.0 * old_val * a[j_range]
    # For j == i: the correct old contribution is old_val^2, but 2*old_val*a_i = 2*old_val^2
    # So we need to fix: for j==i, old_contrib should be old_val^2, not 2*old_val^2
    mask_self = (j_range == i)
    if np.any(mask_self):
        old_contrib[mask_self] = old_val ** 2
    
    base = base - old_contrib  # b_k without a_i's contribution
    
    # Now b_k(d) = base[k] + coeffs[k]*d + (d^2 if k==2i else 0)
    # But coeffs[k] = 2*a_j, and for j==i, the linear part is 2*old_val*d
    # The quadratic part d^2 is separate
    
    # Unaffected max
    affected_mask = np.zeros(2*n-1, dtype=bool)
    affected_mask[k_min:k_max+1] = True
    unaffected_max = np.max(c[~affected_mask]) if np.any(~affected_mask) else 0.0
    
    # Define the objective function h(d)
    def h(d):
        # New b values at affected lags
        b_aff = base + coeffs * d
        # Add quadratic term for k=2i
        if np.any(mask_self):
            self_idx = np.where(affected_mask[k_min:k_max+1] & (k_range == 2*i))[0]
            b_aff[self_idx] += d * d
        # New max
        new_max = max(unaffected_max, np.max(b_aff))
        new_S = S + d
        if new_S < 1e-10:
            return np.inf
        return new_max / (new_S ** 2)
    
    # Golden section search for d in [-old_val, d_upper]
    # d_upper: a reasonable upper bound
    d_upper = max(10.0 * old_val, 1.0)  # Don't increase by more than 10x
    
    # Actually, we should be smarter about the upper bound
    # The optimal d won't be huge because (S+d)^2 in denominator grows
    
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    a_lo = -old_val  # d can be at most -old_val (making a_i = 0)
    a_hi = d_upper
    
    # Evaluate at golden section points
    c_pt = a_hi - gr * (a_hi - a_lo)
    d_pt = a_lo + gr * (a_hi - a_lo)
    c_val = h(c_pt)
    d_val = h(d_pt)
    
    for _ in range(50):  # 50 iterations gives good precision
        if c_val < d_val:
            a_hi = d_pt
            d_pt = c_pt
            d_val = c_val
            c_pt = a_hi - gr * (a_hi - a_lo)
            c_val = h(c_pt)
        else:
            a_lo = c_pt
            c_pt = d_pt
            c_val = d_val
            d_pt = a_lo + gr * (a_hi - a_lo)
            d_val = h(d_pt)
    
    # Optimal d
    d_opt = (a_lo + a_hi) / 2.0
    
    a_new = a.copy()
    a_new[i] = old_val + d_opt
    a_new[i] = max(0, a_new[i])
    
    S_new = S + d_opt
    c_new = c.copy()
    # Update affected lags
    c_new[k_range] = base + coeffs * d_opt
    if np.any(mask_self):
        self_idx = np.where(k_range == 2*i)[0]
        c_new[k_min + self_idx] += d_opt ** 2
    
    return a_new, c_new, S_new
```

The function h(d) = max(A, max_k(base_k + coeff_k * d), base_{2i} + coeff_{2i}*d + d^2) / (S+d)^2

where A = unaffected max.

The numerator is the maximum of linear functions (and one quadratic). The denominator is (S+d)^2.

The optimal d is where the derivative of h is zero, or at a "kink" point.

For the linear part: if the active constraint is base_k + coeff_k * d (for some k), then

= [coeff_k*S - 2*base_k - coeff_k*d] / (S+d)^3

Setting h' = 0: coeff_k*S - 2*base_k - coeff_k*d = 0
d = (coeff_k*S - 2*base_k) / coeff_k = S - 2*base_k/coeff_k
