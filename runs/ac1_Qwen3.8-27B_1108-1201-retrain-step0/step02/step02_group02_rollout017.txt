LP-based peak minimization for non-negative step functions.
Uses linear programming to find directions that reduce the autocorrelation peak,
combined with coordinate descent polishing and multi-start strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Fast local evaluation of the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def build_conv_matrix(f):
    """Build convolution matrix A where A[k,j] = f[j] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        A[k, j_min:j_max + 1] = f[j_min:j_max + 1]
    return A

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

        M = float(np.max(c))
        
        g = solve_lp_direction(f, M)

            # Try relaxed constraint

            if g is None or np.sum(g) < 1e-12:

                f = f * np.exp(rng.normal(0, 0.01, size=n))
                f = np.clip(f, 0, None)
                f = f / np.sum(f) * S_target
                continue

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def build_conv_matrix(f):
    """Build (2n-1) x n matrix where A[k,j] = f[j] if valid for convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        A[k, j_min:j_max + 1] = f[j_min:j_max + 1]
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

                  bounds=[(0, None)] * n, method='highs')

    return None

        else:
            # Global small perturbation
            current *= np.exp(rng.normal(0, 0.003, size=n))

        current = np.clip(current, 0.0, 1000.0)
        if np.sum(current) < 0.01:
            current = best.copy()
            current_score = best_score
            continue

        new_score = local_score(current)

        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = current.copy()
        else:
            # Simulated annealing acceptance
            delta = new_score - current_score
            if delta < 0.001 and rng.random() < np.exp(-delta / max(temp, 1e-15)):
                current_score = new_score
            else:
                current = best.copy()
                current_score = best_score

        temp *= cooling

        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS step={step}, cur={current_score:.6f}, best={best_score:.6f}, t={elapsed:.0f}s")

    return best, best_score


def coordinate_descent_finish(a_init, time_limit, rng, verbose=True):
    """
    Final coordinate descent: for each element, try geometric scaling.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    round_count = 0

    if verbose:
        print(f"  CD start: score={best_score:.6f}")

    while time.monotonic() - t0 < time_limit:
        round_count += 1
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            base_val = current[j]
            if base_val < 1e-8:
                # Try turning on
                current[j] = base_val + 0.01
                new_score = local_score(current)
                if new_score < current_score:
                    current_score = new_score
                    improved = True
                    if new_score < best_score:
                        best_score = new_score
                        best = current.copy()
                else:
                    current[j] = base_val
                continue

            # Try geometric sequence
            found_improve = False
            for frac in [0.85, 0.92, 0.97, 1.03, 1.08, 1.15]:
                trial = base_val * frac
                current[j] = trial
                new_score = local_score(current)
                if new_score < current_score:
                    current_score = new_score
                    improved = True
                    found_improve = True
                    if new_score < best_score:
                        best_score = new_score
                        best = current.copy()
                else:
                    current[j] = base_val
                    break

        if not improved:
            # Random kick to escape
            j = int(rng.integers(0, n))
            current[j] = max(0.0, current[j] * np.exp(rng.normal(0, 0.1)))

        if verbose and round_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD round={round_count}, best={best_score:.6f}, t={elapsed:.0f}s")

    return best, best_score


def make_initial(n, rng, method='broadband'):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)

    if method == 'broadband':
        a = np.ones(n)
        for k in range(1, 12):
            freq = k * (1.0 + 0.7 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.12 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'two_tone':
        a = 1.0 + 0.3 * np.sin(2 * np.pi * t / n * 3) + 0.2 * np.cos(2 * np.pi * t / n * 7)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'random':
        a = 1.0 + 0.4 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'tapered':
        a = 1.0 + 0.4 * np.cos(2 * np.pi * t / n)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    else:
        a = np.ones(n) * (1.0 + 0.15 * rng.standard_normal(n))
        return np.clip(a, 0.01, None)


def propose_candidate(budget_s=2400, **kwargs):
    """
    Multi-phase LP-based optimization for minimizing autocorrelation peak score.

    Phase 1: LP optimization at primary sequence length (provided or 256/384)
    Phase 2: Peak-shave targeted polishing
    Phase 3: Coordinate descent finish
    Phase 4: Secondary length exploration
    Phase 5: Final verification and return
    """
    t_global = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_global + total_budget - 25
    rng = np.random.default_rng(12345)

    best_score = np.inf
    best_seq = None

    # Gather starting sequences
    start_seqs = []
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        s1 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if len(s1) > 10 and np.sum(s1) > 0.01:
            s1_score = local_score(s1)
            start_seqs.append(s1)
            best_score = min(best_score, s1_score)
            best_seq = s1.copy()
            print(f"Provided sequence: n={len(s1)}, score={s1_score:.6f}")

    # Determine primary length
    if start_seqs:
        primary_n = len(start_seqs[0])
    else:
        primary_n = 256

    print(f"\n{'='*60}")
    print(f"Total budget: {total_budget:.0f}s, primary n={primary_n}")
    print(f"{'='*60}")

    # === PHASE 1: LP optimization at primary length ===
    available_time = deadline - time.monotonic()
    t_lp1 = min(1000, available_time * 0.45)
    t_lp1 = max(120, t_lp1)

    print(f"\n--- PHASE 1: LP optimization (n={primary_n}, {t_lp1:.0f}s) ---")

    if best_seq is not None and len(best_seq) == primary_n:
        f_init = best_seq.copy()
    elif start_seqs:
        f_init = np.interp(
            np.linspace(0, 1, primary_n),
            np.linspace(0, 1, len(start_seqs[0])),
            start_seqs[0]
        )
    else:
        f_init = make_initial(primary_n, rng, 'broadband')

    f_init = np.clip(f_init, 0.01, 1000)
    result_f, result_score = lp_optimize(f_init, t_lp1, rng)
    actual1 = evaluate_sequence([float(max(0, min(1000, x))) for x in result_f])
    print(f"  Phase 1 result: local={result_score:.6f}, actual={actual1:.6f}")

    if actual1 < best_score:
        best_score = actual1
        best_seq = result_f.copy()
        print(f"  *** NEW BEST: {best_score:.6f} ***")

    # === PHASE 1b: Second LP run with different init ===
    available_time = deadline - time.monotonic()
    t_lp2 = min(400, available_time * 0.2)
    if t_lp2 > 60:
        print(f"\n--- PHASE 1b: LP with different init ({t_lp2:.0f}s) ---")
        f_init2 = make_initial(primary_n, rng, 'two_tone')
        result_f2, result_score2 = lp_optimize(f_init2, t_lp2, rng)
        actual2 = evaluate_sequence([float(max(0, min(1000, x))) for x in result_f2])
        print(f"  Phase 1b result: local={result_score2:.6f}, actual={actual2:.6f}")
        if actual2 < best_score:
            best_score = actual2
            best_seq = result_f2.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # === PHASE 2: Peak-shave polishing ===
    if best_seq is not None:
        available_time = deadline - time.monotonic()
        t_ps = min(500, available_time * 0.25)
        t_ps = max(60, t_ps)
        print(f"\n--- PHASE 2: Peak-shave polish ({t_ps:.0f}s) ---")
        polished, ps_score = peak_shave_polish(best_seq, t_ps, rng)
        actual_ps = evaluate_sequence([float(max(0, min(1000, x))) for x in polished])
        print(f"  Peak-shave result: local={ps_score:.6f}, actual={actual_ps:.6f}")
        if actual_ps < best_score:
            best_score = actual_ps
            best_seq = polished.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # === PHASE 3: Coordinate descent ===
    if best_seq is not None:
        available_time = deadline - time.monotonic()
        t_cd = min(300, available_time * 0.2)
        t_cd = max(30, t_cd)
        print(f"\n--- PHASE 3: Coordinate descent ({t_cd:.0f}s) ---")
        cd_result, cd_score = coordinate_descent_finish(best_seq, t_cd, rng)
        actual_cd = evaluate_sequence([float(max(0, min(1000, x))) for x in cd_result])
        print(f"  CD result: local={cd_score:.6f}, actual={actual_cd:.6f}")
        if actual_cd < best_score:
            best_score = actual_cd
            best_seq = cd_result.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # === PHASE 4: Secondary length exploration ===
    for n_try in [384, 192, 320]:
        if n_try == primary_n:
            continue
        if time.monotonic() > deadline - 80:
            break
        t_try = min(120, deadline - time.monotonic() - 40)
        if t_try < 40:
            break

        if best_seq is not None:
            f_try = np.interp(
                np.linspace(0, 1, n_try),
                np.linspace(0, 1, len(best_seq)),
                best_seq
            )
        else:
            f_try = make_initial(n_try, rng, 'broadband')

        f_try = np.clip(f_try, 0.01, 1000)
        print(f"\n--- PHASE 4: Try n={n_try} ({t_try:.0f}s) ---")
        res_f, res_s = lp_optimize(f_try, t_try, rng)
        actual_try = evaluate_sequence([float(max(0, min(1000, x))) for x in res_f])
        print(f"  n={n_try} result: local={res_s:.6f}, actual={actual_try:.6f}")
        if actual_try < best_score:
            best_score = actual_try
            best_seq = res_f.copy()
            print(f"  *** NEW BEST: {best_score:.6f} (n={n_try}) ***")

    # === PHASE 5: Final short LP polish if time remains ===
    if best_seq is not None and time.monotonic() < deadline - 40:
        t_final = deadline - time.monotonic() - 30
        if t_final > 60:
            print(f"\n--- PHASE 5: Final LP polish ({t_final:.0f}s) ---")
            final_f, final_s = lp_optimize(best_seq, t_final, rng)
            actual_final = evaluate_sequence([float(max(0, min(1000, x))) for x in final_f])
            print(f"  Final: local={final_s:.6f}, actual={actual_final:.6f}")
            if actual_final < best_score:
                best_score = actual_final
                best_seq = final_f.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")

    # === Return best sequence ===
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL: score={final_eval:.6f}, n={len(final_list)}")
        print(f"{'='*60}")
        return final_list

    # Fallback
    if start_seqs:
        return [float(max(0, min(1000, x))) for x in start_seqs[0]]
    return [1.0] * 256
