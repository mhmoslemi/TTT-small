import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score computation assuming a >= 0."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def project_simplex(v, target_sum=1.0):
    """Project vector v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    u = v.copy()
    tmp = np.sort(u)[::-1]
    cv = np.cumsum(tmp)
    rho = np.nonzero(tmp * np.arange(1, n + 1) > (cv - target_sum))[0][-1]
    theta = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(u - theta, 0.0)

        tau_min = 1.0e-6 / n

    best_score = score_fast(a)
    
    t0 = time.monotonic()
    step = 0
    lr = 0.01
    lr_decay = 0.9999
    patience = 5000
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        
        # Anneal temperature
        tau = tau_init * (tau_min / tau_init) ** elapsed_frac
        tau = max(tau, 1e-12)
        
        # Compute soft subgradient
        soft_val, g, w, c = soft_subgradient(a, tau)
        
        # Gradient step
        a_new = a - lr * g
        a_new = project_simplex(a_new, 1.0)
        
        # Line search (simple backtracking)
        new_score = score_fast(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        step += 1
        
        # Adaptive learning rate
        lr *= lr_decay
        if lr < 1e-8 / n:
            lr = 1e-8 / n
        
        # Escape mechanism
        if no_improve > patience:
            r = rng.random()
            if r < 0.3:
                # Gaussian perturbation
                a = best_a + 0.01 * rng.standard_normal(n)
                a = project_simplex(a, 1.0)
            elif r < 0.6:
                # Peak-directed: perturb elements contributing to peak
                peak_idx = int(np.argmax(c))
                contribs = np.zeros(n)
                for i in range(n):
                    if 0 <= peak_idx - i < n:
                        contribs[i] = a[i] * a[peak_idx - i]
                top_k = np.argsort(contribs)[-max(4, n // 8):]

                a = project_simplex(a, 1.0)
            else:

                block = max(4, n // 12)
                start = int(rng.integers(0, n - block))
                a[start:start+block] *= np.exp(0.03 * rng.standard_normal(block))
                a = project_simplex(a, 1.0)
            no_improve = 0
        
        if step % 2000 == 0:
            cur_score = score_fast(a)
            print(f"      step={step}, cur={cur_score:.6f}, best={best_score:.6f}, tau={tau:.2e}, lr={lr:.2e}")

def coordinate_descent_polish(a_init, time_limit, rng):

    Coordinate descent: for each element, try to find a better value.
    """
    a = a_init.astype(np.float64).copy()
    a = project_simplex(a, 1.0)
    n = len(a)
    best = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            base = a[j]
            if base < 1e-8:
                continue
            
            # Try multiplicative perturbations
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                trial = base * factor
                a[j] = trial
                a[j+1:] = a[j+1:]  # no change needed
                # Re-normalize
                a = project_simplex(a, 1.0)
                s = score_fast(a)
                if s < best_score:
                    best_score = s
                    best = a.copy()
                    improved = True
                else:
                    a[j] = base
                    a = project_simplex(a, 1.0)
                    break
        
        if not improved:
            # Random kick
            for _ in range(10):
                j = int(rng.integers(0, n))
                a[j] = max(0, a[j] * np.exp(rng.normal(0, 0.2)))
                a = project_simplex(a, 1.0)
                s = score_fast(a)
                if s < best_score * 1.01:
                    if s < best_score:
                        best_score = s
                        best = a.copy()
                    break
                a = best.copy()
        
        iteration += 1
        if iteration % 50 == 0:
            print(f"      CD iter={iteration}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")
    
    return best, best_score


def peak_shave(a_init, time_limit, rng):
    """
    Peak-directed local search: identify peak contributors and redistribute mass.
    """
    a = a_init.astype(np.float64).copy()
    a = project_simplex(a, 1.0)
    n = len(a)
    best = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    temp = 0.0005
    cooling = 0.9999
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]
        
        # Find all near-peak positions (within 5% of max)
        threshold = peak_val * 0.95
        near_peaks = np.where(c >= threshold)[0]
        
        # Compute contributions to the main peak
        contribs = np.zeros(n)
        for i in range(n):
            if 0 <= peak_idx - i < n:
                contribs[i] = a[i] * a[peak_idx - i]
        
        r = rng.random()
        
        if r < 0.4:
            # Reduce top peak contributors, boost bottom
            top_k = max(4, n // 10)
            bottom_k = max(4, n // 6)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            bottom_idx = np.argsort(contribs)[:bottom_k]
            
            red_amt = 0.03
            total_reduce = sum(a[i] * red_amt for i in top_idx)
            current[top_idx] = a[top_idx] * (1 - red_amt)
            per_boost = total_reduce / max(1, len(bottom_idx))
            a[bottom_idx] += per_boost / max(len(bottom_idx), 1)
            a = project_simplex(a, 1.0)
            
        elif r < 0.7:
            # Pair transfer: move mass from a top contributor to a random element
            top_k = max(4, n // 8)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            j_from = top_idx[int(rng.integers(0, top_k))]
            j_to = int(rng.integers(0, n))
            if j_to != j_from:
                amt = a[j_from] * rng.uniform(0.02, 0.08)
                a[j_from] -= amt
                a[j_to] += amt
                a = np.maximum(a, 0)
                a = project_simplex(a, 1.0)
                
        elif r < 0.85:
            # Scale a random block
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.05))
            a[start:start+width] *= factor
            a = project_simplex(a, 1.0)
            
        else:
            # Small global noise
            a *= np.exp(rng.normal(0, 0.01, size=n))
            a = project_simplex(a, 1.0)
        
        new_score = score_fast(a)
        if new_score < best_score:
            best_score = new_score
            best = a.copy()
        elif rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-15)):
            pass  # accept worse solution (annealing)
        else:
            a = best.copy()
        
        temp *= cooling
        if step % 2000 == 0:
            print(f"      PS step={step}, cur={score_fast(a):.6f}, best={best_score:.6f}, temp={temp:.6f}")
    
    return best, best_score


def make_initializations(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    
    # Broadband
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 20):
        freq = k * (0.5 + rng.random())
        phase = rng.uniform(0, 2 * np.pi)
        amp = 0.1 / np.sqrt(k)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0.01)
    a /= np.sum(a)
    inits.append(a.copy())
    
    # Two-level with structure
    a = np.ones(n)
    a[::2] = 0.6
    a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    a /= np.sum(a)
    inits.append(a.copy())
    
    # Raised cosine
    a = np.sin(np.pi * t / (n - 1)) ** 2
    a = np.maximum(a, 0.001)
    a /= np.sum(a)
    inits.append(a.copy())
    
    # Random with structure
    a = np.maximum(0.5 + 0.5 * rng.standard_normal(n // 2), 0.01)
    a = np.concatenate([a, a[::-1]])[:n]
    a /= np.sum(a)
    inits.append(a.copy())
    
    # Near-uniform with small perturbation
    a = np.ones(n) * (1 + 0.1 * rng.standard_normal(n))
    a = np.maximum(a, 0.01)
    a /= np.sum(a)
    inits.append(a.copy())
    
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search minimizing 2n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Load provided sequence as warm start
    2. Try multiple n values with subgradient + soft-gradient optimization
    3. Polish with peak-shave and coordinate descent
    4. Return best sequence found
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Load provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = score_fast(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()
    
    # N values to try, ordered by expected quality
    n_candidates = [384, 256, 320, 448, 192, 512]
    if start_seq is not None:
        # Put the provided n first
        n_prog = len(start_seq)
        n_candidates = [n_prog] + [x for x in n_candidates if x != n_prog]
    
    # Allocate time: 50% to first n, 20% to second, rest split
    time_allocs = [0.45, 0.20, 0.10, 0.08, 0.07, 0.05, 0.05]
    
    for n_idx, n in enumerate(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        if n_idx < len(time_allocs):
            t_alloc = time_allocs[n_idx] * time_budget
        else:
            t_alloc = 0.03 * time_budget
        t_alloc = min(t_alloc, deadline - time.monotonic() - 30)
        if t_alloc < 15:
            break
        
        print(f"\n{'='*50}")
        print(f"Optimizing n={n}, time={t_alloc:.0f}s")
        print(f"{'='*50}")
        
        # Get initializations
        if start_seq is not None and len(start_seq) == n:
            a0 = start_seq.copy()
            a0 = project_simplex(a0, 1.0)
            inits = [a0]
        elif start_seq is not None:
            # Interpolate to new length
            x_old = np.linspace(0, 1, len(start_seq))
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, start_seq)
            a0 = project_simplex(a0, 1.0)
            inits = [a0] + make_initializations(n, rng)[:2]
        else:
            inits = make_initializations(n, rng)
        
        # Run optimization on best initialization
        # Split time: 60% subgradient, 25% peak shave, 15% CD
        t_sub = 0.55 * t_alloc
        t_ps = 0.25 * t_alloc
        t_cd = 0.15 * t_alloc
        
        n_inits = min(3, len(inits))
        t_per_init = t_sub / n_inits
        
        local_best = inits[0].copy()
        local_best_score = score_fast(inits[0])
        print(f"  Init score: {local_best_score:.6f}")
        
        for i in range(n_inits):
            if time.monotonic() > deadline - 30:
                break
            t_remain = min(t_per_init, deadline - time.monotonic() - 30)
            if t_remain < 5:
                break
            
            a_opt, s_opt = optimize_subgradient(inits[i], t_remain, rng)
            if s_opt < local_best_score:
                local_best_score = s_opt
                local_best = a_opt.copy()
                print(f"  Subgrad init {i}: score={s_opt:.6f}")
        
        # Peak shave polish
        if time.monotonic() < deadline - 30:
            t_ps_remain = min(t_ps, deadline - time.monotonic() - 20)
            if t_ps_remain > 10:
                a_ps, s_ps = peak_shave(local_best, t_ps_remain, rng)
                if s_ps < local_best_score:
                    local_best_score = s_ps
                    local_best = a_ps.copy()
                    print(f"  After peak shave: {s_ps:.6f}")
        
        # Coordinate descent polish
        if time.monotonic() < deadline - 20:
            t_cd_remain = min(t_cd, deadline - time.monotonic() - 15)
            if t_cd_remain > 5:
                a_cd, s_cd = coordinate_descent_polish(local_best, t_cd_remain, rng)
                if s_cd < local_best_score:
                    local_best_score = s_cd
                    local_best = a_cd.copy()
                    print(f"  After CD: {s_cd:.6f}")
        
        # Update global best
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best.copy()
            print(f"  *** New global best: n={n}, score={best_score:.6f} ***")
    
    # Final polish if time remains
    if best_seq is not None and time.monotonic() < deadline - 20:
        t_final = deadline - time.monotonic() - 15
        if t_final > 20:
            print(f"\n{'='*50}")
            print(f"FINAL POLISH (t={t_final:.0f}s, n={len(best_seq)})")
            print(f"{'='*50}")
            a_fp, s_fp = peak_shave(best_seq, t_final * 0.6, rng)
            if s_fp < best_score:
                best_score = s_fp
                best_seq = a_fp.copy()
            if time.monotonic() < deadline - 15:
                t_cd2 = deadline - time.monotonic() - 10
                if t_cd2 > 10:
                    a_cd2, s_cd2 = coordinate_descent_polish(best_seq, t_cd2, rng)
                    if s_cd2 < best_score:
                        best_score = s_cd2
                        best_seq = a_cd2.copy()
            print(f"  Final: {best_score:.6f}")
    
    # Return the best sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"=== FINAL: evaluated_score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'='*50}")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
