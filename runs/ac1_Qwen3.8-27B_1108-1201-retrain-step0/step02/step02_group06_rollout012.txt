"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions. Uses softplus parametrization, FFT-based smooth gradient
descent with adaptive temperature, peak-targeted local search, and
diverse initializations. Focuses on exploiting the scale invariance of the
objective and targeting flattening of the convolution profile.
"""

import numpy as np
import time


def next_pow2(n):
    """Next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = next_pow2(n + m - 1)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def smooth_max_and_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Returns (smooth_value, gradient_array, convolution, weights).
    """
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    value = max_c + tau * np.log(sum_exp)

    # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1
    # This is 2 * (w cross-corr a)[i] = 2 * ifft(conj(FFT(w)) * FFT(a))[i]
    N = next_pow2(2 * n - 1 + n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(np.conj(W) * A, N)
    grad = 2.0 * corr[:n]

    return value, grad, conv, w


def softplus(x):
    """Numerically stable softplus."""
    return np.log1p(np.exp(np.clip(x, -500, 500)))


def sigmoid(x):
    """Numerically stable sigmoid."""
    x = np.clip(x, -500, 500)
    return 1.0 / (1.0 + np.exp(-x))


def init_triangular(n, rng, noise=0.05):
    """Triangular profile: peaks in the middle."""
    t = np.arange(n) / (n - 1) if n > 1 else np.array([0.5])
    a = 1.0 - np.abs(2.0 * t - 1.0)
    a = a + a.min() + 0.1  # Ensure positive
    a += noise * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def init_exponential_center(n, rng, decay=1.0, noise=0.03):
    """Exponential decay from center."""
    t = np.arange(n) / (n - 1) if n > 1 else np.array([0.5])
    center = 0.5
    a = np.exp(-decay * (t - center) ** 2 / (0.25 / n))
    a += 0.05
    a += noise * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def init_poly(n, rng, power=0.5, noise=0.03):
    """Polynomial profile."""
    t = np.arange(n) / (n - 1) if n > 1 else np.array([0.5])
    a = 1.0 - np.abs(2.0 * t - 1.0) ** power
    a = a + a.min() + 0.1
    a += noise * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def init_random_structured(n, rng, style=0):
    """Various structured random initializations."""
    if style == 0:
        # Symmetric random
        half = n // 2
        h = np.abs(rng.standard_normal(half)) + 0.5
        if n % 2 == 0:
            a = np.concatenate([h, h[::-1]])
        else:
            a = np.concatenate([h, [h[0]], h[::-1]])
    elif style == 1:
        # Low-pass filtered noise
        a = np.ones(n)
        for k in range(1, min(5, n // 2)):
            a += rng.uniform(-0.5, 0.5) * np.sin(2 * np.pi * k * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    elif style == 2:
        # Step function
        a = np.ones(n)
        for _ in range(rng.integers(1, max(2, n // 4))):
            start = rng.integers(0, n)
            width = rng.integers(1, n // 2)
            a[start:start + width] *= rng.uniform(0.5, 1.5)
    else:
        a = np.abs(rng.standard_normal(n)) + 0.3
    return np.clip(a, 0.01, None)


def init_from_prev(prev_seq, n, rng):
    """Upsample/interpolate from a previous sequence."""
    old = np.asarray(prev_seq, dtype=np.float64)
    old = np.clip(old, 0.001, 1000.0)
    x_old = np.linspace(0, 1, len(old))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, old)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def adam_optimize(n, a_init, time_limit, rng, tau_start=None, tau_end=None):
    """
    Adam optimization on softplus parametrization.
    Minimizes 2n * smooth_max(conv) / (sum a)^2.
    """
    if tau_start is None:
        tau_start = 5.0 / n
    if tau_end is None:
        tau_end = 0.05 / n

    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 1e-6, None)
    # Initialize theta from a: theta = log(exp(a) - 1) approximately
    # Use softplus inverse: if a = log(1+exp(theta)), then theta = log(exp(a)-1) for a>0
    # For small a, theta ≈ log(a). For large a, theta ≈ a.
    theta = np.log(np.maximum(a, 1e-10) - 1e-10)
    theta = np.clip(theta, -30, 30)

    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 2.0

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_fast(a)
    no_improve = 0
    max_no_improve = 2000

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Adaptive temperature: start high, decrease
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.6)

        a_curr = softplus(theta)
        a_curr = np.clip(a_curr, 1e-10, 1000.0)
        S = np.sum(a_curr)

        if S < 1e-6:
            break

        value, grad_a, conv, w = smooth_max_and_grad(a_curr, tau)

        # Full gradient of score = 2n * value / S^2
        # d(score)/d(a[i]) = 2n * (grad_a[i] * S - 2 * value) / S^3
        score_grad_a = 2.0 * n * (grad_a * S - 2.0 * value) / (S ** 3)

        # Chain rule through softplus: d(a)/d(theta) = sigmoid(theta)
        sig = sigmoid(theta)
        score_grad_theta = score_grad_a * sig

        step += 1
        m = beta1 * m + (1 - beta1) * score_grad_theta
        v = beta2 * v + (1 - beta2) * (score_grad_theta * score_grad_theta)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # Learning rate schedule
        lr_eff = lr / (1.0 + 0.1 * step ** 0.5)
        theta = theta - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        theta = np.clip(theta, -30, 30)

        # Periodic evaluation
        if step % 20 == 0:
            a_check = softplus(theta)
            a_check = np.clip(a_check, 0.0, 1000.0)
            sc = score_fast(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                no_improve = 0
            else:
                no_improve += 1

            if step % 500 == 0:
                print(f"        adam step={step}, score={sc:.6f}, best={best_score:.6f}, "
                      f"tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")

        if no_improve > max_no_improve:
            break

    return best_a, best_score


def peak_local_search(a_init, time_limit, rng):
    """
    Peak-targeted local search. Works on normalized sequence (sum=1).
    Strategy: identify peak, reduce contributing elements, redistribute.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0.0, None)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    n = len(a)

    current_score = 2.0 * n * np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    patience = 0
    max_patience = 3000
    temp = 0.01  # Metropolis temperature

    while time.monotonic() - t0 < time_limit and patience < max_patience:
        r = rng.random()

        if r < 0.35:
            # Peak-targeted reduction
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Elements contributing to peak: a[i]*a[peak_idx-i]
            contribs = np.zeros(n)
            valid_i = np.arange(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
            for i in valid_i:
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            # Reduce top contributors
            n_reduce = max(2, n // 16)
            top_idx = np.argsort(contribs)[-n_reduce:]
            total_contrib = np.sum(contribs[top_idx])
            if total_contrib < 1e-10:
                patience += 1
                continue
            candidate = a.copy()
            # Reduce proportionally
            reduction = rng.uniform(0.05, 0.2)
            for i in top_idx:
                amt = a[i] * reduction
                candidate[i] -= amt
            # Redistribute to bottom contributors
            bottom_idx = np.argsort(contribs)[:min(n_reduce * 2, n)]
            removed = np.sum(a[top_idx] - candidate[top_idx])
            if removed > 0 and len(bottom_idx) > 0:
                candidate[bottom_idx] += removed / len(bottom_idx)
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                patience += 1
                continue
            candidate = candidate / cs

        elif r < 0.6:
            # Pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.02, 0.1)
            amt = a[i] * frac
            candidate = a.copy()
            candidate[i] = a[i] - amt
            candidate[j] = a[j] + amt
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                patience += 1
                continue
            candidate = candidate / cs

        elif r < 0.8:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                patience += 1
                continue
            candidate = candidate / cs

        else:
            # Edge adjustment: reduce edge elements (they cause edge peaks)
            candidate = a.copy()
            edge_size = max(1, n // 32)
            for side in ['left', 'right']:
                if side == 'left':
                    idx = np.arange(edge_size)
                else:
                    idx = np.arange(n - edge_size, n)
                candidate[idx] *= rng.uniform(0.5, 1.0)
            # Redistribute to middle
            removed = np.sum(a - candidate)
            if removed > 0:
                mid_start = n // 4
                mid_end = 3 * n // 4
                mid_idx = np.arange(mid_start, mid_end)
                candidate[mid_idx] += removed / len(mid_idx)
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                patience += 1
                continue
            candidate = candidate / cs

        new_score = 2.0 * n * np.max(fft_convolve(candidate, a))  # Quick approx
        # Actually compute properly
        new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))

        # Metropolis accept
        if new_score < current_score:
            a = candidate
            current_score = new_score
            patience = 0
        else:
            patience += 1
            # Occasional Metropolis accept for exploration
            delta = new_score - current_score
            if delta < temp and rng.random() < np.exp(-delta / max(temp, 1e-10)):
                a = candidate
                current_score = new_score

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

        # Anneal temperature
        if patience % 500 == 0 and patience > 0:
            temp *= 0.9

    return best_a, best_score


def two_phase_refine(a_init, time_limit, rng):
    """
    Two-phase refinement: first flatten the convolution broadly,
    then target the specific peak.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0.0, None)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    n = len(a)

    t_phase1 = time_limit * 0.4
    t_phase2 = time_limit * 0.6

    # Phase 1: Reduce variance of convolution (make it flatter)
    t0 = time.monotonic()
    lr = 0.01
    while time.monotonic() - t0 < t_phase1:
        conv = fft_convolve(a, a)
        mean_c = np.mean(conv)
        # Gradient of variance: d/var/d(a[i]) = 2 * sum_k (conv[k]-mean_c) * d(conv[k])/d(a[i])
        # d(conv[k])/d(a[i]) = 2*a[k-i] if valid
        diff = conv - mean_c
        # grad[i] = 2 * sum_k diff[k] * 2 * a[k-i] (for valid k-i)
        N = next_pow2(2 * n - 1 + n - 1)
        D = np.fft.rfft(diff, N)
        A = np.fft.rfft(a, N)
        corr = np.fft.irfft(np.conj(D) * A, N)
        grad = 8.0 * corr[:n]

        a = a - lr * grad
        a = np.clip(a, 0.0, None)
        a = a / np.sum(a)
        lr *= 0.999

    # Phase 2: Peak targeting via smooth max gradient
    t1 = time.monotonic()
    tau = 1.0 / n
    lr2 = 0.05
    while time.monotonic() - t1 < t_phase2:
        elapsed = min(1.0, (time.monotonic() - t1) / t_phase2)
        tau = (1.0 / n) * (0.1)**(elapsed)
        value, grad, conv, w = smooth_max_and_grad(a, tau)
        a = a - lr2 * grad / (np.max(np.abs(grad)) + 1e-10)
        a = np.clip(a, 0.0, None)
        S = np.sum(a)
        if S < 1e-10:
            break
        a = a / S
        lr2 *= 0.9995

    final_score = 2.0 * n * np.max(fft_convolve(a, a))
    return a, final_score


def spectral_init(n, rng, n_iter=100):
    """
    Spectral iterative method: alternate between time-domain
    (non-negativity + normalization) and frequency-domain (peak reduction).
    """
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)

    for it in range(n_iter):
        # Compute linear autocorrelation
        conv = fft_convolve(a, a)
        # Target: make conv as flat as possible
        target = np.sum(conv) / len(conv)
        # Reduce peaks: a = a - alpha * (grad of max conv w.r.t. a)
        # Use: subtract proportionally to how much each a[i] contributes to peaks
        peak = np.max(conv)
        alpha = 0.3 / (peak + 1e-10)
        # Gradient of peak conv w.r.t. a
        N = next_pow2(2 * n - 1 + n - 1)
        diff = np.zeros(len(conv))
        diff[np.argmax(conv)] = 1.0  # Focus on the peak
        D = np.fft.rfft(diff, N)
        A = np.fft.rfft(a, N)
        corr = np.fft.irfft(np.conj(D) * A, N)
        grad = 2.0 * corr[:n]

        a = a - alpha * grad
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s

    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for optimal non-negative sequence minimizing
    2n * max(convolve(a,a)) / (sum a)^2.
    
    Strategy:
    1. Generate diverse initializations (spectral, structured, from previous)
    2. Optimize each with Adam on softplus parametrization
    3. Polish best candidates with peak-targeted local search
    4. Explore larger sequence lengths for better asymptotic scores
    """
    t_global = time.monotonic()
    time_budget = max(10.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.001, 1000.0)
        s0 = score_fast(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s0:.6f}")
        if s0 < best_score:
            best_score = s0
            best_seq = start_seq.copy()

    # --- Phase 1: Diversified Adam optimization ---
    print("\n=== Phase 1: Diversified Adam optimization ===")
    phase1_budget = time_budget * 0.55

    # Define configurations: (n, n_init, init_styles)
    configs = []
    # Small to medium sizes with multiple inits
    configs.append((256, 3, [0, 1, 2]))
    configs.append((512, 4, [0, 1, 2, 3]))
    configs.append((768, 3, [0, 1, 2]))
    configs.append((1024, 3, [0, 1, 2]))
    configs.append((1536, 2, [0, 1]))
    configs.append((2048, 2, [0, 1]))

    total_init_count = sum(c[1] for c in configs)
    time_per_init = phase1_budget / total_init_count

    for n, n_init, styles in configs:
        if time.monotonic() > deadline - 120:
            break

        for init_i in range(n_init):
            if time.monotonic() > deadline - 120:
                break

            t_alloc = max(3.0, time_per_init * 1.2)
            style = styles[init_i % len(styles)]

            # Generate initialization
            if style == 0:
                a0 = init_triangular(n, rng)
            elif style == 1:
                a0 = init_exponential_center(n, rng, decay=rng.uniform(0.5, 2.0))
            elif style == 2:
                a0 = init_random_structured(n, rng, style=rng.integers(0, 3))
            elif style == 3:
                a0 = init_poly(n, rng, power=rng.uniform(0.3, 1.5))
            elif style == 4:
                a0 = spectral_init(n, rng, n_iter=50)
            else:
                a0 = init_random_structured(n, rng, style=rng.integers(0, 4))

            # Also try from previous solution if available
            if start_seq is not None and init_i == 0:
                a0 = init_from_prev(start_seq, n, rng)

            # Run Adam
            a_best, sc = adam_optimize(n, a0, t_alloc, rng)
            actual_score = score_fast(a_best)

            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"  *** n={n}, init={style}, score={actual_score:.6f}, "
                      f"CUMULATIVE BEST: {best_score:.6f}, t={time.monotonic()-t_global:.1f}s")
            else:
                print(f"  n={n}, init={style}, score={actual_score:.6f}, t={time.monotonic()-t_global:.1f}s")

        # Try spectral init for this n
        if time.monotonic() < deadline - 120:
            a0 = spectral_init(n, rng, n_iter=80)
            t_alloc = max(3.0, time_per_init)
            a_best, sc = adam_optimize(n, a0, t_alloc, rng)
            actual_score = score_fast(a_best)
            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"  *** n={n}, spectral, score={actual_score:.6f}, t={time.monotonic()-t_global:.1f}s")

    # --- Phase 2: Two-phase refinement on best candidates ---
    print(f"\n=== Phase 2: Two-phase refinement ===")
    print(f"  Best before refinement: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        t_refine = min(120.0, max(20.0, (deadline - time.monotonic() - 60) * 0.4))

        # Refine the best
        a_norm = best_seq / np.sum(best_seq)
        a_ref, ref_score = two_phase_refine(a_norm, t_refine, rng)
        a_ref_scaled = a_ref * np.sqrt(2.0 * n_best)
        final_sc = score_fast(a_ref_scaled)
        print(f"  Two-phase refine (n={n_best}): {final_sc:.6f}")
        if final_sc < best_score:
            best_score = final_sc
            best_seq = a_ref_scaled.copy()
            print(f"  *** New best: {best_score:.6f} ***")

        # Also try refining at a different size
        for n_try in [n_best // 2, n_best * 2]:
            if 128 <= n_try <= 4096 and time.monotonic() < deadline - 60:
                a_try = init_from_prev(best_seq, n_try, rng)
                t_alloc = min(60.0, max(10.0, (deadline - time.monotonic() - 30) * 0.2))
                a_best2, sc2 = adam_optimize(n_try, a_try, t_alloc, rng)
                final_sc2 = score_fast(a_best2)
                print(f"  Cross-size refine (n={n_try}): {final_sc2:.6f}")
                if final_sc2 < best_score:
                    best_score = final_sc2
                    best_seq = a_best2.copy()
                    print(f"  *** New best: {best_score:.6f} ***")

    # --- Phase 3: Peak-targeted local search polish ---
    print(f"\n=== Phase 3: Local search polish ===")
    print(f"  Best before polish: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        t_polish = min(200.0, max(20.0, deadline - time.monotonic() - 30))
        a_norm = best_seq / np.sum(best_seq)
        a_pol, pol_score = peak_local_search(a_norm, t_polish, rng)
        a_pol_scaled = a_pol * np.sqrt(2.0 * len(a_pol))
        final_pol_sc = score_fast(a_pol_scaled)
        print(f"  After polish: {final_pol_sc:.6f}")
        if final_pol_sc < best_score:
            best_score = final_pol_sc
            best_seq = a_pol_scaled.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # --- Phase 4: Final quick exploration ---
    if time.monotonic() < deadline - 20 and best_seq is not None:
        print(f"\n=== Phase 4: Final exploration ===")
        remaining = deadline - time.monotonic() - 20
        n_best = len(best_seq)

        for trial in range(4):
            if time.monotonic() > deadline - 20:
                break
            n_try = [n_best, n_best, n_best * 2, n_best // 2]
            n_try = max(128, min(4096, n_try[trial]))
            t_alloc = min(30.0, remaining / (4 - trial))

            # Perturb the best solution
            a0 = best_seq.copy()
            if len(a0) != n_try:
                a0 = init_from_prev(a0, n_try, rng)
            else:
                a0 = a0 * (1.0 + 0.1 * rng.standard_normal(n_try))
                a0 = np.clip(a0, 0.01, None)

            a_best3, sc3 = adam_optimize(n_try, a0, t_alloc, rng)
            final_sc3 = score_fast(a_best3)
            print(f"  Trial {trial} (n={n_try}): {final_sc3:.6f}")
            if final_sc3 < best_score:
                best_score = final_sc3
                best_seq = a_best3.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Final verification
    print(f"\n=== FINAL RESULTS ===")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"  Score: {final_score:.6f}, n={len(final_list)}")
        print(f"  Total time: {time.monotonic() - t_global:.1f}s")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
