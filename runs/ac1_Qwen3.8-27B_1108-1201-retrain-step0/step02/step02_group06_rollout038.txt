```python
"""
LP-based optimization for minimizing 2n*max(convolve(a,a))/sum(a)^2
over non-negative sequences a. Uses linear programming to identify
optimal blend directions at each iteration, with spectral initialization
and gradient-based polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_convolve(a, b):
    """Compute linear convolution a * b via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * conv.max() / (S * S)


def precompute_lp_indices(n):
    """Return (rows, cols) arrays for the sparse LP constraint matrix.
    The matrix A has shape (2n-1, n) where A[k, j] = a[k-j] if valid."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(a, rows, cols, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    vals = a[rows - cols]
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = np.ones(n)  # we minimize -c, so c = -ones for max... 
    # Actually linprog minimizes c^T x. We want to maximize sum(b).
    # So minimize -sum(b), i.e., c = -ones(n).
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

    if S < 1e-10:
        return a, np.inf
    a = a / S
    n = len(a)
    
    rows, cols = precompute_lp_indices(n)
    
    conv = fft_convolve(a, a)
    M = conv.max()
    best_M = M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = solve_lp(a, rows, cols, M)
        if b is None:
            break
        b = np.clip(b, 0, None)
        S_b = b.sum()
        if S_b < 1e-10:
            break

        if np.max(np.abs(b_norm - a)) < 1e-8:
            stall_count += 1
            if stall_count > 3:

        else:
            stall_count = 0

        best_M_new = M
        for t in [0.5, 0.3, 0.2, 0.5, 0.7, 0.1, 0.8, 0.4]:
```

```python
"""
LP-based optimization for minimizing peak-to-sum ratio of autocorrelation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def max_conv(a):
    return fft_convolve(a, a).max()


def score_fast(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * max_conv(a) / (S * S)


def precompute_lp(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int64), np.array(cols, dtype=np.int64)


def solve_lp(a, rows, cols, M):
    n = len(a)
    vals = a[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def lp_optimize(a_init, time_limit):
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    S = a.sum()
    if S < 1e-10:
        return a_init, np.inf
    a = a / S
    n = len(a)
    rows, cols = precompute_lp(n)
    M = max_conv(a)
    best_M = M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = solve_lp(a, rows, cols, M)
        if b is None:
            break
        b = np.clip(b, 0, None)
        S_b = b.sum()
        if S_b < 1e-10:
            break
        b = b / S_b
        
        # Check for stagnation
        if np.max(np.abs(b - a)) < 1e-10:
            no_improve += 1
            if no_improve > 5:
                break
            # Perturb to escape
            a = a + 0.001 * np.random.default_rng(it).standard_normal(n)
            a = np.clip(a, 0, None)
            a = a / a.sum()
            M = max_conv(a)
            continue
        
        no_improve = 0
        
        # Line search
        best_t = 0.0
        best_M_try = M
        for t in [0.5, 0.25, 0.75, 0.1, 0.9, 0.15, 0.35]:
            a_try = (1 - t) * a + t * b
            a_try = np.clip(a_try, 0, None)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            M_try = max_conv(a_try)
            if M_try < best_M_try:
                best_M_try = M_try
                best_t = t
        
        if best_t < 1e-8:
            break
        
        a = (1 - best_t) * a + best_t * b
        a = np.clip(a, 0, None)
        a = a / a.sum()
        M = max_conv(a)
        it += 1
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
        
        if it % 50 == 0:
            score = 2 * n * best_M
            print(f"      LP iter={it}, M={best_M:.10f}, score={score:.6f}, "
                  f"t={time.monotonic()-t0:.1f}s")
    
    print(f"      LP complete: {it} iters, score={2*n*best_M:.6f}")
    return best_a, best_M
```
