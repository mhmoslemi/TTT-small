import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score computation: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:

    return 2.0 * n * np.max(conv) / (S * S)


def max_conv_smooth(a, tau):
    """Compute smoothed max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    value = max_c + tau * np.log(np.sum(exp_vals))
    # grad_j = 2 * sum_m a[m] * w[m+j] (valid range)
    # = cross_correlation(a, w)[j]
    size = 2 * n
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(fa) * fw, size)[:n]
    grad = 2.0 * corr
    return value, grad, conv


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1} via sorting method."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_init(n, rng, n_iter=50):
    """
    Spectral iterative method to find a good initial sequence.
    Alternates between time-domain (non-negativity) and frequency-domain
    (peak reduction) constraints.
    """
    a = 1.0 + 0.5 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)

    for it in range(n_iter):
        # Compute circular autocorrelation via FFT
        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        corr = np.fft.irfft(power, 2 * n)[:n]
        # Reduce peak: subtract a fraction of the correlation
        alpha = 1.0 / (n + 1.0)
        a = a - alpha * corr
        # Enforce non-negativity and normalize
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s

    return a


def ramp_init(n, rng, style='linear'):

def coordinate_descent(a_init, time_limit, rng):

    Aggressive coordinate descent / local search.
    Tries: single-element changes, pair transfers, block modifications.
    Works on the simplex (sum a = 1).

    a = a_init.copy()

        return a, np.inf

    patience = 0
    max_patience = 5000

    while time.monotonic() - t0 < time_limit and patience < max_patience:
        r = rng.random()

        if r < 0.4:
            # Pair transfer: move mass from i to j
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.15)
            amt = a[i] * frac
            candidate = a.copy()
            candidate[i] = a[i] - amt
            candidate[j] = a[j] + amt
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs * (np.sum(a) / cs) * cs  # keep same sum
            # Actually just re-normalize
            candidate = candidate / np.sum(candidate)

            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        elif r < 0.6:
            # Single element scale
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.05))
            candidate = a.copy()
            candidate[j] *= factor
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs

            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        elif r < 0.8:

            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, size=width))
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs

            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        else:
            # Peak-targeted: reduce elements contributing most to the peak
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Find top contributors to conv[peak_idx]
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
                    if i != j:
                        contribs[j] = a[i] * a[i]  # approximate
            top_k = max(3, n // 32)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate = a.copy()
            candidate[top_idx] *= (1.0 - rng.uniform(0.01, 0.05, size=len(top_idx)))
            # Redistribute to random other elements
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_add = min(5, len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                total_removed = np.sum(a[top_idx] - candidate[top_idx])

            cs = np.sum(candidate)
            if cs < 1e-10:

                a = candidate
                current_score = new_score
                patience = 0
            else:

def optimize_projected_grad(n, a_init, time_limit, rng):
    """
    Projected gradient descent with Nesterov acceleration on the simplex.
    Directly minimizes max_conv(a) subject to sum(a)=1, a>=0.
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a_prev = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_val = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    step = 0
    lr = 1.0 / (4.0 * n)  # Initial learning rate

    tau = 5.0 / n
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = (5.0 / n) * (0.05 / 5.0) ** (elapsed_frac ** 0.8)

        t_prev = t_mom
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        beta = (t_prev - 1.0) / t_mom

        y = a + beta * (a - a_prev)
        y = project_simplex(y)

        value, grad, conv = max_conv_smooth(y, tau)

        # Simple step (no line search for speed)
        candidate = y - lr * grad
        candidate = project_simplex(candidate)

        a_prev = a
        a = candidate
        step += 1

        actual_val = np.max(fft_convolve(a, a))
        if actual_val < best_val:
            best_val = actual_val
            best_a = a.copy()

        if step % 1000 == 0:
            print(f"      PGD step={step}, max_conv={best_val:.8f}, "
                  f"score={2*n*best_val:.6f}, t={time.monotonic()-t0:.1f}s")

    return best_a, 2.0 * n * best_val
