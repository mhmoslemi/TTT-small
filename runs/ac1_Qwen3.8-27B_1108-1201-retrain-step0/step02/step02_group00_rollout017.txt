"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Key improvements over previous approach:
1. Barzilai-Borwein adaptive step sizes for faster convergence
2. Hybrid LSE + exact subgradient for better precision
3. Multi-length exploration (4096, 8192, 16384)
4. Aggressive multi-phase optimization
5. Better restart strategy

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    condition = u * rho_candidates > (cssv - S)
    if not np.any(condition):
        return np.full(n, S / n)
    rho = np.max(rho_candidates[condition]) - 1  # 0-indexed
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_conv_self(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def grad_lse(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def grad_toppk(a, c, k):
    """Subgradient using top-k lags."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    w = top_vals / total
    w_full = np.zeros(2 * n - 1)
    w_full[top_idx] = w
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def normalize(a, n):
    """Clamp and normalize to sum = sqrt(2n)."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * n)
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    Main optimization loop with BB steps and adaptive LSE.
    phase 1: exploration (moderate beta, larger steps)
    phase 2: refinement (high beta, smaller steps)
    phase 3: polish (very high beta, BB steps, very small steps)
    """
    a = normalize(a_init, n)
    y = a.copy()  # For Nesterov (only in phase 1)
    t_mom = 1.0
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    
    # BB step size state
    prev_x = a.copy()
    prev_g = None
    step = 1.0 / (10.0 * 2.0)  # Initial step
    
    it = 0
    beta_log_start = 5.0 if phase == 1 else (50.0 if phase == 2 else 200.0)
    beta_log_end = 50.0 if phase == 1 else (300.0 if phase == 2 else 1000.0)
    
    c = fft_conv_self(a)
    cur_max = c.max()
    best_score = cur_max
    
    use_nesterov = (phase == 1)
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta
        frac = min(1.0, it / (max_time * 50))  # approximate
        beta = beta_log_start * (beta_log_end / beta_log_start) ** frac
        
        # Compute gradient
        if beta < 100:
            grad = grad_lse(y, c, beta)
        else:
            # Switch to top-k for precision
            k = min(64, max(1, int(10 * np.exp(-beta / 200))))
            if k < 1:
                k = 1
            grad = grad_toppk(y, c, max(2, k))
        
        # BB step size
        if prev_g is not None:
            s_vec = y - prev_x
            y_vec = grad - prev_g
            sy = np.dot(s_vec, y_vec)
            yy = np.dot(y_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            if yy > 1e-16 and sy > 1e-16:
                step_bb = ss / sy
                # Clip step size
                step = np.clip(step_bb, 1e-8, 1.0 / (beta * 2.0))
            elif yy > 1e-16:
                step = 0.9 * step  # Dampen
        prev_x = y.copy()
        prev_g = grad.copy()
        
        # Gradient step
        x_new = y - step * grad
        x_new = project_simplex(x_new, S)
        
        # Check improvement
        if it % 5 == 0:
            c_new = fft_conv_self(x_new)
            max_new = c_new.max()
            if max_new < best_score:
                best_score = max_new
                best_a = x_new.copy()
        
        # Nesterov momentum (phase 1 only)
        if use_nesterov:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2))
            momentum = (t_mom - 1.0) / t_new
            y = x_new + momentum * (x_new - a)
            y = np.maximum(y, 0.0)
            y = project_simplex(y, S)
            t_mom = t_new
        else:
            y = x_new
        
        a = x_new
        c = fft_conv_self(y)
        
        # Periodic restart
        if phase == 1 and it % 2000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            y = project_simplex(a + noise, S)
            a = y.copy()
            t_mom = 1.0
            prev_x = a.copy()
            prev_g = None
            c = fft_conv_self(a)
        
        # Check time
        elapsed = time.monotonic() - start
        if elapsed > max_time:
            break
    
    return best_a, best_score

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- Multi-phase proximal gradient with adaptive log-sum-exp smoothing
- Barzilai-Borwein adaptive step sizes for faster convergence
- Multi-length exploration (4096, 8192, 16384)
- Hybrid LSE/exact-subgradient for precision at high beta
- Aggressive polishing in final phase with very high beta
- Periodic restarts to escape local minima during exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1, dtype=np.float64)
    condition = u * rho_candidates > (cssv - S)
    if not np.any(condition):
        return np.full(n, S / n, dtype=np.float64)
    rho = int(np.max(rho_candidates[condition])) - 1
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_conv_self(a):
    """Full self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def grad_lse(a, c, beta):
    """Gradient of LSE_approx(max(conv)) w.r.t. a via FFT cross-correlation."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def grad_toppk(a, c, k):
    """Subgradient using top-k active lags, computed via FFT."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    w = top_vals / total
    w_full = np.zeros(2 * n - 1)
    w_full[top_idx] = w
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def normalize_seq(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def run_single_phase(a_init, n, S, max_time, rng, beta_start, beta_end,
                     use_nesterov=True, restart_prob=0.0, restart_every=3000,
                     noise_scale=0.005, step_scale=1.0, top_k_switch=80):
    """
    Run one phase of optimization.
    Returns (best_a, best_score) where score = max(conv(a,a)).
    """
    a = normalize_seq(a_init, n)
    y = a.copy()
    t_mom = 1.0
    prev_x = a.copy()
    prev_grad = None
    step = step_scale / (2.0 * beta_start)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0

    c = fft_conv_self(a)
    best_score = c.max()

    total_iters_est = max_time / 0.003  # rough estimate of iters per second ~300 for n~8192

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1.0, total_iters_est))
        beta = beta_start * (beta_end / beta_start) ** frac

        # Compute gradient
        if beta < top_k_switch:
            grad = grad_lse(y, c, beta)
        else:
            # For high beta, use top-k with small k for precision
            k = max(2, min(8, int(2 * np.exp(-(beta - top_k_switch) / 200.0))))
            grad = grad_toppk(y, c, k)

        # BB step size adaptation
        if prev_grad is not None:
            s_vec = y - prev_x
            y_vec = grad - prev_grad
            sy = np.dot(s_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            if ss > 1e-20:
                if sy > 1e-20:
                    step_bb = ss / sy
                    step = np.clip(step_bb, 1e-10, 2.0 / (beta * c.max() + 1e-10))
                else:

        prev_x = y.copy()
        prev_grad = grad.copy()

        # Gradient step + projection

        # Periodic evaluation

            if mn < best_score:

        # Nesterov momentum

            t_mom = t_new
        else:
            y = x_new

        a = x_new
        c = fft_conv_self(y)

        # Restart
        if restart_prob > 0 and it % restart_every == 0 and rng.random() < restart_prob:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            y = project_simplex(a + noise, S)
            a = y.copy()
            t_mom = 1.0
            prev_x = a.copy()
            prev_grad = None
            c = fft_conv_self(a)

    # Final check
    c_final = fft_conv_self(a)
    if c_final.max() < best_score:
        best_score = c_final.max()
        best_a = a.copy()

    return normalize_seq(best_a, n), best_score


def make_inits(n, rng):
    """Create diverse initializations of length n."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Random
    inits.append(rng.random(n) + 0.1)

    # Sinusoidal
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt))

    # U-shaped (large at edges)
    inits.append(1.0 + 0.5 * np.cos(2 * np.pi * tt) ** 2)

    # Bump (large in middle)
    inits.append(np.exp(-2.0 * (tt - 0.5) ** 2) + 0.1)

    # Two bumps
    inits.append(np.exp(-8.0 * (tt - 0.25) ** 2) + np.exp(-8.0 * (tt - 0.75) ** 2) + 0.05)

    # Cosine window (raised cosine)
    inits.append(0.5 * (1.0 + np.cos(2 * np.pi * tt)))

    return inits
