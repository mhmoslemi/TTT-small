4. Nesterov momentum in log-parameter space
5. More diverse structured initializations
6. Periodic "re-randomization" to escape local minima
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def a_from_theta(theta, S):
    """Convert log-parameters to sequence: a = S * exp(theta) / sum(exp(theta))."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def compute_soft_weights(c, T):
    """Compute soft-max weights for the convolution values."""
    c_max = c.max()
    if c_max < 1e-15:
        return np.ones(len(c)) / len(c)
    scaled = T * (c - c_max)
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        return np.ones(len(c)) / len(c)
    return exp_scaled / total


def get_grad_a(w, a):
    """Compute grad_a[m] = sum_k w[k] * (a[k-m] + a[m-k]) via FFT."""
    n = len(a)
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    end_idx = min(2 * n - 1, len(full_conv))
    grad = 2.0 * full_conv[n - 1: end_idx][:n]
    return grad


def objective_true(a):
    """Compute true max of self-convolution."""
    c = fft_convolve_self(a)
    return c.max()


def objective_and_grad(theta, S, T, top_k=1):
    """
    Compute soft-max (or hard top-k) objective and gradient w.r.t. theta.
    Returns (obj, grad_theta, c_max, w).
    """
    n = len(theta)
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()

    if T < 1e6:
        shifted = T * (c - c_max)
        log_sum_exp = np.log(np.sum(np.exp(shifted)))
        obj = c_max + log_sum_exp / T
        w = np.exp(shifted) / np.sum(np.exp(shifted))
    else:
        obj = c_max
        k = max(1, top_k)
        k = min(k, len(c))
        idx = np.argpartition(c, -k)[-k:]
        w = np.zeros(len(c))
        w[idx] = 1.0 / k

    grad_a = get_grad_a(w, a)
    dot = np.dot(grad_a, a)
    grad_theta = a * grad_a - (a / S) * dot

    return obj, grad_theta, c_max, w


def line_search_adam(theta, S, T, top_k, lr_init, max_iter_ls=10):
    """
    Adam step with backtracking line search in log-parameter space.
    Returns (new_theta, improved, c_max_new).
    """
    obj0, grad, c_max0, _ = objective_and_grad(theta, S, T, top_k)

    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-14:
        return theta, False, c_max0

    # Direction is -grad (normalized)
    direction = -grad / gnorm

    # Backtracking line search
    alpha = lr_init
    for _ in range(max_iter_ls):
        theta_trial = theta + alpha * direction
        a_trial = a_from_theta(theta_trial, S)
        c_trial = fft_convolve_self(a_trial)
        c_max_trial = c_trial.max()
        if c_max_trial < c_max0:
            return theta_trial, True, c_max_trial
        alpha *= 0.5

    return theta, False, c_max0


def run_mirror_descent(theta_init, S, time_budget, T_start=10.0, T_end=300.0,
                       lr_start=1.0, decay_rate=0.995, top_k_switch=200.0,
                       noise_interval=200, noise_scale=0.1):
    """
    Main optimization loop: mirror descent with adaptive temperature
    and periodic noise injection.
    """
    theta = theta_init.copy().astype(np.float64)
    n = len(theta)
    start = time.monotonic()

    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_cmax = c.max()
    best_theta = theta.copy()
    best_a = a.copy()

    # For noise injection
    rng = np.random.default_rng(12345)
    step = 0
    last_improve = 0

    while time.monotonic() - start < time_budget:
        step += 1
        progress = (time.monotonic() - start) / time_budget

        # Adaptive temperature
        if step < top_k_switch:
            T = T_start * np.exp(2.0 * progress)
        else:
            T = T_end

        # Use hard top-k at high temperature for precision
        if T > 100.0:
            k = min(8, int(8 * (1.0 - 50.0 / T)))
            if k < 1:
                k = 1
            top_k = k
        else:
            top_k = 0

        # Compute objective and gradient
        obj, grad, c_max, w = objective_and_grad(theta, S, T, top_k)

        if c_max < best_cmax:
            best_cmax = c_max
            best_theta = theta.copy()
            best_a = a_from_theta(theta, S).copy()
            last_improve = step

        # Stagnation detection: inject noise if no improvement for a while
        if step - last_improve > noise_interval and step > 100:
            noise = rng.normal(0, noise_scale, size=n)
            theta = theta + noise
            noise_scale *= 0.5
            a_from_theta(theta, S)
            continue

        # Adam-style update with momentum
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Simple gradient step with line search
        lr = lr_start * (decay_rate ** step)
        direction = -grad / gnorm * gnorm  # keep magnitude for adaptive step

        # Try step with line search
        alpha = lr
        improved = False
        for _ in range(8):
            theta_trial = theta + alpha * direction
            a_trial = a_from_theta(theta_trial, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < c_max:
                theta = theta_trial
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try smaller fixed step
            theta = theta - 0.01 * grad / (gnorm + 1e-10)

    return best_theta, best_a, best_cmax


def run_adam_nesterov(theta_init, S, time_budget, T_schedule=None,
                      lr=0.05, beta_momentum=0.9, top_k=4):
    """
    Adam with Nesterov momentum and adaptive temperature.
    """
    theta = theta_init.copy().astype(np.float64)
    n = len(theta)
    start = time.monotonic()

    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_cmax = c.max()
    best_theta = theta.copy()
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    velocity = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    prev_theta = theta.copy()

    while time.monotonic() - start < time_budget:
        t += 1
        progress = (time.monotonic() - start) / time_budget

        # Temperature schedule
        if T_schedule is not None:
            T = T_schedule(progress)
        else:
            T = 10.0 * np.exp(3.0 * progress)

        # Nesterov look-ahead
        theta_nest = theta + beta_momentum * (theta - prev_theta)
        prev_theta = theta.copy()

        obj, grad, c_max, w = objective_and_grad(theta_nest, S, T, top_k)

        if c_max < best_cmax:
            # Verify with actual theta
            a_check = a_from_theta(theta, S)
            c_check = fft_convolve_self(a_check)
            if c_check.max() < best_cmax:
                best_cmax = c_check.max()
                best_theta = theta.copy()
                best_a = a_check.copy()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr_t = lr * (0.8 ** (progress * 3.0))
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)

    return best_theta, best_a, best_cmax


def solve_lp_direction(f, M, n, top_k_constraints=40):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0.
    Returns normalized g and LP value.
    """
    from scipy.optimize import linprog

    c = fft_convolve_self(f)
    if top_k_constraints < len(c):
        active_lags = np.argsort(c)[::-1][:top_k_constraints]
    else:
        active_lags = np.arange(len(c))

    # Build constraint matrix: A[k_idx, j] = f[k-j] if 0<=k-j<n
    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[idx, j] = f[k - j]

    c_obj = -np.ones(n)
    b = np.full(len(active_lags), M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': 15, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return None, None


def coordinate_descent_polish(a, n, S, time_budget, max_sweeps=30):
    """Coordinate descent polish: try reducing each element."""
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    # Identify "important" indices (those contributing most to peaks)
    c_full = fft_convolve_self(a)
    peak_indices = np.argsort(c_full)[::-1][:20]

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False

        # Process indices in order of their contribution to peaks
        contributions = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                contributions[j] += c_full[k] * a[j] * a[k - j]

        order = np.argsort(contributions)[::-1]

        for i in order:
            if time.monotonic() - start > time_budget:
                break
            original = a[i]
            if original <= 1e-10:
                continue

            # Try reducing a[i]
            best_local = original
            best_local_obj = best_obj
            lo, hi = 0.0, original
            for _ in range(10):
                mid = (lo + hi) / 2.0
                a[i] = mid
                s = a.sum()
                if s < 1e-10:
                    a[i] = original
                    break
                a_norm = a * (S / s)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_local_obj:
                    best_local_obj = obj_check
                    best_local = mid
                    hi = mid
                else:
                    lo = mid

            a[i] = best_local
            if abs(best_local - original) > 1e-8:
                improved = True

            # Renormalize
            s = a.sum()
            if s > 1e-10:
                a = a * (S / s)

            c = fft_convolve_self(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()

        if not improved:
            break

    return best_a, best_obj


def get_initialization(name, n, rng, S):
    """Generate various initializations."""
    i = np.arange(n, dtype=np.float64)
    if name == "uniform":
        a = np.full(n, 1.0)
    elif name == "random":
        a = rng.random(n)
    elif name == "gaussian":
        a = np.exp(-0.5 * ((i - n / 2) / (n / 5)) ** 2)
    elif name == "cosine_high":
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * i / n)
        a = np.maximum(a, 0.01)
    elif name == "cosine_low":
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 2 * i / n)
        a = np.maximum(a, 0.01)
    elif name == "anti_peak":
        a = 1.0 - 0.3 * np.cos(2 * np.pi * (n - 1) * i / n)
        a = np.maximum(a, 0.01)
    elif name == "two_humps":
        a = (np.exp(-0.5 * ((i - n * 0.25) / (n / 8)) ** 2) +
             np.exp(-0.5 * ((i - n * 0.75) / (n / 8)) ** 2))
    elif name == "three_humps":
        a = (np.exp(-0.5 * ((i - n * 0.2) / (n / 10)) ** 2) +
             np.exp(-0.5 * ((i - n * 0.5) / (n / 10)) ** 2) +
             np.exp(-0.5 * ((i - n * 0.8) / (n / 10)) ** 2))
    elif name == "sinc_like":
        x = (i - n / 2) / (n / 20)
        with np.errstate(divide='ignore', invalid='ignore'):
            a = np.sinc(x)
        a = np.abs(a)
        a = np.maximum(a, 0.001)
    elif name == "plateau_edges":
        a = np.ones(n)
        edge = n // 8
        a[:edge] = 0.3 * (1 - np.linspace(1, 0, edge))
        a[-edge:] = 0.3 * (1 - np.linspace(1, 0, edge))[::-1]
    else:
        a = np.full(n, 1.0)

    a = np.maximum(a, 1e-6)
    return a * (S / a.sum())

            theta = np.log(a / a.sum())
            t0 = time.monotonic()
            theta, a, cmax = run_adam_nesterov(
                theta, S, remaining,
                T_schedule=lambda p: 200.0 + 200.0 * p,
                lr=0.01, beta_momentum=0.97, top_k=8
            )
            elapsed = time.monotonic() - t0
            if cmax < global_best:
                global_best = cmax
                global_a = a.copy()
            print("Final polish: obj=%.8f (t=%.1fs)" % (cmax, elapsed))

    # ===== Final verification =====
    print("=== Final ===")
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("FINAL: internal=%.8f, verified=%.8f" % (global_best,
          ver if np.isfinite(ver) else np.inf))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback used: obj=%.6f" % ver)

    return final_a
