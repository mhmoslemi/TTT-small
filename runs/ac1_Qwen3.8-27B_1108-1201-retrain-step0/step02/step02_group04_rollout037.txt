Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam gradient descent with analytical gradient, diverse initializations
(periodic, structured, random, provided sequence), and iterative refinement.
Key insight: d(max(conv))/d(a[m]) = 2*a[k*-m] where k* = argmax(conv).

def fft_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_gradient(a):
    """Compute score and analytical gradient."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf, np.inf, -1, None
    conv = fft_convolve(a)
    M = float(np.max(conv))
    ks = int(np.argmax(conv))
    score = 2.0 * n * M / (S * S)
    grad = np.full(n, -4.0 * n * M / (S ** 3))
    lo = max(0, ks - n + 1)
    hi = min(n - 1, ks)
    if hi >= lo:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 4.0 * n * (S * a[ks - idx] - M) / (S ** 3)
    return score, M, ks, grad
