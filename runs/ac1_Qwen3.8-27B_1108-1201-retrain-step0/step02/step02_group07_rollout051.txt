def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_value(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_exact(conv, a, peak_indices):
    """Exact subgradient at specified peak indices."""
    n = len(a)
    sg = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                if j == kj:
                    sg[j] += a[j]
                else:
                    sg[j] += 2.0 * a[kj]
    return sg


def subgradient_smoothed(conv, a, temperature):
    """Smoothed subgradient using softmax weighting over peaks."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = temperature * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # Only use significant weights
    thresh = w.max() * 0.01
    active = np.where(w > thresh)[0]
    sg = np.zeros(n)
    for ki in active:
        k = int(ki)
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                if j == kj:
                    sg[j] += w[ki] * a[j]
                else:
                    sg[j] += w[ki] * 2.0 * a[kj]
    return sg

    best_a = a.copy()

            if 0 <= kj < n:
                if j == kj:
                    contributions[j] = a[j]
                else:
                    contributions[j] = 2.0 * a[kj]
        
        total = contributions.sum()
        if total < 1e-15:
            continue
        probs = contributions / total
        j_source = int(rng.choice(n, p=probs))
        
        # Choose target: prefer positions that don't contribute to peaks
        # Compute "peak exposure" for each position
        peak_exposure = np.zeros(n)
        for pk in peak_positions[:10]:
            for j in range(n):
                kj = pk - j
                if 0 <= kj < n:
                    peak_exposure[j] += a[kj]
        
        # Inverse-proportional selection (low exposure = good target)
        target_weights = 1.0 / (peak_exposure + 0.01)
        target_weights[j_source] = 0
        tw_sum = target_weights.sum()
        if tw_sum < 1e-15:
            continue
        target_probs = target_weights / tw_sum
        j_target = int(rng.choice(n, p=target_probs))
        
        if j_target == j_source:
            continue
        
        delta = a[j_source] * rng.uniform(0.02, 0.15)
        if delta < 1e-10:
            continue
        
        a[j_source] -= delta
        a[j_target] += delta
        if a[j_source] < 0:
            a[j_source] += delta
            a[j_target] -= delta
            continue
        
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            accept_prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < accept_prob:
                cur_obj = new_obj
                conv = new_conv
            else:
                a[j_source] += delta
                a[j_target] -= delta
    
    return best_a, best_obj


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    # Raised cosine (Hann)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    hann = np.maximum(hann, 0)
    if hann.sum() > 1e-10:
        inits['hann'] = hann / hann.sum()
    
    # Tukey windows with different tapers
    for taper in [0.3, 0.5, 0.7]:
        tw = np.ones(n)
        w = int(taper * n / 2)
        if w > 0:
            tw[:w] = 0.5 * (1 + np.cos(np.pi * (np.arange(w) + 0.5) / w))
            tw[-w:] = 0.5 * (1 + np.cos(np.pi * (np.arange(w) + 0.5) / w))
        tw = np.maximum(tw, 0)
        if tw.sum() > 1e-10:
            inits['tukey_%d' % int(taper * 10)] = tw / tw.sum()
    
    # Beta distribution shapes
    for alpha, beta in [(2, 2), (3, 1), (1, 3), (4, 4), (5, 2), (2, 5)]:
        vals = (x / max(n - 1, 1)) ** (alpha - 1) * (1 - x / max(n - 1, 1)) ** (beta - 1)
        vals = np.maximum(vals, 1e-10)
        inits['beta_%d_%d' % (alpha, beta)] = vals / vals.sum()
    
    # Sinusoidal modulations
    for freq in [1, 2, 3]:
        s = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n)
        s = np.maximum(s, 1e-10)
        inits['cos_%d' % freq] = s / s.sum()
    
    # Skewed gaussian
    for mu in [0.3, 0.5, 0.7]:
        g = np.exp(-0.5 * ((x - mu * n) / (0.15 * n)) ** 2)
        g = np.maximum(g, 1e-10)
        inits['gauss_%d' % int(mu * 100)] = g / g.sum()
    
    # Random dirichlet
    for alpha in [0.5, 1.0, 2.0, 0.2]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['rand_a%d' % int(alpha * 10)] = r
    
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi = np.maximum(bi, 1e-10)
    inits['bimodal'] = bi / bi.sum()
    
    # Chirp
    chirp = 0.5 * (1 + np.cos(2 * np.pi * x * x / (2.0 * n)))
    chirp = np.maximum(chirp, 1e-10)
    inits['chirp'] = chirp / chirp.sum()
    
    # Parent initialization
    if parent is not None:
        p = np.maximum(parent, 0.0).copy()
        if p.size >= n:
            p = p[:n]
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        p = np.maximum(p, 0)
        if p.sum() > 1e-10:
            inits['parent'] = p / p.sum()
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    return inits


def get_parent():
    """Try to access parent sequence from global variable."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def safe_evaluate(a):
    """Safely evaluate a sequence."""
    try:
        a_clean = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a_clean = np.minimum(a_clean, 1000.0)
        if a_clean.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a_clean.tolist()))
        return score
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Multi-stage hybrid optimizer:
    Phase 1: Quick multi-start scan across lengths and initializations
    Phase 2: Deep mirror descent + coordinate descent
    Phase 3: Intensive annealing and local search
    Phase 4: Final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    
    parent = get_parent()
    
    best_sequence = None
    best_score = float('inf')
    
    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d" % (score, len(a)))
    
    # Phase 1: Quick scan
    print("Phase 1: Quick multi-start scan...")
    scan_lengths = [128, 192, 256, 320, 384]
    scan_results = []
    phase1_budget = min(400.0, (deadline - time.monotonic()) * 0.2)
    
    all_configs = []
    for n in scan_lengths:
        inits = generate_diverse_inits(n, rng, parent)
        for name, a0 in inits.items():
            all_configs.append((n, name, a0))
    
    # Sort by likelihood of being good (parent first, then structured inits)
    priority = {'parent': 0, 'hann': 1, 'tukey_50': 1, 'beta_2_2': 1,
                'cos_1': 2, 'gauss_50': 2, 'uniform': 3, 'chirp': 3}
    all_configs.sort(key=lambda x: (priority.get(x[1], 5), x[0]))
    
    n_configs = len(all_configs)
    time_per = max(2.0, phase1_budget / n_configs)
    
    for idx, (n, name, a0) in enumerate(all_configs):
        if time.monotonic() > deadline - 1800:
            break
        t_alloc = min(time_per, deadline - time.monotonic() - 1800)
        if t_alloc < 2.0:
            break
        a_opt, _ = mirror_descent_v2(a0, time_limit=t_alloc, t_init=20.0, eta_init=0.4)
        score = safe_evaluate(a_opt)
        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
        record(a_opt)
        
        if (idx + 1) % 20 == 0:
            print("  Progress: %d/%d configs, best so far: %.6f" % (idx + 1, n_configs, best_score))
    
    print("  Scan complete. Best: %.6f" % best_score)
    
    # Sort results and pick top candidates
    scan_results.sort(key=lambda x: x[0])
    
    # Phase 2: Deep optimization
    print("Phase 2: Deep optimization...")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 12:
            top_configs.append((n, name, a_opt))
            seen.add(key)
    
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))
    
    phase2_budget = min(1200.0, (deadline - time.monotonic()) * 0.55)
    n_deep = len(top_configs)
    time_per_deep = max(15.0, phase2_budget / max(1, n_deep))
    
    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 800:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 800)
        if t_alloc < 10.0:
            break
        print("  [%d/%d] n=%d, %s, budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))
        
        a_cur = a0.copy()
        a_cur = np.maximum(a_cur, 0)
        a_cur /= a_cur.sum()
        
        # Alternating optimization stages
        sub = t_alloc / 6.0
        
        # Stage 1: Mirror descent (hot)
        a_cur, _ = mirror_descent_v2(a_cur, time_limit=sub, t_init=40.0, eta_init=0.3, momentum=0.6)
        record(a_cur)
        
        # Stage 2: Coordinate descent
        a_cur, _ = peak_coord_descent(a_cur, time_limit=sub)
        record(a_cur)
        
        # Stage 3: Mirror descent (cooler)
        a_cur, _ = mirror_descent_v2(a_cur, time_limit=sub, t_init=80.0, eta_init=0.15, momentum=0.5)
        record(a_cur)
        
        # Stage 4: Coordinate descent
        a_cur, _ = peak_coord_descent(a_cur, time_limit=sub)
        record(a_cur)
        
        # Stage 5: Mirror descent (cold)
        a_cur, _ = mirror_descent_v2(a_cur, time_limit=sub, t_init=150.0, eta_init=0.08, momentum=0.4)
        record(a_cur)
        
        # Stage 6: Coordinate descent (final)
        a_cur, _ = peak_coord_descent(a_cur, time_limit=sub)
        record(a_cur)
    
    print("  Deep opt complete. Best: %.6f" % best_score)
    
    # Phase 3: Intensive local search on best
    if best_sequence is not None and time.monotonic() < deadline - 200:
        print("Phase 3: Intensive local search...")
        phase3_budget = min(600.0, deadline - time.monotonic() - 100)
        
        # Multiple annealing rounds
        n_anneal = 5
        anneal_per = phase3_budget / (n_anneal + 2)
        
        for i in range(n_anneal):
            if time.monotonic() > deadline - 200:
                break
            a_start = best_sequence.copy()
            a_start = np.maximum(a_start, 0)
            a_start /= a_start.sum()
            # Add small random perturbation for diversity
            if i > 0:
                pert = rng.random(len(a_start)) * 0.05
                a_start = a_start + pert
                a_start = np.maximum(a_start, 0)
                a_start /= a_start.sum()
            
            rng_a = np.random.default_rng(seed + 7000 + i * 137)
            a_opt, _ = informed_anneal_v2(a_start, time_limit=anneal_per, rng=rng_a)
            record(a_opt)
            
            # Follow up with coordinate descent
            if time.monotonic() < deadline - 200:
                cd_time = min(30.0, deadline - time.monotonic() - 200)
                if cd_time > 5:
                    a_opt2, _ = peak_coord_descent(a_opt, time_limit=cd_time)
                    record(a_opt2)
        
        # Extended coordinate descent
        if time.monotonic() < deadline - 100:
            cd_budget = min(200.0, deadline - time.monotonic() - 50)
            a_opt, _ = peak_coord_descent(best_sequence, time_limit=cd_budget, max_peaks=8)
            record(a_opt)
        
        print("  Local search complete. Best: %.6f" % best_score)
    
    # Phase 4: Final polish
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 10
        a_best = best_sequence.copy()
        a_best = np.maximum(a_best, 0)
        a_best /= a_best.sum()
        
        # Very cold mirror descent
        half = remaining * 0.4
        if half > 10:
            a_opt, _ = mirror_descent_v2(a_best, time_limit=half, t_init=500.0, eta_init=0.03, momentum=0.3)
            record(a_opt)
        
        # Final coordinate descent
        if time.monotonic() < deadline - 10:
            rem = deadline - time.monotonic() - 10
            if rem > 10:
                a_opt, _ = peak_coord_descent(best_sequence, time_limit=rem, max_peaks=10)
                record(a_opt)
        
        # Tiny annealing
        if time.monotonic() < deadline - 10:
            rem = deadline - time.monotonic() - 10
            if rem > 5:
                rng_f = np.random.default_rng(seed + 12345)
                a_opt, _ = informed_anneal_v2(best_sequence, time_limit=rem, rng=rng_f)
                record(a_opt)
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = safe_evaluate(best_sequence)
    
    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))
    print("Total time: %.1fs" % (time.monotonic() - t_start))
    
    return [float(x) for x in best_sequence]
