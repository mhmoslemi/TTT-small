Now, one approach: think of it as a zero-sum game. We want to make `conv(a,a)` as flat as possible. This is related to the "minimum peak-to-average power ratio" (PAPR) problem.

In the communications literature, the Papoulis algorithm and its variants are used for PAPR reduction. The key idea is:
1. Identify the peaks in the frequency domain (or time domain for our case)
2. Apply a phase adjustment (in our case, amplitude adjustment since we must stay non-negative)
3. Iterate

For our non-negative constraint, a natural approach is:
1. Compute `c = conv(a, a)`
2. Identify peaks: positions where `c[k] > threshold`
3. For each peak position k, the contribution is `sum_i a[i]*a[k-i]`. To reduce this, we should decrease the products `a[i]*a[k-i]` for the pairs that contribute most.
4. A simple approach: for each i, compute the "peak contribution" of a[i] as `sum_k w[k] * a[k-i]` where w is a weight vector emphasizing peaks. Then reduce a[i] proportionally to its peak contribution and increase it elsewhere.

This is similar to the "peak-cancelling" or "selective mapping" approach.

    """Reduce peaks in autocorrelation by redistributing mass."""
    c = fft_convolve(a, a)
    s = np.sum(a)
    a = a / s  # normalize
    
    # Weight: emphasize high values
    c_normalized = c / np.max(c)
    weights = c_normalized ** 4  # strongly emphasize peaks
    
    # For each position i, compute its "peak burden"
    # peak_burden[i] = sum_k weights[k] * a[k-i] (where a[k-i] is 0 if out of range)
    # This is essentially cross-correlation of weights with a
    a_rev = a[::-1]
    burden = fft_convolve(weights, a_rev)[n-1:2*n-1]  # this gives correlation
    
    # Reduce elements with high burden, increase those with low burden
    burden_normalized = burden / np.max(burden)
    adjustment = 1.0 - alpha * burden_normalized
    adjustment = np.maximum(adjustment, 0.1)  # don't reduce too much
    
    new_a = a * adjustment
    new_a = np.maximum(new_a, 1e-10)
    new_a = new_a / np.sum(new_a)
    
    return new_a

```python
def compute_grad(a, w):
    """Compute grad_a[j] = 2 * sum_k w[k] * a[k-j] (with bounds checking)."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        # sum over k where 0<=k-j<n and 0<=k<2n-1
        # k ranges from j to min(j+n-1, 2n-2)
        k_min = j
        k_max = min(j + n - 1, 2*n - 2)
        if k_min <= k_max:
            m = np.arange(k_max - k_min + 1)
            grad[j] = np.dot(w[k_min:k_max+1], a[m])
    return 2.0 * grad
```

            base = current[idx]
            if base < 1e-12:
                continue

            best_mult = 1.0
            best_s = current_score

            # Try a few multipliers (log-spaced around 1)
            mults = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]
            for mult in mults:
                trial = current.copy()
                trial[idx] = base * mult
                s = np.sum(trial)
                if s < 1e-10:
                    continue
                trial = trial / s
                sc = score_normalized(trial)
                if sc < best_s:
                    best_s = sc
                    best_mult = mult

            if best_mult != 1.0:
                current[idx] = base * best_mult
                current = np.maximum(current, 1e-12)
                current = current / np.sum(current)
                current_score = best_s
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()

        n_passes += 1
        # Shuffle order for next pass
        order = rng.permutation(n)

    return best, best_score


def spectral_init(n, rng):
    """
    Initialize using spectral method: create a sequence whose DFT has
    approximately flat magnitude (like a CAZAC sequence), then take magnitude.
    """
    # Random phase sequence
    phases = rng.uniform(0, 2 * np.pi, n)
    s = np.exp(1j * phases)
    # IFFT gives time-domain signal
    a = np.abs(np.fft.ifft(s))
    a = np.maximum(a, 0.01)
    # Add some smoothing
    kernel = np.ones(3) / 3.0
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def chirp_init(n, rng, alpha=None):
    """Chirp-like initialization."""
    if alpha is None:
        alpha = 0.3 + 0.5 * rng.random()
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.maximum(a, 0.05)
    a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def run_lbgfs_softmax(theta, tau_schedule, deadline, max_iter=400, verbose=True):
    """L-BFGS with annealed soft-max temperature."""
    best_theta = theta.copy()
    a_best = softmax_stable(best_theta)
    best_score = score_normalized(a_best)

    for i, tau in enumerate(tau_schedule):
        if time.monotonic() > deadline - 5:
            break

        result = minimize(
            soft_max_score_grad, best_theta,
            method='L-BFGS-B', jac=True, args=(tau,),
            options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14,
                     'maxfun': max_iter * 8}
        )
        a_new = softmax_stable(result.x)
        s = score_normalized(a_new)

        if s < best_score:
            best_score = s
            best_theta = result.x.copy()
            if verbose:
                print(f"      tau={tau:.6f}: {s:.7f} *")
        elif verbose:
            print(f"      tau={tau:.6f}: {s:.7f}")

    return best_theta, best_score


def run_lbgfs_pnorm(theta, p_schedule, deadline, max_iter=300, verbose=True):
    """L-BFGS with annealed p-norm."""
    best_theta = theta.copy()
    a_best = softmax_stable(best_theta)
    best_score = score_normalized(a_best)

    for p in p_schedule:
        if time.monotonic() > deadline - 5:
            break

        result = minimize(
            pnorm_score_grad, best_theta,
            method='L-BFGS-B', jac=True, args=(p,),
            options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14,
                     'maxfun': max_iter * 8}
        )
        a_new = softmax_stable(result.x)
        s = score_normalized(a_new)

        if s < best_score:
            best_score = s
            best_theta = result.x.copy()
            if verbose:
                print(f"      p={p}: {s:.7f} *")
        elif verbose:
            print(f"      p={p}: {s:.7f}")

    return best_theta, best_score


def propose_candidate():
    """
    Main search function. Multi-phase optimization:
    1. L-BFGS with p-norm then soft-max (fine-grained annealing)
    2. Peak-reduction local search
    3. Coordinate descent polish
    4. Multiple length exploration
    """
    t_global = time.monotonic()
    BUDGET = 2400
    deadline = t_global + BUDGET - 60  # safety margin
    rng = np.random.default_rng(12345)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.array(height_sequence_1, dtype=np.float64)
        start_seq = np.clip(start_seq, 1e-10, 1000.0)
        if np.sum(start_seq) > 0:
            start_seq = start_seq / np.sum(start_seq)
            s = score_normalized(start_seq)
            print(f"Starting sequence: n={len(start_seq)}, score={s:.7f}")
            best_score = s
            best_seq = start_seq.copy()

    # === Phase 1: L-BFGS p-norm from starting sequence ===
    print(f"\n=== Phase 1: L-BFGS p-norm (starting score={best_score:.7f}) ===")
    if best_seq is not None:
        n = len(best_seq)
        theta = np.log(best_seq + 1e-30)
        p_schedule = [2, 3, 4, 6, 8, 12, 16, 24, 32, 48, 64]
        d = min(deadline, t_global + 400)
        theta, s = run_lbgfs_pnorm(theta, p_schedule, d, max_iter=300)
        if s < best_score:
            best_score = s
            best_seq = softmax_stable(theta).copy()
            print(f"  After p-norm: {best_score:.7f} ***")
        print(f"  Phase 1 done: {best_score:.7f}, time={time.monotonic()-t_global:.0f}s")

    # === Phase 2: L-BFGS soft-max with very fine annealing ===
    print(f"\n=== Phase 2: L-BFGS soft-max fine annealing (score={best_score:.7f}) ===")
    if best_seq is not None:
        n = len(best_seq)
        theta = np.log(best_seq + 1e-30)
        # Very fine tau schedule
        tau_schedule = [0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.004,
                        0.003, 0.0025, 0.002, 0.0015, 0.001, 0.0007, 0.0005,
                        0.0003, 0.0002, 0.0001, 0.00005]
        d = min(deadline, t_global + 900)
        theta, s = run_lbgfs_softmax(theta, tau_schedule, d, max_iter=500)
        if s < best_score:
            best_score = s
            best_seq = softmax_stable(theta).copy()
            print(f"  After soft-max: {best_score:.7f} ***")
        print(f"  Phase 2 done: {best_score:.7f}, time={time.monotonic()-t_global:.0f}s")

    # === Phase 3: Peak-reduction local search ===
    print(f"\n=== Phase 3: Peak-reduction local search (score={best_score:.7f}) ===")
    if best_seq is not None:
        # Multiple rounds with different strengths
        for strength in [0.08, 0.05, 0.03, 0.02, 0.01]:
            if time.monotonic() > deadline - 120:
                break
            a_new, s = peak_reduction_step(best_seq, rng, strength=strength, n_rounds=50)
            if s < best_score:
                best_score = s
                best_seq = a_new.copy()
                print(f"    strength={strength}: {best_score:.7f} ***")
            else:
                print(f"    strength={strength}: {s:.7f}")
        print(f"  Phase 3 done: {best_score:.7f}, time={time.monotonic()-t_global:.0f}s")

    # === Phase 4: Another round of soft-max (finer) ===
    print(f"\n=== Phase 4: Fine soft-max refinement (score={best_score:.7f}) ===")
    if best_seq is not None:
        theta = np.log(best_seq + 1e-30)
        tau_fine = [0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00003, 0.00001]
        d = min(deadline, t_global + 1200)
        theta, s = run_lbgfs_softmax(theta, tau_fine, d, max_iter=600)
        if s < best_score:
            best_score = s
            best_seq = softmax_stable(theta).copy()
            print(f"  After fine soft-max: {best_score:.7f} ***")
        print(f"  Phase 4 done: {best_score:.7f}, time={time.monotonic()-t_global:.0f}s")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coordinate descent (score={best_score:.7f}) ===")
    if best_seq is not None:
        cd_time = min(120.0, max(30.0, deadline - time.monotonic() - 60))
        a_new, s = coord_descent_peak(best_seq, rng, time_limit=cd_time)
        if s < best_score:
            best_score = s
            best_seq = a_new.copy()
            print(f"  After coord descent: {best_score:.7f} ***")
        else:
            print(f"  Coord descent: {s:.7f} (no improvement)")
        print(f"  Phase 5 done: time={time.monotonic()-t_global:.0f}s")

    # === Phase 6: Try different lengths with fresh starts ===
    print(f"\n=== Phase 6: Length exploration (best={best_score:.7f}) ===")
    if time.monotonic() < deadline - 180:
        extra_lengths = [200, 224, 288, 320, 384]
        for n_try in extra_lengths:
            if time.monotonic() > deadline - 90:
                break
            print(f"    Trying n={n_try}...")
            # Multiple inits
            for init_fn, init_name in [(spectral_init, 'spectral'),
                                       (lambda nn, rr: chirp_init(nn, rr, 0.5), 'chirp'),
                                       (lambda nn, rr: np.maximum(1 + 0.3*rr.standard_normal(nn), 0.01), 'random')]:
                if time.monotonic() > deadline - 60:
                    break
                a0 = init_fn(n_try, rng)
                a0 = np.maximum(a0, 1e-10)
                a0 = a0 / np.sum(a0)
                s0 = score_normalized(a0)
                if s0 > best_score + 0.1:
                    continue  # Skip if initial is too far off

                theta = np.log(a0 + 1e-30)
                # Quick p-norm optimization
                p_sched = [4, 8, 16]
                d_sub = min(deadline - 60, time.monotonic() + 60)
                theta, s_sub = run_lbgfs_pnorm(theta, p_sched, d_sub, max_iter=200, verbose=False)
                a_sub = softmax_stable(theta)
                s_sub = score_normalized(a_sub)

                if s_sub < best_score:
                    best_score = s_sub
                    best_seq = a_sub.copy()
                    print(f"      n={n_try}, {init_name}: {best_score:.7f} ***")
                # If promising, refine with soft-max
                if s_sub < best_score + 0.05 and time.monotonic() < deadline - 60:
                    tau_sub = [0.01, 0.005, 0.002, 0.001]
                    d_sub2 = min(deadline - 60, time.monotonic() + 40)
                    theta, s_sub2 = run_lbgfs_softmax(theta, tau_sub, d_sub2, max_iter=300, verbose=False)
                    a_sub2 = softmax_stable(theta)
                    s_sub2 = score_normalized(a_sub2)
                    if s_sub2 < best_score:
                        best_score = s_sub2
                        best_seq = a_sub2.copy()
                        print(f"      n={n_try}, {init_name} refined: {best_score:.7f} ***")

    # === Phase 7: Final polish on best ===
    print(f"\n=== Phase 7: Final polish (score={best_score:.7f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        # One more peak reduction + soft-max round
        a_new, s = peak_reduction_step(best_seq, rng, strength=0.02, n_rounds=30)
        if s < best_score:
            best_score = s
            best_seq = a_new.copy()

        theta = np.log(best_seq + 1e-30)
        tau_final = [0.0001, 0.00005, 0.00002]
        d = deadline
        theta, s = run_lbgfs_softmax(theta, tau_final, d, max_iter=400)
        if s < best_score:
            best_score = s
            best_seq = softmax_stable(theta).copy()
            print(f"  Final: {best_score:.7f} ***")

    # Final verification
    print(f"\n=== FINAL: n={len(best_seq)}, local_score={best_score:.7f} ===")
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_eval = evaluate_sequence(final_list)
    print(f"evaluate_sequence = {final_eval:.7f}")

    return final_list
