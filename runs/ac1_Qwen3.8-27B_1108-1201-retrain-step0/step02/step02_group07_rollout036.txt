"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based gradient computation with Nesterov momentum, mirror descent,
informed peak-shaving annealing, and multi-scale multi-start search.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_smoothed_gradient(a, conv, beta):
    """
    Compute gradient of log(sum exp(beta * C[k])) w.r.t. a.
    Returns 2*beta*(cross-correlation of p and a) where p = softmax(beta*conv).
    """
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        p = np.zeros_like(conv)
        p[np.argmax(conv)] = 1.0
    else:
        p = w / w_sum

    # grad[i] = 2*beta * sum_j p[i+j] * a[j]
    # = 2*beta * conv(a_rev, p)[n-1+i] for i=0..n-1
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_result = np.fft.irfft(A_rev * P, fft_size)
    # Extract elements at indices n-1 to 2n-2
    r = conv_result[n - 1:2 * n - 1]
    grad = 2.0 * beta * r
    return grad, p


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def nesterov_smoothed(a0, time_limit, beta_init=10.0, beta_max=500.0):
    """Nesterov-accelerated gradient descent on smoothed max over simplex."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    y = a.copy()  # Nesterov momentum point
    t_mom = 1.0
    beta = beta_init
    patience = 0
    max_pat = 30
    lr = 0.1 / beta
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Compute gradient at y
        conv_y = fast_convolve(y)
        grad, p = compute_smoothed_gradient(y, conv_y, beta)

        # Project gradient onto simplex tangent (zero mean)
        grad_centered = grad - grad.mean()

        # Nesterov update
        t_new = 1.0 + (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        momentum = (t_mom - 1.0) / t_new

        a_new = a - lr * grad_centered
        a_new = project_simplex(a_new)

        # Update momentum point
        y_new = a_new + momentum * (a_new - a)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-12:
            y_new = a_new.copy()
        else:
            y_new /= s_y

        a = a_new
        y = y_new
        t_mom = t_new

        # Evaluate
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Adaptive parameters
        if patience >= max_pat:
            patience = 0
            beta = min(beta * 1.5, beta_max)
            lr = 0.1 / beta
            # Occasional reset of momentum
            if np.random.random() < 0.3:
                y = a.copy()
                t_mom = 1.0

        # Gradually increase beta
        elapsed_frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        beta = min(beta * (1.0 + 0.001), beta_init * (1.0 + 50.0 * elapsed_frac))
        lr = 0.1 / beta

    return best_a, best_obj


def mirror_descent_smoothed(a0, time_limit, eta_init=0.5):
    """Mirror descent (multiplicative weights) on smoothed max over simplex."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    beta = 20.0
    eta = eta_init
    patience = 0
    max_pat = 40
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        grad, p = compute_smoothed_gradient(a, conv, beta)
        # Center the gradient for simplex
        grad_c = grad - grad.mean()

        # Multiplicative update
        log_update = -eta * grad_c
        log_update -= log_update.max()
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-30)
        s = a.sum()
        a /= s

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_pat:
            patience = 0
            eta *= 0.6
            beta *= 1.4
            if eta < 0.01:
                elapsed = (time.monotonic() - t0) / max(time_limit, 1.0)
                eta = eta_init * (1.0 - 0.7 * min(1.0, elapsed))
                beta = 20.0 * (1.0 + 20.0 * min(1.0, elapsed))

    return best_a, best_obj


def informed_anneal(a0, time_limit, rng, temperature_scale=0.01):
    """
    Informed simulated annealing that targets peak reduction.
    Moves mass from positions contributing to peaks to valley positions.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = temperature_scale * (1.0 - frac) ** 2 + 0.0003

        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Find peak lags (top few)
        peak_thresh = max_c * 0.8
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])

        # Pick a random peak
        k = int(rng.choice(peak_positions))

        # Find contributing pairs for this lag
        # C[k] = sum_j a[j]*a[k-j] for valid j
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            js, je = 0, 0
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue

        # Pick source position proportional to contribution
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))

        # Pick target: prefer positions with low autocorrelation contribution
        # Compute "valley_score" for each position
        target_candidates = []
        for _ in range(10):
            cand = int(rng.integers(n))
            if cand == top_j:
                continue
            # Estimate contribution of cand to peaks
            contrib = 0.0
            for pk in peak_positions[:5]:
                # cand contributes to C[pk] via a[cand]*a[pk-cand] and a[pk-cand+cand]...
                d1 = pk - cand
                d2 = cand - pk
                if 0 <= d1 < n:
                    contrib += a[d1]
                if 0 <= d2 < n:
                    contrib += a[d2]
            target_candidates.append((contrib, cand))

        if target_candidates:
            target_candidates.sort()
            # Pick from top (lowest contribution) candidates
            target_j = target_candidates[0][1]
        else:
            target_j = int(rng.integers(n))
            if target_j == top_j:
                continue

        # Move mass
        delta = a[top_j] * rng.uniform(0.02, 0.08)
        if delta < 1e-10:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue

        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj

        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta

    return best_a, best_obj


def coordinate_descent(a0, time_limit, rng):
    """Coordinate-wise descent: try moving mass from one position to another."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Find the peak
        k_peak = int(np.argmax(conv))
        max_c = conv[k_peak]

        # Find the position that contributes most to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        seg_a = a[js:je + 1]
        seg_b = a[k_peak - je:k_peak - js + 1][::-1]
        products = seg_a * seg_b
        if products.sum() < 1e-15:
            break
        top_local = int(np.argmax(products))
        top_j = js + top_local

        # Try moving a fraction to each of several random targets
        best_move_obj = cur_obj = 2.0 * n * conv.max()
        best_target = None
        best_delta = 0

        for _ in range(5):
            target = int(rng.integers(n))
            if target == top_j:
                continue
            for frac in [0.1, 0.3, 0.5]:
                delta = a[top_j] * frac
                if delta < 1e-10:
                    continue
                a[top_j] -= delta
                a[target] += delta
                new_conv = fast_convolve(a)
                new_obj = 2.0 * n * new_conv.max()
                if new_obj < best_move_obj:
                    best_move_obj = new_obj
                    best_target = target
                    best_delta = delta
                a[top_j] += delta
                a[target] -= delta

        if best_target is not None:
            a[top_j] -= best_delta
            a[best_target] += best_delta
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            # No improvement found; try random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                delta = a[i] * 0.05
                a[i] -= delta
                a[j] += delta
                if a[i] < 0:
                    a[i] += delta
                    a[j] -= delta
                else:
                    new_conv = fast_convolve(a)
                    if 2.0 * n * new_conv.max() > cur_obj * 1.05:
                        # Revert if too bad
                        a[i] += delta
                        a[j] -= delta
                    else:
                        conv = new_conv

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Gaussian bumps
    for sf in [0.05, 0.1, 0.2, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        if g.sum() > 1e-10:
            inits['gauss_%s' % sf] = g / g.sum()

    # Off-center Gaussian
    for center in [0.25, 0.75]:
        g = np.exp(-0.5 * ((x - n * center) / (0.15 * n)) ** 2)
        if g.sum() > 1e-10:
            inits['gauss_c%s' % center] = g / g.sum()

    # Sine patterns
    for freq in [1, 2, 3, 5, 8]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        s = np.maximum(s, 0.01)
        inits['sin_%d' % freq] = s / s.sum()

    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    if bi.sum() > 0:
        inits['bimodal'] = bi / bi.sum()

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        inits['rand_%s' % alpha] = rng.dirichlet(np.ones(n) * alpha)

    # Sparse comb patterns
    for period in [2, 3, 4]:
        comb = np.zeros(n)
        comb[::period] = 1.0
        if comb.sum() > 0:
            inits['comb_%d' % period] = comb / comb.sum()

    # Parent (if available)
    if parent is not None:
        p = np.maximum(parent, 0.0)
        if p.size >= n:
            pp = p[:n].copy()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pp = np.maximum(pp, 0.0)
        if pp.sum() > 1e-10:
            inits['parent'] = pp / pp.sum()

    # Parent with noise
    if parent is not None and inits.get('parent') is not None:
        noisy = inits['parent'] * (1.0 + 0.1 * rng.standard_normal(n))
        noisy = np.maximum(noisy, 0.0)
        if noisy.sum() > 1e-10:
            inits['parent_noisy'] = noisy / noisy.sum()

    return inits

        return float('inf')


def multi_phase_optimize(a0, time_budget, rng):
    """Run multiple optimization phases sequentially on a single sequence."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_obj = obj_normalized(a)
    phases = int(max(3, time_budget / 30))
    phase_time = time_budget / phases

    for phase in range(phases):
        remaining = time_budget - phase * phase_time
        if remaining < 5:
            break
        t_alloc = min(phase_time, remaining)
        method = (phase * 7 + n) % 4  # Vary method based on phase and length

        if method == 0:
            t_bet = min(50.0 + phase * 50.0, 300.0)
            a_new, _ = nesterov_smoothed(a, time_limit=t_alloc, beta_init=t_bet)
        elif method == 1:
            t_eta = max(0.05, 0.5 / (1.0 + phase * 0.5))
            a_new, _ = mirror_descent_smoothed(a, time_limit=t_alloc, eta_init=t_eta)
        elif method == 2:
            a_new, _ = informed_anneal(a, time_limit=t_alloc, rng=rng)
        else:
            a_new, _ = coordinate_descent(a, time_limit=t_alloc, rng=rng)

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-12:
            a_new /= s_new
            obj_new = obj_normalized(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Multi-scale, multi-start hybrid optimizer.
    Uses Nesterov momentum, mirror descent, informed annealing,
    and coordinate descent with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)

    parent = get_parent_sequence()

    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d, t=%.0fs" % (score, len(a), time.monotonic() - t_start))

    # Phase 1: Quick exploration across multiple lengths
    print("Phase 1: Quick exploration...")
    scan_lengths = [128, 256, 512, 1024]
    if parent is not None:
        # Add parent's length and nearby lengths
        pn = len(parent)
        for extra in [pn // 2, pn * 2]:
            if extra not in scan_lengths and 64 <= extra <= 4096:
                scan_lengths.append(extra)
        scan_lengths = sorted(set(scan_lengths))

    phase1_budget = min(400.0, (deadline - time.monotonic()) * 0.2)
    scan_results = []

    for n in scan_lengths:
        if time.monotonic() > deadline - 1500:
            break
        inits = generate_initializations(n, rng, parent)
        n_configs = len(inits)
        time_per = max(3.0, phase1_budget / max(1, len(scan_lengths) * max(n_configs, 1)))

        for name, a0 in list(inits.items())[:8]:  # Limit to 8 per length
            if time.monotonic() > deadline - 1500:
                break
            t_alloc = min(time_per, deadline - time.monotonic() - 1500)
            if t_alloc < 3.0:
                break

            # Quick optimization
            rng_sub = np.random.default_rng(seed + n * 100 + hash(name) % 1000)
            a_opt, _ = nesterov_smoothed(a0, time_limit=t_alloc, beta_init=30.0)
            score = safe_evaluate(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt))
            record(a_opt)

    print("  Phase 1 done. Best: %.6f" % best_score)
    scan_results.sort(key=lambda x: x[0])

    # Phase 2: Deep optimization of top candidates
    print("Phase 2: Deep optimization...")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 10:
            top_configs.append((n, name, a_opt))
            seen.add(key)

    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), 'best', best_sequence.copy()))

    # Remove duplicates (same n)
    seen_n = set()
    unique_configs = []
    for cfg in top_configs:
        if cfg[0] not in seen_n:
            unique_configs.append(cfg)
            seen_n.add(cfg[0])
    top_configs = unique_configs[:8]

    phase2_budget = min(1200.0, (deadline - time.monotonic()) * 0.45)
    n_deep = len(top_configs)
    time_per_deep = max(20.0, phase2_budget / max(1, n_deep))

    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 600:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 600)
        if t_alloc < 15.0:
            break
        print("  [%d/%d] n=%d, %s, budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))

        rng_sub = np.random.default_rng(seed + 10000 + idx * 7)
        a_opt, _ = multi_phase_optimize(a0, time_budget=t_alloc, rng=rng_sub)
        record(a_opt)

    print("  Phase 2 done. Best: %.6f" % best_score)

    # Phase 3: Targeted annealing on best
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("Phase 3: Targeted annealing...")
        phase3_budget = min(500.0, deadline - time.monotonic() - 80)
        n_anneal = max(2, int(phase3_budget / 120))
        anneal_per = phase3_budget / n_anneal

        for i in range(n_anneal):
            if time.monotonic() > deadline - 80:
                break
            a_start = best_sequence.copy()
            # Add small random perturbation for diversity
            if i > 0:
                pert = np.zeros_like(a_start)
                n_pert = max(5, len(a_start) // 50)
                idxs = rng.choice(len(a_start), size=n_pert, replace=False)
                pert[idxs] = rng.uniform(0.01, 0.05, size=n_pert)
                a_start = a_start + pert
                a_start = np.maximum(a_start, 0.0)
                a_start /= a_start.sum()

            rng_a = np.random.default_rng(seed + 50000 + i * 13)
            a_opt, _ = informed_anneal(a_start, time_limit=anneal_per, rng=rng_a,
                                        temperature_scale=0.008 + 0.005 * i)
            record(a_opt)

            # Also try coordinate descent
            if time.monotonic() < deadline - 80:
                rng_c = np.random.default_rng(seed + 60000 + i * 17)
                a_opt2, _ = coordinate_descent(a_start, time_limit=min(anneal_per * 0.5, deadline - time.monotonic() - 80), rng=rng_c)
                record(a_opt2)

        print("  Phase 3 done. Best: %.6f" % best_score)

    # Phase 4: Final polish with high-precision optimization
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            a_best = best_sequence.copy()
            a_best = np.maximum(a_best, 0.0)
            a_best /= a_best.sum()

            # Nesterov with very high beta (near-exact max)
            half = remaining * 0.4
            a_opt, _ = nesterov_smoothed(a_best, time_limit=half, beta_init=200.0, beta_max=1000.0)
            record(a_opt)

            if time.monotonic() < deadline - 10:
                rem2 = deadline - time.monotonic() - 10
                if rem2 > 10:
                    # Mirror descent with very small eta
                    a_opt2, _ = mirror_descent_smoothed(a_best, time_limit=rem2, eta_init=0.05)
                    record(a_opt2)

            if time.monotonic() < deadline - 10:
                rem3 = deadline - time.monotonic() - 10
                if rem3 > 5:
                    rng_f = np.random.default_rng(seed + 99999)
                    a_opt3, _ = informed_anneal(best_sequence, time_limit=rem3, rng=rng_f,
                                                 temperature_scale=0.003)
                    record(a_opt3)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_sequence /= best_sequence.sum()
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))
    print("Total time: %.1fs" % (time.monotonic() - t_start))

    return [float(x) for x in best_sequence]
