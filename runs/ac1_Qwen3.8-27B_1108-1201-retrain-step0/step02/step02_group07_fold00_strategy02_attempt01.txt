def update_sequence_with_perturbation(sequence, n, factor=0.05):
    """Update sequence with a swirl perturbation."""
    sequence_copy = sequence.copy()
    for i in range(n):
        sequence_copy[i] = max(0.0, sequence_copy[i] * factor)
    np.clip(sequence_copy, 0, 1000)
    return sequence_copy

def project_sequence_to_simplex(sequence):
    """Project sequence to simplex by redistributing mass."""
    n = len(sequence)
    return sequence / np.cbrt(np.sum(sequence))[:n]
