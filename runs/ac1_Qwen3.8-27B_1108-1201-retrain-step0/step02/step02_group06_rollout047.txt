import numpy as np

def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

        power = np.abs(A) ** 2

def random_smooth_init(n, rng, n_octaves=4):

def structured_init(n, rng, style='ramp'):
    """Create structured initializations."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if style == 'ramp':
        a = 1.0 + rng.uniform(0.1, 0.5) * t
    elif style == 'triangle':
        a = 1.0 + 0.3 * (1.0 - 2.0 * np.abs(t - 0.5))
    elif style == 'bimodal':
        a = 1.0 + 0.3 * np.exp(-4.0 * (t - 0.3) ** 2) + 0.3 * np.exp(-4.0 * (t - 0.7) ** 2)
    elif style == 'plateau':
        width = rng.uniform(0.2, 0.5)
        center = rng.uniform(0.3, 0.7)
        a = 1.0 + 0.4 * np.exp(-((t - center) / width) ** 2)
    elif style == 'sawtooth':

        a = 1.0 + 0.2 * t
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    a = a / np.sum(a)
    return a

def local_search_peak(a_init, time_limit, rng, n_top=5):

                    # Else:
                    #   derivative = a[k-j] + a[k-j] = 2*a[k-j]
                else:
                    g[j] = 2.0 * a[k]
        
        # Hmm wait, I think I'm overcomplicating this. Let me just use the FFT method.
        # The gradient of conv[k] w.r.t. a is: for each j, g[j] = a[k-j] (if 0<=k-j<n) + a[k-j] (if 0<=k-j<n and k-j != j)
        # Actually the cleanest way: conv is a quadratic form a^T Q a where Q is symmetric.
        # grad = 2 * Q * a. And (Q*a)[j] = sum_i Q[j,i] * a[i].
        # Q[j,i] = 1 if j+i = k (and both valid). So (Q*a)[j] = a[k-j] if 0<=k-j<n.
        # So grad[j] = 2 * a[k-j] if 0<=k-j<n, else 0.
        # But this double-counts when j = k-j! Because Q[j,j] = 1 (not 1/2), so (Q*a)[j] includes a[j] once.
        # And grad[j] = 2*(Q*a)[j] = 2*a[j] when 2j=k.
        # But d(a^T Q a)/d a[j] = 2*(Q*a)[j] = 2*sum_i Q[j,i]*a[i].
        # When Q is not symmetric... wait, Q IS symmetric (Q[j,i] = 1 iff j+i=k, which is symmetric in j,i).
        # So the gradient is 2*Q*a, period.
        # (Q*a)[j] = sum_i Q[j,i] a[i] = a[k-j] (if 0<=k-j<n)
        # So grad[j] = 2*a[k-j] if 0<=k-j<n, else 0.
        
        # Let me verify with a small example. a = [a0, a1], n=2.
        # conv has length 3: [a0^2, 2*a0*a1, a1^2]
        # conv[1] = 2*a0*a1. d(conv[1])/d(a0) = 2*a1, d(conv[1])/d(a1) = 2*a0.
        # Using formula: for k=1, j=0: g[0] = 2*a[1-0] = 2*a1. ✓
        # j=1: g[1] = 2*a[1-1] = 2*a0. ✓
        # conv[0] = a0^2. d/d(a0) = 2*a0, d/d(a1) = 0.
        # Using formula: for k=0, j=0: g[0] = 2*a[0-0] = 2*a0. ✓
        # j=1: g[1] = 2*a[0-1] = 2*a[-1] -> invalid, so 0. ✓
        
        # Great, so the formula is simply: g[j] = 2*a[k-j] if 0<=k-j<n, else 0.
        
        g = np.zeros(n)
        valid_j = np.arange(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
        g[valid_j] = 2.0 * a[peak_idx - valid_j]
        
        # Step
        lr = lr0 / (1.0 + step * 0.001)
        candidate = y - lr * g
        candidate = project_simplex(candidate)
        
        a_prev = a
        a = candidate
        step += 1
        
        actual_val = np.max(fft_convolve(a, a))
        if actual_val < best_val:

            best_a = a.copy()
        
        if step % 5000 == 0:
            print(f"        PGD: step={step}, best_val={best_val:.8f}, score={2*n*best_val:.6f}")
    
    return best_a, 2.0 * n * best_val
