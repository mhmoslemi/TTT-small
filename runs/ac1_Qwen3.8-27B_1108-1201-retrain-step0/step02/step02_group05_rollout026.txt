```python
import numpy as np
import time
from scipy.optimize import minimize

# FFT-based convolution
def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n+m-1]

# Score function (normalized: sum a = 1)
def score(a):
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

# Softmax
def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

# P-norm objective + gradient for L-BFGS
def pnorm_obj_grad(theta, p):
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    cp = c ** (p-1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_cp ** (1.0/p)
    score_val = 2.0 * n * pnorm
    factor = 4.0 * n * (sum_cp ** (1.0/p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n-1:2*n-1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score_val, grad_theta

# Soft-max objective + gradient
def softmax_obj_grad(theta, tau):
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score_val = 2.0 * n * soft_max_val
```

```python
"""
Multi-phase optimization for minimum peak autocorrelation non-negative sequences.
Combines L-BFGS with annealed p-norm/soft-max objectives, adaptive evolutionary
strategies, and intensive targeted local search to minimize 2n*max(conv(a,a))/(sum a)^2.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def local_score_and_conv(a):
    """Score and convolution for normalized sequence."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c), c


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_cp ** (1.0 / p)
    score_val = 2.0 * n * pnorm
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score_val, grad_theta


def soft_max_obj_grad(theta, tau):
    """Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau)."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score_val = 2.0 * n * soft_max_val
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score_val, grad_theta


def run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p, deadline):
    """Run L-BFGS with annealed p-norm. Returns best theta and score."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))

    for p in p_schedule:
        if time.monotonic() > deadline - 15:
            break
        result = minimize(
            pnorm_obj_grad, theta.copy(),
            method='L-BFGS-B', jac=True, args=(p,),
            options={'maxiter': maxiter_per_p, 'ftol': 1e-15, 'gtol': 1e-13,
                     'maxfun': maxiter_per_p * 5}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)
        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        theta = best_theta

    return best_theta, best_score


def run_lbgfs_softmax_obj(theta, tau_schedule, maxiter_per_tau, deadline):
    """Run L-BFGS with annealed soft-max temperature."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))

    for tau in tau_schedule:
        if time.monotonic() > deadline - 15:
            break
        result = minimize(
            soft_max_obj_grad, theta.copy(),
            method='L-BFGS-B', jac=True, args=(tau,),
            options={'maxiter': maxiter_per_tau, 'ftol': 1e-15, 'gtol': 1e-13,
                     'maxfun': maxiter_per_tau * 5}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)
        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        theta = best_theta

    return best_theta, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Bell-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def make_ramp_init(n, slope, rng):
    """Ramp-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + slope * t / n
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def subgradient_peak_move(a, alpha, rng):
    """
    Move in the negative subgradient direction of the max of conv(a,a).
    The subgradient of max_k conv(a,a)[k] w.r.t. a[j] is 2*a[k_peak - j]
    (for the peak at k_peak). We step against this.
    """
    n = len(a)
    c = fft_convolve(a, a)
    # Find near-peaks (within 10% of max)
    c_max = np.max(c)
    near_peak_threshold = c_max * 0.9
    near_peaks = np.where(c >= near_peak_threshold)[0]

    # Compute the subgradient (average over near-peaks for robustness)
    subgrad = np.zeros(n)
    for k in near_peaks:
        for j in range(n):
            partner = k - j
            if 0 <= partner < n:
                subgrad[j] += a[partner]

    # Normalize subgradient
    sg_max = np.max(np.abs(subgrad))
    if sg_max < 1e-15:
        return a.copy()

    subgrad_norm = subgrad / sg_max

    # Step against the subgradient (reduces the peak)
    # a_new = a - alpha * subgrad_norm, but keep non-negative
    a_new = a - alpha * subgrad_norm
    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / np.sum(a_new)
    return a_new


def peak_targeted_move(a, rng, reduction=0.05):
    """
    Targeted move: reduce elements that contribute most to the peak.
    """
    n = len(a)
    c = fft_convolve(a, a)
    peak_idx = int(np.argmax(c))

    # Compute contributions to the peak
    contribs = np.zeros(n)
    lo = max(0, peak_idx - n + 1)
    hi = min(n - 1, peak_idx)
    for i in range(lo, hi + 1):
        j = peak_idx - i
        contribs[i] = a[i] * a[j]

    # Reduce top contributors, increase others
    top_k = max(5, n // 8)
    top_idx = np.argsort(contribs)[-top_k:]
    other_idx = np.setdiff1d(np.arange(n), top_idx)

    a_new = a.copy()
    # Reduce top contributors
    reduce_factor = np.exp(rng.normal(-reduction, reduction * 0.3, size=len(top_idx)))
    a_new[top_idx] *= reduce_factor

    # Increase others proportionally
    if len(other_idx) > 0:
        n_inc = min(15, len(other_idx))
        inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
        increase_factor = np.exp(rng.normal(reduction, reduction * 0.3, size=n_inc))
        a_new[inc_idx] *= increase_factor

    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / np.sum(a_new)
    return a_new


def block_perturb_move(a, rng):
    """Perturb a random block of the sequence."""
    n = len(a)
    width = max(2, int(rng.integers(2, max(3, n // 6))))
    start = int(rng.integers(0, max(1, n - width)))
    a_new = a.copy()
    a_new[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))
    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / np.sum(a_new)
    return a_new


def pair_exchange_move(a, rng):
    """Exchange mass between two pairs of elements."""
    n = len(a)
    if n < 4:
        return a.copy()
    i1, i2 = rng.choice(n, size=2, replace=False)
    i3, i4 = rng.choice(n, size=2, replace=False)
    if len(set([i1, i2, i3, i4])) < 4:
        return a.copy()

    a_new = a.copy()
    # Transfer some mass from (i1, i2) to (i3, i4)
    amount = min(a[i1], a[i2]) * rng.uniform(0.05, 0.15)
    a_new[i1] -= amount
    a_new[i3] += amount
    amount2 = min(a_new[i2], a_new[i4] if i4 != i2 else a_new[i2]) * rng.uniform(0.05, 0.15)
    a_new[i2] -= amount2
    a_new[i4] += amount2

    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / np.sum(a_new)
    return a_new


def smooth_move(a, rng, sigma=2.0):
    """Apply Gaussian smoothing to the sequence."""
    n = len(a)
    kernel_size = int(2 * 3 * sigma) | 1  # odd
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / sigma ** 2)
    kernel = kernel / np.sum(kernel)
    a_new = np.convolve(a, kernel, mode='same')
    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / np.sum(a_new)
    return a_new


def large_jump_move(a, rng, magnitude=0.3):
    """Large random perturbation to escape local minima."""
    n = len(a)
    a_new = a * np.exp(rng.normal(0.0, magnitude, size=n))
    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / np.sum(a_new)
    return a_new


def greedy_local_search(a_init, time_limit, rng, verbose=True):
    """
    Greedy local search with multiple move types.
    Only accepts improving moves. Periodic large jumps to escape local minima.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    last_improve_t = t0
    stagnation_counter = 0
    iter_count = 0
    alpha = 0.02  # step size for subgradient move

    if verbose:
        last_print = t0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        candidate = None
        r = rng.random()

        if r < 0.25:
            candidate = subgradient_peak_move(current, alpha, rng)
        elif r < 0.45:
            candidate = peak_targeted_move(current, rng, reduction=0.04)
        elif r < 0.60:
            candidate = block_perturb_move(current, rng)
        elif r < 0.75:
            candidate = pair_exchange_move(current, rng)
        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.02, size=n))
            candidate = np.maximum(candidate, 1e-10)
            candidate = candidate / np.sum(candidate)

        new_score = local_score(candidate)

        if new_score < current_score - 1e-14:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
            last_improve_t = time.monotonic()
            stagnation_counter = 0
            # Slightly increase step size on improvement
            alpha = min(0.05, alpha * 1.01)
        else:
            stagnation_counter += 1
            alpha = max(0.005, alpha * 0.99)

        # Periodic large jump if stuck
        if stagnation_counter > 2000:
            jumped = large_jump_move(current, rng, magnitude=0.2)
            jumped_score = local_score(jumped)
            if jumped_score < current_score:
                current = jumped
                current_score = jumped_score
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
            stagnation_counter = 0

        if verbose and time.monotonic() - last_print > 10.0:
            now = time.monotonic()
            print(f"    GLS: iter={iter_count}, score={current_score:.8f}, "
                  f"best={best_score:.8f}, alpha={alpha:.4f}, t={now-t0:.0f}s")
            last_print = now

    return best, best_score


def multi_start_local_search(a_init, time_limit, rng, verbose=True):
    """
    Generate diverse perturbations of a_init and run local search on each.
    """
    n = len(a_init)
    best = a_init.copy()
    best_score = local_score(a_init)

    if verbose:
        print(f"    Multi-start: initial score={best_score:.8f}")

    # Generate 10 diverse starting points
    start_points = [a_init.copy()]
    for i in range(10):
        magnitude = 0.1 + 0.15 * i / 9.0
        perturbed = a_init * np.exp(rng.normal(0.0, magnitude, size=n))
        perturbed = np.maximum(perturbed, 1e-10)
        perturbed = perturbed / np.sum(perturbed)
        start_points.append(perturbed)

    # Allocate time to each start point
    per_start_time = time_limit / len(start_points)
    t0 = time.monotonic()

    for i, sp in enumerate(start_points):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        # First point gets more time
        if i == 0:
            t_alloc = min(per_start_time * 3, remaining * 0.4)
        else:
            t_alloc = min(per_start_time, remaining * 0.6 / max(1, len(start_points) - i))

        result, score = greedy_local_search(sp, t_alloc, rng, verbose=False)
        if score < best_score:
            best_score = score
            best = result.copy()
            if verbose:
                print(f"    Multi-start {i}: NEW BEST score={best_score:.8f}")

    return best, best_score


def interpolate_sequence(src, dst_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    x_old = np.linspace(0, 1, src_n)
    x_new = np.linspace(0, 1, dst_n)
    interp = np.interp(x_new, x_old, src)
    interp = np.maximum(interp, 1e-6)
    return interp / np.sum(interp)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimum peak autocorrelation non-negative sequence.
    Phases: L-BFGS p-norm, multi-length exploration, soft-max refinement,
    intensive greedy local search, multi-start refinement.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0  # safety margin
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        s = local_score(start_seq)
        print(f"Starting sequence: n={len(start_seq)}, score={s:.8f}")
        best_score = s
        best_seq = start_seq.copy()

    # =========================================================
    # Phase 1: L-BFGS with p-norm annealing on n=256
    # =========================================================
    print("\n=== Phase 1: L-BFGS p-norm on n=256 ===")
    n_main = 256

    # Build initializations
    inits_256 = []
    if start_seq is not None:
        inits_256.append(('provided', start_seq.copy()))

    inits_256.append(('chirp1', make_chirp_init(n_main, 0.3, rng)))
    inits_256.append(('chirp2', make_chirp_init(n_main, 0.5, rng)))
    inits_256.append(('random', make_random_init(n_main, rng, spread=0.25)))
    inits_256.append(('bell', make_bell_init(n_main, n_main * 0.2, rng)))
    inits_256.append(('ramp', make_ramp_init(n_main, 0.8, rng)))

    p_schedule = [8, 16, 32, 64, 128, 256, 512, 1024]
    phase1_deadline = min(time.monotonic() + 500, deadline - 600)

    for init_name, a0 in inits_256:
        if time.monotonic() > phase1_deadline:
            break
        theta = np.log(a0 + 1e-30)
        theta_best, s = run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p=150,
                                        deadline=phase1_deadline)
        a_result = softmax(theta_best)
        s_final = local_score(a_result)

        if s_final < best_score:
            best_score = s_final
            best_seq = a_result.copy()
            print(f"  n=256, init='{init_name}': score={s_final:.8f} *** NEW BEST ***")
        else:
            print(f"  n=256, init='{init_name}': score={s_final:.8f}")

    # =========================================================
    # Phase 2: Try other lengths
    # =========================================================
    print(f"\n=== Phase 2: Other lengths (current best={best_score:.8f}) ===")
    other_lengths = [200, 300, 384, 512]
    phase2_deadline = min(time.monotonic() + 400, deadline - 500)

    for n in other_lengths:
        if time.monotonic() > phase2_deadline:
            break

        # Interpolate best sequence to this length
        if best_seq is not None:
            a0 = interpolate_sequence(best_seq, n)
        else:
            a0 = make_chirp_init(n, 0.4, rng)

        theta = np.log(a0 + 1e-30)
        p_sched_short = [16, 32, 64, 128, 256, 512]
        theta_best, s = run_lbgfs_pnorm(theta, p_sched_short, maxiter_per_p=100,
                                        deadline=phase2_deadline)
        a_result = softmax(theta_best)
        s_final = local_score(a_result)

        if s_final < best_score:
            best_score = s_final
            best_seq = a_result.copy()
            print(f"  n={n}: score={s_final:.8f} *** NEW BEST ***")
        else:
            print(f"  n={n}: score={s_final:.8f}")

    # =========================================================
    # Phase 3: Soft-max refinement
    # =========================================================
    print(f"\n=== Phase 3: Soft-max refinement (score={best_score:.8f}) ===")
    if best_seq is not None:
        n = len(best_seq)
        theta = np.log(best_seq + 1e-30)
        tau_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        sm_deadline = min(time.monotonic() + 250, deadline - 400)
        theta_best, s = run_lbgfs_softmax_obj(theta, tau_schedule, maxiter_per_tau=200,
                                              deadline=sm_deadline)
        a_result = softmax(theta_best)
        s_final = local_score(a_result)

        if s_final < best_score:
            best_score = s_final
            best_seq = a_result.copy()
            print(f"  Soft-max improved: {best_score:.8f} *** NEW BEST ***")
        else:
            print(f"  Soft-max best: {s_final:.8f} (no improvement)")

    # =========================================================
    # Phase 4: Intensive greedy local search
    # =========================================================
    print(f"\n=== Phase 4: Intensive local search (score={best_score:.8f}) ===")
    if best_seq is not None:
        n = len(best_seq)
        gls_time = min(900.0, max(100.0, deadline - time.monotonic() - 200))
        print(f"  GLS time: {gls_time:.0f}s")

        # Run greedy local search
        polished, p_score = greedy_local_search(best_seq, gls_time, rng, verbose=True)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After GLS: {best_score:.8f} *** NEW BEST ***")
        else:
            print(f"  After GLS: {p_score:.8f} (no improvement)")

    # =========================================================
    # Phase 5: Multi-start refinement if time permits
    # =========================================================
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Multi-start (remaining time={remaining:.0f}s) ===")
    if best_seq is not None and remaining > 100:
        ms_time = min(400.0, remaining - 50)
        refined, r_score = multi_start_local_search(best_seq, ms_time, rng, verbose=True)
        if r_score < best_score:
            best_score = r_score
            best_seq = refined.copy()
            print(f"  After multi-start: {best_score:.8f} *** NEW BEST ***")
        else:
            print(f"  After multi-start: {r_score:.8f} (no improvement)")

    # =========================================================
    # Phase 6: Final quick polish
    # =========================================================
    remaining = deadline - time.monotonic()
    if best_seq is not None and remaining > 30:
        final_polish_time = min(100.0, remaining - 10)
        if final_polish_time > 10:
            print(f"\n=== Phase 6: Final polish ({final_polish_time:.0f}s) ===")
            polished, p_score = greedy_local_search(best_seq, final_polish_time, rng,
                                                    verbose=False)
            if p_score < best_score:
                best_score = p_score
                best_seq = polished.copy()
                print(f"  Final polish: {best_score:.8f} *** NEW BEST ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: n={len(final_list)}, local_score={best_score:.8f}, "
              f"evaluate_sequence={final_score:.8f} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
