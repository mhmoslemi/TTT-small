```python
import numpy as np
import time

def fft_convolve(a, a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def score(a):
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * len(a) * np.max(c) / (S * S)
```

```python
    size = 1

def compute_score(a):
    """Compute the objective: 2n * max(conv) / (sum)^2."""
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * len(a) * np.max(c) / (S*S)
```
