```python
"""
LP-guided optimization for minimizing peak autocorrelation of non-negative
step functions. Uses linear programming to find optimal update directions,
combined with gradient-based refinement and multiplicative weight updates.
Key innovation: LP solves give exact optimal directions for reducing the
convolution peak while maintaining the sum constraint.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def gs_init(n, rng, n_iter=60):
    """Gerchberg-Saxton style initialization."""
    a = np.ones(n) / n
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def chirp_init(n, rng, alpha=0.5):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def multi_tone_init(n, rng, n_tones=4):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(n_tones):
        freq = int(rng.integers(1, max(2, n // 3)))
        amp = 0.4 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def lp_step(a, M, use_sparse=True):
    """
    Solve LP: maximize sum(b) subject to conv(a,b)_k <= M for all k, b >= 0.
    Returns optimal b or None if LP fails.
    """
    n = len(a)
    # Build constraint matrix: A_ub[k, i] = a[i] if 0 <= k-i <= n-1
    # This is a banded matrix (bandwidth n)
    # For efficiency, build it as a dense matrix for small n, sparse for large n
    
    # Number of constraints: 2n-1
    n_constraints = 2 * n - 1
    
    if use_sparse and n > 200:
        from scipy.sparse import lil_matrix
        A = lil_matrix((n_constraints, n))
        for i in range(n):
            if a[i] < 1e-15:
                continue
            lo = i  # k-i >= 0 => k >= i
            hi = i + n - 1  # k-i <= n-1 => k <= i+n-1
            lo = max(0, lo)
            hi = min(n_constraints - 1, hi)
            A[lo:hi+1, i] = a[i]
        A = A.tocsc()
    else:
        A = np.zeros((n_constraints, n))
        for i in range(n):
            if a[i] < 1e-15:
                continue
            lo = max(0, i)
            hi = min(n_constraints - 1, i + n - 1)
            A[lo:hi+1, i] = a[i]
    
    # Objective: minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: A @ b <= M
    b_ub = np.full(n_constraints, M)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'time_limit': 30})
        if result.success and result.x is not None:
            return result.x
        else:
            return None
    except Exception:
        return None


def lp_optimize(a_init, time_limit, rng, verbose=False):
    """
    LP-based iterative optimization.
    At each step: find optimal direction via LP, then convex combine.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init / np.sum(a_init) * S
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    step_count = 0
    stagnation = 0
    
    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(a, a))
        b_opt = lp_step(a, M)
        
        if b_opt is None:
            # Fallback: gradient step
            a = gradient_step_fallback(a, S, rng)
            step_count += 1
            continue
        
        # Rescale b_opt to have same sum as a
        s_b = np.sum(b_opt)
        if s_b < 1e-10:
            break
        g = b_opt / s_b * S
        
        # Try different step sizes via line search
        best_t = 0.2
        best_new_score = np.inf
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            candidate = (1 - t) * a + t * g
            candidate = np.clip(candidate, 0, None)
            s_c = np.sum(candidate)
            if s_c < 1e-10:
                continue
            candidate = candidate / s_c * S
            sc = local_score(candidate)
            if sc < best_new_score:
                best_new_score = sc
                best_t = t
        
        if best_new_score < best_score - 1e-8:
            a = (1 - best_t) * a + best_t * g
            a = np.clip(a, 0, None)
            s_a = np.sum(a)
            if s_a > 1e-10:
                a = a / s_a * S
            best_a = a.copy()
            best_score = best_new_score
            stagnation = 0
        else:
            # Try smaller step or different approach
            stagnation += 1
            if stagnation > 10:
                # Perturb and restart
                a = a * np.exp(rng.normal(0, 0.05, size=n))
                a = np.clip(a, 1e-6, None)
                a = a / np.sum(a) * S
                stagnation = 0
        
        step_count += 1
        if verbose and step_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"          LP step {step_count}, score={best_score:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_score


def gradient_step_fallback(a, S, rng):
    """Simple gradient step as fallback when LP fails."""
    n = len(a)
    c = fft_convolve(a, a)
    # Soft-max approximation of max
    tau = S / n * 0.1
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)
    
    # Gradient of soft-max(conv) w.r.t. a
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, weights)
    g = (2.0 / tau) * corr[n - 1:2 * n - 1]
    
    # Normalize gradient
    g_norm = np.linalg.norm(g)
    if g_norm > 1e-10:
        g = g / g_norm
    
    # Gradient descent step
    lr = 0.1 * S / n
    a_new = a - lr * g
    a_new = np.clip(a_new, 0, None)
    s = np.sum(a_new)
    if s < 1e-10:
        return a
    return a_new / s * S


def mwf_optimize(a_init, time_limit, rng, alpha=0.3):
    """
    Multiplicative Weight Fitting: iteratively reduce elements contributing
    to convolution peaks and boost others.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init / np.sum(a_init) * S
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        c_max = np.max(c)
        
        # Compute contribution of each element to the peak
        contrib = np.zeros(n)
        # Peak lags (top few)
        n_top = min(5, len(c))
        top_lags = np.argsort(c)[-n_top:]
        
        for lag in top_lags:
            weight = c[lag] / c_max
            i_min = max(0, lag - n + 1)
            i_max = min(n - 1, lag)
            for i in range(i_min, i_max + 1):
                j = lag - i
                contrib[i] += weight * a[i] * a[j]
        
        # Multiplicative update: reduce high contributors, boost low ones
        max_contrib = np.max(contrib)
        if max_contrib < 1e-15:
            break
        
        # Smooth the contribution to avoid too-aggressive updates
        contrib_smooth = contrib / max_contrib
        update = 1.0 - alpha * contrib_smooth + alpha * 0.3
        a_new = a * update
        a_new = np.clip(a_new, 1e-8, None)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s * S
        
        sc = local_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Halve alpha temporarily
            alpha = max(0.01, alpha * 0.8)
    
    return best_a, best_score


def gradient_optimize(a_init, time_limit, rng):
    """Adam gradient descent with soft-max on convolution peak."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init / np.sum(a_init) * S
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    # Use log-parameterization
    theta = np.log(a + 1e-12)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 1.0
    step = 0
    
    tau_start = 5.0 / n
    tau_end = 0.3 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)
        
        c = fft_convolve(a, a)
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        e = np.exp(shifted)
        s_e = np.sum(e)
        weights = e / s_e
        
        # Gradient
        a_rev = a[::-1]
        corr = fft_convolve(a_rev, weights)
        g_a = (2.0 / tau) * corr[n - 1:2 * n - 1]
        
        # Normalize to sum constraint
        inner = np.dot(a, g_a)
        g = g_a - inner / S
        
        step += 1
        m1 = b1 * m1 + (1 - b1) * g
        m2 = b2 * m2 + (1 - b2) * (g ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr_t = lr / np.sqrt(1.0 + 0.3 * step)
        
        a = a - lr_t * m1_hat / (np.sqrt(m2_hat) + eps)
        a = np.clip(a, 1e-10, None)
        a = a / np.sum(a) * S
        
        if step % 200 == 0:
            sc = local_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = local_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def local_search_polish(a_init, time_limit, rng):
    """Simulated annealing polish with peak-directed mutations."""
    n = len(a_init)
    S = np.sum(a_init)
    current = a_init.copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temp = 0.003
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.4:
            # Peak-directed: reduce top contributors to peak lags
            c = fft_convolve(current, current)
            n_top = min(6, len(c))
            top_lags = np.argsort(c)[-n_top:]
            contrib = np.zeros(n)
            for lag in top_lags:
                w = c[lag] / np.max(c)
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += w * current[i] * current[j]
            n_reduce = max(2, n // 12)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.04, 0.015, size=len(bot_idx)))
        elif r < 0.6:
            # Block swap: reduce one block, boost adjacent
            width = max(1, int(rng.integers(1, max(2, n // 30))))
            start = int(rng.integers(0, max(1, n - 2 * width)))
            candidate[start:start + width] *= np.exp(rng.normal(-0.06, 0.02, size=width))
            candidate[start + width:start + 2 * width] *= np.exp(rng.normal(0.06, 0.02, size=width))
        elif r < 0.8:
            # Symmetrize
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.1 * (sym - current)
            candidate += 0.005 * rng.standard_normal(n)
        else:
            # Small noise
            candidate = current * np.exp(rng.normal(0, 0.01, size=n))
        
        candidate = np.clip(candidate, 1e-8, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current
        else:
            candidate = candidate / s * S
        
        new_score = local_score(candidate)
        delta = new_score - current_score
        
        if delta < 0:
            current = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = candidate.copy()
        else:
            accept_prob = np.exp(-delta / max(temp, 1e-12))
            if rng.random() < accept_prob:
                current = candidate
                current_score = new_score
        
        temp *= 0.9998
    
    return best, best_score


def upsample_sequence(a, new_n, rng):
    """Upsample a sequence via interpolation + small noise."""
    n = len(a)
    if new_n <= n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    LP-guided multi-phase search for optimal non-negative step function.
    Key innovations over previous approach:

    2. Multiplicative weight fitting for fast convergence
    3. Adaptive length selection
    4. Combined LP + gradient + MWF pipeline
    """

    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf

    start_seq = None

    print("\n=== Phase 1: Exploration ===")

        print(f"  LP on provided n={n_prov}, alloc={t_alloc:.0f}s")
        a_lp, s_lp = lp_optimize(start_seq, t_alloc, rng)
        print(f"    LP result: {s_lp:.6f}")

        a_mwf, s_mwf = mwf_optimize(start_seq, t_alloc, rng)
        print(f"    MWF result: {s_mwf:.6f}")
        if s_mwf < best_score:
            best_score = s_mwf

        if time.monotonic() > deadline - 120:

            print(f"    Gradient: {s_grad:.6f}")
            if s_grad < best_score:
                best_score = s_grad
                best_seq = a_grad.copy()
                print(f"    *** New best: {best_score:.6f} (n={n}) ***")
    
    # Phase 2: Focused optimization on best candidate
    print(f"\n=== Phase 2: Focused optimization (best={best_score:.6f}) ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Try LP optimization on best
        t_lp = min(300.0, (deadline - time.monotonic()) * 0.35)
        if t_lp > 30:
            print(f"  LP optimization n={n_best}, alloc={t_lp:.0f}s")
            a_lp2, s_lp2 = lp_optimize(best_seq, t_lp, rng, verbose=True)
            print(f"    LP result: {s_lp2:.6f}")
            if s_lp2 < best_score:
                best_score = s_lp2

            print(f"    MWF result: {s_mwf2:.6f}")

        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    
    # Last resort: uniform sequence
    a = np.ones(256)
    return [float(x) for x in a]
```
