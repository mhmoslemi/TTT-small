```python
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.optimize import minimize

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_convolve(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]

def build_sparse_A(a, n, lags=None):
    """Build sparse constraint matrix for LP.
    A[k, j] = a[k-j] if 0 <= k-j < n
    """
    from scipy.sparse import lil_matrix
    if lags is None:
        lags = range(2 * n - 1)
    A = lil_matrix((len(lags), n))
    for idx, k in enumerate(lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            val = a[k - j]
            if val > 1e-15:
                A[idx, j] = val
    return A.tocsr()

def solve_lp_direction(a, M, n, top_k=None, time_limit=30):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for lags, g >= 0."""
    if top_k is not None and top_k < 2 * n - 1:
        c = fft_convolve_self(a)
        active = np.argsort(c)[::-1][:top_k]
    else:
        active = np.arange(2 * n - 1)
    
    A = build_sparse_A(a, n, lags=active)
    c_obj = -np.ones(n)
    b = np.full(len(active), M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, 
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception as e:
        print("LP error: %s" % str(e))
    return None, None

def a_from_theta(theta, S):
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total

def compute_soft_weights(c, T):
    c_max = c.max()
    if c_max < 1e-15:
        return np.ones(len(c)) / len(c)
    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        return np.ones(len(c)) / len(c)
    return exp_scaled / total

def compute_grad_a(w, a):
    n = len(a)
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    end_idx = min(2 * n - 1, len(full_conv))
    grad = np.zeros(n)
    grad[:] = 2.0 * full_conv[n - 1: end_idx][:n]
    return grad

def softmax_objective_and_grad(theta, n, T, S):
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T
    w = compute_soft_weights(c, T)
    grad_a = compute_grad_a(w, a)
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    return obj, grad_theta

def run_adam_anneal(theta_init, n, S, time_budget, T_start=5.0, T_end=200.0, lr=0.05):
    theta = theta_init.copy().astype(np.float64)
    n_par = len(theta)
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    start = time.monotonic()
    
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    best_theta = theta.copy()
    
    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** progress
        
        a = a_from_theta(theta, S)
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()
            best_theta = theta.copy()
        
        obj, grad = softmax_objective_and_grad(theta, n, T, S)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr_t = lr * (0.7 ** (progress * 2.0))
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
    
    return best_theta, best_a, best_obj

def iterative_lp(a, n, S, time_budget, top_k=None, time_limit_lp=20):
    """Iterative LP-based optimization."""
```

```python
"""
Iterative LP-based optimization with gradient refinement for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses the SoTA LP direction method iteratively
with adaptive step sizes, combined with Adam gradient steps and coordinate
descent polishing. Tries multiple sequence lengths.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import lil_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def a_from_theta(theta, S):
    """Convert log-parameters to sequence: a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def compute_soft_weights(c, T):
    """Compute soft-max weights."""
    c_max = c.max()
    if c_max < 1e-15:
        return np.ones(len(c)) / len(c)
    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        return np.ones(len(c)) / len(c)
    return exp_scaled / total


def compute_grad_a(w, a):
    """Compute grad_a[m] = 2 * sum_j w[m+j] * a[j] for m=0..n-1 using FFT."""
    n = len(a)
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    end_idx = min(2 * n - 1, len(full_conv))
    grad = np.zeros(n)
    grad[:] = 2.0 * full_conv[n - 1: end_idx][:n]
    return grad


def softmax_grad(theta, n, T, S):
    """Compute gradient of soft-max objective w.r.t. theta."""
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    w = compute_soft_weights(c, T)
    grad_a = compute_grad_a(w, a)
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    return grad_theta


def build_sparse_A(a, n, lags):
    """Build sparse constraint matrix: A[idx, j] = a[k-j] if 0<=k-j<n."""
    A = lil_matrix((len(lags), n))
    for idx, k in enumerate(lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            val = a[k - j]
            if val > 1e-12:
                A[idx, j] = val
    return A.tocsr()


def solve_lp_direction(a, M, n, top_k=None, time_limit=25):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for selected lags, g >= 0."""
    if top_k is not None and top_k < 2 * n - 1:
        c = fft_convolve_self(a)
        active = np.argsort(c)[::-1][:top_k]
    else:
        active = np.arange(2 * n - 1)

    A = build_sparse_A(a, n, active)
    c_obj = -np.ones(n)
    b = np.full(len(active), M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return None, None


def run_adam_anneal(theta_init, n, S, time_budget, T_start=5.0, T_end=300.0, lr=0.05):
    """Adam optimizer with temperature annealing."""
    theta = theta_init.copy().astype(np.float64)
    n_par = len(theta)
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    best_a = a.copy()
    best_theta = theta.copy()

    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** progress

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:

    n_par = len(theta)
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    for _ in range(steps):
        t += 1
        grad = softmax_grad(theta, n, T, S)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

    a_new = a_from_theta(theta, S)
    return a_new

        a = global_a.copy()
        a = a * (S / a.sum())

        # Coordinate descent
        t0 = time.monotonic()
        a_best, obj_best = coordinate_polish(a, n, S, phase4_budget * 0.5)
        elapsed = time.monotonic() - t0
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print("Coord polish: obj=%.6f (t=%.1fs)" % (obj_best, elapsed))

        # Final high-temp gradient
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 5 and global_a is not None:
            a = global_a.copy()
            a = np.maximum(a, 1e-10)
            a = a * (S / a.sum())
            theta = np.log(a / a.sum())
            t0 = time.monotonic()
            theta_best, a_best, obj_best = run_adam_anneal(
                theta, n, S, remaining, T_start=100.0, T_end=500.0, lr=0.015
            )
            elapsed = time.monotonic() - t0
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
            print("Final gradient: obj=%.6f (t=%.1fs)" % (obj_best, elapsed))

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: n=%d, internal=%.6f, verified=%.6f" %
          (len(global_a), global_best, ver if np.isfinite(ver) else float('inf')))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.6f" % ver)

    return final_a
```
