"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LSE-annealed mirror descent with L2 pre-optimization
and coordinate descent polish. Uses multiple diverse initializations
and scales to n=1024+ for better performance.

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def lse_subgradient(conv, a, t):
    """
    Gradient of LSE_t(conv) with respect to a.
    LSE_t(conv) = t * log(sum(exp(conv/t)))
    Gradient[j] = 2 * sum_k softmax(conv/t)[k] * a[k-j]
                 = 2 * correlate(softmax(conv/t), a, 'valid')[j]
    """
    n = len(a)
    if t < 1e-10:
        k_best = int(np.argmax(conv))
        sg = np.zeros(n)
        # Subgradient: 2*a[k_best - j] for valid j
        # a[k_best - j] is nonzero when 0 <= k_best - j < n
        lo = max(0, k_best - n + 1)
        hi = min(n - 1, k_best)
        if lo <= hi:
            sg[lo:hi+1] = 2.0 * a[k_best-hi:k_best-lo+1][::-1]
        return sg
    scaled = conv / t
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # grad[j] = 2 * sum_m w[m+j] * a[m] for m=0..n-1
    sg = 2.0 * np.correlate(w, a, mode='valid')
    return sg

def l2_subgradient(conv, a):
    """
    Gradient of sum(conv^2) with respect to a.
    = 4 * sum_k conv[k] * a[k-j]
    = 4 * correlate(conv, a, 'valid')
    """
    n = len(a)
    # conv has length 2n-1, a has length n
    # correlate(conv, a, 'valid') has length (2n-1) - n + 1 = n
    sg = 4.0 * np.correlate(conv, a, mode='valid')
    return sg

def mirror_descent_lse(a0, time_limit, t_start=None, t_end=None, eta_init=0.5, seed=0):
    """
    LSE-annealed mirror descent on the simplex.
    Minimizes LSE_t(conv) with t annealed from t_start to t_end.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    max_c = conv.max()
    
    if t_start is None:
        t_start = max_c * 2.0
    if t_end is None:
        t_end = max_c * 0.01
    
    t = t_start
    eta = eta_init
    best_obj = 2.0 * n * max_c
    best_a = a.copy()
    patience = 0
    max_pat = 30
    t0 = time.monotonic()
    elapsed = 0.0
    total_steps = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient
        sg = lse_subgradient(conv, a, t)
        # Center for stability
        sg_c = sg - sg.mean()
        
        # Multiplicative update
        log_update = -eta * sg_c
        log_update -= log_update.max()  # numerical stability
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-30)  # prevent underflow
        a_sum = a.sum()
        if a_sum < 1e-15:
            a = np.ones(n) / n
            a_sum = 1.0
        a /= a_sum
        
        # Compute new conv and objective
        conv = fast_convolve(a)
        cur_max = conv.max()
        cur_obj = 2.0 * n * cur_max
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        total_steps += 1
        
        # Anneal t
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        t = t_start * (t_end / t_start) ** frac if frac > 0 else t_start
        
        # Adaptive learning rate
        if patience >= max_pat:
            patience = 0
            eta *= 0.7
            if eta < 0.01:
                # Restart with higher t for exploration
                eta = eta_init * 0.5
                t = t_start * 0.5
    
    return best_a, best_obj

def mirror_descent_l2(a0, time_limit, eta_init=0.3, seed=0):
    """Mirror descent to minimize sum(conv^2) (L2 norm proxy)."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = np.sum(conv**2)
    best_a = a.copy()
    eta = eta_init
    patience = 0
    max_pat = 20
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        sg = l2_subgradient(conv, a)
        sg_c = sg - sg.mean()
        
        log_update = -eta * sg_c
        log_update -= log_update.max()
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        if a_sum < 1e-15:
            a = np.ones(n) / n
        else:
            a /= a_sum
        
        conv = fast_convolve(a)
        cur_obj = np.sum(conv**2)
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_pat:
            patience = 0
            eta *= 0.7
            if eta < 0.01:
                break
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, rng, max_passes=100):
    """
    Greedy coordinate descent with mass transfer.
    For each coordinate, try transferring mass to other coordinates
    and keep the best improvement.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    step_frac = 0.05  # fraction of mass to transfer
    
    while time.monotonic() - t0 < time_limit:
        # Pick a random coordinate to modify
        i = int(rng.integers(n))
        if a[i] < 1e-8:
            continue
        
        # Compute sensitivity: how each conv[k] changes if we remove a[i]
        # delta_conv[k] = -2 * a[i] * a[k-i] (removing a[i])
        # Then we add it to position j: +2 * a[i] * a[k-j]
        # Net change: 2*a[i]*a[k-j] - 2*a[i]*a[k-i]
        
        delta = a[i] * step_frac
        
        # Try several candidate target positions
        best_gain = 0
        best_j = -1
        
        # Candidate positions: weighted by "low contribution to peaks"
        # For speed, try a few random positions plus some structured ones
        candidates = set()
        # Random candidates
        for _ in range(20):
            candidates.add(int(rng.integers(n)))
        # Structured: positions far from peak-contributing positions
        peak_k = int(np.argmax(conv))
        for offset in [1, 2, 3, n//4, n//2, 3*n//4]:
            j = (peak_k + offset) % (2*n-2)
            if 0 <= j < n:
                candidates.add(j)
            j2 = (peak_k - offset) % (2*n-2)
            if 0 <= j2 < n:
                candidates.add(j2)
        
        for j in candidates:
            if j == i:
                continue
            # Estimated change in peak conv value
            # conv_new[k] ≈ conv[k] + 2*delta*a[k-j] - 2*delta*a[k-i]
            # (ignoring the delta^2 term)
            # We want to minimize max(conv_new)
            # Approximate: look at the current max position
            k_peak = int(np.argmax(conv))
            # Change at peak: 2*delta*(a[k_peak-j] - a[k_peak-i])
            # a[k-j] = a[k_peak-j] if 0<=k_peak-j<n else 0
            val_j = a[k_peak - j] if 0 <= k_peak - j < n else 0.0
            val_i = a[k_peak - i] if 0 <= k_peak - i < n else 0.0
            gain = 2.0 * delta * (val_i - val_j)  # positive = improvement
            if gain > best_gain:
                best_gain = gain
                best_j = j
        
        if best_j < 0 or best_gain <= 0:
            continue
        
        # Apply the transfer
        a[i] -= delta
        a[best_j] += delta
        if a[i] < 0:
            a[i] += delta
            a[best_j] -= delta
            continue
        
        # Check if it actually improves
        conv = fast_convolve(a)
        new_obj = 2.0 * n * conv.max()
        if new_obj < cur_obj:
            cur_obj = new_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            # Revert
            a[i] += delta
            a[best_j] -= delta
    
    return best_a, best_obj

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n
    
    # Gaussians
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n*0.5) / (sf*n))**2)
        g /= g.sum()
        inits['gauss_%s' % sf] = g
    
    # Sine patterns
    for freq in [1, 2, 3, 5]:
        s = np.abs(np.sin(np.pi * freq * x / n)) + 0.1
        s /= s.sum()
        inits['sin_%d' % freq] = s
    
    # Bimodal
    w = max(1, n // 8)
    bi = np.zeros(n)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Ramp
    ramp = np.arange(1, n+1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['dirichlet_%s' % alpha] = r
    
    # "Code-like": random binary pattern, smoothed
    binary = (rng.random(n) > 0.5).astype(float)
    binary[0] = 1.0  # ensure non-zero
    # Smooth with a small kernel
    kernel = np.ones(3) / 3.0
    smoothed = np.convolve(binary, kernel, mode='same')
    smoothed = np.maximum(smoothed, 0)
    if smoothed.sum() > 0:
        smoothed /= smoothed.sum()
        inits['code_like'] = smoothed
    
    # Parent interpolation
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.ndim == 1 and p.size > 0 and p.sum() > 1e-10:
            if p.size >= n:
                pp = p[:n].copy()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
    
    return inits
