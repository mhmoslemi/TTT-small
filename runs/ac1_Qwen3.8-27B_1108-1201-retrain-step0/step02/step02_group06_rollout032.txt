```python
def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
```

```python
def optimize_lbfgs(n, theta_init, tau_schedule, time_limit):
    """Optimize using L-BFGS-B with annealed tau."""
    best_theta = theta_init.copy()
    best_actual_score = np.inf
    t0 = time.monotonic()
    
    for tau in tau_schedule:
        if time.monotonic() - t0 >= time_limit:
            break
        
        def obj(theta):
            val, _ = smooth_obj_and_grad(theta, tau)
            return val
        
        def grad(theta):
            _, g = smooth_obj_and_grad(theta, tau)
            return g
        
        result = minimize(obj, theta, jac=grad, method='L-BFGS-B',
                         options={'maxiter': 1000, 'gtol': 1e-6})
        theta = result.x
        
        # Evaluate actual score
        a = softmax(theta)
        actual = 2.0 * n * np.max(fft_convolve(a, a))
        if actual < best_actual_score:
            best_actual_score = actual
            best_theta = theta.copy()
    
    return best_theta, best_actual_score
```

```python
"""
Optimization for minimizing peak autocorrelation of non-negative step functions.
Uses L-BFGS-B on softmax parameterization with log-sum-exp smoothing and annealing.
Multiple initializations including structured (U-shape, V-shape, cosine) and random.
Polish with structured local search.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def softmax(x):
    """Stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)


def smooth_max_and_grad(a, tau):
    """
    Compute smoothed max of conv(a,a) and gradient w.r.t. a.
    Returns (value, grad_a).
    """
    n = len(a)
    conv = fft_convolve(a, a)
    
    scaled = conv / tau
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    w = exp_vals / np.sum(exp_vals)
    value = tau * (m + np.log(np.sum(exp_vals)))
    
    # grad_a[j] = 2 * sum_i w[j+i] * a[i] = 2 * (w * reverse(a))[j + n - 1]
    b = a[::-1].copy()
    conv_wb = fft_convolve(w, b)
    grad_a = 2.0 * conv_wb[n-1:2*n-1]
    
    return value, grad_a


def softmax_obj_and_grad(theta, tau):
    """Objective and gradient for softmax parameterization."""
    a = softmax(theta)
    value, grad_a = smooth_max_and_grad(a, tau)
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return value, grad_theta


def make_init_u_shape(n, rng, alpha=1.0):
    """U-shaped initialization (large at ends, small in middle)."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_init_v_shape(n, rng, alpha=1.0):
    """V-shaped initialization."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = 1.0 + alpha * np.abs(2.0 * t - 1.0)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_init_cosine(n, rng, alpha=0.5):
    """Cosine modulation."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * t / n)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_init_random(n, rng, sigma=0.5):
    """Random log-normal initialization."""
    a = np.exp(rng.normal(0, sigma, n))
    return a


def make_init_spectral(n, rng, n_iter=30):
    """Spectral initialization."""
    a = 1.0 + 0.5 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    for it in range(n_iter):
        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        corr = np.fft.irfft(power, 2 * n)[:n]
        alpha = 1.0 / (n + 1.0)
        a = a - alpha * corr
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def make_init_from_provided(provided, n, rng):
    """Resample provided sequence to length n."""
    if provided is None:
        return make_init_u_shape(n, rng)
    old = np.asarray(provided, dtype=np.float64)
    old = np.clip(old, 0.001, None)
    x_old = np.linspace(0, 1, len(old))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, old)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def optimize_lbfgs(n, theta0, tau_list, time_limit, verbose=False):
    """
    Run L-BFGS-B optimization with annealed tau.
    Returns (best_theta, best_score).
    """
    best_theta = theta0.copy()
    best_score = np.inf
    t0 = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - t0 >= time_limit:
            break
        
        def obj(theta):
            val, _ = softmax_obj_and_grad(theta, tau)
            return val
        
        def grad(theta):
            _, g = softmax_obj_and_grad(theta, tau)
            return g
        
        try:
            result = minimize(obj, theta0, jac=grad, method='L-BFGS-B',
                            options={'maxiter': 500, 'gtol': 1e-5, 'iprint': -1})
            theta0 = result.x
        except Exception:
            break
        
        # Evaluate actual score
        a = softmax(theta0)
        conv = fft_convolve(a, a)
        actual_score = 2.0 * n * np.max(conv)
        if actual_score < best_score:
            best_score = actual_score
            best_theta = theta0.copy()
        
        if verbose:
            print(f"        tau={tau:.4f}, score={actual_score:.6f}, best={best_score:.6f}, "
                  f"t={time.monotonic()-t0:.1f}s")
    
    return best_theta, best_score


def local_polish(a, time_limit, rng):
    """
    Local search polish on a sequence in the simplex.
    Uses structured moves: pair transfers, block modifications.
    """
    a = a.copy()
    n = len(a)
    s = np.sum(a)
    a = a / s  # normalize to simplex
    
    current_max = np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_max = current_max
    t0 = time.monotonic()
    patience = 0
    max_patience = 3000
    
    while time.monotonic() - t0 < time_limit and patience < max_patience:
        r = rng.random()
        
        if r < 0.3:
            # Pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.05, 0.3)
            amt = a[i] * frac
            if amt > a[i]:
                amt = a[i] * 0.5
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
            if candidate[i] < 0:
                continue
            cs = np.sum(candidate)
            candidate = candidate / cs
            new_max = np.max(fft_convolve(candidate, candidate))
            if new_max < current_max - 1e-12:
                a = candidate
                current_max = new_max
                patience = 0
            else:
                patience += 1
                
        elif r < 0.5:
            # Single element scale
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.1))
            candidate = a.copy()
            candidate[j] *= factor
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            new_max = np.max(fft_convolve(candidate, candidate))
            if new_max < current_max - 1e-12:
                a = candidate
                current_max = new_max
                patience = 0
            else:
                patience += 1
                
        elif r < 0.7:
            # Block perturbation
            width = max(2, int(n // 50))
            start = int(rng.integers(0, n - width))
            candidate = a.copy()
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.05, width))
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            new_max = np.max(fft_convolve(candidate, candidate))
            if new_max < current_max - 1e-12:
                a = candidate
                current_max = new_max
                patience = 0
            else:
                patience += 1
                
        else:
            # Peak-directed: reduce elements most contributing to peak
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Identify top contributors
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * abs(a[j])
            top_k = max(5, n // 30)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate = a.copy()
            reduction = rng.uniform(0.02, 0.1)
            candidate[top_idx] *= (1.0 - reduction)
            # Redistribute
            removed = np.sum(a[top_idx] - candidate[top_idx])
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_add = min(10, len(others))
                add_idx = rng.choice(others, n_add, replace=False)
                candidate[add_idx] += removed / n_add
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            new_max = np.max(fft_convolve(candidate, candidate))
            if new_max < current_max - 1e-12:
                a = candidate
                current_max = new_max
                patience = 0
            else:
                patience += 1
        
        if current_max < best_max:
            best_max = current_max
            best_a = a.copy()
    
    return best_a, 2.0 * n * best_max
```

```python
            polished, psc = local_polish(a, polish_time, rng)
            if psc < best_score:
                best_score = psc
                best_seq = polished.copy()
                print(f"  Polish improved to {best_score:.6f}")
    
    # Final
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]
    return [1.0] * 256
```
