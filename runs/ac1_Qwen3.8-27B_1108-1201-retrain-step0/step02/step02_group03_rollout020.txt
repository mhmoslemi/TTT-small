```python
import scipy.optimize as opt
import numpy as np
import time

def lp_direction(f, M):
    """Find direction g that maximizes sum(g) subject to f*g <= M and g >= 0."""
    n = len(f)
    # Convolution of f with g has length 2n-1
    # (f*g)[k] = sum_j f[j] * g[k-j] <= M for k = 0, ..., 2n-2
    
    # Build the constraint matrix: A @ g <= M
    # A has shape (2n-1, n)
    # A[k, j] = f[j] if k-j >= 0 and k-j < n, else 0
    # i.e., A[k, j] = f[j] if max(0, k-n+1) <= j <= min(k, n-1), else 0
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        A[k, j_min:j_max+1] = f[j_min:j_max+1]
    
    # Maximize sum(g) = c @ g where c = ones
    c = np.ones(n)
    
    # Scipy minimizes, so negate the objective
    res = opt.linprog(-c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return -res.fun  # This is sum(g)
    else:
        # Fallback: return a direction
        return None
```

```python
import numpy as np
import time
import scipy.optimize as opt

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def fft_phase_init(n, rng, n_iter=100):
    a = np.ones(n) / n
    N_fft = 1
    while N_fft < 2*n - 1:
        N_fft <<= 1
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a

def soft_max_and_weights(x, tau):
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    return x_max + tau * np.log(s), e / s

def grad_soft_max_conv(a, tau):
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_and_weights(c, tau)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = (2.0 / tau) * corr[n - 1:2 * n - 1]
    return value, g, c

def optimize_gradient(n, a_init, time_limit, rng, tau_start_mult=30.0, tau_end_mult=0.1):
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S
    # Use log parameterization: a[i] = exp(theta[i]) / sum(exp(theta)) * S
    theta = np.log(a + 1e-12)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr0 = 3.0
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)
    tau_start = tau_start_mult / n
    tau_end = tau_end_mult / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)
        
        # Compute a from theta
        a = np.exp(theta - np.max(theta))
        a = a / np.sum(a) * S
        
        value, g_a, c = grad_soft_max_conv(a, tau)
        
        # Gradient w.r.t. theta
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)
        
        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        if step % 200 == 0:
            a_check = np.exp(theta - np.max(theta))
            a_check = a_check / np.sum(a_check) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
            if step % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={elapsed:.1f}s")
    
    a_final = np.exp(best_theta - np.max(best_theta))
    a_final = a_final / np.sum(a_final) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
        a_final_local = a_final
    else:
        a_final = np.exp(best_theta - np.max(best_theta))
        a_final = a_final / np.sum(a_final) * S
    return a_final, best_score

def solve_lp_direction(f, M, f_sum=1.0):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0. Returns g."""
    n = len(f)
    # Build constraint matrix
    # (f*g)[k] = sum_j f[j] * g[k-j] <= M for k = 0..2n-2
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        A[k, j_min:j_max+1] = f[j_min:j_max+1]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    res = opt.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                      bounds=[(0, None)]*n, method='highs',
                      options={'presolve': True})
    
    if res.success:
        g = res.x
        g = g / np.sum(g) * f_sum if np.sum(g) > 1e-10 else g
        return g
    return None

def lp_refine(f, time_limit, rng, n_iters_max=100):
    """Refine using LP-based direction finding."""
    n = len(f)
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    f_norm = f / S
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t_blend = 0.1
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit and iter_count < n_iters_max:
        M = np.max(fft_convolve(f_norm, f_norm))
        g = solve_lp_direction(f_norm, M, f_sum=1.0)
        if g is None:
            break
        
        # Check if g is meaningfully different from f_norm
        if np.sum(np.abs(g - f_norm)) < 1e-8:
            # Try smaller blend or break
            t_blend *= 0.5
            if t_blend < 0.01:
                break
            iter_count += 1
            continue
        
        # Blend
        f_new = (1 - t_blend) * f_norm + t_blend * g
        f_new = np.clip(f_new, 0, None)
        s = np.sum(f_new)
        if s < 1e-10:
            break
        f_new = f_new / s
        
        M_new = np.max(fft_convolve(f_new, f_new))
        score_new = 2 * n * M_new  # with sum=1
        
        if score_new < best_score:
            best_score = score_new
            best_f = f_new * S
            t_blend = min(0.3, t_blend * 1.1)  # increase blend slightly
        else:
            t_blend *= 0.7  # decrease blend
        
        f_norm = f_new
        iter_count += 1
        
        if iter_count % 20 == 0:
            print(f"    LP iter={iter_count}, M={M_new:.6f}, best={best_score/(2*n):.6f}, t_blend={t_blend:.4f}")
    
    return best_f, best_score

def local_search_directed(a_init, time_limit, rng):
    """Local search that specifically targets peak reduction."""
    current = a_init.astype(np.float64).copy()
    n = len(current)
    S = np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.4:
            # Peak-directed: reduce elements that contribute to the top peaks
            conv = fft_convolve(current, current)
            # Find top k lags
            k_top = 5
            top_lags = np.argsort(conv)[-k_top:]
            
            # For each top lag, find the elements that contribute most
            contrib = np.zeros(n)
            for lag in top_lags:
                j_min = max(0, lag - n + 1)
                j_max = min(n - 1, lag)
                for j in range(j_min, j_max + 1):
                    other = lag - j
                    if 0 <= other <= n - 1:
                        contrib[j] += current[j] * current[other]
            
            # Reduce top contributors, increase bottom
            n_reduce = max(2, n // 15)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            
            scale = rng.uniform(0.7, 0.9)
            candidate[top_idx] *= scale
            candidate[bot_idx] /= scale
            
        elif r < 0.6:
            # Pair swap: reduce a[i], increase a[j] where i contributes to peak
            conv = fft_convolve(current, current)
            peak_lag = np.argmax(conv)
            j_min = max(0, peak_lag - n + 1)
            j_max = min(n - 1, peak_lag)
            
            # Find the pair with max product at the peak lag
            products = np.array([current[j] * current[peak_lag - j] for j in range(j_min, j_max + 1)])
            worst_j = j_min + np.argmax(products)
            
            # Reduce worst_j, increase a random other element
            i = worst_j
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            
            amount = current[i] * rng.uniform(0.05, 0.2)
            candidate[i] -= amount
            candidate[j] += amount
            
        elif r < 0.75:
            # Block perturbation
            block_len = max(1, n // 20)
            start = rng.integers(0, n - block_len)
            candidate[start:start+block_len] *= np.exp(rng.normal(0, 0.05, block_len))
            
        else:
            # Small global perturbation
            candidate *= np.exp(rng.normal(0, 0.01, n))
        
        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current
        else:
            candidate = candidate / s * S
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 50:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    current = candidate
                    current_score = new_score
        
        temp *= cooling
    
    return best, best_score
```
