def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def compute_spectrum(a):

    n = len(theta)
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()

    # Soft-max objective
    shifted = T * (c - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T

    # Weights for gradient
    w = compute_soft_weights(c, T, top_k)
    grad_a = compute_grad_a(w, a)

    # Chain rule for softmax parameterization
    # a = S * softmax(theta), so da/dtheta[j] = a[j] * (delta_{ij} - a[i]/S)
    # dL/dtheta[i] = sum_j grad_a[j] * da[j]/dtheta[i]
    #               = grad_a[i] * a[i] - (a[i]/S) * sum_j grad_a[j] * a[j]
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot

    return obj, grad_theta


def true_objective_value(a):
    """Compute the true max of self-convolution."""
    c = fft_convolve_self(a)
    return c.max()

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using Nesterov-accelerated Adam with
multi-peak softmax, adaptive temperature, periodic LP kicks, and
peak-deflation coordinate descent. Explores multiple sequence lengths.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def a_from_theta(theta, S):
    """Convert log-parameters to sequence: a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def compute_soft_weights(c, T, top_k=None):
    """Compute soft-max weights for gradient."""
    c_max = c.max()
    if c_max < 1e-15:
        return np.ones(len(c)) / len(c)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_top = c[idx]
        scaled = T * (c_top - c_top.max())
        exp_scaled = np.exp(scaled)
        total = exp_scaled.sum()
        if total < 1e-300:
            return np.ones(len(c)) / len(c)
        w = np.zeros(len(c))
        w[idx] = exp_scaled / total
        return w
    else:
        scaled = T * (c - c_max)
        scaled = scaled - scaled.max()
        exp_scaled = np.exp(scaled)
        total = exp_scaled.sum()
        if total < 1e-300:
            return np.ones(len(c)) / len(c)
        return exp_scaled / total


def compute_grad_a(w, a):
    """Compute gradient of sum_k w[k]*c[k] w.r.t. a.
    d(a*a)[k]/da[m] = 2*a[k-m] if 0<=k-m<n, else 0.
    grad_a[m] = 2 * sum_k w[k] * a[k-m] * 1_{0<=k-m<n}
    = 2 * (w * a_rev)[m+n-1] where a_rev = a[::-1]
    """
    n = len(a)
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    start = n - 1
    end = start + n
    if end > len(full_conv):
        end = len(full_conv)
    grad = np.zeros(n)
    actual_len = min(n, end - start)
    grad[:actual_len] = 2.0 * full_conv[start:start + actual_len]
    return grad


def softmax_objective_and_grad(theta, T, S, top_k=None):
    """Soft-smoothed objective and gradient w.r.t. theta."""
    n = len(theta)
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()

    shifted = T * (c - c_max)
    # Numerically stable log-sum-exp
    log_sum = np.log(np.sum(np.exp(shifted - shifted.max()))) + shifted.max()
    obj = c_max + log_sum / T

    w = compute_soft_weights(c, T, top_k)
    grad_a = compute_grad_a(w, a)

    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot

    return obj, grad_theta


def hard_subgradient(a, S, top_k=1):
    """Compute subgradient at the (top-k) peak(s) of the self-convolution."""
    c = fft_convolve_self(a)
    if top_k >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argsort(c)[::-1][:top_k]
    
    c_top = c[idx]
    total = np.sum(c_top)
    if total < 1e-30:
        return np.zeros(len(a)), c.max()
    w = np.zeros(len(c))
    w[idx] = c_top / total
    
    grad_a = compute_grad_a(w, a)
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    return grad_theta, c.max()


def run_adam_nesterov(theta_init, n, S, time_budget, lr=0.05, momentum=0.9,
                       T_start=10.0, T_end=5000.0, top_k=None, 
                       T_ramp_fraction=0.5):
    """Adam with Nesterov momentum and temperature annealing."""
    theta = theta_init.copy().astype(np.float64)
    n_par = len(theta)
    
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    start = time.monotonic()
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    best_theta = theta.copy()
    
    # Nesterov look-ahead
    y = theta.copy()
    rho = momentum
    
    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Temperature schedule: exponential ramp
        if progress < T_ramp_fraction:
            T = T_start * np.exp(3.0 * progress / T_ramp_fraction)
        else:
            T = T_start * np.exp(3.0) * ((T_end / (T_start * np.exp(3.0))) ** ((progress - T_ramp_fraction) / (1.0 - T_ramp_fraction + 1e-10)))
            T = max(T, T_start * np.exp(3.0))
        
        # Evaluate at look-ahead point
        a_y = a_from_theta(y, S)
        c_y = fft_convolve_self(a_y)
        cur_obj = c_y.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_y.copy()
            best_theta = y.copy()
        
        # Compute gradient at look-ahead point
        if T < 1e8:
            obj, grad = softmax_objective_and_grad(y, T, S, top_k)
        else:
            grad, _ = hard_subgradient(a_y, S, top_k=max(1, top_k or 4))
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        # Cosine learning rate schedule
        lr_cur = lr * 0.5 * (1.0 + np.cos(np.pi * progress))
        
        theta_new = theta - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        
        # Nesterov look-ahead
        y = theta_new + rho * (theta_new - theta)
        theta = theta_new
    
    return best_theta, best_a, best_obj


def coordinate_descent_fast(a, S, time_budget, max_sweeps=20):
    """Fast coordinate descent: directly reduce peak by adjusting elements.
    Uses analytical formula for the effect of changing a[m]."""
    a = a.copy().astype(np.float64)
    n = len(a)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        # Find current peak
        peak_idx = c.argmax()
        peak_val = c[peak_idx]
        
        # Elements that contribute to c[peak_idx] = sum_j a[j]*a[peak_idx-j]
        # For lag k = peak_idx - (n-1) (converting from array index to lag)
        k = peak_idx - (n - 1)
        
        # Elements j where both j and j+k are in [0, n-1]
        if k >= 0:
            j_range = range(0, n - k)
        else:
            j_range = range(-k, n)
        
        # For each such element, try reducing it
        improvement = 0
        for j in j_range:
            if time.monotonic() - start > time_budget:
                break
            partner = j + k
            if partner < 0 or partner >= n:
                continue
            orig = a[j]
            if orig < 1e-10:
                continue
            
            # Try reducing a[j] by factor
            # New c[peak] changes by: -delta * a[partner] (first term)
            # But other lags also change
            # Quick check: if we set a[j] = 0, does peak decrease?
            delta = a[j] * 0.5
            a[j] = orig - delta
            s = a.sum()
            if s < 0.01:
                a[j] = orig
                continue
            a_norm = a * (S / s)
            c_check = fft_convolve_self(a_norm)
            obj_check = c_check.max()
            
            if obj_check < best_obj - 1e-10:
                best_obj = obj_check
                best_a = a.copy()
                a = a_norm
                improvement += 1
            else:
                a[j] = orig
        
        if improvement == 0:
            break
    
    return best_a, best_obj


def solve_lp_direction(f, M, n, top_k_constraints=20, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0."""
    c = fft_convolve_self(f)
    if top_k_constraints < len(c):
        active_lags = np.argsort(c)[::-1][:top_k_constraints]
    else:
        active_lags = np.arange(len(c))
    
    # A[idx, j] = f[k - j] if 0 <= k-j < n, where k is the lag at active_lags[idx]
    # Note: c has length 2n-1, c[idx] corresponds to lag k = idx - (n-1)
    A = np.zeros((len(active_lags), n))
    for idx_arr, arr_idx in enumerate(active_lags):
        k = arr_idx - (n - 1)  # actual lag
        for j in range(n):
            val = j + k  # the index into f for the partner
            if 0 <= val < n:
                A[idx_arr, j] = f[val]
    
    c_obj = -np.ones(n)
    b = np.full(len(active_lags), M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                         method='highs', options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    
    # Fallback: try with fewer constraints
    if top_k_constraints > 5:
        try:
            active_lags_small = np.argsort(c)[::-1][:5]
            A_small = np.zeros((5, n))
            for idx_arr, arr_idx in enumerate(active_lags_small):
                k = arr_idx - (n - 1)
                for j in range(n):
                    val = j + k
                    if 0 <= val < n:
                        A_small[idx_arr, j] = f[val]
            b_small = np.full(5, M)
            result = linprog(c_obj, A_ub=A_small, b_ub=b_small, bounds=[(0, None)] * n,
                            method='highs', options={'time_limit': time_limit, 'presolve': True})
            if result.success and result.x is not None:
                g = result.x
                g_sum = g.sum()
                if g_sum > 1e-10:
                    return g / g_sum, g_sum
        except Exception:
            pass
    
    return None, None


def get_initialization(name, n, rng, S):

    if name == "uniform":

    elif name == "random":

        return a * (S / a.sum())

        i = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * 5 * i / n)
        a = np.maximum(a, 0.05)
        return a * (S / a.sum())
    elif name == "sine_square":
        i = np.arange(n)
        t = i / (n - 1)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-6)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function.
    
    Uses multi-stage optimization:
    1. Exploration with multiple initializations and sizes
    2. Exploitation with Nesterov-accelerated Adam + temperature annealing
    3. LP-based directional improvements
    4. Coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    global_best = np.inf
    global_a = None
    
    # Try to load the given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given sequence: n=%d, obj=%.6f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given sequence: %s" % str(e))
    
    # Phase 1: Quick exploration (20% of time)
    explore_time = 0.2 * (deadline - time.monotonic())
    
    candidate_configs = []
    if given_a is not None:
        candidate_configs.append((len(given_a), "given"))
    candidate_configs.append((4096, "uniform"))
    candidate_configs.append((4096, "random"))
    candidate_configs.append=(4096, "cosine_mod"))
    candidate_configs.append((8192, "uniform"))
    candidate_configs.append((8192, "random"))
    
    per_config_time = max(2.0, explore_time / len(candidate_configs))
    
    for n, init_name in candidate_configs:
        if time.monotonic() > deadline - 120.0:
            break
        S = np.sqrt(2.0 * float(n))
        
        if init_name == "given" and given_a is not None:
            a_init = given_a.copy()
        else:
            a_init = get_initialization(init_name, n, rng, S)
        
        a_init = np.maximum(a_init, 1e-10)
        a_init = a_init * (S / a_init.sum())
        theta_init = np.log(a_init / a_init.sum())
        
        t0 = time.monotonic()
        theta_best, a_best, obj_best = run_adam_nesterov(
            theta_init, n, S, per_config_time,
            lr=0.08, momentum=0.9, T_start=5.0, T_end=500.0,
            top_k=8, T_ramp_fraction=0.6
        )
        elapsed = time.monotonic() - t0
        
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print("Exploration: n=%d init=%s obj=%.6f (t=%.1fs)" %
                  (n, init_name, obj_best, elapsed))
    
    # Phase 2: Main exploitation (50% of time)
    exploit_time = 0.5 * (deadline - time.monotonic())
    if global_a is not None and exploit_time > 10.0:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a = global_a.copy()
        a = np.maximum(a, 1e-10)
        a = a * (S / a.sum())
        theta = np.log(a / a.sum())
        
        # Multiple rounds with increasing temperature
        temp_schedule = [
            (20, 200, 0.06, 0.35),   # (T_start, T_end, lr, fraction)
            (100, 1000, 0.04, 0.30),
            (500, 5000, 0.02, 0.20),
            (2000, 20000, 0.01, 0.15),
        ]
        
        for T_s, T_e, lr_val, frac in temp_schedule:
            if time.monotonic() > deadline - 60.0:
                break
            t_budget = exploit_time * frac
            if t_budget < 3.0:
                continue
            
            t0 = time.monotonic()
            # Use top_k that increases with temperature
            top_k_use = max(1, int(16 / np.log1p(T_s / 10)))
            
            theta, a_best, obj_best = run_adam_nesterov(
                theta, n, S, t_budget,
                lr=lr_val, momentum=0.92,
                T_start=T_s, T_end=T_e,
                top_k=top_k_use, T_ramp_fraction=0.5
            )
            elapsed = time.monotonic() - t0
            
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print("Exploit T=%.0f-%.0f: obj=%.6f (t=%.1fs)" %
                      (T_s, T_e, obj_best, elapsed))
            
            # Update theta from best
            a_best = np.maximum(a_best, 1e-10)
            a_best = a_best * (S / a_best.sum())
            theta = np.log(a_best / a_best.sum())
        
        # LP-based improvements (interleaved)
        if time.monotonic() < deadline - 30.0 and global_a is not None:
            a = global_a.copy()
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            M = c.max()
            
            for lp_round in range(3):
                if time.monotonic() > deadline - 30.0:
                    break
                g, lp_val = solve_lp_direction(a, M * 1.01, n, 
                                               top_k_constraints=15, time_limit=8.0)
                if g is not None:
                    for t_blend in [0.2, 0.4, 0.6, 0.8]:
                        a_new = (1 - t_blend) * a + t_blend * g * S
                        a_new = np.maximum(a_new, 1e-10)
                        s_new = a_new.sum()
                        if s_new > 0.01:
                            a_new = a_new * (S / s_new)
                            c_new = fft_convolve_self(a_new)
                            obj_new = c_new.max()
                            if obj_new < global_best:
                                global_best = obj_new
                                global_a = a_new.copy()
                                a = a_new.copy()
                                M = obj_new
                                print("LP blend t=%.1f: obj=%.6f" % (t_blend, obj_new))
                                break
    
    # Phase 3: Polish (remaining time)
    polish_time = deadline - time.monotonic()
    if global_a is not None and polish_time > 5.0:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        
        # Coordinate descent
        t0 = time.monotonic()
        a_cd, obj_cd = coordinate_descent_fast(
            global_a.copy(), S, polish_time * 0.4, max_sweeps=10
        )
        elapsed = time.monotonic() - t0
        if obj_cd < global_best:
            global_best = obj_cd
            global_a = a_cd.copy()
            print("Coord descent: obj=%.6f (t=%.1fs)" % (obj_cd, elapsed))
        
        # Final high-T Adam
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 5.0 and global_a is not None:
            a = global_a.copy()
            a = np.maximum(a, 1e-10)
            a = a * (S / a.sum())
            theta = np.log(a / a.sum())
            
            t0 = time.monotonic()
            theta_best, a_best, obj_best = run_adam_nesterov(
                theta, n, S, remaining,
                lr=0.01, momentum=0.93,
                T_start=5000.0, T_end=50000.0,
                top_k=4, T_ramp_fraction=0.8
            )
            elapsed = time.monotonic() - t0
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print("Final Adam: obj=%.6f (t=%.1fs)" % (obj_best, elapsed))
        
        # One more LP pass
        remaining = deadline - time.monotonic() - 3.0
        if remaining > 10.0 and global_a is not None:
            a = global_a.copy()
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            M = c.max()
            g, lp_val = solve_lp_direction(a, M * 1.005, n,
                                           top_k_constraints=10, time_limit=5.0)
            if g is not None:
                for t_blend in [0.1, 0.2, 0.3]:
                    a_new = (1 - t_blend) * a + t_blend * g * S
                    a_new = np.maximum(a_new, 1e-10)
                    s_new = a_new.sum()
                    if s_new > 0.01:
                        a_new = a_new * (S / s_new)
                        obj_new = fft_convolve_self(a_new).max()
                        if obj_new < global_best:
                            global_best = obj_new
                            global_a = a_new.copy()
                            print("Final LP t=%.1f: obj=%.6f" % (t_blend, obj_new))
    
    # Final verification
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("FINAL: obj=%.6f (verified=%.6f)" % (global_best, ver))
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.6f" % ver)
    
    return final_a
