import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s

def soft_max(x, tau):
    """Compute soft maximum and its gradient (weights)."""
    shifted = (x - np.max(x)) / tau
    exp_vals = np.exp(shifted)
    log_sum = np.sum(exp_vals)
    w = exp_vals / log_sum
    value = np.max(x) + tau * np.log(log_sum)
    return value, w

    """
    Compute objective and gradient for the soft-max smoothed peak convolution.
    Objective: soft_max(conv(a,a), tau) where a = softmax(theta)
    Returns: (value, gradient_wrt_theta)
    """
    n = len(theta)
    a = softmax(theta)
    conv = fft_convolve(a, a)
    value, w = soft_max(conv, tau)
    
    # Gradient of soft_max w.r.t. a: g[i] = 2 * cross_corr(w, a)[i]
    # cross_corr[i] = sum_j w[i+j] * a[j]

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]

def softmax(theta):
    """Numerically stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones_like(theta) / len(theta)
    return e / s

def compute_score(a):
    """Compute the actual score: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)

def obj_and_grad(theta, tau, n):
    """
    Compute soft-max smoothed objective and its gradient w.r.t. theta.
    a = softmax(theta), objective = soft_max(conv(a,a), tau)
    """
    a = softmax(theta)
    conv = fft_convolve(a, a)
    
    # Soft max
    cmax = np.max(conv)
    shifted = (conv - cmax) / tau
    exp_vals = np.exp(shifted)
    log_sum_exp = np.sum(exp_vals)
    w = exp_vals / log_sum_exp
    value = cmax + tau * np.log(log_sum_exp)
    
    # Gradient of soft_max w.r.t. a:
    # g[i] = 2 * sum_j w[i+j] * a[j]  (cross-correlation)
    size = 1
    while size < len(conv) + n - 1:
        size <<= 1
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(fw * np.conj(fa), size)[:n]
    g_a = 2.0 * corr
    
    # Gradient w.r.t. theta (on simplex, sum a = 1)
    # d(soft_max)/d(theta[j]) = a[j] * (g_a[j] - dot(a, g_a))
    mean_g = np.dot(a, g_a)
    grad_theta = a * (g_a - mean_g)
    
    return value, grad_theta

        if current_score < best_score:
            best_score = current_score
