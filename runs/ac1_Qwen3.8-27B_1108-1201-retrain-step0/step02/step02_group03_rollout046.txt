multi-resolution upscaling, diverse structured initializations, and

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def true_score(a):
    """Compute the exact evaluation score."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s ** 2)


def softmax_normalize(theta):
    """Stable softmax normalization."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def soft_max_conv_and_grad(theta, tau):
    """
    Compute soft-max of conv(a,a) where a = softmax(theta),
    and return (value, gradient_wrt_theta).
    """
    a = softmax_normalize(theta)
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    value = c_max + tau * np.log(s)
    w = e / s

    # Gradient of soft_max(conv(a,a)) w.r.t. a:
    # grad_a[k] = 2 * sum_j w[j+k] * a[j]
    # = 2 * (w * a_rev)[n-1+k] where a_rev = a[::-1]
    a_rev = a[::-1]
    w_a_rev = fft_convolve(w, a_rev)
    grad_a = 2.0 * w_a_rev[n - 1:2 * n - 1]

    # Chain rule: a = softmax(theta), so da_i/dtheta_j = a_i*(delta_ij - a_j)
    # grad_theta_j = sum_i grad_a_i * da_i/dtheta_j
    #              = sum_i grad_a_i * a_i * (delta_ij - a_j)
    #              = a_j * (grad_a_j - sum_i grad_a_i * a_i)
    inner = np.dot(a, grad_a)
    grad_theta = a * (grad_a - inner)

    return value, grad_theta

def l_bfgs_optimize(theta_init, tau_schedule, max_iter_total, time_limit, verbose=False):
    """
    Run L-BFGS-B optimization with a schedule of decreasing temperatures.
    """
    t_start = time.monotonic()
    theta = theta_init.copy()
    best_theta = theta.copy()
    best_score = true_score(softmax_normalize(theta))

    for tau in tau_schedule:
        if time.monotonic() - t_start > time_limit:
            break

        def obj_and_grad(th):
            val, g = soft_max_conv_and_grad(th, tau)
            return val, g

        result = minimize(
            obj_and_grad, theta, method='L-BFGS-B',
            jac=True, options={'maxiter': max_iter_total, 'gtol': 1e-6,
                               'maxfun': max_iter_total * 2}
        )
        theta = result.x

        a = softmax_normalize(theta)
        score = true_score(a)
        if score < best_score:
            best_score = score
            best_theta = theta.copy()

        if verbose:
            elapsed = time.monotonic() - t_start
            print(f"    tau={tau:.1e}: score={score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")

    return best_theta, best_score


def palindromic_init(n, rng, smoothness=0.3):

def upsample(a, new_n, rng, noise_level=0.02):
    """Upsample sequence via interpolation + noise."""
    n = len(a)
    if new_n <= n:
        # Downsample
        x_old = np.linspace(0, 1, new_n)
        return np.interp(x_old, np.linspace(0, 1, n), a)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += noise_level * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 1e-4, None)
    return a_interp / np.sum(a_interp)

            current_score = new_score
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            no_improve += 1
            delta = new_score - current_score
            if delta < temp * 5:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score

        temp *= cooling

        # If stuck for too long, do a restart perturbation
        if no_improve > max_no_improve:
            current = best * np.exp(rng.normal(0, 0.05, size=n))
            current = np.clip(current, 1e-6, None)
            current = current / np.sum(current) * np.sum(best)
            current_score = true_score(current)
            no_improve = 0

    return best, best_score


def gradient_local_search(a_init, time_limit, rng, lr_start=0.1, lr_end=0.001):
    """
    Gradient-based local search using the true (non-smooth) objective
    with a subgradient approximation via the peak lag.
    """
    current = a_init.astype(np.float64).copy()
    current_score = true_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        lr = lr_start * (lr_end / lr_start) ** progress

        # Compute convolution and find peak
        conv = fft_convolve(current, current)
        peak_idx = np.argmax(conv)

        # Subgradient of max at the peak: gradient of conv[peak_idx] w.r.t. a
        # conv[peak_idx] = sum_i a[i] * a[peak_idx - i]
        # d/dx_k = 2 * a[peak_idx - k] (if valid)
        grad = np.zeros(n)
        for k in range(n):
            j = peak_idx - k
            if 0 <= j < n:
                grad[k] = 2.0 * current[j]

        # Add small randomization to escape degenerate subgradients
        top_lags = np.argsort(conv)[-3:]
        for lag in top_lags:
            g_lag = np.zeros(n)
            for k in range(n):
                j = lag - k
                if 0 <= j < n:
                    g_lag[k] = 2.0 * current[j]
            grad += 0.3 * g_lag

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-10:
            grad = grad / g_norm

        # Update: reduce peak by moving mass
        candidate = current - lr * grad
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s > 1e-10:
            candidate = candidate / s * np.sum(current)

        # Add small noise
        candidate += lr * 0.1 * rng.standard_normal(n)
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s > 1e-10:
            candidate = candidate / s * np.sum(current)

        new_score = true_score(candidate)
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < 0.1:
            # Accept with some probability
            current = candidate
            current_score = new_score

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for optimal non-negative step function minimizing
    2*n*max(conv)/sum^2. Uses L-BFGS-B with annealed soft-max,
    multi-resolution upscaling, and peak-directed local search.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = true_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # ============================================================
    # Phase 1: Generate diverse initial candidates
    # ============================================================
    print("\n=== Phase 1: Initialization ===")
    candidates = []

    n_values = [256, 384, 512, 768, 1024]

    for n in n_values:
        if time.monotonic() > deadline - 120:
            break

        inits = []

        # Palindromic init
        for sm in [0.2, 0.4, 0.6]:
            a_p = palindromic_init(n, rng, smoothness=sm)
            inits.append(('pal', a_p))

        # Smooth bump
        a_b = smooth_bump_init(n, rng)
        inits.append(('bump', a_b))

        # Cosine window
        for alpha in [0.8, 1.5, 2.5]:
            a_c = cosine_window_init(n, rng, alpha=alpha)
            inits.append(('cos', a_c))

        # M-sequence inspired
        a_m = msequence_init(n, rng)
        inits.append(('mse', a_m))

        # Near-uniform with slight structure
        t = np.arange(n) / n
        a_u = 1.0 + 0.15 * np.cos(2 * np.pi * t * (1 + rng.integers(0, 4)))
        a_u += 0.02 * rng.standard_normal(n)
        a_u = np.clip(a_u, 0.05, None)
        inits.append(('structured', a_u / np.sum(a_u)))

        for name, a in inits:
            score = true_score(a)
            candidates.append((score, n, a.copy(), name))

    # Add provided sequence candidate
    if start_seq is not None:
        candidates.append((true_score(start_seq), len(start_seq), start_seq.copy(), 'provided'))

    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates")
    print(f"  Top 5: {[(f'{c[0]:.5f}', c[1], c[3]) for c in candidates[:5]]}")

    # ============================================================
    # Phase 2: L-BFGS-B optimization with annealed soft-max
    # ============================================================
    print("\n=== Phase 2: L-BFGS-B gradient optimization ===")
    n_cand = min(10, len(candidates))
    time_left = deadline - time.monotonic() - 180
    time_per_cand = max(10.0, time_left / n_cand)

    optimized = []
    for i in range(n_cand):
        if time.monotonic() > deadline - 90:
            break
        score0, n, a, name = candidates[i]

        # Scale time allocation by n
        t_alloc = time_per_cand * min(1.0, 512.0 / n)
        t_alloc = max(8.0, min(t_alloc, deadline - time.monotonic() - 40))

        print(f"  [{i+1}/{n_cand}] n={n}, init={name}, score0={score0:.6f}, t={t_alloc:.1f}s")

        # Convert to log parameters
        a_norm = a / np.sum(a)
        theta = np.log(a_norm + 1e-10)

        # Temperature schedule: start high, end low
        n_stages = 5
        tau_start = 8.0 / n
        tau_end = 0.05 / n
        tau_schedule = np.geomspace(tau_start, tau_end, n_stages)

        max_iter = max(200, int(t_alloc * 50))  # Estimate iterations

        theta_opt, s_opt = l_bfgs_optimize(
            theta, tau_schedule, max_iter, t_alloc, verbose=False
        )

        a_opt = softmax_normalize(theta_opt)
        s_final = true_score(a_opt)

        print(f"    -> score={s_final:.6f}")

        if s_final < best_score:
            best_score = s_final
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")

        optimized.append((s_final, n, a_opt))

    # ============================================================
    # Phase 3: Multi-resolution upscaling
    # ============================================================
    print("\n=== Phase 3: Multi-resolution upscaling ===")
    if optimized:
        optimized.sort(key=lambda x: x[0])

        # Try upscaling the best solution
        for up_scale in [2, 4]:
            if time.monotonic() > deadline - 90:
                break
            best_short = optimized[0]
            target_n = min(best_short[1] * up_scale, 4096)
            if target_n <= best_short[1]:
                continue

            a_up = upsample(best_short[2], target_n, rng, noise_level=0.01)
            t_alloc = min(120.0, deadline - time.monotonic() - 60)
            if t_alloc < 15:
                break

            print(f"  Upsample n={best_short[1]} -> {target_n}, score0={true_score(a_up):.6f}, t={t_alloc:.1f}s")

            theta = np.log(a_up / np.sum(a_up) + 1e-10)
            tau_schedule = np.geomspace(8.0 / target_n, 0.05 / target_n, 4)
            max_iter = max(100, int(t_alloc * 30))

            theta_opt, s_opt = l_bfgs_optimize(theta, tau_schedule, max_iter, t_alloc)
            a_opt = softmax_normalize(theta_opt)
            s_final = true_score(a_opt)
            print(f"    -> score={s_final:.6f}")

            if s_final < best_score:
                best_score = s_final
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} (n={target_n}) ***")
            optimized.append((s_final, target_n, a_opt))

    # ============================================================
    # Phase 4: Local search polish
    # ============================================================
    print(f"\n=== Phase 4: Local search polish ===")
    if best_seq is not None:
        # Split remaining time between gradient and stochastic local search
        remaining = deadline - time.monotonic() - 30
        if remaining > 20:
            t_grad_search = min(60.0, remaining * 0.3)
            t_stoch_search = min(300.0, remaining - t_grad_search - 20)

            # Gradient-based local search
            if t_grad_search > 10:
                print(f"  Gradient local search: t={t_grad_search:.1f}s")
                a_g, s_g = gradient_local_search(best_seq, t_grad_search, rng)
                print(f"    -> score={s_g:.6f}")
                if s_g < best_score:
                    best_score = s_g
                    best_seq = a_g.copy()
                    print(f"    *** New best: {best_score:.6f} ***")

            # Stochastic local search
            if t_stoch_search > 10:
                print(f"  Stochastic local search: t={t_stoch_search:.1f}s, score={best_score:.6f}")
                a_s, s_s = local_search_peak_directed(best_seq, t_stoch_search, rng)
                print(f"    -> score={s_s:.6f}")
                if s_s < best_score:
                    best_score = s_s
                    best_seq = a_s.copy()
                    print(f"    *** New best: {best_score:.6f} ***")

    # ============================================================
    # Phase 5: Final fine-tuning with very small perturbations
    # ============================================================
    remaining = deadline - time.monotonic() - 25
    if remaining > 10 and best_seq is not None:
        print(f"\n=== Phase 5: Final fine-tuning (t={remaining:.1f}s) ===")
        n = len(best_seq)

        # Very fine local search
        a_f, s_f = local_search_peak_directed(
            best_seq, min(remaining - 5, 100.0), rng,
            base_temp=0.001, cooling=0.9999
        )
        print(f"  Fine search: score={s_f:.6f}")
        if s_f < best_score:
            best_score = s_f
            best_seq = a_f.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # Final: verify and return
    print(f"\n{'='*50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
