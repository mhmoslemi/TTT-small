"""
Power-law initialized optimization for minimizing peak autocorrelation of
non-negative step functions. Uses power-law sequences (i+1)^(-alpha) as
initialization, Adam gradient descent with annealed smooth-max, and
aggressive coordinate descent polish. Sweeps over alpha and sequence length.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def smooth_max(value, tau, top_k=1):
    """Compute smoothed max and gradient for a vector."""
    if top_k == 1:
        max_v = np.max(value)
        shifted = (value - max_v) / tau
        exp_v = np.exp(shifted)
        w = exp_v / np.sum(exp_v)
        smax = max_v + tau * np.log(np.sum(exp_v))
        return smax, w
    else:
        # Top-k smooth max
        idx = np.argpartition(value, -top_k)[-top_k:]
        top_vals = value[idx]
        max_v = np.max(top_vals)
        shifted = (top_vals - max_v) / tau
        exp_v = np.exp(shifted)
        w_top = exp_v / np.sum(exp_v)
        smax = max_v + tau * np.log(np.sum(exp_v))
        w = np.zeros(len(value))
        w[idx] = w_top
        return smax, w


def grad_conv_max(a, w):
    """Gradient of sum_k w_k * conv_k(a,a) w.r.t. a."""
    n = len(a)
    # conv_k = sum_i a_i * a_{k-i}
    # d(conv_k)/d(a_j) = a_{k-j} (if 0 <= k-j < n)
    # So d(sum_k w_k conv_k)/d(a_j) = sum_k w_k * a_{k-j} (valid k)
    # = (w * a)_correlated with a
    # Specifically: result[j] = sum_{k} w[k] * a[k-j] where 0<=k-j<n
    # = sum_{m} w[j+m] * a[m] where 0<=j+m<2n-1 and 0<=m<n
    # This is a cross-correlation of w and a
    size = 2 * n
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    # We need sum_k w[k] * a[k-j] for j = 0..n-1
    # This is the cross-correlation of w and a, evaluated at lag j
    corr = np.fft.irfft(np.conj(fa) * fw, size)[:n]
    return 2.0 * corr  # factor of 2 because conv = a*a (both depend on a)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def power_law_init(n, alpha, rng, noise_level=0.0):
    """Generate power-law initialization a_i = (i+1)^(-alpha)."""
    t = np.arange(1, n + 1, dtype=np.float64)
    a = t ** (-alpha)
    if noise_level > 0:
        a += noise_level * rng.standard_normal(n)
    a = np.clip(a, 1e-10, None)
    return a / a.sum()


def optimize_adam(n, a_init, time_limit, rng, tau_start=None, tau_end=None):
    """
    Adam optimization on softmax parameterization.
    Minimizes max_conv(a) where a = softmax(theta).
    """
    if tau_start is None:
        tau_start = 5.0 / n
    if tau_end is None:
        tau_end = 0.02 / n

    theta = np.log(a_init + 1e-15)
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.5
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)

        a = softmax_normalize(theta)
        conv = fft_convolve(a, a)
        smax, w = smooth_max(conv, tau)

        # Gradient w.r.t. a
        g_a = grad_conv_max(a, w)

        # Gradient w.r.t. theta: grad_a * (diag(p) - p*p^T) where p = softmax(theta)
        p = a
        inner = np.dot(p, g_a)
        g_theta = p * (g_a - inner)

        step += 1
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * (g_theta * g_theta)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr_eff = lr * np.sqrt(1 - beta2 ** step) / (1 - beta1 ** step)
        # Actually standard Adam: lr / sqrt(step) decay
        lr_eff = lr / (1.0 + step * 0.001)
        theta = theta - lr_eff * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate periodically
        if step % 100 == 0:
            a_check = softmax_normalize(theta)
            sc = 2.0 * n * np.max(fft_convolve(a_check, a_check))
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()

        if step % 2000 == 0:
            a_check = softmax_normalize(theta)
            sc = 2.0 * n * np.max(fft_convolve(a_check, a_check))
            print(f"      Adam step={step}, score={sc:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.2e}, t={elapsed:.1f}s")

    return best_theta, best_score


def coordinate_descent(a_init, time_limit, rng):
    """
    Aggressive coordinate descent on the simplex.
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    n = len(a)

    current_score = 2.0 * n * np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    patience = 0
    max_patience = 3000
    it = 0

    while time.monotonic() - t0 < time_limit and patience < max_patience:
        it += 1
        r = rng.random()

        if r < 0.35:
            # Pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.005, 0.2)
            amt = a[i] * frac
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
            if candidate[i] < 0:
                continue
            candidate = candidate / candidate.sum()
            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        elif r < 0.55:
            # Single element scale
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.03))
            candidate = a.copy()
            candidate[j] *= factor
            candidate = np.clip(candidate, 0.0, None)
            cs = candidate.sum()
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        elif r < 0.75:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 50))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.02, size=width))
            candidate = np.clip(candidate, 0.0, None)
            cs = candidate.sum()
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        else:
            # Peak-targeted modification
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Find top contributors
            i_start = max(0, peak_idx - (n - 1))
            i_end = min(n - 1, peak_idx)
            contribs = np.zeros(n)
            for i in range(i_start, i_end + 1):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
                    if i != j:
                        contribs[j] = max(contribs[j], a[i] * a[j])
            top_k = max(5, n // 16)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate = a.copy()
            shrink = 1.0 - rng.uniform(0.01, 0.05)
            candidate[top_idx] *= shrink
            # Redistribute
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_add = min(10, len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                total_removed = np.sum(a[top_idx]) - np.sum(candidate[top_idx])
                candidate[add_idx] += total_removed / n_add
            candidate = np.clip(candidate, 0.0, None)
            cs = candidate.sum()
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"      CD iter={it}, score={best_score:.6f}, "
                  f"patience={patience}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def optimize_lp_blend(n, a_init, M_target, rng, time_limit):
    """
    LP-based blending: find g maximizing sum(g) s.t. (a*g)_k <= M_target, g >= 0.
    Blend a with g.
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    a = a / a.sum()
    conv = fft_convolve(a, a)
    M = np.max(conv)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * M

    # Use scipy to solve the LP
    try:
        from scipy.optimize import linprog
        # Maximize sum(g) subject to (a*g)_k <= M, g >= 0
        # LP: minimize -sum(g) s.t. (a*g)_k <= M for all k, g >= 0
        # (a*g)_k = sum_i a_i * g_{k-i} (with appropriate bounds)
        # This is a large LP for large n, so we limit n
        
        if n <= 512:
            # Build constraint matrix
            # (a*g)_k = sum_{i: 0<=i<n, 0<=k-i<n} a_i * g_{k-i}
            # For each k, the constraint involves g indices from max(0,k-n+1) to min(n-1,k)
            # This gives a (2n-1) x n constraint matrix
            c = -np.ones(n)  # minimize -sum(g)
            A_ub = np.zeros((2 * n - 1, n))
            b_ub = np.full(2 * n - 1, M)
            for k in range(2 * n - 1):
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                for i in range(i_start, i_end + 1):
                    j = k - i
                    A_ub[k, j] = a[i]
            
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs',
                         options={'time_limit': min(30, time_limit - (time.monotonic() - t0))})
            if res.success:
                g = res.x
                g = np.clip(g, 0, None)
                g_sum = g.sum()
                if g_sum > 1e-10:
                    g = g / g_sum
                    # Blend
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
                        blended = (1 - t) * a + t * g
                        blended = np.clip(blended, 0, None)
                        bs = blended.sum()
                        if bs < 1e-10:
                            continue
                        blended = blended / bs
                        conv_b = fft_convolve(blended, blended)
                        M_b = np.max(conv_b)
                        sc = 2.0 * n * M_b
                        if sc < best_score:
                            best_score = sc
                            best_a = blended.copy()
                            a = blended.copy()
    except Exception as e:
        print(f"      LP failed: {e}")

    return best_a, best_score

"""
Power-law initialized optimization for minimizing peak autocorrelation of
non-negative step functions. Uses a_i ~ (i+1)^(-alpha) as initialization
(sweeping alpha), Adam gradient descent with annealed smooth-max on softmax
parameterization, and aggressive coordinate descent polish. Key insight:
power-law sequences give an n-independent score of ~2*B(1-a,1-a)*(1-a)^2
which is near-optimal, so we start near the theoretical optimum and refine.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_normalized(a):
    """Score for a sequence normalized to sum=1: 2n * max(conv(a,a))."""
    n = len(a)
    if np.sum(a) < 1e-15:
        return np.inf
    a = a / np.sum(a)
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv)


def score_fast(a):
    """Full score: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def grad_conv_weighted(a, w):
    """
    Gradient of sum_k w[k] * conv_k(a,a) w.r.t. a.
    w has length 2n-1 (the convolution values).
    Uses FFT for efficiency.
    result[m] = 2 * sum_l w[m+l] * a[l] for m=0..n-1
    """
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = np.fft.rfft(a[::-1], size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fw, size)
    return 2.0 * conv[n - 1:2 * n - 1]


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def power_law_init(n, alpha, rng, noise_level=0.0):
    """Generate power-law initialization a_i = (i+1)^(-alpha), normalized to sum=1."""
    t = np.arange(1, n + 1, dtype=np.float64)
    a = t ** (-alpha)
    if noise_level > 0:
        a = a * (1.0 + noise_level * rng.standard_normal(n))
    a = np.clip(a, 1e-15, None)
    return a / a.sum()


def ramp_init(n, rng, style='linear'):
    """Create various ramp-like initializations."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if style == 'linear':
        a = 1.0 + 0.3 * t
    elif style == 'quadratic':
        a = 1.0 + 0.4 * t ** 2
    elif style == 'sqrt':
        a = 1.0 + 0.5 * np.sqrt(t + 0.01)
    elif style == 'inverted':
        a = 1.0 + 0.3 * (1.0 - t)
    else:
        a = 1.0 + 0.3 * t
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None) / np.clip(a, 0.01, None).sum()


def optimize_adam(n, a_init, time_limit, rng):
    """
    Adam optimization on softmax parameterization.
    Minimizes max_conv(a) where a = softmax(theta), sum(a)=1.
    """
    tau_start = 10.0 / n
    tau_end = 0.01 / n

    theta = np.log(a_init + 1e-15).astype(np.float64)
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 1.0
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf
    best_eval_step = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)

        a = softmax_normalize(theta)
        conv = fft_convolve(a, a)

        # Smooth max with top-k for numerical stability
        max_c = np.max(conv)
        shifted = (conv - max_c) / tau
        # Clip to avoid overflow
        shifted = np.clip(shifted, -50, 0)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)

        # Gradient w.r.t. a
        g_a = grad_conv_weighted(a, w)

        # Gradient w.r.t. theta
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner)

        step += 1
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * (g_theta * g_theta)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # Learning rate with decay
        lr = base_lr / (1.0 + 0.005 * step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate periodically (more often as we approach the end)
        eval_interval = 50 if frac < 0.5 else 20
        if step % eval_interval == 0 or step < 10:
            a_check = softmax_normalize(theta)
            sc = 2.0 * n * np.max(fft_convolve(a_check, a_check))
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()
                best_eval_step = step

        if step % 3000 == 0:
            a_check = softmax_normalize(theta)
            sc = 2.0 * n * np.max(fft_convolve(a_check, a_check))
            print(f"      Adam n={n} step={step} score={sc:.6f} best={best_score:.6f} "
                  f"tau={tau:.2e} t={elapsed:.1f}s")

    # Final evaluation
    a_final = softmax_normalize(best_theta)
    final_sc = 2.0 * n * np.max(fft_convolve(a_final, a_final))
    return a_final, final_sc


def coordinate_descent(a_init, time_limit, rng):
    """
    Aggressive coordinate descent on the simplex (sum=1, a>=0).
    Uses multiple move types: pair transfer, single scale, block, peak-targeted.
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    n = len(a)

    current_score = 2.0 * n * np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    patience = 0
    max_patience = 5000
    it = 0
    step_decay = 0.9999

    while time.monotonic() - t0 < time_limit and patience < max_patience:
        it += 1
        r = rng.random()
        scale = step_decay ** (it // 1000)

        if r < 0.35:
            # Pair transfer
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            frac = rng.uniform(0.005, 0.15) * scale
            amt = a[i] * frac
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
            if candidate[i] < 0:
                continue
            candidate = candidate / candidate.sum()
            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        elif r < 0.55:
            # Single element multiplicative update
            j = int(rng.integers(n))
            factor = np.exp(rng.normal(0, 0.04 * scale))
            candidate = a.copy()
            candidate[j] *= factor
            candidate = np.clip(candidate, 0.0, None)
            cs = candidate.sum()
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        elif r < 0.75:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 64))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.03 * scale, size=width))
            candidate = np.clip(candidate, 0.0, None)
            cs = candidate.sum()
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        else:
            # Peak-targeted: reduce elements contributing most to the peak
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            i_start = max(0, peak_idx - (n - 1))
            i_end = min(n - 1, peak_idx)
            contribs = np.zeros(n)
            for i in range(i_start, i_end + 1):
                j = peak_idx - i
                if 0 <= j < n:
                    c = a[i] * a[j]
                    contribs[i] = max(contribs[i], c)
                    if i != j:
                        contribs[j] = max(contribs[j], c)
            top_k = max(5, n // 20)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate = a.copy()
            shrink = 1.0 - rng.uniform(0.005, 0.04) * scale
            candidate[top_idx] *= shrink
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_add = min(20, len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                total_removed = np.sum(a[top_idx]) - np.sum(candidate[top_idx])
                candidate[add_idx] += total_removed / n_add
            candidate = np.clip(candidate, 0.0, None)
            cs = candidate.sum()
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

        if it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD n={n} iter={it} best={best_score:.6f} "
                  f"patience={patience} t={elapsed:.1f}s")

    return best_a, best_score


def projected_grad_descent(n, a_init, time_limit, rng):
    """
    Projected gradient descent with Nesterov momentum on the simplex.
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    a = a / a.sum()
    a_prev = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_val = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    step = 0
    tau = 5.0 / n

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        tau = (5.0 / n) * (0.02 / 5.0) ** (frac ** 0.7)

        t_prev = t_mom
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        beta = (t_prev - 1.0) / t_mom

        y = a + beta * (a - a_prev)
        # Project to simplex
        y = np.clip(y, 0, None)
        ys = y.sum()
        if ys < 1e-10:
            y = np.ones(n) / n
        else:
            y = y / ys

        conv = fft_convolve(y, y)
        max_c = np.max(conv)
        shifted = np.clip((conv - max_c) / tau, -50, 0)
        exp_v = np.exp(shifted)
        w = exp_v / exp_v.sum()

        g = grad_conv_weighted(y, w)

        # Simple step with adaptive LR
        lr = 1.0 / (4.0 * n * (1.0 + 0.1 * step))
        candidate = y - lr * g
        # Project to simplex via clipping and renormalization
        candidate = np.clip(candidate, 0, None)
        cs = candidate.sum()
        if cs < 1e-10:
            candidate = np.ones(n) / n
        else:
            candidate = candidate / cs

        a_prev = a
        a = candidate
        step += 1

        actual_val = np.max(fft_convolve(a, a))
        if actual_val < best_val:
            best_val = actual_val
            best_a = a.copy()

        if step % 2000 == 0:
            print(f"      PGD n={n} step={step} best_score={2*n*best_val:.6f} "
                  f"t={elapsed:.1f}s")

    return best_a, 2.0 * n * best_val


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: power-law init, Adam, projected gradient, coordinate descent.
    Sweeps alpha and sequence length.
    """
    t_global = time.monotonic()
    time_budget = max(10.0, float(budget_s) - 60.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.001, 1000.0)
        s0 = score_fast(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s0:.6f}")
        if s0 < best_score:
            best_score = s0
            best_seq = start_seq / start_seq.sum()

    alphas = [0.35, 0.40, 0.45, 0.48, 0.50, 0.52, 0.55, 0.60, 0.65]
    lengths = [256, 512, 1024, 2048]

    # Phase 1: Adam optimization with power-law inits
    print("\n=== Phase 1: Adam with power-law inits ===")
    phase1_time = time_budget * 0.55

    # Allocate time proportional to n^1.5 (larger n is more expensive per step
    # but also more valuable)
    weights = [n ** 0.5 for n in lengths]
    total_w = sum(weights)

    for idx_n, n in enumerate(lengths):
        if time.monotonic() > deadline - 180:
            break
        t_remaining_phase = deadline - 180 - time.monotonic()
        if t_remaining_phase < 20:
            break
        # Time budget for this n
        t_n = min(phase1_time * weights[idx_n] / total_w, t_remaining_phase)
        n_alphas = len(alphas)
        t_per_alpha = max(15.0, t_n / (n_alphas + 1))

        for alpha in alphas:
            if time.monotonic() > deadline - 180:
                break

            a0 = power_law_init(n, alpha, rng, noise_level=0.01)
            t_alloc = min(t_per_alpha, deadline - 180 - time.monotonic())
            if t_alloc < 10:
                break

            a_best, sc = optimize_adam(n, a0, t_alloc, rng)
            print(f"  n={n} alpha={alpha:.2f} score={sc:.6f} t={time.monotonic()-t_global:.1f}s")

            if sc < best_score:
                best_score = sc
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} (n={n}, alpha={alpha:.2f}) ***")

    # Phase 2: Projected gradient on best few
    print(f"\n=== Phase 2: Projected gradient polish ===")
    print(f"  Best before PGD: {best_score:.6f}")

    if best_seq is not None:
        n_best = len(best_seq)
        pgd_time = min(time_budget * 0.15, max(30.0, deadline - time.monotonic() - 120))
        a_pgd, pgd_score = projected_grad_descent(n_best, best_seq, pgd_time, rng)
        print(f"  After PGD: {pgd_score:.6f}")
        if pgd_score < best_score:
            best_score = pgd_score
            best_seq = a_pgd.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 3: Coordinate descent polish
    print(f"\n=== Phase 3: Coordinate descent polish ===")
    print(f"  Best before CD: {best_score:.6f}")

    if best_seq is not None:
        cd_time = min(time_budget * 0.20, max(30.0, deadline - time.monotonic() - 90))
        a_cd, cd_score = coordinate_descent(best_seq, cd_time, rng)
        print(f"  After CD: {cd_score:.6f}")
        if cd_score < best_score:
            best_score = cd_score
            best_seq = a_cd.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 4: Additional exploration with diverse inits
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 4: Additional exploration ===")
        remaining = deadline - 60 - time.monotonic()
        if remaining > 30 and best_seq is not None:
            n_exp = len(best_seq)
            # Try interpolating power laws with best
            for alpha in [0.38, 0.42, 0.53, 0.70]:
                if time.monotonic() > deadline - 60:
                    break
                a_pl = power_law_init(n_exp, alpha, rng, noise_level=0.0)
                for mix in [0.3, 0.5, 0.7]:
                    if time.monotonic() > deadline - 60:
                        break
                    a_mix = mix * a_pl + (1 - mix) * best_seq
                    a_mix = np.clip(a_mix, 1e-10, None)
                    a_mix = a_mix / a_mix.sum()
                    t_alloc = min(45.0, (deadline - 60 - time.monotonic()) * 0.4)
                    if t_alloc < 5:
                        break
                    a_cand, sc = optimize_adam(n_exp, a_mix, t_alloc, rng)
                    if sc < best_score:
                        best_score = sc
                        best_seq = a_cand.copy()
                        print(f"  alpha={alpha:.2f} mix={mix:.1f} score={sc:.6f} *** NEW BEST ***")

            # Try a ramp init
            for style in ['linear', 'quadratic', 'inverted']:
                if time.monotonic() > deadline - 60:
                    break
                a_ramp = ramp_init(n_exp, rng, style)
                t_alloc = min(30.0, (deadline - 60 - time.monotonic()) * 0.5)
                if t_alloc < 5:
                    break
                a_cand, sc = optimize_adam(n_exp, a_ramp, t_alloc, rng)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_cand.copy()
                    print(f"  ramp={style} score={sc:.6f} *** NEW BEST ***")

    # Final: one more CD pass on the best
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Final CD pass ===")
        t_final = max(10.0, deadline - 30 - time.monotonic())
        a_final, sc_final = coordinate_descent(best_seq, min(t_final, 120), rng)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"  After final CD: {best_score:.6f} *** NEW BEST ***")

    # Scale and return
    if best_seq is not None:
        n = len(best_seq)
        # Scale to sum = sqrt(2n) (the natural normalization)
        target_sum = np.sqrt(2.0 * n)
        final_seq = best_seq * (target_sum / best_seq.sum())
        final_list = [float(max(0.0, min(1000.0, x))) for x in final_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    a = power_law_init(256, 0.5, rng)
    final = a * np.sqrt(512.0)
    return [float(max(0.0, min(1000.0, x))) for x in final]
