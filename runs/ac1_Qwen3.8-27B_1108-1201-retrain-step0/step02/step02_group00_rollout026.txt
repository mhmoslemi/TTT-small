import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_gradient(a, c, top_k, temperature=1.0):
    """Gradient using top-k lags with softmax weighting over the k values."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices].copy()
    # Softmax weighting for smoother gradient
    tv_max = top_vals.max()
    weights = np.exp(top_vals / max(temperature, 1e-10) - tv_max / max(temperature, 1e-10))
    weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def hard_topk_gradient(a, c, top_k):
    """Gradient using exactly top-k active lags (uniform weighting)."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    w_full = np.zeros(2 * n - 1)
    w_full[top_indices] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista_adaptive(a_init, n, S, max_time, rng, beta_init=2.0, beta_max=400.0):
    """FISTA with adaptive beta, backtracking, and periodic perturbations."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = beta_init * 2.0  # Lipschitz estimate
    stall_count = 0
    last_improve_score = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.003 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stall_count = 0
        elif it % 50 == 0:
            if cur < last_improve_score - 1e-10:
                last_improve_score = cur
                stall_count = 0
            else:
                stall_count += 1

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / L
        c_y = fft_convolve_self(y)
        f_y = c_y.max()
        x_new = None
        for bt in range(5):
            x_candidate = project_simplex(y - step * grad, S)
            if bt > 0:
                c_cand = fft_convolve_self(x_candidate)
                f_cand = c_cand.max()
                if f_cand <= f_y - 1e-4 * step * gnorm * gnorm:
                    x_new = x_candidate
                    break
                step *= 0.5
            else:
                x_new = x_candidate

        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        # Adaptive L update
        c_x = fft_convolve_self(x_new)
        if c_x.max() > f_y:
            L *= 1.5
        elif it % 20 == 0:
            L = max(L * 0.95, beta * 2.0)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Periodic perturbation to escape local minima
        if stall_count > 30 and rng.random() < 0.3:
            noise_scale = 0.01 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_momentum = 1.0
            stall_count = 0

    return best_local, best_score

        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Use top-k gradient with temperature for smoothness
        temp = max(0.5, 5.0 * (1.0 - it / max_time))
        grad = topk_gradient(y, c, top_k, temperature=temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 2.0 * S / (n * (1.0 + 0.002 * it) * max(gnorm / (S / n), 0.1))
        x_new = project_simplex(y - step * grad, S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
            # Backtrack if not improving
            if c_x.max() > cur * 1.01:
                step *= 0.7

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        if it % 5000 == 0 and rng.random() < 0.15:
            noise = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_momentum = 1.0

    return best_local, best_score


def run_mirror(a_init, n, S, max_time, rng, eta_init=0.2, top_k=10):
    """Mirror descent (multiplicative weights) with top-k subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        eta = eta_init / (1.0 + 0.001 * it)
        grad = hard_topk_gradient(a, c, top_k)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        log_update = -eta * grad / gmax
        # Clip for numerical stability
        log_update = np.clip(log_update, -5.0, 5.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_new * (S / s_new)

    return best_local, best_score


def run_peak_target(a_init, n, S, max_time, rng):
    """Directly target the peak lag by reducing contributing pairs."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Find top 3 lags
        top_lags = np.argsort(c)[::-1][:3]

        # For each top lag, find the contributing pairs and reduce them
        for lag in top_lags:
            lag = int(lag)
            # Find pairs (i, lag-i) where both are in [0, n-1]
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue
            indices = np.arange(i_min, i_max + 1)
            partner = lag - indices
            products = a[indices] * a[partner]
            # Reduce the elements with largest product
            top_product_idx = np.argsort(products)[::-1][:min(50, len(indices))]
            # Create perturbation: reduce large elements, increase small ones
            perturbation = np.zeros(n)
            for ii in top_product_idx:
                idx = indices[ii]
                pidx = partner[ii]
                # Reduce the larger of the pair
                if a[idx] >= a[pidx]:
                    perturbation[idx] -= 1.0
                else:
                    perturbation[pidx] -= 1.0
                # Compensate elsewhere (add to small elements)
                small_mask = a < np.median(a)
                if np.any(small_mask):
                    candidates = np.where(small_mask)[0]
                    add_idx = candidates[rng.integers(len(candidates))]
                    perturbation[add_idx] += 1.0

            scale = 0.01 * S / n
            a_new = project_simplex(a + scale * perturbation, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur:
                a = a_new
            else:
                # Try opposite direction
                a_new2 = project_simplex(a - scale * perturbation, S)
                c_new2 = fft_convolve_self(a_new2)
                if c_new2.max() < cur:
                    a = a_new2

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_anti_palindromic_init(n, rng, S):
    """Create an anti-palindromic initialization: a[i] + a[n-1-i] = const."""
    half = n // 2
    # Random first half
    h = rng.random(half) + 0.1
    # Choose mu so that sum = S and all values non-negative
    # sum(a) = sum(h) + sum(mu - h) = n*mu - 2*sum(h[0:half]) + (a[n/2] if n odd)
    # For simplicity, use: a[i] = h[i] for i < n/2, a[n-1-i] = mu - h[i]
    # sum = 2*sum(h) ... wait, let me think more carefully.
    # a[0..half-1] = h, a[n-1..n-half] = mu - h (reversed)
    # If n is even: a = [h[0],...,h[half-1], mu-h[half-1],...,mu-h[0]]
    # sum = 2*sum(h) + half*mu ... no.
    # sum = sum(h) + sum(mu - h) = sum(h) + half*mu - sum(h) = half*mu
    # So mu = S / half
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    # a[n-1-i] = mu - h[i] for i = 0, ..., half-1
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    # Handle middle element for odd n
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum()) if a.sum() > 0 else np.ones(n) * S / n
    return a


def make_periodic_init(n, rng, S, num_freqs=3):
    """Create a periodic initialization with multiple sinusoidal components."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 20)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.4
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a * (S / a.sum())
    return a


def make_ramp_init(n, rng, S):
    """Create a ramp/power-law initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    p = rng.random() * 2.0 + 0.5
    a = tt ** p + 0.05
    # Also try reverse
    if rng.random() < 0.5:
        a = (1 - tt) ** p + 0.05
    a = a * (S / a.sum())
    return a


def make_w_shape_init(n, rng, S):
    """Create a W-shaped or M-shaped initialization (anti-symmetric pairs)."""
    tt = np.arange(n, dtype=np.float64) / n
    # W shape: |sin(2*pi*k*tt)| for k=2 or 3
    k = rng.integers(2, 5)
    a = np.abs(np.sin(2 * np.pi * k * tt)) + 0.05
    a = a * (S / a.sum())
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for optimization."""
    S = np.sqrt(2.0 * n)
    inits = []

    if parent is not None and len(parent) >= 64:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if p.sum() > 0:
            inits.append(("parent", p * (S / p.sum())))

    # Anti-palindromic (key structural insight)
    for i in range(3):
        inits.append(("anti_pal_%d" % i, make_anti_palindromic_init(n, rng, S)))

    # Periodic
    for i in range(3):
        inits.append(("periodic_%d" % i, make_periodic_init(n, rng, S)))

    # W-shape
    for i in range(2):
        inits.append(("wshape_%d" % i, make_w_shape_init(n, rng, S)))

    # Ramp
    for i in range(2):
        inits.append(("ramp_%d" % i, make_ramp_init(n, rng, S)))

    # Random
    inits.append(("random", (rng.random(n) + 0.05) * (S / (rng.random(n) + 0.05).sum() * n) * n / n))
    rand_a = rng.random(n) + 0.05
    inits.append(("random2", rand_a * (S / rand_a.sum())))

    # Sinusoidal single frequency
    tt = np.linspace(0, 1, n, endpoint=False)
    for freq in [3, 7, 13]:
        sin_a = 1.0 + 0.35 * np.sin(2 * np.pi * freq * tt)
        sin_a = np.maximum(sin_a, 0.05)
        inits.append(("sin_%d" % freq, sin_a * (S / sin_a.sum())))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 10.0)

    global_best = np.inf
    global_a = None

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    primary_n = 8192
    if parent is not None and len(parent) >= 64:
        primary_n = len(parent)

    print("Primary length: %d, parent available: %s" % (primary_n, parent is not None))

    # === Phase 1: Exploratory optimization ===
    inits = generate_initializations(primary_n, rng, parent)
    n_inits = len(inits)
    usable_time = max(10.0, deadline - time.monotonic())
    per_init_time = usable_time * 0.70 / n_inits

    print("=== Phase 1: %d initializations, %.1f sec each ===" % (n_inits, per_init_time))

    for name, arr in inits:
        if time.monotonic() > deadline - 120:
            break
        n = len(arr)
        if n < 64:
            continue
        S = np.sqrt(2.0 * n)
        t0 = time.monotonic()

        # FISTA with adaptive step
        seg1 = per_init_time * 0.50
        a_f, s_f = run_fista_adaptive(arr, n, S, seg1, rng,
                                       beta_init=2.0, beta_max=300.0)
        a_f = safe_normalize(a_f, n)
        c_f = fft_convolve_self(a_f)
        score_f = c_f.max()
        if score_f < global_best:
            global_best = score_f
            global_a = a_f.copy()
            print("  FISTA [%s]: %.6f (elapsed %.0fs)" % (name, global_best, time.monotonic() - t0))

        # Active-set method
        rem = per_init_time - (time.monotonic() - t0)
        if rem > 2:
            a_as, s_as = run_active_set(a_f, n, S, rem, rng,
                                         top_k_init=3, top_k_max=100)
            a_as = safe_normalize(a_as, n)
            c_as = fft_convolve_self(a_as)
            if c_as.max() < global_best:
                global_best = c_as.max()
                global_a = a_as.copy()
                print("  ActiveSet [%s]: %.6f" % (name, global_best))

    print("=== After Phase 1: best=%.6f, n=%d ===" % (global_best, len(global_a) if global_a is not None else 0))

    # === Phase 2: Try a longer sequence for more attack surface ===
    if time.monotonic() < deadline - 180:
        n_long = primary_n * 2 if primary_n <= 16384 else primary_n
        if n_long > 32768:
            n_long = 16384
        if n_long != primary_n:
            t0 = time.monotonic()
            print("=== Phase 2: Trying n=%d ===" % n_long)
            S_long = np.sqrt(2.0 * n_long)
            # Interpolate parent or use anti-palindromic
            if parent is not None and len(parent) >= 64:
                # Upsample parent
                from numpy import interp
                tt_old = np.linspace(0, 1, len(parent))
                tt_new = np.linspace(0, 1, n_long)
                a_long_init = interp(tt_new, tt_old, parent)
                a_long_init = np.maximum(a_long_init, 0.0)
            else:
                a_long_init = make_anti_palindromic_init(n_long, rng, S_long)

            seg_long = min(150.0, max(30.0, (deadline - time.monotonic()) * 0.2))
            a_l, s_l = run_fista_adaptive(a_long_init, n_long, S_long, seg_long, rng,
                                           beta_init=2.0, beta_max=200.0)
            a_l = safe_normalize(a_l, n_long)
            c_l = fft_convolve_self(a_l)
            score_l = c_l.max()
            # Compare objective values (not just max_b, since n differs)
            obj_l = 2.0 * n_long * score_l / S_long ** 2
            obj_best = 2.0 * len(global_a) * global_best / (np.sqrt(2.0 * len(global_a))) ** 2 if global_a is not None else np.inf

            if obj_l < obj_best:
                global_best = score_l
                global_a = a_l.copy()
                print("  Long seq n=%d: obj=%.6f" % (n_long, obj_l))
            print("  Phase 2 done in %.1fs" % (time.monotonic() - t0))

    # === Phase 3: Intensive polishing ===
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()

        print("=== Phase 3: Polishing (remaining %.0f sec) ===" % remaining)

        # 3a: High-beta FISTA polish
        seg3a = max(5.0, remaining * 0.20)
        a_p1, s_p1 = run_fista_adaptive(global_a, n, S, seg3a, rng,
                                         beta_init=50.0, beta_max=500.0)
        a_p1 = safe_normalize(a_p1, n)
        c_p1 = fft_convolve_self(a_p1)
        if c_p1.max() < global_best:
            global_best = c_p1.max()
            global_a = a_p1.copy()
            print("  Polish FISTA hi-beta: %.6f" % global_best)

        # 3b: Active-set with high k
        remaining2 = deadline - time.monotonic()
        seg3b = max(5.0, remaining2 * 0.20)
        a_p2, s_p2 = run_active_set(global_a, n, S, seg3b, rng,
                                     top_k_init=10, top_k_max=200)
        a_p2 = safe_normalize(a_p2, n)
        c_p2 = fft_convolve_self(a_p2)
        if c_p2.max() < global_best:
            global_best = c_p2.max()
            global_a = a_p2.copy()
            print("  Polish ActiveSet hi-k: %.6f" % global_best)

        # 3c: Mirror descent polish
        remaining3 = deadline - time.monotonic()
        seg3c = max(5.0, remaining3 * 0.15)
        a_p3, s_p3 = run_mirror(global_a, n, S, seg3c, rng,
                                 eta_init=0.05, top_k=20)
        a_p3 = safe_normalize(a_p3, n)
        c_p3 = fft_convolve_self(a_p3)
        if c_p3.max() < global_best:
            global_best = c_p3.max()
            global_a = a_p3.copy()
            print("  Polish Mirror: %.6f" % global_best)

        # 3d: Peak targeting
        remaining4 = deadline - time.monotonic()
        seg3d = max(5.0, remaining4 * 0.15)
        a_p4, s_p4 = run_peak_target(global_a, n, S, seg3d, rng)
        a_p4 = safe_normalize(a_p4, n)
        c_p4 = fft_convolve_self(a_p4)
        if c_p4.max() < global_best:
            global_best = c_p4.max()
            global_a = a_p4.copy()
            print("  Polish PeakTarget: %.6f" % global_best)

        # 3e: Final fine FISTA
        remaining5 = deadline - time.monotonic()
        if remaining5 > 5:
            a_p5, s_p5 = run_fista_adaptive(global_a, n, S, remaining5 * 0.8, rng,
                                             beta_init=100.0, beta_max=1000.0)
            a_p5 = safe_normalize(a_p5, n)
            c_p5 = fft_convolve_self(a_p5)
            if c_p5.max() < global_best:
                global_best = c_p5.max()
                global_a = a_p5.copy()
                print("  Final FISTA: %.6f" % global_best)

    # === Phase 4: Verification ===
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Verification: %.6f (internal: %.6f), length=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: evaluate_sequence rejected, using fallback")
        global_a = np.ones(1024, dtype=np.float64) * 1.0

    # One more check
    final_list = [float(x) for x in global_a]
    ver2 = evaluate_sequence(final_list)
    print("=== Final: %.6f ===" % (ver2 if np.isfinite(ver2) else float('inf')))

    return final_list
