def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using mirror descent with adaptive subgradient.
Key features: multiplicative updates on the simplex, adaptive top-k peak chasing,
backtracking line search, momentum, stagnation detection with perturbations,
and multi-phase optimization (explore -> exploit -> polish).
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_ab(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection."""
    n = len(v)
    if n == 0:
        return v
    v_clipped = np.clip(v, 0, None)
    s = v_clipped.sum()
    if s <= S:
        # Already sum <= S, just scale up (uniform addition)
        return v_clipped + (S - s) / n
    # Need to find tau such that sum(max(v - tau, 0)) = S
    lo, hi = 0.0, v.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_mid = np.maximum(v - mid, 0.0).sum()
        if s_mid > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.maximum(v - tau, 0.0)


def compute_subgrad(c, a, top_k, soft_temperature=0.0):
    """
    Compute subgradient of max_k c[k] (approximated by top-k weighted average).
    grad[m] = 2 * sum_k w[k] * a[k-m] for valid indices.
    """
    n = len(a)
    top_k = max(1, min(top_k, len(c)))
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    c_max = c[top_idx[0]]

    if soft_temperature > 0:
        w = np.exp(soft_temperature * (c[top_idx] - c_max))
        w /= w.sum()
    else:
        w = np.full(top_k, 1.0 / top_k)

    # Build full weight vector of length 2n-1
    w_full = np.zeros(len(c))
    w_full[top_idx] = w

    # grad[m] = 2 * sum_k w_full[k] * a[k-m]
    # = 2 * conv(w_full, a_rev)[n-1 + m]
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1 - 1).bit_length()
    W = fft(w_full, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))

    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad

                   top_k_start=100, top_k_end=1, top_k_ramp_iters=20000,

                   soft_T=0.0, backtrack_ratio=0.7, verbose=True,
                   stagnation_limit=1500):

    Mirror descent (multiplicative updates) with subgradient for peak chasing.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)  # momentum buffer
    eta = eta0
    iter_no = 0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        # Compute subgradient
        progress = min(1.0, iter_no / max(1, top_k_ramp_iters))
        # Ramp top_k from start to end
        top_k_cur = int(top_k_start + (top_k_end - top_k_start) * progress)
        top_k_cur = max(1, top_k_cur)

        grad = compute_subgrad(c, a, top_k_cur, soft_T)

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_norm = grad / gnorm

        # Momentum
        v = momentum * v + (1.0 - momentum) * grad_norm
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        # Mirror descent step with backtracking
        accepted = False
        for bt in range(8):
            eta_try = eta * (backtrack_ratio ** bt)
            # Multiplicative update
            a_trial = a * np.exp(-eta_try * v)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial * (S / a_trial.sum())

            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)

                a = a_trial
                c = c_trial
                accepted = True
                break

            a = a * (S / a.sum())
            c = fft_convolve_self(a)

        cur_obj = obj_value(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
            # Increase eta slightly on success

            c = fft_convolve_self(a)

            if verbose:
                elapsed = time.monotonic() - start
                print(f"    [perturb] iter={iter_no}, obj={cur_obj:.6f}, "
                      f"best={best_obj:.6f}, t={elapsed:.1f}s")

        # Decay eta over time
        eta *= eta_decay

        iter_no += 1
        if verbose and iter_no % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"eta={eta:.4f}, top_k={top_k_cur}, t={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_polish(a0, S, time_budget, max_sweeps=200):
    """Coordinate descent polishing: try reducing each a[i] to lower the max."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        # Process indices in order of their contribution to peaks
        c_max = c.max()
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            # Try a few multiplicative factors
            for factor in [0.7, 0.85, 1.15, 1.3]:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_norm = a * (S / s)
                c_new = fft_convolve_self(a_norm)
                obj_new = c_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_norm.copy()
                    a = a_norm
                    c = c_new
                    improved = True
                    break
            a[i] = a[i]  # keep current
            # Renormalize
            s = a.sum()
            if s > 0:
                a = a * (S / s)

        if not improved:
            break

    return best_a, best_obj


def rl_polish(a0, S, time_budget, lr=0.001, top_k=1):
    """Reinforcement-learning style polish: focus on the single worst peak."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        # Find the single worst peak
        k_max = np.argmax(c)
        c_max_val = c[k_max]

        # Gradient of c[k_max] w.r.t. a
        # grad[m] = 2 * a[k_max - m] for valid m (0 <= k_max - m < n)
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for m in range(lo, hi + 1):
            grad[m] += 2.0 * a[k_max - m]

        # Normalized gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Momentum
        v = 0.9 * v + 0.1 * grad

        # Multiplicative update
        a = a * np.exp(-lr * v)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())

        c = fft_convolve_self(a)
        cur_obj = obj_value(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    global_best = np.inf
    global_a = None

    # --- Load given sequence ---
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print(f"Given sequence: n={n_given}, obj={ver:.6f}")
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print(f"Could not load given sequence: {e}")

    # --- Phase 1: Quick exploration (10% of time) ---
    explore_time = 0.10 * (deadline - time.monotonic())
    print(f"\nPhase 1: Exploration ({explore_time:.0f}s)")

    # Try different lengths
    candidates = []
    if given_a is not None:
        candidates.append(('given', given_a))

    # Generate alternative initializations
    for n_try in [4096, 8192]:
        S_try = np.sqrt(2.0 * n_try)
        # Cosine-initialized
        i = np.arange(n_try)
        a_cos = 1.0 + 0.5 * np.cos(2 * np.pi * 7 * i / n_try)
        a_cos = np.maximum(a_cos, 0.01)
        a_cos *= S_try / a_cos.sum()
        candidates.append((f'cos_n{n_try}', a_cos))

        # Random
        a_rand = rng.random(n_try)
        a_rand = np.maximum(a_rand, 0.01)
        a_rand *= S_try / a_rand.sum()
        candidates.append((f'rand_n{n_try}', a_rand))

        # Two-hump
        a_2h = (np.exp(-0.5 * ((i - n_try * 0.25) / (n_try * 0.08)) ** 2) +
                np.exp(-0.5 * ((i - n_try * 0.75) / (n_try * 0.08)) ** 2))
        a_2h = np.maximum(a_2h, 0.01)
        a_2h *= S_try / a_2h.sum()
        candidates.append((f'2hump_n{n_try}', a_2h))

    per_candidate_time = explore_time / len(candidates)
    for name, a_cand in candidates:
        if time.monotonic() > deadline - 60:
            break
        t0 = time.monotonic()
        a_best, obj_best = mirror_descent(
            a_cand, per_candidate_time,
            eta0=0.3, eta_decay=0.9999,
            top_k_start=200, top_k_end=50,
            top_k_ramp_iters=5000,
            momentum=0.8, verbose=False
        )
        elapsed = time.monotonic() - t0
        print(f"  {name}: obj={obj_best:.6f} ({elapsed:.1f}s)")
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

    # --- Phase 2: Main optimization (72% of time) ---
    main_time = 0.72 * (deadline - time.monotonic())
    print(f"\nPhase 2: Main optimization ({main_time:.0f}s)")

    if global_a is not None and main_time > 10:
        n = len(global_a)
        S = global_a.sum()

        # Split main time into 3 sub-phases with different strategies
        sub_time = main_time / 3.0

        # Sub-phase 2a: Broad flattening (many peaks)
        print("  2a: Broad flattening")
        a_best, obj_best = mirror_descent(
            global_a, sub_time,
            eta0=0.3, eta_decay=0.99995,
            top_k_start=500, top_k_end=200,
            top_k_ramp_iters=int(sub_time * 300),
            momentum=0.85, stagnation_limit=2000,
            soft_T=5.0, verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

        # Sub-phase 2b: Medium focus
        print("  2b: Medium focus")
        a_best, obj_best = mirror_descent(
            global_a, sub_time,
            eta0=0.2, eta_decay=0.99997,
            top_k_start=50, top_k_end=10,
            top_k_ramp_iters=int(sub_time * 300),
            momentum=0.9, stagnation_limit=1000,

        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

        # Sub-phase 2c: Tight focus on single peak
        print("  2c: Tight focus")
        a_best, obj_best = mirror_descent(

        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

    # --- Phase 3: Polish (15% of time) ---
    polish_time = deadline - time.monotonic() - 10.0
    print(f"\nPhase 3: Polish ({polish_time:.0f}s)")

    if global_a is not None and polish_time > 5:
        n = len(global_a)
        S = global_a.sum()

        # 3a: RL-style polish (single peak chasing)
        polish_3a = polish_time * 0.4
        print(f"  3a: RL polish ({polish_3a:.0f}s)")
        a_best, obj_best = rl_polish(global_a, S, polish_3a, lr=0.005, top_k=1)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

        # 3b: Coordinate descent
        polish_3b = polish_time * 0.3
        print(f"  3b: Coordinate descent ({polish_3b:.0f}s)")
        a_best, obj_best = coordinate_polish(global_a, S, polish_3b)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

        # 3c: Final fine mirror descent
        polish_3c = polish_time * 0.3

        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.6f}")

    return final_a
