def make_random_init(n, rng, spread=0.3):

def compute_gradient_at_lag(a, k):

        a: current sequence (normalized, non-negative)
        active_lags: list of lags to consider (where R(k) is near max)
        max_iterations: max number of LP iterations
    
    Returns:
        new_a: updated sequence
        improved: whether we made progress
    """
    n = len(a)
    R = fft_convolve(a, a)
    R_max = np.max(R)
    
    # Compute gradients for active lags
    grads = []
    for k in active_lags:
        # Use the effective lag (accounting for symmetry)
        if k >= n:
            k_eff = 2 * n - 2 - k
        else:
            k_eff = k
        g = compute_gradient_at_lag(a, k_eff)
        grads.append(g)
    
    grads = np.array(grads)  # shape: (len(active_lags), n)
    R_active = np.array([R[k] for k in active_lags])
    
    # LP: minimize t
    # subject to:
    #   grads[i] @ d <= t - R_active[i]  for all i (active lags)
    #   sum(d) = 0
    #   d_i >= -a_i  for all i
    
    # Variables: d_0, ..., d_{n-1}, t  (total n+1 variables)
    # Objective: minimize t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ vars <= b_ub
    # For each active lag i: grads[i] @ d - t <= -R_active[i]
    # i.e., [grads[i], -1] @ [d, t] <= -R_active[i]
    A_ub = np.zeros((len(active_lags), n + 1))
    for i in range(len(active_lags)):
        A_ub[i, :n] = grads[i]
        A_ub[i, -1] = -1.0
    b_ub = -R_active
    
    # d_i >= -a_i  =>  -d_i <= a_i
    # Add to A_ub:
    A_ub2 = np.zeros((n, n + 1))
    A_ub2[:, :n] = -np.eye(n)
    b_ub2 = a  # -d_i <= a_i means d_i >= -a_i
    
    A_ub = np.vstack([A_ub, A_ub2])
    b_ub = np.concatenate([b_ub, b_ub2])
    
    # Equality constraint: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: we already handle d_i >= -a_i through A_ub2
    # But we also need d_i to have some upper bound for numerical stability
    # t should be bounded below (it represents the new max, should be <= R_max)
    bounds = [(-a[i], a[i]) for i in range(n)] + [(0, R_max)]
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success:
            d = result.x[:n]
            t_new = result.x[-1]
            # Step size: we can take the full step if it's feasible
            # a_new = a + d (check non-negativity)
            a_new = a + d
            if np.all(a_new >= -1e-10):
                a_new = np.maximum(a_new, 0)
                a_new = a_new / np.sum(a_new)
                return a_new, True
    except Exception as e:
        pass
    
    return a, False


def active_set_method(a_init, time_limit, rng, delta=0.05):
    """
    Active set method: iteratively find and shave peaks using LP.
    
    Args:
        a_init: initial normalized non-negative sequence
        time_limit: seconds to run
        rng: random number generator
        delta: fraction of R_max to use for active set (lags within (1-delta)*R_max)
    
    Returns:
        best: best sequence found
        best_score: best score found
    """
    n = len(a_init)
    best = a_init.copy()
    best_score = local_score(best)
    current = a_init.copy()
    current_score = best_score
    
    t0 = time.monotonic()
    last_print = t0
    stuck_count = 0
    delta_current = delta
    
    while time.monotonic() - t0 < time_limit:
        # Compute autocorrelation
        R = fft_convolve(current, current)
        R_max = np.max(R)
        
        # Find active lags (those within delta of R_max)
        threshold = R_max * (1 - delta_current)
        active_lags = np.where(R >= threshold)[0]
        
        # Limit active set size for LP efficiency
        if len(active_lags) > 50:
            # Keep only the top 50
            top_idx = np.argsort(R[active_lags])[-50:]
            active_lags = active_lags[top_idx]
        
        # Solve LP
        a_new, improved = lp_peak_shave(current, active_lags)
        
        if improved:
            new_score = local_score(a_new)
            if new_score < current_score:
                current = a_new
                current_score = new_score
                stuck_count = 0
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
            else:
                # LP didn't help directly, try a fraction of the step
                # (This shouldn't happen if LP is correct, but just in case)
                stuck_count += 1
        else:
            stuck_count += 1
        
        # Adaptively adjust delta
        if stuck_count > 20:
            delta_current = max(0.01, delta_current * 0.8)
            stuck_count = 0
        elif stuck_count > 50:
            # Try a random perturbation to escape
            current = current * np.exp(rng.normal(0, 0.01, n))
            current = np.maximum(current, 0)
            current = current / np.sum(current)
            stuck_count = 0
            delta_current = delta
        
        # Print progress
        if time.monotonic() - last_print > 10.0:
            last_print = time.monotonic()
            print(f"    active_set: score={current_score:.8f}, best={best_score:.8f}, "
                  f"R_max={R_max:.8f}, {len(active_lags)} active, t={time.monotonic()-t0:.0f}s")
    
    return best, best_score


def run_lbgfs_softmax(theta, n, tau_schedule, maxiter_per_tau, deadline):
    """Run L-BFGS with annealed soft-max temperature."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))
    
    for tau in tau_schedule:
        if time.monotonic() > deadline - 5:
            break
        
        result = minimize(
            soft_max_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(tau,),
            options={'maxiter': maxiter_per_tau, 'ftol': 1e-16, 'gtol': 1e-14, 'maxfun': maxiter_per_tau * 3}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)
        
        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        theta = best_theta
    
    return best_theta, best_score


def project_simplex(v):
    """Project v onto the probability simplex (non-negative, sums to 1)."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def gradient_descent_simplex(a_init, time_limit, rng, lr=0.01):
    """
    Gradient descent on the simplex with a smoothed max objective.
    """
    n = len(a_init)
    best = a_init.copy()
    best_score = local_score(best)
    current = a_init.copy()
    current_score = best_score
    tau = 0.01
    t0 = time.monotonic()
    last_print = t0
    
    while time.monotonic() - t0 < time_limit:
        R = fft_convolve(current, current)
        R_max = np.max(R)
        
        # Soft-max weights
        shifted = (R - R_max) / tau
        w = np.exp(shifted)
        w /= np.sum(w)
        
        # Gradient of soft_max w.r.t. a
        grad_a = 2.0 * w[0] * current  # k=0 term
        for k in range(1, n):
            if w[k] < 1e-15:
                continue
            wk = w[k]
            grad_a[:n-k] += wk * current[k:]
            grad_a[k:] += wk * current[:n-k]
        
        # Project gradient onto tangent space of simplex
        grad_projected = grad_a - np.mean(grad_a)
        
        # Step
        candidate = current - lr * grad_projected
        candidate = project_simplex(candidate)
        
        new_score = local_score(candidate)
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                tau = max(tau * 0.95, 0.001)
        else:
            lr = max(lr * 0.7, 1e-8)
        
        # Adaptively decrease tau
        if int((time.monotonic() - t0)) % 50 == 0 and int((time.monotonic() - t0)) > 0:
            tau = max(tau * 0.9, 0.0005)
        
        if time.monotonic() - last_print > 10.0:
            last_print = time.monotonic()
            print(f"    grad_descent: score={current_score:.8f}, best={best_score:.8f}, "
                  f"tau={tau:.6f}, lr={lr:.8f}, t={time.monotonic()-t0:.0f}s")
    
    return best, best_score

"""
LP-based active set optimization for minimum peak autocorrelation non-negative sequences.
Combines L-BFGS gradient optimization with log-sum-exp objective and linear programming
based peak shaving for efficient refinement. Uses multiple phases: gradient descent,
LP-based active set refinement, and final polishing.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def compute_gradient_at_lag(a, k):
    """Compute dR(k)/da for a given lag k (0 <= k < n)."""
    n = len(a)
    if k == 0:
        return 2.0 * a
    grad = np.zeros(n)
    grad[:n - k] = a[k:]
    grad[k:] += a[:n - k]
    return grad


def lse_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * LSE(conv(softmax(theta), softmax(theta)) / tau).
    LSE(x/tau) = tau * log(sum exp(x/tau)) approximates max(x).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_len = len(c)  # 2n-1
    
    # Compute LSE and weights
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    lse_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    
    score = 2.0 * n * lse_val
    
    # Gradient: 2n * sum_k w_k * dR(k)/da
    # For lag k in [0, 2n-2], effective lag is min(k, 2n-2-k)
    grad_a = np.zeros(n)
    for k in range(c_len):
        if w[k] < 1e-15:
            continue
        k_eff = k if k < n else 2 * n - 2 - k
        grad_a += w[k] * compute_gradient_at_lag(a, k_eff)
    
    grad_a *= 2.0 * n
    
    # Convert to theta gradient (Jacobian of softmax)
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def lp_direction(a, R, active_lags):
    """
    Solve LP to find best descent direction for reducing max autocorrelation.
    
    Returns (direction, t_value) or (None, None) if LP fails.
    """
    n = len(a)
    
    # Compute gradients for active lags
    m = len(active_lags)
    if m == 0:
        return None, None
    
    # Build constraint matrix
    # Variables: d_0, ..., d_{n-1}, t  (n+1 total)
    # Minimize t
    # Subject to:
    #   sum_k w_k_active * (grad_k @ d) <= t - R(k)  for each active k
    #   sum(d) = 0
    #   -a_i <= d_i <= a_i (keep relative change bounded)
    #   0 <= t <= R_max
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: for each active lag k: grad_k @ d - t <= -R(k)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for i, k in enumerate(active_lags):
        k_eff = k if k < n else 2 * n - 2 - k
        g = compute_gradient_at_lag(a, k_eff)
        A_ub[i, :n] = g
        A_ub[i, -1] = -1.0
        b_ub[i] = -R[k]
    
    # Bounds on d: -a_i <= d_i <= a_i
    # (Already handled by bounds parameter, not A_ub)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    R_max = np.max(R)
    bounds = [(-a[i], a[i]) for i in range(n)] + [(0.0, R_max)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        d = result.x[:n]
        t_val = result.x[-1]
        return d, t_val
    
    return None, None


def active_set_method(a_init, time_limit, rng, delta=0.05, max_active=30):
    """
    LP-based active set method for peak reduction.
    """
    n = len(a_init)
    best = a_init.copy()
    best_score = local_score(best)
    current = a_init.copy()
    current_score = best_score
    
    t0 = time.monotonic()
    last_print = t0
    stuck_count = 0
    delta_current = delta
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        R = fft_convolve(current, current)
        R_max = np.max(R)
        
        # Find active lags
        threshold = R_max * (1 - delta_current)
        active_mask = R >= threshold
        active_lags = np.where(active_mask)[0]
        
        # Limit active set
        if len(active_lags) > max_active:
            top_idx = np.argsort(R[active_lags])[-max_active:]
            active_lags = active_lags[top_idx]
        
        # Solve LP
        d, t_val = lp_direction(current, R, active_lags)
        
        if d is not None:
            # Try multiple step sizes
            best_step_score = current_score
            best_a = current
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
                candidate = current + alpha * d
                if np.any(candidate < -1e-10):
                    continue
                candidate = np.maximum(candidate, 0)
                candidate = candidate / np.sum(candidate)
                s = local_score(candidate)
                if s < best_step_score:
                    best_step_score = s
                    best_a = candidate
            
            if best_step_score < current_score - 1e-12:
                current = best_a
                current_score = best_step_score
                stuck_count = 0
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
            else:
                stuck_count += 1
        else:
            stuck_count += 1
        
        # Adaptively adjust
        if stuck_count > 30:
            delta_current = max(0.01, delta_current * 0.7)
            stuck_count = 15
        if stuck_count > 60:
            # Random kick
            kick = rng.normal(0, 0.02, n)
            current = current * np.exp(kick)
            current = np.maximum(current, 0)
            current = current / np.sum(current)
            stuck_count = 0
            delta_current = max(delta, 0.03)
        
        iteration += 1
        if time.monotonic() - last_print > 10.0:
            last_print = time.monotonic()
            print(f"    active_set: iter={iteration}, score={current_score:.8f}, "
                  f"best={best_score:.8f}, R_max={R_max:.8f}, "
                  f"active={len(active_lags)}, delta={delta_current:.3f}, "
                  f"t={time.monotonic()-t0:.0f}s")
    
    return best, best_score


def run_lbgfs(theta, n, tau_schedule, maxiter_per_tau, deadline):
    """Run L-BFGS with annealed LSE temperature."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))
    
    for tau in tau_schedule:
        if time.monotonic() > deadline - 5:
            break
        
        result = minimize(
            lse_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(tau,),
            options={'maxiter': maxiter_per_tau, 'ftol': 1e-16, 'gtol': 1e-14,
                     'maxfun': maxiter_per_tau * 3}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)
        
        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
            print(f"      tau={tau:.5f}: score={s:.8f}")
        theta = best_theta
    
    return best_theta, best_score


def gradient_descent_refine(a_init, time_limit, rng):
    """
    Gradient descent on the simplex with adaptive LSE temperature.
    """
    n = len(a_init)
    best = a_init.copy()
    best_score = local_score(best)
    current = a_init.copy()
    current_score = best_score
    tau = 0.005
    lr = 0.01
    t0 = time.monotonic()
    last_print = t0
    
    while time.monotonic() - t0 < time_limit:
        R = fft_convolve(current, current)
        R_max = np.max(R)
        
        # LSE weights
        shifted = (R - R_max) / tau
        w = np.exp(shifted)
        w /= np.sum(w)
        
        # Gradient
        grad_a = np.zeros(n)
        for k in range(len(R)):
            if w[k] < 1e-15:
                continue
            k_eff = k if k < n else 2 * n - 2 - k
            grad_a += w[k] * compute_gradient_at_lag(current, k_eff)
        
        # Project to tangent space
        grad_a -= np.mean(grad_a)
        
        # Step with backtracking
        found = False
        for alpha in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = current - alpha * grad_a
            candidate = project_simplex(candidate)
            s = local_score(candidate)
            if s < current_score:
                current = candidate
                current_score = s
                found = True
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
                    tau = max(tau * 0.97, 0.0005)
                break
        
        if not found:
            lr = max(lr * 0.5, 1e-10)
        
        if time.monotonic() - last_print > 10.0:
            last_print = time.monotonic()
            print(f"    GD: score={current_score:.8f}, best={best_score:.8f}, "
                  f"tau={tau:.6f}, lr={lr:.8f}, t={time.monotonic()-t0:.0f}s")
    
    return best, best_score

"""
LP-based active set optimization for minimum peak autocorrelation non-negative sequences.
Uses correct gradient of full convolution (shifted reversed array), combines L-BFGS with
log-sum-exp objective, LP-based active set peak shaving, and gradient descent refinement.
Key insight: d(a*a)(k)/da[j] = 2*a[k-j] if 0<=k-j<n, else 0.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(full_conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def full_conv_gradient_at_lag(a, k):
    """
    Compute d(a*a)(k)/da. Correct formula:
    grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    if j_start <= j_end:
        indices = np.arange(j_start, j_end + 1)
        a_indices = k - indices
        grad[indices] = 2.0 * a[a_indices]
    return grad


def lse_obj_grad(theta, tau):
    """
    Objective: 2n * LSE(full_conv(softmax(theta)) / tau)
    Gradient: 2n * sum_k w_k * grad_k = 2n * 2 * sum_i w[j+i] * a[i] for each j.
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_len = len(c)  # 2n-1

    # LSE and weights
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    lse_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score = 2.0 * n * lse_val

    # Gradient: grad_a[j] = 2n * 2 * sum_{i=0}^{n-1} w[j+i] * a[i]
    grad_a = np.zeros(n)
    for j in range(n):
        grad_a[j] = 2.0 * np.dot(w[j:j + n], a)
    grad_a *= 2.0 * n

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def make_random_init(n, rng, spread=0.3):

    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def project_simplex(v):
    """Project v onto the probability simplex (non-negative, sums to 1)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def lp_peak_direction(a, R, active_lags):
    """
    Solve LP to find best descent direction for reducing max full convolution.
    
    Minimize t
    Subject to: grad_k @ d - t <= -R(k) for active k
                sum(d) = 0
                -0.5*a_i <= d_i <= a_i
                0 <= t
    """
    n = len(a)
    m = len(active_lags)
    if m == 0:
        return None

    # Objective: minimize t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality constraints for active lags
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)

    for i, k in enumerate(active_lags):
        g = full_conv_gradient_at_lag(a, k)
        A_ub[i, :n] = g
        A_ub[i, -1] = -1.0
        b_ub[i] = -R[k]

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    # Bounds
    R_max = np.max(R)
    bounds = [(-0.5 * a[i], a[i]) for i in range(n)] + [(0.0, R_max)]

    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')

    if result.success:
        return result.x[:n]
    return None


def active_set_method(a_init, time_limit, rng, delta=0.05, max_active=25):
    """
    LP-based active set method for peak reduction of full convolution.
    Iteratively identifies near-peak lags and uses LP to find optimal descent direction.
    """
    n = len(a_init)
    best = a_init.copy()
    best_score = local_score(best)
    current = a_init.copy()
    current_score = best_score

    t0 = time.monotonic()
    last_print = t0
    stuck_count = 0
    delta_current = delta
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        R = fft_convolve(current, current)
        R_max = np.max(R)

        # Find active lags (near-peak)
        threshold = R_max * (1 - delta_current)
        active_lags = np.where(R >= threshold)[0]

        # Limit active set for LP efficiency
        if len(active_lags) > max_active:
            top_idx = np.argsort(R[active_lags])[-max_active:]
            active_lags = active_lags[top_idx]

        # Solve LP for descent direction
        d = lp_peak_direction(current, R, active_lags)

        if d is not None and np.any(np.abs(d) > 1e-12):
            # Line search: try multiple step sizes
            best_step_score = current_score
            best_a = current
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
                candidate = current + alpha * d
                if np.any(candidate < -1e-10):
                    continue
                candidate = np.maximum(candidate, 0.0)
                s_sum = np.sum(candidate)
                if s_sum < 0.01:
                    continue
                candidate = candidate / s_sum
                s = local_score(candidate)
                if s < best_step_score:
                    best_step_score = s
                    best_a = candidate

            if best_step_score < current_score - 1e-14:
                current = best_a
                current_score = best_step_score
                stuck_count = 0
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
            else:
                stuck_count += 1
        else:
            stuck_count += 1

        # Adaptive adjustment
        if stuck_count > 20:
            delta_current = max(0.01, delta_current * 0.7)
            stuck_count = 10
        if stuck_count > 50:
            # Random perturbation to escape local minimum
            kick_sigma = 0.01 + 0.01 * min(stuck_count / 100.0, 1.0)
            kick = rng.normal(0, kick_sigma, n)
            current = current * np.exp(kick)
            current = np.maximum(current, 1e-10)
            current = current / np.sum(current)
            stuck_count = 0
            delta_current = max(delta, 0.03)

        iteration += 1
        if time.monotonic() - last_print > 15.0:
            last_print = time.monotonic()
            elapsed = time.monotonic() - t0
            print(f"    LP: iter={iteration}, score={current_score:.8f}, "
                  f"best={best_score:.8f}, R_max={R_max:.6f}, "
                  f"active={len(active_lags)}, delta={delta_current:.3f}, "
                  f"t={elapsed:.0f}s")

    return best, best_score


def run_lbgfs(theta, n, tau_schedule, maxiter_per_tau, deadline):
    """Run L-BFGS with annealed LSE temperature."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))

    for tau in tau_schedule:
        if time.monotonic() > deadline - 5:
            break

        try:
            result = minimize(
                lse_obj_grad, theta.copy(),
                method='L-BFGS-B',
                jac=True, args=(tau,),
                options={'maxiter': maxiter_per_tau, 'ftol': 1e-18, 'gtol': 1e-16,
                         'maxfun': maxiter_per_tau * 3}
            )
            theta_new = result.x
            a_new = softmax(theta_new)
            s = local_score(a_new)

            if s < best_score:
                best_score = s
                best_theta = theta_new.copy()
                print(f"      tau={tau:.5f}: score={s:.8f}")
            theta = best_theta
        except Exception:
            break

    return best_theta, best_score


def gradient_descent_refine(a_init, time_limit, rng):
    """
    Gradient descent on the simplex with adaptive LSE temperature.
    Uses the correct full convolution gradient.
    """
    n = len(a_init)
    best = a_init.copy()
    best_score = local_score(best)
    current = a_init.copy()
    current_score = best_score
    tau = 0.005
    lr = 0.005
    t0 = time.monotonic()
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        R = fft_convolve(current, current)
        R_max = np.max(R)

        # LSE weights
        shifted = (R - R_max) / tau
        w = np.exp(shifted)
        w /= np.sum(w)

        # Gradient: grad_a[j] = 2 * sum_{i=0}^{n-1} w[j+i] * a[i]
        grad_a = np.zeros(n)
        for j in range(n):
            grad_a[j] = 2.0 * np.dot(w[j:j + n], current)

        # Project to tangent space of simplex (sum=0)
        grad_a -= np.mean(grad_a)

        # Backtracking line search
        found = False
        for alpha_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            alpha = lr * alpha_mult
            candidate = current - alpha * grad_a
            candidate = project_simplex(candidate)
            s = local_score(candidate)
            if s < current_score - 1e-14:
                current = candidate
                current_score = s
                found = True
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
                    tau = max(tau * 0.97, 0.0005)
                break

        if not found:
            lr = max(lr * 0.5, 1e-10)

        if time.monotonic() - last_print > 15.0:
            last_print = time.monotonic()
            elapsed = time.monotonic() - t0
            print(f"    GD: score={current_score:.8f}, best={best_score:.8f}, "
                  f"tau={tau:.6f}, lr={lr:.8f}, t={elapsed:.0f}s")

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization for minimum peak full-convolution non-negative sequences.
    Phase 1: L-BFGS with LSE objective (multiple initializations, aggressive annealing)
    Phase 2: LP-based active set peak shaving (main refinement)
    Phase 3: Gradient descent refinement on simplex
    Phase 4: Final LP polish with tighter active set
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0001, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.8f}")
        best_score = s
        best_seq = start_seq.copy()

    n = 256

    # Phase 1: L-BFGS optimization with LSE objective
    print(f"\n=== Phase 1: L-BFGS LSE optimization (n={n}) ===")

    inits = []
    if start_seq is not None:
        if len(start_seq) == n:
            inits.append(('parent', start_seq.copy()))
            # Perturbed parent
            pert = start_seq * np.exp(rng.normal(0, 0.05, n))
            pert = np.maximum(pert, 0.0001)
            pert = pert / np.sum(pert)
            inits.append(('parent_pert', pert))
            # Strongly perturbed
            pert2 = start_seq * np.exp(rng.normal(0, 0.15, n))
            pert2 = np.maximum(pert2, 0.0001)
            pert2 = pert2 / np.sum(pert2)
            inits.append(('parent_pert2', pert2))
        else:
            x_old = np.linspace(0, 1, len(start_seq))
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, start_seq)
            a_interp = np.maximum(a_interp, 0.0001)
            a_interp = a_interp / np.sum(a_interp)
            inits.append(('parent_interp', a_interp))

    inits.append(('random1', make_random_init(n, rng, 0.3)))
    inits.append(('random2', make_random_init(n, rng, 0.5)))
    inits.append(('chirp1', make_chirp_init(n, 0.4, rng)))
    inits.append(('bell1', make_bell_init(n, n * 0.2, rng)))

    tau_schedule_full = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    tau_schedule_short = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    for init_name, a0 in inits:
        if time.monotonic() > deadline - 900:
            break

        s0 = local_score(a0)
        if s0 < best_score:
            best_score = s0
            best_seq = a0.copy()
            print(f"  init '{init_name}': score={s0:.8f} *** NEW BEST ***")
        else:
            print(f"  init '{init_name}': score={s0:.8f}")

        theta = np.log(a0 + 1e-30)

        if init_name.startswith('parent'):
            tau_sub = tau_schedule_short
            maxiter = 500
        else:
            tau_sub = tau_schedule_full
            maxiter = 300

        dl_sub = min(time.monotonic() + 180, deadline - 700)
        theta_best, s = run_lbgfs(theta, n, tau_sub, maxiter, dl_sub)

        if s < best_score:
            best_score = s
            best_seq = softmax(theta_best).copy()
            print(f"  L-BFGS '{init_name}': score={s:.8f} *** NEW BEST ***")
        else:
            print(f"  L-BFGS '{init_name}': score={s:.8f}")

    # Phase 2: LP active set method (main refinement)
    print(f"\n=== Phase 2: LP active set (score={best_score:.8f}) ===")
    if best_seq is not None:
        lp_time = min(1100.0, max(300.0, deadline - time.monotonic() - 500))
        print(f"  Allocated LP time: {lp_time:.0f}s")
        lp_best, lp_score = active_set_method(best_seq, lp_time, rng, delta=0.05, max_active=25)
        if lp_score < best_score:
            best_score = lp_score
            best_seq = lp_best.copy()
            print(f"  After LP: {best_score:.8f} *** NEW BEST ***")
        else:
            print(f"  After LP: {lp_score:.8f} (no improvement)")

    # Phase 3: Gradient descent refinement
    print(f"\n=== Phase 3: GD refinement (score={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        gd_time = min(400.0, max(60.0, deadline - time.monotonic() - 200))
        print(f"  Allocated GD time: {gd_time:.0f}s")
        gd_best, gd_score = gradient_descent_refine(best_seq, gd_time, rng)
        if gd_score < best_score:
            best_score = gd_score
            best_seq = gd_best.copy()
            print(f"  After GD: {best_score:.8f} *** NEW BEST ***")
        else:
            print(f"  After GD: {gd_score:.8f} (no improvement)")

    # Phase 4: Final LP polish with tighter active set
    print(f"\n=== Phase 4: Final LP polish (score={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        final_time = max(30.0, min(300.0, deadline - time.monotonic() - 10))
        final_best, final_score = active_set_method(best_seq, final_time, rng,
                                                    delta=0.02, max_active=15)
        if final_score < best_score:
            best_score = final_score
            best_seq = final_best.copy()
            print(f"  After final LP: {best_score:.8f} *** NEW BEST ***")
        else:
            print(f"  After final LP: {final_score:.8f} (no improvement)")

    # Final output
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score_check = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score_check:.8f}, n={len(final_list)} ===")
        return final_list

    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
