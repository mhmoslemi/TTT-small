"""
Multi-phase optimization using L-BFGS-B with spectral annealing to minimize
peak self-convolution of non-negative sequences. Uses exponential parameterization,
FFT-based gradient computation, and multi-start strategy with diverse initializations.
Focuses on longer sequences (n=2048-4096) for better asymptotic scores.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def smooth_max_value_and_weights(conv, tau):
    """Compute smooth max and softmax weights."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    value = max_c + tau * np.log(np.sum(exp_vals))
    return value, w


def grad_wrt_a(conv_weights, a, n):
    """
    Compute gradient of smooth_max(conv(a,a)) w.r.t. a.
    grad_j = sum_m w_{m+j} * a_m
    """
    # conv_weights has length 2n-1, a has length n
    # We want g[j] = sum_{m=0}^{n-1} w[m+j] * a[m] for j = 0, ..., n-1
    N = 1
    while N < 3 * n:
        N <<= 1
    FA = np.fft.rfft(a, N)
    FW = np.fft.rfft(conv_weights, N)
    corr = np.fft.irfft(np.conj(FA) * FW, N)[:n]
    return corr


def obj_and_grad(z, tau, n):
    """
    Objective and gradient for L-BFGS-B.
    z: parameter vector (length n)
    a = exp(z) / sum(exp(z))
    Minimize smooth_max(conv(a,a), tau)
    """
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    a = e / np.sum(e)
    
    conv = fft_convolve(a, a)
    value, w = smooth_max_value_and_weights(conv, tau)
    
    # Gradient w.r.t. a
    g_a = grad_wrt_a(w, a, n)
    
    # Gradient w.r.t. z: a_k * (g_a_k - sum_j a_j * g_a_j)
    inner = np.dot(a, g_a)
    g_z = a * (g_a - inner)
    
    return value, g_z


def make_init_cosine(n, n_harmonics, rng):
    """Initialize with a cosine series."""
    t = np.arange(n) / n
    a = np.ones(n)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.05, 0.3) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.clip(a, 0.01, None)
    return a


def make_init_lowpass(n, rng, bandwidth=0.1):
    """Initialize with band-limited noise."""
    # Generate noise in frequency domain, keep only low frequencies
    N = 1
    while N < 2 * n:
        N <<= 1
    F = rng.standard_normal(N // 2 + 1) + 1j * rng.standard_normal(N // 2 + 1)
    # Zero out high frequencies
    cutoff = int(bandwidth * N // 2)
    F[cutoff:] = 0
    F[:1] = 0  # Remove DC
    f = np.fft.irfft(F, N)[:n]
    a = 1.0 + 0.5 * f / (np.std(f) + 1e-10)
    a = np.clip(a, 0.01, None)
    return a


def make_init_ramp(n, rng, slope=0.5):
    """Ramp initialization."""
    t = np.arange(n) / n
    a = 1.0 + slope * (2 * t - 1)  # Goes from 1-slope to 1+slope
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a


def make_init_bipolar(n, rng, sharpness=5.0):
    """Bipolar initialization: high on one side, low on the other."""
    t = np.arange(n) / n
    a = 1.0 + np.tanh(sharpness * (2 * t - 1))  # Transitions from ~0 to ~2
    a = a / np.max(a) * 0.5 + 0.5  # Scale to [0.5, 1.5] roughly
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a

    Run L-BFGS-B optimization with tau annealing.
    tau_schedule: list of (tau, num_iter) pairs.
    """

    best_z = z.copy()
    best_score = np.inf
    t0 = time.monotonic()
    
    for tau, n_iter in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Run L-BFGS-B for n_iter iterations or until time runs out
        n_done = 0
        while n_done < n_iter:
            if time.monotonic() - t0 > time_limit:
                break
            
            val, grad = obj_and_grad(z, tau, n)
            
            if np.any(np.isnan(grad)) or np.any(np.isinf(grad)):
                break
            
            # Use a single L-BFGS-B step (or a few)
            # Actually, let's use scipy's minimize in 'batch' mode
            # But that would recompute the gradient internally...
            # Let's just do manual L-BFGS-B steps.
            
            break  # We'll use scipy.minimize instead
        
        # Use scipy minimize for a batch
        if time.monotonic() - t0 > time_limit:
            break
        
        result = minimize(

def stochastic_refine(a, time_limit, rng, tau=0.01):
    """
    Stochastic local search: randomly perturb elements and accept if score improves.
    Works on normalized a (sum=1).
    """

    current_score = 2.0 * n * np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    # Adaptive step size
    step = 0.05
    
    while time.monotonic() - t0 < time_limit:
        # Perturb a random subset
        n_perturb = max(1, int(n * rng.uniform(0.001, 0.01)))
        idx = rng.choice(n, size=n_perturb, replace=False)

        candidate[idx] *= np.exp(rng.normal(0, step, size=n_perturb))

            current_score = new_score

            step = min(0.1, step * 1.01)
