LP-based Frank-Wolfe optimization for minimum peak autocorrelation sequences.
Uses bilinear linearization with LP subproblem, combined with p-norm polishing.

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score for sequence: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def build_lp_matrix(f, n):
    """Build the (2n-1) x n constraint matrix for (f*g)_k <= M."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        for i in range(i_min, i_max + 1):
            A[k, i] = f[i]
    return A

    """Solve LP: maximize sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""

    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n

        return result.x
    return None

def alternating_lp(f_init, n, max_iters=5000, time_limit=60.0, damping=0.5):

"""
Dual-method LP optimization for minimum peak autocorrelation non-negative sequences.
Combines alternating LP (block coordinate descent) with Frank-Wolfe LP direction steps,
followed by p-norm and soft-max L-BFGS polishing. Operates at n=256.

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def eval_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def soft_max_obj_grad(theta, tau):
    """Objective and gradient for 2n * logsumexp(conv(softmax(theta), softmax(theta))/tau)*tau."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    
    score = 2.0 * n * soft_max_val
    
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def build_conv_matrix(other, n):
    """
    Build the (2n-1) x n matrix A such that (A @ x)_k = (other * x)_k.
    """
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            A[k, i] = other[i]
    return A


def solve_lp_min_M(other, target_sum, n):
    """
    Solve: minimize M s.t. sum(x)=target_sum, x>=0, (other*x)_k <= M for all k=0,...,2n-2.
    Returns (x, M) or None.
    """
    # Variables: x_0,...,x_{n-1}, M  (n+1 vars)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A = build_conv_matrix(other, n)
    # Augment with -1 column for M
    A_aug = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_aug, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None


def solve_lp_max_sum(other, M, n):
    """
    Solve: maximize sum(x) s.t. x>=0, (other*x)_k <= M for all k.
    Returns x or None.
    """
    c = -np.ones(n)
    
    A = build_conv_matrix(other, n)
    b = np.full(2 * n - 1, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    Alternating LP: minimize max(f*f) via block coordinate descent on (f,g)
    where at each step we solve an LP for one variable.

    Frank-Wolfe with LP direction finding.
    """
