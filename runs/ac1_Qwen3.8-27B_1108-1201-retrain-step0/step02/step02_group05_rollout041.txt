def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_normalized(a):
    """Score for sequence with sum=1: 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def make_conv_matrix(a):
    """Build matrix A such that (A @ b)[k] = convolve(a, b)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp_direction(a, M):
    """
    Solve LP: max sum(b) s.t. convolve(a,b) <= M, b >= 0.
    Returns optimal b or None if infeasible.
    """
    n = len(a)
    A = make_conv_matrix(a)
    
    # Primal: max 1'b s.t. Ab <= M*1, b >= 0
    # In scipy linprog form: minimize c'x s.t. A_ub x <= b_ub
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.status == 0:
        return result.x
    return None

    t = theta - np.max(theta)

non-negative sequences. Combines LP oracle direction finding with L-BFGS
refinement and coordinate descent for a multi-phase search.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def score_norm1(a):
    """Score assuming sum(a)=1: 2n*max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)

    """Convolution matrix: (A @ b)[k] = convolve(a,b)[k]."""

    for i in range(n):

    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    return a * (grad_a - inner)

import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def score_norm1(a):
    """Score for sequence with sum=1: 2n*max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def make_A(a):
    """Build convolution matrix A where (A@b)[k] = convolve(a,b)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def lp_solve_direction(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M elementwise, b >= 0."""
    n = len(a)
    A = make_A(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.status == 0 and result.x is not None:
        return result.x
    return None

    """Try symmetric version of a."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    ss = a_sym.sum()
    if ss < 1e-15:
        return a, score_norm1(a)
    a_sym = a_sym / ss
    s_sym = score_norm1(a_sym)
    s_orig = score_norm1(a)
    if s_sym < s_orig:
        return a_sym, s_sym
    return a, s_orig


def make_initializer(n, init_type, rng):
    """Create various initializations."""
    t = np.arange(n, dtype=np.float64)

    if init_type == 'chirp':
        alpha = 0.2 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(2.0 * np.pi * t * (t + 0.5) / (4.0 * n))
    elif init_type == 'chirp2':
        alpha = 0.15 + 0.25 * rng.random()
        a = 1.0 + alpha * np.cos(2.0 * np.pi * t * t / (3.0 * n))
    elif init_type == 'bell':
        width = n * (0.12 + 0.15 * rng.random())
        center = n * 0.5
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
    elif init_type == 'two_lobe':
        w = n * 0.12
        a = np.exp(-0.5 * (t / w) ** 2) + np.exp(-0.5 * ((t - n + 1) / w) ** 2)
    elif init_type == 'flat_cos':
        freq = 1 + int(rng.integers(1, 5))
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * freq * t / n)
    elif init_type == 'random_smooth':
        a = 1.0 + 0.4 * rng.standard_normal(n)
        kernel = np.exp(-0.5 * (np.arange(-4, 5) / 1.5) ** 2)
        a = np.convolve(a, kernel / kernel.sum(), mode='same')
    else:
        a = np.ones(n) * (1.0 + 0.2 * rng.random())

    a = np.clip(a, 0.01, None)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimum peak autocorrelation non-negative sequence.
    Phase 1: LP-guided Frank-Wolfe from multiple initializations
    Phase 2: L-BFGS with p-norm and soft-max refinement
    Phase 3: Coordinate descent with peak-targeted moves
    Phase 4: Symmetrization and final polish
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 40.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        a_prov = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-10, None)
        a_prov = a_prov / a_prov.sum()
        s = score_norm1(a_prov)
        print(f"Provided sequence: n={len(a_prov)}, score={s:.6f}")
        best_score = s
        best_seq = a_prov.copy()

    n = len(best_seq) if best_seq is not None else 256

    # === PHASE 1: LP-guided optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 1: LP-guided Frank-Wolfe (n={n})")
    print(f"{'='*60}")
    p1_deadline = t_global + (deadline - t_global) * 0.45

    inits = []
    if best_seq is not None:
        inits.append(('provided', best_seq.copy()))
    inits.append(('chirp', make_initializer(n, 'chirp', rng)))
    inits.append(('chirp2', make_initializer(n, 'chirp2', rng)))
    inits.append(('bell', make_initializer(n, 'bell', rng)))
    inits.append(('two_lobe', make_initializer(n, 'two_lobe', rng)))
    inits.append(('random_smooth', make_initializer(n, 'random_smooth', rng)))
    inits.append(('flat_cos', make_initializer(n, 'flat_cos', rng)))

    for name, a0 in inits:
        remaining = p1_deadline - time.monotonic()
        if remaining < 30:
            break
        # Allocate time: more for 'provided', less for others
        if name == 'provided':
            alloc = min(300.0, remaining * 0.4)
        else:
            alloc = min(150.0, remaining * 0.2)
        alloc = max(alloc, 20.0)

        t_start = time.monotonic()
        a_res, s = lp_guided_optimize(a0, alloc, seed=seed + hash(name) % 10000)
        elapsed = time.monotonic() - t_start

        # Try symmetrization
        a_res, s = try_symmetrize(a_res)

        if s < best_score:
            best_score = s
            best_seq = a_res.copy()
            print(f"  [{name}] NEW BEST: {s:.8f} (t={elapsed:.0f}s)")
        else:
            print(f"  [{name}] score={s:.6f} (best={best_score:.6f}, t={elapsed:.0f}s)")

    # === PHASE 2: L-BFGS refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS refinement")
    print(f"{'='*60}")
    print(f"  Best entering: {best_score:.8f}")

    if best_seq is not None and time.monotonic() < deadline - 150:
        p2_time = min(500.0, deadline - time.monotonic() - 80)

        # p-norm refinement
        p2a_time = p2_time * 0.6
        a_res, s = lbgfs_pnorm_refine(best_seq, p2a_time,
                                      p_schedule=[16, 32, 64, 128, 256, 512])
        if s < best_score:
            best_score = s
            best_seq = a_res.copy()
            print(f"  p-norm: NEW BEST {s:.8f}")
        else:
            print(f"  p-norm: {s:.8f} (no improvement)")

        # Soft-max refinement
        if time.monotonic() < deadline - 80:
            p2b_time = min(200.0, deadline - time.monotonic() - 60)
            a_res, s = lbgfs_softmax_refine(best_seq, p2b_time,
                                            tau_schedule=[0.005, 0.002, 0.001, 0.0005, 0.0002])
            if s < best_score:
                best_score = s
                best_seq = a_res.copy()
                print(f"  soft-max: NEW BEST {s:.8f}")
            else:
                print(f"  soft-max: {s:.8f} (no improvement)")

    # === PHASE 3: Second LP round (if improved) ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Second LP-guided round")
    print(f"{'='*60}")
    print(f"  Best entering: {best_score:.8f}")

    if best_seq is not None and time.monotonic() < deadline - 100:
        p3_time = min(300.0, deadline - time.monotonic() - 60)
        a_res, s = lp_guided_optimize(best_seq, p3_time, seed=seed + 7777)
        a_res, s = try_symmetrize(a_res)
        if s < best_score:
            best_score = s
            best_seq = a_res.copy()
            print(f"  LP round 2: NEW BEST {s:.8f}")
        else:
            print(f"  LP round 2: {s:.8f} (no improvement)")

    # === PHASE 4: Coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Coordinate descent polish")
    print(f"{'='*60}")
    print(f"  Best entering: {best_score:.8f}")

    if best_seq is not None and time.monotonic() < deadline - 25:
        p4_time = min(400.0, deadline - time.monotonic() - 15)
        a_res, s = coordinate_descent(best_seq, p4_time, seed=seed + 31337)
        if s < best_score:
            best_score = s
            best_seq = a_res.copy()
            print(f"  CD: NEW BEST {s:.8f}")
        else:
            print(f"  CD: {s:.8f} (no improvement)")

    # === PHASE 5: Final symmetrization and tiny refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        # Try symmetrization
        a_sym, s_sym = try_symmetrize(best_seq)
        if s_sym < best_score:
            best_score = s_sym
            best_seq = a_sym
            print(f"  Symmetrization improved: {best_score:.8f}")

        # One more tiny L-BFGS
        if time.monotonic() < deadline - 10:
            final_time = max(10, deadline - time.monotonic() - 5)
            a_res, s = lbgfs_pnorm_refine(best_seq, final_time, p_schedule=[256, 512])
            if s < best_score:
                best_score = s
                best_seq = a_res
                print(f"  Final L-BFGS: NEW BEST {s:.8f}")

    # Final verification and return
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL RESULT: evaluate_sequence score = {final_score:.8f}")
        print(f"Sequence length: {len(final_list)}")
        print(f"Sum: {sum(final_list):.4f}")
        print(f"Max element: {max(final_list):.6f}")
        print(f"Min element: {min(final_list):.6f}")
        print(f"{'='*60}")
        return final_list

    # Fallback
    print("WARNING: No valid sequence found, returning fallback.")
    return [1.0] * 256
