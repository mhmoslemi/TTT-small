"""
Optimization of non-negative step functions to minimize peak self-convolution.
Uses mirror descent with smooth-max approximation, structured initializations,
and coordinate descent polishing. Exploits FFT-based gradient computation for
efficient iterations at large sequence lengths.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def compute_grad_smooth(a, tau):
    """
    Compute the smooth-max gradient of f(a) = max(conv(a,a)).
    Returns (smooth_value, gradient) where gradient has length n.
    """
    n = len(a)
    conv = fft_convolve(a, a)  # length 2n-1
    # Smooth max: tau * log(sum exp(conv/tau))
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / tau
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    smooth_val = max_conv + tau * np.log(sum_exp)
    
    # Gradient: (1/tau) * sum_k w[k] * (a[k-i] + a[i-k])
    # = (1/tau) * (cross_correlate(w, a) + correlate(w, a))[0:n]
    L = 1
    while L < 3 * n - 2:
        L <<= 1
    w_padded = np.zeros(L)
    w_padded[:len(conv)] = w
    a_padded = np.zeros(L)
    a_padded[:n] = a
    
    fft_w = np.fft.rfft(w_padded)
    fft_a = np.fft.rfft(a_padded)
    # 2*real(fft_w) * fft_a gives the combined gradient kernel
    grad_full = np.fft.irfft(2.0 * np.real(fft_w) * fft_a, L)[:n]
    grad = grad_full / tau
    return smooth_val, grad


def mirror_descent_optimize(a_init, time_budget, tau_start=1.0, tau_end=0.01, 
                            alpha=0.5, momentum=0.9, check_interval=10, seed=0):
    """
    Mirror descent optimization on the simplex.
    a_init: initial point on simplex (sum=1, a>=0)
    Returns (best_a, best_max_conv)
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.clip(a, 1e-15, None)
    a = a / np.sum(a)
    
    # Precompute FFT size
    L = 1
    while L < 3 * n - 2:
        L <<= 1
    a_padded_base = np.zeros(L)
    a_padded_base[:n] = a  # will be updated each iteration
    
    best_a = a.copy()
    best_val = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    step = 0
    g_mom = np.zeros(n)  # momentum for gradient
    
    while time.monotonic() - t0 < time_budget:
        elapsed = min(1.0, (time.monotonic() - t0) / time_budget)
        # Anneal tau
        tau = tau_start * (tau_end / tau_start) ** (elapsed ** 0.6)
        # Adaptive alpha: decrease as tau decreases to keep step size reasonable
        alpha_eff = alpha * (tau / tau_start) ** 0.5
        
        conv = fft_convolve(a, a)
        max_conv = np.max(conv)
        shifted = (conv - max_conv) / tau
        exp_vals = np.exp(shifted)
        sum_exp = np.sum(exp_vals)
        w = exp_vals / sum_exp
        
        # Gradient computation
        w_padded = np.zeros(L)
        w_padded[:len(conv)] = w
        # Update a_padded
        a_padded_base[:n] = a
        
        fft_w = np.fft.rfft(w_padded)
        fft_a = np.fft.rfft(a_padded_base)
        grad = np.fft.irfft(2.0 * np.real(fft_w) * fft_a, L)[:n] / tau
        
        # Momentum
        g_mom = momentum * g_mom + (1 - momentum) * grad
        
        # Mirror descent step
        step_size = alpha_eff * g_mom
        step_size = np.clip(step_size, -20, 20)
        a_new = a * np.exp(-step_size)
        s = np.sum(a_new)
        if s < 1e-10 or not np.isfinite(s):
            # Reset
            a_new = np.ones(n) / n
            g_mom = np.zeros(n)
            continue
        a_new = a_new / s
        a = a_new
        step += 1
        
        # Check improvement periodically
        if step % check_interval == 0:
            actual_val = np.max(fft_convolve(a, a))
            if actual_val < best_val:
                best_val = actual_val
                best_a = a.copy()
            if step % (check_interval * 50) == 0:
                score = 2.0 * n * actual_val
                print(f"    MD step={step}, tau={tau:.4f}, max_conv={actual_val:.6e}, "
                      f"score={score:.6f}, best_score={2.0*n*best_val:.6f}, "
                      f"t={time.monotonic()-t0:.1f}s")
    
    # Final check
    final_val = np.max(fft_convolve(a, a))
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
    
    return best_a, best_val


def coordinate_polish(a_init, time_budget, seed=42):
    """
    Coordinate descent polishing on the simplex.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_val = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 3000
    
    while time.monotonic() - t0 < time_budget and no_improve < max_no_improve:
        r = rng.random()
        candidate = a.copy()
        
        if r < 0.35:
            # Pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.005, 0.1)
            amt = a[i] * frac
            candidate[i] -= amt
            candidate[j] += amt
        elif r < 0.55:
            # Single element scale
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.03))
            candidate[j] *= factor
        elif r < 0.75:
            # Block modification
            width = max(1, int(rng.integers(1, max(2, n // 32))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.02, size=width))
        else:
            # Peak reduction: reduce elements at the peak location
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Elements i where i and peak_idx-i both in range contribute
            contrib_indices = []
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contrib_indices.append(i)
            # Pick a random contributing element to reduce
            if contrib_indices:
                i = contrib_indices[rng.integers(len(contrib_indices))]
                candidate[i] *= (1 - rng.uniform(0.01, 0.05))
                # Redistribute to non-contributing elements
                non_contrib = np.setdiff1d(np.arange(n), contrib_indices)
                if len(non_contrib) > 0:
                    j = non_contrib[rng.integers(len(non_contrib))]
                    candidate[j] += a[i] - candidate[i]
        
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s
        
        new_val = np.max(fft_convolve(candidate, candidate))
        if new_val < best_val:
            best_val = new_val
            best_a = candidate.copy()
            a = candidate
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve % 500 == 0:
            print(f"    CD polish: no_improve={no_improve}, best_val={best_val:.6e}, "
                  f"score={2.0*n*best_val:.6f}, t={time.monotonic()-t0:.1f}s")
    
    return best_a, best_val


def make_init(n, a_ref, style, rng):
    """Create various initializations."""
    if style == 'ref':
        if a_ref is not None and len(a_ref) > 0:
            x_old = np.linspace(0, 1, len(a_ref))
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, a_ref)
        else:
            a = np.ones(n)
    elif style == 'uniform':
        a = np.ones(n)
    elif style == 'ramp':
        t = np.arange(n) / max(n-1, 1)
        a = 0.5 + 0.5 * t
    elif style == 'inv_ramp':
        t = np.arange(n) / max(n-1, 1)
        a = 0.5 + 0.5 * (1 - t)
    elif style == 'sinusoidal':
        t = np.arange(n) / max(n-1, 1)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * t)
    elif style == 'two_bump':
        t = np.arange(n) / max(n-1, 1)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 2 * t)
        a = np.clip(a, 0.1, None)
    elif style == 'half':
        a = np.ones(n)
        a[n//2:] *= 0.7
    elif style == 'geometric':
        r = 0.995
        a = r ** np.arange(n, dtype=np.float64)
    elif style == 'noise':
        a = 1.0 + 0.2 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    else:
        a = np.ones(n)
    
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 1e-6, None)
    return a / np.sum(a)

"""
Optimization of non-negative step functions to minimize peak self-convolution.
Uses mirror descent with smooth-max approximation and FFT-based gradient computation,
structured initializations, adaptive step sizes, and coordinate descent polishing.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def compute_smooth_grad(a, tau):
    """
    Compute smooth-max value and gradient for f(a) = max(conv(a,a)).
    Returns (smooth_value, gradient_vector_of_length_n).
    """
    n = len(a)
    conv = fft_convolve(a, a)
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / tau
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    smooth_val = max_conv + tau * np.log(sum_exp)

    # Gradient via FFT: (1/tau) * IFFT(2*real(FFT(w_padded)) * FFT(a_padded))[:n]
    L = 1
    while L < 3 * n - 2:
        L <<= 1
    w_padded = np.zeros(L)
    w_padded[:len(conv)] = w
    a_padded = np.zeros(L)
    a_padded[:n] = a

    fft_w = np.fft.rfft(w_padded)
    fft_a = np.fft.rfft(a_padded)
    grad = np.fft.irfft(2.0 * np.real(fft_w) * fft_a, L)[:n] / tau
    return smooth_val, grad


def mirror_descent_optimize(a_init, time_budget, tau_start=1.0, tau_end=0.005,
                            alpha=0.8, momentum=0.85, check_interval=20,
                            noise_period=2000, noise_scale=0.01, verbose=True):
    """
    Mirror descent optimization on the simplex with Nesterov-like momentum.
    a_init: initial point on simplex (sum=1, a>=0).
    Returns (best_a, best_max_conv).
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.clip(a, 1e-15, None)
    a = a / np.sum(a)

    L = 1
    while L < 3 * n - 2:
        L <<= 1

    best_a = a.copy()
    best_val = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    step = 0
    g_mom = np.zeros(n)
    a_prev = a.copy()
    stagnation = 0

    while time.monotonic() - t0 < time_budget:
        elapsed = min(1.0, (time.monotonic() - t0) / time_budget)
        tau = tau_start * (tau_end / tau_start) ** (elapsed ** 0.6)
        alpha_eff = alpha * (tau / tau_start) ** 0.4

        conv = fft_convolve(a, a)
        max_conv = np.max(conv)
        shifted = (conv - max_conv) / tau
        exp_vals = np.exp(shifted)
        sum_exp = np.sum(exp_vals)
        w = exp_vals / sum_exp

        w_padded = np.zeros(L)
        w_padded[:len(conv)] = w
        a_padded = np.zeros(L)
        a_padded[:n] = a

        fft_w = np.fft.rfft(w_padded)
        fft_a = np.fft.rfft(a_padded)
        grad = np.fft.irfft(2.0 * np.real(fft_w) * fft_a, L)[:n] / tau

        # Momentum
        g_mom = momentum * g_mom + (1.0 - momentum) * grad

        # Nesterov-like lookahead
        a_lookahead = a - (momentum / (1.0 + np.sqrt(1.0 + 4.0 * step))) * (a - a_prev)
        a_lookahead = np.clip(a_lookahead, 1e-15, None)
        a_lookahead = a_lookahead / np.sum(a_lookahead)

        # Mirror descent step
        step_size = alpha_eff * g_mom
        step_size = np.clip(step_size, -25.0, 25.0)
        a_new = a_lookahead * np.exp(-step_size)
        s = np.sum(a_new)
        if s < 1e-10 or not np.isfinite(s):
            a_new = np.ones(n) / n
            g_mom *= 0.5
        else:
            a_new = a_new / s

        a_prev = a
        a = a_new
        step += 1

        # Periodic noise injection to escape local minima
        if noise_period > 0 and step % noise_period == 0:
            a += noise_scale * np.random.standard_normal(n)
            a = np.clip(a, 1e-15, None)
            a = a / np.sum(a)

        if step % check_interval == 0:
            actual_val = np.max(fft_convolve(a, a))
            if actual_val < best_val:
                best_val = actual_val
                best_a = a.copy()
                stagnation = 0
            else:

            if verbose and step % (check_interval * 50) == 0:
                sc = 2.0 * n * actual_val

                      f"stag={stagnation}, t={time.monotonic()-t0:.1f}s")

    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


def coordinate_polish(a_init, time_budget, seed=42, verbose=True):
    """
    Coordinate descent / local search polishing on the simplex.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    a = a / np.sum(a)

    best_a = a.copy()
    best_val = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 5000
    it = 0

    while time.monotonic() - t0 < time_budget and no_improve < max_no_improve:
        r = rng.random()
        candidate = a.copy()

        if r < 0.30:
            # Pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.005, 0.08)
            amt = a[i] * frac
            candidate[i] -= amt
            candidate[j] += amt
        elif r < 0.50:
            # Single element scale
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.02))
            candidate[j] *= factor
        elif r < 0.70:
            # Block modification
            width = max(1, int(rng.integers(1, max(2, n // 64))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.015, size=width))
        elif r < 0.85:
            # Peak-targeted reduction
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Find contributing elements
            contrib = []
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contrib.append(i)
            if contrib:
                i = contrib[rng.integers(len(contrib))]
                red = rng.uniform(0.01, 0.04)
                candidate[i] *= (1.0 - red)
                # Redistribute
                non_contrib = np.setdiff1d(np.arange(n), contrib)
                if len(non_contrib) > 0:
                    n_add = min(3, len(non_contrib))
                    add_idx = rng.choice(non_contrib, size=n_add, replace=False)
                    removed = a[i] * red
                    candidate[add_idx] += removed / n_add
        else:
            # Random perturbation of a few elements
            n_mod = max(2, n // 100)
            idx = rng.choice(n, size=n_mod, replace=False)
            candidate[idx] *= np.exp(rng.normal(0, 0.01, size=n_mod))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s

        new_val = np.max(fft_convolve(candidate, candidate))
        if new_val < best_val:
            best_val = new_val
            best_a = candidate.copy()
            a = candidate
            no_improve = 0
        else:
            no_improve += 1
        it += 1

        if verbose and it % 2000 == 0:
            print(f"      CD n={n} it={it}, best_score={2.0*n*best_val:.6f}, "
                  f"no_impr={no_improve}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_val
