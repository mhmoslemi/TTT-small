from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def objective_normalized(a):
    """Compute max of self-convolution (with sum(a) = sqrt(2n) normalization)."""
    c = fft_convolve_self(a)
    return c.max()

    for _ in range(n_samples):

        if k == 1:
            boundaries = [n]
        else:
            # Sort k-1 random split points
            splits = sorted(rng.integers(1, n, size=k-1).tolist())

            pos += boundaries[i]

        score = objective_normalized(a)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
1. Multi-resolution initialization:
   - Optimize piecewise-constant templates at coarse resolution
   - Upsample to fine resolution for initialization
2. FISTA with adaptive log-sum-exp:
   - FFT-accelerated gradient in O(n log n)
   - Adaptive beta (smooth -> sharp)
   - Periodic random perturbations
3. Active-set Nesterov for polishing:
   - Focus on lags near the maximum
   - Adaptive step size
4. Multiple sequence lengths (4096, 8192, 16384)
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def grad_lse(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    bc = beta * (c - c_max)
    # Numerical stability
    bc = np.clip(bc, -500, 500)
    exp_bc = np.exp(bc)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    
    fsize = 1 << (2 * n - 1 + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def grad_active(a, c, n_active, threshold_frac=0.0):
    """Subgradient using active set (lags within threshold_frac of max)."""
    n = len(a)
    c_max = c.max()
    threshold = c_max * (1.0 - threshold_frac) if threshold_frac > 0 else c_max - 1e-10
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    if len(active) > n_active:
        # Keep the top n_active
        active_scores = c[active]
        top_idx = np.argsort(active_scores)[::-1][:n_active]
        active = active[top_idx]
    
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)
    
    fsize = 1 << (2 * n - 1 + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fista_optimize(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=300.0,
                    beta_rate=0.003, perturb_every=2000, perturb_prob=0.1):
    """FISTA with adaptive log-sum-exp."""
    a = safe_normalize(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = beta_init  # Lipschitz constant estimate (upper bound)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + beta_rate * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        grad = grad_lse(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (beta * 2.0)
        x_new = project_simplex(y - step * grad, S)
        
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-project y to simplex for stability

            y_sum = y.sum()
            if y_sum > 0:
                y = y * (S / y_sum)
            else:
                y = a.copy()
        a = x_new
        t = t_new
        
        # Periodic perturbation

            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
    
    return best_local, best_score
