```python
import numpy as np
import time

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def local_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s**2)
```

```python
def fft_convolve(a, b):

    while size < n + m - 1:

    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):

def softmax_normalize(theta):

def fft_phase_init(n, rng, n_iter=100):
    """Gerchberg-Saxton style initialization."""
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft <<= 1
    
    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def chirp_init(n, rng, alpha=0.5):

    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)

def random_structured_init(n, rng, n_tones=5):

def optimize_lbfgs(n, a_init, time_limit, tau=0.5):

    theta = np.log(a / S + 1e-12)
    
    t0 = time.monotonic()

        obj, grad_a, _ = soft_max_obj_grad(a_cur, tau)

def upsample(a, new_n, rng):
    """Upsample via interpolation + noise."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.015 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)

        inits.append(('fft80', a_fft1))
        
        a_fft2 = fft_phase_init(n, rng, n_iter=150)
        inits.append(('fft150', a_fft2))
        
        a_chirp = chirp_init(n, rng, alpha=0.3 + 0.5 * rng.random())
        inits.append(('chirp', a_chirp))
        
        a_mt = random_structured_init(n, rng, n_tones=3 + rng.integers(0, 5))
        inits.append(('multiline', a_mt))
        
        # From provided sequence if available
        if start_seq is not None and abs(n - len(start_seq)) < len(start_seq) // 2:
            a_from_seq = upsample(start_seq, n, rng)
            inits.append(('from_prev', a_from_seq))
        
        for name, a in inits:
            score = local_score(a)
            print(f"  n={n}, {name}: score={score:.6f}")
            candidates.append((score, n, a.copy(), name))
    
    candidates.sort(key=lambda x: x[0])
    print(f"\n  Top 5: {[(f'{c[0]:.4f}', c[1], c[3]) for c in candidates[:5]]}")
    
    # Phase 2: Gradient optimization (Adam) on best candidates
    print("\n=== Phase 2: Adam gradient optimization ===")
    n_cand = min(6, len(candidates))
    time_left = deadline - time.monotonic() - 180
    time_per = max(10.0, time_left / n_cand)
    
    optimized = []
    for i in range(n_cand):
        if time.monotonic() > deadline - 90:
            break
        score0, n, a, name = candidates[i]
        # Allocate more time to smaller n (faster iterations)
        t_alloc = time_per * min(1.5, max(0.3, 512.0 / n))
        t_alloc = max(8.0, min(t_alloc, deadline - time.monotonic() - 60))
        
        print(f"  [{i+1}/{n_cand}] n={n}, init={name}, score0={score0:.6f}, t={t_alloc:.1f}s")
        a_opt, s_opt = optimize_adam(n, a, t_alloc, rng)
        print(f"    -> score={s_opt:.6f}")
        
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")
        
        optimized.append((s_opt, n, a_opt, name))
    
    # Phase 3: L-BFGS-B refinement on best Adam results
    print("\n=== Phase 3: L-BFGS-B refinement ===")
    if optimized:
        optimized.sort(key=lambda x: x[0])
        for i in range(min(4, len(optimized))):
            if time.monotonic() > deadline - 60:
                break
            s0, n, a, name = optimized[i]
            t_alloc = min(60.0, max(10.0, deadline - time.monotonic() - 40))
            print(f"  L-BFGS on n={n}, score0={s0:.6f}, t={t_alloc:.1f}s")
            
            # Try multiple tau values
            for tau_mult in [1.0, 3.0, 10.0]:
                tau = tau_mult / np.sqrt(n)
                a_lb, s_lb = optimize_lbfgs(n, a, min(t_alloc / 3, deadline - time.monotonic() - 20), tau=tau)
                print(f"    tau={tau:.3f}: score={s_lb:.6f}")
                if s_lb < best_score:
                    best_score = s_lb
                    best_seq = a_lb.copy()
                    print(f"    *** New best: {best_score:.6f} (n={n}) ***")
    
    # Phase 4: Coarse-to-fine upsample
    print("\n=== Phase 4: Coarse-to-fine upsample ===")
    if optimized:
        optimized.sort(key=lambda x: x[0])
        best_short = optimized[0]
        target_ns = sorted(set([512, 1024, 2048, 4096]) - set([len(best_short[2])]))
        for target_n in target_ns:
            if time.monotonic() > deadline - 60:
                break
            if len(best_short[2]) >= target_n:
                continue
            a_up = upsample(best_short[2], target_n, rng)
            t_alloc = min(80.0, max(15.0, deadline - time.monotonic() - 50))
            t_alloc *= min(1.0, 1024.0 / target_n)
            t_alloc = max(10.0, t_alloc)
            print(f"  Upsample to n={target_n}, t={t_alloc:.1f}s")
            a_opt, s_opt = optimize_adam(target_n, a_up, t_alloc, rng)
            print(f"    -> score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} (n={target_n}) ***")
    
    # Phase 5: Peak-directed local search polish
    print(f"\n=== Phase 5: Local search polish ===")
    if best_seq is not None:
        polish_time = min(300.0, max(30.0, deadline - time.monotonic() - 30))
        print(f"  Polishing n={len(best_seq)}, score={best_score:.6f}, time={polish_time:.1f}s")
        polished, p_score = peak_directed_search(best_seq, polish_time, rng, n_top=8)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
    
    # Phase 6: Final Adam refinement on the best sequence found
    remaining = deadline - time.monotonic() - 25
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Phase 6: Final refinement ===")
        n_best = len(best_seq)
        a_init = best_seq * np.exp(rng.normal(0, 0.05, size=n_best))
        a_init = np.clip(a_init, 0.01, None)
        t_alloc = min(remaining - 10, 120.0)
        print(f"  Refining n={n_best}, t={t_alloc:.1f}s")
        a_ref, s_ref = optimize_adam(n_best, a_init, t_alloc, rng,
                                      tau_start=20.0/n_best, tau_end=0.05/n_best)
        if s_ref < best_score:
            best_score = s_ref
            best_seq = a_ref.copy()
            print(f"    *** New best: {best_score:.6f} ***")
        
        # Quick polish
        if deadline - time.monotonic() > 20:
            p2, s2 = peak_directed_search(best_seq, min(40.0, deadline - time.monotonic() - 15), rng)
            if s2 < best_score:
                best_score = s2
                best_seq = p2.copy()
                print(f"    After final polish: {best_score:.6f}")
    
    # Final verification
    print(f"\n{'='*50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        
        if final_score == np.inf or final_score > best_score + 0.01:
            print("WARNING: evaluate_sequence gave unexpected result!")
            # Fallback to previous best if available
            if start_seq is not None:
                return [float(max(0.0, min(1000.0, x))) for x in start_seq]
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
