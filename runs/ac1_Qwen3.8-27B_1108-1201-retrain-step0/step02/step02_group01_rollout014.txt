`min_w ||G(w)||^2` subject to `w >= 0`, `sum(w) = 1`, `supp(w) ⊆ active_set`

where `G(w)[m] = 2 * sum_k w[k] * a[k-m]`.

This is a QP in `w`! We can solve it with scipy.

Actually, `||G(w)||^2 = w^T (2*C) w` where `C` is the "Gram matrix" of the gradient functions: `C[i,j] = sum_m grad_c[i][m] * grad_c[j][m] = 4 * sum_m a[i-m]*a[j-m]`.

But this matrix is `n×n` (or `K×K` where K is the active set size), which could be large. For K=20, it's 20×20, which is fine.

Let me compute `C[i,j] = 4 * sum_m a[i-m] * a[j-m]` (sum over valid m). This is `4 * c_{|i-j|}`... wait, no.

`sum_m a[i-m]*a[j-m]` = cross-correlation of `a` with itself at lag `|i-j|` = `c[|i-j|]` (if we define `c` appropriately).

Actually, `sum_m a[m-i]*a[m-j]` (sum over valid m) is the autocorrelation of `a` at lag `|i-j|`. And `c[k] = sum_m a[m]*a[k-m]` is the "central" autocorrelation.

Hmm, these are related but not the same. Let me just compute `C` directly for small K.

For K active lags `k_1, ..., k_K`:
`C[i,j] = sum_m (2*a[k_i-m]) * (2*a[k_j-m]) = 4 * sum_m a[k_i-m] * a[k_j-m]`

The sum is over `m` where both `0 <= k_i-m < n` and `0 <= k_j-m < n`.

Let `d = k_i - k_j`. Then `a[k_i-m] * a[k_j-m] = a[k_j + d - m] * a[k_j-m]`. Let `l = k_j - m`: sum over `l` where `0 <= l < n` and `0 <= l+d < n`.

This is `sum_l a[l+d]*a[l]` which is the autocorrelation at lag `d = k_i - k_j`.

And `sum_l a[l+d]*a[l]` can be computed from the self-convolution: if `d >= 0`, it's `c[2k-j+d]`... hmm, let me just compute it directly for small K.

For K up to 50, computing the K×K matrix takes O(K * n) operations, which is fine.

OK, let me just write the code. I'll keep it practical and focus on what will actually work:

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog, nnls

def fft_convolve(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:out_len]

def fft_convolve_self(a):
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:2 * n - 1]

def get_a_from_u(u, S):
    """Softmax parameterization."""
    u_shifted = u - u.max()
    e = np.exp(u_shifted)
    total = e.sum()
    if total < 1e-300:
        return np.full(len(u), S / len(u))
    return S * e / total

def obj_and_grad_p(u, p, S):
    """
    F(u) = sum(c^p), grad w.r.t. u.
    a = S * softmax(u), c = conv(a,a)
    """
    a = get_a_from_u(u, S)
    c = fft_convolve_self(a)
    
    if p == 1:
        obj = c.sum()
        c_pow = np.ones_like(c)
    elif p == 2:
        obj = np.sum(c * c)
        c_pow = c.copy()
    elif p == 4:
        obj = np.sum(c**4)
        c_pow = c**3
    else:
        obj = np.sum(c**p)
        c_pow = c**(p-1)
    
    # grad_F/da[m] = 2p * sum_k c[k]^{p-1} * a[k-m]
    # = 2p * conv(c_pow, a_rev)[m + n - 1]
    n = len(a)
    a_rev = a[::-1]
    full = fft_convolve(c_pow, a_rev)
    grad_a = 2.0 * p * full[n-1:2*n-1]
    
    # Chain rule for softmax
    dot = np.dot(grad_a, a)
    grad_u = grad_a * a - (a / S) * dot
    
    return obj, grad_u

def subgradient_max(u, S, top_k=10):
    """
    Subgradient of max(c) w.r.t. u.
    Uses top-k lags with uniform weights.
    """
    a = get_a_from_u(u, S)
    c = fft_convolve_self(a)
    n = len(a)
    
    # Get top-k lags
    idx = np.argsort(c)[::-1][:top_k]
    w = np.zeros(len(c))
    w[idx] = 1.0 / top_k
    
    # grad_a[m] = 2 * sum_k w[k] * a[k-m]
    # = 2 * conv(w, a_rev)[m + n - 1]
    a_rev = a[::-1]
    full = fft_convolve(w, a_rev)
    grad_a = 2.0 * full[n-1:2*n-1]
    
    dot = np.dot(grad_a, a)
    grad_u = grad_a * a - (a / S) * dot
    
    return c.max(), grad_u

def solve_optimal_weights(a, c, active_lags):
    """Solve for optimal subgradient weights to make grad as flat as possible."""
    n = len(a)
    K = len(active_lags)
    
    # G[m] = 2 * sum_k w[k] * a[k-m]
    # We want G to be as flat as possible: minimize ||G - mean(G)||^2
    # G = A @ w where A[m, k] = 2 * a[active_lags[k] - m] (with zero-padding)
    
    # A has size n x K
    A = np.zeros((n, K))
    for k_idx, k in enumerate(active_lags):
        # A[m, k_idx] = 2 * a[k - m] for 0 <= k-m < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for m in range(lo, hi + 1):
            A[m, k_idx] = 2.0 * a[k - m]
    
    # Minimize ||A @ w - mean(A @ w)||^2 s.t. w >= 0, sum(w) = 1
    # = minimize ||(A - 1@A.mean(axis=1)) @ w||^2 (approximately)
    # Actually: mean(A@w) = mean(A) @ w (row means)
    # Let B = A - np.tile(A.mean(axis=1).reshape(-1,1), (1, K))
    # Then minimize ||B @ w||^2 s.t. w >= 0, sum(w) = 1
    
    B = A - A.mean(axis=1, keepdims=True)
    
    # min ||Bw||^2 = w^T B^T B w s.t. w >= 0, sum(w) = 1
    # This is a QP. For small K, solve directly.
    # Use Lagrangian: min w^T(B^T B)w + lambda(sum(w) - 1)
    # KKT: 2(B^T B)w + lambda*1 = 0 (for w > 0 components)
    #      2(B^T B)w + lambda*1 >= 0 (for w = 0 components)
    
    BB = B.T @ B  # K x K
    
    # Try all subsets (only feasible for small K)
    if K <= 15:
        best_val = np.inf
        best_w = np.ones(K) / K
        from itertools import combinations
        for r in range(1, K + 1):
            for subset in combinations(range(K), r):
                BB_s = BB[list(subset), list(subset)]
                try:
                    w_s = np.linalg.solve(2 * BB_s + np.eye(r) * 1e-10, np.ones(r))
                    # Check w_s >= 0
                    if np.all(w_s >= -1e-10):
                        w_s = np.maximum(w_s, 0)
                        w_s = w_s / w_s.sum()
                        val = w_s @ BB_s @ w_s
                        if val < best_val:
                            best_val = val
                            best_w = np.zeros(K)
                            best_w[list(subset)] = w_s
                except np.linalg.LinAlgError:
                    continue
        return best_w
    else:
        # Fallback: use NNLS or uniform
        # min 0.5*w^T BB w s.t. sum(w)=1, w>=0
        # Use scipy
        from scipy.optimize import minimize as sp_min
        res = sp_min(
            lambda w: 0.5 * w @ BB @ w,
            np.ones(K) / K,
            method='SLSQP',
            bounds=[(0, 1)] * K,
            constraints=[{'type': 'eq', 'fun': lambda w: w.sum() - 1}]
        )
        if res.success:
            return res.x
        return np.ones(K) / K

def _run_lbfgs_p(u_init, p, n, S, time_budget, max_iter=200):
    """Run L-BFGS-B on sum(c^p) objective."""
    state = {
        'best_true_obj': np.inf,
        'best_a': None,
        'start': time.monotonic(),
        'time_budget': time_budget
    }
    
    # Initialize
    a0 = get_a_from_u(u_init, S)
    c0 = fft_convolve_self(a0)
    state['best_a'] = a0.copy()
    state['best_true_obj'] = 2.0 * n * c0.max() / S**2
    
    def fun(u):
        state['elapsed'] = time.monotonic() - state['start']
        obj, grad = obj_and_grad_p(u, p, S)
        a = get_a_from_u(u, S)
        c = fft_convolve_self(a)
        true_obj = 2.0 * n * c.max() / S**2
        if true_obj < state['best_true_obj']:
            state['best_true_obj'] = true_obj
            state['best_a'] = a.copy()
        return obj, grad
    
    try:
        minimize(fun, u_init, jac=True, method='L-BFGS-B',
                 options={'maxiter': max_iter, 'maxfun': max_iter * 3, 'gtol': 1e-6})
    except Exception as e:
        print("L-BFGS p=%d error: %s" % (p, str(e)))
    
    return state['best_a'], state['best_true_obj']
