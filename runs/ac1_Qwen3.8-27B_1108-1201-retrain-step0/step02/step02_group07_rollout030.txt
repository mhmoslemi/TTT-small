import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_from_conv(conv, n):
    """Objective value from convolution, assuming sum(a)=1."""
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project vector onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_topk(conv, a, n_peaks=30, power=4.0):
    """Smoothed subgradient using top-K peaks with power weighting."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-30:
        return np.zeros(n)
    
    # Get top peaks
    if n_peaks >= len(conv):
        topk_idx = np.arange(len(conv))
    else:
        topk_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
    
    topk_vals = conv[topk_idx].copy()
    # Power weighting: sharper focus on max
    w = (topk_vals / max_c) ** power
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(w)
        w[0] = 1.0
    else:
        w /= w_sum
    
    sg = np.zeros(n)
    for idx in range(len(topk_idx)):
        wt = w[idx]
        if wt < 1e-12:
            continue
        k = topk_idx[idx]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += wt * a[k - je:k - js + 1][::-1]
    
    return 2.0 * sg

    a = np.maximum(a0, 0.0).copy()

        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = obj_from_conv(conv, n)
    best_obj = cur_obj
    best_a = a.copy()
    
    eta = eta_init
    patience = 0
    max_pat = 40
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient_topk(conv, a, n_peaks=n_peaks, power=power)
        # Center the subgradient
        sg_c = sg - sg.mean()
        
        # Multiplicative update
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 1e-15)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s

        if cur_obj < best_obj - 1e-10:

            best_a = a.copy()
            patience = 0
        else:

def pgd_optimize(a0, time_limit, lr_init=0.05, n_peaks=30, power=4.0, momentum=0.9):
    """Projected gradient descent with Nesterov momentum."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = obj_from_conv(conv, n)
    best_obj = cur_obj
    best_a = a.copy()
    
    lr = lr_init
    pat = momentum
    v = np.zeros(n)  # momentum
    patience = 0
    max_pat = 35
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        a_look = a - pat * v
        a_look = np.maximum(a_look, 0)
        a_look /= a_look.sum()
        conv_look = fast_convolve(a_look)
        sg = compute_subgradient_topk(conv_look, a_look, n_peaks=n_peaks, power=power)
        
        # Momentum update
        v = pat * v + lr * sg
        a_new = a - v
        a_new = project_simplex(a_new)
        
        if a_new.sum() < 0.5:
            a_new = 0.5 * a + 0.5 * project_simplex(a)
        
        a = a_new
        conv = fast_convolve(a)
        cur_obj = obj_from_conv(conv, n)
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_pat:
            patience = 0
            lr *= 0.65
            if lr < 0.003:
                lr = lr_init * 0.7
                v = np.zeros(n)
    
    return best_a, best_obj


def anneal_optimize(a0, time_limit, rng, mode="informed"):
    """Simulated annealing with peak-targeted moves."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = obj_from_conv(conv, n)
    best_obj = cur_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    steps = 0
    
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.01 * (1.0 - frac) ** 2 + 0.0003
        
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        if mode == "informed":
            # Target high peaks
            peak_thresh = max_c * 0.8
            peak_positions = np.where(conv >= peak_thresh)[0]
            if len(peak_positions) == 0:
                peak_positions = np.array([np.argmax(conv)])
            k = int(rng.choice(peak_positions))
            
            # Find contributing elements
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            seg_a = a[js:je + 1]
            seg_b = a[k - je:k - js + 1][::-1]
            products = seg_a * seg_b
            total_prod = products.sum()
            if total_prod < 1e-15:
                continue
            probs = products / total_prod
            top_j = js + int(rng.choice(je - js + 1, p=probs))
        else:
            top_j = int(rng.integers(n))
        
        # Choose destination: prefer low-contribution positions
        target_j = int(rng.integers(n))
        
        if target_j == top_j:
            continue
        
        delta = a[top_j] * rng.uniform(0.01, 0.05)
        if delta < 1e-10:
            continue
        
        a[top_j] -= delta
        a[target_j] += delta
        
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        
        new_conv = fast_convolve(a)
        new_obj = obj_from_conv(new_conv, n)
        delta_obj = new_obj - cur_obj
        
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
        
        steps += 1
        # Time check every 100 steps
        if steps % 200 == 0 and time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_obj


def spectral_perturb(a, rng, strength=0.05):
    """Perturb in frequency domain for diversity."""
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    
    # Perturb magnitudes slightly
    phase = np.angle(A)
    mag = np.abs(A)
    
    # Add small random perturbation to magnitudes
    noise = 1.0 + strength * rng.randn(len(mag))
    mag_new = mag * np.maximum(noise, 0.5)
    
    # Reconstruct
    A_new = mag_new * np.exp(1j * phase)
    A_new[0] = mag[0]  # Keep DC component (sum) fixed
    
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    a_new /= s
    return a_new

def structured_init(n, rng, parent=None):

    # Bimodal

    inits.append(bi / bi.sum())
    
    # Sinusoidal

    inits.append(s / s.sum())
    
    # Dirichlet with different alphas

        inits.append(random_init(n, rng, alpha))
    
    # Random with structure
    r = 0.5 + 0.5 * rng.randn(n)
    r = np.maximum(r, 0.01)
    inits.append(r / r.sum())
    
    # Parent interpolation if available
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0).copy()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            inits.append(pp / pp.sum())
        # Perturbed parent

        pp2 = np.maximum(pp2, 0)
        if pp2.sum() > 1e-10:
            inits.append(pp2 / pp2.sum())
    
    return inits

    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Multi-phase optimizer focusing on longer sequences
    with peak-equalization mirror descent, momentum PGD, and spectral
    perturbation annealing.
    """
    global _best_sequence, _best_score
    _best_sequence = None
    _best_score = float('inf')
    
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    
    def record(a):
        nonlocal _best_sequence, _best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < _best_score:
            _best_score = score
            _best_sequence = a.copy()
            print("  NEW BEST: score=%.6f, n=%d, t=%.0fs" % (score, len(a), time.monotonic() - t_start))
    
    parent = get_parent()
    
    # Phase 1: Quick scan across lengths
    print("=" * 60)
    print("Phase 1: Multi-length scan")
    print("=" * 60)
    
    scan_lengths = [256, 512, 1024, 2048, 4096]
    # If we have parent, prioritize its length and neighbors
    if parent is not None:
        pl = len(parent)
        scan_lengths = [pl] + [l for l in scan_lengths if l != pl]
        # Ensure we have a good spread
        if 1024 not in scan_lengths:
            scan_lengths.append(1024)
    
    phase1_budget = min(400.0, (deadline - time.monotonic()) * 0.15)
    per_length = phase1_budget / len(scan_lengths)
    
    scan_results = []
    for n in scan_lengths:
        if time.monotonic() > deadline - 1800:
            break
        if per_length < 5:
            continue
        
        inits = structured_init(n, rng, parent)
        # Limit inits to save time
        max_inits = min(len(inits), max(2, int(per_length / 15)))
        
        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline - 1800:
                break
            t_alloc = min(20.0, per_length / max(1, max_inits))
            if t_alloc < 3:
                break
            
            # Quick MD
            a_opt, _ = mirror_descent_opt(a0, time_limit=t_alloc, eta_init=0.4, n_peaks=20, power=3.0)
            conv = fast_convolve(a_opt)
            obj = obj_from_conv(conv, len(a_opt))
            scan_results.append((obj, n, a_opt.copy()))
            record(a_opt)
    
    scan_results.sort(key=lambda x: x[0])
    print("  Scan done. Top 5:")
    for obj, n, _ in scan_results[:5]:
        print("    n=%d, obj=%.6f" % (n, obj))
    print("  Current best: %.6f" % _best_score)
    
    # Phase 2: Deep optimization on top candidates
    print("=" * 60)
    print("Phase 2: Deep optimization")
    print("=" * 60)
    
    # Select diverse top candidates
    top_candidates = []
    seen_lengths = set()
    for obj, n, a in scan_results:
        if n not in seen_lengths and len(top_candidates) < 6:
            top_candidates.append((n, a))
            seen_lengths.add(n)
    
    # Always include best sequence
    if _best_sequence is not None:
        bl = len(_best_sequence)
        if bl not in seen_lengths:
            top_candidates.append((bl, _best_sequence.copy()))
    
    remaining = deadline - time.monotonic() - 600
    phase2_budget = max(200.0, min(1200.0, remaining * 0.6))
    per_cand = phase2_budget / max(1, len(top_candidates))
    
    for idx, (n, a0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 600:
            break
        t_alloc = min(per_cand, deadline - time.monotonic() - 600)
        if t_alloc < 10:
            break
        
        print("  [%d/%d] n=%d, budget=%.0fs" % (idx + 1, len(top_candidates), n, t_alloc))
        
        a_cur = a0.copy()
        sub = t_alloc / 6.0
        
        # Alternating MD and PGD with decreasing parameters
        configs = [
            (mirror_descent_opt, {'eta_init': 0.3, 'n_peaks': 30, 'power': 4.0}),
            (pgd_optimize, {'lr_init': 0.05, 'n_peaks': 30, 'power': 4.0, 'momentum': 0.9}),
            (mirror_descent_opt, {'eta_init': 0.15, 'n_peaks': 50, 'power': 6.0}),
            (pgd_optimize, {'lr_init': 0.03, 'n_peaks': 50, 'power': 6.0, 'momentum': 0.85}),
            (mirror_descent_opt, {'eta_init': 0.08, 'n_peaks': 80, 'power': 8.0}),
            (pgd_optimize, {'lr_init': 0.02, 'n_peaks': 80, 'power': 8.0, 'momentum': 0.8}),
        ]
        
        for func, params in configs:
            if time.monotonic() > deadline - 600:
                break
            t_sub = min(sub, deadline - time.monotonic() - 600)
            if t_sub < 2:
                break
            a_cur, _ = func(a_cur, time_limit=t_sub, **params)
            record(a_cur)
    
    print("  Deep opt done. Best: %.6f" % _best_score)
    
    # Phase 3: Intensive local search on best
    print("=" * 60)
    print("Phase 3: Intensive local search")
    print("=" * 60)
    
    if _best_sequence is not None and time.monotonic() < deadline - 100:
        phase3_budget = min(500.0, deadline - time.monotonic() - 100)
        a_best = _best_sequence.copy()
        
        # Multiple annealing rounds with different modes
        n_rounds = 4
        per_round = phase3_budget / n_rounds
        
        for i in range(n_rounds):
            if time.monotonic() > deadline - 100:
                break
            rng_i = np.random.default_rng(seed + 10000 + i * 7)
            a_start = a_best.copy()
            
            # Occasionally apply spectral perturbation for diversity
            if i == 1 or i == 3:
                a_start = spectral_perturb(a_best, rng_i, strength=0.03)
            
            mode = "informed" if i % 2 == 0 else "random"
            a_opt, _ = anneal_optimize(a_start, time_limit=per_round, rng=rng_i, mode=mode)
            record(a_opt)
            
            # Follow up with short MD
            if time.monotonic() < deadline - 100:
                a_opt2, _ = mirror_descent_opt(a_opt, time_limit=min(30.0, per_round * 0.3),
                                               eta_init=0.1, n_peaks=50, power=6.0)
                record(a_opt2)
    
    # Phase 4: Final polish
    print("=" * 60)
    print("Phase 4: Final polish")
    print("=" * 60)
    
    if _best_sequence is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 5
        a_best = _best_sequence.copy()
        
        # Very fine MD
        t_md = remaining * 0.4
        if t_md > 5:
            a_opt, _ = mirror_descent_opt(a_best, time_limit=t_md,
                                          eta_init=0.05, n_peaks=100, power=10.0)
            record(a_opt)
        
        if time.monotonic() < deadline - 5:
            t_pgd = min(remaining * 0.3, deadline - time.monotonic() - 5)
            if t_pgd > 3:
                a_opt2, _ = pgd_optimize(a_best, time_limit=t_pgd,
                                         lr_init=0.01, n_peaks=100, power=10.0, momentum=0.7)
                record(a_opt2)
        
        # Final short anneal
        if time.monotonic() < deadline - 3:
            t_an = min(remaining * 0.2, deadline - time.monotonic() - 3)
            if t_an > 2:
                rng_f = np.random.default_rng(seed + 99999)
                a_opt3, _ = anneal_optimize(_best_sequence, time_limit=t_an,
                                            rng=rng_f, mode="informed")
                record(a_opt3)
    
    # Fallback
    if _best_sequence is None:
        print("  WARNING: No valid sequence found, using fallback!")
        _best_sequence = np.ones(256)
        _best_score = safe_evaluate(_best_sequence)
    
    print("=" * 60)
    print("FINAL SCORE: %.6f" % _best_score)
    print("Sequence length: %d" % len(_best_sequence))
    print("=" * 60)
    
    return [float(x) for x in _best_sequence]
