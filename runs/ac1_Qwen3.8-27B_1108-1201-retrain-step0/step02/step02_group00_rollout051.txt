    else:
        rho = int(np.max(np.where(valid)[0]) + 1)

    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_grad(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def toppk_grad(a, c, top_k):
    """Gradient using top-k active lags via FFT."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

Strategy:
1. Segment-based parameterization: divide sequence into k constant-height segments,
   optimize via Nesterov with log-sum-exp smoothing.
2. Start from given sequence and structured initializations.

4. Multi-length exploration for diversity.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_grad(a, c, beta):

    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def toppk_grad(a, c, top_k):
    """Gradient using top-k active lags via FFT."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=300.0,
              use_backtrack=True):
    """FISTA with adaptive log-sum-exp smoothing and optional backtracking."""
    a = np.clip(a_init, 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    x = a.copy()
    y = a.copy()
    t = 1.0
    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    step = 1.0 / (beta_init * 2.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))

        c_y = fft_convolve_self(y)
        cur = c_y.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()

        grad = lse_grad(y, c_y, beta)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        if use_backtrack:
            # Backtracking line search
            obj_y = c_y.max()
            step_bt = step
            for _ in range(10):
                x_candidate = project_simplex(y - step_bt * grad, S)
                c_x = fft_convolve_self(x_candidate)
                obj_x = c_x.max()
                # Check descent: f(x_new) <= f(y) - (1/(2*step))||x_new - y||^2
                diff = obj_y - obj_x
                quad = (1.0 / (2.0 * step_bt)) * np.sum((x_candidate - y) ** 2)
                if diff >= quad * 0.5:
                    break
                step_bt *= 0.5
            step = step_bt
            x_new = x_candidate
        else:
            step = 1.0 / (beta * gmax + 1e-15)
            x_new = project_simplex(y - step * grad, S)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0.0)
        # Renormalize to maintain sum constraint
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        x = x_new
        t = t_new

        # Periodic restart with noise
        if it % 3000 == 0 and rng.random() < 0.12:
            noise_scale = 0.004 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            x = y.copy()
            t = 1.0
            step = 1.0 / (beta * 2.0)

    return best_a, best_score


def run_nesterov_toppk(a_init, n, S, max_time, rng,
                       top_k_init=5, top_k_max=256, ramp_rate=100):
    """Nesterov with top-k subgradient, ramping top_k over time."""
    a = np.clip(a_init, 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    x = a.copy()
    y = a.copy()
    t = 1.0
    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // ramp_rate)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()

        grad = toppk_grad(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Polyak-type step
        step = 2.0 * (cur - 0.5) / (gnorm * gnorm + 1e-15)
        step = min(step, 10.0 * S / gnorm)
        step = max(step, 1e-8)

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        x = x_new
        t = t_new

        if it % 4000 == 0 and rng.random() < 0.08:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            x = y.copy()
            t = 1.0

    return best_a, best_score


def run_segments(h_init, k, n, max_time, rng, beta_init=5.0, beta_max=150.0):
    """
    Optimize k segment heights for a sequence of length n.
    Returns (best_heights, best_score).
    """
    L = n // k
    actual_n = L * k
    S_actual = np.sqrt(2.0 * float(actual_n))
    S_h = S_actual / L  # target sum of segment heights

    h = np.clip(h_init, 0, 1000).copy()
    h = h * (S_h / max(h.sum(), 1e-15))

    x = h.copy()
    y = h.copy()
    t = 1.0
    best_score = np.inf
    best_h = h.copy()
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.02 * it))

        a = np.repeat(y, L)
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_h = y.copy()

        grad_a = lse_grad(a, c, beta)
        # Average gradient over each segment
        grad_h = np.array([grad_a[j*L:(j+1)*L].mean() for j in range(k)])
        gmax = np.abs(grad_h).max()
        if gmax < 1e-14:
            break

        step = 1.0 / (beta * gmax + 1e-15)
        x_new = project_simplex(y - step * grad_h, S_h)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S_h / s_y)
        x = x_new
        t = t_new

        if it % 500 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.05 * S_h / np.sqrt(k), k), S_h)
            x = y.copy()
            t = 1.0

    return best_h, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses segment-based search + full-resolution FISTA/Nesterov polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Get given sequence
    given_seq = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given_seq = np.clip(given, 0, 1000)
    except Exception:
        pass

    # === Phase 1: Intensive polishing from given sequence ===
    if given_seq is not None:
        print("=== Phase 1: Polish given sequence (n=%d) ===" % len(given_seq))
        n = len(given_seq)
        S = np.sqrt(2.0 * float(n))
        a_norm = given_seq * (S / given_seq.sum())

        c0 = fft_convolve_self(a_norm)
        sc0 = c0.max()
        print("  Initial score: %.6f" % sc0)
        global_best = min(global_best, sc0)
        global_a = a_norm.copy()

        # Sub-phase 1a: FISTA with low beta (smooth)
        rem = deadline - time.monotonic()
        seg_time = min(rem * 0.2, 300.0)
        print("  FISTA smooth (beta 5->100): %.0fs" % seg_time)
        a1, s1 = run_fista(a_norm, n, S, seg_time, rng,
                           beta_init=5.0, beta_max=100.0, use_backtrack=True)
        c1 = fft_convolve_self(a1)
        if c1.max() < global_best:
            global_best = c1.max()
            global_a = a1.copy()
            print("  After FISTA smooth: %.6f" % global_best)

        # Sub-phase 1b: FISTA with high beta (sharp)
        rem = deadline - time.monotonic()
        seg_time = min(rem * 0.25, 400.0)
        print("  FISTA sharp (beta 50->300): %.0fs" % seg_time)
        a2, s2 = run_fista(global_a, n, S, seg_time, rng,
                           beta_init=50.0, beta_max=300.0, use_backtrack=False)
        c2 = fft_convolve_self(a2)
        if c2.max() < global_best:
            global_best = c2.max()
            global_a = a2.copy()
            print("  After FISTA sharp: %.6f" % global_best)

        # Sub-phase 1c: Top-k Nesterov with ramping k
        rem = deadline - time.monotonic()
        seg_time = min(rem * 0.3, 450.0)
        print("  Top-k Nesterov (k 10->256): %.0fs" % seg_time)
        a3, s3 = run_nesterov_toppk(global_a, n, S, seg_time, rng,
                                     top_k_init=10, top_k_max=256, ramp_rate=80)
        c3 = fft_convolve_self(a3)
        if c3.max() < global_best:
            global_best = c3.max()
            global_a = a3.copy()
            print("  After Top-k Nesterov: %.6f" % global_best)

        # Sub-phase 1d: Final high-k polish
        rem = deadline - time.monotonic()
        if rem > 60:
            seg_time = min(rem * 0.4, 300.0)
            print("  Final polish (k 50->512): %.0fs" % seg_time)
            a4, s4 = run_nesterov_toppk(global_a, n, S, seg_time, rng,
                                         top_k_init=50, top_k_max=512, ramp_rate=50)
            c4 = fft_convolve_self(a4)
            if c4.max() < global_best:
                global_best = c4.max()
                global_a = a4.copy()
                print("  After final polish: %.6f" % global_best)

    else:
        print("No given sequence available, starting from scratch.")
        # Use a default starting point
        n = 8192
        global_a = np.ones(n)
        global_best = 2.0

    print("After Phase 1: global_best = %.6f" % global_best)

    # === Phase 2: Segment-based exploration for better structure ===
    print("=== Phase 2: Segment-based exploration ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        phase2_budget = min(rem * 0.3, 500.0)
        n_main = len(global_a) if global_a is not None else 8192

        # Try segment-based optimization at the main length
        k_values = [3, 4, 5, 6, 8, 10, 12, 16, 20]
        per_seg = phase2_budget / (len(k_values) * 2)

        found_better = False
        for k in k_values:
            if time.monotonic() > deadline - 120:
                break
            if n_main // k < 8:
                continue

            # Two different initializations
            for trial in range(2):
                if time.monotonic() > deadline - 120:
                    break

                if trial == 0:
                    h_init = np.linspace(0.2, 1.0, k)
                else:
                    # Try a "bump" shape
                    tt = np.linspace(0, 1, k)
                    h_init = 0.5 + 0.5 * np.exp(-4.0 * (tt - 0.4) ** 2)

                seg_time = min(per_seg, 20.0)
                h_opt, seg_score = run_segments(h_init, k, n_main, seg_time, rng)

                L = n_main // k
                actual_n = L * k
                a_seg = np.repeat(h_opt, L)
                S_act = np.sqrt(2.0 * float(actual_n))
                a_seg = a_seg * (S_act / max(a_seg.sum(), 1e-15))

                c_seg = fft_convolve_self(a_seg)
                sc_seg = c_seg.max()
                if sc_seg < global_best:
                    global_best = sc_seg
                    global_a = a_seg.copy()
                    found_better = True
                    print("  Segment k=%d: %.6f" % (k, global_best))

        # If we found a better segment structure, polish it
        if found_better:
            n = len(global_a)
            S = np.sqrt(2.0 * float(n))
            rem = deadline - time.monotonic()
            if rem > 60:
                seg_time = min(rem * 0.5, 300.0)
                a_pol, s_pol = run_fista(global_a, n, S, seg_time, rng,
                                          beta_init=20.0, beta_max=300.0)
                c_pol = fft_convolve_self(a_pol)
                if c_pol.max() < global_best:
                    global_best = c_pol.max()
                    global_a = a_pol.copy()
                    print("  Polished segment: %.6f" % global_best)

    # === Phase 3: Try different lengths ===
    print("=== Phase 3: Multi-length exploration ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        alt_lengths = [4096, 2048]
        if n_main not in alt_lengths and n_main <= 16384:
            pass  # skip main length

        per_alt = min(rem * 0.25, 200.0)

        for n_alt in alt_lengths:
            if time.monotonic() > deadline - 60:
                break
            S_alt = np.sqrt(2.0 * float(n_alt))
            tt = np.linspace(0, 1, n_alt, endpoint=False)

            # Structured initialization
            a_init = 0.5 + 0.3 * np.cos(2 * np.pi * tt) + 0.2 * np.sin(4 * np.pi * tt)
            a_init = np.maximum(a_init, 0.01)
            a_init = a_init * (S_alt / a_init.sum())

            seg_time = min(per_alt, 100.0)
            a_alt, s_alt = run_fista(a_init, n_alt, S_alt, seg_time, rng,
                                      beta_init=10.0, beta_max=200.0)
            c_alt = fft_convolve_self(a_alt)
            if c_alt.max() < global_best:
                global_best = c_alt.max()
                global_a = a_alt.copy()
                print("  Alt length %d: %.6f" % (n_alt, global_best))

    # === Phase 4: Final intensive polish ===
    print("=== Phase 4: Final polish ===")
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        rem = deadline - time.monotonic()

        if rem > 30:
            # One more round of top-k Nesterov with very high k
            seg_time = min(rem * 0.6, 200.0)
            a_fin, s_fin = run_nesterov_toppk(global_a, n, S, seg_time, rng,
                                               top_k_init=100, top_k_max=512,
                                               ramp_rate=30)
            c_fin = fft_convolve_self(a_fin)
            if c_fin.max() < global_best:
                global_best = c_fin.max()
                global_a = a_fin.copy()
                print("  Final Nesterov: %.6f" % global_best)

            rem = deadline - time.monotonic()
            if rem > 20:
                # Quick FISTA with very high beta
                seg_time = min(rem * 0.7, 100.0)
                a_fin2, s_fin2 = run_fista(global_a, n, S, seg_time, rng,
                                            beta_init=200.0, beta_max=500.0,
                                            use_backtrack=False)
                c_fin2 = fft_convolve_self(a_fin2)
                if c_fin2.max() < global_best:
                    global_best = c_fin2.max()
                    global_a = a_fin2.copy()
                    print("  Final FISTA: %.6f" % global_best)

    # === Verification and return ===
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL ===")
    print("Internal score: %.6f" % global_best)
    print("evaluate_sequence: %.6f" % ver if np.isfinite(ver) else "REJECTED")
    print("Length: %d" % len(global_a))

    if not np.isfinite(ver):
        print("Fallback to constant sequence")
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]

    return final_list
