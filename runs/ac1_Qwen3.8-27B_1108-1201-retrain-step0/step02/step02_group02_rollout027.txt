    if S < 0.01:

    c = fft_convolve(a, a)

def solve_lp(f, M, n):
    """Solve the LP: max sum(b) subject to (f*b)[k] <= M for all k, b >= 0."""
    # c = -ones (we want to maximize sum, linprog minimizes)
    c = -np.ones(n)
    
    # A_ub: (2n-1) x n
    # A_ub[k, i] = f[k-i] if max(0, k-n+1) <= i <= min(k, n-1)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            A_ub[k, i] = f[k - i]
    
    b_ub = np.full(2*n-1, M)
    
    # b >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        return None

"""
LP-based peak reduction for minimizing autocorrelation peak of non-negative sequences.
Uses the state-of-the-art LP iteration with line search, combined with local
polishing and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def build_A(f, n):
    """Build constraint matrix: (A @ b)[k] = (f * b)[k] for linear convolution."""
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A

def lp_iteration(f, M, A):
    """Solve LP: max sum(b) s.t. A @ b <= M, b >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_search(f_init, time_limit, rng, verbose=True):
    """
    State-of-the-art LP-based peak reduction.
    """
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sum(f)
    
    # Normalize
    if S < 0.01:
        f = np.ones(n)
        S = n
    
    t0 = time.monotonic()
    best_f = f.copy()
    best_score = local_score(f)
    
    if verbose:
        print(f"  LP search start: n={n}, score={best_score:.6f}")
    
    A = build_A(f, n)  # Will be rebuilt when f changes
    
    stuck_count = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        
        # Recompute A since f changes
        A = build_A(f, n)
        
        c_ff = fft_convolve(f, f)
        M = np.max(c_ff)
        
        if M <= 0:
            break
        
        b = lp_iteration(f, M, A)
        if b is None:
            # LP failed, try perturbation
            f = np.clip(f + 0.01 * rng.standard_normal(n), 0, None)
            continue
        
        # Rescale b to match sum of f
        sum_b = np.sum(b)
        if sum_b < 1e-10:
            continue
        b = b * S / sum_b
        
        # Line search on t
        best_t = 0
        best_t_score = local_score(f)
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            f_trial = (1 - t) * f + t * b
            f_trial = np.clip(f_trial, 0, None)
            s = local_score(f_trial)
            if s < best_t_score:
                best_t_score = s
                best_t = t
        
        if best_t > 0:
            f = (1 - best_t) * f + best_t * b
            f = np.clip(f, 0, None)
            S = np.sum(f)
            
            score = local_score(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
                stuck_count = 0
            else:
                stuck_count += 1
        else:
            stuck_count += 1
        
        if stuck_count % 20 == 0 and stuck_count > 0:
            # Perturbation to escape
            if verbose and stuck_count % 100 == 0:
                print(f"    stuck={stuck_count}, score={local_score(f):.6f}, best={best_score:.6f}, t={elapsed:.0f}s")
            f = best_f + 0.05 * rng.standard_normal(n)
            f = np.clip(f, 0, None)
            S = np.sum(f)
        elif stuck_count > 5:
            # Reset to best
            f = best_f.copy()
            S = np.sum(f)
            stuck_count = 0
            # Small perturbation
            f[int(rng.integers(0, n))] *= 1.05
            S = np.sum(f)
        
        if verbose and int(elapsed) % 60 < 1:
            print(f"    t={elapsed:.0f}s, score={local_score(f):.6f}, best={best_score:.6f}")
    
    return best_f, best_score


def polish_sequence(f_init, time_limit, rng):
    """Local polishing using targeted peak reduction."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    t0 = time.monotonic()
    best_f = f.copy()
    best_score = local_score(f)
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(f, f)
        peak_idx = int(np.argmax(c))
        
        # Find elements contributing to the peak
        contribs = np.zeros(n)
        for i in range(n):
            if 0 <= peak_idx - i < n:
                contribs[i] = f[i] * f[peak_idx - i]
        
        r = rng.random()
        if r < 0.4:
            # Reduce top contributors
            top_k = max(3, n // 20)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            factor = rng.uniform(0.92, 0.98)

LP-based peak reduction with gradient refinement for minimizing autocorrelation
peak of non-negative sequences. Combines convex LP iterations (state-of-the-art 
approach) with smooth gradient descent and local polishing.

import numpy as np
import time
from scipy.optimize import linprog, minimize

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i+n, i] = f
    return A

def lp_solve(f, M, A, n):
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None
