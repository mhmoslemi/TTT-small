import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT, length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_subgrad(a, w):
    """Compute 2 * cross-correlation of w (length 2n-1) with a (length n)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def subgrad_toppk(a, c, top_k):
    """Subgradient via top-k lags weighted by their values."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = np.maximum(c[top_idx], 0)
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = top_vals[i] / total
    return fft_subgrad(a, w_full)


def subgrad_lse(a, c, beta):
    """Subgradient via log-sum-exp weighting."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    return fft_subgrad(a, w)


def subgrad_power(a, c, p):
    """Subgradient via power weighting."""
    n = len(a)
    c_pos = np.maximum(c, 1e-10)
    cp = c_pos ** p
    total = cp.sum()
    if total < 1e-30:
        return np.zeros(n)
    w = cp / total
    return fft_subgrad(a, w)


def fista_smooth(a0, n, S, max_time, rng, mode='lse', beta_init=5.0, beta_max=500.0,
                 top_k_init=10, top_k_max=512, p_init=16, p_max=1024):
    """
    FISTA with adaptive smoothing.
    mode: 'lse', 'topk', 'power', or 'adaptive'
    """
    a = np.clip(a0, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 0.5  # 1/L where L ≈ 2
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Choose smoothing parameters
        if mode == 'adaptive':
            if it < 500:
                sm = 'lse'
            elif it < 2000:
                sm = 'topk'
            else:
                sm = 'topk'
        else:
            sm = mode
        
        if sm == 'lse':
            beta = min(beta_max, beta_init * (1.0 + 0.01 * it))
            c = fft_convolve_self(y)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = y.copy()
            grad = subgrad_lse(y, c, beta)
        elif sm == 'topk':
            top_k = min(top_k_max, top_k_init + it // 50)
            c = fft_convolve_self(y)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = y.copy()
            grad = subgrad_toppk(y, c, top_k)
        else:  # power
            p = min(p_max, p_init * (1 + it // 100))
            c = fft_convolve_self(y)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = y.copy()
            grad = subgrad_power(y, c, p)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # FISTA step
        x_new = project_simplex(y - step * grad, S)
        
        # Backtracking: check if objective improved
        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Adaptive step: increase if things are going well
        if it % 100 == 0:
            # Check if we've been improving
            pass
        
        # Random restart to escape local minima
        if it % 5000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score


def peak_shaving(a0, n, S, max_time, rng, n_inner=10, n_outer=50):
    """
    Coordinate descent that targets the peak lags.
    For each peak lag, move mass from high-partner positions to low-partner positions.
    """
    a = np.clip(a0, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    outer = 0
    while time.monotonic() - start < max_time:
        outer += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Find top lags
        n_peaks = min(5, 2 * n - 1)
        peak_lags = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[peak_lags]
        
        # For each peak lag, do coordinate updates
        for n_iter in range(n_inner):
            if time.monotonic() - start > max_time:
                break
            
            # Use the weighted subgradient direction
            w_full = np.zeros(2 * n - 1)
            total = peak_vals.sum()
            for i, k in enumerate(peak_lags):
                w_full[int(k)] = peak_vals[i] / total
            
            grad = fft_subgrad(a, w_full)
            
            # Find best coordinate move: move from high-grad to low-grad
            # (high grad means this coordinate contributes most to the peaks)
            i_src = np.argmax(grad)
            i_dst = np.argmin(grad[np.where(a > 0.01 * S / n)[0]]) if np.any(a > 0.01 * S / n) else np.argmin(grad)
            
            if i_src == i_dst:
                break
            
            # Maximum transfer: limited by a[i_src] and the gradient difference
            delta = a[i_src] * 0.1  # transfer 10% of source
            if delta <= 0:
                break
            
            a_copy = a.copy()
            a_copy[i_src] -= delta
            a_copy[i_dst] += delta
            
            # Check improvement
            c_new = fft_convolve_self(a_copy)
            if c_new.max() < cur:
                a = a_copy
                cur = c_new.max()
                if cur < best_score:
                    best_score = cur
                    best_local = a.copy()
            else:
                # Try smaller delta
                delta *= 0.5
                a_copy = a.copy()
                a_copy[i_src] -= delta
                a_copy[i_dst] += delta
                c_new = fft_convolve_self(a_copy)
                if c_new.max() < cur:
                    a = a_copy
                    cur = c_new.max()
                    if cur < best_score:
                        best_score = cur
                        best_local = a.copy()
        
        # Periodic random perturbation
        if outer % 20 == 0 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.02 * S / np.sqrt(n), n), S)
    
    return best_local, best_score


def combine_solutions(a1, a2, n, S):
    """Combine two solutions via element-wise min and renormalize."""
    c = np.minimum(a1, a2)
    s = c.sum()
    if s < 1e-15:
        return a1.copy()
    return c * (S / s)


def safe_normalize(a, n):
    """Clamp and normalize to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
