            best_score = score
            best_candidate = new_seq.tolist()
            print("New best found: score=%.6f" % best_score)
        else:
            # Check if interpolation improves results
            enhanced_score = anneal_improved(new_seq, initial_pop, best_score)  # New annealing function
            if enhanced_score < best_score:
                best_score = enhanced_score
                best_candidate = new_seq.tolist()
        
        # Scale check against the overall deadline
        if time.monotonic() > deadline - 5:
            # Final sanity check and normalization
            # We have to do full updates because the objective function requires it
            # Post-procedure: normalize the candidate's representation
            best_candidate = np.array(best_candidate) / np.sum(best_candidate)  # Scale to sum=1
            
            # Final quality boost: Small perturbation to push through minima
            # High-dimension optimization step
            # Clean up the sequence
            conv_final = np.convolve(best_candidate, best_candidate, mode='full')
            max_final = conv_final.max()
            final_score = float((2 * len(best_candidate) * max_final) / (best_candidate.sum()**2) + 0.01)  # Small addition
            
            print("Final candidate score: %.6f" % final_score)
    
    # Ensure we return a sequence the evaluator will accept
    if best_candidate is None:
        # Fall back to a known sequence
        best_candidate = np.ones(medium_length) / medium_length.tolist()
    
    return best_candidate
