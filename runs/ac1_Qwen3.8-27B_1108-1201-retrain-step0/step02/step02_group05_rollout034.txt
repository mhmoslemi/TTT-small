Coordinate descent + gradient hybrid for minimum peak autocorrelation.
Uses O(n) per-coordinate updates with 1D sub-optimization, combined with
L-BFGS p-norm refinement and peak-annihilation local search.

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    size = max(size, n + m - 1)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (np.sum(a)**2)

def full_convolve(a):
    return fft_convolve(a, a)

def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

"""
Hybrid coordinate descent + high-p L-BFGS optimization for minimum peak autocorrelation.
Uses O(n) per-coordinate ternary search updates combined with very high p-norm
gradient refinement, multi-start diversification, and peak-annihilation heuristics.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def full_conv(a):
    """Full linear autocorrelation."""
    return fft_convolve(a, a)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def score_from_conv(c, n):
    """Score from convolution assuming sum(a)=1."""
    return 2.0 * n * np.max(c)


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c = np.maximum(c, 1e-30)
    cp = c ** (p - 1)
    sum_cpw = np.sum(cp * c)
    if sum_cpw < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_cpw ** (1.0 / p)
    score = 2.0 * n * pnorm
    # d/da[j] of sum_k c[k]^p = 2 * sum_{m=0}^{n-1} a[m] * c[m+j]^(p-1)
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    factor = 4.0 * n * sum_cpw ** (1.0 / p - 1.0)
    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def eval_vp(v_prime, j, c, a, n, v_old):
    """Evaluate max convolution if a[j] were set to v_prime instead of v_old."""
    delta = v_prime - v_old
    # Constant peaks: k not in [j, j+n-1]
    const_max = 0.0
    if j > 0:
        const_max = np.max(c[:j])
    end = j + n
    if end < 2 * n - 1:
        const_max = max(const_max, np.max(c[end:]))
    # Linear peaks: c[m+j] + 2*delta*a[m] for m in [0,n-1], m != j
    linear_vals = c[j:j + n].copy() + 2.0 * delta * a
    linear_vals[j] = -np.inf
    linear_max = np.max(linear_vals)
    # Quadratic peak at k=2j
    quad_peak = c[2 * j] + delta * (v_prime + v_old)
    return max(const_max, linear_max, quad_peak)


def coordinate_descent(a, n, num_sweeps, time_limit=None, time_start=None, verbose=False):
    """
    Coordinate descent with ternary search for optimal v' at each coordinate.
    Modifies a in place. Returns a and final score.
    """
    c = full_conv(a)
    best_score = 2.0 * n * np.max(c) / (np.sum(a) ** 2)
    best_a = a.copy()
    
    for sweep in range(num_sweeps):
        if time_limit is not None and time.monotonic() - time_start > time_limit:
            break
        improved = 0
        for j in range(n):
            v = a[j]
            if v < 1e-14:
                # Try a small positive value
                v_upper = 0.01 * np.max(a)
            else:
                v_upper = max(2.0 * v + 0.05 * np.max(a), 0.02 * np.max(a))
            
            lo = 0.0
            hi = v_upper
            # Ternary search for minimum of f(v') = max conv
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = eval_vp(m1, j, c, a, n, v)
                f2 = eval_vp(m2, j, c, a, n, v)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            v_new = (lo + hi) / 2.0
            
            # Check improvement
            f_new = eval_vp(v_new, j, c, a, n, v)
            f_cur = np.max(c)
            
            if f_new < f_cur - 1e-15:
                delta = v_new - v
                a[j] = v_new
                # Update c efficiently
                c[j:j + n] += 2.0 * delta * a
                c[2 * j] += delta * delta
                c = np.maximum(c, 0.0)
                improved += 1
                new_score = 2.0 * n * c[2 * j] if False else 2.0 * n * np.max(c) / (np.sum(a) ** 2)
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
        
        if verbose and (sweep + 1) % 20 == 0:
            cur_score = 2.0 * n * np.max(c) / (np.sum(a) ** 2)
            print(f"    CD sweep {sweep+1}/{num_sweeps}: score={cur_score:.8f}, best={best_score:.8f}, improved_coords={improved}")
    
    return best_a, best_score


def run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p, deadline):
    """Run L-BFGS with annealed p-norm schedule."""
    best_theta = theta.copy()
    a_test = softmax(theta)
    best_score = 2.0 * len(theta) * np.max(full_conv(a_test))
    
    for p in p_schedule:
        if time.monotonic() > deadline - 20:
            break
        result = minimize(
            pnorm_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(p,),
            options={'maxiter': maxiter_per_p, 'ftol': 1e-15, 'gtol': 1e-13,
                     'maxfun': maxiter_per_p * 8, 'maxls': 50}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = 2.0 * len(theta_new) * np.max(full_conv(a_new))
        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        theta = best_theta
    
    return best_theta, best_score


def peak_annihilation(a, n, rng, num_rounds=20):
    """
    Targeted peak reduction: identify top contributing pairs to the peak lag,
    reduce them, and re-optimize via coordinate descent.
    """
    a = a.copy()
    c = full_conv(a)
    best_score = 2.0 * n * np.max(c) / (np.sum(a) ** 2)
    best_a = a.copy()
    
    for round_i in range(num_rounds):
        c = full_conv(a)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]
        
        # Find top contributing pairs
        contribs = np.zeros(n)
        for i in range(min(n, peak_idx + 1)):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        top_k = max(3, n // 16)
        top_idx = np.argsort(contribs)[-top_k:]
        
        # Reduce top contributors and increase others
        a_new = a.copy()
        a_new[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
        other = np.setdiff1d(np.arange(n), top_idx)
        if len(other) > 0:
            n_inc = min(8, len(other))
            inc_idx = rng.choice(other, size=n_inc, replace=False)
            a_new[inc_idx] *= np.exp(rng.normal(0.04, 0.02, size=n_inc))
        
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / np.sum(a_new)
        
        # Quick coordinate descent polish
        a_new, s = coordinate_descent(a_new, n, num_sweeps=30, verbose=False)
        
        if s < best_score:
            best_score = s
            best_a = a_new.copy()
    
    return best_a, best_score
