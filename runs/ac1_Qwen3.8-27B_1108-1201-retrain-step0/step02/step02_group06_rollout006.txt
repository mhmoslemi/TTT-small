Uses block coordinate descent (LP in the "dual" variable), gradient refinement, and local search.

from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)

    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def max_conv_smooth(a, tau):
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    value = max_c + tau * np.log(np.sum(exp_vals))

    fw = np.fft.rfft(w, size)

    grad = 2.0 * corr
    return value, grad, conv

    cols = []

def spectral_init(n, rng, n_iter=40):
    """Spectral initialization via time-frequency alternation."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)

    for it in range(n_iter):
        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        corr = np.fft.irfft(power, 2 * n)[:n]
        alpha = 1.0 / (n * 0.5)
        a = a - alpha * corr
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s

    return a


def raised_cosine_init(n, rng, beta=2.0):
    """Raised cosine window as initialization."""
    t = np.arange(n) / n
    a = 0.5 * (1 + np.cos(2 * np.pi * t - np.pi))
    a = a ** beta
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def gaussian_init(n, rng, sigma_frac=0.3):
    """Gaussian bump initialization."""
    t = np.arange(n) / n
    center = 0.5
    a = np.exp(-0.5 * ((t - center) / sigma_frac) ** 2)
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def optimize_adam_simplex(a_init, time_limit, rng, tau_start_frac=10.0):
    """Adam on simplex with smoothed max, using log-barrier for non-negativity."""
    n = len(a_init)
    a = a_init.copy()
    a = np.clip(a, 1e-8, None)
    a = a / np.sum(a)

    # Use log-parameterization: a_i = exp(theta_i) / sum(exp(theta_j))
    theta = np.log(a + 1e-15)
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr_base = 0.5
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = score_fast(a)

    tau_start = tau_start_frac / n
    tau_end = 0.05 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.6)

        a = softmax(theta)
        value, g_a, conv = max_conv_smooth(a, tau)

        # Gradient w.r.t. theta for softmax: p*(g - p@g)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner)

        step += 1
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * (g_theta * g_theta)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr_base / (1.0 + 0.1 * np.sqrt(step))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if step % 20 == 0:
            a_check = softmax(theta)
            sc = score_fast(a_check)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()

        if step % 500 == 0:
            a_check = softmax(theta)
            sc = score_fast(a_check)
            print(f"      Adam step={step}, score={sc:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")

    return softmax(best_theta), best_score


def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_search(a_init, time_limit, rng, max_stall=3000):
    """Peak-targeted local search on the simplex."""
    a = a_init.copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    current_score = score_fast(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        r = rng.random()

        if r < 0.3:
            # Pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.02, 0.2)
            amt = a[i] * frac
            cand = a.copy()
            cand[i] -= amt
            cand[j] += amt
            cand = np.clip(cand, 0, None)
            cs = np.sum(cand)
            if cs < 1e-10:
                continue
            cand = cand / cs

        elif r < 0.5:
            # Single element scale
            j = int(rng.integers(n))
            factor = np.exp(rng.normal(0, 0.08))
            cand = a.copy()
            cand[j] *= factor
            cand = np.clip(cand, 0, None)
            cs = np.sum(cand)
            if cs < 1e-10:
                continue
            cand = cand / cs

        elif r < 0.7:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 32))))
            start = int(rng.integers(0, max(1, n - width)))
            cand = a.copy()
            cand[start:start + width] *= np.exp(rng.normal(0, 0.05, size=width))
            cand = np.clip(cand, 0, None)
            cs = np.sum(cand)
            if cs < 1e-10:
                continue
            cand = cand / cs

        elif r < 0.85:
            # Peak-targeted: find peak of convolution, reduce contributors
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Contributors to conv[peak_idx]: a[i]*a[peak_idx-i] for valid i
            contribs = np.zeros(n)
            for i in range(max(0, peak_idx - n + 1), min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] += a[i] * a[j]
            # Find top contributors
            n_top = max(2, n // 64)
            top_idx = np.argsort(contribs)[-n_top:]
            cand = a.copy()
            reduction = rng.uniform(0.05, 0.2)
            cand[top_idx] *= (1 - reduction * rng.uniform(0.5, 1.0, size=len(top_idx)))
            cand = np.clip(cand, 0, None)
            # Redistribute mass
            removed = np.sum(a) - np.sum(cand)
            if removed > 0:
                others = np.setdiff1d(np.arange(n), top_idx)
                if len(others) > 0:
                    add_idx = rng.choice(others, size=min(5, len(others)), replace=False)
                    cand[add_idx] += removed / len(add_idx)
            cs = np.sum(cand)
            if cs < 1e-10:
                continue
            cand = cand / cs

        else:
            # Antisymmetric perturbation (helps flatten convolution)
            j = int(rng.integers(n))
            partner = n - 1 - j
            if partner == j:
                continue
            delta = rng.uniform(-0.05, 0.05) * (a[j] + a[partner])
            cand = a.copy()
            cand[j] += delta
            cand[partner] -= delta
            cand = np.clip(cand, 0, None)
            cs = np.sum(cand)
            if cs < 1e-10:
                continue
            cand = cand / cs

        new_score = score_fast(cand)
        if new_score < current_score:
            a = cand
            current_score = new_score
            stall = 0
        else:
            stall += 1

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

    return best_a, best_score


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - 1.0)
    rho = np.flatnonzero(cond)
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def resize_sequence(seq, new_n, rng):
    """Resize a sequence to new length via interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.001, None)
    return new_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP alternating -> Adam gradient -> Local search.
    Focuses on the LP-based block coordinate descent as the primary optimizer.
    """
    t_global = time.monotonic()
    time_budget = max(100.0, float(budget_s) - 60.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.001, 1000.0)
        s0 = score_fast(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s0:.6f}")
        if s0 < best_score:
            best_score = s0
            best_seq = start_seq.copy()

    # Define search configurations: (n, n_lp_iters, adam_time_frac, local_time_frac)
    # Total time per config = lp_time + adam_time + local_time
    configs = [
        (512, 80, 0.25, 0.25),
        (768, 60, 0.3, 0.3),
        (1024, 80, 0.3, 0.3),
        (1024, 60, 0.3, 0.3),
        (2048, 40, 0.35, 0.35),
    ]

    # Allocate time proportionally
    total_units = sum(1.0 / c[0] for c in configs)  # weight toward smaller n being faster
    time_per_config = time_budget * 0.85  # leave 15% for final exploration

    init_strategies = [
        ('spectral', spectral_init),
        ('raised_cos', raised_cosine_init),
        ('gaussian', gaussian_init),
    ]

    config_idx = 0
    for n, n_lp, adam_frac, local_frac in configs:
        if time.monotonic() > deadline - 60:
            break

        # Determine how much time to give this config
        remaining_configs = len(configs) - config_idx
        t_left = deadline - time.monotonic() - 60
        t_alloc = min(t_left / remaining_configs, time_budget * 0.3)
        t_alloc = max(60.0, t_alloc)

        t_lp = t_alloc * 0.35
        t_adam = t_alloc * adam_frac
        t_local = t_alloc * local_frac

        print(f"\n=== Config n={n}: lp={t_lp:.0f}s, adam={t_adam:.0f}s, local={t_local:.0f}s ===")

        # Try multiple inits
        n_inits = max(1, int(t_alloc / 120))
        n_inits = min(n_inits, 3)

        for init_i in range(n_inits):
            if time.monotonic() > deadline - 60:
                break

            init_name, init_func = init_strategies[(init_i + config_idx) % len(init_strategies)]
            a0 = init_func(n, rng)
            a0 = a0 / np.sum(a0)

            # Phase 1: LP Alternating
            print(f"  Init '{init_name}', n={n}")
            t_lp_actual = min(t_lp, max(10.0, (deadline - time.monotonic() - 60) * 0.3))
            a_lp = lp_alternating(a0, n_iter=n_lp, blend_rate=0.7,
                                  verbose=True, time_limit=t_lp_actual)
            sc_lp = score_fast(a_lp)
            print(f"  After LP: score={sc_lp:.6f}")

            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
                print(f"    *** New best: {best_score:.6f} ***")

            # Phase 2: Adam gradient refinement
            if time.monotonic() < deadline - 30 and t_adam > 5:
                t_adam_actual = min(t_adam, max(5.0, (deadline - time.monotonic() - 60) * 0.3))
                a_grad, sc_grad = optimize_adam_simplex(a_lp, t_adam_actual, rng)
                sc_grad = score_fast(a_grad)
                print(f"  After Adam: score={sc_grad:.6f}")

                if sc_grad < best_score:
                    best_score = sc_grad
                    best_seq = a_grad.copy()
                    print(f"    *** New best: {best_score:.6f} ***")

                a_for_local = a_grad
            else:
                a_for_local = a_lp

            # Phase 3: Local search
            if time.monotonic() < deadline - 30 and t_local > 5:
                t_local_actual = min(t_local, max(5.0, deadline - time.monotonic() - 30))
                a_local, sc_local = local_search(a_for_local, t_local_actual, rng)
                sc_local = score_fast(a_local)
                print(f"  After local: score={sc_local:.6f}")

                if sc_local < best_score:
                    best_score = sc_local
                    best_seq = a_local.copy()
                    print(f"    *** New best: {best_score:.6f} ***")

        config_idx += 1

    # Final phase: intensive local search on best
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Final intensive local search (budget: {deadline - time.monotonic() - 30:.0f}s) ===")
        t_final = min(300.0, deadline - time.monotonic() - 30)
        a_final, sc_final = local_search(best_seq / np.sum(best_seq), t_final, rng)
        sc_final = score_fast(a_final)
        print(f"  Final local search score: {sc_final:.6f}")
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Final intensive LP on best
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = len(best_seq)
        t_lp_final = min(120.0, deadline - time.monotonic() - 20)
        if t_lp_final > 10:
            print(f"\n=== Final LP polish (budget: {t_lp_final:.0f}s) ===")
            a_polished = lp_alternating(best_seq / np.sum(best_seq),
                                        n_iter=50, blend_rate=0.5,
                                        verbose=True, time_limit=t_lp_final)
            sc_pol = score_fast(a_polished)
            print(f"  After final LP: score={sc_pol:.6f}")
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_polished.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Verify with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        if final_score < best_score or np.isinf(best_score):
            best_score = final_score
            best_seq = np.array(final_list)

    # Fallback
    if best_seq is None:
        if start_seq is not None:
            return [float(max(0.0, min(1000.0, x))) for x in start_seq]
        return [1.0] * 256

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
