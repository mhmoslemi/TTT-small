```python

Multi-strategy optimization for minimizing peak self-convolution of non-negative sequences.
Uses L-BFGS-B with annealed soft-max, symmetric Gerchberg-Saxton initialization,
and peak-directed local search.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    c = fft_convolve(a, a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def smooth_objective(theta, tau):
    """
    Objective in terms of theta where a = exp(theta).
    Uses soft-max approximation of max.
    Returns (value, gradient).
    """
    n = len(theta)
    a = np.exp(theta - np.max(theta))  # for numerical stability
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf, np.zeros_like(theta)
    
    c = fft_convolve(a, a)
    
    # Soft-max
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    w = e / np.sum(e)
    soft_max_val = c_max + tau * np.log(np.sum(e))
    
    # Objective: 2*n*soft_max / sum_a^2
    value = 2.0 * n * soft_max_val / (sum_a ** 2)
    
    # Gradient w.r.t. a:
    # d(value)/d(a) = 2*n/(sum_a^2) * d(soft_max)/d(a) - 2*value/sum_a * 1 (from derivative of 1/sum_a^2)
    # d(soft_max)/d(a_i) = 2 * sum_k w_k * a[k-i] * [valid]
    # = 2 * (w cross-correlate a)[i]
    
    # Compute gradient of soft_max w.r.t. a
    # g_a[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]
    # This is 2 * np.correlate(w, a, 'valid')
    
    # Using FFT for efficiency:
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    # cross-correlation of w and a: IFFT(W * conj(A))
    corr = np.fft.irfft(W * np.conj(A), N)
    g_a = 2.0 * corr[n-1:2*n-1]
    
    # d(value)/d(a_i) = 2*n/(sum_a^2) * g_a[i] - 2*value/sum_a
    grad_a = (2.0 * n / (sum_a ** 2)) * g_a - (2.0 * value / sum_a) * np.ones(n)
    
    # Chain rule: d(value)/d(theta_i) = a_i * grad_a[i]
    grad_theta = a * grad_a
    
    return value, grad_theta

def gradient_optimize(a_init, time_limit, tau_start, tau_end, rng):
    """
    L-BFGS-B optimization with annealed soft-max tau.
    a_init: initial non-negative sequence
    """
    n = len(a_init)
    # Normalize
    sum_a = np.sum(a_init)
    if sum_a < 1e-10:
        a_init = np.ones(n) / n
        sum_a = 1.0
    a_init = a_init / sum_a  # normalize to sum=1
    
    theta = np.log(a_init + 1e-15)
    best_theta = theta.copy()
    best_value = compute_score(a_init)
    
    t0 = time.monotonic()
    
    # Anneal tau and use L-BFGS-B with multiple phases
    n_phases = 5
    phase_times = [time_limit * f for f in [0.25, 0.2, 0.2, 0.2, 0.15]]
    
    for phase in range(n_phases):
        if time.monotonic() - t0 >= time_limit:
            break
        phase_end = t0 + sum(phase_times[:phase+1])
        phase_time = min(time_limit, phase_end - t0)
        if time.monotonic() > phase_end - 5:
            continue
        
        frac = phase / max(1, n_phases - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Define objective and gradient for this phase
        def obj(val):
            v, g = smooth_objective(val, tau)
            return v
        
        def grad(val):
            v, g = smooth_objective(val, tau)
            return g
        
        # Run L-BFGS-B
        result = minimize(
            obj, theta, jac=grad, method='L-BFGS-B',
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-8,
                     'maxfun': 500, 'disp': False}
        )
        
        if result.fun < best_value:
            a_check = np.exp(result.x - np.max(result.x))
            a_check = a_check / np.sum(a_check)
            score = compute_score(a_check)
            if score < best_value:
                best_value = score
                best_theta = result.x.copy()
        
        theta = result.x
    
    # Final evaluation
    a_final = np.exp(best_theta - np.max(best_theta))
    a_final = a_final / np.sum(a_final)
    final_score = compute_score(a_final)
    
    return a_final, min(best_value, final_score)

def gs_symmetric(n, n_iter=200, rng=None):
    """
    Gerchberg-Saxton for symmetric sequences to get flat convolution.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    # Initialize
    a = np.ones(n) / n
    # Enforce symmetry
    a = (a + a[::-1]) / 2
    
    for it in range(n_iter):
        # Forward: go to frequency domain
        a_padded = np.zeros(N)
        a_padded[:n] = a
        A = np.fft.rfft(a_padded)
        
        # Modify magnitude to flatten
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        # Target: flat magnitude
        target_mag = np.sqrt(np.sum(mag**2) / N)
        # Partial update (relaxation)
        new_mag = 0.5 * mag + 0.5 * target_mag
        phase = A / mag
        A_new = new_mag * phase
        
        # Inverse
        a_new = np.fft.irfft(A_new, N).real
        a = a_new[:n]
        
        # Enforce symmetry
        a = (a + a[::-1]) / 2
        
        # Enforce non-negativity
        a = np.maximum(a, 0)
        
        # Normalize
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
    
    return a

def peak_directed_search(a, time_limit, rng, top_k=10):
    """
    Local search that targets the peak convolution lags.
    """
    n = len(a)
    current = a.copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    # Temperature for simulated annealing
    temp = 0.003
    cooling = 0.9997
    
    while time.monotonic() - t0 < time_limit:
        # Compute current convolution
        c = fft_convolve(current, current)
        
        # Find top peak lags (excluding the trivial ones at the very ends)
        # The main peak is typically at lag n-1
        top_lags = np.argsort(c)[-top_k:]
        
        # For each peak lag, find which elements contribute most
        r = rng.random()
        
        if r < 0.4:
            # Reduce contribution to top lags
            contrib = np.zeros(n)
            for lag in top_lags[:5]:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    if 0 <= j < n:
                        contrib[i] += current[i] * current[j]
            
            # Reduce high-contributing elements, boost low-contributing
            n_reduce = max(3, n // 15)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            
            candidate = current.copy()
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=len(top_idx)))
            sum_reduced = np.sum(current[top_idx]) - np.sum(candidate[top_idx])
            if sum_reduced > 0 and len(bot_idx) > 0:
                boost = sum_reduced / np.sum(current[bot_idx])
                candidate[bot_idx] *= (1 + boost * 0.5)
            
        elif r < 0.6:
            # Random perturbation focused on peak regions
            candidate = current.copy()
            for lag in top_lags[:3]:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                region = slice(i_min, i_max + 1)
                candidate[region] *= np.exp(rng.normal(-0.02, 0.01, 
                    size=i_max - i_min + 1))
            # Compensate elsewhere
            other = np.setdiff1d(np.arange(n), 
                                 np.arange(max(0, top_lags[0]-n+1), 
                                          min(n-1, top_lags[0])+1))
            if len(other) > 0:
                n_boost = min(len(other), n//10)
                boost_idx = rng.choice(other, size=n_boost, replace=False)
                candidate[boost_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_boost))
                
        elif r < 0.75:
            # Block permutation: swap two blocks
            candidate = current.copy()
            block_size = max(2, n // 32)
            p1 = int(rng.integers(0, n - 2*block_size))
            p2 = int(rng.integers(p1 + block_size, n - block_size))
            candidate[p1:p1+block_size], candidate[p2:p2+block_size] = \
                candidate[p2:p2+block_size].copy(), candidate[p1:p1+block_size].copy()
            
        elif r < 0.9:
            # Gaussian perturbation
            candidate = current * np.exp(rng.normal(0, 0.01, size=n))
        else:
            # Symmetrization step
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.1 * (sym - current) + 0.005 * rng.standard_normal(n)
        
        candidate = np.maximum(candidate, 0)
        s = np.sum(candidate)
        if s > 1e-10:
            candidate = candidate / s * np.sum(current)
        
        new_score = compute_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 5:
                if rng.random() < np.exp(-delta / max(temp, 1e-15)):
                    current = candidate
                    current_score = new_score
        
        temp *= cooling
    
    return best, best_score

def upsample(a, new_n, rng):
    """Upsample sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.maximum(a_interp, 0.001)
    return a_interp / np.sum(a_interp)
```

Multi-scale L-BFGS-B optimization with annealed soft-max for minimizing
peak self-convolution of non-negative sequences. Uses coarse-to-fine
upsampling, quasi-Newton optimization, and subgradient refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    n = len(a)
    c = fft_convolve(a, a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    return 2.0 * n * np.max(c) / (s ** 2)

def smooth_obj_and_grad(theta, tau):
    """
    a = exp(theta - max(theta)) for stability.
    Objective: 2*n*softmax(conv(a,a), tau) / (sum a)^2
    Returns (value, gradient w.r.t. theta)
    """
    n = len(theta)
    a = np.exp(theta - np.max(theta))
    s = np.sum(a)
    if s < 1e-10:
        return np.inf, np.zeros(n)
    
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    w = e / np.sum(e)
    sm = c_max + tau * np.log(np.sum(e))
    
    val = 2.0 * n * sm / (s * s)
    
    # Gradient of soft-max w.r.t. a: g_a[i] = 2 * sum_m w[i+m] * a[m]
    g_a = 2.0 * np.correlate(w, a, 'valid')
    
    # d(val)/d(a_i) = 2n/s^2 * g_a[i] - 2*val/s
    grad_a = (2.0 * n / (s * s)) * g_a - (2.0 * val / s)
    
    # Chain rule
    grad_theta = a * grad_a
    
    return val, grad_theta

def optimize_lbfgsb(a_init, time_limit, tau_seq, maxiter_per_phase=300):
    """
    Multi-phase L-BFGS-B with different tau values.
    tau_seq: list of tau values to use in sequence.
    """
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    theta = np.log(a + 1e-15)
    
    best_theta = theta.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for idx, tau in enumerate(tau_seq):
        if time.monotonic() - t0 >= time_limit:
            break
        
        def f(val):
            v, _ = smooth_obj_and_grad(val, tau)
            return v
        
        def g(val):
            _, gr = smooth_obj_and_grad(val, tau)
            return gr
        
        res = minimize(f, theta, jac=g, method='L-BFGS-B',
                       options={'maxiter': maxiter_per_phase, 'ftol': 1e-12,
                                'gtol': 1e-10, 'maxfun': 1000})
        
        if res.fun < np.inf:
            a_new = np.exp(res.x - np.max(res.x))
            a_new = a_new / np.sum(a_new)
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_theta = res.x.copy()
        theta = res.x
    
    a_final = np.exp(best_theta - np.max(best_theta))
    a_final = a_final / np.sum(a_final)
    
    return a_final, best_score

def subgradient_polish(a, time_limit, rng, n_steps=None):
    """
    Subgradient method: at each step, identify the peak lag and
    reduce contributing elements.
    """
    n = len(a)
    current = a.copy()
    best = current.copy()
    best_score = compute_score(current)
    t0 = time.monotonic()
    lr = 0.02
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        peak_lag = np.argmax(c)
        
        # Compute contribution of each element to the peak
        contrib = np.zeros(n)
        i_min = max(0, peak_lag - n + 1)
        i_max = min(n - 1, peak_lag)
        for i in range(i_min, i_max + 1):
            j = peak_lag - i
            contrib[i] = current[i] * current[j]

        # Subgradient step
        s = np.sum(current)
        grad = (2.0 * n / s**2) * 2.0 * contrib - (2.0 * best_score * s / s**2)
        # Actually, the subgradient of max at the peak is just the gradient at that lag
        # d/d(a_i) [c[peak_lag]] = 2*a[peak_lag - i] (if valid)
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            update = lr * grad / g_norm
            # Apply with projection
            candidate = current - update
            candidate = np.maximum(candidate, 0)
            cs = np.sum(candidate)
            if cs > 1e-10:

        cs = np.sum(candidate)
        candidate = candidate / cs * s

            best = candidate.copy()
            current = candidate
        elif rng.random() < np.exp(-(sc - best_score) / (lr * 0.5)):
            current = candidate
        
        lr *= 0.999
    
    return best, best_score
