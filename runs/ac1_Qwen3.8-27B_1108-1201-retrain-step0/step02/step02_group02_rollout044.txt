Multi-phase peak-reduction optimizer for minimizing 2n*max(convolve(a,a))/(sum(a))^2.
Uses softmax-parameterized gradient descent with adaptive temperature,
Remez-like peak equalization, and coordinate-wise local search.

import time
from scipy.optimize import minimize

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def score_and_gradients(theta, S_target, tau):
    """
    Compute smoothed max score and gradient w.r.t. theta.
    a = softmax(theta) * S_target
    objective = smoothed_max(convolve(a,a)) with temperature tau
    """
    n = len(theta)
    a = softmax_normalize(theta) * S_target
    c = fft_convolve(a, a)
    
    # Log-sum-exp for smoothed max
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    smooth_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp  # softmax weights over convolution lags
    
    # Gradient of smooth_val w.r.t. a
    # d(conv(a,a))[k]/da[j] = 2 * a[k-j] (with bounds)
    # = 2 * corr(a, w)[j] where corr is cross-correlation
    # Actually: d/db[k] da[j] = da[k-j]/da[j] + da[j-(k-n+1+n-1)]... 
    # Let me be more careful.
    # b[k] = sum_i a[i]*a[k-i] (where 0<=i<n, 0<=k-i<n)
    # db[k]/da[j] = sum_i [delta(i,j)*a[k-i] + a[i]*delta(k-i,j)]
    #             = a[k-j] (if 0<=k-j<n) + a[j-k+n-1]... 
    # Hmm, let me think differently.
    # b = convolve(a, a), so b[k] = sum_i a[i] * a_flip[k-i] where a_flip = a reversed
    # Actually np.convolve(a, b)[k] = sum_i a[i] * b[k-i]
    # So b[k] = sum_i a[i] * a[k-i]
    # db[k]/da[j] = a[k-j] (from first factor) + a[j-k] * (-1)... 
    # No. b[k] = sum_i a[i]*a[k-i]. 
    # db[k]/da[j] = sum_i [da[i]/da[j] * a[k-i] + a[i] * da[k-i]/da[j]]
    # = a[k-j] + sum_i a[i] * (-1) * delta(k-i-j, 0)  [since d(k-i)/da[j] doesn't make sense]
    # 
    # Wait, I need to be more careful. b[k] = sum_i a[i]*a[k-i] where both a[i] and a[k-i] are functions of a.
    # db[k]/da[j] = sum_i [delta_{i,j} * a[k-i] + a[i] * delta_{k-i,j}]
    # = a[k-j] * I(0<=k-j<n) + a[k-j] * I(0<=j<n and 0<=k-j<n)
    # 
    # Wait no. The second term: a[i] * d(a[k-i])/da[j] = a[i] * delta_{k-i,j} = a[k-j] * delta_{i,k-j}
    # So the sum over i of a[i]*delta_{k-i,j} = a[k-j] (if 0<=k-j<n)
    # 
    # So db[k]/da[j] = a[k-j]*I(0<=k-j<n) + a[k-j]*I(0<=k-j<n) = 2*a[k-j]*I(0<=k-j<n)?
    # 
    # No wait, that's only if k-j is a valid index. Let me redo.
    # b[k] = sum_i a[i]*a[k-i] where the sum is over i such that 0<=i<n and 0<=k-i<n.
    # 
    # db[k]/da[j] = sum_i [I(i=j)*a[k-i] + a[i]*I(k-i=j)]
    # = a[k-j]*I(0<=k-j<n) + sum_i a[i]*I(k-i=j)*I(0<=i<n)*I(0<=k-i<n)
    # The second sum: k-i=j means i=k-j. So we get a[k-j]*I(0<=k-j<n)*I(0<=k-j<n) = a[k-j]*I(0<=k-j<n)
    # 
    # So db[k]/da[j] = 2*a[k-j]*I(0<=k-j<n)
    # 
    # This makes sense! The convolution is symmetric in its arguments, so the gradient gets a factor of 2.
    
    # Gradient of smooth_val w.r.t. a:
    # d(smooth_val)/da[j] = sum_k w[k] * db[k]/da[j] = 2 * sum_k w[k] * a[k-j] * I(0<=k-j<n)
    # = 2 * sum_k w[k] * a[k-j] where k-j is in [0,n-1], i.e., k in [j, j+n-1]
    # = 2 * (w cross-correlated with a)[j]
    # 
    # More precisely: let's define g[j] = 2 * sum_{k=j}^{j+n-1} w[k] * a[k-j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # This is (2 * (a_flip * w))[j + n - 1] where a_flip = a reversed and * is convolution
    # Or equivalently: g = 2 * fft_convolve(a[::-1], w)[n-1:2n-1]
    
    g_a = 2.0 * fft_convolve(a[::-1], w)[n-1:2*n-1]
    
    # Now convert to theta-space: a = softmax(theta) * S_target
    # da/dtheta: da[i]/dtheta[j] = a[i]*(delta_{ij} - a[i]/S_target) * S_target... 
    # Actually a[i] = S_target * exp(theta[i]) / sum(exp(theta))
    # da[i]/dtheta[j] = a[i] * (delta_{ij} - a[i]/S_target)
    # 
    # d(smooth_val)/dtheta[j] = sum_i (d smooth_val/da[i]) * (da[i]/dtheta[j])
    # = sum_i g_a[i] * a[i] * (delta_{ij} - a[i]/S_target)
    # = g_a[j]*a[j] - a[j] * sum_i g_a[i]*a[i] / S_target
    # = a[j] * (g_a[j] - sum(g_a*a)/S_target)
    
    inner = np.dot(g_a, a)
    g_theta = a * (g_a - inner / S_target)
    
    return smooth_val, g_theta, c, w

def optimize_smoothed(n, theta_init, time_budget, rng, S_target=None):
    """
    Optimize using smoothed max objective with adaptive temperature.
    Returns best theta and score.
    """
    if S_target is None:
        S_target = np.sqrt(2.0 * n)
    
    theta = theta_init.astype(np.float64).copy()
    best_theta = theta.copy()
    best_score = compute_score(softmax_normalize(theta) * S_target)
    
    # Adam
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0
    no_improve = 0
    patience = 2000
    t0 = time.monotonic()
    
    # Temperature schedule: start high, end very low
    tau_start = 5.0 / n
    tau_end = 0.001 / n
    
    while time.monotonic() - t0 < time_budget:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_budget)
        
        # Log-linear temperature schedule
        log_tau = np.log(tau_start) + elapsed_frac * (np.log(tau_end) - np.log(tau_start))
        tau = np.exp(log_tau)
        
        value, g_theta, c, w = score_and_gradients(theta, S_target, tau)
        
        # Adam update
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g_theta
        v_buf = beta2 * v_buf + (1 - beta2) * g_theta ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)
        
        # Learning rate with warmup and decay
        lr = 1.0 / np.sqrt(step + 10)
        if elapsed_frac < 0.05:
            lr *= elapsed_frac / 0.05  # warmup
        elif elapsed_frac > 0.8:
            lr *= 0.5  # slower at end
        
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic evaluation
        if step % 100 == 0:
            a = softmax_normalize(theta) * S_target
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
                no_improve = 0
            else:
                no_improve += 100
            
            if no_improve >= patience:
                # Escape: perturb from best
                if rng.random() < 0.5:
                    theta = best_theta + 0.2 * rng.standard_normal(n)
                else:
                    # Targeted perturbation on peak contributors
                    a_best = softmax_normalize(best_theta) * S_target
                    c_best = fft_convolve(a_best, a_best)
                    peak_idx = int(np.argmax(c_best))
                    # Perturb elements that don't contribute much to peak
                    contribs = np.zeros(n)
                    for i in range(n):
                        if 0 <= peak_idx - i < n:
                            contribs[i] = a_best[i] * a_best[peak_idx - i]
                    low_contrib = np.argsort(contribs)[:n//4]
                    theta[low_contrib] += 0.15 * rng.standard_normal(len(low_contrib))
                no_improve = 0
            
            if step % 2000 == 0:
                print(f"    step={step}, smoothed={value:.6f}, exact={score:.6f}, best={best_score:.6f}, tau={tau:.2e}")
    
    return best_theta, best_score
