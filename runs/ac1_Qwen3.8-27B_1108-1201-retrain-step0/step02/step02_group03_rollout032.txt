"""
LP-based iterative optimization for minimizing peak autocorrelation of
non-negative step functions. Uses an alternating LP approach where at each
step we solve a linear program to find the best "response" sequence, then
blend with the current sequence. Combined with gradient-based refinement
and local search for final polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def solve_lp(f, M):
    """
    Solve: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0
    
    Returns optimal g and status.
    """
    n = len(f)
    
    # Build constraint matrix A_ub such that A_ub @ g <= M_vec
    # For each lag k (0 to 2n-2), the constraint is:
    #   sum_{i=max(0,k-n+1)}^{min(n-1,k)} f_i * g_{k-i} <= M
    # which is: sum_j A[k,j] * g_j <= M
    # where A[k,j] = f_{k-j} if 0 <= k-j < n, else 0
    
    # A_ub has shape (2n-1, n)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A_ub[k, j] = f[k-j]
    
    b_ub = np.full(2*n-1, M)
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0:
        g = res.x
        return g, True
    else:
        return None, False

def lp_iterate(f, n_iter, t0, time_limit, t_start=0.3, t_end=0.05):
    """
    Run LP-based iteration.
    At each step:
    1. Compute M = max(f*f)
    2. Solve LP to get g
    3. Rescale g to match sum(f)
    4. Update f = (1-t)*f + t*g
    """
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal t
        frac = it / max(1, n_iter)
        t = t_start * (t_end / t_start) ** frac
        
        # Compute convolution and peak
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        # Solve LP
        g, success = solve_lp(f, M)
        if not success:
            # If LP fails, try with slightly larger M
            g, success = solve_lp(f, M * 1.01)
            if not success:
                break
        
        # Rescale g to match sum
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        g = g / sum_g * S
        
        # Update
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        
        # Check score
        score = local_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        
        f = f_new
        
        if it % 10 == 0:
            print(f"    LP iter {it}: score={score:.6f}, best={best_score:.6f}, t={t:.4f}, M={M:.6f}")
    
    return best_f, best_score

def fft_phase_init(n, rng, n_iter=80):
    """Gerchberg-Saxton style initialization."""
    N_fft = 1
    while N_fft < 2*n-1:
        N_fft <<= 1
    
    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a

def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s

def grad_soft_max_conv(a, tau):
    """Compute soft-max(conv(a,a)) and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    value = c_max + tau * np.log(s)
    w = e / s
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = (2.0 / tau) * corr[n - 1:2 * n - 1]
    return value, g, c

def optimize_gradient(n, a_init, time_limit, rng, tau_start_mult=20.0, tau_end_mult=0.1):
    """Adam gradient descent on softmax-parameterized sequence."""
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S
    theta = np.log(a / S + 1e-12)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr0 = 3.0
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)
    tau_start = tau_start_mult / n
    tau_end = tau_end_mult / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)

        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)

        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 200 == 0:
            a_check = softmax_normalize(theta) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
            if step % 1000 == 0:
                print(f"        grad step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}")

    a_final = softmax_normalize(best_theta) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
    return a_final, best_score

def local_search(a_init, time_limit, rng, top_k=8):
    """Simulated annealing with targeted mutations."""
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9997
    conv = fft_convolve(current, current)

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.3:
            # Peak-directed
            top_lags = np.argsort(conv)[-top_k:]
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            n_reduce = max(2, n // 12)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.02, size=len(bot_idx)))
        elif r < 0.5:
            # Block perturb
            n_blocks = int(rng.integers(2, 5))
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 20))))
                bs = int(rng.integers(0, n - bw))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0.0, 0.04, size=bw))
        elif r < 0.7:
            # Symmetrization
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.2 * (sym - current)
            candidate += 0.01 * rng.standard_normal(n)
        elif r < 0.85:
            # Sparse modification
            n_mod = max(1, n // 8)
            idx = rng.choice(n, size=n_mod, replace=False)
            candidate[idx] *= np.exp(rng.normal(0.0, 0.15, size=n_mod))
        else:
            # Global noise
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current
        else:
            candidate = candidate / s * np.sum(current)

        new_score = local_score(candidate)
        if new_score < current_score:
            current = candidate
            current_score = new_score
            conv = fft_convolve(current, current)
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 5:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score
                    conv = fft_convolve(current, current)

        temp *= cooling

    return best, best_score

"""
LP-based iterative optimization for minimizing peak autocorrelation of
non-negative step functions. The core idea is an alternating LP approach:
at each step, solve a linear program to find the best "response" sequence g
that has cross-correlation with current f bounded by f's peak autocorrelation,
then blend f with g to reduce peaks. Combined with FFT-based initialization,
gradient refinement, and local search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_lp_matrix(f):
    """Build the LP constraint matrix for the cross-correlation constraints."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # For efficiency, use the Toeplitz structure
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def build_lp_matrix_fast(f):
    """Build LP matrix using vectorized operations."""
    n = len(f)
    # Create indices
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, None] - j[None, :]  # shape (2n-1, n)
    mask = (diff >= 0) & (diff < n)
    # For valid entries, A[k,j] = f[k-j]
    # We need to index f with diff values
    # Clip to valid range for indexing
    diff_clipped = np.clip(diff, 0, n - 1)
    A = np.where(mask, f[diff_clipped], 0.0)
    return A


def solve_lp(f, M, verbose=False):
    """
    Solve: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Returns (g, success).
    """
    n = len(f)
    A_ub = build_lp_matrix_fast(f)
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if res.status == 0:
            return res.x, True
        else:
            return None, False
    except Exception as e:
        if verbose:
            print(f"  LP error: {e}")
        return None, False


def lp_optimize(f_init, time_limit, t0=None, t_start=0.4, t_end=0.03, verbose=True):
    """
    LP-based iterative optimization.
    """
    if t0 is None:
        t0 = time.monotonic()
    
    f = f_init.astype(np.float64).copy()
    S = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    n = len(f)
    it = 0
    stagnation = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Anneal t
        # Use a schedule that starts aggressive and becomes conservative
        progress = min(1.0, it / max(1, int(time_limit / 0.1)))  # rough progress
        t = t_start * (t_end / t_start) ** (progress ** 0.7)
        
        # Compute current peak
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        # Solve LP
        g, success = solve_lp(f, M)
        if not success:
            # Try with relaxed constraint
            g, success = solve_lp(f, M * 1.001)
            if not success:
                g, success = solve_lp(f, M * 1.01)
                if not success:
                    break
        
        # Rescale g
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        g = g / sum_g * S
        
        # Update with blend
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0.0, None)
        
        # Ensure sum is preserved
        sum_new = np.sum(f_new)
        if sum_new < 1e-10:
            f_new = f
        else:
            f_new = f_new / sum_new * S
        
        score = local_score(f_new)
        
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            stagnation = 0
        else:
            stagnation += 1
            # If stuck for too long, try more aggressive step
            if stagnation > 20:
                t = t * 2.0
                stagnation = 0
        
        f = f_new
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: score={score:.6f}, best={best_score:.6f}, t={t:.4f}, M={M:.4f}, time={elapsed:.1f}s")
    
    return best_f, best_score


def fft_phase_init(n, rng, n_iter=100):
    """Gerchberg-Saxton style initialization."""
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    
    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def chirp_init(n, rng, alpha=0.5):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def grad_soft_max_conv(a, tau):
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    value = c_max + tau * np.log(s)
    w = e / s
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = (2.0 / tau) * corr[n - 1:2 * n - 1]
    return value, g, c


def optimize_gradient(a_init, time_limit, rng):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S
    theta = np.log(a / S + 1e-12)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr0 = 3.0
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)
    tau_start = 20.0 / n
    tau_end = 0.1 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)

        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)

        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 200 == 0:
            a_check = softmax_normalize(theta) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()

    a_final = softmax_normalize(best_theta) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
    return a_final, best_score


def local_search(a_init, time_limit, rng):
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9997
    conv = fft_convolve(current, current)
    top_k = max(4, n // 32)

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.35:
            # Peak-directed reduction
            top_lags = np.argsort(conv)[-top_k:]
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            n_reduce = max(2, n // 10)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.02, size=len(bot_idx)))
        elif r < 0.55:
            # Multi-block perturbation
            n_blocks = int(rng.integers(2, 6))
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 15))))
                bs = int(rng.integers(0, n - bw))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0.0, 0.04, size=bw))
        elif r < 0.7:
            # Partial symmetrization
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.15 * (sym - current)
            candidate += 0.008 * rng.standard_normal(n)
        elif r < 0.85:
            # Sparse modification
            n_mod = max(1, n // 6)
            idx = rng.choice(n, size=n_mod, replace=False)
            candidate[idx] *= np.exp(rng.normal(0.0, 0.12, size=n_mod))
        else:
            # Global noise
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current
        else:
            candidate = candidate / s * np.sum(current)

        new_score = local_score(candidate)
        if new_score < current_score:
            current = candidate
            current_score = new_score
            conv = fft_convolve(current, current)
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 10:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score
                    conv = fft_convolve(current, current)

        temp *= cooling

    return best, best_score

def fft_phase_init(n, rng, n_iter=100):
    """Gerchberg-Saxton: flatten magnitude spectrum, clip to non-negative."""
    N_fft = 1

        N_fft <<= 1
    a = np.ones(n) / n
    for _ in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag

        s = np.sum(a)
        if s < 1e-10:

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)
        step += 1
        m1 = b1 * m1 + (1.0 - b1) * g_theta
        m2 = b2 * m2 + (1.0 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        if step % 200 == 0:
            a_check = softmax_normalize(theta) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()

    a_final = softmax_normalize(best_theta) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
        best_theta = theta.copy()
    return softmax_normalize(best_theta) * S, best_score


def local_search(a_init, time_limit, rng):
    """Simulated annealing with convolution-peak-directed perturbations."""
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.004
    cooling = 0.9996
    conv = fft_convolve(current, current)
    top_k = max(4, n // 32)

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.35:
            top_lags = np.argsort(conv)[-top_k:]
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            n_red = max(2, n // 10)
            top_idx = np.argsort(contrib)[-n_red:]
            bot_idx = np.argsort(contrib)[:n_red]
            candidate[top_idx] *= np.exp(rng.normal(-0.06, 0.02, n_red))
            candidate[bot_idx] *= np.exp(rng.normal(0.06, 0.02, n_red))
        elif r < 0.55:
            n_blks = int(rng.integers(2, 6))
            for _ in range(n_blks):
                bw = max(1, int(rng.integers(1, max(2, n // 15))))
                bs = int(rng.integers(0, n - bw))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0.0, 0.04, bw))
        elif r < 0.7:
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.2 * (sym - current)
            candidate += 0.008 * rng.standard_normal(n)
        elif r < 0.85:
            n_mod = max(1, n // 6)
            idx = rng.choice(n, size=n_mod, replace=False)
            candidate[idx] *= np.exp(rng.normal(0.0, 0.15, n_mod))
        else:
            candidate = current * np.exp(rng.normal(0.0, 0.012, n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / s * np.sum(current)

        new_score = local_score(candidate)
        if new_score < current_score:
            current = candidate
            current_score = new_score
            conv = fft_convolve(current, current)
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 10:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    current = candidate
                    current_score = new_score
                    conv = fft_convolve(current, current)
        temp *= cooling

    return best, best_score


def _sort_key(x):
    return x[0]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP-based optimization (main), gradient refinement,
    and local search polish for minimizing 2n*max(conv)/sum^2.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # === Phase 1: Initialization ===
    print("\n=== Phase 1: Initialization ===")
    candidates = []
    n_list = [256, 384, 512]

    for n in n_list:
        if time.monotonic() > deadline - 600:
            break
        a1 = fft_phase_init(n, rng, n_iter=100)
        s1 = local_score(a1)
        print(f"  n={n} fft: {s1:.6f}")
        candidates.append((s1, n, a1.copy()))

        a2 = random_init(n, rng, 0.2 + 0.3 * rng.random())
        s2 = local_score(a2)
        print(f"  n={n} rand: {s2:.6f}")
        candidates.append((s2, n, a2.copy()))

        a3 = chirp_init(n, rng, 0.3 + 0.4 * rng.random())
        s3 = local_score(a3)
        print(f"  n={n} chirp: {s3:.6f}")
        candidates.append((s3, n, a3.copy()))

        a4 = cosine_init(n, rng, n_freqs=3)
        s4 = local_score(a4)
        print(f"  n={n} cos: {s4:.6f}")
        candidates.append((s4, n, a4.copy()))

    if start_seq is not None:
        candidates.append((local_score(start_seq), len(start_seq), start_seq.copy()))

    candidates.sort(key=_sort_key)
    print(f"  Best init: {candidates[0][0]:.6f} (n={cand0[1] if (cand0:=candidates[0]) else '?'})")

    # Fix: just print properly
    print(f"  Top candidates: " +
          ", ".join(f"[{c[0]:.4f},n={c[1]}]" for c in candidates[:5]))

    if candidates[0][0] < best_score:
        best_score = candidates[0][0]
        best_seq = candidates[0][2].copy()

    # === Phase 2: LP-based optimization (main) ===
    print("\n=== Phase 2: LP optimization ===")
    tried = set()
    max_per_lp = 450.0

    for i in range(min(6, len(candidates))):
        if time.monotonic() > deadline - 300:
            break
        s0, n, a = candidates[i]
        if n in tried:
            continue
        tried.add(n)
        t_alloc = min(max_per_lp, max(30.0, deadline - time.monotonic() - 300))
        print(f"  LP n={n} score={s0:.6f} budget={t_alloc:.0f}s")
        t_lp = time.monotonic()
        a_opt, s_opt = lp_optimize(a, t_alloc, t_start=0.5, t_end=0.02)
        el = time.monotonic() - t_lp
        print(f"  LP n={n} -> {s_opt:.6f} ({el:.0f}s)")
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}) ***")

    # === Phase 3: Second LP round on best if it came from a different length ===
    remaining = deadline - time.monotonic() - 200
    if remaining > 120 and best_seq is not None:
        print(f"\n=== Phase 3: Additional LP on best ===")
        n_best = len(best_seq)
        t_alloc = min(300.0, remaining * 0.5)
        print(f"  LP n={n_best} score={best_score:.6f} budget={t_alloc:.0f}s")
        a_opt, s_opt = lp_optimize(best_seq, t_alloc, t_start=0.3, t_end=0.01)
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: Gradient refinement ===
    remaining = deadline - time.monotonic() - 120
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Phase 4: Gradient refinement ===")
        t_alloc = min(200.0, remaining * 0.5)
        print(f"  Grad n={len(best_seq)} score={best_score:.6f} budget={t_alloc:.0f}s")
        a_grad, s_grad = optimize_gradient(best_seq, t_alloc, rng)
        print(f"  Grad -> {s_grad:.6f}")
        if s_grad < best_score:
            best_score = s_grad
            best_seq = a_grad.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 5: Local search polish ===
    remaining = deadline - time.monotonic() - 30
    if remaining > 20 and best_seq is not None:
        print(f"\n=== Phase 5: Local search ===")
        t_alloc = min(200.0, remaining)
        print(f"  LS n={len(best_seq)} score={best_score:.6f} budget={t_alloc:.0f}s")
        a_ls, s_ls = local_search(best_seq, t_alloc, rng)
        print(f"  LS -> {s_ls:.6f}")
        if s_ls < best_score:
            best_score = s_ls
            best_seq = a_ls.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 6: Final LP + quick polish if time remains ===
    remaining = deadline - time.monotonic() - 25
    if remaining > 60 and best_seq is not None:
        print(f"\n=== Phase 6: Final refinement ===")
        t_alloc = min(120.0, remaining * 0.6)
        a_opt, s_opt = lp_optimize(best_seq, t_alloc, t_start=0.15, t_end=0.01)
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"  LP -> {best_score:.6f} *** New best ***")
        remaining2 = deadline - time.monotonic() - 20
        if remaining2 > 15:
            a_ls2, s_ls2 = local_search(best_seq, min(40.0, remaining2), rng)
            if s_ls2 < best_score:
                best_score = s_ls2
                best_seq = a_ls2.copy()
                print(f"  LS -> {best_score:.6f} *** New best ***")

    # Final
    print(f"\n{'=' * 50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
