import time

"""

local search polishing and multi-length exploration.

import numpy as np
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_norm(a):
    """Objective with sum(a)=sqrt(2n): max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return conv.max()


def full_objective(a):
    """The actual objective: 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def safe_evaluate(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if sum(a_list) < 0.01:
            scale = 1.0 / max(sum(a_list), 1e-15)
            a_list = [x * scale for x in a_list]
        score = float(evaluate_sequence(a_list))
        return score
    except Exception:
        return float('inf')


def solve_lp_best_response(a, M, solver=cp.ECOS):
    """
    Solve LP: maximize sum(b) subject to b >= 0, (a*b)(k) <= M for all k.
    Returns b* or None on failure.
    """
    n = len(a)
    b = cp.Variable(n)

    constraints = [b >= 0]
    # Build convolution constraints
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        a_seg = a[js:je + 1]
        b_idx = np.arange(k - je, k - js + 1)
        constraints.append(a_seg @ b[b_idx] <= M)

    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)

    try:
        if solver == cp.ECOS:
            prob.solve(solver=cp.ECOS, max_iter=10000, abstol=1e-8, reltol=1e-8)
        elif solver == cp.SCS:
            prob.solve(solver=cp.SCS, max_iters=20000, eps=1e-8)
        else:
            prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=20000, eps=1e-7)
        except Exception:
            return None

    if b.value is None:
        return None
    return np.maximum(b.value, 0.0)


def solve_lp_dual(a, M):
    """
    Solve dual LP: minimize M * sum(y) subject to (A^T y)_j >= 1, y >= 0.
    Returns y (dual variables for each lag) or None.
    """
    n = len(a)
    y = cp.Variable(2 * n - 1)

    constraints = [y >= 0]
    for j in range(n):
        # (A^T y)_j = sum_k y_k * A[k,j] = sum_k y_k * a_j if 0<=k-j<=n-1
        # = a_j * sum_{k=j}^{j+n-1} y_k
        k_min = j
        k_max = j + n - 1
        constraints.append(a[j] * cp.sum(y[k_min:k_max + 1]) >= 1)

    obj = cp.Minimize(M * cp.sum(y))
    prob = cp.Problem(obj, constraints)

    try:
        prob.solve(solver=cp.ECOS, max_iter=10000, abstol=1e-8, reltol=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=20000, eps=1e-7)
        except Exception:
            return None

    if y.value is None:
        return None
    return np.maximum(y.value, 0.0)


def lp_iterate(a, M, t, solver=cp.ECOS, eps_floor=1e-10):
    """One LP iteration: solve LP, update a."""
    n = len(a)
    # Ensure a is positive for LP well-posedness
    a_pos = np.maximum(a, eps_floor)
    
    b = solve_lp_best_response(a_pos, M, solver=solver)
    if b is None or b.sum() < 1e-12:
        return a, M, False

    g = b / b.sum()
    # Rescale to match normalization sum(a) = sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    g *= target_sum

    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        return a, M, False
    a_new *= target_sum / s

    conv = fast_convolve(a_new)
    M_new = conv.max()
    return a_new, M_new, True

def mirror_descent_step(a, M, t_sm=50.0, eta=0.3):
    """One mirror descent step (for polishing)."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return a
    scaled = t_sm * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    thresh = w.max() * 0.005
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]

    sg_c = sg - sg.mean()
    log_f = -eta * sg_c
    log_f -= log_f.max()
    a_new = a * np.exp(log_f)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        return a
    a_new /= s
    return a_new


def md_optimize(a0, time_limit, t_sm_init=50.0, eta_init=0.3):
    """Mirror descent optimization (for polishing)."""
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0).astype(np.float64).copy()

def local_search(a0, time_limit, rng, n_peaks=10):
    """Informed local search: move mass from high-contribution to low-contribution positions."""
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) * target_sum / n
    else:
        a *= target_sum / s

    conv = fast_convolve(a)
    best_M = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    T = 0.01
    count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1.0))
        T = 0.01 * (1.0 - frac) ** 2 + 0.0003

        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Find peak positions
        thresh = max_c * 0.8
        peak_pos = np.where(conv >= thresh)[0]
        if len(peak_pos) == 0:
            peak_pos = np.array([np.argmax(conv)])
        peak_pos = peak_pos[:n_peaks]

        # Choose a peak to target
        k = int(rng.choice(peak_pos))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total = products.sum()
        if total < 1e-15:
            continue
        probs = products / total
        top_j = js + int(rng.choice(je - js + 1, p=probs))

        # Choose target: random position with low contribution to peaks
        target_j = int(rng.integers(n))
        for _ in range(3):
            cand = int(rng.integers(n))
            contrib = 0.0
            for pk in peak_pos[:5]:
                d = pk - cand
                if 0 <= d <= n - 1:
                    contrib += a[d]
            if contrib < a.mean():
                target_j = cand
                break

        if target_j == top_j:
            continue

        delta = a[top_j] * rng.uniform(0.01, 0.05)
        if delta < 1e-12:
            continue

        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue

        new_conv = fast_convolve(a)
        new_M = new_conv.max()
        delta_obj = new_M - conv.max()

        if delta_obj < 0:
            conv = new_conv
            if new_M < best_M:
                best_M = new_M
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
                conv = fast_convolve(a)
        count += 1

    return best_a, best_M


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n)
    for sf in [0.1, 0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'gauss_{sf}'] = g
    for freq in [1, 2, 3, 5]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        inits[f'sin_{freq}'] = s
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bimodal'] = bi
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits[f'rand_{alpha}'] = r
    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0)
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
        if pp.sum() > 1e-10:
            inits['parent'] = pp
    return inits

                    record(a_opt2)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = safe_evaluate(best_sequence)

    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    elapsed_total = time.monotonic() - t_start
    print(f"Total time: {elapsed_total:.1f}s")

    return [float(x) for x in best_sequence]
