```python
from scipy.optimize import linprog

def lp_direction(f, M):
    """Find direction g that maximizes sum(g) subject to conv(f,g) <= M, g >= 0."""
    n = len(f)
    # conv(f, g)[k] = sum_i f[i] * g[k-i] for valid i
    # We need: for each k in 0..2n-2, sum_i f[i] * g[k-i] <= M
    # The matrix A has rows corresponding to k, columns to g[j]
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_constraint = np.full(2*n-1, M)
    
    result = linprog(c, A_ub=A, b_ub=b_constraint, bounds=[(0, None)]*n, method='highs')
    if result.success:
        return result.x
    return np.ones(n) / n
```

```python
Uses the state-of-the-art LP direction-finding method combined with
subgradient polishing, multi-length exploration, and adaptive step sizes.

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ b = convolve(a, b)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        C[k, js:je + 1] = a[k - je:k - js + 1][::-1]
    return C


def lp_solve_direction(a, M_val):
    """Solve LP: max 1^T b s.t. C(a) b <= M_val, b >= 0. Returns b or None."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_val)
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return None
```
