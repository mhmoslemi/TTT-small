def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def power_sum_gradient(a, c, p):
    """
    Gradient of (sum c[k]^p)^(1/p) w.r.t. a, using FFT cross-correlation.
    c must be the full self-convolution of a (length 2n-1).
    """
    n = len(a)
    c_pos = np.maximum(c, 0.0)
    c_p = c_pos ** p
    total = c_p.sum()
    if total < 1e-300:
        return np.zeros(n)
    scale = total ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1) * scale

    fsize = 1 << (3 * n - 2).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    # R[m] = sum_j a[j] * w[j+m] = ifft(conj(fft(a)) * fft(w))[m]
    R = np.real(ifft(np.conj(A) * W))[:n]
    grad = 2.0 * R
    return grad


def toppk_subgradient(a, c, top_k):
    """Subgradient using top-k active lags via FFT cross-correlation."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)

    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = top_vals[i] / total

    fsize = 1 << (3 * n - 2).bit_length()
    A = fft(a, fsize)
    W = fft(w_full, fsize)
    R = np.real(ifft(np.conj(A) * W))[:n]
    return 2.0 * R

    inits.append(("bump_asym", np.exp(-0.5 * ((tt - 0.3) * 5.0) ** 2) + 0.05))

    # Try a secondary length for diversity
    n2 = min(n * 2, 16384) if n <= 8192 else n // 2
    if n2 != n and 1024 <= n2 <= 16384:
        tt2 = np.linspace(0, 1, n2, endpoint=False)
        inits.append(("ramp1_n2", (tt2 + 0.01) ** 1))
        inits.append(("ramp2_n2", (tt2 + 0.01) ** 2))
        inits.append(("rand_n2", rng.random(n2) + 0.05))

    n_inits = len(inits)
    usable = max(10.0, deadline - time.monotonic())
    per_init = usable * 0.65 / n_inits

    print("=== Phase 1: Exploration (%d inits, ~%.1f sec each) ===" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 120:
            break
        ni = len(arr)
        if ni < 64:
            continue
        Si = np.sqrt(2.0 * float(ni))
        t0 = time.monotonic()

        # Power-sum Nesterov
        seg1 = min(per_init * 0.6, max(2.0, time_budget_rem(deadline) * 0.1))
        a_ps, score_ps = run_nesterov_powersum(arr, ni, Si, seg1, rng,
                                                p_init=4.0, p_max=96.0,
                                                step_scale=1.0)
        a_norm = safe_normalize(a_ps, ni)
        c_chk = fft_convolve_self(a_norm)
        sc = c_chk.max()
        if sc < global_best:
            global_best = sc
            global_a = a_norm.copy()
            print("  PSum [%s] n=%d: %.6f" % (name, ni, global_best))

        # Top-k polish from power-sum result
        rem = per_init - (time.monotonic() - t0)
        if rem > 3:
            a_pk, score_pk = run_nesterov_toppk(a_ps, ni, Si, rem, rng,
                                                 top_k_init=5, top_k_max=128)
            a_norm2 = safe_normalize(a_pk, ni)
            c2 = fft_convolve_self(a_norm2)
            if c2.max() < global_best:
                global_best = c2.max()
                global_a = a_norm2.copy()
                print("  TopK [%s] n=%d: %.6f" % (name, ni, global_best))

    print("=== After Phase 1: global_best=%.6f, best_n=%s ===" %
          (global_best, len(global_a) if global_a is not None else 0))

    # --- Phase 2: Intensive polish on the best candidate ---
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = deadline - time.monotonic()

        # High-p power-sum polish
        seg = max(2.0, remaining * 0.2)
        a_p1, s1 = run_nesterov_powersum(global_a, n_best, S_best, seg, rng,
                                          p_init=16.0, p_max=256.0,
                                          step_scale=0.8)
        a_p1 = safe_normalize(a_p1, n_best)
        c1 = fft_convolve_self(a_p1)
        if c1.max() < global_best:
            global_best = c1.max()
            global_a = a_p1.copy()
            print("  Polish PSum high-p: %.6f" % global_best)

        # Top-k with large k
        remaining2 = deadline - time.monotonic()
        seg2 = max(2.0, remaining2 * 0.2)
        a_p2, s2 = run_nesterov_toppk(global_a, n_best, S_best, seg2, rng,
                                       top_k_init=16, top_k_max=384)
        a_p2 = safe_normalize(a_p2, n_best)
        c2 = fft_convolve_self(a_p2)
        if c2.max() < global_best:
            global_best = c2.max()
            global_a = a_p2.copy()
            print("  Polish TopK high-k: %.6f" % global_best)

        # Mirror descent for different geometry
        remaining3 = deadline - time.monotonic()
        if remaining3 > 5:
            seg3 = max(2.0, remaining3 * 0.15)
            a_p3, s3 = run_mirror_descent(global_a, n_best, S_best, seg3, rng,
                                           eta_init=0.1, top_k=20)
            a_p3 = safe_normalize(a_p3, n_best)
            c3 = fft_convolve_self(a_p3)
            if c3.max() < global_best:
                global_best = c3.max()
                global_a = a_p3.copy()
                print("  Polish Mirror: %.6f" % global_best)

        # Final aggressive top-k polish
        remaining4 = deadline - time.monotonic()
        if remaining4 > 3:
            a_p4, s4 = run_nesterov_toppk(global_a, n_best, S_best, remaining4, rng,
                                           top_k_init=64, top_k_max=512)
            a_p4 = safe_normalize(a_p4, n_best)
            c4 = fft_convolve_self(a_p4)
            if c4.max() < global_best:
                global_best = c4.max()
                global_a = a_p4.copy()
                print("  Final polish: %.6f" % global_best)

    # --- Phase 3: Additional exploration with different strategies ---
    remaining5 = deadline - time.monotonic()
    if remaining5 > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))

        # Try perturbed versions of the best
        for trial in range(5):
            if time.monotonic() > deadline - 30:
                break
            perturbation = rng.normal(0, 0.02 * S_best / np.sqrt(n_best), n_best)
            a_trial = project_simplex(global_a + perturbation, S_best)
            seg = max(3.0, remaining5 * 0.15)
            a_tr, sc_tr = run_nesterov_powersum(a_trial, n_best, S_best, seg, rng,
                                                 p_init=8.0, p_max=128.0,
                                                 step_scale=1.0)
            a_tr = safe_normalize(a_tr, n_best)
            c_tr = fft_convolve_self(a_tr)
            if c_tr.max() < global_best:
                global_best = c_tr.max()
                global_a = a_tr.copy()
                print("  Trial %d: %.6f" % (trial, global_best))

    # --- Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(1024, dtype=np.float64) * np.sqrt(2.0 * 1024) / 1024

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal max_conv=%.6f" %
          (len(global_a), global_best))
    return [float(x) for x in global_a]


def time_budget_rem(deadline):
    return max(0.0, deadline - time.monotonic())
