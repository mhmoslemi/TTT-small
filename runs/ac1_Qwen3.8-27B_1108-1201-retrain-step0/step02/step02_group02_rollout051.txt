    cols = []

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def build_conv_matrix(a):
    """Build sparse convolution matrix C where (C @ b)[k] = (a*b)[k]."""
    n = len(a)
    rows, cols, vals = [], [], []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_direction(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0."""
    n = len(a)
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    A_ub = np.vstack([C.toarray(), -np.eye(n)])
    b_ub = np.concatenate([np.full(2*n-1, M), np.zeros(n)])
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return a.copy()  # fallback

    a = a * (np.sqrt(2*len(a)) / S0)
    n = len(a)
    
    best_a = a.copy()
    best_score = local_score(a)
    
    t0 = time.monotonic()
    patience = 0
    max_patience = 50
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:

        g = lp_direction(a, M)
        g_sum = np.sum(g)
        if g_sum < 1e-10:

        g = g * (np.sum(a) / g_sum)
        
        # Line search for t

        best_s = local_score(a)
        for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:

            a_trial = np.clip(a_trial, 0, 1000)

from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Fast evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s

def build_conv_indices(n):
    """Precompute (row, col) indices for the convolution matrix."""
    rows = []
    cols = []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)

def make_conv_matrix(a, row_idx, col_idx):
    """Build sparse convolution matrix from precomputed indices."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)])
    return csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))

def solve_lp(a, M, row_idx, col_idx):
    """Solve: max sum(b) s.t. C@b <= M*1, b >= 0."""
    n = len(a)
    C = make_conv_matrix(a, row_idx, col_idx)
    c_obj = -np.ones(n)
    # A_ub = [C; -I]
    # b_ub = [M*1(2n-1); 0(n)]
    A_ub = C.toarray()
    b_ub_top = np.full(2*n-1, M)
    # Add -I constraints
    full_A = np.vstack([A_ub, -np.eye(n)])
    full_b = np.concatenate([b_ub_top, np.zeros(n)])
    res = linprog(c_obj, A_ub=full_A, b_ub=full_b, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

def topk_grad(a, tau):
    """Soft-max gradient of max convolution."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n-1:2*n-1]
    return value, g

def gradient_polish(a_init, time_limit, rng):
    """Gradient-based polish using soft-max parameterization."""
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a0 = a_init.copy()
    theta = np.log(a0 / np.sum(a0) + 1e-10)
    
    best_score = local_score(a0)
    best_theta = theta.copy()
    no_improve = 0
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        
        # Temperature schedule
        if elapsed_frac < 0.5:
            tau = (30.0/n) * (0.02/n) ** (elapsed_frac / 0.5)
        else:
            tau = (0.02/n) * (0.003/n) ** ((elapsed_frac-0.5)/0.5)
        
        a = softmax_normalize(theta) * S_target
        value, g_a = topk_grad(a, tau)
        
        inner = np.dot(a, g_a) / S_target
        g_theta = a * (g_a - inner)
        
        step += 1
        m_buf = beta1*m_buf + (1-beta1)*g_theta
        v_buf = beta2*v_buf + (1-beta2)*g_theta**2
        m_hat = m_buf / (1 - beta1**step)
        v_hat = v_buf / (1 - beta2**step)
        
        lr = 2.0 / np.sqrt(step + 1)
        if elapsed_frac > 0.7:
            lr *= 0.5
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if step % 100 == 0:
            a_check = softmax_normalize(theta) * S_target
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
                no_improve = 0
            else:
                no_improve += 100
            
            if no_improve > 2000:
                theta = best_theta + 0.2 * rng.standard_normal(n)
                no_improve = 0
        
        if step % 1000 == 0:
            a_c = softmax_normalize(theta) * S_target
            print(f"    Grad iter={step}, score={local_score(a_c):.6f}, best={best_score:.6f}")
    
    best_a = softmax_normalize(best_theta) * S_target
    return best_a, best_score

def coordinate_polish(a_init, time_limit, rng):
    """Coordinate descent polish."""
    current = a_init.copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            base = current[j]
            if base < 1e-8:
                continue
            for frac in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                trial = base * frac
                current[j] = trial
                new_s = local_score(current)
                if new_s < current_score:
                    current_score = new_s
                    improved = True
                    if new_s < best_score:
                        best_score = new_s
                        best = current.copy()
                else:
                    current[j] = base
                    break
        
        if not improved:
            # Random perturbation
            mask = rng.random(n) < 0.1
            current[mask] *= np.exp(rng.normal(0, 0.1, size=mask.sum()))
            current = np.clip(current, 0, 1000)
            new_s = local_score(current)
            if new_s < current_score * 1.001:
                current_score = new_s
        
        iteration += 1
        if iteration % 10 == 0:
            print(f"    CD iter={iteration}, cur={current_score:.6f}, best={best_score:.6f}")
    
    return best, best_score

def peak_shave(a_init, time_limit, rng):
    """Peak-directed mass redistribution."""
    current = a_init.copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.001
    cooling = 0.9999
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))
        
        # Compute peak contributions
        contribs = np.zeros(n)
        for i in range(n):
            if 0 <= peak_idx - i < n:
                contribs[i] = current[i] * current[peak_idx - i]
        
        r = rng.random()
        if r < 0.4:
            top_k = max(4, n // 16)
            bottom_k = max(4, n // 8)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            bottom_idx = np.argsort(contribs)[:bottom_k]
            red = rng.uniform(0.01, 0.04)
            current[top_idx] *= (1 - red)
            current[bottom_idx] *= (1 + red * len(top_idx) / len(bottom_idx))
        elif r < 0.6:
            # LP-informed: reduce top contributor, add to random
            top_idx = np.argsort(contribs)[::-1][:max(4, n//12)]
            j_red = top_idx[int(rng.integers(0, len(top_idx)))]
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red:
                amt = current[j_red] * rng.uniform(0.01, 0.03)
                current[j_red] = max(0, current[j_red] - amt)
                current[j_inc] += amt
        elif r < 0.8:
            width = max(2, int(rng.integers(2, n//6)))
            start = int(rng.integers(0, max(1, n-width)))
            factor = np.exp(rng.normal(0, 0.02))
            current[start:start+width] *= factor
        else:
            current *= np.exp(rng.normal(0, 0.003, size=n))
        
        current = np.clip(current, 0, 1000)
        new_score = local_score(current)
        
        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current_score = new_score
        
        temp *= cooling
        if step % 500 == 0:
            print(f"    PS step={step}, cur={current_score:.6f}, best={best_score:.6f}")
    
    return best, best_score

def make_init(n, rng, method='broadband'):
    """Create structured initialization."""
    t = np.arange(n, dtype=np.float64)
    if method == 'broadband':
        a = np.ones(n)
        for k in range(1, 20):
            freq = k * (1 + 0.7 * rng.random())
            phase = rng.uniform(0, 2*np.pi)
            amp = 0.12 / np.sqrt(k)
            a += amp * np.cos(2*np.pi*freq*t/n + phase)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'smooth':
        a = 1.0 + 0.3 * np.sin(2*np.pi*3*t/n) + 0.2 * np.cos(2*np.pi*7*t/n + 0.5)
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'random':
        return np.clip(1.0 + 0.5 * rng.standard_normal(n), 0.01, None)
    else:
        return np.ones(n)

def _lp_refine(a_init, row_idx, col_idx, time_limit, rng):
    """LP-based iterative refinement with line search."""
    a = a_init.copy()
    n = len(a)
    best_a = a.copy()
    best_score = local_score(a)
    
    t0 = time.monotonic()
    patience = 0
    max_patience = 100
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        c = fft_convolve(a, a)
        M = np.max(c)
        
        # Solve LP
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            # LP failed, use gradient direction instead
            tau = 1.0 / n
            _, g = topk_grad(a, tau)
            g = np.abs(g)  # Ensure non-negative
            g_sum = np.sum(g)
            if g_sum < 1e-10:
                break
            g = g / g_sum
        
        a_sum = np.sum(a)
        g = g * (a_sum / np.sum(g))
        
        # Line search
        best_t = 0.0
        best_s = best_score if local_score(a) < best_score else local_score(a)
        cur_s = local_score(a)
        
        for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            a_trial = (1-t)*a + t*g
            a_trial = np.clip(a_trial, 0, 1000)
            s_trial = local_score(a_trial)
            if s_trial < cur_s:
                cur_s = s_trial
                best_t = t
        
        if best_t == 0:
            patience += 1
            if patience > max_patience:
                # Try a larger perturbation
                a = a * (1 + 0.05 * rng.standard_normal(n))
                a = np.clip(a, 0.01, 1000)
                patience = 0
                continue
            # Small random perturbation
            a += 0.005 * rng.standard_normal(n)
            a = np.clip(a, 0.01, 1000)
        else:
            a = (1-best_t)*a + best_t*g
            a = np.clip(a, 0, 1000)
            patience = 0
        
        s = local_score(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
        
        iteration += 1
        if iteration % 100 == 0:
            print(f"    LP iter={iteration}, score={s:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score

LP-guided iterative refinement for minimizing peak autocorrelation score.
Combines LP-based direction finding (solving max sum(b) s.t. (a*b)[k]<=M, b>=0)
with adaptive line search, gradient polish, coordinate descent, and peak-directed
local search. The LP approach provides exact optimal descent directions considering
all convolution lags simultaneously, unlike gradient methods that only use the peak.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def topk_grad(a, tau):
    """
    Compute soft-max approximation and gradient of max(convolution).
    Returns (value, gradient).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return value, g


def softmax_normalize(theta):
    """Stable softmax returning values summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def build_conv_indices(n):
    """Precompute (row, col) indices for the convolution matrix C[k,j] = a[k-j]."""
    rows = []
    cols = []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_direction(a, M, row_idx, col_idx):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, and b >= 0.
    Returns the optimal b vector, or None if LP fails.
    """
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    C = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))

    c_obj = -np.ones(n)
    A_top = C.toarray()
    b_top = np.full(2 * n - 1, M)
    A_full = np.vstack([A_top, -np.eye(n)])
    b_full = np.concatenate([b_top, np.zeros(n)])

    try:
        res = linprog(c_obj, A_ub=A_full, b_ub=b_full, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_refine(a_init, row_idx, col_idx, time_limit, rng):
    """
    LP-based iterative refinement with adaptive line search.
    At each step, solve the LP to find the best direction, then line-search for step size.
    """
    a = a_init.copy()
    n = len(a)
    best_a = a.copy()
    best_score = local_score(a)

    t0 = time.monotonic()
    patience = 0
    max_patience = 80
    iteration = 0
    lp_count = 0
    lp_fail_count = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        c = fft_convolve(a, a)
        M = np.max(c)

        # Solve LP for direction
        g = solve_lp_direction(a, M, row_idx, col_idx)
        lp_count += 1

        if g is None:
            lp_fail_count += 1
            # Fall back to gradient direction
            tau = max(1.0 / n, 0.01 / n)
            _, g_raw = topk_grad(a, tau)
            g = np.abs(g_raw)
            gs = np.sum(g)
            if gs < 1e-10:
                break
            g = g / gs
        else:
            g_sum = np.sum(g)
            if g_sum < 1e-10:
                break

        a_sum = np.sum(a)
        g = g * (a_sum / np.sum(g))

        # Adaptive line search
        cur_s = local_score(a)
        best_t = 0.0
        best_s = cur_s

        # Try moderate to large steps first, then small
        for t in [0.1, 0.2, 0.3, 0.05, 0.02, 0.5, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            s_trial = local_score(a_trial)
            if s_trial < best_s - 1e-12:
                best_s = s_trial
                best_t = t

        if best_t == 0.0:
            patience += 1
            if patience > max_patience:
                # Escape: try a more aggressive perturbation
                r = rng.random()
                if r < 0.3:
                    a = a * np.exp(0.05 * rng.standard_normal(n))
                elif r < 0.6:
                    c_cur = fft_convolve(a, a)
                    peak_idx = int(np.argmax(c_cur))
                    contribs = np.zeros(n)
                    for i in range(n):
                        if 0 <= peak_idx - i < n:
                            contribs[i] = a[i] * a[peak_idx - i]
                    top_idx = np.argsort(contribs)[::-1][:max(4, n // 10)]
                    a[top_idx] *= 0.95
                    a[int(rng.integers(0, n))] *= 1.1
                else:
                    # Multi-frequency perturbation
                    freq = rng.integers(1, n // 2)
                    phase = rng.uniform(0, 2 * np.pi)
                    pert = 0.02 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
                    a = a * (1.0 + pert)
                a = np.clip(a, 0.01, 1000.0)
                patience = 0
                continue
            else:
                # Small random nudge
                a += 0.003 * rng.standard_normal(n)
                a = np.clip(a, 0.01, 1000.0)
        else:
            a = (1.0 - best_t) * a + best_t * g
            a = np.clip(a, 0.0, 1000.0)
            patience = 0

        s = local_score(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()

        iteration += 1
        if iteration % 200 == 0:
            print(f"    LP iter={iteration}, score={s:.6f}, best={best_score:.6f}, "
                  f"lp_ok={lp_count-lp_fail_count}/{lp_count}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def gradient_polish(a_init, time_limit, rng):
    """
    Soft-max gradient descent with adaptive temperature and Adam optimizer.
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a0 = a_init.copy()
    theta = np.log(a0 / np.sum(a0) + 1e-10)

    best_score = local_score(a0)
    best_theta = theta.copy()
    no_improve = 0
    patience = 3000

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit

        # Temperature annealing
        if elapsed_frac < 0.6:
            tau = (30.0 / n) * (0.01 / n) ** (elapsed_frac / 0.6)
        else:
            tau = (0.01 / n) * (0.001 / n) ** ((elapsed_frac - 0.6) / 0.4)

        a = softmax_normalize(theta) * S_target
        _, g_a = topk_grad(a, tau)

        inner = np.dot(a, g_a) / S_target
        g_theta = a * (g_a - inner)

        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g_theta
        v_buf = beta2 * v_buf + (1 - beta2) * g_theta ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)

        base_lr = 2.0 / np.sqrt(step + 1)
        if elapsed_frac > 0.8:
            lr = base_lr * 0.4
        elif elapsed_frac < 0.2:
            lr = base_lr * 1.3
        else:
            lr = base_lr

        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if step % 150 == 0:
            a_check = softmax_normalize(theta) * S_target
            score = local_score(a_check)
            if score < best_score - 1e-10:
                best_score = score
                best_theta = theta.copy()
                no_improve = 0
            else:
                no_improve += 150

            if no_improve > patience:
                r = rng.random()
                if r < 0.4:
                    theta = best_theta + 0.3 * rng.standard_normal(n)
                elif r < 0.7:
                    a_cur = softmax_normalize(best_theta) * S_target
                    c_cur = fft_convolve(a_cur, a_cur)
                    peak_idx = int(np.argmax(c_cur))
                    contribs = np.zeros(n)
                    for i in range(n):
                        if 0 <= peak_idx - i < n:
                            contribs[i] = a_cur[i] * a_cur[peak_idx - i]
                    top_k = np.argsort(contribs)[::-1][:max(4, n // 8)]
                    theta[top_k] += 0.25 * rng.standard_normal(len(top_k))
                else:
                    block = max(1, n // 16)
                    start = int(rng.integers(0, n - block))
                    theta[start:start + block] += 0.2 * rng.standard_normal(block)
                no_improve = 0

            if step % 2000 == 0:
                a_c = softmax_normalize(theta) * S_target
                print(f"    Grad step={step}, score={local_score(a_c):.6f}, best={best_score:.6f}, tau={tau:.2e}")

    best_a = softmax_normalize(best_theta) * S_target
    final_s = local_score(best_a)
    if final_s < best_score:
        best_score = final_s
        best_a = best_a.copy()
    return best_a, best_score


def peak_shave(a_init, time_limit, rng):
    """
    Peak-directed mass redistribution with simulated annealing acceptance.
    """
    current = a_init.copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.0005
    cooling = 0.99995
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))

        # Compute peak contributions
        contribs = np.zeros(n)
        for i in range(n):
            if 0 <= peak_idx - i < n:
                contribs[i] = current[i] * current[peak_idx - i]

        r = rng.random()
        if r < 0.35:
            # Reduce top contributors, boost bottom
            top_k = max(4, n // 16)
            bot_k = max(4, n // 8)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            bot_idx = np.argsort(contribs)[:bot_k]
            red = rng.uniform(0.01, 0.04)
            current[top_idx] *= (1.0 - red)
            current[bot_idx] *= (1.0 + red * len(top_idx) / len(bot_idx))
        elif r < 0.55:
            # Targeted pair: reduce top, grow random
            top_k = max(4, n // 12)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            j_red = top_idx[int(rng.integers(0, top_k))]
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red:
                amt = current[j_red] * rng.uniform(0.01, 0.04)
                current[j_red] = max(0.0, current[j_red] - amt)
                current[j_inc] += amt
        elif r < 0.75:
            # Block scaling
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, max(1, n - width)))
            factor = np.exp(rng.normal(0, 0.025))
            current[start:start + width] *= factor
        elif r < 0.9:
            # Frequency-domain perturbation
            fa = np.fft.rfft(current)
            mask = rng.random(len(fa)) < 0.05
            fa[mask] *= np.exp(1j * rng.normal(0, 0.3, size=mask.sum()))
            current = np.real(np.fft.irfft(fa, n))
            current = np.clip(current, 0.0, 1000.0)
        else:
            # Small global perturbation
            current *= np.exp(rng.normal(0, 0.003, size=n))

        current = np.clip(current, 0.0, 1000.0)
        new_score = local_score(current)

        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = current.copy()
        else:
            gap = new_score - current_score
            if gap > 0 and rng.random() < np.exp(-gap / max(temp, 1e-12)):
                current_score = new_score

        temp *= cooling
        if step % 1000 == 0:
            print(f"    PS step={step}, cur={current_score:.6f}, best={best_score:.6f}, temp={temp:.6f}")

    return best, best_score


def coordinate_polish(a_init, time_limit, rng):
    """
    Coordinate descent with geometric trial values and random restarts.
    """
    current = a_init.copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    iteration = 0

    fracs = [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4]

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            base = current[j]
            if base < 1e-8:
                continue

            # Try geometric values around base
            for frac in fracs:
                trial = base * frac
                current[j] = trial
                new_s = local_score(current)
                if new_s < current_score - 1e-12:
                    current_score = new_s
                    improved = True
                    if new_s < best_score:
                        best_score = new_s
                        best = current.copy()
                else:
                    current[j] = base
                    break

        if not improved:
            # Perturb a random subset
            k = max(1, n // 10)
            idx = rng.choice(n, size=k, replace=False)
            current[idx] *= np.exp(rng.normal(0, 0.1, size=k))
            current = np.clip(current, 0.0, 1000.0)
            new_s = local_score(current)
            if new_s < current_score * 1.002:
                current_score = new_s
            else:
                current[idx] /= np.exp(rng.normal(0, 0.1, size=k))
                current = np.clip(current, 0.0, 1000.0)

        iteration += 1
        if iteration % 20 == 0:
            print(f"    CD iter={iteration}, cur={current_score:.6f}, best={best_score:.6f}")

    return best, best_score


def make_structured_init(n, rng, method='broadband'):
    """Create diverse structured initializations."""
    t = np.arange(n, dtype=np.float64)
    if method == 'broadband':
        a = np.ones(n)
        for k in range(1, 20):
            freq = k * (1.0 + 0.7 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.12 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'two_block':
        a = np.ones(n)
        a[:n // 3] = 1.4
        a[2 * n // 3:] = 1.4
        a[n // 3:2 * n // 3] = 0.6
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'smooth':
        a = 1.0 + 0.25 * np.sin(2 * np.pi * 3.0 * t / n)
        a += 0.15 * np.cos(2 * np.pi * 7.0 * t / n + 0.7)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    else:
        return np.clip(1.0 + 0.4 * rng.standard_normal(n), 0.01, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search combining LP-guided refinement, gradient polish,
    peak-directed local search, and coordinate descent to minimize the
    evaluation score of non-negative step functions.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 35.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    n_main = len(start_seq) if start_seq is not None else 384
    a_start = start_seq.copy() if start_seq is not None else make_structured_init(n_main, rng, 'broadband')

    # Precompute LP matrix structure
    row_idx, col_idx = build_conv_indices(n_main)

    # === PHASE 1: LP-based iterative refinement (primary) ===
    print(f"\n{'=' * 50}")
    print(f"PHASE 1: LP-based refinement (n={n_main})")
    print(f"{'=' * 50}")

    t_phase1 = 0.45 * time_budget
    a_lp, score_lp = lp_refine(a_start, row_idx, col_idx, t_phase1, rng)
    print(f"  LP result: score={score_lp:.6f}")

    if score_lp < best_score:
        best_score = score_lp
        best_seq = a_lp.copy()
        print(f"  *** New best: {best_score:.6f} ***")

    # Try gradient from original start in parallel
    if time.monotonic() < deadline - 60:
        t_gr = min(0.12 * time_budget, deadline - time.monotonic() - 30)
        if t_gr > 10:
            a_gr, score_gr = gradient_polish(a_start, t_gr, rng)
            print(f"  Gradient (from start): score={score_gr:.6f}")
            if score_gr < best_score:
                best_score = score_gr
                best_seq = a_gr.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === PHASE 2: Gradient polish from current best ===
    print(f"\n{'=' * 50}")
    print(f"PHASE 2: Gradient polish (n={n_main})")
    print(f"{'=' * 50}")

    if best_seq is not None and time.monotonic() < deadline - 50:
        t_phase2 = min(0.15 * time_budget, max(20.0, deadline - time.monotonic() - 40))
        if t_phase2 > 10:
            a_g2, score_g2 = gradient_polish(best_seq, t_phase2, rng)
            if score_g2 < best_score:
                best_score = score_g2
                best_seq = a_g2.copy()
                print(f"  After gradient: {best_score:.6f}")
            else:
                print(f"  No improvement (best={best_score:.6f})")

    # === PHASE 3: Fine local search ===
    print(f"\n{'=' * 50}")
    print(f"PHASE 3: Fine local search")
    print(f"{'=' * 50}")

    if best_seq is not None and time.monotonic() < deadline - 40:
        # Peak shave
        t_ps = min(0.10 * time_budget, max(15.0, deadline - time.monotonic() - 30))
        if t_ps > 5:
            a_ps, score_ps = peak_shave(best_seq, t_ps, rng)
            if score_ps < best_score:
                best_score = score_ps
                best_seq = a_ps.copy()
                print(f"  After peak-shave: {best_score:.6f}")
            else:
                print(f"  Peak-shave: no improvement (best={best_score:.6f})")

        # Coordinate descent
        t_cd = min(0.08 * time_budget, max(8.0, deadline - time.monotonic() - 20))
        if t_cd > 5:
            a_cd, score_cd = coordinate_polish(best_seq, t_cd, rng)
            if score_cd < best_score:
                best_score = score_cd
                best_seq = a_cd.copy()
                print(f"  After CD: {best_score:.6f}")
            else:
                print(f"  CD: no improvement (best={best_score:.6f})")

    # === PHASE 4: LP refinement from a different start ===
    print(f"\n{'=' * 50}")
    print(f"PHASE 4: LP from alternate start")
    print(f"{'=' * 50}")

    if time.monotonic() < deadline - 40:
        a_alt = make_structured_init(n_main, rng, 'two_block')
        # Blend with current best
        a_blend = 0.5 * a_alt + 0.5 * best_seq
        a_blend = np.clip(a_blend, 0.01, 1000.0)
        t_alt = min(0.08 * time_budget, max(10.0, deadline - time.monotonic() - 20))
        if t_alt > 5:
            a_alt_res, score_alt = lp_refine(a_blend, row_idx, col_idx, t_alt, rng)
            print(f"  LP (alt): score={score_alt:.6f}")
            if score_alt < best_score:
                best_score = score_alt
                best_seq = a_alt_res.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === PHASE 5: Try nearby lengths ===
    print(f"\n{'=' * 50}")
    print(f"PHASE 5: Other lengths")
    print(f"{'=' * 50}")

    for n_try in [n_main + 32, n_main - 32, n_main + 64, n_main - 64]:
        if time.monotonic() > deadline - 45:
            break
        if n_try < 64 or n_try > 1024:
            continue
        t_try = min(18.0, deadline - time.monotonic() - 25)
        if t_try < 8:
            break

        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a0_try = np.interp(x_new, x_old, best_seq)
        else:
            a0_try = make_structured_init(n_try, rng, 'broadband')

        a_res, s_res = gradient_polish(a0_try, t_try, rng)
        print(f"  n={n_try}: score={s_res:.6f}")
        if s_res < best_score:
            best_score = s_res
            best_seq = a_res.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === FINAL: Quick polish if time remains ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_final = deadline - time.monotonic() - 10
        if t_final > 10:
            print(f"\nFINAL POLISH ({t_final:.0f}s remaining)")
            a_fin, score_fin = peak_shave(best_seq, t_final, rng)
            if score_fin < best_score:
                best_score = score_fin
                best_seq = a_fin.copy()
                print(f"  Final: {best_score:.6f}")

    # Verify and return
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'=' * 50}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'=' * 50}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 384
