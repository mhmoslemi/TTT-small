def adam_softmax(a_init, time_limit, rng):
    """Adam gradient descent on softmax-parameterized sequence."""
    n = len(a_init)
    S = 1.0  # normalize to sum=1
    a = a_init.copy()
    a = a / np.sum(a)  # normalize
    
    # Initialize theta from a
    theta = np.log(np.maximum(a, 1e-10))
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    # Temperature schedule
    tau_start = 50.0 / n
    tau_end = 0.1 / n
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(fft_convolve(a, a))
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Logarithmic annealing (faster than power law)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)
        
        # Compute a from theta
        a = softmax(theta)
        
        # Compute soft-max of conv and gradient
        conv = fft_convolve(a, a)
        obj, g_a, _, _ = soft_max_conv(a, tau)

        # Because da/dtheta[i] = a[i]*(delta_ij - a[j])
        inner = np.dot(a, g_a)

"""
Focused multi-phase optimization for minimizing peak autocorrelation of
non-negative step functions. Starts from provided sequence, uses Adam gradient
descent with aggressive soft-max annealing, exact subgradient refinement,
peak-directed local search, and sparse-support exploration.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence (normalized to sum=1 internally)."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def normalized_score(a):
    """Score assuming sum(a)=1, i.e., 2*n*max(conv)."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def softmax_stable(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def soft_max_conv_and_grad(a, tau):
    """Compute soft-max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    w = e / s
    value = c_max + tau * np.log(s)
    # Gradient via cross-correlation
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return value, g, c


def exact_subgrad(a):
    """Compute exact subgradient of max(conv(a,a)) at the argmax lag."""
    n = len(a)
    c = fft_convolve(a, a)
    k_star = np.argmax(c)
    g = np.zeros(n)
    # g[m] = 2*a[k_star - m] for 0 <= k_star - m < n
    for m in range(n):
        j = k_star - m
        if 0 <= j < n:
            g[m] = 2.0 * a[j]
    return g, k_star, c


def adam_softmax_optimize(a_init, time_limit, rng, tau_start_mult=60.0, tau_end_mult=0.05):
    """
    Adam gradient descent on softmax-parameterized sequence.
    Minimizes 2*n*soft_max(conv, tau) with sum(a)=1.
    Returns (best_a, best_normalized_score).
    """
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    theta = np.log(np.maximum(a, 1e-12))
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr0 = 1.5

    tau_start = tau_start_mult / n
    tau_end = tau_end_mult / n
    t0 = time.monotonic()

    best_a = a.copy()
    best_score = normalized_score(a)
    last_print = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        # Aggressive power-law annealing
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.4)

        a = softmax_stable(theta)
        _, g_a, _ = soft_max_conv_and_grad(a, tau)

        # Gradient in theta-space for softmax parameterization
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.4 * step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        theta = np.clip(theta, -40.0, 40.0)

        if step % 30 == 0:
            score = normalized_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            now = time.monotonic()
            if now - last_print > 5.0:
                last_print = now
                print(f"      adam: step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={elapsed:.1f}s")

    return best_a, best_score


def exact_subgrad_refine(a_init, time_limit, rng):
    """
    Projected subgradient refinement using exact max.
    Works in a-space with simplex projection.
    """
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = normalized_score(a)
    lr = 0.02
    step = 0

    while time.monotonic() - t0 < time_limit:
        g, k_star, c = exact_subgrad(a)

        # Project gradient to simplex tangent: remove mean component
        g_proj = g - np.mean(g)

        step += 1
        lr_current = lr / np.sqrt(1.0 + 0.1 * step)

        a_new = a - lr_current * g_proj
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = normalized_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        # Accept with some probability even if worse (to explore)
        if score < best_score + 0.001:
            a = a_new
        else:
            # Occasionally take worse steps for exploration
            if rng.random() < 0.05:
                a = a_new

        if step % 200 == 0:
            cur_score = normalized_score(a)
            print(f"      subgrad: step={step}, score={cur_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def peak_local_search(a_init, time_limit, rng, top_k=5):
    """
    Local search with peak-directed moves.
    Targets the top lags of the convolution for reduction.
    """
    n = len(a_init)
    current = a_init.copy()
    s = np.sum(current)
    if s < 1e-10:
        current = np.ones(n) / n
    else:
        current = current / s

    current_score = normalized_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9997
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        iter_count += 1

        if r < 0.30:
            # Peak-directed: reduce top contributors to peak lag
            conv = fft_convolve(current, current)
            k_star = np.argmax(conv)
            # Contribution vector for this lag
            contrib = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contrib[i] = current[i] * current[j]
            # Reduce top contributors, increase bottom
            n_mod = max(3, n // 12)
            top_idx = np.argsort(contrib)[-n_mod:]
            bot_idx = np.argsort(contrib)[:n_mod]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=n_mod))
            candidate[bot_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_mod))
        elif r < 0.50:
            # Multi-peak directed: target top-3 lags simultaneously
            conv = fft_convolve(current, current)
            top_lags = np.argsort(conv)[-top_k:]
            contrib = np.zeros(n)
            for lag in top_lags:
                for i in range(n):
                    j = lag - i
                    if 0 <= j < n:
                        contrib[i] += current[i] * current[j]
            n_mod = max(3, n // 10)
            top_idx = np.argsort(contrib)[-n_mod:]
            bot_idx = np.argsort(contrib)[:n_mod]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=n_mod))
            candidate[bot_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_mod))
        elif r < 0.60:
            # Adjacent block swap-like
            width = max(1, n // 30)
            start = int(rng.integers(0, max(1, n - 2 * width)))
            candidate[start:start + width] *= np.exp(rng.normal(-0.06, 0.02, size=width))
            candidate[start + width:start + 2 * width] *= np.exp(rng.normal(0.06, 0.02, size=width))
        elif r < 0.70:
            # Sparse support: zero out some small elements
            n_zero = max(1, n // 20)
            sort_idx = np.argsort(current)
            zero_candidates = sort_idx[:n * 2 // 5]
            zero_idx = rng.choice(zero_candidates, size=min(n_zero, len(zero_candidates)), replace=False)
            candidate[zero_idx] *= 0.1
            # Redistribute to random other positions
            inc_idx = rng.choice(n, size=n_zero, replace=False)
            candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_zero))
        elif r < 0.85:
            # Smooth noise
            candidate *= np.exp(rng.normal(0.0, 0.01, size=n))
        else:
            # Larger perturbation for escape
            n_mod = max(5, n // 8)
            idx = rng.choice(n, size=n_mod, replace=False)
            candidate[idx] *= np.exp(rng.normal(0.0, 0.05, size=n_mod))

        candidate = np.maximum(candidate, 0.0)
        s_c = np.sum(candidate)
        if s_c < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / s_c

        new_score = normalized_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < 0.01:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score

        temp *= cooling

        if iter_count % 500 == 0:
            print(f"      local: iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.1f}s")

    return best, best_score


def upsample_sequence(a, new_n, rng):
    """Upsample via interpolation with small noise."""
    n = len(a)
    if new_n <= n:
        return a[:new_n].copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.maximum(a_interp, 0.001)
    a_interp = a_interp / np.sum(a_interp)
    return a_interp
