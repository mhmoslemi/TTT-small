```python
import time

from scipy.sparse import csr_matrix

    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)

def lp_direction(f, M):
    """
    Find g that maximizes sum(g) subject to g >= 0 and (f*g)(k) <= M for all k.
    Returns the optimal g.
    """
    n = len(f)
    
    # Build the constraint matrix A such that A @ g <= M * ones
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0

    return None
```

```python
def fft_convolve(a, b):

    while size < len(a) + len(b) - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:len(a) + len(b) - 1]

    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

    x = x - np.max(x)
    e = np.exp(x)
    return e / e.sum()

def ln softmax(x):
    return np.log(x + 1e-30)

# LP-based direction
def lp_direction(a, M):
    n = len(a)
    # Build constraint matrix: (a * g)(k) <= M for all k
    # (a*g)(k) = sum_i a[i] * g[k-i] for valid i, k-i
    # So A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):

        for j in range(lo, hi):

    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)

    if res.success:
        return res.x
    return None

def lp_iteration(a, t, max_lp_iters=50):

    a = a.copy()
    n = len(a)
    s = np.sum(a)
    best = a.copy()
    best_score = score(a)
    
    for it in range(max_lp_iters):
        c = fft_convolve(a, a)
        M = np.max(c)
        
        g0 = lp_direction(a, M)
        if g0 is None:

        if s_g0 < 1e-10:
            break

        for tt in [t, t*0.5, t*2, t*0.1, t*5]:

            a_trial = np.maximum(a_trial, 0)

                best_t = tt

            a = a_new

            if sc2 < best_score:

                best = a_new2.copy()
                a = a_new2
            else:

    return best, best_score
```

  = sum_j 4n * a[k*-j] * a[j] * delta_{j,m} - a[m] * sum_j 4n * a[k*-j] * a[j]
  = 4n * a[k*-m] * a[m] - a[m] * 4n * sum_j a[k*-j] * a[j]
  = 4n * a[m] * (a[k*-m] - sum_j a[j] * a[k*-j])
  = 4n * a[m] * (a[k*-m] - (a*a)(k*))

Wait, sum_j a[j] * a[k*-j] = (a*a)(k*) = c[k*].

So d score / d theta[m] = 4n * a[m] * (a[max(0, min(n-1, k*-m))] - c[k*]) (with a[i]=0 if out of range)

Simplifying: let b = flip(a), so b[m] = a[n-1-m]. Then a[k*-m] is valid when 0 <= k*-m < n.

Actually, let me just compute it numerically. The gradient vector g_a of score wrt a is:
g_a[m] = 4n * a[k*-m] if 0 <= k*-m < n, else 0

Then the gradient wrt theta is:
g_theta = a * (g_a - dot(g_a, a))

This is the standard chain rule for softmax.

OK let me now write the full solution. I'll combine:
1. LP-based direction finding
2. L-BFGS with the max gradient
3. A soft-max smoothing objective (to handle the non-differentiability of max)
4. Local search with simulated annealing
5. Multiple sequence lengths

```python
def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def lp_update(a, step=0.1, verbose=False):
    """
    LP-based direction: find g maximizing sum(g) s.t. g>=0, (a*g)(k)<=M.
    Then blend: a_new = (1-step)*a + step*g (rescaled).
    """
    n = len(a)
    c = fft_convolve(a, a)
    M = np.max(c)
    
    # Build constraint matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[k, j] = a[k - j]
    
    # Maximize sum(g) s.t. A@g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if not res.success:
        return a, score(a)
    
    g0 = res.x
    s_g0 = np.sum(g0)
    if s_g0 < 1e-10:
        return a, score(a)
    
    # Rescale to same sum as a
    s_a = np.sum(a)
    g = g0 * (s_a / s_g0)
    
    # Try different step sizes
    best_a = a.copy()
    best_s = score(a)
    
    for step in [step, step*0.5, step*0.2, step*2]:
        a_trial = (1 - step) * a + step * g
        a_trial = np.maximum(a_trial, 0)
        if np.sum(a_trial) < 1e-10:
            continue
        a_trial = a_trial / np.sum(a_trial) * s_a
        s_trial = score(a_trial)
        if s_trial < best_s:
            best_s = s_trial
            best_a = a_trial.copy()
    
    return best_a, best_s


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_binary_init(n, density, rng):
    """Initialize with binary-like values."""
    a = (rng.random(n) < density).astype(float)
    a += 0.01 * rng.random(n)
    return a / np.sum(a)


def make_triangle_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(2*t/n - 1)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)

def local_search(a_init, time_limit, rng, verbose=True):
    """Simulated annealing local search with multiple move types."""
    current = a_init.astype(np.float64).copy()
    s = np.sum(current)
    current = current / s  # normalize to sum=1
    current_score = score(current) *  s**2 / s**2  # score is scale invariant
    # Actually score is scale invariant, so just use score(current)
    current_score = score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 1e-5
    cooling = 0.99995
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.25:
            # Targeted peak reduction
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            # Find elements that contribute most to the peak
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n, peak_idx + 1)
            for i in range(lo, hi):
                contribs[i] = current[i] * current[peak_idx - i]
            top_k = max(5, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            # Boost some other elements
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(20, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.45:
            # Pair swap: move mass from one position to another
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amt = current[i] * rng.uniform(0.05, 0.3)
                candidate[i] -= amt
                candidate[j] += amt
        elif r < 0.60:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))
        elif r < 0.75:
            # Symmetrize partially
            frac = rng.uniform(0.3, 0.8)
            candidate = (1-frac) * current + frac * (0.5*(current + current[::-1]))
            candidate *= np.exp(rng.normal(0.0, 0.01, size=n))
        elif r < 0.85:
            # Gaussian perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.02, size=n))
        else:
            # Larger random restart (partial)
            n_perturb = int(n * rng.uniform(0.1, 0.4))
            idx = rng.choice(n, size=n_perturb, replace=False)
            candidate[idx] = current[idx] * np.exp(rng.normal(0.0, 0.1, size=n_perturb))
        
        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)
        new_score = score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score
        
        temperature = max(temperature * cooling, 1e-10)
        iter_count += 1
        
        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA: iter={iter_count}, score={current_score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best, best_score
```

```python
"""
Multi-phase optimization for minimum peak autocorrelation non-negative sequences.
Combines LP-based direction finding, L-BFGS with p-norm/soft-max annealing,
and simulated annealing local search. Explores multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def score_grad_pnorm(theta, p):
    """Score and gradient for 2n * (sum c^p)^(1/p) with softmax parameterization."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    s = 2.0 * n * pnorm

    factor = sum_cp ** (1.0 / p - 1.0)

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    g_a = 4.0 * n * factor * cross[n - 1:2 * n - 1]

    inner = np.dot(g_a, a)
    g_theta = a * (g_a - inner)

    return s, g_theta


def score_grad_softmax(theta, tau):
    """Score and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau)."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    s = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    g_a = 4.0 * n * cross[n - 1:2 * n - 1]

    inner = np.dot(g_a, a)
    g_theta = a * (g_a - inner)

    return s, g_theta


def score_grad_max(theta):
    """Score and subgradient for 2n * max(conv(softmax(theta), softmax(theta)))."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    k_max = int(np.argmax(c))
    cm = c[k_max]
    s = 2.0 * n * cm

    g_a = np.zeros(n)
    lo = max(0, k_max - n + 1)
    hi = min(n, k_max + 1)
    for m in range(lo, hi):
        g_a[m] = 2.0 * a[k_max - m]

    inner = np.dot(g_a, a)
    g_theta = a * (g_a - inner)
    g_theta *= 2.0 * n

    return s, g_theta


def lp_update(a, step=0.1):
    """
    LP-based direction: find g maximizing sum(g) s.t. g>=0, (a*g)(k)<=M for all k.
    Blend: try several step sizes, return best.
    """
    n = len(a)
    c = fft_convolve(a, a)
    M = np.max(c)

    # Build constraint matrix: (a*g)(k) = sum_j A[k,j] * g[j] <= M
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[k, j] = a[k - j]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')

    if not res.success:
        return a, score(a)

    g0 = res.x
    s_g0 = np.sum(g0)
    if s_g0 < 1e-10:
        return a, score(a)

    s_a = np.sum(a)
    g = g0 * (s_a / s_g0)

    best_a = a.copy()
    best_s = score(a)

    for st in [step, step * 0.5, step * 0.25, step * 2.0]:
        a_trial = (1.0 - st) * a + st * g
        a_trial = np.maximum(a_trial, 0.0)
        sa = np.sum(a_trial)
        if sa < 1e-10:
            continue
        a_trial = a_trial / sa * s_a
        s_trial = score(a_trial)
        if s_trial < best_s:
            best_s = s_trial
            best_a = a_trial.copy()

    return best_a, best_s


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_binary_init(n, density, rng):
    """Binary-like initialization."""
    a = (rng.random(n) < density).astype(np.float64)
    a += 0.01 * rng.random(n)
    return a / np.sum(a)


def make_triangle_init(n, rng):
    """Triangle-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(2.0 * t / n - 1.0)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Bell-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)

    cooling = 0.99997
    iter_count = 0
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.25:
            # Targeted peak reduction: reduce elements contributing to peak
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)

            for i in range(lo, hi):
                contribs[i] = current[i] * current[peak_idx - i]
            top_k = max(5, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(20, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.40:
            # Pair transfer: move mass between two positions
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amt = current[i] * rng.uniform(0.05, 0.4)
                candidate[i] = max(0.0, current[i] - amt)
                candidate[j] = current[j] + amt
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        elif r < 0.70:

        else:

            continue
        candidate = candidate / cs
        new_score = score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score

        if verbose and time.monotonic() - last_print > 10.0:
            last_print = time.monotonic()
            elapsed = time.monotonic() - t0
            print(f"    SA: iter={iter_count}, score={current_score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")

        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
