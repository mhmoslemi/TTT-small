Minimize 2*n*max(conv(a,a))/sum(a)^2 using subgradient descent with momentum,
adaptive step sizes, top-K active lag tracking, and coordinate descent polishing.

Key differences from previous approach:

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient_fft(a, active_lags, weights):
    """Compute weighted subgradient sum_k w_k * (∇c[k]) using FFT cross-correlation.
    
    (∇c[k])[m] = 2*a[k-m] if 0 <= k-m < n, else 0.
    So g = 2 * cross_corr(w_seq, a) where w_seq[k] = w_k for active lags.
    """
    n = len(a)
    if len(active_lags) == 0:
        return np.zeros(n)
    
    max_k = int(max(active_lags))
    # w_seq has indices 0..max_k
    w_seq = np.zeros(max_k + 1)
    for idx, k in enumerate(active_lags):
        w_seq[int(k)] = weights[idx]
    
    # Cross-correlation: R[m] = sum_k w_seq[k] * a[k-m]
    # = ifft(conj(FFT(a)) * FFT(w_seq)) taken appropriately
    N = n + max_k + 1
    fsize = 1 << (N - 1).bit_length()
    
    A = fft(a, fsize)
    W = fft(w_seq, fsize)
    R = np.real(ifft(np.conj(A) * W))
    
    return 2.0 * R[:n]


def project_simplex(a, S):
    """Project a onto the non-negative simplex with sum = S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    """Compute the true objective: 2*n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (s * s))


def max_conv(a):
    """Compute max of self-convolution."""
    return fft_convolve_self(a).max()


def subgradient_descent(a_init, time_budget, top_k=20, momentum=0.9, 
                        base_lr=0.01, verbose=True):
    """Subgradient descent with momentum on the simplex.
    
    Minimizes max_k c[k] where c = conv(a,a), subject to a>=0, sum(a)=S.
    """
    n = len(a_init)
    S = float(a_init.sum())
    a = a_init.copy()
    a = project_simplex(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Lower bound on max(c) given sum(c) = S^2 and len(c) = 2n-1
    lb = S * S / (2 * n - 1)
    
    v = np.zeros(n)  # momentum
    t = 0
    start = time.monotonic()
    stagnation = 0
    backtrack_count = 0
    
    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_obj - 1e-12:
            best_obj = cur_max
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Adaptive top-k: use more lags as we converge
        cur_top_k = min(top_k, max(5, int(top_k * (1 + progress))))
        
        # Find top-K active lags
        if cur_top_k >= len(c):
            active = np.arange(len(c))
        else:
            # Use argsort for top-K
            active = np.argpartition(c, -cur_top_k)[-cur_top_k:]
        
        # Weights: proportional to excess over lower bound, with floor
        excess = np.maximum(c[active] - lb, 0.0)
        if excess.sum() < 1e-15:
            # All lags at or below lower bound (shouldn't happen)
            excess = np.ones(len(active))
        weights = excess / excess.sum()
        
        # Compute subgradient
        g = compute_subgradient_fft(a, active, weights)
        
        # Project gradient to tangent space of simplex (remove mean)
        g = g - np.mean(g)
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adaptive learning rate: decrease over time
        lr = base_lr * (0.5 ** (t / 200.0))
        
        # Momentum update
        v = momentum * v - lr * g
        a_new = a + v
        
        # Project to simplex
        a_new = project_simplex(a_new, S)
        
        # Backtracking: check if objective improved
        c_new = fft_convolve_self(a_new)
        new_max = c_new.max()
        
        if new_max < cur_max + 1e-10:
            a = a_new
        else:
            # Backtrack: reduce momentum
            v = 0.5 * v
            a_new = a + 0.5 * v
            a_new = project_simplex(a_new, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max:
                a = a_new
            else:
                # Just take a smaller step
                a_temp = a - 0.1 * lr * g / gnorm
                a_temp = project_simplex(a_temp, S)
                if fft_convolve_self(a_temp).max() < cur_max:
                    a = a_temp
                v = 0.0
        
        # Periodic restart: if stagnant for too long, add small random perturbation
        if stagnation > 500:
            a_pert = a + 0.01 * np.random.randn(n)
            a_pert = project_simplex(a_pert, S)
            if fft_convolve_self(a_pert).max() < best_obj:
                a = a_pert
            stagnation = 0
        
        if verbose and t % 5000 == 0:
            print("  SGD iter %d: max_c=%.6f, best=%.6f, lr=%.6f, t=%.1fs" % 
                  (t, cur_max, best_obj, lr, elapsed))
        
        # Convergence check
        if best_obj - lb < 1e-6:
            break
    
    return best_a, best_obj


def coordinate_descent_peaks(a, time_budget, n_peaks=5, max_sweeps=100):
    """Coordinate descent that specifically targets peak reduction.
    
    For each peak lag, try to reduce the contributing elements.
    """
    n = len(a)
    S = float(a.sum())
    a = a.copy()
    a = project_simplex(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_obj:
            best_obj = cur_max
            best_a = a.copy()
        
        # Find top peaks
        peak_lags = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        improved = False
        for k in peak_lags:
            if time.monotonic() - start > time_budget:
                break
            
            # Elements contributing to c[k]: a[j] where 0<=k-j<n
            lo = max(0, int(k) - n + 1)
            hi = min(n - 1, int(k))
            
            for j in range(lo, hi + 1):
                if a[j] < 1e-10:
                    continue
                
                # Try reducing a[j] by a small amount
                # The reduction in c[k] from reducing a[j] by delta is approximately:
                # -delta * (a[k-j] + a[k-j]) = -2*delta*a[k-j] (if j != k-j)
                # or -delta * a[k-j] (if j == k-j)
                
                # But reducing a[j] also affects other lags.
                # Heuristic: try a small reduction and see if max improves.
                orig_val = a[j]
                
                # Try several reduction factors
                best_local = orig_val
                best_local_obj = cur_max
                
                for factor in [0.7, 0.5, 0.3, 0.1]:
                    a[j] = orig_val * factor
                    a = project_simplex(a, S)
                    c_check = fft_convolve_self(a)
                    obj_check = c_check.max()
                    if obj_check < best_local_obj:
                        best_local_obj = obj_check
                        best_local = orig_val * factor
                
                if best_local < orig_val - 1e-12:
                    a[j] = best_local
                    a = project_simplex(a, S)
                    improved = True
                else:
                    a[j] = orig_val  # revert
        
        if not improved:
            # Try increasing some small elements to help flatten
            c = fft_convolve_self(a)
            min_lags = np.argpartition(c, n_peaks)[:n_peaks]
            for k in min_lags:
                if a[np.argmin(a)] > 0:
                    j_min = np.argmin(a)
                    # Try increasing a[j_min] slightly
                    orig_val = a[j_min]
                    for factor in [2.0, 5.0, 10.0]:
                        a[j_min] = orig_val * factor
                        a = project_simplex(a, S)
                        c_check = fft_convolve_self(a)
                        if c_check.max() < best_obj:
                            break
                        a[j_min] = orig_val
    
    c = fft_convolve_self(a)
    if c.max() < best_obj:
        best_obj = c.max()
        best_a = a.copy()
    
    return best_a, best_obj


def accelerated_subgradient(a_init, time_budget, top_k=30, verbose=True):
    """Nesterov-accelerated subgradient descent with adaptive step size.
    
    Uses the Fenchel duality gap to estimate remaining distance to optimum.
    """
    n = len(a_init)
    S = float(a_init.sum())
    a = a_init.copy()
    a = project_simplex(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    lb = S * S / (2 * n - 1)  # lower bound
    
    # Nesterov momentum
    y = a.copy()
    v = np.zeros(n)
    beta = 0.9  # momentum
    
    t = 0
    start = time.monotonic()
    
    # Initial step size estimate
    alpha = 0.1
    
    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        c = fft_convolve_self(y)
        y_max = c.max()
        
        # Find active lags
        cur_top_k = min(top_k, max(10, int(top_k * (1 + 2 * progress))))
        if cur_top_k >= len(c):
            active = np.arange(len(c))
        else:
            active = np.argpartition(c, -cur_top_k)[-cur_top_k:]
        
        # Weighted subgradient
        excess = np.maximum(c[active] - lb, 1e-10)
        weights = excess ** 2 / (excess ** 2).sum()  # emphasize larger peaks
        
        g = compute_subgradient_fft(y, active, weights)
        g = g - np.mean(g)  # projection to tangent space
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Polyak-type step size
        # alpha_t = (F(y) - F*) / ||g||^2 * 2
        alpha_t = 2.0 * (y_max - lb) / (gnorm * gnorm + 1e-15)
        alpha_t = min(alpha_t, 1.0 / (t + 1))  # cap step size
        alpha_t = max(alpha_t, 1e-6)  # minimum step
        
        # Nesterov update
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        v_new = (beta_t - 1.0) / (beta + 1.0) * v - alpha_t * g
        y_new = y + v_new
        y_new = project_simplex(y_new, S)
        
        # Check improvement
        c_new = fft_convolve_self(y_new)
        new_max = c_new.max()
        
        if new_max < y_max:
            a = y_new
            v = v_new
            y = y_new
            if new_max < best_obj:
                best_obj = new_max
                best_a = a.copy()
        else:
            # Backtrack
            alpha_t = alpha_t * 0.5
            y_new = y - alpha_t * g
            y_new = project_simplex(y_new, S)
            c_new = fft_convolve_self(y_new)
            if c_new.max() < y_max:
                a = y_new
                y = y_new
                v = -alpha_t * g
                if c_new.max() < best_obj:
                    best_obj = c_new.max()
                    best_a = a.copy()
            else:
                v = 0.0
        
        if verbose and t % 10000 == 0:
            print("  ASGD iter %d: max_c=%.6f, best=%.6f, lb=%.6f, alpha=%.6f, t=%.1fs" %
                  (t, y_max, best_obj, lb, alpha_t, elapsed))
        
        if best_obj - lb < 1e-7:
            break
    
    return best_a, best_obj


def spectral_initialization(n, S, rng):
    """Generate a sequence with spectral properties favorable for flat convolution.
    
    Idea: choose a sequence whose DFT has approximately constant magnitude
    but varying phase, which tends to flatten the self-convolution.
    """
    # Method 1: Chirp-like sequence
    i = np.arange(n)
    # Linear chirp
    phase = 2 * np.pi * (i ** 2) / (4 * n)
    a = 1.0 + 0.3 * np.cos(phase)
    a = np.maximum(a, 0.01)
    
    # Method 2: Add some structured noise
    a += 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    
    return a * (S / a.sum())


def diverse_initialization(n, S, rng, method="blend"):
    """Generate diverse initializations."""
    i = np.arange(n)
    t = i / (n - 1)
    
    if method == "uniform":
        a = np.ones(n)
    elif method == "sine":
        a = 1.0 + 0.5 * np.sin(2 * np.pi * 3 * t)
    elif method == "sine2":
        a = 1.0 + 0.5 * np.sin(2 * np.pi * 7 * t) + 0.2 * np.cos(2 * np.pi * 13 * t)
    elif method == "gaussian":
        a = np.exp(-4 * (t - 0.5) ** 2)
    elif method == "bimodal":
        a = np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2)
    elif method == "poly":
        a = 1.0 + 0.3 * t + 0.2 * t ** 2
    elif method == "random":
        a = rng.random(n)
    elif method == "chirp":
        a = 1.0 + 0.4 * np.cos(2 * np.pi * i ** 2 / (2 * n))
    elif method == "blend":
        # Blend of structured and random
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 5 * t) + 0.2 * rng.standard_normal(n)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.01)
    return a * (S / a.sum())


def fine_tune_subgradient(a_init, time_budget, top_k=50, lr_decay=0.999, verbose=True):
    """Fine-tuning subgradient with very small steps and many active lags.
    
    Designed to be run after a coarse optimization to squeeze out last bits.
    """
    n = len(a_init)
    S = float(a_init.sum())
    a = a_init.copy()
    a = project_simplex(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    lb = S * S / (2 * n - 1)
    
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    lr = 0.001
    
    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        # Use all lags above the mean (more aggressive for fine-tuning)
        threshold = c.mean() + 0.5 * (c.max() - c.mean())
        active = np.where(c > threshold)[0]
        
        if len(active) < 5:
            active = np.argpartition(c, -10)[-10:]
        if len(active) > top_k:
            # Keep top top_k
            top_idx = np.argsort(c[active])[-top_k:]
            active = active[top_idx]
        
        # Smooth weights
        excess = np.maximum(c[active] - lb, 0)
        if excess.sum() < 1e-15:
            break
        weights = excess / excess.sum()
        
        g = compute_subgradient_fft(a, active, weights)
        g = g - np.mean(g)
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Very small adaptive step
        alpha = lr * (lr_decay ** t)
        
        # Momentum
        v = 0.85 * v - alpha * g
        a_new = a + v
        a_new = project_simplex(a_new, S)
        
        c_new = fft_convolve_self(a_new)
        if c_new.max() <= cur_max:
            a = a_new
            if c_new.max() < best_obj:
                best_obj = c_new.max()
                best_a = a.copy()
        else:
            v = 0.5 * v
            
        if verbose and t % 20000 == 0:
            print("    Fine-tune iter %d: max_c=%.8f, best=%.8f, t=%.1fs" %
                  (t, cur_max, best_obj, elapsed))
        
        if best_obj - lb < 1e-8:
            break
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function.
    
    Strategy:
    1. Load previous best if available
    2. Coarse optimization with accelerated subgradient (multiple starts)
    3. Fine-tuning with small-step subgradient
    4. Coordinate descent polish targeting specific peaks
    5. Final verification
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    global_best_obj = np.inf
    global_a = None
    global_n = None
    
    def try_update(a, source=""):
        nonlocal global_best_obj, global_a, global_n
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        obj = max_conv(a) / (a.sum() ** 2) * 2 * len(a)
        if obj < global_best_obj:
            global_best_obj = obj
            global_a = a.copy()
            global_n = len(a)
            if source:
                print("  NEW BEST [%s]: obj=%.6f, n=%d" % (source, obj, len(a)))
    
    # Load previous best
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw.copy()
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given sequence: n=%d, obj=%.6f" % (n_given, ver))
            try_update(given_a, "given")
    except Exception as e:
        print("Could not load given sequence: %s" % str(e))
    
    # Phase 1: Coarse optimization (60% of time)
    phase1_budget = 0.6 * (deadline - time.monotonic())
    print("Phase 1: Coarse optimization (budget=%.0fs)" % phase1_budget)
    
    # Determine which lengths to try
    lengths_to_try = []
    if given_a is not None:
        lengths_to_try.append(len(given_a))
    else:
        lengths_to_try = [4096, 2048]
    
    # If we have time, also try a different length
    remaining_after_phase1 = (deadline - phase1_budget)
    if remaining_after_phase1 > 60:
        if 4096 not in lengths_to_try:
            lengths_to_try.append(4096)
        if 8192 not in lengths_to_try and remaining_after_phase1 > 120:
            lengths_to_try.append(8192)
    
    per_length_time = phase1_budget / len(lengths_to_try)
    
    for n in lengths_to_try:
        if time.monotonic() > deadline - 30:
            break
        
        S = np.sqrt(2.0 * n)
        remaining = min(per_length_time, deadline - time.monotonic() - 20)
        if remaining < 10:
            continue
        
        print("  Trying n=%d (time=%.0fs)" % (n, remaining))
        
        # Generate multiple initializations
        if n == (len(given_a) if given_a is not None else -1) and given_a is not None:
            inits = [given_a.copy()]
            # Add perturbed versions
            for _ in range(2):
                pert = given_a * (1 + 0.05 * rng.standard_normal(n))
                inits.append(project_simplex(pert, S))
        else:
            inits = []
            methods = ["blend", "sine", "chirp", "random", "bimodal"]
            for m in methods[:3]:
                inits.append(diverse_initialization(n, S, rng, m))
        
        per_init_time = remaining / len(inits)
        
        for idx, a_init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            if per_init_time < 5:
                break
            
            print("    Init %d/%d (time=%.0fs)" % (idx + 1, len(inits), per_init_time))
            
            # Use accelerated subgradient
            t0 = time.monotonic()
            a_best, obj_best = accelerated_subgradient(
                a_init, per_init_time * 0.7, top_k=30, verbose=False
            )
            
            # Fine-tune
            remaining_here = per_init_time - (time.monotonic() - t0) - 5
            if remaining_here > 3:
                a_best2, obj_best2 = fine_tune_subgradient(
                    a_best, remaining_here * 0.5, top_k=50, verbose=False
                )
                if obj_best2 < obj_best:
                    a_best = a_best2
                    obj_best = obj_best2
            
            # Convert to objective form
            # obj_best is max(c), objective = 2*n*max(c)/S^2 = 2*n*max(c)/(2n) = max(c)
            # Wait: S = sqrt(2n), so S^2 = 2n, objective = 2*n*max(c)/(2*n) = max(c)
            # So obj_best IS the objective value (when sum(a) = sqrt(2n))
            a_best = project_simplex(a_best, S)
            try_update(a_best, "ASGD n=%d init=%d" % (n, idx))
    
    # Phase 2: Fine-tuning (25% of time)
    phase2_budget = 0.25 * (deadline - time.monotonic())
    print("Phase 2: Fine-tuning (budget=%.0fs)" % phase2_budget)
    
    if global_a is not None and phase2_budget > 5:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = project_simplex(global_a.copy(), S)
        
        # Run fine-tuning subgradient with very small steps
        t0 = time.monotonic()
        a_ft, obj_ft = fine_tune_subgradient(a, phase2_budget * 0.6, top_k=80, verbose=True)
        if obj_ft < global_best_obj:
            global_best_obj = obj_ft
            global_a = a_ft.copy()
            print("  Fine-tune improved: %.6f" % obj_ft)
        
        a = project_simplex(global_a.copy(), S)
        
        # Coordinate descent on peaks
        remaining = phase2_budget - (time.monotonic() - t0) - 5
        if remaining > 3:
            t0 = time.monotonic()
            a_cd, obj_cd = coordinate_descent_peaks(a, remaining * 0.7, n_peaks=3, max_sweeps=200)
            if obj_cd < global_best_obj:
                global_best_obj = obj_cd
                global_a = a_cd.copy()
                print("  Coord descent improved: %.6f" % obj_cd)
    
    # Phase 3: Final polish and verification (15% of time)
    phase3_budget = deadline - time.monotonic()
    print("Phase 3: Final polish (budget=%.0fs)" % phase3_budget)
    
    if global_a is not None and phase3_budget > 3:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = project_simplex(global_a.copy(), S)
        
        # One more round of very fine subgradient
        t0 = time.monotonic()
        a_final, obj_final = fine_tune_subgradient(
            a, min(phase3_budget * 0.5, 30), top_k=100, lr_decay=0.9999, verbose=False
        )
        if obj_final < global_best_obj:
            global_best_obj = obj_final
            global_a = a_final.copy()
            print("  Final subgradient improved: %.6f" % obj_final)
        
        # Try a "smoothing" step: blend with a slightly smoothed version
        a = project_simplex(global_a.copy(), S)
        # Simple box smoothing
        kernel_size = 3
        kernel = np.ones(kernel_size) / kernel_size
        from scipy.ndimage import uniform_filter1d
        a_smooth = uniform_filter1d(a, size=kernel_size, mode='reflect')
        a_smooth = np.maximum(a_smooth, 0)
        
        for mix in [0.1, 0.2, 0.3]:
            a_blend = (1 - mix) * a + mix * a_smooth
            a_blend = project_simplex(a_blend, S)
            obj_blend = max_conv(a_blend)
            if obj_blend < global_best_obj:
                global_best_obj = obj_blend
                global_a = a_blend.copy()
                print("  Smoothing mix=%.1f improved: %.6f" % (mix, obj_blend))
                a = a_blend
    
    # Final verification
    if global_a is None:
        global_a = np.ones(256)
        print("WARNING: No valid solution found, using fallback")
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("FINAL: obj=%.6f (verified=%.6f), n=%d" % (global_best_obj, ver, len(global_a)))
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.6f" % ver)
    
    return final_a
