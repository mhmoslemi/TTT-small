def gradient_for_lags(a, top_idx, center):

    g=np.zeros(n, dtype=np.float64)

"""
Search for low-peak non-negative step functions.

The program uses:
  - direct black-box scoring through evaluate_sequence() when available,
  - a large bank of smooth / sparse / chirp / window / sidon-like candidates,
  - length exploration,
  - coarse-to-fine resizing,
  - true-evaluation local search with simulated annealing,
  - peak-lag directed gradient moves (including a variant that suppresses
    non-zero lags) as candidate generators,
  - final verification and fallback to a constant sequence.
"""

def fft_convolve(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.array([], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:need]


def fast_full_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    c = fft_convolve(a, a)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * np.max(c) / (s * s))


def fast_off_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1:
        return 0.0
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    c = fft_convolve(a, a)
    if c.size <= n:
        return np.inf
    # positive lags only: indices n ... 2n-2
    off_max = float(np.max(c[n:]))
    return float(2.0 * n * max(0.0, off_max) / (s * s))


def fast_score_mode(a, off_mode):
    if off_mode:
        return fast_off_score(a)
    return fast_full_score(a)

def normalize_pos(a, target_sum=1.0):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64) * float(target_sum)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-15:
        return np.full(a.shape, float(target_sum) / float(a.size), dtype=np.float64)
    return (a / s) * float(target_sum)


def build_list(a, target_sum=1.0):
    arr = normalize_pos(a, target_sum)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    if (not np.isfinite(s)) or s < 0.01:
        if arr.size == 0:
            return None
        if s > 0.0 and np.isfinite(s):
            arr = arr / s * max(1.0, float(target_sum))
        else:
            arr = np.full(arr.shape, 1.0 / float(arr.size), dtype=np.float64)
        arr = np.clip(arr, 0.0, 1000.0)
    return [float(x) for x in arr]


def score_list(seq):
    if not isinstance(seq, list) or len(seq) == 0:
        return np.inf
    try:
        val = evaluate_sequence(seq)
    except Exception:

            val = fast_full_score(np.asarray(seq, dtype=np.float64))
        except Exception:
            val = np.inf
    if not np.isfinite(val):
        return np.inf
    return float(val)


def score_array(a, target_sum=1.0):

    if seq is None:

def gen_uniform(n):
    n = int(max(1, n))
    return np.ones(n, dtype=np.float64) / float(n)


def gen_gauss(n, rng):
    n = int(max(1, n))
    x = np.linspace(0.0, 1.0, n)
    sigma = float(rng.uniform(0.05, 0.45))
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a += 0.01 * rng.standard_normal(n)
    return normalize_pos(a, 1.0)


def gen_super_gauss(n, rng):
    n = int(max(1, n))
    x = np.linspace(0.0, 1.0, n)
    p = float(rng.uniform(2.0, 25.0))
    a = np.exp(-p * np.power(np.abs(x - 0.5), 2.0))
    a += 0.01 * rng.standard_normal(n)
    return normalize_pos(a, 1.0)


def gen_tapered(n, rng):
    n = int(max(1, n))
    if n < 4:
        return gen_uniform(n)
    edge_frac = float(rng.uniform(0.10, 0.40))
    edge = int(np.clip(edge_frac * n, 2, n // 2))
    t = np.arange(edge, dtype=np.float64)
    ramp = 0.5 * (1.0 - np.cos(np.pi * t / float(edge)))
    a = np.ones(n, dtype=np.float64)
    a[:edge] = ramp
    a[-edge:] = ramp[::-1]
    a += 0.005 * rng.standard_normal(n)
    return normalize_pos(a, 1.0)


def gen_kaiser(n, rng):
    n = int(max(1, n))
    beta = float(rng.uniform(0.0, 12.0))
    a = np.kaiser(n, beta)
    a += 0.005 * rng.standard_normal(n)
    return normalize_pos(a, 1.0)


def gen_chirp(n, rng):
    n = int(max(1, n))
    if n < 3:
        return gen_uniform(n)
    x = np.linspace(0.0, 1.0, n)
    f = float(rng.uniform(1.0, 5.0))
    chirp = float(rng.uniform(0.0, 18.0))
    phase = float(rng.uniform(0.0, 2.0 * np.pi))
    theta = 2.0 * np.pi * (f * x + chirp * x * x + phase)
    base = 0.5 + 0.5 * np.cos(theta)
    a = np.exp(float(rng.uniform(0.0, 2.0)) * base)
    a += 0.01 * rng.standard_normal(n)
    return normalize_pos(a, 1.0)


def gen_exp_chirp(n, rng):
    n = int(max(1, n))
    if n < 3:
        return gen_uniform(n)
    i = np.arange(n, dtype=np.float64)
    spread = float(rng.uniform(0.5, 5.0))
    phase = float(rng.uniform(0.0, 2.0 * np.pi))
    quad = np.mod(i * (i + 1.0) / 2.0, float(n)) / float(n)
    theta = 2.0 * np.pi * spread * quad + phase
    base = 0.5 + 0.5 * np.cos(theta)
    a = np.exp(float(rng.uniform(0.0, 2.0)) * base)
    a += 0.01 * rng.standard_normal(n)
    return normalize_pos(a, 1.0)


def gen_multitone(n, rng):
    n = int(max(1, n))
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n, dtype=np.float64)
    m = int(rng.integers(1, 5))
    max_f = max(2, n // 3)
    for k in range(m):
        f = int(rng.integers(1, max_f + 1))
        amp = float(rng.uniform(0.15, 0.9)) / float(k + 1)
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        a += amp * np.cos(2.0 * np.pi * f * t + ph)
    a += 0.02 * rng.standard_normal(n)
    return normalize_pos(a, 1.0)


def gen_lognormal(n, rng):
    n = int(max(1, n))
    sigma = float(rng.uniform(0.1, 2.0))
    z = rng.standard_normal(n)
    a = np.exp(sigma * z)
    if n >= 8 and rng.random() < 0.5:
        x = np.linspace(0.0, 1.0, n)
        f = int(rng.integers(1, max(2, n // 4) + 1))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        a *= 0.5 + 0.5 * np.cos(2.0 * np.pi * f * x + ph)
    return normalize_pos(a, 1.0)


def gen_ou(n, rng):
    n = int(max(1, n))
    phi = float(rng.uniform(0.2, 0.95))
    x = 0.0
    a = np.empty(n, dtype=np.float64)
    for i in range(n):
        x = phi * x + float(rng.standard_normal())
        a[i] = abs(x) + 1e-6
    return normalize_pos(a, 1.0)


def gen_smooth_bump(n, rng):
    n = int(max(1, n))
    x = np.linspace(0.0, 1.0, n)
    p = float(rng.uniform(1.0, 25.0))
    a = 1.0 - np.power(np.abs(2.0 * x - 1.0), p)
    a = np.clip(a, 1e-5, 1.0)
    a += 0.02 * rng.standard_normal(n)
    return normalize_pos(a, 1.0)


def gen_sparse_floor(n, rng):
    n = int(max(1, n))
    p = float(rng.uniform(0.01, 0.90))
    floor = float(rng.uniform(0.0, 1.0))
    a = np.full(n, floor, dtype=np.float64)
    mask = rng.random(n) < p
    a[mask] = 1.0
    a += 0.002 * rng.standard_normal(n)
    return normalize_pos(a, 1.0)


def greedy_sidon_indicator(n, k, rng):
    n = int(max(1, n))
    k = int(max(0, min(k, n)))
    if k == 0:
        return np.zeros(n, dtype=np.float64)

    in_set = np.zeros(n, dtype=bool)
    seen_diff = np.zeros(n, dtype=bool)
    pos = []

    attempts = 0
    limit = 1000 + 2000 * k
    while len(pos) < k and attempts < limit:
        attempts += 1
        p = int(rng.integers(0, n))
        if in_set[p]:
            continue
        bad = False
        for q in pos:
            d = abs(p - q)
            if d == 0 or seen_diff[d]:
                bad = True
                break
        if bad:
            continue
        for q in pos:
            d = abs(p - q)
            seen_diff[d] = True
        in_set[p] = True
        pos.append(p)

    a = np.zeros(n, dtype=np.float64)
    for p in pos:
        a[p] = 1.0
    return a


def gen_sidon_floor(n, rng):
    n = int(max(1, n))
    k = int(max(1, min(32, int(math.sqrt(n)))))
    ind = greedy_sidon_indicator(n, k, rng)
    floor = float(rng.uniform(0.0, 1.0))
    a = floor + (1.0 - floor) * ind
    a += 0.005 * rng.standard_normal(n)
    return normalize_pos(a, 1.0)


def gen_product_cos(n, rng):

    )
    a += 0.03 * rng.standard_normal(n)
    return normalize_pos(a, 1.0)

def resize_sequence(a, n, rng, noise=0.0):
    a = np.asarray(a, dtype=np.float64)
    n = int(max(1, n))
    if a.size == 0:
        return gen_uniform(n)
    if a.size == n:
        b = a.copy()
    else:
        x_old = np.linspace(0.0, 1.0, a.size)
        x_new = np.linspace(0.0, 1.0, n)
        b = np.interp(x_new, x_old, a)
    if noise > 0.0 and rng is not None:
        b = b + float(noise) * rng.standard_normal(n)
    b = np.clip(b, 0.0, None)
    return normalize_pos(b, 1.0)


def select_top_lags(c, center, top_count, off_mode):
    vals = np.array(c, dtype=np.float64, copy=True)
    if off_mode and 0 <= center < vals.size:
        vals[center] = -np.inf
    if top_count >= vals.size:
        idx = np.arange(vals.size, dtype=np.int64)
    else:
        idx = np.argpartition(vals, -top_count)[-top_count:]
    if off_mode:
        idx = idx[idx != center]
    return idx


def gradient_for_lags(a, top_idx, center):
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    used = set()
    for ii in top_idx:
        d = abs(int(ii) - int(center))
        if d in used:
            continue
        used.add(d)
        if d == 0:
            g += 2.0 * a
        elif d < n:
            seg = a[: n - d]
            g[: n - d] += seg
            g[d:] += seg
    return g


def local_search_true(a_init, time_limit, rng, off_mode=False):
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = normalize_pos(a, 1.0)
    n = a.size
    if n <= 1:
        return a, 0.0
    if time_limit <= 0.0:
        return a, fast_score_mode(a, off_mode)

    t0 = time.monotonic()
    use_true = (n <= 1024)

    cur = a.copy()
    cur_score = score_array(cur, 1.0) if use_true else fast_score_mode(cur, off_mode)
    best = cur.copy()
    best_score = float(cur_score)

    temp = 0.02
    cooling = 0.9995
    last_print = t0
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(1e-6, time_limit))
        cand = cur.copy()
        r = float(rng.random())

        if r < 0.22:
            sigma = 0.08 * (0.03 ** frac) + 0.002
            cand = cur * np.exp(rng.normal(0.0, sigma, n))

        elif r < 0.47:
            c = fft_convolve(cur, cur)
            center = n - 1
            top_count = 12 if n > 60 else 5
            top_idx = select_top_lags(c, center, top_count, off_mode)
            g = gradient_for_lags(cur, top_idx, center)
            maxg = float(np.max(np.abs(g))) if g.size else 0.0
            if maxg > 1e-15:
                base_fast = fast_score_mode(cur, off_mode)
                best_trial = None
                best_fast = base_fast
                for factor in (0.05, 0.02, 0.01, 0.005, 0.002):
                    trial = cur - (factor / maxg) * g
                    trial = normalize_pos(trial, 1.0)
                    fs = fast_score_mode(trial, off_mode)
                    if fs < best_fast - 1e-12:
                        best_fast = fs
                        best_trial = trial
                if best_trial is not None:
                    cand = best_trial
                else:
                    cand = cur * np.exp(rng.normal(0.0, 0.02, n))
            else:
                cand = cur * np.exp(rng.normal(0.0, 0.02, n))

        elif r < 0.64:

                    width = 1
                start = int(rng.integers(0, max(1, n - width + 1)))
                logf = float(rng.normal(0.0, 0.25))
                cand[start:start + width] *= np.exp(logf)

        elif r < 0.74:
            sigma = float(rng.uniform(0.01, 0.12))
            cand = cur * np.exp(rng.normal(0.0, sigma, n))
            cand = 0.5 * (cand + cand[::-1])

        elif r < 0.84:
            q = float(rng.uniform(0.10, 0.40))
            thr = float(np.quantile(cur, q)) if n > 1 else 0.0
            cand = cur.copy()
            cand[cand < thr] *= float(rng.uniform(0.0, 0.7))
            idx = int(rng.integers(0, n))
            cand[idx] *= float(rng.uniform(1.0, 2.0))

        elif r < 0.93:
            max_width = max(2, n // 8)
            width = int(rng.integers(1, max_width + 1))
            if width < n:
                i = int(rng.integers(0, max(1, n - width + 1)))
                j = int(rng.integers(0, max(1, n - width + 1)))
                b1 = cand[i:i + width].copy()
                b2 = cand[j:j + width].copy()
                cand[i:i + width] = b2
                cand[j:j + width] = b1

        else:
            cand = cur.copy()
            if n > 1:
                vals = np.sort(np.unique(cand))
                if vals.size > 2:
                    idx = int(rng.integers(1, vals.size))
                    thr = vals[idx]
                    cand[cand > thr] = thr
            cand += 1e-8

        cand = np.clip(cand, 0.0, None)
        cs = float(np.sum(cand))
        if (not np.isfinite(cs)) or cs <= 1e-15:
            cand = cur.copy()
        else:
            cand = cand / cs

        new_score = score_array(cand, 1.0) if use_true else fast_score_mode(cand, off_mode)

        if new_score < cur_score:
            cur = cand
            cur_score = float(new_score)
            if new_score < best_score:
                best = cand.copy()
                best_score = float(new_score)
                if time.monotonic() - last_print > 10.0:
                    print(
                        f"      local_search off_mode={int(off_mode)} n={n} "
                        f"score={best_score:.6f} elapsed={elapsed:.1f}s"
                    )
                    last_print = time.monotonic()
        else:
            delta = new_score - cur_score
            if np.isfinite(delta) and delta < temp:
                accept_prob = math.exp(-delta / max(temp, 1e-12))
                if float(rng.random()) < accept_prob:
                    cur = cand
                    cur_score = float(new_score)
                    if new_score < best_score:
                        best = cand.copy()
                        best_score = float(new_score)

        temp *= cooling
        if temp < 1e-5:
            temp = 1e-5

    return best, best_score

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a low-scoring non-negative step sequence within the time budget."""
    t0 = time.monotonic()

    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 2400.0
    if not np.isfinite(total_budget) or total_budget <= 0.0:
        total_budget = 2400.0

    # Reserve a safety margin for final verification and return.
    budget = max(120.0, total_budget - 40.0)
    deadline = t0 + budget
    rng = np.random.default_rng(int(seed))

    best_score = np.inf
    best_seq = None
    records = []

    print(f"\n=== propose_candidate === seed={seed}, budget={budget:.1f}s")

    # ------------------------------------------------------------------
    # 0. Use any previously discovered sequence, if present.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            raw_prov = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        except Exception:
            raw_prov = None

        if raw_prov is not None and raw_prov.size > 0:
            prov = normalize_pos(raw_prov, 1.0)
            scales = [1.0, float(math.sqrt(2.0 * raw_prov.size)), 0.1, 10.0]
            best_score, best_seq = update_from_array(prov, best_score, best_seq, scales)
            add_record(records, best_score if best_seq is best_seq and np.isfinite(best_score) else np.inf, prov)

            raw_abs = np.abs(np.nan_to_num(raw_prov, nan=0.0, posinf=0.0, neginf=0.0))
            raw_sum = float(np.sum(raw_abs))
            if np.isfinite(raw_sum) and 0.02 <= raw_sum <= 1e6:
                best_score, best_seq = update_from_array(prov, best_score, best_seq, [raw_sum])

            print(f"Provided sequence scored. current_best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Diverse randomized candidate generation.
    # ------------------------------------------------------------------
    lengths = [32, 64, 96, 128, 192, 256, 384, 512, 768, 1024]
    if budget > 1200.0:
        lengths.append(1536)
    if budget > 1800.0:
        lengths.append(2048)

    phase1_deadline = t0 + min(900.0, max(30.0, 0.40 * budget))
    phase1_deadline = min(phase1_deadline, max(t0 + 20.0, deadline - 420.0))

    print(f"\n=== Phase 1: diverse candidates, deadline offset "
          f"{phase1_deadline - t0:.1f}s ===")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break

        if n <= 256:
            n_try = 40
        elif n <= 768:
            n_try = 24
        else:
            n_try = 12

        for _ in range(n_try):
            if time.monotonic() > phase1_deadline:
                break
            a = generate_random_parametric(n, rng)
            sc, seq = eval_candidate(a, 1.0)
            add_record(records, sc, a)
            best_score, best_seq = update_best_tuple(sc, seq, best_score, best_seq)

        print(f"  length {n}: current_best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Multi-scale evaluation of promising shapes.
    # ------------------------------------------------------------------
    if records:
        records.sort(key=record_score_key)
        print(f"\n=== Phase 2: multi-scale top candidates, best={records[0][0]:.6f} ===")

        for rec in records[:20]:
            if time.monotonic() > deadline - 180:
                break
            a = rec[2]
            n = int(a.size)
            if n <= 0:
                continue
            scales = [1.0, 0.1, float(math.sqrt(2.0 * n)), 10.0]
            best_score, best_seq = update_from_array(a, best_score, best_seq, scales)

        print(f"  after multi-scale: best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Local search from the best diverse candidates.
    # ------------------------------------------------------------------
    local_inputs = []
    if records:
        # Best per length for diversity.
        best_per_len = {}
        for rec in records:
            ln = rec[1]
            if ln not in best_per_len:
                best_per_len[ln] = rec[2]
            if len(best_per_len) >= 6:
                break
        for a in best_per_len.values():
            local_inputs.append(a)

        # Also add a few absolute top shapes.
        for rec in records[:6]:
            local_inputs.append(rec[2])

        if len(local_inputs) > 8:
            local_inputs = local_inputs[:8]

    if local_inputs:
        remaining = deadline - time.monotonic() - 60.0
        if remaining > 80.0:
            print(f"\n=== Phase 3: local search on {len(local_inputs)} shapes, "
                  f"remaining={remaining:.1f}s ===")

            # Two modes: full max suppression, and non-zero-lag suppression.
            for base_guess, a in enumerate(local_inputs):
                if time.monotonic() > deadline - 60:
                    break
                n = int(a.size)
                per_candidate = max(8.0, (deadline - time.monotonic() - 60.0) /
                                    max(1, (len(local_inputs) - base_guess) * 2))
                t_mode = min(per_candidate, 220.0)
                if t_mode < 6.0:
                    break

                print(f"  local {base_guess + 1}/{len(local_inputs)} n={n}, "
                      f"alloc={t_mode:.1f}s, off=False")
                res, _ = local_search_true(a, t_mode, rng, off_mode=False)
                scales = [1.0, 0.1, float(math.sqrt(2.0 * n)), 10.0]
                best_score, best_seq = update_from_array(res, best_score, best_seq, scales)
                print(f"    -> best={best_score:.6f}")

                if time.monotonic() < deadline - 30:
                    t_mode2 = min(t_mode, deadline - time.monotonic() - 20.0)
                    if t_mode2 >= 6.0 and n <= 1536:
                        print(f"  local {base_guess + 1}/{len(local_inputs)} n={n}, "
                              f"alloc={t_mode2:.1f}s, off=True")
                        res, _ = local_search_true(a, t_mode2, rng, off_mode=True)
                        scales = [1.0, 0.1, float(math.sqrt(2.0 * n)), 10.0]
                        best_score, best_seq = update_from_array(res, best_score, best_seq, scales)
                        print(f"    -> best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Coarse-to-fine resizing around the current best.
    # ------------------------------------------------------------------
    if best_seq is not None:
        best_arr = normalize_pos(np.asarray(best_seq, dtype=np.float64), 1.0)
        if best_arr.size > 0:
            remaining = deadline - time.monotonic() - 60.0
            if remaining > 120.0:
                print(f"\n=== Phase 4: coarse-to-fine, best={best_score:.6f} ===")

                n0 = int(best_arr.size)
                target_lengths = []

                # Downsample.
                half = max(32, n0 // 2)
                if half != n0:
                    target_lengths.append(half)

                # Upsample.
                for cand_n in (n0 * 2, 1024, 2048, 4096):
                    if cand_n > n0 and cand_n <= 4096:
                        target_lengths.append(cand_n)

                for n in target_lengths:
                    if time.monotonic() > deadline - 60:
                        break
                    a_up = resize_sequence(best_arr, n, rng, noise=0.05)
                    t_polish = min(120.0, max(10.0, deadline - time.monotonic() - 40.0))
                    res, _ = local_search_true(a_up, t_polish, rng, off_mode=False)
                    scales = [1.0, 0.1, float(math.sqrt(2.0 * n)), 10.0]
                    best_score, best_seq = update_from_array(res, best_score, best_seq, scales)
                    print(f"  resized to n={n}: best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polish on the global best.
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 12.0
        if remaining > 35.0:
            best_arr = normalize_pos(np.asarray(best_seq, dtype=np.float64), 1.0)
            n = int(best_arr.size)
            t_polish = min(300.0, max(8.0, remaining / 2.0))
            print(f"\n=== Phase 5: final polish n={n}, t={t_polish:.1f}s ===")

            res, _ = local_search_true(best_arr, t_polish, rng, off_mode=False)
            scales = [1.0, 0.1, float(math.sqrt(2.0 * n)), 10.0]
            best_score, best_seq = update_from_array(res, best_score, best_seq, scales)

            remaining = deadline - time.monotonic() - 10.0
            if remaining > 20.0 and n <= 2048:
                res, _ = local_search_true(best_arr, min(120.0, remaining), rng, off_mode=True)
                scales = [1.0, 0.1, float(math.sqrt(2.0 * n)), 10.0]
                best_score, best_seq = update_from_array(res, best_score, best_seq, scales)

    # ------------------------------------------------------------------
    # 6. Final fallback and verification.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback = gen_uniform(256)
        best_score, best_seq = eval_candidate(fallback, 1.0)

    # Ensure the returned object is a normal Python list.
    try:
        final_seq = build_list(best_seq, 1.0)
    except Exception:
        final_seq = [1.0] * 256

    if final_seq is None or len(final_seq) == 0:
        final_seq = [1.0] * 256

    final_score = score_list(final_seq)
    if not np.isfinite(final_score):
        final_seq = build_list(gen_uniform(256), 1.0)
        final_score = score_list(final_seq)

    print(f"\n{'=' * 60}")
    print(f"FINAL: n={len(final_seq)}, score={final_score:.6f}")
    print(f"{'=' * 60}")

    return final_seq
