```
import numpy as np
import time

TARGET = 1.5027

def next_pow2(m):
    size=1
    while size < m: size <<=1
    return size

def fft_convolve(a):
    n = len(a)
    if n == 0: return np.array([])
    size = next_pow2(2*n-1)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa*fa, size)[:2*n-1]

def score_c(c, n, s):
    mx = float(np.max(c)) if c.size else 0.0
    s2 = s*s
    if s2 <= 1e-24: return np.inf
    return float(2.0*n*mx/s2)

def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.ones(len(a), dtype=np.float64) if len(a)>0 else np.ones(1)
    return a/s
```

```
def resample_seq(old, new_n, rng, noise=0.02):
   old=np.asarray(old,float)
   if new_n == len(old): return normalize_nonneg(old)
   x_old = np.linspace(0.0,1.0,len(old))
   x_new = np.linspace(0.0,1.0,new_n)
   b = np.interp(x_new,x_old,old)
   if noise>0: b += noise*rng.standard_normal(new_n)
   b = np.clip(b,1e-10,None)
   b = b / float(np.sum(b))
   return b
```

```python
"""
Search for non-negative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a)**2).

Strategy:
- Normalize to sum = 1 so the objective is 2*n*max(convolve(a,a)).
- Start from the provided sequence if available.
- Use a short exponentiated-gradient-like exposure phase that moves mass away
  from indices strongly contributing to the largest convolution peaks.
- Use exact O(n) rank-two convolution updates for local pair moves.  For a move
  a_i -> a_i + d, a_j -> a_j - d, the entire change in the convolution can be
  built in O(n), evaluated for many small moves, and applied exactly.
- Greedy/limited-annealing pair search repeatedly tries:
    * peak pair moves: for a dominant lag L=i+j, unbalance the pair (i,j),
    * exposure moves: move mass from high top-peak exposure to low exposure,
    * amplitude moves: move mass from large coordinates to small coordinates,
    * random moves for escape.
- Explore several sequence lengths and finish with a long polish of the best
  found sequence.
"""

import numpy as np
import time

TARGET_UPPER = 1.5027


def next_pow2(m):
    size = 1
    while size < m:
        size <<= 1
    return size


def fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def score_c(c, n, s):
    if n <= 0 or c.size == 0:
        return np.inf
    s2 = float(s) * float(s)
    if s2 <= 1e-24:
        return np.inf
    return float(2.0 * n * float(np.max(c)) / s2)


def score_a(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.inf
    c = fft_convolve(a)
    return score_c(c, n, s)


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        a = np.ones(1, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.ones(len(a), dtype=np.float64)
    return a / s


def resample_seq(old, new_n, rng, noise=0.02):
    old = np.asarray(old, dtype=np.float64)
    if len(old) == 0:
        old = np.ones(1, dtype=np.float64)
    if new_n == len(old):
        return normalize_nonneg(old)
    x_old = np.linspace(0.0, 1.0, len(old))
    x_new = np.linspace(0.0, 1.0, new_n)
    b = np.interp(x_new, x_old, old)
    if noise > 0.0:
        b = b + noise * rng.standard_normal(new_n)
    b = np.clip(b, 1e-10, None)
    s = float(np.sum(b))
    if s <= 1e-12:
        b = np.ones(new_n, dtype=np.float64)
    else:
        b = b / s
    return b


def make_candidate_seeds(n, base_seq, best_seq, rng):
    seeds = []

    if base_seq is not None:
        seeds.append(resample_seq(base_seq, n, rng, 0.01))
    if best_seq is not None:
        seeds.append(resample_seq(best_seq, n, rng, 0.02))

    t = np.arange(n, dtype=np.float64)

    # Smooth cosine seeds with several frequencies/phases.
    frequencies = (1, 2, 3, 5, 8)
    for f in frequencies:
        phase = rng.uniform(0.0, 2.0 * np.pi)
        a = 1.0 + 0.85 * np.cos(2.0 * np.pi * f * t / max(1, n) + phase)
        a = np.clip(a, 0.01, None)
        s = float(np.sum(a))
        if s > 0.0:
            seeds.append(a / s)

    # Zero-endpoint beta-like seed.
    if n > 1:
        u = t / float(n - 1)
        a = np.sin(np.pi * u)
        a = a * a + 0.05
    else:
        a = np.ones(1, dtype=np.float64)
    a = np.clip(a, 0.01, None)
    s = float(np.sum(a))
    if s > 0.0:
        seeds.append(a / s)

    # One-sided soft ramp seed.
    if n > 1:
        u = t / float(n - 1)
        a = 1.0 + 0.9 * np.cos(np.pi * u)
    else:
        a = np.ones(1, dtype=np.float64)
    a = np.clip(a, 0.01, None)
    s = float(np.sum(a))
    if s > 0.0:
        seeds.append(a / s)

    # Random seed.
    a = 1.0 + 0.35 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    s = float(np.sum(a))
    if s > 0.0:
        seeds.append(a / s)

    scored = []
    scores = np.empty(len(seeds), dtype=np.float64)
    for idx in range(len(seeds)):
        a = seeds[idx]
        s = float(np.sum(a))
        c = fft_convolve(a)
        scores[idx] = score_c(c, n, s)
    order = np.argsort(scores)
    for k in (0, 1, 2):
        if k < len(order):
            scored.append(seeds[int(order[k])])
    return scored


def move_candidate_max(c, tmp, buf, two_a, n, d1, d2, i, j):
    """Return max convolution coefficient after a two-coordinate move."""
    np.fill(tmp, 0.0)

    # Change a_i by d1.
    tmp[i:i + n] += d1 * two_a
    tmp[2 * i] += d1 * d1

    # Change a_j by d2 = -d1.
    tmp[j:j + n] += d2 * two_a
    tmp[2 * j] += d2 * d2

    # Cross interaction term for the lag i+j.
    tmp[i + j] += 2.0 * d1 * d2

    np.add(c, tmp, out=buf)
    return float(buf.max())


def apply_pair_move(c, tmp, two_a, d1, d2, i, j, n):
    """Apply an exact two-coordinate convolution update to c in-place."""
    np.fill(tmp, 0.0)

    tmp[i:i + n] += d1 * two_a
    tmp[2 * i] += d1 * d1

    tmp[j:j + n] += d2 * two_a
    tmp[2 * j] += d2 * d2

    tmp[i + j] += 2.0 * d1 * d2

    np.add(c, tmp, out=c)


def exposure_optimize(a, c, time_budget, rng):
    """Exponentiated-gradient-like moves away from high top-peak exposure."""
    a = np.asarray(a, dtype=np.float64).copy()
    c = np.asarray(c, dtype=np.float64).copy()
    n = len(a)
    if n == 0:
        return a, np.inf

    s0 = float(np.sum(a))
    if s0 <= 1e-12:
        return a, np.inf
    a = a / s0

    current = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    best_score = current

    if time_budget <= 0.0:
        return best_a, best_score

    t0 = time.monotonic()
    if n <= 512:
        kpeaks = 14
    else:
        kpeaks = 9
    alphas = (0.05, 0.15, 0.4, 1.0, 2.0, 4.0)

    while time.monotonic() - t0 < time_budget:
        m = 2 * n - 1
        k = min(kpeaks, m)
        if k >= m:
            top = np.argsort(c)[::-1]
        else:
            top = np.argpartition(c, -k)[-k:]
            top = top[np.argsort(c[top])[::-1]]

        h = np.zeros(n, dtype=np.float64)
        for L in top:
            Li = int(L)
            i_min = Li - n + 1
            if i_min < 0:
                i_min = 0
            i_max = Li
            if i_max >= n:
                i_max = n - 1
            if i_min <= i_max:
                ia = np.arange(i_min, i_max + 1)
                ja = Li - ia
                h[ia] += a[ja]

        hmax = float(np.max(h))
        hmin = float(np.min(h))
        if hmax - hmin <= 1e-15:
            h = rng.standard_normal(n)
            hmax = float(np.max(h))
            hmin = float(np.min(h))
        if hmax - hmin <= 1e-15:
            break
        h = (h - hmin) / (hmax - hmin)

        best_sc = None
        best_an = None
        best_cn = None

        for alpha in alphas:
            an = a * np.exp(-alpha * h)
            an = np.clip(an, 1e-12, None)
            sn = float(np.sum(an))
            if sn <= 1e-12:
                continue
            an = an / sn
            cn = fft_convolve(an)
            sc = 2.0 * n * float(np.max(cn))
            if sc < current - 1e-14:
                if best_sc is None or sc < best_sc:
                    best_sc = sc
                    best_an = an
                    best_cn = cn

        if best_an is None:
            # Small random descent direction before giving up.
            hi = rng.standard_normal(n)
            hi = (hi - np.min(hi)) / (np.max(hi) - np.min(hi) + 1e-15)
            for alpha in (0.05, 0.2, 0.6):
                an = a * np.exp(-alpha * hi)
                an = np.clip(an, 1e-12, None)
                sn = float(np.sum(an))
                if sn <= 1e-12:
                    continue
                an = an / sn
                cn = fft_convolve(an)
                sc = 2.0 * n * float(np.max(cn))
                if sc < current - 1e-14:
                    if best_sc is None or sc < best_sc:
                        best_sc = sc
                        best_an = an
                        best_cn = cn
            if best_an is None:
                break

        current = best_sc
        a = best_an
        c = best_cn
        if current < best_score:
            best_score = current
            best_a = a.copy()

    # Return a confidently exact score.
    sbest = float(np.sum(best_a))
    if sbest <= 1e-12:
        return best_a, np.inf
    cbest = fft_convolve(best_a / sbest)
    final_score = score_c(cbest, len(best_a), 1.0)
    return best_a / sbest, final_score


def pair_local_search(a, c, time_budget, rng):
    """Exact pair-move local search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    if n == 0:
        return a, np.inf
    s = float(np.sum(a))
    if s <= 1e-12:
        return a, np.inf
    a = a / s

    if c is None:
        c = fft_convolve(a)
    else:
        c = np.asarray(c, dtype=np.float64)
        if c.size != 2 * n - 1:
            c = fft_convolve(a)

    m = 2 * n - 1
    current_max = float(np.max(c))
    current_score = 2.0 * n * current_max
    best_a = a.copy()
    best_score = current_score

    if time_budget <= 0.0:
        return best_a, best_score

    t0 = time.monotonic()

    tmp = np.zeros(m, dtype=np.float64)
    buf = np.empty(m, dtype=np.float64)
    two_a = 2.0 * a

    if n <= 320:
        base_cand = 48
        kpeaks = 3
        pairs_per_peak = 2
    elif n <= 768:
        base_cand = 36
        kpeaks = 3
        pairs_per_peak = 2
    else:
        base_cand = 28
        kpeaks = 2
        pairs_per_peak = 2

    fracs_peak = (0.003, 0.01, 0.03, 0.08, 0.20, 0.50, 0.80)
    fracs_high = (0.01, 0.03, 0.10, 0.30)
    fracs_rand = (0.01, 0.05, 0.20)

    no_improve = 0
    accept_count = 0
    recompute_count = 0
    perturb_count = 0
    max_perturb = 6

    while time.monotonic() - t0 < time_budget:
        if no_improve > 180:
            if perturb_count >= max_perturb:
                break
            # Escape the ridge by a small multiplicative perturbation.
            sigma = 0.07 / (2.0 ** perturb_count)
            a = a * np.exp(rng.normal(0.0, sigma, n))
            a = np.clip(a, 1e-10, None)
            sa = float(np.sum(a))
            if sa <= 1e-12:
                a = normalize_nonneg(a)
            else:
                a = a / sa
            c = fft_convolve(a)
            two_a = 2.0 * a
            current_max = float(np.max(c))
            current_score = 2.0 * n * current_max
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
            no_improve = 0
            accept_count = 0
            recompute_count = 0
            perturb_count += 1
            continue

        cand_limit = base_cand
        if no_improve > 25:
            cand_limit = base_cand * 2
        if no_improve > 80:
            cand_limit = base_cand * 3

        # Current dominant convolution lags.
        if kpeaks >= m:
            lag_idx = np.argsort(c)[::-1]
        else:
            lag_idx = np.argpartition(c, -kpeaks)[-kpeaks:]
            lag_idx = lag_idx[np.argsort(c[lag_idx])[::-1]]

        best_new = current_max
        best_i = -1
        best_j = -1
        best_d1 = 0.0
        evaluated = 0
        eps = 1e-14

        # 1) Moves directly involving pairs that sum to a dominant lag.
        for L in lag_idx:
            if evaluated >= cand_limit:
                break
            Li = int(L)
            i_min = Li - n + 1
            if i_min < 0:
                i_min = 0
            i_max = Li
            if i_max >= n:
                i_max = n - 1
            if i_min > i_max:
                continue

            ia = np.arange(i_min, i_max + 1)
            ja = Li - ia
            mask = ia < ja
            if not np.any(mask):
                continue
            ia = ia[mask]
            ja = ja[mask]
            prod = a[ia] * a[ja]

            p = min(pairs_per_peak, len(prod))
            if p <= 0:
                continue
            if p >= len(prod):
                part = np.argsort(prod)[::-1]
            else:
                part = np.argpartition(prod, -p)[-p:]
                part = part[np.argsort(prod[part])[::-1]]

            for pos in part:
                if evaluated >= cand_limit:
                    break
                i = int(ia[pos])
                j = int(ja[pos])
                A = float(a[i])
                B = float(a[j])
                if A <= 1e-15 or B <= 1e-15:
                    continue
                bound = A if A < B else B
                if A > B:
                    sign = 1.0
                elif B > A:
                    sign = -1.0
                else:
                    sign = 1.0 if rng.random() < 0.5 else -1.0

                for frac in fracs_peak:
                    d1 = sign * bound * frac
                    d2 = -d1
                    nm = move_candidate_max(c, tmp, buf, two_a, n, d1, d2, i, j)
                    evaluated += 1
                    if nm < best_new - eps:
                        best_new = nm
                        best_i = i
                        best_j = j
                        best_d1 = d1
                    if evaluated >= cand_limit:
                        break

                # Occasionally try the opposite small move (peaks elsewhere).
                if evaluated < cand_limit:
                    d1 = -sign * bound * 0.02
                    d2 = -d1
                    nm = move_candidate_max(c, tmp, buf, two_a, n, d1, d2, i, j)
                    evaluated += 1
                    if nm < best_new - eps:
                        best_new = nm
                        best_i = i
                        best_j = j
                        best_d1 = d1

        # 2) Exposure-based mass transfer: high top-peak exposure -> low.
        if evaluated < cand_limit:
            h = np.zeros(n, dtype=np.float64)
            for L in lag_idx:
                Li = int(L)
                i_min = Li - n + 1
                if i_min < 0:
                    i_min = 0
                i_max = Li
                if i_max >= n:
                    i_max = n - 1
                if i_min <= i_max:
                    ia = np.arange(i_min, i_max + 1)
                    ja = Li - ia
                    h[ia] += a[ja]

            hiK = 2 if evaluated + 16 < cand_limit else 1
            loK = 2 if evaluated + 16 < cand_limit else 1

            if hiK > 0 and loK > 0:
                src_idx = np.argpartition(h, -min(hiK, n))[-min(hiK, n):]
                src_idx = src_idx[np.argsort(h[src_idx])[::-1]]
                dst_idx = np.argpartition(h, min(loK, n))[:min(loK, n)]
                dst_idx = dst_idx[np.argsort(h[dst_idx])]

                for s_idx in src_idx:
                    if evaluated >= cand_limit:
                        break
                    si = int(s_idx)
                    As = float(a[si])
                    if As <= 1e-15:
                        continue
                    for t_idx in dst_idx:
                        if evaluated >= cand_limit:
                            break
                        ti = int(t_idx)
                        if ti == si:
                            continue
                        for frac in fracs_high:
                            if evaluated >= cand_limit:
                                break
                            d1 = -As * frac
                            d2 = As * frac
                            nm = move_candidate_max(c, tmp, buf, two_a, n, d1, d2, si, ti)
                            evaluated += 1
                            if nm < best_new - eps:
                                best_new = nm
                                best_i = si
                                best_j = ti
                                best_d1 = d1

        # 3) Large-amplitude -> small-amplitude transfer.
        if evaluated < cand_limit:
            top_count = 2 if evaluated + 12 < cand_limit else 1
            low_count = 2 if evaluated + 12 < cand_limit else 1
            topa = np.argpartition(a, -min(top_count, n))[-min(top_count, n):]
            topa = topa[np.argsort(a[topa])[::-1]]
            lowa = np.argpartition(a, min(low_count, n))[:min(low_count, n)]
            lowa = lowa[np.argsort(a[lowa])]

            for s_idx in topa:
                if evaluated >= cand_limit:
                    break
                si = int(s_idx)
                As = float(a[si])
                if As <= 1e-15:
                    continue
                for t_idx in lowa:
                    if evaluated >= cand_limit:
                        break
                    ti = int(t_idx)
                    if ti == si:
                        continue
                    for frac in fracs_high:
                        if evaluated >= cand_limit:
                            break
                        d1 = -As * frac
                        d2 = As * frac
                        nm = move_candidate_max(c, tmp, buf, two_a, n, d1, d2, si, ti)
                        evaluated += 1
                        if nm < best_new - eps:
                            best_new = nm
                            best_i = si
                            best_j = ti
                            best_d1 = d1

        # 4) Random pair moves.
        remaining = cand_limit - evaluated
        attempt = 0
        while remaining > 0 and attempt < remaining * 3:
            attempt += 1
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                j = (j + 1) % n
            A = float(a[i])
            B = float(a[j])
            if A <= 1e-15 or B <= 1e-15:
                continue
            bound = A if A < B else B
            sign = 1.0 if rng.random() < 0.5 else -1.0
            r = int(rng.integers(0, len(fracs_rand)))
            frac = float(fracs_rand[r])
            d1 = sign * bound * frac
            d2 = -d1
            nm = move_candidate_max(c, tmp, buf, two_a, n, d1, d2, i, j)
            evaluated += 1
            remaining = cand_limit - evaluated
            if nm < best_new - eps:
                best_new = nm
                best_i = i
                best_j = j
                best_d1 = d1

        if best_i >= 0:
            d1 = best_d1
            d2 = -d1
            i = best_i
            j = best_j

            apply_pair_move(c, tmp, two_a, d1, d2, i, j, n)
            a[i] += d1
            a[j] += d2
            two_a[i] += 2.0 * d1
            two_a[j] += 2.0 * d2

            current_max = best_new
            current_score = 2.0 * n * current_max
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()

            no_improve = 0
            accept_count += 1
            recompute_count += 1

            if recompute_count >= 120:
                sa = float(np.sum(a))
                if abs(sa - 1.0) > 1e-8:
                    a = a / sa
                c = fft_convolve(a)
                two_a = 2.0 * a
                current_max = float(np.max(c))
                current_score = 2.0 * n * current_max
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
                recompute_count = 0
                accept_count = 0
        else:
            no_improve += 1
            if no_improve > 25 and no_improve % 15 == 0:
                # Refresh convolution to protect against accumulated drift.
                sa = float(np.sum(a))
                if abs(sa - 1.0) > 1e-8:
                    a = a / sa
                c = fft_convolve(a)
                two_a = 2.0 * a
                current_max = float(np.max(c))
                current_score = 2.0 * n * current_max
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
                    no_improve = 0

    # Exact final score for the returned local best.
    sb = float(np.sum(best_a))
    if sb <= 1e-12:
        return best_a, np.inf
    cb = fft_convolve(best_a / sb)
    final_score = score_c(cb, len(best_a), 1.0)
    return best_a / sb, final_score


def local_optimize(a0, time_budget, rng):
    """Run a short exposure phase followed by exact pair local search."""
    t_start = time.monotonic()
    a = normalize_nonneg(a0)
    n = len(a)
    c = fft_convolve(a)
    s = float(np.sum(a))
    current_score = score_c(c, n, s)
    best_a = a.copy()
    best_score = current_score

    # Exposure phase budget.
    if n > 1024:
        expo = min(25.0, 0.08 * float(time_budget))
    elif n > 512:
        expo = min(45.0, 0.12 * float(time_budget))
    else:
        expo = min(70.0, 0.18 * float(time_budget))

    if expo > 3.0:
        a2, sc2 = exposure_optimize(a, c, expo, rng)
        if np.isfinite(sc2):
            a2 = normalize_nonneg(a2)
            c2 = fft_convolve(a2)
            sc2 = score_c(c2, len(a2), 1.0)
            a = a2
            c = c2
        else:
            a = a2
            c = fft_convolve(a)
            sc2 = score_c(c, n, 1.0)
        if sc2 < best_score:
            best_score = sc2
            best_a = a.copy()

    pair_budget = (t_start + float(time_budget)) - time.monotonic() - 2.0
    if pair_budget > 4.0:
        a3, sc3 = pair_local_search(a, c, pair_budget, rng)
        if np.isfinite(sc3):
            a3 = normalize_nonneg(a3)
            c3 = fft_convolve(a3)
            sc3 = score_c(c3, len(a3), 1.0)
        if sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()

    # Final exactness check.
    sb = float(np.sum(best_a))
    if sb > 1e-12:
        cb = fft_convolve(best_a / sb)
        final_score = score_c(cb, len(best_a), 1.0)
        return best_a / sb, final_score
    return best_a, best_score


def propose_candidate(seed=0, budget_s=2400):
    """Main entry point: search for the best non-negative step-function heights."""
    global_start = time.monotonic()
    hard_deadline = global_start + max(60.0, float(budget_s) - 45.0)
    rng = np.random.default_rng(seed)

    # Load provided starting sequence if available.
    base_a = None
    if "height_sequence_1" in globals():
        supplied = height_sequence_1
        if supplied is not None:
            base_a = normalize_nonneg(supplied)

    if base_a is None:
        n0 = 256
        base_a = np.ones(n0, dtype=np.float64) / float(n0)

    best_a = base_a.copy()
    best_score = score_a(best_a)
    best_n = len(best_a)
    print("Initial provided sequence: n=%d score=%.8f" % (best_n, best_score), flush=True)

    # Lengths to explore.  Prefer the provided length first, then common lengths.
    preferred = [256, 512, 1024, 2048]
    lengths = [best_n]
    for n in preferred:
        if n not in lengths:
            lengths.append(n)

    stage_caps = (450.0, 380.0, 330.0, 270.0, 220.0)

    for idx, n in enumerate(lengths):
        now = time.monotonic()
        time_left = hard_deadline - now
        if time_left < 180.0:
            break

        # Reserve time for remaining stages and a final polish.
        remaining_after = len(lengths) - idx - 1
        reserve = 130.0 + 70.0 * remaining_after
        max_stage = time_left - reserve
        if max_stage < 60.0:
            break

        cap = float(stage_caps[min(idx, len(stage_caps) - 1)])
        t_stage = min(cap, max_stage)

        if idx == 0 and n == best_n:
            seed_a = best_a
            # For the first stage, reuse current best directly.
        else:
            seeds = make_candidate_seeds(n, base_a, best_a, rng)
            if seeds:
                seed_a = seeds[0]
            else:
                seed_a = normalize_nonneg(best_a)

        print(
            "Stage %d: n=%d, budget=%.1fs, current best=%.8f"
            % (idx + 1, n, t_stage, best_score),
            flush=True
        )

        opt_a, opt_score = local_optimize(seed_a, t_stage, rng)
        if np.isfinite(opt_score) and opt_score < best_score:
            opt_a = normalize_nonneg(opt_a)
            best_score = opt_score
            best_a = opt_a.copy()
            best_n = len(best_a)
            print("  New best: n=%d score=%.8f" % (best_n, best_score), flush=True)
        else:
            print("  No improvement at n=%d" % n, flush=True)

        if best_score < TARGET_UPPER - 1e-7:
            print("Reached below target; use remaining time for final polish.")
            break

    # Final long polish on the global best.
    now = time.monotonic()
    final_time = hard_deadline - now - 8.0
    if final_time > 15.0:
        print("Final polish: n=%d, budget=%.1fs, current best=%.8f"
              % (best_n, final_time, best_score), flush=True)
        opt_a, opt_score = local_optimize(best_a, final_time, rng)
        if np.isfinite(opt_score) and opt_score < best_score:
            opt_a = normalize_nonneg(opt_a)
            best_score = opt_score
            best_a = opt_a.copy()
            best_n = len(best_a)
            print("  After final polish: n=%d score=%.8f" % (best_n, best_score), flush=True)

    # Final safety normalization/clipping.
    best_a = normalize_nonneg(best_a)
    final_list = []
    for x in best_a:
        y = float(x)
        if not np.isfinite(y) or y < 0.0:
            y = 0.0
        elif y > 1000.0:
            y = 1000.0
        final_list.append(y)

    # Verify with the actual evaluator when available.
    try:
        final_eval = evaluate_sequence(final_list)
    except Exception:
        final_eval = score_a(np.asarray(final_list, dtype=np.float64))

    print("FINAL: n=%d local_score=%.8f evaluator_score=%s"
          % (len(final_list), local_score_of_final if ('local_score_of_final' in globals()) else score_a(np.asarray(final_list, dtype=np.float64)), final_eval),
          flush=True)

    return final_list
```
