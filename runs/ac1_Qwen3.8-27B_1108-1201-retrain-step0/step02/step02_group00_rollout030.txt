def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences using
continuation FISTA with log-sum-exp smoothing, adaptive top-k subgradient,
and local polishing. Tries multiple sequence lengths and initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_to(a, S):
    """Clip to [0, 1000] and rescale to sum = S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def obj_from_maxb(n, max_b, sum_a):
    """Compute objective from components."""
    if sum_a < 1e-15:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

                        beta_list=None, iters_per_beta=200):

    Continuation FISTA with log-sum-exp.
    beta_list: list of beta values to sweep through (geometric).
    Returns (best_a, best_maxb).

    if beta_list is None:
        beta_list = np.geomspace(1.0, 300.0, 20)
    
    a = normalize_to(a_init, S).copy()
    best_maxb = np.inf
    best_a = a.copy()

    time_per_beta = time_budget / len(beta_list)
    
    for beta in beta_list:
        y = a.copy()
        t_mom = 1.0
        step = 1.0 / (beta * 2.0)
        start_beta = time.monotonic()
        
        it = 0
        while time.monotonic() - start_beta < time_per_beta:
            it += 1
            c = fft_convolve_self(y)
            cur_max = c.max()

                best_maxb = cur_max
                best_a = y.copy()

            exp_bc = np.exp(beta * c_shift)
            total = exp_bc.sum()
            if total < 1e-300:
                break
            w = exp_bc / total
            
            # Cross-correlation for gradient
            fsize = 1 << (len(w) + n - 2).bit_length()
            W = fft(w, fsize)
            A_y = fft(y, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A_y)))[:n]
            
            gnorm = np.linalg.norm(grad)

            a = x_new
            t_mom = t_new

def fista_toppk(a_init, n, S, time_budget, rng, top_k=10):

    a = normalize_to(a_init, S).copy()

    step = 1.0 / (2.0 * np.sqrt(n))
    best_maxb = np.inf
    best_a = a.copy()
    start = time.monotonic()
    
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_maxb:
            best_maxb = cur_max
            best_a = y.copy()

def local_polish(a, n, S, time_budget, rng):
    """
    Local polishing: iteratively reduce coordinates that contribute most
    to the maximum convolution value, then renormalize.
    """
    a = normalize_to(a, S).copy()
    best_maxb = fft_convolve_self(a).max()
    best_a = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_maxb:
            best_maxb = cur_max
            best_a = a.copy()
        
        max_lag = int(np.argmax(c))
        
        if max_lag == 0:
            # Reduce the largest elements (they contribute most to c[0] = sum(a^2))
            top_k = 30
            top_coords = np.argsort(a)[::-1][:top_k]
            for i in top_coords:
                a[i] *= (1.0 - 0.01 * rng.random())
        else:
            # Reduce pairs (a[i], a[i+max_lag]) with largest product
            valid = np.arange(n - max_lag)
            products = a[valid] * a[valid + max_lag]
            top_k = 30
            top_pairs = valid[np.argsort(products)[::-1][:top_k]]

                j = i + max_lag
                # Reduce the larger one
                if a[j] > a[i]:
                    a[j] *= (1.0 - 0.01 * rng.random())
                else:
                    a[i] *= (1.0 - 0.01 * rng.random())
        
        a = np.maximum(a, 0.0)
        a = normalize_to(a, S)
    
    return best_a, best_maxb

Let me also add a "dual" perspective: at the optimum, the active constraints (lags where c[k] = max_b) should have specific properties. We can use this to guide the optimization.

Actually, let me reconsider the problem from scratch. We want to minimize:
f(a) = max_k (a*a)[k] / (sum(a))^2 * 2n

Since this is scale-invariant, let's fix sum(a) = 1 and minimize 2n * max_k (a*a)[k].

Or fix sum(a) = sqrt(2n) and minimize max_k (a*a)[k].

The FISTA approach minimizes the log-sum-exp approximation of max_k (a*a)[k] subject to sum(a) = sqrt(2n) and a >= 0.

The main issue is: how good is the approximation at a given beta, and how many iterations does FISTA need to converge?

For the log-sum-exp with parameter beta:
- The approximation error is at most (1/beta) * log(2n-1) ≈ (log(2n-1))/beta
- For n=8192, log(16383) ≈ 9.7, so error ≈ 9.7/beta
- For beta=100, error ≈ 0.097
- For beta=500, error ≈ 0.019

But this is the error in the surrogate, not in the actual max. The actual max is what we care about.

The FISTA convergence rate is O(f_0 - f*) / k^2 for the smooth part. The "smooth part" has Lipschitz constant L ≈ beta * 2 * max(a_i)^2 (roughly). So the step size is 1/L, and convergence depends on L.

For a sequence normalized to sum = sqrt(2n) ≈ 128 (for n=8192), if elements are roughly uniform, a_i ≈ 128/8192 ≈ 0.0156. Then max(a_i)^2 ≈ 0.00024, and L ≈ beta * 2 * 0.00024 = 0.00048 * beta. For beta=100, L ≈ 0.048, step ≈ 20. That's a reasonable step size.

But as the sequence becomes more peaked (which it will during optimization), max(a_i) increases, making L larger and steps smaller. This is the fundamental difficulty.

**Alternative: Use the subgradient of the actual max (not the smooth approximation).**

The subgradient of max_k (a*a)[k] with respect to a is: 2 * (a convolved with the indicator of the argmax lag). If there are multiple lags achieving the max, the subgradient is a convex combination.

FISTA (or accelerated proximal gradient) can work with subgradients too (it becomes the accelerated subgradient method with rate O(1/k) instead of O(1/k^2), but it's exact).

Actually, for the non-smooth case, the accelerated method (Nesterov's) still gives O(1/k) rate, same as plain subgradient with the right step schedule. The O(1/k^2) rate requires smoothness.

So the trade-off is:
- Smooth approximation (log-sum-exp): O(1/k^2) rate but with approximation error
- Exact subgradient: O(1/k) rate but exact

For the final polish, the exact subgradient might be better since we're close to the optimum and the approximation error matters more.

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Uses continuation FISTA with log-sum-exp smoothing for the main optimization,
exact top-1 subgradient for final polishing, and local coordinate search.
Tries multiple initializations and sequence lengths for diversity.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_to(a, S):
    """Clip to [0,1000] and rescale to sum = S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def compute_obj(a):
    """Compute the objective 2*n*max_b/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def grad_lse(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a via FFT cross-correlation."""
    n = len(a)
    c_shift = c - c.max()
    exp_bc = np.exp(beta * c_shift)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def grad_toppk(a, c, k):
    """Gradient using top-k active lags via FFT cross-correlation."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    w_full[top_idx] = weights
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fista_smooth(a_init, n, S, time_budget, rng, beta_start=1.0, beta_end=200.0, n_stages=15):
    """
    Continuation FISTA with log-sum-exp.
    Geometric beta schedule, FISTA with momentum at each stage.
    Returns (best_a, best_maxb).
    """
    a = normalize_to(a_init, S).copy()
    best_maxb = np.inf
    best_a = a.copy()
    
    # Weight stages more heavily at higher beta
    betas = np.geomspace(beta_start, beta_end, n_stages)
    # Time weights: give more time to later (higher beta) stages
    w = np.linspace(0.5, 2.0, n_stages)
    w = w / w.sum()
    
    for si, beta in enumerate(betas):
        t_alloc = time_budget * w[si]
        if t_alloc < 0.5:
            t_alloc = 0.5
        y = a.copy()
        t_mom = 1.0
        # Initial step: 1/(beta * 2 * max_possible_grad_factor)
        # Conservative estimate: L ~ beta * 2 * ||a||_inf * ||a||_2
        a_inf = np.abs(y).max()
        a_2 = np.linalg.norm(y)
        L_est = beta * 2.0 * a_inf * a_2
        if L_est < 1e-10:
            L_est = beta * 2.0
        step = 1.0 / L_est
        
        start_s = time.monotonic()
        it = 0
        while time.monotonic() - start_s < t_alloc:
            it += 1
            c = fft_convolve_self(y)
            cur = c.max()
            if cur < best_maxb:
                best_maxb = cur
                best_a = y.copy()
            
            grad = grad_lse(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-16:
                break
            
            # Backtracking: check if step is too large
            # (approximate: if the log-sum-exp value increased, halve step)
            lse_val = c.max() + np.log(np.exp(beta * (c - c.max())).sum()) / beta
            
            x_new = project_simplex(y - step * grad, S)
            
            # Simple backtracking (every few iterations)
            if it % 10 == 0:
                c_x = fft_convolve_self(x_new)
                lse_x = c_x.max() + np.log(np.exp(beta * (c_x - c_x.max())).sum()) / beta
                if lse_x > lse_val * 1.001 + 0.01:
                    step *= 0.5
                    x_new = project_simplex(y - step * grad, S)
            
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new
            
            # Slowly decay step
            step *= (1.0 - 0.0002)
            step = max(step, 1e-10)
        
        # Renormalize
        a = normalize_to(a, S)
    
    c = fft_convolve_self(best_a)
    if c.max() < best_maxb:
        best_maxb = c.max()
    
    return best_a, best_maxb


def fista_toppk_polish(a_init, n, S, time_budget, rng, top_k_start=5, top_k_end=100):
    """
    FISTA with growing top-k subgradient for polishing.
    Returns (best_a, best_maxb).
    """
    a = normalize_to(a_init, S).copy()
    y = a.copy()
    t_mom = 1.0
    step = 2.0 / (np.sqrt(n) * np.linalg.norm(a, ord=np.inf) + 1e-10)
    best_maxb = np.inf
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_maxb:
            best_maxb = cur
            best_a = y.copy()
        
        # Grow top_k linearly
        frac = min(1.0, it / max(1, it + 1000))
        k = int(top_k_start + frac * (top_k_end - top_k_start))
        k = max(1, min(k, 2 * n - 1))
        
        grad = grad_toppk(y, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Decay step slowly
        step = 2.0 / (np.sqrt(n) * (np.linalg.norm(a, ord=np.inf) + 1e-10) * (1.0 + 0.0003 * it))
        
        # Noise injection for escape
        if it % 3000 == 0 and rng.random() < 0.2:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = normalize_to(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
    
    c = fft_convolve_self(best_a)
    if c.max() < best_maxb:
        best_maxb = c.max()
    
    return best_a, best_maxb


def local_coordinate_polish(a_init, n, S, time_budget, rng, reduce_rate=0.02, top_pairs=50):
    """
    Local polish: iteratively reduce coordinates contributing most to max_b.
    Returns (best_a, best_maxb).
    """
    a = normalize_to(a_init, S).copy()
    best_maxb = np.inf
    best_a = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_maxb:
            best_maxb = cur
            best_a = a.copy()
        
        max_lag = int(np.argmax(c))
        
        if max_lag == 0:
            top_coords = np.argsort(a)[::-1][:top_pairs]
            for i in top_coords:
                a[int(i)] *= (1.0 - reduce_rate * (0.5 + 0.5 * rng.random()))
        else:
            valid = np.arange(n - max_lag)
            products = a[valid] * a[valid + max_lag]
            top_pairs_idx = valid[np.argsort(products)[::-1][:top_pairs]]
            for i in top_pairs_idx:
                j = int(i) + max_lag
                if j < n:
                    if a[j] > a[int(i)]:
                        a[j] *= (1.0 - reduce_rate * (0.5 + 0.5 * rng.random()))
                    else:
                        a[int(i)] *= (1.0 - reduce_rate * (0.5 + 0.5 * rng.random()))
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            break
        a = a * (S / s)
    
    return best_a, best_maxb
