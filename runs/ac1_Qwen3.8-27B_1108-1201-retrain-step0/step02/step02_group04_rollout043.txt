    m = len(b)

    while size < n + m - 1:

from scipy.optimize import linprog

def lp_step(f, M, n):
    """
    Solve: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0.
    Returns the optimal b.
    """
    # c = -1 (we want to maximize sum, linprog minimizes)
    c = -np.ones(n)
    
    # A_ub: constraints (f*b)[k] <= M
    # (f*b)[k] = sum_{i} f[i] * b[k-i] where 0<=i<n, 0<=k-i<n
    # = sum_{i: 0<=k-i<n} f[i] * b[k-i]
    # Let j = k-i, so i = k-j
    # = sum_{j: 0<=k-j<n} f[k-j] * b[j]
    # The coefficient of b[j] in constraint k is f[k-j] (if 0<=k-j<n)
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = f[i]
    
    b_ub = np.full(2*n-1, M)
    
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return None
```

    M = c.max()
    S = f.sum()
    
    # Solve LP: max sum(b) s.t. (f*b)[k] <= M, b >= 0
    c_obj = -np.ones(n)
    A = build_A(f, n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0:
        return f, True
    
    g = res.x.copy()
    g_sum = g.sum()
    if g_sum < 1e-10:
        return f, True
    
    # Rescale to match sum
    g = g * S / g_sum
    
    # Blend
    f_new = (1-t)*f + t*g
    f_new = np.clip(f_new, 0, 1000)
    
    return f_new, False
```
