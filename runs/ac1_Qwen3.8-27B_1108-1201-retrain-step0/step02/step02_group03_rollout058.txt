import numpy as np
import time

def conv_aa(a):
    """Compute a * a (linear convolution) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = conv_aa(a)
    return 2.0 * n * c.max() / (s * s)

def soft_max_grad(a, tau):
    """
    Compute soft-max of conv(a,a) and its gradient.
    Returns (value, grad) where grad is d/d a of soft-max.
    """
    n = len(a)
    c = conv_aa(a)
    # Soft max
    c_max = c.max()
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    w = e / e.sum()
    value = c_max + tau * np.log(e.sum())
    
    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j] for i=0,...,n-1
    # This is the valid cross-correlation of w (length 2n-1) and a (length n)
    # = conv(w, a[::-1])[n-1 : 2n-1]
    size2 = 1
    while size2 < (2*n-1) + n - 1:
        size2 <<= 1
    W = np.fft.rfft(w, size2)
    A = np.fft.rfft(a[::-1], size2)
    conv_w_arev = np.fft.irfft(W * A, size2)[:3*n-2]
    grad = 2.0 * conv_w_arev[n-1:2*n-1]
    
    return value, grad, c

def gs_init(n, rng, n_iter=200, n_restarts=5):
    """Gerchberg-Saxton initialization with multiple restarts."""
    best = None
    best_sc = np.inf
    for r in range(n_restarts):
        a = np.clip(rng.standard_normal(n), 0.1, None)
        a /= a.sum()
        for it in range(n_iter):
            A = np.fft.fft(a, 2*n-1)
            # Target: flat magnitude spectrum
            mag = np.abs(A)
            mag[mag < 1e-15] = 1e-15
            phase = A / mag
            a_new = np.fft.ifft(phase).real[:n]
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                a_new = a  # keep old
            else:
                a_new /= s
            # Check improvement
            if it % 20 == 0:
                sc = score(a_new)
                if sc < best_sc:
                    best_sc = sc
                    best = a_new.copy()
            a = a_new
        sc = score(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
    return best, best_sc

def gradient_descent(a, time_budget, rng, tau_init=None):
    """Adam gradient descent on the soft-max smoothed objective."""
    n = len(a)
    if tau_init is None:
        tau_init = 50.0 / n
    
    # Normalize
    a = a.copy()
    a /= a.sum()  # sum = 1
    
    theta = np.log(a + 1e-15)  # parameterize as a = softmax(theta) to ensure positivity
    # Actually, let's parameterize differently: a_i = exp(theta_i) / sum(exp(theta))
    # This ensures a_i > 0 and sum(a) = 1 automatically.
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.5
    
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    
    tau_end = 0.1 / n
    
    while time.monotonic() - t0 < time_budget:
        progress = min(1.0, (time.monotonic() - t0) / time_budget)
        # Annealing: tau decreases from tau_init to tau_end
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        
        # Compute a from theta
        theta_shifted = theta - theta.max()
        a = np.exp(theta_shifted)
        a /= a.sum()
        
        # Compute gradient of soft-max(conv(a,a)) w.r.t. theta
        value, grad_a, c = soft_max_grad(a, tau)
        
        # d soft-max / d theta_i = sum_j (d soft-max / d a_j) * (d a_j / d theta_i)
        # a_j = exp(theta_j) / sum(exp(theta))
        # d a_j / d theta_i = a_j * (delta_{ji} - a_i)
        # So grad_theta_i = sum_j grad_a_j * a_j * (delta_{ji} - a_i)
        # = grad_a_i * a_i - a_i * sum_j grad_a_j * a_j
        
        inner = np.dot(grad_a, a)
        grad_theta = a * (grad_a - inner)
        
        # Adam update
        t += 1
        m = b1 * m + (1 - b1) * grad_theta
        v = b2 * v + (1 - b2) * grad_theta ** 2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Check score periodically
        if t % 100 == 0:
            s = score(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
            # Adaptive learning rate
            if t % 1000 == 0:
                print(f"    t={t}, score={s:.6f}, best={best_score:.6f}, tau={tau:.2e}")
    
    return best_a, best_score

def local_search(a, time_budget, rng):
    """Simulated annealing local search."""
    n = len(a)
    current = a.copy()
    current_score = score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9997
    
    while time.monotonic() - t0 < time_budget:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.4:
            # Target peaks
            c = conv_aa(current)
            top_k = np.argsort(c)[-10:]
            # Reduce elements that contribute most to peaks
            contrib = np.zeros(n)
            for lag in top_k:
                i_lo = max(0, lag - n + 1)
                i_hi = min(n - 1, lag)
                for i in range(i_lo, i_hi + 1):
                    contrib[i] += current[i] * current[lag - i]
            n_mod = max(2, n // 8)
            top_idx = np.argsort(contrib)[-n_mod:]
            bot_idx = np.argsort(contrib)[:n_mod]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.02, len(bot_idx)))
        elif r < 0.6:
            # Random block perturbation
            bw = max(1, rng.integers(1, max(2, n // 10)))
            bs = rng.integers(0, n - bw)
            candidate[bs:bs+bw] *= np.exp(rng.normal(0, 0.04, bw))
        elif r < 0.8:
            # Global noise
            candidate *= np.exp(rng.normal(0, 0.01, n))
        else:
            # Swap-like perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                swap = 0.1 * (candidate[i] - candidate[j])
                candidate[i] -= swap
                candidate[j] += swap
        
        candidate = np.clip(candidate, 0.0, None)
        if candidate.sum() < 1e-10:
            continue
        candidate /= candidate.sum()
        
        new_score = score(candidate)
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                current = candidate
                current_score = new_score
        
        temp *= cooling
    
    return best, best_score

"""
Optimized step function search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Gerchberg-Saxton initialization with momentum, Adam gradient descent with
soft-max annealing, targeted local search, and coarse-to-fine upscaling.
"""

import numpy as np
import time
import math


def conv_aa(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = conv_aa(a)
    return float(2.0 * n * np.max(c) / (s * s))


def soft_max_and_weights(c, tau):
    """Compute soft-max value and normalized weights on convolution."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    w = e / s
    value = c_max + tau * np.log(s)
    return value, w


def grad_soft_max(a, tau):
    """
    Compute soft-max(conv(a,a)) and its gradient w.r.t. a.
    grad_a[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j] for i=0,...,n-1
    """
    n = len(a)
    c = conv_aa(a)
    value, w = soft_max_and_weights(c, tau)

    # grad_a[i] = 2 * sum_j a[j] * w[i+j]
    # This is the valid cross-correlation of w (len 2n-1) and a (len n)
    # = conv(w, a[::-1])[n-1 : 2n-1]
    size2 = 1
    while size2 < (2 * n - 1) + n - 1:
        size2 <<= 1
    W = np.fft.rfft(w, size2)
    A_rev = np.fft.rfft(a[::-1], size2)
    conv_result = np.fft.irfft(W * A_rev, size2)[:3 * n - 2]
    grad = 2.0 * conv_result[n - 1:2 * n - 1]

    return value, grad, c


def gs_init(n, rng, n_iter=300, n_restarts=8, momentum=0.5):
    """Gerchberg-Saxton with momentum and multiple restarts."""
    best_a = None
    best_score = np.inf

    for r in range(n_restarts):
        # Random non-negative initialization
        a = np.abs(rng.standard_normal(n)) + 0.1
        a /= a.sum()

        for it in range(n_iter):
            # Get current DFT
            size = 1
            while size < 2 * n - 1:
                size <<= 1
            A = np.fft.rfft(a, size)
            mag = np.abs(A)
            mag[mag < 1e-15] = 1e-15
            phase = A / mag

            # Inverse transform
            a_new = np.fft.irfft(phase, size)[:n]
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                a_new = a
            else:
                a_new = a_new / s

            # Momentum blend
            a = momentum * a + (1.0 - momentum) * a_new
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s < 1e-10:
                a = a_new
            else:
                a /= s

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score

def gradient_descent(a_init, time_budget, rng, verbose=True):
    """
    Adam gradient descent on softmax-parameterized sequence.
    Minimizes soft-max(conv(a,a)) with annealed temperature.
    Sequence is normalized to sum=1 automatically.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.clip(a, 1e-10, None)
    a /= a.sum()

    # Parameterize: a_i = exp(theta_i) / sum(exp(theta_j))
    theta = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    step_count = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    base_lr = 1.5

    tau_init = 80.0 / n
    tau_end = 0.05 / n

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    no_improve = 0
    lr_scale = 1.0

    if verbose:
        print(f"    GD start: n={n}, score={best_score:.6f}")

    while time.monotonic() - t0 < time_budget:
        progress = min(1.0, (time.monotonic() - t0) / time_budget)
        # Slow annealing with power law
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        # Compute a from theta
        theta_shifted = theta - np.max(theta)
        a = np.exp(theta_shifted)
        a_sum = a.sum()
        a = a / a_sum

        # Gradient
        value, grad_a, c = grad_soft_max(a, tau)

        # Convert gradient w.r.t. a to gradient w.r.t. theta
        inner = np.dot(grad_a, a)
        grad_theta = a * (grad_a - inner)

        # Adam update
        step_count += 1
        m = b1 * m + (1.0 - b1) * grad_theta
        v = b2 * v + (1.0 - b2) * grad_theta ** 2
        m_hat = m / (1.0 - b1 ** step_count)
        v_hat = v / (1.0 - b2 ** step_count)
        lr = base_lr * lr_scale / (1.0 + 0.1 * step_count ** 0.5)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic score check
        if step_count % 200 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
                if verbose:
                    print(f"      step={step_count}, score={sc:.6f}, tau={tau:.2e}")
            else:
                no_improve += 1
                if no_improve == 20:
                    lr_scale *= 0.7
                    no_improve = 0
                elif no_improve == 50:
                    # Add noise to escape local minimum
                    theta += 0.05 * rng.standard_normal(n)
                    no_improve = 0

    final_sc = compute_score(best_a)
    if verbose:
        print(f"    GD end: best_score={final_sc:.6f}")
    return best_a, final_sc

    temp = 0.005

            continue
        candidate = candidate / s

        new_score = compute_score(candidate)
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if verbose and iter_count % 5000 == 0:
                    print(f"      LS iter={iter_count}, score={best_score:.6f}, temp={temp:.6f}")
        else:
            delta = new_score - current_score
            if delta < temp * 20:
                accept_prob = math.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score

        temp *= cooling

    if verbose:
        print(f"    LS end: best_score={best_score:.6f}")
    return best, best_score


def upsample_seq(a, new_n, rng):
    """Upsample sequence to new length via interpolation."""
    n = len(a)
    if new_n <= n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    a_interp /= a_interp.sum()
    return a_interp


def chirp_init(n, rng, alpha=0.5):
    """Quadratic phase (chirp) initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (2.0 * n))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a /= a.sum()
    return a


def multi_perturb_init(n, rng):
    """Multi-scale perturbation initialization."""
    a = np.ones(n) / n
    for scale in [n // 2, n // 4, n // 8, n // 16]:
        if scale < 2:
            continue
        freq = rng.integers(1, max(2, n // scale))
        amp = 0.2 / (1.0 + math.log(scale))
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search for optimal non-negative step function.
    Returns the list of heights that minimizes the evaluation score.
    """
    t_global = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_global + total_budget - 30.0  # reserve 30s for safety
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided starting sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start) > 0 and start.sum() > 0.01:
            sc = compute_score(start)
            print(f"Provided sequence: n={len(start)}, score={sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = start.copy()

    # Phase 1: Initialization
    print("\n=== Phase 1: Initialization ===")
    candidates = []
    lengths = [256, 384, 512, 768, 1024]

    for n in lengths:
        if time.monotonic() > deadline - 120:
            break
        t_phase1_start = time.monotonic()

        # G-S with momentum
        a_gs, sc_gs = gs_init(n, rng, n_iter=200, n_restarts=5, momentum=0.4)
        print(f"  n={n} G-S(mom): score={sc_gs:.6f}")
        candidates.append((sc_gs, n, a_gs, "gs_mom"))

        # G-S flat target
        if time.monotonic() < deadline - 60:
            a_gs2, sc_gs2 = gs_init_flat_target(n, rng, n_iter=150, n_restarts=3)
            print(f"  n={n} G-S(target): score={sc_gs2:.6f}")
            candidates.append((sc_gs2, n, a_gs2, "gs_target"))

        # Chirp init
        a_chirp = chirp_init(n, rng, alpha=0.3 + 0.4 * rng.random())
        sc_chirp = compute_score(a_chirp)
        print(f"  n={n} Chirp: score={sc_chirp:.6f}")
        candidates.append((sc_chirp, n, a_chirp, "chirp"))

        # Multi-perturb
        a_mp = multi_perturb_init(n, rng)
        sc_mp = compute_score(a_mp)
        print(f"  n={n} MultiPerturb: score={sc_mp:.6f}")
        candidates.append((sc_mp, n, a_mp, "multiperturb"))

    candidates.sort(key=lambda x: x[0])
    print(f"  Top 5 candidates:")
    for i, (sc, n, a, name) in enumerate(candidates[:5]):
        print(f"    [{i}] n={n}, score={sc:.6f}, type={name}")

    # Phase 2: Gradient descent on top candidates
    print("\n=== Phase 2: Gradient Descent ===")
    n_cand = min(6, len(candidates))
    time_for_gd = (deadline - time.monotonic()) * 0.55
    time_per_gd = time_for_gd / n_cand

    optimized = []
    for i in range(n_cand):
        if time.monotonic() > deadline - 200:
            break
        sc0, n, a, name = candidates[i]

        # Allocate more time to shorter sequences (faster iterations)
        t_alloc = time_per_gd * min(1.5, 384.0 / n)
        t_alloc = max(10.0, min(t_alloc, deadline - time.monotonic() - 100))

        print(f"  [{i+1}/{n_cand}] n={n}, init={name}, score0={sc0:.6f}, time={t_alloc:.0f}s")
        a_opt, sc_opt = gradient_descent(a, t_alloc, rng, verbose=True)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")
        optimized.append((sc_opt, n, a_opt))

    # Phase 3: Local search on best candidate
    print("\n=== Phase 3: Local Search ===")
    if best_seq is not None:
        ls_time = min(300.0, max(30.0, (deadline - time.monotonic()) * 0.3))
        print(f"  Polishing n={len(best_seq)}, score={best_score:.6f}, time={ls_time:.0f}s")
        polished, sc_polished = local_search(best_seq, ls_time, rng)
        if sc_polished < best_score:
            best_score = sc_polished
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
        else:
            print(f"  No improvement from polish: {sc_polished:.6f}")

    # Phase 4: Upscale and optimize
    print("\n=== Phase 4: Upscaling ===")
    if best_seq is not None:
        current_n = len(best_seq)
        target_ns = [x for x in [512, 768, 1024, 1536, 2048] if x > current_n]
        for target_n in target_ns[:2]:
            if time.monotonic() > deadline - 80:
                break
            a_up = upsample_seq(best_seq, target_n, rng)
            t_alloc = min(120.0, max(15.0, (deadline - time.monotonic()) * 0.25))
            print(f"  Upsample to n={target_n}, time={t_alloc:.0f}s")
            a_opt, sc_opt = gradient_descent(a_up, t_alloc, rng)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  *** New best after upsample: {best_score:.6f} (n={target_n}) ***")

                # Quick polish
                if deadline - time.monotonic() > 40:
                    p, sc_p = local_search(best_seq, min(40.0, deadline - time.monotonic() - 30), rng)
                    if sc_p < best_score:
                        best_score = sc_p
                        best_seq = p.copy()
                        print(f"  After polish: {best_score:.6f}")
            else:
                print(f"  Upsample n={target_n}: score={sc_opt:.6f} (no improvement)")

    # Phase 5: Additional restarts if time remains
    remaining = deadline - time.monotonic()
    if remaining > 60 and best_seq is not None:
        print(f"\n=== Phase 5: Additional restarts (time={remaining:.0f}s) ===")
        n_best = len(best_seq)
        n_restarts = min(4, int(remaining // 50))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            # Perturb best and re-optimize
            a_new = best_seq * np.exp(rng.normal(0, 0.15, size=n_best))
            a_new = np.clip(a_new, 0.001, None)
            a_new /= a_new.sum()
            t_alloc = min(60.0, (deadline - time.monotonic() - 20) / max(1, n_restarts - r))
            a_opt, sc_opt = gradient_descent(a_new, t_alloc, rng, verbose=False)
            print(f"  Restart {r+1}: score={sc_opt:.6f}")
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  *** New best: {best_score:.6f} ***")
                # Quick local search
                if deadline - time.monotonic() > 30:
                    p, sc_p = local_search(best_seq, min(30.0, deadline - time.monotonic() - 20), rng)
                    if sc_p < best_score:
                        best_score = sc_p
                        best_seq = p.copy()
                        print(f"  After polish: {best_score:.6f}")

    # Final verification
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")

        # If evaluate gives different score (due to clamping), try to fix
        if final_score != compute_score(best_seq) and final_score < best_score:
            print(f"  (evaluate gives better score due to clamping)")
        return final_list

    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
