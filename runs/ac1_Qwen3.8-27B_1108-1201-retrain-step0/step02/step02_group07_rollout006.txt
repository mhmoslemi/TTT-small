import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_value(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (a.sum()**2)

def lp_step(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0"""
    n = len(a)
    # Constraints: for each k in 0..2n-2, sum_i g_i * a[k-i] <= M
    # where a[k-i] = 0 if k-i not in [0, n-1]
    # Variables: g_0, ..., g_{n-1}
    # Maximize: sum g_i => minimize: -sum g_i
    
    # Build constraint matrix A_ub (2n-1 x n)
    # A_ub[k, i] = a[k-i] if 0 <= k-i <= n-1 else 0
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = a[j]
    
    b_ub = np.full(2*n-1, M)
    c_obj = -np.ones(n)  # minimize -sum(g)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return -res.x, res.fun
    return None, None

Uses LP-based iterative refinement, informed annealing, and multi-length search.

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_norm(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def build_lp_matrix(a):
    """Build the constraint matrix for the LP step.
    A[k, i] = a[k-i] if 0 <= k-i < n else 0
    """
    n = len(a)
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = a
    return A

def lp_step(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns the optimal g vector and the optimal value.
    """
    n = len(a)
    A = build_lp_matrix(a)
    b = np.full(2*n-1, M)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x, -res.fun
    return a.copy(), 0.0  # fallback

def lp_optimize(a0, time_limit, t_start=0.3, t_min=0.01, max_iters=200):
    """LP-based iterative optimization."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    t = t_start
    
    t0 = time.monotonic()
    iter_num = 0
    while time.monotonic() - t0 < time_limit and iter_num < max_iters:
        g, _ = lp_step(a, M)
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-12:
            break
        g /= gs
        
        # Damped update
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        
        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()
        obj_new = 2.0 * n * M_new
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            M = M_new
            # Slowly decrease step size
            t = max(t_min, t * 0.95)
        else:
            # Reduce step size
            t = max(t_min, t * 0.7)
        
        iter_num += 1
    
    return best_a, best_obj

def anneal_informed(a0, time_limit, rng):
    """Informed simulated annealing."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.01 * (1.0 - frac) ** 2 + 0.0003
        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.8
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        target_j = int(rng.integers(n))
        delta = a[top_j] * rng.uniform(0.01, 0.05)
        if delta < 1e-9:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
    return best_a, best_obj

def mirror_descent(a0, time_limit, t_init=50.0, eta_init=0.3):
    """Mirror descent on the simplex."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        # Softmax-weighted subgradient
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        sg = np.zeros(n)
        thresh = w.max() * 0.002
        active = np.where(w > thresh)[0]
        for ki in active:
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                sg[js:je+1] += w[k] * a[k-je:k-js+1][::-1]
        # Center the subgradient
        sg_c = sg - sg.mean()
        # Multiplicative update
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= 50:
            patience = 0
            eta *= 0.65
            t *= 1.3
            if eta < 0.004:
                break
    return best_a, best_obj

Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based iterative refinement (SOTA approach), mirror descent, and informed annealing.
Explores multiple sequence lengths with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_norm(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_lp_matrix(a):
    """Build constraint matrix A where A[k,i] = a[k-i] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return a.copy()


def lp_optimize(a0, time_limit, t_start=0.4, t_min=0.05, n_restart=3):
    """LP-based iterative optimization with restarts."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    t = t_start
    patience = 0
    max_pat = 30
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        if patience >= max_pat:
            # Restart with perturbation
            if n_restart <= 0:
                break
            n_restart -= 1
            patience = 0
            t = max(t_start * 0.5, t)
            noise = np.random.default_rng(int(time.monotonic() * 1e6)).uniform(0, 0.05 * a.max(), n)
            a = a + noise
            a = np.maximum(a, 0)
            a /= a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            continue

        g = lp_solve(a, M)
        gs = g.sum()
        if gs < 1e-12:
            break
        g /= gs

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        ns = a_new.sum()
        if ns < 1e-12:
            break
        a_new /= ns

        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()
        obj_new = 2.0 * n * M_new

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            M = M_new
            patience = 0
            t = max(t_min, t * 0.97)
        else:
            # Try with larger step
            a_new2 = (1.0 - min(1.0, t * 2)) * a + min(1.0, t * 2) * g
            a_new2 = np.maximum(a_new2, 0.0)
            if a_new2.sum() > 1e-12:
                a_new2 /= a_new2.sum()
                conv2 = fast_convolve(a_new2)
                obj2 = 2.0 * n * conv2.max()
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a_new2.copy()
                    a = a_new2
                    conv = conv2
                    M = conv2.max()
                    patience = 0
                else:
                    patience += 1
                    t = max(t_min, t * 0.8)
            else:
                patience += 1
                t = max(t_min, t * 0.8)

    return best_a, best_obj


def mirror_descent(a0, time_limit, t_init=50.0, eta_init=0.3):
    """Multiplicative mirror descent on the simplex."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        sg = np.zeros(n)
        thresh = w.max() * 0.003
        active = np.where(w > thresh)[0]
        for ki in active:
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= 40:
            patience = 0
            eta *= 0.6
            t *= 1.4
            if eta < 0.003:
                break
    return best_a, best_obj


def anneal_informed(a0, time_limit, rng):
    """Informed simulated annealing targeting peak reduction."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.008 * (1.0 - frac) ** 2 + 0.0002
        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.78
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        # Choose target with low peak contribution
        target_j = int(rng.integers(n))
        for _ in range(3):
            candidate = int(rng.integers(n))
            contrib = 0.0
            for pk in peak_positions[:6]:
                d = pk - candidate
                if 0 <= d <= n - 1:
                    contrib += a[d]
                d2 = candidate - pk + 2 * n - 2
                if 0 <= d2 <= n - 1:
                    contrib += a[d2]
            if contrib < a.mean() * 1.5:
                target_j = candidate
                break
        if target_j == top_j:
            continue
        delta = a[top_j] * rng.uniform(0.005, 0.04)
        if delta < 1e-9:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
    return best_a, best_obj

def gen_inits(n, rng):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n
    for sf in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%.2f' % sf] = g
    for freq in [1, 2, 3]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        s /= s.sum()
        inits['cos_%d' % freq] = s
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    rc[rc < 0] = 0
    rc /= rc.sum()
    inits['raised_cos'] = rc
    # Hann window
    hw = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n - 1)))
    hw /= hw.sum()
    inits['hann'] = hw
    for alpha in [0.2, 0.5, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['dir_%.1f' % alpha] = r
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    # Chirp
    chirp = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * x / (2.0 * n)))
    chirp = np.maximum(chirp, 0.0)
    if chirp.sum() > 1e-10:
        chirp /= chirp.sum()
        inits['chirp'] = chirp
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Uses LP-based iterative refinement combined with
    mirror descent and informed annealing to minimize the auto-convolution peak.
    Explores multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)

    parent = get_parent()
    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d" % (score, len(a)))

    # Collect initial candidates
    candidates = []
    if parent is not None:
        p = np.maximum(parent, 0.0).copy()
        p /= p.sum()
        candidates.append(('parent', len(p), p.copy()))

    # Add candidates for various lengths with best initializations
    for n in [192, 256, 320, 384, 512]:
        inits = gen_inits(n, rng)
        best_init = None
        best_obj = float('inf')
        for name, a0 in inits.items():
            o = obj_norm(a0)
            if o < best_obj:
                best_obj = o
                best_init = a0.copy()
        if best_init is not None:
            candidates.append(('best_init_%d' % n, n, best_init))

    print("Total candidates: %d" % len(candidates))

    # Phase 1: LP optimization on all candidates
    print("Phase 1: LP-based optimization...")
    phase1_budget = min(1400.0, (deadline - time.monotonic()) * 0.45)
    n_cands = len(candidates)
    time_per = phase1_budget / max(1, n_cands)

    for idx, (name, n, a0) in enumerate(candidates):
        if time.monotonic() > deadline - 1400:
            break
        t_alloc = min(time_per * 1.5, deadline - time.monotonic() - 1400)
        if t_alloc < 8:
            break
        print("  [%d/%d] %s, n=%d, budget=%.0fs" % (idx + 1, n_cands, name, n, t_alloc))

        a_opt, obj = lp_optimize(a0, time_limit=t_alloc, t_start=0.4)
        record(a_opt)

        # Mirror descent polish
        if time.monotonic() < deadline - 1400:
            rem = min(min(40.0, t_alloc * 0.3), deadline - time.monotonic() - 1400)
            if rem > 5:
                a_opt2, _ = mirror_descent(a_opt, time_limit=rem, t_init=80, eta_init=0.15)
                record(a_opt2)

    print("  Phase 1 complete. Best: %.6f" % best_score)

    # Phase 2: LP refinement on best
    if best_sequence is not None and time.monotonic() < deadline - 600:
        print("Phase 2: Deep LP refinement on best...")
        phase2_budget = min(600.0, deadline - time.monotonic() - 600)
        a_best = best_sequence.copy()
        a_best /= a_best.sum()
        a_opt, _ = lp_optimize(a_best, time_limit=phase2_budget, t_start=0.2, t_min=0.02)
        record(a_opt)
        print("  Phase 2 complete. Best: %.6f" % best_score)

    # Phase 3: Annealing
    if best_sequence is not None and time.monotonic() < deadline - 200:
        print("Phase 3: Informed annealing...")
        phase3_budget = min(500.0, deadline - time.monotonic() - 200)
        n_anneal = 4
        anneal_per = phase3_budget / n_anneal
        for i in range(n_anneal):
            if time.monotonic() > deadline - 200:
                break
            a_start = best_sequence.copy()
            a_start /= a_start.sum()
            # Add small noise for diversity
            noise_scale = 0.02 * (i + 1) / n_anneal
            noise = np.random.default_rng(seed + 10000 + i * 7).uniform(-noise_scale, noise_scale, len(a_start))
            a_start = np.maximum(a_start + noise, 0)
            a_start /= a_start.sum()
            rng_a = np.random.default_rng(seed + 5000 + i * 3)
            a_opt, _ = anneal_informed(a_start, time_limit=anneal_per, rng=rng_a)
            record(a_opt)
        print("  Phase 3 complete. Best: %.6f" % best_score)

    # Phase 4: Final LP + mirror polish
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("Phase 4: Final polish...")
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            a_best = best_sequence.copy()
            a_best /= a_best.sum()
            half = rem * 0.5
            a_opt, _ = lp_optimize(a_best, time_limit=half, t_start=0.1, t_min=0.01)
            record(a_opt)
            if time.monotonic() < deadline - 10:
                rem2 = deadline - time.monotonic() - 10
                if rem2 > 5:
                    a_opt2, _ = mirror_descent(best_sequence, time_limit=rem2, t_init=200, eta_init=0.05)
                    record(a_opt2)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))
    return [float(x) for x in best_sequence]
