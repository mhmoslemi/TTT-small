Uses L-BFGS with annealed p-norm/soft-max objectives, direct subgradient refinement,

import numpy as np

from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def pnorm_obj_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p.
    For p > 32, uses soft-max approximation.
    Returns (score, grad_theta).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    if p > 32:
        tau = max(np.max(c) * 0.02, 1e-10)
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(np.clip(shifted, -50, 0))
        sum_exp = np.sum(exp_shifted)
        soft_max_val = c_max + tau * np.log(sum_exp)
        w = exp_shifted / sum_exp
        score = 2.0 * n * soft_max_val
        a_rev = a[::-1]
        cross = fft_convolve(a_rev, w)
        grad_a = 2.0 * cross[n - 1:2 * n - 1]
        inner = np.dot(grad_a, a)
        grad_theta = a * (grad_a - inner)
        return score, grad_theta

    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def soft_max_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(np.clip(shifted, -50, 0))
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def compute_subgradient(a):
    """
    Compute subgradient of max(conv(a,a)) with respect to a.
    Uses FFT for efficiency.
    Returns (peak_value, subgradient_vector).
    """
    n = len(a)
    c = fft_convolve(a, a)
    peak_val = np.max(c)

    # Find all near-peak positions
    tol = peak_val * 1e-8 + 1e-15
    peak_mask = (c >= peak_val - tol).astype(np.float64)

    # subgrad_j = sum_k peak_mask[k] * 2 * a[k-j] for valid k-j in [0,n-1]
    # This is 2 * (cross-correlation of peak_mask with a) at lag j
    # Using FFT: IFFT(conj(FFT(peak_mask_padded)) * FFT(a_padded))
    size = 1
    while size < 3 * n:
        size <<= 1

    W = np.fft.rfft(peak_mask, size)
    A = np.fft.rfft(a, size)
    cross = np.fft.irfft(np.conj(W) * A, size)

    subgrad = 2.0 * cross[:n]
    return peak_val, subgrad

    print(f"\n=== Phase 3: Subgradient polish (score={best_score:.6f}) ===")
    if best_seq is not None:
        sg_time = min(180.0, max(60.0, deadline - time.monotonic() - 120))
        polished, p_score = subgradient_polish(best_seq, sg_time, rng)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  Subgrad: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Subgrad: {p_score:.6f} (no improvement)")

    # Phase 4: Peak peeling search
    print(f"\n=== Phase 4: Peak peeling (score={best_score:.6f}) ===")
    if best_seq is not None:
        peel_time = min(150.0, max(40.0, deadline - time.monotonic() - 60))
        peeled, p_score = peak_peeling_search(best_seq, peel_time, rng)
        if p_score < best_score:
            best_score = p_score
            best_seq = peeled.copy()
            print(f"  Peel: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Peel: {p_score:.6f} (no improvement)")

    # Phase 5: Final multi-restart L-BFGS on best length
    print(f"\n=== Phase 5: Final restarts (score={best_score:.6f}, n={best_n}) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n = len(best_seq)
        final_time = deadline - time.monotonic() - 30
        n_restarts = max(2, int(final_time / 40))
        print(f"  {n_restarts} restarts, time={final_time:.0f}s")

        for r in range(n_restarts):
            if time.monotonic() > deadline - 40:
                break
            # Perturb the best solution
            scale = 0.03 + 0.02 * r
            a_pert = best_seq * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 1e-8)
            a_pert = a_pert / np.sum(a_pert)
            theta = np.log(a_pert + 1e-30)

            t_rem = deadline - time.monotonic() - 40
            maxiter = max(50, int(t_rem / 2))
            theta_best, s = run_lbgfs_pnorm(theta, n, [8, 16, 32, 64], maxiter, deadline - 40)

            if s < best_score:
                best_score = s
                best_seq = softmax(theta_best).copy()
                print(f"  restart {r}: {s:.6f} *** NEW BEST ***")
            else:
                print(f"  restart {r}: {s:.6f}")

    # Phase 6: Very fine subgradient on final best
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Phase 6: Final micro-polish ===")
        final_polish_time = deadline - time.monotonic() - 15
        if final_polish_time > 10:
            polished, p_score = subgradient_polish(best_seq, final_polish_time, rng)
            if p_score < best_score:
                best_score = p_score
                best_seq = polished.copy()
                print(f"  Final polish: {best_score:.6f} *** NEW BEST ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
