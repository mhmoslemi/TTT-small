Strategy:
- Start from parent sequence (already at ~1.5088)
- Multi-phase FISTA with adaptive log-sum-exp beta ramp
- Phase 1: broad exploration (low beta)
- Phase 2: refinement (medium beta) 
- Phase 3: high-beta polishing
- Phase 4: exact subgradient micro-polish
- Alternative initializations for diversity
- Frequency of noise injection decreases over time
- Backtracking-aware step size

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def grad_lse(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def grad_toppk(a, c, top_k):
    """Gradient using top-k lags weighted by their values."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices].astype(np.float64)
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fista_phase(a_init, n, S, time_budget, rng, beta_start, beta_end,
                step_scale=1.0, noise_prob=0.1, noise_scale=0.003):
    """Run one phase of FISTA with beta ramping from beta_start to beta_end."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    total_iters_est = max(1, int(time_budget / 0.05))

    while time.monotonic() - start < time_budget:
        it += 1
        frac = min(1.0, it / total_iters_est)
        beta = beta_start + (beta_end - beta_start) * frac

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = grad_lse(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = step_scale * S / (beta * 2.0 * (1.0 + 0.01 * frac * it))
        step = max(step, 1e-8)

        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic noise injection to escape local minima
        if it % 500 == 0 and rng.random() < noise_prob:
            y = project_simplex(y + rng.normal(0, noise_scale * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score, best_score

    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)

    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    # --- Phase 4: Exact subgradient polish (remaining time) ---
    rem = deadline - time.monotonic()
    if rem > 5:
        t4 = rem * 0.7
        print("Phase 4: Exact subgrad polish, budget=%.0fs" % t4)
        a4, s4 = exact_subgrad_polish(a3, n, S, t4, rng, top_k=1, step_init=0.05)
        a4 = safe_normalize(a4, n)
        obj4 = get_objective(a4)
        if obj4 < global_best:
            global_best = obj4
            global_a = a4.copy()
            print("  Phase 4 best: %.6f" % global_best)
        else:
            a4 = global_a.copy()
        print("  Phase 4 done: %.6f" % obj4)

    # --- Alternative initializations (if time remains) ---
    rem = deadline - time.monotonic()
    if rem > 20:
        print("Trying alternative initializations (remaining=%.0fs)" % rem)
        tt = np.linspace(0, 1, n, endpoint=False)

        alts = [
            ("raised_cos", 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.3),
            ("gauss_center", np.exp(-2.0 * (tt - 0.5) ** 2) + 0.2),
            ("flat_wings", np.where(np.abs(tt - 0.5) < 0.4, 1.0, 0.5)),
        ]

        for name, arr in alts:
            rem = deadline - time.monotonic()
            if rem < 8:
                break
            arr = np.clip(arr, 0.01, 100.0)
            arr = safe_normalize(arr, n)
            alloc = min(rem * 0.4, 60.0)
            alt_a, alt_s, _ = fista_phase(arr, n, S, alloc, rng,
                                          beta_start=5.0, beta_end=200.0,
                                          step_scale=1.0, noise_prob=0.1, noise_scale=0.003)
            alt_a = safe_normalize(alt_a, n)
            alt_obj = get_objective(alt_a)
            if alt_obj < global_best:
                global_best = alt_obj
                global_a = alt_a.copy()
                print("  Alt [%s]: %.6f (NEW BEST)" % (name, global_best))
            else:
                print("  Alt [%s]: %.6f" % (name, alt_obj))

    # --- Final micro-polish ---
    rem = deadline - time.monotonic()
    if rem > 3 and global_a is not None:
        a_fp, s_fp = exact_subgrad_polish(global_a, n, S, min(rem, 30.0), rng,
                                          top_k=2, step_init=0.02)
        a_fp = safe_normalize(a_fp, n)
        obj_fp = get_objective(a_fp)
        if obj_fp < global_best:
            global_best = obj_fp
            global_a = a_fp.copy()
            print("  Final micro-polish: %.6f" % global_best)

    # --- Verify ---
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    ver = evaluate_sequence(final_list)
    print("\nFinal verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, internal=%.6f, eval=%.6f" %
          (len(final_list), global_best, ver if np.isfinite(ver) else np.inf))
    return final_list
