"""
Population-based optimization for minimizing peak autocorrelation of non-negative
step functions. Uses soft-max gradient descent in the probability simplex,
adaptive temperature annealing, top-k peak targeting, and diverse initialization
strategies including frequency-domain and structured random approaches.

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def topk_soft_max(c, tau, k):
    """
    Compute a top-k soft-max approximation of the max.
    Returns (value, weights) where value approximates max(c) and
    weights are distributed over the top-k elements.
    """
    # Get top-k indices
    k = min(k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx]
    
    # Soft-max over top-k
    t = top_vals - np.max(top_vals)
    e = np.exp(t / tau)
    s = np.sum(e)
    value = np.max(top_vals) + tau * np.log(s)
    w = e / s
    
    # Map back to full vector
    full_w = np.zeros(len(c))
    full_w[top_idx] = w
    return value, full_w


def grad_soft_max_conv(a, tau, top_k=32):
    """Compute soft-max(conv(a,a)) and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    value, w = topk_soft_max(c, tau, top_k)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = (1.0 / tau) * corr[n - 1:2 * n - 1] * 2.0
    return value, g, c


def init_uniform(n):
    return np.ones(n) / n


def init_random(n, rng, alpha=0.5):
    a = 1.0 + alpha * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def init_smoothed_random(n, rng, bandwidth=0.1):
    """Random values passed through a circular low-pass filter."""
    a = rng.standard_normal(n)
    A = np.fft.rfft(a)
    freqs = np.fft.rfftfreq(n)
    mask = np.exp(-((freqs / bandwidth) ** 2))
    A *= mask
    a = np.fft.irfft(A, n)
    a = np.abs(a) + 0.1
    return a / np.sum(a)


def init_fft_phase(n, rng, n_iter=50):
    """Gerchberg-Saxton: flatten magnitude spectrum."""
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    
    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.01, None)
        a = a / np.sum(a)
    return a


def init_multitone(n, rng, n_tones=5):
    """Sum of random cosine tones."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n)
    for k in range(n_tones):
        freq = rng.integers(1, max(2, n // 2))
        amp = 1.0 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.3
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def init_block(n, rng, n_blocks=8):
    """Block-constant initialization with random block values."""
    block_size = n // n_blocks
    remaining = n % n_blocks
    a = np.zeros(n)
    pos = 0
    for i in range(n_blocks):
        bs = block_size + (1 if i < remaining else 0)
        a[pos:pos+bs] = rng.uniform(0.1, 1.0)
        pos += bs
    return a / np.sum(a)

    Adam gradient descent on softmax-parameterized sequence.
    Uses adaptive soft-max temperature and top-k peak targeting.
    """

    a = a_init.copy()
    a = a / np.sum(a) * S
    theta = np.log(a + 1e-12)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr0 = 3.0
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)

    tau_min = 0.1 / n
    tau_decay = 0.9995
    
    # Top-k schedule: start with larger k, shrink over time
    k_start = min(64, 2 * n - 1)
    k_end = max(4, n // 64)
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        
        # Anneal temperature
        tau = max(tau_min, tau * tau_decay)
        
        # Anneal top-k
        k = int(k_start + (k_end - k_start) * elapsed_frac)
        k = max(4, k)
        
        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau, top_k=k)

        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)
        
        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

def local_search_direct(a_init, time_limit, rng, top_k_lags=6):
    """
    Direct local search on the sequence (not parameterized).
    Uses peak-directed mutations.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9998
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        iter_count += 1
        
        if r < 0.4:
            # Peak-directed: find top lags and reduce contributing pairs
            conv = fft_convolve(current, current)
            n_top = min(top_k_lags, len(conv))
            top_lags = np.argsort(conv)[-n_top:]
            
            # For each top lag, find the pair with max contribution
            for lag in top_lags[:3]:
                if lag < 0 or lag >= n:
                    continue
                if lag == 0:
                    # For lag 0, reduce the largest elements
                    top_idx = np.argsort(current)[-max(2, n//20):]
                    candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
                else:
                    # Find i that maximizes current[i]*current[i+lag]
                    i_max = 0
                    max_val = 0
                    for i in range(n - lag):
                        v = current[i] * current[i + lag]
                        if v > max_val:
                            max_val = v
                            i_max = i
                    # Reduce both elements
                    candidate[i_max] *= np.exp(rng.normal(-0.04, 0.02))
                    candidate[i_max + lag] *= np.exp(rng.normal(-0.04, 0.02))
        elif r < 0.6:
            # Block perturbation
            bw = max(1, int(rng.integers(1, max(2, n // 30))))
            bs = int(rng.integers(0, n - bw))
            candidate[bs:bs+bw] *= np.exp(rng.normal(-0.05, 0.03, size=bw))
            # Compensate elsewhere
            other = np.setdiff1d(np.arange(n), np.arange(bs, bs+bw))
            if len(other) > 0:
                n_inc = min(bw, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.03, size=n_inc))
        elif r < 0.75:
            # Symmetric perturbation
            idx = int(rng.integers(0, n // 2))
            delta = rng.normal(0, 0.03)
            candidate[idx] *= np.exp(delta)
            candidate[n-1-idx] *= np.exp(delta)
        elif r < 0.9:
            # Global small noise
            candidate *= np.exp(rng.normal(0, 0.01, size=n))
        else:
            # Sparsification: zero out a few elements
            n_zero = max(1, n // 50)
            zero_idx = rng.choice(n, size=n_zero, replace=False)
            candidate[zero_idx] *= 0.1
            dense_idx = np.setdiff1d(np.arange(n), zero_idx)
            n_comp = min(n_zero * 2, len(dense_idx))
            comp_idx = rng.choice(dense_idx, size=n_comp, replace=False)
            candidate[comp_idx] *= np.exp(rng.normal(0.1, 0.05, size=n_comp))
        
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / s * np.sum(current)
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 5:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score
        
        temp = max(1e-6, temp * cooling)
    
    return best, best_score

import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def topk_softmax_weights(c, tau, k):
    """
    Compute top-k soft-max value and weights over the convolution.
    Returns (value, w) where w has the same shape as c.
    """
    k = min(k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx]
    shifted = (top_vals - np.max(top_vals)) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    value = np.max(top_vals) + tau * np.log(s)
    w = np.zeros(len(c))
    w[top_idx] = e / s
    return value, w

def init_fft_phase(n, rng, n_iter=50):

        amp = 1.0 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.3
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def init_block(n, rng, n_blocks=8):
    block_size = n // n_blocks
    remaining = n % n_blocks
    a = np.zeros(n)
    pos = 0
    for i in range(n_blocks):
        bs = block_size + (1 if i < remaining else 0)
        a[pos:pos + bs] = rng.uniform(0.1, 1.0)
        pos += bs
    return a / np.sum(a)


def init_binary_smoothed(n, rng, p=0.5, bandwidth=0.2):
    """Random binary pattern smoothed by a low-pass filter."""
    a = (rng.random(n) < p).astype(np.float64)
    A = np.fft.rfft(a)
    freqs = np.fft.rfftfreq(n)
    mask = np.exp(-((freqs / max(bandwidth, 1e-6)) ** 2))
    A *= mask
    a = np.fft.irfft(A, n)
    a = np.abs(a) + 0.05
    return a / np.sum(a)


def optimize_adam(n, a_init, time_limit, rng, top_k_start=32, top_k_end=4,
                  tau_start=None, tau_end=None, lr0=3.0):
    """
    Adam gradient descent on softmax-parameterized sequence with corrected gradient.
    """
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S
    theta = np.log(a + 1e-12)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)
    
    if tau_start is None:
        tau_start = 2.0 / n
    if tau_end is None:
        tau_end = 0.05 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)
        k = max(top_k_end, int(top_k_start + (top_k_end - top_k_start) * elapsed_frac))
        
        a = softmax_normalize(theta) * S
        value, g_a, c = grad_conv_softmax(a, tau, top_k=k)
        
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)
        
        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.2 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        if step % 500 == 0:
            a_check = softmax_normalize(theta) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
            if step % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"      step={step}, score={score:.6f}, best={best_score:.6f}, "
                      f"tau={tau:.2e}, k={k}, t={elapsed:.1f}s")
    
    a_final = softmax_normalize(best_theta) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
    return a_final, best_score


def local_search_corrected(a_init, time_limit, rng, top_k_lags=4):
    """
    Local search with corrected gradient-guided mutations.
    Uses the true contribution of each element to each lag.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9999
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        iter_count += 1
        
        if r < 0.35 and n <= 4096:
            # Gradient-guided: compute which elements contribute most to peak lags
            conv = fft_convolve(current, current)
            n_top = min(top_k_lags, len(conv))
            top_lags = np.argsort(conv)[-n_top:]
            
            # Compute per-element contribution to top lags
            contrib = np.zeros(n)
            for lag in top_lags[:3]:
                if 0 <= lag < n:
                    if lag == 0:
                        contrib += current ** 2
                    else:
                        for i in range(n - lag):
                            c_ij = current[i] * current[i + lag]
                            contrib[i] += c_ij
                            contrib[i + lag] += c_ij
            
            # Reduce top contributors, increase bottom
            n_modify = max(2, n // 25)
            top_idx = np.argsort(contrib)[-n_modify:]
            bot_idx = np.argsort(contrib)[:n_modify]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.02, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.04, 0.02, size=len(bot_idx)))
        elif r < 0.55:
            # Block reduction with compensation
            bw = max(1, int(rng.integers(1, max(2, n // 20))))
            bs = int(rng.integers(0, n - bw))
            factor = np.exp(rng.normal(-0.06, 0.02, size=bw))
            candidate[bs:bs + bw] *= factor
            # Compensate
            other = np.setdiff1d(np.arange(n), np.arange(bs, bs + bw))
            if len(other) > 0:
                n_comp = min(bw + 1, len(other))
                comp_idx = rng.choice(other, size=n_comp, replace=False)
                candidate[comp_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_comp))
        elif r < 0.7:
            # Symmetric pair mutation
            idx = int(rng.integers(0, n // 2))
            delta = rng.normal(0, 0.02)
            candidate[idx] *= np.exp(delta)
            candidate[n - 1 - idx] *= np.exp(-delta)
        elif r < 0.85:
            # Small global perturbation
            candidate *= np.exp(rng.normal(0, 0.008, size=n))
        else:
            # Sparse-zero mutation
            n_zero = max(1, n // 40)
            zero_idx = rng.choice(n, size=n_zero, replace=False)
            candidate[zero_idx] *= 0.05
            dense_idx = np.setdiff1d(np.arange(n), zero_idx)
            if len(dense_idx) > 0:
                n_comp = min(n_zero * 3, len(dense_idx))
                comp_idx = rng.choice(dense_idx, size=n_comp, replace=False)
                candidate[comp_idx] *= np.exp(rng.normal(0.15, 0.05, size=n_comp))
        
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / s * np.sum(current)
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < 0.1:
                accept_prob = np.exp(-delta / max(temp, 1e-10))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score
        
        temp = max(1e-8, temp * cooling)
    
    return best, best_score


def upsample_sequence(a, target_n, rng, noise_scale=0.01):
    """Upsample a sequence to a target length via interpolation + noise."""
    n = len(a)
    if n >= target_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += noise_scale * rng.standard_normal(target_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for optimal non-negative step function using corrected gradient
    descent with diverse initialization and peak-directed local search.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 35.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Check provided starting sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()
    
    # Phase 1: Diverse initialization
    print("\n=== Phase 1: Diverse candidates ===")
    candidates = []
    n_values = [256, 384, 512, 768, 1024, 1536, 2048]
    
    for n in n_values:
        if time.monotonic() > deadline - 180:
            break
        inits = []
        inits.append(('uniform', init_uniform(n)))
        inits.append(('random', init_random(n, rng, alpha=0.3)))
        inits.append(('smoothed', init_smoothed_random(n, rng, bandwidth=0.03 + 0.08 * rng.random())))
        inits.append(('fft', init_fft_phase(n, rng, n_iter=40)))
        inits.append(('multitone', init_multitone(n, rng, n_tones=4 + rng.integers(0, 4))))
        inits.append(('block', init_block(n, rng, n_blocks=6 + rng.integers(0, 10))))
        inits.append(('bin_smooth', init_binary_smoothed(n, rng, p=0.3 + 0.4 * rng.random(),
                                                         bandwidth=0.1 + 0.2 * rng.random())))
        
        for name, a in inits:
            score = local_score(a)
            candidates.append((score, n, a.copy(), name))
        
        # Sort and print top for this n
        n_cands = [(s, nm) for s, nn, a, nm in candidates if nn == n]
        n_cands.sort()
        if n_cands:
            print(f"  n={n}: best={n_cands[0][0]:.6f} ({n_cands[0][1]}), "
                  f"2nd={n_cands[1][0]:.6f} ({n_cands[1][1]})" if len(n_cands) > 1
                  else f"  n={n}: best={n_cands[0][0]:.6f} ({n_cands[0][1]})")
    
    # Quick optimization on diverse candidates
    print("\n=== Phase 1b: Quick optimization ===")
    candidates.sort(key=lambda x: x[0])
    n_quick = min(15, len(candidates))
    time_left = deadline - time.monotonic() - 120
    time_per_quick = max(10.0, time_left / (n_quick + 3))
    
    quick_results = []
    for i in range(n_quick):
        if time.monotonic() > deadline - 100:
            break
        score0, n, a, name = candidates[i]
        t_alloc = min(time_per_quick, 45.0)
        if t_alloc < 8:
            continue
        a_opt, s_opt = optimize_adam(n, a, t_alloc, rng,
                                     top_k_start=32, top_k_end=6,
                                     tau_start=2.0/n, tau_end=0.1/n)
        quick_results.append((s_opt, n, a_opt, name))
        print(f"  [{i+1}/{n_quick}] n={n}, {name:12s}: {score0:.6f} -> {s_opt:.6f}")
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}) ***")
    
    # Phase 2: Deep optimization
    print(f"\n=== Phase 2: Deep optimization ===")
    quick_results.sort(key=lambda x: x[0])
    n_deep = min(5, len(quick_results))
    remaining = deadline - time.monotonic() - 100
    time_per_deep = max(30.0, remaining / (n_deep + 2))
    
    deep_results = []
    for i in range(n_deep):
        if time.monotonic() > deadline - 80:
            break
        score0, n, a, name = quick_results[i]
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 60)
        if t_alloc < 20:
            continue
        print(f"  [{i+1}/{n_deep}] n={n}, {name}: {score0:.6f}, t={t_alloc:.0f}s")
        a_opt, s_opt = optimize_adam(n, a, t_alloc, rng,
                                     top_k_start=16, top_k_end=4,
                                     tau_start=1.0/n, tau_end=0.05/n,
                                     lr0=2.0)
        deep_results.append((s_opt, n, a_opt, name))
        print(f"    -> {s_opt:.6f}")
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}) ***")
    
    # Phase 3: Upsampling
    print(f"\n=== Phase 3: Upsampling ===")
    all_results = quick_results + deep_results
    all_results.sort(key=lambda x: x[0])
    
    for target_n in [1024, 2048, 4096]:
        if time.monotonic() > deadline - 80:
            break
        best_short = None
        for s, n, a, name in all_results:
            if n < target_n:
                if best_short is None or s < best_short[0]:
                    best_short = (s, n, a, name)
        if best_short is None:
            continue
        s0, n0, a0, name0 = best_short
        a_up = upsample_sequence(a0, target_n, rng, noise_scale=0.015)
        t_alloc = min(80.0, deadline - time.monotonic() - 50)
        if t_alloc < 15:
            break
        print(f"  n={n0} ({name0}) -> n={target_n}, t={t_alloc:.0f}s")
        a_opt, s_opt = optimize_adam(target_n, a_up, t_alloc, rng,
                                     top_k_start=32, top_k_end=8,
                                     tau_start=2.0/target_n, tau_end=0.1/target_n)
        print(f"    -> {s_opt:.6f}")
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"  *** New best: {best_score:.6f} (n={target_n}) ***")
    
    # Phase 4: Local search polish
    print(f"\n=== Phase 4: Local search polish ===")
    if best_seq is not None:
        polish_time = min(200.0, max(30.0, deadline - time.monotonic() - 25))
        print(f"  n={len(best_seq)}, score={best_score:.6f}, t={polish_time:.0f}s")
        polished, p_score = local_search_corrected(best_seq, polish_time, rng, top_k_lags=4)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
    
    # Phase 5: Final restarts
    remaining = deadline - time.monotonic() - 20
    if remaining > 40 and best_seq is not None:
        print(f"\n=== Phase 5: Final restarts (time left: {remaining:.0f}s) ===")
        n_best = len(best_seq)
        n_restarts = min(5, int(remaining // 30))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break
            a_new = best_seq * np.exp(rng.normal(0, 0.2, size=n_best))
            a_new = np.clip(a_new, 0.01, None)
            a_new = a_new / np.sum(a_new)
            t_alloc = min(30.0, (deadline - time.monotonic() - 20) / max(1, n_restarts - r))
            a_opt, s_opt = optimize_adam(n_best, a_new, t_alloc, rng,
                                         top_k_start=16, top_k_end=4,
                                         tau_start=1.0/n_best, tau_end=0.05/n_best)
            print(f"  Restart {r+1}: {s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"  *** New best: {best_score:.6f} ***")
                # Quick polish
                if deadline - time.monotonic() > 40:
                    p2, s2 = local_search_corrected(best_seq,
                                                    min(50.0, deadline - time.monotonic() - 25),
                                                    rng)
                    if s2 < best_score:
                        best_score = s2
                        best_seq = p2.copy()
                        print(f"  After quick polish: {best_score:.6f}")
    
    # Final verification
    print(f"\n{'='*50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
