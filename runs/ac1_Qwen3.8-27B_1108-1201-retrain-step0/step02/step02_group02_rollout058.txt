2n*max(autocorr)/sum^2. Uses projected subgradient descent with soft-max

import numpy as np

def fft_convolve(a, b=None):
    """Fast linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf, 0
    c = fft_convolve(a, a)
    max_b = np.max(c)
    return 2.0 * n * max_b / (S * S), c

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 1 and np.all(v >= 0):
        return v / max(np.sum(v), 1e-15)
    
    # Standard projection algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
