def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)

        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)

    c_max = np.max(c)

def project_simplex(v, S=1.0):
    """Project v onto the probability simplex {x >= 0, sum(x) = S}."""
    n = len(v)
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def convolve_self(a):
    """Compute self-convolution of a using direct or FFT method."""
    n = len(a)
    if n <= 600:
        return np.convolve(a, a)
    else:
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = np.fft.rfft(a, size)
        result = np.fft.irfft(fa * fa, size)[:2 * n - 1]
        return np.maximum(result, 0)


def compute_gradient(a, weights):
    """
    Compute gradient of sum_k weights[k]*conv[k] w.r.t. a.
    Uses convolution identity: grad = 2 * convolve(weights, a_rev)[n-1:2n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    if n <= 600:
        full = np.convolve(weights, a_rev)
    else:
        size = 1
        while size < (2 * n - 1) + n - 1:

        fw = np.fft.rfft(weights, size)
        fa = np.fft.rfft(a_rev, size)
        full = np.fft.irfft(fw * fa, size)
    return 2.0 * full[n - 1:2 * n - 1]


def project_simplex(v):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
