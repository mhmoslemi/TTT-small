import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_subgradient(a, w):
    """Compute subgradient of weighted sum of c[k].
    grad[m] = 2 * sum_k w[k] * a[k-m] (with bounds)
    This equals 2 * (conv(w, a_rev))[m + n - 1] for valid m.
    """
    n = len(a)
    a_rev = a[::-1]
    # conv(w, a_rev) has length len(w) + n - 1 = (2n-1) + n - 1 = 3n-2
    out_len = len(w) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    conv = np.real(ifft(W * R))[:out_len]
    # We need conv[n-1 : 2n-1] for the gradient
    grad = np.zeros(n)
    end = min(2 * n - 1, out_len)
    grad[:] = 2.0 * conv[n-1: end][:n]
    return grad

def project_sum_nonneg(a, S):
    """Project onto {x : sum(x) = S, x >= 0}."""
    # Bisection on threshold
    a = np.maximum(a, 0)
    s = a.sum()
    if s <= S:
        # Need to add (S - s) while keeping non-negativity
        # The projection is a + (S - s)/n (add uniformly)
        # But this might not be the Euclidean projection...
        # Actually for the constraint sum(x) = S, x >= 0:
        # If sum(a) <= S, the projection is a + (S-s)/n * ones (if all stay non-neg)
        # which they do since a >= 0 and we're adding.
        return a + (S - s) / len(a)
    else:
        # Need to subtract. Binary search on threshold t.
        a_sorted = np.sort(a)[::-1]
        cumsum = np.cumsum(a_sorted)
        # Find largest k such that (cumsum[k-1] - (k-1)*a_sorted[k]) / k < a_sorted[k]
        # i.e., the "water level" is below a_sorted[k]
        # Standard water-filling algorithm
        n = len(a)
        # Find the water level
        # (cumsum[k] - S) / k < a_sorted[k] for the right k
        # Binary search
        lo, hi = 0, n - 1
        k = n  # default: all elements get clipped
        for _ in range(50):
            mid = (lo + hi) // 2
            if mid == 0:
                lo = mid + 1
                continue
            water_level = (cumsum[mid] - S) / (mid + 1)
            if water_level < a_sorted[mid]:
                k = mid + 1
                hi = mid
            else:
                lo = mid + 1
        if k == 0:
            k = 1
        t = (cumsum[k-1] - S) / k
        result = np.maximum(a - t, 0)
        # Fix numerical issues
        result = np.maximum(result, 0)
        s2 = result.sum()
        if s2 > 0:
            result *= S / s2
        else:
            result = np.full(n, S / n)
        return result

    best_a = a.copy()

    L = 4.0  # conservative
    
    if lr is None:
        lr = 1.0 / L

        t += 1

        progress = min(1.0, elapsed / time_budget)

        T = T_start * np.exp(np.log(T_end / T_start) * progress)

        else:

        c_y = fft_convolve_self(y)

"""
Minimize 2n*max(conv(a,a))/sum(a)^2 using Nesterov-accelerated projected
subgradient descent (FISTA) in primal space with water-filling projection,
adaptive top-K subgradient, LP-based directional improvements, and
multi-phase search strategy.

Key innovations over previous approach:
1. Direct primal-space optimization (no log-parameterization)
2. FISTA/Nesterov acceleration for faster convergence
3. Water-filling O(n log n) projection onto simplex
4. Phase-based strategy: FISTA → LP-enhanced → Multi-restart → Polish
5. Analytical coordinate descent for final polish (no FFT needed)
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution a*a via FFT, length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgrad(w, a):
    """Weighted subgradient: grad[m] = 2 * sum_k w[k] * a[k-m].
    Computed as 2 * (conv(w, a_rev))[m + n - 1]."""
    n = len(a)
    a_rev = a[::-1]
    out_len = len(w) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    conv_result = np.real(ifft(W * R))[:out_len]
    grad = np.zeros(n)
    end = min(2 * n - 1, out_len)
    grad[:] = 2.0 * conv_result[n - 1:end][:n]
    return grad


def project_simplex(a, S):
    """Euclidean projection onto {x: sum(x) = S, x >= 0}."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    np.maximum(a, 0.0, out=a)
    s = a.sum()
    if s <= S + 1e-12:
        return a + (S - s) / n
    # Water-filling: find threshold t such that sum(max(a - t, 0)) = S
    a_sorted = np.sort(a)[::-1]
    cumsum = np.cumsum(a_sorted)
    # Binary search for k: largest k with (cumsum[k-1] - S) / k < a_sorted[k]
    k = n
    lo, hi = 1, n - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        t_candidate = (cumsum[mid - 1] - S) / mid
        if t_candidate < a_sorted[mid]:
            k = mid
            lo = mid + 1
        else:
            hi = mid - 1
    t = (cumsum[k - 1] - S) / k
    result = np.maximum(a - t, 0.0)
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        mask = result > 0
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            np.maximum(result, 0.0, out=result)
        else:
            result.fill(S / n)
    return result


def get_weights(c, K, T):
    """Compute soft-top-K weights over the convolution."""
    n_c = len(c)
    K_eff = min(K, n_c)
    if K_eff == n_c:
        top_idx = np.arange(n_c)
    else:
        top_idx = np.argpartition(-c, K_eff - 1)[:K_eff]
    w = np.zeros(n_c)
    vals = c[top_idx]
    vmax = vals.max()
    if vmax < 1e-15:
        w[top_idx] = 1.0 / K_eff
    else:
        # Weights proportional to exp(T * (vals - vmax) / vmax)
        # = exp(-T * (vmax - vals) / vmax)
        scaled = -T * (vmax - vals) / vmax
        # Clamp for numerical stability
        scaled = np.maximum(scaled, -700.0)
        exp_v = np.exp(scaled)
        total = exp_v.sum()
        if total < 1e-300:
            w[top_idx] = 1.0 / K_eff
        else:
            w[top_idx] = exp_v / total
    return w


def fista_core(a_init, S, n, time_budget, lr=0.3, K_start=5, K_end=100,
               T_start=3.0, T_end=500.0, momentum_start=0.0, lr_decay=0.5):
    """Core FISTA loop with adaptive parameters."""
    a = a_init.copy().astype(np.float64)
    np.maximum(a, 1e-15, out=a)
    a *= S / a.sum()
    
    a_prev = a.copy()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    t = 0
    current_lr = lr
    stall_count = 0
    stall_patience = 500
    
    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Adaptive schedule
        K = max(1, int(K_start + (K_end - K_start) * progress))
        T = T_start * (T_end / T_start) ** progress
        current_lr = lr * (1.0 + lr_decay * (1.0 - progress))
        
        # Nesterov momentum
        mu = (t - 1) / (t + 2.0) if t > 1 else 0.0
        y = a + mu * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
        else:
            y *= S / y_sum
        
        # Gradient at y
        c_y = fft_convolve_self(y)
        w = get_weights(c_y, K, T)
        grad = compute_subgrad(w, y)
        
        # Gradient step + projection
        a_new = y - current_lr * grad
        a_new = project_simplex(a_new, S)
        
        # Track best (use the actual a_new, not y)
        c_new = fft_convolve_self(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive learning rate: reduce if stalled
        if stall_count > stall_patience:
            current_lr *= 0.7
            stall_count = 0
        
        a_prev = a
        a = a_new
    
    return best_a, best_obj


def solve_lp_direction(f, M, n, top_k=50, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0.
    Returns normalized g (sum=1) or None."""
    from scipy.optimize import linprog
    
    c = fft_convolve_self(f)
    if top_k < len(c):
        active_lags = np.argpartition(-c, top_k - 1)[:top_k]
    else:
        active_lags = np.arange(len(c))
    
    # Build constraint matrix A: A[idx, j] = f[k - j] if 0 <= k-j < n
    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[idx, j] = f[k - j]
    
    c_obj = -np.ones(n)
    b = np.full(len(active_lags), M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

    best_a = a.copy()

            break

            # Try a few candidate values and evaluate.
