"""
Minimize peak-to-average ratio of linear autocorrelation for non-negative sequences.
Uses smoothed-Lp optimization with Nesterov acceleration, spectral initialization,
and targeted peak-reduction local search. Explores multiple sequence lengths.
"""

import numpy as np
import time


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def smoothed_lp(a, p, tau=1.0):
    """Compute smoothed p-norm of conv(a,a) and its gradient."""
    n = len(a)
    conv = fft_convolve(a, a)
    if tau < 1e-10:
        max_val = np.max(conv)
        peak_idx = np.argmax(conv)
        # Gradient of max is the gradient of that element
        # d(conv[peak_idx])/d(a[j]) = a[peak_idx - j] + (if peak_idx+j < 2n-1) a[peak_idx + j]
        # Actually for linear conv: conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/d(a[j]) = a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n)
        # = a[|k-j|] if the index is valid... 
        # More precisely: conv[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j]*a[k-j]
        # d(conv[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n and j!=k-j)
        # Wait, that's double counting. Let me be more careful.
        # conv[k] = sum_j a[j] * a[k-j] where 0<=j<n and 0<=k-j<n
        # d(conv[k])/d(a[j]) = a[k-j] (from the first factor) 
        # But if k-j = j (i.e., k = 2j), then we get 2*a[j].
        # Actually no: conv[k] = sum_j a[j]*a[k-j]. 
        # d/d(a[j0]) = a[k-j0] (from the term where j=j0 in the first factor)
        #              + a[j0-k] (from the term where k-j=j0, i.e., j=k-j0, in the second factor)
        # But the second term is only valid if 0 <= k-j0 < n AND j0 != k-j0 (to avoid double counting)
        # If j0 = k-j0 (i.e., k=2*j0), then d/d(a[j0]) = 2*a[j0]
        
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= peak_idx - j < n:
                val += a[peak_idx - j]
            if 0 <= j - peak_idx < n and j != peak_idx - j:
                val += a[j - peak_idx]
            elif 0 <= j - peak_idx < n and j == peak_idx - j:
                val += a[j - peak_idx]  # This adds another a[j], giving 2*a[j]
            grad[j] = val
        # Simpler: use the FFT-based cross-correlation
        # d(conv[k])/d(a[j]) for all j is the cross-correlation of a with a delta at k
        # = a shifted... 
        
        # Actually the simplest way: the gradient of conv[k] w.r.t. a is:
        # (grad)[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        # But this counts the j term twice when k=2j. Let me just use the formula:
        # grad = autocorr of a at lag (k-j) ... 
        
        # Easiest: use the fact that conv = a * a (convolution), so:
        # d(conv[k])/d(a[j]) = a[k-j] + a[j-k] (with appropriate bounds)
        # where a[x] = 0 if x < 0 or x >= n
        
        # Let me just compute it directly for the peak:
        grad = np.zeros(n)
        for j in range(n):
            # Contribution from a[j] as the first element
            if 0 <= peak_idx - j < n:
                grad[j] += a[peak_idx - j]
            # Contribution from a[j] as the second element (a[k-j] where k-j = j, so k = 2j)
            # This is the same as: a[j] appears as the second factor when k - j' = j, i.e., j' = k - j
            if 0 <= peak_idx - j < n:
                pass  # already counted
            # Actually, let me think again.
            # conv[k] = sum_{j'} a[j'] * a[k-j']
            # d(conv[k])/d(a[j]) = sum_{j'} [delta(j',j)*a[k-j'] + a[j']*delta(k-j',j)]
            # = a[k-j] (if 0<=j<n and 0<=k-j<n) + a[j-k] (if 0<=k-j<n and j-j... 
            # = a[k-j] (if valid) + a[j-k] (if valid, and j != k-j to avoid double count)
            # If k-j = j (k=2j): d/d(a[j]) = 2*a[j]
            
            if 0 <= peak_idx - j < n:
                grad[j] += a[peak_idx - j]
            if 0 <= j - peak_idx < n:
                grad[j] += a[j - peak_idx]
            # But if peak_idx - j == j (i.e., peak_idx = 2*j), we've added a[j] twice, which is correct (2*a[j])
        
        return max_val, grad
    
    # Smoothed case: use softmax weighting
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    value = max_c + tau * np.log(np.sum(exp_vals))
    
    # Gradient: grad = 2 * (w * a) convolved... 
    # d(conv[k])/d(a[j]) * w[k] summed over k
    # = 2 * sum_k w[k] * (a[k-j] + a[j-k]) with bounds
    # = 2 * cross_correlation(a, w)[j] approximately
    # More precisely: sum_k w[k] * d(conv[k])/d(a[j])
    # = sum_k w[k] * (a[k-j] + a[j-k]) [with bounds]
    # = sum_k w[k] * a[k-j] + sum_k w[k] * a[j-k]
    # First sum: k-j in [0,n-1] => k in [j, j+n-1], and k in [0, 2n-2]
    #   = sum_{k=j}^{min(j+n-1, 2n-2)} w[k] * a[k-j]
    #   = (w convolved with a)[j] (approximately)
    # Second sum: j-k in [0,n-1] => k in [j-n+1, j], and k in [0, 2n-2]
    #   = sum_{k=max(0,j-n+1)}^{min(j, 2n-2)} w[k] * a[j-k]
    #   = (w correlated with a)[j]
    
    # Use FFT for efficiency
    size = 2 * n
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    # First term: circular convolution of w and a, take first n
    conv1 = np.fft.irfft(fw * fa, size)[:n]
    # Second term: circular correlation of w and a
    # = ifft(conj(fw) * fa) ... hmm, let me think
    # correlation: sum_k w[k] * a[j-k] = ifft(conj(FW) * FA)[j]
    corr2 = np.fft.irfft(np.conj(fw) * fa, size)[:n]
    grad = 2.0 * (conv1 + corr2)
    
    return value, grad, conv


def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    indic = np.arange(1, n + 1)
    cond = u * indic > (cv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def cosine_taper(n, alpha=1.0, rng=None):
    """Create a cosine-tapered sequence."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = 1.0 + alpha * (1.0 - t)  # Linear taper from 1+alpha to 1
    # Or cosine:
    # a = 1.0 + alpha * 0.5 * (1 + np.cos(np.pi * t))
    if rng is not None:
        a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def spectral_hio_init(n, rng, n_iter=100, taper='cosine'):
    """HIO-style spectral initialization."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if taper == 'cosine':
        a = 0.5 + 0.5 * np.cos(np.pi * (t - 0.5))
        a = 1.0 + 0.5 * (1.0 - a)  # Inverted: high at edges
    elif taper == 'linear':
        a = 1.0 + 0.5 * (1.0 - t)
    elif taper == 'inverted_cosine':
        a = 1.0 + 0.5 * np.cos(np.pi * t)  # High in middle
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
    
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    
    for it in range(n_iter):
        # Compute circular autocorrelation (zero-padded for linear)
        size = 2 * n
        A = np.fft.rfft(a, size)
        power = np.abs(A) ** 2
        corr = np.fft.irfft(power, size)[:2*n-1]
        
        # Target: reduce peaks above average
        avg = np.sum(corr) / (2*n - 1)  # = 1 (since sum(a)=1)
        # Soft threshold: reduce values above avg
        alpha = 0.1
        excess = np.maximum(corr - avg * 1.01, 0)
        
        # Modify spectrum to reduce time-domain peaks
        # Simple approach: scale down the time domain
        a = a - alpha * np.interp(np.arange(n), np.arange(2*n-1), corr)
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    
    return a


def optimize_smoothed_lp(n, a_init, time_limit, rng, p=20):
    """
    Minimize smoothed p-norm of conv(a,a) on the simplex using
    Nesterov-accelerated projected gradient descent.
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    a_prev = a.copy()
    t_nesterov = 1.0
    best_a = a.copy()
    best_max_conv = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    step = 0
    
    # Learning rate schedule
    lr_base = 0.5 / np.sqrt(n)
    
    # Temperature schedule
    tau_start = 1.0 / n
    tau_end = 0.01 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)
        
        # Nesterov step
        t_prev = t_nesterov
        t_nesterov = (1.0 + np.sqrt(1.0 + 4.0 * t_nesterov ** 2)) / 2.0
        beta = (t_prev - 1.0) / t_nesterov
        
        y = a + beta * (a - a_prev)
        y = project_simplex(y)
        
        value, grad, conv = smoothed_lp(y, p, tau)
        
        # Adaptive learning rate
        lr = lr_base / (1.0 + 0.1 * step / n)
        
        candidate = y - lr * grad
        candidate = project_simplex(candidate)
        
        a_prev = a
        a = candidate
        step += 1
        
        # Track best
        current_max = np.max(fft_convolve(a, a))
        if current_max < best_max_conv:
            best_max_conv = current_max
            best_a = a.copy()
        
        if step % 2000 == 0:
            sc = 2.0 * n * best_max_conv  # since sum=1
            print(f"      SLP step={step}, score={sc:.6f}, t={time.monotonic()-t0:.1f}s")
    
    return best_a, 2.0 * n * best_max_conv


def optimize_adam_taper(n, a_init, time_limit, rng):
    """
    Adam optimization directly on the sequence (with simplex projection).
    Uses smoothed max with annealing.
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Work in log-space for non-negativity: a = exp(theta) / sum(exp(theta))
    theta = np.log(a + 1e-10)
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.1
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    
    tau_start = 5.0 / n
    tau_end = 0.05 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)
        
        # Compute a from theta (softmax)
        theta_shifted = theta - np.max(theta)
        exp_theta = np.exp(theta_shifted)
        a = exp_theta / np.sum(exp_theta)
        
        # Compute smoothed max and gradient w.r.t. a
        value, g_a, conv = smoothed_lp(a, 2, tau)
        
        # Gradient w.r.t. theta: softmax Jacobian
        # d(a_i)/d(theta_j) = a_i * (delta_ij - a_j)
        # grad_theta = S * a * (g_a - a @ g_a) where S = sum(a) = 1
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner)
        
        step += 1
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * (g_theta * g_theta)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate periodically
        if step % 100 == 0:
            current_score = 2.0 * n * np.max(fft_convolve(a, a))
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        
        if step % 3000 == 0:
            current_score = 2.0 * n * np.max(fft_convolve(a, a))
            print(f"      Adam step={step}, curr={current_score:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")
    
    return best_a, best_score


def peak_targeted_search(a_init, time_limit, rng):
    """
    Local search that specifically targets the peak of the convolution.
    Uses a "push-down" strategy: find the peak lag, identify contributors,
    and redistribute mass away from them.
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    n = len(a)
    
    current_score = 2.0 * n * np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 500
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        conv = fft_convolve(a, a)
        peak_val = np.max(conv)
        
        # Find top peak lags
        top_k = min(5, 2*n-1)
        peak_lags = np.argsort(conv)[-top_k:][::-1]
        
        improved = False
        
        for k in peak_lags:
            # Find contributors to conv[k]
            # conv[k] = sum_j a[j]*a[k-j] for valid j
            contribs = np.zeros(n)
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            for j in range(low, high + 1):
                partner = k - j
                contribs[j] += a[partner]
                if partner != j:
                    contribs[partner] += a[j]
            
            # Find the top contributors
            n_top = max(3, min(n // 8, 20))
            top_idx = np.argsort(contribs)[-n_top:]
            
            # Strategy: reduce top contributors, add to non-contributors
            total_reduce = 0.0
            candidate = a.copy()
            for idx in top_idx:
                frac = rng.uniform(0.02, 0.1)
                amt = a[idx] * frac
                candidate[idx] -= amt
                total_reduce += amt
            
            # Redistribute to non-contributor indices
            non_contrib = np.setdiff1d(np.arange(n), top_idx)
            if len(non_contrib) > 0:
                # Prefer indices that don't contribute to the peak
                n_add = min(10, len(non_contrib))
                add_idx = rng.choice(non_contrib, size=n_add, replace=False)
                candidate[add_idx] += total_reduce / n_add
            
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            
            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                improved = True
                if current_score < best_score:
                    best_score = current_score
                    best_a
