"""
Optimization of non-negative step functions to minimize peak autocorrelation ratio.
Uses L-BFGS with log-sum-exp smoothing and progressive temperature cooling.
Explores multiple initializations and sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def smooth_max_and_weights(c, tau):
    """Compute smooth max and its weights (gradient w.r.t. c)."""
    m = np.max(c)
    shifted = (c - m) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    value = m + tau * np.log(s)
    w = e / s  # gradient of smooth_max w.r.t. c
    return value, w


def gradient_wrt_p(w, p):
    """
    Compute gradient of <w, conv(p,p)> w.r.t. p.
    w has length 2n-1, p has length n.
    Returns gradient of length n.
    """
    n = len(p)
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    W = rfft(w, N)
    P = rfft(p, N)
    grad = irfft(W * np.conj(P), N)[:n]
    return 2.0 * grad


def softmax_grad_objective(params, tau, n):
    """
    Objective and gradient for L-BFGS.
    params: theta (length n), a = softmax(theta)
    Returns: (value, gradient)
    """
    # Compute a = softmax(params)
    t = params - np.max(params)
    e = np.exp(t)
    p = e / np.sum(e)
    
    # Compute convolution
    c = fft_convolve(p, p)
    
    # Smooth max
    val, w = smooth_max_and_weights(c, tau)
    
    # Gradient w.r.t. p
    grad_p = gradient_wrt_p(w, p)
    
    # Gradient w.r.t. params (theta)
    # p = softmax(theta), so dp_i/dtheta_j = p_i * (delta_ij - p_j)
    # grad_theta = p * (grad_p - sum(p * grad_p))
    inner = np.dot(p, grad_p)
    grad_theta = p * (grad_p - inner)
    
    return val, grad_theta


def optimize_with_lbfgs(n, theta_init, tau_schedule, max_iter_per_phase=200):
    """
    Run L-BFGS optimization with progressive temperature cooling.
    tau_schedule: list of tau values to try (decreasing).
    """
    theta = theta_init.copy()
    best_score = np.inf
    best_theta = theta.copy()
    
    for tau in tau_schedule:
        # Run L-BFGS for this tau
        result = minimize(
            softmax_grad_objective,
            theta,
            args=(tau, n),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter_per_phase, 'gtol': 1e-6, 'ftol': 1e-8}
        )
        theta = result.x
        
        # Evaluate actual score
        t = theta - np.max(theta)
        e = np.exp(t)
        p = e / np.sum(e)
        S = np.sqrt(2.0 * n)
        a = p * S
        c = fft_convolve(a, a)
        score = 2.0 * n * np.max(c) / (np.sum(a) ** 2)
        
        if score < best_score:
            best_score = score
            best_theta = theta.copy()
    
    return best_theta, best_score
