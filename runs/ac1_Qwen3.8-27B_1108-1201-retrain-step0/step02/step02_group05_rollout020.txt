"""
LP-iterative + L-BFGS hybrid optimization for minimum peak autocorrelation
non-negative sequences. Combines a linear-programming-based direction-finding
method (solving max sum(g) s.t. f*g <= M) with gradient-based p-norm annealing,
subgradient peak-shaving, and targeted local search. Explores multiple sequence
lengths with diverse initializations.

import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def local_score(a):
    """Score for sequence: 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = fft_convolve(a, a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def local_score_normalized(a):
    """Score assuming sum(a)=1: 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def build_conv_matrix(f, n):
    """Build the convolution matrix A where (A @ g)[k] = (f*g)(k)."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = f[k - j]
    return A


def solve_lp_for_direction(f, M):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    Returns (optimal_value, g) or (None, None) on failure.
    """
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return None, None


def lp_refine(f_init, time_limit, rng, t_init=0.15, t_decay=0.9995, t_min=0.005):
    """
    Iterative LP-based refinement.
    Normalizes f to have sum = sqrt(2n), then iteratively solves LP and updates.
    Returns (best_f, best_score).
    """
    f = f_init.astype(np.float64).copy()
    f = np.maximum(f, 0.0)
    f_sum = np.sum(f)
    if f_sum < 1e-10:
        return f_init, local_score(f_init)
    
    n = len(f)
    target_sum = np.sqrt(2.0 * n)
    f = f * (target_sum / f_sum)
    
    best_f = f.copy()
    best_score = local_score(f)
    t = t_init
    t0 = time.monotonic()
    last_improve = t0
    stall_count = 0
    max_stall = 5000
    
    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(f, f))
        if M < 1e-15:
            break
        
        val, g = solve_lp_for_direction(f, M)
        if g is None or val is None:
            break
        
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        # Rescale g to match target sum
        g = g * (target_sum / g_sum)
        
        # Update
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new_sum = np.sum(f_new)
        if f_new_sum < 1e-10:
            break
        f_new = f_new * (target_sum / f_new_sum)
        
        new_score = local_score(f_new)
        
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            last_improve = time.monotonic()
            stall_count = 0
        else:
            stall_count += 1
        
        f = f_new
        t = max(t * t_decay, t_min)
        
        if stall_count > max_stall:
            # Perturb and reset
            f = best_f * np.exp(rng.normal(0, 0.05, n))
            f = np.maximum(f, 0)
            f = f * (target_sum / np.sum(f))
            t = min(t_init, t * 2)
            stall_count = 0
        
        if time.monotonic() - last_improve > 120:
            # Long stall - try restart
            f = best_f * np.exp(rng.normal(0, 0.1, n))
            f = np.maximum(f, 0)
            f = f * (target_sum / np.sum(f))
            t = t_init
            stall_count = 0
    
    return best_f, best_score


def pnorm_obj_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p.
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    if p >= 1e6:
        # Just use max
        c_max = np.max(c)
        score = 2.0 * n * c_max
        idx = np.argmax(c)
        # Subgradient: 2 * f(idx - j) for valid j
        grad_a = np.zeros(n)
        for j in range(n):
            i = idx - j
            if 0 <= i < n:
                grad_a[j] += 2.0 * a[i]
        inner = np.dot(grad_a, a)
        grad_theta = a * (grad_a - inner)
        return score, grad_theta
    
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    
    factor = 2.0 * n * p * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def run_lbgfs_pnorm(theta, p_schedule, maxiter, deadline):
    """Run L-BFGS with annealed p-norm."""
    best_theta = theta.copy()
    best_score = local_score_normalized(softmax(theta))
    
    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                pnorm_obj_grad, theta.copy(),
                method='L-BFGS-B',
                jac=True, args=(p,),
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxfun': maxiter * 3, 'maxls': 50}

                best_theta = theta_new.copy()
            theta = best_theta
        except Exception:

    return best_theta, best_score


def soft_max_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    
    score = 2.0 * n * soft_max_val
    
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta

def subgrad_descent(a, time_limit, lr_init=0.01, lr_decay=0.999, rng=None):
    """
    Subgradient descent on 2n * max(conv(a,a)) with projection onto sum=1, a>=0.
    """
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    n = len(a)
    best_a = a.copy()
    best_score = local_score_normalized(a)
    lr = lr_init
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))
        
        # Subgradient: 2 * a(peak_idx - j) for valid j
        grad = np.zeros(n)
        for j in range(n):
            i = peak_idx - j
            if 0 <= i < n:
                grad[j] = 2.0 * a[i]
        
        # Projected gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        new_score = local_score_normalized(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        
        a = a_new
        lr *= lr_decay
    
    return best_a, best_score

def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Bell-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def make_cosine_init(n, rng, freqs=None):
    """Cosine-based initialization with multiple frequencies."""
    if freqs is None:
        freqs = [1.0, 2.0, 3.0, 0.5]
    t = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for i, f in enumerate(freqs):
        a += 0.3 / (i + 1) * np.cos(2 * np.pi * f * t + rng.uniform(0, 2 * np.pi))
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def upscale_sequence(a, new_n):
    """Upscale a sequence to a new length using interpolation."""
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-6)
    return a_new / np.sum(a_new)

    t0 = time.monotonic()
    lp_result, lp_score = lp_refine(lp_init, lp_deadline - time.monotonic(), rng)
    print(f"  LP refinement: score={lp_score:.6f} (took {time.monotonic()-t0:.1f}s)")
    update_best(lp_result, lp_score)
    
    # Try LP on a slightly different init
    if start_seq is not None and time.monotonic() < lp_deadline - 60:
        lp_init2 = start_seq * np.exp(rng.normal(0, 0.08, n_lp))
        lp_init2 = np.maximum(lp_init2, 1e-6)
        lp_init2 = lp_init2 / np.sum(lp_init2)
        t0 = time.monotonic()
        lp_result2, lp_score2 = lp_refine(lp_init2, lp_deadline - time.monotonic(), rng)
        print(f"  LP refinement (perturbed): score={lp_score2:.6f}")
        update_best(lp_result2, lp_score2)
    
    # ================================================================
    # Phase 2: L-BFGS with p-norm annealing (multiple lengths/inits)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS p-norm annealing")
    print(f"{'='*60}")
    
    lbgfs_deadline = t_global + min(1400.0, budget_s * 0.65)
    lengths = [192, 256, 320, 384]
    
    for n in lengths:
        if time.monotonic() > lbgfs_deadline - 60:
            break
        
        # Allocate time based on priority
        if n == best_len or (n == 256 and best_len == 0):
            time_alloc = min(350.0, (lbgfs_deadline - time.monotonic()) * 0.4)
        else:
            time_alloc = min(200.0, (lbgfs_deadline - time.monotonic()) * 0.2)
        
        n_deadline = time.monotonic() + time_alloc
        print(f"\n  --- n={n} (time_alloc={time_alloc:.0f}s) ---")
        
        # Generate initializations
        inits = []
        if start_seq is not None:
            a_interp = upscale_sequence(start_seq, n)
            inits.append(('parent_interp', a_interp))
        if best_seq is not None and len(best_seq) != n:
            a_best_interp = upscale_sequence(best_seq, n)
            inits.append(('best_interp', a_best_interp))
            a_best_pert = a_best_interp * np.exp(rng.normal(0, 0.1, n))
            a_best_pert = np.maximum(a_best_pert, 1e-6)
            a_best_pert = a_best_pert / np.sum(a_best_pert)
            inits.append(('best_pert', a_best_pert))
        
        inits.append(('chirp1', make_chirp_init(n, 0.3 + 0.2 * rng.random(), rng)))
        inits.append(('chirp2', make_chirp_init(n, 0.5 + 0.3 * rng.random(), rng)))
        inits.append(('random', make_random_init(n, rng, spread=0.2 + 0.3 * rng.random())))
        inits.append(('bell', make_bell_init(n, n * (0.12 + 0.12 * rng.random()), rng)))
        inits.append(('cosine', make_cosine_init(n, rng)))
        
        p_schedule = [2, 4, 8, 16, 32, 64, 128, 256]
        
        for init_name, a0 in inits:
            if time.monotonic() > n_deadline - 10:
                break
            
            theta = np.log(a0 + 1e-30)
            t_init_now = time.monotonic()
            
            # Run L-BFGS with p-norm
            theta_best, s = run_lbgfs_pnorm(theta, p_schedule, maxiter=300,
                                            deadline=n_deadline - 5)
            
            if s < best_score:
                a_best = softmax(theta_best)
                update_best(a_best, s)
            else:
                print(f"    {init_name}: {s:.6f}")
            
            # If we have time, also try soft-max on the best theta
            if time.monotonic() < n_deadline - 30:
                tau_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
                theta_sm, s_sm = run_lbgfs_softmax(theta_best, tau_schedule,
                                                   maxiter=400, deadline=n_deadline - 3)
                if s_sm < best_score:
                    a_best = softmax(theta_sm)
                    update_best(a_best, s_sm)
                else:
                    # Check if soft-max improved over p-norm
                    if s_sm < s:
                        print(f"    soft-max improved: {s_sm:.6f}")
        
        print(f"  Best after n={n}: {best_score:.6f}")
    
    # ================================================================
    # Phase 3: LP refinement on best candidate
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: LP refinement on best candidate (score={best_score:.6f})")
    print(f"{'='*60}")
    
    if best_seq is not None:
        lp3_deadline = t_global + min(1900.0, budget_s * 0.82)
        time_remaining = lp3_deadline - time.monotonic()
        if time_remaining > 30:
            print(f"  Time available: {time_remaining:.0f}s")
            a3 = best_seq.astype(np.float64).copy()
            a3 = np.maximum(a3, 0)
            a3 = a3 / np.sum(a3)
            t0 = time.monotonic()
            lp3_result, lp3_score = lp_refine(a3, time_remaining - 5, rng,
                                               t_init=0.1, t_decay=0.9998, t_min=0.003)
            print(f"  LP on best: score={lp3_score:.6f} (took {time.monotonic()-t0:.1f}s)")
            update_best(lp3_result, lp3_score)
            
            # Also try subgradient descent
            if time.monotonic() < lp3_deadline - 30:
                sg_deadline = lp3_deadline - time.monotonic()
                if sg_deadline > 20:
                    a_sg, sg_score = subgrad_descent(a3, min(sg_deadline, 100),
                                                     lr_init=0.005, lr_decay=0.9995)
                    print(f"  Subgrad on best: score={sg_score:.6f}")
                    update_best(a_sg, sg_score)
    
    # ================================================================
    # Phase 4: Final local search polish
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Local search polish (score={best_score:.6f})")
    print(f"{'='*60}")
    
    if best_seq is not None:
        polish_deadline = deadline - 20
        time_remaining = polish_deadline - time.monotonic()
        if time_remaining > 30:
            # Run multiple polish iterations from slightly different starting points
            polish_inits = [best_seq.astype(np.float64).copy()]
            polish_inits.append(best_seq * np.exp(rng.normal(0, 0.03, len(best_seq))))
            polish_inits[-1] = np.maximum(polish_inits[-1], 1e-6)
            polish_inits[-1] = polish_inits[-1] / np.sum(polish_inits[-1])
            
            per_init_time = time_remaining / len(polish_inits)
            
            for i, p_init in enumerate(polish_inits):
                if time.monotonic() > polish_deadline - 10:
                    break
                t0 = time.monotonic()
                polished, p_score = polish_local_search(p_init, per_init_time - 5, rng)
                print(f"  Polish {i+1}/{len(polish_inits)}: score={p_score:.6f} "
                      f"(took {time.monotonic()-t0:.1f}s)")
                update_best(polished, p_score)
    
    # ================================================================
    # Final: Verify and return
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Final result:")
    print(f"{'='*60}")
    
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"  n={len(final_list)}, local_score={best_score:.6f}, "
              f"evaluate_sequence={final_eval:.6f}")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
