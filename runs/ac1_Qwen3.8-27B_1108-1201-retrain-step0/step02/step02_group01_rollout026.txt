    # grad[m] = sum_{k in active} a[k-m]*(0<=k-m<n) + sum_{k in active} a[m-k]*(0<=m-k<n)
    # = sum_{k in active, k>=m, k-m<n} a[k-m] + sum_{k in active, k<m, m-k<n} a[m-k]
    
    # Term 1: sum_{k in active, k>=m} a[k-m] (k-m < n is automatic since k <= 2n-2 and m >= 0, so k-m <= 2n-2 < 3n... hmm not quite)
    # Actually k ranges from 0 to 2n-2, m from 0 to n-1. k-m ranges from -(n-1) to 2n-2. We need 0 <= k-m < n, so m <= k <= m+n-1.
    # Term 1: sum_{k=max(m,0) to min(m+n-1, 2n-2), k in active} a[k-m]
    # Let j = k-m: sum_{j=0 to min(n-1, 2n-2-m), j+m in active} a[j]
    # = sum_j a[j] * w[j+m] (where w is indicator of active)
    
    # Term 2: m-k in [0, n-1], so k in [m-n+1, m]. Since k >= 0: k in [max(0, m-n+1), m].
    # sum_{k in active, max(0,m-n+1) <= k <= m} a[m-k]
    # Let j = m-k: sum_{j=0 to min(m, n-1)} a[j] * w[m-j]
    # = sum_j a[j] * w[m-j]
    
    # So grad[m] = sum_j a[j]*w[j+m] + sum_j a[j]*w[m-j]
    # = (correlation of a and w at lag m) + (convolution of a and w at m)
    # where correlation: sum_j a[j]*w[j+m] and convolution: sum_j a[j]*w[m-j]
    
    # For FFT computation:
    # conv(a, w)[m] = sum_j a[j]*w[m-j] → this is standard convolution, easily computed via FFT.
    # corr(a, w)[m] = sum_j a[j]*w[j+m] → this is cross-correlation.
    # corr(a,w)[m] = conv(a, w_flipped)[len(w)-1+m] where w_flipped[k] = w[len(w)-1-k]
    # Because conv(a, w_flipped)[p] = sum_j a[j]*w_flipped[p-j] = sum_j a[j]*w[len(w)-1-p+j]
    # Set p = len(w)-1+m: sum_j a[j]*w[len(w)-1-(len(w)-1-m)+j] = sum_j a[j]*w[m+j] ✓
    
    # Great! So:
    # grad[m] = conv(a, w)[m] + conv(a, w_flipped)[len(w)-1+m]
    # where w has length 2n-1, w_flipped has length 2n-1.
    # conv(a, w) has length n + (2n-1) - 1 = 3n-2.
    # conv(a, w_flipped) has length 3n-2.
    # We need elements [0:n] of the first and [2n-2 : 2n-2+n] = [2n-2 : 3n-2] of the second.
    
    # Let me implement this with FFT.
    w = np.zeros(2*n-1)
    w[active] = 1.0 / len(active)
    
    # conv(a, w): length 3n-2
    # conv(a, w_flipped): we need elements 2n-2 to 3n-3
    w_flipped = w[::-1]
    
    # Use FFT for both
    fsize = 1 << (3*n - 2 - 1).bit_length()  # need at least 3n-2
    fsize = 1 << (3*n - 2).bit_length()  # power of 2 >= 3n-2
    A = fft(a, fsize)
    W = fft(w, fsize)
    Wf = fft(w_flipped, fsize)
    
    conv_aw = np.real(ifft(A * W))[:3*n-2]
    conv_awf = np.real(ifft(A * Wf))[:3*n-2]
    
    t2 = conv_aw[:n]  # sum_j a[j]*w[m-j] for m=0..n-1
    t1 = conv_awf[2*n-2 : 3*n-2]  # sum_j a[j]*w[j+m] for m=0..n-1
    
    grad = t1 + t2
    return grad


def backtracking_line_search(a, grad, S, max_iter=50):

    n = len(a)
    c0 = fft_convolve_self(a)
    f0 = c0.max()
    alpha = 1.0
    c_alpha = 1e-4  # Armijo condition

        alpha *= 0.5
    
    return a, alpha

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using hybrid optimization:
1. Exact subgradient descent with Nesterov acceleration
2. Periodic LP-based direction finding (SoTA approach)
3. Coordinate descent polish
4. Multi-start with various initializations

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_subgradient(a, c, active_threshold=None):
    """Compute subgradient of max(conv(a,a)) at a.
    
    Args:
        a: sequence of length n
        c: self-convolution of a (length 2n-1)
        active_threshold: fraction of max to include in active set (default 0.95)
    
    Returns:
        grad: subgradient vector of length n
    """
    n = len(a)
    c_max = c.max()
    if c_max <= 0:
        return np.zeros(n)
    
    if active_threshold is None:
        active_threshold = 0.95
    
    threshold = c_max * active_threshold
    active = np.where(c >= threshold)[0]
    n_active = len(active)
    
    if n_active == 0:
        # Shouldn't happen, but fallback
        active = np.array([np.argmax(c)])
        n_active = 1
    
    # w[k] = 1/n_active if k in active, else 0
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / n_active
    
    # grad[m] = sum_j a[j]*w[j+m] + sum_j a[j]*w[m-j]
    # = conv(a, w_flipped)[2n-2+m] + conv(a, w)[m]
    w_flipped = w[::-1]
    
    fsize = 1 << (3 * n - 2).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    Wf = fft(w_flipped, fsize)
    
    conv_aw = np.real(ifft(A * W))
    conv_awf = np.real(ifft(A * Wf))
    
    t2 = conv_aw[:n]  # sum_j a[j]*w[m-j] for m=0..n-1
    t1 = conv_awf[2*n-2 : 3*n-2]  # sum_j a[j]*w[j+m] for m=0..n-1
    
    grad = t1 + t2
    return grad

def normalize(a, S):
    """Normalize a to have sum S."""
    s = a.sum()
    if s < 1e-300:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def nesterov_subgradient(a, S, time_budget, initial_lr=0.1, active_threshold=0.95):
    """Nesterov accelerated subgradient descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Nesterov state
    y = a.copy()
    beta = 0.9  # momentum
    
    start = time.monotonic()
    it = 0
    lr = initial_lr
    
    while time.monotonic() - start < time_budget:
        # Nesterov extrapolation
        if it > 0:
            t_k = (1 + np.sqrt(1 + 4 * beta**2)) / 2
            beta_new = (1 - t_k) / (1 + t_k)
            y = a + beta_new * (a - y)
        else:
            y = a.copy()
        
        # Compute gradient at y
        c_y = fft_convolve_self(y)
        y_max = c_y.max()
        
        if y_max < best_obj:
            best_obj = y_max
            best_a = y.copy()
        
        grad = compute_subgradient(y, c_y, active_threshold)
        
        # Project gradient onto sum=0 subspace
        grad_mean = grad.mean()
        grad_proj = grad - grad_mean
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad_proj)
        if grad_norm < 1e-15:
            break
        
        # Decrease lr over time
        progress = (time.monotonic() - start) / time_budget
        lr_t = lr * (1.0 - 0.5 * progress)
        
        # Step
        a_new = y - lr_t * grad_proj / grad_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S)
        
        # Check if we improved
        c_new = fft_convolve_self(a_new)
        f_new = c_new.max()
        
        if f_new > best_obj * 1.5:
            # Step was too large, reduce lr
            lr *= 0.5
            a_new = a  # revert
        else:
            a = a_new
        
        it += 1
    
    return best_a, best_obj

def solve_lp_direction(f, M, n, top_k=100, time_limit=5):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0.
    Returns normalized g and LP objective."""
    c = fft_convolve_self(f)
    # Get top-k lags
    active_lags = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix A: A[i,j] = f[active_lags[i] - j] if 0 <= active_lags[i]-j < n
    K = len(active_lags)
    A = np.zeros((K, n))
    for i, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[i, j] = f[k - j]
    
    # Maximize sum(g) = minimize -sum(g)
    c_obj = -np.ones(n)
    b = np.full(K, M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                         method='highs', options={'time_limit': time_limit})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum * np.sqrt(2.0 * n), g_sum
    except Exception:
        pass
    return None, None

def lp_based_iteration(a, S, n, top_k=100, alpha=0.3):
    """One LP-based iteration: solve LP, blend with current."""
    c = fft_convolve_self(a)
    M = c.max()
    
    g, lp_val = solve_lp_direction(a, M, n, top_k=top_k)
    if g is None:
        return a, M
    
    # Blend
    a_new = (1 - alpha) * a + alpha * g
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S)
    
    c_new = fft_convolve_self(a_new)
    obj_new = c_new.max()
    
    if obj_new < M:
        return a_new, obj_new
    
    # Try smaller alpha
    for alpha_try in [0.1, 0.05, 0.01]:
        a_try = (1 - alpha_try) * a + alpha_try * g
        a_try = np.maximum(a_try, 0.0)
        a_try = normalize(a_try, S)
        c_try = fft_convolve_self(a_try)
        if c_try.max() < M:
            return a_try, c_try.max()
    
    return a, M

def coordinate_descent(a, S, time_budget, max_sweeps=100):
    """Coordinate descent polish."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            
            # Try reducing a[i]
            # The contribution of a[i] to conv[k] is a[i]*a[k-i] + a[i]*a[i-k]
            # So reducing a[i] reduces all conv values that involve a[i]
            # Try a few values
            best_local = orig
            best_local_obj = best_obj
            
            for frac in [0.5, 0.7, 0.85, 0.3, 0.1]:
                a[i] = orig * frac
                a_tmp = normalize(a, S)
                c_tmp = fft_convolve_self(a_tmp)
                obj_tmp = c_tmp.max()
                if obj_tmp < best_local_obj:
                    best_local_obj = obj_tmp
                    best_local = orig * frac
            
            a[i] = best_local
            a = normalize(a, S)
            
            if best_local < orig - 1e-10:
                improved = True
                c = fft_convolve_self(a)
                if c.max() < best_obj:
                    best_obj = c.max()
                    best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_obj

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2:
- Phase 1: Nesterov subgradient descent (exact max, no soft-max approximation)
- Phase 2: LP-based directional refinement (SoTA approach with iterative blending)
- Phase 3: Fine Nesterov polish with adaptive active set
- Phase 4: Coordinate descent + final subgradient polish
Key improvements: exact subgradient (no soft-max bias), aggressive LP usage,
adaptive active set management, and Nesterov acceleration.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, c, active_fraction=0.95):
    """Compute subgradient of max(conv(a,a)) at a using FFT.
    
    grad[m] = sum_{k in active} (a[k-m] + a[m-k]) / |active|
    Computed as: conv(a, w_flipped)[2n-2+m] + conv(a, w)[m]
    """
    n = len(a)
    c_max = c.max()
    if c_max <= 0:
        return np.zeros(n)
    
    threshold = c_max * active_fraction
    active = np.where(c >= threshold)[0]
    n_active = len(active)
    if n_active == 0:
        active = np.array([np.argmax(c)])
        n_active = 1
    
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / n_active
    
    w_flipped = w[::-1]
    
    fsize = 1 << (3 * n - 2).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    Wf = fft(w_flipped, fsize)
    
    conv_aw = np.real(ifft(A * W))
    conv_awf = np.real(ifft(A * Wf))
    
    t2 = conv_aw[:n]
    t1 = conv_awf[2 * n - 2: 3 * n - 2]
    
    grad = t1 + t2
    return grad


def normalize_seq(a, S):
    """Normalize a to have sum S, keeping non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def subgradient_descent(a, S, time_budget, initial_lr=0.5, active_fraction=0.98,
                        use_nesterov=True, restart_interval=200):
    """Subgradient descent with optional Nesterov acceleration."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize_seq(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    lr = initial_lr
    momentum = 0.9
    v = np.zeros(n)  # for momentum
    stuck_count = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Compute subgradient
        grad = compute_subgradient(a, c, active_fraction)
        
        # Project onto sum=0 subspace
        grad = grad - grad.mean()
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Adaptive learning rate
        lr_t = lr * (1.0 + 0.5 * (1.0 - progress))
        
        # Momentum update
        v = momentum * v - lr_t * grad / grad_norm
        a_new = a + v
        a_new = normalize_seq(a_new, S)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        f_new = c_new.max()
        
        if f_new > best_obj * 1.3:
            lr *= 0.5
            v *= 0.5
            a = a  # keep old
        else:
            a = a_new
        
        # Periodic restart to escape local minima
        if stuck_count > restart_interval and progress < 0.8:
            # Perturb current best
            noise = np.random.default_rng(it).random(n) * 0.01 * best_a.max()
            a = normalize_seq(best_a + noise, S)
            v = np.zeros(n)
            stuck_count = 0
            lr = lr * 0.7
            restart_interval = int(restart_interval * 1.5)
        
        it += 1
    
    return best_a, best_obj


def solve_lp_direction(f, M, n, top_k=80, time_limit=3.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top_k lags, g >= 0."""
    c = fft_convolve_self(f)
    active_lags = np.argsort(c)[::-1][:top_k]
    K = len(active_lags)
    
    # Build constraint matrix efficiently
    A = np.zeros((K, n))
    for i, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            A[i, lo:hi+1] = f[k - hi:k - lo + 1][::-1]
    
    c_obj = -np.ones(n)
    b = np.full(K, M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                         method='highs', options={'time_limit': time_limit,
                                                    'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum * np.sqrt(2.0 * n), g_sum
    except Exception:
        pass
    return None, None


def lp_refinement(a, S, n, time_budget, initial_top_k=150, min_top_k=30):
    """Iterative LP-based refinement."""
    a = a.copy().astype(np.float64)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

        progress = (time.monotonic() - start) / time_budget
        top_k = max(min_top_k, int(initial_top_k - (initial_top_k - min_top_k) * progress))
        
        c = fft_convolve_self(a)
        M = c.max()
        
        g, _ = solve_lp_direction(a, M, n, top_k=top_k, time_limit=2.0)
        if g is None:
            it += 10  # skip ahead
            continue
        
        # Try multiple blend alphas
        improved = False
        for alpha in alphas:
            a_try = (1.0 - alpha) * a + alpha * g
            a_try = normalize_seq(a_try, S)
            c_try = fft_convolve_self(a_try)
            obj_try = c_try.max()
            if obj_try < best_obj:

                best_obj = obj_try
                best_a = a_try.copy()
                improved = True

def coordinate_descent(a, S, time_budget):
    """Coordinate descent for final polish."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize_seq(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    sweep = 0
    
    # Precompute which positions contribute most to the peak
    peak_lag = np.argmax(c)
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        improved = False
        
        # Focus on positions that contribute to the peak
        # grad[m] for peak at k: a[k-m] + a[m-k]
        # So positions m where a[k-m] or a[m-k] is large are most important
        contributions = np.zeros(n)
        for m in range(n):
            if 0 <= peak_lag - m < n:
                contributions[m] += a[peak_lag - m]
            if 0 <= m - peak_lag < n:
                contributions[m] += a[m - peak_lag]
        
        # Sort by contribution (try high-contribution positions first)
        order = np.argsort(contributions)[::-1]
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            
            # Try reducing a[i] by various fractions
            best_local = orig
            best_local_obj = best_obj
            
            for frac in [0.7, 0.5, 0.8, 0.3, 0.9]:
                a[i] = orig * frac
                a_tmp = normalize_seq(a, S)
                c_tmp = fft_convolve_self(a_tmp)
                obj_tmp = c_tmp.max()
                if obj_tmp < best_local_obj:
                    best_local_obj = obj_tmp
                    best_local = orig * frac
            
            a[i] = best_local
            
            if abs(best_local - orig) > 1e-12:
                improved = True
                a = normalize_seq(a, S)
                c = fft_convolve_self(a)
                if c.max() < best_obj:
                    best_obj = c.max()
                    best_a = a.copy()
                    peak_lag = np.argmax(c)
        
        if not improved:
            break
    
    return best_a, best_obj

            a_best, obj_best = coordinate_descent(a, S, remaining2)
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print("Phase 5b: obj=%.6f" % obj_best)
    
    # Final verification and output
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: internal obj=%.6f, verified=%.6f" % (global_best, ver))
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.6f" % ver)
    
    return final_a
