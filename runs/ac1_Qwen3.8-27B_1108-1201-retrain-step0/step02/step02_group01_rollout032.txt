    # This is ifft(fft(w) * conj(fft(a)))
    fsize = 1 << (len(w) + n - 1 - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    xcorr = np.real(ifft(W * np.conj(A)))

def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    # Find lambda such that sum(max(0, v[i] + lambda)) = S
    # f(lambda) = sum(max(0, v[i] + lambda)) is monotonically increasing
    lo = -v.max() - 1.0  # f(lo) <= 0

        s = np.sum(np.maximum(0.0, v + mid))
        if s < S:

        else:

def softmax_obj_and_weights(c, T):
    """Compute soft-max objective and weights."""
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = shifted - shifted.max()  # numerical stability (max is 0)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    obj = c_max + np.log(total) / T
    w = exp_shifted / total
    return obj, w

    c = fft_convolve_self(a)
    return c.max()

def projected_gradient_step(a, grad, lr, S):
    """One step of projected gradient descent."""
    v = a - lr * grad
    return project_nonneg_sum(v, S)

def armijo_line_search(a, grad, f_val, T, S, lr_init=0.01, c1=1e-4, max_trials=20):
    """Armijo backtracking line search."""
    n = len(a)
    # Project gradient
    grad_proj = grad - np.mean(grad)
    lr = lr_init
    for _ in range(max_trials):
        a_new = projected_gradient_step(a, grad_proj, lr, S)
        c_new = fft_convolve_self(a_new)
        obj_new, _ = softmax_obj_and_weights(c_new, T)
        # Armijo condition
        if obj_new <= f_val - c1 * lr * np.dot(grad_proj, a - a_new):
            return a_new, obj_new, lr
        lr *= 0.5
    return a, f_val, lr

    a = a_init.copy()

    best_a = a.copy()

    lr = lr_init

        progress = elapsed / time_budget
        T = T_start * (T_end / T_start) ** progress  # exponential annealing
        
        c = fft_convolve_self(a)
        true_max = c.max()
        if true_max < best_obj:

        grad_proj = grad - np.mean(grad)

        a, obj, lr = armijo_line_search(a, grad_proj, obj, T, S, lr_init=lr)
        
        # Adapt learning rate
        if obj < best_obj:
            lr = min(lr * 1.05, 0.1)  # increase slightly on success
        else:
            lr *= 0.9  # decrease on failure
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Projected gradient descent with soft-max loss and temperature annealing
2. Active-set (top-K) subgradient with adaptive K for high-temperature phase
3. LP-based directional improvements for escape from local minima
4. Efficient gradient computation via cross-correlation (single FFT)
5. Armijo line search for robust step size selection

import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_weights(w, a):
    """Compute grad_a[i] = 2 * sum_k w[k] * a[k-i] for i=0..n-1.
    
    Uses cross-correlation: xcorr(w, a)[i] = sum_j w[j+i] * a[j]
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 1 - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    xcorr = np.real(ifft(W * np.conj(A)))
    return 2.0 * xcorr[:n]


def project_onto_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection on lambda."""
    # Find lambda such that sum(max(0, v[i] + lambda)) = S
    # f(lambda) = sum(max(0, v + lambda)) is monotonically increasing in lambda
    lo = -v.max() - 1.0
    hi = (S - v.sum()) / len(v) + 1.0  # rough upper bound
    
    # Ensure f(hi) >= S
    while np.sum(np.maximum(0.0, v + hi)) < S:
        hi = hi * 2 + 1.0
    
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.maximum(0.0, v + mid)) < S:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    return np.maximum(0.0, v + lam)


def soft_plus_approx(c, T):
    """Compute log-sum-exp approximation to max and the weights."""
    c_max = c.max()
    shifted = T * (c - c_max)
    # shifted.max() = 0
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    obj = c_max + log_sum / T
    w = exp_shifted / exp_shifted.sum()
    return obj, w


def top_k_subgradient(c, a, K):
    """Compute subgradient using top-K peaks."""
    n = len(a)
    idx = np.argsort(c)[::-1][:K]
    w = np.zeros(len(c))
    w[idx] = 1.0 / K
    return compute_grad_weights(w, a)


def solve_lp(f, M, n, top_k=50, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for top-k lags, g >= 0."""
    c = fft_convolve_self(f)
    active_lags = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix A: A[k,j] = f[k-j] if 0<=k-j<n
    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[idx, j] = f[k - j]
    
    c_obj = -np.ones(n)
    b = np.full(len(active_lags), M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

        if true_max < best_true_obj:
            best_true_obj = true_max
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

def coordinate_descent_efficient(a, S, time_budget, max_sweeps=100):
    """Efficient coordinate descent: only update coordinates that contribute to the peak."""
    a = a.copy().astype(np.float64)
    n = len(a)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        c = fft_convolve_self(a)
        peak_idx = c.argmax()
        
        # Find coordinates that contribute to the peak
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        active_coords = list(range(lo, hi + 1))
        
        # Randomize order
        np.random.shuffle(active_coords)
        
        for i in active_coords:
            if time.monotonic() - start > time_budget:
                break
            
            original = a[i]
            if original < 1e-12:
                continue
            
            # Try reducing a[i]
            # The contribution to c[peak_idx] from a[i] is a[i]*a[peak_idx-i]
            # Reducing a[i] by delta reduces c[peak_idx] by approximately 2*delta*a[peak_idx-i]
            # (if i != peak_idx-i) or delta*a[peak_idx-i] (if i == peak_idx-i)
            
            # Binary search for best value
            lo_b, hi_b = 0.0, original
            best_local = original
            best_local_obj = c[peak_idx]
            
            for _ in range(10):
                mid = (lo_b + hi_b) / 2.0
                a[i] = mid
                s = a.sum()
                if s < 1e-10:
                    a[i] = original
                    break
                a_norm = a * (S / s)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_local_obj:
                    best_local_obj = obj_check
                    best_local = mid
                    lo_b = mid
                else:
                    hi_b = mid
            
            a[i] = best_local
            a = a * (S / a.sum())
        
        c = fft_convolve_self(a)
        if c.max() < best_obj:
            best_obj = c.max()
            best_a = a.copy()
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a multi-phase approach:
1. Projected gradient descent with soft-max loss and temperature annealing
2. Top-K active-set subgradient for high-temperature refinement  
3. LP-based directional improvements for local minimum escape
4. Efficient coordinate descent for final polish
Uses FFT-based gradient computation via cross-correlation.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import time


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_weights(w, a):
    """grad_a[i] = 2 * sum_k w[k] * a[k-i] for i=0..n-1 (cross-correlation)."""
    n = len(a)
    N = 1 << (3 * n - 2 - 1).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    corr = np.real(ifft(np.conj(A) * W))
    return 2.0 * corr[:n]


def project_onto_simplex(v, S):
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    lo = -v.max() - 1.0
    hi = S / n * 2.0 + abs(v).max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.maximum(0.0, v + mid)) < S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.maximum(0.0, v + lam)


def soft_plus(c, T):
    c_max = c.max()
    if T < 1e10:
        shifted = T * (c - c_max)
        shifted = shifted - shifted.max()
        exp_s = np.exp(shifted)
        obj = c_max + np.log(exp_s.sum()) / T
        w = exp_s / exp_s.sum()
    else:
        idx = c.argmax()
        obj = c_max
        w = np.zeros(len(c))
        w[idx] = 1.0
    return obj, w


def run_pgd(a_init, S, time_budget, T_start=5.0, T_end=800.0, lr0=0.003):
    a = a_init.copy().astype(np.float64)
    n = len(a)
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best_obj = c.max()
    
    lr = lr0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_start * (T_end / T_start) ** progress
        
        c = fft_convolve_self(a)
        true_max = c.max()
        if true_max < best_obj - 1e-12:
            best_obj = true_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            lr = lr0 * (0.5 ** (no_improve / 200.0))
            no_improve = 0
            if lr < 1e-8:
                break
        
        obj, w = soft_plus(c, T)
        grad = compute_grad_weights(w, a)
        grad_proj = grad - np.mean(grad)
        
        # Try a few step sizes
        accepted = False
        for scale in [1.0, 0.5, 0.25, 0.1]:
            lr_t = lr * scale
            v = a - lr_t * grad_proj
            a_new = project_onto_simplex(v, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < true_max:
                a = a_new
                accepted = True
                break
        
        if accepted:
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.5
    
    return best_a, best_obj


def run_topk_pgd(a_init, S, time_budget, K=8, lr0=0.001):
    a = a_init.copy().astype(np.float64)
    n = len(a)
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best_obj = c.max()
    
    lr = lr0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        true_max = c.max()
        if true_max < best_obj - 1e-12:
            best_obj = true_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            break
        
        # Top-K subgradient
        idx = np.argsort(c)[::-1][:K]
        w = np.zeros(len(c))
        w[idx] = 1.0 / K
        grad = compute_grad_weights(w, a)
        grad_proj = grad - np.mean(grad)
        
        v = a - lr * grad_proj
        a_new = project_onto_simplex(v, S)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() < true_max:
            a = a_new
            lr = min(lr * 1.01, 0.02)
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return best_a, best_obj


def solve_lp_direction(f, M, n, top_k=40, time_limit=8):
    c = fft_convolve_self(f)
    active_lags = np.argsort(c)[::-1][:top_k]
    
    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[idx, lo:hi+1] = f[hi:lo-1:-1]  # f[k-j] for j in lo..hi
    
    c_obj = -np.ones(n)
    b = np.full(len(active_lags), M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            if g.sum() > 1e-10:
                return g / g.sum()
    except Exception:
        pass
    return None


def coord_polish(a_init, S, time_budget):
    a = a_init.copy().astype(np.float64)
    n = len(a)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        peak_val = c.max()
        peak_idx = c.argmax()
        
        # Active coordinates for the peak
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        active = list(range(lo, hi + 1))
        rng.shuffle(active)
        
        for i in active:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            
            # Find best a[i] by trying a few values
            best_val = orig
            best_local = peak_val
            
            # Try reducing
            for frac in [0.7, 0.5, 0.3, 0.1]:
                trial = orig * frac
                a[i] = trial
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_t = a * (S / s)
                c_t = fft_convolve_self(a_t)
                if c_t.max() < best_local:
                    best_local = c_t.max()
                    best_val = trial
                a[i] = orig
            
            # Try increasing
            for frac in [1.5, 2.0]:
                trial = orig * frac
                a[i] = trial
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_t = a * (S / s)
                c_t = fft_convolve_self(a_t)
                if c_t.max() < best_local:
                    best_local = c_t.max()
                    best_val = trial
                a[i] = orig
            
            a[i] = best_val
            a = a * (S / a.sum())
        
        c = fft_convolve_self(a)
        if c.max() < best_obj - 1e-10:
            best_obj = c.max()
            best_a = a.copy()
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a multi-phase approach:
1. Projected gradient descent with soft-max loss, momentum, and temperature annealing
2. Top-K active-set subgradient for high-temperature refinement
3. LP-based directional improvements for local minimum escape
4. Efficient coordinate descent for final polish
Uses FFT-based cross-correlation for fast gradient computation.

import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_weights(w, a):
    """grad_a[i] = 2 * sum_k w[k] * a[k-i] for i=0..n-1 via cross-correlation."""
    n = len(a)
    N = 1 << (3 * n - 2 - 1).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    corr = np.real(ifft(np.conj(A) * W))
    return 2.0 * corr[:n]


def project_onto_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} via bisection on lambda."""
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    # f(lam) = sum(max(0, v + lam)) is monotonically increasing
    # We need f(lam) = S
    lo = -v.max() - 1.0
    hi = (S / n) + abs(v).max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.maximum(0.0, v + mid)) < S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.maximum(0.0, v + lam)
    # Correct for floating point
    s = result.sum()
    if s > 0 and abs(s - S) > 1e-10:
        result = result * (S / s)
    return result


def soft_plus(c, T):
    """Compute soft-max approximation to max and normalized weights."""
    c_max = c.max()
    if T >= 1e8:
        obj = c_max
        w = np.zeros(len(c))
        # Find all indices at max (within tolerance)
        idx = np.where(c >= c_max - 1e-10)[0]
        w[idx] = 1.0 / len(idx)
        return obj, w
    shifted = T * (c - c_max)
    shifted = shifted - shifted.max()
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    obj = c_max + np.log(total) / T
    w = exp_s / total
    return obj, w


def run_pgd_momentum(a_init, S, time_budget, T_start=3.0, T_end=600.0, lr0=0.003, beta=0.85):
    """PGD with heavy-ball momentum, soft-max annealing, and backtracking."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best_obj = c.max()

    lr = lr0
    momentum = np.zeros(n)
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_start * (T_end / T_start) ** progress

        c = fft_convolve_self(a)
        true_max = c.max()
        if true_max < best_obj - 1e-14:
            best_obj = true_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr = lr0 * (0.7 ** (no_improve / 150.0))
            momentum *= 0.5
            no_improve = 0
            if lr < 1e-8:
                break

        obj, w = soft_plus(c, T)
        grad = compute_grad_weights(w, a)
        grad_proj = grad - np.mean(grad)

def solve_lp_direction(f, M, n, top_k=30, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for top-k lags, g >= 0."""
    c = fft_convolve_self(f)
    active_lags = np.argsort(c)[::-1][:top_k]

    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            # A[idx, lo:hi+1] = f[k-hi : k-lo+1] reversed
            A[idx, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

    c_obj = -np.ones(n)
    b = np.full(len(active_lags), M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception as e:
        print("LP error: %s" % str(e))
    return None


def coord_polish(a_init, S, time_budget):
    """Efficient coordinate descent focusing on peak-contributing coordinates."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(777)
    stall = 0

    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        peak_val = c.max()
        peak_idx = c.argmax()

        if peak_val < best_obj - 1e-12:
            best_obj = peak_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 20:
            break

        # Active coordinates for the current peak
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        active = list(range(lo, hi + 1))
        rng.shuffle(active)

        sweep_improved = False
        for i in active:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue

            best_val = orig
            best_local = c.max()

            # Try several multipliers
            fracs = [0.5, 0.7, 0.85, 0.3, 1.3, 1.1]
            rng.shuffle(fracs)
            for frac in fracs:
                trial = orig * frac
                if trial < 0:
                    continue
                a[i] = trial
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    continue
                a_t = a * (S / s)
                c_t = fft_convolve_self(a_t)
                obj_t = c_t.max()
                if obj_t < best_local:
                    best_local = obj_t
                    best_val = trial
            a[i] = best_val
            if abs(best_val - orig) > 1e-10 * orig:
                sweep_improved = True

        a = a * (S / a.sum())

        if not sweep_improved:
            break

    return best_a, best_obj
