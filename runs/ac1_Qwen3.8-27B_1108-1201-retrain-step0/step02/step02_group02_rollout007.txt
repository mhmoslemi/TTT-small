"""
Optimization of non-negative step functions to minimize the peak autocorrelation ratio.
Uses gradient descent with smoothed max, coordinate descent polish, and multi-start strategy.
Focuses on n=384 with secondary exploration at nearby lengths.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Fast computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, a
    c = fft_convolve(a, a)
    score = 2.0 * n * np.max(c) / (S * S)
    return float(score), a

def gradient_smoothed(a, tau):
    """
    Compute score and gradient using softmax smoothing of the max.
    Returns (score, gradient, conv).
    """
    n = len(a)
    S = np.sum(a)
    c = fft_convolve(a, a)
    
    # Softmax weights
    c_shifted = (c - np.max(c)) / tau
    exp_c = np.exp(c_shifted)
    w = exp_c / np.sum(exp_c)
    
    # Smoothed max
    peak = np.max(c) + tau * np.log(np.sum(exp_c))
    
    # Gradient of smoothed max w.r.t. a
    # d(peak)/da_j = sum_k w_k * dc_k/da_j
    # dc_k/da_j = 2 * a_{k-j} * I[0 <= k-j <= n-1]
    # So d(peak)/da_j = 2 * sum_k w_k * a_{k-j} * I[0 <= k-j <= n-1]
    # This is 2 * (w_reversed * a)[j + offset]... let me think.
    # sum_k w_k * a_{k-j} where 0 <= k-j <= n-1
    # Let m = k-j, then k = m+j, and m ranges from 0 to n-1.
    # sum_{m=0}^{n-1} w_{m+j} * a_m
    # This is the cross-correlation of w (shifted) with a.
    
    # Actually, let's use the FFT approach:
    # d(peak)/da_j = 2 * (w * a_reversed)[j + n - 1] ... let me verify.
    # (w * a_reversed)_k = sum_i w_i * a_reversed_{k-i} = sum_i w_i * a_{n-1-(k-i)} = sum_i w_i * a_{n-1-k+i}
    # For k = j + n - 1: = sum_i w_i * a_{n-1-j-n+1+i} = sum_i w_i * a_{i-j}
    # = sum_i w_i * a_{i-j} where 0 <= i-j <= n-1, i.e., j <= i <= j+n-1
    # Let m = i-j: sum_{m=0}^{n-1} w_{m+j} * a_m
    # Yes! That's what we want.
    
    a_rev = a[::-1]
    corr = fft_convolve(w, a_rev)
    g_peak = 2.0 * corr[n-1:2*n-1]
    
    # Now compute gradient of full score = 2n * peak / S^2
    # d(score)/da_j = 2n * g_peak_j / S^2 - 2n * peak * 2 / S^3
    # = (2n/S^2) * g_peak_j - (4n*peak/S^3)
    
    grad = (2.0 * n / (S * S)) * g_peak - (4.0 * n * peak / (S ** 3)) * np.ones(n)
    
    return float(2.0 * n * peak / (S * S)), grad, c

def exact_subgradient(a):
    """Compute exact subgradient at the argmax lag."""
    n = len(a)
    S = np.sum(a)
    c = fft_convolve(a, a)
    k_star = int(np.argmax(c))
    peak_val = c[k_star]
    
    # Gradient of c_k_star w.r.t. a_j = 2 * a_{k_star-j} * I[0 <= k_star-j <= n-1]
    g = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            g[j] = 2.0 * a[idx]
    
    # Score gradient
    grad = (2.0 * n / (S * S)) * g - (4.0 * n * peak_val / (S ** 3)) * np.ones(n)
    
    return float(2.0 * n * peak_val / (S * S)), grad, k_star

def gradient_descent(a_init, time_limit, rng, verbose=True):
    """
    Gradient descent with adaptive temperature and backtracking line search.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    a = a * (S_target / np.sum(a))  # Normalize
    
    score, _, _ = compute_score(a)
    best_a = a.copy()
    best_score = score
    
    t0 = time.monotonic()
    step = 0
    patience_counter = 0
    PATIENCE = 500
    
    # Adam buffers
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = elapsed / time_limit
        
        # Adaptive temperature
        if frac < 0.5:
            tau = (50.0 / n) * (1.0 / n) ** (frac / 0.5)
            score, grad, c = gradient_smoothed(a, tau)
        else:
            # Use exact subgradient for the last half
            score, grad, k_star = exact_subgradient(a)
            # Add small random tie-breaking if multiple peaks
            peak_indices = np.where(c > c.max() * 0.999)[0]
            if len(peak_indices) > 1:
                k_rand = peak_indices[rng.integers(len(peak_indices))]
                g_rand = np.zeros(n)
                for j in range(n):
                    idx = k_rand - j
                    if 0 <= idx < n:
                        g_rand[j] = 2.0 * a[idx]
                grad = (2.0 * n / (np.sum(a) ** 2)) * g_rand - (4.0 * n * c[k_rand] / (np.sum(a) ** 3)) * np.ones(n)
        
        # Adam update
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * grad ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)
        
        # Learning rate
        base_lr = 5.0 / np.sqrt(step + 1)
        if frac < 0.3:
            lr = base_lr
        elif frac < 0.7:
            lr = base_lr * 0.5
        else:
            lr = base_lr * 0.2
        
        # Backtracking line search
        a_trial = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_trial = np.clip(a_trial, 0.0, None)
        
        trial_score, _ = compute_score(a_trial)
        
        # If trial is worse, try smaller step
        if trial_score > score * 1.01:
            # Try 0.5x step
            a_trial = a - 0.5 * lr * m_hat / (np.sqrt(v_hat) + eps)
            a_trial = np.clip(a_trial, 0.0, None)
            trial_score, _ = compute_score(a_trial)
        
        if trial_score < score:
            a = a_trial
            score = trial_score
            if score < best_score:
                best_score = score
                best_a = a.copy()
                patience_counter = 0
            # Renormalize occasionally
            if step % 100 == 0:
                a = a * (S_target / np.sum(a))
                score, _ = compute_score(a)
        else:
            patience_counter += 1
        
        if patience_counter > PATIENCE:
            # Restart from best with perturbation
            a = best_a + 0.05 * best_a * rng.standard_normal(n)
            a = np.clip(a, 0.0, None)
            score, _ = compute_score(a)
            patience_counter = 0
        
        if verbose and step % 500 == 0:
            print(f"    GD step={step}, score={score:.6f}, best={best_score:.6f}, tau={'exact' if frac >= 0.5 else f'{tau:.3f}'}")
    
    return best_a, best_score

def coordinate_descent(a_init, time_limit, rng, verbose=True):
    """
    Coordinate descent: for each element, try a range of values and keep the best.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    score, _ = compute_score(a)
    best_a = a.copy()
    best_score = score
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    
    # Precompute the "rest" contributions for efficiency
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        improved = False
        
        # Process coordinates in random order
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-10:
                # Try turning on this coordinate
                trial_vals = [S / n * 0.5, S / n, S / n * 1.5]
            else:
                # Try geometric around current value
                trial_vals = [old_val * f for f in [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.6, 2.0]]
                # Also try zero
                trial_vals.append(0.0)
            
            best_trial = old_val
            best_trial_score = score
            
            for tv in trial_vals:
                a[j] = max(0.0, tv)
                s_trial, _ = compute_score(a)
                if s_trial < best_trial_score:
                    best_trial_score = s_trial
                    best_trial = max(0.0, tv)
            
            a[j] = best_trial
            if best_trial_score < score - 1e-12:
                score = best_trial_score
                improved = True
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        no_improve = 0 if improved else no_improve + 1
        
        if no_improve >= 5:
            # Perturb to escape local minimum
            n_perturb = max(5, n // 10)
            idx = rng.choice(n, n_perturb, replace=False)
            a[idx] *= np.exp(rng.normal(0, 0.2, n_perturb))
            a = np.clip(a, 0.0, None)
            score, _ = compute_score(a)
            no_improve = 0
        
        if verbose and iteration % 5 == 0:
            print(f"    CD iter={iteration}, score={score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score

def peak_redistribution(a_init, time_limit, rng, verbose=True):
    """
    Targeted peak reduction: identify which elements contribute most to the peak
    and redistribute mass from them to less contributing elements.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    score, _ = compute_score(a)
    best_a = a.copy()
    best_score = score
    
    t0 = time.monotonic()
    temp = 0.01
    cooling = 0.9999
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))
        
        # Compute contribution of each element to the peak
        contrib = np.zeros(n)
        for j in range(n):
            idx = peak_idx - j
            if 0 <= idx < n:
                contrib[j] = a[j] * a[idx]
        
        # Strategy: reduce top contributors, boost bottom
        top_k = max(3, n // 20)
        bottom_k = max(5, n // 10)
        
        sorted_idx = np.argsort(contrib)
        top_idx = sorted_idx[-top_k:]
        bot_idx = sorted_idx[:bottom_k]
        
        r = rng.random()
        if r < 0.4:
            # Reduce top, boost bottom
            amt = 0.03 * np.sum(a[top_idx])
            a[top_idx] -= amt / len(top_idx)
            a[bot_idx] += amt / len(bot_idx)
        elif r < 0.6:
            # Swap: reduce one top, boost one bottom
            j_top = top_idx[rng.integers(len(top_idx))]
            j_bot = bot_idx[rng.integers(len(bot_idx))]
            amt = a[j_top] * 0.1
            a[j_top] -= amt
            a[j_bot] += amt
        elif r < 0.8:
            # Scale a random block
            width = max(2, min(n // 4, int(rng.integers(2, n // 3 + 1))))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.05))
            a[start:start+width] *= factor
        else:
            # Global perturbation
            a *= np.exp(rng.normal(0, 0.01, n))
        
        a = np.clip(a, 0.0, None)
        new_score, _ = compute_score(a)
        
        if new_score < score:
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        elif rng.random() < np.exp(-(new_score - score) / max(temp, 1e-10)):
            score = new_score
        
        temp *= cooling
    
    return best_a, best_score

def make_initial(n, rng, method='random'):
    """Create initial sequences with various methods."""
    if method == 'random':
        a = 1.0 + 0.5 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'broadband':
        t = np.arange(n, dtype=np.float64)
        a = np.ones(n)
        for k in range(1, 20):
            freq = k * (1.0 + 0.5 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.1 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'binary_random':
        # Binary-like: mostly 0 or 1 with some variation
        a = rng.random(n)
        a = np.where(a > 0.5, 1.0, 0.1) + 0.1 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'sine_mod':
        t = np.arange(n, dtype=np.float64) / n
        a = 1.0 + 0.4 * np.sin(2 * np.pi * rng.uniform(1, 5) * t + rng.uniform(0, 2*np.pi))
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    else:
        return np.ones(n)
