"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.
Uses multi-scale FISTA with adaptive top-K log-sum-exp,
segment-based initialization, and perturbation-based restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_grad_lse(a, c, beta):
    """Gradient of log-sum-exp smooth max via FFT cross-correlation."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def fft_grad_toppk(a, c, top_k):
    """Gradient using top-k active lags via FFT cross-correlation."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def make_segment_sequence(n, n_segments, rng):
    """Create a sequence with smooth segment transitions."""
    # Choose boundaries
    if n_segments == 1:
        return np.ones(n) * 0.5
    boundaries = np.sort(rng.permutation(np.arange(1, n))[:n_segments-1])
    boundaries = np.concatenate([[0], boundaries, [n]])
    
    # Choose heights with smooth variation
    base_heights = rng.random(n_segments) + 0.1
    # Add a global shape (bump)
    centers = (boundaries[:-1] + boundaries[1:]) / 2.0
    bump = np.exp(-0.5 * ((centers - n/2) / (n/4))**2)
    heights = base_heights * (0.5 + 0.5 * bump)
    
    seq = np.zeros(n)
    for i in range(n_segments):
        lo, hi = boundaries[i], boundaries[i+1]
        seq[lo:hi] = heights[i]
    
    # Smooth transitions
    kernel_size = min(11, n_segments)
    if kernel_size >= 3 and kernel_size % 2 == 1:
        kernel = np.ones(kernel_size) / kernel_size
        from scipy.ndimage import uniform_filter1d
        seq = uniform_filter1d(seq, size=kernel_size)
    
    return np.maximum(seq, 0.0)

def run_fista_adaptive(a_init, n, S, max_time, rng, beta_init=3.0, beta_max=500.0, K_init=50, K_min=1):
    """FISTA with adaptive top-K log-sum-exp."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    last_improve = 0
    stagnation = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive K: start large, decrease over time
        K = max(K_min, K_init - it // 100)
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            last_improve = it
            stagnation = 0
        else:
            stagnation += 1

        # Compute gradient using top-K with smoothing
        if K >= len(c):
            grad = fft_grad_lse(y, c, beta)
        else:
            # Use top-K but with soft weights
            top_indices = np.argsort(c)[::-1][:K]
            top_vals = c[top_indices]
            # Apply temperature to top-K values
            t_max = top_vals.max()
            exp_t = np.exp(beta * (top_vals - t_max))
            weights = exp_t / exp_t.sum()
            
            w_full = np.zeros(2 * n - 1)
            for i, k in enumerate(top_indices):
                w_full[int(k)] = weights[i]
            
            fsize = 1 << (len(w_full) + n - 2).bit_length()
            W = fft(w_full, fsize)
            A = fft(a, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta * 2.0 * (1.0 + 0.001 * it))
        step = max(step, 1e-7)
        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
                last_improve = it
                stagnation = 0

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Restart if stagnating
        if stagnation > 2000:
            y = project_simplex(y + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            t = 1.0
            stagnation = 0

    return best_local, best_score

def run_multiscale(a_init, n, S, max_time, rng, coarse_factor=4):
    """Multi-scale: optimize on coarse grid, then refine on fine grid."""
    start = time.monotonic()
    
    # Coarse grid
    nc = n // coarse_factor
    if nc < 64:
        nc = 64
    a_coarse = np.zeros(nc)
    for i in range(nc):
        lo = i * (n // nc)
        hi = lo + (n // nc)
        a_coarse[i] = a_init[lo:hi].mean()
    
    S_c = np.sqrt(2.0 * float(nc))
    a_coarse = a_coarse * (S_c / a_coarse.sum())
    
    seg_coarse = max_time * 0.3
    a_c, score_c = run_fista_adaptive(a_coarse, nc, S_c, seg_coarse, rng, 
                                         beta_init=5.0, beta_max=200.0, K_init=nc//2, K_min=1)
    
    # Upsample to fine grid
    a_upsampled = np.zeros(n)
    for i in range(nc):
        lo = i * (n // nc)
        hi = lo + (n // nc)
        a_upsampled[lo:hi] = a_c[i]
    # Blend with original
    a_blend = 0.5 * a_upsampled + 0.5 * a_init
    a_blend = safe_normalize(a_blend, n)
    
    # Fine optimization
    remaining = max_time - (time.monotonic() - start)
    a_f, score_f = run_fista_adaptive(a_blend, n, S, remaining, rng,
                                         beta_init=20.0, beta_max=400.0, K_init=n//4, K_min=3)
    
    return a_f, score_f

            best_score = c.max()
            best = a_opt.copy()

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.
Uses multi-strategy approach: parametric initialization, adaptive FISTA with
hybrid top-K/log-sum-exp gradients, multi-scale optimization, structured
perturbation search, and intensive polishing with plateau-aware descent.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_weighted(a, w_full):
    """Compute gradient given precomputed weights on full convolution lag indices."""
    n = len(a)
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def grad_lse(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c))."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    return compute_grad_weighted(a, w)


def grad_toppk_soft(a, c, K, beta):
    """Gradient using top-K lags with softmax weighting within top-K."""
    n = len(a)
    if K >= len(c):
        return grad_lse(a, c, beta)
    top_indices = np.argsort(c)[::-1][:K]
    top_vals = c[top_indices]
    t_max = top_vals.max()
    exp_t = np.exp(beta * (top_vals - t_max))
    total = exp_t.sum()
    if total < 1e-30:
        weights = np.ones(K) / K
    else:
        weights = exp_t / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    return compute_grad_weighted(a, w_full)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_parametric_seq(n, rng, style):
    """Create a parametric initial sequence with specific structure."""
    tt = np.linspace(0, 1, n, endpoint=False)
    if style == 0:
        # Modulated Gaussian bump
        sigma = rng.uniform(0.15, 0.35)
        eps = rng.uniform(0, 0.6)
        f = rng.uniform(2, 15)
        phi = rng.uniform(0, 2 * np.pi)
        seq = (1.0 + eps * np.sin(2 * np.pi * f * tt + phi)) * np.exp(-0.5 * ((tt - 0.5) / sigma) ** 2)
    elif style == 1:
        # Double bump
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        s1 = rng.uniform(0.08, 0.15)
        s2 = rng.uniform(0.08, 0.15)
        h1 = rng.uniform(0.5, 1.5)
        h2 = rng.uniform(0.5, 1.5)
        seq = h1 * np.exp(-0.5 * ((tt - c1) / s1) ** 2) + h2 * np.exp(-0.5 * ((tt - c2) / s2) ** 2) + 0.05
    elif style == 2:
        # Tapered flat-top with modulation
        taper = rng.uniform(0.1, 0.3)
        f = rng.uniform(3, 20)
        eps = rng.uniform(0, 0.4)
        window = np.ones(n)
        m = int(taper * n)
        for i in range(m):
            val = 0.5 * (1 - np.cos(np.pi * i / m))
            window[i] = val
            window[n - 1 - i] = val
        seq = window * (1.0 + eps * np.sin(2 * np.pi * f * tt))
    elif style == 3:
        # Multi-frequency (chirp-like amplitude modulation)
        f1 = rng.uniform(1, 5)
        f2 = rng.uniform(5, 15)
        a1 = rng.uniform(0, 0.3)
        a2 = rng.uniform(0, 0.2)

    else:
        # Smooth random (low-pass filtered random)
        raw = rng.random(n)
        # Simple moving average as low-pass
        k = max(3, n // 200)

        seq = np.convolve(raw, kernel, mode='same')

def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=500.0,
              K_init=None, K_min=1, restart_every=3000, noise_scale=0.005):
    """FISTA with adaptive top-K log-sum-exp and restarts."""
    if K_init is None:
        K_init = n // 4
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        K = max(K_min, K_init - it // 80)
        beta = min(beta_max, beta_init * (1.0 + 0.003 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute gradient
        grad = grad_toppk_soft(y, c, K, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 1.0 / (beta * 2.0 * (1.0 + 0.0008 * it))
        step = max(step, 1e-8)
        x_new = project_simplex(y - step * grad, S)

        # Periodic expensive check
        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
                no_improve = 0

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart on stagnation
        if no_improve > restart_every:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
            no_improve = 0

    return best_local, best_score


def run_epoch_descent(a_init, n, S, max_time, rng):
    """Epoch-based descent: identify peak lags, solve subproblem to reduce them."""
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time - 1:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Identify top peak lags
        n_peaks = min(8, 2 * n - 1)
        peak_lags = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[peak_lags]
        avg_val = c.mean()

        # If peaks are close to average, we're near optimal
        if peak_vals[0] - avg_val < 0.01:
            break

        # For each peak lag k, the gradient component at position i is a[(i-k) mod n]
        # (approximately, for the linear convolution)
        # We want to reduce a_i where it contributes to peak lags
        # Build a "pressure" vector: sum of a_j over all peak lags k where i and j are at distance k
        pressure = np.zeros(n)
        for lag in peak_lags:
            lag = int(lag)
            if lag == 0:
                pressure += a * (peak_vals[np.where(peak_lags == lag)[0][0]] if len(np.where(peak_lags == lag)[0]) > 0 else c[0])
            elif lag < n:
                # b[lag] = sum_i a[i] * a[i+lag] (approximately)
                # gradient w.r.t. a[i] is a[i+lag] for i in [0, n-lag-1]
                # and a[i-lag] for i in [lag, n-1]
                w = (c[int(lag)] - avg_val) / max(peak_vals[0] - avg_val, 1e-10)
                pressure[:n - lag] += w * a[lag:]
                pressure[lag:] += w * a[:n - lag]

        # Subtract pressure, renormalize
        alpha = 0.1 / max(1.0, it * 0.1)
        a_new = a - alpha * pressure
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s)
        a = a_new

    return best_local, best_score

        seg2 = rem * 0.5

        # High-beta FISTA
        a_p1, _ = run_fista(global_a, n, S, seg1, rng,
                             beta_init=50.0, beta_max=800.0,
                             K_init=n // 16, K_min=1,
                             restart_every=2000, noise_scale=0.003)
        a_p1 = safe_normalize(a_p1, n)
        c1 = fft_convolve_self(a_p1)
        if c1.max() < global_best:
            global_best = c1.max()
            global_a = a_p1.copy()
            print("  Polish1: %.6f" % global_best)

        # Very high beta, very low K (near-exact max)
        a_p2, _ = run_fista(global_a, n, S, seg2, rng,
                             beta_init=100.0, beta_max=1500.0,
                             K_init=n // 32, K_min=1,
                             restart_every=1500, noise_scale=0.002)
        a_p2 = safe_normalize(a_p2, n)
        c2 = fft_convolve_self(a_p2)
        if c2.max() < global_best:
            global_best = c2.max()
            global_a = a_p2.copy()
            print("  Polish2: %.6f" % global_best)

    # ============ Final verification ============
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Verify: %.6f" % ver if np.isfinite(ver) else "Verify: INVALID")
        if not np.isfinite(ver):
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("Returning n=%d, score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
