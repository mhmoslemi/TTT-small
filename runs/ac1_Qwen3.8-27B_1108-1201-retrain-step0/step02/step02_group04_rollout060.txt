"""
Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based iterative refinement (inspired by multiplicative weight updates)
combined with projected subgradient descent for polishing.
Key insight: optimal sequences are asymmetric (ramp-like), not uniform.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import next_fast_len, rfft, irfft


def fft_convolve(a, b=None):
    """Compute linear convolution of two 1-D arrays using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = next_fast_len(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_candidates = np.where(u * idx > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        # All elements are equal
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score_normalized(a):
    """Compute score for a normalized sequence (sum=1, a>=0)."""
    n = len(a)
    c = fft_convolve(a)

def fft_convolve(a, b=None):
    """Compute linear convolution of two 1-D arrays using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)

    fa = rfft(a, size)

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    mask = u * idx > (cssv - 1.0)
    if not np.any(mask):
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        if sc < best_score:
            best_score = sc
            best_seq = ramp.copy()
            print(f"  Ramp n={n}: score={sc:.8f}")

        # Exponential decay
        exp_seq = np.exp(-3.0 * np.linspace(0, 1, n))
        exp_seq = exp_seq / np.sum(exp_seq)
        sc = compute_score_normalized(exp_seq)
        if sc < best_score:
            best_score = sc
            best_seq = exp_seq.copy()
            print(f"  Exp n={n}: score={sc:.8f}")

        # Quadratic ramp
        q_ramp = np.linspace(1.0, 0.01, n) ** 2
        q_ramp = q_ramp / np.sum(q_ramp)
        sc = compute_score_normalized(q_ramp)
        if sc < best_score:
            best_score = sc
            best_seq = q_ramp.copy()
            print(f"  Q-ramp n={n}: score={sc:.8f}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # --- Phase 2: LP Optimization ---
    print("\n=== Phase 2: LP Optimization ===")

    # Try LP from multiple initializations and sizes
    configs = [
        (256, "ramp"),
        (512, "ramp"),
        (1024, "ramp"),
        (2048, "ramp"),
        (512, "exp"),
        (1024, "exp"),
        (256, "uniform"),
        (512, "uniform"),
        (1024, "uniform"),
    ]

    for n, shape in configs:
        time_left = deadline - time.monotonic()
        if time_left < 15:
            break

        # Build initialization
        if shape == "ramp":
            init = np.linspace(2.0, 0.05, n)
        elif shape == "exp":
            init = np.exp(-3.0 * np.linspace(0, 1, n))
        else:
            init = np.ones(n)
        init = init / np.sum(init)

        # If we have a good sequence of this length, also try from it
        if best_seq is not None and len(best_seq) == n:
            init = best_seq.copy()

        time_budget = min(180.0, time_left / 2.0)
        print(f"  LP n={n}, shape={shape}, budget={time_budget:.0f}s")
        opt_f, opt_score = try_lp_optimization(init, n, time_budget, verbose=(n <= 512))
        print(f"    Result: score={opt_score:.8f}")

        if opt_score < best_score:
            best_score = opt_score
            best_seq = opt_f.copy()
            print(f"    *** NEW BEST: {best_score:.8f} (n={n}) ***")

    # --- Phase 3: Further LP iterations on best ---
    if best_seq is not None:
        time_left = deadline - time.monotonic()
        if time_left > 60:
            print(f"\n=== Phase 3: Continued LP on best (n={len(best_seq)}), time={time_left:.0f}s ===")
            n_best = len(best_seq)
            # Try smaller step sizes for more convergence
            for t_try in [0.05, 0.03, 0.01]:
                time_left = deadline - time.monotonic()
                if time_left < 30:
                    break
                t0 = time.monotonic()
                opt_f, opt_score = optimize_lp(
                    best_seq, max_iter=300, t_init=t_try,
                    time_limit=min(time_left - 10, 120), t0=t0
                )
                if opt_score < best_score:
                    best_score = opt_score
                    best_seq = opt_f.copy()
                    print(f"    t={t_try}: NEW BEST {best_score:.8f}")
                else:
                    print(f"    t={t_try}: no improvement ({opt_score:.8f})")

    # --- Phase 4: Subgradient polishing ---
    if best_seq is not None:
        time_left = deadline - time.monotonic()
        if time_left > 30:
            print(f"\n=== Phase 4: Subgradient polish, time={time_left:.0f}s ===")
            t0 = time.monotonic()
            opt_a, opt_score = subgradient_polish(
                best_seq, max_iter=20000,
                time_limit=min(time_left - 10, 120), t0=t0
            )
            if opt_score < best_score:
                best_score = opt_score
                best_seq = opt_a.copy()
                print(f"  Subgradient: NEW BEST {best_score:.8f}")
            else:
                print(f"  Subgradient: no improvement ({opt_score:.8f})")

    # --- Phase 5: Try larger n with LP if time remains ---
    time_left = deadline - time.monotonic()
    if time_left > 60:
        print(f"\n=== Phase 5: Larger n attempts, time={time_left:.0f}s ===")
        for n in [4096, 8192]:
            time_left = deadline - time.monotonic()
            if time_left < 30:
                break
            # Use a ramp initialization
            init = np.linspace(2.0, 0.01, n)
            init = init / np.sum(init)
            time_budget = min(120.0, time_left - 10)
            print(f"  LP n={n}, budget={time_budget:.0f}s")
            opt_f, opt_score = try_lp_optimization(init, n, time_budget)
            print(f"    Result: score={opt_score:.8f}")
            if opt_score < best_score:
                best_score = opt_score
                best_seq = opt_f.copy()
                print(f"    *** NEW BEST: {best_score:.8f} (n={n}) ***")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(256)

    # Scale: use sum = sqrt(2*n) for the final answer (score is scale-invariant)
    n_final = len(best_seq)
    scale = np.sqrt(2.0 * n_final)
    final_scaled = best_seq * scale
    # Clip to [0, 1000]
    final_list = [float(max(0.0, min(1000.0, x))) for x in final_scaled]

    final_score = evaluate_sequence(final_list)
    print(f"\n=== FINAL: score={final_score:.8f}, n={n_final} ===")

    return final_list
