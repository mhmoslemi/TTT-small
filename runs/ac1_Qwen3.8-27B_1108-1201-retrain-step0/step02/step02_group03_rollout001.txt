"""
Refined peak-directed optimization for non-negative step functions minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses active-set peak tracking, subgradient
methods with adaptive soft-max temperature, symmetry-breaking initialization,
and iterative coarse-to-fine refinement.
"""

import numpy as np
import time


def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax normalization."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def grad_soft_max_conv(a, tau):
    """Compute soft-max(conv(a,a)) and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    w = e / s
    value = c_max + tau * np.log(s)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = (2.0 / tau) * corr[n - 1:2 * n - 1]
    return value, g, c, w


def get_peak_lags(c, n, top_k=5):
    """Get the indices of the top-k peaks in the convolution."""
    k = min(top_k, len(c))
    return np.argsort(c)[-k:]


def compute_peak_contributions(a, peak_lags):
    """For each peak lag, compute which elements contribute most."""
    n = len(a)
    contrib = np.zeros(n)
    for lag in peak_lags:
        i_min = max(0, lag - n + 1)
        i_max = min(n - 1, lag)
        for i in range(i_min, i_max + 1):
            j = lag - i
            contrib[i] += a[i] * a[j]
    return contrib


def gradient_optimize(a_init, time_limit, rng, tau_mode='adaptive'):
    """
    Adam gradient descent on softmax-parameterized sequence.
    Uses adaptive temperature based on peak concentration.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(a_init, 1e-10, None)
    a = a / np.sum(a) * S
    theta = np.log(a + 1e-15)

    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr0 = 2.0
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)
    best_a = a.copy()

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)

        a = softmax_normalize(theta) * S
        value, g_a, c, w = grad_soft_max_conv(a, tau)

        # Adaptive: if peaks are concentrated, use lower tau
        c_max = np.max(c)
        c_mean = np.mean(c)
        concentration = c_max / max(c_mean, 1e-10)
        # Higher concentration -> lower tau (sharper)
        base_tau = 30.0 / n
        if tau_mode == 'adaptive':
            tau = base_tau * max(0.1, 1.0 - 0.5 * elapsed_frac) * max(0.3, 2.0 - concentration / 1.6)
        else:
            tau = base_tau * (0.3) ** (elapsed_frac ** 0.7)

        # Recompute with final tau
        value, g_a, c, w = grad_soft_max_conv(a, tau)

        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 150 == 0:
            a_check = softmax_normalize(theta) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
                best_a = a_check.copy()
            if step % 750 == 0:
                print(f"          step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}")

    a_final = softmax_normalize(best_theta) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
    return a_final, best_score


def peak_directed_search(a_init, time_limit, rng):
    """
    Iterative peak-directed local search. Identifies peak lags, computes
    element contributions, and performs targeted mass redistribution.
    """
    n = len(a_init)
    current = np.clip(a_init, 0.0, None).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(current)
        peak_lags = get_peak_lags(conv, n, top_k=4)
        contrib = compute_peak_contributions(current, peak_lags)

        r = rng.random()
        if r < 0.4:
            # Reduce top contributors, increase bottom
            n_mod = max(3, n // 12)
            top_idx = np.argsort(contrib)[-n_mod:]
            bot_idx = np.argsort(contrib)[:n_mod]
            total_top = np.sum(current[top_idx])
            total_bot = np.sum(current[bot_idx])
            if total_top > 1e-10 and total_bot > 1e-10:
                alpha = rng.uniform(0.02, 0.06)
                transfer = alpha * total_top
                current[top_idx] *= (1.0 - alpha)
                current[bot_idx] *= (1.0 + transfer / total_bot)
        elif r < 0.6:
            # Anti-symmetry: reduce where a[i]*a[n-1-i] is large
            prod = current * current[::-1]
            n_mod = max(3, n // 10)
            top_idx = np.argsort(prod)[-n_mod:]
            bot_idx = np.argsort(prod)[:n_mod]
            total_top = np.sum(current[top_idx])
            total_bot = np.sum(current[bot_idx])
            if total_top > 1e-10 and total_bot > 1e-10:
                alpha = rng.uniform(0.01, 0.04)
                transfer = alpha * total_top
                current[top_idx] *= (1.0 - alpha)
                current[bot_idx] *= (1.0 + transfer / total_bot)
        elif r < 0.75:
            # Block swap: move mass from high-contrib blocks to low-contrib
            n_blocks = rng.integers(2, 5)
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 20))))
                bs = int(rng.integers(0, n - bw))
                factor = np.exp(rng.normal(-0.05, 0.02))
                current[bs:bs + bw] *= factor
                # Redistribute to a random other block
                bs2 = int(rng.integers(0, n - bw))
                if bs2 != bs:
                    current[bs2:bs2 + bw] *= 1.0 / factor
        elif r < 0.9:
            # Targeted single-element reduction at peak contributors
            n_reduce = rng.integers(3, 10)
            idx = np.argsort(contrib)[-n_reduce:]
            for i in idx:
                current[i] *= np.exp(rng.normal(-0.04, 0.02))
            # Compensate
            idx_inc = np.argsort(contrib)[:n_reduce]
            total_red = 1.0 - np.prod([np.exp(rng.normal(-0.04, 0.02)) for _ in range(n_reduce)])
            for i in idx_inc:
                current[i] *= (1.0 + total_red / 2.0)
        else:
            # Global small perturbation
            current *= np.exp(rng.normal(0, 0.01, size=n))

        current = np.clip(current, 0.0, None)
        s = np.sum(current)
        if s < 1e-10:
            current = best.copy() * np.exp(rng.normal(0, 0.01, size=n))
            current = np.clip(current, 0.0, None)
            current = current / np.sum(current) * np.sum(best)
            continue
        target_sum = np.sum(best)
        current = current / s * target_sum

        new_score = local_score(current)

        if new_score < current_score:
            current_score = new_score
            stall_count = 0
            if new_score < best_score:
                best_score = new_score
                best = current.copy()
        else:
            stall_count += 1
            # Sometimes accept worse to escape local min
            if stall_count > 50 and rng.random() < 0.1:
                current = current
                current_score = new_score
                stall_count = 0
            else:
                stall_count = min(stall_count, 100)

    return best, best_score


def anti_sym_init(n, rng, alpha=0.3):
    """Initialize with anti-symmetric structure to reduce center peak."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    # High at edges, low in middle, with oscillation
    base = 1.0 + alpha * np.cos(np.pi * t / n)
    # Add anti-symmetric component
    anti = 0.2 * np.sin(np.pi * (t - center) / n * 4)
    a = base + anti + 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def fft_phase_init(n, rng, n_iter=80):
    """Gerchberg-Saxton style flat-spectrum initialization."""
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def upsample(a, new_n, rng):
    """Upsample sequence via interpolation + noise."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)
