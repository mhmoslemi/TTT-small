If the maximum of conv(a,a) is at lag k*, we can try to reduce it by adjusting a. The derivative of conv(a,a)[k] with respect to a[j] is:

- Find the lag k* where conv is maximized
- Find the j that most contributes to conv[k*] (i.e., where a[k*-j] is largest)

Let me code this up:

```python
def peak_shave_move(a, c, M, n, S, step_size):
    """Try to reduce the maximum of c by moving mass from peak-contributing entries."""
    k_star = np.argmax(c)  # lag where max occurs
    
    # Find which j contributes most to c[k_star]
    # c[k_star] = sum_j a[j] * a[k_star - j] for valid j
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    
    contributions = np.zeros(n)
    for j in range(lo, hi + 1):
        contributions[j] = a[j] * a[k_star - j]
    
    # The j with the largest contribution
    j_peak = lo + np.argmax(contributions[lo:hi+1])
    
    # Find a j' to move mass to (one that doesn't contribute to peaks)
    # Heuristic: move to a position where a[j'] is small and doesn't create new peaks
    # Simple: move to the position with minimum a
    j_target = np.argmin(a)
    
    # Move step_size worth of mass
    delta = min(step_size, a[j_peak] * 0.1)
    a[j_peak] -= delta
    a[j_target] += delta
    
    # Renormalize
    a = a * (S / a.sum())
    return a
```

```python
def backtracking_line_search(theta, grad, n, T, S, alpha=0.1, beta=0.5, max_iter=20):
    """Backtracking line search for the soft-max objective."""
    obj0, _ = softmax_objective_and_grad(theta, n, T, S)
    step = 1.0
    for _ in range(max_iter):
        theta_new = theta - step * grad
        obj_new, _ = softmax_objective_and_grad(theta_new, n, T, S)
        if obj_new <= obj0 - alpha * step * np.dot(grad, grad):
            return theta_new, step
        step *= beta
    return theta - step * grad, step
```

Let me now think about the **peak shaving** move more carefully:

The idea is: if the max of conv(a,a) is at lag k*, we want to reduce it. The value at lag k* is:
c[k*] = sum_j a[j] * a[k*-j] (for valid j)

To reduce c[k*], we can:
1. Decrease a[j] for j where a[k*-j] is large
2. Increase a[j'] for j' where j' + j'' ≠ k* for any large a[j'']

But we need to maintain sum(a) = S, so we transfer mass from one position to another.

A simple heuristic:
- Find k* = argmax(c)
- Find j* = argmax_j a[j] * a[k*-j] (the j that contributes most to c[k*])
- Find j' = argmin_j a[j] (the position with the least mass)
- Transfer a small amount from j* to j'
- Check if the max decreased

This is essentially a **2-opt** type move.

```python
def normalize_a(a, S):
    """Normalize a to have sum S, clip to non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def a_to_theta(a, S):
    """Convert a to log-parameters."""
    a = np.maximum(a, 1e-15)
    return np.log(a / a.sum())


def theta_to_a(theta, S):
    """Convert log-parameters to a."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total

"""
```

```python
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    n, m = len(a), len(b)
    out_len = n + m - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:out_len]


def fft_conv_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def objective_of_a(a):
    """Compute the true objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_conv_self(a)
    return float(2.0 * n * c.max() / (s * s))


def softmax_obj_grad_a(a, T):
    """
    Compute soft-max of self-convolution and its gradient w.r.t. a.
    a: non-negative sequence of length n
    T: temperature parameter
    Returns: (soft_max_value, grad_a of length n)
    """
    n = len(a)
    c = fft_conv_self(a)
    c_max = c.max()
    if c_max < 1e-15:
        return 0.0, np.zeros(n)

    shifted = T * (c - c_max)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    log_sum_exp = np.log(sum_exp)
    obj = c_max + log_sum_exp / T

    # Soft-max weight vector (length 2n-1)
    w = exp_shifted / sum_exp

    # grad_a[m] = 2 * sum_j a[j] * w[j+m] for j=0..n-1
    # = 2 * conv(a, w_rev)[2n-2-m] for m=0..n-1
    w_rev = w[::-1]
    conv_aw = fft_conv(a, w_rev)  # length 3n-2
    grad_a = 2.0 * conv_aw[n - 1: 2 * n - 1][::-1]

    return obj, grad_a


def theta_to_a(theta, S):
    """Convert log-parameters to sequence: a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def a_to_theta(a, S):
    """Convert sequence to log-parameters."""
    a = np.maximum(a, 1e-15)
    return np.log(a / a.sum())


def softmax_obj_grad_theta(theta, T, S):
    """
    Soft-max objective and gradient w.r.t. log-parameters theta.
    a = S * softmax(theta)
    """
    a = theta_to_a(theta, S)
    obj, grad_a = softmax_obj_grad_a(a, T)

    # Chain rule: dL/dtheta_j = grad_a_j * a_j - (a_j/S) * dot(grad_a, a)
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot

    return obj, grad_theta

def coordinate_descent_grad(a, n, S, time_budget, max_sweeps=100):
    """
    Gradient-based coordinate descent polish.
    Uses the gradient to determine which coordinate to adjust and in which direction.
    """
    a = a.copy().astype(np.float64)
    a = a * (S / a.sum())
    start_time = time.monotonic()

    c = fft_conv_self(a)
    best_obj = c.max()
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() - start_time > time_budget:
            break

        # Compute soft-max gradient at high temperature (approximates max gradient)
        T = 500.0
        _, grad_a = softmax_obj_grad_a(a, T)

        # Find coordinate with largest positive gradient (most to decrease)
        i_dec = int(np.argmax(grad_a))
        # Find coordinate with smallest gradient (most to increase)
        i_inc = int(np.argmin(grad_a))

        if i_dec == i_inc:
            break

        if grad_a[i_dec] <= grad_a[i_inc]:
            break

        # Binary search for optimal transfer amount
        orig_dec = a[i_dec]
        orig_inc = a[i_inc]
        lo, hi = 0.0, orig_dec * 0.5
        best_delta = 0.0
        best_local_obj = best_obj

        for _ in range(15):
            mid = (lo + hi) / 2.0
            a[i_dec] = orig_dec - mid
            a[i_inc] = orig_inc + mid
            s = a.sum()
            if s < 1e-10:
                a[i_dec] = orig_dec
                a[i_inc] = orig_inc
                break
            a_norm = a * (S / s)
            c_check = fft_conv_self(a_norm)
            obj_check = c_check.max()
            if obj_check < best_local_obj:
                best_local_obj = obj_check
                best_delta = mid
            if obj_check < best_obj:
                lo = mid
            else:
                if obj_check < best_local_obj * 1.001:
                    lo = mid
                else:
                    hi = mid

        if best_delta > 1e-10:
            a[i_dec] = orig_dec - best_delta
            a[i_inc] = orig_inc + best_delta
            a = a * (S / a.sum())
            c = fft_conv_self(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()
        else:
            a[i_dec] = orig_dec
            a[i_inc] = orig_inc
            # Try random pair
            mask = np.ones(n, dtype=bool)
            mask[i_dec] = False
            if mask.sum() < 2:
                break
            candidates = np.where(mask)[0]
            i2 = candidates[np.random.randint(len(candidates))]
            delta = 0.01 * S / n
            if a[i_dec] > delta:
                a[i_dec] -= delta
                a[i2] += delta
                a = a * (S / a.sum())
                c = fft_conv_self(a)
                if c.max() < best_obj:
                    best_obj = c.max()
                    best_a = a.copy()
                else:
                    a[i_dec] += delta
                    a[i2] -= delta
                    a = a * (S / a.sum())

    return best_a, best_obj


def get_init_exponential(n, S, alpha, rng):
    """Exponential decay initialization (non-symmetric)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i / (n - 1.0))
    a += 0.01 * rng.random(n)
    return a * (S / a.sum())


def get_init_ramp(n, S, beta, rng):
    """Linear ramp initialization (non-symmetric)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + beta * i / (n - 1.0)
    a += 0.01 * rng.random(n)
    return a * (S / a.sum())


def get_init_sine(n, S, freq, phase, amp, rng):
    """Sine-modulated initialization (non-symmetric due to phase)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.sin(2.0 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.01)
    a += 0.01 * rng.random(n)
    return a * (S / a.sum())


def get_init_random(n, S, rng):
    """Random initialization."""
    a = rng.random(n) + 0.01
    return a * (S / a.sum())


def get_init_multiply(n, S, rng):
    """Product of two decays (creates asymmetric bump)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-2.0 * i / n) * np.exp(-8.0 * (1.0 - i / n))
    a += 0.01 * rng.random(n)
    return a * (S / a.sum())


def get_init_power(n, S, p, rng):
    """Power law initialization."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a += 0.01 * rng.random(n)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns the best sequence found."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    global_best = np.inf
    global_a = None

    # Try to load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.8f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("No given sequence: %s" % str(e))

    # === Phase 1: Quick exploration (25% of time) ===
    explore_budget = 0.25 * (deadline - time.monotonic())
    print("Phase 1: Exploration (budget=%.0fs)" % explore_budget)

    # Define initializations to try
    n_candidates = [2048, 4096, 8192]
    if given_a is not None:
        n_candidates = [len(given_a)] + n_candidates

    init_fns = [
        ('exp_1', lambda n, S: get_init_exponential(n, S, 1.0, rng)),
        ('exp_3', lambda n, S: get_init_exponential(n, S, 3.0, rng)),
        ('exp_5', lambda n, S: get_init_exponential(n, S, 5.0, rng)),
        ('ramp_1', lambda n, S: get_init_ramp(n, S, 1.0, rng)),
        ('ramp_2', lambda n, S: get_init_ramp(n, S, 2.0, rng)),
        ('sine_1', lambda n, S: get_init_sine(n, S, 1.0, 0.5, 0.3, rng)),
        ('sine_3', lambda n, S: get_init_sine(n, S, 3.0, 1.0, 0.2, rng)),
        ('multiply', lambda n, S: get_init_multiply(n, S, rng)),
        ('power_05', lambda n, S: get_init_power(n, S, 0.5, rng)),
        ('power_neg', lambda n, S: get_init_power(n, S, -0.3, rng)),
        ('random', lambda n, S: get_init_random(n, S, rng)),
    ]

    # If we have a given sequence, start with it
    if given_a is not None:
        n_g = len(given_a)
        S_g = np.sqrt(2.0 * n_g)
        a_g = given_a.copy() * (S_g / given_a.sum())
        theta_g = a_to_theta(a_g, S_g)
        t0 = time.monotonic()
        theta_b, a_b, obj_b = run_momentum_backtrack(
            theta_g, n_g, S_g, min(5.0, explore_budget * 0.3),
            T_start=10.0, T_end=100.0, momentum=0.9
        )
        if obj_b < global_best:
            global_best = obj_b
            global_a = a_b.copy()
            print("  Given + short opt: obj=%.8f (%.1fs)" % (obj_b, time.monotonic() - t0))

    # Quick exploration
    per_start_time = max(1.5, explore_budget / (len(n_candidates) * len(init_fns) * 0.7))

    for n in n_candidates:
        if time.monotonic() > deadline - 60.0:
            break
        S = np.sqrt(2.0 * n)

        for init_name, init_fn in init_fns:
            if time.monotonic() > deadline - 60.0:
                break
            if time.monotonic() > deadline - explore_budget * 0.15:
                break

            a_init = init_fn(n, S)
            theta_init = a_to_theta(a_init, S)

            t0 = time.monotonic()
            t_actual = min(per_start_time, deadline - time.monotonic() - 60.0)
            if t_actual < 0.5:
                break
            theta_b, a_b, obj_b = run_momentum_backtrack(
                theta_init, n, S, t_actual,
                T_start=5.0, T_end=100.0, momentum=0.9,
                perturb_every=200
            )
            elapsed = time.monotonic() - t0

            if obj_b < global_best:
                global_best = obj_b
                global_a = a_b.copy()
                print("  Phase1: n=%d %s: obj=%.8f (%.1fs)" % (n, init_name, obj_b, elapsed))

    print("After Phase 1: best=%.8f" % global_best)

    # === Phase 2: Main optimization (45% of time) ===
    exploit_budget = 0.45 * (deadline - time.monotonic())
    print("Phase 2: Main optimization (budget=%.0fs)" % exploit_budget)

    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = global_a.copy()
        a = a * (S / a.sum())
        theta = a_to_theta(a, S)

        # Multiple rounds with different temperatures
        temp_schedule = [
            (10.0, 50.0),
            (20.0, 100.0),
            (50.0, 200.0),
            (100.0, 500.0),
            (200.0, 1000.0),
        ]

        # Allocate time proportionally
        weights = [0.25, 0.25, 0.2, 0.15, 0.15]
        for idx, (T_lo, T_hi) in enumerate(temp_schedule):
            if time.monotonic() > deadline - 30.0:
                break
            t_avail = exploit_budget * weights[idx]
            t_actual = min(t_avail, deadline - time.monotonic() - 30.0)
            if t_actual < 1.0:
                break

            t0 = time.monotonic()
            theta_b, a_b, obj_b = run_momentum_backtrack(
                theta, n, S, t_actual,
                T_start=T_lo, T_end=T_hi,
                momentum=0.85 + 0.05 * idx,
                perturb_every=400,
                perturb_scale=0.03 + 0.02 * idx
            )
            elapsed = time.monotonic() - t0

            if obj_b < global_best:
                global_best = obj_b
                global_a = a_b.copy()
                print("  Phase2 T=[%d,%d]: obj=%.8f (%.1fs)" % (T_lo, T_hi, obj_b, elapsed))

            # Update theta from best
            a_clean = np.maximum(a_b, 1e-10)
            a_clean = a_clean * (S / a_clean.sum())
            theta = a_to_theta(a_clean, S)

    # === Phase 3: L-BFGS refinement (15% of time) ===
    lbfgs_budget = 0.15 * (deadline - time.monotonic())
    print("Phase 3: L-BFGS (budget=%.0fs)" % lbfgs_budget)

    if global_a is not None and lbfgs_budget > 3.0:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = global_a.copy()
        a = a * (S / a.sum())
        theta = a_to_theta(a, S)

        lbfgs_temps = [50, 100, 200, 500]
        per_lbfgs_time = lbfgs_budget / len(lbfgs_temps)

        for T in lbfgs_temps:
            if time.monotonic() > deadline - 20.0:
                break
            t_actual = min(per_lbfgs_time, deadline - time.monotonic() - 20.0)
            if t_actual < 1.0:
                break

            t0 = time.monotonic()
            a_b, obj_b = run_lbfgs(theta, n, S, t_actual, T=T, max_iter=200)
            elapsed = time.monotonic() - t0

            if obj_b < global_best:
                global_best = obj_b
                global_a = a_b.copy()
                print("  L-BFGS T=%d: obj=%.8f (%.1fs)" % (T, obj_b, elapsed))

            # Update theta
            if a_b is not None:
                a_clean = np.maximum(a_b, 1e-10)
                a_clean = a_clean * (S / a_clean.sum())
                theta = a_to_theta(a_clean, S)

    # === Phase 4: Coordinate descent polish (15% of time) ===
    cd_budget = 0.15 * (deadline - time.monotonic())
    print("Phase 4: Coord descent (budget=%.0fs)" % cd_budget)

    if global_a is not None and cd_budget > 2.0:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = global_a.copy()
        a = a * (S / a.sum())

        # Multiple CD rounds with high-T momentum in between
        for round_i in range(3):
            if time.monotonic() > deadline - 10.0:
                break
            t_cd = cd_budget / 3.0 * 0.6
            t_cd = min(t_cd, deadline - time.monotonic() - 10.0)
            if t_cd < 1.0:
                break

            t0 = time.monotonic()
            a_b, obj_b = coordinate_descent_grad(a, n, S, t_cd, max_sweeps=500)
            elapsed = time.monotonic() - t0
            if obj_b < global_best:
                global_best = obj_b
                global_a = a_b.copy()
                a = a_b.copy()
                print("  CD round %d: obj=%.8f (%.1fs)" % (round_i, obj_b, elapsed))

            # Brief momentum optimization between CD rounds
            if time.monotonic() < deadline - 10.0:
                t_mom = min(2.0, (deadline - time.monotonic() - cd_budget / 3.0 * 0.4))
                if t_mom > 0.5:
                    a = a * (S / a.sum())
                    theta = a_to_theta(a, S)
                    theta_b, a_b, obj_b = run_momentum_backtrack(
                        theta, n, S, t_mom,
                        T_start=500.0, T_end=1000.0,
                        momentum=0.8
                    )
                    if obj_b < global_best:
                        global_best = obj_b
                        global_a = a_b.copy()
                        a = a_b.copy()
                        print("  Post-CD momentum: obj=%.8f" % obj_b)

    # === Final: Try a different n if it might help ===
    remaining = deadline - time.monotonic() - 5.0
    if remaining > 30.0 and global_a is not None:
        # Try with a slightly different n by truncating or padding
        n_cur = len(global_a)
        for n_try in [n_cur * 2, n_cur // 2]:
            if n_try < 128 or n_try > 32768:
                continue
            if time.monotonic() > deadline - 10.0:
                break
            S_try = np.sqrt(2.0 * n_try)
            # Upsample or downsample
            if n_try > n_cur:
                # Interpolate
                x_old = np.linspace(0, 1, n_cur)
                x_new = np.linspace(0, 1, n_try)
                a_try = np.interp(x_new, x_old, global_a)
            else:
                # Decimate
                indices = np.linspace(0, n_cur - 1, n_try).astype(int)
                a_try = global_a[indices]
            a_try = np.maximum(a_try, 0.0)
            a_try = a_try * (S_try / a_try.sum())

            t0 = time.monotonic()
            theta_try = a_to_theta(a_try, S_try)
            theta_b, a_b, obj_b = run_momentum_backtrack(
                theta_try, n_try, S_try, min(20.0, remaining * 0.3),
                T_start=50.0, T_end=300.0
            )
            if obj_b < global_best:
                global_best = obj_b
                global_a = a_b.copy()
                print("  Size change n=%d: obj=%.8f" % (n_try, obj_b))
            remaining = deadline - time.monotonic() - 5.0

    # === Final verification ===
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_list = global_a.tolist()
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print("FINAL: n=%d, tracked=%.8f, verified=%.8f" % (n_final, global_best,
          ver if np.isfinite(ver) else float('inf')))

    if not np.isfinite(ver):
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
        print("Fallback: obj=%.6f" % ver)

    return final_list
```
