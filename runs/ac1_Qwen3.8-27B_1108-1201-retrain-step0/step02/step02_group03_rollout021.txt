Multi-phase optimization for minimizing peak convolution of non-negative sequences.
Uses FFT-based initialization, gradient descent with soft-min-max, and targeted
local search with peak-directed moves. Key improvements: larger n, log-parameterization,
better move generation in local search, and adaptive time allocation.

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def soft_max_conv(a, tau):
    """Compute soft-max(conv(a,a), tau) and gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    w = e / np.sum(e)
    value = c_max + tau * np.log(np.sum(e))
    
    # Gradient w.r.t. a: g[i] = 2 * sum_k w[k] * a[k-i]
    # This is 2 * cross_correlation(w, a)[i]
    # Using FFT:
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(W * np.conj(A), size)
    g = 2.0 * corr[0:n]
    
    return value, g, c


def fft_phase_init(n, rng, n_iter=80):
    """Gerchberg-Saxton style initialization."""
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    
    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.irfft(phase, N_fft)
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def chirp_init(n, rng, alpha=0.5):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def optimize_gradient(n, a_init, time_limit, rng, tau_start_mult=30.0, tau_end_mult=0.2):
    """
    Adam gradient descent on log-parameterized sequence with annealed soft-max.
    """
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S
    theta = np.log(a / S + 1e-12)
    
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr0 = 3.0
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)
    tau_start = tau_start_mult / np.sqrt(n)
    tau_end = tau_end_mult / np.sqrt(n)
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)
        
        a = np.exp(theta - np.max(theta))
        a = a / np.sum(a) * S
        
        value, g_a, c = soft_max_conv(a, tau)
        
        # Gradient w.r.t. theta with normalization constraint
        # a = exp(theta) * S / sum(exp(theta))
        # da/dtheta[i] = a[i] * (1 - a[i]/S)
        # g_theta = sum_i g_a[i] * da/dtheta[i] ... no
        # Actually: d(value)/d(theta[i]) = sum_j g_a[j] * da[j]/d(theta[i])
        # da[j]/d(theta[i]) = a[j]*delta(j,i) - a[j]*a[i]/S = a[j]*(delta(j,i) - a[i]/S)
        # d(value)/d(theta[i]) = g_a[i]*a[i] - a[i]/S * sum_j g_a[j]*a[j]
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)
        
        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        if step % 200 == 0:
            a_check = np.exp(theta - np.max(theta))
            a_check = a_check / np.sum(a_check) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
            if step % 1000 == 0:
                print(f"        step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")
    
    a_final = np.exp(best_theta - np.max(best_theta))
    a_final = a_final / np.sum(a_final) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
        a_final = a_final
    return a_final, best_score


def local_search(a_init, time_limit, rng, top_k=10):
    """
    Simulated annealing with diverse move types.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9997
    iter_count = 0
    
    # Precompute for speed
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        iter_count += 1
        
        if r < 0.30:
            # Peak-directed: identify top contributing pairs and reduce
            conv = fft_convolve(current, current)
            n_top = min(top_k, len(conv))
            top_lags = np.argsort(conv)[-n_top:]
            
            # Compute contribution of each element to top lags
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            
            n_reduce = max(2, n // 15)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.015, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.015, size=len(bot_idx)))
            
        elif r < 0.45:
            # Block swap: reduce one block, increase adjacent
            width = max(1, int(rng.integers(1, max(2, n // 30))))
            start = int(rng.integers(0, n - 2 * width))
            candidate[start:start + width] *= np.exp(rng.normal(-0.07, 0.02, size=width))
            candidate[start + width:start + 2 * width] *= np.exp(rng.normal(0.07, 0.02, size=width))
            
        elif r < 0.60:
            # Multi-block random perturbation
            n_blocks = int(rng.integers(2, 6))
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 20))))
                bs = int(rng.integers(0, n - bw))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0.0, 0.04, size=bw))
                
        elif r < 0.75:
            # Partial symmetrization + noise
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.1 * (sym - current)
            candidate += 0.005 * rng.standard_normal(n)
            
        elif r < 0.85:
            # Sparsify/invert sparse regions
            n_sparse = max(1, n // 4)
            sparse_idx = rng.choice(n, size=n_sparse, replace=False)
            candidate[sparse_idx] *= np.exp(rng.normal(-0.2, 0.1, size=n_sparse))
            
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))
        
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current
        else:
            candidate = candidate / s * np.sum(current)
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 5:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score
        
        temp *= cooling
    
    return best, best_score


def upsample(a, new_n, rng):
    """Upsample a sequence to a longer length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)
