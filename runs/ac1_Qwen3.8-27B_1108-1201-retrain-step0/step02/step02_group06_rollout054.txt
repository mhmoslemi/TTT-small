def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def autocorr(p):
    """Autocorrelation of probability vector p."""
    return fft_convolve(p, p)


def score_from_p(p):
    """Score = 2n * max(autocorr(p))."""
    n = len(p)
    c = autocorr(p)
    return 2.0 * n * np.max(c)


def score_from_a(a):
    """Score = 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)

def spectral_init(n, rng, n_iter=100):

        # Reduce peak: push mass away from high-contribution indices
        # The peak is at some lag k*. The contribution to c[k*] from p[i] is p[k*-i].
        # We want to reduce p at indices that contribute to the peak.
        peak_k = np.argmax(c)

        # c[peak_k] = sum_i p[i] * p[peak_k - i]
        # d/dc dp[j] = sum_i (delta_{i,j} * p[peak_k-i] + delta_{peak_k-i,j} * p[i])
        # = p[peak_k - j] + p[peak_k - j] = 2*p[peak_k-j] (if valid)

def lp_optimize(p_init, time_budget, rng):
    """
    Main LP-based optimization loop.
    p is a probability vector (sum = 1).
    """
    p = p_init.copy()
    p = np.clip(p, 0, None)
    p = p / p.sum()
    n = len(p)
    
    c = autocorr(p)
    M = np.max(c)
    best_p = p.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    stagnant = 0
    step_size = 1.0
    
    it = 0
    while time.monotonic() - t0 < time_budget:
        # Find LP direction
        b = lp_direction(p, M, time_limit=30.0)
        if b is None:
            break
        
        # Check if b is meaningfully different from p
        b_norm = b / b.sum() if b.sum() > 1e-15 else p.copy()
        
        # Line search for step size
        best_t = 0
        best_new_M = M
        for t in [1.0, 0.75, 0.5, 0.3, 0.15, 0.05, 0.01]:
            p_candidate = (1 - t) * p + t * b_norm
            p_candidate = np.clip(p_candidate, 0, None)
            s = p_candidate.sum()
            if s < 1e-15:
                continue
            p_candidate = p_candidate / s
            c_new = autocorr(p_candidate)
            new_M = np.max(c_new)
            if new_M < best_new_M:
                best_new_M = new_M
                best_t = t
        
        if best_t == 0:
            stagnant += 1
            if stagnant > 20:
                # Perturb and try again
                p += 0.01 * rng.standard_normal(n)
                p = np.clip(p, 0, None)
                p = p / p.sum()
                c = autocorr(p)
                M = np.max(c)
                stagnant = 0
            continue
        
        p = p / p.sum() if p is not None else p
        # Apply best step
        p = (1 - best_t) * p + best_t * b_norm
        p = np.clip(p, 0, None)
        p = p / p.sum()
        
        c = autocorr(p)
        new_M = np.max(c)
        
        if new_M < M:
            M = new_M
            stagnant = 0
            if 2.0 * n * M < best_score:
                best_score = 2.0 * n * M
                best_p = p.copy()
                if it % 20 == 0:

            # Reduce step size
            step_size *= 0.5

def adam_optimize(p_init, time_budget, rng):
    """
    Adam optimization on log-parameterization.
    Minimizes max of autocorrelation.
    """
    n = len(p_init)
    theta = np.log(p_init + 1e-15)
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.5
    
    t0 = time.monotonic()
    best_p = p_init.copy()
    best_score = 2.0 * n * np.max(autocorr(p_init))
    
    tau_start = 50.0 / n
    tau_end = 0.5 / n
    
    while time.monotonic() - t0 < time_budget:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_budget)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)
        
        p = softmax(theta)
        c = autocorr(p)
        max_c = np.max(c)
        
        # Smoothed max and its gradient
        shifted = (c - max_c) / tau
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        
        # Gradient of max_c w.r.t. p: approx 2 * (p * w) [cross-correlation]
        # c(k) = sum_i p[i] * p[k-i], so dc(k)/dp[j] = p[k-j] + p[k-j] = 2*p[k-j]
        # d max_c / dp[j] = sum_k w[k] * 2 * p[k-j] (with proper bounds)
        # This is 2 * (w * p)(j) reversed... actually it's the cross-correlation
        # Let me compute it via FFT
        size = 2 * n
        fp = np.fft.rfft(p, size)
        fw = np.fft.rfft(w, size)
        # gradient[j] = 2 * sum_k w[k] * p[k-j] where k-j in [0,n-1]
        # = 2 * (w convolved with p_reversed)[something]
        # Actually: sum_k w[k] * p[k-j] for valid k,j
        # Let q[k] = w[k] for k in [0, 2n-2], 0 otherwise
        # Then sum_k w[k] * p[k-j] = sum_k q[k] * p[k-j] = (q * p_rev)(n-1-j) ... 
        # This is getting complicated. Let me just use the FFT approach:
        # The gradient of c(k) w.r.t. p[j] is 2*p[k-j] (when 0 <= k-j < n)
        # So the gradient of the smoothed max is:
        # grad[j] = sum_k w[k] * 2 * p[k-j] (when 0 <= k-j < n)
        #         = 2 * sum_i w[i+j] * p[i] (when i+j < 2n-1 and i < n)
        # This is 2 * cross_correlation(w, p) evaluated appropriately
        
        # Simpler: use the fact that grad = 2 * fft_corr(p, w)
        # where fft_corr gives the cross-correlation
        # cross_corr[j] = sum_k p[k] * w[k-j] (circular)
        # We want: sum_k w[k] * p[k-j] = cross_corr of w and p at lag j
        # = circular cross-correlation
        
        # Let me just compute it directly for clarity (O(n^2) is too slow for large n)
        # Use FFT: 
        # grad[j] = 2 * sum_{k} w[k] * p[k-j] where both indices valid
        # = 2 * (w ⊛ p)[j] (cross-correlation, not convolution)
        # In FFT: cross_corr(w, p) = ifft(conj(FFT(w)) * FFT(p))
        
        corr = np.fft.irfft(np.conj(fw) * fp, size)[:n]
        grad_p = 2.0 * corr
        
        # Gradient w.r.t. theta: p * (grad_p - dot(p, grad_p))
        inner = np.dot(p, grad_p)
        grad_theta = p * (grad_p - inner)
        
        step += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * (grad_theta * grad_theta)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        
        lr_eff = lr / np.sqrt(1 + 0.001 * step)
        theta = theta - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        
        if step % 100 == 0:
            p_check = softmax(theta)
            sc = 2.0 * n * np.max(autocorr(p_check))
            if sc < best_score:
                best_score = sc
                best_p = p_check.copy()
                if step % 500 == 0:
                    print(f"    Adam step={step}, score={best_score:.6f}, tau={tau:.2e}")

                continue

    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def normalize_p(p):
    p = np.clip(p, 0, None)
    s = p.sum()
    if s < 1e-15:
        return np.ones(len(p)) / len(p)
    return p / s


def spectral_init(n, rng, n_iter=80):
    p = 1.0 + 0.3 * rng.standard_normal(n)
    p = np.clip(p, 0.01, None)
    p = p / p.sum()
    for it in range(n_iter):
        c = autocorr(p)
        alpha = 0.05 / (1.0 + 0.02 * it)
        p = p - alpha * c
        p = normalize_p(p)
    return p


def structured_init(n, rng, style):
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if style == 'linear':
        p = 1.0 + 0.5 * t
    elif style == 'quadratic':
        p = 1.0 + 0.8 * t ** 2
    elif style == 'bathtub':
        p = 0.5 + 1.5 * t ** 2
    elif style == 'hill':
        p = 0.5 + 1.5 * np.sin(np.pi * t)
    elif style == 'exponential':
        p = np.exp(1.5 * (t - 0.5))
    elif style == 'sawtooth':
        p = 1.0 + 0.4 * np.sin(4 * np.pi * t)
    elif style == 'two_peak':
        p = 1.0 + 1.0 * np.exp(-((t - 0.2) ** 2) / 0.02) + 1.0 * np.exp(-((t - 0.8) ** 2) / 0.02)
    elif style == 'inverse_bathtub':
        p = 1.0 + 1.0 * (1 - t) ** 2
    else:
        p = 1.0 + 0.3 * rng.standard_normal(n)
    p += 0.01 * rng.standard_normal(n)
    p = np.clip(p, 0.001, None)
    return p / p.sum()


def build_lp_matrix(p, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = p[:k_end - k_start]
    return A


def solve_lp_direction(p, M, time_limit=20.0):
    n = len(p)
    A = build_lp_matrix(p, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    max_p = np.max(p)
    upper = M / (max_p + 1e-20)
    bounds = [(0.0, upper)] * n
    t_start = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None
    if time.monotonic() - t_start > time_limit:
        return None
    if res.success:
        return res.x
    return None


def lp_optimize(p_init, time_budget, rng, verbose=True):
    p = p_init.copy()
    p = normalize_p(p)
    n = len(p)
    c = autocorr(p)
    M = np.max(c)
    best_p = p.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    stagnant = 0
    it = 0
    while time.monotonic() - t0 < time_budget:
        b = solve_lp_direction(p, M, time_limit=min(30.0, time_budget - (time.monotonic() - t0)))
        if b is None:
            break
        if b.sum() < 1e-15:
            break
        b_norm = b / b.sum()
        cosine_sim = np.dot(p, b_norm) / (np.linalg.norm(p) * np.linalg.norm(b_norm) + 1e-15)
        if cosine_sim > 0.9999:
            stagnant += 1
            if stagnant > 10:
                p = p + 0.05 * rng.standard_normal(n)
                p = normalize_p(p)
                c = autocorr(p)
                M = np.max(c)
                stagnant = 0
            continue
        best_t = 0.0
        best_new_M = M * 1.01
        for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05, 0.02]:
            p_cand = (1.0 - t) * p + t * b_norm
            p_cand = normalize_p(p_cand)
            c_new = autocorr(p_cand)
            new_M = np.max(c_new)
            if new_M < best_new_M:
                best_new_M = new_M
                best_t = t
        if best_t > 0:
            p = (1.0 - best_t) * p + best_t * b_norm
            p = normalize_p(p)
            c = autocorr(p)
            new_M = np.max(c)
            if new_M < M:
                M = new_M
                stagnant = 0
                sc = 2.0 * n * M
                if sc < best_score:
                    best_score = sc
                    best_p = p.copy()
                    if verbose and it % 30 == 0:
                        print(f"      LP it={it}, score={best_score:.6f}, t={time.monotonic()-t0:.0f}s")
            else:
                stagnant += 1
        else:
            stagnant += 1
            if stagnant > 15:
                p = p + 0.08 * rng.standard_normal(n)
                p = normalize_p(p)
                c = autocorr(p)
                M = np.max(c)
                stagnant = 0
        it += 1
    return best_p, best_score


def adam_optimize(p_init, time_budget, rng, verbose=True):
    n = len(p_init)
    theta = np.log(p_init + 1e-15)
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.3
    t0 = time.monotonic()
    best_p = p_init.copy()
    best_score = score_from_p(p_init)
    tau_start = 50.0 / n
    tau_end = 0.3 / n
    while time.monotonic() - t0 < time_budget:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_budget)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.6)
        p = softmax_stable(theta)
        c = autocorr(p)
        max_c = np.max(c)
        shifted = (c - max_c) / tau
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()
        size = 2 * n
        fp = np.fft.rfft(p, size)
        fw = np.fft.rfft(w, size)
        corr = np.fft.irfft(np.conj(fw) * fp, size)[:n]
        grad_p = 2.0 * corr
        inner = np.dot(p, grad_p)
        grad_theta = p * (grad_p - inner)
        step += 1
        m = beta1 * m + (1.0 - beta1) * grad_theta
        v = beta2 * v + (1.0 - beta2) * (grad_theta * grad_theta)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr_eff = lr / np.sqrt(1.0 + 0.005 * step)
        theta = theta - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        if step % 200 == 0:
            p_check = softmax_stable(theta)
            sc = score_from_p(p_check)
            if sc < best_score:
                best_score = sc
                best_p = p_check.copy()
                if verbose and step % 1000 == 0:
                    print(f"      Adam step={step}, score={best_score:.6f}, tau={tau:.2e}, t={time.monotonic()-t0:.0f}s")
    p_final = softmax_stable(theta)
    sc_final = score_from_p(p_final)
    if sc_final < best_score:
        best_score = sc_final
        best_p = p_final.copy()
    return best_p, best_score


def coordinate_descent(p_init, time_budget, rng, verbose=True):
    p = p_init.copy()
    p = normalize_p(p)
    n = len(p)
    c = autocorr(p)
    cur_M = np.max(c)
    best_p = p.copy()
    best_M = cur_M
    t0 = time.monotonic()
    stagnant = 0
    while time.monotonic() - t0 < time_budget and stagnant < 5000:
        r = rng.random()
        if r < 0.45:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.02, 0.12)
            amt = p[i] * frac
            p_new = p.copy()
            p_new[i] -= amt
            p_new[j] += amt
        elif r < 0.75:
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.04))
            p_new = p.copy()
            p_new[j] *= factor
        elif r < 0.9:
            width = max(1, int(rng.integers(1, max(2, n // 40))))
            start = int(rng.integers(0, max(1, n - width)))
            p_new = p.copy()
            p_new[start:start + width] *= np.exp(rng.normal(0, 0.025, size=width))
        else:
            # Peak-targeted: reduce contributors to the peak lag
            c = autocorr(p)
            peak_k = int(np.argmax(c))
            # Contributors to c[peak_k]: p[i]*p[peak_k-i] for valid i
            contrib_indices = []
            for i in range(n):
                j = peak_k - i
                if 0 <= j < n:
                    contrib_indices.append(i)
            if len(contrib_indices) > 0:
                top_k = max(2, len(contrib_indices) // 10)
                contribs = [p[i] * p[min(peak_k - i, n - 1)] if 0 <= peak_k - i < n else 0 for i in contrib_indices]
                sorted_idx = np.argsort(contribs)[::-1][:top_k]
                top_pos = [contrib_indices[i] for i in sorted_idx]
                p_new = p.copy()
                for idx in top_pos:
                    p_new[idx] *= (1.0 - rng.uniform(0.01, 0.04))
                others = [i for i in range(n) if i not in top_pos]
                if others:
                    n_add = min(3, len(others))
                    add_idx = list(rng.choice(others, size=n_add, replace=False))
                    removed = sum(p[idx] - p_new[idx] for idx in top_pos)
                    for idx in add_idx:
                        p_new[idx] += removed / n_add
        p_new = np.clip(p_new, 0, None)
        s = p_new.sum()
        if s < 1e-15:
            continue
        p_new = p_new / s
        c_new = autocorr(p_new)
        new_M = np.max(c_new)
        if new_M < cur_M * (1.0 - 1e-9):
            p = p_new
            cur_M = new_M
            stagnant = 0
            if cur_M < best_M:
                best_M = cur_M
                best_p = p.copy()
        else:
            stagnant += 1
    return best_p, 2.0 * n * best_M


def multi_start_lp(n, time_budget, rng, n_starts=4):
    best_p = None
    best_score = np.inf
    t0 = time.monotonic()
    styles = ['linear', 'bathtub', 'exponential', 'two_peak', 'inverse_bathtub', 'quadratic']
    time_per_start = time_budget / n_starts
    for i in range(n_starts):
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 5:
            break
        style = styles[i % len(styles)]
        p0 = structured_init(n, rng, style)
        p_opt, sc = lp_optimize(p0, min(time_per_start, remaining), rng, verbose=False)
        if sc < best_score:
            best_score = sc
            best_p = p_opt.copy()
            print(f"    LP start {i} (n={n}, {style}): score={sc:.6f}, t={time.monotonic()-t0:.0f}s")
    if best_p is None:
        p0 = spectral_init(n, rng, n_iter=50)
        p_opt, sc = lp_optimize(p0, time_budget * 0.5, rng, verbose=False)
        best_p = p_opt
        best_score = sc
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_global = time.monotonic()
    time_budget = max(10.0, float(budget_s) - 60.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_p = None
    best_n = 0

    # Check provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, None)
        if len(parent) > 0 and np.sum(parent) > 0:
            p_parent = parent / np.sum(parent)
            sc_parent = score_from_p(p_parent)
            print(f"Parent: n={len(parent)}, score={sc_parent:.6f}")
            if sc_parent < best_score:
                best_score = sc_parent
                best_p = p_parent.copy()
                best_n = len(parent)

    # Phase 1: LP-based optimization with multiple n values
    print("\n=== Phase 1: LP-based optimization ===")
    n_configs = [
        (256, 0.25, 3),
        (512, 0.30, 4),
        (1024, 0.25, 3),
    ]
    phase1_budget = time_budget * 0.50
    t1_start = time.monotonic()
    for n, frac, n_starts in n_configs:
        remaining = deadline - time.monotonic() - 60
        if remaining < 30:
            break
        t_alloc = min(phase1_budget * frac, remaining)
        print(f"\n  n={n}, time={t_alloc:.0f}s, starts={n_starts}")
        p_opt, sc = multi_start_lp(n, t_alloc, rng, n_starts=n_starts)
        if p_opt is not None and sc < best_score:
            best_score = sc
            best_p = p_opt.copy()
            best_n = n
            print(f"    *** New best: n={n}, score={best_score:.6f} ***")

    # Phase 2: Adam optimization from best found
    print(f"\n=== Phase 2: Adam optimization ===")
    print(f"  Best before Adam: {best_score:.6f} (n={best_n})")
    phase2_budget = min(time_budget * 0.25, max(60.0, deadline - time.monotonic() - 90))
    if phase2_budget > 30 and best_p is not None:
        # Try Adam from best and from spectral init
        for trial in range(2):
            if time.monotonic() > deadline - 60:
                break
            if trial == 0:
                p_init = best_p.copy()
            else:
                p_init = spectral_init(best_n, rng, n_iter=100)
            t_alloc = phase2_budget / 2
            p_opt, sc = adam_optimize(p_init, t_alloc, rng, verbose=True)
            if sc < best_score:
                best_score = sc
                best_p = p_opt.copy()
                best_n = len(p_opt)
                print(f"    *** New best after Adam: n={best_n}, score={best_score:.6f} ***")

    # Phase 3: Coordinate descent polish
    print(f"\n=== Phase 3: Coordinate descent polish ===")
    print(f"  Best before polish: {best_score:.6f} (n={best_n})")
    polish_budget = min(300.0, max(30.0, deadline - time.monotonic() - 30))
    if polish_budget > 20 and best_p is not None:
        p_polished, sc_polished = coordinate_descent(best_p, polish_budget, rng, verbose=True)
        if sc_polished < best_score:
            best_score = sc_polished
            best_p = p_polished.copy()
            best_n = len(p_polished)
            print(f"    *** New best after polish: n={best_n}, score={best_score:.6f} ***")
        else:
            print(f"    No improvement from polish (best={best_score:.6f})")

    # Phase 4: Final quick LP polish if time remains
    if time.monotonic() < deadline - 30 and best_p is not None:
        print(f"\n=== Phase 4: Final LP polish ===")
        final_budget = min(120.0, deadline - time.monotonic() - 10)
        if final_budget > 15:
            p_final, sc_final = lp_optimize(best_p, final_budget, rng, verbose=True)
            if sc_final < best_score:
                best_score = sc_final
                best_p = p_final.copy()
                best_n = len(p_final)
                print(f"    *** New best after final LP: n={best_n}, score={best_score:.6f} ***")

    # Convert to original scale and validate
    if best_p is not None:
        n = len(best_p)
        scale = np.sqrt(2.0 * n)
        final_seq = best_p * scale
        final_list = [float(max(0.0, min(1000.0, x))) for x in final_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n=== FINAL: n={n}, internal_score={best_score:.6f}, eval_score={final_eval:.6f} ===")
        print(f"Total time: {time.monotonic() - t_global:.1f}s")
        if final_eval == np.inf:
            # Fallback: return uniform sequence
            return [1.0] * 256
        return final_list

    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, None)
        return [float(x) for x in parent]
    return [1.0] * 256
