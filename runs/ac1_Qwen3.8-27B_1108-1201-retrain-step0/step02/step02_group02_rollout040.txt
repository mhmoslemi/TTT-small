LP-directed peak optimization for minimizing the evaluation score of non-negative
step functions. Uses linear programming-based direction finding (cross-correlation LP
and linearized self-correlation LP), combined with soft-max gradient descent refinement,
coordinate descent polish, and multi-start with diverse initializations.
Key innovation: the LP directly identifies the best feasible direction to reduce the
convolution peak, providing more principled updates than pure gradient methods.

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def build_conv_matrix(f):
    """Build the (2n-1) x n matrix A such that (Ab)_k = (f*b)_k."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            A[k, j] = f[k - j]
    return A


def build_conv_matrix_fast(f):
    """Build convolution matrix using vectorized operations."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # f[j] contributes to A[k, j] where k-j = j, i.e., k = 2j... no
        # A[k, j] = f[k-j] if 0 <= k-j < n
        # So for fixed j, A[k, j] = f[k-j] for k in [j, j+n-1]
        lo = j
        hi = min(j + n, 2 * n - 1)
        idx = np.arange(lo, hi)
        A[idx, j] = f[idx - j]
    return A


def lp_cross_direction(f, M, timeout=5.0):
    """
    Solve LP: maximize 1^T b subject to (f*b)_k <= M for all k, b >= 0.
    Returns the optimal b (rescaled to have same sum as f) or None if failed.
    """
    n = len(f)
    A = build_conv_matrix_fast(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n

    t0 = time.monotonic()
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': timeout, 'presolve': True})

    if res.status != 0:
        return None
    if time.monotonic() - t0 > timeout * 1.5:
        return None

    b_star = res.x.copy()
    s = np.sum(b_star)
    if s < 1e-10:
        return None
    # Rescale to match sum of f
    b_rescaled = b_star * (np.sum(f) / s)
    return b_rescaled


def lp_linearized_direction(f, M, S_target, timeout=5.0):
    """
    Solve LP: minimize M' subject to
      2(f*b)_k - (f*f)_k <= M' for all k
      sum(b) = S_target
      b >= 0
    This linearizes the self-convolution peak around f.
    Returns optimal b or None.
    """
    n = len(f)
    c_f = fft_convolve(f, f)
    A_f = A_b = build_conv_matrix_fast(f)

    # Variables: b_0, ..., b_{n-1}, M'  (n+1 variables)
    # Objective: minimize M' -> c = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Constraints: 2*A_f*b - c_f <= M'
    # => 2*A_f*b - M' <= c_f
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = 2.0 * A_f
    A_ub[:, n] = -1.0
    b_ub = c_f.copy()

    # Equality: sum(b) = S_target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S_target])

    bounds = [(0.0, None)] * n + [(-np.inf, np.inf)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})

    if res.status != 0:
        return None
    return res.x[:n].copy()


def optimize_lp_method(n, f_init, time_limit, rng, use_linearized=False):
    """
    Main LP-based optimization loop.
    Alternates between cross-correlation LP and (optionally) linearized LP.
    """
    f = f_init.astype(np.float64).copy()
    S = np.sum(f)
    f = f * (S / S)  # keep S unchanged
    best_f = f.copy()
    best_score = local_score(f)
    current_M = np.max(fft_convolve(f, f))

    t0 = time.monotonic()
    step = 0
    no_improve = 0
    patience = 500
    step_size = 0.3

    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        if elapsed > 0.95:
            break

        # Recompute M
        c = fft_convolve(f, f)
        current_M = np.max(c)

        if use_linearized and step % 3 == 0:
            b_dir = lp_linearized_direction(f, current_M, S, timeout=3.0)
        else:
            b_dir = lp_cross_direction(f, current_M, timeout=3.0)

        if b_dir is None:
            # Fallback: small random perturbation
            f = f + 0.01 * rng.standard_normal(n)
            f = np.clip(f, 0.0, None)
            f = f * S / np.sum(f)
            step += 1
            continue

        # Check if direction is useful
        diff = np.linalg.norm(b_dir - f) / (np.linalg.norm(f) + 1e-10)
        if diff < 1e-6:
            # Degenerate: perturb and continue
            f = f + 0.05 * rng.standard_normal(n)
            f = np.clip(f, 0.0, None)
            f = f * S / np.sum(f)
            step += 1
            continue

        # Adaptive step size
        # Try the update with current step_size
        g = (1.0 - step_size) * f + step_size * b_dir
        g = np.clip(g, 0.0, None)
        g = g * S / np.sum(g)

        new_M = np.max(fft_convolve(g, g))

        if new_M < current_M:
            f = g
            current_M = new_M
            # Try to increase step size
            if step_size < 0.5:
                step_size = min(0.5, step_size * 1.1)
            no_improve = 0
        else:
            # Reduce step size
            step_size *= 0.5
            if step_size < 0.01:
                # Stuck: try a larger perturbation
                r = rng.random()
                if r < 0.5:
                    f = f * np.exp(rng.normal(0, 0.1, size=n))
                else:
                    # Block perturbation
                    block = max(2, n // 8)
                    start = int(rng.integers(0, n - block))
                    f[start:start+block] *= np.exp(rng.normal(0, 0.2, size=block))
                f = np.clip(f, 0.0, None)
                f = f * S / np.sum(f)
                step_size = 0.1
                no_improve += 100
                step += 1
                continue

        step += 1
        score = local_score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Escape mechanism
        if no_improve > patience:
            r = rng.random()
            if r < 0.3:
                f = best_f + 0.2 * rng.standard_normal(n)
            elif r < 0.6:
                # Peak-directed perturbation
                c = fft_convolve(f, f)
                peak_idx = int(np.argmax(c))
                contribs = np.zeros(n)
                for i in range(n):
                    if 0 <= peak_idx - i < n:
                        contribs[i] = f[i] * f[peak_idx - i]
                top_k = np.argsort(contribs)[::-1][:max(2, n // 10)]
                f[top_k] *= 0.9
                f = np.clip(f, 0.0, None)
                f = f * S / np.sum(f)
            else:
                # Fresh random reinitialization blended with best
                f_rand = np.abs(rng.standard_normal(n)) + 0.1
                f = 0.5 * best_f / np.sum(best_f) + 0.5 * f_rand / np.sum(f_rand)
                f = f * S / np.sum(f)
            step_size = 0.3
            no_improve = 0

        if step % 500 == 0:
            print(f"      LP step={step}, score={best_score:.6f}, M={current_M:.4f}, "
                  f"step_size={step_size:.3f}, t={(time.monotonic()-t0):.0f}s")

    # Final check
    final_score = local_score(f)
    if final_score < best_score:
        best_score = final_score
        best_f = f.copy()

    return best_f, best_score


def optimize_gradient(n, theta_init, time_limit, rng):
    """
    Improved soft-max gradient descent with adaptive temperature.
    """
    S_target = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0

    best_theta = theta.copy()
    best_score = local_score(softmax_normalize(theta) * S_target)
    no_improve = 0
    patience = 2000

    t0 = time.monotonic()
    print(f"    Initial score: {best_score:.6f}")

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit

        # Adaptive temperature: starts warm, cools down
        if elapsed_frac < 0.5:
            tau = (30.0 / n) * (1.0 / n) ** (elapsed_frac / 0.5)
        else:
            tau = (1.0 / n) * (0.01 / n) ** ((elapsed_frac - 0.5) / 0.5)

        a = softmax_normalize(theta) * S_target
        c = fft_convolve(a, a)
        c_max = np.max(c)
        shifted = (c - c_max) / max(tau, 1e-10)
        # Clip to avoid overflow
        shifted = np.clip(shifted, -50, 50)
        exp_shifted = np.exp(shifted)
        sum_exp = np.sum(exp_shifted)
        w = exp_shifted / sum_exp

        # Gradient: g[j] = 2 * sum_k w[k] * dc[k]/da[j]
        # dc[k]/da[j] = a[k-j] if 0<=k-j<n plus a[j-k] if 0<=j-k<n
        # More efficiently: g = 2 * (flip(a) * w)[n-1:2n-1] via correlation
        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g = 2.0 * corr[n - 1:2 * n - 1]

        # Convert to theta-space
        inner = np.dot(a, g)
        g_theta = a * (g - inner / S_target)

        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g_theta
        v_buf = beta2 * v_buf + (1 - beta2) * g_theta ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)

        base_lr = 3.0 / np.sqrt(step + 1)
        if elapsed_frac < 0.2:
            lr = base_lr * 1.5
        elif elapsed_frac < 0.6:
            lr = base_lr
        else:
            lr = base_lr * 0.3

        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if step % 300 == 0:
            a_check = softmax_normalize(theta) * S_target
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
                no_improve = 0
            else:
                no_improve += 300

            if no_improve > patience:
                r = rng.random()
                if r < 0.4:
                    theta = best_theta + 0.3 * rng.standard_normal(n)
                elif r < 0.7:
                    a_cur = softmax_normalize(best_theta) * S_target
                    c_cur = fft_convolve(a_cur, a_cur)
                    peak_idx = int(np.argmax(c_cur))
                    contribs = np.zeros(n)
                    for i in range(n):
                        if 0 <= peak_idx - i < n:
                            contribs[i] = a_cur[i] * a_cur[peak_idx - i]
                    top_k = np.argsort(contribs)[::-1][:n // 6]
                    theta[top_k] += 0.2 * rng.standard_normal(len(top_k))
                else:
                    block = max(2, n // 12)
                    start = int(rng.integers(0, n - block))
                    theta[start:start+block] += 0.15 * rng.standard_normal(block)
                no_improve = 0

            if step % 3000 == 0:
                print(f"    Grad step={step}, score={score:.6f}, best={best_score:.6f}, "
                      f"tau={tau:.2e}, t={(time.monotonic()-t0):.0f}s")

    a_final = softmax_normalize(theta) * S_target
    final_score = local_score(a_final)
    if final_score < best_score:
        best_score = final_score
        best_theta = theta.copy()

    return best_theta, best_score


def peak_shave_polish(f_init, time_limit, rng):
    """
    Targeted peak shaving: identify peak contributors and redistribute mass.
    Uses simulated annealing acceptance.
    """
    current = f_init.astype(np.float64).copy()
    S = np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.0005
    cooling = 0.99998

    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]

        # Compute contribution to peak
        contribs = np.zeros(n)
        for i in range(n):
            if 0 <= peak_idx - i < n:
                contribs[i] = current[i] * current[peak_idx - i]

        r = rng.random()
        trial = current.copy()

        if r < 0.35:
            # Reduce top contributors, increase bottom
            top_k = max(3, n // 20)
            bot_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            bot_idx = np.argsort(contribs)[:bot_k]
            red = rng.uniform(0.01, 0.05)
            trial[top_idx] *= (1.0 - red)
            trial[bot_idx] *= (1.0 + red * len(top_idx) / len(bot_idx))
        elif r < 0.55:
            # Pair transfer
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            j_red = top_idx[int(rng.integers(0, top_k))]
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red:
                amt = current[j_red] * rng.uniform(0.01, 0.04)
                trial[j_red] -= amt
                trial[j_inc] += amt
        elif r < 0.75:
            # Block scale
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.02))
            trial[start:start+width] *= factor
        elif r < 0.9:
            # Small global
            trial *= np.exp(rng.normal(0, 0.003, size=n))
        else:
            # Second-peak targeting
            # Find second highest peak
            c_sorted = c.copy()
            c_sorted[peak_idx] = -1
            # Also remove neighbors of peak
            for d in range(-2, 3):
                idx2 = peak_idx + d
                if 0 <= idx2 < len(c_sorted):
                    c_sorted[idx2] = min(c_sorted[idx2], c[peak_idx] * 0.8)
            peak2_idx = int(np.argmax(c_sorted))
            contribs2 = np.zeros(n)
            for i in range(n):
                if 0 <= peak2_idx - i < n:
                    contribs2[i] = current[i] * current[peak2_idx - i]
            top_idx2 = np.argsort(contribs2)[::-1][:max(2, n // 20)]
            trial[top_idx2] *= (1.0 - rng.uniform(0.01, 0.03))

        trial = np.clip(trial, 0.0, None)
        trial = trial * S / np.sum(trial)
        new_score = local_score(trial)

        if new_score < current_score:
            current = trial
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current = trial
            current_score = new_score

        temp *= cooling
        if step % 2000 == 0:
            print(f"    PS step={step}, cur={current_score:.6f}, best={best_score:.6f}, "
                  f"temp={temp:.6f}, t={(time.monotonic()-t0):.0f}s")

    return best, best_score


def coordinate_polish(f_init, time_limit, rng):
    """
    Multi-round coordinate descent with geometric line search.
    """
    current = f_init.astype(np.float64).copy()
    S = np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            base_val = current[j]
            if base_val < 1e-8:
                # Try turning on a zero element
                trial = current.copy()
                trial[j] = base_val + S * 0.001
                trial = trial * S / np.sum(trial)
                ns = local_score(trial)
                if ns < current_score:
                    current = trial
                    current_score = ns
                    improved = True
                    if ns < best_score:
                        best_score = ns
                        best = current.copy()
                continue

            # Geometric search
            for frac in [0.7, 0.85, 0.95, 1.05, 1.15, 1.35]:
                trial = current.copy()
                trial[j] = base_val * frac
                trial = np.clip(trial, 0.0, None)
                trial = trial * S / np.sum(trial)
                ns = local_score(trial)
                if ns < current_score:
                    current = trial
                    current_score = ns
                    improved = True
                    if ns < best_score:
                        best_score = ns
                        best = current.copy()
                else:
                    break

        if not improved:
            # Random kick
            j = int(rng.integers(0, n))
            trial = current.copy()
            trial[j] *= np.exp(rng.normal(0, 0.1))
            trial = np.clip(trial, 0.0, None)
            trial = trial * S / np.sum(trial)
            ns = local_score(trial)
            if ns < current_score * 1.002:
                current = trial
                current_score = ns

        iteration += 1
        if iteration % 30 == 0:
            print(f"    CD iter={iteration}, cur={current_score:.6f}, best={best_score:.6f}, "
                  f"t={(time.monotonic()-t0):.0f}s")

    return best, best_score


def make_structured_init(n, rng, method='broadband'):
    """Create structured initial sequences with good spectral properties."""
    t = np.arange(n, dtype=np.float64)

    if method == 'broadband':
        a = np.ones(n)
        for k in range(1, 20):
            freq = k * (1.0 + 0.5 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.12 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'zilching':
        # Start with a pattern that has low autocorrelation
        # Use a "nearly orthogonal" pattern
        a = np.ones(n)
        # Add a periodic modulation with incommensurate frequency
        freq1 = 1.0 + 0.6180339887  # golden ratio
        freq2 = 1.0 + 0.3819660113
        a[::2] *= 0.7
        a[1::2] *= 1.3
        a += 0.1 * np.cos(2 * np.pi * freq1 * t / n)
        a += 0.1 * np.cos(2 * np.pi * freq2 * t / n)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'tapered':
        # Tapered ends to reduce edge effects
        a = np.ones(n)
        taper = np.sqrt(np.sin(np.pi * t / (2 * n)))
        a *= 0.5 + 0.5 * taper
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'bipolar_smooth':
        # Smooth bipolar-like pattern
        period = max(3, int(np.round(np.sqrt(n) * 1.5)))
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.05, None)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a different length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, f)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search combining LP-directed optimization, gradient descent,
    and local polish. Explores multiple sequence lengths and initializations.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    def try_optimize(n, f_init, t_alloc, method='lp'):
        """Run an optimization method and return (sequence, score)."""
        nonlocal best_score, best_seq
        if t_alloc < 10:
            return f_init, local_score(f_init)

        S = np.sum(f_init)
        f_init = f_init * (np.sqrt(2.0 * n) / S)

        if method == 'lp':
            f_res, score = optimize_lp_method(n, f_init, t_alloc, rng, use_linearized=False)
        elif method == 'lp_lin':
            f_res, score = optimize_lp_method(n, f_init, t_alloc, rng, use_linearized=True)
        elif method == 'grad':
            theta = np.log(f_init / np.sum(f_init) + 1e-10)
            theta_res, score = optimize_gradient(n, theta, t_alloc, rng)
            f_res = softmax_normalize(theta_res) * np.sqrt(2.0 * n)
        else:
            f_res = f_init
            score = local_score(f_res)

        # Keep consistent scaling
        f_res = np.clip(f_res, 0.0, None)
        actual = local_score(f_res)
        if actual < best_score:
            best_score = actual
            best_seq = f_res.copy()
            print(f"    *** New global best: {best_score:.6f} (n={n}, method={method}) ***")
        return f_res, actual

    # === PHASE 1: LP-based optimization at n=256 (primary) ===
    print(f"\n{'='*60}")
    print(f"PHASE 1: LP optimization (n=256)")
    print(f"{'='*60}")

    n_main = 256

    # Start from provided sequence if available
    if start_seq is not None:
        f0 = interpolate_sequence(start_seq, n_main)
    else:
        f0 = make_structured_init(n_main, rng, 'broadband')

    t_phase1 = 0.40 * time_budget
    f1, s1 = try_optimize(n_main, f0, t_phase1, 'lp')
    print(f"  LP result: score={s1:.6f}")

    # Second LP run with linearized variant
    if time.monotonic() < deadline - 60:
        t_p1b = min(0.15 * time_budget, deadline - time.monotonic() - 60)
        if t_p1b > 20:
            f1b, s1b = try_optimize(n_main, f1, t_p1b, 'lp_lin')
            print(f"  LP-lin result: score={s1b:.6f}")

    # Fresh start with different init
    if time.monotonic() < deadline - 60:
        t_p1c = min(0.10 * time_budget, deadline - time.monotonic() - 60)
        if t_p1c > 15:
            f0c = make_structured_init(n_main, rng, 'zilching')
            f1c, s1c = try_optimize(n_main, f0c, t_p1c, 'lp')
            print(f"  LP fresh-start result: score={s1c:.6f}")

    # === PHASE 2: Gradient refinement ===
    print(f"\n{'='*60}")
    print(f"PHASE 2: Gradient refinement (n=256)")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 60:
        f2_init = intercept = interpolate_sequence(best_seq, n_main)
        t_phase2 = min(0.15 * time_budget, deadline - time.monotonic() - 60)
        if t_phase2 > 15:
            f2, s2 = try_optimize(n_main, f2_init, t_phase2, 'grad')
            print(f"  Gradient result: score={s2:.6f}")

    # === PHASE 3: Peak-shave polish ===
    print(f"\n{'='*60}")
    print(f"PHASE 3: Peak-shave polish")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        t_phase3 = min(0.10 * time_budget, deadline - time.monotonic() - 30)
        if t_phase3 > 10:
            polished, ps_score = peak_shave_polish(best_seq, t_phase3, rng)
            if ps_score < best_score:
                best_score = ps_score
                best_seq = polished.copy()
                print(f"  After peak-shave: {best_score:.6f}")
            else:
                print(f"  After peak-shave: no improvement (still {best_score:.6f})")

    # === PHASE 4: Try other lengths ===
    print(f"\n{'='*60}")
    print(f"PHASE 4: Multi-length exploration")
    print(f"{'='*60}")

    for n_try in [320, 384, 192, 448]:
        if time.monotonic() > deadline - 40:
            break
        t_try = min(30.0, deadline - time.monotonic() - 20)
        if t_try < 12:
            break

        if best_seq is not None:
            f0_try = interpolate_sequence(best_seq, n_try)
        else:
            f0_try = make_structured_init(n_try, rng, 'broadband')

        # Quick LP optimization
        f_try, s_try = try_optimize(n_try, f0_try, t_try, 'lp')
        print(f"  n={n_try}: score={s_try:.6f}")

    # === PHASE 5: Final polish ===
    print(f"\n{'='*60}")
