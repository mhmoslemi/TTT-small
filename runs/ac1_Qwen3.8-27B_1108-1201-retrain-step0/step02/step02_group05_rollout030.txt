"""
LP-guided optimization for minimum peak convolution non-negative sequences.
Combines L-BFGS p-norm optimization with LP-based directional refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix, csr_matrix


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fn(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def build_F_sparse(a):
    """Build sparse (2n-1) x n matrix F where F[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    F = lil_matrix((2*n-1, n))
    for j in range(n):
        F[j:j+n, j] = a
    return F.tocsr()


def solve_lp_direction(a):
    """
    Solve LP: minimize t subject to F@g <= t*1, sum(g)=1, g>=0, t>=0.
    Returns (g_star, t_star) or (None, None) if infeasible.
    """
    n = len(a)
    F = build_F_sparse(a)
    
    # Variables: [g_0, ..., g_{n-1}, t] (n+1 total)
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize t
    
    # Inequality: F@g - t <= 0 for each of the 2n-1 rows
    # Matrix: [F | -1] (size (2n-1) x (n+1))
    A_ub = lil_matrix((2*n-1, n+1))
    A_ub[:, :n] = F
    A_ub[:, n] = -1
    A_ub = A_ub.tocsr()
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n+1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs', options={'presolve': True})
    
    if result.status != 0:
        return None, None
    
    g_star = result.x[:n]
    t_star = result.x[n]
    
    # Ensure g is valid
    g_star = np.maximum(g_star, 0)
    if np.sum(g_star) < 1e-10:
        return None, None
    g_star = g_star / np.sum(g_star)
    
    return g_star, t_star


def lp_refine(a, budget=600, verbose=True):
    """
    LP-based iterative refinement.
    """
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    current_score = score_fn(a)
    best = a.copy()
    best_score = current_score
    n = len(a)
    t0 = time.monotonic()
    stagnant = 0
    
    while time.monotonic() - t0 < budget:
        g_star, t_star = solve_lp_direction(a)
        if g_star is None:
            break
        
        if t_star >= 1.0 - 1e-6:
            # No improvement possible from this direction
            stagnant += 1
            if stagnant > 5:
                break
            # Perturb slightly
            a = a * np.exp(np.random.default_rng(stagnant).normal(0, 0.01, n))
            a = np.maximum(a, 1e-10)
            a = a / np.sum(a)
            continue
        
        stagnant = 0
        
        # Line search: try different step sizes
        best_s = 0
        best_new_score = current_score
        for s in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            a_new = (1-s)*a + s*g_star
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)
            s_score = score_fn(a_new)
            if s_score < best_new_score:
                best_new_score = s_score
                best_s = s
        
        if best_s > 0:
            a = (1-best_s)*a + best_s*g_star
            a = np.maximum(a, 1e-15)

            current_score = best_new_score
            if current_score < best_score:

                if verbose:

def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def pnorm_obj_grad(theta, p):
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta

            if s < best_score:

                best_theta = theta_new.copy()

            pass
        theta = best_theta

"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix, csr_matrix


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fn(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def build_F_sparse(a):
    """Build sparse (2n-1) x n convolution matrix F where F[k,j] = a[k-j] if valid."""
    n = len(a)
    F = lil_matrix((2 * n - 1, n))
    for j in range(n):
        F[j:j + n, j] = a
    return F.tocsr()


def solve_lp_direction(a):
    """
    Solve LP: minimize t subject to F@g <= t*1, sum(g)=1, g>=0, t>=0.
    Returns (g_star, t_star) or (None, None).
    """
    n = len(a)
    F = build_F_sparse(a)

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = lil_matrix((2 * n - 1, n + 1))
    A_ub[:, :n] = F
    A_ub[:, n] = -1
    A_ub = A_ub.tocsr()
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'presolve': True})
    except Exception:
        return None, None

    if result.status != 0:
        return None, None

    g_star = result.x[:n]
    t_star = result.x[n]

    g_star = np.maximum(g_star, 0)
    sg = np.sum(g_star)
    if sg < 1e-10:
        return None, None
    g_star = g_star / sg

    return g_star, t_star


def lp_refine(a, budget, rng, verbose=True):
    """LP-based iterative refinement with adaptive line search."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    current_score = score_fn(a)
    best = a.copy()
    best_score = current_score
    n = len(a)
    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < budget:
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: score={current_score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")

        g_star, t_star = solve_lp_direction(a)
        if g_star is None:
            if verbose:
                print(f"    LP infeasible at iter {it}, stopping.")
            break

        it += 1

        if t_star >= 1.0 - 1e-5:
            stagnant += 1
            if stagnant > 10:
                if verbose:
                    print(f"    LP stagnant (t*={t_star:.6f}), stopping.")
                break
            # Perturb to escape
            perturb = rng.normal(0, 0.02, n)
            a_pert = a * np.exp(perturb)
            a_pert = np.maximum(a_pert, 1e-12)
            a_pert = a_pert / np.sum(a_pert)
            sp = score_fn(a_pert)
            if sp < current_score + 0.01:
                a = a_pert
                current_score = sp
            continue

        stagnant = 0

        # Line search over step sizes
        best_s = 0.0
        best_new_score = current_score
        step_candidates = [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
        for s in step_candidates:
            a_new = (1 - s) * a + s * g_star
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)
            sc = score_fn(a_new)
            if sc < best_new_score:
                best_new_score = sc
                best_s = s

        if best_s > 0:
            a = (1 - best_s) * a + best_s * g_star
            a = np.maximum(a, 1e-15)
            a = a / np.sum(a)
            current_score = best_new_score
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            # Tiny step
            a = 0.99 * a + 0.01 * g_star
            a = np.maximum(a, 1e-15)
            a = a / np.sum(a)
            current_score = score_fn(a)

    return best, best_score


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    # Avoid numerical issues
    c_clipped = np.maximum(c, 1e-30)
    cp = c_clipped ** (p - 1)
    sum_cp = np.sum(cp * c_clipped)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score_val = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score_val, grad_theta


def soft_max_obj_grad(theta, tau):
    """Objective and gradient for 2n * soft_max(conv(a,a), tau)."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    if sum_exp < 1e-300:
        return 2.0 * n * c_max, np.zeros(n)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score_val = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score_val, grad_theta


def run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p=300, deadline=None):
    """Run L-BFGS with annealed p-norm. Returns best theta and score."""
    best_theta = theta.copy()
    best_score = score_fn(softmax(theta))

    for p in p_schedule:
        if deadline is not None and time.monotonic() > deadline - 10:
            break
        try:
            result = minimize(
                pnorm_obj_grad, theta.copy(),
                method='L-BFGS-B', jac=True, args=(p,),
                options={'maxiter': maxiter_per_p, 'ftol': 1e-15, 'gtol': 1e-13,
                         'maxfun': maxiter_per_p * 10}
            )
            theta_new = result.x
            s = score_fn(softmax(theta_new))
            if s < best_score:
                best_score = s
                best_theta = theta_new.copy()
        except Exception:
            pass
        theta = best_theta

    return best_theta, best_score

        elif r < 0.85:

        new_score = score_fn(candidate)

            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-14)):
            current = candidate
            current_score = new_score

        temp = max(temp * cooling, 1e-9)

    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Bell-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase hybrid optimization:
    Phase 1: L-BFGS p-norm from provided sequence + diverse starts
    Phase 2: LP-based directional refinement (key innovation)
    Phase 3: L-BFGS cleanup + soft-max
    Phase 4: LP refinement round 2
    Phase 5: Targeted local search polish
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 45.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        s = score_fn(start_seq)
        print(f"Starting sequence: n={len(start_seq)}, score={s:.8f}")
        best_score = s
        best_seq = start_seq.copy()

    # Phase 1: L-BFGS initial optimization
    print("\n=== Phase 1: L-BFGS p-norm optimization ===")
    n_main = len(best_seq) if best_seq is not None else 256

    if best_seq is not None:
        theta = np.log(best_seq + 1e-30)
        p1_deadline = time.monotonic() + 300  # 5 min
        p_schedule = [4, 8, 16, 32, 64, 128, 256]
        theta_best, s = run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p=400, deadline=p1_deadline)
        a_new = softmax(theta_best)
        if s < best_score:
            best_score = s
            best_seq = a_new.copy()
            print(f"  Phase 1 (p-norm): {best_score:.8f} *** IMPROVED ***")
        else:
            print(f"  Phase 1 (p-norm): {s:.8f} (no improvement)")

    # Try a couple of alternative initializations
    for init_type in ['chirp', 'random']:
        if time.monotonic() > deadline - 1800:
            break
        if init_type == 'chirp':
            a0 = make_chirp_init(n_main, 0.3 + 0.4 * rng.random(), rng)
        else:
            a0 = make_random_init(n_main, rng, spread=0.3 + 0.4 * rng.random())
        theta = np.log(a0 + 1e-30)
        p1b_deadline = time.monotonic() + 120
        theta_best, s = run_lbgfs_pnorm(theta, [8, 16, 32, 64], maxiter_per_p=200, deadline=p1b_deadline)
        if s < best_score:
            best_score = s
            best_seq = softmax(theta_best).copy()
            print(f"  Phase 1 ({init_type} init): {best_score:.8f} *** NEW BEST ***")
        else:
            print(f"  Phase 1 ({init_type} init): {s:.8f}")

    # Phase 2: LP-based refinement (main workhorse)
    print(f"\n=== Phase 2: LP refinement (current best={best_score:.8f}) ===")
    if best_seq is not None:
        lp2_budget = min(900.0, deadline - time.monotonic() - 300)
        if lp2_budget > 60:
            polished, p_score = lp_refine(best_seq, lp2_budget, rng, verbose=True)
            if p_score < best_score:
                best_score = p_score
                best_seq = polished.copy()
                print(f"  Phase 2 (LP): {best_score:.8f} *** IMPROVED ***")
            else:
                print(f"  Phase 2 (LP): {p_score:.8f} (no improvement)")

    # Phase 3: L-BFGS cleanup with soft-max
    print(f"\n=== Phase 3: L-BFGS cleanup + soft-max (best={best_score:.8f}) ===")
    if best_seq is not None:
        theta = np.log(best_seq + 1e-30)
        p3_deadline = time.monotonic() + 240
        theta_best, s = run_lbgfs_pnorm(theta, [16, 32, 64, 128, 256], maxiter_per_p=300, deadline=p3_deadline)
        if s < best_score:
            best_score = s
            best_seq = softmax(theta_best).copy()
            print(f"  Phase 3 (p-norm): {best_score:.8f} *** IMPROVED ***")

        # Soft-max refinement
        if time.monotonic() < deadline - 200:
            theta = np.log(best_seq + 1e-30)
            tau_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005]
            sm_deadline = min(time.monotonic() + 180, deadline - 120)
            theta_best, s = run_lbgfs_softmax_obj(theta, tau_schedule, maxiter_per_tau=300, deadline=sm_deadline)
            if s < best_score:
                best_score = s
                best_seq = softmax(theta_best).copy()
                print(f"  Phase 3 (soft-max): {best_score:.8f} *** IMPROVED ***")
            else:
                print(f"  Phase 3 (soft-max): {s:.8f}")

    # Phase 4: LP refinement round 2
    print(f"\n=== Phase 4: LP refinement round 2 (best={best_score:.8f}) ===")
    if best_seq is not None:
        lp4_budget = min(480.0, deadline - time.monotonic() - 120)
        if lp4_budget > 60:
            polished, p_score = lp_refine(best_seq, lp4_budget, rng, verbose=True)
            if p_score < best_score:
                best_score = p_score
                best_seq = polished.copy()
                print(f"  Phase 4 (LP): {best_score:.8f} *** IMPROVED ***")
            else:
                print(f"  Phase 4 (LP): {p_score:.8f} (no improvement)")

    # Phase 5: Targeted local search
    print(f"\n=== Phase 5: Targeted local search (best={best_score:.8f}) ===")
    if best_seq is not None:
        ls_budget = max(30.0, min(120.0, deadline - time.monotonic() - 10))
        polished, p_score = targeted_local_search(best_seq, ls_budget, rng)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  Phase 5 (local): {best_score:.8f} *** IMPROVED ***")
        else:
            print(f"  Phase 5 (local): {p_score:.8f}")

    # Final short LP pass if time permits
    if time.monotonic() < deadline - 30 and best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            print(f"\n=== Final LP pass ({remaining:.0f}s remaining) ===")
            polished, p_score = lp_refine(best_seq, remaining, rng, verbose=False)
            if p_score < best_score:
                best_score = p_score
                best_seq = polished.copy()
                print(f"  Final LP: {best_score:.8f} *** IMPROVED ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.8f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
