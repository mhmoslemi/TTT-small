    # convolve w (length 2n-1) with a_rev (length n), result length 3n-2
    conv_w_a = fft_convolve(w, a_rev)
    grad = 2.0 * conv_w_a[n-1:2*n-1]
    return c, w, grad

def evaluate_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = convolve_square(a)
    return 2.0 * n * np.max(c) / (s * s)

def project_simplex(a, target_sum=1.0):
    """Project a onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a / s * target_sum
    else:
        a = np.full_like(a, target_sum / len(a))
    return a

            score = evaluate_score(a_check)

                best_a = a_check.copy()

"""
Optimization of non-negative step functions to minimize 2n*max(conv)/sum^2.
Uses softmax-parameterized gradient descent with annealed soft-max,
upsampling for convergence, and peak-directed local search.
"""

import numpy as np
import time


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def conv_sq(a):
    return fft_convolve(a, a)


def score(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    c = conv_sq(a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax(x):
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)


def linear_interp(a, new_n):
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def optimize(a_init, time_limit, rng, tau_init_mult=30.0, tau_end_mult=0.1, lr0=1.5):
    """Softmax-parameterized Adam gradient descent with annealed soft-max."""
    n = len(a_init)
    S = 1.0
    a = a_init.copy()
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a) * S
    theta = np.log(a)
    
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0
    best_theta = theta.copy()
    best_score = score(a)
    
    t0 = time.monotonic()
    tau_init = tau_init_mult / n
    tau_end = tau_end_mult / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        tau = tau_init * (tau_end / tau_init) ** (frac ** 0.6)
        
        a = softmax(theta) * S
        c = conv_sq(a)
        
        # Soft-max weights
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        e = np.exp(shifted)
        w = e / np.sum(e)
        
        # Gradient: grad[p] = 2 * sum_{j=0}^{n-1} w[p+j] * a[j]
        # = 2 * (w * a_rev)[n-1 + p]
        a_rev = a[::-1]
        conv_w = fft_convolve(w, a_rev)
        grad_a = 2.0 * conv_w[n - 1:2 * n - 1]
        
        # Transform to theta space
        inner = np.dot(grad_a, a) / S
        grad_theta = a * (grad_a - inner)
        
        step += 1
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        
        m_v = beta1 * m_v + (1 - beta1) * grad_theta
        v_v = beta2 * v_v + (1 - beta2) * (grad_theta ** 2)
        m_hat = m_v / (1.0 - beta1 ** step)
        v_hat = v_v / (1.0 - beta2 ** step)
        
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if step % 1000 == 0:
            a_check = softmax(theta) * S
            s = score(a_check)
            if s < best_score:
                best_score = s
                best_theta = theta.copy()
            if step % 5000 == 0:
                print(f"    step={step}, score={s:.6f}, best={best_score:.6f}, tau={tau:.2e}")
    
    a_final = softmax(best_theta) * S
    s_final = score(a_final)
    if s_final < best_score:
        best_score = s_final
    return a_final, best_score


def local_search(a_init, time_limit, rng):
    """Peak-directed local search with simulated annealing."""
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return a, np.inf
    current_score = score(a)
    best = a.copy()
    best_score = current_score
    n = len(a)
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9997
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.4:
            # Peak-directed: reduce contributions to top peaks
            c = conv_sq(a)
            n_peaks = min(5, len(c) // 4)
            top_lags = np.argsort(c)[-n_peaks:]
            
            # Find which (i,j) pairs contribute most to these peaks
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += a[i] * a[j]
            
            # Reduce high contributors, increase low contributors
            n_reduce = max(2, n // 15)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, n_reduce))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.02, n_reduce))
        elif r < 0.6:
            # Adjacent block swap
            w = max(1, n // 40)
            start = int(rng.integers(0, n - 2 * w))
            candidate[start:start + w] *= np.exp(rng.normal(-0.07, 0.02, w))
            candidate[start + w:start + 2 * w] *= np.exp(rng.normal(0.07, 0.02, w))
        elif r < 0.75:
            # Multi-point perturbation
            n_pts = int(rng.integers(3, 10))
            pts = rng.choice(n, size=n_pts, replace=False)
            deltas = rng.normal(0, 0.04, n_pts)
            candidate[pts] *= np.exp(deltas)
        elif r < 0.9:
            # Antisymmetric nudge (reduce center peak)
            mid = n // 2
            candidate[:mid] *= np.exp(rng.normal(0.02, 0.01, mid))
            candidate[mid:] *= np.exp(rng.normal(-0.02, 0.01, n - mid))
        else:
            # Global noise
            candidate *= np.exp(rng.normal(0, 0.01, n))
        
        candidate = np.maximum(candidate, 0)
        cs = np.sum(candidate)
        if cs < 1e-15:
            candidate = a.copy()
        else:
            candidate = candidate / cs * s
        
        new_score = score(candidate)
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 50:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    a = candidate
                    current_score = new_score
        
        temp *= cooling
    
    return best, best_score

Now, one concern: the local search in the previous code used a loop over individual indices for the peak-directed mutation, which is O(n) per peak. For large n (1024+), this could be slow. Let me vectorize it.

For the peak-directed mutation:
For each lag k in top_lags, the contribution vector is:
contrib_from_lag[i] = a[i] * a[k-i] (for valid i, j=k-i)

This is a "sliding product" which can be computed as:
For lag k, valid i range from max(0, k-n+1) to min(n-1, k).
contrib_from_lag[i] = a[i] * a[k-i]

This is like a "masked element-wise product" of a with a shifted/reversed version.

For a single lag k:
i ranges from max(0, k-n+1) to min(n-1, k)
j = k - i ranges from k - min(n-1, k) to k - max(0, k-n+1)

If k < n: i from 0 to k, j from k to 0. So a[i] * a[k-i] for i=0..k.

Actually, for the linear convolution, flat power spectrum doesn't directly give flat convolution. The relationship is:
- Circular autocorrelation is the IFFT of |FFT(a)|^2
- Linear autocorrelation is the circular autocorrelation "leaked" (zero-padded)

So a flat power spectrum gives flat circular autocorrelation, which means the linear autocorrelation will be flat in the middle and drop off at the edges. The peak will be at the center (where the circular and linear autocorrelations agree).

To reduce the center peak, we need the circular autocorrelation to have a "dip" at lag 0, which means the power spectrum should NOT be flat but rather have a "hole" at DC.

This is getting into deep signal processing. Let me just rely on the gradient optimization to find the right structure.

Since I'm working with sum = 1 (or any positive constant), this is automatically satisfied as long as n is reasonable and values don't exceed 1000.

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences using
softmax-parameterized gradient descent with annealed soft-max, progressive
upsampling, and peak-directed local search with efficient FFT-based gradients.
"""

import numpy as np
import time


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def conv_sq(a):
    return fft_convolve(a, a)


def score(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    c = conv_sq(a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax_fn(x):
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)


def linear_interp(a, new_n):
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def compute_grad(a, tau):
    """Compute gradient of soft_max(conv(a,a), tau) w.r.t. a using FFT."""
    n = len(a)
    c = conv_sq(a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    w = e / np.sum(e)
    a_rev = a[::-1]
    conv_w = fft_convolve(w, a_rev)
    grad_a = 2.0 * conv_w[n - 1:2 * n - 1]
    return c, w, grad_a


def gradient_descent(a_init, time_limit, rng, tau_init_mult=40.0, tau_end_mult=0.05, lr0=2.0):
    """Softmax-parameterized Adam with annealed soft-max temperature."""
    n = len(a_init)
    S = 1.0
    a = np.maximum(a_init, 1e-12)
    a = a / np.sum(a) * S
    theta = np.log(a + 1e-15)

    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0
    best_theta = theta.copy()
    best_score_val = score(a)
    no_improve = 0
    patience = 20000

    t0 = time.monotonic()
    tau_init = tau_init_mult / n
    tau_end = tau_end_mult / n

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        tau = tau_init * (tau_end / tau_init) ** (frac ** 0.5)

        a = softmax_fn(theta) * S
        c, w, grad_a = compute_grad(a, tau)

        inner = np.dot(grad_a, a) / S
        grad_theta = a * (grad_a - inner)

        step += 1
        lr = lr0 / np.sqrt(1.0 + 0.2 * step)

        m_v = beta1 * m_v + (1 - beta1) * grad_theta
        v_v = beta2 * v_v + (1 - beta2) * (grad_theta ** 2)
        m_hat = m_v / (1.0 - beta1 ** step)
        v_hat = v_v / (1.0 - beta2 ** step)

        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if step % 2000 == 0:
            a_check = softmax_fn(theta) * S
            s = score(a_check)
            if s < best_score_val:
                best_score_val = s
                best_theta = theta.copy()
                no_improve = 0
            else:
                no_improve += 2000

            if no_improve >= patience and frac < 0.9:
                # Random restart in theta space
                theta = theta + 0.5 * np.log(a + 1e-15) * rng.standard_normal(n)
                no_improve = 0

            if step % 10000 == 0:
                print(f"    step={step}, score={s:.6f}, best={best_score_val:.6f}, tau={tau:.2e}, elapsed={elapsed:.0f}s")

    a_final = softmax_fn(best_theta) * S
    s_final = score(a_final)
    if s_final < best_score_val:
        best_score_val = s_final
    return a_final, best_score_val


def local_search(a_init, time_limit, rng):
    """Peak-directed local search with simulated annealing."""
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = np.sum(a)
    if s < 1e-15:
        return a, np.inf
    current_score = score(a)
    best = a.copy()
    best_score = current_score
    n = len(a)
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9998

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.35:
            # Peak-directed reduction
            c = conv_sq(a)
            n_peaks = min(4, max(2, n // 64))
            top_lags = np.argsort(c)[-n_peaks:]

            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                if i_max >= i_min:
                    idx = np.arange(i_min, i_max + 1)
                    contrib[idx] += a[idx] * a[lag - idx]

            n_reduce = max(3, n // 20)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, n_reduce))
            candidate[bot_idx] *= np.exp(rng.normal(0.04, 0.015, n_reduce))
        elif r < 0.55:
            # Adjacent block redistribution
            w = max(1, n // 50)
            start = int(rng.integers(0, n - 2 * w))
            candidate[start:start + w] *= np.exp(rng.normal(-0.06, 0.02, w))
            candidate[start + w:start + 2 * w] *= np.exp(rng.normal(0.06, 0.02, w))
        elif r < 0.70:
            # Multi-point perturbation
            n_pts = int(rng.integers(5, max(6, n // 20)))
            pts = rng.choice(n, size=n_pts, replace=False)
            candidate[pts] *= np.exp(rng.normal(0, 0.03, n_pts))
        elif r < 0.85:
            # Anti-symmetric nudge to reduce center peak
            mid = n // 2
            candidate[:mid] *= np.exp(rng.normal(0.015, 0.008, mid))
            rest_size = n - mid
            candidate[mid:] *= np.exp(rng.normal(-0.015, 0.008, rest_size))
        else:
            # Small global noise
            candidate *= np.exp(rng.normal(0, 0.008, n))

        candidate = np.maximum(candidate, 0.0)
        cs = np.sum(candidate)
        if cs < 1e-15:
            candidate = a.copy()
            new_score = current_score
        else:
            candidate = candidate / cs * s
            new_score = score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 30:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    a = candidate
                    current_score = new_score

        temp *= cooling

    return best, best_score


def fft_phase_init(n, rng, n_iter=50):
    """Gerchberg-Saxton: flatten magnitude spectrum for good initial autocorrelation."""
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    a = np.ones(n) / n
    for _ in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-14] = 1e-14
        phase = A / mag
        a_new = np.fft.ifft(phase).real
        a = np.clip(a_new[:n], 0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def propose_candidate(seed=42, budget_s=2400):
    t_global = time.monotonic()
    deadline = t_global + budget_s - 20
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_a = None

    # Load provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        a0 = np.array(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0, 1000)
        s0 = score(a0)
        print(f"Provided sequence: n={len(a0)}, score={s0:.6f}")
        if s0 < best_score:
            best_score = s0
            best_a = a0.copy()
        n0 = len(a0)
    else:
        a0 = np.ones(256)
        best_a = a0.copy()
        best_score = score(a0)
        n0 = 256
        print(f"No provided sequence, using uniform n={n0}, score={best_score:.6f}")

    # Phase 1: Gradient optimize at original length
    print(f"\n=== Phase 1: Gradient descent n={n0} ===")
    t_alloc = min(350.0, deadline - time.monotonic() - 200)
    if t_alloc > 20:
        a1, s1 = gradient_descent(a0, t_alloc, rng)
        print(f"  Result: score={s1:.6f}")
        if s1 < best_score:
            best_score = s1
            best_a = a1.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # Phase 2: Upsample to larger lengths for better 2n/(2n-1) factor
    upsample_targets = []
    for factor in [2, 4, 8]:
        target_n = n0 * factor
        if target_n <= 4096:
            upsample_targets.append(target_n)

    for target_n in upsample_targets:
        if time.monotonic() > deadline - 100:
            break

        print(f"\n=== Phase 2: Upsample & optimize n={target_n} ===")
        a_up = linear_interp(best_a, target_n)
        # Add small noise to escape local minima
        a_up += 0.005 * rng.standard_normal(target_n)
        a_up = np.maximum(a_up, 1e-8)

        t_alloc = min(500.0, deadline - time.monotonic() - 80)
        if t_alloc < 20:
            break

        a2, s2 = gradient_descent(a_up, t_alloc, rng)
        print(f"  Result: score={s2:.6f}")
        if s2 < best_score:
            best_score = s2
            best_a = a2.copy()
            print(f"  *** New best: {best_score:.6f} (n={target_n}) ***")
        else:
            print(f"  (no improvement, keeping n={len(best_a)})")

    # Phase 3: If we have time, try FFT phase init at a good length
    remaining = deadline - time.monotonic() - 60
    if remaining > 120:
        n_fft = min(1024, max(n0, 2 * n0))
        print(f"\n=== Phase 3: FFT init + optimize n={n_fft} ===")
        a_fft = fft_phase_init(n_fft, rng, n_iter=60)
        s_fft = score(a_fft)
        print(f"  FFT init score: {s_fft:.6f}")
        if s_fft < best_score:
            best_score = s_fft
            best_a = a_fft.copy()

        t_alloc = min(remaining - 30, 400.0)
        if t_alloc > 20:
            a3, s3 = gradient_descent(a_fft, t_alloc, rng)
            print(f"  After optimize: score={s3:.6f}")
            if s3 < best_score:
                best_score = s3
                best_a = a3.copy()
                print(f"  *** New best: {best_score:.6f} (n={n_fft}) ***")

    # Phase 4: Local search polish
    remaining = deadline - time.monotonic() - 10
    if remaining > 15:
        print(f"\n=== Phase 4: Local search polish n={len(best_a)} ===")
        t_alloc = min(remaining, 200.0)
        a4, s4 = local_search(best_a, t_alloc, rng)
        print(f"  After local search: score={s4:.6f}")
        if s4 < best_score:
            best_score = s4
            best_a = a4.copy()
            print(f"  *** Final best: {best_score:.6f} ***")

    # Final: verify with evaluate_sequence
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    final_score = evaluate_sequence(final_list)
    print(f"\n{'='*50}")
    print(f"=== FINAL: n={len(final_list)}, evaluate_score={final_score:.6f} ===")
    return final_list
