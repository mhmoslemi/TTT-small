import numpy as np

def fft_linconv(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_linconv(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def softmax_safe(theta):
    """Numerically stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def smooth_objective_and_grad(theta, tau):
    """
    Compute log-sum-exp smoothed objective and gradient w.r.t. theta.
    a = softmax(theta), so sum(a) = 1.
    Objective: tau * log(sum_k exp(c_k / tau)) where c = a*a
    """
    a = softmax_safe(theta)
    n = len(a)
    c = fft_linconv(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    obj = c_max + tau * np.log(sum_exp)
    
    # Weights over lags
    v = exp_shifted / sum_exp  # length 2n-1
    
    # Gradient of smoothed objective w.r.t. a:
    # d/da_j [tau log sum_k exp(c_k/tau)] = 2 * sum_k v_k * a_{k-j} * I[0<=k-j<n]
    # This equals 2 * (cross-correlation of a and v)[j]
    
    # Compute cross-correlation: h[j] = sum_m v[m+j] * a[m] for m=0..n-1
    # Use FFT: h = ifft(conj(FFT(a_pad)) * FFT(v_pad))
    size = 1
    while size < 3 * n:
        size <<= 1
    A = np.fft.rfft(a, size)
    V = np.fft.rfft(v, size)
    h = np.fft.irfft(np.conj(A) * V, size)[:n]
    
    grad_a = 2.0 * h
    
    # Gradient w.r.t. theta: grad_theta = a * (grad_a - dot(a, grad_a))
    inner = np.dot(a, grad_a)
    grad_theta = a * (grad_a - inner)
    
    return obj, grad_theta


def exact_subgrad(theta):
    """Compute exact subgradient of max(a*a) w.r.t. theta."""
    a = softmax_safe(theta)
    n = len(a)
    c = fft_linconv(a, a)
    k_star = int(np.argmax(c))
    
    # d/da_j max(c) = 2 * a_{k_star - j} if 0 <= k_star - j < n
    grad_a = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad_a[j] = 2.0 * a[idx]
    
    inner = np.dot(a, grad_a)
    grad_theta = a * (grad_a - inner)
    return c[k_star], grad_theta

def lbfgs_smoothed(a_init, n, tau_sequence, max_iter=500, time_limit=30.0):
    """
    Run L-BFGS-B on the log-sum-exp smoothed objective with decreasing temperatures.
    Returns best theta and best score.
    """
    theta = np.log(a_init / np.sum(a_init) + 1e-12).astype(np.float64)
    best_score = np.inf
    best_theta = theta.copy()
    t0 = time.monotonic()
    
    for tau in tau_sequence:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Define objective and gradient for this tau
        def fun(x):
            obj, _ = smooth_objective_and_grad(x, tau)
            return obj
        
        def grad(x):
            _, g = smooth_objective_and_grad(x, tau)
            return g
        
        try:
            result = minimize(fun, theta, jac=grad, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-8,
                                      'maxls': 50, 'maxfun': 2000})
            theta = result.x
        except Exception:
            pass
        
        # Evaluate actual score
        a = softmax_safe(theta)
        score = 2.0 * n * np.max(fft_linconv(a, a))  # sum(a)=1
        if score < best_score:
            best_score = score
            best_theta = theta.copy()
    
    return best_theta, best_score / (2.0 * n)  # Convert back to proper score

            top_k = max(3, n // 20)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            # Find bottom contributors
            bottom_idx = np.argsort(contribs)[:max(3, n // 10)]
            
            reduction = rng.uniform(0.03, 0.08)
            amt = np.sum(current[top_idx]) * reduction
            current[top_idx] *= (1.0 - reduction)
            current[bottom_idx] += amt / len(bottom_idx)
            
        elif r < 0.55:
            # Targeted swap: move mass from peak contributor to non-contributor
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            j_red = top_idx[int(rng.integers(0, top_k))]
            # Find positions that DON'T contribute to the peak
            non_contrib = np.where(contribs < np.median(contribs))[0]
            if len(non_contrib) > 0:
                j_inc = non_contrib[int(rng.integers(0, len(non_contrib)))]
                if j_inc != j_red:
                    amt = current[j_red] * rng.uniform(0.05, 0.15)
                    current[j_red] -= amt
                    current[j_inc] += amt
        
        elif r < 0.7:
            # Block scaling: scale a random contiguous block
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.08))
            current[start:start + width] *= factor
        
        elif r < 0.85:
            # Pair flip: find a pair (i, j) where i+j == peak_idx, reduce the larger
            valid_pairs = [(j, peak_idx - j) for j in range(n) if 0 <= peak_idx - j < n]
            if valid_pairs:
                # Sort by product, take top few
                valid_pairs.sort(key=lambda p: current[p[0]] * current[p[1]], reverse=True)
                i, j = valid_pairs[0]
                if i != j:
                    larger = i if current[i] > current[j] else j
                    smaller = j if current[i] > current[j] else i
                    transfer = current[larger] * rng.uniform(0.05, 0.2)
                    current[larger] -= transfer
                    current[smaller] += transfer
        
        else:
            # Small global perturbation
            current *= np.exp(rng.normal(0, 0.01, size=n))
        
        current = np.clip(current, 0.0, None)
        s = np.sum(current)
        if s < 1e-6:
            current = a_init / np.sum(a_init)
            continue
        current = current / s  # Renormalize
        
        new_score = 2.0 * n * np.max(fft_linconv(current, current))
        
        if new_score < current_score:
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-10)):
            current_score = new_score
        
        temp *= cooling
    
    return best, best_score / (2.0 * n)


def coordinate_polish(a_init, time_limit, rng, max_rounds=200):
    """
    Coordinate descent: for each element, try multiplying by factors
    and keep the best.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current = current / np.sum(current)
    current_score = 2.0 * n * np.max(fft_linconv(current, current))
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    factors = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                    break
            if current[j] < 1e-8:
                continue
            
            base = current[j]
            for f in factors:
                trial = base * f
                if trial < 0:
                    continue
                current[j] = trial
                sc = 2.0 * n * np.max(fft_linconv(current, current))
                if sc < current_score:
                    current_score = sc
                    improved = True
                    if current_score < best_score:
                        best_score = current_score
                        best = current.copy()
                else:
                    break
            # Restore if we broke early (keep the best factor tried)
            # Actually we already kept the best, so just continue
        
        if not improved:
            # Random perturbation to escape
            j = int(rng.integers(0, n))
            current[j] = max(0.0, current[j] * np.exp(rng.normal(0, 0.3)))
            s = np.sum(current)
            if s > 1e-6:
                current = current / s
                sc = 2.0 * n * np.max(fft_linconv(current, current))
                if sc < current_score * 1.01:
                    current_score = sc
    
    return best, best_score / (2.0 * n)


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search:
    Phase 1: Multi-start L-BFGS with temperature annealing (60% time)
    Phase 2: Peak-directed local search on best result (20% time)
    Phase 3: Coordinate descent polish (10% time)
    Phase 4: Additional attempts if time remains (10% time)
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 40.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Load provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = compute_score(start_seq)
        print(f"Provided: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()
    
    # Target n values to explore
    n_targets = [512, 384, 640]
    
    # === PHASE 1: Multi-start L-BFGS ===
    print(f"\n{'='*60}")
    print(f"PHASE 1: Multi-start L-BFGS optimization")
    print(f"{'='*60}")
    
    t_phase1 = 0.60 * time_budget
    phase1_deadline = t_global + t_phase1
    
    n = n_targets[0]  # Start with 512
    
    initializations = []
    
    # Generate diverse initializations
    init_methods = [
        make_init_ramp, make_init_bimodal, make_init_cosine,
        make_init_random_binary, make_init_broadband,
        make_init_sawtooth, make_init_quadratic
    ]
    
    for method in init_methods:
        a_init = method(n, rng)
        initializations.append(a_init)
    
    # Also add interpolated version of provided sequence
    if start_seq is not None and len(start_seq) != n:
        x_old = np.linspace(0, 1, len(start_seq))
        x_new = np.linspace(0, 1, n)
        a_interp = np.interp(x_new, x_old, start_seq)
        initializations.append(a_interp)
    elif start_seq is not None and len(start_seq) == n:
        initializations.append(start_seq.copy())
    
    # Add a few random ones
    for _ in range(3):
        a_rand = np.abs(rng.standard_normal(n)) + 0.1
        initializations.append(a_rand)
    
    print(f"  n={n}, {len(initializations)} initializations")
    
    # Temperature sequences (from smooth to sharp)
    # The "spread" of c values is roughly max(c) * 0.1 for a good sequence
    tau_schedules = [
        [0.05, 0.02, 0.01, 0.005, 0.002, 0.001],
        [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001],
        [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005],
    ]
    
    for idx, a_init in enumerate(initializations):
        if time.monotonic() > phase1_deadline:
            break
        
        remaining = phase1_deadline - time.monotonic()
        per_init_time = max(5.0, min(remaining / max(1, len(initializations) - idx), 30.0))
        
        tau_sched = tau_schedules[idx % len(tau_schedules)]
        
        theta_best, score = lbfgs_smoothed(a_init, n, tau_sched, 
                                           max_iter=300, time_limit=per_init_time)
        a_result = softmax_safe(theta_best)
        actual_score = compute_score(a_result)
        
        print(f"  Init {idx}: score={actual_score:.6f}" + 
              (f" *** NEW BEST ***" if actual_score < best_score else ""))
        
        if actual_score < best_score:
            best_score = actual_score
            best_seq = a_result.copy()
    
    # Try n=384 if time permits
    if time.monotonic() < phase1_deadline - 30:
        n2 = n_targets[1]
        print(f"\n  Trying n={n2}...")
        t_n2 = min(40.0, phase1_deadline - time.monotonic() - 10)
        
        for method in [make_init_broadband, make_init_ramp, make_init_sawtooth]:
            if time.monotonic() > phase1_deadline - 10:
                break
            a_init = method(n2, rng)
            tau_sched = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            theta_best, score = lbfgs_smoothed(a_init, n2, tau_sched,
                                             max_iter=300, time_limit=15.0)
            a_result = softmax_safe(theta_best)
            actual_score = compute_score(a_result)
            print(f"  n={n2}, init={method.__name__}: score={actual_score:.6f}" +
                  (f" *** NEW BEST ***" if actual_score < best_score else ""))
            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_result.copy()
    
    # Try n=640 if time permits
    if time.monotonic() < phase1_deadline - 20:
        n3 = n_targets[2]
        print(f"\n  Trying n={n3}...")
        
        for method in [make_init_broadband, make_init_ramp]:
            if time.monotonic() > phase1_deadline - 10:
                break
            a_init = method(n3, rng)
            tau_sched = [0.05, 0.02, 0.01, 0.005, 0.002]
            theta_best, score = lbfgs_smoothed(a_init, n3, tau_sched,
                                             max_iter=200, time_limit=20.0)
            a_result = softmax_safe(theta_best)
            actual_score = compute_score(a_result)
            print(f"  n={n3}, init={method.__name__}: score={actual_score:.6f}" +
                  (f" *** NEW BEST ***" if actual_score < best_score else ""))
            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_result.copy()
    
    print(f"\n  Phase 1 best: {best_score:.6f}")
    
    # === PHASE 2: Peak-directed local search ===
    print(f"\n{'='*60}")
    print(f"PHASE 2: Peak-directed local search")
    print(f"{'='*60}")
    
    if best_seq is not None:
        t_phase2 = min(0.20 * time_budget, max(30.0, deadline - time.monotonic() - 60))
        print(f"  Time: {t_phase2:.0f}s, starting score: {best_score:.6f}, n={len(best_seq)}")
        
        polished, ps_score = local_search_peak_directed(
            best_seq, t_phase2, rng, n_max_evals=80000)
        
        if ps_score < best_score:
            best_score = ps_score
            best_seq = polished.copy()
            print(f"  After local search: {best_score:.6f} *** IMPROVED ***")
        else:
            print(f"  After local search: no improvement ({best_score:.6f})")
    
    # === PHASE 3: Coordinate descent polish ===
    print(f"\n{'='*60}")
    print(f"PHASE 3: Coordinate descent polish")
    print(f"{'='*60}")
    
    if best_seq is not None:
        t_phase3 = min(0.10 * time_budget, max(20.0, deadline - time.monotonic() - 30))
        if t_phase3 > 10:
            print(f"  Time: {t_phase3:.0f}s, starting score: {best_score:.6f}")
            cd_result, cd_score = coordinate_polish(best_seq, t_phase3, rng)
            if cd_score < best_score:
                best_score = cd_score
                best_seq = cd_result.copy()
                print(f"  After CD: {best_score:.6f} *** IMPROVED ***")
            else:
                print(f"  After CD: no improvement ({best_score:.6f})")
    
    # === PHASE 4: Additional attempts ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        t_phase4 = deadline - time.monotonic() - 15
        if t_phase4 > 20:
            print(f"\n{'='*60}")
            print(f"PHASE 4: Additional local search (t={t_phase4:.0f}s)")
            print(f"{'='*60}")
            
            # Run another round of local search with different random seed
            rng2 = np.random.default_rng(seed + 12345)
            polished2, ps2_score = local_search_peak_directed(
                best_seq, t_phase4 * 0.5, rng2, n_max_evals=50000)
            if ps2_score < best_score:
                best_score = ps2_score
                best_seq = polished2.copy()
                print(f"  After additional LS: {best_score:.6f} *** IMPROVED ***")
            
            if time.monotonic() < deadline - 15:
                t_cd2 = deadline - time.monotonic() - 15
                if t_cd2 > 10:
                    cd2, cd2_score = coordinate_polish(best_seq, t_cd2, rng2)
                    if cd2_score < best_score:
                        best_score = cd2_score
                        best_seq = cd2.copy()
                        print(f"  After final CD: {best_score:.6f} *** IMPROVED ***")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'='*60}")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
