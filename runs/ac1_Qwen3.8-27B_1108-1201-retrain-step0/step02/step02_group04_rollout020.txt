```python
import numpy as np
import time

def fft_convolve_square(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def project_simplex(v, S):
    """Project v onto {a >= 0, sum(a) = S}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_step(a, c, lr, active_frac=0.1):
    """Take one subgradient step to minimize max(c)."""
    n = len(a)
    c_max = np.max(c)

    active = np.where(c >= c_max * (1 - active_frac))[0]
    # Compute subgradient: average of gradients at active peaks
```

Looking at this problem, I need to minimize `2*n*max(conv(a,a)) / (sum(a))^2` where the key insight is that we want the self-convolution to be as "flat" as possible. The theoretical lower bound is `2n/(2n-1)` (approaching 1 for large n), and the previous best was 1.512.

My approach uses projected subgradient descent with FFT-accelerated gradient computation, momentum, and multiple restarts with diverse initializations. The gradient of `conv(a,a)[k]` w.r.t. `a[j]` is `2*a[k-j]` (zero-padded), and the subgradient of the max is computed via cross-correlation with the active peak indicator (also FFT-accelerated).

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using FFT-accelerated projected subgradient
descent with momentum, adaptive active sets, and multi-restart strategy.
Key: flatten the self-convolution as much as possible under non-negativity.
"""

import numpy as np
import time


def fft_conv_square(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    if not np.any(cond):
        return np.full(n, S / n)
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fft(a, active_indices, fft_size):
    """Compute average subgradient at active peaks using FFT cross-correlation."""
    n = len(a)
    h = np.zeros(fft_size)
    h[active_indices] = 1.0
    Fa = np.fft.rfft(a, fft_size)
    Fh = np.fft.rfft(h, fft_size)
    corr = np.fft.irfft(Fh * np.conj(Fa), fft_size)
    g = corr[:n] * (2.0 / len(active_indices))
    return g


def score_from_conv(n, c_max, S):
    """Compute score from precomputed values."""
    return 2.0 * n * c_max / (S * S)


def optimize_subgrad(a_init, time_limit, lr_init=0.1, active_frac=0.05,
                     momentum=0.5, seed=42):
    """Projected subgradient descent with momentum to minimize max(conv(a,a))."""
    rng = np.random.default_rng(seed)
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = a / np.sum(a) * S

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    c = fft_conv_square(a)
    best = a.copy()
    best_score = score_from_conv(n, np.max(c), S)

    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    step = 0
    no_improve = 0
    patience = 3000
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_conv_square(a)
        c_max = np.max(c)
        cur_score = score_from_conv(n, c_max, S)

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Active set: peaks within a fraction of max
        threshold = c_max * (1.0 - active_frac)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Compute subgradient via FFT
        g = compute_subgradient_fft(a, active, fft_size)

        # Adaptive step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        lr = lr_init / (1.0 + 0.005 * step)

        # Momentum update
        v = momentum * v - lr * g
        a_new = a + v
        a_new = project_simplex(a_new, S)
        a = a_new
        step += 1

        # Restart if stuck
        if no_improve > patience:
            restart_count += 1
            pert_scale = 0.05 * S / np.sqrt(n)
            a = best + pert_scale * rng.standard_normal(n)
            a = project_simplex(a, S)
            v = np.zeros(n)
            no_improve = 0
            # Vary active fraction on restart
            active_frac = min(0.3, active_frac * 1.5)

        if step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      step={step}, cur={cur_score:.8f}, best={best_score:.8f}, "
                  f"restarts={restart_count}, time={elapsed:.1f}s")

    return best, best_score


def try_initialization(n, init_type, rng):
    """Generate an initial sequence of length n."""
    if init_type == 'uniform':
        a = np.ones(n)
    elif init_type == 'uniform_noise':

    elif init_type == 'gaussian_center':
        x = np.arange(n)
        center = (n - 1) / 2.0
        a = np.exp(-0.5 * ((x - center) / (n * 0.3)) ** 2)
    elif init_type == 'cosine':
        x = np.arange(n)
        a = 0.5 + 0.5 * np.cos(np.pi * x / (n - 1 + 1e-10))
        a = np.abs(a) + 0.1
    elif init_type == 'bimodal':
        x = np.arange(n)
        c1 = n * 0.25
        c2 = n * 0.75
        a = np.exp(-0.5 * ((x - c1) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((x - c2) / (n * 0.1)) ** 2)
    elif init_type == 'linear_ramp':
        a = np.linspace(1, 3, n)
    elif init_type == 'inverted_v':
        x = np.arange(n)
        center = (n - 1) / 2.0
        a = 1.0 + np.abs(x - center) / center
    elif init_type == 'sine_mod':
        x = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x / n)

        a = 1.0 + 0.5 * np.sin(2 * np.pi * x / n) + 0.2 * np.sin(4 * np.pi * x / n)

    else:
        a = np.ones(n)
    a = np.clip(a, 0.0, 1000.0)
    if np.sum(a) < 1e-6:
        a = np.ones(n)
    return a
```
