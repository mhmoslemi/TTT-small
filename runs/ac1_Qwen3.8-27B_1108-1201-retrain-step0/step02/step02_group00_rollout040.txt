def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w, idx = np.sort(v)[::-1], np.arange(1, n+1, dtype=np.float64)
    cumsum = np.cumsum(w)
    theta = (cumsum - S) / idx
    valid = theta <= w
    rho = int(np.max(np.where(valid)[0])) + 1 if np.any(valid) else n
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0])) + 1
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_weighted(a, w):
    """
    Gradient of sum_k w[k]*c[k] w.r.t. a, where c = a*a (self-convolution).
    Uses FFT: grad = 2 * ifft(FFT(w) * conj(FFT(a)))[:n]
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

```python
"""
Minimize 2*n*max(f*f)/(sum f)^2 over non-negative sequences of length n.

Algorithm:
1. Collect diverse initializations (given sequence, random, structured).
2. Score all initializations.
3. Run FISTA with adaptive log-sum-exp (beta: 50→3000) on top candidates.
4. Polish the best with:
   a. Very high-beta FISTA (beta: 1000→10000)
   b. Top-k subgradient (k: 50→2048)
   c. Targeted deflation
   d. Final fine FISTA with perturbation
5. Verify with evaluate_sequence and fallback if needed.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_c = (cumsum - S) / idx
    valid = theta_c <= w
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0])) + 1
    else:
        rho = n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Self-convolution via FFT, length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_weighted(a, w):
    """Gradient of sum_k w[k]*(a*a)[k] w.r.t. a, via FFT."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def score_sequence(a):
    """Compute 2n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (s * s))


def normalize_to_sum(a, S):
    """Clip to non-negative and rescale to sum S."""
    a = np.clip(a, 0.0, None).copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def fista_lse(a_init, n, S, max_time, rng, beta_start, beta_end, step_decay=0.001):
    """FISTA with log-sum-exp smooth approximation to max."""
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(max_time, 1e-6))
        beta = beta_start * (beta_end / beta_start) ** (frac ** 0.7)
        
        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
        
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        tot = exp_bc.sum()
        if tot < 1e-300:
            break
        w = exp_bc / tot
        
        grad = fft_grad_weighted(y, w)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        it = int(elapsed / 0.002)  # approximate iteration count
        step = 1.0 / (2.0 * beta * (1.0 + step_decay * it))
        step = max(step, 1e-12)
        
        x = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        a = x
        t = t_new
    
    return best_a, best


def toppk_subgrad(a_init, n, S, max_time, rng, k_start=20, k_end=1024):
    """Subgradient with top-k weighting, adaptive step."""
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(max_time, 1e-6))
        k = int(k_start + (k_end - k_start) * frac)
        
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()
        
        top_idx = np.argsort(c)[::-1][:k]
        top_vals = c[top_idx]
        tw = top_vals ** 2
        tw_sum = tw.sum()
        if tw_sum < 1e-30:
            break
        weights = tw / tw_sum
        
        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = weights
        
        grad = fft_grad_weighted(a, w_full)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        step = 1.5 * S / (n * (1.0 + 0.02 * elapsed))
        a = project_simplex(a - step * grad / max(gn, 1e-15), S)
    
    return best_a, best


def deflation_step(a_init, n, S, max_time, rng):
    """Targeted deflation: reduce peaks while maintaining sum."""
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()
        
        c_max = c.max()
        c_mean = c.mean()
        # Weight: above-mean values get more weight, peaks get extra
        w = np.maximum(c - c_mean, 0) ** 2
        top_n = max(5, len(c) // 50)
        top_idx = np.argsort(c)[::-1][:top_n]
        w[top_idx] += c_max
        ws = w.sum()
        if ws < 1e-30:
            break
        w = w / ws
        
        grad = fft_grad_weighted(a, w)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        step = 1.0 * S / (n * (1.0 + 0.01 * (time.monotonic() - start)))
        a = project_simplex(a - step * grad / max(gn, 1e-15), S)
    
    return best_a, best

```python
"""
Minimize 2*n*max(f*f)/(sum f)^2 over non-negative sequences.

Algorithm:
- Phase 1: Score diverse initializations (given, random, structured, multiple lengths).
- Phase 2: FISTA with adaptive log-sum-exp (beta 50→3000) on top candidates.
- Phase 3: Multi-method polishing (high-beta FISTA, top-k subgrad, deflation, final fine FISTA).
- FFT-accelerated gradients throughout for O(n log n) per iteration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_c = (cumsum - S) / idx
    valid = theta_c <= w
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0])) + 1
    else:
        rho = n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a, fsize=None):
    """Self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    if fsize is None:
        fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_weighted(a, w, fsize=None):
    """Gradient of sum_k w[k]*(a*a)[k] w.r.t. a, via FFT cross-correlation."""
    n = len(a)
    if fsize is None:
        fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def score_sequence(a):
    """Compute 2n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (s * s))


def normalize_to_sum(a, S):
    """Clip to non-negative and rescale to sum S."""
    a = np.clip(a, 0.0, None).copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) * S / len(a)
    return a * (S / s)


def fista_lse(a_init, n, S, max_time, rng, beta_start, beta_end, step_decay=0.001,
              noise_period=4000, noise_prob=0.08):
    """FISTA with log-sum-exp smooth approximation to max(self-conv)."""
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    fsize_c = 1 << (2 * n - 2).bit_length()
    fsize_g = 1 << (3 * n - 3).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(max_time, 1e-6))
        beta = beta_start * (beta_end / beta_start) ** (frac ** 0.65)

        A_y = fft(y, fsize_c)
        c = np.real(ifft(A_y * A_y))[:2 * n - 1]

        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()

        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        tot = exp_bc.sum()
        if tot < 1e-300:
            break
        w = exp_bc / tot

        W = fft(w, fsize_g)
        A_y_g = fft(y, fsize_g)
        grad = 2.0 * np.real(ifft(W * np.conj(A_y_g)))[:n]
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        step = 1.0 / (2.0 * beta * (1.0 + step_decay * it))
        step = max(step, 1e-12)

        x = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        a = x
        t = t_new

        if it % noise_period == 0 and rng.random() < noise_prob:
            ns = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, ns, n), S)
            t = 1.0

    return best_a, best
