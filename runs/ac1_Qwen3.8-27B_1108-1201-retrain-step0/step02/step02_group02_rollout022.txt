- `theta_high = max(v)` (ensures sum < S... actually sum = 0 at theta = max(v))

Actually, `sum(max(v_i - theta, 0))` is a continuous, monotonically decreasing function of `theta` that goes from `sum(v)` (at `theta = min(v)`) to `0` (at `theta = max(v)`). We want it to equal `S`.

If `sum(v) < S`, then we need `theta < min(v)` (which means all `v_i - theta > 0`), so `theta = mean(v) - S/n`... wait no.

If all `v_i - theta > 0` (i.e., `theta < min(v)`), then `sum(v_i - theta) = sum(v) - n*theta = S`, so `theta = (sum(v) - S) / n`.

This is the solution when no coordinates are at zero. If `theta >= min(v)`, then some coordinates are clamped to zero, and we need to solve implicitly.

The standard algorithm for projection onto the simplex (with non-negativity) is:
1. Sort `v` in descending order: `v_(1) >= v_(2) >= ... >= v_(n)`
2. For `k = 1, ..., n`, define `theta_k = (sum_{i=1}^k v_(i) - S) / k`
3. Find the largest `k` such that `v_(k) > theta_k`
4. Set `theta = theta_k`
5. Return `max(v - theta, 0)`

This is an O(n log n) algorithm (dominated by the sort).

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def project_simplex(v, S):
    """Project v onto {a >= 0, sum(a) = S}."""
    n = len(v)
    # Sort in descending order
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    # Find largest k such that sorted_v[k-1] > (cumsum[k-1] - S) / k
    # i.e., k * sorted_v[k-1] > cumsum[k-1] - S
    # i.e., k * sorted_v[k-1] + S > cumsum[k-1]
    k_range = np.arange(1, n+1)
    condition = sorted_v * k_range + S > cumsum
    k = np.max(np.where(condition)[0]) + 1  # 1-indexed
    theta = (cumsum[k-1] - S) / k
    return np.maximum(v - theta, 0.0)

def softmax_gradient(a, tau):
    """Compute soft-max value and gradient."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    
    # Gradient: 2 * (a_rev * w) evaluated at positions n-1 to 2n-2
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    grad = 2.0 * corr[n-1:2*n-1]
    
    return value, grad, c, w

def optimize_projected(n, a_init, time_limit, rng):
    """Projected gradient descent with adaptive temperature."""
    S = np.sqrt(2.0 * n)
    a = project_simplex(a_init, S)
    current_score = score(a)
    best_a = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    step = 0
    momentum = np.zeros(n)
    beta = 0.9
    
    # Adam-like buffering
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    no_improve = 0
    patience = 2000
    
    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        
        # Temperature schedule
        if elapsed < 0.5:
            tau = (50.0 / n) * (0.1 / n) ** (elapsed / 0.5)
        elif elapsed < 0.8:
            tau = (0.1 / n) * (0.01 / n) ** ((elapsed - 0.5) / 0.3)
        else:
            tau = 0.001 / n
        
        value, grad, c, w = softmax_gradient(a, tau)
        
        # Adam update
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * grad**2
        m_hat = m_buf / (1 - beta1**step)
        v_hat = v_buf / (1 - beta2**step)
        
        # Learning rate
        base_lr = 0.5 / np.sqrt(step + 1)
        if elapsed < 0.3:
            lr = base_lr * 2.0
        elif elapsed < 0.7:
            lr = base_lr
        else:
            lr = base_lr * 0.3
        
        # Step + project
        v_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(v_new, S)
        
        # Check score periodically
        if step % 100 == 0:
            s = score(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 100
            
            # Escape mechanism
            if no_improve > patience:
                r = rng.random()
                if r < 0.3:
                    a = project_simplex(best_a + 0.1 * rng.standard_normal(n), S)
                elif r < 0.6:
                    # Target peak reduction
                    peak_idx = np.argmax(c)
                    # Reduce elements that contribute to peak
                    contrib = np.zeros(n)
                    for i in range(n):
                        if 0 <= peak_idx - i < 2*n-1:
                            contrib[i] = a[i] * (a[peak_idx - i] if 0 <= peak_idx - i < n else 0)
                    top_k = np.argsort(contrib)[-max(4, n//8):]
                    a[top_k] *= 0.9
                    a = project_simplex(a, S)
                else:
                    # Random block perturbation
                    width = max(2, n // 16)
                    start = rng.integers(0, n - width)
                    a[start:start+width] *= np.exp(rng.normal(0, 0.1, width))
                    a = project_simplex(a, S)
                no_improve = 0
            
            if step % 5000 == 0:
                print(f"  step={step}, score={s:.6f}, best={best_score:.6f}, tau={tau:.2e}")
    
    # Final check
    s = score(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    
    return best_a, best_score

                else:
                    # Invert a random block (swap with another block)
                    w1 = max(3, n // 16)
                    s1 = int(rng.integers(0, max(1, n - w1)))
                    s2 = int(rng.integers(0, max(1, n - w1)))
                    if abs(s1 - s2) > w1:
                        tmp = a[s1:s1 + w1].copy()
                        a[s1:s1 + w1] = a[s2:s2 + w1]
                        a[s2:s2 + w1] = tmp
                        a = project_simplex(a, S)
                no_improve = 0

            if step % 5000 == 0:
                print(f"    PG step={step}, score={s:.6f}, best={best_score:.6f}, tau={tau:.2e}")

    s = compute_score(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    return best_a, best_score


def scipy_polish(n, a_init, time_limit, rng):
    """Polish using scipy L-BFGS-B on log-parameterized variables."""
    S = np.sqrt(2.0 * n)

    def objective_and_grad(theta):
        a = np.exp(theta - np.max(theta))
        a = a / np.sum(a) * S
        tau = 0.005 / n
        value, grad_a, _, _ = softmax_gradient(a, tau)
        inner = np.dot(a, grad_a)
        grad_theta = a * (grad_a - inner / S)
        return value, grad_theta

    a0 = np.clip(a_init, 1e-10, None)
    a0 = a0 / np.sum(a0) * S
    theta0 = np.log(a0 + 1e-15)

    t0 = time.monotonic()
    maxiter = max(100, int(time_limit * 50))

    result = minimize(
        objective_and_grad, theta0, method='L-BFGS-B', jac=True,
        options={'maxiter': maxiter, 'maxfun': maxiter * 2,
                 'ftol': 1e-14, 'gtol': 1e-12, 'iprint': -1}
    )

    a_final = np.exp(result.x - np.max(result.x))
    a_final = a_final / np.sum(a_final) * S
    a_final = np.clip(a_final, 0.0, 1000.0)
    return a_final, compute_score(a_final)


def simulated_anneal(n, a_init, time_limit, rng):
    """Simulated annealing with structured moves for peak reduction."""
    S = np.sqrt(2.0 * n)
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score

    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.99998
    step = 0

    while time.monotonic() - t0 < time_limit and temp > 1e-8:
        step += 1
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))

        r = rng.random()
        if r < 0.35:
            # Reduce peak contributors, redistribute
            contrib = np.zeros(n)
            for i in range(n):
                km = peak_idx - i
                if 0 <= km < n:
                    contrib[i] = a[i] * a[km]
            top_k = max(3, n // 20)
            bottom_k = max(6, n // 10)
            top_idx = np.argsort(contrib)[::-1][:top_k]
            bottom_idx = np.argsort(contrib)[:bottom_k]
            reduction = rng.uniform(0.01, 0.05)
            a[top_idx] *= (1.0 - reduction)
            a[bottom_idx] *= (1.0 + reduction * len(top_idx) / len(bottom_idx))
        elif r < 0.55:
            # Pair swap
            top_k = max(4, n // 16)
            contrib = np.zeros(n)
            for i in range(n):
                km = peak_idx - i
                if 0 <= km < n:
                    contrib[i] = a[i] * a[km]
            top_idx = np.argsort(contrib)[::-1][:top_k]
            j_red = top_idx[int(rng.integers(0, top_k))]
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red:
                amt = a[j_red] * rng.uniform(0.01, 0.06)
                a[j_red] -= amt
                a[j_inc] += amt
        elif r < 0.7:
            # Scale random block
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, max(1, n - width)))
            factor = np.exp(rng.normal(0, 0.04))
            a[start:start + width] *= factor
        elif r < 0.85:
            # Authenticate: smooth a random segment
            width = max(3, n // 32)
            start = int(rng.integers(0, max(1, n - width)))
            seg = a[start:start + width]
            alpha = rng.uniform(0.1, 0.4)
            for i in range(1, width - 1):
                seg[i] = (1 - alpha) * seg[i] + alpha * 0.5 * (seg[i - 1] + seg[i + 1])
            a[start:start + width] = seg
        else:
            # Global gentle perturbation
            a *= np.exp(rng.normal(0, 0.003, size=n))

        a = np.clip(a, 0.0, 1000.0)
        a = project_simplex(a, S)
        new_score = compute_score(a)

        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
                current_score = new_score

        temp *= cooling
        if step % 2000 == 0:
            print(f"    SA step={step}, cur={current_score:.6f}, best={best_score:.6f}, temp={temp:.2e}")

    return best_a, best_score


def make_init(n, rng, method):
    """Create diverse initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if method == 'flat_noise':
        a = 1.0 + 0.15 * rng.standard_normal(n)
    elif method == 'u_shaped':
        x = 2.0 * t / (n - 1) - 1.0
        alpha = rng.uniform(0.3, 1.0)
        a = 1.0 + alpha * x ** 2
    elif method == 'broadband':
        a = np.ones(n)
        for k in range(1, 12):
            freq = k * (1.0 + 0.5 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.12 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.03 * rng.standard_normal(n)
    elif method == 'sinusoidal':
        freq = rng.uniform(1.0, 3.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
    elif method == 'two_tone':
        f1 = rng.uniform(1.0, 2.0)
        f2 = rng.uniform(3.0, 8.0)
        a = 1.0 + 0.2 * np.cos(2 * np.pi * f1 * t / n) + 0.1 * np.cos(2 * np.pi * f2 * t / n)
    elif method == 'arch':
        x = 2.0 * t / (n - 1) - 1.0
        a = 1.0 - rng.uniform(0.2, 0.5) * x ** 2
        a += 0.02 * rng.standard_normal(n)
    else:
        a = 1.0 + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: projected gradient, scipy polish, simulated annealing,
    across multiple sequence lengths and initializations.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = compute_score(start_seq)
        print(f"Provided: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Define lengths to try and time allocation
    n_values = [256, 320, 384, 192]
    # Weight by priority: more time for promising lengths
    weights = [0.30, 0.25, 0.25, 0.20]

    methods = ['flat_noise', 'u_shaped', 'broadband', 'sinusoidal', 'two_tone', 'arch']

    for n_idx, n in enumerate(n_values):
        if time.monotonic() > deadline - 20:
            break
        t_alloc = weights[n_idx] * time_budget
        t_alloc = min(t_alloc, deadline - time.monotonic() - 10)
        if t_alloc < 15:
            continue

        print(f"\n{'=' * 50}")
        print(f"Optimizing n={n}, time_budget={t_alloc:.0f}s")
        print(f"{'=' * 50}")

        # Generate initializations
        inits = []
        if start_seq is not None and len(start_seq) > 0:
            x_old = np.linspace(0, 1, len(start_seq))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, start_seq))
        inits.append(make_init(n, rng, 'flat_noise'))
        inits.append(make_init(n, rng, 'u_shaped'))
        inits.append(make_init(n, rng, 'broadband'))
        if n_idx == 0:
            inits.append(make_init(n, rng, 'sinusoidal'))
            inits.append(make_init(n, rng, 'arch'))

        # Split time among initializations and methods
        n_starts = len(inits)
        t_per_start = t_alloc / (2 * n_starts)  # Half for gradient, half for polish

        for init_idx, a_init in enumerate(inits):
            if time.monotonic() > deadline - 15:
                break

            t_grad = min(t_per_start, deadline - time.monotonic() - 10)
            if t_grad < 5:
                continue

            print(f"  Start {init_idx + 1}/{n_starts} (t={t_grad:.0f}s)")

            # Phase A: Projected gradient
            a_res, s_res = optimize_projected(n, a_init, t_grad, rng)
            if s_res < best_score:
                best_score = s_res
                best_seq = a_res.copy()
                print(f"    PG improved: {best_score:.6f}")

            # Phase B: SA polish on the best
            if time.monotonic() > deadline - 15:
                break
            t_sa = min(t_per_start * 0.5, deadline - time.monotonic() - 10)
            if t_sa > 5:
                a_sa, s_sa = simulated_anneal(n, a_res, t_sa, rng)
                if s_sa < best_score:
                    best_score = s_sa
                    best_seq = a_sa.copy()
                    print(f"    SA improved: {best_score:.6f}")

            # Phase C: Scipy polish
            if time.monotonic() > deadline - 15:
                break
            t_sci = min(t_per_start * 0.5, deadline - time.monotonic() - 10)
            if t_sci > 5:
                a_sci, s_sci = scipy_polish(n, a_res, t_sci, rng)
                if s_sci < best_score:
                    best_score = s_sci
                    best_seq = a_sci.copy()
                    print(f"    SciPy improved: {best_score:.6f}")

    # Final polish on the overall best
    if best_seq is not None and time.monotonic() < deadline - 15:
        t_final1 = min(deadline - time.monotonic() - 30, 120.0)
        if t_final1 > 10:
            print(f"\n{'=' * 50}")
            print(f"FINAL SA POLISH (t={t_final1:.0f}s)")
            print(f"{'=' * 50}")
            a_fp, s_fp = simulated_anneal(len(best_seq), best_seq, t_final1, rng)
            if s_fp < best_score:
                best_score = s_fp
                best_seq = a_fp.copy()
                print(f"  SA final: {best_score:.6f}")

        if time.monotonic() < deadline - 15:
            t_final2 = min(deadline - time.monotonic() - 15, 60.0)
            if t_final2 > 10:
                print(f"\nFINAL SCIPY POLISH (t={t_final2:.0f}s)")
                a_fp2, s_fp2 = scipy_polish(len(best_seq), best_seq, t_final2, rng)
                if s_fp2 < best_score:
                    best_score = s_fp2
                    best_seq = a_fp2.copy()
                    print(f"  SciPy final: {best_score:.6f}")

    # Verify with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'=' * 50}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'=' * 50}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
