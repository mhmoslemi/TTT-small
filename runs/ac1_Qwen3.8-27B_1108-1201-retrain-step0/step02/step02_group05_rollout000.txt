"""
LP-based iterative optimization for minimum peak convolution of non-negative sequences.
Core idea: At each step, solve an LP to find b that maximizes sum(b) subject to conv(f,b) <= M,
then interpolate f toward b. This monotonically reduces the peak. Combined with gradient-based
refinement and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def build_conv_matrix(f):
    """Build A where A[k,j] = f[k-j] if 0<=k-j<n, else 0. Shape (2n-1, n)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[k, j] = f[k - j]
    return A


def solve_lp_for_b(f, M):
    """Solve: max sum(b) s.t. conv(f,b) <= M (componentwise), b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 5})
    if res.status == 0:
        return res.x
    return None


def lp_phase(f_init, time_limit, verbose=True):
    """Run LP-based iterations to reduce peak. Returns (best_f, best_M)."""
    f = f_init.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = np.sum(f)
    if s < 1e-15:
        f = np.ones(len(f))
        s = 1.0
    f = f / s
    M = np.max(fft_convolve(f, f))
    best_f = f.copy()
    best_M = M
    t0 = time.monotonic()
    no_improve_count = 0
    step_sizes = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        b = solve_lp_for_b(f, M)
        if b is None or np.sum(b) < 1e-10:
            break

        b_norm = b / np.sum(b)
        improved = False

        for t in step_sizes:
            f_try = (1 - t) * f + t * b_norm
            f_try = np.maximum(f_try, 0)
            s_try = np.sum(f_try)
            if s_try < 1e-10:
                continue
            f_try = f_try / s_try
            M_try = np.max(fft_convolve(f_try, f_try))
            if M_try < M * (1 - 1e-10):
                f = f_try
                M = M_try
                improved = True
                break

        if improved:
            if M < best_M:
                best_M = M
                best_f = f.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 20:
                break

        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * len(f) * M
            print(f"    LP iter {iter_count}: score={score:.6f}, M={M:.8f}, elapsed={elapsed:.0f}s")

    if verbose:
        print(f"    LP phase done: {iter_count} iters, best score = {2.0*len(f)*best_M:.6f}")
    return best_f, best_M


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def soft_max_obj_grad(theta, tau):
    """Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau)."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score = 2.0 * n * soft_max_val
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def gradient_refine(f_init, time_limit, verbose=True):
    """L-BFGS refinement using p-norm annealing then soft-max."""
    n = len(f_init)
    theta = np.log(np.maximum(f_init, 1e-30))
    best_score = local_score(softmax(theta))
    best_theta = theta.copy()
    t0 = time.monotonic()

    # Phase A: p-norm annealing
    p_schedule = [4, 8, 16, 32, 64, 128, 256]
    for p in p_schedule:
        if time.monotonic() - t0 > time_limit * 0.6:
            break
        try:
            result = minimize(pnorm_obj_grad, theta.copy(), method='L-BFGS-B', jac=True,
                              args=(p,),
                              options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': 1000})
            s = local_score(softmax(result.x))
            if s < best_score:
                best_score = s
                best_theta = result.x.copy()
                theta = result.x
                if verbose:
                    print(f"      p={p}: score={s:.6f}")
        except Exception:
            break

    # Phase B: soft-max refinement
    tau_schedule = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        try:
            result = minimize(soft_max_obj_grad, best_theta.copy(), method='L-BFGS-B', jac=True,
                              args=(tau,),
                              options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': 1500})
            s = local_score(softmax(result.x))
            if s < best_score:
                best_score = s
                best_theta = result.x.copy()
                if verbose:
                    print(f"      tau={tau}: score={s:.6f}")
        except Exception:
            break

    best_a = softmax(best_theta)
    return best_a, best_score


def local_search_polish(f_init, time_limit, rng):
    """Targeted local search with peak-aware perturbations."""
    f = f_init.astype(np.float64).copy()
    f = np.maximum(f, 1e-12)
    f = f / np.sum(f)
    best = f.copy()
    best_score = local_score(f)
    current_score = best_score
    n = len(f)
    t0 = time.monotonic()
    temperature = 0.001
    cooling = 0.9997

    while time.monotonic() - t0 < time_limit:
        candidate = f.copy()
        r = rng.random()

        if r < 0.35:
            # Peak-targeted: reduce contributors to the peak
            c = fft_convolve(f, f)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(5, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.55:
            # Symmetric perturbation
            candidate = 0.7 * f + 0.3 * f[::-1]
            candidate *= np.exp(rng.normal(0, 0.02, n))
        elif r < 0.70:
            # Block reduction at high-contribution regions
            c = fft_convolve(f, f)
            c_threshold = np.mean(c) + 0.5 * np.std(c)
            high_regions = []
            for k in range(len(c)):
                if c[k] > c_threshold:
                    for i in range(n):
                        j = k - i
                        if 0 <= j < n and f[i] * f[j] > 0:
                            pass
            # Reduce values that contribute to multiple high conv positions
            pull = np.zeros(n)
            for k in range(len(c)):
                if c[k] > c_threshold:
                    for i in range(min(n, k + 1)):
                        j = k - i
                        if 0 <= j < n:
                            pull[i] += f[j]
                            pull[j] += f[i]
            if np.max(pull) > 1e-15:
                pull = pull / np.max(pull)
                candidate *= np.exp(-0.03 * pull)
        elif r < 0.85:
            # Smooth global perturbation
            candidate *= np.exp(rng.normal(0, 0.02, n))
        else:
            # Mirror + noise
            mirror = 0.5 * (f + f[::-1])
            candidate = mirror * np.exp(rng.normal(0, 0.01, n))

        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = f.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            f = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 1e-10)

    return best, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_u_shape_init(n, depth, rng):
    """U-shaped initialization (high at edges, low in middle)."""
    t = np.linspace(-1, 1, n)
    a = 1.0 + depth * t * t
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def interpolate_sequence(src, n):
    """Interpolate a sequence to length n."""
    x_old = np.linspace(0, 1, len(src))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, src)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)

                best_score = score_final
                best_seq = f_final.copy()
                print(f"  Final LP: {best_score:.6f} *** NEW BEST ***")

    # Final verification
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_seq)} ===")
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    eval_score = evaluate_sequence(final_list)
    print(f"  evaluate_sequence confirms: {eval_score:.6f}")
    return final_list
