"""
LP-based block coordinate descent with mirror descent and local search
for minimizing 2*n*max(auto-convolution)/(sum)^2.
Key innovation: uses linear programming to compute optimal updates
in a block coordinate descent framework.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_val(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_block(a):
    """
    Solve the LP: minimize M subject to (a*b)_k <= M for all k,
    sum(b) = 1, b >= 0.
    Returns the optimal b.
    """
    n = len(a)
    # Variables: [b_0, ..., b_{n-1}, M]
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: (a*b)_k - M <= 0 for k = 0, ..., 2n-2
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        i_lo = max(0, k - (n-1))
        i_hi = min(n-1, k)
        for i in range(i_lo, i_hi+1):
            j = k - i  # index of b
            A_ub[k, j] = a[i]
        A_ub[k, -1] = -1.0
    
    # Equality constraint: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_j in [0, inf], M in [0, inf]
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    b = res.x[:n]
    b = np.maximum(b, 0)
    s = b.sum()
    if s < 1e-12:
        return None
    b /= s
    return b

def lp_descent(a, time_limit, t_init=0.5, max_iters=5000):
    """
    LP-based block coordinate descent.
    Alternates: fix a, solve for b; then mix.
    """
    a = np.maximum(a, 0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    t0 = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = solve_lp_block(a)
        if b is None:
            t *= 0.5
            continue
        
        # Try full update first
        new_a = b
        new_conv = fast_convolve(new_a)
        new_obj = 2.0 * n * new_conv.max()
        
        if new_obj < best_obj:
            best_obj = new_obj
            best_a = new_a.copy()
        
        # Mix with current
        if t < 1.0:
            mixed_a = (1 - t) * a + t * b
            mixed_a = np.maximum(mixed_a, 0)
            ms = mixed_a.sum()
            if ms > 1e-12:
                mixed_a /= ms
            mixed_conv = fast_convolve(mixed_a)
            mixed_obj = 2.0 * n * mixed_conv.max()
            
            if mixed_obj < new_obj:
                new_a = mixed_a
                new_obj = mixed_obj
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_a = new_a.copy()
        
        a = new_a
        iter_count += 1
        
        # Adapt step size
        if iter_count % 100 == 0:
            t = min(1.0, t * 1.02)  # gradually increase to full update
    
    return best_a, best_obj

def mirror_descent(a0, time_limit, t_init=50.0, eta_init=0.3):
    """Multiplicative mirror descent on the simplex."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    max_pat = 80
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient
        sg = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-10:
                continue
            i_lo = max(0, k - n + 1)
            i_hi = min(n-1, k)
            for i in range(i_lo, i_hi+1):
                j = k - i
                sg[i] += w[k] * a[j]
        
        # Center the subgradient
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_pat:
            patience = 0
            eta *= 0.6
            t *= 1.4
            if eta < 0.003:
                elapsed = time.monotonic() - t0
                frac = min(1.0, elapsed / max(time_limit, 1.0))
                eta = eta_init * (1.0 - 0.4 * frac)
                t = t_init * (1.0 + 4.0 * frac)
    
    return best_a, best_obj

def local_search(a0, time_limit, rng):
    """Informed local search targeting peak reduction."""
    a = np.maximum(a0, 0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        frac = (time.monotonic() - t0) / max(time_limit, 1.0)
        frac = min(1.0, frac)
        T = 0.01 * (1.0 - frac)**2 + 0.0003
        
        # Find peak positions
        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.7
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        
        k = int(rng.choice(peak_positions))
        i_lo = max(0, k - n + 1)
        i_hi = min(n-1, k)
        
        # Contribution of each a_i to c_k
        contrib = np.zeros(n)
        for i in range(i_lo, i_hi+1):
            j = k - i
            contrib[i] = a[i] * a[j]
        
        total = contrib.sum()
        if total < 1e-15:
            continue
        
        # Source: weighted by contribution to peak
        probs = contrib / total
        probs /= probs.sum()
        src = int(rng.choice(n, p=probs))
        
        # Destination: random, preferring low-activity positions
        dst = int(rng.integers(n))
        if dst == src:
            continue
        
        # Transfer amount
        frac_transfer = rng.uniform(0.02, 0.1)
        delta = a[src] * frac_transfer
        if delta < 1e-12:
            continue
        
        a[src] -= delta
        a[dst] += delta
        if a[src] < 0:
            a[src] += delta
            a[dst] -= delta
            continue
        
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            accept_prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < accept_prob:
                cur_obj = new_obj
                conv = new_conv
            else:
                a[src] += delta
                a[dst] -= delta
    
    return best_a, best_obj

"""
LP-based block coordinate descent combined with mirror descent and informed local search
for minimizing 2*n*max(auto-convolution)/(sum)^2.
Key innovation: uses linear programming to compute exact optimal updates in a
block coordinate descent framework, alternating between the two factors in the
bilinear auto-convolution form.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_val(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project vector onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_maxsum(a, M):
    """
    Maximize sum(b) subject to (a*b)_k <= M for all k, b >= 0.
    This is the LP subproblem in block coordinate descent.
    """
    n = len(a)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)

    for k in range(2 * n - 1):
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            A_ub[k, j] = a[i]

    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=None, b_eq=None,
                  bounds=bounds, method='highs')

    if res.status != 0:
        return None

    b = np.maximum(res.x, 0.0)
    s = b.sum()
    if s < 1e-12:
        return None
    return b


def solve_lp_minM(a):
    """
    Minimize M subject to (a*b)_k <= M for all k, sum(b)=1, b >= 0.
    Returns optimal b and M.
    """
    n = len(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    for k in range(2 * n - 1):
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            A_ub[k, j] = a[i]
        A_ub[k, -1] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status != 0:
        return None, None

    b = np.maximum(res.x[:n], 0.0)
    s = b.sum()
    if s < 1e-12:
        return None, None
    b /= s
    return b, res.x[-1]


def lp_descent(a0, time_limit, t_start=0.15, use_minM=False):
    """
    LP-based block coordinate descent.
    Alternates: fix a, solve LP for b; then mix a and b.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_start
    t0 = time.monotonic()
    iter_count = 0
    stale_count = 0

    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break

        if use_minM:
            b, lp_M = solve_lp_minM(a)
            if b is None:
                t *= 0.5
                if t < 0.01:
                    break
                continue
        else:
            b = solve_lp_maxsum(a, max_c)
            if b is None:
                t *= 0.5
                if t < 0.01:
                    break
                continue

        # Try full update
        new_a_full = b
        new_conv_full = fast_convolve(new_a_full)
        new_obj_full = 2.0 * n * new_conv_full.max()

        if new_obj_full < best_obj:
            best_obj = new_obj_full
            best_a = new_a_full.copy()

        # Try mixed update
        if t < 0.999:
            mixed = (1.0 - t) * a + t * b
            mixed = np.maximum(mixed, 0.0)
            ms = mixed.sum()
            if ms > 1e-12:
                mixed = mixed / ms
            mixed_conv = fast_convolve(mixed)
            mixed_obj = 2.0 * n * mixed_conv.max()

            if mixed_obj < new_obj_full:
                new_a_full = mixed
                new_obj_full = mixed_obj
                if new_obj_full < best_obj:
                    best_obj = new_obj_full
                    best_a = new_a_full.copy()

        # Accept if improved from current, or always (for exploration)
        a = new_a_full
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        iter_count += 1

        # Adapt step size
        if cur_obj < 2.0 * n * 0.8 / n:
            t = min(1.0, t * 1.05)
            stale_count = 0
        else:
            if iter_count % 50 == 0:
                stale_count += 1
                if stale_count > 5:
                    t = max(0.05, t * 0.7)
                    stale_count = 0

    return best_a, best_obj


def mirror_descent(a0, time_limit, t_init=50.0, eta_init=0.3):
    """Multiplicative mirror descent on the simplex."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:

        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum

        # Compute subgradient efficiently
        sg = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-12:
                continue
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                j = k - i
                if i == j:
                    sg[i] += wk * a[j]
                else:
                    sg[i] += wk * a[j]

        # Symmetrize: sg should account for both a_i and a_{k-i} contributions
        # Actually the above is correct: dc_k/da_i = a_{k-i} (plus a_i if k=2i)
        # Let me redo: c_k = sum_i a_i * a_{k-i}
        # dc_k/da_j = a_{k-j} (if valid) + a_{k-j} (if j != k-j and valid)
        # = a_{k-j} if k=2j, else 2*a_{k-j}

        sg = np.zeros(n)

            wk = w[k]
            if wk < 1e-12:

                sg[i] += wk * a[j]

        # But dc_k/da_j = a_{k-j} + a_{k-j} (if j != k-j) = 2*a_{k-j} (usually)

def local_search(a0, time_limit, rng):
    """Informed local search targeting peak reduction with multiple moves."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.008 * (1.0 - frac) ** 2 + 0.0002

        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Find peak positions
        peak_thresh = max_c * 0.65
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])

        k = int(rng.choice(peak_positions))
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)

        # Contribution pattern for this peak
        contrib = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            contrib[i] = a[i] * a[j]

        total = contrib.sum()
        if total < 1e-15:
            continue

        # Source: weighted by contribution
        probs = contrib.copy()
        probs /= probs.sum()
        src = int(rng.choice(n, p=probs))

        # Destination: prefer positions with low "activity"
        # Activity = sum of contributions to all peaks
        activity = np.zeros(n)
        for pk in peak_positions[:12]:
            plo = max(0, pk - n + 1)
            phi = min(n - 1, pk)
            for i in range(plo, phi + 1):
                activity[i] += a[i] * a[pk - i]

        # Weighted random destination (inverse activity)
        inv_act = 1.0 / (activity + 1e-10)
        inv_act /= inv_act.sum()
        dst = int(rng.choice(n, p=inv_act))

        if dst == src:
            continue

        # Transfer
        frac_t = rng.uniform(0.02, 0.08)
        delta = a[src] * frac_t
        if delta < 1e-12:
            continue

        a[src] -= delta
        a[dst] += delta
        if a[src] < 0:
            a[src] += delta
            a[dst] -= delta
            continue

        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj

        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[src] += delta
                a[dst] -= delta

    return best_a, best_obj

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % sf] = g

    for freq in [1, 2, 3, 5]:
        s_val = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        s_val /= s_val.sum()
        inits['sin_%d' % freq] = s_val

    for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['dir_%s' % alpha] = r

    chirp = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * x / (2.0 * n)))
    chirp = np.maximum(chirp, 0.0)
    if chirp.sum() > 1e-10:
        chirp /= chirp.sum()
        inits['chirp'] = chirp

    # Product of Gaussians
    g1 = np.exp(-0.5 * ((x - n * 0.3) / (0.15 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - n * 0.7) / (0.15 * n)) ** 2)
    bi = g1 + g2
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Exponential ramp
    ramp = np.exp(np.linspace(0, 2, n))
    ramp /= ramp.sum()
    inits['exp_ramp'] = ramp

    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0).copy()
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size),
                           np.maximum(parent, 0.0))
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            pp /= pp.sum()
            inits['parent'] = pp

        # Perturbed parent
        pp2 = pp * (1.0 + 0.3 * rng.standard_normal(n))
        pp2 = np.maximum(pp2, 0)
        if pp2.sum() > 1e-10:
            pp2 /= pp2.sum()
            inits['parent_pert'] = pp2

    return inits

                                      t_start=0.3 + 0.15 * r, use_minM=False)
            else:
                a_opt, _ = lp_descent(a_cur, time_limit=t_alloc,
                                      t_start=0.5 + 0.15 * r, use_minM=True)
            record(a_opt)

    print("Phase 2 complete. Best: %.6f" % best_score)

    # Phase 3: Mirror descent polish
    if best_sequence is not None and time.monotonic() < deadline - 300:
        print("Phase 3: Mirror descent polish...")
        phase3_budget = min(300.0, (deadline - time.monotonic()) * 0.3)
        n_md = 3
        per_md = phase3_budget / n_md

        for i in range(n_md):
            if time.monotonic() > deadline - 300:
                break
            a_cur = best_sequence.copy()
            a_cur /= a_cur.sum()
            t_alloc = min(per_md, deadline - time.monotonic() - 300)
            if t_alloc < 3:
                break
            a_opt, _ = mirror_descent(a_cur, time_limit=t_alloc,
                                      t_init=200.0 + 200 * i,
                                      eta_init=0.1 / (i + 1))
            record(a_opt)

    print("Phase 3 complete. Best: %.6f" % best_score)

    # Phase 4: Local search
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("Phase 4: Local search...")
        phase4_budget = min(250.0, deadline - time.monotonic() - 100)
        n_ls = 4
        per_ls = phase4_budget / n_ls

        for i in range(n_ls):
            if time.monotonic() > deadline - 100:
                break
            rng_ls = np.random.default_rng(seed + 20000 + i * 7)
            t_alloc = min(per_ls, deadline - time.monotonic() - 100)
            if t_alloc < 3:
                break
            if i % 2 == 0:
                a_opt, _ = local_search(best_sequence, time_limit=t_alloc, rng=rng_ls)
            else:
                a_opt, _ = multi_point_local_search(best_sequence, time_limit=t_alloc, rng=rng_ls)
            record(a_opt)

    print("Phase 4 complete. Best: %.6f" % best_score)

    # Phase 5: Final combined polish
    if best_sequence is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 10
        if remaining > 15:
            print("Phase 5: Final polish...")
            a_cur = best_sequence.copy()
            a_cur /= a_cur.sum()

            # LP with large step
            t1 = remaining * 0.4
            a_opt, _ = lp_descent(a_cur, time_limit=t1, t_start=0.7, use_minM=False)
            record(a_opt)

            if time.monotonic() < deadline - 10:
                rem2 = deadline - time.monotonic() - 10
                if rem2 > 5:
                    a_opt2, _ = mirror_descent(best_sequence, time_limit=rem2 * 0.6,
                                               t_init=1000.0, eta_init=0.03)
                    record(a_opt2)

            if time.monotonic() < deadline - 5:
                rem3 = deadline - time.monotonic() - 5
                if rem3 > 3:
                    rng_f = np.random.default_rng(seed + 99999)
                    a_opt3, _ = local_search(best_sequence, time_limit=rem3, rng=rng_f)
                    record(a_opt3)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
