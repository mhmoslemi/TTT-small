Let me also consider using a different approach entirely: **alternating minimization** or **coordinate descent**.

For coordinate descent: fix all variables except a[i], and optimize a[i] to minimize the max peak. Given a[i-1], a[i+1], etc., the peak C[k] is a quadratic function of a[i] (since C[k] contains the term a[i]*a[k-i]). The optimal a[i] can be found in closed form for each k, and we want the a[i] that minimizes the max over k.

This is a 1D minimax problem which can be solved by bisection or by finding the "bottleneck" k.

Actually, for a fixed a[i] = x, C[k] is linear in x for each k (since C[k] = a[i]*a[k-i] + rest, and "rest" doesn't depend on x). So max_k C[k] is the max of linear functions of x, which is convex in x. We can minimize it by finding the point where the two "active" constraints (the two largest C[k] values) are equal.

This gives an efficient coordinate descent! At each step:
1. For each i (or a random subset), compute the optimal a[i] given all other variables fixed.
2. The optimal a[i] is where the maximum of the two largest "slopes" (a[k-i] values for the top peaks) balances out.

Specifically, if the current peak is at k*, then increasing a[i] by dx increases C[k*] by dx * a[k*-i]. To reduce the peak, we want to decrease a[i] if a[i] contributes significantly to C[k*].

Uses log-sum-exp soft-max objective with Adam optimizer, exact subgradient refinement,

Key difference from previous: uses adaptive gradient methods (Adam) instead of L-BFGS,

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def logsumexp_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), tau)).
    Returns (score, grad_theta).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.exp

    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score = 2.0 * n * soft_max_val

    # Gradient: grad_a = 2 * conv(w, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_w_ar = fft_convolve(w, a_rev)
    grad_a = 2.0 * conv_w_ar[n - 1:2 * n - 1]

    # Convert to theta gradient via softmax Jacobian
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def exact_peak_grad(a):

    Compute exact subgradient of max_k C[k] w.r.t. a.
    Returns (max_val, peak_idx, grad_a).

    max_val = c[peak_idx]

    a_rev = a[::-1]
    k = peak_idx
    s = n - 1 - k

        grad_a[:n - s] = 2.0 * a_rev[s:]
    else:
        grad_a = np.zeros(n)
        grad_a[-s:] = 2.0 * a_rev[:n + s]

    return max_val, peak_idx, grad_a

            score, grad = logsumexp_obj_grad(theta, tau)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t_total)
            v_hat = v / (1 - beta2 ** t_total)
            theta -= lr_eff * m_hat / (np.sqrt(v_hat) + eps)

            # Evaluate actual score
            a = softmax(theta)
            actual_score = local_score(a)
            if actual_score < best_score:
                best_score = actual_score
                best_theta = theta.copy()

        if verbose:
            print(f"    tau={tau:.6f}, score={best_score:.6f}, t={time.monotonic()-t0:.0f}s")

    return best_theta, best_score


def subgradient_refine(a_init, n_steps, time_limit, deadline):
    """
    Exact subgradient method with adaptive step size.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    a = a / np.sum(a)
    current_score = local_score(a)
    best = a.copy()
    best_score = current_score

    # Use Polyak step size
    t0 = time.monotonic()
    gamma = 0.01
    best_val = current_score / (2.0 * n)  # this is max(conv) for sum=1
    B = 1.0 / n  # upper bound on max(conv) for sum=1 (uniform gives 1/n)
    # Actually B should be an upper bound; use initial score
    B = current_score / (2.0 * n) * 2.0

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        max_val, peak_idx, grad_a = exact_peak_grad(a)

        # Polyak step: gamma_k = (f(a) - f*) / ||g||^2, but we don't know f*
        # Use: gamma_k = 2 * (f(a) - f_lower) / (k * ||g||^2)
        # f_lower = 2 (theoretical minimum)
        f_lower = 2.0
        grad_norm_sq = np.dot(grad_a, grad_a)
        if grad_norm_sq < 1e-20:
            break

        a_val = current_score  # this is 2n * max(conv) with sum=1
        gamma_k = 2.0 * max(0.0, a_val - f_lower) / ((step + 1) * grad_norm_sq)
        gamma_k = min(gamma_k, 0.05)  # cap step size

        a_new = a - gamma_k * grad_a

        # Project onto simplex {a >= 0, sum a = 1}
        a_new = simplex_project(a_new)

        new_score = local_score(a_new)
        if new_score < current_score:
            a = a_new
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            # Reduce step size
            gamma_k *= 0.5
            a_new = a - gamma_k * grad_a
            a_new = simplex_project(a_new)
            new_score = local_score(a_new)
            if new_score < current_score:
                a = a_new
                current_score = new_score
                if current_score < best_score:
                    best_score = current_score
                    best = a.copy()

    return best, best_score


def simplex_project(x):
    """Project vector onto probability simplex {a >= 0, sum a = 1}."""
    n = len(x)
    if np.sum(x) <= 0:
        return np.ones(n) / n
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.findlast((u * np.arange(1, n + 1) > (cssv - 1)))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)


def coordinate_descent_refine(a_init, time_limit, deadline, verbose=True):
    """
    Coordinate descent: for each coordinate, find optimal value given others fixed.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    a = a / np.sum(a)
    current_score = local_score(a)
    best = a.copy()
    best_score = current_score

    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    order = np.arange(n)

    while time.monotonic() < deadline:
        np.random.shuffle(order) if False else rng.shuffle(order)
        improved = False
        for i in order:
            if time.monotonic() > deadline:
                break
            # Compute C[k] as function of a[i] = x
            # C[k] = a[i]*a[k-i] + (rest not involving a[i]) + a[i-k]*a[k-(i-k)] ...
            # Actually C[k] = sum_j a[j]*a[k-j]. Terms involving a[i]:
            #   j=i: a[i]*a[k-i]
            #   j=k-i: a[k-i]*a[i] (same product if k-i == i)
            # So C[k] = a[i] * (a[k-i] + a[k-i] if k!=2i, else a[i]) + rest
            # More precisely: C[k] = a[i] * a[k-i] + a[k-i] * a[i] - (overlap if k=2i) + rest
            # = 2*a[i]*a[k-i] - (a[i]^2 if k=2i) + rest_k
            # where rest_k = C[k] - 2*a[i]*a[k-i] + (a[i]^2 if k=2i)

            # For simplicity, compute C for current a, then for each k:
            # C[k] as function of x (=a[i]): C[k](x) = x * s_k + r_k
            # where s_k = 2*a[k-i] (or a[k-i] if k=2i, but let's be careful)
            # Actually: C[k](x) = x*a[k-i] + a[k-i]*x + (x^2 if k=2i else 0) + rest
            # Wait no. Let me redo.
            # C[k] = sum_j a[j]*a[k-j]
            # Setting a[i] = x:
            # Terms with a[i] as first factor (j=i): x * a[k-i]
            # Terms with a[i] as second factor (k-j=i, j=k-i): a[k-i] * x
            # If k-i == i (k=2i): these are the same term, so just x * a[i] = x^2
            #   Actually no: when j=i and k-j=i, we have j=i and k-i=i, so a[j]*a[k-j] = a[i]*a[i] = x^2
            #   This appears only once in the sum.
            # If k-i != i: we get two separate terms: x*a[k-i] + a[k-i]*x = 2*x*a[k-i]

            # So: C[k](x) = x * coeff_k + const_k
            # where coeff_k = 2*a[k-i] (if k != 2i and 0 <= k-i < n)
            #                  a[i] = x (if k = 2i) -> this makes it quadratic!
            #                  0 (if k-i out of range)
            # const_k = C[k] - x * coeff_k (computed from current a)

            # For k = 2i: C[2i](x) = x^2 + const_2i (quadratic)
            # For k != 2i: C[k](x) = x * 2*a[k-i] + const_k (linear)

            # To minimize max_k C[k](x) subject to x >= 0 and sum constraint:
            # The sum constraint: sum_j a[j] = 1, so if we change a[i] to x,
            # we must adjust other elements. Simplest: redistribute uniformly.
            # delta = x - a[i], then all other elements decrease by delta/(n-1).
            # But this complicates things. Instead, let's just optimize x freely
            # and then re-normalize.

            # For a quick heuristic: find x that minimizes the peak contribution.
            # The peak is at the k where C[k] is currently largest.
            # To reduce C[peak]: if a[peak-i] > 0, decreasing x helps.
            # But decreasing x increases other C[k] values.

            # Simple approach: try a few candidate x values and pick the best.
            old_x = a[i]
            current_c = fft_convolve(a, a)
            peak_k = int(np.argmax(current_c))

            # Try reducing a[i] if it contributes to the peak
            if 0 <= peak_k - i < n:
                # Reducing a[i] reduces C[peak_k]
                for frac in [0.5, 0.7, 0.8, 0.9]:
                    candidate = a.copy()
                    candidate[i] = old_x * frac
                    candidate = np.maximum(candidate, 0)
                    candidate = candidate / np.sum(candidate)
                    sc = local_score(candidate)
                    if sc < current_score:
                        a = candidate
                        current_score = sc
                        improved = True
                        break

            # Try increasing a[i] if it's small (might help spread)
            if old_x < 1.0 / n * 1.5:
                for frac in [1.5, 2.0, 3.0]:
                    candidate = a.copy()
                    candidate[i] = old_x * frac
                    candidate = np.maximum(candidate, 0)
                    candidate = candidate / np.sum(candidate)
                    sc = local_score(candidate)
                    if sc < current_score:
                        a = candidate
                        current_score = sc
                        improved = True
                        break

            if current_score < best_score:
                best_score = current_score
                best = a.copy()

        if not improved:
            # Try a full sweep with smaller steps
            for i in range(n):
                if time.monotonic() > deadline:
                    break
                old_x = a[i]
                for frac in [0.85, 0.7, 1.2, 1.4]:
                    candidate = a.copy()
                    candidate[i] = old_x * frac
                    candidate = np.maximum(candidate, 0)
                    s = np.sum(candidate)
                    if s < 1e-10:
                        continue
                    candidate = candidate / s
                    sc = local_score(candidate)
                    if sc < current_score:
                        a = candidate
                        current_score = sc
                        if sc < best_score:
                            best_score = sc
                            best = a.copy()
                        break

    return best, best_score


def make_diverse_inits(n, rng, base_seq=None):
    """Generate diverse initializations."""
    inits = []

    # Uniform
    inits.append(('uniform', np.ones(n) / n))

    # Chirp
    t = np.arange(n, dtype=np.float64)
    for alpha in [0.2, 0.4, 0.6]:
        a = 1.0 + alpha * np.cos(2 * np.pi * t * t / (3 * n))
        a = np.clip(a, 0.05, None)
        a = a / np.sum(a)
        inits.append((f'chirp_{alpha}', a.copy()))

    # Sine modulation
    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * t / n)
        a = np.clip(a, 0.05, None)
        a = a / np.sum(a)
        inits.append((f'sine_{freq}', a.copy()))

    # Random
    for spread in [0.1, 0.3, 0.5]:
        a = 1.0 + spread * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
        a = a / np.sum(a)
        inits.append((f'rand_{spread}', a.copy()))

    # Bell
    center = (n - 1) / 2.0
    for width in [n * 0.1, n * 0.2, n * 0.4]:
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
        a = np.clip(a, 0.01, None)
        a = a / np.sum(a)
        inits.append((f'bell_{width/n:.1f}', a.copy()))

    # Interpolation from base
    if base_seq is not None:
        x_old = np.linspace(0, 1, len(base_seq))
        x_new = np.linspace(0, 1, n)
        a_interp = np.interp(x_new, x_old, base_seq)
        a_interp = np.maximum(a_interp, 1e-6)
        a_interp = a_interp / np.sum(a_interp)
        inits.append(('interp_base', a_interp))

        # Perturbed base
        a_pert = a_interp * np.exp(rng.normal(0, 0.2, n))
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert = a_pert / np.sum(a_pert)
        inits.append(('pert_base', a_pert))

        # More perturbation
        a_pert2 = a_interp * np.exp(rng.normal(0, 0.5, n))
        a_pert2 = np.maximum(a_pert2, 1e-6)
        a_pert2 = a_pert2 / np.sum(a_pert2)
        inits.append(('pert2_base', a_pert2))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search combining Adam optimization, subgradient refinement,
    and coordinate descent. Uses curriculum annealing on soft-max temperature.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    base_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        base_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        base_seq = base_seq / np.sum(base_seq)
        s = local_score(base_seq)
        print(f"Base sequence: n={len(base_seq)}, score={s:.6f}")
        best_score = s
        best_seq = base_seq.copy()

    # Focus on n=256 primarily, with some exploration of nearby lengths
    primary_n = 256
    if base_seq is not None:
        primary_n = len(base_seq)

    print(f"\n=== Phase 1: Adam optimization with soft-max (n={primary_n}) ===")

    # Generate initializations
    inits = make_diverse_inits(primary_n, rng, base_seq)
    print(f"  {len(inits)} initializations generated")

    # Phase 1: Run Adam on multiple initializations with time allocation
    n_inits = len(inits)
    time_per_init = min(200.0, (deadline - t_global) * 0.5 / max(1, min(n_inits, 8)))

    # Sort inits: put base-perturbed ones first (likely to be good)
    init_order = sorted(range(n_inits), key=lambda i: 0 if 'base' in inits[i][0] else 1)

    for idx in init_order:
        if time.monotonic() > deadline - 300:
            break
        name, a0 = inits[idx]
        theta0 = np.log(a0 + 1e-30)

        # Adam with annealed tau
        tau_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        max_iter = max(500, int(200 * time_per_init))
        lr = 0.01

        theta_best, score = adam_optimize(
            theta0, tau_schedule, max_iter, lr,
            beta1=0.9, beta2=0.999, eps=1e-8,
            time_limit=time_per_init,
            deadline=min(time.monotonic() + time_per_init, deadline - 300),
            verbose=False
        )

        a_best = softmax(theta_best)
        actual_score = local_score(a_best)

        if actual_score < best_score:
            best_score = actual_score
            best_seq = a_best.copy()
            print(f"  '{name}': score={actual_score:.6f} *** NEW BEST ***")
        else:
            print(f"  '{name}': score={actual_score:.6f}")

    print(f"\n  Best after Phase 1: {best_score:.6f}")

    # Phase 2: Subgradient refinement on best
    print(f"\n=== Phase 2: Subgradient refinement ===")
    if best_seq is not None:
        sg_deadline = min(time.monotonic() + 120, deadline - 100)
        refined, sg_score = subgradient_refine(best_seq, n_steps=5000,
                                               time_limit=120, deadline=sg_deadline)
        if sg_score < best_score:
            best_score = sg_score
            best_seq = refined.copy()
            print(f"  Subgradient improved: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Subgradient best: {sg_score:.6f}")

    # Phase 3: Coordinate descent refinement
    print(f"\n=== Phase 3: Coordinate descent refinement ===")
    if best_seq is not None:
        cd_deadline = min(time.monotonic() + 180, deadline - 60)
        refined, cd_score = coordinate_descent_refine(best_seq,
                                                      time_limit=180, deadline=cd_deadline)
        if cd_score < best_score:
            best_score = cd_score
            best_seq = refined.copy()
            print(f"  Coord descent improved: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Coord descent best: {cd_score:.6f}")

    # Phase 4: Try a second round of Adam from the refined solution
    print(f"\n=== Phase 4: Second-pass Adam from refined solution ===")
    if best_seq is not None:
        theta = np.log(best_seq + 1e-30)
        tau_schedule2 = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        max_iter2 = 3000
        p4_deadline = min(time.monotonic() + 200, deadline - 30)
        theta_best2, score2 = adam_optimize(
            theta, tau_schedule2, max_iter2, 0.005,
            beta1=0.9, beta2=0.999, eps=1e-8,
            time_limit=200, deadline=p4_deadline, verbose=True
        )
        a_best2 = softmax(theta_best2)
        actual_score2 = local_score(a_best2)
        if actual_score2 < best_score:
            best_score = actual_score2
            best_seq = a_best2.copy()
            print(f"  Second-pass Adam: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Second-pass Adam: {actual_score2:.6f}")

    # Phase 5: Try different length if time permits
    if time.monotonic() < deadline - 150:
        print(f"\n=== Phase 5: Length exploration ===")
        for test_n in [primary_n - 32, primary_n + 32, 320, 384]:
            if time.monotonic() > deadline - 80:
                break
            if test_n < 32:
                continue
            a0 = make_diverse_inits(test_n, rng, best_seq)
            a0 = a0[0][1]  # uniform
            # Also try interpolated
            if best_seq is not None:
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, test_n)
                a0 = np.interp(x_new, x_old, best_seq)
                a0 = np.maximum(a0, 1e-6)
                a0 = a0 / np.sum(a0)

            theta0 = np.log(a0 + 1e-30)
            tau_sched = [0.05, 0.02, 0.01, 0.005]
            p5_deadline = min(time.monotonic() + 60, deadline - 40)
            theta_best5, s5 = adam_optimize(
                theta0, tau_sched, 2000, 0.01,
                beta1=0.9, beta2=0.999, eps=1e-8,
                time_limit=60, deadline=p5_deadline, verbose=False
            )
            a5 = softmax(theta_best5)
            score5 = local_score(a5)
            if score5 < best_score:
                best_score = score5
                best_seq = a5.copy()
                print(f"  n={test_n}: {score5:.6f} *** NEW BEST ***")
            else:
                print(f"  n={test_n}: {score5:.6f}")

    # Phase 6: Final micro-optimization
    print(f"\n=== Phase 6: Final refinement ===")
    if best_seq is not None:
        # Short subgradient pass
        final_deadline = deadline - 5
        refined, fin_score = subgradient_refine(best_seq, n_steps=2000,
                                                time_limit=60, deadline=final_deadline)
        if fin_score < best_score:
            best_score = fin_score
            best_seq = refined.copy()
            print(f"  Final refinement: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Final refinement: {fin_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if base_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in base_seq]
    return [1.0] * 256
