def fft_convolve(a):
    a=np.asarray(a, dtype=np.float64)
    n=a.size
    if n==0:
        return np.array([], dtype=np.float64)
    size=1 << (2*n - 2).bit_length()
    fa=np.fft.rfft(a, size)
    out=np.fft.irfft(fa*fa, size)[:2*n-1]
    np.maximum(out, 0.0, out=out)
    return out

"""
Multi-scale optimizer for minimizing 2n*max(conv(a,a))/(sum(a)^2) over
non-negative step functions.

Main components:
1) A safe global initialization bank of polynomial/exponential/grid shapes,
   plus resampling of any provided parent sequence.
2) A constrained L-BFGS-B optimizer on a softmax parameterization, using a
   log-sum-exp smoothing of the active convolution peaks with annealed tau.
3) Several length scales (512, 1024, optionally 2048) and restarting phases.
4) An exact local pair-transfer coordinate polish with line search over
   transfer amount, based on exact updates of the convolution histogram.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as scipy_minimize
except Exception:
    scipy_minimize = None


class TimeUp(Exception):
    pass


_TIME_STATE = {"deadline": 0.0}
_BOUND = np.array([-80.0, 80.0], dtype=np.float64)
_TARGET = 1.5027
_FRACS = (
    0.5,
    0.25,

def fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    out = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    np.maximum(out, 0.0, out=out)
    return out


def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    conv = fft_convolve(a)
    if conv.size == 0:
        return np.inf
    val = 2.0 * a.size * float(np.max(conv)) / (s * s)
    if not np.isfinite(val):
        return np.inf
    return float(val)


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def try_update_best(a, best_a, best_s):
    a = normalize_nonneg(a)
    if a.size == 0:
        return best_a, best_s
    s = score_fast(a)
    if np.isfinite(s) and s < best_s:
        return a.copy(), float(s)
    return best_a, best_s


def gradient_weights(a, w):
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(fa) * fw, size)[:n]
    corr = np.maximum(corr, 0.0)
    return 2.0 * corr


def smooth_compute(x, base, nonbase, tau_rel, state):
    if time.monotonic() > _TIME_STATE["deadline"]:
        raise TimeUp()

    x = np.asarray(x, dtype=np.float64)
    n = len(nonbase) + 1

    if len(x) > 0:
        m = float(np.max(x))
        e = np.exp(x - m)
    else:
        e = np.empty(0, dtype=np.float64)

    den = 1.0 + float(np.sum(e))
    a = np.empty(n, dtype=np.float64)
    a[base] = 1.0 / den
    if len(e) > 0:
        a[nonbase] = e / den

    c = fft_convolve(a)
    maxc = float(np.max(c))
    exact = 2.0 * float(n) * maxc

    if exact < state["best_score"]:
        state["best_score"] = float(exact)
        state["best_a"] = a.copy()

    state["last_a"] = a
    state["last_exact"] = float(exact)

    if maxc <= 1e-300:
        value = 0.0
        g = np.zeros(n, dtype=np.float64)
    else:
        tau = float(tau_rel) * maxc
        if tau <= 1e-300:
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
            value = maxc
        else:
            shifted = (c - maxc) / tau
            ex = np.exp(shifted)
            ss = float(np.sum(ex))
            if ss <= 0.0:
                p = np.full_like(c, 1.0 / float(c.size))
                value = maxc
            else:
                p = ex / ss
                value = float(maxc + tau * np.log(ss))
        g = gradient_weights(a, p)

    scalar = float(np.dot(a, g))
    grad = a[nonbase] * (g[nonbase] - scalar)
    if not np.all(np.isfinite(grad)):
        grad = np.where(np.isfinite(grad), grad, 0.0)

    state["last_value"] = float(value)
    state["last_grad"] = grad
    state["cache_valid"] = True
    state["x_cache"] = x.copy()
    return float(value), grad


def value_fun(x, base, nonbase, tau_rel, state):
    if state.get("cache_valid", False):
        xc = state.get("x_cache")
        if xc is not None and np.array_equal(np.asarray(x, dtype=np.float64), xc):
            return state["last_value"]
    val, grad = smooth_compute(x, base, nonbase, tau_rel, state)
    return val


def grad_fun(x, base, nonbase, tau_rel, state):
    if state.get("cache_valid", False):
        xc = state.get("x_cache")
        if xc is not None and np.array_equal(np.asarray(x, dtype=np.float64), xc):
            return state["last_grad"]
    val, grad = smooth_compute(x, base, nonbase, tau_rel, state)
    return grad


def x_from_a(a, base):
    n = a.size
    if n <= 1:
        return np.zeros(0, dtype=np.float64), np.zeros(0, dtype=int)

    b = float(a[base])
    if b <= 1e-300:
        base = int(np.argmax(a))
        b = float(a[base])
    if b <= 1e-300:
        return np.zeros(n - 1, dtype=np.float64), np.zeros(n - 1, dtype=int)

    nonbase = np.empty(n - 1, dtype=int)
    k = 0
    for i in range(n):
        if i != base:
            nonbase[k] = i
            k += 1

    pos = np.maximum(a, 1e-300)
    x = np.log(pos[nonbase] / b)
    x = np.clip(x, -80.0, 80.0)
    return x, nonbase


def lbfgs_optimize(a, time_budget, global_deadline, tau_list, tau_weights, label):
    a = normalize_nonneg(a)
    n = a.size
    best_score = score_fast(a)
    best_a = a.copy()

    if n <= 1 or time_budget <= 5.0 or scipy_minimize is None:
        return best_a, best_score

    total_w = float(sum(tau_weights))
    no_improve = 0

    for idx, tau_rel in enumerate(tau_list):
        now = time.monotonic()
        remaining = global_deadline - now
        if remaining <= 15.0:
            break

        tau_budget = time_budget * (float(tau_weights[idx]) / total_w)
        tau_budget = min(float(tau_budget), max(0.0, remaining - 5.0))
        if tau_budget <= 10.0:
            continue

        run_deadline = now + tau_budget
        base = int(np.argmax(a))
        x, nonbase = x_from_a(a, base)

        state = {
            "best_score": best_score,
            "best_a": best_a.copy(),
            "last_a": a.copy(),
            "last_exact": best_score,
            "last_value": 0.0,
            "last_grad": None,
            "cache_valid": False,
            "x_cache": None,
        }

        _TIME_STATE["deadline"] = run_deadline
        bounds = np.tile(_BOUND, (max(1, x.size), 1))

        try:
            scipy_minimize(
                value_fun,
                x,
                args=(base, nonbase, float(tau_rel), state),
                jac=grad_fun,
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": 400,
                    "maxfun": 20000,
                    "maxls": 50,
                    "ftol": 1e-12,
                    "gtol": 1e-7,
                    "disp": False,
                },
            )
        except TimeUp:
            pass
        except Exception:
            pass

        st_best_score = state.get("best_score")
        st_best_a = state.get("best_a")
        if st_best_a is not None and st_best_score is not None:
            if float(st_best_score) < best_score:
                best_score = float(st_best_score)
                best_a = st_best_a

        last_a = state.get("last_a")
        last_exact = state.get("last_exact")
        if last_a is not None and last_exact is not None:
            le = float(last_exact)
            ref = float(st_best_score) if st_best_score is not None else le
            if le <= ref + 1e-7:
                cand = normalize_nonneg(last_a)
                cs = score_fast(cand)
                if cs < best_score:
                    best_score = float(cs)
                    best_a = cand.copy()
                a = cand
            else:
                if st_best_a is not None:
                    best_a = normalize_nonneg(st_best_a)
                    best_score = score_fast(best_a)
                    a = best_a.copy()
        elif st_best_a is not None:
            a = normalize_nonneg(st_best_a)
            best_score = score_fast(a)
            best_a = a.copy()

        if best_score >= best_score - 1e-12:
            no_improve += 1
        else:
            no_improve = 0

        print(
            f"[{label}] tau={tau_rel:.4f} n={n} "
            f"score={best_score:.7f} elapsed={time.monotonic() - 0:.0f}s"
        )

    final_score = score_fast(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a
    return best_a, best_score


def resample_sequence(a, m):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(m, dtype=np.float64)
    if a.size == 1:
        return np.full(m, float(a[0]), dtype=np.float64)
    old = np.linspace(0.0, 1.0, a.size)
    new = np.linspace(0.0, 1.0, m)
    return np.interp(new, old, a)

def prefix_suffix(c):
    c = np.asarray(c, dtype=np.float64)
    pref = np.maximum.accumulate(c)
    suff = np.maximum.accumulate(c[::-1])[::-1]
    return pref, suff


def add_shift_seg(seg, seg_lo, seg_hi_excl, start, coeff, a, n):
    if seg.size == 0 or coeff == 0.0:
        return
    s0 = max(0, seg_lo - start)
    s1 = min(n, seg_hi_excl - start)
    if s1 > s0:
        k0 = start + s0 - seg_lo
        k1 = k0 + (s1 - s0)
        seg[k0:k1] += coeff * a[s0:s1]


def add_point_seg(seg, seg_lo, k, val):
    idx = k - seg_lo
    if 0 <= idx < seg.size:
        seg[idx] += val


def max_transfer_eval(c, a, i, j, delta, pref, suff):
    n = len(a)
    L = len(c)
    if i == j or delta <= 0.0 or i < 0 or j < 0 or i >= n or j >= n:
        return float(np.max(c))
    if float(a[i]) < delta:
        return float(np.max(c))

    if i < j:
        lo = i
        hi_excl = j + n
    else:
        lo = j
        hi_excl = i + n

    hi_excl = min(L, hi_excl)
    if hi_excl <= lo:
        return float(np.max(c))

    seg = c[lo:hi_excl].copy()
    d = float(delta)
    add_shift_seg(seg, lo, hi_excl, i, -2.0 * d, a, n)
    add_shift_seg(seg, lo, hi_excl, j, 2.0 * d, a, n)

    d2 = d * d
    add_point_seg(seg, lo, 2 * i, d2)
    add_point_seg(seg, lo, 2 * j, d2)
    add_point_seg(seg, lo, i + j, -2.0 * d2)

    out = 0.0
    if lo > 0:
        out = float(pref[lo - 1])
    if hi_excl < L:
        sv = float(suff[hi_excl])
        if sv > out:
            out = sv

    m = float(np.max(seg))
    return m if m > out else out


def apply_transfer(c, a, i, j, delta):
    n = len(a)
    L = len(c)
    newc = c.copy()
    if i == j or delta <= 0.0:
        return newc

    if i < j:
        lo = i
        hi_excl = j + n
    else:
        lo = j
        hi_excl = i + n

    hi_excl = min(L, hi_excl)
    if hi_excl > lo:
        seg = newc[lo:hi_excl]
        d = float(delta)
        add_shift_seg(seg, lo, hi_excl, i, -2.0 * d, a, n)
        add_shift_seg(seg, lo, hi_excl, j, 2.0 * d, a, n)

        d2 = d * d
        add_point_seg(seg, lo, 2 * i, d2)
        add_point_seg(seg, lo, 2 * j, d2)
        add_point_seg(seg, lo, i + j, -2.0 * d2)

    np.maximum(newc, 0.0, out=newc)
    return newc


def find_best_delta(c, a, i, j, pref, suff, current_max):
    ai = float(a[i])
    if ai <= 1e-12:
        return 0.0, float(current_max)

    best_d = 0.0
    best_m = float(current_max)
    for frac in _FRACS:
        d = ai * frac
        if d <= 0.0:
            continue
        m = max_transfer_eval(c, a, i, j, d, pref, suff)
        if m < best_m - 1e-14:
            best_m = m
            best_d = d
    return best_d, best_m


def select_pairs(a, G, rng, n):
    if n <= 1:
        return []

    pairs = set()

    donor_scores = G * a
    donor_order = np.argsort(-donor_scores)
    donors = []
    for idx in donor_order:
        idx = int(idx)
        if a[idx] > 1e-8:
            donors.append(idx)
        if len(donors) >= 18:
            break

    donor_set = set(donors)
    recipient_order = np.argsort(G)
    recipients = []
    for idx in recipient_order:
        idx = int(idx)
        if idx not in donor_set:
            recipients.append(idx)
        if len(recipients) >= 12:
            break

    if len(recipients) < 5:
        small = np.argsort(a)[: max(10, n // 2)]
        for idx in small:
            idx = int(idx)
            if idx not in donor_set:
                recipients.append(idx)
            if len(recipients) >= 10:
                break

    if len(recipients) == 0:
        recipients.append(int(np.argmin(G)))

    for d in donors:
        for r in recipients[:5]:
            if d != r:
                pairs.add((d, r))

    if G.size > 0:
        rmin = int(np.argmin(G))
        for d in donors:
            if d != rmin:
                pairs.add((d, rmin))

    for _ in range(50):
        d = int(rng.integers(n))
        r = int(rng.integers(n))
        if d != r and a[d] > 1e-8:
            pairs.add((d, r))

    pairs = list(pairs)
    if len(pairs) > 150:
        pairs = pairs[:150]
    return pairs


def polish_sequence(a, time_limit, rng):

            if d > 0.0 and m < current_max - 1e-13:

                a[j] += d
                if a[i] < 0.0:
                    a[i] = 0.0
                if a[j] < 0.0:
                    a[j] = 0.0

                accepts += 1
                improved = True
                current_max = float(np.max(c))

                if current_max < best_max:
                    best_max = current_max
                    best_a = a.copy()

                pref, suff = prefix_suffix(c)

                if accepts % 200 == 0:
                    a = normalize_nonneg(a)

                    current_max = float(np.max(c))
                    pref, suff = prefix_suffix(c)
                    if current_max < best_max:

        if improved:

        else:
            stagnation += 1
            if stagnation == 2:

                a = normalize_nonneg(a)

                pref, suff = prefix_suffix(c)

                    best_max = current_max
                    best_a = a.copy()

    best_a = normalize_nonneg(best_a)
    best_score = score_fast(best_a)
    return best_a, best_score

    hard_deadline = t0 + max(60.0, budget - 25.0)

    try:
        seed_int = int(seed)
    except Exception:
        seed_int = 42
    rng = np.random.default_rng(seed_int)

    best_a = None
    best_score = np.inf

    parent = globals().get("height_sequence_1", None)
    if parent is not None:
        try:
            pp = np.asarray(parent, dtype=np.float64)
            if pp.size > 0:
                best_a, best_score = try_update_best(pp, best_a, best_score)
        except Exception:
            pass

    if best_a is None:
        best_a = generate_initial(512, rng, None)
        best_score = score_fast(best_a)

    print(f"Initial best: score={best_score:.7f}, n={best_a.size}")

    # Phase 1: length 512 optimization.
    n1 = 512
    a1 = generate_initial(n1, rng, best_a)
    s1 = score_fast(a1)
    if s1 < best_score:
        best_score = s1
        best_a = a1.copy()

    remaining = hard_deadline - time.monotonic()
    alloc = allocate_budget(remaining, 700.0, 320.0)
    if alloc > 35.0:
        print(f"Starting n=512 L-BFGS phase, budget={alloc:.0f}s")
        tau_full = [0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
        w_full = [1, 1, 1, 1, 1, 2, 3]
        a1, s1 = lbfgs_optimize(a1, alloc, hard_deadline, tau_full, w_full, "n512")
        if s1 < best_score:
            best_score = s1
            best_a = a1.copy()
        print(f"After n=512 phase: best={best_score:.7f}, n={best_a.size}")

    # Phase 2: length 1024 optimization.
    remaining = hard_deadline - time.monotonic()
    if remaining > 420.0:
        n2 = 1024
        a2 = generate_initial(n2, rng, best_a)
        s2 = score_fast(a2)
        if s2 < best_score:
            best_score = s2
            best_a = a2.copy()

        alloc = allocate_budget(remaining, 1050.0, 380.0)
        if alloc > 70.0:
            print(f"Starting n=1024 L-BFGS phase, budget={alloc:.0f}s")
            tau_full = [0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
            w_full = [1, 1, 1, 1, 1, 2, 3]
            a2, s2 = lbfgs_optimize(a2, alloc, hard_deadline, tau_full, w_full, "n1024")
            if s2 < best_score:
                best_score = s2
                best_a = a2.copy()
        print(f"After n=1024 phase: best={best_score:.7f}, n={best_a.size}")

    # Optional phase: length 2048 if still above target and enough time.
    remaining = hard_deadline - time.monotonic()
    if remaining > 950.0 and best_score > _TARGET + 0.001:
        n3 = 2048
        a3 = generate_initial(n3, rng, best_a)
        s3 = score_fast(a3)
        if s3 < best_score:
            best_score = s3
            best_a = a3.copy()

        alloc = allocate_budget(remaining, 520.0, 520.0)
        if alloc > 90.0:
            print(f"Starting n=2048 L-BFGS phase, budget={alloc:.0f}s")
            tau_mid = [0.15, 0.08, 0.04, 0.02, 0.01]
            w_mid = [1, 1, 1, 2, 3]
            a3, s3 = lbfgs_optimize(a3, alloc, hard_deadline, tau_mid, w_mid, "n2048")
            if s3 < best_score:
                best_score = s3
                best_a = a3.copy()
        print(f"After optional 2048 phase: best={best_score:.7f}, n={best_a.size}")

    # Phase 3: refine the current global best length.
    remaining = hard_deadline - time.monotonic()
    if (
        remaining > 320.0
        and best_a is not None
        and best_a.size <= 3000
    ):
        a_r = best_a.copy()
        alloc = allocate_budget(remaining, 650.0, 270.0)
        if alloc > 65.0:
            print(f"Starting refinement phase, budget={alloc:.0f}s, n={a_r.size}")
            tau_small = [0.05, 0.02, 0.01, 0.005, 0.002]
            w_small = [1, 1, 2, 3, 4]
            a_r, s_r = lbfgs_optimize(
                a_r, alloc, hard_deadline, tau_small, w_small, "refine"
            )
            if s_r < best_score:
                best_score = s_r
                best_a = a_r.copy()
        print(f"After refinement: best={best_score:.7f}, n={best_a.size}")

    # Phase 4: small random restart if still above target.
    remaining = hard_deadline - time.monotonic()
    if (
        remaining > 260.0
        and best_a is not None
        and best_score > _TARGET + 0.0005
        and best_a.size <= 3000
    ):
        a_restart = best_a * (1.0 + 0.01 * rng.standard_normal(best_a.size))
        a_restart = normalize_nonneg(a_restart)
        s_restart = score_fast(a_restart)
        alloc = allocate_budget(remaining, 320.0, 190.0)
        if alloc > 50.0:
            print(f"Starting restart phase, budget={alloc:.0f}s")
            tau_restart = [0.02, 0.01, 0.005, 0.002]
            w_restart = [1, 2, 3, 4]
            a_restart, s_restart = lbfgs_optimize(
                a_restart,
                alloc,
                hard_deadline,
                tau_restart,
                w_restart,
                "restart",
            )
            if s_restart < best_score:
                best_score = s_restart
                best_a = a_restart.copy()
        print(f"After restart: best={best_score:.7f}, n={best_a.size}")

    # Final exact coordinate polish.
    remaining = hard_deadline - time.monotonic()
    if best_a is not None and remaining > 40.0 and best_a.size <= 4096:
        polish_time = max(15.0, remaining - 15.0)
        print(f"Starting polish phase, budget={polish_time:.0f}s, n={best_a.size}")
        a_pol, s_pol = polish_sequence(best_a, polish_time, rng)
        if s_pol < best_score:
            best_score = s_pol
            best_a = a_pol.copy()
        print(f"After polish: best={best_score:.7f}, n={best_a.size}")

    if best_a is None:
        best_a = np.ones(256, dtype=np.float64)

    final_arr = normalize_nonneg(best_a)
    final_arr = np.clip(final_arr, 0.0, 1000.0)
    ss = float(np.sum(final_arr))
    if ss <= 1e-12:
        final_arr = np.ones(max(final_arr.size, 1), dtype=np.float64)
    else:
        final_arr = final_arr / ss

    final_list = [float(x) for x in final_arr]

    try:
        final_score = evaluate_sequence(final_list)
        if not np.isfinite(final_score):
            final_list = [1.0]
            final_score = evaluate_sequence(final_list)
    except Exception:
        final_score = best_score

    print(f"FINAL internal score={best_score:.7f}, evaluator={final_score:.7f}, n={len(final_list)}")
    return final_list
