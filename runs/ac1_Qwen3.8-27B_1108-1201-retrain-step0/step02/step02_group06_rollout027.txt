"""
Multi-phase optimization for minimizing peak linear convolution of non-negative
step functions. Uses smoothed max (log-sum-exp) with adaptive temperature,
Adam gradient descent in log-parameterization, and targeted coordinate descent
polish. Leverages the provided starting sequence and explores multiple
initializations and sequence lengths.

import numpy as np
import time


def _fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def _score(a):
    """Compute score: 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = _fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def _score_from_max_conv(a, max_conv_val):
    """Score given precomputed max of convolution."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    return 2.0 * n * max_conv_val / (S * S)


def _soft_max_and_weights(conv, tau):
    """Compute soft max value and probability weights over conv values."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    shifted = np.clip(shifted, -100.0, 100.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    w = exp_vals / total
    value = max_c + tau * np.log(total)
    return value, w


def _grad_soft_max(a, w):
    """
    Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    d/da[j] = 2 * sum_{l=0}^{n-1} a[l] * w[l+j]
    Computed via FFT cross-correlation.
    """
    n = len(a)
    # Need S >= n + len(w) - 1 = n + (2n-1) - 1 = 3n-2 for no aliasing
    S = 1
    while S < 3 * n - 2:
        S <<= 1
    fa = np.fft.rfft(a, S)
    fw = np.fft.rfft(w, S)
    cross = np.fft.irfft(np.conj(fa) * fw, S)
    grad = 2.0 * cross[:n]
    return grad


def _softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def _project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    mask = u * np.arange(1, n + 1) > (cv - 1.0)
    rho = np.where(mask)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def _exponential_init(n, r, rng):
    """Initialize with exponential shape a[i] = exp(r*i/n)."""
    t = np.arange(n, dtype=np.float64)
    a = np.exp(r * t / max(n - 1, 1))
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 1e-6, None)
    return a / np.sum(a)


def _ramp_init(n, power, rng):
    """Initialize with power ramp a[i] = ((i+1)/n)^power."""
    t = (np.arange(n) + 1) / n
    a = t ** power
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 1e-6, None)
    return a / np.sum(a)


def _spectral_init(n, rng, n_iter=40):
    """Spectral iterative method for good initial sequence."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)

    for it in range(n_iter):
        conv = _fft_convolve(a, a)
        # Reduce peaks: subtract weighted convolution
        alpha = 2.0 / (n + 10.0)
        a = a - alpha * conv[:n]
        a = a - alpha * conv[n-1:n-1+n]
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def _interp_sequence(a_old, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a_old)
    if n_old == n_new:
        return a_old.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.clip(a_new, 1e-8, None)
    return a_new / np.sum(a_new)


def _adam_optimize(n, a_init, time_limit, rng):
    """
    Adam optimization on softmax parameterization.
    Minimizes soft max of conv(a,a) where a = softmax(theta) (sum=1).
    """
    S_target = 1.0  # Work on the simplex
    theta = np.log(a_init + 1e-15).astype(np.float64)
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = _score(a_init)

    # Adaptive temperature: start high, decay over time
    tau_start = 50.0 / n
    tau_end = 0.5 / n

    # Learning rate schedule
    lr_base = 0.5

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        # Smooth temperature decay
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)

        a = _softmax_normalize(theta)
        conv = _fft_convolve(a, a)
        value, w = _soft_max_and_weights(conv, tau)

        # Gradient w.r.t. theta
        g_a = _grad_soft_max(a, w)
        p = a  # already sum=1
        inner = np.dot(p, g_a)
        g_theta = p * (g_a - inner)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta * g_theta)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # Learning rate with decay
        lr = lr_base / (1.0 + 0.1 * step ** 0.5)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic evaluation
        if step % 100 == 0:
            a_eval = _softmax_normalize(theta)
            sc = _score(a_eval)
            if sc < best_score:
                best_score = sc
                best_a = a_eval.copy()

            if step % 2000 == 0:
                actual_max = np.max(_fft_convolve(a_eval, a_eval))
                print(f"        step={step}, tau={tau:.2e}, score={sc:.8f}, "
                      f"best={best_score:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def _proj_grad_optimize(n, a_init, time_limit, rng):
    """
    Projected gradient descent with Nesterov momentum on the simplex.
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    a_prev = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_max = np.max(_fft_convolve(a, a))
    t0 = time.monotonic()
    step = 0

    tau_start = 20.0 / n
    tau_end = 0.3 / n
    lr = 1.0 / (8.0 * n)

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)

        t_prev = t_mom
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        beta = (t_prev - 1.0) / t_mom

        y = a + beta * (a - a_prev)
        y = _project_simplex(y)

        conv = _fft_convolve(y, y)
        value, w = _soft_max_and_weights(conv, tau)
        grad = _grad_soft_max(y, w)

        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            step_size = lr / grad_norm * np.max(y)
        else:
            step_size = 0.0

        candidate = y - step_size * grad
        candidate = _project_simplex(candidate)

        a_prev = a
        a = candidate
        step += 1

        if step % 200 == 0:
            actual_max = np.max(_fft_convolve(a, a))
            if actual_max < best_max:
                best_max = actual_max
                best_a = a.copy()

                print(f"        PGD step={step}, max_conv={best_max:.8f}, "
                      f"score={2*n*best_max:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_a, 2.0 * n * best_max


def _coordinate_descent(a_init, time_limit, rng):

    Targeted coordinate descent / local search on the simplex.
    Focuses on reducing the peak of the convolution.
    """

    n = len(a)
    current_max = np.max(_fft_convolve(a, a))
    best_a = a.copy()
    best_max = current_max
    t0 = time.monotonic()
    patience = 0
    max_patience = 8000

    while time.monotonic() - t0 < time_limit and patience < max_patience:
        r = rng.random()

        if r < 0.35:
            # Peak-targeted: find elements contributing most to the peak
            conv = _fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Contribution of a[j] to conv[peak_idx]:
            # conv[peak_idx] = sum_i a[i]*a[peak_idx-i]
            # d/da[j] (when j and peak_idx-j are both in [0,n-1])
            # The element a[j] appears in the term a[j]*a[peak_idx-j] (if valid)
            # and a[peak_idx-j]*a[j] (same term)
            # So reducing a[j] reduces the peak by ~2*a[peak_idx-j]*delta
            contribs = np.zeros(n)
            for j in range(n):
                other = peak_idx - j
                if 0 <= other < n:
                    contribs[j] = a[other]
                # Also check if j contributes via other lags near peak
            # Try reducing top contributors
            n_top = max(3, n // 32)
            top_idx = np.argsort(contribs)[-n_top:]
            candidate = a.copy()
            reduc = rng.uniform(0.005, 0.04, size=len(top_idx))
            candidate[top_idx] *= (1.0 - reduc)
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                patience += 1
                continue
            candidate = candidate / cs
            new_max = np.max(_fft_convolve(candidate, candidate))
            if new_max < current_max:
                a = candidate
                current_max = new_max
                patience = 0
            else:
                patience += 1

        elif r < 0.60:
            # Pair transfer: move mass from high-contrib to low-contrib element
            i = rng.integers(n)
            j = rng.integers(n)

            amt = a[i] * frac
            candidate = a.copy()
            candidate[i] = a[i] - amt
            candidate[j] = a[j] + amt
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                patience += 1
                continue
            candidate = candidate / cs
            new_max = np.max(_fft_convolve(candidate, candidate))

def _two_phase_optimize(n, a_init, time_limit, rng):
    """
    Two-phase: Adam gradient then coordinate descent polish.
    """
    # Phase A: Adam
    adam_time = time_limit * 0.6
    t_start = time.monotonic()
    a_adam, score_a = _adam_optimize(n, a_init, adam_time, rng)
    t_elapsed = time.monotonic() - t_start

    # Phase B: Coordinate descent
    cd_time = max(5.0, time_limit - t_elapsed - 5.0)
    a_final, score_final = _coordinate_descent(a_adam, cd_time, rng)

    # Also try CD directly on the initial (in case Adam made it worse)
    cd_time2 = max(5.0, time_limit - t_elapsed - cd_time - 5.0)
    if cd_time2 > 10:
        a_alt, score_alt = _coordinate_descent(a_init, cd_time2, rng)
        if 2.0 * n * np.max(_fft_convolve(a_alt, a_alt)) < score_final:
            a_final = a_alt
            score_final = 2.0 * n * np.max(_fft_convolve(a_alt, a_alt))

    return a_final, score_final

            print(f"  *** New best: {best_score:.8f} ***")

    # Final polish on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Final polish on best sequence ===")
        print(f"  Best before polish: {best_score:.8f}")
        n_best = len(best_seq)
        a_norm = best_seq / np.sum(best_seq)
        polish_time = min(200.0, deadline - 20 - time.monotonic())

        if polish_time > 10:
            a_polished, polish_score = _coordinate_descent(a_norm, polish_time, rng)
            a_polished_scaled = a_polished * np.sqrt(2.0 * n_best)
            polished_list = [float(max(0.0, min(1000.0, x))) for x in a_polished_scaled]
            polish_eval = evaluate_sequence(polished_list)
            print(f"  After polish: {polish_eval:.8f}")

            if polish_eval < best_score:
                best_score = polish_eval
                best_seq = a_polished_scaled.copy()
                print(f"  *** New best after polish: {best_score:.8f} ***")

    # Output final result
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.8f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
