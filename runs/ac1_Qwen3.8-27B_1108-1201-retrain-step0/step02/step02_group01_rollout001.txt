6. Block-coordinate descent for final polishing

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def a_from_theta(theta, S):
    """Convert parameters to sequence: a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def compute_soft_weights(c, T):
    """Compute soft-max weights: w_k = exp(T*c_k/c_max) / sum."""
    c_max = c.max()
    if c_max < 1e-15:
        return np.ones(len(c)) / len(c)
    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        return np.ones(len(c)) / len(c)
    return exp_scaled / total


def compute_grad_a(w, a):
    """Compute grad_a[m] = 2 * sum_j w[m+j] * a[j] for m=0..n-1 using FFT."""
    n = len(a)
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    end_idx = min(2 * n - 1, len(full_conv))
    grad = np.zeros(n)
    grad[:] = 2.0 * full_conv[n - 1: end_idx][:n]
    return grad


def softmax_objective_and_grad(theta, T, S, c_max_ref=None):
    """Objective and gradient w.r.t. theta."""
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()

    shifted = T * (c - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T

    w = compute_soft_weights(c, T)
    grad_a = compute_grad_a(w, a)

    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot

    return obj, grad_theta, c, c_max, a


def gradient_with_topk(theta, T, S, top_k=16):
    """Gradient using hard top-k weights (for very high T)."""
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()

    w = np.zeros(len(c))
    idx = np.argsort(c)[::-1][:top_k]
    w[idx] = 1.0 / top_k

    grad_a = compute_grad_a(w, a)
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot

    return c_max + (c[idx].sum() - top_k * c_max + 1e-30) / top_k, grad_theta, c, c_max, a


def solve_lp_direction(f, M, n, top_k_constraints=40, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0."""
    c = fft_convolve_self(f)
    if top_k_constraints < len(c):
        active_lags = np.argsort(c)[::-1][:top_k_constraints]
    else:
        active_lags = np.arange(len(c))

    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[idx, j] = f[k - j]

    c_obj = -np.ones(n)
    b = np.full(len(active_lags), M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return None, None


def nesterov_adam_hybrid(theta_init, n, S, time_budget, lr_base=0.04,
                         momentum=0.9, T_start=20.0, T_end=3000.0):
    """Hybrid Nesterov+Adam optimizer with adaptive temperature."""
    theta = theta_init.copy().astype(np.float64)
    n_par = len(theta)

    # Adam state
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # Nesterov momentum
    v_mom = np.zeros(n_par)

    start = time.monotonic()
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    best_theta = theta.copy()
    stall_counter = 0
    stall_threshold = 2000

    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget

        # Adaptive temperature
        a = a_from_theta(theta, S)
        c = fft_convolve_self(a)
        c_max = c.max()
        sorted_c = np.sort(c)[::-1]
        gap = sorted_c[0] - sorted_c[min(20, len(sorted_c) - 1)]
        # Temperature scales with how "flat" the top of the distribution is
        T = T_start + (T_end - T_start) * (1.0 - np.exp(-3.0 * progress))
        # Adjust based on gap: if peaks are well-separated, use higher T
        if gap > c_max * 0.01:
            T = min(T * 2.0, 10000.0)
        elif gap < c_max * 0.001:
            T = max(T * 0.5, T_start)

        # Compute gradient
        if T < 5000.0:
            _, grad, _, _, _ = softmax_objective_and_grad(theta, T, S)
        else:
            _, grad, _, _, _ = gradient_with_topk(theta, T, S, top_k=32)

        # Update best
        cur_obj = c.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_theta = theta.copy()
            stall_counter = 0
        else:
            stall_counter += 1

        # Adaptive learning rate with decay
        lr = lr_base * (1.0 - 0.5 * progress) * (0.8 ** (stall_counter / 1000.0))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        adam_step = m_hat / (np.sqrt(v_hat) + eps)

        # Nesterov momentum on top of Adam
        v_mom = momentum * v_mom - lr * adam_step
        theta = theta + v_mom

        # Periodic restart: if stalled, add perturbation
        if stall_counter > stall_threshold:
            noise_scale = 0.1 * (1.0 - progress)
            theta += noise_scale * np.random.randn(n_par)
            v_mom = np.zeros(n_par)
            m = np.zeros(n_par)
            v = np.zeros(n_par)
            stall_counter = 0
            stall_threshold = int(stall_threshold * 0.7)

    return best_theta, best_a, best_obj


def block_coordinate_descent(a, n, S, time_budget, block_size=64):
    """Block coordinate descent: optimize blocks of the sequence."""
    a = a.copy().astype(np.float64)
    n_par = len(a)
    n_blocks = (n_par + block_size - 1) // block_size
    start = time.monotonic()

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    for sweep in range(100):
        if time.monotonic() - start > time_budget:
            break
        for b in range(n_blocks):
            if time.monotonic() - start > time_budget:
                break
            lo = b * block_size
            hi = min(lo + block_size, n_par)
            if lo >= n_par:
                break

            # Optimize a[lo:hi] while keeping rest fixed
            a_block = a[lo:hi].copy()
            a_rest = np.concatenate([a[:lo], a[hi:]])

            # Try several perturbations of the block
            best_block = a_block.copy()
            best_block_obj = best_obj

            for trial in range(20):
                # Random direction in the block
                if trial == 0:
                    perturbation = np.zeros(hi - lo)
                elif trial == 1:
                    perturbation = np.ones(hi - lo) * 0.1
                else:
                    perturbation = np.random.randn(hi - lo) * 0.05

                a_trial = a.copy()
                a_trial[lo:hi] = np.maximum(a_block + perturbation * a_block.mean(), 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial * (S / s)
                c_trial = fft_convolve_self(a_trial)
                obj_trial = c_trial.max()

                if obj_trial < best_block_obj:
                    best_block_obj = obj_trial
                    best_block = a_trial[lo:hi].copy()

            a[lo:hi] = best_block
            s = a.sum()
            if s > 1e-10:
                a = a * (S / s)
            c = fft_convolve_self(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()

    return best_a, best_obj


def coordinate_descent_polish(a, n, S, time_budget):
    """Full coordinate descent polish."""
    a = a.copy().astype(np.float64)
    n_par = len(a)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    # Sort indices by contribution to peak (process most important first)
    c = fft_convolve_self(a)
    # Contribution of a[i] to peak: sum of c peaks that involve a[i]
    contributions = np.zeros(n_par)
    top_lags = np.argsort(c)[::-1][:64]
    for k in top_lags:
        lo = max(0, k - n_par + 1)
        hi = min(n_par - 1, k)
        for j in range(lo, hi + 1):
            contributions[j] += c[k]
    order = np.argsort(contributions)[::-1]

    for sweep in range(50):
        if time.monotonic() - start > time_budget:
            break
        any_improved = False
        for idx_pos in range(min(n_par, 512)):
            i = order[idx_pos % n_par]
            if time.monotonic() - start > time_budget:
                break
            original = a[i]
            if original <= 1e-12:
                continue

            # Binary search for best a[i]
            best_local = original
            best_local_obj = c.max()
            lo_v, hi_v = 0.0, original * 2.0
            for _ in range(10):
                mid = (lo_v + hi_v) / 2.0
                a[i] = mid
                s = a.sum()
                if s < 1e-10:
                    a[i] = original
                    break
                a_norm = a * (S / s)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_local_obj:
                    best_local_obj = obj_check
                    best_local = mid
                    lo_v = mid
                else:
                    hi_v = mid
            a[i] = best_local
            s = a.sum()
            if s > 1e-10:
                a = a * (S / s)
            c = fft_convolve_self(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()
                any_improved = True

        if not any_improved:
            break

    return best_a, best_obj


def run_lp_improvements(a, n, S, time_budget):
    """Apply LP-based direction improvements."""
    start = time.monotonic()
    best_a = a.copy()
    best_obj = fft_convolve_self(a).max()

    for iteration in range(20):
        if time.monotonic() - start > time_budget:
            break
        c = fft_convolve_self(a)
        M = c.max()

        # Try different numbers of active constraints
        for k_constr in [20, 40, 80]:
            if time.monotonic() - start > time_budget:
                break
            g, lp_val = solve_lp_direction(a, M, n, top_k_constraints=k_constr,
                                           time_limit=3.0)
            if g is not None:
                # Blend with different factors
                for t_blend in [0.2, 0.4, 0.6, 0.8]:
                    a_new = (1 - t_blend) * a + t_blend * g * S
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 0.01:
                        a_new = a_new * (S / s_new)
                        c_new = fft_convolve_self(a_new)
                        obj_new = c_new.max()
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_a = a_new.copy()
                            a = a_new
                            print("  LP iter %d k=%d t=%.1f: obj=%.6f" %
                                  (iteration, k_constr, t_blend, obj_new))

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses Nesterov-Adam hybrid optimization with adaptive temperature,
    LP-based direction finding, and coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 12.0)

    global_best = np.inf
    global_a = None

    # Load the given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given sequence: n=%d, obj=%.6f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given sequence: %s" % str(e))

    # Determine primary working length
    if given_a is not None:
        n = len(given_a)
    else:
        n = 4096
    S = np.sqrt(2.0 * float(n))

    # Convert to log-parameters
    a = global_a.copy() if global_a is not None else np.ones(n) * (S / n)
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())

    # === Phase 1: Main optimization (55% of time) ===
    phase1_budget = 0.55 * (deadline - time.monotonic())
    print("Phase 1: Nesterov-Adam hybrid, budget=%.1fs, n=%d" % (phase1_budget, n))

    # Split into sub-runs with decreasing learning rate
    sub_runs = 3
    for sub in range(sub_runs):
        remaining = deadline - time.monotonic() - (sub_runs - sub - 1) * 5.0
        if remaining < 10.0:
            break
        sub_budget = min(phase1_budget / sub_runs, remaining * 0.4)

        lr = 0.05 * (0.6 ** sub)
        print("  Sub-run %d: lr=%.4f, budget=%.1fs" % (sub, lr, sub_budget))

        theta_best, a_best, obj_best = nesterov_adam_hybrid(
            theta, n, S, sub_budget, lr_base=lr
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print("  After sub-run %d: obj=%.6f" % (sub, obj_best))
        theta = theta_best

        # Interleave LP improvements
        if time.monotonic() < deadline - 60.0:
            a_work = global_a.copy()
            a_work = a_work * (S / a_work.sum())
            lp_budget = min(30.0, (deadline - time.monotonic()) * 0.1)
            a_lp, obj_lp = run_lp_improvements(a_work, n, S, lp_budget)
            if obj_lp < global_best:
                global_best = obj_lp
                global_a = a_lp.copy()
                print("  After LP: obj=%.6f" % obj_lp)
            theta = np.log(np.maximum(global_a, 1e-10) / np.maximum(global_a, 1e-10).sum())

    # === Phase 2: L-BFGS refinement (20% of time) ===
    phase2_budget = 0.20 * (deadline - time.monotonic())
    print("\nPhase 2: L-BFGS refinement, budget=%.1fs" % phase2_budget)

    if global_a is not None and phase2_budget > 10.0:
        a = global_a.copy()
        a = np.maximum(a, 1e-10)
        a = a * (S / a.sum())
        theta = np.log(a / a.sum())

        # Try different temperatures
        for T_val in [50, 200, 800]:
            if time.monotonic() > deadline - 30.0:
                break
            T_budget = min(30.0, phase2_budget / 3.0)
            a_best, obj_best = nesterov_adam_hybrid(
                theta, n, S, T_budget,
                lr_base=0.02, T_start=T_val * 0.5, T_end=T_val * 2.0
            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print("  L-BFGS T=%d: obj=%.6f" % (T_val, obj_best))
            theta = np.log(np.maximum(global_a, 1e-10) /
                           np.maximum(global_a, 1e-10).sum())

    # === Phase 3: Block coordinate descent (15% of time) ===
    phase3_budget = 0.15 * (deadline - time.monotonic())
    print("\nPhase 3: Block coordinate descent, budget=%.1fs" % phase3_budget)

    if global_a is not None and phase3_budget > 5.0:
        a = global_a.copy() * (S / global_a.sum())
        a_best, obj_best = block_coordinate_descent(a, n, S, phase3_budget,
                                                     block_size=32)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print("  After block CD: obj=%.6f" % obj_best)

    # === Phase 4: Coordinate descent polish (10% of time) ===
    phase4_budget = 0.10 * (deadline - time.monotonic())
    print("\nPhase 4: Coordinate descent polish, budget=%.1fs" % phase4_budget)

    if global_a is not None and phase4_budget > 5.0:
        a = global_a.copy() * (S / global_a.sum())
        a_best, obj_best = coordinate_descent_polish(a, n, S, phase4_budget)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print("  After coord CD: obj=%.6f" % obj_best)

    # === Phase 5: Final LP sweep + verification ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 10.0:
        a = global_a.copy() * (S / global_a.sum())
        a_best, obj_best = run_lp_improvements(a, n, S, remaining - 10.0)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print("  Final LP: obj=%.6f" % obj_best)

    # === Final verification ===
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.6f (verified=%.6f)" % (global_best, ver))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.6f" % ver)
        return final_a

    return final_a
