```python
def solve_lp_direction(a, S, n, K_size=50):
    """Solve LP to find direction g that minimizes max active autocorrelation."""
    c = fft_convolve_self(a)
    M = c.max()
    
    # Get active lags
    active = np.argsort(c)[::-1][:K_size]
    
    # Build constraint matrix
    # (a*g)[k] = sum_j a[k-j]*g[j] for 0<=k-j<n
    A = np.zeros((K_size, n))
    for i, k in enumerate(active):
        for j in range(max(0, k-n+1), min(k+1, n)):
            A[i, j] = a[k-j]
    
    # LP: min t s.t. A*g <= t, sum(g) = S, g >= 0
    # Variables: [t, g[0], ..., g[n-1]]
    # Objective: minimize t
    c_obj = np.zeros(n+1)
    c_obj[0] = 1.0
    
    # Inequality constraints: A*g - t <= 0 for each active lag
    # A_ub * x <= b_ub
    A_ub = np.zeros((K_size, n+1))
    A_ub[:, 0] = -1.0  # -t
    A_ub[:, 1:] = A
    b_ub = np.zeros(K_size)
    
    # Equality: sum(g) = S
    A_eq = np.zeros((1, n+1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([S])
    
    bounds = [(None, None)] + [(0, None)] * n  # t unbounded, g >= 0
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'time_limit': 10})
    
    if result.success:
        return result.x[1:], result.x[0]
    return None, None
```

```python
LP-based fixed-point iteration for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Main idea:
- Normalize sum(a) = sqrt(2n) so objective = max(conv(a,a))
- Iteratively solve LP to find direction g that minimizes max autocorrelation at active lags
- Take steps in direction g with line search
- Periodically perturb to escape local minima
- Polish with coordinate descent

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def compute_objective(a):
    """Compute the true objective: 2*n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def get_active_lags(c, K, threshold_frac=0.99):
    """Get active lags: top K by value, plus any lags within threshold_frac of max."""
    c_max = c.max()

    top_k = np.argsort(c)[::-1][:K]

    near_max = np.where(c >= threshold_frac * c_max)[0]
    # Combine
    active = np.unique(np.concatenate([top_k, near_max]))
    return active

def line_search(a, g, S, current_obj, max_iter=30):
    """Find step size alpha for a_new = (1-alpha)*a + alpha*g."""
    best_alpha = 0.0
    best_obj = current_obj
    alpha = 1.0

    best_obj = c.max()

            break

            if orig < 1e-12:
                continue

                    a[i] = orig

                    a[i] = orig

            s = a.sum()
            if s > 1e-10:

def subgradient_direction(a, active_lags):
    """Compute subgradient direction for the max of autocorrelation at active lags."""
    n = len(a)
    c = fft_convolve_self(a)
    K = len(active_lags)
    
    # g[m] = 2/K * sum_{k in active} a[k-m] (with 0 outside [0,n-1])
    # This equals 2/K * conv(indicator_active, a_rev)[m + n - 1]
    a_rev = a[::-1]
    ind = np.zeros(2 * n - 1)
    ind[active_lags] = 1.0

    # full[p] = sum_k ind[k] * a_rev[p-k]
    # We want full[m + n - 1] for m = 0, ..., n-1
    grad = 2.0 / K * full[n-1:2*n-1]
    return grad

            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    # Standard algorithm: find tau such that x_i = max(v_i - tau, 0) and sum(x) = S
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0][-1]
    tau = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - tau, 0)
```

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/sum(a)^2` for a non-negative sequence. The key insight is that with normalization `sum(a) = sqrt(2n)`, the objective becomes just `max(conv(a,a))`, and we want the autocorrelation to be as flat as possible.

My approach differs from the previous code by making the **LP-based fixed-point iteration** the primary driver (rather than a secondary refinement), with:
1. Cycle through active set sizes to avoid getting trapped

```python
"""
LP-based fixed-point iteration for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Primary driver is solving LPs to find directions that reduce the max autocorrelation
at active lags, with geometric line search, cycling active-set sizes, and
periodic perturbations to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - S)
    if not np.any(cond):
        return np.full(n, S / n)
    rho = np.nonzero(cond)[0][-1]
    tau = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - tau, 0)


def build_lp_matrix(a, active_lags):
    n = len(a)
    ka = len(active_lags)
    A = np.zeros((ka, n))
    for i in range(ka):
        k = active_lags[i]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            A[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return A


def solve_lp(a, S, active_lags, time_limit=12):
    n = len(a)
    ka = len(active_lags)
    A = build_lp_matrix(a, active_lags)

    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0

    A_ub = np.zeros((ka, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = A
    b_ub = np.zeros(ka)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([S])

    bounds = [(None, None)] + [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x[1:], 0.0)
            gs = g.sum()
            if gs > 1e-10:
                g = g * (S / gs)
            return g, float(res.x[0])
    except Exception:
        pass
    return None, None


def get_active_lags(c, k_top, thresh_frac):
    cmax = c.max()
    above = np.where(c >= thresh_frac * cmax)[0]
    topk = np.argsort(c)[::-1][:min(k_top, len(c))]
    active = np.unique(np.concatenate([above, topk]))
    # Sort by value descending, cap
    if len(active) > 200:
        vals = c[active]
        order = np.argsort(vals)[::-1]
        active = active[order[:200]]
    return active

def coord_descent_polish(a, S, time_budget=60):
    a = a.copy().astype(np.float64)
    n = len(a)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            # Try reducing
            for frac in [0.0, 0.2, 0.5, 0.8]:
                a[i] = orig * frac
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_n = a * (S / s)
                c_n = fft_convolve_self(a_n)
                if c_n.max() < best_obj - 1e-12:
                    best_obj = c_n.max()
                    best_a = a_n.copy()
                    a = a_n.copy()
                    break
            # Try increasing
            a[i] = orig
            for frac in [1.3, 1.6, 2.0]:
                a[i] = orig * frac
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_n = a * (S / s)
                c_n = fft_convolve_self(a_n)
                if c_n.max() < best_obj - 1e-12:
                    best_obj = c_n.max()
                    best_a = a_n.copy()
                    a = a_n.copy()
                    break
                a[i] = orig
            s = a.sum()
            if s > 1e-10:
                a = a * (S / s)

    return best_a, best_obj


def subgradient_step(a, S, active_lags, lr):
    n = len(a)
    c = fft_convolve_self(a)
    K = len(active_lags)
    a_rev = a[::-1]
    ind = np.zeros(2 * n - 1)
    ind[active_lags] = 1.0
    full = fft_convolve(ind, a_rev)
    grad = 2.0 / K * full[n - 1:2 * n - 1]
    a_new = a - lr * grad
    a_new = project_simplex(a_new, S)
    return a_new

        a = rng.random(n)

    elif name == "cosine":
        i = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * i / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "gaussian":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n / 2) / (n / 5)) ** 2)
        return a * (S / a.sum())
    elif name == "bumps":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n * 0.2) / (n / 8)) ** 2) + \
            np.exp(-0.5 * ((i - n * 0.5) / (n / 8)) ** 2) + \
            np.exp(-0.5 * ((i - n * 0.8) / (n / 8)) ** 2)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)

                    break
                a_try, S_try = candidates[cand_idx]
                sub_budget = min(400.0, max(30.0, deadline - time.monotonic() - 150.0))
                a_b, obj_b, _ = run_lp_iteration(a_try, S_try, sub_budget, seed=seed + 7)
                if obj_b < global_best:
                    global_best = obj_b
                    global_a = a_b.copy()
                    print("Alt candidate improved: obj=%.8f" % obj_b)
                    break

    # Coordinate descent polish
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = global_a * (S / global_a.sum())
        polish_budget = min(180.0, max(10.0, deadline - time.monotonic() - 60.0))
        print("Polish budget: %.0fs" % polish_budget)
        a_p, obj_p = coord_descent_polish(a, S, polish_budget)
        if obj_p < global_best:
            global_best = obj_p
            global_a = a_p.copy()
        print("After CD polish: obj=%.8f" % obj_p)

        # Final LP burst with remaining time
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 40 and global_a is not None:
            a = global_a * (S / global_a.sum())
            a_f, obj_f, _ = run_lp_iteration(a, S, remaining, seed=seed + 200)
            if obj_f < global_best:
                global_best = obj_f
                global_a = a_f.copy()
            print("After final LP: obj=%.8f" % obj_f)

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("FINAL: obj=%.8f verified=%s" % (global_best,
          "%.8f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        print("Using fallback")

    return final_a
```
