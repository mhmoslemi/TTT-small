Let me also think about the NESTEROV acceleration more carefully. For minimizing f(a) = max_k conv[k] on the simplex:

The Nesterov method:
- y_k = a_k + (beta_k - 1) * (a_k - a_{k-1})  [extrapolation]
- Compute grad at y_k
- a_{k+1} = proj_simplex(y_k - step * grad)

Where beta_k = (1 + sqrt(1 + 4*beta_{k-1}^2)) / 2 (standard Nesterov schedule)

Or simpler: use momentum:
- v_k = momentum * v_{k-1} + grad_k
- a_{k+1} = proj_simplex(a_k - step * v_k)

The learning rate schedule is crucial. For subgradient methods, a good schedule is:
- step_k = C / sqrt(k) for subgradient
- step_k = C / k for gradient (if smooth)

Since the max function is not smooth, we use the subgradient rate.

But in practice, we can do better by:
- Reducing the step when the score decreases (adaptive)
- Increasing the step when stuck (boost)
- Using a "barrier" to avoid the boundary of the simplex (keeps all elements positive)

import numpy as np
import time

def fft_convolve(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (s*s)

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_optimize(a_init, time_limit, rng, lr_init=0.5, momentum_init=0.9):

def soft_max_optimize(a_init, time_limit, rng):
    """Gradient descent on soft-max approximation of max-conv."""
    n = len(a_init)
    a = project_simplex(a_init.astype(np.float64).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.1
    step = 0
    patience = 0
    
    # tau schedule
    tau_start = 5.0
    tau_end = 0.1
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.7)
        
        conv = fft_convolve(a)
        conv_max = np.max(conv)
        w = np.exp((conv - conv_max) / tau)
        w /= np.sum(w)
        
        # Gradient of soft-max w.r.t. a
        # grad[m] = 2 * sum_{j=0}^{n-1} w[j+m] * a[j]
        # Use the fact that this is a cross-correlation
        # Efficient computation:
        # Let's pad w to length 3n-2 and use FFT
        size = 1
        while size < 3*n - 1:
            size <<= 1
        fw = np.fft.rfft(w, size)
        fa = np.fft.rfft(a, size)
        # ifft(fw * fa)[m] = sum_k w[k] * a[(k-m) mod size]
        # With sufficient padding, for m in 0..n-1:
        # sum_k w[k] * a[k-m] (where a[k-m] = 0 if k-m < 0 or k-m >= n)
        # = sum_{k=m}^{m+n-1} w[k] * a[k-m]  (valid range)
        # This is NOT simply ifft(fw*fa)[m] because of the zero-padding
        
        # Actually, ifft(fw * fa)[m] = sum_{k=0}^{size-1} w[k] * a[(k-m) % size]
        # For m < n and with size >= 2n: a[(k-m) % size] is non-zero only when 0 <= (k-m)%size < n
        # For k in 0..2n-2 (where w is non-zero) and m in 0..n-1:
        # (k-m) is in range -(n-1) to 2n-2
        # If k >= m: (k-m) is in 0 to 2n-2, and a[k-m] is valid if k-m < n
        # If k < m: (k-m) is negative, (k-m)%size = size + k - m, which is >= size - n + 1 > n (for size >= 2n)
        # So a[(k-m)%size] = 0 in that case
        
        # Therefore: ifft(fw*fa)[m] = sum_{k=m}^{min(m+n-1, 2n-2)} w[k] * a[k-m]
        # = sum_{j=0}^{min(n-1, 2n-2-m)} w[j+m] * a[j]
        # = sum_{j=0}^{n-1} w[j+m] * a[j] (since w has length 2n-1, w[j+m] = 0 for j+m >= 2n-1)
        
        # So grad[m] = 2 * ifft(fw*fa)[m] for m = 0, ..., n-1
        
        corr = np.fft.irfft(fw * fa, size)
        grad = 2.0 * corr[:n]
        
        grad_c = grad - np.mean(grad)
        v = momentum * v + grad_c
        a = a - lr * v
        a = project_simplex(a)

    n = len(a_init)
    a = project_simplex(a_init.astype(np.float64).copy())

    best_score = compute_score(a)

        contrib = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        # Find top and bottom contributors
        n_move = max(1, n // 100)
        top = np.argsort(contrib)[-n_move:]
        bot = np.argsort(contrib)[:n_move]

        improved = False
        for _ in range(10):
            candidate = a.copy()
            frac = rng.uniform(0.01, 0.05)
            # Move mass from top to bottom
            for i in top:
                amount = candidate[i] * frac
                candidate[i] -= amount
                candidate[rng.choice(bot)] += amount
        
        candidate = np.maximum(candidate, 0)
        s_total = np.sum(candidate)
        if s_total > 1e-10:
            candidate /= s_total
        
        s = compute_score(candidate)
        if s < best_score:
            best_score = s
            best_a = candidate.copy()
            a = candidate
            conv = fft_convolve(a)
            improved = True
        
        # Also try random single-pair moves
        for _ in range(5):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            candidate = a.copy()
            amount = min(candidate[i], rng.uniform(0.001, 0.01))
            candidate[i] -= amount
            candidate[j] += amount
            s = compute_score(candidate)
            if s < best_score:
                best_score = s
                best_a = candidate.copy()
                a = candidate
                conv = fft_convolve(a)
                improved = True
        
        step += 1
        if step % 100 == 0 and not improved:
            # Perturb
            a = a + 0.005 * rng.standard_normal(n)
            a = project_simplex(a)
            conv = fft_convolve(a)
    
    return best_a, best_score
