def soft_max_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    
    score = 2.0 * n * soft_max_val
    
    # dc[k]/da[j] = 2*a[k-j] if 0<=k-j<n
    # grad_a[j] = 2 * 2n * sum_k w[k] * a[k-j] * 1[0<=k-j<n]  (the outer 2 is from d(soft_max)/d(c[max]) chain)
    # Wait: score = 2n * soft_max and soft_max = c_max + tau*log(sum exp((c-c_max)/tau))
    # d(score)/d(a[j]) = 2n * sum_k w[k] * dc[k]/da[j] = 2n * sum_k w[k] * 2*a[k-j]*1[0<=k-j<n]
    # = 4n * sum_k w[k] * a[k-j]*1[0<=k-j<n]
    # = 4n * cross_corr_w[j]
    
    a_rev = a[::-1]
    size2 = 1
    while size2 < 2 * (2*n - 1) - 1:
        size2 <<= 1
    fa_rev = np.fft.rfft(a_rev, size2)
    fw = np.fft.rfft(w, size2)
    conv_aw = np.fft.irfft(fa_rev * fw, size2)[:len(a_rev) + len(w) - 1]
    grad_a = 4.0 * n * conv_aw[n-1:2*n-1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def multiplicative_gradient_step(a, eta, c_peak_idx):
    """
    Multiplicative weight update targeting the peak.
    a[j] *= exp(-eta * grad_j) where grad_j is the subgradient of the max.
    """
    n = len(a)
    c = fft_convolve(a, a)
    peak_val = np.max(c)
    
    # Subgradient: find all positions where c is at (or near) the max
    # Use soft version: weight by c[k]/peak_val
    threshold = 0.99 * peak_val
    active = np.where(c >= threshold)[0]
    
    # grad_a[j] = 2 * sum_{k in active} a[k-j] * 1[0<=k-j<n] / len(active)
    # (averaged subgradient)
    grad_a = np.zeros(n)
    a_rev = a[::-1]
    for k in active:
        # dc[k]/da[j] = 2*a[k-j] for 0<=k-j<n
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        j_indices = np.arange(lo, hi+1)
        k_minus_j = k - j_indices
        grad_a[j_indices] += 2.0 * a[k_minus_j]
    
    grad_a /= len(active)
    
    # Multiplicative update
    a_new = a * np.exp(-eta * grad_a)
    return a_new


def targeted_local_search(a_init, time_limit, rng, verbose=True):
    """
    Local search with targeted peak reduction using cooperative moves.
    """
    current = a_init.astype(np.float64).copy()
    s = np.sum(current)
    current = current / s
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    # Temperature for simulated annealing
    temperature = 0.001
    cooling = 0.99995
    best_iter = 0
    no_improve = 0
    
    iters = 0
    last_print = t0
    
    while time.monotonic() - t0 < time_limit:
        batch_start = time.monotonic()
        
        # Generate a batch of moves
        r = rng.random()
        
        if r < 0.25:
            # Peak-directed move: reduce elements contributing most to the peak
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            
            # Compute contribution of each element to the peak
            contribs = np.zeros(n)
            lo = max(0, peak_idx - (n-1))
            hi = min(n-1, peak_idx)
            for j in range(lo, hi+1):
                k_minus_j = peak_idx - j
                contribs[j] = current[j] * current[k_minus_j]
            
            # Find top contributors
            top_k = max(5, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            
            candidate = current.copy()
            # Reduce top contributors
            reductions = np.exp(rng.normal(-0.02, 0.01, size=len(top_idx)))
            candidate[top_idx] *= reductions
            # Increase some non-contributing elements
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(15, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.01, size=n_inc))
        
        elif r < 0.45:
            # Gradient-based move using soft subgradient
            c = fft_convolve(current, current)
            c_max = np.max(c)
            tau = c_max * 0.05  # soft threshold
            w = np.exp((c - c_max) / tau)
            w /= np.sum(w)
            
            # Compute gradient
            a_rev = current[::-1]
            size2 = 1
            while size2 < 2 * (2*n-1) - 1:
                size2 <<= 1
            fa = np.fft.rfft(a_rev, size2)
            fw = np.fft.rfft(w, size2)
            conv_aw = np.fft.irfft(fa * fw, size2)[:3*n-2]
            grad = 4.0 * conv_aw[n-1:2*n-1]
            
            # Normalized gradient step
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 1e-10:
                eta = np.max(current) * 0.02 / grad_norm * rng.uniform(0.5, 1.5)
                candidate = current * np.exp(-eta * grad)
            else:
                candidate = current.copy()
        
        elif r < 0.60:
            # Block swap: move a block from one region to another
            candidate = current.copy()
            block_size = max(2, int(rng.integers(2, max(3, n // 10))))
            src_start = int(rng.integers(0, n - block_size))
            dst_start = int(rng.integers(0, n - block_size))
            if src_start != dst_start and not (src_start < src_start + block_size and dst_start < dst_start + block_size and dst_start < src_start + block_size and dst_start + block_size > src_start):
                # Non-overlapping
                if not (src_start < dst_start + block_size and dst_start < src_start + block_size):
                    temp = candidate[dst_start:dst_start+block_size].copy()
                    candidate[dst_start:dst_start+block_size] = candidate[src_start:src_start+block_size]
                    candidate[src_start:src_start+block_size] = temp
        
        elif r < 0.75:
            # Smooth perturbation
            sigma = 0.02 * rng.uniform(0.5, 2.0)
            candidate = current * np.exp(rng.normal(0, sigma, size=n))
        
        elif r < 0.85:
            # Anti-symmetric perturbation (breaks palindromic symmetry)
            candidate = current.copy()
            half = n // 2
            scale = rng.uniform(0.01, 0.05)
            candidate[:half] *= (1 + scale * rng.standard_normal(half))
            candidate[half:] *= (1 - scale * rng.standard_normal(n - half))
        
        else:
            # Random restart of a small region
            candidate = current.copy()
            region_size = max(3, int(rng.integers(3, max(4, n // 5))))
            start = int(rng.integers(0, n - region_size))
            candidate[start:start+region_size] = rng.dirichlet(np.ones(region_size) * rng.uniform(1, 5))
            candidate[start:start+region_size] *= current[start:start+region_size].sum() / candidate[start:start+region_size].sum()
        
        candidate = np.maximum(candidate, 1e-15)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                best_iter = iters
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score
            no_improve += 1
        else:
            no_improve += 1
        
        temperature = max(temperature * cooling, 1e-10)
        
        iters += 1
        if verbose and time.monotonic() - last_print > 10.0:
            last_print = time.monotonic()
            print(f"    TLA: iter={iters}, score={current_score:.8f}, best={best_score:.8f}, t={time.monotonic()-t0:.0f}s")
    
    return best, best_score


def fine_hill_climb(a_init, time_limit, rng):
    """
    Very fine-grained hill climbing: try small perturbations of individual elements.
    """
    current = a_init.astype(np.float64).copy()
    s = np.sum(current)
    current = current / s
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    # Try perturbations of decreasing magnitude
    magnitudes = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    mag_idx = 0
    
    while time.monotonic() - t0 < time_limit and mag_idx < len(magnitudes):
        mag = magnitudes[mag_idx]
        improved = False
        
        # Try each element
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            for delta_factor in [mag, -mag, mag*0.5, -mag*0.5]:
                candidate = current.copy()
                candidate[j] = max(candidate[j] * (1 + delta_factor), 1e-15)
                candidate = candidate / np.sum(candidate)
                new_score = local_score(candidate)
                
                if new_score < current_score - 1e-12:
                    current = candidate
                    current_score = new_score
                    improved = True
                    if current_score < best_score:
                        best_score = current_score
                        best = current.copy()
                    break
        
        if not improved:
            mag_idx += 1
        
        print(f"    HHC: mag={mag}, score={current_score:.8f}, t={time.monotonic()-t0:.0f}s")
    
    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2 * np.pi * t * t / (2*n))
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Bell-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def make_ramp_init(n, rng, bias=0.3):
    """Ramp-shaped initialization (one side heavier)."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + bias * (t / (n-1) - 0.5) * 2
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def run_optimization(theta_init, n, deadline, p_schedule, maxiter_per_p, rng):
    """Run L-BFGS with p-norm annealing."""
    best_theta = theta_init.copy()
    best_score = local_score(softmax(theta_init))
    
    for p in p_schedule:
        if time.monotonic() > deadline - 20:
            break
        
        try:
            result = minimize(
                pnorm_obj_grad, theta_init.copy(),
                method='L-BFGS-B',
                jac=True, args=(p,),
                options={'maxiter': maxiter_per_p, 'ftol': 1e-15, 'gtol': 1e-13, 'maxfun': maxiter_per_p * 5}
            )
            a_new = softmax(result.x)
            s = local_score(a_new)
            
            if s < best_score:
                best_score = s
                best_theta = result.x.copy()
                theta_init = result.x.copy()
        except Exception as e:
            print(f"    L-BFGS error at p={p}: {e}")
    
    return best_theta, best_score


def run_softmax_optimization(theta_init, n, deadline, tau_schedule, maxiter_per_tau):
    """Run L-BFGS with soft-max temperature annealing."""
    best_theta = theta_init.copy()
    best_score = local_score(softmax(theta_init))
    
    for tau in tau_schedule:
        if time.monotonic() > deadline - 20:
            break
        
        try:
            result = minimize(
                soft_max_obj_grad, theta_init.copy(),
                method='L-BFGS-B',
                jac=True, args=(tau,),
                options={'maxiter': maxiter_per_tau, 'ftol': 1e-15, 'gtol': 1e-13, 'maxfun': maxiter_per_tau * 5}
            )
            a_new = softmax(result.x)
            s = local_score(a_new)
            
            if s < best_score:
                best_score = s
                best_theta = result.x.copy()
                theta_init = result.x.copy()
        except Exception as e:
            print(f"    Softmax L-BFGS error at tau={tau}: {e}")
    
    return best_theta, best_score
