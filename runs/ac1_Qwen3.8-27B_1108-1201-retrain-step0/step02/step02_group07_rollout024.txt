"""
Spectral-subgradient hybrid optimizer for minimizing the autocorrelation peak ratio.
Uses Gerchberg-Saxton spectral flattening, Nesterov-accelerated subgradient descent,
and peak-targeted simulated annealing.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_norm(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(conv, a, t, top_k=20):
    """
    Smoothed subgradient of max(conv) using softmax weighting over top peaks.
    Only computes contributions from the top_k peaks to save time.
    """
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    
    # Get top_k peaks
    if top_k < len(conv):
        top_indices = np.argsort(conv)[::-1][:top_k]
        peak_vals = conv[top_indices]
    else:
        top_indices = np.arange(len(conv))
        peak_vals = conv
    
    # Softmax weighting over peaks
    scaled = t * peak_vals / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(peak_vals)
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    # Compute subgradient contributions
    sg = np.zeros(n)
    for i, ki in enumerate(top_indices):
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[i] * a[k - je:k - js + 1][::-1]
    return sg


def spectral_flatten(a, n_iter=200, target_flatness=0.1, time_limit=None):
    """
    Gerchberg-Saxton style spectral flattening.
    Iteratively modifies the DFT magnitude to be more uniform,
    then projects back to non-negative time domain.
    """
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    alpha = 0.5  # Mixing parameter
    
    for it in range(n_iter):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        # Compute DFT
        A = np.fft.rfft(a, 2 * n - 1)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        
        # Target: flatten the magnitude (except DC component)
        # Use a power-law flattening
        target_mag = np.ones_like(mag)
        target_mag[0] = mag[0]  # Keep DC
        # Mix current with target
        new_mag = (1 - alpha) * mag + alpha * target_mag
        new_mag[0] = mag[0]  # Preserve DC (sum)
        
        # Reconstruct
        A_new = new_mag * phase
        a_new = np.fft.irfft(A_new, 2 * n - 1)[:n]
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            a_new = a.copy()
        else:
            a_new /= s_new
        
        # Evaluate
        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_new.copy()
        
        # Adaptive alpha
        if cur_obj > best_obj * 1.001:
            alpha *= 0.95
            a = a.copy()  # Don't update if worse
        else:
            a = a_new.copy()
    
    return best_a, best_obj


def nesterov_subgradient(a0, time_limit, t_init=80.0, lr_init=0.5, momentum=0.9):
    """
    Nesterov-accelerated subgradient method on the simplex.
    Uses backtracking for step size.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)  # Momentum
    t = t_init
    lr = lr_init
    patience = 0
    max_pat = 30
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov look-ahead
        a_look = project_simplex(a - lr * v)
        
        conv = fast_convolve(a_look)
        sg = compute_subgradient(conv, a_look, t, top_k=min(30, n))
        
        # Update momentum
        v = momentum * v + (1 - momentum) * sg
        
        # Step
        a_new = project_simplex(a - lr * v)
        
        # Check
        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
        
        if patience >= max_pat:
            patience = 0
            lr *= 0.7
            t *= 1.2
            momentum = min(0.99, momentum + 0.01)
            if lr < 0.005:
                lr = lr_init * 0.3
                t = t_init * 2.0
                momentum = 0.9
    
    return best_a, best_obj


def peak_targeted_anneal(a0, time_limit, rng, n_peaks=10):
    """
    Simulated annealing that specifically targets the top peaks.
    Moves probability mass from positions contributing to peaks
    to positions with low peak contribution.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1.0))
        T = 0.02 * (1.0 - frac) ** 2 + 0.0003
        
        # Find current peaks
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # Get top peaks
        peak_thresh = max_c * 0.85
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        
        # Pick a peak
        k = int(rng.choice(peak_positions))
        
        # Find positions contributing most to this peak
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        
        # Source: weighted by contribution to peak
        probs_src = products / total_prod
        # Boost high contributors
        probs_src = probs_src ** 1.5
        probs_src = probs_src / probs_src.sum()
        
        src_idx = js + int(rng.choice(je - js + 1, p=probs_src))
        
        # Target: positions with low total peak contribution
        # Compute "peak exposure" for each position
        exposure = np.zeros(n)
        for pk in peak_positions[:min(n_peaks, len(peak_positions))]:
            pk = int(pk)
            js2 = max(0, pk - n + 1)
            je2 = min(n - 1, pk)
            exp_src = a[js2:je2 + 1] * 0 + 1.0  # uniform within range
            exp_tgt = a[pk - je2:pk - js2 + 1][::-1]
            exposure[js2:je2 + 1] += exp_tgt
            exposure[pk - je2:pk - js2 + 1] += exp_src
        
        # Normalize
        exposure = np.maximum(exposure, 0)
        exp_sum = exposure.sum()
        if exp_sum < 1e-15:
            target_idx = int(rng.integers(n))
        else:
            # Prefer low exposure, with some randomness
            probs_tgt = 1.0 / (exposure + exposure.mean() * 0.5)
            probs_tgt = probs_tgt / probs_tgt.sum()
            target_idx = int(rng.choice(n, p=probs_tgt))
        
        if target_idx == src_idx:
            continue
        
        # Amount to move
        delta = a[src_idx] * rng.uniform(0.005, 0.04)
        if delta < 1e-10:
            continue
        
        a[src_idx] -= delta
        a[target_idx] += delta
        
        if a[src_idx] < 0:
            a[src_idx] += delta
            a[target_idx] -= delta
            continue
        
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()

            conv = new_conv

                best_a = a.copy()
        else:
            accept_prob = np.exp(-max(0, delta_obj) / max(T, 1e-12))
            if rng.random() < accept_prob:
                cur_obj = new_obj
                conv = new_conv
            else:
                a[src_idx] += delta
                a[target_idx] -= delta
    
    return best_a, best_obj

"""
Spectral-subgradient hybrid optimizer for minimizing autocorrelation peak ratio.
Combines Gerchberg-Saxton spectral flattening, Nesterov-accelerated subgradient
descent, efficient coordinate descent, and peak-targeted simulated annealing.
Explores multiple sequence lengths to avoid local minima.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_norm(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Normalize to probability vector."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_subgradient(conv, a, t, top_k=25):
    """
    Smoothed subgradient of max(conv) using softmax over top peaks.
    """
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)

    if top_k < len(conv):
        top_indices = np.argsort(conv)[::-1][:top_k]
        peak_vals = conv[top_indices]
    else:
        top_indices = np.arange(len(conv))
        peak_vals = conv

    scaled = t * peak_vals / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(peak_vals)
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    for i_idx in range(len(top_indices)):
        k = int(top_indices[i_idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[i_idx] * a[k - je:k - js + 1][::-1]
    return sg


def spectral_flatten(a, n_iter=300, time_limit=60.0, alpha_init=0.3):
    """
    Gerchberg-Saxton style spectral flattening.
    Iteratively flattens the DFT magnitude then projects back to non-negative.
    """
    a = normalize(a.copy())
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    alpha = alpha_init
    stall = 0

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 50 == 0 and stall > 100:
            break

        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-30)
        phase = A / mag_safe

        # Flatten magnitude: push towards geometric mean (excluding DC)
        dc = mag[0]
        non_dc_mag = mag[1:]
        target_level = np.geometric_mean(non_dc_mag)
        
        # Mix: new magnitude = (1-alpha)*current + alpha*target
        new_mag = np.empty_like(mag)
        new_mag[0] = dc
        new_mag[1:] = (1.0 - alpha) * non_dc_mag + alpha * target_level

        # Reconstruct
        A_new = new_mag * phase
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        # Project
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            a_new = a.copy()
        else:
            a_new /= s_new

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            alpha = min(0.6, alpha * 1.02)
        else:
            stall += 1
            alpha = max(0.05, alpha * 0.97)

    return best_a, best_obj


def nesterov_subgradient(a0, time_limit=60.0, t_init=80.0, lr_init=0.4, momentum_init=0.9):
    """
    Nesterov-accelerated subgradient method on the simplex.
    """
    a = normalize(a0.copy())
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t = t_init
    lr = lr_init
    momentum = momentum_init
    patience = 0
    max_pat = 35
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov look-ahead
        a_look = project_simplex(a - lr * v)

        conv_look = fast_convolve(a_look)
        sg = compute_subgradient(conv_look, a_look, t, top_k=min(30, n))

        # Momentum update
        v = momentum * v + (1.0 - momentum) * sg

        # Step
        a_new = project_simplex(a - lr * v)

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1

        if patience >= max_pat:
            patience = 0
            lr *= 0.7
            t *= 1.15
            momentum = min(0.99, momentum + 0.005)
            if lr < 0.004:
                lr = max(lr_init * 0.2, 0.003)
                t = t_init * 1.5
                momentum = momentum_init
                v = v * 0.5

    return best_a, best_obj


def coordinate_descent(a0, time_limit=60.0, num_candidates=5):

def peak_targeted_anneal(a0, time_limit=120.0, rng=None, n_peaks=12):

        elapsed = time.monotonic() - t0

        if max_c < 1e-15:

        peak_thresh = max_c * (0.80 + 0.10 * frac)

        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)

        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b

    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    inits['uniform'] = np.ones(n)

    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            inits['parent'] = pp

    for sf in [0.12, 0.2, 0.35]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        inits['g' + str(sf)] = g

    for freq in [1, 2, 3]:
        c = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        inits['c' + str(freq)] = c

    w = max(1, n // 5)
    bi = np.zeros(n)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bimodal'] = bi

    inits['rand1'] = rng.dirichlet(np.ones(n) * 1.0) * n
    inits['rand2'] = rng.dirichlet(np.ones(n) * 0.5) * n

    # Sawtooth
    saw = np.arange(1, n + 1, dtype=np.float64)
    inits['saw'] = saw

    # Chirp
    chirp = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * x / (2.0 * n)))
    chirp = np.maximum(chirp, 0)
    if chirp.sum() > 1e-10:
        inits['chirp'] = chirp

    # Triangular
    tri = 1.0 - np.abs(2.0 * x / n - 1.0)
    inits['tri'] = np.maximum(tri, 0.01)

    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Main entry point. Hybrid optimizer using spectral flattening,
    Nesterov subgradient, coordinate descent, and peak-targeted annealing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)

    parent = get_parent()

    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d" % (score, len(a)))

    # Phase 1: Multi-length spectral exploration
    print("Phase 1: Multi-length spectral exploration...")
    t1 = time.monotonic()

    scan_lengths = []
    if parent is not None:
        scan_lengths.append(len(parent))
    for n in [128, 192, 256, 320, 384, 512]:
        if n not in scan_lengths:
            scan_lengths.append(n)

    phase1_budget = min(700.0, (deadline - t_start) * 0.24)
    candidates = []

    for n in sorted(scan_lengths):
        if time.monotonic() > deadline - 1500:
            break
        t_alloc = max(60.0, phase1_budget / len(scan_lengths))
        if time.monotonic() + t_alloc > deadline - 1500:
            break

        inits = generate_initializations(n, rng, parent)
        n_inits = len(inits)
        t_per_init = max(5.0, t_alloc / n_inits * 0.7)

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 1500:
                break
            a0 = normalize(a0)
            t_spec = min(t_per_init * 0.5, 25.0)
            t_sg = min(t_per_init * 0.5, 20.0)

            a_spec, obj_spec = spectral_flatten(a0, n_iter=150, time_limit=t_spec, alpha_init=0.3)
            a_sg, obj_sg = nesterov_subgradient(a_spec, time_limit=t_sg, t_init=60.0, lr_init=0.35)

            record(a_spec)
            record(a_sg)
            candidates.append((min(obj_spec, obj_sg), n, name, a_sg))

    print("  Phase 1 done. Best: %.6f, candidates: %d" % (best_score, len(candidates)))
    candidates.sort(key=lambda x: x[0])

    # Phase 2: Deep optimization on top candidates
    print("Phase 2: Deep optimization...")
    top_k = min(8, len(candidates))
    phase2_budget = min(1100.0, deadline - time.monotonic() - 500)
    time_per = max(40.0, phase2_budget / max(1, top_k))

    for i in range(top_k):
        if time.monotonic() > deadline - 500:
            break
        sc, n, name, a0 = candidates[i]
        t_alloc = min(time_per, deadline - time.monotonic() - 500)
        if t_alloc < 30:
            break

        print("  [%d/%d] n=%d, %s, obj=%.6f, budget=%.0fs" % (i + 1, top_k, n, name, sc, t_alloc))
        a_cur = normalize(a0)

        # Multi-stage pipeline
        stages = [
            ('spec', 0.18),
            ('sg', 0.22),
            ('spec', 0.15),
            ('cd', 0.20),
            ('sg', 0.18),
            ('spec', 0.07),
        ]

        for st_name, st_frac in stages:
            if time.monotonic() > deadline - 500:
                break
            t_sub = t_alloc * st_frac
            if t_sub < 3.0:
                continue

            if st_name == 'spec':
                a_cur, _ = spectral_flatten(a_cur, n_iter=200, time_limit=t_sub, alpha_init=0.25)
            elif st_name == 'sg':
                a_cur, _ = nesterov_subgradient(a_cur, time_limit=t_sub, t_init=80.0, lr_init=0.3, momentum_init=0.9)
            elif st_name == 'cd':
                a_cur, _ = coordinate_descent(a_cur, time_limit=t_sub)

            record(a_cur)

    print("  Phase 2 done. Best: %.6f" % best_score)

    # Phase 3: Peak-targeted annealing on best
    if best_sequence is not None and time.monotonic() < deadline - 150:
        print("Phase 3: Peak-targeted annealing...")
        phase3_budget = min(400.0, deadline - time.monotonic() - 100)
        n_anneal = 5
        anneal_per = phase3_budget / n_anneal

        for i in range(n_anneal):
            if time.monotonic() > deadline - 100:
                break
            a_start = normalize(best_sequence.copy())
            rng_a = np.random.default_rng(seed + 12345 + i * 31)
            
            if i < n_anneal // 2:
                a_opt, _ = peak_targeted_anneal(a_start, time_limit=anneal_per, rng=rng_a, n_peaks=15)
            else:
                # Combine: brief spectral then anneal
                t_half = anneal_per * 0.3
                a_start2, _ = spectral_flatten(a_start, n_iter=80, time_limit=t_half, alpha_init=0.2)
                a_opt, _ = peak_targeted_anneal(a_start2, time_limit=anneal_per * 0.7, rng=rng_a, n_peaks=20)
            
            record(a_opt)

    print("  Phase 3 done. Best: %.6f" % best_score)

    # Phase 4: Final polish
    if best_sequence is not None and time.monotonic() < deadline - 15:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 15
        if remaining > 40:
            a_best = normalize(best_sequence.copy())

            # Fine spectral
            t_sp = remaining * 0.30
            a_opt, _ = spectral_flatten(a_best, n_iter=400, time_limit=t_sp, alpha_init=0.15)
            record(a_opt)

            # Fine subgradient
            if time.monotonic() < deadline - 15:
                rem = deadline - time.monotonic() - 15
                t_sg2 = rem * 0.35
                a_opt2, _ = nesterov_subgradient(a_best, time_limit=t_sg2, t_init=150.0, lr_init=0.08, momentum_init=0.95)
                record(a_opt2)

            # Fine anneal
            if time.monotonic() < deadline - 15:
                rem = deadline - time.monotonic() - 15
                if rem > 8:
                    rng_f = np.random.default_rng(seed + 77777)
                    a_opt3, _ = peak_targeted_anneal(a_best, time_limit=rem, rng=rng_f, n_peaks=25)
                    record(a_opt3)

            # Coordinate descent final
            if time.monotonic() < deadline - 15:
                rem = deadline - time.monotonic() - 15
                if rem > 10:
                    a_opt4, _ = coordinate_descent(a_best, time_limit=rem)
                    record(a_opt4)

    # Fallback
    if best_sequence is None:
        print("WARNING: No valid sequence found, using fallback.")
        best_sequence = np.ones(128)
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))
    elapsed = time.monotonic() - t_start
    print("Total time: %.1f s" % elapsed)

    return [float(x) for x in best_sequence]
