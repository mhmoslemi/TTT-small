```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.

Strategy:
- Start from the provided sequence if available, plus strong analytic seeds.
  A one-sided discrete 1/sqrt(x) density is close to continuous optimal
  and typically starts near 1.57.
- Use L-BFGS-B on log-softmax parameters with a smooth max (log-sum-exp)
  of the self-convolution.
- Search sparse supports derived from the current best, optimize weights on
  those supports, and perturb supports combinatorially.
- Add an alternating-projection / polynomial square-root step for flat
  convolution targets, plus coordinate polish and Adam polish as backups.
"""
import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def _next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def _fft_size(n):
    # Large enough for self convolution (2n-1) and gradient convolution (3n-2).
    return _next_pow2(3 * int(n) - 2)


def _conv_self(a, F):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    c = irfft(rfft(a, F) * rfft(a, F), F)
    out = c[:2 * n - 1]
    np.maximum(out, 0.0, out=out)
    return out


def _conv_two(x, y, F):
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    out_len = len(x) + len(y) - 1
    c = irfft(rfft(x, F) * rfft(y, F), F)[:out_len]
    np.maximum(c, 0.0, out=c)
    return c


def _grad_from_q(q, a, F):
    """grad_w = 2 * sum_k q_k a_{k-i} for each i."""
    n = len(a)
    c = _conv_two(q, a[::-1], F)
    return 2.0 * c[n - 1:2 * n - 1]


def _softmax(v):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v.copy()
    v = v - np.max(v)
    e = np.exp(v)
    s = e.sum()
    if not np.isfinite(s) or s <= 0:
        return np.full(v.shape, 1.0 / v.size, dtype=np.float64)
    return e / s


def _normalize(a, S):
    a = np.array(a, dtype=np.float64, copy=True)
    np.clip(a, 0.0, None, out=a)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        a[0] = float(S)
    else:
        a *= float(S) / s
    return a


def _maxb(a, F):
    b = _conv_self(a, F)
    return float(np.max(b))


def _objective_general(a, n, F):
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum())
    if s <= 1e-12 or not np.isfinite(s):
        return np.inf
    b = _conv_self(a, F)
    return float(2.0 * int(n) * float(np.max(b)) / (s * s))


def _theta_from(a, S):
    a = np.asarray(a, dtype=np.float64)
    eps = max(1e-12, 1e-8 * float(S))
    safe = np.maximum(a, eps)
    s = safe.sum()
    if s <= 0:
        return np.zeros(len(a), dtype=np.float64)
    return np.log(safe / s)


def _soft_obj_grad(a, b, T, F):
    cmax = float(np.max(b))
    n = len(a)
    if not np.isfinite(cmax) or cmax <= 0:
        return cmax, np.zeros(n, dtype=np.float64)
    shifted = float(T) * (b - cmax)
    exp = np.exp(shifted)
    lse = float(exp.sum())
    if not np.isfinite(lse) or lse <= 1e-300:
        q = np.zeros(len(b), dtype=np.float64)
        q[int(np.argmax(b))] = 1.0
    else:
        q = exp / lse
    soft = cmax + np.log(lse) / float(T)
    gw = _grad_from_q(q, a, F)
    return soft, gw


def _full_theta_obj(theta, n, S, F, T, state):
    elapsed = time.monotonic() - state['start']
    if elapsed > state['budget']:
        return state['last_obj'], np.zeros(len(theta), dtype=np.float64)

    a = float(S) * _softmax(theta)
    b = _conv_self(a, F)
    obj = float(np.max(b))

    if np.isfinite(obj) and obj < state['best_obj']:
        state['best_obj'] = obj
        state['best_a'] = a.copy()

    soft, gw = _soft_obj_grad(a, b, T, F)
    dot = float(np.dot(gw, a))
    grad = gw * a - (a / float(S)) * dot
    state['last_obj'] = soft
    return soft, grad


def _sparse_theta_obj(theta, state):
    elapsed = time.monotonic() - state['start']
    if elapsed > state['budget']:
        return state['last_obj'], np.zeros(len(theta), dtype=np.float64)

    idx = state['idx']
    n = state['n']
    S = state['S']
    F = state['F']
    T = state['T']

    w = float(S) * _softmax(theta)
    a = state['a_buf']
    a[:] = 0.0
    a[idx] = w

    b = _conv_self(a, F)
    obj = float(np.max(b))

    if np.isfinite(obj) and obj < state['best_obj']:
        state['best_obj'] = obj
        state['best_a'] = a.copy()

    soft, gw_full = _soft_obj_grad(a, b, T, F)
    gw = gw_full[idx]
    dot = float(np.dot(gw, w))
    grad = gw * w - (w / float(S)) * dot
    state['last_obj'] = soft
    return soft, grad

    return {

        'maxiter': 3000,
        'maxls': 50,
        'gtol': 1e-10,
        'ftol': 1e-12,
    }

    if budget <= 2.0:
        a = _normalize(float(S) * _softmax(theta), S)
        return a, _maxb(a, F)

    state = {
        'start': time.monotonic(),
        'budget': float(budget),
        'best_obj': np.inf,
        'best_a': None,
        'last_obj': 0.0,
    }
    try:
        minimize(
            _full_theta_obj,
            np.asarray(theta, dtype=np.float64),

            method='L-BFGS-B',
            jac=True,

    except Exception:
        pass

    if state['best_a'] is None:
        a = _normalize(float(S) * _softmax(theta), S)
        return a, _maxb(a, F)
    return state['best_a'], state['best_obj']

        orig * 2.0,
        orig * 4.0,
        mean,
        0.5 * mean,
        2.0 * mean,
        float(S),
    ]
    out = []
    last = -1.0
    for x in cands:
        if not np.isfinite(x):
            continue
        x = float(min(max(x, 0.0), float(S)))
        if x < 0.0:
            x = 0.0
        if last >= 0.0 and abs(x - last) <= 1e-12 * max(1.0, float(S)):
            continue
        out.append(x)
        last = x
        if len(out) >= 10:
            break
    return out


def _coordinate_polish(a, idx, S, F, budget):
    a = _normalize(a, S).copy()
    if len(idx) == 0:
        return a, _maxb(a, F)

    t0 = time.monotonic()
    deadline = t0 + float(budget)
    best_a = a.copy()
    best_obj = _maxb(a, F)

    K = len(idx)
    while time.monotonic() < deadline - 0.5:
        improved = False
        for ipos in np.asarray(idx, dtype=int):
            if time.monotonic() >= deadline - 0.5:
                break
            ipos = int(ipos)
            orig = float(a[ipos])
            cands = _unique_positive_candidates(orig, S, K)
            local_best_x = orig
            local_best_obj = np.inf
            local_best_a = a.copy()
            base = a.copy()
            for x in cands:
                cand = base.copy()
                cand[ipos] = float(x)
                cand = _normalize(cand, S)
                m = _maxb(cand, F)
                if m < local_best_obj:
                    local_best_obj = m
                    local_best_x = float(x)
                    local_best_a = cand.copy()
            a = local_best_a.copy()
            if local_best_obj < best_obj - 1e-12:
                best_a = a.copy()
                best_obj = float(local_best_obj)
                improved = True
        if not improved:
            break
    return best_a, best_obj


def _shape_power(n, S, alpha, reverse=False, osc=0.0, phase=0.0):
    i = np.arange(n, dtype=np.float64)
    x = i + 0.5
    if reverse:
        x = (n - 1.0) - i + 0.5
    v = np.power(x, -float(alpha))
    if abs(osc) > 0.0:
        den = max(1, n - 1)
        v = v * (1.0 + float(osc) * np.sin(2.0 * np.pi * i / den + float(phase)))
        np.maximum(v, 1e-8, out=v)
    return _normalize(v, S)


def _best_initial_shape(n, S, F):
    cands = [np.ones(n, dtype=np.float64)]
    for alpha in (0.30, 0.45, 0.50, 0.60, 0.80, 1.20):
        cands.append(_shape_power(n, S, alpha, False))
        cands.append(_shape_power(n, S, alpha, True))
    for phase in (0.4, 1.1, 1.9, 2.8, 3.9, 5.1):
        cands.append(_shape_power(n, S, 0.50, False, 0.35, phase))
        cands.append(_shape_power(n, S, 0.55, True, 0.30, phase + 0.7))
    best = None
    best_obj = np.inf
    for c in cands:
        a = _normalize(c, S)
        obj = _maxb(a, F)
        if obj < best_obj:
            best_obj = obj
            best = a.copy()
    if best is None:
        best = _normalize(np.ones(n), S)
        best_obj = _maxb(best, F)
    return best, best_obj


def _top_idx(a, K):
    a = np.asarray(a, dtype=np.float64)
    K = min(max(0, int(K)), len(a))
    if K == 0:
        return np.array([], dtype=int)
    return np.argsort(-a)[:K].astype(int)


def _modify_indices(idx, n, rng, r):
    idx = np.asarray(idx, dtype=int).copy()
    K = len(idx)
    if K <= 2 or int(n) <= K:
        return idx
    r = int(min(max(1, r), K - 2))
    rem = rng.choice(K, size=r, replace=False)
    rem_set = set(int(x) for x in rem)
    new = []
    for i, p in enumerate(idx):
        if i not in rem_set:
            new.append(int(p))
    uniq = set(new)
    attempts = 0
    while len(uniq) < K and attempts < 20000:
        j = int(rng.integers(0, int(n)))
        if j not in uniq:
            uniq.add(j)
        attempts += 1
    j = 0
    while len(uniq) < K:
        if j not in uniq:
            uniq.add(j)
        j += 1
    arr = np.array(list(uniq), dtype=int)
    rng.shuffle(arr)
    arr_set = set(int(x) for x in arr)
    if 0 not in arr_set and rng.random() < 0.85:
        p = int(rng.integers(0, arr.shape[0]))
        arr[p] = 0
        arr_set = set(int(x) for x in arr)
    if int(n - 1) not in arr_set and rng.random() < 0.85:
        cands = np.nonzero(arr != (n - 1))[0]
        if len(cands) > 0:
            arr[int(cands[int(rng.integers(len(cands)))]]) = n - 1
    return arr


def _optimize_support(theta, idx, n, S, F, budget):
    t0 = time.monotonic()
    deadline = t0 + float(budget)
    idx = np.asarray(idx, dtype=int)
    if len(idx) == 0:
        a = np.zeros(n, dtype=np.float64)
        a[0] = float(S)
        return a, _maxb(a, F)

    theta = np.asarray(theta, dtype=np.float64).copy()
    w = float(S) * _softmax(theta)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    best_a = _normalize(a, S).copy()
    best_obj = _maxb(best_a, F)

    T_list = (30.0, 100.0, 300.0)
    L_cap = 0.55 * float(budget)
    C_cap = 0.45 * float(budget)
    l_used = 0.0
    for T in T_list:
        if time.monotonic() >= deadline - max(3.0, 0.2 * C_cap):
            break
        per = max(0.0, min(L_cap / len(T_list), deadline - time.monotonic() - max(3.0, 0.2 * C_cap)))
        if per < 3.0:
            continue
        a2, obj2 = _run_sparse_lbfgs(theta, idx, n, S, F, T, per)
        l_used += per
        if a2 is not None:
            a2 = _normalize(a2, S)
            obj2 = _maxb(a2, F)
            if obj2 < best_obj:
                best_obj = obj2
                best_a = a2.copy()
        w = best_a[idx]
        if w.sum() > 1e-20:
            theta = np.log(np.maximum(w, 1e-300) / w.sum())

    c_budget = max(0.0, deadline - time.monotonic() - 1.0)
    c_budget = min(c_budget, C_cap)
    if c_budget > 4.0:
        a3, obj3 = _coordinate_polish(best_a, idx, S, F, c_budget)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
    return best_a, best_obj


def _support_search(base_a, n, S, F, budget, rng):
    t0 = time.monotonic()
    deadline = t0 + float(budget)
    best_a = _normalize(base_a, S).copy()
    best_obj = _maxb(best_a, F)

    Ks = (512, 1024)
    no_improve_total = 0

    for K in Ks:
        K = min(int(K), int(n))
        if time.monotonic() >= deadline - 40.0:
            break

        # Direct top-support optimum.
        idx = _top_idx(best_a, K)
        w = best_a[idx]
        if w.sum() <= 1e-20:
            w = np.full(len(idx), float(S) / max(1, len(idx)), dtype=np.float64)
        theta = np.log(w / w.sum())
        trial = min(90.0, max(20.0, 0.25 * (deadline - time.monotonic())))
        a_opt, obj_opt = _optimize_support(theta, idx, n, S, F, trial)
        if a_opt is not None and obj_opt < best_obj:
            best_a = _normalize(a_opt, S).copy()
            best_obj = float(obj_opt)
            no_improve_total = 0
        else:
            no_improve_total += 1

        # Perturbed support search.
        trial = min(35.0, max(12.0, 0.08 * (deadline - time.monotonic())))
        local_no = 0
        while time.monotonic() < deadline - 35.0:
            idx_curr = _top_idx(best_a, K)
            new_idx = _modify_indices(idx_curr, n, rng, max(1, int(0.05 * K)))
            wmap = np.zeros(n, dtype=np.float64)
            wmap[idx_curr] = best_a[idx_curr]
            w = wmap[new_idx].astype(np.float64).copy()
            pos = w[w > 1e-300]
            if pos.size:
                small = max(1e-12, 0.1 * float(np.median(pos)))
            else:
                small = float(S) / max(1, K)
            w[w <= 1e-300] = small
            wsum = float(w.sum())
            if wsum <= 1e-20:
                w[:] = float(S) / max(1, K)
            else:
                theta = np.log(w / wsum)

            a_opt, obj_opt = _optimize_support(theta, new_idx, n, S, F, trial)
            if a_opt is not None and obj_opt < best_obj:
                best_a = _normalize(a_opt, S).copy()
                best_obj = float(obj_opt)
                local_no = 0
                no_improve_total = 0
            else:
                local_no += 1
                no_improve_total += 1

            if local_no >= 8:
                break
            if no_improve_total >= 18:
                break
            trial = min(trial, 28.0)
        if no_improve_total >= 18:
            break

    return best_a, best_obj


def _polish_full(a, n, S, F, budget, rng):
    t0 = time.monotonic()
    deadline = t0 + float(budget)
    best_a = _normalize(a, S).copy()
    best_obj = _maxb(best_a, F)

    T_list = (50.0, 150.0, 400.0)
    rem = float(budget)
    for i, T in enumerate(T_list):
        if time.monotonic() >= deadline - 15.0:
            break
        m = len(T_list) - i
        perL = (rem / m) * 0.70
        perC = (rem / m) * 0.30
        rem -= perL + perC

        if perL > 4.0:
            theta = _theta_from(best_a, S)
            if rng is not None and i > 0:
                theta = theta + rng.normal(0.0, 0.012, size=n)
            a2, obj2 = _run_full_lbfgs(theta, n, S, F, T, perL)
            if a2 is not None:
                a2 = _normalize(a2, S)
                obj2 = _maxb(a2, F)
                if obj2 < best_obj:
                    best_obj = obj2
                    best_a = a2.copy()

        if time.monotonic() < deadline - 10.0 and perC > 3.0:
            idx = _top_idx(best_a, min(384, n))
            a3, obj3 = _coordinate_polish(best_a, idx, S, F, perC)
            if obj3 < best_obj:
                best_obj = obj3
                best_a = a3.copy()

    return best_a, best_obj


def _project_box_sum(y, M, total):
    y = np.asarray(y, dtype=np.float64)
    m = len(y)
    total = float(total)
    M = float(M)
    if M <= 0.0 or total <= 0.0:
        return np.zeros(m, dtype=np.float64)
    cap = m * M
    if total > cap * (1.0 - 1e-12):
        return None

    low = float(np.min(y)) - M - 1.0
    high = float(np.max(y)) + 1.0
    for _ in range(64):
        mid = 0.5 * (low + high)
        p = np.clip(y - mid, 0.0, M)
        s = float(p.sum())
        if s > total:
            low = mid
        else:
            high = mid

    lam = 0.5 * (low + high)
    p = np.clip(y - lam, 0.0, M)
    s = float(p.sum())
    diff = total - s
    if abs(diff) > 1e-9:
        if diff > 0:
            mask = p < (M - 1e-9)
        else:
            mask = p > 1e-9
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            p[mask] += diff / cnt
    p = np.clip(p, 0.0, M)
    s = float(p.sum())
    if s > 0 and abs(s - total) > 1e-7:
        p *= total / s
        p = np.clip(p, 0.0, M)
    return p


def _formal_sqrt(B, S, cap):
    m = len(B)
    n = (m + 1) // 2
    a0 = 1e-3
    if m > 0:
        b0 = float(B[0])
        if np.isfinite(b0) and b0 > 0.0:
            a0 = math.sqrt(b0)
        if not np.isfinite(a0) or a0 <= 1e-12:
            a0 = math.sqrt(max(1e-4, min(1e-4, float(cap))))
    a0 = min(max(a0, 1e-6), float(cap))

    a = np.zeros(n, dtype=np.float64)
    a[0] = a0
    denom = 2.0 * a0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        val = (float(B[k]) - s) / denom
        if not np.isfinite(val):
            val = 0.0
        if val < 0.0:
            val = 0.0
        if val > cap:
            val = cap
        a[k] = val

    s = float(a.sum())
    if s <= 1e-12:
        a[0] = float(S)
    else:
        a *= float(S) / s
    a = np.clip(a, 0.0, max(float(cap), float(S)))
    s = float(a.sum())
    if s > 0:
        a *= float(S) / s
    else:
        a[0] = float(S)
    return a


def _run_ap(a, n, S, F, budget, target):
    t0 = time.monotonic()
    deadline = t0 + float(budget)
    best_a = _normalize(a, S).copy()
    best_obj = _maxb(best_a, F)
    a = best_a.copy()
    m = 2 * int(n) - 1
    min_possible = (float(S) * float(S) / float(m)) * 1.0005
    no_improve = 0

    while time.monotonic() < deadline - 1.0:
        B = _conv_self(a, F)
        curopj = float(np.max(B))
        if curopj < best_obj:
            best_obj = curopj
            best_a = a.copy()

        if best_obj > target:
            Mproj = max(min_possible, target, best_obj - max(0.02, 0.012 * best_obj))
        else:
            Mproj = max(min_possible, target - 0.0015 * (no_improve + 1))

        Bp = _project_box_sum(B, Mproj, float(S) * float(S))
        if Bp is None:
            Bp = B.copy()
            np.minimum(Bp, Mproj, out=Bp)
            s = float(Bp.sum())
            if s > 0:
                Bp *= float(S) * float(S) / s

        if Bp[0] < 1e-3:
            Bp[0] = 1e-3

        cap = math.sqrt(max(Mproj, 1.0)) * 1.35
        A2 = _formal_sqrt(Bp, S, cap)
        improved = False
        for alpha in (0.25, 0.5, 1.0, 0.75, 0.35, 0.85):
            cand = (1.0 - alpha) * best_a + alpha * A2
            cand = _normalize(cand, S)
            obj = _maxb(cand, F)
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = cand.copy()
                a = cand.copy()
                improved = True
                no_improve = 0
                break

        if not improved:
            obj = _maxb(_normalize(A2, S), F)
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = _normalize(A2, S).copy()
                a = best_a.copy()
                no_improve = 0
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 35:
                break

    return best_a, best_obj


def _adam_full(a, n, S, F, budget, T=100.0):
    t0 = time.monotonic()
    deadline = t0 + float(budget)
    best_a = _normalize(a, S).copy()
    best_obj = _maxb(best_a, F)
    theta = _theta_from(best_a, S)
    N = len(theta)
    m = np.zeros(N, dtype=np.float64)
    v = np.zeros(N, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = 0.04

    while time.monotonic() < deadline - 1.0:
        t += 1
        a = float(S) * _softmax(theta)
        b = _conv_self(a, F)
        obj = float(np.max(b))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        _, gw = _soft_obj_grad(a, b, T, F)
        dot = float(np.dot(gw, a))
        grad = gw * a - (a / float(S)) * dot

        gn = float(np.linalg.norm(grad))
        if not np.isfinite(gn) or gn < 1e-16:
            break

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr_t = lr * (0.985 ** (t // 100))
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_obj


def _try_evaluate(final_seq):
    try:
        val = evaluate_sequence(final_seq)
    except Exception:
        return np.inf
    try:
        return float(val)
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    """Return a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(int(seed))
    total_budget = max(60.0, min(float(budget_s), 2400.0))
    deadline = time.monotonic() + total_budget - 25.0

    global_a = None
    global_obj = np.inf

    # Load the previous state-of-the-art sequence if present.
    given_raw = None
    try:
        given = globals().get('height_sequence_1', None)
        if given is not None:
            given_raw = np.asarray(given, dtype=np.float64).ravel()
            given_raw = np.clip(given_raw, 0.0, 1000.0)
            given_raw = given_raw[np.isfinite(given_raw)]
            if given_raw.size > 0 and given_raw.sum() > 0.01:
                n_given = int(given_raw.size)
                if n_given <= 20000:
                    F_given = _fft_size(n_given)
                    obj_given = _objective_general(given_raw, n_given, F_given)
                    if np.isfinite(obj_given):
                        global_a = given_raw.copy()
                        global_obj = float(obj_given)
                        print("Given sequence: n=%d obj=%.8f" % (n_given, global_obj))
    except Exception as e:
        print("Could not load given sequence: %s" % str(e))

    # Primary length: keep the expected construction length 4096.
    n = 4096
    if given_raw is not None and 1024 <= len(given_raw) <= 8192:
        # Prefer the given length if it already differs and looks stronger.
        n = int(given_raw.size)
    S = math.sqrt(2.0 * float(n))
    F = _fft_size(n)

    # Primary initial sequence.
    primary_a = None
    primary_obj = np.inf
    if given_raw is not None and len(given_raw) == n:
        primary_a = _normalize(given_raw, S)
        primary_obj = _maxb(primary_a, F)
        if primary_obj < global_obj:
            global_obj = primary_obj
            global_a = primary_a.copy()

    if primary_a is None:
        primary_a, primary_obj = _best_initial_shape(n, S, F)
        print("Primary shape init: n=%d obj=%.8f" % (n, primary_obj))
        if primary_obj < global_obj:
            global_obj = primary_obj
            global_a = primary_a.copy()

    def update_primary(a, obj):
        nonlocal primary_a, primary_obj
        if a is None or not np.isfinite(obj):
            return
        a = _normalize(a, S)
        obj = _maxb(a, F)
        if obj < primary_obj:
            primary_a = a.copy()
            primary_obj = float(obj)

    def update_global_len(a, obj):
        nonlocal global_a, global_obj
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.clip(a, 0.0, 1000.0)
        a = a[np.isfinite(a)]
        if a.size == 0 or a.sum() <= 0.01:
            return
        Fg = _fft_size(a.size) if a.size <= 20000 else None
        if Fg is None:
            return
        obj = _objective_general(a, a.size, Fg)
        if np.isfinite(obj) and obj < global_obj:
            global_obj = float(obj)
            global_a = a.copy()

    # Stage 1: full L-BFGS + coordinate polish from the best primary sequence.
    if time.monotonic() < deadline - 250.0:
        b = min(850.0, 0.30 * (deadline - time.monotonic()))
        t0 = time.monotonic()
        print("Full L-BFGS, budget=%.1fs, start=%.8f" % (b, primary_obj))
        primary_a, primary_obj = _polish_full(primary_a, n, S, F, b, rng)
        update_local = primary_obj
        print("Full L-BFGS done in %.1fs, best=%.8f" % (time.monotonic() - t0, update_local))
        update_global_len(primary_a, primary_obj)

    # Stage 2: sparse support search from the current primary best.
    if time.monotonic() < deadline - 220.0:
        b = min(1100.0, 0.35 * (deadline - time.monotonic()))
        t0 = time.monotonic()
        print("Support search, budget=%.1fs, start=%.8f" % (b, primary_obj))
        primary_a, primary_obj = _support_search(primary_a, n, S, F, b, rng)
        print("Support search done in %.1fs, best=%.8f" % (time.monotonic() - t0, primary_obj))
        update_global_len(primary_a, primary_obj)

    # Stage 3: full polish from the possibly sparser best.
    if time.monotonic() < deadline - 150.0:
        b = min(500.0, 0.18 * (deadline - time.monotonic()))
        t0 = time.monotonic()
        print("Second full polish, budget=%.1fs, start=%.8f" % (b, primary_obj))
        primary_a, primary_obj = _polish_full(primary_a, n, S, F, b, rng)
        print("Second full polish done in %.1fs, best=%.8f" % (time.monotonic() - t0, primary_obj))
        update_global_len(primary_a, primary_obj)

    # Stage 4: alternating projection / square-root search.
    if time.monotonic() < deadline - 90.0:
        target = min(1.5015, max(1.49, primary_obj - 0.002))
        b = min(450.0, 0.15 * (deadline - time.monotonic()))
        t0 = time.monotonic()
        print("AP search, budget=%.1fs, target=%.6f" % (b, target))
        a4, o4 = _run_ap(primary_a, n, S, F, b, target)
        if o4 < primary_obj:
            primary_a, primary_obj = a4, o4
        print("AP done in %.1fs, best=%.8f" % (time.monotonic() - t0, primary_obj))
        update_global_len(primary_a, primary_obj)

    # Stage 5: Adam polish and final coordinate polish.
    if time.monotonic() < deadline - 25.0:
        b = min(250.0, 0.5 * (deadline - time.monotonic()))
        t0 = time.monotonic()
        a5, o5 = _adam_full(primary_a, n, S, F, min(120.0, b), T=120.0)
        if o5 < primary_obj:
            primary_a, primary_obj = a5, o5
        update_global_len(primary_a, primary_obj)

    if time.monotonic() < deadline - 5.0:
        b = max(0.0, deadline - time.monotonic() - 5.0)
        idx = _top_idx(primary_a, min(512, n))
        a6, o6 = _coordinate_polish(primary_a, idx, S, F, b)
        if o6 < primary_obj:
            primary_a, primary_obj = a6, o6
        update_global_len(primary_a, primary_obj)

    # Also allow the provided sequence itself to be returned if it is strongest.
    update_global_len(primary_a, primary_obj)

    # Final fallback.
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)
        global_obj = _objective_general(global_a, 256, _fft_size(256))

    # Normalize the returned sequence on its own length for safety.
    fa = np.asarray(global_a, dtype=np.float64).ravel()
    fa = np.clip(fa, 0.0, 1000.0)
    fa = fa[np.isfinite(fa)]
    if fa.size == 0 or fa.sum() <= 0.01:
        fa = np.ones(256, dtype=np.float64)
    else:
        S_final = math.sqrt(2.0 * float(fa.size))
        fa = fa * (S_final / float(fa.sum()))

    final_seq = [float(x) for x in fa]
    final_val = _try_evaluate(final_seq)
    print("Final numeric: n=%d obj_internal=%.8f evaluate=%.8f" % (
        len(final_seq), float(global_obj),
        final_val if np.isfinite(final_val) else float('inf')))

    if not np.isfinite(final_val):
        final_seq = [1.0] * 256
        print("Fallback uniform: evaluate=%.8f" % _try_evaluate(final_seq))

    return final_seq
```
