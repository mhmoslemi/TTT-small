from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_for_lag(k, a, n):
    """Compute gradient of (a*a)[k] w.r.t. a.
    d/d a[m] (a*a)[k] = a[k-m] + a[k-m] (if k-m != m) = 2*a[k-m]

        if idx == m:
            grad[m] += a[idx]
        else:
            grad[m] += 2.0 * a[idx]
    return grad

    n = len(a)
    c = fft_convolve_self(a)

    # A[i, j] = a[k_i - j] if 0 <= k_i - j < n

            if g_sum > 1e-10:

        pass

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based directional optimization,
FFT-based subgradient steps, and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, weights, n):
    """Compute weighted sum of gradients of active lags.
    weights[k] = weight for lag k (length 2n-1).
    Returns gradient of length n.
    """
    # subgrad[m] = 2 * sum_k weights[k] * a[k-m] (approx, ignoring diagonal)
    # = 2 * xcorr(weights, a)[m]
    N = 1 << (2 * (2 * n - 1) - 1).bit_length()
    W = fft(weights, N)
    A = fft(a, N)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def solve_lp_direction(a, M, top_k=50):
    """LP: max sum(g) s.t. (a*g)[k] <= M for top-k lags, g >= 0."""
    n = len(a)
    c = fft_convolve_self(a)
    active_lags = np.argsort(c)[::-1][:top_k]
    
    A_mat = np.zeros((top_k, n))
    for i, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        A_mat[i, idx] = a[k - idx]
    
    c_obj = -np.ones(n)
    b_ub = np.full(top_k, M)
    
    try:
        result = linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                        bounds=[(0, None)] * n, method='highs')
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except:
        pass
    return None


def coordinate_descent_polish(a, S, time_budget, max_sweeps=100):
    """Polish using coordinate descent."""
    a = a.copy()
    n = len(a)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig <= 1e-12:
                continue
            # Try reducing a[i]
            # Binary search for best value in [0, orig*2]
            lo, hi = 0.0, orig * 2.0
            best_val = orig
            best_local = best_obj
            for _ in range(10):
                mid = (lo + hi) / 2.0
                a[i] = mid
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_test = a * (S / s)
                c_test = fft_convolve_self(a_test)
                obj_test = c_test.max()
                if obj_test < best_local:
                    best_local = obj_test
                    best_val = mid
                    lo = mid
                else:
                    hi = mid
            a[i] = best_val
            s = a.sum()
            if s < 1e-10:
                a = np.full(n, S / n)
            else:
                a = a * (S / s)
            c = fft_convolve_self(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        if not improved:
            break
    
    return best_a, best_obj
```

    """Fast coordinate descent using O(n) convolution update."""
    a = a.copy()
    n = len(a)
    start = time.monotonic()
    
    a_normalized = a * (S / a.sum())
    c = fft_convolve_self(a_normalized)
    best_obj = c.max()
    best_a = a_normalized.copy()
    
    # Precompute: if a[i] changes by delta, c[k] changes by ~2*delta*a[k-i]
    # (for the normalized version)
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            orig = a_normalized[i]
            if orig <= 1e-12:
                continue
            
            # Try a few candidate values
            candidates = [orig * 0.5, orig * 0.8, orig * 1.2, orig * 1.5, 0.0]
            best_val = orig
            best_c = c.copy()
            
            for val in candidates:
                if val < 0:
                    continue
                # Estimate new c
                delta = val - orig
                # c_new[k] ≈ c[k] + 2*delta*a[k-i] (for valid k-i)
                # But this is only approximate for large delta
                # For small delta, it's a good linear approximation
                
                # For accuracy, let's just do the full FFT for the best candidate
                a_test = a_normalized.copy()
                a_test[i] = val
                s = a_test.sum()
                if s < 1e-10:
                    continue
                a_test = a_test * (S / s)
                c_test = fft_convolve_self(a_test)
                obj_test = c_test.max()
                if obj_test < best_c.max():
                    best_c = c_test
                    best_val = val
            
            if best_val != orig:
                a_normalized[i] = best_val
                s = a_normalized.sum()
                a_normalized = a_normalized * (S / s)
                c = best_c  # Already computed
                if c.max() < best_obj:
                    best_obj = c.max()
                    best_a = a_normalized.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based directional optimization,
FFT-accelerated subgradient steps with active-set identification, and adaptive
coordinate descent polishing. Key improvements: (1) LP directions solve the
linearized constraint exactly, (2) subgradient uses precise active set via
FFT cross-correlation, (3) backtracking line search for robust step selection,
(4) periodic perturbation restarts to escape local minima, (5) multi-phase
timescheduling for optimal use of the 2400s budget.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient_fft(a, weights, n):
    """Compute weighted subgradient of max self-conv using FFT cross-correlation.
    weights[k] for k=0..2n-2. Returns grad of length n.
    subgrad[m] = 2 * sum_k weights[k] * a[k-m] (approx, ignoring diagonal)
    """
    N = 1 << (2 * (2 * n - 1) - 1).bit_length()
    W = fft(weights, N)
    A = fft(a, N)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def solve_lp_direction(a, M, top_k=40):
    """LP: max sum(g) s.t. (a*g)[k] <= M for top-k lags, g >= 0.
    Returns normalized g (sum=1) or None."""
    n = len(a)
    c = fft_convolve_self(a)
    active_lags = np.argsort(c)[::-1][:top_k]

    A_mat = np.zeros((top_k, n))
    for i, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        A_mat[i, idx] = a[k - idx]

    c_obj = -np.ones(n)
    b_ub = np.full(top_k, M)

    try:
        result = linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs')
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def gradient_step(a, S, c, lr_scale=1.0):
    """Take a subgradient step. Returns (new_a, improved)."""
    n = len(a)
    M = c.max()
    # Active set: lags within 0.5% of max, or top-10 if fewer
    threshold = M * 0.995
    active = np.where(c >= threshold)[0]
    if len(active) < 3:
        active = np.argsort(c)[::-1][:10]
    K = len(active)

    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / K

    grad = compute_subgradient_fft(a, w, n)
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return a, False

    # Backtracking line search
    lr = lr_scale * S / (grad_norm * np.sqrt(n))
    for attempt in range(6):
        a_try = a - lr * grad
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            lr *= 0.5
            continue
        a_try = a_try * (S / s_try)
        c_try = fft_convolve_self(a_try)
        if c_try.max() < M:
            return a_try, True
        lr *= 0.5
    return a, False

            print("Polish: M=%.8f obj=%.8f" % (best_M, global_best))

    # Final: run a few more LP steps if time permits
    if time.monotonic() < deadline - 5.0:
        a = global_a.copy()
        a = a * (S / a.sum())
        c = fft_convolve_self(a)
        M = c.max()
        for _ in range(20):
            if time.monotonic() > deadline - 5.0:
                break
            a, improved = lp_step(a, S, c, M)
            if improved:
                c = fft_convolve_self(a)
                M = c.max()
                if M < best_M:
                    best_M = M
                    global_best = 2.0 * n * best_M / (S * S)
                    global_a = a.copy()

    # Verify and return
    final_a = np.clip(global_a, 0.0, 1000.0)
    final_list = final_a.tolist()
    ver = evaluate_sequence(final_list)
    print("FINAL: verified=%.8f (tracked=%.8f)" % (ver, global_best))

    if not np.isfinite(ver):
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
        print("Fallback: obj=%.8f" % ver)

    return final_list
