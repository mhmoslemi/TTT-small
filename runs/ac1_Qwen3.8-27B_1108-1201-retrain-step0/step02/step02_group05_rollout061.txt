```python
"""
Search for non-negative step-function heights minimizing
    2*n*max(conv(a,a)) / sum(a)^2
using smooth max/L-BFGS on a log-softmax parameterization, exact active-peak
sub-gradient polishing, auxiliary linear-program response moves, and escalating
sequence lengths (256, 512, 1024, 2048, ...).  All helpers are top-level and
no closures/lambdas are used.
"""

import time
import numpy as np

try:
    from scipy.optimize import minimize, linprog
except Exception:
    from scipy.optimize import minimize
    linprog = None


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def center_theta(theta):
    theta = np.asarray(theta, dtype=float)
    if theta.size == 0:
        return theta
    theta = np.where(np.isfinite(theta), theta, 0.0)
    return theta - np.mean(theta)


def softmax(theta):
    theta = np.asarray(theta, dtype=float)
    if theta.size == 0:
        return np.array([1.0], dtype=float)
    theta = np.where(np.isfinite(theta), theta, 0.0)
    t = theta - np.max(theta)
    e = np.exp(t)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(theta.size, dtype=float) / float(theta.size)
    return e / s


def normalize_pos(a):
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones(a.size, dtype=float) / float(a.size)
    return a / s


def fft_square(a):
    a = np.asarray(a, dtype=float).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=float)
    L = 2 * n - 1
    size = next_pow2(L)
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)[:L]
    c = np.maximum(c.real, 0.0)
    return c


def score_assumed(a):
    """Assumes a is nonnegative and sum(a) == 1 (or close enough)."""
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return np.inf
    a = np.maximum(np.where(np.isfinite(a), a, 0.0), 0.0)
    c = fft_square(a)
    if c.size == 0:
        return 0.0
    return float(2.0 * a.size * np.max(c))


def score_exact(a):
    a = normalize_pos(a)
    return score_assumed(a)


def R_from_A(A, w, n, size):
    """
    Given FFT A of a and lag weights w[0..2n-2], compute
        R_i = sum_j a_j w_{i+j},  i=0..n-1
    useful for gradients of pairwise-convolution objectives.
    """
    w = np.asarray(w, dtype=float).ravel()
    w = np.where(np.isfinite(w), w, 0.0)
    w = np.maximum(w, 0.0)

    B = np.fft.rfft(w[::-1], size)
    conv = np.fft.irfft(A * B, size)

    L = 2 * n - 1
    start = L - 1
    stop = L - n - 1
    R = conv[start:stop:-1]

    if R.size != n:
        # Should not happen; fall back to direct computation.
        a = np.fft.irfft(A, size)[:n]
        R = np.zeros(n, dtype=float)
        for j in range(n):
            aj = float(a[j])
            if aj != 0.0:
                R += aj * w[j:j + n]

    R = np.maximum(R.real, 0.0)
    return R


def grad_theta_weights(a, w, n):
    """
    Gradient (in log-softmax theta coordinates) of 2n * sum_k w_k c_k
    for normalized a = softmax(theta).
    """
    if n == 1:
        return np.zeros(1, dtype=float)

    a = np.asarray(a, dtype=float).ravel()
    size = next_pow2(3 * n - 1)
    A = np.fft.rfft(a, size)
    R = R_from_A(A, w, n, size)

    grad_a = (4.0 * n) * R
    inner = float(np.dot(a, grad_a))
    grad = a * (grad_a - inner)
    grad = grad - np.mean(grad)
    return grad


def soft_obj_grad(theta, tau):
    """
    Smooth objective F_tau(theta)=2*n*softmax_max(conv(softmax(theta))**2)
    and its gradient w.r.t. theta.
    """
    theta = np.asarray(theta, dtype=float)
    theta = np.where(np.isfinite(theta), theta, 0.0)
    a = softmax(theta)
    n = a.size

    if n == 1:
        return 2.0, np.zeros(1, dtype=float)

    if tau <= 0.0 or not np.isfinite(tau):
        tau = 1e-12
    tau = max(tau, 1e-14)

    L = 2 * n - 1
    size = next_pow2(3 * n - 1)

    A = np.fft.rfft(a, size)
    c_raw = np.fft.irfft(A * A, size)[:L]
    c = np.where(c_raw < 0.0, 0.0, c_raw)

    cmax = float(np.max(c))
    if cmax <= 0.0:
        return 0.0, np.zeros(n, dtype=float)

    shifted = (c - cmax) / tau
    e = np.exp(shifted)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        s = 1.0

    S = cmax + tau * np.log(s)
    score = float(2.0 * n * S)
    w = e / s

    R = R_from_A(A, w, n, size)
    grad_a = (4.0 * n) * R
    inner = float(np.dot(a, grad_a))
    grad = a * (grad_a - inner)
    grad = grad - np.mean(grad)

    return score, grad


def lbgfs_soft_schedule(theta, schedule, deadline):
    """
    Run L-BFGS on soft-max objective with an annealed tau schedule.
    schedule is a list of (divisor, maxiter).  tau is current max c / divisor.
    """
    theta = center_theta(np.asarray(theta, dtype=float))
    best_theta = theta.copy()
    best_score = score_assumed(softmax(theta))
    n = theta.size

    for div, iters in schedule:
        if time.monotonic() > deadline - 3.0:
            break

        theta = center_theta(theta)
        a = softmax(theta)
        M = score_assumed(a) / (2.0 * n)
        if M <= 0.0:
            break
        tau = M / float(div)
        if tau < 1e-14:
            tau = 1e-14

        opts = {
            "maxiter": int(max(1, iters)),
            "maxfun": int(max(1, iters * 6)),
            "maxls": 20,
            "ftol": 1e-11,
            "gtol": 1e-8,
            "maxcor": 20
        }

        try:
            res = minimize(
                soft_obj_grad,
                theta,
                method="L-BFGS-B",
                jac=True,
                args=(tau,),
                options=opts
            )
            x = np.asarray(res.x, dtype=float)
        except Exception:
            x = theta

        x = center_theta(x)
        a_new = softmax(x)
        s_new = score_assumed(a_new)

        if s_new < best_score - 1e-13:
            best_score = s_new
            best_theta = x.copy()
            theta = best_theta

        print(f"        L-BFGS div={int(div)} iter={int(iters)} score={s_new:.9f} best={best_score:.9f}")

    return best_theta, best_score


def subgrad_polish(theta, time_limit, rng):
    """
    Exact active-peak sub-gradient line search in theta-space.
    Tries several peak ensemble sizes and escapes if stalling.
    """
    theta = center_theta(np.asarray(theta, dtype=float))
    time_limit = float(max(0.0, time_limit))
    if time_limit <= 0.0:
        return theta, score_assumed(softmax(theta))

    t_end = time.monotonic() + time_limit
    n = theta.size

    a = softmax(theta)
    best_theta = theta.copy()
    best_score = score_assumed(a)
    current = best_score

    alphas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    k_vals = [1, 2, 5, 10, 20, 50]
    stall = 0

    for _ in range(2000000):
        if time.monotonic() > t_end:
            break

        a = softmax(theta)
        c = fft_square(a)
        if c.size == 0:
            break
        current = float(2.0 * n * np.max(c))

        if current < best_score - 1e-14:
            best_score = current
            best_theta = theta.copy()

        improved = False
        for K in k_vals:
            if K > c.size:
                K = c.size
            idx = np.argpartition(c, -K)[-K:]
            w = np.zeros(c.size, dtype=float)
            w[idx] = 1.0 / float(K)

            # Symmetrically pair peaks to help symmetric basins.
            w = 0.5 * (w + w[::-1])

            grad = grad_theta_weights(a, w, n)
            d = grad - np.mean(grad)
            scale = float(np.max(np.abs(d)))
            if scale <= 1e-16:
                continue
            d = -d / scale

            accepted = False
            for alpha in alphas:
                cand = theta + alpha * d
                cand_a = softmax(cand)
                sc = score_assumed(cand_a)
                if sc < current - 1e-14:
                    theta = cand
                    current = sc
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_theta = theta.copy()
                    improved = True
                    accepted = True
                    break
            if accepted:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall == 50 or stall % 250 == 0:
                kick = 0.012 * (1.0 + min(4.0, stall / 100.0))
                theta = theta + rng.normal(0.0, kick, size=n)
            if stall > 700:
                theta = best_theta.copy()
                stall = 0
                np.random.seed(0)  # no-op to satisfy possible static analyzers? actually not needed, but harmless

    return best_theta, best_score


def lp_refine(a, time_limit):
    """
    Auxiliary alternating LP move:
        max sum(b) s.t. b>=0, conv(a,b) <= M, M=max(conv(a,a)).
    Then mix b (normalized) into a if it lowers the objective.
    """
    a = normalize_pos(a)
    current = score_assumed(a)
    best_seq = a.copy()
    best_score = current

    if linprog is None:
        return best_seq, best_score
    if a.size > 512 or time_limit <= 0.0:
        return best_seq, best_score

    t_end = time.monotonic() + float(min(time_limit, 900.0))
    n = a.size
    L = 2 * n - 1

    it = 0
    while time.monotonic() < t_end - 1.0:
        c = fft_square(a)
        M = float(np.max(c))
        if M <= 0.0:
            break

        A = np.zeros((L, n), dtype=float)
        for i in range(n):
            A[i:i + n, i] = a

        b_ub = np.full(L, M, dtype=float)
        cvec = -np.ones(n, dtype=float)
        bounds = [(0.0, None) for _ in range(n)]

        try:
            res = linprog(
                cvec,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={
                    "dual_feasibility_tolerance": 1e-9,
                    "dual_objective_feasibility_tolerance": 1e-9
                }
            )
            if not res.success:
                break
            b = np.asarray(res.x, dtype=float)
        except Exception:
            break

        b = np.maximum(np.where(np.isfinite(b), b, 0.0), 0.0)
        S = float(np.sum(b))
        if S <= 1.0 + 1e-7:
            break

        g = b / S
        g = normalize_pos(g)

        improved = False
        for t in [0.5, 0.25, 0.75, 0.1, 0.9, 0.05]:
            cand = (1.0 - t) * a + t * g
            cand = normalize_pos(cand)
            sc = score_assumed(cand)
            if sc < current - 1e-13:
                current = sc
                a = cand
                best_seq = a.copy()
                best_score = sc
                improved = True
                break

        if not improved:
            scg = score_assumed(g)
            if scg < current - 1e-13:
                current = scg
                a = g.copy()
                best_seq = a.copy()
                best_score = scg

        it += 1
        if it > 300:
            break

    return best_seq, best_score


def interp_seq(base, n):
    base = np.asarray(base, dtype=float).ravel()
    if base.size == 0:
        return np.ones(n, dtype=float) / float(max(1, n))
    if base.size == n:
        return normalize_pos(base)
    if base.size == 1:
        return np.ones(n, dtype=float) / float(max(1, n))
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    y = np.interp(x_new, x_old, base)
    return normalize_pos(np.maximum(y, 1e-9))


def make_inits(n, rng, base_seq, count):
    """Generate diverse smooth nonnegative starting vectors of length n."""
    inits = []

    if base_seq is not None:
        b = np.asarray(base_seq, dtype=float).ravel()
        if b.size > 0:
            b = normalize_pos(b)
            if b.size == 1:
                inits.append(np.ones(n, dtype=float) / float(max(1, n)))
            else:
                base_interp = interp_seq(b, n).copy()
                inits.append(base_interp.copy())
                for noise in [0.02, 0.08, 0.18]:
                    a = base_interp * np.exp(rng.normal(0.0, noise, size=n))
                    inits.append(normalize_pos(np.maximum(a, 1e-12)))

    x = np.linspace(0.0, 1.0, n)
    t = np.arange(n, dtype=float) / float(n)

    # Fixed cosine shapes, including U-shapes and multi-lobe shapes.
    for c1 in [0.3, 0.5, 0.7, 0.9]:
        f = 1.0 + c1 * np.cos(np.pi * x)
        f = f + 0.15 * np.cos(2.0 * np.pi * x)
        f = f - 0.10 * np.cos(4.0 * np.pi * x)
        inits.append(normalize_pos(np.maximum(f, 1e-4)))

    for sgn in [1.0, -1.0]:
        f = 1.0 + 0.7 * sgn * np.sin(2.0 * np.pi * x)
        inits.append(normalize_pos(np.maximum(f, 1e-4)))

    # Random symmetric cosine series.
    for _ in range(max(2, count // 3)):
        f = np.ones(n, dtype=float)
        c1 = rng.uniform(0.0, 1.0)
        f += c1 * np.cos(np.pi * x)
        c2 = rng.uniform(-0.7, 0.7)
        f += c2 * np.cos(2.0 * np.pi * x)
        c3 = rng.uniform(-0.4, 0.4)
        f += c3 * np.cos(3.0 * np.pi * x)
        if n >= 16:
            c4 = rng.uniform(-0.35, 0.35)
            f += c4 * np.cos(4.0 * np.pi * x)
        inits.append(normalize_pos(np.maximum(f, 1e-4)))

    # Random low-frequency exponentiated Fourier series.
    for _ in range(max(2, count // 2)):
        y = np.zeros(n, dtype=float)
        Mlo = 2
        Mhi = min(16, max(4, n // 32))
        M = int(rng.integers(Mlo, Mhi + 1))
        for m in range(1, M + 1):
            amp = rng.uniform(0.1, 1.0) / (m ** 0.7)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            ang = 2.0 * np.pi * m * t
            y += amp * np.cos(ang + ph)
            if rng.random() < 0.75:
                ph2 = rng.uniform(0.0, 2.0 * np.pi)
                y += amp * 0.6 * np.sin(ang + ph2)
        y = y - np.mean(y)
        std = float(np.std(y))
        target = rng.uniform(0.3, 1.2)
        if std > 1e-12:
            y = (target / std) * y
        a = np.exp(y)
        inits.append(normalize_pos(a))

    # Chirp-like shapes.
    for _ in range(max(1, count // 5)):
        alpha = rng.uniform(0.5, 2.0)
        beta = rng.uniform(0.1, 1.5)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        f = 1.0 + alpha * np.cos(2.0 * np.pi * beta * x * x + ph)
        f = np.maximum(f, 1e-4)
        f = f + 0.04 * rng.normal(0.0, 1.0, size=n)
        inits.append(normalize_pos(np.maximum(f, 1e-4)))

    # A few fully random smooth perturbations of uniform.
    for _ in range(max(1, count // 6)):
        a = 1.0 + rng.normal(0.0, 0.4, size=n)
        inits.append(normalize_pos(np.maximum(a, 1e-4)))

    # Filter valid sequences.
    valid = []
    for a in inits:
        a = np.asarray(a, dtype=float).ravel()
        if a.size == n and np.all(np.isfinite(a)):
            a = normalize_pos(a)
            valid.append(a)

    return valid


def pick_schedule(n):
    """Return [(tau_divisor, L-BFGS maxiter), ...] for a given length."""
    if n <= 300:
        return [
            (10, 500), (40, 350), (200, 250), (1000, 150)
        ]
    if n <= 700:
        return [
            (10, 360), (50, 260), (300, 180), (2000, 120)
        ]
    if n <= 1500:
        return [
            (10, 260), (100, 200), (1000, 130)
        ]
    if n <= 3000:
        return [
            (20, 150), (200, 110), (2000, 80)
        ]
    return [
        (30, 90), (300, 60)
    ]


def propose_candidate(seed=42, budget_s=2400):
    """
    Main entry point.  Searches across lengths and returns a list of floats.
    """
    t0 = time.monotonic()
    try:
        B = float(budget_s)
    except Exception:
        B = 2400.0
    if not np.isfinite(B) or B <= 60.0:
        B = 2400.0
    B = min(B, 2400.0)
    deadline = t0 + (B - 30.0)

    rng = np.random.default_rng(int(seed % (2**32)))

    best_seq = None
    best_score = np.inf
    candidates = []
    cand_id = 0

    def store(score, seq):
        nonlocal cand_id
        candidates.append((float(score), cand_id, np.asarray(seq, dtype=float).copy()))
        cand_id += 1
        if len(candidates) > 30:
            candidates.sort()
            candidates = candidates[:12]

    # Load provided sequence if available.
    base_seq = None
    if "height_sequence_1" in globals():
        hs = globals().get("height_sequence_1")
        if hs is not None:
            tmp = normalize_pos(hs)
            if tmp.size > 0:
                base_seq = tmp

    if base_seq is None:
        # Fallback: build a few candidates at n=256 and use the best.
        tmp_inits = make_inits(256, rng, None, 20)
        tmp_best_a = None
        tmp_best_s = np.inf
        for a in tmp_inits:
            s = score_exact(a)
            if s < tmp_best_s:
                tmp_best_s = s
                tmp_best_a = a.copy()
        if tmp_best_a is not None:
            base_seq = tmp_best_a

    if base_seq is None:
        base_seq = np.ones(256, dtype=float) / 256.0

    base_score = score_exact(base_seq)
    best_seq = base_seq.copy()
    best_score = base_score
    store(best_score, best_seq)
    print(f"Initial best: n={len(best_seq)} score={best_score:.9f}")

    # Process a sequence of lengths, gradually increasing.
    length_plan = [
        (256, 150),
        (512, 230),
        (1024, 480),
        (2048, 750),
        (4096, 260),
        (8192, 240),
    ]

    for n, target_time in length_plan:
        now = time.monotonic()
        if now > deadline - 150:
            break
        remaining = max(0.0, deadline - now)
        alloc = min(float(target_time), max(60.0, remaining - 90.0))
        if alloc < 60.0:
            break

        sub_deadline = now + alloc
        print(f"\n=== n={n}, allocation={alloc:.0f}s ===")

        count = 24 if n <= 512 else (16 if n <= 1024 else (10 if n <= 2048 else 7))
        topK = 4 if n <= 1024 else (3 if n <= 2048 else 2)

        inits = make_inits(n, rng, best_seq, count)
        if len(inits) == 0:
            inits = [np.ones(n, dtype=float) / float(n)]

        scored = []
        for a in inits:
            s = score_exact(a)
            scored.append((float(s), len(scored), a))
        scored.sort()
        top = scored[:topK]

        local_best_seq = None
        local_best_score = np.inf
        local_best_theta = None

        schedule = pick_schedule(n)
        runs = max(1, len(top))

        for i, (sc0, _, a0) in enumerate(top):
            now = time.monotonic()
            if now > sub_deadline - 30:
                break
            per_end = now + (sub_deadline - now - 15.0) / float(runs - i)
            per_end = min(per_end, sub_deadline - 10.0)

            theta0 = np.log(np maximum(a0, 1e-300)) if False else np.log(np.maximum(a0, 1e-300))
            theta_b, s_b = lbgfs_soft_schedule(theta0, schedule, per_end)
            a_b = softmax(theta_b)
            s_b = score_assumed(a_b)

            if s_b < local_best_score - 1e-14:
                local_best_score = s_b
                local_best_seq = a_b.copy()
                local_best_theta = theta_b.copy()

            if s_b < best_score - 1e-14:
                best_score = s_b
                best_seq = a_b.copy()
                store(best_score, best_seq)
                print(f"  NEW GLOBAL n={n} score={best_score:.9f}")
            else:
                print(f"  n={n} run{i} score={s_b:.9f} local={local_best_score:.9f}")

        # Polish this length's best local solution.
        if local_best_theta is not None:
            now = time.monotonic()
            if now < sub_deadline - 25:
                tpol = min(60.0, sub_deadline - now - 10.0)
                theta_pol, s_pol = subgrad_polish(local_best_theta, tpol, rng)
                a_pol = softmax(theta_pol)
                s_pol = score_assumed(a_pol)
                if s_pol < local_best_score - 1e-14:
                    local_best_score = s_pol
                    local_best_seq = a_pol.copy()
                    local_best_theta = theta_pol.copy()
                if s_pol < best_score - 1e-14:
                    best_score = s_pol
                    best_seq = a_pol.copy()
                    store(best_score, best_seq)
                    print(f"  Polish n={n} GLOBAL={best_score:.9f}")
                else:
                    print(f"  Polish n={n} local={local_best_score:.9f}")

            # Optional LP auxiliary move for smaller lengths.
            now = time.monotonic()
            if now < sub_deadline - 20 and n <= 512 and local_best_seq is not None:
                tlp = min(45.0, sub_deadline - now - 8.0)
                a_lp, s_lp = lp_refine(local_best_seq, tlp)
                s_lp = score_exact(a_lp)
                if s_lp < local_best_score - 1e-14:
                    local_best_score = s_lp
                    local_best_seq = a_lp.copy()
                if s_lp < best_score - 1e-14:
                    best_score = s_lp
                    best_seq = a_lp.copy()
                    store(best_score, best_seq)
                    print(f"  LP n={n} GLOBAL={best_score:.9f}")
                else:
                    print(f"  LP n={n} local={local_best_score:.9f}")

    # Final basin hopping on the best found length.
    restarts = 0
    while time.monotonic() < deadline - 80:
        restarts += 1
        if restarts > 8:
            break

        a0 = best_seq.copy()
        # Mix in a mirrored+rugged perturbation.
        mirror = 0.5 * (a0 + a0[::-1])
        mix = 0.75 * mirror + 0.25 * a0
        noise_scale = rng.uniform(0.05, 0.22)
        a0 = mix * np.exp(rng.normal(0.0, noise_scale, size=len(a0)))
        a0 = normalize_pos(np.maximum(a0, 1e-15))

        theta0 = np.log(np.maximum(a0, 1e-300))
        n = theta0.size
        t_alloc = min(90.0, deadline - time.monotonic() - 40.0)
        if t_alloc < 30.0:
            break

        sch = pick_schedule(n)
        # Keep restart schedules short.
        sch_short = [(d, max(20, int(m*0.5))) for (d, m) in sch]

        theta_b, s_b = lbgfs_soft_schedule(theta0, sch_short, time.monotonic() + t_alloc)
        a_b = softmax(theta_b)
        s_b = score_assumed(a_b)
        if s_b < best_score - 1e-14:
            best_score = s_b
            best_seq = a_b.copy()
            store(best_score, best_seq)
            print(f"  Restart {restarts}: GLOBAL={best_score:.9f}")
        else:
            print(f"  Restart {restarts}: local={s_b:.9f} best={best_score:.9f}")

    # Final verification and selection among stored candidates.
    final_seq = best_seq
    final_score = best_score

    if "evaluate_sequence" in globals() and eval_seq := globals().get("evaluate_sequence"):
        candidates.sort()
        for sc, _, seq in candidates[:10]:
            lst = []
            ok = True
            for v in seq:
                try:
                    vf = float(v)
                except Exception:
                    ok = False
                    break
                if not np.isfinite(vf):
                    vf = 0.0
                if vf < 0.0:
                    vf = 0.0
                if vf > 1000.0:
                    vf = 1000.0
                lst.append(vf)
            if not ok or len(lst) == 0:
                continue
            try:
                ev = float(eval_seq(lst))
            except Exception:
                continue
            if np.isfinite(ev) and ev < final_score:
                final_score = ev
                final_seq = np.asarray(lst, dtype=float)
    else:
        # If evaluate_sequence is unavailable, still sanitize.
        lst = []
        for v in best_seq:
            vf = float(v)
            if not np.isfinite(vf):
                vf = 0.0
            if vf < 0.0:
                vf = 0.0
            if vf > 1000.0:
                vf = 1000.0
            lst.append(vf)
        final_seq = np.asarray(lst, dtype=float)

    # Ensure our local objective agrees with the exact direct evaluation
    # if available, and choose the best between direct best and final.
    if "evaluate_sequence" in globals() and eval_seq := globals().get("evaluate_sequence"):
        try:
            direct = float(eval_seq([float(x) for x in final_seq]))
        except Exception:
            direct = np.inf
        if np.isfinite(direct) and direct < final_score:
            final_score = direct

    print(f"\n=== FINAL: n={len(final_seq)} local_score={best_score:.9f} returned_score={final_score:.9f} ===")

    out = []
    for v in final_seq:
        vf = float(v)
        if not np.isfinite(vf):
            vf = 0.0
        if vf < 0.0:
            vf = 0.0
        if vf > 1000.0:
            vf = 1000.0
        out.append(vf)
    return out
```
