def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)

"""
LP-based optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: uses linear programming to find optimal descent directions
at each iteration, combined with adaptive line search, peak-specific local
search, and subgradient polishing. Targets n values from 128 to 2048.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_conv_matrix(a):
    """Build convolution matrix A where (A @ b)[k] = (a*b)(k) for k=0..2n-2."""
    n = len(a)
    k = np.arange(2 * n - 1)[:, None].astype(np.int64)
    l = np.arange(n)[None, :].astype(np.int64)
    idx = k - l
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = a[idx[mask]]
    return A


def solve_lp_step(a, M, relax=1.0):
    """Solve LP: max sum(b) s.t. (a*b)(k) <= M*relax, b >= 0. Returns b or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = (M * relax) * np.ones(2 * n - 1)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(a0, time_limit, relax=1.0, t_init=0.5, verbose=False):
    """LP-based iterative optimization with adaptive step size."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    stagnation = 0
    max_stag = 30
    t0 = time.monotonic()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        if M < 1e-15:
            break

        b = solve_lp_step(a, M, relax=relax)
        if b is None or b.sum() < 1e-12:
            stagnation += 1
            if stagnation > 50:
                break
            continue

        b = b / b.sum()

        # Line search: try several step sizes
        trial_ts = [t, min(t * 1.5, 1.0), t * 0.5, t * 0.25, 0.9, 0.7, 0.4, 0.15, 0.05]
        trial_ts = [tt for tt in trial_ts if 0.01 <= tt <= 1.0]
        best_t = t
        best_trial_obj = best_obj
        for trial_t in trial_ts:
            a_trial = (1.0 - trial_t) * a + trial_t * b
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial /= st
            obj_trial = 2.0 * n * fast_convolve(a_trial).max()
            if obj_trial < best_trial_obj:
                best_trial_obj = obj_trial
                best_t = trial_t

        a = (1.0 - best_t) * a + best_t * b
        a = np.maximum(a, 0.0)
        a /= a.sum()
        cur_obj = 2.0 * n * fast_convolve(a).max()
        iter_count += 1

def compute_subgradient_smooth(conv, a, temp):
    """Smoothed subgradient using softmax weighting over peak positions."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = temp * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    # Only process top weighted peaks for efficiency
    threshold = w.max() * 0.05
    active = np.where(w > threshold)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    return sg


def subgradient_polish(a0, time_limit, temp_init=50.0, eta_init=0.3):

    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()

        a = np.ones(n) / n
    else:

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta_init
    patience = 0
    max_pat = 40
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient_smooth(conv, a, temp)
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            eta *= 0.6
            temp *= 1.4
            if eta < 0.003:
                elapsed = time.monotonic() - t0
                frac = min(1.0, elapsed / max(time_limit, 1.0))
                eta = eta_init * (1.0 - 0.4 * frac)
                temp = temp_init * (1.0 + 4.0 * frac)

    """Targeted local search: reduce the largest peaks via coordinate moves."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.01 * (1.0 - frac) ** 2 + 0.0002
        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Find peak positions (top few)
        peak_thresh = max_c * 0.9
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])

        # Pick a peak to attack
        k = int(peak_positions[rng.integers(len(peak_positions))])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue

        # Source: proportional to contribution to this peak
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))

        # Target: position that contributes least to any peak
        # Precompute peak positions (top 10)
        top_peaks = np.argsort(conv)[::-1][:10]
        best_target = int(rng.integers(n))
        best_contrib = float('inf')
        for _ in range(8):
            cand = int(rng.integers(n))
            contrib = 0.0
            for pk in top_peaks:
                d = int(pk) - cand
                if 0 <= d <= n - 1:
                    contrib += a[d]
                d2 = cand - int(pk)
                if 0 <= d2 <= n - 1:
                    contrib += a[d2]
            if contrib < best_contrib:
                best_contrib = contrib
                best_target = cand

        target_j = best_target
        if target_j == top_j:
            no_improve += 1
            continue

        delta = a[top_j] * rng.uniform(0.02, 0.08)
        if delta < 1e-9:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            no_improve += 1
            continue

        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
                no_improve += 1
            else:
                a[top_j] += delta
                a[target_j] -= delta
                no_improve += 1

    return best_a, best_obj


def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % str(sf)] = g

    for freq in [1, 2, 3, 5]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        s /= s.sum()
        inits['sin_%d' % freq] = s

    # Bimodal
    for w_frac in [0.15, 0.25, 0.35]:
        bi = np.zeros(n)
        w = max(1, int(n * w_frac))
        bi[:w] = 1.0
        bi[-w:] = 1.0
        bi /= bi.sum()
        inits['bimodal_%s' % str(w_frac)] = bi

    # Power law
    for alpha in [0.5, 2.0, 5.0]:
        p = np.power(np.arange(1, n + 1, dtype=np.float64), -alpha)
        p /= p.sum()
        inits['pow_%s' % str(alpha)] = p

    # Random
    for alpha in [0.2, 1.0, 5.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['rand_%s' % str(alpha)] = r

    # Chirp
    chirp = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * x / (2.0 * n)))
    chirp = np.maximum(chirp, 0.0)
    if chirp.sum() > 1e-10:
        chirp /= chirp.sum()
        inits['chirp'] = chirp

    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0).copy()
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size),
                           np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp

    return inits

        phase3_budget = min(400.0, deadline - time.monotonic() - 80)
        n_anneal = 5
        anneal_per = phase3_budget / n_anneal

        for i in range(n_anneal):
            if time.monotonic() > deadline - 80:
                break
            a_start = best_sequence.copy()
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
            rng_a = np.random.default_rng(seed + 7000 + i * 137)
            a_opt, _ = peak_local_search(a_start, time_limit=anneal_per, rng=rng_a)
            record(a_opt)

            # Follow up with LP if improved
            if time.monotonic() < deadline - 80:
                lp_rem = min(anneal_per * 0.5, deadline - time.monotonic() - 80)
                if lp_rem > 3:
                    a_opt2, _ = lp_optimize(a_opt, time_limit=lp_rem, relax=1.0, t_init=0.2)
                    record(a_opt2)

        print("  Phase 3 done. Best: %.6f" % best_score)

    # === Phase 4: Final polish ===
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_best = best_sequence.copy()
            a_best = np.maximum(a_best, 0.0)
            a_best /= a_best.sum()

            # LP with very small step
            rem1 = remaining * 0.4
            a_opt, _ = lp_optimize(a_best, time_limit=rem1, relax=1.0, t_init=0.05)
            record(a_opt)

            # Subgradient
            if time.monotonic() < deadline - 5:
                rem2 = deadline - time.monotonic() - 5
                a_opt2, _ = subgradient_polish(a_best, time_limit=rem2 * 0.6,
                                               temp_init=200.0, eta_init=0.05)
                record(a_opt2)

            # One more peak search
            if time.monotonic() < deadline - 5:
                rem3 = deadline - time.monotonic() - 5
                if rem3 > 5:
                    rng_f = np.random.default_rng(seed + 12345)
                    a_opt3, _ = peak_local_search(a_best, time_limit=rem3, rng=rng_f)
                    record(a_opt3)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
