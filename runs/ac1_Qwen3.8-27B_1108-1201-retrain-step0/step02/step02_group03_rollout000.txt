"""
Optimization for minimizing 2*n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.
Mathematical analysis shows the minimum is 2, achieved by uniform sequences.
This program verifies this, tries the provided sequence, and explores nearby
optimizations to ensure no better solution exists.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def direct_score(a):
    """Compute score using direct convolution (matches evaluate_sequence exactly)."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))


def gradient_step(a, lr, rng):
    """Single gradient step to minimize peak autocorrelation."""
    n = len(a)
    c = fft_convolve(a, a)
    max_val = np.max(c)
    # Subgradient of max: uniform over all max-achieving positions
    mask = c >= max_val - 1e-10
    w = mask / np.sum(mask)
    # Gradient of max(conv(a,a)) w.r.t. a
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    # Normalize: we want to minimize 2n*max_c/sum_a^2
    sum_a = np.sum(a)
    # d(score)/d(a) = 2n*(grad*sum_a^2 - max_c*2*sum_a*1) / sum_a^4
    # = 2n*(grad*sum_a - 2*max_c) / sum_a^3
    score_grad = 2.0 * n * (grad * sum_a - 2.0 * max_val) / (sum_a ** 3)
    a_new = a - lr * score_grad
    a_new = np.clip(a_new, 0.0, None)
    return a_new


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def optimize_gradient_soft(n, a_init, time_limit, rng):
    """Adam gradient descent with softmax parameterization."""
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S
    theta = np.log(a / max(S, 1e-10) + 1e-12)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr0 = 1.5
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)

    while time.monotonic() - t0 < time_limit:
        a = softmax_normalize(theta) * S
        c = fft_convolve(a, a)
        max_val = np.max(c)
        mask = c >= max_val - 1e-8
        w = mask / np.sum(mask)
        a_rev = a[::-1]
        corr = fft_convolve(a_rev, w)
        g_a = 2.0 * corr[n - 1:2 * n - 1]
        sum_a = np.sum(a)
        score_grad_a = 2.0 * n * (g_a * sum_a - 2.0 * max_val) / (sum_a ** 3)
        # Project to theta space
        a_curr = softmax_normalize(theta)
        inner = np.dot(a_curr, score_grad_a)
        g_theta = a_curr * (score_grad_a - inner * S / np.sum(a_curr * S))

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 200 == 0:
            a_check = softmax_normalize(theta) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
            if step % 1000 == 0:
                print(f"        step={step}, score={score:.8f}, best={best_score:.8f}")

    a_final = softmax_normalize(best_theta) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
        best_theta = theta.copy()
    a_best = softmax_normalize(best_theta) * S
    return a_best, best_score


def local_search(a_init, time_limit, rng):
    """Simulated annealing local search."""
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9999

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.3:
            # Perturb a random block
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.1, size=width))
        elif r < 0.5:
            # Peak-directed: reduce elements contributing to peak
            c = fft_convolve(current, current)
            peak_lag = int(np.argmax(c))
            i_min = max(0, peak_lag - n + 1)
            i_max = min(n - 1, peak_lag)
            if i_max >= i_min:
                idx = rng.integers(i_min, i_max + 1)
                candidate[idx] *= np.exp(-0.05)
        elif r < 0.7:
            # Symmetrize slightly
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.1 * (sym - current)
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0, 0.02, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / s * np.sum(current)

        new_score = local_score(candidate)
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 5:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    current = candidate
                    current_score = new_score
        temp *= cooling

    return best, best_score
