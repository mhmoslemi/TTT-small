LP-based iterative peak reduction for non-negative step functions.
At each step, solves a linear program to find the optimal perturbation
direction that minimizes the peak of the convolution (linearized).
Uses backtracking line search for finite steps.

import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    """Compute the evaluation score."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def lp_step(a, max_time_per_lp=None):
    """
    Solve one LP step to find the optimal perturbation direction.
    
    Parameters:
    a: current sequence (length n), non-negative, sum = 1 (normalized)
    
    Returns:
    da: optimal perturbation direction (length n)
    t_opt: predicted new max (linearized)
    """
    n = len(a)
    c = fft_convolve(a, a)
    
    # Variable vector: x = [da_0, da_1, ..., da_{n-1}, t]
    # Length: n + 1
    N = n + 1
    
    # Objective: minimize t
    # c_obj = [0, 0, ..., 0, 1]
    c_obj = np.zeros(N)
    c_obj[-1] = 1.0

    # Constraint 1: For each k = 0, ..., 2n-2:
    #   2 * (a*da)[k] - t <= -c[k]
    # (a*da)[k] = sum_{i} a[i] * da[k-i] (with da[j] = 0 outside [0,n-1])
    
    num_c1 = 2 * n - 1
    A_rb = np.zeros((num_c1, n))  # will be A_ub without the t column
    b_rb = np.zeros(num_c1)
    
    for k in range(2 * n - 1):
        # (a*da)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * da[k-i]
        # = sum_{j: 0<=j<n, 0<=k-j<n} a[k-j] * da[j]
        # Coefficient of da[j] is a[k-j] if 0 <= k-j <= n-1, else 0
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A_rb[k, j] = 2.0 * a[k - j]
        b_rb[k] = -c[k]

    num_c2 = n
    A_nb = np.zeros((num_c2, n))
    b_nb = np.zeros(num_c2)
    for i in range(n):
        A_nb[i, i] = -1.0
        b_nb[i] = a[i]
    
    # Combine inequality constraints
    A_ub = np.hstack([np.vstack([A_rb, A_nb]), 
                       np.column_stack([-np.ones(num_c1), np.zeros(num_c2)])])
    b_ub = np.concatenate([b_rb, b_nb])
    
    # Equality constraint: sum(da) = 0
    A_eq = np.zeros((1, N))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = [(-a[i], None) for i in range(n)] + [(0, None)]

LP-based iterative peak reduction for minimizing 2*n*max(conv(a,a))/sum(a)^2
over non-negative sequences. Uses a linear program at each step to find the
optimal perturbation direction that minimizes the (linearized) convolution peak,
combined with backtracking line search.

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT, zero-padded to next power of 2."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

    Build the LP for one optimization step.
    
    a: current sequence (length n, sum=1)
    c: current convolution (length 2n-1)
    
    Returns: c_obj, A_ub, b_ub, A_eq, b_eq, bounds

        for j in range(j_min, j_max + 1):
            A1[k, j] = 2.0 * a[k - j]
        b1[k] = -c[k]
    
    # Part 2: Non-negativity constraints (n of them)
    # a[i] + da[i] >= 0 => -da[i] <= a[i]
    A2 = -np.eye(n)
    b2 = a.copy()
    
    # Combine
    A_ub = np.vstack([
        np.hstack([A1, -np.ones((num_c1, 1))]),
        np.hstack([A2, np.zeros((n, 1))])
    ])
    b_ub = np.concatenate([b1, b2])
    
    # Equality: sum(da) = 0
    A_eq = np.zeros((1, N))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-a[i], None) for i in range(n)] + [(None, None)]
    
    return c_obj, A_ub, b_ub, A_eq, b_eq, bounds


def solve_lp_step(a):
    """Solve one LP step. Returns da (perturbation direction) or None."""
    n = len(a)
    c = fft_convolve(a, a)
    
    c_obj, A_ub, b_ub, A_eq, b_eq, bounds = build_lp(a, c)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    
    da = result.x[:n]
    t_new = result.x[n]
    
    return da, t_new


def lp_optimize(a_init, time_limit, verbose=True, eps=1e-12):
    """
    Main LP-based optimization loop.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, eps, None)
    a = a / np.sum(a)
    
    current_score = compute_score(a)
    best_score = current_score
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        da, t_new = solve_lp_step(a)
        
        if da is None:
            # LP failed, try a random perturbation
            da = rng.standard_normal(n)
            da -= da.mean()
        
        # Line search
        best_alpha = 0.0
        best_trial_score = current_score
        
        for alpha in [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]:
            a_trial = a + alpha * da
            a_trial = np.clip(a_trial, 0.0, None)
            s = np.sum(a_trial)
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            trial_score = compute_score(a_trial)
            if trial_score < best_trial_score:
                best_trial_score = trial_score
                best_alpha = alpha
        
        if best_alpha == 0.0:
            no_improve += 1
            if no_improve > 200:
                if verbose:
                    print(f"    Stalled after {it} iterations")
                break
            # Random kick
            da = rng.standard_normal(n) * 0.01
            da -= da.mean()
            best_alpha = 1.0
            best_trial_score = None
        else:
            no_improve = 0
        
        # Apply step
        a = a + best_alpha * da
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = best_a.copy()
            current_score = best_score
            continue
        a = a / s
        
        current_score = compute_score(a)
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            if verbose and it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    it={it}, score={current_score:.8f}, elapsed={elapsed:.1f}s")
        
        it += 1
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    DONE: {it} iterations, best_score={best_score:.8f}, time={elapsed:.1f}s")
    
    return best_a, best_score


def resize_sequence(a, new_n):
    """Resize a sequence to new length via interpolation."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.clip(a_new, 0.0, None)
    s = np.sum(a_new)
    if s < 1e-10:
        a_new = np.ones(new_n) / new_n
    else:
        a_new = a_new / s
    return a_new


def proposal_local_search(a_init, time_limit, rng, top_k=5):
    """Simple local search with peak-directed mutations."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    current_score = compute_score(a)
    best_a = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.4:
            # Peak-directed: reduce elements contributing to the peak
            c = fft_convolve(a, a)
            peak_lag = np.argmax(c)
            # Elements contributing to peak
            contrib = np.zeros(n)
            j_min = max(0, peak_lag - n + 1)
            j_max = min(n - 1, peak_lag)
            for j in range(j_min, j_max + 1):
                contrib[j] = a[j] * a[peak_lag - j]
            # Reduce top contributors, boost bottom
            n_mod = max(2, n // 8)
            top_idx = np.argsort(contrib)[-n_mod:]
            bot_idx = np.argsort(contrib)[:n_mod]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, n_mod))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.02, n_mod))
        elif r < 0.6:
            # Block perturbation
            width = max(1, n // 32)
            start = rng.integers(0, n - width)
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.1, width))
        elif r < 0.8:
            # Symmetrization
            sym = 0.5 * (a + a[::-1])
            candidate = 0.8 * a + 0.2 * sym
            candidate += 0.005 * rng.standard_normal(n)
        else:
            # Global noise
            candidate = a * np.exp(rng.normal(0, 0.02, n))
        
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s
        
        new_score = compute_score(candidate)
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = candidate.copy()
        else:
            delta = new_score - current_score
            accept_prob = np.exp(-delta / max(temp, 1e-12))
            if rng.random() < accept_prob:
                a = candidate
                current_score = new_score
        
        temp *= cooling
    
    return best_a, best_score
