        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = 0.3
    K = K_init
    patience = 0
    t0 = time.monotonic()
    n_iter = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = elapsed / max(time_limit, 1.0)
        
        # Adapt K: start small, grow over time
        K_cur = min(K_max, max(K_init, int(K_init + (K_max - K_init) * frac)))
        
        # Get LP direction
        b_star = lp_direction(a, K_cur)
        
        # Line search: try several step sizes
        improved = False
        for t_try in [t, 0.5 * t, 0.7 * t, 0.3 * t, 0.1 * t]:
            a_new = (1.0 - t_try) * a + t_try * b_star
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-12:
                a_new = np.ones(n) / n
            else:
                a_new /= s_new
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            
            if obj_new < best_obj * 1.0005:
                a = a_new
                conv = conv_new
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a.copy()
                    patience = 0
                else:
                    patience += 1
                improved = True
                break
        else:
            patience += 1
        
        if not improved:
            t *= 0.6
            if t < 0.01:
                # Perturb and reset
                a = a + 0.02 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                conv = fast_convolve(a)
                t = 0.2
                patience = 0
        
        n_iter += 1
    
    return best_a, best_obj, n_iter


def subgradient_optimize(a0, time_limit, seed=42, t_init=50.0, eta_init=0.35):
    """Mirror descent with adaptive parameters."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient(conv, a, t)
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= 40:
            patience = 0
            eta *= 0.6
            t *= 1.4
            if eta < 0.005:
                # Reset with perturbation
                a = best_a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                conv = fast_convolve(a)
                eta = eta_init * 0.5
                t = t_init * 1.5
    
    return best_a, best_obj


def pgd_optimize(a0, time_limit, seed=42, t_init=30.0, lr_init=0.08):
    """Projected gradient descent."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    lr = lr_init
    patience = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient(conv, a, t)
        a_new = a - lr * sg
        a_new = project_simplex(a_new)
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= 35:
            patience = 0
            lr *= 0.65
            t *= 1.3
            if lr < 0.002:
                a = best_a + 0.01 * rng.standard_normal(n)
                a = project_simplex(a)
                conv = fast_convolve(a)
                lr = lr_init * 0.4
                t = t_init * 2.0
    
    return best_a, best_obj


def anneal_informed(a0, time_limit, seed=42):
    """Informed simulated annealing: transfer mass from peak contributors to low-impact positions."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.02 * (1.0 - frac) ** 2 + 0.0003
        
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # Find peak positions
        peak_thresh = max_c * 0.8
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        
        # Pick a peak
        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        
        # Find the source position (max contributor to this peak)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        
        # Choose target: position that minimally affects peaks
        target_j = -1
        for _ in range(8):
            candidate = int(rng.integers(n))
            if candidate == top_j:
                continue
            # Estimate impact: sum of a[d] for d where candidate contributes to peaks
            impact = 0.0
            for pk in peak_positions[:10]:
                d = pk - candidate
                if 0 <= d < n:
                    impact += a[d]
                d2 = candidate - pk + 2 * n - 2
                if 0 <= d2 < 2 * n - 1:
                    pass  # linear conv, so d2 is relative
                # For linear conv: candidate contributes to (a*a)[k] if a[k-candidate] is valid
                d3 = pk - candidate
                if 0 <= d3 < n:
                    impact += a[d3]
            if impact < a.mean() * 3.0:
                target_j = candidate
                break
        
        if target_j < 0:
            target_j = int(rng.integers(n))
            if target_j == top_j:
                continue
        
        # Transfer
        delta = a[top_j] * rng.uniform(0.02, 0.08)
        if delta < 1e-10:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
    
    return best_a, best_obj


def spectral_init(n, seed=42):
    """Initialize using flat-spectrum design in frequency domain."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Random phases for frequency bins
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    # Amplitudes: try to make |A(k)|^2 flat
    amplitudes = np.ones(fft_size // 2 + 1)
    amplitudes[0] = np.sqrt(n)  # A(0) = sum(a) = 1, so |A(0)| = 1... 
    # Actually A(0) = sum(a) = 1, so |A(0)|^2 = 1
    # For flat spectrum: |A(k)|^2 = 1/(2n-1) for k != 0
    # But sum |A(k)|^2 = n * sum(a[j]^2) >= n * (sum a)^2 / n = 1 (by Cauchy)
    # So we need sum |A(k)|^2 >= n
    # If flat: (2n-1) * 1/(2n-1) = 1... that's < n for n > 1
    # So perfectly flat is impossible. Let's just use random phases with |A(0)|=1.
    
    A = np.zeros(fft_size // 2 + 1, dtype=complex)
    A[0] = 1.0  # A(0) = sum(a) = 1
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.sqrt(1.0 / n) * np.exp(1j * phases[k])
    
    # Make it conjugate symmetric for real signal
    A_full = np.zeros(fft_size, dtype=complex)
    A_full[:fft_size // 2 + 1] = A
    for k in range(1, fft_size // 2 + 1):
        A_full[fft_size - k] = np.conj(A[k])
    
    a = np.fft.irfft(A_full, fft_size)
    a = a[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a /= s
    return a


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    a_new /= s
    return a_new


def propose_candidate(seed=42, budget_s=2400):
    """
    Main optimization: LP-driven direction finding + subgradient + annealing.
    Multi-length exploration with adaptive time allocation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng_main = np.random.default_rng(seed)
    
    parent = get_parent_sequence()
    
    best_sequence = None
    best_score = float('inf')
    
    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a_arr = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a_arr.sum()
        if s < 1e-10:
            return
        a_norm = a_arr / s
        score = safe_eval(a_norm)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_arr.copy()  # keep unnormalized for evaluate_sequence
            print("  New best: score=%.6f, n=%d" % (score, len(a_arr)))
    
    # === Phase 1: LP optimization from parent (n=256) ===
    print("Phase 1: LP optimization from parent...")
    if parent is not None and time.monotonic() < deadline - 1500:
        n_parent = len(parent)
        a_init = np.maximum(parent, 0.0)
        a_init = a_init / a_init.sum()
        
        t_alloc = min(350.0, deadline - time.monotonic() - 1500)
        if t_alloc > 30:
            a_opt, obj, iters = lp_optimize(a_init, time_limit=t_alloc, seed=seed,
                                            K_init=40, K_max=120)
            record(a_opt)
            print("  LP done: obj=%.6f, iters=%d, time=%.0fs" % (obj, iters, time.monotonic() - t_start))
    
    # === Phase 2: Subgradient refinement ===
    if best_sequence is not None and time.monotonic() < deadline - 1200:
        print("Phase 2: Subgradient refinement...")
        a_cur = best_sequence / best_sequence.sum()
        t_alloc = min(200.0, deadline - time.monotonic() - 1200)
        if t_alloc > 20:
            a_opt, obj = subgradient_optimize(a_cur, time_limit=t_alloc, seed=seed + 100)
            record(a_opt)
            
            if time.monotonic() < deadline - 1200:
                t_alloc2 = min(100.0, deadline - time.monotonic() - 1200)
                if t_alloc2 > 15:
                    a_opt2, obj2 = pgd_optimize(a_opt, time_limit=t_alloc2, seed=seed + 200)
                    record(a_opt2)
    
    # === Phase 3: Try longer sequences (n=512) ===
    if time.monotonic() < deadline - 800:
        print("Phase 3: Longer sequences...")
        
        if parent is not None:
            a_512 = interpolate_sequence(parent, 512)
        else:
            a_512 = spectral_init(512, seed)
        
        t_alloc = min(300.0, deadline - time.monotonic() - 800)
        if t_alloc > 30:
            a_opt, obj, iters = lp_optimize(a_512, time_limit=t_alloc, seed=seed + 300,
                                            K_init=50, K_max=150)
            record(a_opt)
            
            if time.monotonic() < deadline - 800:
                t_alloc2 = min(100.0, deadline - time.monotonic() - 800)
                if t_alloc2 > 15:
                    a_opt2, obj2 = subgradient_optimize(a_opt, time_limit=t_alloc2, seed=seed + 400)
                    record(a_opt2)
    
    # === Phase 4: LP optimization with different initializations ===
    if time.monotonic() < deadline - 500:
        print("Phase 4: Alternative initializations...")
        n_try = len(best_sequence) if best_sequence is not None else 256
        n_try = min(n_try, 512)  # Keep manageable
        
        inits = {}
        x = np.arange(n_try, dtype=np.float64)
        
        # Gaussian
        g = np.exp(-0.5 * ((x - n_try * 0.5) / (0.2 * n_try)) ** 2)
        g /= g.sum()
        inits['gauss'] = g
        
        # Bimodal (mass at edges)
        bi = np.zeros(n_try)
        w = max(1, n_try // 5)
        bi[:w] = 1.0
        bi[-w:] = 1.0
        bi /= bi.sum()
        inits['bimodal'] = bi
        
        # Spectral
        inits['spectral'] = spectral_init(n_try, seed=seed + 500)
        
        # Random
        r = rng_main.dirichlet(np.ones(n_try) * 0.5)
        inits['random'] = r
        
        # Interpolated parent
        if parent is not None:
            inits['parent_interp'] = interpolate_sequence(parent, n_try)
        
        t_total = min(300.0, deadline - time.monotonic() - 500)
        t_per = t_total / max(1, len(inits))
        
        for name, a0 in inits.items():
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 500)
            if t_alloc < 15:

            a_opt, obj, iters = lp_optimize(a0, time_limit=t_alloc, seed=seed + 600,

            record(a_opt)

    # === Phase 5: Annealing polish ===
    if best_sequence is not None and time.monotonic() < deadline - 50:

        for i in range(n_anneal):

            if t_alloc < 10:
                break
            rng_a = np.random.default_rng(seed + 1000 + i * 100)
            a_opt, obj = anneal_informed(a_best, time_limit=t_alloc, seed=seed + 1000 + i)
            record(a_opt)
            a_best = best_sequence / best_sequence.sum()  # restart from best
    
    # === Phase 6: Final LP polish on best ===
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("Phase 6: Final LP polish...")
        a_best = best_sequence / best_sequence.sum()
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_opt, obj, iters = lp_optimize(a_best, time_limit=remaining, seed=seed + 2000,
                                            K_init=80, K_max=200)
            record(a_opt)
    
    # Final output
    if best_sequence is None:
        # Fallback
        best_sequence = np.ones(128)
        best_score = safe_eval(best_sequence)
    
    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))
    print("Total time: %.1fs" % (time.monotonic() - t_start))
    
    return [float(x) for x in best_sequence]
