def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_normalized(a):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_gradient(conv, a, temperature):
    """
    Compute the smoothed gradient of max(conv) using softmax weighting.
    Returns gradient of T*log(sum(exp(conv/T))) which approximates max(conv) as T->0.
    """
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    # Softmax weights
    scaled = conv / max(temperature, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        idx = np.argmax(conv)
        w = np.zeros_like(conv)
        w[idx] = 1.0
    else:
        w /= w_sum
    
    # Gradient: for each i, grad[i] = sum_k w[k] * a[k-i] (where valid)
    grad = np.zeros(n)
    for k in range(len(conv)):

            continue

        i_hi = min(n - 1, k)

        # Let j = k-i, so j ranges from k-i_hi to k-i_lo
        j_lo = k - i_hi
        j_hi = k - i_lo
        grad[i_lo:i_hi+1] += w[k] * a[j_lo:j_hi+1][::-1]
    
    return grad

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_gradient(conv, a, T):
    """Compute (w ⊛ a) where w = softmax(conv/T)."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = conv / max(T, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # Compute (w ⊛ a)[j] = sum_m w[m] * a[m-j] for j=0..n-1
    # = ifft(fft(w, N) * conj(fft(a, N)))[:n]
    N = 1 << (len(conv) + n - 2).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    result = np.fft.irfft(W * np.conj(A), N)
    return result[:n]

def mirror_descent(a0, time_limit, T_start, T_end, eta_start, eta_end, n_iter_max=10000):
    """Mirror descent on simplex with adaptive temperature and step size."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** frac
        eta = eta_start * (eta_end / eta_start) ** frac
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        grad = softmax_gradient(conv, a, T)
        # Center the gradient (remove mean for simplex)
        grad -= grad.mean()
        # Mirror descent update
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    if cur_obj < best_obj:
        best_obj = cur_obj
        best_a = a.copy()
    
    return best_a, best_obj

def pgd_optimize(a0, time_limit, T_start, T_end, lr_start, lr_end, n_iter_max=10000):
    """Projected gradient descent on simplex."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

            best_a = a.copy()
        
        grad = softmax_gradient(conv, a, T)
        grad -= grad.mean()
        a = a - lr * grad
        a = project_simplex(a)

    cur_obj = 2.0 * n * conv.max()
    if cur_obj < best_obj:

"""
Step function optimizer using FFT-accelerated gradient methods with adaptive
temperature mirror descent, projected gradient descent, and peak-targeting
local search. Explores multiple sequence lengths (256-4096) with diverse
initializations to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_gradient(conv, a, T):
    """
    Compute (w * a) cross-correlation where w = softmax(conv/T).
    This gives the direction of the smoothed gradient of max(conv).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = conv / max(T, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        idx = int(np.argmax(conv))
        w = np.zeros_like(conv)
        w[idx] = 1.0
    else:
        w /= w_sum
    # (w * a)[j] = sum_m w[m] * a[m-j] for j=0..n-1
    # = ifft(fft(w, N) * conj(fft(a, N)))[:n]
    N = 1 << (len(conv) + n - 2).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    result = np.fft.irfft(W * np.conj(A), N)
    return result[:n]


def obj_val(a):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def normalize_simplex(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)

    if s < 1e-15:

def mirror_descent(a0, time_limit, T_start, T_end, eta_start, eta_end):
    """
    Mirror descent on the simplex with entropy regularization.
    Uses adaptive temperature and step size with exponential decay.
    """
    a = normalize_simplex(a0.copy())
    n = len(a)
    best_obj = obj_val(a)
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            break
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1.0))
        T = T_start * (T_end / T_start) ** frac
        eta = eta_start * (eta_end / eta_start) ** frac

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        grad = softmax_gradient(conv, a, T)
        grad -= grad.mean()
        # Mirror descent update in log-space
        log_update = -eta * grad

def make_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussian (various widths)
    for sigma in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sigma * n)) ** 2)
        g /= g.sum()
        inits['gauss_%g' % sigma] = g

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1) if n > 1 else 1))
    if rc.sum() > 0:
        inits['raised_cos'] = rc / rc.sum()

    # Power law (decaying)
    for alpha in [0.5, 1.0, 2.0]:
        p = (x + 1) ** (-alpha)
        inits['pow_%g' % alpha] = p / p.sum()

    # U-shaped (more mass at edges)
    u = (x - n * 0.5) ** 2 + 1
        inits['u_shape'] = u / u.sum()

    # Bimodal
    bi = np.zeros(n)
    w = max(2, n // 8)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    if bi.sum() > 0:
        inits['bimodal'] = bi / bi.sum()

    # Random Dirichlet
    for alpha in [0.5, 1.0, 5.0]:
        r = rng.dirichlet(np.ones(n) * max(0.01, alpha))
        inits['rand_%g' % alpha] = r

    # Chirp
    chirp = 0.5 * (1 + np.cos(2 * np.pi * x * x / (2 * n)))
    chirp = np.maximum(chirp, 0)
    if chirp.sum() > 1e-10:
        inits['chirp'] = chirp / chirp.sum()

    # Parent sequence (resampled to length n)
    if parent is not None and parent.size > 0:
        p = np.maximum(parent, 0.0)
        if p.sum() > 1e-10:
            if p.size == n:
                inits['parent'] = p / p.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent'] = pp / pp.sum()

    return inits

    time_per_deep = max(10.0, phase2_budget / max(1, n_deep))

    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 600:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 600)
        if t_alloc < 8.0:
            break

        print("  [%d/%d] n=%d, %s, budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))

        # Multi-stage optimization
        sub = t_alloc / 6.0
        a_cur = a0.copy()

        # Stage 1: Mirror descent (broad)
        a_cur, _ = mirror_descent(a_cur, time_limit=sub,
                                  T_start=200.0, T_end=5.0,
                                  eta_start=0.4, eta_end=0.05)
        record(a_cur)

        # Stage 2: PGD (medium)
        a_cur, _ = pgd_optimize(a_cur, time_limit=sub,
                                T_start=50.0, T_end=1.0,
                                lr_start=0.05, lr_end=0.001)
        record(a_cur)

        # Stage 3: Mirror descent (fine)
        a_cur, _ = mirror_descent(a_cur, time_limit=sub,
                                  T_start=20.0, T_end=0.5,
                                  eta_start=0.1, eta_end=0.005)
        record(a_cur)

        # Stage 4: PGD (fine)
        a_cur, _ = pgd_optimize(a_cur, time_limit=sub,
                                T_start=10.0, T_end=0.3,
                                lr_start=0.02, lr_end=0.0005)
        record(a_cur)

        # Stage 5: More mirror descent with very low temperature
        a_cur, _ = mirror_descent(a_cur, time_limit=sub,
                                  T_start=5.0, T_end=0.1,
                                  eta_start=0.05, eta_end=0.002)
        record(a_cur)

        # Stage 6: PGD final
        a_cur, _ = pgd_optimize(a_cur, time_limit=sub,
                                T_start=5.0, T_end=0.1,
                                lr_start=0.01, lr_end=0.0003)
        record(a_cur)

    print("  Deep opt complete. Best: %.6f" % best_score)

    # Phase 3: Peak-targeting local search
    print("=== Phase 3: Peak-targeting local search ===")
    if best_sequence is not None and time.monotonic() < deadline - 100:
        phase3_budget = min(600.0, deadline - time.monotonic() - 100)
        n_anneal = 4
        anneal_per = phase3_budget / n_anneal

        for i in range(n_anneal):
            if time.monotonic() > deadline - 100:
                break
            a_start = best_sequence.copy()
            a_start = normalize_simplex(a_start)
            # Add small noise for diversity
            noise_level = 0.02 * (1 - i * 0.2)
            a_start = a_start + noise_level * rng.standard_normal(len(a_start))
            a_start = normalize_simplex(a_start)

            rng_local = np.random.default_rng(seed + 7000 + i * 1000)
            a_opt, _ = peak_targeting_search(a_start, time_limit=anneal_per, rng=rng_local)
            record(a_opt)

        print("  Local search complete. Best: %.6f" % best_score)

    # Phase 4: Coordinate descent polish
    print("=== Phase 4: Coordinate descent polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 50:
        remaining = deadline - time.monotonic() - 50
        if remaining > 20:
            a_start = normalize_simplex(best_sequence.copy())
            rng_polish = np.random.default_rng(seed + 12345)
            a_opt, _ = coordinate_descent_polish(a_start, time_limit=remaining * 0.6,
                                                 rng=rng_polish)
            record(a_opt)

            if time.monotonic() < deadline - 30:
                rem2 = deadline - time.monotonic() - 30
                if rem2 > 10:
                    a_opt2, _ = peak_targeting_search(
                        normalize_simplex(best_sequence.copy()),
                        time_limit=rem2,
                        rng=np.random.default_rng(seed + 54321))
                    record(a_opt2)

    # Phase 5: Final gradient polish on the very best
    print("=== Phase 5: Final polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            a_best = normalize_simplex(best_sequence.copy())
            a_opt, _ = mirror_descent(a_best, time_limit=remaining * 0.5,
                                      T_start=1.0, T_end=0.05,
                                      eta_start=0.01, eta_end=0.001)
            record(a_opt)
            if time.monotonic() < deadline - 3:
                rem = deadline - time.monotonic() - 3
                if rem > 3:
                    a_opt2, _ = pgd_optimize(a_best, time_limit=rem,
                                             T_start=0.5, T_end=0.05,
                                             lr_start=0.005, lr_end=0.0001)
                    record(a_opt2)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = safe_evaluate(best_sequence)

    print("=== FINAL ===")
    print("Score: %.6f" % best_score)
    print("Length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
