    g = np.zeros(n)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using projected gradient descent on the simplex,
with adaptive smoothing, Frank-Wolfe steps, LP-based directions, and coordinate descent.

Key improvements over previous approach:
1. Work directly on the simplex (allows exact zeros, unlike log-param)
2. Adaptive top-k subgradient with decreasing k for sharp peak targeting
3. Projected gradient + Frank-Wolfe hybrid for fast convergence
4. Frequent LP-based directional improvements
5. FFT-accelerated coordinate descent without binary search
6. Multi-start with diverse structured initializations

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 1:
        return np.maximum(v, 0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_subgradient_topk(a, b, top_k=10):
    """Compute subgradient of max(b) using top-k peak lags. O(n * top_k)."""
    n = len(a)
    top_lags = np.argsort(b)[::-1][:top_k]
    g = np.zeros(n)
    for lag in top_lags:
        if lag < n:
            g[:lag + 1] += 2.0 * a[lag::-1]
        else:
            m_start = lag - n + 1
            m_end = n
            length = lag - m_start
            g[m_start:m_end] += 2.0 * a[length - 1::-1]
    g /= top_k
    return g


def get_smoothed_grad(a, b, T):
    """Compute gradient of log-sum-exp smoothing: (1/T) log(sum exp(T*b))."""
    n = len(a)
    b_shifted = T * (b - b.max())
    exp_b = np.exp(b_shifted)
    p = exp_b / exp_b.sum()
    # Gradient = sum_k p[k] * grad_b[k]
    # grad_b[k] = 2 * a[lag - m] for valid m
    g = np.zeros(n)
    # For efficiency, only use lags where p is significant
    threshold = p.max() * 0.01
    active = np.where(p > threshold)[0]
    for k in active:
        pk = p[k]
        if k < n:
            g[:k + 1] += 2.0 * pk * a[k::-1]
        else:
            m_start = k - n + 1
            length = k - m_start
            g[m_start:n] += 2.0 * pk * a[length - 1::-1]
    return g


def run_projected_gradient(a_init, n, time_budget, top_k_schedule=None, lr_initial=0.01):
    """
    Projected gradient descent on the simplex with adaptive top-k subgradient.
    a_init: sequence on simplex (sum=1, non-negative)
    """
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    start = time.monotonic()
    b = fft_convolve_self(a)
    best_obj = b.max()
    best_a = a.copy()

    # Momentum (Nesterov-like)
    v = np.zeros(n)
    beta = 0.9
    lr = lr_initial
    iteration = 0
    stale_count = 0
    patience = 200

    while time.monotonic() - start < time_budget:
        iteration += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Adaptive top-k: start broad, narrow down
        if top_k_schedule is None:
            top_k = max(3, int(50 * (1 - progress) + 3 * progress))
        else:
            top_k = top_k_schedule

        # Compute subgradient
        g = get_subgradient_topk(a, b, top_k)

        # Momentum update
        v = beta * v - lr * g
        a_candidate = a + v

        # Project to simplex
        a_candidate = project_simplex(a_candidate)

        # Adaptive learning rate
        if a_candidate.max() > 0.5:
            lr *= 0.9
        elif progress > 0.5:
            lr = max(0.001, lr_initial * (1 - progress) * 0.5)

        a = a_candidate
        b = fft_convolve_self(a)
        cur_obj = b.max()

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            stale_count = 0
        else:
            stale_count += 1

        # If stuck, try a Frank-Wolfe step
        if stale_count >= patience:
            stale_count = 0
            patience = max(50, patience // 2)
            # Frank-Wolfe: move mass from high-g to low-g
            i_from = np.argmax(g * np.maximum(a, 1e-10))
            i_to = np.argmin(g)
            if i_from != i_to and a[i_from] > 1e-10:
                # Try several step sizes
                for t in [0.05, 0.1, 0.2, 0.3]:
                    a_try = a.copy()
                    delta = t * a[i_from]
                    a_try[i_from] -= delta
                    a_try[i_to] += delta
                    b_try = fft_convolve_self(a_try)
                    if b_try.max() < cur_obj:
                        a = a_try
                        b = b_try
                        cur_obj = b_try.max()
                        v = np.zeros(n)
                        break

        if iteration % 500 == 0:
            for _ in range(5):
                a_try = a + 0.01 * np.random.randn(n)
                a_try = project_simplex(a_try)
                b_try = fft_convolve_self(a_try)
                if b_try.max() < best_obj:
                    a = a_try
                    best_a = a_try.copy()
                    best_obj = b_try.max()

    return best_a, best_obj


def run_exp_gradient(a_init, n, time_budget, eta_initial=0.5):
    """Exponentiated gradient (mirror descent) on the simplex."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-300)
    a /= a.sum()

    start = time.monotonic()
    b = fft_convolve_self(a)
    best_obj = b.max()
    best_a = a.copy()
    eta = eta_initial
    iteration = 0

    while time.monotonic() - start < time_budget:
        iteration += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Adaptive top-k
        top_k = max(3, int(30 * (1 - progress) + 3 * progress))
        g = get_subgradient_topk(a, b, top_k)

        # Mirror descent step
        log_a = np.log(a)
        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = project_simplex(a_new)

        a = a_new
        b = fft_convolve_self(a)
        cur_obj = b.max()

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # Decay eta
        eta = eta_initial * (1 - 0.3 * progress)

        # Periodic zeroing: set very small elements to 0
        if iteration % 200 == 0:
            threshold = 1e-6
            a[a < threshold] = 0
            s = a.sum()
            if s > 0:
                a /= s

    return best_a, best_obj


def solve_lp_direction(f, M, n, top_k_constraints=30, time_limit=5):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0.
    Returns optimal g (unnormalized) or None.
    """
    c = fft_convolve_self(f)
    active_lags = np.argsort(c)[::-1][:top_k_constraints]

    # Build constraint matrix
    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        if k < n:
            # (f*g)[k] = sum_{j=0}^{k} f[k-j] * g[j]
            for j in range(k + 1):
                A[idx, j] = f[k - j]
        else:
            # (f*g)[k] = sum_{j=k-n+1}^{n-1} f[k-j] * g[j]
            for j in range(k - n + 1, n):
                A[idx, j] = f[k - j]

    c_obj = -np.ones(n)
    b_vec = np.full(len(active_lags), M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_vec,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def apply_lp_direction(a, time_budget):
    """Apply LP-based directional improvement iteratively."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s

    start = time.monotonic()
    b = fft_convolve_self(a)
    best_obj = b.max()
    best_a = a.copy()

    while time.monotonic() - start < time_budget:
        M = b.max()
        # Solve LP
        g = solve_lp_direction(a, M, n, top_k_constraints=25, time_limit=3)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-10:
            break

        # Try blending
        improved = False
        for t_blend in [0.1, 0.2, 0.3, 0.5, 0.7]:
            a_new = (1 - t_blend) * a + t_blend * (g / g_sum)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new /= s_new
            b_new = fft_convolve_self(a_new)
            if b_new.max() < best_obj:
                best_obj = b_new.max()
                best_a = a_new.copy()
                a = a_new
                b = b_new
                improved = True
                break

        if not improved:
            break

    return best_a, best_obj

def get_initialization(name, n, rng):
    """Generate various initializations on the simplex."""
    if name == "uniform":
        return np.full(n, 1.0 / n)
    elif name == "random":
        a = rng.random(n) + 0.01
        return a / a.sum()
    elif name == "anti_symmetric":
        # Large in first half, small in second half (reduces center autocorrelation)
        i = np.arange(n)
        a = 1.0 + 0.5 * np.cos(np.pi * i / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif name == "two_humps":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n * 0.2) / (n * 0.1)) ** 2) + \
            np.exp(-0.5 * ((i - n * 0.8) / (n * 0.1)) ** 2) + 0.01
        return a / a.sum()
    elif name == "cosine_mod":
        i = np.arange(n)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 3 * i / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif name == "edge_concentrated":
        i = np.arange(n)
        a = np.exp(-0.5 * (i / (n * 0.15)) ** 2) + np.exp(-0.5 * ((n - 1 - i) / (n * 0.15)) ** 2)
        a += 0.01
        return a / a.sum()
    elif name == "alternating":
        i = np.arange(n)
        a = np.where(i % 2 == 0, 1.0, 0.3)
        return a / a.sum()
    elif name == "gaussian_center":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n / 2) / (n * 0.2)) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.full(n, 1.0 / n)

    print("Phase 2: Exploitation, budget=%.0fs" % exploit_budget)

    if global_a is not None:
        n = len(global_a)
        a = global_a.copy()

        # 2a: Projected gradient with longer time
        t0 = time.monotonic()
        a_best, obj_best = run_projected_gradient(a, n, exploit_budget * 0.3,
                                                   top_k_schedule=5)
        elapsed = time.monotonic() - t0
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print("  PG refined: obj=%.6f (%.1fs)" % (obj_best, elapsed))

        # 2b: Exponentiated gradient
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 10:
            a = global_a.copy()
            t0 = time.monotonic()
            a_best, obj_best = run_exp_gradient(a, n, min(remaining * 0.3, 300))
            elapsed = time.monotonic() - t0
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print("  Exp-grad: obj=%.6f (%.1fs)" % (obj_best, elapsed))

        # 2c: LP-based direction
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 10 and global_a is not None:
            a = global_a.copy()
            t0 = time.monotonic()
            a_best, obj_best = apply_lp_direction(a, min(remaining * 0.3, 200))
            elapsed = time.monotonic() - t0
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print("  LP direction: obj=%.6f (%.1fs)" % (obj_best, elapsed))

    # Phase 3: Coordinate descent polish (15% of time)
    polish_budget = 0.15 * (deadline - time.monotonic())
    print("Phase 3: Coordinate descent, budget=%.0fs" % polish_budget)

    if global_a is not None and polish_budget > 5.0:
        a = global_a.copy()
        t0 = time.monotonic()
        a_best, obj_best = coordinate_descent_fast(a, polish_budget)
        elapsed = time.monotonic() - t0
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print("  Coord descent: obj=%.6f (%.1fs)" % (obj_best, elapsed))

    # Phase 4: Final combined polish
    remaining = deadline - time.monotonic() - 10.0
    if global_a is not None and remaining > 10.0:
        a = global_a.copy()
        # Quick projected gradient
        t0 = time.monotonic()
        a_best, obj_best = run_projected_gradient(a, len(a), remaining * 0.4,
                                                   top_k_schedule=3)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

        # Final LP
        remaining2 = deadline - time.monotonic() - 5.0
        if remaining2 > 5.0 and global_a is not None:
            a = global_a.copy()
            a_best, obj_best = apply_lp_direction(a, remaining2 * 0.5)
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()

        # Final coord descent
        remaining3 = deadline - time.monotonic() - 5.0
        if remaining3 > 3.0 and global_a is not None:
            a = global_a.copy()
            a_best, obj_best = coordinate_descent_fast(a, remaining3 * 0.8)
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("FINAL: obj=%.6f (verified=%.6f)" % (global_best, ver if np.isfinite(ver) else np.inf))

    if not np.isfinite(ver):
        # Try with different scaling
        for scale in [1.0, 0.1, 10.0, 100.0]:
            test_a = (global_a * scale).tolist()
            ver = evaluate_sequence(test_a)
            if np.isfinite(ver):
                final_a = test_a
                print("Scaled: scale=%.1f, obj=%.6f" % (scale, ver))
                break
        if not np.isfinite(ver):
            final_a = [1.0] * 256
            ver = evaluate_sequence(final_a)
            print("Fallback: obj=%.6f" % ver)

    return final_a
