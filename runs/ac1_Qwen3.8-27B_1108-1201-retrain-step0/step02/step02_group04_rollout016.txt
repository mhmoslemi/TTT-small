import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S ** 2)

def compute_gradient(a, tau):
    """Compute gradient of logsumexp(autocorrelation, tau) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    # Softmax weights
    c_shifted = c - np.max(c)
    w = np.exp(c_shifted / tau)
    w = w / np.sum(w)
    # Gradient: grad[p] = 2 * sum_{j=0}^{n-1} w[p+j] * a[j]
    # This is a "sliding dot product" of w and a
    grad = np.zeros(n)
    for p in range(n):
        # w[p:p+n] dotted with a[0:n]
        if p + n <= len(w):
            grad[p] = 2.0 * np.dot(w[p:p+n], a)
        else:
            # Handle boundary (w might be shorter)
            valid_len = min(n, len(w) - p)
            grad[p] = 2.0 * np.dot(w[p:p+valid_len], a[:valid_len])
    return grad, c

def optimize(a_init, time_limit, rng, target_sum=None):
    """Optimize sequence to minimize max autocorrelation."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    
    if target_sum is None:
        target_sum = np.sum(a)
    
    # Normalize
    a = a * (target_sum / np.sum(a))
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t0 = time.monotonic()
    lr = 0.01
    tau = 0.1
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        a = np.clip(a, 0.0, 1000.0)
        # Renormalize
        S = np.sum(a)
        if S < 1e-10:
            break
        a = a * (target_sum / S)
        
        grad, c = compute_gradient(a, tau)
        
        # Update
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new * (target_sum / S_new)
        
        new_score = compute_score(a_new)
        
        if np.isfinite(new_score) and new_score < compute_score(a):
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
            # Decrease lr slightly and tau
            tau = max(0.01, tau * 0.999)
        else:
            lr = max(lr * 0.9, 1e-6)
        
        step += 1
        if step % 100 == 0:
            print(f"    step={step}, score={compute_score(a):.8f}, best={best_score:.8f}, tau={tau:.4f}, lr={lr:.6f}")
    
    return best_a, best_score
