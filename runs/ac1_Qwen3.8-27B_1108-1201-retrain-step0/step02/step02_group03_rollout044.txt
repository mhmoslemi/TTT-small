Optimization of non-negative step functions to minimize peak autocorrelation ratio.
Uses gradient descent with annealed max approximation, FFT-phase initialization,
coarse-to-fine upscaling, and peak-directed local search.

import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute the evaluation score."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def fft_phase_init(n, rng, n_iter=100):
    """Gerchberg-Saxton style initialization for flat magnitude spectrum."""
    a = np.ones(n) / n
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-14] = 1e-14
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a
