def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n+m-1]

    """Compute the evaluation score."""

        return np.inf

    return 2.0 * n * np.max(c) / (S * S)

def build_A(a):
    """Build the constraint matrix for the LP.
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(a)
    k = np.arange(2*n-1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    valid = (idx >= 0) & (idx < n)
    safe_idx = np.clip(idx, 0, n-1)
    A = np.where(valid, a[safe_idx], 0.0)
    return A

def lp_solve(a, M):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns optimal b or None.
    """
    n = len(a)
    A = build_A(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

"""
Optimization of non-negative step function to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses LP-based peak reduction (iteratively solving a linear program to find the
best sequence under a peak constraint) combined with spectral initializations
and local search. The key insight is that the uniform sequence is a fixed point,
so we start from spectrally-shaped non-uniform sequences to escape it.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score: 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def build_A(a):
    """Build constraint matrix A for the LP.
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(a)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    valid = (idx >= 0) & (idx < n)
    safe_idx = np.clip(idx, 0, n - 1)
    A = np.where(valid, a[safe_idx], 0.0)
    return A


def lp_solve(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M element-wise, b >= 0.
    Returns optimal b or None on failure.
    """
    n = len(a)
    A = build_A(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def optimize_lp_iteration(a_init, time_limit, rng, verbose=True):
    """Run LP-based peak reduction iterations on a normalized sequence."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return a_init.copy(), compute_score(a_init)
    a = a / S * n

    best_score = compute_score(a)
    best_a = a.copy()
    t0 = time.monotonic()
    iteration = 0
    prev_M = np.max(fft_convolve(a, a))
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        M = np.max(c)
        current_score = 2.0 * M / n

        if current_score < best_score - 1e-10:
            best_score = current_score
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if prev_M - M < 1e-9 * max(M, 1e-10):
            if verbose:
                print(f"    Converged at iter {iteration}, M={M:.8f}, score={2*M/n:.8f}")
            break

        if no_improve_count > 20:
            if verbose:
                print(f"    Stuck at iter {iteration}, perturbing...")
            a += 0.05 * (M / n) * rng.standard_normal(n)
            a = np.clip(a, 0, None)
            S = np.sum(a)
            if S > 1e-10:
                a = a / S * n
            no_improve_count = 0
            prev_M = np.max(fft_convolve(a, a))
            continue

        prev_M = M

        b = lp_solve(a, M)
        if b is None:
            if verbose:
                print(f"    LP failed at iter {iteration}")
            break

        sum_b = np.sum(b)
        if sum_b < 1e-10:
            break

        b = b / sum_b * n

        # Try full step first
        a_new = b
        c_new = fft_convolve(a_new, a_new)
        M_new = np.max(c_new)

        if M_new < M * (1 - 1e-10):
            a = a_new
        else:
            # Try geometric steps

                a_new = (1 - t) * a + t * b
                c_new = fft_convolve(a_new, a_new)
                M_new = np.max(c_new)
                if M_new < M * (1 - 1e-10):
                    a = a_new
                    break

                a = np.clip(a, 0, None)
                S = np.sum(a)
                if S > 1e-10:
                    a = a / S * n

        iteration += 1
        if verbose and iteration % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration}: M={M:.6f}, score={2*M/n:.8f}, best={best_score:.8f}, t={elapsed:.1f}s")

    return best_a, best_score


def local_search(a, time_limit, rng, verbose=False):
    """Local search: coordinate descent with random restarts."""
    a = a.astype(np.float64).copy()
    n = len(a)
    mean_val = np.mean(a)
    best_score = compute_score(a)
    best_a = a.copy()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Try random coordinate changes
        for _ in range(min(500, n)):
            i = rng.integers(0, n)
            old_val = a[i]
            # Try multiplicative changes
            multipliers = [0.5, 0.7, 1.3, 1.5, 0.3, 2.0, 0.1]
            for m in multipliers:
                a[i] = max(0.0, old_val * m)
                s = compute_score(a)
                if s < best_score - 1e-10:
                    best_score = s
                    best_a = a.copy()
                    improved = True
                    break
            # If no improvement, revert
            a[i] = old_val

        if not improved:
            # Random perturbation
            a = best_a.copy()
            scale = 0.01 * mean_val
            a += scale * rng.standard_normal(n)
            a = np.clip(a, 0, None)
            S = np.sum(a)
            if S > 1e-10:
                a = a / S * n
            s = compute_score(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()

        iters += 1
        if verbose and iters % 20 == 0:
            print(f"    Local search iter {iters}: best={best_score:.8f}")

    return best_a, best_score


def make_spectral_init(n, harmonics, rng):
    """Create a spectrally-shaped initialization using cosine harmonics."""
    t = np.arange(n)
    a = np.ones(n)
    for freq, amp in harmonics:
        a += amp * np.cos(2 * np.pi * freq * t / n)
    a = np.clip(a, 0.01, None)
    a = a / a.sum() * n
    return a
