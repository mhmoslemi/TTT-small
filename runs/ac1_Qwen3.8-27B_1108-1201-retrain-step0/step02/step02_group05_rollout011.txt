"""
LP-based alternating minimization combined with L-BFGS p-norm refinement
for minimum peak autocorrelation non-negative sequences. The LP approach
exploits the linear structure of cross-correlation constraints to find
guaranteed descent directions, complementing gradient-based methods.
Includes multi-length exploration and targeted local search.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def autocorr_peak(a):
    """max(conv(a,a)) assuming sum(a) is tracked separately."""
    return np.max(fft_convolve(a, a))


def objective_normalized(a):
    """2n * max(conv(a,a)) assuming sum(a) = 1."""
    n = len(a)
    return 2.0 * n * autocorr_peak(a)


def build_conv_constraint_matrix(f, n):
    """Build matrix A such that (A @ b)[k] = conv(f, b)[k] for k=0..2n-2."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = f[k - j]
    return A


def lp_direction(f, M, n):
    """Solve LP: max sum(b) s.t. b>=0, conv(f,b) <= M. Returns normalized b or None."""
    A = build_conv_constraint_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None

    if result.status != 0:
        return None

    g0 = result.x
    s = np.sum(g0)
    if s < 1e-10:
        return None
    return g0 / s


def lp_optimize(f_init, n, max_iters=50, deadline=None):
    """Run LP-based alternating minimization with line search."""
    current = f_init.astype(np.float64).copy()
    current = np.maximum(current, 0)
    s = np.sum(current)
    if s < 1e-10:
        return f_init, objective_normalized(f_init)
    current = current / s
    current_score = objective_normalized(current)
    best = current.copy()
    best_score = current_score

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        M = autocorr_peak(current)
        g = lp_direction(current, M, n)
        if g is None:
            break

        # Check if direction is meaningful
        direction_norm = np.linalg.norm(g - current)
        if direction_norm < 1e-6:
            break

        # Line search: try multiple step sizes
        improved = False
        for alpha in [0.3, 0.5, 0.7, 1.0, 0.2, 0.1, 0.05]:
            f_new = (1.0 - alpha) * current + alpha * g
            f_new = np.maximum(f_new, 0)
            s = np.sum(f_new)
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = objective_normalized(f_new)
            if new_score < current_score - 1e-10:
                current = f_new
                current_score = new_score
                improved = True
                break

        if not improved:
            # Try very small steps
            for alpha in [0.01, 0.02, 0.03]:
                f_new = (1.0 - alpha) * current + alpha * g
                f_new = np.maximum(f_new, 0)

                f_new = f_new / s
                new_score = objective_normalized(f_new)
                if new_score < current_score - 1e-10:

                    current_score = new_score
                    break
            else:
                # No improvement, check convergence
                if it > 5:

        if current_score < best_score:
            best_score = current_score
            best = current.copy()

    return best, best_score


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    if p > 50:
        # For very large p, use soft-max style
        c_max = np.max(c)
        tau = 1.0 / p
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        sum_exp = np.sum(exp_shifted)
        soft_max_val = c_max + tau * np.log(sum_exp)
        w = exp_shifted / sum_exp
        score = 2.0 * n * soft_max_val
        a_rev = a[::-1]
        cross = fft_convolve(a_rev, w)
        grad_a = 2.0 * cross[n - 1:2 * n - 1]
        inner = np.dot(grad_a, a)
        grad_theta = a * (grad_a - inner)
        return score, grad_theta

    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def softmax_obj_grad(theta, tau):
    """Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau)."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def lbgfs_pnorm(theta, n, p_schedule, maxiter, deadline):

def interpolate_to_length(a, n):
    """Interpolate sequence to new length."""
    if len(a) == n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp = np.maximum(a_interp, 1e-6)
    return a_interp / np.sum(a_interp)

                if c_above_avg[k] > 0:
                    w = c_above_avg[k]
                    for i in range(min(n, k + 1)):
                        j = k - i
                        if 0 <= j < n:
                            pull[i] += w * current[j]
            if np.max(pull) > 1e-14:
                pull = pull / np.max(pull)
                candidate *= np.exp(-0.025 * pull)

        elif r < 0.65:
            # Block perturbation
            candidate = current.copy()
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))

        elif r < 0.75:
            # Partial symmetrization
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.02, size=n))

        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.015, size=n))

        candidate = np.maximum(candidate, 1e-10)
        candidate = candidate / np.sum(candidate)
        new_score = objective_normalized(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 1e-8)

        if time.monotonic() - last_print > 10.0:
            last_print = time.monotonic()
            print(f"      local search: current={current_score:.6f}, best={best_score:.6f}, "
                  f"t={time.monotonic()-t0:.0f}s")

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: LP-based alternating minimization, L-BFGS with
    p-norm and soft-max objectives, multi-length exploration, and
    targeted local search polish.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 60.0  # safety margin
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            start_seq = np.array(height_sequence_1, dtype=np.float64)
            start_seq = np.clip(start_seq, 0, None)
            s = np.sum(start_seq)
            if s > 0.01:
                start_seq = start_seq / s
                best_seq = start_seq.copy()
                best_score = objective_normalized(start_seq)
                print(f"Starting sequence: n={len(start_seq)}, score={best_score:.6f}")
        except Exception:
            start_seq = None

    if start_seq is None:
        n = 256
        start_seq = make_random_init(n, rng, spread=0.3)
        best_seq = start_seq.copy()
        best_score = objective_normalized(start_seq)
        print(f"Random start: n={n}, score={best_score:.6f}")

    # === Phase 1: LP-based alternating minimization ===
    print("\n=== Phase 1: LP-based alternating minimization ===")

    n_main = len(best_seq)
    lp_deadline = min(t_global + 400, deadline - 600)
    print(f"  Starting LP: n={n_main}, initial score={best_score:.6f}, "
          f"deadline in {lp_deadline - time.monotonic():.0f}s")

    # Run LP from the best sequence
    lp_result, lp_score = lp_optimize(best_seq, n_main, max_iters=80, deadline=lp_deadline)
    if lp_score < best_score:
        best_score = lp_score
        best_seq = lp_result.copy()
        print(f"  After LP: score={best_score:.6f}")
    else:
        print(f"  After LP: score={lp_score:.6f} (no improvement)")

    # Try LP from a few other initializations
    for init_name in ['chirp', 'cosine']:
        if time.monotonic() > lp_deadline:
            break
        if init_name == 'chirp':
            a0 = make_chirp_init(n_main, 0.4, rng)
        else:
            a0 = make_cosine_init(n_main, 3 + 2 * rng.random(), rng)

        lp_result, lp_score = lp_optimize(a0, n_main, max_iters=40, deadline=lp_deadline)
        if lp_score < best_score:
            best_score = lp_score
            best_seq = lp_result.copy()
            print(f"  LP from '{init_name}': score={best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  LP from '{init_name}': score={lp_score:.6f}")

    # === Phase 2: L-BFGS with p-norm annealing ===
    print(f"\n=== Phase 2: L-BFGS p-norm refinement (score={best_score:.6f}) ===")

    n = len(best_seq)
    p_deadline = min(time.monotonic() + 500, deadline - 400)

    theta = np.log(best_seq + 1e-30)
    p_schedule = [8, 16, 32, 64, 128, 256]
    theta_best, s2 = lbgfs_pnorm(theta, n, p_schedule, maxiter=300, deadline=p_deadline)

    if s2 < best_score:
        best_score = s2
        best_seq = softmax(theta_best).copy()
        print(f"  After p-norm L-BFGS: {best_score:.6f} *** NEW BEST ***")
    else:
        print(f"  After p-norm L-BFGS: {s2:.6f} (no improvement)")

    # === Phase 3: Soft-max refinement ===
    print(f"\n=== Phase 3: Soft-max L-BFGS refinement (score={best_score:.6f}) ===")

    tau_deadline = min(time.monotonic() + 400, deadline - 300)
    theta = np.log(best_seq + 1e-30)
    tau_schedule = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    theta_best, s3 = lbgfs_softmax(theta, n, tau_schedule, maxiter=300, deadline=tau_deadline)

    if s3 < best_score:
        best_score = s3
        best_seq = softmax(theta_best).copy()
        print(f"  After soft-max: {best_score:.6f} *** NEW BEST ***")
    else:
        print(f"  After soft-max: {s3:.6f} (no improvement)")

    # === Phase 4: LP refinement on the improved sequence ===
    print(f"\n=== Phase 4: LP refinement round 2 (score={best_score:.6f}) ===")

    if time.monotonic() < deadline - 300:
        lp2_deadline = min(time.monotonic() + 200, deadline - 200)
        lp_result, lp_score = lp_optimize(best_seq, n, max_iters=60, deadline=lp2_deadline)
        if lp_score < best_score:
            best_score = lp_score
            best_seq = lp_result.copy()
            print(f"  After LP round 2: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  After LP round 2: {lp_score:.6f} (no improvement)")

    # === Phase 5: Try other sequence lengths ===
    print(f"\n=== Phase 5: Multi-length exploration (score={best_score:.6f}) ===")

    if time.monotonic() < deadline - 300:
        other_lengths = [192, 288, 320, 384]
        for n_alt in other_lengths:
            if time.monotonic() > deadline - 200:
                break

            a_interp = interpolate_to_length(best_seq, n_alt)
            # Add perturbation
            a_test = a_interp * np.exp(rng.normal(0, 0.1, n_alt))
            a_test = np.maximum(a_test, 1e-6)
            a_test = a_test / np.sum(a_test)

            # Quick LP
            if time.monotonic() < deadline - 150:
                lp_res, lp_s = lp_optimize(a_test, n_alt, max_iters=30,
                                           deadline=min(time.monotonic() + 100, deadline - 100))
                if lp_s < objective_normalized(a_test):
                    a_test = lp_res

            # Quick L-BFGS
            theta_alt = np.log(a_test + 1e-30)
            if time.monotonic() < deadline - 100:
                theta_alt, s_alt = lbgfs_pnorm(theta_alt, n_alt, [16, 32, 64],
                                               maxiter=200,
                                               deadline=min(time.monotonic() + 150, deadline - 60))
                a_candidate = softmax(theta_alt)
            else:
                a_candidate = a_test
                s_alt = objective_normalized(a_candidate)

            if s_alt < best_score:
                best_score = s_alt
                best_seq = a_candidate.copy()
                print(f"  n={n_alt}: score={s_alt:.6f} *** NEW BEST ***")
            else:
                print(f"  n={n_alt}: score={s_alt:.6f}")

    # === Phase 6: Local search polish ===
    print(f"\n=== Phase 6: Local search polish (score={best_score:.6f}) ===")

    if time.monotonic() < deadline - 30:
        polish_time = min(300.0, deadline - time.monotonic() - 20)
        if polish_time > 30:
            print(f"  Polish time: {polish_time:.0f}s")
            polished, p_score = local_search_polish(best_seq, polish_time, rng)
            if p_score < best_score:
                best_score = p_score
                best_seq = polished.copy()
                print(f"  After polish: {best_score:.6f} *** NEW BEST ***")
            else:
                print(f"  After polish: {p_score:.6f} (no improvement)")

    # === Final: one more LP pass if time allows ===
    if time.monotonic() < deadline - 10:
        n_final = len(best_seq)
        _, final_lp_score = lp_optimize(best_seq, n_final, max_iters=20, deadline=deadline)
        lp_res_final, final_lp_score = lp_optimize(best_seq, n_final, max_iters=20, deadline=deadline)
        if final_lp_score < best_score:
            best_score = final_lp_score
            best_seq = lp_res_final
            print(f"  Final LP: {best_score:.6f} *** NEW BEST ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
