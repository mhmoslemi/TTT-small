```python
def subgradient_step(a, step_size):
    n = len(a)
    b = conv(a, a)
    max_b = np.max(b)
    # Find all positions where b is close to max
    threshold = max_b - 1e-6
    peak_positions = np.where(b >= threshold)[0]
    
    # Subgradient is 2 * sum of conv(e_k, a) for peak positions, normalized
    # conv(e_k, a)[i] = a[k-i] if 0 <= k-i < n, else 0
    # = a[k-i] = a at position k-i
    # So conv(e_k, a) is just a shifted version of a
    
    grad = np.zeros(n)
    for k in peak_positions:
        # conv(e_k, a)[i] = a[k-i] for valid i
        # This means grad[i] += a[k-i] for each peak k
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += a[j]
    
    grad *= 2.0 / len(peak_positions)
    
    # Take step
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0)
    a_new = a_new / np.sum(a_new) * np.sqrt(2 * n)
    
    return a_new
```

Optimization of non-negative step functions to minimize peak autocorrelation.
Uses peak-directed gradient descent, FFT-based initialization, and targeted local search.

import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    b = fft_convolve(a, a)
    return 2.0 * n * np.max(b) / (s * s)

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s

    # the "full" cross-correlation, which can be computed as:
    # ifft(fft(a, N) * conj(fft(w, N))) gives the circular cross-correlation
    # where N >= n + len(w) - 1

    N_fft = 1
    while N_fft < N:

    grad = 2.0 * corr[n-1:n-1+n]  # extract the relevant n elements

def optimize_direct(a_init, time_limit, rng, S=None):
    """
    Direct optimization using peak-directed gradient with adaptive step size.
    a is constrained to be >= 0 and sum(a) = S (default: sqrt(2n)).
    """
    n = len(a_init)
    if S is None:
        S = np.sqrt(2.0 * n)
    
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s * S
    
    best_a = a.copy()
    best_score = score(a)
    
    t0 = time.monotonic()
    step = 0
    lr = 0.1 * S / n  # initial learning rate
    peak_width = 0.1  # starts wide, narrows over time
    
    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        # Anneal peak width
        pw = 0.1 * (0.01) ** (elapsed ** 0.5)  # from 0.1 to 0.01
        
        value, grad, b = peak_gradient(a, pw)
        
        # Adaptive step size
        # Use the change in score to adjust
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new * S
        
        new_score = score(a_new)
        
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        
        # Simple adaptive: if score improved, increase lr slightly; else decrease
        if new_score < score(a):
            lr *= 1.05
        else:
            lr *= 0.9
            a_new = a  # don't update if worse
        
        # Add small noise to escape local minima
        if step % 100 == 99 and elapsed > 0.3:
            noise = rng.normal(0, 0.001 * S / n, n)
            a_new = np.maximum(a_new + noise, 0)
            a_new = a_new / np.sum(a_new) * S
        
        a = a_new
        step += 1
        
        if step % 500 == 0:
            print(f"    t={time.monotonic()-t0:.1f}s, score={score(a):.6f}, best={best_score:.6f}, lr={lr:.2e}")
    
    return best_a, best_score


def optimize_adam(a_init, time_limit, rng):
    """Adam-based optimization with soft-max peak tracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    
    a = a_init.copy()
    a = a / np.sum(a) * S
    theta = np.log(a / S + 1e-15)
    
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr0 = 1.0
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = score(a)
    
    while time.monotonic() - t0 < time_limit:
        elapsed = min(1.0, (time.monotonic() - t0) / time_limit)
        # Temperature schedule
        tau = (0.1 * (0.01) ** (elapsed ** 0.7)) / n  # from 0.1/n to 0.001/n
        
        a = softmax_normalize(theta) * S
        b = fft_convolve(a, a)
        max_b = np.max(b)
        
        # Soft-max weights
        w = np.exp((b - max_b) / tau)
        w = w / np.sum(w)
        value = np.dot(w, b)
        
        # Gradient
        a_rev = a[::-1]
        corr = fft_convolve(w, a_rev)
        g_a = 2.0 * corr[n-1:2*n-1]
        
        # Project gradient onto tangent space of the constraint sum(a) = S
        inner = np.dot(a, g_a) / S
        g_theta = a * (g_a - inner / S)
        
        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1 - b1 ** step)
        m2_hat = m2 / (1 - b2 ** step)
        
        lr = lr0 / np.sqrt(1 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        if step % 500 == 0:
            a_check = softmax_normalize(theta) * S
            sc = score(a_check)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()
            print(f"    t={time.monotonic()-t0:.1f}s, score={sc:.6f}, best={best_score:.6f}, tau={tau:.2e}")
    
    a_final = softmax_normalize(best_theta) * S
    return a_final, best_score


def local_search_v2(a_init, time_limit, rng):
    """
    Improved local search with focus on peak reduction.
    """
    n = len(a_init)
    S = np.sum(a_init)
    current = a_init.astype(np.float64).copy()
    current_score = score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    # Temperature for simulated annealing
    temp = 0.001
    min_temp = 1e-7
    
    while time.monotonic() - t0 < time_limit:
        # Every 1000 iterations, reduce temperature
        r = rng.random()
        
        if r < 0.4:
            # Peak-directed: reduce elements contributing most to the peak
            b = fft_convolve(current, current)
            peak_idx = np.argmax(b)
            
            # Find elements contributing to the peak
            i_min = max(0, peak_idx - n + 1)
            i_max = min(n - 1, peak_idx)
            contrib = np.zeros(n)
            for i in range(i_min, i_max + 1):
                j = peak_idx - i
                contrib[i] = current[i] * current[j]
            
            # Reduce top contributors, boost low contributors
            n_reduce = max(1, n // 20)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:max(1, n - n_reduce)]
            
            alpha = 0.05 + 0.1 * rng.random()
            current[top_idx] *= (1 - alpha)
            # Redistribute to bottom
            redistribution = np.sum(current[top_idx] * 0) + alpha * np.sum(current[top_idx] * (1/(1-alpha)))
            # Simpler: just scale up the bottom
            current[bot_idx] *= (1 + alpha * n_reduce / max(1, len(bot_idx)))
            
        elif r < 0.6:
            # Random block perturbation
            block_size = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - block_size))
            current[start:start+block_size] *= np.exp(rng.normal(-0.05, 0.05, block_size))
            # Boost a different block
            start2 = int(rng.integers(0, n - block_size))
            current[start2:start2+block_size] *= np.exp(rng.normal(0.05, 0.05, block_size))
        
        elif r < 0.75:
            # Global scaling perturbation
            current *= np.exp(rng.normal(0, 0.02, n))
        
        elif r < 0.9:
            # Swap-based: move mass from high to low regions
            n_move = max(1, n // 30)
            # Move mass from elements with high local convolution contribution
            b = fft_convolve(current, current)
            # Weight each element by the max convolution value it participates in
            elem_weight = np.zeros(n)
            for i in range(n):
                # Element i participates in conv positions from i to i+n-1
                # Take the max of those
                lo = max(0, i)
                hi = min(len(b), i + n)
                elem_weight[i] = np.max(b[lo:hi]) if hi > lo else 0
            
            src_idx = np.argsort(elem_weight)[-n_move:]
            dst_idx = np.argsort(elem_weight)[:n_move]
            
            for si, di in zip(src_idx, dst_idx):
                amount = current[si] * 0.1
                current[si] -= amount
                current[di] += amount
        
        else:
            # Mirror perturbation
            sym = 0.5 * (current + current[::-1])
            current = current + 0.1 * (sym - current)
            current += 0.005 * rng.standard_normal(n)
        
        current = np.maximum(current, 0.0)
        s = np.sum(current)
        if s < 1e-10:
            current = best.copy()
            current_score = best_score
            continue
        current = current / s * S
        
        new_score = score(current)
        
        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 100:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current_score = new_score
        
        # Cool down
        temp *= 0.9999
        if temp < min_temp:
            temp = min_temp
    
    return best, best_score


def fft_phase_init(n, rng, n_iter=120):
    """Gerchberg-Saxton style initialization."""
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft <<= 1
    
    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        # Target: flat magnitude
        target_mag = np.ones_like(mag)
        phase = A / mag
        a_full = np.fft.ifft(phase * target_mag).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def proportionate_init(n, rng, n_iter=200):
    """
    Proportionate reduction: iteratively reduce elements that contribute most to peaks.
    """
    a = np.ones(n) / n
    S = 1.0
    
    for it in range(n_iter):
        b = fft_convolve(a, a)
        max_b = np.max(b)
        peak_idx = np.argmax(b)
        
        # Find elements contributing to the peak
        i_min = max(0, peak_idx - n + 1)
        i_max = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        for i in range(i_min, i_max + 1):
            j = peak_idx - i
            contrib[i] = a[i] * a[j]
        
        # Reduce top contributors
        n_reduce = max(1, n // 10)
        top_idx = np.argsort(contrib)[-n_reduce:]
        alpha = 0.02
        a[top_idx] *= (1 - alpha)
        
        # Boost elements with low contribution
        bot_idx = np.argsort(contrib)[:n_reduce]
        a[bot_idx] *= (1 + alpha)
        
        # Normalize
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    
    return a
