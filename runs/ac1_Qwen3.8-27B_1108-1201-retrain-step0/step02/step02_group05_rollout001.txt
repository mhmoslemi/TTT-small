"""
Optimized peak autocorrelation minimization for non-negative sequences.
Uses subgradient descent with simplex projection, L-BFGS with soft-max,
and multi-restart across multiple sequence lengths. Focuses on flattening
the self-convolution profile to minimize the peak-to-mean ratio.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def simplex_proj(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    rho = np.where(sv * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    mu = (sv[rho] - 1) / (rho + 1)
    return np.maximum(v - mu, 0)


def conv_gradient_at_peak(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    m = np.arange(n)
    idx = k - m
    valid = (idx >= 0) & (idx < n)
    grad = np.zeros(n)
    grad[valid] = 2.0 * a[idx[valid]]
    return grad


def subgradient_step(a, lr, top_k=5, smooth_tau=0.001):
    """
    One subgradient descent step for minimizing max(conv(a,a))
    subject to a on the probability simplex.
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    
    # Smoothed subgradient: weight peaks by exp((c - c_max) / tau)
    weights = np.exp((c - c_max) / smooth_tau)
    weights /= np.sum(weights)
    
    # Focus on top peaks
    top_idx = np.argsort(c)[-top_k:]
    top_weights = weights[top_idx]
    top_weights /= np.sum(top_weights)
    
    # Compute combined gradient
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        grad += top_weights[i] * conv_gradient_at_peak(a, k)
    
    # Take step and project
    a_new = simplex_proj(a - lr * grad)
    return a_new, c, c_max


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def soft_max_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_fourier_init(n, rng):
    """
    Initialize using random Fourier features to create structured patterns.
    """
    n_harm = rng.integers(3, 10)
    freqs = rng.integers(1, n // 2 + 1, size=n_harm)
    phases = rng.uniform(0, 2 * np.pi, size=n_harm)
    amps = rng.uniform(0.3, 1.0, size=n_harm)
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for f, p, amp in zip(freqs, phases, amps):
        a += amp * np.cos(2 * np.pi * f * t / n + p)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def run_subgradient(a_init, time_limit, rng, initial_lr=0.01):
    """
    Subgradient descent with adaptive step size and simplex projection.
    """
    current = a_init.astype(np.float64).copy()
    current /= np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    lr = initial_lr
    lr_decay = 0.9995
    top_k = 5
    smooth_tau = 1.0 / (n * 0.5)
    iterations = 0
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        # Adaptive top_k: start focused, widen later
        progress = (time.monotonic() - t0) / time_limit
        if progress < 0.3:
            top_k = 3
            smooth_tau = 1.0 / (n * 1.0)
        elif progress < 0.6:
            top_k = 5
            smooth_tau = 1.0 / (n * 0.5)
        else:
            top_k = 8
            smooth_tau = 1.0 / (n * 0.3)

        candidate, c, c_max = subgradient_step(current, lr, top_k, smooth_tau)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
            # Slightly increase lr on success
            lr *= 1.01
        else:
            # Accept with some probability (simulated annealing flavor)
            delta = new_score - current_score
            accept_prob = np.exp(-delta * n * 2.0)

                current = candidate
                current_score = new_score
                lr *= 0.98
            else:
                lr *= 0.95

        lr = max(lr, 1e-6)
        lr = min(lr, 0.1)
        lr *= lr_decay
        iterations += 1

        if time.monotonic() - last_print > 5.0:

def run_lbgfs_smoothing(a_init, time_limit, rng):
    """
    L-BFGS with annealed soft-max temperature for fine optimization.
    """
    n = len(a_init)
    theta = np.log(a_init + 1e-30)
    best_theta = theta.copy()
    best_score = local_score(a_init)
    t0 = time.monotonic()

    tau_schedule = [0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002]

            break

        maxiter = max(50, int(200 * (time_limit - (time.monotonic() - t0)) / time_limit))
        result = minimize(
            soft_max_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(tau,),
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12,

        theta = best_theta

    return softmax(best_theta), best_score


def coordinate_descent_polish(a_init, time_limit, rng):
    """

    For each coordinate i, find the optimal a_i given others fixed.
    """

        j = int(rng.integers(0, n))
        if i == j:
            j = (i + 1) % n

        # Current values
        ai = current[i]
        aj = current[j]
        rest_sum = 1.0 - ai - aj

        #conv(a,a) is quadratic in (ai, aj)
        # We want to minimize max_k conv(a,a)[k] subject to ai + aj = ai + aj, ai >= 0, aj >= 0
        
        # Compute the current convolution contributions
        # For efficiency, compute the base convolution (with ai=aj=0) and the cross terms
        base = current.copy()
        base[i] = 0
        base[j] = 0
        c_base = fft_convolve(base, base)
        
        # Cross terms: conv involves ai*a[j+k-i], aj*a[i+k-j], etc.
        # This is complex, so use a simpler approach: 
        # try a few values along the line ai + aj = const
        
        total = ai + aj

            test = np.maximum(test, 0)

def run_multi_peak_reduction(a_init, time_limit, rng):
    """
    Directly target the top peaks: identify which a_i values contribute most
    to each peak and reduce them (while increasing others to maintain sum=1).
    """
    current = a_init.astype(np.float64).copy()
    current /= np.sum(current)
    best = current.copy()
    best_score = local_score(current)
    n = len(current)
    t0 = time.monotonic()
    last_print = t0
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        c_sorted = np.sort(c)[::-1]
        # Get top 5 peaks
        top_k = 5
        peak_indices = np.argsort(c)[-top_k:]
        
        # For each peak, find the top contributors
        reduction = np.zeros(n)
        for k in peak_indices:
            weight = max(0, c[k] - c_sorted[min(top_k, len(c_sorted)-1)])
            if weight <= 0:
                continue
            # Contributors to conv[k]: pairs (i, k-i) with 0<=i<n, 0<=k-i<n
            contrib = np.zeros(n)
            for i in range(max(0, k - n + 1), min(k + 1, n)):
                j = k - i
                if 0 <= j < n:
                    contrib[i] += current[i] * current[j]
            # Reduce top contributors
            top_contrib = np.argsort(contrib)[-max(3, n//20):]
            reduction[top_contrib] += weight * contrib[top_contrib]
        
        if np.max(reduction) < 1e-15:
            break
        
        reduction = reduction / np.max(reduction)
        
        # Apply reduction: decrease high-reduction elements, increase others
        factor = 0.01 + 0.02 * rng.random()
        new_current = current * np.exp(-factor * reduction)
        
        # Redistribute: increase elements with low reduction
        low_mask = reduction < np.median(reduction)
        n_low = np.sum(low_mask)
        if n_low > 0:
            increase_amount = (1.0 - np.sum(new_current)) / n_low
            new_current[low_mask] += increase_amount
        
        new_current = np.maximum(new_current, 1e-10)
        new_current = new_current / np.sum(new_current)
        
        new_score = local_score(new_current)
        if new_score < best_score:
            best_score = new_score
            best = new_current.copy()
            current = new_current
        else:
            # Sometimes accept
            if rng.random() < 0.1:
                current = new_current

        if time.monotonic() - last_print > 5.0:
            last_print = time.monotonic()
            print(f"      MP: score={local_score(current):.6f}, best={best_score:.6f}, "
                  f"t={time.monotonic()-t0:.0f}s")

    return best, best_score

        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        elapsed = time.monotonic() - t_global
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)}, elapsed={elapsed:.0f}s ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
