```python
import time
from scipy.optimize import linprog
from scipy.signal import fftconvolve

def compute_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    a = np.clip(a, 0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fftconvolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def lp_direction(a, top_k=100):
    """
    Solve LP to find the best direction to reduce max convolution.
    a: normalized sequence (sum = 1, all >= 0)
    Returns: delta (direction to add to a)
    """
    n = len(a)
    c = fftconvolve(a, a)
    
    # Select top K peaks
    num_peaks = 2 * n - 1
    if top_k < num_peaks:
        # Get indices of top peaks
        top_indices = np.argsort(c)[::-1][:top_k]
    else:
        top_indices = np.arange(num_peaks)
    
    # Build constraint matrix A for the LP
    # We want: c[k] + sum_j A[k,j] * delta[j] <= t
    # where A[k,j] = dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    
    A_rows = np.zeros((len(top_indices), n))
    b_const = np.zeros(len(top_indices))
    
    for idx, k in enumerate(top_indices):
        for j in range(n):
            val = 0.0
            if 0 <= k - j < n:
                val += a[k - j]
            if 0 <= j - k < n:
                val += a[j - k]
            A_rows[idx, j] = val
        b_const[idx] = -c[k]  # c[k] + A[k,:] @ delta <= t → A[k,:] @ delta - t <= -c[k]
    
    # Variables: [delta_0, ..., delta_{n-1}, t]
    # Minimize t → c_obj = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_rows @ delta - t <= -c[k]
    # → [A_rows, -I] @ [delta; t] <= -c[k]
    A_ineq = np.hstack([A_rows, -np.eye(len(top_indices))])
    b_ineq = b_const
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds: delta[j] >= -a[j], t >= 0
    bounds = [(-a[j], None) for j in range(n)] + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if not result.success:
        return None, c
    
    delta = result.x[:n]
    return delta, c
```

```python
"""
LP-guided optimization for minimizing the peak autocorrelation of non-negative sequences.
Uses linear programming to find the optimal direction to reduce the maximum convolution
element, combined with gradient-based and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.signal import fftconvolve

def compute_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fftconvolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def fft_convolve(a, b):
    """FFT-based linear convolution."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def lp_step(a, top_k=80, step_frac=0.5):
    """
    One LP step: find direction to reduce max convolution, take a step.
    a: non-negative array with sum=1
    Returns: new_a (or None if LP fails)
    """
    n = len(a)
    c = fft_convolve(a, a)
    num_c = 2 * n - 1
    
    # Select which peaks to constrain
    if top_k >= num_c:
        peak_indices = np.arange(num_c)
    else:
        peak_indices = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix
    # For each peak k: c[k] + sum_j A[k,j] * delta[j] <= t
    # A[k,j] = dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    
    m = len(peak_indices)
    A = np.zeros((m, n))
    
    for idx in range(m):
        k = int(peak_indices[idx])
        # Part 1: a[k-j] for valid j
        # Valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k, intersected with [0, n-1]
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            # a[k-j] for j = j_start, ..., j_end
            # When j = j_start: a[k - j_start], when j = j_end: a[k - j_end]
            # This is a[k-j_end : k-j_start+1] in reverse
            A[idx, j_start:j_end+1] = a[k-j_end:k-j_start+1][::-1]
        
        # Part 2: a[j-k] for valid j
        # Valid when 0 <= j-k < n, i.e., k <= j <= k+n-1, intersected with [0, n-1]
        j_start2 = max(0, k)
        j_end2 = min(k + n - 1, n - 1)
        if j_start2 <= j_end2:
            # a[j-k] for j = j_start2, ..., j_end2
            A[idx, j_start2:j_end2+1] += a[j_start2-k:j_end2-k+1]
    
    # LP: minimize t
    # subject to: A @ delta - t <= -c[peak_indices]  (i.e., A @ delta[peak_idx] - t <= -c[peak_idx])
    #             sum(delta) = 0
    #             delta[j] >= -a[j]
    #             t >= 0
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.hstack([A, -np.eye(m)])
    b_ub = -c[peak_indices]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-a[j], None) for j in range(n)] + [(0, None)]
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', 
                         options={'max_iter': 5000})
    except Exception:
        return None
    
    if not result.success:
        return None
    
    delta = result.x[:n]
    
    # Determine step size
    # Max step maintaining non-negativity
    neg_mask = delta < 0
    if np.any(neg_mask):
        alpha_max = np.min(a[neg_mask] / (-delta[neg_mask]))
    else:
        alpha_max = 1.0
    
    # Try a few step sizes
    best_a = a.copy()
    best_score = compute_score(a)
    
    for frac in [step_frac, 0.3, 0.1, 0.05]:
        alpha = alpha_max * frac
        new_a = a + alpha * delta
        new_a = np.clip(new_a, 0, None)
        # Renormalize
        S = np.sum(new_a)
        if S < 1e-10:
            continue
        new_a = new_a / S
        new_score = compute_score(new_a)
        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
    
    return best_a

def gradient_step(a, tau=0.01, lr=0.1):
    """
    One gradient step using soft-max approximation of the max.
    a: non-negative array (not necessarily normalized)
    """
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a_norm = a / S  # work with normalized version
    
    c = fft_convolve(a_norm, a_norm)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    # Gradient of the soft-max with respect to a_norm
    # d(sum w[k]*c[k])/da[j] ≈ 2 * sum_k w[k] * dc[k]/da[j]
    # = 2 * (w * a_reversed)[j] convolved appropriately
    
    # Simpler: use the fact that dc[k]/da[j] = a[k-j] + a[j-k] (if valid)
    # So gradient[j] = 2 * sum_k w[k] * (a[k-j] + a[j-k]) (if valid)
    # = 2 * (w * a_rev)[j] + 2 * (w * a)[j] (with appropriate shifts)
    
    # Actually, let's just compute it directly using FFT
    # grad[j] = 2 * sum_k w[k] * (a[k-j] if 0<=k-j<n + a[j-k] if 0<=j-k<n)
    # = 2 * sum_k w[k] * a[k-j] + 2 * sum_k w[k] * a[j-k]
    # First term: for fixed j, sum over k where 0<=k-j<n, i.e., j<=k<j+n
    #   = sum_{k=j}^{min(j+n-1, 2n-2)} w[k] * a[k-j]
    #   This is a cross-correlation of w and a
    # Second term: for fixed j, sum over k where 0<=j-k<n, i.e., j-n+1<=k<=j
    #   = sum_{k=max(0,j-n+1)}^{min(j, 2n-2)} w[k] * a[j-k]
    #   This is a convolution of w and a
    
    # Let's use FFT
    w_padded = np.zeros(2*n-1)
    w_padded[:len(w)] = w  # w has length 2n-1 already
    
    # First term: cross-correlate w with a
    # (w ⋆ a)[j] = sum_k w[k] * a[k-j] (with a extended by 0)
    # This is the same as: reverse a, convolve with w, then take appropriate part
    
    a_rev = a_norm[::-1]
    corr1 = fft_convolve(w_padded, a_rev)
    # corr1[j + n - 1] = sum_k w_padded[k] * a_rev[k + n - 1 - j]
    # = sum_k w[k] * a[n-1 - (k+n-1-j)] = sum_k w[k] * a[j-k]
    # Hmm, let me just compute it directly for clarity (n is small enough)
    
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        for k in range(2*n-1):
            if 0 <= k - j < n:
                val += w[k] * a_norm[k - j]
            if 0 <= j - k < n:
                val += w[k] * a_norm[j - k]
        grad[j] = 2.0 * val
    
    # Project gradient to maintain sum=1: subtract mean * a_norm
    # (Riemannian gradient on the simplex)
    alpha = np.dot(a_norm, grad) / S  # Actually just mean in normalized space
    # We want to update a_norm in the direction -grad projected onto the tangent space of the simplex
    # Tangent space: sum of changes = 0
    grad_proj = grad - np.mean(grad)
    
    # Update
    new_a = a_norm - lr * grad_proj
    new_a = np.clip(new_a, 0, None)
    S_new = np.sum(new_a)
    if S_new > 1e-10:
        new_a = new_a / S_new
    
    # Scale back to original scale
    return new_a * S

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def coordinate_descent(a_init, time_limit, rng):

    current = a_init.astype(np.float64).copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            base = current[j]
            if base < 1e-8:
                continue
            for frac in [0.7, 0.85, 0.95, 1.05, 1.15, 1.4]:
                trial = base * frac
                current[j] = trial
                s = compute_score(current)
                if s < current_score:
                    current_score = s
                    improved = True
                    if s < best_score:
                        best_score = s
                        best = current.copy()
                else:
                    break
            current[j] = current[j] if current[j] != trial else base
        
        if not improved:
            # Small random perturbation
            j = int(rng.integers(0, n))
            current[j] = max(0.0, current[j] * np.exp(rng.normal(0, 0.1)))
            s = compute_score(current)
            if s < current_score * 1.01:
                current_score = s
        
        if (it := int((time.monotonic() - t0) // max(1, time_limit/50))) % 10 == 0:
            pass
    
    return best, best_score
```

```python
LP-guided peak reduction for non-negative step functions.
Uses linear programming to find optimal directions for reducing the maximum
convolution element at the current point (linearized), combined with adaptive
peak tracking, multi-phase optimization, and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.signal import fftconvolve


def compute_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fftconvolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def build_grad_matrix(a, peak_indices):
    """
    Build the gradient matrix for the LP.
    dc[k]/da[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    Returns matrix of shape (len(peak_indices), n).
    """
    n = len(a)
    m = len(peak_indices)
    A = np.zeros((m, n), dtype=np.float64)
    for idx in range(m):
        k = int(peak_indices[idx])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[idx, j_start:j_end + 1] = 2.0 * a[k - j_end:k - j_start + 1][::-1]
    return A


def lp_direction(a, peak_indices, step_frac=0.5):

    Solve LP to find best direction to reduce max convolution value.
    a: normalized non-negative array (sum=1)

    """
    n = len(a)

    m = len(peak_indices)

    except Exception:
        return None

    if not res.success:

    return best_a


def exact_subgrad_step(a, lr=0.01):
    """Step using exact subgradient of max convolution value."""
    n = len(a)
    c = fftconvolve(a, a)
    k_star = int(np.argmax(c))

    # dc[k_star]/da[j] = 2*a[k_star-j] if 0<=k_star-j<n
    grad = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_start <= j_end:
        grad[j_start:j_end + 1] = 2.0 * a[k_star - j_end:k_star - j_start + 1][::-1]

    # Project to tangent space of simplex (sum=0 direction)
    grad_proj = grad - np.mean(grad)

    new_a = a - lr * grad_proj
    new_a = np.clip(new_a, 0.0, None)
    s = np.sum(new_a)
    if s > 1e-10:
        new_a = new_a / s
    return new_a


def softmax_grad_step(a, tau=0.01, lr=0.05):
    """Step using soft-max approximation of max (smoother)."""
    n = len(a)
    c = fftconvolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)

    # Gradient: grad[j] = sum_k w[k] * dc[k]/da[j] = sum_k w[k] * 2*a[k-j] (if valid)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        k_lo = max(0, j)
        k_hi = min(j + n - 1, 2 * n - 2)
        if k_lo <= k_hi:
            # a[k-j] for k in [k_lo, k_hi], which is a[k_lo-j : k_hi-j+1]
            grad[j] = 2.0 * np.dot(w[k_lo:k_hi + 1], a[k_lo - j:k_hi - j + 1])

    grad_proj = grad - np.mean(grad)
    new_a = a - lr * grad_proj
    new_a = np.clip(new_a, 0.0, None)
    s = np.sum(new_a)
    if s > 1e-10:
        new_a = new_a / s
    return new_a


def optimize_lp_phase(a_init, time_limit, top_k=80, step_frac=0.5, rng=None):
    """Main LP-based optimization loop with adaptive strategies."""
    if rng is None:
        rng = np.random.default_rng(123)

    a = a_init.astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(len(a))
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    patience = 80
    cur_top_k = top_k

    print(f"    Start: score={best_score:.6f}, n={len(a)}, top_k={top_k}")

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Select peaks
        c_curr = fftconvolve(a, a)
        num_c = 2 * len(a) - 1
        if cur_top_k >= num_c:
            peak_indices = np.arange(num_c)
        else:
            peak_indices = np.argsort(c_curr)[::-1][:cur_top_k]

        # Try LP step
        new_a = lp_direction(a, peak_indices, step_frac=step_frac)

        if new_a is not None:
            a = new_a
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Fallback: subgradient step
            a = exact_subgrad_step(a, lr=0.005)
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Escape mechanism
        if no_improve >= patience:
            r = rng.random()
            if r < 0.3:
                # Perturb peak contributors
                c_b = fftconvolve(best_a, best_a)
                pk = int(np.argmax(c_b))
                contribs = np.zeros(len(best_a))
                nn = len(best_a)
                js = max(0, pk - nn + 1)
                je = min(pk, nn - 1)
                if js <= je:
                    contribs[js:je + 1] = best_a[pk - je:pk - js + 1][::-1]
                top_c = np.argsort(contribs)[::-1][:max(4, nn // 16)]
                bot_c = np.argsort(contribs)[:max(4, nn // 16)]
                a = best_a.copy()
                a[top_c] *= 0.85
                a[bot_c] *= 1.2
            elif r < 0.5:
                # Gaussian perturbation
                a = best_a * np.exp(0.08 * rng.standard_normal(len(best_a)))
            elif r < 0.7:
                # Block perturbation
                a = best_a.copy()
                bs = max(2, len(a) // 8)
                st = int(rng.integers(0, len(a) - bs))
                a[st:st + bs] *= np.exp(rng.normal(0, 0.1))
            else:
                # Increase top_k for more constraints
                cur_top_k = min(2 * len(a) - 1, int(cur_top_k * 1.5))

            a = np.clip(a, 0, None)
            ss = np.sum(a)
            if ss > 1e-10:
                a = a / ss
            no_improve = 0

        # Occasionally reduce top_k to be less conservative
        if it % 500 == 0 and cur_top_k > 30:
            cur_top_k = max(30, cur_top_k // 2)

        if it % 200 == 0:
            print(f"    it={it}, score={compute_score(a):.6f}, best={best_score:.6f}, "
                  f"top_k={cur_top_k}, t={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def gradient_polish(a_init, time_limit, rng=None):
    """Soft-max gradient descent polish with adaptive temperature."""
    if rng is None:
        rng = np.random.default_rng(77)

    a = a_init.astype(np.float64).copy()
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        frac = (time.monotonic() - t0) / time_limit
        tau = 0.05 / len(a) * (0.01 ** (frac ** 0.5))
        lr = 0.05 * (1.0 - 0.5 * frac)

        a = softmax_grad_step(a, tau=tau, lr=lr)
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if it % 500 == 0:
            print(f"    GP it={it}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}")

    return best_a, best_score


def coordinate_descent_polish(a_init, time_limit, rng=None):
    """Coordinate descent with geometric trial values."""
    if rng is None:
        rng = np.random.default_rng(55)

    current = a_init.astype(np.float64).copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            base = current[j]
            if base < 1e-10:
                continue
            for frac in [0.7, 0.85, 0.95, 1.05, 1.15, 1.35]:
                current[j] = base * frac
                sc = compute_score(current)
                if sc < current_score:
                    current_score = sc
                    improved = True
                    if sc < best_score:
                        best_score = sc
                        best = current.copy()
                else:
                    break
            if not improved:
                current[j] = base

        if not improved:
            j = int(rng.integers(0, n))
            current[j] = max(0.0, current[j] * np.exp(rng.normal(0, 0.08)))
            sc = compute_score(current)
            if sc < current_score * 1.005:
                current_score = sc

    return best, best_score


def make_structured_init(n, rng):
    """Create a structured initial sequence with low autocorrelation tendency."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 15):
        freq = k * (0.7 + 0.6 * rng.random())
        phase = rng.uniform(0, 2 * np.pi)
        amp = 0.25 / np.sqrt(k)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase LP-guided optimization for minimizing the peak convolution score.
    Phase 1: LP optimization on main length (55% time)
    Phase 2: LP with fresh initialization (15% time)
    Phase 3: Gradient polish (10% time)
    Phase 4: Coordinate descent (5% time)
    Phase 5: Explore other lengths (15% time)
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 40.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = compute_score(start_seq)
        print(f"Provided: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    n_main = len(start_seq) if start_seq is not None else 256

    # === PHASE 1: LP optimization on main length ===
    print(f"\n{'='*60}")
    print(f"PHASE 1: LP optimization (n={n_main}, budget={0.55*time_budget:.0f}s)")
    print(f"{'='*60}")

    if start_seq is not None:
        a0 = start_seq.copy()
    else:
        a0 = make_structured_init(n_main, rng)

    t_p1 = 0.55 * time_budget
    a_res, score_res = optimize_lp_phase(a0, t_p1, top_k=80, step_frac=0.5, rng=rng)
    actual = compute_score(a_res)
    print(f"  Phase 1 result: score={actual:.6f}")
    if actual < best_score:
        best_score = actual
        best_seq = a_res.copy()
        print(f"  *** New best: {best_score:.6f} ***")

    # === PHASE 2: LP with fresh initialization ===
    if time.monotonic() < deadline - 60:
        print(f"\n{'='*60}")
        print(f"PHASE 2: LP fresh init (n={n_main})")
        print(f"{'='*60}")

        a0_2 = make_structured_init(n_main, rng)
        t_p2 = min(0.15 * time_budget, max(10.0, deadline - time.monotonic() - 60))
        if t_p2 > 10:
            a_res2, score_res2 = optimize_lp_phase(a0_2, t_p2, top_k=60, step_frac=0.4, rng=rng)
            actual2 = compute_score(a_res2)
            print(f"  Phase 2 result: score={actual2:.6f}")
            if actual2 < best_score:
                best_score = actual2
                best_seq = a_res2.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === PHASE 3: Gradient polish ===
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\n{'='*60}")
        print(f"PHASE 3: Gradient polish")
        print(f"{'='*60}")

        t_p3 = min(0.10 * time_budget, max(10.0, deadline - time.monotonic() - 40))
        if t_p3 > 5:
            a_g, score_g = gradient_polish(best_seq, t_p3, rng=rng)
            actual_g = compute_score(a_g)
            print(f"  Phase 3 result: score={actual_g:.6f}")
            if actual_g < best_score:
                best_score = actual_g
                best_seq = a_g.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === PHASE 4: Coordinate descent ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n{'='*60}")
        print(f"PHASE 4: Coordinate descent")
        print(f"{'='*60}")

        t_p4 = min(0.08 * time_budget, max(5.0, deadline - time.monotonic() - 20))
        if t_p4 > 5:
            a_cd, score_cd = coordinate_descent_polish(best_seq, t_p4, rng=rng)
            actual_cd = compute_score(a_cd)
            print(f"  Phase 4 result: score={actual_cd:.6f}")
            if actual_cd < best_score:
                best_score = actual_cd
                best_seq = a_cd.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === PHASE 5: Explore other lengths ===
    if time.monotonic() < deadline - 30:
        print(f"\n{'='*60}")
        print(f"PHASE 5: Explore other lengths")
        print(f"{'='*60}")

        for n_try in [192, 320, 384, 512, 448]:
            if time.monotonic() > deadline - 30:
                break
            t_try = min(25.0, deadline - time.monotonic() - 20)
            if t_try < 8:
                break

            if best_seq is not None:
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, n_try)
                a0_try = np.interp(x_new, x_old, best_seq)
            else:
                a0_try = make_structured_init(n_try, rng)

            a_t, s_t = optimize_lp_phase(a0_try, t_try, top_k=50, step_frac=0.4, rng=rng)
            actual_t = compute_score(a_t)
            print(f"  n={n_try}: score={actual_t:.6f}")
            if actual_t < best_score:
                best_score = actual_t
                best_seq = a_t.copy()
                print(f"  *** New best at n={n_try}: {best_score:.6f} ***")

    # === FINAL POLISH ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        t_final = deadline - time.monotonic() - 15
        if t_final > 10:
            print(f"\n{'='*60}")
            print(f"FINAL POLISH (t={t_final:.0f}s)")
            print(f"{'='*60}")
            a_fp, s_fp = coordinate_descent_polish(best_seq, t_final, rng=rng)
            if s_fp < best_score:
                best_score = s_fp
                best_seq = a_fp.copy()
                print(f"  Final score: {best_score:.6f}")

    # Return result
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'='*60}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
