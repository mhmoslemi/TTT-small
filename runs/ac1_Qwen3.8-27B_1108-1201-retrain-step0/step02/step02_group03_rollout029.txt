def fft_convolve(a, b):

    while size < n + m - 1:

    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    if S < 1e-10:

    c = fft_convolve(a, a)
    c_shifted = c - np.max(c)
    w = np.exp(c_shifted / T)
    w /= np.sum(w)
    val = np.max(c) + T * np.log(np.sum(np.exp(c_shifted / T)))
    return val, w, c

def grad_soft_max(a, T):

    w = np.exp((c - c_max) / T)

    corr = fft_convolve(w_rev, a)
    n = len(a)

    return g, c

    while time.monotonic() - t0 < time_budget:

"""
Optimization of non-negative step functions minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses gradient descent with annealed soft-max, peak-directed local search,
multi-scale initialization, and coarse-to-fine upscaling.

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the objective score."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def soft_max_and_weights(c, T):
    """Compute soft-max value and normalized weights."""
    c_max = np.max(c)
    shifted = (c - c_max) / T
    e = np.exp(shifted)
    s = np.sum(e)
    return c_max + T * np.log(s), e / s


def grad_soft_max_conv(a, T):
    """Compute gradient of T*soft_max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    _, w = soft_max_and_weights(c, T)
    a_rev = a[::-1].copy()
    corr = fft_convolve(a_rev, w)
    g = (2.0 / T) * corr[n - 1:2 * n - 1]
    return g, c


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def smooth_init(n, rng, n_modes=5):
    """Generate a smooth random initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = rng.uniform(0.3, 0.8)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.1 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def gaussian_window_init(n, rng):
    """Gaussian window with random center and width."""
    t = np.arange(n, dtype=np.float64)
    center = rng.uniform(0.2 * n, 0.8 * n)
    width = rng.uniform(0.1 * n, 0.4 * n)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def upsample_seq(a, new_n, rng):
    """Upsample a sequence to a longer length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_standard_normal(new_n) if hasattr(rng, 'standard_standard_normal') else 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)


def optimize_gradient_softmax(n, a_init, time_budget, rng, T_start=None, T_end=None):

    m2 = np.zeros(n)
