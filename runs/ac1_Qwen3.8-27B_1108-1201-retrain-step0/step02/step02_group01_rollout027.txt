    n = len(a)
    S = a.sum()
    a = a / S  # normalize to sum = 1
    
    # Precompute
    C = fft_convolve_self(a)
    max_C = C.max()
    best_max_C = max_C
    
    eta = eta0
    for it in range(max_iter):
        # Find active lags
        threshold = max_C * 0.999  # lags within 0.1% of max
        active = np.where(C >= threshold)[0]
        m = len(active)
        
        # Compute subgradient of max(C)
        # grad[i] = (1/m) * sum_{k in active} 2*a[k-i] (with out-of-bounds = 0)
        grad = np.zeros(n)
        for k in active:
            # a[k-i] for i = 0, ..., n-1: this is a reversed and shifted
            # a[k-i] = a_rev[i-k+n-1] where a_rev = a[::-1]
            # Actually, let's just compute it directly
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # For i in [lo, hi]: a[k-i] is in bounds
            # grad[i] += 2*a[k-i] / m
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i] / m
        
        # Gradient of the full objective (with sum=1 constraint handled by projection)
        # We're minimizing max(C) subject to sum(a)=1, a>=0
        # The gradient of max(C) is grad (computed above)
        # We project the negative gradient onto the tangent space of the simplex
        # (i.e., subtract the mean to ensure the step sums to 0)
        grad_mean = grad.mean()
        grad_tangent = grad - grad_mean
        
        # Step
        a_new = a - eta * grad_tangent
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()  # renormalize
        
        # Evaluate
        C_new = fft_convolve_self(a_new)
        max_C_new = C_new.max()
        
        # Backtracking if we didn't improve
        if max_C_new > max_C * 1.01:
            eta *= 0.5
            continue
        
        a = a_new
        C = C_new
        max_C = max_C_new
        eta = max(eta * decay, 1e-8)
        
        if max_C < best_max_C:
            best_max_C = max_C
    
    return a, best_max_C

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using active-set projected gradient
with LP-based updates, multiplicative updates, and multi-scale exploration.

Key ideas:
1. The gradient of max(conv(a,a)) at the active lags is simple: 2*a[k-i]
2. Use projected subgradient descent on the simplex (a>=0, sum=1)
3. Periodically solve an LP to get a "global" update
4. Use multiplicative (exponential) updates for stability
5. Multi-start with different lengths and initializations
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csc_matrix

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_gradient_at_active(a, active_lags):
    """Compute grad[i] = sum_{k in active} a[k-i] (out-of-bounds = 0)."""
    n = len(a)
    m = len(active_lags)
    # Use the FFT trick: grad = conv(v, a[::-1])[n-1 : 2n-1]
    # where v[k] = 1 for k in active
    v = np.zeros(2 * n - 1)
    for k in active_lags:
        v[k] = 1.0
    a_rev = a[::-1]
    # conv(v, a_rev) has length (2n-1) + n - 1 = 3n-2
    out_len = (2 * n - 1) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    V = fft(v, fsize)
    AR = fft(a_rev, fsize)
    conv_result = np.real(ifft(V * AR))
    grad = conv_result[n - 1 : 2 * n - 1]
    return grad / m

def objective_value(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    C = fft_convolve_self(a)
    return float(2 * n * C.max() / (S * S))

def projected_gradient_step(a, grad, eta):
    """Take a projected gradient step on the simplex."""
    # Center the gradient (projection onto tangent space of simplex)
    grad_centered = grad - grad.mean()
    a_new = a - eta * grad_centered
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-30:
        return a
    return a_new / s

def multiplicative_step(a, grad, eta):
    """Take a multiplicative (exponential) step."""
    grad_centered = grad - grad.mean()
    # Scale eta to avoid overflow
    max_grad = np.abs(grad_centered).max()
    if max_grad < 1e-15:
        return a
    scaled_eta = min(eta, 1.0 / max_grad)
    a_new = a * np.exp(-scaled_eta * grad_centered)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-30:
        return a
    return a_new / s

def solve_lp_active(active_lags, n, time_limit=5.0):
    """Solve LP: min t s.t. b[i] <= t for all i, sum(a)=1, a>=0.
    Returns (a_optimal, t_optimal) or (None, None)."""
    m = len(active_lags)
    if m == 0 or m >= n:
        return None, None
    
    # Build sparse constraint matrix A_ub (n x (n+1))
    # Row i: b[i] - t <= 0
    # b[i] = sum_{k in active} a[k-i] * I(0<=k-i<n)
    rows = []
    cols = []
    vals = []
    
    for i in range(n):
        for k in active_lags:
            j = k - i
            if 0 <= j < n:
                rows.append(i)
                cols.append(j)
                vals.append(1.0)
        # -t term
        rows.append(i)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csc_matrix((vals, (rows, cols)), shape=(n, n + 1))
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Equality: sum(a) = 1
    A_eq_data = np.ones(n)
    A_eq = csc_matrix((A_eq_data, (np.zeros(n, dtype=int), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=np.zeros(n),
                         A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                         method='highs',
                         options={'presolve': True, 'time_limit': time_limit})
        if result.success and result.x is not None:
            return result.x[:n], result.x[n]
    except Exception:
        pass
    return None, None

def run_projected_gradient(a_init, n, time_budget, initial_eta=0.01, 
                           active_fraction=0.02, lp_every=200,
                           perturb_every=500, perturb_scale=0.01):
    """Run projected subgradient method with periodic LP updates."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    S = a.sum()
    a = a / S  # normalize to sum = 1
    
    C = fft_convolve_self(a)
    max_C = C.max()
    best_max_C = max_C
    best_a = a.copy()
    
    eta = initial_eta
    start = time.monotonic()
    it = 0
    stuck_count = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        
        # Compute convolution
        C = fft_convolve_self(a)
        max_C = C.max()
        
        if max_C < best_max_C:
            best_max_C = max_C
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Find active lags
        threshold = max_C * (1 - active_fraction)
        active = np.where(C >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(C)])
        
        # Compute gradient
        grad = compute_gradient_at_active(a, active)
        
        # Projected gradient step
        a_new = projected_gradient_step(a, grad, eta)
        
        # Check improvement
        C_new = fft_convolve_self(a_new)
        max_C_new = C_new.max()
        
        if max_C_new < max_C:
            a = a_new
            max_C = max_C_new
            eta = min(eta * 1.05, 0.1)  # slightly increase step size
        else:
            eta *= 0.7
            if eta < 1e-8:
                break
        
        # Periodic LP update
        if it % lp_every == 0 and n <= 8192:
            a_lp, t_lp = solve_lp_active(active, n, time_limit=3.0)
            if a_lp is not None:
                C_lp = fft_convolve_self(a_lp)
                if C_lp.max() < max_C:
                    a = a_lp
                    max_C = C_lp.max()
                    eta = initial_eta  # reset step size
                else:
                    # Blend
                    for blend in [0.5, 0.3, 0.1]:
                        a_blend = (1 - blend) * a + blend * a_lp
                        a_blend = np.maximum(a_blend, 0)
                        a_blend = a_blend / a_blend.sum()
                        C_blend = fft_convolve_self(a_blend)
                        if C_blend.max() < max_C:
                            a = a_blend
                            max_C = C_blend.max()
                            break
        
        # Periodic perturbation
        if it % perturb_every == 0:
            rng = np.random.default_rng(it)
            a_pert = a + perturb_scale * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            C_pert = fft_convolve_self(a_pert)
            if C_pert.max() < max_C:
                a = a_pert
                max_C = C_pert.max()
            # Decay perturbation scale
            perturb_scale *= 0.9
    
    return best_a, best_max_C

def coordinate_descent(a, n, time_budget, max_sweeps=100):
    """Coordinate descent polish."""
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    C = fft_convolve_self(a)
    best_max_C = C.max()
    best_a = a.copy()
    start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            
            # Try reducing a[i]
            best_local = orig
            best_local_C = best_max_C
            for frac in [0.5, 0.7, 0.3, 0.8, 0.2]:
                a[i] = orig * frac
                s = a.sum()
                a_try = a / s
                C_try = fft_convolve_self(a_try)
                if C_try.max() < best_local_C:
                    best_local_C = C_try.max()
                    best_local = orig * frac
            a[i] = best_local
            if best_local != orig:
                improved = True
                a = a / a.sum()
                C = fft_convolve_self(a)
                if C.max() < best_max_C:
                    best_max_C = C.max()
                    best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_max_C
