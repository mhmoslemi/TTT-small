Minimize 2*n*max(conv(a,a))/sum(a)^2 using Nesterov accelerated projected gradient
with softmax surrogate, adaptive temperature, and active-set polishing.

Key improvements:
1. Nesterov acceleration for O(1/sqrt(k)) convergence rate
2. Efficient FFT-based gradient computation for the softmax surrogate
3. Proper projection onto {a >= 0, sum(a) = S}
4. Adaptive temperature with restarts
5. Active-set identification and local polishing
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def next_pow2(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = next_pow2(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_softmax_grad(p, a, fsize):
    """Compute grad of softmax objective: grad[j] = 2 * sum_m p[j+m] * a[m].
    This is 2 * cross_correlation(p, a)[0:n]."""
    # Cross-correlation: (p ⊛ a)[j] = sum_m p[m+j] * a[m]
    # Using FFT: ifft(fft(p) * conj(fft(a)))
    P = fft(p, fsize)
    A = fft(a, fsize)
    cc = np.real(ifft(P * np.conj(A)))
    return 2.0 * cc[:len(a)]

def project_nonneg_sum(v, S):
    """Project v onto {a >= 0, sum(a) = S}."""
    n = len(v)
    # Find tau such that sum(max(v_i - tau, 0)) = S
    # This is a decreasing function of tau
    lo = min(v) - 1.0
    hi = max(v) + 1.0
    
    # Bisection
    for _ in range(60):
        tau = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - tau, 0.0))
        if s > S:
            lo = tau
        else:
            hi = tau
    
    tau = (lo + hi) / 2.0
    result = np.maximum(v - tau, 0.0)
    # Fix any numerical issues
    s = result.sum()
    if s > 1e-15:
        result = result * (S / s)
    else:
        result = np.full(n, S / n)
    return result

def nesterov_softmax(a_init, S, T, time_budget, step_size=None, verbose=False):
    """Nesterov accelerated projected gradient on softmax surrogate."""
    n = len(a_init)
    fsize = next_pow2(3 * n - 2)  # for cross-correlation
    
    a = a_init.copy().astype(np.float64)
    y = a.copy()

    start = time.monotonic()
    
    best_obj = np.inf
    best_a = a.copy()
    
    # Compute initial objective
    c = fft_convolve_self(a)
    cur_max = c.max()
    best_obj = cur_max
    
    if step_size is None:
        # Initial step size estimate
        step_size = S / (n * T) * 0.5
    
    momentum = 0.0
    
    for it in range(1000000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute gradient at y
        c_y = fft_convolve_self(y)
        c_max = c_y.max()
        shifted = T * (c_y - c_max)
        exp_shifted = np.exp(shifted)
        total = exp_shifted.sum()
        p = exp_shifted / total
        
        grad = compute_softmax_grad(p, y, fsize)
        
        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k**2)) / 2.0
        beta = (t_k - 1.0) / t_new
        
        # Gradient step from y
        x_new = y - step_size * grad
        x_new = project_nonneg_sum(x_new, S)
        
        # Update y for next iteration
        y = x_new + beta * (x_new - a)
        y = np.maximum(y, 0.0)  # Keep y non-negative for numerical stability
        # Renormalize y (soft constraint)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        
        a = x_new
        t_k = t_new
        
        # Check improvement
        c_a = fft_convolve_self(a)
        cur_max = c_a.max()
        if cur_max < best_obj:
            best_obj = cur_max
            best_a = a.copy()
    
    return best_a, best_obj

def annealed_nesterov(a_init, S, time_budget):
    """Run Nesterov with adaptive temperature annealing."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    
    best_obj = np.inf
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    
    total_start = time.monotonic()
    
    # Temperature schedule
    temperatures = [5, 10, 20, 40, 80, 160, 320, 640, 1280, 2560, 5120]
    
    for T in temperatures:
        elapsed = time.monotonic() - total_start
        remaining = time_budget - elapsed
        if remaining < 2.0:
            break
        
        # Adaptive step size: roughly S / (n * sqrt(T))
        step_size = S / (n * np.sqrt(T)) * 0.3
        
        a, obj = nesterov_softmax(a, S, T, remaining, step_size)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print("  T=%d: obj=%.8f" % (T, obj))
        
        a = a * (S / a.sum())
    
    return best_a, best_obj

def active_set_polish(a, S, time_budget):
    """Polish by focusing on active constraints (near-max lags)."""
    n = len(a)
    fsize = next_pow2(3 * n - 2)
    start = time.monotonic()
    
    a = a.copy().astype(np.float64)
    best_obj = fft_convolve_self(a).max()
    best_a = a.copy()
    
    # Use very high temperature (essentially active set)
    # With top-k active lags
    for top_k in [4, 8, 16, 32]:
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break
        
        # Use high T with the top-k lags
        T_eff = 5000.0
        step_size = S / (n * np.sqrt(T_eff) * top_k) * 0.5
        
        a, obj = nesterov_softmax(a, S, T_eff, remaining, step_size)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print("  Active polish top_k=%d: obj=%.8f" % (top_k, obj))
        a = a * (S / a.sum())
    
    return best_a, best_obj

def coordinate_descent_polish(a, S, time_budget, max_sweeps=20):
    """Coordinate descent: for each coordinate, find the best value via golden section."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    
    best_obj = fft_convolve_self(a).max()
    best_a = a.copy()
    
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            # Try range [0, orig * 2]
            lo = 0.0
            hi = orig * 2.0 if orig > 1e-10 else 1.0
            if hi <= lo:
                hi = lo + 1.0
            
            # Golden section
            c1 = hi - inv_phi * (hi - lo)
            c2 = lo + inv_phi * (hi - lo)
            
            for _ in range(25):
                a[i] = c1
                s = a.sum()
                if s > 1e-10:
                    a_norm = a * (S / s)
                    f1 = fft_convolve_self(a_norm).max()
                else:
                    f1 = np.inf
                
                a[i] = c2
                s = a.sum()
                if s > 1e-10:
                    a_norm = a * (S / s)
                    f2 = fft_convolve_self(a_norm).max()
                else:
                    f2 = np.inf
                
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - inv_phi * (hi - lo)
                    a[i] = c1
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + inv_phi * (hi - lo)
                    a[i] = c2
            
            # Set to best
            a[i] = (lo + hi) / 2.0
            s = a.sum()
            if s > 1e-10:
                a = a * (S / s)
            else:
                a[i] = orig
                a = a * (S / a.sum())
            
            cur_obj = fft_convolve_self(a).max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

def get_initializations(n, S, rng):
    """Generate various initializations."""
    inits = {}
    
    # Uniform
    inits['uniform'] = np.full(n, S / n)
    
    # Random
    a = rng.random(n)
    inits['random'] = a * (S / a.sum())
    
    # Gaussian centered
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - n/2) / (n/5))**2)
    inits['gaussian'] = a * (S / a.sum())
    
    # Cosine ripple
    a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * i / n)
    a = np.maximum(a, 0.01)
    inits['cosine'] = a * (S / a.sum())
    
    # Anti-peak (valley in middle)
    a = 1.0 - 0.4 * np.exp(-0.5 * ((i - n/2) / (n/4))**2)
    a = np.maximum(a, 0.01)
    inits['anti_peak'] = a * (S / a.sum())
    
    # Power law decay from center
    dist = np.abs(i - n/2) / (n/2)
    a = (1.0 + dist)**(-0.5)
    inits['power'] = a * (S / a.sum())
    
    return inits

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/sum(a)^2` for non-negative sequences. The previous approach achieved 1.510224 using log-parameterized Adam with softmax. I need to get below 1.5027.

Key insights for improvement:
1. **Nesterov acceleration** gives better convergence rates than plain gradient descent
2. **Barzilai-Borwein adaptive step size** is more robust than fixed/decaying steps
3. **Direct variable optimization** with projection (rather than log-parameterization) avoids numerical issues
4. **Active-set leveling**: once near the optimum, focus equalization on the top lags

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using Nesterov accelerated projected gradient
with softmax surrogate, Barzilai-Borwein adaptive steps, active-set leveling,
and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def next_pow2(n):
    return 1 << (n - 1).bit_length()


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = next_pow2(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_softmax(p, a, fsize):
    """grad[j] = 2 * sum_m p[j+m] * a[m] = 2 * cross_corr(p,a)[j]"""
    P = fft(p, fsize)
    A = fft(a, fsize)
    cc = np.real(ifft(P * np.conj(A)))
    return 2.0 * cc[:len(a)]


def project_nonneg_sum(v, S):
    """Project onto {a >= 0, sum(a) = S} via bisection on threshold."""
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(50):
        tau = 0.5 * (lo + hi)
        s = np.sum(np.maximum(v - tau, 0.0))
        if s > S:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    result = np.maximum(v - tau, 0.0)
    s = result.sum()
    if s > 1e-15:
        result *= S / s
    else:
        result = np.full(n, S / n)
    return result


def nesterov_bb(a_init, S, T, time_budget, step_init=None, max_iter=200000):
    """Nesterov + Barzilai-Borwein step on softmax(T * max(c))."""
    n = len(a_init)
    fsize = next_pow2(3 * n - 2)

    a = a_init.copy().astype(np.float64)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()

    # Initial step
    if step_init is None:
        step = S / (n * T) * 0.5
    else:
        step = step_init

    # BB state
    s_prev = None
    y_prev_grad = None

    best_obj = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best_obj = c.max()

    g_prev = None

    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        # Compute gradient at y
        c_y = fft_convolve_self(y)
        c_max = c_y.max()
        shifted = T * (c_y - c_max)
        p = np.exp(shifted - shifted.max())
        p /= p.sum()
        grad = compute_grad_softmax(p, y, fsize)

        # BB step update
        if s_prev is not None and y_prev_grad is not None and g_prev is not None:
            y_bb = grad - g_prev
            s_dot_y = np.dot(s_prev, s_prev)
            y_dot_y = np.dot(y_bb, y_bb)
            s_dot_
            = np.dot(s_prev, y_bb)
            if y_dot_y > 1e-30 and abs(s_dot_y) > 1e-30:
                step_new = s_dot_y / y_dot_y
                if step_new > 1e-15 and step_new < 10.0 * step:
                    step = 0.5 * step + 0.5 * min(step_new, 2.0 * step)
        s_prev = y - a  # will update after step
        y_prev_grad = grad.copy()
        g_prev = grad.copy()

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        beta = (t_k - 1.0) / t_new

        x_new = y - step * grad
        x_new = project_nonneg_sum(x_new, S)

        y = x_new + beta * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y *= S / y_sum

        s_prev = x_new - a
        a = x_new
        t_k = t_new

        # Track best
        c_a = fft_convolve_self(a)
        cur = c_a.max()
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()

    return best_a, best_obj


def multi_temp_nesterov(a_init, S, time_budget):
    """Run Nesterov with increasing temperatures."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    best_obj = fft_convolve_self(a).max()
    best_a = a.copy()
    start = time.monotonic()

    # Temperature sequence with time allocation
    phases = [
        (30, 0.15),
        (100, 0.20),
        (300, 0.20),
        (1000, 0.20),
        (3000, 0.15),
        (10000, 0.10),
    ]

    for T, frac in phases:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break
        t_alloc = frac * time_budget
        if t_alloc > remaining:
            t_alloc = remaining

        step = S / (n * np.sqrt(T)) * 0.2
        a, obj = nesterov_bb(a, S, T, t_alloc, step_init=step)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print("    T=%d: obj=%.8f" % (T, obj))
        a *= S / a.sum()

    return best_a, best_obj


def leveling_step(a, S, top_k=20, iters=200):
    """Try to equalize the top-k lags of the autocorrelation."""
    n = len(a)
    fsize = next_pow2(3 * n - 2)
    a = a.copy().astype(np.float64)
    best_obj = fft_convolve_self(a).max()
    best_a = a.copy()

    for _ in range(iters):
        c = fft_convolve_self(a)
        # Get top-k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        max_val = top_vals[0]

        # If spread is tiny, stop
        if top_vals[-1] > max_val * 0.9999:
            break

        # Compute gradient targeting equalization
        # We want to reduce the spread: use weights proportional to (c_k - mean_top)
        mean_top = top_vals.mean()
        weights = np.zeros(2 * n - 1)
        for i, idx in enumerate(top_idx):
            weights[idx] = (top_vals[i] - mean_top) / (max_val - mean_top + 1e-15)

        # Gradient: sum_k w_k * 2*a_{k-j}
        W = fft(weights, fsize)
        A = fft(a, fsize)
        cc = np.real(ifft(W * np.conj(A)))
        grad = 2.0 * cc[:n]

        # Small step
        step = 0.01 * S / n
        a_new = a - step * grad
        a_new = project_nonneg_sum(a_new, S)

        c_new = fft_convolve_self(a_new)
        if c_new.max() < best_obj:
            best_obj = c_new.max()
            best_a = a_new.copy()
            a = a_new
        else:
            a_new = a - step * 0.5 * grad
            a_new = project_nonneg_sum(a_new, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < best_obj:
                best_obj = c_new.max()
                best_a = a_new.copy()
                a = a_new
            else:
                step *= 0.5
                if step < 1e-12:
                    break

    return best_a, best_obj


def coord_desc_polish(a, S, time_budget, max_sweeps=3):
    """Coordinate descent with golden section search per coordinate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    best_obj = fft_convolve_self(a).max()
    best_a = a.copy()
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-12:
                continue
            orig = a[i]
            lo = 0.0
            hi = orig * 1.8

            for _ in range(20):
                m1 = hi - inv_phi * (hi - lo)
                m2 = lo + inv_phi * (hi - lo)

                a[i] = m1
                s = a.sum()
                a_n = a * (S / s) if s > 1e-10 else a
                f1 = fft_convolve_self(a_n).max()

                a[i] = m2
                s = a.sum()
                a_n = a * (S / s) if s > 1e-10 else a
                f2 = fft_convolve_self(a_n).max()

                if f1 < f2:
                    hi = m2
                    m2 = m1
                    f2 = f1
                    m1 = hi - inv_phi * (hi - lo)
                else:
                    lo = m1
                    m1 = m2
                    f1 = f2
                    m2 = lo + inv_phi * (hi - lo)

            a[i] = (lo + hi) * 0.5
            s = a.sum()
            if s > 1e-10:
                a *= S / s
            cur = fft_convolve_self(a).max()
            if cur < best_obj:
                best_obj = cur
                best_a = a.copy()

    return best_a, best_obj


def get_inits(n, S, rng):
    inits = {}
    i = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.full(n, S / n)
    a = rng.random(n)
    inits['random'] = a * (S / a.sum())
    a = np.exp(-0.5 * ((i - n / 2) / (n / 5.0)) ** 2)
    inits['gauss'] = a * (S / a.sum())
    a = 1.0 + 0.3 * np.cos(2 * np.pi * 7.0 * i / n)
    a = np.maximum(a, 0.01)
    inits['ripple'] = a * (S / a.sum())
    a = 1.0 - 0.3 * np.exp(-0.5 * ((i - n / 2) / (n / 3.0)) ** 2)
    a = np.maximum(a, 0.05)
    inits['valley'] = a * (S / a.sum())
    return inits
