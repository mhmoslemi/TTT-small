**Approach: Iterative peak shaving with Newton-like corrections**

The key idea: if the peak is at lag k*, we want to reduce conv(k*). The gradient of conv(k*) with respect to a[i] is:
- If k* >= i and k* - i < n: a[k*-i]
- If k* < i and i - k* < n: a[i-k*]
- (Both can contribute if both conditions hold)

More precisely: d conv(k*) / d a[i] = a[k*-i] + a[i-k*] (with a[j] = 0 for j outside [0,n-1]).

So the "influence" of a[i] on the peak is `a[k*-i] + a[i-k*]`.

A natural update is:
a[i] <- a[i] * (1 - eta * (a[k*-i] + a[i-k*]))

But this might drive some a[i] to zero. We should use an exponential form:
a[i] <- a[i] * exp(-eta * (a[k*-i] + a[i-k*]))

Then normalize: a <- a / sum(a).

This is essentially a gradient step in the log-space (same as the soft-max parameterization but applied directly).

**Multiple peaks**: If there are multiple near-peak lags, we can sum their gradients:
total_grad[i] = sum_{k in hot_lags} (a[k-i] + a[i-k])
a[i] <- a[i] * exp(-eta * total_grad[i])

**Adaptive step size**: Use line search or backtracking to ensure the objective decreases.

The Flat Top window is designed to have minimal sidelobes in its frequency response. Dually, we want a sequence with minimal sidelobes in its time-domain autocorrelation.

"""
Optimization of non-negative sequences to minimize peak autocorrelation.
Uses a combination of:
1. Gradient-based optimization (L-BFGS with soft-max objective)
2. Iterative peak-shaving with multiplicative updates
3. Coordinate descent refinement
4. Exploration of asymmetric initializations and multiple sequence lengths
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def soft_max_obj_grad(theta, tau):
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score = 2.0 * n * soft_max_val
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def pnorm_obj_grad(theta, p):
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def peak_shave(a, n_iters=1000, eta_init=0.1, eta_min=1e-6, cooling=0.999, deadline=None):
    """
    Iterative peak shaving: reduce values that contribute most to peak lags.
    """
    current = a.astype(np.float64).copy()
    current = current / np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    eta = eta_init
    
    t0 = time.monotonic()
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fft_convolve(current, current)
        c_max = np.max(c)
        # Find hot lags (within 5% of max)
        hot_threshold = 0.95 * c_max
        hot_lags = np.where(c >= hot_threshold)[0]
        
        # Compute total gradient from hot lags
        total_grad = np.zeros(n)
        for k in hot_lags:
            # d conv(k) / d a[i] = a[k-i] + a[i-k] (with bounds)
            # For each i, the contribution is a[k-i] if 0 <= k-i < n, 
            #                    plus a[i-k] if 0 <= i-k < n
            # This is: (reversed a) convolved with a delta at k, evaluated at i
            # Or equivalently: cross-correlation
            # a[k-i] for valid k-i: this is a shifted/reversed version
            # a[i-k] for valid i-k: this is a shifted version
            
            # Let's compute it differently:
            # d conv(k) / d a[i] = sum_j delta(j,i) * a[k-j] + a[j] * delta(k-j, i)
            #                     = a[k-i] (if valid) + a[i-k] (if valid)
            
            # a[k-i] is valid when 0 <= k-i < n, i.e., max(0, k-n+1) <= i <= min(k, n-1)
            # a[i-k] is valid when 0 <= i-k < n, i.e., k <= i <= k+n-1, combined with 0 <= i < n: max(0,k) <= i <= min(n-1, k+n-1) = n-1
            
            # So total_grad[i] += a[k-i] + a[i-k] where defined
            
            # Efficient computation:
            # a[k-i] for valid i: this is a reversed and shifted
            # Let's just do it directly for small n
            for i in range(n):
                contrib = 0.0
                if 0 <= k - i < n:
                    contrib += current[k - i]
                if 0 <= i - k < n:
                    contrib += current[i - k]
                total_grad[i] += contrib
        
        # Multiplicative update
        if np.max(total_grad) > 0:
            current = current * np.exp(-eta * total_grad / np.max(total_grad))
            current = np.maximum(current, 1e-12)
            current = current / np.sum(current)
        
        new_score = local_score(current)
        if new_score < best_score:
            best_score = new_score
            best = current.copy()
        
        # Adaptive step size
        if new_score < current_score:
            eta = min(eta * 1.05, 1.0)  # increase slightly
        else:
            eta = max(eta * 0.7, eta_min)  # decrease
        
        current_score = new_score
        eta = max(eta * cooling, eta_min)
        
        if it % 200 == 0:
            print(f"    peak_shave iter {it}: score={current_score:.6f}, best={best_score:.6f}, eta={eta:.6f}")
    
    return best, best_score

Multi-phase optimization for minimum peak autocorrelation non-negative sequences.
Key innovations:
1. Asymmetric initializations (exponential, skewed Gaussian)
2. Iterative peak-shaving with adaptive multiplicative updates
3. Multi-length exploration

import numpy as np

from scipy.optimize import minimize


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def soft_max_obj_grad(theta, tau):
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score = 2.0 * n * soft_max_val
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def make_exp_init(n, rate, side='left', rng=None):
    """Exponential decay initialization."""
    t = np.arange(n, dtype=np.float64)
    if side == 'left':
        a = np.exp(-rate * t / n)
    else:
        a = np.exp(-rate * (n - 1 - t) / n)
    if rng is not None:
        a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_skew_gauss_init(n, center_frac, width, rng):
    """Skewed Gaussian initialization."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    center = center_frac
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    # Add skew
    a = a * (1 + 0.5 * np.sin(np.pi * t))
    if rng is not None:
        a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def peak_shave(a, deadline, rng, eta_init=0.3, eta_min=1e-5):
    """
    Iterative peak shaving with multiplicative updates.
    Directly reduces values contributing to peak autocorrelation lags.
    """
    current = a.astype(np.float64).copy()
    current = np.maximum(current, 1e-10)
    current = current / np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    eta = eta_init
    n_iters = 0
    
    while time.monotonic() < deadline:
        c = fft_convolve(current, current)
        c_max = np.max(c)
        
        # Find hot lags
        hot_threshold = c_max * (1.0 - 0.05)  # within 5% of max
        hot_lags = np.where(c >= hot_threshold)[0]
        
        # Compute gradient from all hot lags
        total_grad = np.zeros(n)
        for k in hot_lags:
            if k < n:
                # g_k[i] = a[k-i] (for i=0..k) + a[i-k] (for i=k..n-1)
                total_grad[:k+1] += current[k::-1]
                total_grad[k:] += current[:n-k]
            else:
                # k >= n: only first term, a[k-i] for i=k-n+1..n-1
                m = k - n + 1  # m >= 1
                total_grad[m:] += current[n-1:m-1:-1]
        
        # Multiplicative update
        max_grad = np.max(total_grad)
        if max_grad > 1e-15:
            current = current * np.exp(-eta * total_grad / max_grad)
            current = np.maximum(current, 1e-12)
            current = current / np.sum(current)
        
        new_score = local_score(current)
        if new_score < best_score:
            best_score = new_score
            best = current.copy()
        
        # Adaptive step size
        if new_score < current_score * (1 - 1e-4):
            eta = min(eta * 1.02, 1.0)
        elif new_score > current_score * (1 + 1e-3):
            eta = max(eta * 0.8, eta_min)
        
        current_score = new_score
        n_iters += 1
        
        if n_iters % 500 == 0:
            print(f"    peak_shave iter {n_iters}: score={current_score:.6f}, best={best_score:.6f}, eta={eta:.4f}, n_hot={len(hot_lags)}")
    
    return best, best_score

    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    
    round_num = 0
    while time.monotonic() < deadline and round_num < max_rounds:
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break

            # Try several candidate values

                    current_score = s
                    improved = True
                    break

            best = current.copy()

            break
    
    return best, best_score

            result = minimize(
                soft_max_obj_grad, theta.copy(),

                options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-13, 'maxfun': 2000}

            s = local_score(softmax(result.x))

"""
Multi-phase optimization for minimum peak autocorrelation non-negative sequences.
Key innovations over previous approach:
1. Asymmetric initializations (exponential decay, skewed Gaussians) to escape symmetric local minima
2. Iterative peak-shaving with adaptive multiplicative updates (directly targets peak lags)
3. Multi-length exploration focusing on n=192-320 range
4. Combined L-BFGS (soft-max + p-norm) with peak-shaving in alternating phases
5. Coordinate descent polish using 1D optimization per coordinate
"""

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def local_score_and_conv(a):
    """Score and full convolution."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c), c


def soft_max_obj_grad(theta, tau):
    """Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau)."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score = 2.0 * n * soft_max_val
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def make_exp_init(n, rate, side, rng):
    """Exponential decay initialization (asymmetric)."""
    t = np.arange(n, dtype=np.float64)
    if side == 'left':
        a = np.exp(-rate * t / n)
