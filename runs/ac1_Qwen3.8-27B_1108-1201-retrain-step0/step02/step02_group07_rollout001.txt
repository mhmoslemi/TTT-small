"""

descent, swap-based local search, and informed annealing.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_value(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_peak_info(conv, min_frac=0.90):
    """Get peak positions and their weights (sharp softmax)."""
    max_c = conv.max()
    if max_c < 1e-15:
        return [int(np.argmax(conv))], np.array([1.0])
    thresh = max_c * min_frac
    peaks = np.where(conv >= thresh)[0]
    if len(peaks) == 0:
        return [int(np.argmax(conv))], np.array([1.0])
    vals = conv[peaks] / max_c
    weights = vals ** 8.0
    wsum = weights.sum()
    if wsum < 1e-30:
        return [int(peaks[0])], np.array([1.0])
    return [int(p) for p in peaks], (weights / wsum)


def compute_gradient_peaks(a, peak_indices, peak_weights):
    """Compute weighted subgradient from specified peaks."""
    n = len(a)
    sg = np.zeros(n)
    for idx, (k, w) in enumerate(zip(peak_indices, peak_weights)):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]
    return sg


def multiplicative_optimize(a0, time_limit, eta_scale=1.0, peak_frac=0.88):
    """
    Multiplicative (mirror) descent with adaptive peak tracking.
    Returns (best_a, best_obj).
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta_scale / n
    t0 = time.monotonic()
    patience = 0
    max_pat = 200

    while time.monotonic() - t0 < time_limit:
        peaks, weights = get_peak_info(conv, min_frac=peak_frac)
        sg = compute_gradient_peaks(a, peaks, weights)
        sg_c = sg - sg.mean()
        # Clamp for stability
        sg_c = np.clip(sg_c, -10.0 / n, 10.0 / n)
        a_new = a * np.exp(-eta * sg_c)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        new_conv = fast_convolve(a_new)
        new_obj = 2.0 * n * new_conv.max()

        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new
        conv = new_conv

        if patience >= max_pat:
            patience = 0
            eta *= 0.5
            peak_frac = min(0.98, peak_frac * 1.02)
            if eta < 1e-10:
                break

    return best_a, best_obj


def coordinate_descent_swap(a0, time_limit, rng):

    Swap-based local search: move mass from positions contributing most

    """
    a = np.maximum(a0, 0.0).copy()

    no_improve = 0
    max_no_improve = 500

    while time.monotonic() - t0 < time_limit:
        peaks, _ = get_peak_info(conv, min_frac=0.92)
        if len(peaks) == 0:
            break

        # Compute contribution of each position to total peak mass
        contrib = np.zeros(n)

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            contrib[js:je + 1] += a[js:je + 1] * a[k - je:k - js + 1][::-1]

        # Find high-contrib source
        if contrib.max() < 1e-15:

        source_candidates = np.where(contrib > 0.7 * contrib.max())[0]
        if len(source_candidates) == 0:
            source_candidates = np.array([int(np.argmax(contrib))])
        i = int(source_candidates[rng.integers(len(source_candidates))])

        # Find low-contrib target
        min_contrib = contrib.min()
        target_candidates = np.where(contrib < 0.3 * max(contrib.max(), 1e-10))[0]
        if len(target_candidates) == 0:
            target_candidates = np.array([int(np.argmin(contrib))])
        j = int(target_candidates[rng.integers(len(target_candidates))])

        if i == j:
            continue

        # Try different transfer amounts
        best_delta = 0.0
        best_new_obj = cur_obj
        frac_options = [0.1, 0.25, 0.4, 0.6]
        for frac in frac_options:
            delta = a[i] * frac
            if delta < 1e-12:
                continue
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            if a_try[i] < 0:
                continue
            a_try = a_try / a_try.sum()
            new_conv = fast_convolve(a_try)
            new_obj = 2.0 * n * new_conv.max()
            if new_obj < best_new_obj:

            cur_obj = best_new_obj

def informed_anneal(a0, time_limit, rng):
    """
    Informed simulated annealing: prefer moves that reduce peak contributions.
    Uses adaptive temperature and peak-aware proposal.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        T = 0.02 * (1.0 - frac) ** 2.5 + 0.0003

        peaks, _ = get_peak_info(conv, min_frac=0.85)
        if len(peaks) == 0:
            break

        # Build contribution map
        contrib = np.zeros(n)
        for k in peaks[:min(8, len(peaks))]:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            contrib[js:je + 1] += a[js:je + 1] * a[k - je:k - js + 1][::-1]

        max_contrib = contrib.max()
        if max_contrib < 1e-15:
            break

        # Choose source: biased toward high contribution
        source_w = contrib / max_contrib
        source_w = np.maximum(source_w, 0.01)
        source_w = source_w / source_w.sum()
        i = int(rng.choice(n, p=source_w))

        # Choose target: biased toward low contribution
        target_w = 1.0 / (contrib + 0.1 * max_contrib)
        target_w = target_w / target_w.sum()
        j = int(rng.choice(n, p=target_w))

        if i == j:
            continue

        delta = a[i] * rng.uniform(0.02, 0.15)
        if delta < 1e-13:
            continue

        a[i] -= delta
        a[j] += delta
        if a[i] < 0:
            a[i] = 0
            delta = a[j] - a[j] + delta  # revert
            a[i] += delta
            a[j] -= delta
            continue

        a = np.maximum(a, 0)
        a = a / a.sum()

        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj

        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < math.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[i] += delta
                a[j] -= delta
                a = np.maximum(a, 0)
                a = a / a.sum()

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Raised cosine (window) - good spectral properties
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    rc = np.maximum(rc, 0) / np.maximum(rc, 0).sum()
    inits['raised_cos'] = rc

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    inits['hann'] = hann / hann.sum()

    # Gaussian (narrow)
    g = np.exp(-0.5 * ((x - mid) / (0.15 * n)) ** 2)
    inits['narrow_gauss'] = g / g.sum()

    # Gaussian (wide)
    g2 = np.exp(-0.5 * ((x - mid) / (0.4 * n)) ** 2)
    inits['wide_gauss'] = g2 / g2.sum()

    # Bimodal (mass at center and edges)
    bi = np.zeros(n)
    w1 = max(1, n // 8)
    w2 = max(1, n // 4)
    bi[:w1] = 1.0
    bi[-w1:] = 1.0
    bi[mid - w2 // 2:mid + w2 // 2 + 1] += 1.0
    inits['bimodal'] = bi / bi.sum()

    # Symmetric tent
    tent = 1.0 - np.abs(x - mid) / (mid + 1)
    tent = np.maximum(tent, 0.01)
    inits['tent'] = tent / tent.sum()

    # Spaced comb (mass at regularly spaced positions)
    comb = np.zeros(n)
    period = max(2, n // 32)
    for pos in range(0, n, period):
        comb[pos] = 1.0
    inits['comb'] = comb / comb.sum()

    # Random Dirichlet
    for alpha in [0.5, 2.0, 10.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['dirichlet_' + str(alpha)] = r

    # Parent-based
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        if len(p) == n:
            inits['parent'] = p / p.sum()
        elif len(p) > n:
            pp = p[:n]
            inits['parent_crop'] = pp / pp.sum()
        else:
            # Interpolate to length n
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits['parent_interp'] = pp / pp.sum()

    # Spectrally-flat init: try to make |FFT(a)| as flat as possible
    # Start with random phases, take magnitude squared of IDFT
    for trial in range(3):
        phases = rng.uniform(0, 2 * np.pi, n)
        c = np.exp(1j * phases) / np.sqrt(n)
        a_spec = np.abs(np.fft.ifft(c)) ** 2
        a_spec = np.maximum(a_spec, 0)
        if a_spec.sum() > 1e-10:
            inits['spectral_' + str(trial)] = a_spec / a_spec.sum()

    return inits

        print("  [%d/%d] n=%d, budget=%.0fs" % (idx + 1, n_cand, n, t_alloc))

        a_cur = a0.copy()
        # Step 1: Multiplicative descent
        sub1 = t_alloc * 0.3
        a_cur, _ = multiplicative_optimize(a_cur, time_limit=sub1, eta_scale=1.0, peak_frac=0.90)
        record(a_cur)

        # Step 2: Targeted local search
        if time.monotonic() < deadline - 800:
            sub2 = min(t_alloc * 0.25, deadline - time.monotonic() - 800)
            if sub2 > 3.0:
                a_cur, _ = two_step_refine(a_cur, time_limit=sub2)
                record(a_cur)

        # Step 3: More multiplicative descent (finer)
        if time.monotonic() < deadline - 800:
            sub3 = min(t_alloc * 0.25, deadline - time.monotonic() - 800)
            if sub3 > 3.0:
                a_cur, _ = multiplicative_optimize(a_cur, time_limit=sub3, eta_scale=0.5, peak_frac=0.95)
                record(a_cur)

        # Step 4: Coordinate descent swap
        if time.monotonic() < deadline - 800:
            sub4 = min(t_alloc * 0.2, deadline - time.monotonic() - 800)
            if sub4 > 3.0:
                rng_cd = np.random.default_rng(seed + idx * 7919)
                a_cur, _ = coordinate_descent_swap(a_cur, time_limit=sub4, rng=rng_cd)
                record(a_cur)

    print("  Deep opt complete. Best: %.6f" % best_score)

    # Phase 3: Informed annealing
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("Phase 3: Informed annealing...")
        phase3_budget = min(500.0, deadline - time.monotonic() - 100)
        n_anneal = 3
        anneal_per = phase3_budget / n_anneal

        for i in range(n_anneal):
            if time.monotonic() > deadline - 100:
                break
            a_start = best_sequence.copy()
            rng_a = np.random.default_rng(seed + 10000 + i * 31)
            a_opt, _ = informed_anneal(a_start, time_limit=anneal_per, rng=rng_a)
            record(a_opt)

        print("  Annealing done. Best: %.6f" % best_score)

    # Phase 4: Final fine optimization
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 10
        a_best = best_sequence.copy()

        sub = remaining * 0.4
        a_opt, _ = multiplicative_optimize(a_best, time_limit=sub, eta_scale=0.2, peak_frac=0.97)
        record(a_opt)

        if time.monotonic() < deadline - 10:
            rem2 = deadline - time.monotonic() - 10
            if rem2 > 5.0:
                a_opt2, _ = two_step_refine(a_best, time_limit=rem2 * 0.6)
                record(a_opt2)

        if time.monotonic() < deadline - 10:
            rem3 = deadline - time.monotonic() - 10
            if rem3 > 3.0:
                rng_f = np.random.default_rng(seed + 99999)
                a_opt3, _ = informed_anneal(a_best, time_limit=rem3, rng=rng_f)
                record(a_opt3)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = safe_eval(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
