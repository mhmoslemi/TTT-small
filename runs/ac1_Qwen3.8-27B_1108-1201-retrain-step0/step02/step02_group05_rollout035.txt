def target_peak_reduction(a, rng, iters=100):
    """
    Targeted local search: identify peaks and reduce them.
    """
    a = a.astype(np.float64).copy()
    a = a / np.sum(a)
    n = len(a)
    current_score = local_score(a)
    best = a.copy()
    best_score = current_score
    
    for it in range(iters):
        c = fft_convolve(a, a)
        # Find top peaks
        sorted_idx = np.argsort(c)[::-1]
        top_peaks = sorted_idx[:5]
        
        # For each peak, compute which a[i] contribute most
        # and reduce them while increasing others
        total_reduction = np.zeros(n)
        total_increase = np.zeros(n)
        
        for pk in top_peaks:
            weight = c[pk] / c[sorted_idx[0]]  # normalized weight
            # Contribution: a[i] * a[pk-i] for valid i
            contribs = np.zeros(n)
            for i in range(n):
                j = pk - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            
            # Reduce top contributors
            top_k = max(2, n // 50)
            top_idx = np.argsort(contribs)[-top_k:]
            total_reduction[top_idx] += weight * contribs[top_idx]
            
            # Increase bottom contributors (or random others)
            bot_idx = np.argsort(contribs)[:top_k]
            total_increase[bot_idx] += weight * np.mean(contribs[top_idx]) / max(1, np.mean(contribs[bot_idx]))
        
        # Apply modifications
        eps = 0.01
        a_new = a.copy()
        total_reduction = np.maximum(total_reduction, 0)
        total_increase = np.maximum(total_increase, 0)
        
        # Normalize the adjustments
        if np.max(total_reduction) > 1e-15:
            a_new -= eps * total_reduction / np.max(total_reduction)
        if np.max(total_increase) > 1e-15:
            a_new += eps * total_increase / np.max(total_increase)
        
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / np.sum(a_new)
        new_score = local_score(a_new)
        
        if new_score < current_score:
            a = a_new
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
    
    return best, best_score


def make_window_init(n, window_type, rng):
    """Create various window-function initializations."""
    t = np.arange(n) / (n - 1)  # 0 to 1
    if window_type == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif window_type == 'hamming':
        a = 0.54 - 0.46 * np.cos(2 * np.pi * t)
    elif window_type == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    elif window_type == 'tukey':
        r = 0.25  # parameter
        a = np.ones(n)
        m = int(r * n)
        for i in range(m):
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / m - 1)))
            a[n-1-i] = a[i]
    elif window_type == 'gaussian':
        center = (n - 1) / 2.0
        width = n * 0.2
        a = np.exp(-0.5 * ((t * (n-1) - center) / width) ** 2)
    elif window_type == 'inverted_gaussian':
        center = (n - 1) / 2.0
        width = n * 0.3
        a = 1.0 - 0.5 * np.exp(-0.5 * ((t * (n-1) - center) / width) ** 2)
    elif window_type == 'linear_decrease':
        a = np.linspace(1.0, 0.1, n)
    elif window_type == 'bilateral':
        center = (n - 1) / 2.0
        a = np.exp(-0.3 * np.abs(t * (n-1) - center) / center)
    else:
        a = np.ones(n)
    
    # Add small noise
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)

"""
Multi-stage optimization for minimum peak autocorrelation non-negative sequences.
Combines L-BFGS with adaptive soft-max smoothing, LP-based directional updates,
projected gradient descent, and targeted peak-reduction local search.
Explores multiple sequence lengths with diverse structural initializations.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def soft_max_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def pnorm_obj_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p.
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum} using water-filling."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - target_sum) / rho > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, target_sum / n)
    rho_star = idx[-1] + 1
    theta = (cssv[rho_star - 1] - target_sum) / rho_star
    return np.maximum(v - theta, 0)


def fft_grad_conv_max(a, tau=0.01):
    """
    Compute gradient of 2n * soft_max(conv(a,a), tau) with respect to a.
    a is assumed non-negative. Returns grad (same shape as a).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    return grad_a


def projected_grad_search(a_init, time_limit, rng):
    """
    Projected gradient descent on the simplex with adaptive tau and step size.
    """
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    n = len(a)
    current_score = local_score(a)
    best = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    lr = 0.005
    tau = 0.02
    patience = 0
    best_patience_score = current_score
    best_patience_tau = tau

    while time.monotonic() - t0 < time_limit:
        grad = fft_grad_conv_max(a, tau)
        a_new = project_simplex(a - lr * grad)
        new_score = local_score(a_new)

        if new_score < current_score:
            a = a_new
            current_score = new_score
            lr = min(lr * 1.05, 0.02)
            patience = 0
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            lr *= 0.8
            patience += 1
            if patience > 30:
                tau *= 0.7
                lr = 0.005
                patience = 0

        if time.monotonic() - t0 > time_limit * 0.85 and patience > 80:
            break

    return best, best_score


def lp_direction_step(f, solver=None):
    """
    Solve LP: maximize sum(g) subject to g >= 0, conv(f,g) <= M.
    Returns the optimal g (or None if LP fails).
    """
    n = len(f)
    M = np.max(fft_convolve(f, f))

    if M <= 0:
        return None

    try:
        import cvxpy as cp
        g = cp.Variable(n)
        constraints = [g >= 0]

        # Build constraint matrix: F @ g <= M * ones(2n-1)
        # F[k, i] = f[i] if 0 <= k-i < n, else 0
        F = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                F[k, i] = f[i]

        constraints.append(F @ g <= M)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        if solver is not None:
            prob.solve(solver=solver)
        else:
            prob.solve(solver=cp.ECOS)

        if g.value is not None:
            return np.maximum(g.value, 0)
    except Exception:
        pass

    return None


def lp_iteration(f, time_limit, max_iterations=50):
    """
    Perform LP-based iterations: solve for direction g, update f.
    """
    f = f.astype(np.float64).copy()
    f = f / np.sum(f)
    n = len(f)
    current_score = local_score(f)
    best = f.copy()
    best_score = current_score
    t0 = time.monotonic()

    # Try to find a working solver
    solver = None
    try:
        import cvxpy as cp
        for s in [cp.ECOS, cp.SCS]:
            solver = s
            break
    except ImportError:
        return best, best_score

    for it in range(max_iterations):
        if time.monotonic() - t0 > time_limit - 30:
            break

        g = lp_direction_step(f, solver)
        if g is None or np.sum(g) <= 0:
            break

        # Rescale g to have same sum as f
        g = g * (np.sum(f) / np.sum(g))

        # Line search for step size t
        best_t = 0.0
        best_s = current_score
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            s_new = local_score(f_new)
            if s_new < best_s:
                best_s = s_new
                best_t = t

        if best_t == 0.0:
            break

        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0)
        f = f / np.sum(f)
        current_score = local_score(f)

        if current_score < best_score:
            best_score = current_score
            best = f.copy()
            if it % 5 == 0:
                print(f"    LP iter {it}: score={current_score:.8f}, t={best_t:.2f}")

    return best, best_score


def target_peak_search(a_init, time_limit, rng):
    """
    Targeted local search: identify peaks in convolution and reduce them
    by adjusting contributing a values.
    """
    a = a_init.astype(np.float64).copy()
    a = a / np.sum(a)
    n = len(a)
    current_score = local_score(a)
    best = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    step_size = 0.015

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        c_max = np.max(c)
        # Find peaks above 0.9 * max
        peak_threshold = 0.92 * c_max
        peak_indices = np.where(c >= peak_threshold)[0]

        if len(peak_indices) == 0:
            peak_indices = np.array([int(np.argmax(c))])

        # For each peak, find top contributors and reduce them
        reduction = np.zeros(n)
        increase = np.zeros(n)

        for pk in peak_indices[:8]:
            weight = c[pk] / c_max
            # Contributors: a[i] * a[pk-i] for valid i
            contribs = np.zeros(n)
            valid_i = []
            for i in range(n):
                j = pk - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
                    valid_i.append(i)

            if len(valid_i) == 0:
                continue

            # Reduce top contributors
            top_k = max(2, n // 40)
            sorted_c = np.argsort(contribs)[::-1]
            top_idx = sorted_c[:top_k]
            reduction[top_idx] += weight * contribs[top_idx]

            # Increase low contributors
            bottom_k = min(top_k, len(valid_i) - top_k)
            if bottom_k > 0:
                bot_idx = sorted_c[-bottom_k:]
                bot_idx = bot_idx[contribs[bot_idx] < np.max(contribs[top_idx])]
                if len(bot_idx) > 0:
                    increase[bot_idx] += weight * np.mean(contribs[top_idx]) / (np.max(contribs[bot_idx]) + 1e-15)

        # Apply with normalization
        max_r = np.max(reduction)
        max_i = np.max(increase)

        a_new = a.copy()
        if max_r > 1e-15:
            a_new -= step_size * reduction / max_r
        if max_i > 1e-15:
            a_new += step_size * increase / max_i

        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / np.sum(a_new)
        new_score = local_score(a_new)

        if new_score < current_score:
            a = a_new
            current_score = new_score
            step_size = min(step_size * 1.02, 0.03)
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            step_size *= 0.9

    return best, best_score


def lbgfs_optimize(a_init, time_limit, rng, use_pnorm=True):
    """
    L-BFGS optimization with decreasing soft-max (or p-norm) smoothing.
    """
    n = len(a_init)
    theta = np.log(a_init + 1e-30)
    best_theta = theta.copy()
    best_score = local_score(a_init)
    t0 = time.monotonic()

    if use_pnorm:
        p_schedule = [8, 16, 32, 64, 128, 256]
        for p in p_schedule:
            if time.monotonic() - t0 > time_limit - 30:
                break
            result = minimize(
                pnorm_obj_grad, theta.copy(),
                method='L-BFGS-B',
                jac=True, args=(p,),
                options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-13, 'maxfun': 4000}
            )
            theta_new = result.x
            a_new = softmax(theta_new)
            s = local_score(a_new)
            if s < best_score:
                best_score = s
                best_theta = theta_new.copy()
                theta = best_theta.copy()

    # Switch to soft-max for finer tuning
    tau_schedule = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit - 20:
            break
        result = minimize(
            soft_max_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(tau,),
            options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-14, 'maxfun': 5000}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)
        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
            theta = best_theta.copy()

    best_a = softmax(best_theta)
    best_a = best_a / np.sum(best_a)
    return best_a, best_score


def make_structured_init(n, init_type, rng):
    """Create various structural initializations."""
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)

    if init_type == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    elif init_type == 'hamming':
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * t)
    elif init_type == 'blackman':
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t)
    elif init_type == 'tukey':
        r = 0.3
        m = max(1, int(r * n))
        a = np.ones(n)
        for i in range(m):
            val = 0.5 * (1.0 + np.cos(np.pi * (2.0 * i / m - 1.0)))
            a[i] = val
            a[n - 1 - i] = val
    elif init_type == 'gauss':
        center = (n - 1) / 2.0
        width = n * (0.15 + 0.15 * rng.random())
        a = np.exp(-0.5 * ((t * (n - 1) - center) / width) ** 2)
    elif init_type == 'inv_gauss':
        center = (n - 1) / 2.0
        width = n * 0.25
        a = 1.0 - 0.6 * np.exp(-0.5 * ((t * (n - 1) - center) / width) ** 2)
    elif init_type == 'bilateral':
        center = (n - 1) / 2.0
        a = np.exp(-0.4 * np.abs(t * (n - 1) - center) / max(1, center))
    elif init_type == 'raised_cos':
        a = np.cos(np.pi * t) ** 2
    elif init_type == 'sinc_like':
        x = (t - 0.5) * np.pi
        a = np.sinc(t - 0.5) ** 2 + 0.1
    elif init_type == 'chirp':
        alpha = 0.3 + 0.4 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t * n / 4.0)
    elif init_type == 'linear':
        a = np.linspace(0.3, 1.0, n)
    elif init_type == 'parabolic':
        a = 1.0 - 0.5 * (2.0 * t - 1.0) ** 2
    else:
        a = np.ones(n)

    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-stage optimization for minimum peak autocorrelation non-negative sequences.
    Phases: L-BFGS refinement, LP-based iterations, targeted peak search,
    multi-start exploration, and final polish.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 80.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        s = local_score(start_seq)
        print(f"Starting sequence: n={len(start_seq)}, score={s:.8f}")
        best_score = s
        best_seq = start_seq.copy()

    # ============ Phase 1: L-BFGS refinement of starting sequence ============
    print("\n=== Phase 1: L-BFGS refinement ===")
    if best_seq is not None:
        n = len(best_seq)
        time_alloc = min(500.0, (deadline - time.monotonic()) * 0.25)
        print(f"  Time allocated: {time_alloc:.0f}s")
        polished, score = lbgfs_optimize(best_seq, time_alloc, rng)
        if score < best_score:
            best_score = score
            best_seq = polished
            print(f"  After L-BFGS: {best_score:.8f} *** IMPROVED ***")
        else:
            print(f"  L-BFGS best: {score:.8f}")

    # ============ Phase 2: LP-based iterations ============
    print(f"\n=== Phase 2: LP-based iterations (score={best_score:.8f}) ===")
    if best_seq is not None:
        time_alloc = min(400.0, (deadline - time.monotonic()) * 0.2)
        if time_alloc > 60:
            polished, score = lp_iteration(best_seq, time_alloc, max_iterations=80)
            if score < best_score:
                best_score = score
                best_seq = polished
                print(f"  After LP: {best_score:.8f} *** IMPROVED ***")
            else:
                print(f"  LP best: {score:.8f}")

    # ============ Phase 3: Targeted peak search ============
    print(f"\n=== Phase 3: Targeted peak search (score={best_score:.8f}) ===")
    if best_seq is not None:
        time_alloc = min(300.0, (deadline - time.monotonic()) * 0.15)
        if time_alloc > 30:
            polished, score = target_peak_search(best_seq, time_alloc, rng)
            if score < best_score:
                best_score = score
                best_seq = polished
                print(f"  After peak search: {best_score:.8f} *** IMPROVED ***")

            # L-BFGS after peak search
            if deadline - time.monotonic() > 120:
                time2 = min(150.0, (deadline - time.monotonic()) * 0.2)
                refined, score2 = lbgfs_optimize(best_seq, time2, rng)
                if score2 < best_score:
                    best_score = score2
                    best_seq = refined
                    print(f"  After post-peak L-BFGS: {best_score:.8f} *** IMPROVED ***")

    # ============ Phase 4: Multi-start exploration ============
    print(f"\n=== Phase 4: Multi-start exploration (best={best_score:.8f}) ===")
    lengths_to_try = [192, 224, 256, 288, 320, 384, 448]
    init_types = ['hann', 'tukey', 'inv_gauss', 'bilateral', 'parabolic', 'chirp', 'blackman']

    for n in lengths_to_try:
        if deadline - time.monotonic() < 180:
            break

        time_for_length = min(400.0, (deadline - time.monotonic()) * 0.25)
        if time_for_length < 100:
            break

        for wt in init_types:
            if deadline - time.monotonic() < 150:
                break
            time_per_init = min(80.0, (deadline - time.monotonic()) * 0.1)
            if time_per_init < 30:
                break

            a0 = make_structured_init(n, wt, rng)

            # Mix with interpolated best if available
            if best_seq is not None and len(best_seq) != n:
                a_interp = np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, len(best_seq)),
                    best_seq
                )
                a_interp = np.maximum(a_interp, 0.01)
                a0 = 0.4 * a0 + 0.6 * a_interp
                a0 = a0 / np.sum(a0)
            elif best_seq is not None:
                a_perturbed = best_seq * np.exp(rng.normal(0, 0.15, n))
                a_perturbed = np.maximum(a_perturbed, 0.01)
                a0 = 0.3 * a0 + 0.7 * a_perturbed
                a0 = a0 / np.sum(a0)

            refined, score = lbgfs_optimize(a0, time_per_init, rng)
            if score < best_score:
                best_score = score
                best_seq = refined
                print(f"  n={n}, {wt}: {score:.8f} *** NEW BEST ***")
            else:
                print(f"  n={n}, {wt}: {score:.8f}")

    # ============ Phase 5: Intensive final optimization ============
    print(f"\n=== Phase 5: Intensive final optimization (score={best_score:.8f}) ===")
    if best_seq is not None:
        for round_num in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 120:
                break

            # L-BFGS with very small tau
            time_lbgfs = min(150.0, remaining * 0.4)
            theta = np.log(best_seq + 1e-30)
            result = minimize(
                soft_max_obj_grad, theta,
                method='L-BFGS-B',
                jac=True, args=(0.0003,),
                options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-15, 'maxfun': 20000}
            )
            a_new = softmax(result.x)
            a_new = a_new / np.sum(a_new)
            s = local_score(a_new)
            if s < best_score:
                best_score = s
                best_seq = a_new
                print(f"  Round {round_num}: L-BFGS fine -> {s:.8f}")

            # Projected gradient
            if deadline - time.monotonic() > 80:
                pg_time = min(80.0, (deadline - time.monotonic()) * 0.25)
                pg_result, pg_score = projected_grad_search(best_seq, pg_time, rng)
                if pg_score < best_score:
                    best_score = pg_score
                    best_seq = pg_result
                    print(f"  Round {round_num}: PGD -> {s:.8f}")

            # Targeted peak search
            if deadline - time.monotonic() > 60:
                pk_time = min(60.0, (deadline - time.monotonic()) * 0.2)
                pk_result, pk_score = target_peak_search(best_seq, pk_time, rng)
                if pk_score < best_score:
                    best_score = pk_score
                    best_seq = pk_result
                    print(f"  Round {round_num}: peak -> {pk_score:.8f}")

    # ============ Phase 6: Final micro-polish ============
    print(f"\n=== Phase 6: Final micro-polish (score={best_score:.8f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            # Very targeted: just reduce the single worst peak
            a = best_seq.copy()
            current_s = best_score
            for micro_iter in range(500):
                if time.monotonic() > deadline - 20:
                    break
                c = fft_convolve(a, a)
                peak_idx = int(np.argmax(c))
                # Find top contributors to this peak
                contribs = np.zeros(len(a))
                for i in range(len(a)):
                    j = peak_idx - i
                    if 0 <= j < len(a):
                        contribs[i] = a[i] * a[j]
                # Reduce top 3, increase bottom 5
                top_idx = np.argsort(contribs)[-3:]
                bot_idx = np.argsort(contribs)[:5]
                a_new = a.copy()
                a_new[top_idx] *= 0.9995
                a_new[bot_idx] *= 1.0005
                a_new = np.maximum(a_new, 1e-10)
                a_new = a_new / np.sum(a_new)
                s_new = local_score(a_new)
                if s_new < current_s:
                    a = a_new
                    current_s = s_new
                    if current_s < best_score:
                        best_score = current_s
                        best_seq = a.copy()

            if best_score < (best_seq is not None and local_score(best_seq)):
                pass  # already updated
            print(f"  Micro-polish final: {best_score:.8f}")

    # Final verification and return
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL RESULT: n={len(final_list)}, score={final_score:.8f}")
        print(f"{'='*60}")
        return final_list

    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
