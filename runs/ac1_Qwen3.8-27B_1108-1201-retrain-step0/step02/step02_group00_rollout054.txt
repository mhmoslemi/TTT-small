Minimize 2*n*max(self_conv)/sum^2 using L-BFGS with softmax parameterization.

Key innovations over previous approach:
1. Softmax parameterization ensures non-negativity and fixed sum without projection.
2. L-BFGS (quasi-Newton) converges much faster than FISTA for smooth objectives.
3. Segment-based initializations provide structured starting points.
4. Adaptive log-sum-exp beta for smooth-to-sharp transition.
5. Multiple restarts with diverse initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_value_grad(z, n, beta):
    """Compute log-sum-exp value and gradient w.r.t. z."""
    S = np.sqrt(2.0 * n)
    p = softmax(z)
    f = p * S
    c = fft_convolve_self(f)
    
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    lse_val = (c_max + np.log(total)) / beta
    
    w = exp_bc / total
    
    # cross-correlation: (w star f)[i] = sum_j w[i+j] * f[j]
    fsize = 1 << (len(w) + n - 1 - 1).bit_length()
    W = fft(w, fsize)
    F = fft(f, fsize)
    cross = np.real(ifft(W * np.conj(F)))[:n]
    
    grad_f = f + cross
    sum_gf_f = np.dot(grad_f, f)
    grad_z = f * (grad_f - sum_gf_f / S)
    
    return lse_val, grad_z


def run_lbfgs(z_init, n, beta_schedule, max_iter, time_limit, rng):
    """Run L-BFGS with adaptive beta."""
    start = time.monotonic()
    best_val = np.inf
    best_z = z_init.copy()
    
    # We'll run L-BFGS in stages with increasing beta
    stages = len(beta_schedule)
    time_per_stage = time_limit / stages
    
    for stage in range(stages):
        beta = beta_schedule[stage]
        
        def fun(z):
            val, grad = compute_lse_value_grad(z, n, beta)
            return val, grad
        
        if time.monotonic() - start > time_limit * (stage + 1) / stages:
            break
        
        stage_time = time.limit(time_per_stage - 0.1)  # Can't use this directly
        
        # Use scipy L-BFGS-B
        try:
            result = minimize(fun, z_init, method='L-BFGS-B', jac=True,
                             options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                      'ftol': 1e-12, 'gtol': 1e-8,
                                      'maxls': 50})
            z_init = result.x
            final_val, _ = compute_lse_value_grad(z_init, n, beta)
            if final_val < best_val:
                best_val = final_val
                best_z = z_init.copy()
        except Exception as e:
            print("  L-BFGS error: %s" % str(e))
            break
    
    return best_z, best_val

    """Expand segment parameters to a full sequence."""
    f = np.zeros(n)

        start = int(boundaries[i])
        end = int(boundaries[i+1])
        f[start:end] = heights[i]
    return f

"""
Minimize 2*n*max(self_conv)/sum^2 using L-BFGS with softmax parameterization.

Key improvements over previous FISTA approach:
1. Softmax parameterization: eliminates need for simplex projection, makes problem unconstrained.
2. L-BFGS (quasi-Newton): converges superlinearly vs. FISTA's O(1/k) rate.
3. Segment-structured initializations: provide diverse, informed starting points.
4. Multi-stage adaptive beta: smooth-to-sharp log-sum-exp for progressive refinement.
5. Final subgradient polish: sharpens the max using top-k active set.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_stable(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_value_grad(z, n, beta):
    """
    Compute log-sum-exp of self-convolution and its gradient w.r.t. z.
    f = softmax(z) * sqrt(2n)
    Returns (lse_value, gradient_vector).
    """
    S = np.sqrt(2.0 * n)
    p = softmax_stable(z)
    f = p * S
    c = fft_convolve_self(f)

    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    lse_val = (c_max + np.log(total)) / beta

    w = exp_bc / total

    # Cross-correlation: (w star f)[i] = sum_j w[i+j] * f[j]
    fsize = 1 << (len(w) + n - 1 - 1).bit_length()
    W = fft(w, fsize)
    F = fft(f, fsize)
    cross = np.real(ifft(W * np.conj(F)))[:n]

    grad_f = f + cross
    sum_gf_f = np.dot(grad_f, f)
    grad_z = f * (grad_f - sum_gf_f / S)

    return lse_val, grad_z


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def f_to_z(f, n):
    """Convert non-negative sequence to log-space (softmax parameterization)."""
    S = np.sqrt(2.0 * n)
    f = np.maximum(f, 1e-15)
    s = f.sum()
    if s < 1e-15:
        return np.zeros(n)
    f = f / s * S
    z = np.log(f / S)
    return z


def run_lbfgs_staged(z_init, n, beta_list, time_limit, max_iter_per_stage):
    """
    Run L-BFGS in stages with increasing beta.
    beta_list: list of beta values to try in sequence.
    time_limit: total time limit in seconds.
    """
    start = time.monotonic()
    z_current = z_init.copy()
    best_val = np.inf
    best_z = z_init.copy()

    n_stages = len(beta_list)
    time_per_stage = time_limit / n_stages

    for stage in range(n_stages):
        beta = beta_list[stage]
        remaining = time_limit - (time.monotonic() - start)
        if remaining < 1.0:
            break

        stage_start = time.monotonic()
        stop_events = []

        def time_callback(z):
            if time.monotonic() - stage_start > time_per_stage - 0.2:
                stop_events.append(1)
                return True
            return None

        def fun(z):
            return compute_lse_value_grad(z, n, beta)

        try:
            result = minimize(fun, z_current, method='L-BFGS-B', jac=True,
                             options={'maxiter': max_iter_per_stage,
                                      'maxfun': max_iter_per_stage * 5,
                                      'ftol': 1e-14, 'gtol': 1e-10,
                                      'maxls': 40},
                             callback=time_callback)
            z_current = result.x
        except Exception:
            pass

        # Evaluate at current point
        val, _ = compute_lse_value_grad(z_current, n, beta)
        if val < best_val:
            best_val = val
            best_z = z_current.copy()

        elapsed = time.monotonic() - start
        print("    Stage %d (beta=%.0f): lse=%.6f, time=%.1fs" %
              (stage + 1, beta, val, elapsed))

    return best_z, best_val


def expand_segments(n, k, boundaries_frac, heights):
    """
    Expand segment parameters to a full sequence.
    boundaries_frac: array of k-1 fractions in (0,1) for boundary positions.
    heights: array of k heights.
    """
    f = np.zeros(n)
    bounds = [0] + [int(b * n) for b in np.sort(boundaries_frac)] + [n]
    for i in range(k):
        start = bounds[i]
        end = bounds[i + 1]
        if end > start:
            f[start:end] = heights[i]
    return f


def subgradient_polish(f_init, n, time_limit, rng):
    """Nesterov-accelerated top-k subgradient for final polishing."""
    S = np.sqrt(2.0 * n)
    f = np.clip(f_init, 0, 1000.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) * S / n
    else:
        f = f * (S / s)

    y = f.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_f = f.copy()
    it = 0

    while time.monotonic() - start < time_limit:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        sum_y = y.sum()
        score = 2.0 * n * cur_max / (sum_y ** 2)
        if score < best_score:
            best_score = score
            best_f = y.copy()

        # Top-k subgradient
        top_k = min(512, 10 + it // 300)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = weights

        fsize = 1 << (len(w_full) + n - 1 - 1).bit_length()
        W = fft(w_full, fsize)
        F = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(F)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 2.0 / (1.0 + 0.01 * it)
        x_new = y - step * grad / gnorm
        x_new = project_simplex(x_new, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        else:
            y = x_new
        f = x_new
        t = t_new

        # Periodic noise to escape local minima
        if it % 5000 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0

    return best_f, best_score
