"""
Hybrid peak-equalization optimization for minimizing the self-convolution peak
of non-negative step functions. Uses direct subgradient descent with multi-peak
active set management, adaptive step sizes, and coordinate descent polishing.
"""

import numpy as np
import time

def fft_convolve_self(a):
    """Compute a*a (linear self-convolution) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_score(a):
    """Compute the evaluation score directly."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * np.max(c) / (S * S)

def gradient_of_conv_at_peak(a, peak_idx, n):
    """
    Compute gradient of (a*a)[peak_idx] with respect to a.
    (a*a)[k] = sum_j a_j * a_{k-j} for valid j.
    d/d a_i of (a*a)[k] = a_{k-i} + a_{i-k} (with bounds check)
    """
    g = np.zeros(n)
    k = peak_idx
    for i in range(n):
        # a_{k-i} is valid if 0 <= k-i < n
        if 0 <= k - i < n:
            g[i] += a[k - i]
        # a_{i-k} is valid if 0 <= i-k < n
        if 0 <= i - k < n:
            g[i] += a[i - k]
    return g

def optimize_direct(a_init, time_limit, rng, verbose=True):
    """
    Direct subgradient descent minimizing max(a*a) subject to sum(a)=const, a>=0.
    Uses multi-peak active set and adaptive step sizes.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S0 = np.sum(a)
    
    # Normalize to sum = sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / S0)
    
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score
    
    t0 = time.monotonic()
    step = 0
    lr = 0.01  # initial learning rate
    no_improve = 0
    patience = 500
    
    print(f"  Starting score: {best_score:.6f}")
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve_self(a)
        max_c = np.max(c)
        current_score = 2.0 * n * max_c / (np.sum(a))**2
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
            if verbose and step % 100 == 0:
                print(f"  step={step}, score={best_score:.6f}")
        else:
            no_improve += 1
        
        # Find active set: indices where c_k is within 5% of max
        threshold = max_c * 0.95
        active = np.where(c >= threshold)[0]
        
        # Compute weighted gradient (weights proportional to c_k)
        weights = c[active] / np.sum(c[active])
        g = np.zeros(n)
        for idx in active:
            g += weights[list(active).index(idx)] * gradient_of_conv_at_peak(a, idx, n)
        
        # Project gradient onto the constraint sum(a) = target_sum
        # The feasible direction is any v with sum(v) = 0
        g_mean = np.mean(g)
        g_proj = g - g_mean

        if no_improve % 100 == 0 and no_improve > 0:
            lr *= 0.95

        # Project back: clip to [0, inf) and renormalize

        else:

            a_new = np.clip(a_new, 0, None)
            s_new = np.sum(a_new)

                break

def make_structured_init(n, rng, method='broadband'):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)
    
    if method == 'broadband':
        a = np.ones(n)
        for k in range(1, 12):
            freq = k * (1 + 0.5 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.2 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    
    elif method == 'ramp':
        # Ramp: high at one end, low at other
        a = np.linspace(2, 0.5, n)
        a += 0.05 * np.sin(2 * np.pi * 3 * t / n)
        return np.clip(a, 0.01, None)
    
    elif method == 'two_ramp':
        # Two ramps: high-low-high
        a = 1.0 + 0.5 * np.cos(2 * np.pi * t / n)
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
