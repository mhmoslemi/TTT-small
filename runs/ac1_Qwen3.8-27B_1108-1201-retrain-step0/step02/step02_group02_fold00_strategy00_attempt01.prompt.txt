import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):

    S = np.sum(a)

        return np.inf

    return 2.0 * n * np.max(c) / (S * S)


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def topk_grad(a, tau):

    Compute soft-max value and gradient of the convolution with temperature tau.
    Smaller tau focuses more on the peak.
    Returns (value, gradient_wrt_a, convolution, weights).

    c = fft_convolve(a, a)

    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)

    # Gradient via FFT: g[j] = 2 * (a_reversed * w)[j + n - 1]
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1:2 * n - 1]

    return value, g, c, w

def coordinate_descent(a_init, time_limit, rng):
    """
    Coordinate descent polish: for each element, find optimal value via
    gradient-informed line search.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        improved_this_round = False

        # Randomize order for diversity
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            base_val = current[j]
            if base_val < 1e-6:
                continue

            # Compute gradient direction for this coordinate
            # Use a small perturbation to estimate the 1D landscape
            # Try geometric sequence of values
            for frac in [0.6, 0.8, 0.9, 1.0, 1.1, 1.25, 1.5]:
                trial = base_val * frac
                current[j] = trial
                new_score = local_score(current)

                if new_score < current_score:
                    current_score = new_score
                    improved_this_round = True
                    if current_score < best_score:
                        best_score = current_score
                        best = current.copy()
                else:
                    current[j] = base_val
                    break

        # If no improvement in a full pass, try a random perturbation
        if not improved_this_round:
            j = int(rng.integers(0, n))
            current[j] = max(0.0, current[j] * np.exp(rng.normal(0, 0.15)))
            new_score = local_score(current)
            if new_score < current_score * 1.005:
                current_score = new_score

        iteration += 1
        if iteration % 20 == 0:
            print(f"    CD iter={iteration}, cur={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")

    return best, best_score


def peak_shave_polish(a_init, time_limit, rng):
    """
    Peak-directed polishing: identify peak contributors and perform
    targeted mass redistribution.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.001
    cooling = 0.99995

    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))

        # Compute contribution of each element to the peak
        contribs = np.zeros(n)
        for i in range(n):
            if 0 <= peak_idx - i < n:
                contribs[i] = current[i] * current[peak_idx - i]

        # Strategy: reduce top contributors, increase low contributors
        r = rng.random()

        if r < 0.5:
            # Reduce top contributors, redistribute to bottom
            top_k = max(4, n // 16)
            bottom_k = max(4, n // 8)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            bottom_idx = np.argsort(contribs)[:bottom_k]

            reduction = rng.uniform(0.02, 0.06)
            current[top_idx] *= (1.0 - reduction)
            current[bottom_idx] *= (1.0 + reduction * len(top_idx) / len(bottom_idx))

        elif r < 0.7:
            # Pair swap: reduce a top contributor, increase a random other
            top_k = max(4, n // 12)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            j_red = top_idx[int(rng.integers(0, top_k))]
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red:
                amt = current[j_red] * rng.uniform(0.02, 0.05)
                current[j_red] -= amt
                current[j_inc] += amt
                current[j_red] = max(0.0, current[j_red])

        elif r < 0.85:
            # Scale a random block
            width = max(2, int(rng.integers(2, n // 6)))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.03))
            current[start:start + width] *= factor

        else:
            # Small global perturbation
            current *= np.exp(rng.normal(0, 0.005, size=n))

        current = np.clip(current, 0.0, 1000.0)
        new_score = local_score(current)

        if new_score < current_score:
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current_score = new_score

        temp *= cooling
        if step % 1000 == 0:
            print(f"    PS step={step}, cur={current_score:.6f}, best={best_score:.6f}, temp={temp:.5f}")

    return best, best_score


def make_structured_init(n, rng, method='broadband'):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)

    if method == 'broadband':
        # Broadband: sum of many incommensurate frequencies
        a = np.ones(n)
        for k in range(1, 15):
            freq = k * (1 + 0.7 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.15 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'two_block':
        # Two-block: high values in two separated regions
        a = np.ones(n)
        block1 = slice(0, n // 3)
        block2 = slice(2 * n // 3, n)
        a[block1] = 1.5
        a[block2] = 1.5
        a[n // 3:2 * n // 3] = 0.5
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'periodic':
        # Based on a simple periodic pattern with low autocorrelation
        period = max(3, int(np.sqrt(n)))
        pattern = np.ones(period)
        pattern[0] = 1.5
        pattern[period // 2] = 0.5
        a = np.tile(pattern, n // period + 1)[:n]
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
