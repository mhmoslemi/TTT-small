def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def mirror_descent_quick(a0, time_limit):
    """Quick mirror descent for initialization."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    t_smooth = 30.0
    eta = 0.3
    t0 = time.monotonic()
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        # Softmax subgradient
        scaled = t_smooth * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        w /= w_sum
        
        # Subgradient
        sg = np.zeros(n)
        for k in range(len(conv)):
            if w[k] > 1e-4:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                sg[js:je+1] += w[k] * a[k-je:k-js+1][::-1]
        
        sg_c = sg - sg.mean()
        a = a * np.exp(-eta * sg_c)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = best_a.copy()
        else:
            a /= s
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 30 == 0:
                eta *= 0.7
                t_smooth *= 1.3
    
    return best_a, best_obj

"""
LP-guided optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find safe directions that avoid increasing the
auto-convolution peak, combined with adaptive line search, smart perturbations,
peak-shaving local search, and multi-scale exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]


def obj_val(a):
    """Objective value assuming a is non-negative with positive sum."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def safe_evaluate(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        if not np.all(np.isfinite(a)):
            return float('inf')
        return float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')


def build_lp_matrix(a):
    """Build sparse constraint matrix for conv(a,b)(k) <= M."""
    n = len(a)
    n_row = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(n_row):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(n_row, n))
    return A


def solve_direction_lp(a, M, A=None):
    """
    Solve LP: maximize sum(b) s.t. b >= 0, conv(a,b) <= M.
    Returns normalized b (sum=1) or None.
    """
    n = len(a)
    c = -np.ones(n)
    
    if A is None:
        A = build_lp_matrix(a)
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 5.0})
    
    if result.success and result.x is not None:
        b = np.maximum(result.x, 0.0)
        s = b.sum()
        if s > 1e-10:
            return b / s
    return None

    LP-guided optimization with adaptive line search and smart restarts.
    """

    a = np.maximum(a0, 0.0).copy()

    s = a.sum()
    if s < 1e-12:

    else:
        a /= s
    
    conv = fast_convolve(a)
    M = conv.max()
    cur_obj = 2.0 * n * M
    best_obj = cur_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    lp_reuse_count = 0

            else:

                found = True
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()

                t_step *= 0.5

            stall += 1

    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    stall = 0
    T = 0.02
    
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = 0.02 * (1.0 - frac)**2 + 0.001
        
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # Find peak positions (top few)
        peak_thresh = max_c * 0.9
        peaks = np.where(conv >= peak_thresh)[0]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(conv)])
        
        # Pick a peak to attack
        k = int(peaks[rng.integers(len(peaks))])
        
        # Find contributing pairs
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        contributions = a[js] * a[k - js]
        total = contributions.sum()
        if total < 1e-15:
            continue
        
        # Pick the pair with highest contribution
        probs = contributions / total
        j_src = js[rng.choice(len(js), p=probs)]
        j_dst_val = k - j_src
        
        # Choose destination: minimize contribution to all peaks
        best_dest = -1
        best_score = float('inf')
        candidates = rng.choice(n, size=min(20, n), replace=False)
        for cand in candidates:
            if cand == j_src:
                continue
            # Estimate contribution to peaks if we add mass here
            contrib = 0.0
            for pk in peaks[:5]:
                d = pk - cand
                if 0 <= d < n:
                    contrib += a[d]
                d2 = cand - pk + 2 * n - 2
                if 0 <= d2 < n:
                    contrib += a[d2]
            if contrib < best_score:
                best_score = contrib
                best_dest = int(cand)
        
        if best_dest < 0 or best_dest == j_src:
            continue
        
        # Move some mass
        delta = a[j_src] * rng.uniform(0.02, 0.1)
        if delta < 1e-10 or delta > a[j_src]:
            continue
        
        a[j_src] -= delta
        a[best_dest] += delta
        
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
                stall += 1
            else:
                a[j_src] += delta
                a[best_dest] -= delta
                stall += 1
        
        if stall > 500:
            a = best_a.copy()
            stall = 0
    
    return best_a, best_obj


def mirror_descent_quick(a0, time_limit, seed=42):
    """Fast mirror descent for initial optimization."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    t_smooth = 20.0
    eta = 0.4
    t0 = time.monotonic()
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # Softmax-weighted subgradient
        scaled = t_smooth * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient
        sg = np.zeros(n)
        active = np.where(w > 1e-3)[0]
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je+1] += w[k] * a[k-je:k-js+1][::-1]
        
        sg_c = sg - sg.mean()
        a = a * np.exp(-eta * sg_c)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = best_a.copy()
        else:
            a /= s
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 20 == 0:
                eta *= 0.75
                t_smooth *= 1.2
                if eta < 0.01:
                    break
    
    return best_a, best_obj


def get_parent():
    """Access parent sequence from global variable."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Multi-phase LP-guided optimizer.
    
    Phase 1: Quick exploration across multiple n values and initializations.
    Phase 2: Deep LP-guided optimization on top candidates.
    Phase 3: Peak-shaving local search and fine-tuning.
    Phase 4: Final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    
    parent = get_parent()
    
    best_sequence = None
    best_score = float('inf')
    
    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d" % (score, len(a)))
    
    # Phase 1: Quick exploration
    print("Phase 1: Quick exploration...")
    t_phase1_start = time.monotonic()
    
    lengths = [128, 192, 256, 320, 384, 512, 768]
    
    # Include parent length
    if parent is not None:
        p_len = len(parent)
        if p_len not in lengths:
            lengths.insert(0, p_len)
    
    candidates = []
    
    # Start with parent if available
    if parent is not None:
        p = np.maximum(parent, 0.0).copy()
        p_len = len(p)
        p /= p.sum()
        a_opt, obj = mirror_descent_quick(p, time_limit=40, seed=seed)
        candidates.append((obj, p_len, 'parent_md', a_opt))
        record(a_opt)
        candidates.append((obj, p_len, 'parent', p))
    
    phase1_budget = min(280.0, (deadline - time.monotonic()) * 0.14)
    
    for n in lengths:
        if time.monotonic() > deadline - 1900:
            break
        
        inits = {}
        x = np.arange(n, dtype=np.float64)
        inits['uniform'] = np.ones(n) / n
        
        g = np.exp(-0.5 * ((x - 0.5*n) / (0.15*n))**2)
        inits['gauss'] = g / g.sum()
        
        # Multi-frequency
        s1 = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
        s2 = 0.5 * (1.0 + np.cos(4.0 * np.pi * x / n + 1.0))
        combo = (s1 * s2)
        combo = np.maximum(combo, 0.0)
        if combo.sum() > 1e-10:
            inits['multi_freq'] = combo / combo.sum()
        
        # Chirp
        chirp = 0.5 * (1.0 + np.cos(2.0 * np.pi * x**2 / (3.0 * n)))
        chirp = np.maximum(chirp, 0.0)
        if chirp.sum() > 1e-10:
            inits['chirp'] = chirp / chirp.sum()
        
        # Random
        r = rng.dirichlet(np.ones(n) * 0.4)
        inits['rand'] = r
        
        # Bimodal
        bi = np.zeros(n)
        w = max(1, n // 5)
        bi[:w] = 1.0
        bi[-w:] = 1.0
        bi /= bi.sum()
        inits['bimodal'] = bi
        
        mid = n // 2
        n_configs = len(inits)
        time_per = max(2.0, phase1_budget / max(1, len(lengths) * n_configs))
        
        for name, a0 in inits.items():
            if time.monotonic() > deadline - 1900:
                break
            t_alloc = min(time_per, deadline - time.monotonic() - 1900)
            if t_alloc < 1.5:
                break
            a_opt, obj = mirror_descent_quick(a0, time_limit=t_alloc, seed=seed + hash(name + str(n)))
            candidates.append((obj, n, name, a_opt))
            # Also record without MD (sometimes random starts are good for LP)
            candidates.append((obj, n, name + '_raw', a0))
    
    candidates.sort(key=lambda x: x[0])
    print("  Phase 1 done (%.0fs). Best: %.6f" % (time.monotonic() - t_phase1_start,
                                                   candidates[0][0] if candidates else float('inf')))
    
    # Phase 2: LP-guided deep optimization
    print("Phase 2: LP-guided optimization...")
    t_phase2_start = time.monotonic()
    
    top_candidates = []
    seen = set()
    for obj, n, name, a_opt in candidates:
        key = (n, name)
        if key not in seen and len(top_candidates) < 10:
            top_candidates.append((n, name, a_opt.copy()))
            seen.add(key)
    
    # Ensure best is included
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), 'global_best', best_sequence.copy()))
    
    phase2_budget = min(1400.0, (deadline - time.monotonic()) * 0.55)
    n_cands = len(top_candidates)
    time_per_cand = max(10.0, phase2_budget / max(1, n_cands))
    
    for idx, (n, name, a0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 900:
            break
        t_alloc = min(time_per_cand, deadline - time.monotonic() - 900)
        if t_alloc < 5.0:
            break
        
        print("  [%d/%d] n=%d, %s, budget=%.0fs" % (idx+1, n_cands, n, name, t_alloc))
        
        # First: LP-guided
        a_opt, obj = lp_guided_optimize(a0, time_limit=t_alloc * 0.6, seed=seed + idx)
        record(a_opt)
        
        # Second: Peak-shaving
        if time.monotonic() < deadline - 900:
            t2 = min(t_alloc * 0.4, deadline - time.monotonic() - 900)
            if t2 > 5:
                a_opt2, obj2 = peak_shave_local_search(a_opt, time_limit=t2, seed=seed + 100 + idx)
                record(a_opt2)
                # Third: LP again on refined
                if obj2 < obj and time.monotonic() < deadline - 900:
                    t3 = (deadline - time.monotonic() - 900) * 0.2
                    if t3 > 5:
                        a_opt3, obj3 = lp_guided_optimize(a_opt2, time_limit=t3, seed=seed + 200 + idx)
                        record(a_opt3)
    
    print("  Phase 2 done (%.0fs). Best: %.6f" % (time.monotonic() - t_phase2_start, best_score))
    
    # Phase 3: Intensive local search on best
    print("Phase 3: Intensive local search...")
    t_phase3_start = time.monotonic()
    
    if best_sequence is not None and time.monotonic() < deadline - 50:
        remaining = deadline - time.monotonic() - 50
        a_best = best_sequence.copy()
        
        # Alternating LP and peak-shaving
        n_rounds = 6
        time_per_round = remaining / n_rounds
        
        for rnd in range(n_rounds):
            if time.monotonic() > deadline - 50:
                break
            
            if rnd % 2 == 0:
                a_opt, obj = lp_guided_optimize(a_best, time_limit=time_per_round,
                                                seed=seed + 5000 + rnd)
            else:
                a_opt, obj = peak_shave_local_search(a_best, time_limit=time_per_round,
                                                     seed=seed + 6000 + rnd)
            record(a_opt)
            if best_sequence is not None:
                a_best = best_sequence.copy()
        
        # Try a few random restarts from best with perturbation
        if time.monotonic() < deadline - 50:
            rem = deadline - time.monotonic() - 50
            n_restarts = 3
            t_restart = rem / n_restarts
            for i in range(n_restarts):
                if time.monotonic() > deadline - 50:
                    break
                a_pert = best_sequence.copy()
                n_pert = max(3, len(a_pert) // 8)
                idx_pert = rng.choice(len(a_pert), size=n_pert, replace=False)
                a_pert[idx_pert] *= rng.uniform(0.3, 2.5, size=n_pert)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert /= a_pert.sum()
                
                a_opt, obj = lp_guided_optimize(a_pert, time_limit=t_restart,
                                                seed=seed + 7000 + i)
                record(a_opt)
    
    print("  Phase 3 done (%.0fs). Best: %.6f" % (time.monotonic() - t_phase3_start, best_score))
    
    # Phase 4: Final polish
    print("Phase 4: Final polish...")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            # Very fine LP
            half = remaining * 0.5
            a_opt, obj = lp_guided_optimize(best_sequence, time_limit=half,
                                            seed=seed + 9999)
            record(a_opt)
            
            if time.monotonic() < deadline - 5:
                rem2 = deadline - time.monotonic() - 5
                a_opt2, obj2 = peak_shave_local_search(best_sequence, time_limit=rem2,
                                                       seed=seed + 9998)
                record(a_opt2)
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = safe_evaluate(best_sequence)
    
    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))
    
    return [float(x) for x in best_sequence]
