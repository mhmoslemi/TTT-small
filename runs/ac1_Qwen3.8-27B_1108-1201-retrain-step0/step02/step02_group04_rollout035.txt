```python
def optimize_lp_iteration(a, M, rng):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    # This is a linear program
    # Variables: b[0], ..., b[n-1] >= 0
    # Constraints: sum_i a[i]*b[k-i] <= M for k = 0, ..., 2n-2
    #             (where a[j] = 0 for j < 0 or j >= n)
    # Objective: maximize sum(b)
    n = len(a)
    # Use scipy.optimize.linprog
    # Maximize sum(b) = -minimize -sum(b)
    c = -np.ones(n)
    # Constraints: conv(a,b)[k] <= M
    # conv(a,b)[k] = sum_{i} a[i] * b[k-i] for valid i and k-i
    # For a given k, the constraint is: sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * b[k-i] <= M
    # Let j = k-i, so j ranges from max(0, k-(n-1)) to min(n-1, k)
    # The constraint becomes: sum_j a[k-j] * b[j] <= M
    
    # Build constraint matrix A_ub where A_ub @ b <= M * ones
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n))
    for k in range(n_constraints):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = a[i]
    
    b_ub = np.full(n_constraints, M)
    
    from scipy.optimize import linprog
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if result.success:
        return result.x
    return None
```

```python
"""
Optimization using iterative LP-based best response for non-negative step functions.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 by iteratively solving LPs to find
constrained best responses, with blending for convergence.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S ** 2)

def build_constraint_matrix(a):
    """Build A_ub matrix for the LP: A_ub[k,j] = a[k-j] if valid."""
    n = len(a)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        # a[k-j] for j in [j_min, j_max]
        # k-j ranges from k-j_max to k-j_min
        A_ub[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A_ub

def solve_lp_best_response(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    c = -np.ones(n)
    A_ub = build_constraint_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, 
                     bounds=[(0, None)] * n, method='highs',
                     options={'time_limit': 30})
    if result.success:
        return result.x
    return None

def lp_iteration(a, t, rng):
    """One iteration of the LP-based optimization."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return a, compute_score(a)
    
    # Normalize to sum 1
    a_norm = a / S
    conv = fft_convolve(a_norm, a_norm)
    M = np.max(conv)
    
    # Solve LP
    b = solve_lp_best_response(a_norm, M)
    if b is None:
        return a, compute_score(a)
    
    # Normalize b to sum 1
    S_b = np.sum(b)
    if S_b < 1e-10:
        return a, compute_score(a)
    b_norm = b / S_b
    
    # Blend
    a_new = (1 - t) * a_norm + t * b_norm
    a_new = np.clip(a_new, 0.0, None)
    
    # Renormalize
    S_new = np.sum(a_new)
    if S_new < 1e-10:
        return a, compute_score(a)
    a_new = a_new / S_new
    
    return a_new, compute_score(a)
```

```python
"""
Optimization of non-negative step functions to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based iterative best-response, coordinate descent, and soft-max gradients.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score for a sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S ** 2)


def build_A_ub(n):
    """Build the constraint matrix structure (indices only) for LP."""
    # For constraint k, column j is nonzero iff 0 <= k-j < n
    # A_ub[k, j] = a[k-j]
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_best_response(a, M, row_idx, col_idx):
    """Solve: max sum(b) s.t. b >= 0, A@b <= M*1 where A[k,j]=a[k-j]."""
    n = len(a)
    c = -np.ones(n)
    # Fill in the values of A
    vals = a[row_idx - col_idx]  # a[k-j]
    # Build sparse-like representation for HiGHS
    # Actually for moderate n, just build dense
    A_ub = np.zeros((2 * n - 1, n))
    A_ub[row_idx, col_idx] = vals
    b_ub = np.full(2 * n - 1, M)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub,
                     bounds=[(0, None)] * n, method='highs',
                     options={'time_limit': 20})
    if result.success:
        return result.x
    return None


def coordinate_descent(a, max_iter=50, tol=1e-10):
    """Polish with coordinate descent. Optimize one coordinate at a time."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return a, compute_score(a)
    a = a / S  # normalize to sum 1
    conv = fft_convolve(a, a)
    
    for it in range(max_iter):
        improved = False
        for i in range(n):
            # Current contribution of a[i] to conv: a[i]*a[k-i] at position k
            # If we change a[i] from v to v', conv[k] changes by (v'-v)*a[k-i]
            v = a[i]
            # The "gain vector": conv[k] += (v'-v) * a[k-i]
            # a[k-i] is nonzero when 0 <= k-i < n, i.e., i <= k <= i+n-1
            k_min = i
            k_max = i + n - 1
            gain = np.zeros(2 * n - 1)
            # For k in [i, i+n-1], a[k-i] is the gain coefficient
            gain[i:i + n] = a  # a[k-i] for k from i to i+n-1 is a[0], a[1], ..., a[n-1]
            
            # We want to minimize max(conv + (v'-v)*gain) over v' >= 0
            # This is a 1D minimax problem
            # The optimal v' equalizes the top values
            # Try: the max is achieved at some set of k's where conv[k] + (v'-v)*gain[k] = M_new
            # For k where gain[k] > 0: conv[k] + (v'-v)*gain[k] is increasing in v'
            # For k where gain[k] < 0: decreasing (but gain >= 0 here since a >= 0)
            # For k where gain[k] = 0: unchanged
            
            if v < 1e-15:
                # Currently zero, check if increasing helps
                # Try small increase
                best_v = v
                best_M = np.max(conv)
                for trial in [0.001 * S, 0.01 * S, 0.05 * S, 0.1 * S]:
                    new_conv = conv + (trial - v) * gain
                    new_M = np.max(new_conv)
                    if new_M < best_M:
                        best_M = new_M
                        best_v = trial
                if best_v != v:
                    a[i] = best_v
                    conv = fft_convolve(a, a)
                    improved = True
                continue
            
            # Binary search for optimal v'
            # At v'=v, M = max(conv)
            # As v' increases, entries with gain>0 increase, gain=0 stay same
            # As v' decreases (toward 0), entries with gain>0 decrease
            # Optimal is to reduce v' until the max is balanced
            
            lo, hi = 0.0, v * 2 + S
            for _ in range(40):
                mid = (lo + hi) / 2
                new_conv = conv + (mid - v) * gain
                M_new = np.max(new_conv)
                M_lo = np.max(conv + (lo - v) * gain)
                M_hi = np.max(conv + (hi - v) * gain)
                if M_lo < M_hi:
                    # Reducing v helps, go lower
                    hi = mid
                else:
                    lo = mid
            v_new = (lo + hi) / 2
            
            if abs(v_new - v) > tol * S:
                a[i] = v_new
                conv = fft_convolve(a, a)
                improved = True
        
        if not improved:
            break
    
    S = np.sum(a)
    a = a / S
    return a, compute_score(a)


def softmax_gradient_step(a, epoch, lr=None):
    """One step of gradient descent using soft-max approximation of max."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return a, compute_score(a)
    a_norm = a / S
    conv = fft_convolve(a_norm, a_norm)
    
    # Soft-max: max_k conv[k] ≈ (1/eps) * log(sum(exp(eps*conv)))
    # eps decreases over iterations for sharpness
    eps = max(0.5, 20.0 / (epoch + 1))
    # Scale to avoid overflow
    conv_shifted = conv - np.max(conv)
    log_sum_exp = np.max(conv) + np.log(np.sum(np.exp(eps * conv_shifted))) / eps
    # Gradient of log_sum_exp w.r.t. a[i]:
    # d/d a[i] of conv[k] = a[k-i] (for valid k)
    # So grad_soft[i] = eps * sum_k (exp(eps*conv[k]) / sum(exp(eps*conv))) * a[k-i]
    weights = np.exp(eps * conv_shifted)
    weights = weights / np.sum(weights)
    
    # grad[i] = eps * sum_k weights[k] * a[k-i] for valid k
    # This is a convolution of weights with a, evaluated at... let's compute directly
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += eps * weights[k] * a[j]
    
    # Gradient of score = 2n * max(conv) / S^2 w.r.t. a (with S fixed since we normalize)
    # Actually, we're working with a_norm (sum=1), so score = 2n * max(conv(a_norm, a_norm))
    # Gradient w.r.t. a_norm[i] is just the gradient of max(conv) w.r.t. a_norm[i]
    
    if lr is None:
        lr = 1.0 / (epoch + 10)
    
    a_new = a_norm - lr * grad
    a_new = np.clip(a_new, 0.0, None)
    S_new = np.sum(a_new)
    if S_new < 1e-10:
        return a, compute_score(a)
    a_new = a_new / S_new
    return a_new, compute_score(a_new)


def lp_optimize(a_init, deadline, rng, verbose=True):
    """Main LP-based optimization loop."""
    n = len(a_init)
    S = np.sum(a_init)
    if S < 0.01:
        return a_init, compute_score(a_init)
    a = a_init / S
    current_score = compute_score(a)
    best = a.copy()
    best_score = current_score
    
    # Precompute constraint matrix indices
    row_idx, col_idx = build_A_ub(n)
    
    t = 0.5
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        conv = fft_convolve(a, a)
        M = np.max(conv)
        
        # Solve LP
        b = solve_lp_best_response(a, M, row_idx, col_idx)
        
        if b is not None and np.sum(b) > 1e-10:
            b_norm = b / np.sum(b)
            # Adaptive blending
            a_new = (1 - t) * a + t * b_norm
            a_new = np.clip(a_new, 0.0, None)
            S_new = np.sum(a_new)
            if S_new > 1e-10:
                a_new = a_new / S_new
                new_score = compute_score(a_new)
                
                if new_score < current_score - 1e-12:
                    a = a_new
                    current_score = new_score
                    no_improve = 0
                    if current_score < best_score:
                        best_score = current_score
                        best = a.copy()
                        if verbose and it % 5 == 0:
                            print(f"    LP it={it}, score={current_score:.8f} (improved)")
                else:
                    no_improve += 1
                    if no_improve % 5 == 0:
                        t = max(0.05, t * 0.8)
                        # Try smaller t
                        a_try = (1 - t) * a + t * b_norm
                        a_try = np.clip(a_try, 0.0, None)
                        S_try = np.sum(a_try)
                        if S_try > 1e-10:
                            a_try = a_try / S_try
                            s_try = compute_score(a_try)
                            if s_try < current_score:
                                a = a_try
                                current_score = s_try
                                no_improve = 0
                                if current_score < best_score:
                                    best_score = current_score
                                    best = a.copy()
        else:
            no_improve += 1
            t = max(0.05, t * 0.8)
        
        # Occasional coordinate descent polish
        if it % 20 == 0 and time.monotonic() < deadline:
            a_cd, score_cd = coordinate_descent(a, max_iter=20)
            if score_cd < current_score:
                a = a_cd
                current_score = score_cd
                if current_score < best_score:
                    best_score = current_score
                    best = a.copy()
                    if verbose:
                        print(f"    CD polish it={it}, score={current_score:.8f}")
        
        # Add small noise to escape local minima occasionally
        if no_improve > 30:
            noise = 0.01 * rng.standard_normal(n)
            a_noisy = a + noise
            a_noisy = np.clip(a_noisy, 0.0, None)
            S_n = np.sum(a_noisy)
            if S_n > 1e-10:
                a_noisy = a_noisy / S_n
                s_n = compute_score(a_noisy)
                if s_n < current_score:
                    a = a_noisy
                    current_score = s_n
                    no_improve = 0
            no_improve = 0  # Reset to give noise a chance
        
        if verbose and it % 10 == 0:
            print(f"    LP it={it}, t={t:.3f}, score={current_score:.8f}, best={best_score:.8f}")
    
    return best, best_score


def softmax_optimize(a_init, deadline, rng, verbose=True):
    """Soft-max gradient descent optimization."""
    n = len(a_init)
    S = np.sum(a_init)
    if S < 0.01:

        print(f"  Result: score={opt_score:.8f}")
        
        if opt_score < best_score:
            best_score = opt_score
            best_seq = opt_seq.copy()
            print(f"  ** NEW BEST: {best_score:.8f} **")
    
    # Phase 3: Soft-max gradient on best found
    print(f"\n=== Phase 3: Soft-max refinement (best={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        sm_deadline = min(time.monotonic() + 120, deadline - 60)
        sm_seq, sm_score = softmax_optimize(best_seq, sm_deadline, rng)
        if sm_score < best_score:
            best_score = sm_score
            best_seq = sm_seq.copy()
            print(f"  ** NEW BEST: {best_score:.8f} **")
    
    # Phase 4: Coordinate descent polish on best
    print(f"\n=== Phase 4: Final CD polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        cd_seq, cd_score = coordinate_descent(best_seq, max_iter=200)
        if cd_score < best_score:
            best_score = cd_score
            best_seq = cd_seq.copy()
            print(f"  ** NEW BEST: {best_score:.8f} **")
    
    # Phase 5: Try to improve with more iterations if time remains
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Phase 5: Extended LP optimization (time_left={deadline-time.monotonic():.0f}s) ===")
        ext_deadline = deadline - 30
        ext_seq, ext_score = lp_optimize(best_seq, ext_deadline, rng, verbose=True)
        if ext_score < best_score:
            best_score = ext_score
            best_seq = ext_seq.copy()
            print(f"  ** NEW BEST: {best_score:.8f} **")
        # Final CD
        cd_seq2, cd_score2 = coordinate_descent(best_seq, max_iter=100)
        if cd_score2 < best_score:
            best_score = cd_score2
            best_seq = cd_seq2.copy()
    
    # Final answer
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = 2.0
    
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(final_list)
    print(f"\n=== FINAL: score={final_score:.8f}, n={len(final_list)} ===")
    return final_list
```
