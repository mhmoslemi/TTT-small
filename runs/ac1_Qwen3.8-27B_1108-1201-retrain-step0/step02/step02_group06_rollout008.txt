import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
    """Fast score: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)

def solve_lp_peak(f, M, method='highs'):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns the optimal g.
    """
    n = len(f)
    # Constraint matrix: A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # This is a convolution matrix
    # Build it efficiently
    # Row k has non-zeros at columns j where 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(k, n-1)
    
    # For large n, use sparse
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.ones(2 * n - 1) * M
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method=method)
    
    if result.success:
        return result.x
    else:
        return None

def lp_iteration(f, M, t, method='highs'):

    g = solve_lp_peak(f, M, method)
    if g is None:
        return f
    g_sum = np.sum(g)
    if g_sum < 1e-10:
        return f
    g = g / g_sum
    f_new = (1 - t) * f + t * g
    f_new = np.clip(f_new, 0, None)
    f_new = f_new / np.sum(f_new)
    return f_new

            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.02, 0.2)
            amt = a[i] * frac
            if a[i] - amt < 0:
                continue
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
            candidate = candidate / np.sum(candidate)
            
        elif r < 0.55:
            # Single element scale
            j = int(rng.integers(n))
            factor = np.exp(rng.normal(0, 0.1))
            candidate = a.copy()
            candidate[j] *= factor
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-15:
                continue
            candidate = candidate / cs
            
        elif r < 0.75:
            # Peak-targeted: reduce contributors to peak lag
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Find contributing pairs for the peak lag
            n_good = min(n, peak_idx + 1)
            contribs = np.zeros(n)
            for i in range(n_good):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
                    if j != i:
                        contribs[j] = a[j] * a[i]
            top_k = max(3, n // 16)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate = a.copy()
            reductions = rng.uniform(0.05, 0.2, size=len(top_idx))
            for idx, red in zip(top_idx, reductions):
                candidate[idx] = a[idx] * (1 - red)
            # Redistribute to random safe positions
            removed = np.sum(a[top_idx] - candidate[top_idx])
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0 and removed > 0:
                n_add = min(10, len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                candidate[add_idx] += removed / n_add
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-15:
                continue
            candidate = candidate / cs
            
        else:
            # Block perturbation
            width = max(2, int(rng.integers(2, max(3, n // 16))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.05, size=width))
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-15:
                continue
            candidate = candidate / cs
        
        new_conv = fft_convolve(candidate, candidate)
        new_max = np.max(new_conv)
        new_score = 2.0 * n * new_max
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            patience = 0
        else:
            patience += 1
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
    
    return best_a, best_score

def spectral_init(n, rng, n_iter=50):
    """Spectral iterative method for initialization."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    
    for it in range(n_iter):
        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        corr = np.fft.irfft(power, 2 * n)[:n]
        alpha = 1.0 / (n + 10.0)
        a = a - alpha * corr
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    
    return a

def cosine_init(n, rng, freq=1.0):
    """Cosine-modulated initialization."""
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)

def ramp_init(n, rng):
    """Linear ramp initialization."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = 1.0 + 0.5 * (2 * t - 1)  # from 0.5 to 1.5
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)
