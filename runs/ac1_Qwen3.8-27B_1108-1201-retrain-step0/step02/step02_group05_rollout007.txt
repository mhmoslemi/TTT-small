from scipy.fft import next_fast_len

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = next_fast_len(n + m - 1)
    fa = np.fft.rfft(a, size)

dc(0)/db = 0. Formula gives: 2*f(0-1) = 2*f(-1) = 0. ✓
dc(1)/da = 2b. Formula gives: 2*f(1-0) = 2*f(1) = 2b. ✓
dc(1)/db = 2a. Formula gives: 2*f(1-1) = 2*f(0) = 2a. ✓
dc(2)/da = 0. Formula gives: 2*f(2-0) = 2*f(2) = 0 (out of range). ✓
dc(2)/db = 2b. Formula gives: 2*f(2-1) = 2*f(1) = 2b. ✓

So the formula is: dc(k)/df(m) = 2*f(k-m) if 0 <= k-m < n, else 0.

This is simply 2 * (f reversed and padded) evaluated at position (k-m). Or equivalently, the gradient vector for lag k is 2 * flip(f) shifted.

More precisely, if we define the gradient vector grad_k (of length n) as:
grad_k[m] = 2 * f[k-m] if 0 <= k-m < n, else 0

This is: grad_k = 2 * ifft... no, it's just a shifted version of f.

For k=0: grad_0 = [2*f(0), 0, 0, ..., 0]
For k=1: grad_1 = [2*f(1), 2*f(0), 0, ..., 0]
For k=n-1: grad_{n-1} = [2*f(n-1), 2*f(n-2), ..., 2*f(0)]
For k=n: grad_n = [0, 2*f(n-1), 2*f(n-2), ..., 2*f(1)]
...
For k=2n-2: grad_{2n-2} = [0, 0, ..., 0, 2*f(n-1)]

So grad_k is a shifted and flipped version of 2*f.

Now, the subgradient of max_k c(k) is the average of the gradients at the active set (where c(k) = max). If the max is unique at k*, the subgradient is grad_{k*}.

So the steepest descent direction (negative subgradient) is:
- d[m] = -2*f(k*-m) if 0 <= k*-m < n, else 0

This means we should decrease f(m) the most where f(k*-m) is large, i.e., where the cross-correlation contribution to the peak is large.

For the normalized problem (sum(f) = 1), we need to project the descent direction onto the hyperplane sum = 0 (to maintain normalization) and onto the non-negative orthant.

The projected direction is:
d_proj[m] = d[m] - mean(d)  [to ensure sum = 0]
Then we need f + t * d_proj >= 0 for all m.

This gives us a simple gradient descent algorithm!

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = next_fast_len(n + m - 1)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score for sequence: 2n * max(conv(a,a)) / sum(a)^2."""

def solve_lp_direction(f, M, solver=cp.ECOS):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and (f*g)(k) <= M for all k.
    Returns g or None if infeasible.
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build constraint: for each k=0..2n-2, sum_i f[i]*g[k-i] <= M
    # where the sum is over i with 0<=i<n and 0<=k-i<n
    constraints = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_min > i_max:
            continue
        expr = 0.0
        for i in range(i_min, i_max + 1):
            expr += f[i] * g[k - i]
        constraints.append(expr <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None
    
    if g.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
        result = np.maximum(0, np.asarray(g.value).flatten())
        return result
    return None

    # = (sum c^p)^(1/p - 1) * c^(p-1)
    # Let S = sum c^p, then ||c||_p = S^(1/p)
    # d||c||_p/d a(m) = (1/p) S^(1/p-1) * p * sum_k c(k)^(p-1) * dc(k)/da(m)
    # = S^(1/p-1) * sum_k c(k)^(p-1) * dc(k)/da(m)
    
    S = np.sum(c_pos ** p)
    factor = S ** (1.0 / p - 1.0)
    
    cp = c_pos ** (p - 1)
    
    # grad_a[m] = factor * sum_k cp[k] * 2*a[k-m] (valid range)
    # Same cross-correlation structure as soft-max gradient
    L = next_fast_len(3 * n)
    CP = np.fft.rfft(cp, L)
    A = np.fft.rfft(a, L)
    cross = np.fft.irfft(CP * np.conj(A), L)
    grad_a = 2.0 * factor * cross[:n]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def run_lbgfs(theta, obj_func, sched_param, param_name, maxiter_per, deadline, verbose=True):
    """Run L-BFGS with annealed parameter."""
    best_theta = theta.copy()
    a_test = softmax(theta)
    best_score = local_score(a_test)
    
    for param in sched_param:
        if time.monotonic() > deadline - 10:
            break
        
        try:
            result = minimize(
                obj_func, theta.copy(),
                method='L-BFGS-B',
                jac=True, args=(param,),
                options={'maxiter': maxiter_per, 'ftol': 1e-15, 'gtol': 1e-12,
                         'maxfun': maxiter_per * 3}
            )
            theta_new = result.x
            a_new = softmax(theta_new)
            s = local_score(a_new)
            
            if s < best_score:
                best_score = s
                best_theta = theta_new.copy()
                if verbose:
                    print(f"      {param_name}={param}: score={s:.8f}")
            else:
                if verbose:
                    print(f"      {param_name}={param}: score={s:.8f} (no improvement)")
        except Exception as e:
            if verbose:
                print(f"      {param_name}={param}: error {e}")
            break
    
    return best_theta, best_score


def make_random_init(n, rng, base_mean=1.0, spread=0.3):
    """Random initialization around a base mean."""
    a = base_mean + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2 * np.pi * t * t / (4 * n))
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def subgradient_refine(f_init, time_limit, verbose=True):
    """Subgradient descent on the exact max objective."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = np.maximum(f, 0)
    f = f / np.sum(f)
    current_score = local_score(f)
    best = f.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    lr = 0.005
    patience = 300
    no_improve_count = 0
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fft_convolve(f, f)
        c_max = np.max(c)
        
        # Find active set (peaks within 0.1% of max)
        tol = 0.001 * c_max
        active = np.where(c >= c_max - tol)[0]
        
        # Compute subgradient (average of gradients at active peaks)
        grad = np.zeros(n)
        idx = np.arange(n)
        for k in active:
            valid = (k - idx >= 0) & (k - idx < n)
            grad[valid] += 2.0 * f[k - idx[valid]]
        grad /= len(active)
        
        # Project onto sum=0 (to maintain normalization)
        grad = grad - np.mean(grad)
        
        # Adaptive learning rate with backtracking
        found_step = False
        for t in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05, lr * 0.01]:
            new_f = f - t * grad
            if np.min(new_f) < 0:
                continue
            new_f = new_f / np.sum(new_f)
            s = local_score(new_f)
            if s < current_score:
                f = new_f
                current_score = s
                found_step = True
                break
        
        if found_step:
            if current_score < best_score - 1e-12:
                best_score = current_score
                best = f.copy()
                no_improve_count = 0
                # Increase lr slightly if we made good progress
                if lr < 0.02:
                    lr = min(0.02, lr * 1.05)
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            lr *= 0.7
        
        if no_improve_count >= patience:
            break
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SD iter {it}: score={current_score:.8f}, best={best_score:.8f}, lr={lr:.5f}, t={elapsed:.0f}s")
    
    return best, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimization: LP-based descent, soft-max L-BFGS,
    subgradient refinement. Explores the provided sequence and random inits.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 60.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Load starting sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-8, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        s = local_score(start_seq)
        print(f"Starting sequence: n={len(start_seq)}, score={s:.8f}")
        best_score = s
        best_seq = start_seq.copy()
    
    # ===== Phase 1: Quick L-BFGS exploration (300s) =====
    print(f"\n=== Phase 1: L-BFGS exploration (score={best_score:.6f}) ===")
    p1_deadline = t_global + 300.0
    
    lengths_to_try = [192, 256, 320]
    
    for n in lengths_to_try:
        if time.monotonic() > p1_deadline - 30:
            break
        
        p1_n_deadline = min(p1_deadline - 10, time.monotonic() + 90)
        
        inits = []
        # Perturbation of current best
        if best_seq is not None:
            if len(best_seq) == n:
                a_pert = best_seq * np.exp(rng.normal(0, 0.15, n))
                a_pert = np.maximum(a_pert, 1e-6)
                a_pert = a_pert / np.sum(a_pert)
                inits.append(('best_pert', a_pert))
            else:
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, n)
                a_interp = np.interp(x_new, x_old, best_seq)
                a_interp = np.maximum(a_interp, 1e-6)
                a_interp = a_interp / np.sum(a_interp)
                inits.append(('interp', a_interp))
        
        inits.append(('chirp1', make_chirp_init(n, 0.4, rng)))
        inits.append(('chirp2', make_chirp_init(n, -0.3, rng)))
        inits.append(('random', make_random_init(n, rng, spread=0.4)))
        
        for init_name, a0 in inits:
            if time.monotonic() > p1_n_deadline:
                break
            
            theta = np.log(a0 + 1e-30)
            p_schedule = [8, 16, 32, 64]
            theta_best, s = run_lbgfs(theta, pnorm_obj_grad, p_schedule, 'p', 
                                       maxiter_per=150, deadline=p1_n_deadline, verbose=False)
            
            if s < best_score:
                best_score = s
                best_seq = softmax(theta_best).copy()
                print(f"  n={n}, '{init_name}': {s:.8f} *** NEW BEST ***")
            else:
                print(f"  n={n}, '{init_name}': {s:.8f}")
    
    print(f"  After Phase 1: best={best_score:.8f}, n={len(best_seq)}")
    
    # ===== Phase 2: LP-based descent (1200s) =====
    print(f"\n=== Phase 2: LP-based descent (score={best_score:.6f}) ===")
    
    if best_seq is not None:
        n = len(best_seq)
        p2_time = min(1200.0, max(300.0, deadline - time.monotonic() - 400.0))
        print(f"  LP descent time budget: {p2_time:.0f}s, n={n}")
        
        if n <= 320:  # LP is feasible for reasonable sizes
            lp_best, lp_score = lp_descent(best_seq, p2_time, verbose=True)
            
            if lp_score < best_score:
                best_score = lp_score
                best_seq = lp_best.copy()
                print(f"  After LP: {best_score:.8f} *** IMPROVED ***")
            else:
                print(f"  After LP: {lp_score:.8f} (no improvement)")
        else:
            print(f"  Skipping LP for n={n} (too large)")
    
    # ===== Phase 3: Soft-max L-BFGS fine-tuning (400s) =====
    print(f"\n=== Phase 3: Soft-max L-BFGS (score={best_score:.6f}) ===")
    
    if best_seq is not None:
        n = len(best_seq)
        p3_deadline = min(deadline - 200, time.monotonic() + 400.0)
        
        # Multiple tau values for fine-tuning
        tau_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        theta = np.log(best_seq + 1e-30)
        
        theta_best, s = run_lbgfs(theta, soft_max_obj_grad_fast, tau_schedule, 'tau',
                                   maxiter_per=500, deadline=p3_deadline, verbose=True)
        
        if s < best_score:
            best_score = s
            best_seq = softmax(theta_best).copy()
            print(f"  Soft-max improved: {best_score:.8f} *** NEW BEST ***")
        
        # Try another round with smaller tau
        if time.monotonic() < p3_deadline - 30:
            theta = np.log(best_seq + 1e-30)
            tau_schedule2 = [0.0003, 0.0001, 0.00005, 0.00002]
            theta_best2, s2 = run_lbgfs(theta, soft_max_obj_grad_fast, tau_schedule2, 'tau',
                                         maxiter_per=500, deadline=p3_deadline, verbose=True)
            if s2 < best_score:
                best_score = s2
                best_seq = softmax(theta_best2).copy()
                print(f"  Soft-max round 2: {best_score:.8f} *** NEW BEST ***")
    
    # ===== Phase 4: Subgradient refinement (remaining time) =====
    print(f"\n=== Phase 4: Subgradient refinement (score={best_score:.6f}) ===")
    
    if best_seq is not None:
        remaining = max(60.0, deadline - time.monotonic() - 20.0)
        print(f"  Subgradient time: {remaining:.0f}s")
        
        sd_best, sd_score = subgradient_refine(best_seq, remaining, verbose=True)
        
        if sd_score < best_score:
            best_score = sd_score
            best_seq = sd_best.copy()
            print(f"  After subgradient: {best_score:.8f} *** NEW BEST ***")
    
    # ===== Phase 5: One more LP round if time permits =====
    if time.monotonic() < deadline - 100 and best_seq is not None and len(best_seq) <= 320:
        print(f"\n=== Phase 5: Final LP polish ===")
        final_lp_time = min(120.0, deadline - time.monotonic() - 20.0)
        lp_best2, lp_score2 = lp_descent(best_seq, final_lp_time, verbose=True)
        if lp_score2 < best_score:
            best_score = lp_score2
            best_seq = lp_best2.copy()
            print(f"  Final LP: {best_score:.8f} *** NEW BEST ***")
    
    # ===== Final verification =====
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.8f}, n={len(final_list)} ===")
        print(f"    (local_score was {best_score:.8f})")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
