import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
from scipy.fft import fft, ifft

def fft_convolve_self(a):

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_convolve(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]

def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def build_lp_matrix(f, n, selected_lags):
    """Build sparse matrix A where A[idx, j] = f[k-j] if 0<=k-j<n, for k=selected_lags[idx]."""
    rows = []
    cols = []
    data = []
    for idx, k in enumerate(selected_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)

            cols.extend(js[mask].tolist())
            data.extend(vals[mask].tolist())

def solve_lp(f, M, n, time_limit=10, num_constr=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for selected k, g >= 0."""
    c_self = fft_convolve_self(f)

        sorted_idx = np.argsort(c_self)[::-1]
        selected_lags = sorted_idx[:num_constr]

    c_obj = -np.ones(n)
    b = np.full(len(selected_lags), M)

                        bounds=[(0, None)] * n,

            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum
    except Exception as e:
        print("LP error: %s" % str(e))
    return None, None

def line_search(a, g, S, M, max_trials=10):
    """Find best t in [0,1] such that max(conv((1-t)*a+t*g, ...)) is minimized."""
    best_obj = np.inf
    best_t = 0.0
    for i in range(max_trials):
        t = 1.0 / (i + 1)
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 0.01:
            continue
        a_norm = a_new * (S / s)
        c = fft_convolve_self(a_norm)
        obj = c.max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
        if obj < M * 0.99:
            # Good improvement, try larger t
            break
    return best_t, best_obj

def softmax_objective_and_grad(theta, n, T, S):
    """Compute soft-max objective and gradient w.r.t. theta (log-softmax parameterization)."""
    a = S * softmax(theta)
    c = fft_convolve_self(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T
    
    # Gradient
    w = np.exp(shifted) / np.exp(shifted).sum()  # soft-max weights
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    grad_a = 2.0 * full_conv[n-1:2*n-1]
    
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    
    return obj, grad_theta

def softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def run_adam_refine(a, n, S, time_budget, T=100.0, lr=0.02):
    """Run Adam optimization to refine the sequence."""
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())
    n_par = len(theta)
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t_count = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    while time.monotonic() - start < time_budget:
        t_count += 1
        a = S * softmax(theta)
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()
        
        obj, grad = softmax_objective_and_grad(theta, n, T, S)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        lr_t = lr * (0.8 ** (progress * 3))
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding with
line search, hybrid Adam refinement, perturbation restarts, and coordinate
descent polish. The LP solves max sum(g) s.t. (f*g)[k]<=M for all k, g>=0
to find structurally informed descent directions.
"""
import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def softmax_fn(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def compute_true_obj(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def build_lp_matrix(f, n, num_constr=None):
    """Build sparse (2n-1 or num_constr) x n matrix for LP constraints."""
    total_lags = 2 * n - 1
    if num_constr is not None and num_constr < total_lags:
        c_self = fft_convolve_self(f)
        sorted_idx = np.argsort(c_self)[::-1]
        selected_lags = sorted_idx[:num_constr]
    else:
        selected_lags = np.arange(total_lags)

    n_rows = len(selected_lags)
    rows_list = []
    cols_list = []
    data_list = []

    for idx, k in enumerate(selected_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        js = np.arange(lo, hi + 1)
        vals = f[k - js]
        mask = vals > 1e-12
        if mask.any():
            cnt = int(mask.sum())
            rows_list.extend([idx] * cnt)
            cols_list.extend(js[mask].tolist())
            data_list.extend(vals[mask].tolist())

    if not rows_list:
        return csr_matrix((n_rows, n)), selected_lags
    return csr_matrix((data_list, (rows_list, cols_list)), shape=(n_rows, n)), selected_lags


def solve_lp_direction(f, M, n, time_limit=12, num_constr=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for selected k, g >= 0.
    Returns (g, g_sum) or (None, None)."""
    A, lags = build_lp_matrix(f, n, num_constr)
    c_obj = -np.ones(n)
    b = np.full(A.shape[0], M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True,
                                  'dual_feasibility_tolerance': 1e-8,
                                  'primal_feasibility_tolerance': 1e-8})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum
    except Exception as e:
        print("  LP solve error: %s" % str(e))
    return None, None


def line_search_for_step(a, g, S, M_current, max_trials=12):
    """Find best t in (0,1] minimizing max(conv((1-t)*a + t*g, itself)).
    Returns (best_t, best_conv_max, best_a)."""
    best_conv_max = np.inf
    best_t = 0.0
    best_a = a.copy()

    # Try a range of step sizes
    ts = [1.0, 0.7, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03]
    for i in range(min(max_trials, len(ts))):
        t = ts[i]
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 0.01:
            continue
        a_norm = a_new * (S / s)
        c = fft_convolve_self(a_norm)
        cm = c.max()
        if cm < best_conv_max:
            best_conv_max = cm
            best_t = t
            best_a = a_norm.copy()

    return best_t, best_conv_max, best_a


def softmax_grad(theta, n, T, S):
    """Compute soft-max objective and its gradient w.r.t. theta."""
    a = S * softmax_fn(theta)
    c = fft_convolve_self(a)
    c_max = c.max()

    shifted = T * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    obj = c_max + log_sum / T

    w = exp_shifted / exp_shifted.sum()
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    grad_a = 2.0 * full_conv[n - 1: 2 * n - 1]

    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot

    return obj, grad_theta


def run_adam_refine(a_start, n, S, time_budget, T_start=20.0, T_end=500.0, lr=0.03):
    """Adam optimization with temperature annealing on log-softmax parameters."""
    a = np.maximum(a_start, 1e-12)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())
    n_par = len(theta)
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t_count = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_conv_max = c.max()
    best_a = a.copy()

    while time.monotonic() - start < time_budget:
        t_count += 1
        a = S * softmax_fn(theta)
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_conv_max:
            best_conv_max = cur
            best_a = a.copy()

        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** progress

        obj, grad = softmax_grad(theta, n, T, S)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)

        lr_t = lr * (0.85 ** (progress * 4.0))
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_conv_max


def coordinate_descent_polish(a_start, n, S, time_budget, max_sweeps=30):
    """Coordinate descent: try reducing each coordinate to lower the peak."""
    a = a_start.astype(np.float64).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_conv_max = c.max()
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            # Binary search: find value of a[i] that minimizes the peak
            lo, hi = 0.0, orig * 1.2
            best_local_val = orig
            best_local_obj = best_conv_max
            for _ in range(10):
                mid = (lo + hi) / 2.0
                a[i] = mid
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_tmp = a * (S / s)
                c_tmp = fft_convolve_self(a_tmp)
                obj_tmp = c_tmp.max()
                if obj_tmp < best_local_obj:
                    best_local_obj = obj_tmp
                    best_local_val = mid
                if obj_tmp < best_conv_max:
                    lo = mid
                else:
                    hi = mid
            a[i] = best_local_val
            s = a.sum()
            if s > 1e-10:
                a = a * (S / s)
            c = fft_convolve_self(a)
            if c.max() < best_conv_max:
                best_conv_max = c.max()
                best_a = a.copy()

    return best_a, best_conv_max


def lp_iteration(a, n, S, M, time_limit_lp=10, num_constr=None):
    """One LP-based iteration: find direction, line search, update.
    Returns (new_a, new_M, improved)."""
    g, g_sum = solve_lp_direction(a, M, n, time_limit=time_limit_lp, num_constr=num_constr)
    if g is None:
        return a, M, False

    # Line search
    best_t, best_conv_max, best_a = line_search_for_step(a, g, S, M)

    if best_conv_max < M * (1.0 - 1e-10):
        return best_a, best_conv_max, True
    return a, M, False


def run_lp_phase(a_start, n, S, time_budget, max_stall=8):
    """Run LP-based iterations with Adam refinement in between."""
    a = a_start.copy()
    S = a.sum()
    c = fft_convolve_self(a)
    M = c.max()
    best_a = a.copy()
    best_M = M
    best_obj = 2.0 * n * M / (S * S)

    start = time.monotonic()
    stall_count = 0
    iter_count = 0

    print("  LP phase start: n=%d, M=%.8f, obj=%.8f" % (n, M, best_obj))

    while time.monotonic() - start < time_budget:
        # Determine how many constraints to use
        if time.monotonic() - start < time_budget * 0.5:
            num_constr = None  # all constraints
        else:
            num_constr = n  # top-n lags

        time_lp = min(8.0, (time_budget - (time.monotonic() - start)) * 0.3)
        new_a, new_M, improved = lp_iteration(a, n, S, M,
                                              time_limit_lp=time_lp,
                                              num_constr=num_constr)

        if improved:
            a = new_a
            M = new_M
            stall_count = 0
            if M < best_M:
                best_M = M
                best_a = a.copy()
                best_obj = 2.0 * n * M / (S * S)
            iter_count += 1
            if iter_count % 5 == 0:
                print("  LP iter %d: M=%.8f, obj=%.8f" % (iter_count, M, best_obj))
        else:
            stall_count += 1

        # Periodic Adam refinement
        if stall_count >= 3 or (iter_count > 0 and iter_count % 6 == 0):
            remaining = time_budget - (time.monotonic() - start)
            if remaining > 5.0:
                adam_budget = min(4.0, remaining * 0.15)
                a_ref, M_ref = run_adam_refine(a, n, S, adam_budget,
                                               T_start=50.0, T_end=200.0, lr=0.02)
                if M_ref < M:
                    a = a_ref
                    M = M_ref
                    if M < best_M:
                        best_M = M
                        best_a = a.copy()
                        best_obj = 2.0 * n * M / (S * S)
                        print("  Adam refine: M=%.8f, obj=%.8f" % (M, best_obj))
                stall_count = 0

        if stall_count >= max_stall:
            print("  LP phase: stalled after %d iterations" % iter_count)
            break

    return best_a, best_obj


def perturb_sequence(a, S, rng, epsilon=0.1):
    """Add multiplicative noise and renormalize."""
    noise = 1.0 + epsilon * (rng.random(len(a)) - 0.5)
    a_new = a * noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 0.01:
        return a.copy()
    return a_new * (S / s)

    if remaining > 15.0 and global_best_a is not None:
        print("\n=== Phase 5: Final LP micro-iterations (budget=%.0fs) ===" % remaining)
        a = global_best_a.copy()
        a = a * (S / a.sum())
        a_f, obj_f = run_lp_phase(a, n, S, remaining, max_stall=3)
        if obj_f < global_best_obj:
            global_best_obj = obj_f
            global_best_a = a_f.copy()
            print("  Final LP: obj=%.8f" % obj_f)

    # Final verification
    if global_best_a is None:
        global_best_a = np.ones(256)

    global_best_a = np.clip(np.asarray(global_best_a, dtype=np.float64), 0.0, 1000.0)
    if global_best_a.sum() < 0.01:
        global_best_a = np.ones(256)

    final_list = global_best_a.tolist()
    ver = evaluate_sequence(final_list)
    print("\n=== FINAL: internal=%.8f, verified=%.8f ===" % (global_best_obj,
          ver if np.isfinite(ver) else float('inf')))

    if not np.isfinite(ver):
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
        print("Fallback to uniform: obj=%.6f" % ver)

    return final_list
