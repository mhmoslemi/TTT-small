```python
def build_conv_matrix(f, n):
    """Build the matrix A where A[k,j] = f_{k-j} if 0 <= k-j < n, else 0."""
    A = np.zeros((2*n-1, n))
    # For k = 0, ..., n-1: A[k, 0..k] = f[k], f[k-1], ..., f[0]
    for k in range(n):
        A[k, :k+1] = f[k::-1]  # f[k], f[k-1], ..., f[0]
    # For k = n, ..., 2n-2: A[k, k-n+1..n-1] = f[n-1], f[n-2], ..., f[k-n+1]
    for k in range(n, 2*n-1):
        A[k, k-n+1:] = f[:2*n-1-k]
```

```python
"""
LP-based iterative optimization for minimizing the peak of autocorrelation
of non-negative step functions. Uses the linear programming approach to find
optimal mixing directions, with line search, restarts, and coordinate descent
polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def build_conv_matrix(f, n):
    """Build matrix A where A[k,j] = f[k-j] if 0<=k-j<n, else 0."""
    A = np.zeros((2 * n - 1, n))
    for k in range(n):
        A[k, :k + 1] = f[k::-1]
    for k in range(n, 2 * n - 1):
        A[k, k - n + 1:] = f[:2 * n - 1 - k]
    return A


def solve_lp(f, M, use_sparse=False):
    """
    Solve LP: max sum(g) s.t. A@g <= M, g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    A = build_conv_matrix(f, n)
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a, time_limit, rng, max_iter=10000):
    """
    LP-based iterative optimization.
    a: initial sequence (will be normalized to sum=1).
    Returns (best_sequence, best_peak).
    """
    a = np.clip(a.copy(), 0.0, None)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S  # normalize to sum 1
    
    n = len(a)
    aa = fft_convolve(a, a)
    best_peak = np.max(aa)
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    patience = 200
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit and iter_count < max_iter:
        iter_count += 1
        M = np.max(aa)
        
        # Solve LP
        g = solve_lp(a, M)
        if g is None or np.sum(g) < 1e-10:
            no_improve += 1
            if no_improve > patience:
                # Perturb and restart
                a = best_a + 0.01 * rng.standard_normal(n)
                a = np.clip(a, 0.01, None)
                a = a / np.sum(a)
                aa = fft_convolve(a, a)
                no_improve = 0
            continue
        
        # Rescale g to sum 1
        g = g / np.sum(g)
        
        # Compute cross and self convolutions
        ag = fft_convolve(a, g)
        gg = fft_convolve(g, g)
        
        # Line search for best t
        best_t = 0.0
        best_t_peak = M
        for t in np.linspace(0.01, 1.0, 30):
            peak_t = np.max((1 - t) ** 2 * aa + 2 * t * (1 - t) * ag + t ** 2 * gg)
            if peak_t < best_t_peak:
                best_t_peak = peak_t
                best_t = t
        
        if best_t > 0 and best_t_peak < M * (1 - 1e-6):
            a = (1 - best_t) * a + best_t * g
            a = np.clip(a, 0.0, None)
            S_new = np.sum(a)
            if S_new > 1e-10:
                a = a / S_new
            aa = fft_convolve(a, a)
            new_peak = np.max(aa)
            
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = a.copy()
                no_improve = 0
            else:
                # Shouldn't happen if line search worked, but just in case
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > patience:
            # Multiple restart strategies
            r = rng.random()
            if r < 0.3:
                # Gaussian perturbation
                a = best_a * (1 + 0.05 * rng.standard_normal(n))
                a = np.clip(a, 0.01, None)
            elif r < 0.6:
                # Peak-directed perturbation
                peak_idx = int(np.argmax(aa))
                # Reduce top contributors
                contribs = np.zeros(n)
                for i in range(n):
                    if 0 <= peak_idx - i < n:
                        contribs[i] = a[i] * a[peak_idx - i]
                top_k = max(5, n // 10)
                top_idx = np.argsort(contribs)[::-1][:top_k]
                a[top_idx] *= 0.95
                # Increase random others
                other_idx = np.setdiff1d(np.arange(n), top_idx)
                inc_idx = rng.choice(other_idx, size=min(top_k, len(other_idx)), replace=False)
                a[inc_idx] *= 1.05
            else:
                # Complete restart from best with noise
                a = best_a + 0.02 * rng.standard_normal(n)
                a = np.clip(a, 0.01, None)
            
            a = a / np.sum(a)
            aa = fft_convolve(a, a)
            no_improve = 0
        
        if iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_peak
            print(f"    LP iter={iter_count}, peak={best_peak:.8f}, score={score:.6f}, t={elapsed:.0f}s")
    
    return best_a, best_peak


def coordinate_descent(a, time_limit, rng):
    """
    Coordinate descent polish: for each element, try to improve the peak.
    """
    current = np.clip(a.copy(), 0.0, None)
    S = np.sum(current)
    if S < 1e-10:
        return current, np.inf
    current = current / S
    n = len(current)
    
    aa = fft_convolve(current, current)
    current_peak = np.max(aa)
    best = current.copy()
    best_peak = current_peak
    
    t0 = time.monotonic()
    round_count = 0
    
    while time.monotonic() - t0 < time_limit:
        round_count += 1
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            base_val = current[j]
            if base_val < 1e-8:
                continue
            
            # Try several multipliers
            for frac in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                trial_val = base_val * frac
                if trial_val < 0:
                    continue
                current[j] = trial_val
                # Recompute peak (need full recompute for accuracy)
                aa_new = fft_convolve(current, current)
                new_peak = np.max(aa_new)
                
                if new_peak < current_peak - 1e-12:
                    current_peak = new_peak
                    aa = aa_new
                    improved = True
                    if current_peak < best_peak:
                        best_peak = current_peak
                        best = current.copy()
                else:
                    current[j] = base_val
                    aa = aa_new  # restore
                    break
        
        if not improved:
            # Try a random perturbation
            j = int(rng.integers(0, n))
            current[j] = max(0.0, current[j] * np.exp(rng.normal(0, 0.1)))
            current = current / np.sum(current)
            aa = fft_convolve(current, current)
            current_peak = np.max(aa)
            if current_peak < best_peak:
                best_peak = current_peak
                best = current.copy()
        
        if round_count % 50 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_peak
            print(f"    CD round={round_count}, peak={best_peak:.8f}, score={score:.6f}, t={elapsed:.0f}s")
    
    return best, best_peak


def make_structured_init(n, rng, method='spread'):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)
    
    if method == 'spread':
        # Broadband with many incommensurate frequencies
        a = np.ones(n)
        for k in range(1, 20):
            freq = k * (1 + 0.618 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.2 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.01 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    
    elif method == 'log_spacing':
        # Log-spaced peaks
        a = np.ones(n)
        num_peaks = max(3, int(np.log2(n)))
        for i in range(num_peaks):
            center = int(n * (i + 0.5) / num_peaks)
            width = max(1, n // (num_peaks * 4))
            a[max(0, center - width):center + width + 1] *= 2.0
        return a
    
    elif method == 'quadratic':
        # Quadratic envelope
        mid = (n - 1) / 2.0
        a = 1.0 + 0.5 * ((t - mid) / mid) ** 2
        return a
    
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
```

Let me optimize: instead of computing the full convolution, I can compute only the change in the peak. When I change $a_j$ by $\delta$, the change in $(a*a)_k$ is:
$\delta (a_{k-j} + a_{k-j}) + \delta^2 \cdot [k = 2j]$ (approximately, for small $\delta$).

Actually, more precisely:
$(a + \delta e_j) * (a + \delta e_j) = a*a + \delta (e_j * a + a * e_j) + \delta^2 e_j * e_j$

$(e_j * a)_k = a_{k-j}$ (with bounds)
$(a * e_j)_k = a_{k-j}$ (same thing, by symmetry of convolution... wait, $e_j * a$ and $a * e_j$ are the same by commutativity of convolution).

Actually, $(e_j * a)_k = \sum_m (e_j)_m a_{k-m} = a_{k-j}$.
And $(a * e_j)_k = \sum_m a_m (e_j)_{k-m} = a_{k-j}$.

So the cross term is $2\delta a_{k-j}$.
And $e_j * e_j$ is a delta at position $2j$, so the $\delta^2$ term is $\delta^2$ at position $2j$ and 0 elsewhere.

So the new peak is:
$\max_k [(a*a)_k + 2\delta a_{k-j} + \delta^2 \cdot \mathbf{1}[k=2j]]$

This is much faster to compute! I just need to update the peak by checking the modified values.
