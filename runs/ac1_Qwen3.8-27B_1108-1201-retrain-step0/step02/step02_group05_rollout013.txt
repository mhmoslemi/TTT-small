```python
    # Actually w has length 2n-1 and a has length n

    # This is a "sliding dot product" which can be done with FFT

    # = sum_k a[k] * w[k+j] which is what we want! g_j = sum_m a_m * w_{m+j}

    # Yes! That's exactly g_j = sum_m a_m * w_{m+j}
    
    grad_a = 2.0 * g
    
    # Convert from grad_a to grad_theta via softmax
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta
```

```python
"""
Multi-strategy optimization for minimum peak autocorrelation of non-negative sequences.
Combines L-BFGS with soft-max smoothing, projected gradient descent with simplex projection,
targeted pairwise mass-transfer local search, and simulated annealing polish.
Focuses on n=256 with multiple diverse starts and careful time allocation.
"""

import numpy as np
import time
from scipy.optimize import minimize

# === Core utilities ===

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_normalized(a):
    n = len(a)
    c = fft_convolve(a)
    return 2.0 * n * np.max(c)

def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s

def proj_simplex(v, target_sum=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta_val = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta_val, 0.0)

# === Objective functions with gradients ===

def soft_max_obj_grad(theta, tau):
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a)
    
    c_max = np.max(c)
    shifted = np.clip((c - c_max) / tau, -50.0, 50.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    
    score = 2.0 * n * soft_max_val
    
    g = np.correlate(w, a, mode='valid')
    grad_a = 2.0 * g
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta

def pnorm_obj_grad(theta, p):
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a)
    
    # p-norm of c
    cp = c ** (p - 1)
    sum_cpw = np.sum(cp * c)
    if sum_cpw < 1e-300:
        return 0.0, np.zeros(n)
    pnorm = sum_cpw ** (1.0 / p)
    score = 2.0 * n * pnorm
    
    factor = 4.0 * n * (sum_cpw ** (1.0/p - 1.0))
    # grad(c_p)_j = 2 * sum_k cp[k] * a_{k-j}
    g = np.correlate(cp, a, mode='valid')
    grad_a = factor * g
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta

# === Optimization routines ===

def run_lbgfs(theta0, n, obj_grad_func, schedule, param_name, deadline, maxiter=200):
    """Run L-BFGS with annealed parameter."""
    best_theta = theta0.copy()
    best_score = score_normalized(softmax(theta0))
    current_theta = theta0.copy()
    
    for param in schedule:
        if time.monotonic() > deadline - 10:
            break
        args = (param,)
        try:
            result = minimize(
                obj_grad_func, current_theta.copy(),
                method='L-BFGS-B', jac=True, args=args,
                options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxfun': maxiter * 10}
            )
            new_theta = result.x
            a_new = softmax(new_theta)
            s = score_normalized(a_new)
            if s < best_score:
                best_score = s
                best_theta = new_theta.copy()
                current_theta = new_theta.copy()
            print(f"    {param_name}={param}: score={s:.6f} (best={best_score:.6f})")
        except Exception as e:
            print(f"    {param_name}={param}: error {e}")
    
    return best_theta, best_score

def projected_gradient(a0, n, time_budget, rng, verbose=True):
    """
    Projected gradient descent on the simplex with peak-balancing objective.
    Uses soft-max approximation with adaptive temperature.
    """
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    current_score = score_normalized(a)
    best_a = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    tau = 0.01  # initial soft-max temperature
    eta = 0.01  # initial step size
    patience = 0
    max_patience = 200
    
    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a)
        c_max = np.max(c)
        shifted = np.clip((c - c_max) / tau, -50.0, 50.0)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        
        # Gradient of soft-max objective
        g = np.correlate(w, a, mode='valid')
        grad = 2.0 * g  # gradient of 2n * soft_max_val w.r.t. a (before normalization)
        
        # Step
        a_new = a - eta * grad
        a_new = proj_simplex(a_new, 1.0)
        
        new_score = score_normalized(a_new)
        
        if new_score < current_score:
            a = a_new
            current_score = new_score
            eta *= 1.05  # increase step slightly
            patience = 0
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        else:
            eta *= 0.7  # decrease step
            patience += 1
        
        # Anneal tau
        tau *= 0.995
        
        if patience > max_patience:
            # Restart with larger tau and fresh step
            tau = max(tau * 10, 0.005)
            eta = 0.01
            patience = 0
            # Add small random perturbation
            a = a * (1.0 + 0.01 * rng.standard_normal(n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            current_score = score_normalized(a)
        
        if verbose and (time.monotonic() - t0) % 5.0 < 0.1:
            print(f"    PG: score={current_score:.6f}, best={best_score:.6f}, tau={tau:.6f}, eta={eta:.4f}")
    
    return best_a, best_score

def pairwise_local_search(a0, time_budget, rng, verbose=True):
    """
    Local search using pairwise mass transfer between entries.
    Focuses on reducing the peak autocorrelation.
    """
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    current_score = score_normalized(a)
    best_a = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    accept_worse = 0.001  # accept moves that increase score by up to this amount
    
    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a)
        peak_idx = int(np.argmax(c))
        
        # Find top contributors to c[peak_idx]
        # c[peak_idx] = sum_i a[i] * a[peak_idx - i]
        n = len(a)
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        # Top K contributors
        K = min(20, n // 4)
        top_idx = np.argsort(contribs)[-K:]
        
        # For each top contributor i, find the best j to transfer to
        # Moving delta from i to j changes c[peak] by approximately:
        # delta * (a[peak-j] - a[peak-i]) (first order)
        # We want a[peak-j] < a[peak-i], so reduce c[peak]
        
        best_move_score = current_score
        best_move_a = None
        
        for i in top_idx:
            # Try transferring a small amount from i to random other indices
            for _ in range(3):
                j = rng.integers(0, n)
                if j == i:
                    continue
                delta = a[i] * 0.1  # transfer 10% of a[i]
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                
                s = score_normalized(a_try)
                if s < best_move_score:
                    best_move_score = s
                    best_move_a = a_try.copy()
        
        if best_move_a is not None:
            a = best_move_a
            current_score = best_move_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        else:
            # No improvement found; try a random perturbation
            a = a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            current_score = score_normalized(a)
    
    if verbose:
        print(f"    Pairwise: best={best_score:.6f}")
    return best_a, best_score

def simulated_annealing(a0, time_budget, rng, verbose=True):
    """Simulated annealing with peak-targeted moves."""
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    current_score = score_normalized(a)
    best_a = a.copy()
    best_score = current_score
    n = len(a)
    
    t0 = time.monotonic()
    temperature = 0.001
    cooling = 0.9999
    
    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a)
        peak_idx = int(np.argmax(c))
        
        candidate = a.copy()
        move_type = rng.random()
        
        if move_type < 0.35:
            # Targeted: reduce top contributors to peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(5, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            other = np.setdiff1d(np.arange(n), top_idx)
            
            n_dec = rng.integers(1, min(5, len(top_idx) + 1))
            dec_idx = rng.choice(top_idx, size=n_dec, replace=False)
            n_inc = rng.integers(1, min(8, len(other) + 1))
            inc_idx = rng.choice(other, size=n_inc, replace=False)
            
            candidate[dec_idx] *= np.exp(rng.normal(-0.05, 0.02, n_dec))
            candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, n_inc))
        elif move_type < 0.55:
            # Block perturbation
            width = rng.integers(2, max(3, n // 10))
            start = rng.integers(0, n - width)
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.03, width))
        elif move_type < 0.75:
            # Symmetrize partially
            sym = 0.5 * (a + a[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.02, n))
        elif move_type < 0.90:
            # Smooth perturbation (Gaussian kernel)
            sigma = rng.uniform(0.5, 3.0)
            kernel = np.exp(-0.5 * (np.arange(n) - n//2)**2 / (2*sigma**2))
            candidate = a + 0.02 * rng.standard_normal(n) * kernel
        else:
            # Global small perturbation
            candidate = a * np.exp(rng.normal(0, 0.01, n))
        
        candidate = np.maximum(candidate, 0)
        cand_sum = candidate.sum()
        if cand_sum < 0.01:
            continue
        candidate = candidate / cand_sum
        
        new_score = score_normalized(candidate)
        
        if new_score < current_score or rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        
        temperature = max(temperature * cooling, 1e-8)
    
    if verbose:
        print(f"    SA: best={best_score:.6f}")
    return best_a, best_score

# === Initialization ===

def make_chirp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2 * np.pi * t**2 / (4*n))
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / a.sum()

def make_random(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / a.sum()

def make_bell(n, width, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width)**2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / a.sum()

def make_perturbed(base, sigma, rng):
    a = base * np.exp(rng.normal(0, sigma, len(base)))
    a = np.maximum(a, 1e-8)
    return a / a.sum()
```

"""
Multi-strategy optimization for minimum peak autocorrelation of non-negative sequences.
Combines: (1) L-BFGS with soft-max and p-norm annealing from diverse starts,
(2) projected gradient descent with backtracking on the simplex,
(3) peak-targeted local search with pairwise mass transfer,
(4) simulated annealing polish. Focuses on n=256 with careful time allocation.
"""

import numpy as np
import time
from scipy.optimize import minimize

# === Core utilities ===

def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fft_convolve(a)
    return 2.0 * n * np.max(c)

def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s

    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np
