from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    n = len(a)

    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def fft_convolve_self(a):
    n = len(a)

    A = fft(a, fsize)

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]

        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_topk(a, c, top_k):
    n = len(a)
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]

        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    w_full[top_indices] = weights
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def segment_optimize(n, K, S):
    """Optimize K-segment piecewise-constant sequence."""
    # Each segment has length n//K (approximately)
    seg_len = n // K
    # Segment i covers indices [i*seg_len, (i+1)*seg_len)
    
    # Self-convolution of K-segment sequence:
    # c_k = sum_i h_i * h_{i + delta(k)} * overlap(i, i+delta(k), k)
    # where delta(k) = round(k / seg_len) approximately
    
    # Actually, let's just compute it directly for a candidate
    def eval_segments(h):
        h = np.maximum(h, 0)
        h = h * (S / h.sum())
        a = np.repeat(h, seg_len)[:n]
        if len(a) < n:
            a = np.concatenate([a, np.zeros(n - len(a))])
        c = fft_convolve_self(a)
        return c.max()
    
    # Use scipy.optimize.minimize
    from scipy.optimize import minimize
    h0 = np.ones(K)
    result = minimize(eval_segments, h0, method='Nelder-Mead', 
                      options={'maxiter': 10000, 'xatol': 1e-10, 'fatol': 1e-10})
    h_opt = np.maximum(result.x, 0)
    h_opt = h_opt * (S / h_opt.sum())
    
    # Build full sequence
    a = np.repeat(h_opt, seg_len)[:n]
    if len(a) < n:
        a = np.concatenate([a, np.zeros(n - len(a))])
    return a, eval_segments(h_opt)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_topk(a, c, top_k):
    """Subgradient using top-k active lags via FFT cross-correlation."""
    n = len(a)
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    w_full[top_indices] = weights
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def lse_subgradient(a, c, beta):
    """Gradient of log-sum-exp smooth approximation."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def segment_init(n, S, K, rng):
    """Create a K-segment piecewise-constant initialization with optimized heights."""
    seg_len = max(1, n // K)
    actual_segs = min(K, n // seg_len)

    def obj(h):
        h = np.maximum(h, 1e-10)
        h = h / h.sum() * S
        a = np.repeat(h, seg_len)[:n]
        if len(a) < n:
            a = np.concatenate([a, np.zeros(n - len(a))])
        c = fft_convolve_self(a)
        return c.max()

    best_h = None
    best_val = np.inf
    for trial in range(3):
        if trial == 0:
            h0 = np.ones(actual_segs)
        elif trial == 1:
            h0 = 1.0 + 0.3 * np.sin(np.pi * np.arange(actual_segs) / actual_segs)
        else:
            h0 = rng.random(actual_segs) + 0.1
        h0 = np.maximum(h0, 0.01)
        res = sp_minimize(obj, h0, method='Nelder-Mead',
                          options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-8,
                                   'adaptive': True})
        h_try = np.maximum(res.x, 1e-10)
        val = obj(h_try)
        if val < best_val:
            best_val = val
            best_h = h_try

    best_h = best_h / best_h.sum() * S
    a = np.repeat(best_h, seg_len)[:n]
    if len(a) < n:
        a = np.concatenate([a, np.zeros(n - len(a))])
    return a, best_val


def run_nesterov(a_init, n, S, max_time, rng, top_k_start=5, top_k_end=512,
                 step_scale=1.0, perturb_period=4000, perturb_scale=0.002,
                 grad_mode="topk", beta_start=5.0, beta_end=300.0):
    """Nesterov-accelerated subgradient with adaptive top-k and optional LSE."""
    a = np.clip(a_init, 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    last_perturb = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, int(max_time * 3)))

        if grad_mode == "topk":
            top_k = int(top_k_start + frac * (top_k_end - top_k_start))
            c = fft_convolve_self(y)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = y.copy()
            grad = subgradient_topk(y, c, top_k)
        elif grad_mode == "lse":
            beta = beta_start + frac * (beta_end - beta_start)
            c = fft_convolve_self(y)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = y.copy()
            grad = lse_subgradient(y, c, beta)
        elif grad_mode == "hybrid":
            top_k = int(min(256, top_k_start + frac * (top_k_end * 0.5)))
            beta = beta_start + frac * (beta_end * 0.5)
            c = fft_convolve_self(y)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = y.copy()
            g1 = subgradient_topk(y, c, top_k)
            g2 = lse_subgradient(y, c, beta)
            grad = 0.6 * g1 + 0.4 * g2
        elif grad_mode == "topk_fine":
            top_k = min(1024, 50 + it // 5)
            c = fft_convolve_self(y)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = y.copy()
            grad = subgradient_topk(y, c, top_k)
        else:
            top_k = 20
            c = fft_convolve_self(y)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = y.copy()
            grad = subgradient_topk(y, c, top_k)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = step_scale * 2.0 * S / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it - last_perturb >= perturb_period and rng.random() < 0.1:
            scale = perturb_scale * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, scale, n), S)
            t_mom = 1.0
            last_perturb = it

    return best_local, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None

    # Get the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given = np.clip(given, 0, 1000)
        if given.size < 64 or given.sum() < 0.01:
            given = None
    except Exception:
        given = None

    n_main = len(given) if given is not None else 8192
    S_main = np.sqrt(2.0 * float(n_main))

    # ===== Phase 1: Segment-based initializations (fast) =====
    print("=== Phase 1: Segment initializations ===")
    t_phase1 = 120.0

    candidates = []
    if given is not None:
        c_g = fft_convolve_self(safe_normalize(given, n_main))
        candidates.append(("given", safe_normalize(given, n_main), c_g.max()))
        print("  Given: score=%.6f" % c_g.max())

    # Try segment inits at main n
    for K in [3, 5, 7, 9]:
        if time.monotonic() > deadline - 120:
            break
        a_seg, score_seg = segment_init(n_main, S_main, K, rng)
        candidates.append(("seg%d" % K, a_seg, score_seg))
        print("  Seg K=%d: score=%.6f" % (K, score_seg))

    # Try a random structured init
    tt = np.linspace(0, 1, n_main, endpoint=False)
    a_struct = 1.0 + 0.4 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 11 * tt)
    a_struct = np.maximum(a_struct, 0.01)
    a_struct = safe_normalize(a_struct, n_main)
    c_s = fft_convolve_self(a_struct)
    candidates.append(("struct", a_struct, c_s.max()))
    print("  Structured: score=%.6f" % c_s.max())

    # Try alternative n values with segment inits
    for n_alt in [4096, 12288, 16384]:
        if n_alt == n_main or time.monotonic() > deadline - 120:
            continue
        S_alt = np.sqrt(2.0 * float(n_alt))
        for K in [5, 9]:
            a_seg, score_seg = segment_init(n_alt, S_alt, K, rng)
            candidates.append(("seg%d_n%d" % (K, n_alt), a_seg, score_seg))
            print("  Seg K=%d n=%d: score=%.6f" % (K, n_alt, score_seg))

    # Sort by score and track best
    candidates.sort(key=lambda x: x[2])
    for name, arr, score in candidates[:3]:
        if score < global_best:
            global_best = score
            global_a = arr.copy()

    print("  Best after Phase 1: %.6f" % global_best)

    # ===== Phase 2: Focused Nesterov polishing (main computation) =====
    print("=== Phase 2: Focused polishing ===")
    n_best = len(global_a)
    S_best = np.sqrt(2.0 * float(n_best))
    time_remaining = deadline - time.monotonic()

    # Run multiple polishing passes with different settings
    polish_configs = [
        {"mode": "topk", "top_k_start": 5, "top_k_end": 200, "step": 1.0, "frac": 0.30},
        {"mode": "hybrid", "top_k_start": 10, "top_k_end": 256, "step": 0.8, "frac": 0.20,
         "beta_start": 5.0, "beta_end": 200.0},
        {"mode": "lse", "step": 1.0, "frac": 0.15, "beta_start": 10.0, "beta_end": 400.0},
        {"mode": "topk_fine", "step": 0.6, "frac": 0.20},
        {"mode": "topk", "top_k_start": 50, "top_k_end": 512, "step": 0.5, "frac": 0.15},
    ]

    # Also polish the best segment init if different from given
    if given is not None:
        # The given sequence might be in a different basin
        pass

    for i, cfg in enumerate(polish_configs):
        if time.monotonic() > deadline - 30:
            break
        seg_time = time_remaining * cfg["frac"]
        seg_time = max(10.0, seg_time)
        seg_time = min(seg_time, deadline - time.monotonic() - 30)
        if seg_time < 10:
            break

        print("  Polish %d/%d [%s] t=%.0fs" % (i + 1, len(polish_configs), cfg["mode"], seg_time))
        kwargs_run = {}
        if cfg["mode"] in ("topk", "topk_fine", "hybrid"):
            kwargs_run["top_k_start"] = cfg.get("top_k_start", 5)
            kwargs_run["top_k_end"] = cfg.get("top_k_end", 256)
        if cfg["mode"] in ("lse", "hybrid"):
            kwargs_run["beta_start"] = cfg.get("beta_start", 5.0)
            kwargs_run["beta_end"] = cfg.get("beta_end", 300.0)

        a_opt, score_opt = run_nesterov(
            global_a, n_best, S_best, seg_time, rng,
            step_scale=cfg["step"], grad_mode=cfg["mode"], **kwargs_run
        )
        a_opt = safe_normalize(a_opt, n_best)
        c_opt = fft_convolve_self(a_opt)
        score_opt = c_opt.max()

        if score_opt < global_best:
            global_best = score_opt
            global_a = a_opt.copy()
            print("    Improved: %.6f" % global_best)
        else:
            print("    No improvement: %.6f (best=%.6f)" % (score_opt, global_best))

    # ===== Phase 3: Multi-start from best + perturbations =====
    print("=== Phase 3: Multi-start refinement ===")
    n_best = len(global_a)
    S_best = np.sqrt(2.0 * float(n_best))
    time_remaining = deadline - time.monotonic()

    if time_remaining > 60:
        n_starts = max(2, int(time_remaining / 60))
        per_start = time_remaining / n_starts
        for s in range(n_starts):
            if time.monotonic() > deadline - 15:
                break
            # Perturb the best solution
            perturb_scale = 0.005 / (s + 1)
            a_pert = global_a + rng.normal(0, perturb_scale * S_best / np.sqrt(n_best), n_best)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = safe_normalize(a_pert, n_best)

            a_opt, score_opt = run_nesterov(
                a_pert, n_best, S_best, min(per_start, 120), rng,
                top_k_start=20, top_k_end=256, step_scale=0.8,
                grad_mode="topk", perturb_period=2000, perturb_scale=0.001
            )
            a_opt = safe_normalize(a_opt, n_best)
            c_opt = fft_convolve_self(a_opt)
            score_opt = c_opt.max()
            if score_opt < global_best:
                global_best = score_opt
                global_a = a_opt.copy()
                print("  Multi-start %d: improved to %.6f" % (s + 1, global_best))

    # ===== Phase 4: Ultra-fine final polish =====
    print("=== Phase 4: Ultra-fine polish ===")
    n_best = len(global_a)
    S_best = np.sqrt(2.0 * float(n_best))
    time_final = max(10.0, deadline - time.monotonic() - 10)

    a = global_a.copy()
    y = a.copy()
    t_mom = 1.0
    start_fin = time.monotonic()
    it_fin = 0
    best_fin = global_best

    while time.monotonic() - start_fin < time_final:
        it_fin += 1
        top_k = min(2048, 100 + it_fin * 2)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_fin:
            best_fin = cur
            global_best = cur
            global_a = y.copy()

        grad = subgradient_topk(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 0.5 * S_best / (n_best * (1.0 + 0.0003 * it_fin))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S_best)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Rare tiny perturbation
        if it_fin % 20000 == 0 and rng.random() < 0.03:
            y = project_simplex(y + rng.normal(0, 0.0002 * S_best / np.sqrt(n_best), n_best), S_best)
            t_mom = 1.0

    print("  Final score: %.6f" % global_best)

    # ===== Verification =====
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * np.sqrt(512.0) / 256

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: %.6f" % ver)
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        global_a = np.ones(256, dtype=np.float64) * np.sqrt(512.0) / 256
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, internal best=%.6f, verified=%.6f" %
          (len(final_list), global_best, ver if np.isfinite(ver) else float('inf')))
    return final_list
