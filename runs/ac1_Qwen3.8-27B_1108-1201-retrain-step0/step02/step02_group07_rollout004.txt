def lp_step(f, M):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)[k] <= M for all k."""
    n = len(f)
    # (f*b)[k] = sum_j f[j] * b[k-j] for valid j
    # This is linear in b: for each k, the coefficient of b[j] is f[k-j] (if valid)
    
    # For each k (0 to 2n-2), and each j (0 to n-1):
    # coefficient of b[j] in constraint k is f[k-j] if 0 <= k-j < n, else 0
    
    # A matrix: (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # For each j, find which k's have k-j in [0, n-1]
        # i.e., k in [j, j+n-1]
        k_start = j
        k_end = min(j + n - 1, 2*n - 2)
        A[k_start:k_end+1, j] = f[0:j+n-1-k_end+... ]
    
    # Actually, let me do it differently:
    # A[k, j] = f[k-j] if 0 <= k-j < n else 0
    # For row k: A[k, :] has non-zero entries where 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(n-1, k)
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end+1] = f[k - j_end:k - j_start + 1][::-1]
    
    # Maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: A @ b <= M (scalar), b >= 0
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:

"""
Optimizer combining LP-based peak reduction, mirror descent, and smooth gradient methods
to minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Compute full linear auto-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    np.maximum(result, 0, out=result)
    return result[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def lp_solve(f, M):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)[k] <= M for all k.
    Returns optimal b or None if infeasible."""
    n = len(f)
    n_cons = 2 * n - 1
    
    # Build A matrix: A[k, j] = f[k-j] if 0 <= k-j < n else 0
    A = np.zeros((n_cons, n))
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, n_cons - 1)
        length = k_hi - k_lo + 1
        A[k_lo:k_hi + 1, j] = f[:length]
    
    c = -np.ones(n)
    b_ub = np.full(n_cons, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def softmax_subgradient(conv, a, temperature):
    """Compute smoothed subgradient of max(conv) using softmax."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = temperature * (conv - max_c) / max_c
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    
    sg = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:

            indices = np.arange(j_start, j_end + 1)
            sg[j_start:j_end + 1] += w[k] * a[k - indices]

        # It's simply: d(conv[k])/da[j] = a[k-j] if 0 <= k-j < n, else 0.
        # Plus we need to account for the j in the other position: a[j]*a[k-j] means
        # d/da[j] gives a[k-j] (from the j position) and a[k-j] (from the k-j position when k-j=j, i.e., k=2j)
        # Actually: conv[k] = sum_i a[i]*a[k-i]. d/da[j] = a[k-j] (from i=j) + a[j] * d(a[k-i])/da[j] at i=k-j

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho_list = np.nonzero(cond)[0]
    if len(rho_list) == 0:
        return np.ones(n) / n
    rho = rho_list[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))
    
    # Gaussian bumps
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g = np.maximum(g, 0)
        if g.sum() > 1e-10:
            inits.append(g / g.sum())
    
    # Sine
    for freq in [1, 2, 3]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        inits.append(s / s.sum())
    
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 8)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits.append(bi / bi.sum())
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    
    # Sparse random
    for density in [0.1, 0.3, 0.5]:
        mask = rng.random(n) < density
        if mask.sum() > 0:
            r = np.zeros(n)
            r[mask] = rng.random(mask.sum())
            if r.sum() > 1e-10:
                inits.append(r / r.sum())
    
    return inits


def get_parent():
    """Try to get parent sequence from global."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def safe_evaluate(a):
    """Safely evaluate a sequence."""
    try:
        a_clean = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a_clean = np.minimum(a_clean, 1000.0)
        s = a_clean.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a_clean.tolist()))
        return score
    except Exception:
        return float('inf')


def coordinate_descent(a, time_limit, max_sweeps=100):
    """Cyclic coordinate descent with proportional rescaling."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a /= a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            
            # Try increasing a[i] and proportionally decreasing others
            # a_new[i] = a[i] + d, a_new[j] = a[j] * (1-(a[i]+d)) / (1-a[i]) for j != i
            # This is complex. Let's use a simpler approach:
            # Binary search for the best d in [-a[i]*0.5, a[i]*0.5 + small]
            
            a_i = a[i]
            if a_i < 1e-10:
                # Try adding a small amount
                d = (1.0 - a_i) * 0.01
                if d > 0:
                    a_try = a.copy()
                    a_try[i] += d
                    others_sum = a_try.sum() - a_try[i]
                    if others_sum > 1e-15:
                        a_try[np.arange(n) != i] *= (1.0 - a_try[i]) / others_sum
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() > 1e-15:
                            a_try /= a_try.sum()
                            new_obj = 2.0 * n * fast_convolve(a_try).max()
                            if new_obj < best_obj:
                                best_obj = new_obj
                                best_a = a_try.copy()
                                a = a_try
                continue
            
            # Golden section search for d in [-0.5*a_i, 0.5*(1-a_i)]
            lo = -0.5 * a_i
            hi = 0.5 * (1.0 - a_i)
            
            if hi <= lo:
                continue
            
            # Do a few golden section iterations
            gr = 0.618
            c1 = lo + gr * (hi - lo)
            c2 = lo + (1.0 - gr) * (hi - lo)
            
            def eval_d(d):
                a_try = a.copy()
                a_try[i] = a_i + d
                if a_try[i] < 0:
                    return np.inf
                others = a_try.sum() - a_try[i]
                if others < 1e-15:
                    return np.inf
                a_try[np.arange(n) != i] *= (1.0 - a_try[i]) / others
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    return np.inf
                a_try /= s_try
                return 2.0 * n * fast_convolve(a_try).max()
            
            # Simplified: just try a few values
            for d_frac in [0.1, -0.1, 0.2, -0.2, 0.3, -0.3, 0.5, -0.5]:
                d = d_frac * (hi - lo)
                a_try = a.copy()
                a_try[i] = a_i + d
                if a_try[i] < 0:
                    continue
                others = a_try.sum() - a_try[i]
                if others < 1e-15:
                    continue
                a_try[np.arange(n) != i] *= (1.0 - a_try[i]) / others
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try /= s_try
                new_obj = 2.0 * n * fast_convolve(a_try).max()
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_a = a_try.copy()
                    a = a_try
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400):
    """Main entry point for the optimization."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    
    parent = get_parent()
    
    best_sequence = None
    best_score = float('inf')
    
    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d" % (score, len(a)))
    
    print("Starting optimization at t=0")
    print("Parent available: %s" % (parent is not None))
    
    # Phase 1: Quick scan with mirror descent
    print("\n=== Phase 1: Quick scan ===")
    scan_lengths = [128, 256, 512, 1024]
    scan_results = []
    phase1_budget = min(300.0, (deadline - time.monotonic()) * 0.15)
    
    for n in scan_lengths:
        if time.monotonic() > deadline - 1500:
            break
        
        inits = generate_inits(n, rng)
        # Add parent-based init
        if parent is not None:
            p = np.maximum(parent, 0).astype(np.float64)
            if p.size >= n:
                p_crop = p[:n]
            else:
                p_crop = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
            p_crop = np.maximum(p_crop, 0)
            if p_crop.sum() > 1e-10:
                inits.append(p_crop / p_crop.sum())
        
        n_configs = len(inits)
        time_per = max(3.0, phase1_budget / max(1, len(scan_lengths) * n_configs))
        
        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 1500:
                break
            t_alloc = min(time_per, deadline - time.monotonic() - 1500)
            if t_alloc < 2.0:
                break
            
            a_opt, obj = mirror_descent_optimize(a0, time_limit=t_alloc, t_init=50.0, eta_init=0.4)
            score = safe_evaluate(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, idx, a_opt))
            record(a_opt)
    
    print("  Phase 1 complete. Best: %.6f" % best_score)
    scan_results.sort(key=lambda x: x[0])
    
    # Phase 2: LP-based deep optimization
    print("\n=== Phase 2: LP-based optimization ===")
    top_configs = []
    seen_keys = set()
    for score, n, idx, a_opt in scan_results:
        key = (n, idx)
        if key not in seen_keys and len(top_configs) < 10:
            top_configs.append((n, a_opt))
            seen_keys.add(key)
    
    if best_sequence is not None:
        top_configs.append((len(best_sequence), best_sequence.copy()))
    
    # Also add some fresh starts for diversity
    for n in [256, 512]:
        a_fresh = rng.dirichlet(np.ones(n) * 0.3)
        top_configs.append((n, a_fresh))
    
    phase2_budget = min(1500.0, (deadline - time.monotonic()) * 0.55)
    n_deep = len(top_configs)
    time_per_deep = max(10.0, phase2_budget / max(1, n_deep))
    
    for idx, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 600:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 600)
        if t_alloc < 10.0:
            break
        
        print("  [LP %d/%d] n=%d, budget=%.0fs" % (idx + 1, n_deep, n, t_alloc))
        
        # Split time: 60% LP, 40% mirror descent polish
        lp_time = t_alloc * 0.6
        md_time = t_alloc * 0.4
        
        a_cur = a0.copy()
        a_cur, _ = lp_optimize(a_cur, time_limit=lp_time, verbose=True)
        record(a_cur)
        
        if time.monotonic() < deadline - 600:
            rem = min(md_time, deadline - time.monotonic() - 600)
            if rem > 3:
                a_cur, _ = mirror_descent_optimize(a_cur, time_limit=rem, t_init=t_alloc, eta_init=0.1)
                record(a_cur)
    
    print("  Phase 2 complete. Best: %.6f" % best_score)
    
    # Phase 3: Polish with different methods
    print("\n=== Phase 3: Polishing ===")
    if best_sequence is not None and time.monotonic() < deadline - 50:
        phase3_budget = min(400.0, deadline - time.monotonic() - 50)
        n_polish = 4
        time_per_polish = phase3_budget / n_polish
        
        for i in range(n_polish):
            if time.monotonic() > deadline - 50:
                break
            
            a_start = best_sequence.copy()
