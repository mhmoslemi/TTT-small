def lp_step(f, M):
    n = len(f)
    # Build constraint matrix: (f * b)[k] <= M for k = 0, ..., 2n-2
    # (f * b)[k] = sum_{i=0}^{n-1} f[i] * b[k-i] (with b[j] = 0 if j < 0 or j >= n)
    # = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[i] * b[k-i]
    
    # For constraint k: sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * b[j] <= M
    # (letting j = k-i, so i = k-j, and we need 0 <= i < n, i.e., k-n < j <= k, and 0 <= j < n)
    
    # So (f*b)[k] = sum_{j=0}^{n-1} f[k-j] * b[j] where f[x] = 0 if x < 0 or x >= n
    # = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * b[j]

```python
import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)

def build_conv_matrix_indices(n):
    """Precompute indices for the convolution constraint matrix."""
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # For constraint k, variable j: A[k,j] = f[k-j] if max(0,k-n+1) <= j <= min(n-1,k)
    rows = []
    cols = []
    for k in range(2*n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)

def lp_direction(f, M):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0."""
    n = len(f)
    # Build sparse constraint matrix
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n
    A_ub_rows = []
    A_ub_cols = []
    A_ub_vals = []
    for k in range(2*n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A_ub_rows.append(k)
            A_ub_cols.append(j)
            A_ub_vals.append(f[k - j])
    
    from scipy.sparse import csr_matrix
    A_ub = csr_matrix((A_ub_vals, (A_ub_rows, A_ub_cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return f  # fallback
```

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    """Fast score: 2n * max(conv) / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)

            cols_list.extend(range(j_lo, j_hi + 1))

def subgradient_polish(f, time_limit=60, verbose=False):

        return f

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(0.0, np.sum(v))])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
LP-based optimization for minimizing peak autocorrelation of non-negative sequences.
Uses Dantzig-Wolfe style iteration: at each step, solve an LP to find a direction
that reduces the convolution peak while maintaining feasibility. Combined with
subgradient polish and multiple sequence lengths/initializations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def build_conv_structure(n):
    """Precompute (rows, cols) indices for the convolution constraint matrix A.
    A[k, j] = f[k-j] when 0 <= k-j < n, else 0. Shape: (2n-1) x n."""
    rows_parts = []
    cols_parts = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        cnt = j_hi - j_lo + 1
        if cnt > 0:
            rows_parts.append(np.full(cnt, k, dtype=np.int32))
            cols_parts.append(np.arange(j_lo, j_hi + 1, dtype=np.int32))
    return np.concatenate(rows_parts), np.concatenate(cols_parts)


def solve_lp_for_direction(f, M, rows, cols):
    """Solve LP: maximize sum(b) subject to A(b) <= M*1, b >= 0.
    Returns optimal b or None on failure."""
    n = len(f)
    vals = f[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': 20, 'feasibility_tolerance': 1e-9})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(0.0, float(np.sum(v)))])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_iterate(f_init, time_limit=300, verbose=False):
    """Main LP-based iteration. f_init should be non-negative.
    Returns (best_f_normalized, best_M)."""
    n = len(f_init)
    f = np.maximum(f_init.astype(np.float64), 0.0)
    S = np.sum(f)
    if S < 1e-15:
        return f, np.inf
    f = f / S

    rows, cols = build_conv_structure(n)

    best_M = np.inf
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = float(np.max(conv))

        if M < best_M:
            best_M = M
            best_f = f.copy()

        if verbose and it % 5 == 0:
            print(f"    LP iter {it}: M={M:.10f}, best={best_M:.10f}, "
                  f"t={time.monotonic()-t0:.1f}s")

        # Solve LP for direction
        b = solve_lp_for_direction(f, M, rows, cols)
        if b is None:
            break
        S_b = float(np.sum(b))
        if S_b < 1e-12:
            break

        g = b / S_b  # normalize to sum=1

        # Try to improve via blending
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            S_new = float(np.sum(f_new))
            if S_new < 1e-15:
                continue
            f_new = f_new / S_new
            M_new = float(np.max(fft_convolve(f_new, f_new)))
            if M_new < M * (1.0 - 1e-10):
                f = f_new
                improved = True
                break

        if not improved:
            # Check convergence
            if it > 3 and M <= best_M * (1.0 + 1e-7):
                break
            # Try direct replacement with damping
            f_new = np.maximum(0.5 * f + 0.5 * g, 0.0)
            S_new = float(np.sum(f_new))
            if S_new > 1e-15:
                f_new = f_new / S_new
                M_new = float(np.max(fft_convolve(f_new, f_new)))
                if M_new < M:
                    f = f_new
                    continue
            # Stuck, try perturbing and continuing
            f = f + 0.001 * np.random.default_rng(it).standard_normal(n)
            f = project_simplex(f)

        it += 1

    return best_f, best_M


def subgradient_polish(f_init, time_limit=60, verbose=False):
    """Subgradient method to polish. f_init normalized to sum=1."""
    n = len(f_init)
    f = np.maximum(f_init.astype(np.float64), 0.0)
    S = np.sum(f)
    if S < 1e-15:
        return f
    f = f / S

    best_M = np.inf
    best_f = f.copy()
    t0 = time.monotonic()
    base_lr = 4.0 / n

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fft_convolve(f, f)
        M = float(np.max(conv))
        if M < best_M:
            best_M = M
            best_f = f.copy()

        # Find active lags
        threshold = M * 0.9
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])

        # Compute subgradient (average over active lags)
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, int(k) - n + 1)
            j_hi = min(n - 1, int(k))
            if j_lo <= j_hi:
                # grad[j] += f[k - j] for j in [j_lo, j_hi]
                # k - j ranges from k - j_hi to k - j_lo
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi + 1] += f[idx]

        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        lr = base_lr / np.sqrt(1.0 + 0.1 * it)
        f_new = f - lr * grad / g_norm
        f = project_simplex(f_new)

        if verbose and it % 1000 == 0:
            print(f"    SG iter {it}: M={M:.10f}, best={best_M:.10f}, "
                  f"t={time.monotonic()-t0:.1f}s")

    return best_f


def make_init(n, method, rng, provided=None):
    """Create initial non-negative sequence of length n, normalized to sum=1."""
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'gaussian':
        x = np.arange(n) / n
        a = np.exp(-3.0 * (x - 0.5) ** 2)
        a = np.maximum(a, 0.001)
        return a / np.sum(a)
    elif method == 'ramp':
        a = np.linspace(0.3, 1.7, n)
        return a / np.sum(a)
    elif method == 'sin':
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif method == 'bump':
        x = np.arange(n) / n
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif method == 'plateau':
        a = np.ones(n)
        a[n * 3 // 10:n * 7 // 10] = 1.5
        return a / np.sum(a)
    elif method == 'provided':
        if provided is not None and len(provided) > 1:
            x_old = np.linspace(0.0, 1.0, len(provided))
            x_new = np.linspace(0.0, 1.0, n)
            a = np.interp(x_new, x_old, provided)
            a = np.maximum(a, 0.001)
            return a / np.sum(a)
        return np.ones(n) / n
    elif method == 'noise':
        a = 1.0 + 0.05 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def multi_start_lp(n, time_limit, provided, rng, n_starts=3):
    """Try multiple initializations and return the best."""
    methods = ['provided', 'uniform', 'gaussian', 'bump', 'sin', 'plateau', 'ramp']
    # Shuffle a bit for diversity
    if len(methods) > n_starts:
        chosen = rng.choice(methods, size=n_starts, replace=False).tolist()
    else:
        chosen = methods[:n_starts]

    best_f = None
    best_M = np.inf
    t0 = time.monotonic()

    for i, method in enumerate(chosen):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        t_per = remaining / (len(chosen) - i)
        f0 = make_init(n, method, rng, provided)
        f_best, M_best = lp_iterate(f0, time_limit=t_per, verbose=(i == 0))
        if M_best < best_M:
            best_M = M_best
            best_f = f_best
        print(f"      n={n}, init='{method}', M={M_best:.10f}, "
              f"score={2.0*n*M_best:.6f}, t={time.monotonic()-t0:.1f}s")

    if best_f is None:
        best_f = make_init(n, 'uniform', rng, provided)
        best_M = np.max(fft_convolve(best_f, best_f))
    return best_f, best_M
