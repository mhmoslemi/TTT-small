def lp_direction(a, S_target, top_k=10):
    """
    Find the steepest descent direction using LP.
    Minimize max_k [sum_j (a[k-j]+a[j-k])*d[j]] over active peaks k
    subject to: d[j] >= -a[j], sum(d) = 0
    """
    n = len(a)
    c = fft_convolve(a, a)
    k_peaks = np.argsort(c)[::-1][:top_k]
    
    # Build LP:
    # Variables: d[0], ..., d[n-1], t
    # Minimize: t
    # Subject to:
    #   sum_j A[k,j] * d[j] <= t  for k in k_peaks
    #   -a[j] <= d[j]  for all j
    #   sum_j d[j] = 0
    
    # c_matrix: n variables + 1 (t)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each peak k: sum_j A[k,j]*d[j] - t <= 0
    # A[k,j] = a[k-j]*I[0<=k-j<n] + a[j-k]*I[0<=j-k<n]
    n_constraints = len(k_peaks) + n
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    
    for i, k in enumerate(k_peaks):
        for j in range(n):
            val = 0.0
            if 0 <= k - j < n:
                val += a[k - j]
            if 0 <= j - k < n:
                val += a[j - k]
            A_ub[i, j] = val
        A_ub[i, -1] = -1.0  # -t
        b_ub[i] = 0.0
    
    # Lower bound constraints: -a[j] <= d[j] => -d[j] <= a[j]
    for j in range(n):
        A_ub[len(k_peaks) + j, j] = -1.0
        b_ub[len(k_peaks) + j] = a[j]
    
    # Equality: sum d[j] = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0, None)] * (n + 1), method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    
    return None

        if step % 50 == 0:
            cur_s = compute_score(current)
            print(f"    LP: step={step}, cur={cur_s:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")
    
    return best, best_score


def spectral_init(n, rng, target_sum=None):
    """Initialize using inverse FFT of a shaped power spectrum."""
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)
    
    # Create a power spectrum that's roughly flat but with some structure
    freqs = np.arange(n // 2 + 1)
    
    # Multiple approaches
    r = rng.random()
    if r < 0.3:
        # Flat spectrum with random phases (white noise-like)
        power = np.ones(n // 2 + 1)
        power[0] = 0  # zero DC
    elif r < 0.6:
        # Broadband with emphasis on mid-frequencies
        power = np.exp(-((freqs - n // 4) / (n // 4)) ** 2)
        power[0] = 0.1
    else:
        # Emphasis on low frequencies (smooth sequences)
        power = 1.0 / (1.0 + freqs ** 0.5)
        power[0] = 0.5
    
    # Random phases
    phases = rng.uniform(0, 2 * np.pi, size=n // 2 + 1)
    phases[0] = 0  # DC is real
    
    # Build complex spectrum
    spec = np.sqrt(power) * np.exp(1j * phases)
    
    # Inverse FFT
    a = np.fft.irfft(spec * np.sqrt(n), n)
    
    # Make non-negative and project to simplex
    a = np.clip(a, 0, None)
    if np.sum(a) < 1e-6:
        a = np.ones(n)
    a = project_simplex(a, target_sum)
    
    return a


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    S = np.sqrt(2.0 * n)
    inits = []
    
    # 1. Uniform
    inits.append(np.full(n, S / n))
    
    # 2. Triangle
    t = np.arange(n)
    tri = np.minimum(t + 1, n - t)
    inits.append(project_simplex(tri, S))
    
    # 3. Sine-based
    for k in range(1, 4):
        a = 1.0 + 0.3 * np.cos(2 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
        inits.append(project_simplex(a, S))
    
    # 4. Spectral
    for _ in range(3):
        inits.append(spectral_init(n, rng, S))
    
    # 5. Two-peak
    a = np.ones(n)
    center = n // 2
    width = n // 6
    a[center - width:center + width] = 0.3
    inits.append(project_simplex(a, S))
    
    # 6. Random binary
    binary = rng.integers(0, 2, size=n).astype(float)
    if np.sum(binary) == 0:
        binary[0] = 1
    inits.append(project_simplex(binary, S))
    
    # 7. Ramp
    ramp = np.linspace(0.1, 2.0, n)
    inits.append(project_simplex(ramp, S))
    
    return inits


def coordinate_descent_polish(a_init, S_target, time_limit, rng):
    """Coordinate descent: for each element, try multiple values."""
    n = len(a_init)
    current = a_init.copy()
    current = project_simplex(current, S_target)
    best = current.copy()
    best_score = compute_score(current)
    
    t0 = time.monotonic()
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            base = current[j]
            if base < 1e-8:
                # Try adding a small amount
                for add in [0.01 * S_target / n, 0.05 * S_target / n, 0.1 * S_target / n]:
                    trial = current.copy()
                    trial[j] = add
                    trial = project_simplex(trial, S_target)
                    s = compute_score(trial)
                    if s < best_score:
                        best_score = s
                        best = trial.copy()
                        current = trial.copy()
                        improved = True
                continue
            
            # Try scaling
            for frac in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                trial = current.copy()
                trial[j] = base * frac
                trial = project_simplex(trial, S_target)
                s = compute_score(trial)
                if s < compute_score(current):
                    current = trial.copy()
                    improved = True
                    if s < best_score:
                        best_score = s
                        best = trial.copy()
                    break
        
        if not improved:
            # Try a random pair swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                amt = np.abs(current[i] - current[j]) * rng.uniform(0.1, 0.4)
                trial = current.copy()
                trial[i] = max(0, current[i] + amt * rng.choice([-1, 1]))
                trial[j] = max(0, current[j] - amt * rng.choice([-1, 1]))
                trial = project_simplex(trial, S_target)
                s = compute_score(trial)
                if s < compute_score(current) + 1e-6:
                    current = trial.copy()
        
        iteration += 1
        if iteration % 10 == 0:
            cur_s = compute_score(current)
            print(f"    CD: iter={iteration}, cur={cur_s:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")
    
    return best, best_score


def peak_directed_polish(a_init, S_target, time_limit, rng):
    """Targeted peak reduction: reduce top contributors to peak, spread to low areas."""
    n = len(a_init)
    current = a_init.copy()
    current = project_simplex(current, S_target)
    best = current.copy()
    best_score = compute_score(current)
    
    t0 = time.monotonic()
    temp = 0.01
    cooling = 0.9999
    
    for step in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))
        
        # Compute per-element contribution to peak
        contribs = np.zeros(n)
        for i in range(n):
            if 0 <= peak_idx - i < n:
                contribs[i] = current[i] * current[peak_idx - i]
        
        r = rng.random()
        if r < 0.4:
            # Reduce top contributors, boost bottom
            k = max(4, n // 16)
            top_idx = np.argsort(contribs)[::-1][:k]
            bot_idx = np.argsort(contribs)[:k]
            red = rng.uniform(0.05, 0.15)
            current[top_idx] *= (1 - red)
            current[bot_idx] *= (1 + red * 2)
        elif r < 0.6:
            # Single element adjustment at peak
            peak_contrib_idx = np.argsort(contribs)[::-1][:max(2, n // 32)]
            j = peak_contrib_idx[int(rng.integers(0, len(peak_contrib_idx)))]
            current[j] *= np.exp(-rng.uniform(0.05, 0.2))
            # Distribute to random low-contribution element
            j2 = int(rng.integers(0, n))
            current[j2] *= np.exp(rng.uniform(0.02, 0.1))
        else:
            # Block scaling
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.05))
            current[start:start + width] *= factor
        
        current = np.clip(current, 0, 1000)
        current = project_simplex(current, S_target)
        
        new_score = compute_score(current)
        if new_score < best_score:
            best_score = new_score
            best = current.copy()
        elif rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-10)):
            pass  # Accept anyway (simulated annealing)
        
        temp *= cooling
        temp = max(temp, 1e-8)
        
        if step % 500 == 0:
            cur_s = compute_score(current)
            print(f"    PDP: step={step}, cur={cur_s:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")
    
    return best, best_score


def optimize_single_n(n, a_start, time_budget, rng):
    """Full optimization pipeline for a single n."""
    S_target = np.sqrt(2.0 * n)
    best = a_start.copy()
    best_score = compute_score(best)
    
    print(f"    Starting score: {best_score:.6f}")
    
    # Phase 1: Variance pre-optimization
    t_var = min(0.2 * time_budget, 100.0)
    if t_var > 5:
        var_result = optimize_variance(best, S_target, t_var, rng)
        s_var = compute_score(var_result)
        if s_var < best_score:
            print(f"    After var pre-opt: {s_var:.6f}")
            best_score = s_var
            best = var_result.copy()
    
    # Phase 2: Multi-peak subgradient
    t_mpsd = min(0.35 * time_budget, 300.0)
    if t_mpsd > 5:
        mpsd_result, s_mpsd = multi_peak_subgrad_descent(best, S_target, t_mpsd, rng, top_k=8)
        if s_mpsd < best_score:
            print(f"    After MPSD: {s_mpsd:.6f}")
            best_score = s_mpsd
            best = mpsd_result.copy()
    
    # Phase 3: LP-based optimization
    t_lp = min(0.2 * time_budget, 120.0)
    remaining = time_budget - (time.monotonic() - time.monotonic())  # approximate
    if t_lp > 5:
        lp_result, s_lp = lp_based_opt(best, S_target, t_lp, rng, top_k=8)
        if s_lp < best_score:
            print(f"    After LP: {s_lp:.6f}")
            best_score = s_lp
            best = lp_result.copy()
    
    # Phase 4: Peak-directed polish
    t_pdp = min(0.15 * time_budget, 60.0)
    if t_pdp > 5:
        pdp_result, s_pdp = peak_directed_polish(best, S_target, t_pdp, rng)
        if s_pdp < best_score:
            print(f"    After PDP: {s_pdp:.6f}")
            best_score = s_pdp
            best = pdp_result.copy()
    
    # Phase 5: Coordinate descent
    t_cd = min(0.1 * time_budget, 40.0)
    if t_cd > 3:
        cd_result, s_cd = coordinate_descent_polish(best, S_target, t_cd, rng)
        if s_cd < best_score:
            print(f"    After CD: {s_cd:.6f}")
            best_score = s_cd
            best = cd_result.copy()
    
    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Uses a multi-phase approach:
    1. Generate diverse initializations
    2. Variance pre-optimization for smooth landscape
    3. Multi-peak subgradient descent with Nesterov acceleration
    4. LP-based steepest descent
    5. Peak-directed polish and coordinate descent
    6. Explore multiple sequence lengths
    """
    t_global = time.monotonic()
    time_budget = max(10.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Use provided sequence if available
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s_start = compute_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s_start:.6f}")
        if s_start < best_score:
            best_score = s_start
            best_seq = start_seq.copy()
    
    # Primary n values to try
    n_values = [384]
    if best_seq is not None and len(best_seq) != 384:
        n_values.insert(0, len(best_seq))
    n_values.extend([256, 320, 448, 512])
    # Remove duplicates while preserving order
    seen = set()
    n_values_unique = []
    for nv in n_values:
        if nv not in seen and nv <= 1024:
            seen.add(nv)
            n_values_unique.append(nv)
    n_values = n_values_unique
    
    print(f"\n{'='*60}")
    print(f"Starting optimization. Target n values: {n_values}")
    print(f"Time budget: {time_budget:.0f}s")
    print(f"{'='*60}")
    
    # Allocate time proportionally to n values
    # Give more time to n=384 (or the provided length)
    time_allocation = {}
    total_weight = 0
    for i, n in enumerate(n_values):
        if i == 0:
            weight = 3.0  # Primary
        elif i == 1:
            weight = 2.0
        else:
            weight = 1.0
        time_allocation[n] = weight
        total_weight += weight
    
    for n_val in n_values:
        if time.monotonic() > deadline - 30:
            break
        
        remaining = deadline - time.monotonic() - 20
        allocated = (time_allocation[n_val] / total_weight) * time_budget
        allocated = min(allocated, remaining)
        if allocated < 10:
            continue
        
        print(f"\n--- Optimizing n={n_val} (budget: {allocated:.0f}s) ---")
        
        # Generate initializations
        inits = diverse_init(n_val, rng)
        
        # If we have a good sequence from another n, interpolate
        if best_seq is not None and len(best_seq) != n_val:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_val)
            interp = np.interp(x_new, x_old, best_seq)
            S_t = np.sqrt(2.0 * n_val)
            inits.insert(0, project_simplex(interp, S_t))
        
        # Also add the provided sequence if it matches
        if best_seq is not None and len(best_seq) == n_val:
            inits.insert(0, best_seq.copy())
        
        # Try top initializations
        S_target = np.sqrt(2.0 * n_val)
        
        # Quick evaluation to rank initializations
        init_scores = []
        for a_init in inits:
            s = compute_score(a_init)
            init_scores.append((s, a_init))
        init_scores.sort(key=lambda x: x[0])
        
        print(f"  Top init scores: {[f'{s:.4f}' for s, _ in init_scores[:3]]}")
        
        # Optimize the top 2-3 initializations with time split
        n_top = min(3, len(init_scores))
        time_per = allocated / n_top
        
        for idx in range(n_top):
            if time.monotonic() > deadline - 20:
                break
            remaining_for_this = min(time_per, deadline - time.monotonic() - 15)
            if remaining_for_this < 5:
                break
            
            a_init = init_scores[idx][1]
            print(f"  Init {idx}: score={init_scores[idx][0]:.6f}, time={remaining_for_this:.0f}s")
            
            result, result_score = optimize_single_n(n_val, a_init, remaining_for_this, rng)
            
            # Verify with evaluate_sequence
            final_list = [float(max(0.0, min(1000.0, x))) for x in result]
            eval_score = evaluate_sequence(final_list)
            
            if eval_score < best_score:
                best_score = eval_score
                best_seq = result.copy()
                print(f"  *** NEW BEST: n={n_val}, score={best_score:.6f} ***")
            
            total_weight -= time_allocation[n_val]  # Don't reuse for other n
            break  # Only try the best init per n to save time
    
    # Final polish on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 20:
        final_time = deadline - time.monotonic() - 15
        if final_time > 10:
            n_final = len(best_seq)
            S_final = np.sqrt(2.0 * n_final)
            print(f"\n--- Final polish: n={n_final}, time={final_time:.0f}s ---")
            
            # Split between PDP and CD
            t1 = final_time * 0.5
            t2 = final_time * 0.5
            
            polished1, s1 = peak_directed_polish(best_seq, S_final, t1, rng)
            if s1 < best_score:
                best_seq = polished1
                best_score = s1
            
            polished2, s2 = coordinate_descent_polish(best_seq, S_final, t2, rng)
            if s2 < best_score:
                best_seq = polished2
                best_score = s2
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL: n={len(final_list)}, score={final_eval:.6f}")
        print(f"{'='*60}")
        if final_eval < best_score:
            best_score = final_eval
        return final_list
    
    # Fallback
    if best_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    print("FALLBACK: returning uniform sequence")
    return [1.0] * 256
