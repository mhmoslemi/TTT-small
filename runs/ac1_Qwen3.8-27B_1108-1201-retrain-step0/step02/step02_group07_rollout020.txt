def solve_lp(a, M):
    n = len(a)
    b = cp.Variable(n)
    # Convolution constraints: (a*b)_k <= M for k = 0,...,2n-2
    # (a*b)_k = sum_i a[i] * b[k-i] for valid i
    # This is a linear function of b
    constraints = [b >= 0]
    # Build the convolution matrix A_conv such that A_conv @ b = conv(a,b)
    # (a*b)_k = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * b[k-i]

    # For efficiency, we can build this matrix
    A_conv = np.zeros((2*n-1, n))
    for k in range(2*n-1):

            i = k - j
            if 0 <= i < n:
                A_conv[k, j] = a[i]
    
    constraints.append(A_conv @ b <= M)
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    return b.value

"""
LP-iterative optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find directions that reduce the peak autocorrelation,
with multiple lengths, initializations, and refinement stages.
"""

import numpy as np
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution a * b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix C such that C @ b = conv(a, b)."""
    n = len(a)
    # C[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # k ranges 0..2n-2, j ranges 0..n-1
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # j must satisfy 0 <= k-j < n, i.e., k-n < j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            C[k, j] = a[k - j]
    return C


def build_conv_matrix_fast(a):
    """Build convolution matrix more efficiently using vectorization."""
    n = len(a)
    C = np.zeros((2*n-1, n))
    for j in range(n):
        # For column j: C[k, j] = a[k-j] for k=j, j+1, ..., j+n-1
        # i.e., k-j ranges from 0 to n-1
        C[j:j+n, j] = a
    return C


def solve_lp_direction(a, M, verbose=False):
    """
    Solve: max sum(b) subject to b >= 0, conv(a,b) <= M * 1.
    Returns the optimal b (unnormalized).
    """
    n = len(a)
    b = cp.Variable(n)
    C = build_conv_matrix_fast(a)
    constraints = [b >= 0, C @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    # Try solvers in order of preference
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                if b.value is not None:
                    return np.maximum(b.value, 0)
        except Exception:
            continue
    
    # Fallback: try without specifying solver
    try:
        prob.solve()
        if prob.status in ['optimal', 'optimal_inaccurate'] and b.value is not None:
            return np.maximum(b.value, 0)
    except Exception:
        pass
    
    return None


def lp_iteration(a, max_iters=100, tol=1e-8, verbose=True, time_limit=None, t_start=None):

    LP-based iterative optimization.
    """

def subgradient_refine(a, time_limit, t_init=50.0, eta_init=0.3):
    """Mirror descent refinement."""
    a = np.maximum(a, 0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    max_pat = 50
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        max_c = conv.max()
        if max_c < 1e-15:
            break
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        sg = np.zeros(n)
        thresh = w.max() * 0.002
        active = np.where(w > thresh)[0]
        for ki in active:
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
        
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_pat:
            patience = 0
            eta *= 0.65
            t *= 1.3
            if eta < 0.004:
                break
    
    return best_a, best_obj


def anneal_refine(a, time_limit, rng):
    """Informed annealing for final refinement."""
    a = np.maximum(a, 0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.005 * (1.0 - frac) ** 2 + 0.0001
        
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # Target the peak positions
        peak_thresh = max_c * 0.8
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        
        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        target_j = int(rng.integers(n))
        
        delta = a[top_j] * rng.uniform(0.005, 0.03)
        if delta < 1e-12:
            continue
        
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
    
    return best_a, best_obj

LP-iterative optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find descent directions that reduce peak autocorrelation,
combined with mirror descent refinement and informed annealing.
Key insight: fixing a and solving max sum(b) s.t. b>=0, conv(a,b)<=M gives
a direction to improve the sequence.

import numpy as np
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix C such that C @ b = conv(a, b)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def solve_lp_direction(a, M):
    """
    Solve: max sum(b) subject to b >= 0, conv(a,b) <= M.
    Returns optimal b or None on failure.
    """
    n = len(a)
    b = cp.Variable(n)
    C = build_conv_matrix(a)
    constraints = [b >= 0, C @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDL_P]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate'] and b.value is not None:
                return np.maximum(b.value, 0.0)
        except Exception:
            continue

    try:
        prob.solve()
        if prob.status in ['optimal', 'optimal_inaccurate'] and b.value is not None:
            return np.maximum(b.value, 0.0)
    except Exception:
        pass
    return None

def mirror_descent_refine(a, time_limit, t_init=80.0, eta_init=0.2):

        max_c = conv.max()

        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        thresh = w.max() * 0.003
        active = np.where(w > thresh)[0]
        for ki in active:
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]

        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()

            a = np.ones(n) / n
        else:
            a /= s

        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:

            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_pat:
            patience = 0
            eta *= 0.7
            t *= 1.4
            if eta < 0.003:

def pgd_refine(a, time_limit, lr_init=0.05):
    """Projected gradient descent with softmax smoothing."""
    a = np.maximum(a, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = 40.0
    lr = lr_init
    patience = 0
    max_pat = 40
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        sg = np.zeros(n)
        thresh = w.max() * 0.003
        active = np.where(w > thresh)[0]
        for ki in active:
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]

        a = a - lr * sg
        # Project onto simplex
        v = np.sort(a)[::-1]
        cssv = np.cumsum(v)
        cond = v * np.arange(1, n + 1) > (cssv - 1.0)
        rho = np.nonzero(cond)[0]
        if len(rho) == 0:
            a = np.ones(n) / n
        else:
            rho = rho[-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a = np.maximum(a - theta, 0.0)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_pat:
            patience = 0
            lr *= 0.7
            t *= 1.3
            if lr < 0.001:
                break

    return best_a, best_obj


def anneal_refine(a, time_limit, rng):
    """Informed simulated annealing targeting peak reduction."""
    a = np.maximum(a, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.004 * (1.0 - frac) ** 2 + 0.00005

        max_c = conv.max()
        if max_c < 1e-15:
            break

        peak_thresh = max_c * 0.75
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])

        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue

        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        target_j = int(rng.integers(n))

        if top_j == target_j:
            continue

        delta = a[top_j] * rng.uniform(0.003, 0.025)
        if delta < 1e-12:
            continue

        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue

        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj

        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta

    return best_a, best_obj

def make_init(name, n, rng, parent):
    """Create an initialization of the given type."""
    x = np.arange(n, dtype=np.float64)
    if name == 'uniform':
        return np.ones(n) / n
    elif name == 'gauss':
        g = np.exp(-0.5 * ((x - n * 0.5) / (0.15 * n)) ** 2)
        return g / g.sum()
    elif name == 'gauss_wide':
        g = np.exp(-0.5 * ((x - n * 0.5) / (0.35 * n)) ** 2)
        return g / g.sum()
    elif name == 'twopk':
        bi = np.zeros(n)
        bi[:n // 4] = 1.0
        bi[-n // 4:] = 1.0
        return bi / bi.sum()
    elif name == 'ramp':
        r = np.arange(1, n + 1, dtype=np.float64)
        return r / r.sum()
    elif name == 'sin':
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
        return s / s.sum()
    elif name == 'sin2':
        s = 0.5 * (1.0 + np.cos(4.0 * np.pi * x / n))
        return s / s.sum()
    elif name == 'bethe':
        # Bethe-like: concentrated at center with decay
        g = np.exp(-0.5 * ((x - n * 0.5) / (0.08 * n)) ** 2)
        return g / g.sum()
    elif name == 'parent':
        if parent is None:
            return None
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0)
            if pp.sum() > 1e-10:
                return pp / pp.sum()
        pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size),
                       np.maximum(parent, 0))
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            return pp / pp.sum()
        return None
    elif name == 'rand_low':
        return rng.dirichlet(np.ones(n) * 0.3)
    elif name == 'rand_high':
        return rng.dirichlet(np.ones(n) * 3.0)
    return None

            c = best_alpha * a + (1.0 - best_alpha) * b_norm
            c = np.maximum(c, 0.0)
            c /= c.sum()
            a = c
            M = best_M_try
            best_a = a.copy()
            best_M = M
            stagnant = 0

        if verbose and (it % 20 == 0 or it < 5):
            print("      LP iter %d: M=%.8f, obj=%.6f, b_sum=%.4f" % (it, M, 2 * n * M, b_sum))

    if verbose:
        print("      LP done: n=%d, M=%.8f, obj=%.6f, iters=%d" % (n, best_M, 2 * n * best_M, it_executed + 1))

    return best_a, 2.0 * n * best_M


def mirror_descent_refine(a, time_limit, t_init=80.0, eta_init=0.2):
    """Mirror descent on simplex with adaptive parameters."""
    a = np.maximum(a, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    max_pat = 60
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        thresh = w.max() * 0.003
        active = np.where(w > thresh)[0]
        for ki in active:
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]

        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_pat:
            patience = 0
            eta *= 0.7
            t *= 1.4
            if eta < 0.003:
                break

    return best_a, best_obj


def pgd_refine(a, time_limit, lr_init=0.05):
    """Projected gradient descent with softmax smoothing."""
    a = np.maximum(a, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = 40.0
    lr = lr_init
    patience = 0
    max_pat = 40
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        sg = np.zeros(n)
        thresh = w.max() * 0.003
        active = np.where(w > thresh)[0]
        for ki in active:
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]

        a_new = a - lr * sg
        # Project onto simplex
        v = np.sort(a_new)[::-1]
        cssv = np.cumsum(v)
        cond = v * np.arange(1, n + 1) > (cssv - 1.0)
        rho = np.nonzero(cond)[0]
        if len(rho) == 0:
            a = np.ones(n) / n
        else:
            rho = rho[-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a = np.maximum(a_new - theta, 0.0)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_pat:
            patience = 0
            lr *= 0.7
            t *= 1.3
            if lr < 0.001:
                break

    return best_a, best_obj
