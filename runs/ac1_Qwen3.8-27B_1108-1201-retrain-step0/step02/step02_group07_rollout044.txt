        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            A[k, j_start:j_end + 1] = f[k - js]
    return A


def solve_lp_direction(f, M, n):
    """Solve: max sum(b) s.t. A@b <= M, b >= 0. Returns b or None."""
    A = build_constraint_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def normalize(a):
    """Normalize to probability vector."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def obj_value(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def lp_optimize(a0, time_limit, t_init=0.2, verbose=False):
    """
    LP-based optimization. At each step:
    1. Compute M = max(conv(a,a))
    2. Solve LP for best direction b
    3. Line search for step size
    4. Update a
    """
    a = normalize(a0.copy())
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    t = t_init
    stagnation = 0
    max_stagnation = 30
    t0 = time.monotonic()

    lp_matrix = build_constraint_matrix(a, n)

    while time.monotonic() - t0 < time_limit:
        # Check for stagnation
        if stagnation >= max_stagnation:
            # Try increasing M slightly to escape
            M *= 1.01
            t = max(0.05, t * 0.5)
            lp_matrix = build_constraint_matrix(a, n)
            b = solve_lp_direction(a, M, n)
            if b is None:
                break
            b = normalize(b)
            a_new = (1 - 0.1) * a + 0.1 * b
            a = normalize(a_new)
            conv = fast_convolve(a)
            M = conv.max()
            lp_matrix = build_constraint_matrix(a, n)
            stagnation = 0
            continue

        # Solve LP
        b = solve_lp_direction(a, M, n, )
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1.0 + 1e-6:
            # LP can't improve - at local optimum for this M
            stagnation += 1
            t *= 0.8
            if t < 0.02:
                # Try a small perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1e6))
                a = normalize(a + rng.normal(0, 0.01, n))
                conv = fast_convolve(a)
                M = conv.max()
                lp_matrix = build_constraint_matrix(a, n)
                stagnation = 0
                continue
            continue

        b = normalize(b)

        # Line search
        best_t = t
        best_new_obj = float('inf')
        for t_candidate in [t * 0.5, t, t * 1.5, t * 2.0]:
            t_candidate = min(t_candidate, 0.9)
            if t_candidate <= 0:
                continue
            a_cand = normalize((1 - t_candidate) * a + t_candidate * b)
            conv_cand = fast_convolve(a_cand)
            obj_cand = 2.0 * n * conv_cand.max()
            if obj_cand < best_new_obj:
                best_new_obj = obj_cand
                best_t = t_candidate

        if best_new_obj < best_obj - 1e-12:
            a = normalize((1 - best_t) * a + best_t * b)
            conv = fast_convolve(a)
            M = conv.max()
            best_obj = 2.0 * n * M
            best_a = a.copy()
            stagnation = 0
            # Adapt step size
            if best_new_obj < best_obj * 0.999:
                t = min(t * 1.2, 0.8)
        else:
            stagnation += 1
            t *= 0.7

    return best_a, best_obj


def smoothed_subgradient(conv, a, temperature=50.0):
    """Compute smoothed subgradient of max(conv) applied to a."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = temperature * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # Sparsify
    thresh = w.max() * 0.01
    active = np.where(w > thresh)[0]

    sg = np.zeros(n)
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    return sg


def gradient_optimize(a0, time_limit, t_init=50.0, lr_init=0.1):
    """Smoothed gradient descent on the simplex."""
    a = normalize(a0.copy())
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    temperature = t_init
    lr = lr_init
    stagnation = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        sg = smoothed_subgradient(conv, a, temperature)
        sg_c = sg - sg.mean()
        # Multiplicative update (mirror descent)
        log_update = -lr * sg_c
        log_update -= log_update.max()
        a = a * np.exp(log_update)
        a = normalize(a)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation >= 30:
            stagnation = 0
            lr *= 0.7
            temperature *= 1.3
            if lr < 0.005:
                lr = lr_init * 0.3
                temperature = t_init * 0.5

    return best_a, best_obj


def peak_local_search(a0, time_limit, rng):
    """Local search targeting specific peaks in the convolution."""
    a = normalize(a0.copy())
    n = len(a)
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 500:
            break

        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Find top peaks
        peak_thresh = max_c * 0.8
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])

        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)

        # Find the pair (i, j) that contributes most to peak k
        # conv[k] = sum_{i=js}^{je} a[i] * a[k-i]
        best_contrib = -1
        best_i = -1
        best_j = -1
        for i in range(js, je + 1):
            j = k - i
            c = a[i] * a[j]
            if c > best_contrib:
                best_contrib = c
                best_i = i
                best_j = j

        if best_i < 0:
            continue

        # Try to reduce the contribution by moving mass
        # Move mass from best_i (or best_j) to a "safe" position
        source = best_i if a[best_i] >= a[best_j] else best_j
        delta = a[source] * rng.uniform(0.02, 0.10)
        if delta < 1e-10:
            continue

        # Find target: position that doesn't contribute much to any peak
        target = -1
        for _ in range(10):
            cand = int(rng.integers(n))
            if cand == source:
                continue
            # Check contribution to top peaks
            contrib = 0.0
            for pk in peak_positions[:10]:
                d = pk - cand
                if 0 <= d < n:
                    contrib += a[d]
                d2 = cand - pk + 2 * n - 2
                if 0 <= d2 < n:
                    contrib += a[d2]
            if contrib < 2.0 * a.mean():
                target = cand
                break
        if target < 0:
            target = int(rng.integers(n))
            if target == source:
                continue

        a[source] -= delta
        a[target] += delta
        if a[source] < 0:
            a[source] += delta
            a[target] -= delta
            continue

        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj

        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.01 * (1.0 - frac) ** 2 + 0.0003

        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
                no_improve += 1
            else:
                a[source] += delta
                a[target] -= delta
                no_improve += 1

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    # Gaussian
    for sf in [0.15, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g = normalize(g)
        inits['gauss_%s' % sf] = g

    # Cosine / window functions
    for alpha in [0.5, 1.0, 2.0]:
        w = np.cos(np.pi * (x / (n - 1) - 0.5) * alpha) ** 2
        w = np.maximum(w, 0)
        if w.sum() > 1e-10:
            inits['cos_%s' % alpha] = normalize(w)

    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    if bi.sum() > 0:
        inits['bimodal'] = bi / bi.sum()

    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['rand_%s' % alpha] = r

    # Chirp-like
    chirp = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * x / (2.0 * n)))
    chirp = np.maximum(chirp, 0)
    if chirp.sum() > 1e-10:
        inits['chirp'] = normalize(chirp)

    # Sparse: random subset
    n_active = max(1, n // 3)
    idx = rng.choice(n, size=n_active, replace=False)
    sp = np.zeros(n)
    sp[idx] = 1.0
    inits['sparse'] = sp / sp.sum()

    # Parent
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.size >= n:
            pp = p[:n]
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            inits['parent'] = normalize(pp)

    return inits


def get_parent():
    """Try to access parent sequence from global variable."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def safe_evaluate(a):
    """Safely evaluate a sequence."""
    try:
        a_clean = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a_clean = np.minimum(a_clean, 1000.0)
        if a_clean.sum() < 0.01:
            return float('inf')
        if not np.all(np.isfinite(a_clean)):
            return float('inf')
        score = float(evaluate_sequence(a_clean.tolist()))
        return score
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. LP-based optimizer with peak-targeting local search.
    Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using linear programming
    for descent directions and adaptive step sizes.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)

    parent = get_parent()

    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d, time=%.0fs" % (
                score, len(a), time.monotonic() - t_start))

    # Phase 1: Quick scan over lengths and initializations
    print("Phase 1: Quick scan...")
    scan_lengths = [128, 192, 256, 384, 512, 768, 1024]
    scan_results = []
    phase1_time = min(400.0, (deadline - time.monotonic()) * 0.20)
    t0 = time.monotonic()

    for n in scan_lengths:
        if time.monotonic() > deadline - 1500:
            break
        inits = generate_inits(n, rng, parent)
        # Limit number of inits per length for speed
        init_names = list(inits.keys())
        if len(init_names) > 6:
            # Prioritize parent, uniform, and a couple others
            priority = []
            for name in ['parent', 'uniform', 'gauss_0.3', 'rand_1.0', 'cos_1.0', 'bimodal']:
                if name in init_names:
                    priority.append(name)
            for name in init_names:
                if name not in priority and len(priority) < 6:
                    priority.append(name)
            init_names = priority

        time_per_init = max(1.0, (phase1_time - (time.monotonic() - t0)) /
                           max(1, (len(scan_lengths) - scan_lengths.index(n)) * len(init_names)))

        for name in init_names:
            if time.monotonic() > deadline - 1500:
                break
            a0 = inits[name]
            t_alloc = min(time_per_init, deadline - time.monotonic() - 1500)
            if t_alloc < 1.0:
                break

            # Use LP for n <= 512, gradient for larger
            if n <= 512:
                a_opt, _ = lp_optimize(a0, time_limit=t_alloc)
            else:
                a_opt, _ = gradient_optimize(a0, time_limit=t_alloc)

            score = safe_evaluate(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
            record(a_opt)

    print("  Scan done. Best so far: %.6f, time=%.0fs" % (best_score, time.monotonic() - t_start))
    scan_results.sort(key=lambda x: x[0])

    # Phase 2: Deep optimization of top candidates
    print("Phase 2: Deep LP optimization...")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 10:
            top_configs.append((n, name, a_opt))
            seen.add(key)

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best_so_far', best_sequence.copy()))

    phase2_time = min(1200.0, (deadline - time.monotonic()) * 0.45)
    t0 = time.monotonic()
    n_deep = len(top_configs)
    time_per = max(5.0, phase2_time / max(1, n_deep))

    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 600:
            break
        t_alloc = min(time_per, deadline - time.monotonic() - 600)
        if t_alloc < 3.0:
            break
        print("  [%d/%d] n=%d, %s, budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))

        # Alternating LP and gradient
        a_cur = a0.copy()
        sub_steps = 4
        sub_time = t_alloc / sub_steps

        for s in range(sub_steps):
            if time.monotonic() > deadline - 600:
                break
            tt = max(2.0, sub_time * (1 - 0.1 * s))  # slightly less time each sub-step
            if s % 2 == 0 and n <= 1024:
                a_cur, _ = lp_optimize(a_cur, time_limit=tt)
            else:
                a_cur, _ = gradient_optimize(a_cur, time_limit=tt)
            record(a_cur)

    print("  Deep opt done. Best: %.6f, time=%.0fs" % (best_score, time.monotonic() - t_start))

    # Phase 3: Peak-targeting local search
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("Phase 3: Peak local search...")
        phase3_time = min(600.0, deadline - time.monotonic() - 100)
        n_anneal = 5
        per_anneal = phase3_time / n_anneal

        for i in range(n_anneal):
            if time.monotonic() > deadline - 100:
                break
            a_start = best_sequence.copy()
            a_start = normalize(a_start)
            # Add small perturbation for diversity
            rng_a = np.random.default_rng(seed + 10000 + i)
            pert = rng_a.normal(0, 0.02, len(a_start))
            a_start = normalize(a_start + pert)

            a_opt, _ = peak_local_search(a_start, time_limit=per_anneal, rng=rng_a)
            record(a_opt)

        print("  Local search done. Best: %.6f, time=%.0fs" % (best_score, time.monotonic() - t_start))

    # Phase 4: Final polish with LP
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 10
        a_best = normalize(best_sequence.copy())
        n = len(a_best)

        if n <= 1024 and remaining > 30:
            half = remaining * 0.5
            a_opt, _ = lp_optimize(a_best, time_limit=half, t_init=0.1)
            record(a_opt)
            if time.monotonic() < deadline - 10:
                rem2 = deadline - time.monotonic() - 10
                rng_f = np.random.default_rng(seed + 99999)
                a_opt2, _ = peak_local_search(a_best, time_limit=rem2, rng=rng_f)
                record(a_opt2)
        elif remaining > 20:
            a_opt, _ = gradient_optimize(a_best, time_limit=remaining * 0.6)
            record(a_opt)
            if time.monotonic() < deadline - 10:
                rem2 = deadline - time.monotonic() - 10
                rng_f = np.random.default_rng(seed + 99999)
                a_opt2, _ = peak_local_search(a_best, time_limit=rem2, rng=rng_f)
                record(a_opt2)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))
    print("Total time: %.0fs" % (time.monotonic() - t_start))

    return [float(x) for x in best_sequence]
