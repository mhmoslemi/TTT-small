       e = np.exp(t)
       return e / np.sum(e)
   
   def power_mean_obj_grad(theta, p, S_target):

       a = softmax_normalize(theta) * S_target

"""
Optimized step function search using L-BFGS-B with power-mean annealing,
exact subgradient polishing, and multi-length exploration.
Key innovation: analytical gradient of the power-mean smoothed objective
with softmax parameterization, combined with scipy L-BFGS-B for efficient
local optimization.
"""
import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score efficiently."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def softmax_normalize(theta):
    """Numerically stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def power_mean_obj_grad(theta, p, S_target):
    """
    Compute F = 2n * (sum c^p)^(1/p) / S^2 and dF/dtheta
    where a = softmax(theta) * S_target, c = convolve(a, a).
    """
    a = softmax_normalize(theta) * S_target
    n = len(a)
    S = S_target

    c = fft_convolve(a, a)
    c = np.maximum(c, 1e-30)

    c_p = c ** p
    sum_cp = np.sum(c_p)
    if sum_cp < 1e-300:
        return 1e10, np.zeros(n)
    G = sum_cp ** (1.0 / p)
    F = 2.0 * n * G / (S * S)

    # Gradient w.r.t. a
    w = c ** (p - 1)
    S_grad = np.zeros(n)
    for j in range(n):
        # Term 1: sum_{m=0}^{n-1} w[m+j] * a[m]
        S_grad[j] += np.dot(w[j:j + n], a)
        # Term 2: sum_k w[k] * a[j-k] where 0<=j-k<n
        lo = max(0, j - n + 1)
        hi = min(j, 2 * n - 2)
        if lo <= hi:
            S_grad[j] += np.dot(w[lo:hi + 1], a[j - hi:j - lo + 1][::-1])

    # dG/da_j = (sum_cp)^(1/p - 1) * S_grad[j]
    dG_da = sum_cp ** (1.0 / p - 1.0) * S_grad

    # dF/da_j = 2n/S^2 * dG/da_j - 4n*G/S^3
    dF_da = (2.0 * n / (S * S)) * dG_da - (4.0 * n * G) / (S ** 3)

    # dF/dtheta_j = a_j * dF_da_j (scale invariance simplification)
    dF_dtheta = a * dF_da

    return F, dF_dtheta


def optimize_power_mean(a_init, p_schedule, time_limit, max_iter_per_p=150):
    """
    Run L-BFGS-B with power-mean objective, annealing p from low to high.
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a_init = a_init * S_target / np.sum(a_init)
    theta = np.log(a_init / np.sum(a_init) + 1e-15)

    best_theta = theta.copy()
    best_score = local_score(a_init)
    t0 = time.monotonic()
    n_evals = 0

    for p in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5.0:
            break

        def obj_wrapper(th):
            nonlocal n_evals
            n_evals += 1
            val, _ = power_mean_obj_grad(th, p, S_target)
            return val

        def grad_wrapper(th):
            nonlocal n_evals
            n_evals += 1
            _, g = power_mean_obj_grad(th, p, S_target)
            return g

        try:
            result = minimize(
                obj_wrapper, theta, jac=grad_wrapper,
                method='L-BFGS-B',
                options={'maxiter': max_iter_per_p, 'maxfun': max_iter_per_p * 3,
                         'gtol': 1e-6, 'iprint': -1}
            )
            theta = result.x.copy()
        except Exception:
            pass

        a = softmax_normalize(theta) * S_target
        a = np.clip(a, 0.0, 1000.0)
        score = local_score(a)
        if score < best_score:
            best_score = score
            best_theta = theta.copy()

        elapsed = time.monotonic() - t0
        print(f"    p={p}: score={score:.8f}, best={best_score:.8f}, time={elapsed:.1f}s")

    return best_theta, best_score


def exact_subgradient(a_init, time_limit, lr_scale=1.0):
    """
    Exact subgradient descent on 2n*max(c)/S^2 with projection.
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a = a_init.astype(np.float64).copy()
    a = a * S_target / np.sum(a)

    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(a, a)
        k_star = int(np.argmax(c))
        peak_val = c[k_star]
        S = np.sum(a)
        score = 2.0 * n * peak_val / (S * S)

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Subgradient of max(c) at k_star
        g = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= k_star - j < n:
                val += a[k_star - j]
            if 0 <= j - k_star < n:
                val += a[j - k_star]
            g[j] = 2.0 * val

        # Full subgradient of F = 2n*max(c)/S^2
        S = np.sum(a)
        grad_F = (2.0 * n / (S * S)) * g - (4.0 * n * peak_val) / (S ** 3)

        # Diminishing step size
        alpha = lr_scale * np.sqrt(2.0 * n) / (np.sqrt(step + 1) * (np.sum(a ** 2) / S + 1e-10))
        alpha = min(alpha, 1.0)

        a_new = a - alpha * grad_F
        # Project onto {x >= 0, sum(x) = S_target}
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = np.ones(n) / n * S_target
        else:
            a_new = a_new * S_target / s_new

        a = a_new

        if step % 500 == 0:
            print(f"    SG step={step}, score={score:.8f}, best={best_score:.8f}")

    return best_a, best_score


def multi_peak_shave(a_init, time_limit, n_peaks=5):
    """
    Minimize the sum of the top n_peaks of the convolution using L-BFGS.
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a_init = a_init * S_target / np.sum(a_init)
    theta = np.log(a_init / np.sum(a_init) + 1e-15)

    best_theta = theta.copy()
    best_score = local_score(a_init)
    t0 = time.monotonic()

    def topk_obj_grad(th):
        a = softmax_normalize(th) * S_target
        c = fft_convolve(a, a)
        c = np.maximum(c, 1e-30)
        # Use soft-max approximation of top-k sum
        tau = 2.0 / n
        shifted = (c - np.max(c)) / tau
        exp_shifted = np.exp(shifted)
        sum_exp = np.sum(exp_shifted)
        w = exp_shifted / sum_exp
        value = np.max(c) + tau * np.log(sum_exp)
        F = 2.0 * n * value / (S_target ** 2)

        # Gradient
        # reuse w as weights
        S_grad = np.zeros(n)
        for j in range(n):
            lo1 = max(0, j)
            hi1 = min(j + n, len(c))
            if lo1 < hi1:
                S_grad[j] += np.dot(w[lo1:hi1], a[j - lo1:hi1 - lo1][::-1] if lo1 > 0 else a[:hi1 - lo1])
            # Simpler: direct computation
            S_grad[j] = 0.0
            for k in range(len(c)):
                if 0 <= k - j < n:
                    S_grad[j] += w[k] * a[k - j]
                if 0 <= j - k < n:
                    S_grad[j] += w[k] * a[j - k]

        dF_da = (2.0 * n / (S_target ** 2)) * S_grad - (4.0 * n * value) / (S_target ** 3)
        dF_dtheta = a * dF_da
        return F, dF_dtheta

    # Use L-BFGS-B
    result = minimize(
        lambda th: topk_obj_grad(th)[0],
        theta,
        jac=lambda th: topk_obj_grad(th)[1],
        method='L-BFGS-B',
        options={'maxiter': 100, 'maxfun': 300, 'gtol': 1e-6, 'iprint': -1}
    )

    a = softmax_normalize(result.x) * S_target
    score = local_score(a)
    if score < best_score:
        best_score = score
        best_theta = result.x.copy()

    return best_theta, best_score


def make_good_init(n, rng):
    """Create a good initial sequence using structured broadband approach."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 20):
        freq = k * (1.0 + 0.7 * rng.random())
        phase = rng.uniform(0, 2 * np.pi)
        amp = 0.12 / np.sqrt(k)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    S = np.sqrt(2.0 * n)
    return a * S / np.sum(a)

        print(f"{'='*60}")
        a_perturbed = best_a * np.exp(rng.normal(0, 0.05, size=n_main))
        a_perturbed = np.clip(a_perturbed, 0.01, 1000.0)
        a_perturbed = a_perturbed * S_target / np.sum(a_perturbed)

        p_schedule2 = [10, 20, 35, 50, 80, 120, 200]
        t_phase1b = min(0.20 * time_budget, remaining - 30)
        if t_phase1b > 10:
            theta2, score2 = optimize_power_mean(a_perturbed, p_schedule2, t_phase1b, max_iter_per_p=80)
            a2 = softmax_normalize(theta2) * S_target
            a2 = np.clip(a2, 0.0, 1000.0)
            actual2 = local_score(a2)
            print(f"  Phase 1b result: {actual2:.8f}")
            if actual2 < best_score:
                best_score = actual2
                best_a = a2.copy()
                print(f"  *** New best: {best_score:.8f} ***")

    # === PHASE 2: Exact subgradient polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n{'='*60}")
        print(f"PHASE 2: Exact subgradient polishing")
        print(f"{'='*60}")
        t_phase2 = min(0.20 * time_budget, remaining - 20)
        if t_phase2 > 10:
            a_sg, score_sg = exact_subgradient(best_a, t_phase2, lr_scale=0.5)
            actual_sg = local_score(a_sg)
            print(f"  Subgradient result: {actual_sg:.8f}")
            if actual_sg < best_score:
                best_score = actual_sg
                best_a = a_sg.copy()
                print(f"  *** New best: {best_score:.8f} ***")

            # Second subgradient run with different lr
            a_sg2, score_sg2 = exact_subgradient(best_a, min(t_phase2, remaining - time.monotonic() - time.monotonic() + deadline - 20), lr_scale=0.2)
            actual_sg2 = local_score(a_sg2)
            if actual_sg2 < best_score:
                best_score = actual_sg2
                best_a = a_sg2.copy()
                print(f"  *** New best (SG2): {best_score:.8f} ***")

    # === PHASE 3: Another L-BFGS pass with very high p ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n{'='*60}")
        print(f"PHASE 3: High-p L-BFGS refinement")
        print(f"{'='*60}")
        p_schedule3 = [20, 40, 80, 150, 300]
        t_phase3 = min(0.10 * time_budget, remaining - 20)
        if t_phase3 > 10:
            theta3, score3 = optimize_power_mean(best_a, p_schedule3, t_phase3, max_iter_per_p=60)
            a3 = softmax_normalize(theta3) * S_target
            a3 = np.clip(a3, 0.0, 1000.0)
            actual3 = local_score(a3)
            print(f"  Phase 3 result: {actual3:.8f}")
            if actual3 < best_score:
                best_score = actual3
                best_a = a3.copy()
                print(f"  *** New best: {best_score:.8f} ***")

    # === PHASE 4: Try other lengths ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n{'='*60}")
        print(f"PHASE 4: Secondary length exploration")
        print(f"{'='*60}")
        for n_try in [320, 256, 448, 512]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            t_try = min(30.0, remaining - 15)
            if t_try < 10:
                break
            print(f"  Trying n={n_try}...")
            a_new, score_new = try_length(best_a, best_score, n_try, t_try, rng)
            if score_new < best_score:
                best_score = score_new
                best_a = a_new
                print(f"  *** New best at n={n_try}: {best_score:.8f} ***")
            else:
                print(f"  n={n_try}: {score_new:.8f} (no improvement)")

    # === FINAL: Quick subgradient polish if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n{'='*60}")
        print(f"FINAL POLISH (t={remaining:.0f}s)")
        print(f"{'='*60}")
        a_final, score_final = exact_subgradient(best_a, remaining - 5, lr_scale=0.3)
        actual_final = local_score(a_final)
        if actual_final < best_score:
            best_score = actual_final
            best_a = a_final
            print(f"  Final: {best_score:.8f}")

    # Construct final output
    best_a = np.clip(best_a, 0.0, 1000.0)
    final_list = [float(x) for x in best_a]

    # Verify
    final_eval = evaluate_sequence(final_list)
    print(f"\n{'='*60}")
    print(f"=== FINAL: score={final_eval:.8f}, n={len(final_list)}, local={best_score:.8f} ===")
    print(f"{'='*60}")

    if final_eval != np.inf and final_eval < best_score + 1e-6:
        return final_list

    # If evaluate_sequence gives different result, fall back
    if final_eval != np.inf:
        return final_list

    return [1.0] * n_main
