    """Compute soft max and its gradient."""
    b_shifted = b - np.max(b)
    exp_b = np.exp(b_shifted / tau)
    sum_exp = np.sum(exp_b)
    soft_max_val = tau * np.log(sum_exp) + np.max(b)
    # Gradient: d(soft_max)/d b[k] = (1/sum_exp) * exp_b[k]
    grad_weights = exp_b / sum_exp
    return soft_max_val, grad_weights
