def softmax_normalize(theta):
    """Numerically stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s

def objective_and_grad(theta, n, tau):
    """
    Objective and gradient for L-BFGS.
    Minimizes smoothed max conv, where a = softmax(theta).
    Score = 2n * max_conv / sum^2, but since sum=1 (softmax), score = 2n * max_conv.
    We return (max_conv_smoothed, grad_wrt_theta) to be used as the objective.
    """
    a = softmax_normalize(theta)
    value, grad_a = smoothed_max_conv_and_grad(a, tau)
    # grad wrt theta: a * (grad_a - dot(a, grad_a))
    inner = np.dot(a, grad_a)
    grad_theta = a * (grad_a - inner)
    return value, grad_theta

def optimize_lbfgs_stage(theta, n, tau, maxiter=200, gtol=1e-8):
    """Run one L-BFGS stage at fixed tau."""
    result = minimize(
        objective_and_grad,
        theta,
        args=(n, tau),
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': maxiter, 'gtol': gtol, 'maxfun': 500}
    )
    return result.x, result.fun

def optimize_adaptive_lbfgs(theta_init, n, time_budget, rng):
    """
    Adaptive L-BFGS with decreasing tau.
    """
    t0 = time.monotonic()
    theta = theta_init.copy()
    best_theta = theta.copy()
    best_score = np.inf
    
    # Tau schedule: start large, decrease geometrically
    tau_start = 5.0 / n
    tau_end = 0.01 / n
    n_stages = 15
    taus = tau_start * (tau_end / tau_start) ** (np.arange(n_stages) / (n_stages - 1))
    
    for i, tau in enumerate(taus):
        elapsed = time.monotonic() - t0
        if elapsed > time_budget * 0.85:
            break
        
        # Allocate more iterations to later (smaller tau) stages
        progress = i / n_stages
        maxiter = int(100 + 200 * progress)
        gtol = 1e-10 if i > n_stages // 2 else 1e-8
        
        result = minimize(
            objective_and_grad,
            theta,
            args=(n, tau),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': maxiter, 'gtol': gtol, 'maxfun': 500}
        )
        
        theta = result.x
        a = softmax_normalize(theta)
        score = 2.0 * n * np.max(fft_convolve_linear(a, a))
        
        if score < best_score:
            best_score = score
            best_theta = theta.copy()
        
        print(f"    Stage {i+1}/{n_stages}: tau={tau:.2e}, score={score:.6f}, "
              f"best={best_score:.6f}, t={time.monotonic()-t0:.1f}s")
    
    return best_theta, best_score

def coordinate_descent_polish(a_init, time_budget, rng):
    """
    Coordinate descent on the simplex for final polish.
    """
    a = a_init.copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    current_score = 2.0 * n * np.max(fft_convolve_linear(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    patience = 0
    max_patience = 2000
    
    while time.monotonic() - t0 < time_budget and patience < max_patience:
        r = rng.random()
        
        if r < 0.5:
            # Pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.005, 0.1)
            amt = a[i] * frac
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
            if candidate[i] < 0:
                continue
            candidate = candidate / np.sum(candidate)
            
        elif r < 0.75:
            # Single element scale
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.03))
            candidate = a.copy()
            candidate[j] *= factor
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            
        else:
            # Small block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 50))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.02, size=width))
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs
        
        new_score = 2.0 * n * np.max(fft_convolve_linear(candidate, candidate))
        if new_score < current_score:
            a = candidate
            current_score = new_score
            patience = 0
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        else:
            patience += 1
    
    return best_a

def spectral_init(n, rng, n_iter=50):
    """Spectral initialization."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    
    for it in range(n_iter):
        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        corr = np.fft.irfft(power, 2 * n)[:n]
        alpha = 1.0 / (n + 1.0)
        a = a - alpha * corr
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a

"""
L-BFGS based optimization minimizing peak autocorrelation of non-negative sequences.
Uses adaptive log-sum-exp smoothing with coarse-to-fine tau schedule,
starting from parent sequence with multiple n values and coordinate polish.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve_linear(a, b):
    """Linear convolution via FFT."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]


def softmax_normalize(theta):
    """Numerically stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def smoothed_max_conv_and_grad(a, tau):
    """
    Compute log-sum-exp smoothed max of conv(a,a) and gradient w.r.t. a.
    a must have sum=1, a>=0.
    """
    n = len(a)
    conv = fft_convolve_linear(a, a)
    c_max = np.max(conv)
    shifted = np.clip((conv - c_max) / tau, -50, 50)
    exp_vals = np.exp(shifted)
    log_sum_exp = np.log(exp_vals.sum())
    value = c_max + tau * log_sum_exp
    w = exp_vals / exp_vals.sum()

    # Gradient computation via FFT
    # f[m] = sum_k a[k] * w[k+m] (cross-correlation)
    # b[m] = sum_j a[j] * w[m-j] = (a*w)[m] (convolution)
    # grad = f + b
    L = 1
    while L < 3 * n - 2:
        L <<= 1
    Fa = np.fft.rfft(a, L)
    Fw = np.fft.rfft(w, L)
    f = np.fft.irfft(np.conj(Fa) * Fw, L)[:n]
    b = np.fft.irfft(Fa * Fw, L)[:n]
    grad = f + b
    return value, grad


def objective_and_grad(theta, n, tau):
    """Objective and gradient for L-BFGS on softmax parameterization."""
    a = softmax_normalize(theta)
    value, grad_a = smoothed_max_conv_and_grad(a, tau)
    inner = np.dot(a, grad_a)
    grad_theta = a * (grad_a - inner)
    return value, grad_theta


def optimize_adaptive_lbfgs(theta_init, n, time_budget, rng, n_stages=18):
    """
    Multi-stage L-BFGS with geometrically decreasing tau.
    """
    t0 = time.monotonic()
    theta = theta_init.copy()
    best_theta = theta.copy()
    best_score = 2.0 * n * np.max(fft_convolve_linear(softmax_normalize(theta), softmax_normalize(theta)))

    tau_start = 10.0 / n
    tau_end = 0.005 / n
    ratios = (tau_end / tau_start) ** (1.0 / (n_stages - 1))
    taus = [tau_start * (ratios ** i) for i in range(n_stages)]

    for i, tau in enumerate(taus):
        elapsed = time.monotonic() - t0
        if elapsed > time_budget * 0.9:
            break

        progress = i / max(1, n_stages - 1)
        maxiter = int(50 + 300 * progress)
        gtol = 1e-12 if progress > 0.5 else 1e-9

        try:
            result = minimize(
                objective_and_grad,
                theta,
                args=(n, tau),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter, 'gtol': gtol, 'maxfun': maxiter * 3}
            )
            theta = result.x
        except Exception as e:
            print(f"    L-BFGS error at stage {i}: {e}")
            break

        a = softmax_normalize(theta)
        score = 2.0 * n * np.max(fft_convolve_linear(a, a))

        if score < best_score:
            best_score = score
            best_theta = theta.copy()

        if (i + 1) % 3 == 0 or i == n_stages - 1:
            print(f"    Stage {i+1:2d}/{n_stages}: tau={tau:.2e}, score={score:.8f}, "
                  f"best={best_score:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_theta, best_score


def coordinate_descent_polish(a_init, time_budget, rng):
    """Coordinate descent on simplex for final precision."""
    a = a_init.copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    current_score = 2.0 * n * np.max(fft_convolve_linear(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    patience = 0
    max_patience = 5000

    while time.monotonic() - t0 < time_budget and patience < max_patience:
        r = rng.random()

        if r < 0.45:
            # Pair transfer
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            frac = rng.uniform(0.003, 0.08)
            amt = a[i] * frac
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
            if candidate[i] < 0:
                continue
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs

        elif r < 0.7:
            # Single element scale
            j = int(rng.integers(n))
            factor = np.exp(rng.normal(0, 0.02))
            candidate = a.copy()
            candidate[j] *= factor
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs

        elif r < 0.85:
            # Small block
            width = max(1, int(rng.integers(1, max(2, n // 64))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.015, size=width))
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs

        else:
            # Peak-targeted reduction
            conv = fft_convolve_linear(a, a)
            top_k = max(2, n // 16)
            peak_indices = np.argsort(conv)[-top_k:]
            # Reduce elements most contributing to peaks
            contrib = np.zeros(n)
            for pk in peak_indices[:5]:
                for j in range(n):
                    k = pk - j
                    if 0 <= k < n:
                        contrib[j] += a[j] * a[k]
            top_contrib = np.argsort(contrib)[-max(3, n // 32):]
            candidate = a.copy()
            candidate[top_contrib] *= (1.0 - rng.uniform(0.005, 0.03, size=len(top_contrib)))
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs

        new_score = 2.0 * n * np.max(fft_convolve_linear(candidate, candidate))
        if new_score < current_score * (1.0 + 1e-12):
            a = candidate
            current_score = new_score
            patience = 0
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        else:
            patience += 1

    return best_a


def spectral_init(n, rng, n_iter=60):
    """Spectral iterative initialization."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    for _ in range(n_iter):
        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        corr = np.fft.irfft(power, 2 * n)[:n]
        alpha = 1.0 / (n + 1.0)
        a = a - alpha * corr
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def multi_scale_init(n_target, rng, n_base=64, n_iter=100):
    """Build initialization by multi-scale spectral method."""
    # Start small and grow
    a = 1.0 + 0.2 * rng.standard_normal(n_base)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)

    for _ in range(30):
        A = np.fft.rfft(a, 2 * n_base)
        corr = np.fft.irfft(np.abs(A) ** 2, 2 * n_base)[:n_base]
        a = a - (1.0 / (n_base + 1.0)) * corr
        a = np.clip(a, 0, None)
        s = np.sum(a)
        a = a / s if s > 1e-10 else np.ones(n_base) / n_base

    # Upsample to target
    if n_target != n_base:
        x_old = np.linspace(0, 1, n_base)
        x_new = np.linspace(0, 1, n_target)
        a = np.interp(x_new, x_old, a)
        a = np.clip(a, 0.001, None)
        a = a / np.sum(a)
        # Refine at target size
        for _ in range(n_iter):
            A = np.fft.rfft(a, 2 * n_target)
            corr = np.fft.irfft(np.abs(A) ** 2, 2 * n_target)[:n_target]
            a = a - (1.0 / (n_target + 1.0)) * corr
            a = np.clip(a, 0, None)
            s = np.sum(a)
            a = a / s if s > 1e-10 else np.ones(n_target) / n_target

    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-stage L-BFGS optimization with adaptive log-sum-exp smoothing.
    Focuses on n=512 (parent length) and n=1024, with multiple initializations
    and coordinate descent polish.
    """
    t_global = time.monotonic()
    deadline = t_global + max(60.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load parent
    parent = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.001, 1000.0)
        p_score = 2.0 * len(parent) * np.max(fft_convolve_linear(parent, parent)) / (np.sum(parent) ** 2)
        print(f"Parent: n={len(parent)}, score={p_score:.8f}")
        if p_score < best_score:
            best_score = p_score
            best_seq = parent.copy()

    # === Phase 1: L-BFGS on parent (n=512) ===
    print("\n=== Phase 1: L-BFGS at n=512 ===")
    n1 = 512
    if parent is not None:
        a_parent = parent / np.sum(parent)
        # Try several small perturbations
        for trial in range(3):
            if time.monotonic() > deadline - 400:
                break
            if trial == 0:
                a0 = a_parent.copy()
            else:
                a0 = a_parent * (1.0 + 0.01 * rng.standard_normal(n1))
                a0 = np.clip(a0, 0.001, None)
                a0 = a0 / np.sum(a0)

            theta0 = np.log(a0 + 1e-12)
            time_alloc = min(400.0, (deadline - time.monotonic() - 350) / (4 - trial))
            if time_alloc < 60:
                time_alloc = 60

            theta_best, _ = optimize_adaptive_lbfgs(theta0, n1, time_alloc, rng, n_stages=18)
            a_best = softmax_normalize(theta_best)
            sc = 2.0 * n1 * np.max(fft_convolve_linear(a_best, a_best))
            print(f"  Trial {trial}: score={sc:.8f}")
            if sc < best_score:
                best_score = sc
                best_seq = a_best.copy()
                print(f"  *** New best: {best_score:.8f} ***")

    # === Phase 2: Try n=1024 ===
    print(f"\n=== Phase 2: L-BFGS at n=1024 ===")
    if time.monotonic() < deadline - 250:
        n2 = 1024
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n2)
            a_init2 = np.interp(x_new, x_old, best_seq)
        else:
            a_init2 = spectral_init(n2, rng, n_iter=80)
        a_init2 = np.clip(a_init2, 0.001, None)
        a_init2 = a_init2 / np.sum(a_init2)
        theta2 = np.log(a_init2 + 1e-12)
        time_alloc = min(400.0, deadline - time.monotonic() - 200)
        if time_alloc > 30:
            theta_best2, _ = optimize_adaptive_lbfgs(theta2, n2, time_alloc, rng, n_stages=15)
            a_best2 = softmax_normalize(theta_best2)
            sc2 = 2.0 * n2 * np.max(fft_convolve_linear(a_best2, a_best2))
            print(f"  n=1024: score={sc2:.8f}")
            if sc2 < best_score:
                best_score = sc2
                best_seq = a_best2.copy()
                print(f"  *** New best: {best_score:.8f} ***")

    # === Phase 3: Alternative initializations at n=512 ===
    print(f"\n=== Phase 3: Alternative inits at n=512 ===")
    n3 = 512
    inits = []
    t_arr = np.linspace(0, 1, n3)
    inits.append(('spec', spectral_init(n3, rng, n_iter=100)))
    inits.append(('cos_hi', 1.0 + 0.4 * np.cos(2 * np.pi * t_arr)))
    inits.append(('cos_lo', 1.4 - 0.4 * np.cos(2 * np.pi * t_arr)))
    inits.append(('sine2', 1.0 + 0.3 * np.sin(np.pi * t_arr) ** 2))
    inits.append(('sqrt', 1.0 + 0.2 * np.sqrt(t_arr) + 0.2 * np.sqrt(1 - t_arr)))
    inits.append(('multi', multi_scale_init(n3, rng)))

    for idx, (name, a0) in enumerate(inits):
        if time.monotonic() > deadline - 120:
            break
        a0 = np.clip(a0, 0.001, None)
        a0 = a0 / np.sum(a0)
        theta0 = np.log(a0 + 1e-12)
        time_alloc = min(150.0, (deadline - time.monotonic() - 100) / max(1, len(inits) - idx))
        if time_alloc < 40:
            break
        theta_best, _ = optimize_adaptive_lbfgs(theta0, n3, time_alloc, rng, n_stages=12)
        a_best = softmax_normalize(theta_best)
        sc = 2.0 * n3 * np.max(fft_convolve_linear(a_best, a_best))
        print(f"  '{name}': score={sc:.8f}")
        if sc < best_score:
            best_score = sc
            best_seq = a_best.copy()
            print(f"  *** New best: {best_score:.8f} ***")

    # === Phase 4: L-BFGS restart from best with new random perturbation ===
    if best_seq is not None and time.monotonic() < deadline - 100:
        print(f"\n=== Phase 4: Restart from best ===")
        a_best_norm = best_seq / np.sum(best_seq)
        n_best = len(a_best_norm)
        # Small perturbation + re-optimize with fine tau
        for trial in range(2):
            if time.monotonic() > deadline - 60:
                break
            a_pert = a_best_norm * (1.0 + 0.005 * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.001, None)
            a_pert = a_pert / np.sum(a_pert)
            theta_pert = np.log(a_pert + 1e-12)
            time_alloc = min(120.0, deadline - time.monotonic() - 50)
            if time_alloc < 30:
                break
            theta_best, _ = optimize_adaptive_lbfgs(theta_pert, n_best, time_alloc, rng, n_stages=10)
            a_final = softmax_normalize(theta_best)
            sc = 2.0 * n_best * np.max(fft_convolve_linear(a_final, a_final))
            print(f"  Restart {trial}: score={sc:.8f}")
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"  *** New best: {best_score:.8f} ***")

    # === Phase 5: Coordinate descent polish ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Phase 5: Coordinate descent polish ===")
        print(f"  Score before: {best_score:.8f}")
        polish_time = min(300.0, deadline - time.monotonic() - 5)
        if polish_time > 20:
            a_norm = best_seq / np.sum(best_seq)
            polished = coordinate_descent_polish(a_norm, polish_time, rng)
            final_score = 2.0 * len(polished) * np.max(fft_convolve_linear(polished, polished))
            print(f"  Score after: {final_score:.8f}")
            if final_score < best_score:
                best_score = final_score
                best_seq = polished.copy()
                print(f"  *** New best: {best_score:.8f} ***")

    # Final evaluation
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.8f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if parent is not None:
        return [float(max(0.0, min(1000.0, x))) for x in parent]
    return [1.0] * 256
