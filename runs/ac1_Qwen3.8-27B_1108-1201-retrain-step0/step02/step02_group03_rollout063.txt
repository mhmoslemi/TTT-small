```python
"""
Hybrid non-negative step-function optimizer.

The objective is scale invariant:
    F(a) = 2*n*max(conv(a,a)) / sum(a)^2.
All local scores are computed with sum(a)=1.

The main search combines:
  1. Spectral / Gerchberg-Saxon and smooth-random initializations.
  2. Active-set primal LP:
       min t
       s.t. sum(b)=1, b>=0,
            (a*b)[k] <= t for the currently active convolution lags k.
     This gives a strong linear direction b that avoids the current peaks.
  3. Simple line-search mixing a <- mix(a,b), randomness, and reset-to-best.
  4. Occasionally, adaptive top-lag softmax subgradient descent on a softmax
     parameterization to polish dense candidates.

Scipy is used for the active LP when available. If unavailable, a heuristic
direction replaces it.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog, Bounds
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False

SAFE_TARGET = 1.5025


def next_pow2(x):
    y = 1
    while y < x:
        y <<= 1
    return y


def item_score(item):
    return item[0]


def normalize_prob(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    n = a.size
    if s <= 1e-300 or n == 0:
        if n == 0:
            return np.ones(1, dtype=np.float64)
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    a = a / s
    a = np.clip(a, 0.0, None)
    s2 = float(np.sum(a))
    if s2 <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return a / s2


def full_conv(a):
    n = a.size
    target = 2 * n - 1
    size = next_pow2(target)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:target]


def score_prob(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    n = a.size
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s <= 1e-300:
        return np.inf
    a = a / s
    c = full_conv(a)
    maxc = float(np.max(c))
    if not np.isfinite(maxc) or maxc <= 0.0:
        return np.inf
    return float(2.0 * n * maxc)


def update_best(cur_seq, cur_score, cand):
    if cand is None:
        return cur_seq, cur_score
    try:
        s = score_prob(cand)
    except Exception:
        return cur_seq, cur_score
    if np.isfinite(s) and s < cur_score:
        return np.asarray(cand, dtype=np.float64).copy(), float(s)
    return cur_seq, cur_score


def smooth_random_init(n, rng, rho=0.78):
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    vals = np.empty(n, dtype=np.float64)
    x = 0.0
    one_minus = 1.0 - rho
    for i in range(n):
        x = rho * x + one_minus * rng.standard_normal()
        vals[i] = x
    a = 1.0 + 0.9 * vals
    a = np.clip(a, 0.01, None)
    return normalize_prob(a)


def resample_seq(src, n, rng=None, noise=0.0):
    src = np.asarray(src, dtype=np.float64)
    m = src.size
    if m == 0:
        return np.ones(max(1, n), dtype=np.float64) / max(1, n)
    if m == n:
        a = src.copy()
    else:
        xold = np.linspace(0.0, 1.0, m)
        xnew = np.linspace(0.0, 1.0, n)
        a = np.interp(xnew, xold, src)
    if rng is not None and noise > 0.0:
        a = a * np.exp(rng.normal(0.0, noise, size=n))
    return normalize_prob(a)


def gerchberg_init(n, rng, n_iter=30):
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    target = 2 * n - 1
    size = next_pow2(target)
    a = np.full(n, 1.0 / float(n), dtype=np.float64)
    for _ in range(n_iter):
        fa = np.fft.rfft(a, size)
        mag = np.abs(fa)
        mag[mag < 1e-12] = 1e-12
        phase = fa / mag
        afull = np.fft.irfft(phase, size)[:target]
        a = np.clip(np.real(afull[:n]), 0.0, None)
        s = float(np.sum(a))
        if s < 1e-10:
            a = np.full(n, 1.0 / float(n), dtype=np.float64)
        else:
            a = a / s
    return a


def spectral_sqrt_init(n, rng, iters=20, damping=0.25):
    """
    Iterate a pointwise Newton step toward P^2 ~= flat target coefficient
    polynomial, with nonnegativity projection. Used only for initialization.
    """
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    target = 2 * n - 1
    size = next_pow2(target)
    tgt = np.full(target, 1.0 / float(target), dtype=np.float64)
    H = np.fft.rfft(tgt, size).astype(np.complex128)

    a = np.full(n, 1.0 / float(n), dtype=np.float64)
    P = np.fft.rfft(a, size).astype(np.complex128)
    P = 0.5 * P + 0.5 * np.sqrt(H + 1e-8)

    for _ in range(max(1, int(iters))):
        afull = np.fft.irfft(P, size)[:target]
        a = np.clip(np.real(afull[:n]), 0.0, None)
        s = float(np.sum(a))
        if s < 1e-10:
            a = np.full(n, 1.0 / float(n), dtype=np.float64)
        else:
            a = a / s
        P = np.fft.rfft(a, size).astype(np.complex128)
        Psafe = P.copy()
        mask = np.abs(Psafe) < 1e-7
        Psafe[mask] = 1e-7 + 0.0j
        Pnew = 0.5 * (P + H / Psafe)
        maxn = float(np.max(np.abs(Pnew)))
        if maxn > 50.0:
            Pnew = Pnew * (50.0 / maxn)
        P = (1.0 - damping) * P + damping * Pnew

    afull = np.fft.irfft(P, size)[:target]
    a = np.clip(np.real(afull[:n]), 0.0, None)
    return normalize_prob(a)


def make_start_candidates(n, base_seq, rng):
    raw = []
    if base_seq is not None:
        a0 = resample_seq(base_seq, n, rng, 0.02)
        raw.append(a0)
        raw.append(resample_seq(base_seq, n, rng, 0.12))
        if n > 2:
            roll = int(rng.integers(1, n))
            raw.append(normalize_prob(np.roll(a0, roll)))

    if n <= 4096:
        giter = 30
        if n > 512:
            giter = 15
        if n > 2048:
            giter = 8
        raw.append(gerchberg_init(n, rng, giter))

    raw.append(smooth_random_init(n, rng))
    raw.append(np.full(n, 1.0 / float(n), dtype=np.float64))

    if n <= 1536:
        sir = 18
        if n > 1024:
            sir = 10
        raw.append(spectral_sqrt_init(n, rng, sir))

    scored = []
    for a in raw:
        try:
            s = score_prob(a)
        except Exception:
            s = np.inf
        scored.append((s, a))

    scored.sort(key=item_score)
    out = []
    for it in scored:
        if np.isfinite(it[0]):
            out.append(it[1])
        if len(out) == 2:
            break
    if not out:
        out.append(np.full(n, 1.0 / float(n), dtype=np.float64))
    return out


def choose_active_lags(c, k, rng):
    m = c.size
    k = int(k)
    if k <= 0:
        return np.zeros(0, dtype=np.int64)
    if k >= m:
        return np.argsort(c)[::-1].astype(np.int64)

    top = np.argpartition(c, -k)[-k:].astype(np.int64)
    repl = max(1, k // 4)
    if k > 3 and repl < k - 1 and rng.random() < 0.6:
        pool_k = min(m, 2 * k)
        pool = np.argpartition(c, -pool_k)[-pool_k:].astype(np.int64)
        keep = top[:k - repl]
        extra = int(min(len(pool), repl * 3))
        if extra > 0:
            randoms = rng.choice(pool, size=extra, replace=False)
            cand = np.unique(np.concatenate((keep, randoms)))
            if cand.size >= k:
                order = np.argsort(c[cand])[::-1]
                top = cand[order[:k]]
            else:
                cand = np.unique(np.concatenate((cand, top)))
                if cand.size > k:
                    order = np.argsort(c[cand])[::-1]
                    top = cand[order[:k]]

    order = np.argsort(c[top])[::-1]
    return top[order].astype(np.int64)


def response_direct(a, active, weights):
    n = a.size
    p = np.zeros(n, dtype=np.float64)
    for idx, k in enumerate(active):
        w = weights[idx]
        if w == 0.0:
            continue
        jmin = max(0, int(k) - n + 1)
        jmax = min(n - 1, int(k))
        if jmax < jmin:
            continue
        low = int(k) - jmax
        high = int(k) - jmin
        p[jmin:jmax + 1] += w * a[low:high + 1][::-1]
    return p


def heuristic_direction(a, active):
    n = a.size
    p = response_direct(a, active, np.ones(len(active), dtype=np.float64))
    med = float(np.median(p))
    if np.max(p) - med <= 1e-15:
        med = 0.0
    denom = float(np.max(p - med)) + 1e-15
    b = np.exp(-0.8 * (p - med) / denom)
    b = np.clip(b, 1e-8, None)
    return normalize_prob(b)


def solve_active_primal(f, active):
    if not HAVE_SCIPY:
        return None
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    active = np.asarray(active, dtype=np.int64)
    if active.size == 0:
        return None
    active = np.unique(active)
    K = int(active.size)

    max_dense_entries = 2_200_000
    if K * (n + 1) > max_dense_entries:
        kmax = max(1, max_dense_entries // (n + 1))
        active = active[:kmax]
        K = int(active.size)

    try:
        A = np.zeros((K, n + 1), dtype=np.float64)
        for idx, k in enumerate(active):
            kk = int(k)
            jmin = max(0, kk - n + 1)
            jmax = min(n - 1, kk)
            low = kk - jmax
            high = kk - jmin
            A[idx, jmin:jmax + 1] = f[low:high + 1][::-1]
            A[idx, n] = -1.0

        c = np.zeros(n + 1, dtype=np.float64)
        c[n] = 1.0
        Aeq = np.zeros((1, n + 1), dtype=np.float64)
        Aeq[0, :n] = 1.0
        b_eq = np.ones(1, dtype=np.float64)

        lb = np.zeros(n + 1, dtype=np.float64)
        ub = np.ones(n + 1, dtype=np.float64)
        ub[n] = np.inf
        bounds = Bounds(lb, ub)

        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=np.zeros(K, dtype=np.float64),
                A_eq=Aeq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
            )
        except Exception:
            res = linprog(
                c,
                A_ub=A,
                b_ub=np.zeros(K, dtype=np.float64),
                A_eq=Aeq,
                b_eq=b_eq,
                bounds=bounds,
            )
    except Exception:
        return None

    if res is not None and getattr(res, "success", False):
        b = np.asarray(res.x[:n], dtype=np.float64)
        b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
        b = np.clip(b, 0.0, None)
        s = float(np.sum(b))
        if s > 1e-12:
            t = float(getattr(res, "fun", 0.0))
            return (b / s, t)
    return None


def softmax_normalize(theta):
    theta = np.asarray(theta, dtype=np.float64)
    n = theta.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    t = theta - float(np.max(theta))
    e = np.exp(t)
    s = float(np.sum(e))
    if s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    a = e / s
    floor = 1e-12 / float(n)
    a = np.maximum(a, floor)
    s2 = float(np.sum(a))
    if s2 <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return a / s2


def gradient_refine(a_init, time_limit, rng, stop_threshold=None):
    if time_limit <= 0:
        a = normalize_prob(a_init)
        return a.copy(), score_prob(a)

    start = time.monotonic()
    deadline = start + max(0.0, float(time_limit))
    a = normalize_prob(a_init)
    n = a.size

    if k_start is None:
        k_start = int(min(64, max(16, n // 32)))

    c = full_conv(a)
    current_score = float(2.0 * n * np.max(c))
    best = a.copy()
    best_score = current_score

    if stop_threshold is not None and current_score <= float(stop_threshold):
        return best, best_score

    theta = np.log(a + 1e-300)
    m1 = np.zeros(n, dtype=np.float64)
    v1 = np.zeros(n, dtype=np.float64)
    step = 0
    lr = 0.7
    k = k_start
    no_improve = 0

    active = choose_active_lags(c, k, rng)
    ctop = c[active]
    spread = float(np.max(ctop) - np.min(ctop))
    if spread <= 1e-15:
        spread = 1.0 / float(n)
    tau = spread / 4.0
    w = np.exp((ctop - float(np.max(ctop))) / tau)
    w = w / float(np.sum(w))

    while time.monotonic() < deadline:
        p = response_direct(a, active, w)
        g = float(np.dot(a, p))
        grad = a * (p - g)

        step += 1
        m1 = 0.85 * m1 + 0.15 * grad
        v1 = 0.999 * v1 + 0.001 * (grad * grad)
        mhat = m1 / (1.0 - 0.85 ** step)
        vhat = v1 / (1.0 - 0.999 ** step)
        eff_lr = lr / np.sqrt(1.0 + 0.001 * step)
        theta = theta - eff_lr * mhat / (np.sqrt(vhat) + 1e-12)

        if step % 100 == 0:
            theta -= float(np.max(theta))

        a = softmax_normalize(theta)

        if step % 10 == 0:
            c = full_conv(a)
            current_score = float(2.0 * n * np.max(c))
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if current_score > best_score * 1.05:
                lr *= 0.6
            elif current_score < best_score * (1.0 - 1e-5):
                lr = min(1.5, lr * 1.15)
            if lr < 0.05:
                lr = 0.05

            if no_improve > 20 and k < 3 * k_start:
                k = min(3 * k_start, int(k * 1.5) + 1)
                no_improve = 0

            active = choose_active_lags(c, k, rng)
            ctop = c[active]
            spread = float(np.max(ctop) - np.min(ctop))
            if spread <= 1e-15:
                spread = 1.0 / float(n)
            tau = spread / 4.0
            w = np.exp((ctop - float(np.max(ctop))) / tau)
            w = w / float(np.sum(w))

            if (no_improve > 0) and (no_improve % 30 == 0):
                theta += rng.normal(0.0, 0.05, size=n)
                theta -= float(np.max(theta))

            if stop_threshold is not None and best_score <= float(stop_threshold):
                break

    c = full_conv(best)
    s = float(2.0 * n * np.max(c))
    if np.isfinite(s) and s < best_score:
        best_score = s
    return best, best_score


def search_with_lp(a_init, time_limit, rng, stop_threshold=None):
    if time_limit <= 0:
        a = normalize_prob(a_init)
        return a.copy(), score_prob(a)

    start = time.monotonic()
    deadline = start + max(0.0, float(time_limit))
    a = normalize_prob(a_init)
    n = a.size

    c = full_conv(a)
    current_score = float(2.0 * n * np.max(c))
    best = a.copy()
    best_score = current_score

    if stop_threshold is not None and best_score <= float(stop_threshold):
        return best, best_score

    if n <= 256:
        max_active = 96
        start_active = 32
    elif n <= 512:
        max_active = 128
        start_active = 36
    elif n <= 1024:
        max_active = 170
        start_active = 48
    elif n <= 2048:
        max_active = 220
        start_active = 64
    else:
        max_active = 256
        start_active = 80

    active_count = min(max_active, start_active)
    stall = 0
    alphas = (0.35, 0.6, 1.0)
    last_grad = start

    while time.monotonic() < deadline:
        if stop_threshold is not None and best_score <= float(stop_threshold):
            break

        c = full_conv(a)
        current_score = float(2.0 * n * np.max(c))
        if not np.isfinite(current_score):
            current_score = np.inf

        if current_score < best_score:
            best_score = current_score
            best = a.copy()

        active = choose_active_lags(c, active_count, rng)
        lp_res = solve_active_primal(a, active)
        if lp_res is None:
            b = heuristic_direction(a, active)
            tlp = None
        else:
            b, tlp = lp_res

        if b is None:
            b = a.copy()
            tlp = None
        else:
            b = normalize_prob(b)

        best_cand = a
        best_cand_score = current_score
        best_cand_is_current = True

        for alpha in alphas:
            x = (1.0 - alpha) * a + alpha * b
            if alpha >= 0.9 and rng.random() < 0.25:
                x = x * np.exp(rng.normal(0.0, 0.03, size=n))
            x = normalize_prob(x)
            xs = score_prob(x)
            if np.isfinite(xs) and xs < best_cand_score:
                best_cand_score = xs
                best_cand = x
                best_cand_is_current = False

        pos_count = int(np.sum(b > 0.0))
        if pos_count < max(10, n // 2):
            x = np.power(b, 0.7)
            x = normalize_prob(x)
            xs = score_prob(x)
            if np.isfinite(xs) and xs < best_cand_score:
                best_cand_score = xs
                best_cand = x
                best_cand_is_current = False

        if not best_cand_is_current and best_cand_score < current_score:
            a = best_cand
            current_score = best_cand_score
            stall = 0
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            stall += 1
            close_ok = (not best_cand_is_current) and (
                np.isfinite(best_cand_score) and best_cand_score < current_score * 1.02
            )
            promising = False
            if tlp is not None and np.isfinite(current_score) and current_score > 0.0:
                current_lag = current_score / (2.0 * float(n))
                promising = float(tlp) < 0.95 * current_lag

            if close_ok and (stall > 4 or rng.random() < 0.15 or promising):
                a = best_cand
                current_score = best_cand_score
            else:
                a = a * np.exp(rng.normal(0.0, 0.04, size=n))
                a = normalize_prob(a)
                current_score = score_prob(a)
                if not np.isfinite(current_score):
                    a = normalize_prob(best)
                    current_score = best_score

            if stall % 3 == 0 and active_count < max_active:
                active_count = min(max_active, int(active_count * 1.5) + 1)

            if stall > 12:
                a = best * np.exp(rng.normal(0.0, 0.15, size=n))
                a = normalize_prob(a)
                current_score = score_prob(a)
                if not np.isfinite(current_score):
                    a = normalize_prob(best)
                    current_score = best_score
                stall = 0

        now = time.monotonic()
        if now - last_grad > 45.0 and (deadline - now) > 60.0:
            gt = min(12.0, max(0.0, deadline - now - 10.0))
            if gt > 5.0:
                a2, s2 = gradient_refine(
                    a, gt, rng, stop_threshold=SAFE_TARGET
                )
                if np.isfinite(s2) and s2 < current_score:
                    a = a2
                    current_score = s2
                    stall = 0
                if np.isfinite(s2) and s2 < best_score:
                    best_score = s2
                    best = a2.copy()
            last_grad = time.monotonic()

    remaining = deadline - time.monotonic()
    if remaining > 20.0:
        gt = min(50.0, max(0.0, remaining - 5.0))
        if gt > 5.0:
            a2, s2 = gradient_refine(
                best, gt, rng, stop_threshold=SAFE_TARGET
            )
            if np.isfinite(s2) and s2 < best_score:
                best = a2.copy()
                best_score = s2

    return best, float(best_score)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Returns a list of non-negative heights.
    """
    t0 = time.monotonic()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 2400.0
    if not np.isfinite(total_budget) or total_budget <= 120.0:
        total_budget = 2400.0

    deadline = t0 + max(120.0, total_budget - 100.0)
    rng = np.random.default_rng(int(seed) if seed is not None else 42)

    best_seq = None
    best_score = np.inf

    init_seq = globals().get("height_sequence_1", None)
    if init_seq is not None:
        try:
            arr = np.asarray(init_seq, dtype=np.float64)
            arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                s0 = score_prob(arr)
                if np.isfinite(s0):
                    best_seq = normalize_prob(arr)
                    best_score = float(s0)
                    print(f"Initial provided sequence: n={arr.size}, local_score={best_score:.6f}")
                else:
                    best_seq = smooth_random_init(256, rng)
                    best_score = score_prob(best_seq)
                    print(f"Initial invalid, using random n=256 score={best_score:.6f}")
        except Exception as e:
            print("Failed to process provided sequence:", e)
            best_seq = smooth_random_init(256, rng)
            best_score = score_prob(best_seq)
            print(f"Fallback n=256 score={best_score:.6f}")

    if best_seq is None:
        best_seq = smooth_random_init(256, rng)
        best_score = score_prob(best_seq)
        print(f"No initial sequence, random n=256 score={best_score:.6f}")

    # Factors are approximate budget shares for each length.
    task_plan = [
        (512, 0.31),
        (1024, 0.28),
        (256, 0.17),
        (2048, 0.15),
        (4096, 0.09),
    ]

    available0 = max(60.0, deadline - time.monotonic())

    for n, frac in task_plan:
        if time.monotonic() > deadline - 160.0:
            break
        if best_score < SAFE_TARGET:
            break

        now = time.monotonic()
        available = max(0.0, deadline - now - 80.0)
        desired = available * frac
        if desired < 45.0:
            continue
        if desired > 0.35 * available:
            desired = 0.35 * available

        # If the requested length is already very large, do not allocate too much.
        if n >= 2048 and desired > 0.22 * available:
            desired = 0.22 * available

        if desired < 35.0:
            continue

        starts = make_start_candidates(n, best_seq, rng)
        if not starts:
            starts = [smooth_random_init(n, rng)]

        time_left_task = min(desired, deadline - time.monotonic() - 100.0)
        if time_left_task < 45.0:
            break

        num_starts = 1
        if n < 2048 and time_left_task > 130.0:
            num_starts = 2
        num_starts = min(num_starts, len(starts))
        per_start = time_left_task / float(num_starts)

        for idx in range(num_starts):
            if time.monotonic() > deadline - 120.0:
                break
            if best_score < SAFE_TARGET:
                break

            this_limit = min(per_start, deadline - time.monotonic() - 120.0)
            if this_limit < 25.0:
                break

            st = starts[idx]
            s_start = score_prob(st)
            print(
                f"Task n={n}, start {idx + 1}/{num_starts}, "
                f"limit={this_limit:.1f}s, start_score={s_start:.6f}"
            )

            a_opt, s_opt = search_with_lp(
                st, this_limit, rng, stop_threshold=SAFE_TARGET
            )

            a_opt, s_opt = update_best(best_seq, best_score, a_opt)
            if np.isfinite(s_opt):
                print(f"  n={n}, start {idx + 1}: optimum={s_opt:.6f}, best={best_score:.6f}")
            else:
                print(f"  n={n}, start {idx + 1}: no finite score")

    # Final polish on the global best if time remains.
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 25.0:
            final_limit = max(10.0, rem - 20.0)
            try:
                a_pol, s_pol = gradient_refine(
                    best_seq, final_limit, rng, stop_threshold=SAFE_TARGET
                )
                a_pol, s_pol = update_best(best_seq, best_score, a_pol)
                if np.isfinite(s_pol):
                    print(f"After final polish: best={best_score:.6f}, n={len(best_seq)}")
            except Exception as e:
                print("Final polish failed:", e)

    if best_seq is None:
        best_seq = smooth_random_init(256, rng)

    # Ensure a Python list of plain floats, nonnegative and bounded as evaluator expects.
    final_list = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_seq, dtype=np.float64).ravel()]

    try:
        fs = evaluate_sequence(final_list)
        print(f"FINAL evaluate_sequence score={fs:.6f}, n={len(final_list)}")
        if not np.isfinite(fs):
            # As a last resort, try a simple known-valid sequence.
            fallback = [1.0 / 256.0] * 256
            return fallback
    except Exception as e:
        print("Final evaluate_sequence failed:", e)

    return final_list
```
