```python
"""
FFT-accelerated optimization for minimum peak self-convolution non-negative sequences.
Uses log-sum-exp smoothing with multi-start L-BFGS, momentum gradient descent,
and FFT-based local search with targeted peak reduction.
"""

import numpy as np
import time
from scipy.optimize import minimize

# All helper functions at top level, no closures, no lambdas

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1 << (n + m - 2).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s

def score_normalized(a):
    """Score for a sequence normalized to sum=1: 2n * max(conv(a,a))"""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

def lse_obj_grad(theta, tau):
    """
    Log-sum-exp smoothed objective and its gradient with respect to theta.
    Objective: 2n * (1/tau) * log(sum_k exp(tau * c[k]))
    where c = convolve(softmax(theta), softmax(theta))
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    c_max = np.max(c)
    shifted = tau * (c - c_max)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    
    # Value: 2n * (c_max + (1/tau)*log(sum_exp))
    val = 2.0 * n * (c_max + np.log(sum_exp) / tau)
    
    # Weights
    w = exp_shifted / sum_exp
    
    # Gradient of L w.r.t. a:
    # dL/da[i] = 2n * sum_k w[k] * dc[k]/da[i] = 2n * 2 * (w ⊛ a)[i] = 4n * (w ⊛ a)[i]
    # where (w ⊛ a)[i] = sum_k w[k] * a[k-i]
    
    N = 1 << (2 * n - 2).bit_length()
    Fa = np.fft.rfft(a, N)
    Fw = np.fft.rfft(w, N)
    
    # Cross-correlation (w ⊛ a)[i] = sum_k w[k] * a[k-i]
    # DFT: F{(w ⊛ a)} = Fw * conj(Fa)
    grad_a = 4.0 * n * np.fft.irfft(Fw * np.conj(Fa), N)[:n]
    
    # Gradient w.r.t. theta (softmax chain rule)
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return val, grad_theta

def lse_minimize(theta0, tau_schedule, deadline):
    """Run L-BFGS with annealing tau schedule."""
    best_theta = theta0.copy()
    a = softmax(theta0)
    best_score = score_normalized(a)
    
    for tau in tau_schedule:
        if time.monotonic() > deadline - 10:
            break
        result = minimize(
            lse_obj_grad, best_theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(tau,),
            options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-14, 'maxfun': 1500}
        )
        a_new = softmax(result.x)
        s = score_normalized(a_new)
        if s < best_score:
            best_score = s
            best_theta = result.x.copy()
    
    return best_theta, best_score

def momentum_gradient_descent(a0, tau_schedule, deadline, lr_init=0.1, momentum=0.9):
    """
    Momentum gradient descent on the log-sum-exp objective,
    working directly on a (with simplex projection).
    """
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    v = np.zeros(n)
    lr = lr_init
    
    for tau in tau_schedule:
        # Inner iterations for this tau
        n_iters = 500
        for it in range(n_iters):
            if time.monotonic() > deadline - 10:
                break
            
            c = fft_convolve(a, a)
            c_max = np.max(c)
            shifted = tau * (c - c_max)
            exp_shifted = np.exp(shifted)
            w = exp_shifted / exp_shifted.sum()
            
            # Gradient w.r.t. a
            N = 1 << (2 * n - 2).bit_length()
            Fa = np.fft.rfft(a, N)
            Fw = np.fft.rfft(w, N)
            grad_a = 4.0 * n * np.fft.irfft(Fw * np.conj(Fa), N)[:n]
            
            # Project gradient to tangent space of simplex (sum preserved)
            grad_a = grad_a - np.mean(grad_a)
            
            # Momentum update
            v = momentum * v + grad_a
            a_new = a - lr * v
            
            # Project to simplex (non-negative, sum=1)
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()
            
            # Adaptive learning rate (reduce if score increases)
            s_new = score_normalized(a_new)
            if s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
            
            if s_new > score_normalized(a) * (1 + 0.01):
                lr *= 0.9
            else:
                lr = min(lr * 1.01, 0.5)
            
            a = a_new
        
        # Anneal
        # (tau decreases in the schedule)
    
    return best_a, best_score

def local_search_fft(a0, deadline, rng):
    """
    Fast local search using FFT for convolution.
    Targets peaks in the convolution.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    temp = 0.005
    N = 1 << (2 * n - 2).bit_length()
    
    while time.monotonic() < deadline:
        c = fft_convolve(a, a)
        c_max = np.max(c)
        
        # Find peak locations
        peak_indices = np.where(c > 0.95 * c_max)[0]
        
        # Strategy: reduce contributions to peaks
        # For peak at position k, the contributing pairs are (i, k-i)
        # We want to reduce a[i]*a[k-i] for these pairs
        
        # Compute "pressure" on each element
        pressure = np.zeros(n)
        for k in peak_indices:
            for i in range(min(n, k + 1)):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += a[j]
                    # Note: if i != j, also pressure[j] += a[i], but we handle that below
        
        # Actually, let's be more careful:
        # c[k] = sum_i a[i]*a[k-i]
        # dc[k]/da[i] = 2*a[k-i] (as derived earlier)
        # So the "pressure" on a[i] from peak k is proportional to a[k-i]
        pressure = np.zeros(n)
        for k in peak_indices:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += a[j]
        
        if np.max(pressure) > 0:
            pressure = pressure / np.max(pressure)
        
        # Perturb: reduce high-pressure elements, increase low-pressure
        r = rng.random()
        if r < 0.5:
            # Targeted peak reduction
            shift = np.zeros(n)
            shift[pressure > 0.7] = -0.05 * (pressure[pressure > 0.7] - 0.7)
            low_mask = pressure < 0.3
            if np.any(low_mask):
                shift[low_mask] = 0.03 * rng.random(np.sum(low_mask))
            a_new = a * np.exp(shift)
        elif r < 0.7:
            # Smooth perturbation
            a_new = a * np.exp(rng.normal(0, 0.02, n))
        elif r < 0.85:
            # Block modification
            width = max(1, n // 16)
            start = rng.integers(0, n - width)
            a_new = a.copy()
            a_new[start:start+width] *= np.exp(rng.normal(0, 0.05, width))
        else:
            # Symmetrize
            a_new = 0.7 * a + 0.3 * a[::-1]
            a_new = a_new * np.exp(rng.normal(0, 0.01, n))
        
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        
        s_new = score_normalized(a_new)
        if s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
        
        if s_new < score_normalized(a):
            a = a_new
        else:
            if rng.random() > np.exp(-(s_new - score_normalized(a)) / max(temp, 1e-10)):
                a = a  # reject
            else:
                a = a_new
        
        temp *= 0.999
    
    return best_a, best_score

def make_init_uniform(n, rng):
    a = np.ones(n) + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def make_init_chirp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2 * np.pi * t * (t + 1) / (2 * n))
    a = np.maximum(a, 0.01)
    a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def make_init_bell(n, width, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.02 * rng.standard_normal(n)
    a = np.maximum(a, 0.005)
    return a / a.sum()

def make_init_random(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def make_init_symmetric(n, rng, spread=0.2):
    """Symmetric initialization: a[i] = a[n-1-i]"""
    half = n // 2
    a_half = 1.0 + spread * rng.standard_normal(half)
    a_half = np.maximum(a_half, 0.01)
    a = np.concatenate([a_half, a_half[::-1]]) if n % 2 == 0 else np.concatenate([a_half, [a_half[-1]], a_half[::-1][:-1]])
    # Ensure exact symmetry
    a = (a + a[::-1]) / 2
    return a / a.sum()
```

```python
"""
FFT-accelerated optimization for minimum peak self-convolution non-negative sequences.
Combines multi-start L-BFGS with log-sum-exp annealing, momentum gradient descent,
and FFT-based local search with targeted peak reduction. Exploits the analytical
gradient of the smoothed objective computed via cross-correlation in frequency domain.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1 << (n + m - 2).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Numerically stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def score_normalized(a):
    """Score for sequence normalized to sum=1: 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def lse_cross_corr(w, a, N):
    """Compute cross-correlation (w * a)[i] = sum_k w[k]*a[k-i] for i=0..n-1 using FFT."""
    n = len(a)
    Fa = np.fft.rfft(a, N)
    Fw = np.fft.rfft(w, N)
    return np.fft.irfft(Fw * np.conj(Fa), N)[:n]


def lse_obj_grad(theta, tau):
    """
    Log-sum-exp smoothed objective and gradient w.r.t. theta (softmax params).
    L = 2n * (1/tau) * log(sum_k exp(tau * c[k]))
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = tau * (c - c_max)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()

    val = 2.0 * n * (c_max + np.log(sum_exp) / tau)
    w = exp_shifted / sum_exp

    N = 1 << (2 * n - 2).bit_length()
    grad_a = 4.0 * n * lse_cross_corr(w, a, N)

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return val, grad_theta


def lse_minimize(theta0, tau_schedule, deadline):
    """Run L-BFGS with annealed log-sum-exp temperature."""
    best_theta = theta0.copy()
    a = softmax(theta0)
    best_score = score_normalized(a)

    for tau in tau_schedule:
        if time.monotonic() > deadline - 5:
            break
        result = minimize(
            lse_obj_grad, best_theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(tau,),
            options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-13, 'maxfun': 2500}
        )
        a_new = softmax(result.x)
        s = score_normalized(a_new)
        if s < best_score:
            best_score = s
            best_theta = result.x.copy()

    return best_theta, best_score


def momentum_gd(a0, tau_schedule, deadline, lr_init=0.05, mom=0.9):
    """Momentum gradient descent with simplex projection, using FFT gradients."""
    n = len(a0)
    a = np.maximum(a0, 1e-10).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    v = np.zeros(n)
    lr = lr_init
    N = 1 << (2 * n - 2).bit_length()

    for tau in tau_schedule:
        n_iters = 300
        stuck_count = 0
        for it in range(n_iters):
            if time.monotonic() > deadline - 5:
                break

            c = fft_convolve(a, a)
            c_max = np.max(c)
            shifted = tau * (c - c_max)
            exp_shifted = np.exp(shifted)
            w = exp_shifted / exp_shifted.sum()

            grad_a = 4.0 * n * lse_cross_corr(w, a, N)
            grad_a -= np.mean(grad_a)

            v = mom * v + grad_a
            a_new = a - lr * v
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()

            s_new = score_normalized(a_new)
            cur_score = score_normalized(a)

            if s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()

            if s_new < cur_score:
                a = a_new
                lr = min(lr * 1.02, 0.3)
                stuck_count = 0
            else:
                stuck_count += 1
                if stuck_count > 20:
                    lr *= 0.85
                    stuck_count = 0
                a = a_new  # still accept for exploration

    return best_a, best_score


def local_search_fft(a0, deadline, rng):
    """FFT-based local search with targeted peak reduction and adaptive temperature."""
    n = len(a0)
    a = np.maximum(a0, 1e-10).astype(np.float64)
    a = a / a.sum()

        c_max = np.max(c)
        cur_score = 2.0 * n * c_max

        peak_indices = np.where(peak_mask)[0]

        # Compute pressure on each element from peaks
        pressure = np.zeros(n)

            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += a[j]

        if np.max(pressure) > 1e-15:
            pressure = pressure / np.max(pressure)

        if r < 0.45:
            # Targeted peak reduction: reduce high-pressure, boost low-pressure
            shift = np.zeros(n)
            high_mask = pressure > 0.6
            if np.any(high_mask):
                shift[high_mask] = -0.04 * (pressure[high_mask] - 0.6)
            low_mask = pressure < 0.25
            if np.any(low_mask):
                shift[low_mask] = 0.02 * rng.random(np.sum(low_mask))
            a_new = a * np.exp(shift)
        elif r < 0.65:

            w[peak_indices] = 1.0 / len(peak_indices)
            grad_a = 4.0 * n * lse_cross_corr(w, a, N)

            norm = np.linalg.norm(grad_a)
            if norm > 1e-15:
                step_size = 0.02 / norm
                a_new = a - step_size * grad_a
            else:
                a_new = a * np.exp(rng.normal(0, 0.02, n))

        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        s_new = score_normalized(a_new)

        if s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
            iters_since_improve = 0
            temp = max(temp * 1.1, 0.005)  # explore more after improvement

            iters_since_improve += 1

        accept = s_new < cur_score or (rng.random() < np.exp(-(s_new - cur_score) / max(temp, 1e-12)))
        if accept:
            a = a_new

        temp *= 0.9997
        if iters_since_improve > max_iters_no_improve:
            temp = max(temp * 2.0, 0.01)
            iters_since_improve = 0

    return best_a, best_score


def make_init_uniform(n, rng):
    a = np.ones(n) + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_init_chirp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * t * (t + 1.0) / (2.0 * n))
    a = np.maximum(a, 0.01)
    a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_init_bell(n, width, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.02 * rng.standard_normal(n)
    a = np.maximum(a, 0.005)
    return a / a.sum()


def make_init_random(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_init_symmetric(n, rng, spread=0.2):
    half = (n + 1) // 2
    a_half = 1.0 + spread * rng.standard_normal(half)
    a_half = np.maximum(a_half, 0.01)
    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[::-1]])
    else:
        a = np.concatenate([a_half, a_half[-2::-1]])
    a = (a + a[::-1]) / 2.0
    return a / a.sum()


def make_init_fractal(n, rng):
    """Initialize using a dyadic-like structure (Hilbert curve inspired)."""
    a = np.ones(n) * 0.5
    scale = n // 2
    while scale >= 1:
        for i in range(0, n, scale):
            a[i:i + scale] *= rng.uniform(0.7, 1.3)
        scale //= 2
    a = np.maximum(a, 0.01)
    a += 0.03 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()

        print(f"\n=== FINAL: evaluate_sequence score={final_score:.8f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if best_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return [1.0] * 256
```
