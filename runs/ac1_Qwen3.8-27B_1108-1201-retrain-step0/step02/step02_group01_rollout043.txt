```python
"""
Search for a nonnegative sequence minimizing
    2*n*max(conv(a,a))/sum(a)^2
starting mainly from the supplied construction, using:
  - fast FFT convolution,
  - soft-max smoothed subgradient / Adam on log-softmax parameters,
  - exact 1-D coordinate descent with full convolution updates,
  - L-BFGS polishing of a smoothed surrogate,
  - a small LP directional step from the top active convolution lags.
"""

import time
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog, minimize


def _fft_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def _score(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    c = _fft_self(a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def _normalize_to_score_scale(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = len(a)
    S = float(np.sqrt(2.0 * n))
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, S / max(1, n))
    return a * (S / s)


def _theta_from_a(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    return np.log(np.maximum(a, 1e-300))


def _softmax(theta):
    theta = np.asarray(theta, dtype=np.float64)
    z = theta - np.max(theta)
    ez = np.exp(z)
    s = float(np.sum(ez))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(theta.shape, 1.0 / theta.size)
    return ez / s


def _grad_from_weights(w, a):
    """Gradient of sum_k w_k * c_k where c = conv(a,a)."""
    n = len(a)
    active = np.flatnonzero(w > 1e-14)
    if active.size == 0:
        return np.zeros(n, dtype=np.float64)

    # Sparse direct summation when only a small active set matters.
    if active.size <= 128:
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            wk = float(w[k])
            if abs(wk) < 1e-18:
                continue
            kk = int(k)
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            if hi >= lo:
                grad[lo:hi + 1] += (2.0 * wk) * a[kk - hi:kk - lo + 1][::-1]
        return grad

    out_len = 3 * n - 2
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    AR = fft(a[::-1], fsize)
    g = np.real(ifft(W * AR))
    return 2.0 * g[n - 1:2 * n - 1]


def _soft_obj_grad(theta, S, beta):
    """Soft-max surrogate gradient on log-softmax parameters."""
    a = S * _softmax(theta)
    c = _fft_self(a)
    Emax = float(np.max(c))

    if beta <= 1e-8:
        weights = np.full(len(c), 1.0 / len(c), dtype=np.float64)
        obj = Emax
    else:
        shifted = beta * (c - Emax)
        shifted = np.clip(shifted, -50.0, 0.0)
        es = np.exp(shifted)
        se = float(np.sum(es))
        if not np.isfinite(se) or se <= 0.0:
            weights = np.zeros(len(c), dtype=np.float64)
            weights[int(np.argmax(c))] = 1.0
            obj = Emax
        else:
            weights = es / se
            obj = Emax + float(np.log(se)) / beta

    grad_a = _grad_from_weights(weights, a)
    if not np.all(np.isfinite(grad_a)):
        return 1e100, np.zeros(len(theta)), Emax, a

    dot = float(np.dot(grad_a, a))
    grad_theta = grad_a * a - (a / S) * dot
    if not np.all(np.isfinite(grad_theta)):
        return 1e100, np.zeros(len(theta)), Emax, a

    return float(obj), grad_theta, Emax, a


def _run_adam(theta, S, budget, beta_start=2.0, beta_end=300.0,
              lr0=0.06, lr1=0.004, beta1=0.90, beta2=0.999):
    theta = np.array(theta, dtype=np.float64, copy=True)
    n = len(theta)

    a = S * _softmax(theta)
    c = _fft_self(a)
    best = float(np.max(c))
    best_a = a.copy()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    start = time.monotonic()
    last_print = start

    while time.monotonic() - start < budget:
        t += 1
        elapsed = time.monotonic() - start
        prog = min(1.0, elapsed / max(1e-12, budget))
        beta = beta_start * (beta_end / max(1e-12, beta_start)) ** prog

        obj, grad, Emax, a = _soft_obj_grad(theta, S, beta)
        if Emax < best:
            best = Emax
            best_a = a.copy()
            now = time.monotonic()
            if now - last_print > 10.0:
                print("Adam improving: score=%.6f beta=%.1f t=%d elapsed=%.1f"
                      % (best, beta, t, elapsed))
                last_print = now

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-16:
            break

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        mhat = m / (1.0 - beta1 ** t)
        vhat = v / (1.0 - beta2 ** t)
        lr = lr0 * (1.0 - prog) + lr1 * prog
        step = mhat / (np.sqrt(vhat) + 1e-8)
        np.clip(step, -10.0, 10.0, out=step)
        theta = theta - lr * step

        mean_theta = float(np.mean(theta))
        if abs(mean_theta) > 50.0:
            theta = theta - mean_theta

    return best_a, best


def _coordinate_descent(a, budget, rng=None, max_sweeps=1000):
    """Exact one-coordinate line search, maintaining convolution incrementally."""
    a = np.array(np.asarray(a, dtype=np.float64).reshape(-1), copy=True)
    n = len(a)
    S = float(np.sqrt(2.0 * n))
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-15:
        a = np.full(n, S / max(1, n))
    else:
        a = a * (S / s)

    c = _fft_self(a)
    E = float(np.max(c))
    best_a = a.copy()
    best_score = E

    L = 2 * n - 1
    start = time.monotonic()
    mean0 = S / max(1, n)

    factors = (0.0, 0.125, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 4.0, 16.0)
    refine = (0.5, 0.75, 0.9, 1.12, 1.35, 2.0)
    last_print = start

    for sweep in range(max_sweeps):
        if time.monotonic() - start > budget:
            break

        improved = False
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for ii in order:
            i = int(ii)
            if time.monotonic() - start > budget:
                break

            ai = a[i]
            slice_start = i
            slice_end = i + n
            old_slice = c[slice_start:slice_end].copy()
            k2 = 2 * i
            old_k2 = c[k2]

            best_x = ai
            best_delta = 0.0
            best_cand = E

            if ai <= 1e-15:
                candidates = (
                    0.0,
                    mean0 * 0.01,
                    mean0 * 0.10,
                    mean0 * 0.50,
                    mean0 * 1.00,
                    mean0 * 4.00,
                    max(mean0 * 10.0, S * 0.01),
                )
            else:
                candidates = tuple(ai * f for f in factors)
                if ai * factors[-1] < mean0 * 5.0:
                    candidates = candidates + (mean0 * 5.0, mean0 * 10.0, S * 0.01)

            for x in candidates:
                if not np.isfinite(x) or x < 0.0:
                    continue
                delta = x - ai
                if abs(delta) <= 1e-18 * max(1.0, ai):
                    cand = E
                else:
                    c[slice_start:slice_end] += (2.0 * delta) * a
                    if k2 < L:
                        c[k2] += delta * delta
                    maxv = float(np.max(c))
                    c[slice_start:slice_end] = old_slice
                    if k2 < L:
                        c[k2] = old_k2

                    new_sum = S + delta
                    if new_sum <= 1e-12:
                        cand = np.inf
                    else:
                        sf = S / new_sum
                        cand = maxv * sf * sf

                if cand < best_cand - 1e-14:
                    best_cand = cand
                    best_x = x
                    best_delta = delta

            if best_x > 0.0:
                for rel in refine:
                    x = best_x * rel
                    if not np.isfinite(x) or x < 0.0:
                        continue
                    delta = x - ai
                    if abs(delta) <= 1e-18 * max(1.0, ai):
                        continue
                    c[slice_start:slice_end] += (2.0 * delta) * a
                    if k2 < L:
                        c[k2] += delta * delta
                    maxv = float(np.max(c))
                    c[slice_start:slice_end] = old_slice
                    if k2 < L:
                        c[k2] = old_k2

                    new_sum = S + delta
                    if new_sum > 1e-12:
                        sf = S / new_sum
                        cand = maxv * sf * sf
                    else:
                        cand = np.inf

                    if cand < best_cand - 1e-14:
                        best_cand = cand
                        best_x = x
                        best_delta = delta

            if best_delta != 0.0 and best_cand < E:
                c[slice_start:slice_end] += (2.0 * best_delta) * a
                if k2 < L:
                    c[k2] += best_delta * best_delta
                a[i] = ai + best_delta

                new_sum = S + best_delta
                if new_sum <= 1e-12:
                    break
                sf = S / new_sum
                if abs(sf - 1.0) > 1e-15:
                    a *= sf
                    c *= sf * sf

                E = float(np.max(c))
                if E < best_score - 1e-14:
                    best_score = E
                    best_a = a.copy()
                    improved = True

                now = time.monotonic()
                if now - start > budget:
                    break

        if time.monotonic() - start > budget:
            break

        if not improved:
            break

        if sweep % 10 == 9:
            c_re = _fft_self(a)
            if np.max(np.abs(c_re - c)) > 1e-7 * max(1.0, E):
                c = c_re
            E = float(np.max(c))
            if E < best_score:
                best_score = E
                best_a = a.copy()

        now = time.monotonic()
        if now - last_print > 20.0:
            print("Coordinate descent: sweep=%d score=%.6f elapsed=%.1f"
                  % (sweep, best_score, now - start))
            last_print = now

    return best_a, best_score


def _lbfgs_obj(theta, S, beta, state):
    if time.monotonic() >= state["deadline"]:
        return state["floor"], np.zeros(len(theta), dtype=np.float64)

    obj, grad, Emax, a = _soft_obj_grad(theta, S, beta)
    if Emax < state["best_score"]:
        state["best_score"] = Emax
        state["best_a"] = a.copy()
        state["floor"] = obj

    if not np.all(np.isfinite(grad)):
        return 1e100, np.zeros(len(theta), dtype=np.float64)

    return obj, grad


def _run_lbfgs(theta, S, budget, beta=50.0):
    state = {
        "deadline": time.monotonic() + budget,
        "best_score": np.inf,
        "best_a": None,
        "floor": 1e100,
    }

    a0 = S * _softmax(theta)
    s0 = float(_score(a0))
    state["best_score"] = s0
    state["best_a"] = a0.copy()
    state["floor"] = s0

    try:
        minimize(
            _lbfgs_obj,
            theta,
            args=(S, beta, state),
            jac=True,
            method="L-BFGS-B",
            options={
                "maxiter": 250,
                "maxfun": 700,
                "maxls": 50,
                "gtol": 1e-8,
                "disp": False,
            },
        )
    except Exception as exc:
        print("L-BFGS error: %s" % str(exc))

    best_a = state.get("best_a")
    best_score = state.get("best_score")
    if best_a is None or not np.isfinite(best_score):
        best_a = a0.copy()
        best_score = s0
    return best_a, float(best_score)


def _lp_improve(a, budget, K=96, time_limit=45.0):
    """LP direction: max sum(g) s.t. conv(a,g) <= M on top active lags."""
    n = len(a)
    base_score = _score(a)
    if budget <= 2.0 or not np.isfinite(base_score):
        return a, base_score

    S = float(np.sqrt(2.0 * n))
    an = np.maximum(np.asarray(a, dtype=np.float64).reshape(-1), 0.0)
    sn = float(np.sum(an))
    if sn <= 1e-12:
        return a, base_score

    p = an / sn
    c = _fft_self(p)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return a, base_score

    K = int(min(K, len(c)))
    top = np.argsort(c)[::-1][:K]
    A = np.zeros((K, n), dtype=np.float64)

    for row, k in enumerate(top):
        kk = int(k)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if hi >= lo:
            A[row, lo:hi + 1] = p[kk - hi:kk - lo + 1][::-1]

    res = None
    try:
        res = linprog(
            -np.ones(n, dtype=np.float64),
            A_ub=A,
            b_ub=np.full(K, M, dtype=np.float64),
            bounds=[(0.0, None)] * n,
            method="highs",
            options={
                "time_limit": float(max(5.0, time_limit)),
                "presolve": True,
                "disp": False,
            },
        )
    except Exception as exc:
        print("LP error: %s" % str(exc))
        return a, base_score

    best_a = a.copy()
    best_score = base_score

    if res.success and res.x is not None:
        g = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
        gs = float(np.sum(g))
        if gs > 1e-12:
            g = g / gs
            for t in (0.1, 0.3, 0.5, 0.7, 0.9):
                pnew = (1.0 - t) * p + t * g
                pnew = np.maximum(pnew, 0.0)
                ps = float(np.sum(pnew))
                if ps <= 1e-12:
                    continue
                pnew = pnew / ps
                anew = S * pnew
                sc = float(_score(aneq))
                if sc < best_score:
                    best_score = sc
                    best_a = anew.copy()

    return best_a, best_score


def _verify_sequence(a):
    try:
        arr = np.asarray(a, dtype=np.float64).reshape(-1)
        return float(evaluate_sequence([float(x) for x in arr]))
    except Exception:
        return np.inf


def _try_accept(a, score, best_a, best_score, label):
    if a is None:
        return best_a, best_score
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0 or not np.all(np.isfinite(a)):
        return best_a, best_score

    a = np.clip(a, 0.0, 1000.0)
    if float(np.sum(a)) <= 0.01:
        return best_a, best_score

    if not np.isfinite(score):
        score = _score(a)

    ver = _verify_sequence(a)
    if np.isfinite(ver) and ver < score:
        score = ver

    if np.isfinite(score) and score < best_score:
        print("Accept %s: score=%.6f" % (label, score))
        return a.copy(), float(score)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)

    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0

    # Reserve a little time for final conversion/verification.
    deadline = time.monotonic() + max(1.0, min(budget, 2370.0) - 15.0)

    global_a = None
    global_best = np.inf

    # Load the supplied/previous construction if available.
    try:
        given = height_sequence_1
    except Exception:
        given = None

    if given is not None:
        try:
            arr = np.asarray(given, dtype=np.float64).reshape(-1)
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size >= 8 and float(np.sum(arr)) > 0.01:
                intern = _score(arr)
                actual = _verify_sequence(arr)
                sc = intern
                if np.isfinite(actual) and actual < sc:
                    sc = actual
                if np.isfinite(sc):
                    global_a = arr.copy()
                    global_best = float(sc)
                    print("Initial supplied: n=%d internal=%.6f actual=%.6f"
                          % (arr.size, intern, actual if np.isfinite(actual) else np.nan))
        except Exception as exc:
            print("Failed to load supplied sequence: %s" % str(exc))

    if global_a is None:
        print("No usable supplied sequence; using uniform fallback.")
        global_a = np.ones(4096, dtype=np.float64)
        global_best = _score(global_a)

    # Ensure a valid finite starting point.
    if not np.all(np.isfinite(global_a)) or global_a.size == 0:
        global_a = np.ones(4096, dtype=np.float64)
        global_best = _score(global_a)
    if not np.isfinite(global_best):
        global_best = 2.0

    # Phase 1: quick coordinate polish from the starting point.
    if time.monotonic() < deadline - 60.0:
        rem = deadline - time.monotonic()
        tc = min(160.0, 0.10 * rem)
        if tc > 20.0:
            a2, sc2 = _coordinate_descent(global_a, tc, rng)
            global_a, global_best = _try_accept(a2, sc2, global_a, global_best, "coord1")

    # Phase 2: broad smoothed Adam run.
    if time.monotonic() < deadline - 120.0:
        rem = deadline - time.monotonic()
        t1 = min(1200.0, 0.55 * rem)
        if t1 > 60.0:
            a_norm = _normalize_to_score_scale(global_a)
            S = float(np.sqrt(2.0 * len(a_norm)))
            theta = _theta_from_a(a_norm)
            a2, sc2 = _run_adam(theta, S, t1,
                                beta_start=2.0, beta_end=300.0,
                                lr0=0.06, lr1=0.004)
            global_a, global_best = _try_accept(a2, sc2, global_a, global_best, "adam1")

    # Phase 3: coordinate polish.
    if time.monotonic() < deadline - 90.0:
        rem = deadline - time.monotonic()
        tc = min(450.0, 0.20 * rem)
        if tc > 30.0:
            a2, sc2 = _coordinate_descent(global_a, tc, rng)
            global_a, global_best = _try_accept(a2, sc2, global_a, global_best, "coord2")

    # Phase 4: focused high-temperature Adam.
    if time.monotonic() < deadline - 90.0:
        rem = deadline - time.monotonic()
        t2 = min(350.0, 0.18 * rem)
        if t2 > 40.0:
            a_norm = _normalize_to_score_scale(global_a)
            S = float(np.sqrt(2.0 * len(a_norm)))
            theta = _theta_from_a(a_norm)
            a2, sc2 = _run_adam(theta, S, t2,
                                beta_start=50.0, beta_end=800.0,
                                lr0=0.025, lr1=0.002)
            global_a, global_best = _try_accept(a2, sc2, global_a, global_best, "adam2")

    # Phase 5: L-BFGS polish of the smooth surrogate.
    if time.monotonic() < deadline - 70.0:
        rem = deadline - time.monotonic()
        t3 = min(120.0, 0.10 * rem)
        if t3 > 30.0:
            a_norm = _normalize_to_score_scale(global_a)
            S = float(np.sqrt(2.0 * len(a_norm)))
            theta = _theta_from_a(a_norm)
            a2, sc2 = _run_lbfgs(theta, S, t3, beta=50.0)
            global_a, global_best = _try_accept(a2, sc2, global_a, global_best, "lbfgs")

    # Phase 6: LP directional step, if time remains.
    if time.monotonic() < deadline - 70.0:
        rem = deadline - time.monotonic()
        t4 = min(100.0, 0.10 * rem)
        if t4 > 30.0:
            a2, sc2 = _lp_improve(global_a, t4, K=96,
                                  time_limit=max(12.0, 0.65 * t4))
            global_a, global_best = _try_accept(a2, sc2, global_a, global_best, "lp")

    # Phase 7: final coordinate polish with remaining time.
    if time.monotonic() < deadline - 8.0:
        rem = max(1.0, deadline - time.monotonic())
        a2, sc2 = _coordinate_descent(global_a, rem, rng)
        global_a, global_best = _try_accept(a2, sc2, global_a, global_best, "coord3")

    # Final validation and safe conversion.
    if global_a is None or global_a.size == 0 or not np.all(np.isfinite(global_a)):
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64).reshape(-1), 0.0, 1000.0)
    if float(np.sum(global_a)) <= 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    final_internal = _score(global_a)
    final_actual = _verify_sequence(final_list)

    print("FINAL internal=%.6f actual=%.s"
          % (final_internal,
             "%.6f" % final_actual if np.isfinite(final_actual) else "inf"))

    if not np.isfinite(final_actual):
        # Extremely defensive fallback.
        final_list = [1.0] * 256
        print("Fallback uniform: actual=%.6f" % _verify_sequence(final_list))

    return final_list
```
