"""
Optimized step function search combining FFT initialization, 
soft-max gradient descent, and LP-based refinement to minimize
peak autocorrelation normalized by sum squared.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def autocorr(a):
    return fft_convolve(a, a)


def compute_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = autocorr(a)
    return 2.0 * n * np.max(c) / (s * s)


def gs_init(n, n_iter=100, rng=None):
    """Gerchberg-Saxton initialization for low-peak autocorrelation."""
    if rng is None:
        rng = np.random.default_rng()
    a = np.ones(n) / n
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    
    for it in range(n_iter):
        A = rfft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        a_full = irfft(phase, N_fft)
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:

def smooth_init(n, rng, n_frequencies=5):
    """Initialize with smooth random oscillations."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(n_frequencies):
        freq = rng.integers(1, max(2, n // 2))
        amp = rng.uniform(0.1, 0.5) / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 1e-4, None)
    return a / np.sum(a)

        grad_full = fft_convolve(a_rev, w)  # length n + (2n-1) - 1 = 3n-2
        grad = 2.0 * grad_full[n-1:2*n-1]  # length n
        
        # Now apply gradient with normalization constraint
        # We want to minimize grad dot a subject to sum(a) = 1 and a > 0
        # Project gradient onto tangent space: grad_tan = grad - mean(grad) * ... 
        # Actually, with log parameterization (a = exp(theta)/sum(exp(theta))), 
        # the gradient w.r.t. theta is: a * (grad - sum(a*grad)/sum(a))
        # = a * (grad - sum(a*grad)) since sum(a)=1
        
        inner = np.dot(a, grad)
        grad_theta = a * (grad - inner)
        
        # Adam update
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * (grad_theta ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic score check
        if step % 500 == 0:
            a_check = np.exp(theta - np.max(theta))
            a_check = a_check / np.sum(a_check)
            sc = compute_score(a_check)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()
            if step % 2000 == 0:
                print(f"          step={step}, score={sc:.6f}, best={best_score:.6f}, tau={tau:.2e}")
    
    a_final = np.exp(best_theta - np.max(best_theta))
    a_final = a_final / np.sum(a_final)
    final_score = compute_score(a_final)
    if final_score < best_score:
        best_score = final_score
    return a_final, best_score
