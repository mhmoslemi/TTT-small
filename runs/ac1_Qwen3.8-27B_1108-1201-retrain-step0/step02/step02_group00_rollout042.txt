def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = 0
        theta = v.min() - S / n  # fallback
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)

def fft_subgradient(a, c, top_k):
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total

    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]

    fsize = 1 << (3 * n - 3).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    
    term1 = np.real(ifft(W * np.conj(A)))[:n]  # correlation
    term2 = np.real(ifft(W * A))[:n]  # convolution

import numpy as np

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full linear self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_subgradient_full(a, c, top_k):
    """
    CORRECT subgradient of max(top_k lags of convolution) w.r.t. a.
    g[j] = sum_k w[k] * (a[k-j] + a[j-k])
    = correlation(w,a)[j] + convolution(w,a)[j]
    """
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total

    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]

    # Need fsize >= (2n-1) + n - 1 = 3n-2 for linear corr/conv
    fsize = 1 << (3 * n - 3).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)

    # First term: sum_k w[k]*a[k-j] = correlation = IFFT(W * conj(A))
    term1 = np.real(ifft(W * np.conj(A)))[:n]
    # Second term: sum_k w[k]*a[j-k] = convolution = IFFT(W * A)
    term2 = np.real(ifft(W * A))[:n]

    return term1 + term2


def fft_subgradient_lse(a, c, beta):
    """Smooth gradient via log-sum-exp. Uses CORRECT two-term formula."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (3 * n - 3).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)

    term1 = np.real(ifft(W * np.conj(A)))[:n]
    term2 = np.real(ifft(W * A))[:n]

    return (term1 + term2)

    S = np.sqrt(2.0 * float(n))

        return np.ones(n) * S / n
    return a * (S / s)

        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = total_deadline - time.monotonic()

        # 2a: Nesterov with high top-k, long run
        seg = max(5.0, remaining * 0.30)
        a_p1, _ = nesterov_optimize(global_a, n_best, S_best, seg, rng,
                                     top_k_init=20, top_k_max=512, use_lse=False)
        a_p1 = safe_normalize(a_p1, n_best)
        c_p1 = fft_convolve_self(a_p1)
        if c_p1.max() < global_best:
            global_best = c_p1.max()
            global_a = a_p1.copy()
            print("  [Polish Nesterov high-k] %.6f" % global_best)

        # 2b: LSE with high beta
        remaining *= 0.65
        seg2 = max(5.0, remaining * 0.35)
        a_p2, _ = lse_optimize(global_a, n_best, S_best, seg2, rng,
                                beta_init=50.0, beta_max=500.0)
        a_p2 = safe_normalize(a_p2, n_best)
        c_p2 = fft_convolve_self(a_p2)
        if c_p2.max() < global_best:
            global_best = c_p2.max()
            global_a = a_p2.copy()
            print("  [Polish LSE high-beta] %.6f" % global_best)

        # 2c: Nesterov again from latest best
        remaining *= 0.6
        seg3 = max(5.0, remaining * 0.4)
        a_p3, _ = nesterov_optimize(global_a, n_best, S_best, seg3, rng,
                                     top_k_init=50, top_k_max=512, use_lse=False)
        a_p3 = safe_normalize(a_p3, n_best)
        c_p3 = fft_convolve_self(a_p3)
        if c_p3.max() < global_best:
            global_best = c_p3.max()
            global_a = a_p3.copy()
            print("  [Polish Nesterov round2] %.6f" % global_best)

        # 2d: Coordinate descent polish
        remaining *= 0.55
        if remaining > 3:
            a_p4, score_p4 = coordinate_descent_polish(global_a, n_best, S_best,
                                                        remaining, rng,
                                                        max_iter=int(remaining * 500))
            a_p4 = safe_normalize(a_p4, n_best)
            c_p4 = fft_convolve_self(a_p4)
            if c_p4.max() < global_best:
                global_best = c_p4.max()
                global_a = a_p4.copy()
                print("  [Coord descent] %.6f" % global_best)

    # --- Phase 3: Final verification ---
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Final verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback score: %.6f" % ver if np.isfinite(ver) else "  Fallback also failed!")

    print("Returning sequence of length %d, internal max_conv=%.6f, obj=%.6f" %
          (len(global_a), global_best, ver))
    return final_list
