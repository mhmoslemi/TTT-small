def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)

"""
Optimization of non-negative step functions to minimize 2n*max(conv(a,a))/(sum a)^2.
Uses Adam gradient descent with softplus parameterization, diverse asymmetric
initializations, multi-scale n, and coordinate descent polishing.
Key insight: asymmetric sequences beat uniform ones because the convolution peak
can be reduced by making a[i]*a[n-1-i] small on average.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Fast linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute 2n*max(conv(a,a))/(sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def smoothed_max_and_weights(conv, tau):
    """Smoothed max via log-sum-exp and corresponding weights."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    shifted = np.clip(shifted, -100.0, 50.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    value = max_c + tau * np.log(sum_exp)
    weights = exp_vals / sum_exp
    return value, weights


def compute_grad_a(a, weights):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a, where conv = a*a (linear)."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(weights, size)
    # Cross-correlation approach: grad_a[j] = sum_k w[k]*(a[k-j]+a[j-k])
    # = 2 * (a cross-correlated with w) for symmetric w
    corr = np.fft.irfft(np.conj(fa) * fw, size)[:n]
    return 2.0 * corr


def softplus(x):
    """Numerically stable softplus."""
    return np.logaddexp(0, x)


def optimize_adam(n, a_init, time_limit, rng, lr_base=1.0):
    """
    Adam optimization on softplus parameterization.
    Minimizes max(conv(a,a)) subject to sum(a)=1, a>=0.
    a = softplus(theta) / sum(softplus(theta))
    """
    a_init = np.clip(a_init, 1e-8, None)
    a_init = a_init / np.sum(a_init)
    # Initialize theta from a_init: theta = log(exp(a_init*S) - 1) ~ log(a_init*S) for small a
    S_target = np.sqrt(2.0 * n)
    theta = np.log(a_init * S_target + 1e-10)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    step = 0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t0 = time.monotonic()

    best_conv_max = np.max(fft_convolve(a_init, a_init))
    best_a = a_init.copy()

    tau_start = 10.0 / n
    tau_end = 0.3 / n

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1.0))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_base * (1.0 + frac) ** (-0.5)

        sp = softplus(theta)
        S = np.sum(sp)
        a = sp / S

        conv = fft_convolve(a, a)
        value, weights = smoothed_max_and_weights(conv, tau)

        grad_a = compute_grad_a(a, weights)

        # Gradient through softplus + normalization
        # a_j = sp_j / S, da_j/dtheta_k = delta(j,k)*sigma_j/S - a_j*sigma_k/S
        # grad_theta_k = sum_j grad_a_j * (delta(j,k)*sigma_j/S - a_j*sigma_k/S)
        #              = sigma_k * (grad_a_k/S - sum_j grad_a_j * a_j / S)
        #              = (sigma_k / S) * (grad_a_k - <grad_a, a>)
        sigma = 1.0 / (1.0 + np.exp(-np.clip(theta, -50, 50)))
        sigma_over_S = sigma / S
        inner = np.dot(grad_a, a)
        grad_theta = sigma_over_S * (grad_a - inner)

        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * grad_theta
        v_buf = beta2 * v_buf + (1 - beta2) * (grad_theta ** 2)
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if step % 80 == 0:
            sp_check = softplus(theta)
            a_check = sp_check / np.sum(sp_check)
            cv = np.max(fft_convolve(a_check, a_check))
            if cv < best_conv_max:
                best_conv_max = cv
                best_a = a_check.copy()

            if step % 2000 == 0:
                sc = 2.0 * n * best_conv_max
                print(f"      adam n={n} step={step} best_score={sc:.6f} tau={tau:.2e} t={elapsed:.0f}s")

    return best_a, 2.0 * n * best_conv_max


def coordinate_descent(a_init, time_limit, rng):
    """
    Coordinate descent on the simplex for final polishing.
    Tries pair transfers and peak-targeted reductions.
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    n = len(a)

    best_a = a.copy()
    best_score = 2.0 * n * np.max(fft_convolve(a, a))
    current_score = best_score
    t0 = time.monotonic()
    patience = 0
    max_patience = 8000

    while time.monotonic() - t0 < time_limit and patience < max_patience:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        r = rng.random()

        if r < 0.5:
            # Pair transfer
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                patience += 1
                continue
            frac = rng.uniform(0.02, 0.2)
            amt = a[i] * frac
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
            if candidate[i] < 0 or candidate[j] < 0:
                patience += 1
                continue
            cs = np.sum(candidate)
            candidate = candidate / cs
            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        elif r < 0.7:
            # Single element scale
            j = int(rng.integers(n))
            factor = np.exp(rng.normal(0, 0.08))
            candidate = a.copy()
            candidate[j] *= factor
            if candidate[j] < 0:
                patience += 1
                continue
            cs = np.sum(candidate)
            candidate = candidate / cs
            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        elif r < 0.85:
            # Block perturbation
            width = max(2, int(rng.integers(2, max(3, n // 50))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            noise = np.exp(rng.normal(0, 0.05, size=width))
            candidate[start:start + width] *= noise
            cs = np.sum(candidate)
            candidate = candidate / cs
            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        else:
            # Peak-targeted: find peak position and reduce contributors
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Identify which pairs (i, peak_idx - i) contribute to conv[peak_idx]
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] += a[i] * a[j]
                    if i != j:
                        contribs[j] += a[i] * a[j]
            top_k = max(3, n // 64)
            top_idx = np.argpartition(contribs, -top_k)[-top_k:]
            candidate = a.copy()
            reduce_amt = 0.0
            for idx in top_idx:
                amt = candidate[idx] * rng.uniform(0.01, 0.08)
                candidate[idx] -= amt
                reduce_amt += amt
            # Redistribute to low-contribution positions
            low_idx = np.argpartition(contribs, top_k)[:min(10, n // 8)]
            low_idx = low_idx[contribs[low_idx] > 0][:10]
            if len(low_idx) == 0:
                low_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(low_idx) > 0:
                n_add = min(5, len(low_idx))
                add_idx = rng.choice(low_idx, size=n_add, replace=False)
                candidate[add_idx] += reduce_amt / n_add
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                patience += 1
                continue
            candidate = candidate / cs
            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                patience = 0
            else:
                patience += 1

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

    return best_a, best_score


def make_ramp_init(n, slope=1.0, phase=0.0, rng=None):
    """Create a ramp-like initialization."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = 0.1 + slope * t
    a += phase * np.sin(2 * np.pi * t)
    if rng is not None:
        a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_step_init(n, ratio=0.5, rng=None):
    """Step function initialization."""
    a = np.ones(n)
    a[int(n * ratio):] *= 0.3
    if rng is not None:
        a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_gaussian_init(n, center=0.3, width=0.2, rng=None):
    """Gaussian bump initialization."""
    t = np.arange(n, dtype=np.float64) / n
    a = 0.1 + np.exp(-0.5 * ((t - center) / width) ** 2)
    if rng is not None:
        a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_quadratic_init(n, rng=None):
    """Quadratic ramp."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = 0.1 + t ** 2
    if rng is not None:
        a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_sine_init(n, freq=1.0, rng=None):
    """Sine-modulated initialization."""
    t = np.arange(n, dtype=np.float64) / n
    a = 0.5 + 0.4 * np.sin(np.pi * t * freq) + 0.3 * t
    if rng is not None:
        a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def spectral_init(n, rng, n_iter=40):
    """Spectral iterative initialization."""
    a = 1.0 + 0.5 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    for it in range(n_iter):
        conv = fft_convolve(a, a)
        alpha = 2.0 / (n + 1.0)
        a = a - alpha * conv
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a

            print(f"  n={n_try}, init={init_type}, score={actual_score:.6f}")

            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_scaled.copy()
                print(f"    *** New best: {best_score:.6f} ***")

            # Quick polish
            if time.monotonic() < deadline - 40:
                a_norm = a_best
                polished, psc = coordinate_descent(a_norm, min(30.0, deadline - 40 - time.monotonic()), rng)
                pscaled = polished * np.sqrt(2.0 * len(polished))
                pscore = score_fast(pscaled)
                if pscore < best_score:
                    best_score = pscore
                    best_seq = pscaled.copy()
                    print(f"    *** New best after polish: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
