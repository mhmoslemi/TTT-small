def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def grad_at_lags(a, lags, values=None):
    """
    Compute gradient of sum_k values[k] * conv[k] w.r.t. a.
    grad[m] = 2 * sum_k values[k] * a[k-m] (for valid k-m)
    
    Actually, conv[k] = sum_i a[i]*a[i+k], so d(conv[k])/da[m] = a[m+k] + a[m-k] (with bounds)

    grad[m] = 2 * sum_k w[k] * corr(a, a, k)[m]
    where corr(a,a,k)[m] = a[m]*a[m+k] + a[m+k]*a[m] ... 
    
    Let me be more careful. conv[k] = sum_{i=0}^{n-1-k} a[i]*a[i+k] for k >= 0
    For k < 0, conv[k] = conv[|k|] by symmetry.
    
    d(conv[k])/da[m] for k >= 0:
    - If m <= n-1-k: a[m+k]
    - If m+k >= n: 0 (since a[i+k] is out of bounds)
    - Also if m >= n-k: the term a[m]*a[m-k] contributes... 
    
    Wait, let me use the full convolution. The full convolution c has length 2n-1, indexed from 0 to 2n-2.
    c[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]
    
    d(c[k])/da[m] = a[k-m] if 0 <= k-m < n, else 0
    
    So for the objective max_k c[k], using log-sum-exp approximation:
    grad[m] = sum_k w[k] * d(c[k])/da[m] = sum_k w[k] * a[k-m] (where 0<=k-m<n)
    
    This is a cross-correlation of w (length 2n-1) with a (length n).
    Specifically: grad[m] = sum_k w[k] * a[k-m] for k where 0 <= k-m < n, i.e., m <= k < m+n
    
    This is equivalent to: (w * a_rev)[offset] or we can use FFT.
    
    Let me define: pad a to length 2n-1, then the cross-correlation of w and a (padded) gives us what we want.
    
    Actually, let's think of it as:
    grad[m] = sum_{k=m}^{m+n-1} w[k] * a[k-m] = (w * a)[m] where * is correlation
    
    In terms of convolution: if we reverse a, then (w * a_rev)[2n-2-m] = sum_k w[k] * a_rev[2n-2-m-k] = sum_k w[k] * a[m-k+2n-2-2n+2] = sum_k w[k] * a[m-k]
    
    Hmm, let me just use the FFT approach directly:
    For the cross-correlation: grad[m] = sum_k w[k] * a[k-m] for valid k

    This is the cross-correlation of w and a at lag m.
    
    Using FFT: cross_corr(w, a)[m] = ifft(fft(w) * conj(fft(a))) at appropriate index
    But we need to be careful about padding and indexing.
    
    Let me use a simpler approach: pad both to length >= 3n-3, compute FFT, multiply, IFFT.
    """

    # (where w is indexed 0 to 2n-2, and we skip out-of-range terms)
    
    # Pad w to length 2n-1, pad a to length n
    # cross-correlation: h[m] = sum_j a[j] * w[m+j]
    # This is like: h = correlate(w, a) at lags 0 to n-1
    # But standard correlate(a, b)[m] = sum_j a[j+m]*b[j]
    
    # Let me just do it with FFT cross-correlation
    # h[m] = sum_j a[j] * w[m+j] = sum_j a[j] * w[(m+j) mod (2n-1)]... no that's circular
    
    # Simpler: define W[j] = w[j] for j in [0, 2n-1), A[j] = a[j] for j in [0, n), 0 otherwise
    # Then h[m] = sum_j A[j] * W[m+j] = (A * W_rev)[m + (2n-2)] where * is linear convolution
    # W_rev[j] = W[2n-2-j]
    
    # Actually the simplest: use the formula from the previous code

    # Which IS the cross-correlation of w and a at lag m (with the convention xcorr(w,a)[m] = sum_j w[j+m]*a[j])
    
    # So the formula gives: ifft(fft(w) * conj(fft(a))) = xcorr(w, a)
    # And xcorr(w, a)[m] = sum_j w[j+m] * a[j] (for linear cross-correlation with proper padding)
    
    fsize = 1 << (2 * (2 * n - 1) - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    result = np.real(ifft(W * np.conj(A)))

def compute_grad(a, c, lags, weights):
    """
    Compute the gradient of sum_k weights[k]*c[k] w.r.t. a.
    lags: indices into c where the peaks are
    weights: weight for each lag
    """
    n = len(a)
    w = np.zeros(2 * n - 1)
    for i, k in enumerate(lags):
        w[int(k)] = weights[i]
    
    fsize = 1 << (2 * (2 * n - 1) - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * grad  # factor of 2 from symmetry


def run_deflation(a_init, n, S, max_time, rng, top_m=10):
    """
    Peak-reduction deflation: iteratively identify the top peaks in the
    self-convolution and reduce them via projected gradient steps.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    y = a.copy()
    t = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Find top-m peaks
        top_m_eff = min(top_m, 2 * n - 1)
        top_indices = np.argsort(c)[::-1][:top_m_eff]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # Compute gradient
        grad = compute_grad(y, c, top_indices, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 2.0 / (it + 10) * S / n
        # Actually, use a more principled step size
        # The gradient of max(c) is bounded by... let's use backtracking
        step = S / (n * np.sqrt(it + 1))
        
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic perturbation
        if it % 5000 == 0 and rng.random() < 0.2:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score

def construct_segment_sequence(n, k, heights, fractions, rng):
    """
    Construct a piecewise-constant sequence with k segments.
    fractions: relative lengths of each segment (will be normalized to sum to 1)
    heights: amplitude of each segment
    """
    seq = np.zeros(n)
    fractions = np.array(fractions, dtype=np.float64)
    fractions = np.abs(fractions)
    if fractions.sum() < 1e-15:
        fractions = np.ones(k)
    fractions = fractions / fractions.sum()
    
    boundaries = np.cumsum(fractions * n).astype(int)
    boundaries = np.maximum.accumulate(boundaries)  # ensure monotone
    boundaries[-1] = n
    
    for i in range(k):
        start = boundaries[i-1] if i > 0 else 0
        end = boundaries[i]
        seq[start:end] = max(0.0, heights[i])
    
    return seq


def optimize_segments(n, k, max_time, rng, start_heights=None):

        seq = construct_segment_sequence(n, k, heights, fractions, rng)

        if score < best_score:
            best_score = score

        for i in range(k):

                    score = c_trial.max()
                    if score < best_score:

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def grad_from_weights(a, w):
    """
    Compute gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a).
    d(c[k])/da[m] = a[k-m] (if 0<=k-m<n, else 0)
    So grad[m] = 2 * sum_k w[k] * a[k-m] (factor 2 from both a[m] and a[k-m] terms)
    
    This is 2 * cross_correlation(w, a) evaluated at lags 0..n-1.
    Using FFT: ifft(fft(w) * conj(fft(a))) gives cross-correlation.
    """
    n = len(a)
    fsize = 1 << (2 * (2 * n - 1) - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * corr


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    return grad_from_weights(a, w)


def toppk_gradient(a, c, top_k):
    """Gradient using top-k peaks with weights proportional to values."""
    n = len(a)
    actual_k = min(top_k, 2 * n - 1)
    top_indices = np.argsort(c)[::-1][:actual_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    return grad_from_weights(a, w_full)

        if cur_max < best_score:

        # Backtracking reduction if needed
        if backtracking:
            c_trial = fft_convolve_self(x_trial)
            max_improve = (prev_score - cur_max)  # expected decrease
            if c_trial.max() > cur_max + max_improve + 1e-10:
                # Reduce step
                for _ in range(5):
                    step *= 0.5
                    x_trial = project_simplex(y - step * grad_normed, S)
                    c_trial = fft_convolve_self(x_trial)
                    if c_trial.max() <= cur_max + 1e-10:
                        break

        # Warm restart
        if warm_restart:
            c_x = fft_convolve_self(x_trial)
            if c_x.max() > prev_score + 1e-6:
                # Reset momentum
                t = 1.0
                y = x_trial.copy()
                prev_score = cur_max
                continue
        
        x_new = x_trial
        if not backtracking:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-normalize y to maintain sum constraint approximately
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        a = x_new
        t = t_new
        prev_score = max(cur_max, c_x.max() if 'c_x' in dir() else cur_max)

        # Periodic perturbation
        if it % 4000 == 0 and rng.random() < 0.15:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0
            prev_score = np.inf

    return best_local, best_score


def run_fista_lse_v2(a_init, n, S, max_time, rng, beta_init=20.0, beta_max=1000.0,
                      backtracking=True):
    """FISTA with log-sum-exp and backtracking line search."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 1.0 / (2.0 * beta_init)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))

        c = fft_convolve_self(y)
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_local = y.copy()

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: 1/(L*beta) where L ~ n (rough)
        step = 1.0 / (beta * (1.0 + 0.0005 * it))
        step = max(step, 1e-8)

        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        # Warm restart for FISTA
        if it % 200 == 0 and rng.random() < 0.3:
            c_check = fft_convolve_self(x_new)
            if c_check.max() > c_max * 1.01:
                t = 1.0

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        a = x_new
        t = t_new

    return best_local, best_score


def run_peak_deflation(a_init, n, S, max_time, rng, num_peaks=5):
    """
    Direct peak deflation: identify dominant peaks and use pairwise
    multiplicative updates to reduce them while preserving sum.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        # Find top peaks
        k_peaks = min(num_peaks, 2 * n - 1)
        top_indices = np.argsort(c)[::-1][:k_peaks]
        
        # For each peak lag, compute the "blame" for each element
        blame = np.zeros(n)
        for lag_idx in top_indices:
            lag = int(lag_idx)
            weight = c[lag] / max(c.max(), 1e-15)
            if lag == 0:
                # c[0] = sum(a[i]^2), blame[i] = 2*a[i]
                blame += weight * 2.0 * a
            elif lag < n:
                # c[lag] = sum_{i=0}^{n-1-lag} a[i]*a[i+lag]
                # blame[i] from this lag: a[i+lag] for i < n-lag, a[i-lag] for i >= lag
                for i in range(n):
                    contrib = 0.0
                    if i + lag < n:
                        contrib += a[i + lag]
                    if i - lag >= 0:
                        contrib += a[i - lag]
                    blame[i] += weight * contrib
            else:
                # lag >= n: c[lag] = c[2n-2-lag] by symmetry
                eff_lag = 2 * n - 2 - lag
                if eff_lag < n:
                    for i in range(n):
                        contrib = 0.0
                        if i + eff_lag < n:
                            contrib += a[i + eff_lag]
                        if i - eff_lag >= 0:
                            contrib += a[i - eff_lag]
                        blame[i] += weight * contrib

        bnorm = np.linalg.norm(blame)
        if bnorm < 1e-14:
            break

        # Multiplicative update (mirror-descent style)
        eta = 0.1 / (1.0 + 0.001 * it)
        a_new = a * np.exp(-eta * blame / bnorm)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_new * (S / s_new)

        # Occasionally add noise to escape
        if it % 1000 == 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

    return best_local, best_score


def construct_piecewise(n, breakpoints, heights, rng=None):
    """
    Construct a piecewise-constant sequence.
    breakpoints: array of length k+1, breakpoints[0]=0, breakpoints[-1]=n
    heights: array of length k
    """
    seq = np.zeros(n)
    for i in range(len(heights)):
        start = int(breakpoints[i])
        end = int(breakpoints[i + 1])
        if start < end:
            seq[start:end] = max(0.0, heights[i])
    return seq


def optimize_piecewise(n, k, max_time, rng):
    """
    Optimize a k-segment piecewise-constant sequence using coordinate descent
    on both heights and breakpoints.
    """
    S = np.sqrt(2.0 * float(n))
    
    # Initialize
    heights = rng.random(k) * 0.8 + 0.2
    fractions = np.ones(k) / k + rng.normal(0, 0.05, k)
    fractions = np.maximum(fractions, 0.05)
    fractions = fractions / fractions.sum()
    
    best_score = np.inf
    best_heights = heights.copy()
    best_fracs = fractions.copy()
    
    start = time.monotonic()
    it = 0
    S = np.sqrt(2.0 * float(n))
    
    while time.monotonic() - start < max_time:
        it += 1
        # Build sequence
        bps = np.zeros(k + 1)
        bps[0] = 0
        for i in range(k):
            bps[i + 1] = bps[i] + fractions[i] * n
        bps[-1] = n
        bps = np.maximum.accumulate(bps.astype(int))
        bps[-1] = n
        
        seq = construct_piecewise(n, bps, heights)
        s = seq.sum()
        if s < 1e-15:
            continue
        seq = seq * (S / s)
        c = fft_convolve_self(seq)
        score = c.max()
        
        if score < best_score:
            best_score = score
            best_heights = heights.copy()
            best_fracs = fractions.copy()
        
        # Optimize heights
        improved = True
        while improved and time.monotonic() - start < max_time:
            improved = False
            for i in range(k):
                best_h = heights[i]
                best_s = score
                for mult in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.3]:
                    h_trial = heights.copy()
                    h_trial[i] = heights[i] * mult
                    if h_trial[i] < 0.01:
                        h_trial[i] = 0.01
                    bps_t = np.zeros(k + 1)
                    bps_t[0] = 0
                    for j in range(k):
                        bps_t[j + 1] = bps_t[j] + fractions[j] * n
                    bps_t[-1] = n
                    bps_t = np.maximum.accumulate(bps_t.astype(int))
                    bps_t[-1] = n
                    seq_t = construct_piecewise(n, bps_t, h_trial)
                    s_t = seq_t.sum()
                    if s_t < 1e-15:
                        continue
                    seq_t = seq_t * (S / s_t)
                    c_t = fft_convolve_self(seq_t)
                    if c_t.max() < best_s:
                        best_s = c_t.max()
                        best_h = h_trial[i]
                if best_h != heights[i]:
                    heights[i] = best_h
                    score = best_s
                    improved = True
                    if score < best_score:
                        best_score = score
                        best_heights = heights.copy()
                        best_fracs = fractions.copy()
        
        # Optimize fractions (occasionally)
        if it % 3 == 0:
            for i in range(k - 1):
                best_s = score
                for delta in [0.02, 0.05, 0.1, -0.02, -0.05, -0.1]:
                    f_trial = fractions.copy()
                    f_trial[i] += delta
                    f_trial[i + 1] -= delta
                    if f_trial[i] < 0.03 or f_trial[i + 1] < 0.03:
                        continue
                    f_trial = f_trial / f_trial.sum()
                    bps_t = np.zeros(k + 1)
                    bps_t[0] = 0
                    for j in range(k):
                        bps_t[j + 1] = bps_t[j] + f_trial[j] * n
                    bps_t[-1] = n
                    bps_t = np.maximum.accumulate(bps_t.astype(int))
                    bps_t[-1] = n
                    seq_t = construct_piecewise(n, bps_t, heights)
                    s_t = seq_t.sum()
                    if s_t < 1e-15:
                        continue
                    seq_t = seq_t * (S / s_t)
                    c_t = fft_convolve_self(seq_t)
                    if c_t.max() < best_s:
                        best_s = c_t.max()
                        fractions = f_trial
                        score = best_s
                        if score < best_score:
                            best_score = score
                            best_heights = heights.copy()
                            best_fracs = fractions.copy()

    # Build final sequence
    bps = np.zeros(k + 1)
    bps[0] = 0
    for i in range(k):
        bps[i + 1] = bps[i] + best_fracs[i] * n
    bps[-1] = n
    bps = np.maximum.accumulate(bps.astype(int))
    bps[-1] = n
    seq = construct_piecewise(n, bps, best_heights)
    s = seq.sum()
    if s > 1e-15:
        seq = seq * (S / s)
    else:
        seq = np.ones(n) * S / n
    return seq, best_score


def multi_scale_optimize(a_init, n, S, max_time, rng):
    """
    Coarse-to-fine: downsample, optimize at half resolution, upsample, refine.
    """
    # Downsample by factor 2
    n_half = n // 2
    a_half = np.zeros(n_half)
    for i in range(n_half):
        a_half[i] = (a_init[2*i] + a_init[2*i+1]) / 2.0
    
    S_half = np.sqrt(2.0 * float(n_half))
    s_h = a_half.sum()
    if s_h > 1e-15:
        a_half = a_half * (S_half / s_h)
    
    # Optimize at half resolution
    t0 = time.monotonic()
    a_half_opt, score_half = run_nesterov_adapt(a_half, n_half, S_half, max_time * 0.4, rng,
                                                  top_k_init=3, top_k_max=32,
                                                  top_k_growth=20)
    
    # Upsample: interpolate
    a_full = np.zeros(n)
    for i in range(n_half):
        a_full[2*i] = a_half_opt[i]
        if 2*i+1 < n:
            a_full[2*i+1] = a_half_opt[i]
    
    s_full = a_full.sum()
    if s_full > 1e-15:
        a_full = a_full * (S / s_full)
    
    # Refine at full resolution
    remaining = max_time - (time.monotonic() - t0)
    a_full_opt, score_full = run_nesterov_adapt(a_full, n, S, remaining, rng,
                                                  top_k_init=5, top_k_max=64,
                                                  top_k_growth=30)
    return a_full_opt, score_full


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    
    Strategy:
    - Start from previous best (height_sequence_1)
    - Multi-phase optimization: FISTA -> Nesterov adaptive -> peak deflation
    - Segment-based exploration for diversity
    - Multi-scale coarse-to-fine
    - Final intensive polishing
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # --- Get starting points ---
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0, 1000)))
    except Exception:
        pass

    if inits:
        n = len(inits[0][1])
    else:
        n = 8192

    S = np.sqrt(2.0 * float(n))

    # Add diverse inits at same length
    tt = np.linspace(0, 1, n, endpoint=False)
    inits.append(("rand", rng.random(n) + 0.05))
    inits.append(("bimodal", np.exp(-0.5*((tt-0.3)*8)**2) + np.exp(-0.5*((tt-0.7)*8)**2) + 0.05))
    inits.append(("sin_comp", 1.0 + 0.25*np.sin(2*np.pi*5*tt) + 0.15*np.cos(2*np.pi*11*tt) + 0.1*np.sin(2*np.pi*3*tt)))

    print("=== Starting: n=%d, %d inits, budget=%.0fs ===" % (n, len(inits), budget_s))

    # --- Phase 1: Evaluate given and run FISTA on it (primary) ---
    if inits[0][0] == "given":
        a0 = inits[0][1].copy()
        s0 = a0.sum()
        if s0 > 1e-15:
            a0 = a0 * (S / s0)
        c0 = fft_convolve_self(a0)
        score0 = c0.max()
        global_best = score0
        global_a = a0.copy()
        print("Given sequence score: %.6f" % score0)

        # FISTA on given
        t0 = time.monotonic()
        fista_time = min(300, (deadline - t0) * 0.15)
        a_f, score_f = run_fista_lse_v2(global_a, n, S, fista_time, rng,
                                          beta_init=20.0, beta_max=500.0)
        a_f = a_f * (S / a_f.sum())
        c_f = fft_convolve_self(a_f)
        score_f = c_f.max()
        if score_f < global_best:
            global_best = score_f
            global_a = a_f.copy()
            print("Phase1 FISTA: %.6f" % global_best)

    # --- Phase 2: Adaptive Nesterov on best ---
    t0 = time.monotonic()
    nesterov_time = min(500, (deadline - t0) * 0.25)
    a_n, score_n = run_nesterov_adapt(global_a, n, S, nesterov_time, rng,
                                       top_k_init=3, top_k_max=100,
                                       top_k_growth=30)
    a_n = a_n * (S / a_n.sum())
    c_n = fft_convolve_self(a_n)
    score_n = c_n.max()
    if score_n < global_best:
        global_best = score_n
        global_a = a_n.copy()
        print("Phase2 Nesterov: %.6f" % global_best)

    # --- Phase 3: Peak deflation ---
    t0 = time.monotonic()
    defl_time = min(300, (deadline - t0) * 0.15)
    a_d, score_d = run_peak_deflation(global_a, n, S, defl_time, rng, num_peaks=5)
    a_d = a_d * (S / a_d.sum())
    c_d = fft_convolve_self(a_d)
    score_d = c_d.max()
    if score_d < global_best:
        global_best = score_d
        global_a = a_d.copy()
        print("Phase3 Deflation: %.6f" % global_best)

    # --- Phase 4: Try other inits with FISTA ---
    for idx in range(1, min(len(inits), 4)):
        if time.monotonic() > deadline - 200:
            break
        name, arr = inits[idx]
        t_init = time.monotonic()
        init_time = min(150, (deadline - t_init - 150) * 0.3)
        if init_time < 10:
            continue
        a_i, score_i = run_fista_lse_v2(arr, n, S, init_time, rng,
                                          beta_init=10.0, beta_max=200.0)
        a_i = a_i * (S / a_i.sum())
        c_i = fft_convolve_self(a_i)
        score_i = c_i.max()
        if score_i < global_best:
            global_best = score_i
            global_a = a_i.copy()
            print("Phase4 FISTA[%s]: %.6f" % (name, global_best))

    # --- Phase 5: Segment-based search ---
    t0 = time.monotonic()
    seg_total = min(300, max(0, (deadline - t0 - 200)))
    if seg_total > 30:
        for k in [3, 5, 7]:
            if time.monotonic() > deadline - 200:
                break
            seg_time = seg_total / 3.5
            if seg_time < 10:
                continue
            a_seg, score_seg = optimize_piecewise(n, k, seg_time, rng)
            c_seg = fft_convolve_self(a_seg)
            score_seg = c_seg.max()
            if score_seg < global_best:
                global_best = score_seg
                global_a = a_seg.copy()
                print("Phase5 Segments k=%d: %.6f" % (k, global_big if False else global_best))

    # --- Phase 6: Multi-scale on best ---
    t0 = time.monotonic()
    ms_time = min(200, max(0, deadline - t0 - 150))
    if ms_time > 30:
        a_ms, score_ms = multi_scale_optimize(global_a, n, S, ms_time, rng)
        a_ms = a_ms * (S / a_ms.sum())
        c_ms = fft_convolve_self(a_ms)
        score_ms = c_ms.max()
        if score_ms < global_best:
            global_best = score_ms
            global_a = a_ms.copy()
            print("Phase6 Multi-scale: %.6f" % global_best)

    # --- Phase 7: Final intensive polishing ---
    t0 = time.monotonic()
    final_time = max(10, deadline - t0)
    
    # Split between Nesterov and deflation
    half_time = final_time * 0.5
    
    a_fin, score_fin = run_nesterov_adapt(global_a, n, S, half_time, rng,
                                            top_k_init=10, top_k_max=200,
                                            top_k_growth=50)
    a_fin = a_fin * (S / a_fin.sum())
    c_fin = fft_convolve_self(a_fin)
    score_fin = c_fin.max()
    if score_fin < global_best:
        global_best = score_fin
        global_a = a_fin.copy()
        print("Phase7a Nesterov polish: %.6f" % global_best)
    
    remaining = deadline - time.monotonic()
    if remaining > 5:
        a_fin2, score_fin2 = run_peak_deflation(global_a, n, S, remaining, rng, num_peaks=8)
        a_fin2 = a_fin2 * (S / a_fin2.sum())
        c_fin2 = fft_convolve_self(a_fin2)
        score_fin2 = c_fin2.max()
        if score_fin2 < global_best:
            global_best = score_fin2
            global_a = a_fin2.copy()
            print("Phase7b Deflation polish: %.6f" % global_best)

    # --- Verification ---
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Final verification: %s ===" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
    
    if not np.isfinite(ver):
        print("evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
    
    print("Returning sequence of length %d, internal score=%.6f" % (len(global_a), global_best))
    return final_list
