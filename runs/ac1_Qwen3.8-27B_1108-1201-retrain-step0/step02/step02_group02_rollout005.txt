import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def project_simplex(v, S):
    """Project v onto {x : sum(x) = S, x >= 0}."""
    n = len(v)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.sum(u > 0)  # number of positive elements
    # Find theta
    # We need: sum(max(v - theta, 0)) = S
    # Using bisection
    lo = np.min(v)
    hi = np.max(v)

        s = np.sum(np.maximum(v - mid, 0))
        if s > S:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    return np.maximum(v - theta, 0)

def subgradient_at_peaks(a, top_k=1):

def soft_subgradient(a, tau):

    n = len(a)

    shifted = (c - c_max) / tau

    rev_a = a[::-1]

    g = 2.0 * corr[n - 1:2 * n - 1]

        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))

        if r < 0.3:
            # Peak-directed: reduce top contributors to peak, increase others
            # Find pairs contributing to peak

                    contribs[i] = a[i] * a[peak_idx - i]

        new_score = compute_score(a)

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score (fast, local)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:

    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def project_simplex(v, S):
    """Project v onto {x : sum(x) = S, x >= 0} using bisection."""
    lo = np.min(v) - abs(S)
    hi = np.max(v) + abs(S)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    return np.maximum(v - theta, 0.0)


def topk_subgradient(a, top_k, tau):
    """
    Compute gradient of soft-top-k objective of max(a*a).
    Uses softmax over convolution values with temperature tau,
    restricted to top_k positions for sharper focus.
    Returns (value, gradient, convolution).

    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)

        k_star = int(np.argmax(c))

                val += a[j - k_star]

        return c_max, g, c
    
    # Soft version: weight all peaks by softmax, but with effective top-k focus

    exp_vals = np.exp(shifted)

    value = c_max + tau * np.log(sum_exp)

    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1:2 * n - 1]
    
    return value, g, c

def make_freq_init(n, rng):

    a += 0.01 * np.abs(rng.standard_normal(n))
    return a
