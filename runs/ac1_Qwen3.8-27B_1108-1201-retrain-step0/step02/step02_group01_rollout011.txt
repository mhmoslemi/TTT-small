    return 2.0 * corr[:n]


def compute_obj(a):
    """Compute the normalized objective: max(conv(a,a)) when sum(a) = sqrt(2n)."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a_norm = a * (S / a.sum())
    c = fft_convolve_self(a_norm)
    return c.max(), a_norm


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_subgradient(a_init, n, S, time_budget, T_init=10.0, T_final=500.0,
                    lr_init=0.1, momentum=0.9, restart_interval=500,
                    noise_scale=0.01):
    """
    Weighted subgradient descent with Nesterov momentum and backtracking.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Nesterov variables
    y = a.copy()  # lookahead point
    beta = momentum  # momentum parameter
    
    start = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** progress  # exponential annealing of T
        
        # Nesterov lookahead
        y = a + beta * (a - best_a)  # Hmm, this isn't quite right
        # Actually, standard Nesterov: y = a + (beta_k - 1)/beta_{k-1} * (a - a_prev)
        # Let me use a simpler momentum approach:
        # v = beta * v + (1-beta) * grad
        # a = a - lr * v
        
        # Compute subgradient at current a
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Restart with noise
            a = best_a + noise_scale * np.random.randn(n)
            a = normalize_nonneg(a, S)
            no_improve = 0
            continue
        
        g = compute_subgrad_weighted(c, a, T)
        
        # Normalize gradient for step size
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g = g / gnorm
        
        # Backtracking line search
        lr = lr_init * (1.0 - 0.5 * progress)
        best_step_obj = cur_obj
        best_step_a = a.copy()
        
        for i in range(15):
            a_trial = a - lr * g
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            
            if obj_trial < cur_obj * (1 - 1e-10):  # sufficient decrease
                best_step_obj = obj_trial
                best_step_a = a_trial.copy()
                break
            lr *= 0.5
        
        a = best_step_a
        step += 1
        
        if step % 1000 == 0:
            print("  Subgrad step %d: obj=%.6f, T=%.1f" % (step, best_obj, T))
    
    return best_a, best_obj


def coordinate_descent(a, n, S, time_budget, max_sweeps=100):
    """Coordinate descent: try adjusting each coordinate individually."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        # Process coordinates in random order
        order = np.random.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            original = a[i]
            if original < 1e-15:
                continue
            
            # Try several values: original * factor
            best_local = original
            best_local_obj = c[i * 2]  # rough: contribution to peak
            
            # Try reducing and increasing
            for factor in [0.5, 0.75, 1.25, 1.5, 0.25, 2.0]:
                a[i] = original * factor
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj:
                    best_local_obj = obj_check
                    best_local = original * factor
                elif obj_check > c.max() * 1.05:
                    break
            
            a[i] = best_local
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        if not improved and sweep > 5:
            break
    
    return best_a, best_obj

Minimize 2*n*max(conv(a,a))/sum(a)^2 using weighted subgradient descent with
exponential peak weighting, backtracking line search, Nesterov momentum,
periodic random restarts, coordinate descent polish, and multi-start strategy.

Key differences from previous approach:
1. Direct optimization (no log-parameterization) allowing coordinates to reach zero
2. Exponential peak weighting with aggressive temperature scheduling
3. Backtracking line search for reliable convergence
4. Nesterov momentum acceleration
5. Periodic restarts with diminishing noise to escape local minima
6. Frequency-domain structured initializations for diverse starts
7. Adaptive peak selection based on autocorrelation flatness

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgrad_weighted(c, a, T):
    """Compute weighted subgradient using exponential peak weighting.
    g[m] = 2 * cross_corr(w, a)[m] where w[k] proportional to exp(T*c[k]/c_max)
    """
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)

    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # g[m] = 2 * sum_{k} w[k] * a[k-m] (a[j]=0 for j outside [0,n-1])
    # This is 2 * cross_corr(w, a)[m]
    # Using FFT with zero-padding for linear cross-correlation
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def evaluate_obj_normalized(a):
    """Compute max(conv(a,a)) assuming sum(a) = sqrt(2n)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    S = np.sqrt(2.0 * n)
    a_norm = a * (S / s)
    c = fft_convolve_self(a_norm)
    return c.max()


def run_subgradient_nesterov(a_init, n, S, time_budget, T_init=10.0, T_final=500.0,
                             lr_init=0.1, momentum=0.9, noise_scale=0.01,
                             restart_every=300, verbose=False):
    """
    Nesterov accelerated weighted subgradient descent with backtracking.
    Optimizes directly in the original space (allows zeros).
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    # Nesterov variables
    a_prev = a.copy()
    beta = momentum

    start = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 150

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        # Exponential temperature schedule (aggressive: T increases fast)
        T = T_init * (T_final / T_init) ** (progress ** 0.7)

        # Nesterov lookahead
        y = a + (beta - 1.0) / (1.0 + beta) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        else:
            y = a.copy()

        # Compute autocorrelation and subgradient at lookahead point
        c_y = fft_convolve_self(y)
        obj_y = c_y.max()

        # Also track at current point
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic restart with noise
        if no_improve > max_no_improve:
            scale = noise_scale * max(0.1, 1.0 - progress)
            a = best_a + scale * np.random.default_rng(step).standard_normal(n)
            a = normalize_nonneg(a, S)
            a_prev = a.copy()
            no_improve = 0
            step += 1
            continue

        # Compute subgradient (use current c for stability)
        g = compute_subgrad_weighted(c, a, T)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g_dir = g / gnorm

        # Backtracking line search
        lr = lr_init * (1.0 - 0.7 * progress)
        best_step_a = a.copy()
        best_step_obj = cur_obj

        for i in range(12):
            a_trial = a - lr * g_dir
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()

            if obj_trial < cur_obj - 1e-10 * cur_obj:
                best_step_a = a_trial
                best_step_obj = obj_trial
                break
            lr *= 0.5

        a_prev = a.copy()
        a = best_step_a
        step += 1

        if verbose and step % 500 == 0:
            print("    SG step %d: obj=%.7f, T=%.1f, no_improve=%d" % (step, best_obj, T, no_improve))

    return best_a, best_obj


def run_adam_log(a_init, n, S, time_budget, T_init=5.0, T_final=1000.0,
                 lr_init=0.05, beta1=0.9, beta2=0.999, verbose=False):
    """
    Adam optimizer on log-parameters with log-sum-exp smooth objective.
    Ensures non-negativity and sum constraint automatically.
    """
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())  # log of proportions

    n_par = n
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    eps = 1e-8

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.6)

        # Convert theta to a
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n_par, S / n_par)
        else:
            a = S * exp_theta / total

        # Compute autocorrelation and smooth objective gradient
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()

        # Compute weighted subgradient
        c_max = c.max()
        if c_max < 1e-30:
            break
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

        # g_a[m] = 2 * sum_k w[k] * a[k-m]
        fsize = 1 << (3 * n_par - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g_a = 2.0 * corr[:n_par]

        # Chain rule for log-parameters: dL/dtheta[j] = g_a[j]*a[j] - (a[j]/S)*sum_i g_a[i]*a[i]
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot

        # Adam update
        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break

        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_init * (0.3 + 0.7 * (1.0 - progress))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if verbose and t % 1000 == 0:
            print("    Adam step %d: obj=%.7f, T=%.1f" % (t, best_obj, T))

    return best_a, best_obj


def coordinate_descent_fast(a, n, S, time_budget, max_sweeps=50, verbose=False):
    """Fast coordinate descent: try adjusting individual coordinates."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            original = a[i]
            if original < 1e-12:
                continue

            best_local = original
            best_local_obj = best_obj

            # Try a few multiplicative changes
            for factor in [0.5, 0.75, 1.33, 1.5, 0.25, 2.0, 0.6, 1.4]:
                a[i] = original * factor
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-12:
                    best_local_obj = obj_check
                    best_local = original * factor

            a[i] = best_local
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)

            if c.max() < best_obj - 1e-12:
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        if not improved and sweep >= 3:
            if verbose:
                print("    CD converged at sweep %d" % sweep)
            break

    return best_a, best_obj


def get_initialization(name, n, S, rng):
    """Generate various initializations."""
    i = np.arange(n)
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "gaussian_peak":
        a = np.exp(-0.5 * ((i - n / 2) / (n / 6)) ** 2)
        return a * (S / a.sum())
    elif name == "cosine_mix":
        a = 1.0 + 0.25 * np.cos(2 * np.pi * 5 * i / n + rng.random() * 2) + \
            0.15 * np.cos(2 * np.pi * 11 * i / n + rng.random() * 2) + \
            0.1 * np.cos(2 * np.pi * 23 * i / n + rng.random() * 2)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "flat_spectrum":
        # Try to create near-flat spectrum
        a = 1.0 + 0.1 * np.sin(2 * np.pi * 3 * i / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "two_hump":
        a = np.exp(-0.5 * ((i - n * 0.3) / (n / 10)) ** 2) + \
            np.exp(-0.5 * ((i - n * 0.7) / (n / 10)) ** 2)
        return a * (S / a.sum())
    elif name == "triangle":
        t = i / n
        a = 1.0 - 2.0 * np.abs(t - 0.5)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def try_zero_masking(a, n, S, n_zero=10, verbose=False):
    """Try setting the smallest coordinates to zero and check if objective improves."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    current_obj = c.max()

    # Find smallest coordinates
    sorted_idx = np.argsort(a)
    candidates = sorted_idx[:n_zero * 3]

    best_obj = current_obj
    best_a = a.copy()

    # Greedily try zeroing
    for idx in candidates:
        if a[idx] < 1e-10:
            continue
        saved = a[idx]
        a[idx] = 0.0
        a = normalize_nonneg(a, S)
        c_check = fft_convolve_self(a)
        obj_check = c_check.max()
        if obj_check < best_obj - 1e-12:
            best_obj = obj_check
            best_a = a.copy()
        else:
            a[idx] = saved
            a = normalize_nonneg(a, S)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 20.0)
    deadline = time.monotonic() + total_budget

    global_best = np.inf
    global_a = None

    # Load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: " + str(e))

    # ============ PHASE 1: Exploration (20% time) ============
    explore_budget = 0.18 * total_budget
    explore_deadline = time.monotonic() + explore_budget

    lengths_to_try = [4096]
    if given_a is not None:
        ng = len(given_a)
        if ng not in lengths_to_try:
            lengths_to_try.insert(0, ng)
    lengths_to_try.extend([2048, 8192])

    init_names = ["uniform", "random", "cosine_mix", "flat_spectrum", "two_hump"]
    n_total_attempts = len(lengths_to_try) * len(init_names)
    per_attempt_time = max(2.0, explore_budget / n_total_attempts)

    print("Phase 1: Exploration (%.0fs budget, %d attempts)" % (explore_budget, n_total_attempts))

    for n in lengths_to_try:
        if time.monotonic() > explore_deadline - 10:
            break
        S = np.sqrt(2.0 * float(n))

        for init_name in init_names:
            if time.monotonic() > explore_deadline - 5:
                break
            if per_attempt_time < 1.5:
                break

            a_init = get_initialization(init_name, n, S, rng)
            t0 = time.monotonic()

            # Use log-Adam for exploration (robust)
            a_best, obj_best = run_adam_log(
                a_init, n, S, per_attempt_time,
                T_init=5.0, T_final=200.0, lr_init=0.04
            )

            a_check = normalize_nonneg(a_best, S)
            final_obj = evaluate_sequence(a_check.tolist())
            if np.isfinite(final_obj) and final_obj < global_best:
                global_best = final_obj
                global_a = a_check.copy()
                print("  Explore: n=%d init=%s obj=%.7f" % (n, init_name, final_obj))

    # ============ PHASE 2: Deep exploitation (50% time) ============
    exploit_budget = 0.50 * total_budget
    exploit_deadline = time.monotonic() + exploit_budget

    if global_a is not None and exploit_budget > 20:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a = global_a.copy()
        a = normalize_nonneg(a, S)

        print("\nPhase 2: Exploitation n=%d (budget=%.0fs)" % (n, exploit_budget))

        # Round 1: Log-Adam with increasing T
        r1_budget = 0.30 * exploit_budget
        a_best, _ = run_adam_log(
            a, n, S, r1_budget,
            T_init=20, T_final=500, lr_init=0.03
        )
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  R1 LogAdam T20-500: obj=%.7f" % obj)

        # Round 2: Direct subgradient with Nesterov
        r2_budget = 0.25 * exploit_budget
        a_best, _ = run_subgradient_nesterov(
            a, n, S, r2_budget,
            T_init=50, T_final=1000, lr_init=0.05,
            momentum=0.85, noise_scale=0.01
        )
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  R2 Subgrad T50-1000: obj=%.7f" % obj)

        # Round 3: Log-Adam at very high T (focus on max)
        r3_budget = 0.20 * exploit_budget
        a_best, _ = run_adam_log(
            a, n, S, r3_budget,
            T_init=200, T_final=2000, lr_init=0.02
        )
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  R3 LogAdam T200-2000: obj=%.7f" % obj)

        # Round 4: Direct subgradient with low T (broader smoothing)
        r4_budget = 0.15 * exploit_budget
        a_best, _ = run_subgradient_nesterov(
            a, n, S, r4_budget,
            T_init=10, T_final=100, lr_init=0.08,
            momentum=0.7, noise_scale=0.02
        )
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  R4 Subgrad T10-100: obj=%.7f" % obj)

        # Try zero masking
        a_z, obj_z = try_zero_masking(global_a, n, S, n_zero=20)
        obj_z = evaluate_sequence(a_z.tolist())
        if np.isfinite(obj_z) and obj_z < global_best:
            global_best = obj_z
            global_a = a_z.copy()
            print("  ZeroMask: obj=%.7f" % obj_z)

    # ============ PHASE 3: Polish (22% time) ============
    polish_budget = 0.22 * total_budget
    polish_deadline = time.monotonic() + polish_budget

    if global_a is not None and polish_budget > 10:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a = global_a.copy()
        a = normalize_nonneg(a, S)

        print("\nPhase 3: Polish (budget=%.0fs)" % polish_budget)

        # Coordinate descent
        cd_budget = 0.4 * polish_budget
        a_best, _ = coordinate_descent_fast(a, n, S, cd_budget, max_sweeps=30)
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  CD: obj=%.7f" % obj)

        # More subgradient with very high T and small steps
        sg_budget = 0.35 * polish_budget
        a_best, _ = run_subgradient_nesterov(
            a, n, S, sg_budget,
            T_init=500, T_final=5000, lr_init=0.02,
            momentum=0.9, noise_scale=0.003
        )
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  HighT SG: obj=%.7f" % obj)

        # Final coordination
        remaining = polish_deadline - time.monotonic()
        if remaining > 5:
            a_best, _ = coordinate_descent_fast(a, n, S, remaining, max_sweeps=20)
            a = normalize_nonneg(a_best, S)
            obj = evaluate_sequence(a.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a.copy()
                print("  Final CD: obj=%.7f" % obj)

    # ============ PHASE 4: Random restart exploration (8% time) ============
    restart_budget = deadline - time.monotonic()
    if global_a is not None and restart_budget > 15:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        best_a = global_a.copy()

        print("\nPhase 4: Restarts (budget=%.0fs)" % restart_budget)
        n_rounds = max(2, int(restart_budget / 25))
        per_round = restart_budget / (n_rounds + 1)

        for r in range(n_rounds):
            if time.monotonic() > deadline - 5:
                break
            # Add noise proportional to current solution quality
            noise = 0.005 * (1 + r * 0.3)
            a_noisy = best_a + noise * rng.standard_normal(n)
            a_noisy = normalize_nonneg(a_noisy, S)

            a_best, _ = run_subgradient_nesterov(
                a_noisy, n, S, per_round,
                T_init=100, T_final=2000, lr_init=0.03,
                momentum=0.85, noise_scale=0.005
            )
            a_best = normalize_nonneg(a_best, S)
            obj = evaluate_sequence(a_best.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_best.copy()
                best_a = a_best.copy()
                print("  Restart %d: obj=%.7f" % (r, obj))

    # Final verification and cleanup
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f (verified=%s)" % (
        global_best, "%.7f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)

    return final_a
