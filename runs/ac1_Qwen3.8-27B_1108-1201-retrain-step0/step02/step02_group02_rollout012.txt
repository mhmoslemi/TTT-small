        a = np.ones(n) / n
        S = 1.0
    
    best_a = a.copy()
    best_score = local_score(a)
    
    t0 = time.monotonic()
    step = 0
    no_improve = 0
    patience = 500
    
    # Adam state
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    # Temperature schedule
    tau_start = 50.0 / n
    tau_end = 0.01 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = elapsed / time_limit
        
        if frac < 0.6:
            # Soft-max phase with decreasing temperature
            tau = tau_start * (tau_end / tau_start) ** (frac / 0.6)
            value, g_c, c, w = softmax_value_grad(a, tau)
            
            # Score and its gradient (with sum(a) dependence)
            S = np.sum(a)
            score_val = 2.0 * n * value / (S * S)
            # d(score)/da = 2n/(S^2) * g_c - 4n*value/S^3 * a
            g = (2.0 * n / (S * S)) * g_c - (4.0 * n * value / (S ** 3)) * a
        elif frac < 0.85:
            # Transition: very low temperature soft-max
            tau = tau_end * 0.1
            value, g_c, c, w = softmax_value_grad(a, tau)
            S = np.sum(a)
            score_val = 2.0 * n * value / (S * S)
            g = (2.0 * n / (S * S)) * g_c - (4.0 * n * value / (S ** 3)) * a
        else:
            # Exact subgradient phase
            peak_val, g_c, k_star, c = exact_subgrad(a)
            S = np.sum(a)
            score_val = 2.0 * n * peak_val / (S * S)
            g = (2.0 * n / (S * S)) * g_c - (4.0 * n * peak_val / (S ** 3)) * a
        
        # Adam update
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g
        v_buf = beta2 * v_buf + (1 - beta2) * g ** 2
        m_hat = m_buf / (1 - beta1 ** step)
        v_hat = v_buf / (1 - beta2 ** step)
        
        # Learning rate schedule
        base_lr = 0.5 / np.sqrt(step + 1)
        if frac < 0.2:
            lr = base_lr * 2.0
        elif frac < 0.6:
            lr = base_lr
        else:
            lr = base_lr * 0.3
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0, None)
        
        # Periodic evaluation
        if step % 50 == 0:
            cur_score = local_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 50
            
            # Escape mechanism
            if no_improve > patience:
                r = rng.random()
                if r < 0.3:
                    # Gaussian perturbation
                    a = best_a + 0.1 * rng.standard_normal(n)
                    a = np.clip(a, 0, None)
                elif r < 0.6:
                    # Targeted: perturb peak contributors
                    c_cur = fft_convolve(best_a, best_a)
                    pk = int(np.argmax(c_cur))
                    contribs = np.zeros(n)
                    for i in range(n):
                        if 0 <= pk - i < n:
                            contribs[i] = best_a[i] * best_a[pk - i]
                    top_k = np.argsort(contribs)[::-1][:max(4, n // 10)]
                    a = best_a.copy()
                    a[top_k] *= (1 - 0.05 * rng.random(len(top_k)))
                    a[np.argsort(contribs)[:max(4, n//10)]] *= (1 + 0.1)
                    a = np.clip(a, 0, None)
                else:
                    # Block perturbation
                    a = best_a.copy()
                    blk = max(2, n // 8)
                    start = int(rng.integers(0, n - blk))
                    a[start:start+blk] *= np.exp(rng.normal(0, 0.1, size=blk))
                    a = np.clip(a, 0, None)
                no_improve = 0
        
        if verbose and step % 2000 == 0:
            print(f"      n={n}, step={step}, frac={frac:.2f}, cur={cur_score:.6f}, best={best_score:.6f}, tau={tau:.2e}")
    
    # Final check
    final_score = local_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score

def coordinate_descent_polish(a_init, time_limit, rng):
    """Polish using coordinate descent."""
    a = a_init.astype(np.float64).copy()
    best_a = a.copy()
    best_score = local_score(a)
    n = len(a)
    t0 = time.monotonic()
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-10:
                # Try activating this coordinate
                trial = a.copy()
                trial[j] = np.sum(a) / n * 0.1
                s = local_score(trial)
                if s < best_score:
                    a = trial
                    best_score = s
                    best_a = a.copy()
                    improved = True
                continue
            
            base = a[j]
            # Try multiplicative perturbations
            for frac in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                trial = a.copy()
                trial[j] = base * frac
                s = local_score(trial)
                if s < best_score:
                    a = trial
                    best_score = s
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    break
        
        if not improved:
            # Random perturbation
            j = int(rng.integers(0, n))
            a[j] = max(0, a[j] * np.exp(rng.normal(0, 0.1)))
        
        iteration += 1
        if iteration % 10 == 0:
            print(f"      CD iter={iteration}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score

def make_inits(n, rng, n_inits=5):
    """Generate diverse initializations."""
    inits = []
    
    # 1. Uniform + noise
    a = 1.0 + 0.3 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))
    
    # 2. Sum of sinusoids
    a = np.ones(n)
    t = np.arange(n)
    for _ in range(5):
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2*np.pi)
        amp = rng.uniform(0.1, 0.3)
        a += amp * np.cos(2*np.pi*freq*t/n + phase)
    inits.append(np.clip(a, 0.01, None))
    
    # 3. Two-block
    a = np.ones(n)
    a[:n//3] = 1.5
    a[2*n//3:] = 1.5
    a[n//3:2*n//3] = 0.5
    a += 0.1 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))
    
    # 4. Periodic
    period = max(3, int(np.sqrt(n) + rng.integers(-2, 3)))
    pattern = 1.0 + 0.5 * rng.standard_normal(period)
    pattern = np.clip(pattern, 0.1, None)
    a = np.tile(pattern, n // period + 1)[:n]
    inits.append(a)
    
    # 5. Decreasing/exponential
    a = np.exp(-t * rng.uniform(0.5, 2.0) / n)
    a += 0.1 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))
    
    return inits
