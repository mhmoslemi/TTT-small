        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Aggressive annealing: start high, end very low
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)

        a = softmax_normalize(theta) * S
        value, g_a, c = compute_soft_max_grad(a, tau)

        # Gradient of the softmax parameterization:
        # a = softmax(theta) * S, so da/dtheta[i] = a[i]*(delta[i] - softmax(theta)[j])
        # grad_theta[i] = a[i] * (g_a[i] - sum_j a[j]*g_a[j] / S)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 2000 == 0:
            a_check = softmax_normalize(theta) * S
            sc = score(a_check)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()
                stall_count = 0
            else:
                stall_count += 2000
            if step % 10000 == 0:
                print(f"          step={step}, score={sc:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")

            # If stalled, restart with slight perturbation
            if stall_count > stall_threshold:
                theta += 0.1 * rng.standard_normal(n)
                stall_count = 0

    a_final = softmax_normalize(best_theta) * S
    sc = score(a_final)
    if sc < best_score:
        best_score = sc
    return a_final, best_score


def local_search(a_init, time_limit, rng, top_k=10):
    """
    Simulated annealing with peak-directed perturbations.
    Specifically targets reducing the convolution peak by redistributing
    mass from high-contribution positions to low-contribution ones.
    """
    current = a_init.astype(np.float64).copy()
    current_score = score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9997
    iter_count = 0
    epoch_best = best_score

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        iter_count += 1

        if r < 0.30:
            # Peak-directed: reduce contribution to peak lags
            conv = fft_convolve(current, current)
            n_top = min(top_k, len(conv))
            top_lags = np.argsort(conv)[-n_top:]
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            n_reduce = max(3, n // 8)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            epsilon = 0.04 + 0.03 * rng.random()
            candidate[top_idx] *= np.exp(rng.normal(-epsilon, 0.015, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(epsilon, 0.015, size=len(bot_idx)))
        elif r < 0.45:
            # Asymmetric shift: move mass toward one end
            direction = rng.choice([-1, 1])
            width = max(2, int(rng.integers(2, max(3, n // 10))))
            if direction > 0:
                start = int(rng.integers(0, n - 2 * width))
                candidate[start:start + width] *= np.exp(rng.normal(-0.05, 0.02, size=width))
                candidate[start + width:start + 2 * width] *= np.exp(rng.normal(0.05, 0.02, size=width))
            else:
                start = int(rng.integers(width, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.05, 0.02, size=width))
                candidate[start - width:start] *= np.exp(rng.normal(-0.05, 0.02, size=width))
        elif r < 0.60:
            # Multi-block perturbation
            n_blocks = int(rng.integers(2, 6))
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 12))))
                bs = int(rng.integers(0, n - bw))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0.0, 0.04, size=bw))
        elif r < 0.75:
            # Asymmetric reshaping: apply a linear tilt
            tilt = rng.normal(0, 0.1 / n)
            t = np.arange(n, dtype=np.float64)
            candidate *= np.exp(tilt * t)
        elif r < 0.85:
            # Sparsification: reduce a random subset
            n_sparse = max(1, n // 6)
            sparse_idx = rng.choice(n, size=n_sparse, replace=False)
            candidate[sparse_idx] *= np.exp(rng.normal(-0.2, 0.1, size=n_sparse))
            dense_idx = np.setdiff1d(np.arange(n), sparse_idx)
            if len(dense_idx) > 1:
                n_inc = min(8, len(dense_idx))
                inc_idx = rng.choice(dense_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.2, 0.1, size=n_inc))
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current
        else:
            candidate = candidate / s * np.sum(current)

        new_score = score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                epoch_best = best_score
        else:
            delta = new_score - current_score
            if delta < temp * 20:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score

        temp *= cooling
        # Occasional reheating
        if iter_count % 20000 == 0 and temp < 0.0001:
            temp = 0.001

    return best, best_score


def lp_polish(a_init, time_limit, rng):
    """
    LP-inspired polish: iteratively find directions that reduce the peak
    while maintaining feasibility (non-negativity, fixed sum).
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    S = np.sum(current)

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(current, current)
        max_val = np.max(conv)
        # Find active lags (within 5% of max)
        threshold = 0.95 * max_val
        active_lags = np.where(conv >= threshold)[0]

        # Compute contribution of each element to active lags
        contrib = np.zeros(n)
        for lag in active_lags:
            i_min = max(0, lag - n + 1)
            i_max = min(n - 1, lag)
            for i in range(i_min, i_max + 1):
                j = lag - i
                contrib[i] += current[i] * current[j]

        # Direction: reduce high-contrib, increase low-contrib
        n_top = max(2, n // 5)
        n_bot = max(2, n // 5)
        top_idx = np.argsort(contrib)[-n_top:]
        bot_idx = np.argsort(contrib)[:n_bot]

        # Compute step size to maintain sum
        reduction = np.sum(current[top_idx]) * 0.02
        if np.sum(current[bot_idx]) > 1e-10:
            increase = reduction / np.sum(current[bot_idx]) * current[bot_idx]
        else:
            increase = np.zeros(n)

        direction = np.zeros(n)
        direction[top_idx] = -current[top_idx] * 0.02
        direction[bot_idx] = increase
        # Correct for sum change
        direction -= np.sum(direction) / n

        candidate = current + direction
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s * S

        new_score = score(candidate)
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                candidate2 = current + factor * direction
                candidate2 = np.clip(candidate2, 0.0, None)
                s2 = np.sum(candidate2)
                if s2 < 1e-10:
                    continue
                candidate2 = candidate2 / s2 * S
                new_score2 = score(candidate2)
                if new_score2 < current_score:
                    current = candidate2
                    current_score = new_score2
                    if current_score < best_score:
                        best_score = current_score
                        best = current.copy()
                    break

        # Add small noise to escape flat regions
        if rng.random() < 0.1:
            current += 0.005 * rng.standard_normal(n)
            current = np.clip(current, 0.0, None)
            s = np.sum(current)
            if s > 1e-10:
                current = current / s * S

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for optimal non-negative step function.
    Phase 1: Diverse initialization (asymmetric shapes, FFT, random)
    Phase 2: Gradient optimization (Adam + soft-max annealing)
    Phase 3: LP-inspired polish + local search
    Phase 4: Coarse-to-fine upscaling
    Phase 5: Final verification
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Phase 1: Generate diverse initial candidates
    print("\n=== Phase 1: Initialization ===")
    candidates = []

    lengths_to_try = [128, 256, 384, 512, 768, 1024, 1536]

    for n in lengths_to_try:
        if time.monotonic() > deadline - 300:
            break
        inits = []

        # Asymmetric ramp (various asymmetries)
        for asym in [0.2, 0.4, 0.6, -0.3, -0.5]:
            a = ramp_init(n, rng, asym=asym)
            inits.append((f'ramp_{asym}', a))

        # Geometric
        for rate in [0.3, 0.6, -0.3, -0.6, 1.0]:
            a = geometric_init(n, rng, rate=rate)
            inits.append((f'geo_{rate}', a))

        # Power law
        for p in [0.5, 1.0, 1.5, -0.5]:
            a = power_law_init(n, rng, p=p)
            inits.append((f'pow_{p}', a))

        # FFT phase
        a_fft = fft_phase_init(n, rng, n_iter=60)
        inits.append(('fft', a_fft))

        # Multi-tone
        a_mt = multi_tone_init(n, rng, n_tones=3 + rng.integers(0, 3))
        inits.append(('multitone', a_mt))

        # Random
        a_rnd = random_init(n, rng, spread=0.2 + 0.3 * rng.random())
        inits.append(('random', a_rnd))

        for name, a in inits:
            sc = score(a)
            candidates.append((sc, n, a.copy(), name))

    # Also add the provided sequence variants
    if start_seq is not None:
        n0 = len(start_seq)
        candidates.append((best_score, n0, start_seq.copy(), 'provided'))
        # Perturbed versions
        for _ in range(3):
            a_pert = start_seq * np.exp(rng.normal(0, 0.05, size=n0))
            a_pert = np.clip(a_pert, 0.01, None)
            sc = score(a_pert)
            candidates.append((sc, n0, a_pert, 'provided_pert'))

    # Sort and keep top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates")
    for c in candidates[:8]:
        print(f"    score={c[0]:.6f}, n={c[1]}, init={c[3]}")

    # Phase 2: Gradient optimization on best candidates
    print("\n=== Phase 2: Gradient optimization ===")
    n_cand = min(10, len(candidates))
    time_left = deadline - time.monotonic() - 500  # reserve for later phases
    time_per = max(10.0, time_left / n_cand)

    optimized = []
    seen_n = set()
    count_per_n = 0

    for i in range(n_cand):
        if time.monotonic() > deadline - 550:
            break
        sc0, n, a, name = candidates[i]
        # Use more time for smaller n (faster per iteration)
        time_factor = min(1.5, 512.0 / max(n, 128))
        t_alloc = min(time_per * time_factor, deadline - time.monotonic() - 500)
        t_alloc = max(10.0, t_alloc)

        print(f"  [{i + 1}/{n_cand}] n={n}, init={name}, score0={sc0:.6f}, t_alloc={t_alloc:.1f}s")
        a_opt, s_opt = optimize_gradient(n, a, t_alloc, rng)
        print(f"    -> score={s_opt:.6f}")

        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")

        optimized.append((s_opt, n, a_opt))

    # Phase 3: Polish with LP-inspired method + local search
    print("\n=== Phase 3: Polish ===")
    if optimized:
        optimized.sort(key=lambda x: x[0])
        # Polish the top 3
        for i in range(min(3, len(optimized))):
            if time.monotonic() > deadline - 300:
                break
            sc, n, a = optimized[i]
            print(f"  Polish [{i + 1}]: n={n}, score={sc:.6f}")

            # LP polish
            t_lp = min(60.0, (deadline - time.monotonic() - 250) / max(1, 3 - i))
            if t_lp > 10:
                a_lp, s_lp = lp_polish(a, t_lp, rng)
                print(f"    LP polish: {s_lp:.6f}")
                if s_lp < best_score:
                    best_score = s_lp
                    best_seq = a_lp.copy()
                    print(f"    *** New best: {best_score:.6f} ***")
                    a = a_lp
                    sc = s_lp

            # Local search
            t_ls = min(80.0, deadline - time.monotonic() - 200)
            if t_ls > 10:
                a_ls, s_ls = local_search(a, t_ls, rng, top_k=8)
                print(f"    Local search: {s_ls:.6f}")
                if s_ls < best_score:
                    best_score = s_ls
                    best_seq = a_ls.copy()
                    print(f"    *** New best: {best_score:.6f} ***")

    # Phase 4: Coarse-to-fine upscaling from best short sequence
    print("\n=== Phase 4: Coarse-to-fine ===")
    if optimized:
        optimized.sort(key=lambda x: x[0])
        best_short = optimized[0]
        current_n = len(best_short[2])

        for target_n in [512, 1024, 2048]:
            if time.monotonic() > deadline - 200:
                break
            if current_n >= target_n:
                continue
            a_up = upsample(best_short[2], target_n, rng)
            t_alloc = min(80.0, deadline - time.monotonic() - 150)
            if t_alloc < 15:
                break
            print(f"  Upsample to n={target_n}, t_alloc={t_alloc:.1f}s")
            a_opt, s_opt = optimize_gradient(target_n, a_up, t_alloc, rng)
            print(f"    -> score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                best_short = (s_opt, target_n, a_opt)
                current_n = target_n
                print(f"    *** New best: {best_score:.6f} (n={target_n}) ***")

    # Phase 5: Final restarts and polish
    print("\n=== Phase 5: Final refinement ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            n_best = len(best_seq)
            # Try a few restarts from the best
            n_restarts = min(3, int(remaining // 60))
            for r in range(n_restarts):
                if time.monotonic() > deadline - 40:
                    break
                a_new = best_seq.copy() * np.exp(rng.normal(0, 0.08, size=n_best))
                a_new = np.clip(a_new, 0.01, None)
                t_alloc = min(40.0, (deadline - time.monotonic() - 35) / max(1, n_restarts - r))
                a_opt, s_opt = optimize_gradient(n_best, a_new, t_alloc, rng)
                print(f"  Restart {r + 1}: score={s_opt:.6f}")
                if s_opt < best_score:
                    best_score = s_opt
                    best_seq = a_opt.copy()
                    print(f"    *** New best: {best_score:.6f} ***")

        # Final local search polish
        if deadline - time.monotonic() > 20:
            t_final = min(40.0, deadline - time.monotonic() - 15)
            if t_final > 10:
                print(f"  Final polish: t={t_final:.1f}s")
                a_pol, s_pol = local_search(best_seq, t_final, rng, top_k=6)
                if s_pol < best_score:
                    best_score = s_pol
                    best_seq = a_pol.copy()
                    print(f"  After final polish: {best_score:.6f}")

    # Final verification
    print(f"\n{'=' * 50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        if final_score < best_score or best_score == np.inf:
            best_score = final_score
            best_seq = np.array(final_list)
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
