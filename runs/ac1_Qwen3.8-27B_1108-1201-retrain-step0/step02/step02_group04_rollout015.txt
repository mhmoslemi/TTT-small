LP-based optimization to minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses linear programming to find descent directions for the peak of the auto-convolution,
combined with multiple initializations, random restarts, and coordinate refinement.

import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a):
    """Compute a*a (full auto-convolution) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

    """Compute the evaluation score for a sequence."""
    n = len(a)
    s = np.sum(a)

        return np.inf

    return 2.0 * n * np.max(c) / (s * s)

    n = len(a)
    A = np.zeros((2*n-1, n))

        A[j:j+n, j] = a
    return A

    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_lp(a_init, time_limit, use_random=False, seed=0):
    """Main LP-based optimization loop."""
    rng = np.random.default_rng(seed)
    a = a_init.astype(np.float64).copy()
    n = len(a)
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    M = np.max(fft_convolve(a))
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    stall = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = lp_find_direction(a, M)
        if b is None or np.sum(b) < 1e-12:
            stall += 10
            if stall > 200:
                break
            a = a + 0.01 * rng.standard_normal(n)
            a = np.clip(a, 0, None)
            a = a / np.sum(a)
            M = np.max(fft_convolve(a))
            iter_count += 1
            continue
        
        b_sum = np.sum(b)
        b = b / b_sum
        
        improved = False
        for t in [0.5, 0.3, 0.7, 1.0, 0.1, 0.2, 0.4, 0.8]:
            a_t = (1.0 - t) * a + t * b
            Mt = np.max(fft_convolve(a_t))
            if Mt < M * (1.0 - 1e-10):
                a = a_t
                a = np.clip(a, 0, None)
                s = np.sum(a)
                if s < 1e-10:
                    break
                a = a / s
                M = np.max(fft_convolve(a))
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                improved = True
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
            perturbation = 0.005 + 0.01 * (stall / 100.0)
            a = a + perturbation * rng.standard_normal(n)
            a = np.clip(a, 0, None)
            a = a / np.sum(a)
            M = np.max(fft_convolve(a))
        else:
            stall = 0
        
        iter_count += 1
    
    return best_a, 2.0 * n * best_M
