    while maximizing the sum increase.

"""
Optimized peak minimization for step function sequences.
Uses multi-peak Nesterov subgradient descent, LP-based refinement,
and coordinate descent polish to minimize the evaluation score.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:len(a) + len(b) - 1]

def score(a):
    """Compute the evaluation score."""
    a = np.maximum(a, 0)
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (S * S)

def softmax(x):
    """Stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def optimize_subgradient(a_init, time_limit, rng):
    """
    Multi-peak Nesterov subgradient descent.
    Works in log-space (softmax parameterization) to maintain positivity.
    """
    n = len(a_init)
    S_init = a_init.sum()
    
    # Initialize log-parameters
    theta = np.log(a_init + 1e-15)
    v = np.zeros(n)  # Nesterov momentum
    
    best_a = a_init.copy()
    best_score = score(a_init)
    
    t0 = time.monotonic()
    step = 0
    base_lr = 1.0
    momentum = 0.9
    top_k = max(4, n // 16)
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        elapsed_frac = (time.monotonic() - t0) / time_limit
        
        # Compute current a from theta
        p = softmax(theta)  # p sums to 1
        a = p * S_init
        
        # Compute convolution and peaks
        c = fft_convolve(a, a)
        c_max = c.max()
        current_score = 2.0 * n * c_max / S_init**2
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
        
        # Find top-k peaks with soft weighting
        c_sorted = np.sort(c)[::-1]
        top_vals = c_sorted[:top_k]
        # Temperature for peak weighting
        tau = c_max * 0.05 * (1 + elapsed_frac)  # Decreasing temperature
        w = np.exp(-(c_max - c) / tau)
        w = w / w.sum()
        
        # Compute weighted gradient
        # grad of (a*a)[k] w.r.t. theta[j] = S_init^2 * p[j] * (grad w.r.t. a)[j] * (1/S_init)
        # = S_init * p[j] * 2*a[k-j] * I[0<=k-j<n]
        # Actually, let's compute grad w.r.t. a, then convert to theta-space.
        
        # grad w.r.t. a:
        g_a = np.zeros(n)
        for k in range(min(top_k, 2*n-1)):
            wk = w[k]
            if wk < 1e-10:
                continue
            # (a*a)[k] = sum_i a[i]*a[k-i], grad w.r.t. a[j] = 2*a[k-j]*I[0<=k-j<n]
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                g_a[j] += wk * 2.0 * a[k - j]
        
        # Convert to theta-space: a = S_init * softmax(theta)
        # da[j]/dtheta[j'] = S_init * p[j] * (delta[j,j'] - p[j'])
        # d_score/dtheta[j'] = sum_j (d_score/da[j]) * (da[j]/dtheta[j'])
        # = S_init * sum_j g_a[j] * p[j] * (delta[j,j'] - p[j'])
        # = S_init * (g_a[j'] * p[j'] - p[j'] * sum_j g_a[j] * p[j])
        
        S_dot_g = np.dot(S_init * p, g_a)  # = S_init * sum p[j] * g_a[j]
        g_theta = S_init * p * (g_a - S_dot_g / S_init)
        # Simplify: g_theta = S_init * p * g_a - p * S_dot_g
        g_theta = S_init * p * g_a - p * S_dot_g
        
        # Nesterov update
        lr = base_lr / np.sqrt(1 + 0.01 * step)
        v = momentum * v + g_theta
        theta = theta - lr * v
        
        # Periodic checks and restarts
        if step % 500 == 0:
            a_check = softmax(theta) * S_init
            s_check = score(a_check)
            if s_check < best_score:
                best_score = s_check
                best_a = a_check.copy()
            
            if step % 5000 == 0:
                print(f"    SG step={step}, score={current_score:.6f}, best={best_score:.6f}")
        
        # Restart if stuck
        if step % 10000 == 0 and step > 0:
            # Small perturbation from best
            theta = np.log(best_a + 1e-15)
            theta += 0.1 * rng.standard_normal(n)
            v = np.zeros(n)
    
    return best_a, best_score
