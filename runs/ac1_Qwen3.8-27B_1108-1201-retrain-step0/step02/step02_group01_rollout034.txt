```python
def spectral_iteration(a, N, alpha=0.5, num_iters=100):
    n = len(a)
    S = a.sum()
    for _ in range(num_iters):
        A = fft(a, N)
        # Compress dynamic range of |A| (keep DC fixed)
        A_magnitude = np.abs(A)
        A_magnitude[0] = A_magnitude[0]  # keep DC
        # Compress: A_magnitude[m] = A_magnitude[m]^alpha for m > 0
        A_magnitude[1:] = A_magnitude[1:] ** alpha
        # Reconstruct with original phases
        A_new = A_magnitude * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new))[:n]
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        # Renormalize
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new * (S / s)
        a = a_new
    return a
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using Nesterov-accelerated subgradient
with adaptive top-R smoothing, spectral shaping restarts, and pair-swap
coordinate descent polish.

Key improvements over previous approach:
1. Exact subgradient (top-R average) instead of soft-max, eliminating bias
2. Nesterov momentum with adaptive step size for accelerated convergence
3. Spectral shaping (flat-spectrum iteration) for generating diverse candidates
4. Pair-swap coordinate descent for final polish
5. Multi-start strategy with length exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def a_from_theta(theta, S):
    """Convert log-parameters to sequence: a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def compute_subgrad_a(a, R):
    """Compute subgradient of max(c) w.r.t. a, averaged over top R lags.
    Returns (g, c) where g[j] = (2/R) * sum_{k in top_R} a[k-j]."""
    n = len(a)
    c = fft_convolve_self(a)
    top_idx = np.argsort(c)[::-1][:R]
    
    h = np.zeros(2 * n - 1)
    h[top_idx] = 1.0
    a_rev = a[::-1]
    conv = fft_convolve(h, a_rev)
    g = (2.0 / R) * conv[n - 1:2 * n - 1]
    
    return g, c


def nesterov_subgrad(theta, S, time_budget, R_start=3, R_end=30, eta_0=2.0, beta=0.9, name=""):
    """Nesterov-accelerated subgradient method on log-parameters."""
    n = len(theta)
    start = time.monotonic()
    
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    best_theta = theta.copy()
    
    z = theta.copy()
    theta_prev = theta.copy()
    t = 0
    counter = 0
    
    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        
        # Adaptive R
        R = max(1, int(R_start + (R_end - R_start) * progress))
        
        # Nesterov look-ahead
        if t > 1:
            z = z + beta * (z - theta_prev)
        theta_prev = z
        
        # Evaluate at look-ahead point
        a_z = a_from_theta(z, S)
        c_z = fft_convolve_self(a_z)
        obj_z = c_z.max()
        
        if obj_z < best_obj:
            best_obj = obj_z
            best_a = a_z.copy()
            best_theta = z.copy()
            counter = 0
        else:
            counter += 1
        
        # Compute subgradient at z
        g_a, _ = compute_subgrad_a(a_z, R)
        dot = np.dot(a_z, g_a)
        grad_theta = a_z * g_a - (a_z / S) * dot
        
        # Step size
        eta = eta_0 / np.sqrt(t)
        
        # Update
        theta_new = z - eta * grad_theta
        
        # Simple safeguard: if objective increased a lot, reduce step
        a_new = a_from_theta(theta_new, S)
        c_new = fft_convolve_self(a_new)
        obj_new = c_new.max()
        
        if obj_new > obj_z * 1.1:
            # Try smaller step
            theta_new = z - eta * 0.1 * grad_theta
            a_new = a_from_theta(theta_new, S)
            c_new = fft_convolve_self(a_new)
            obj_new = c_new.max()
        
        z = theta_new
        
        # Periodic print
        if t % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"  {name} t={t} obj={best_obj:.6f} elapsed={elapsed:.1f}s")
    
    return best_theta, best_a, best_obj


def spectral_shaping(a, alpha=0.5, num_iters=200):
    """Flat-spectrum iteration to flatten the convolution."""
    n = len(a)
    S = a.sum()
    N = 1 << (2 * n - 1).bit_length()
    a = a.copy()
    
    for _ in range(num_iters):
        A = fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress dynamic range (keep DC)
        mag[0] = mag[0]
        mag_non_dc = mag[1:]
```

```python
def spectral_shaping(a, alpha=0.5, num_iters=200):
    n = len(a)
    S = a.sum()
    N = 1 << (2 * n - 1).bit_length()
    a = a.copy().astype(np.float64)
    
    for _ in range(num_iters):
        A = fft(a, N)
        mag = np.abs(A)
        phase = A / (mag + 1e-300)  # unit phasor
        
        # Compress: mag^alpha for non-DC
        mag_compressed = mag ** alpha
        mag_compressed[0] = mag[0]  # keep DC fixed
        
        A_new = mag_compressed * phase
        a = np.real(ifft(A_new))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a * (S / s)
    
    return a
```

For a pair swap (transfer delta from i to j):

- a[j] -> a[j] + delta
- sum is preserved.

The change in c[k] is:
delta_c[k] = -delta * (2*a[k-i]) + delta * (2*a[k-j])
= 2*delta * (a[k-j] - a[k-i])

So the new c[k] = old_c[k] + 2*delta*(a[k-j] - a[k-i]).

We want to choose delta to minimize max(c_new). This is a 1D optimization.

For efficiency, I can:
1. Identify the "critical" lags (where c is near the max).
2. For those lags, compute the slope d[k] = 2*(a[k-j] - a[k-i]).
3. The new max is max_k (c[k] + delta * d[k]).
4. Find the delta that minimizes this max.

The function max_k (c[k] + delta * d[k]) is a convex piecewise-linear function of delta. The minimum occurs where the "upper envelope" has a kink, or at a boundary.

In practice, I can use the "optimal delta" formula:
- For delta > 0 (moving mass from i to j): the lags where d[k] > 0 will increase, and d[k] < 0 will decrease.
- The optimal delta is where the maximum of the increasing lags equals the maximum of the decreasing lags (if such a point exists), otherwise at a boundary.
