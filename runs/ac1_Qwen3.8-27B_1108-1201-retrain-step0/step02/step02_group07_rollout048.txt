"""
Optimized step function search using LP-based iteration (SOTA-inspired),
smoothed log-sum-exp gradient descent, and multi-scale exploration
to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: principled LP blending direction, sparse LP for scalability,
chirp/window initializations, adaptive step sizes, and multi-length search.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_normalized(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_sparse_A(f, n):
    """Build sparse constraint matrix for LP: A[k,j] = f[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_blend_direction(f, M):
    """Solve LP: maximize sum(g) s.t. g>=0, (f*g)[k] <= M for all k.
    Returns normalized g or None if infeasible."""
    n = len(f)
    A = build_sparse_A(f, n)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status != 0:
            return None
        g = res.x
        s = g.sum()
        if s < 1e-12:
            return None
        return g / s
    except Exception:
        return None


def lp_iterate(a, n_iters=200, t_init=0.15, t_min=0.03, time_limit=60.0, verbose=False):
    """LP-based iterative optimization. Returns best sequence found."""
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a)) / len(a)
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = obj_normalized(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_convolve(a)
        M = conv.max()
        if M < 1e-15:
            break
        g = lp_blend_direction(a, M)
        if g is None:
            break
        # Adaptive step size: start moderate, decrease
        frac = it / max(1, n_iters)
        t = t_init * (1.0 - 0.7 * frac) + t_min
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-12:
            break
        a_new = a_new / sn
        obj_new = obj_normalized(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        if verbose and it % 20 == 0:
            print("    LP iter %d: obj=%.6f, t=%.3f" % (it, best_obj, t))
    return best_a


def compute_smoothed_grad(a, conv, temperature):
    """Compute smoothed gradient of 2*n*max(conv) using log-sum-exp."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return 0.0, np.zeros(n)
    scaled = temperature * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    else:
        w = w / w_sum
    smoothed_max = max_c + np.log(w_sum) / temperature
    obj_val = 2.0 * n * smoothed_max
    # Gradient: d(obj)/d(a[j]) = 2*n * sum_k w[k] * 2*a[k-j] (where valid)
    grad = np.zeros(n)
    # Use the structure: for each k, contributes to positions j in [max(0,k-n+1), min(n-1,k)]
    # grad[j] += w[k] * 2*a[k-j] for all k where j is in valid range
    # Equivalently: for each j, sum over k in [j, j+n-1] of w[k] * 2*a[k-j]
    # This is a cross-correlation! grad = 2 * 2*n * ifft(conj(fft(w)) * fft(a_padded))
    # But let's just do it directly for clarity (O(n^2) but n is small enough)
    for k in range(len(conv)):
        if w[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += w[k] * 2.0 * a[k - j]
    grad *= 2.0 * n
    return obj_val, grad

    from scipy.optimize import project

        # Project onto simplex {x >= 0, sum(x) = 1}

            temperature *= 1.5
            if lr < 0.001:

def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def peak_reduction_search(a, time_limit=30.0, seed=0):
    """Targeted local search: attack the peak of the convolution."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a)) / len(a)
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = obj_normalized(a)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        max_c = conv.max()
        if max_c < 1e-15:
            break
        # Find peak positions
        peak_thresh = max_c * 0.8
        peak_pos = np.where(conv >= peak_thresh)[0]
        if len(peak_pos) == 0:
            peak_pos = np.array([np.argmax(conv)])
        k = int(rng.choice(peak_pos))
        # Find which j positions contribute most to conv[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        contribs = np.array([a[j] * a[k - j] for j in range(js, je + 1)])
        total = contribs.sum()
        if total < 1e-15:
            continue
        # Pick a high-contributing position to reduce
        probs = contribs / total
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        # Pick a low-contributing position to increase (prefer positions far from peaks)
        target_j = int(rng.integers(n))
        for _ in range(10):
            cand = int(rng.integers(n))
            # Check if cand contributes to any peak
            contrib_to_peaks = 0.0
            for pk in peak_pos[:5]:
                d = pk - cand
                if 0 <= d < n:
                    contrib_to_peaks += a[d]
                d2 = cand - pk
                if 0 <= d2 < n and d2 != 0:
                    contrib_to_peaks += a[d2]
            if contrib_to_peaks < a.mean():
                target_j = cand
                break
        if target_j == top_j:
            continue
        delta = a[top_j] * rng.uniform(0.02, 0.08)
        if delta < 1e-10:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < -1e-15:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        a[top_j] = max(0.0, a[top_j])
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a.copy()
        else:
            # Accept with some probability (simulated annealing style)
            frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
            T = 0.01 * (1.0 - frac) ** 2 + 0.001
            if rng.random() < np.exp(-(new_obj - cur_obj) / max(T, 1e-12)):
                pass  # accept

                a[top_j] += delta
                a[target_j] -= delta
    return best_a


def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = []
    # Uniform

    # Gaussian windows
    for sigma in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - n / 2.0) / (sigma * n)) ** 2)
        g = np.maximum(g, 0)
        if g.sum() > 1e-10:
            inits.append(('gauss_%.2f' % sigma, g / g.sum()))
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * (x - n / 2.0 + 0.5) / n))
    rc = np.maximum(rc, 0)
    if rc.sum() > 1e-10:
        inits.append(('raised_cos', rc / rc.sum()))
    # Tukey window (alpha=0.5)
    alpha = 0.5
    n_half = n / 2.0
    tw = np.ones(n)
    trans = int(alpha * n_half)
    for i in range(trans):
        tw[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / alpha / n - 1) / 2))
        tw[n - 1 - i] = tw[i]
    tw = np.maximum(tw, 0)
    if tw.sum() > 1e-10:
        inits.append(('tukey', tw / tw.sum()))
    # Bimodal (mass at edges)
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits.append(('bimodal', bi / bi.sum()))
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(('ramp', ramp / ramp.sum()))
    # Chirp-like: use |cos| of quadratic phase
    for mu in [1.0, 2.0, 3.0]:
        phase = mu * x * x / (2.0 * n)
        chirp = np.abs(np.cos(np.pi * phase))
        chirp = np.maximum(chirp, 0)
        if chirp.sum() > 1e-10:
            inits.append(('chirp_%.0f' % mu, chirp / chirp.sum()))
    # Random Dirichlet
    for alpha_r in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha_r)
        inits.append(('rand_%.1f' % alpha_r, r))
    # Parent interpolation
    if parent is not None:
        p = np.maximum(parent, 0).astype(np.float64)
        if p.sum() > 1e-10:
            if len(p) == n:
                inits.append(('parent', p / p.sum()))
            else:

                pp = np.maximum(pp, 0)

    return inits

def coordinate_descent(a, time_limit=30.0):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a)) / len(a)
    a = a / s
    n = len(a)
    best_obj = obj_normalized(a)
    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            # Try adjusting a[j]
            old_val = a[j]
            if old_val < 1e-15:
                # Try small increase
                for frac in [0.001, 0.005, 0.01]:
                    a[j] = old_val + frac / n
                    a = a / a.sum()
                    if obj_normalized(a) < best_obj - 1e-12:
                        best_obj = obj_normalized(a)
                        improved = True
                        break
                a[j] = a[j] * (a[j] / a.sum())  # renormalize properly
                a = a / a.sum()
            else:
                # Try decreases and increases
                for frac in [0.3, 0.5, 0.7, 1.2, 1.5, 2.0]:
                    a[j] = old_val * frac
                    if a[j] < 0:
                        a[j] = 0
                    a = a / a.sum()
                    obj_new = obj_normalized(a)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        improved = True
                        break
                a[j] = old_val  # revert
                a = a / a.sum()
        sweep += 1
        if sweep > 50 and not improved:
            break
    return a

        if a is None:

        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = (a / s).copy()
            print("  [%.1fs] New best: score=%.8f, n=%d" % (time.monotonic() - t_start, score, len(a)))
            return True
        return False

    # Phase 1: Quick scan across lengths and initializations
    print("Phase 1: Quick scan...")
    lengths_to_try = [128, 256, 512, 1024]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths_to_try:
            # Add parent's length and nearby powers of 2
            lengths_to_try = [128, 256, 512, 1024, 2048]

    phase1_budget = min(300.0, (deadline - time.monotonic()) * 0.15)
    candidates = []

    for n in lengths_to_try:
        if time.monotonic() > deadline - 1800:
            break
        inits = generate_inits(n, rng, parent)
        time_per_init = max(3.0, phase1_budget / max(1, len(lengths_to_try) * len(inits)))

        for name, a0 in inits:
            if time.monotonic() > deadline - 1800:
                break
            t_alloc = min(time_per_init, deadline - time.monotonic() - 1800)
            if t_alloc < 2.0:
                break
            # Quick LP optimization
            a_opt = lp_iterate(a0, n_iters=30, t_init=0.12, t_min=0.04, time_limit=t_alloc)
            score = safe_evaluate(a_opt)
            if np.isfinite(score):
                candidates.append((score, n, name, a_opt))
            record(a_opt)

    print("  Phase 1 done. Best so far: %.8f" % best_score)

    # Sort and pick top candidates for deep optimization
    candidates.sort(key=lambda x: x[0])
    top_candidates = []
    seen_keys = set()
    for score, n, name, a_opt in candidates:
        key = (n, name)
        if key not in seen_keys and len(top_candidates) < 10:
            top_candidates.append((n, name, a_opt))
            seen_keys.add(key)

    # Always include current best
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), 'best', best_sequence.copy()))

    # Phase 2: Deep LP optimization
    print("Phase 2: Deep LP optimization...")
    phase2_budget = min(1200.0, (deadline - time.monotonic()) * 0.5)
    n_deep = len(top_candidates)
    time_per_deep = max(10.0, phase2_budget / max(1, n_deep))

    for idx, (n, name, a0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 800:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 800)
        if t_alloc < 5.0:
            break
        print("  [%d/%d] n=%d, init=%s, budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))

        # Multi-round LP with varying step sizes
        a_cur = a0.copy()
        sub_time = t_alloc / 4.0

        a_cur = lp_iterate(a_cur, n_iters=100, t_init=0.2, t_min=0.05, time_limit=sub_time)
        record(a_cur)
        a_cur = lp_iterate(a_cur, n_iters=100, t_init=0.1, t_min=0.03, time_limit=sub_time)
        record(a_cur)
        a_cur = lp_iterate(a_cur, n_iters=150, t_init=0.05, t_min=0.02, time_limit=sub_time)
        record(a_cur)

        # Smoothed gradient polishing
        if time.monotonic() < deadline - 800:
            rem = deadline - time.monotonic() - 800
            a_cur = smoothed_gradient_descent(a_cur, time_limit=min(sub_time, rem), t_init=100.0, lr_init=0.03)
            record(a_cur)

    print("  Phase 2 done. Best: %.8f" % best_score)

    # Phase 3: Peak reduction and coordinate descent
    if best_sequence is not None and time.monotonic() < deadline - 50:
        print("Phase 3: Local refinement...")
        phase3_budget = min(500.0, deadline - time.monotonic() - 50)
        a_best = best_sequence.copy()

        n_refine = 5
        time_per_refine = phase3_budget / n_refine

        for i in range(n_refine):
            if time.monotonic() > deadline - 50:
                break
            t_alloc = min(time_per_refine, deadline - time.monotonic() - 50)
            if t_alloc < 3.0:
                break

            if i % 3 == 0:
                a_opt = peak_reduction_search(a_best, time_limit=t_alloc, seed=seed + 100 + i)
            elif i % 3 == 1:
                a_opt = coordinate_descent(a_best, time_limit=t_alloc)
            else:
                a_opt = lp_iterate(a_best, n_iters=50, t_init=0.03, t_min=0.01, time_limit=t_alloc)
            record(a_opt)
            if best_sequence is not None:
                a_best = best_sequence.copy()

        print("  Phase 3 done. Best: %.8f" % best_score)

    # Phase 4: Final polish with very fine LP steps
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_best = best_sequence.copy()
            # Very fine LP steps
            a_opt = lp_iterate(a_best, n_iters=300, t_init=0.02, t_min=0.005, time_limit=remaining * 0.5)
            record(a_opt)
            if time.monotonic() < deadline - 5:
                rem2 = deadline - time.monotonic() - 5
                a_opt2 = smoothed_gradient_descent(
                    best_sequence if best_sequence is not None else a_opt,
                    time_limit=rem2, t_init=200.0, lr_init=0.01)
                record(a_opt2)

        print("  Phase 4 done. Best: %.8f" % best_score)

    # Fallback
    if best_sequence is None:
        print("  WARNING: No valid sequence found, using fallback.")
        best_sequence = np.ones(128)
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.8f" % best_score)
    print("Sequence length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
