LP-based iterative optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses direct linear programming for best-response computation, spectral
initialization via Fienup method, and coordinate descent for refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def normalize_simplex(a):
    """Project onto probability simplex."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(n):
    """Build binary matrix M where M[k,i]=1 if b[i] appears in conv[k].
    Actually we need: coefficient of b[i] in conv(a,b)[k] is a[k-i] if valid.
    So we build the index structure once."""
    # Return lists of (k, j) pairs or just the structure
    # For efficiency, precompute the ranges
    k_starts = np.arange(n)
    k_ends = np.minimum(np.arange(n) + n, 2*n - 1)
    return k_starts, k_ends

    Returns normalized b or None on failure."""

    for i in range(n):
        k_start = i
        k_end = min(i + n, 2*n - 1)
        A[k_start:k_end, i] = a[:k_end - k_start]
    
    b_ub = np.full(2*n-1, C)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, 
                  method='highs', options={'presolve': True, 'time_limit': 30})
    
    if res.success:
        b = res.x
        s = b.sum()
        if s > 1e-12:
            return b / s
    return None


def fienup_init(n, n_iters=200, rng=None, alpha=0.7):
    """Initialize using Fienup/Gerchberg-Saxton spectral flattening."""
    if rng is None:
        rng = np.random.default_rng()
    
    # Start with random or uniform
    a = rng.dirichlet(np.ones(n))
    # Or: a = np.ones(n) / n
    
    fft_size = 1 << (2*n - 2).bit_length()
    
    for it in range(n_iters):
        # Forward transform
        A = np.fft.rfft(a, fft_size)
        PSD = np.abs(A)**2
        phase = np.angle(A)
        
        # Flatten PSD
        PSD_flat = PSD ** alpha
        # Add small regularization to avoid exact zeros
        PSD_flat = PSD_flat + 1e-10
        
        # Inverse transform
        A_new = np.sqrt(PSD_flat) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to simplex
        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a = a / s
    
    return a


def lp_iterate(a, n_iters=100, t=0.3, C_scale=1.0, verbose=False):
    """Iterate LP-based best response method.
    a: initial point on simplex
    Returns optimized a."""
    n = len(a)
    a = normalize_simplex(a.copy())
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    for it in range(n_iters):
        conv = fast_convolve(a)
        C = conv.max() * C_scale
        
        if C < 1e-15:
            break
        
        b = solve_lp_response(a, C)
        if b is None:
            # Fallback: try with larger C
            b = solve_lp_response(a, C * 1.1)
            if b is None:
                break
        
        a_new = (1 - t) * a + t * b
        a = normalize_simplex(a_new)
        
        cur_conv = fast_convolve(a)
        cur_obj = 2.0 * n * cur_conv.max()
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        if verbose and it % 20 == 0:
            print("  LP iter %d: obj=%.8f" % (it, cur_obj))
    
    return best_a, best_obj


def coordinate_descent(a, n_passes=10, n_trials=50):
    """Local refinement via coordinate descent with pairwise transfers."""
    n = len(a)
    a = normalize_simplex(a.copy())
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    best_obj = cur_obj
    
    for p in range(n_passes):
        improved = False
        for i in range(n):
            # Try transferring from i to random j's
            for trial in range(n_trials):
                j = np.random.randint(n)
                if i == j:
                    continue
                # Transfer small amount from i to j
                for frac in [0.01, 0.05, 0.1, 0.2]:
                    delta = a[i] * frac
                    if delta < 1e-15:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    if a[i] < 0:
                        a[i] += delta
                        a[j] -= delta
                        continue
                    new_conv = fast_convolve(a)
                    new_obj = 2.0 * n * new_conv.max()
                    if new_obj < cur_obj:
                        cur_obj = new_obj
                        conv = new_conv
                        improved = True
                        if cur_obj < best_obj:
                            best_obj = cur_obj
                            best_a = a.copy()
                    else:
                        a[i] += delta
                        a[j] -= delta
                # Also try transferring from j to i
                for frac in [0.01, 0.05, 0.1, 0.2]:
                    delta = a[j] * frac
                    if delta < 1e-15:
                        continue
                    a[j] -= delta
                    a[i] += delta
                    if a[j] < 0:
                        a[j] += delta
                        a[i] -= delta
                        continue
                    new_conv = fast_convolve(a)
                    new_obj = 2.0 * n * new_conv.max()
                    if new_obj < cur_obj:
                        cur_obj = new_obj
                        conv = new_conv
                        improved = True
                        if cur_obj < best_obj:
                            best_obj = cur_obj
                            best_a = a.copy()
                    else:
                        a[j] += delta
                        a[i] -= delta
        
        if not improved:
            break
    
    return best_a, best_obj

LP-based iterative optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses direct linear programming for best-response computation, spectral
initialization, and coordinate descent for local refinement.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def normalize_simplex(a):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def solve_lp_response(a, C):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= C for all k, b >= 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_end = min(i + n, 2 * n - 1)
        A[i:k_end, i] = a[:k_end - i]

LP-based iterative optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses direct linear programming for best-response computation, Fienup spectral
initialization, adaptive step sizes, and coordinate descent for local refinement.
Key improvement over subgradient methods: exact LP best-response at each step.

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def normalize_simplex(a):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def solve_lp_response(a, C):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= C for all k, b >= 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_end = min(i + n, 2 * n - 1)
        A[i:k_end, i] = a[:k_end - i]

    b_ub = np.full(2 * n - 1, C)
    c_obj = -np.ones(n)

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 10})

    if res.success:
        b = res.x
        s = b.sum()
        if s > 1e-12:
            return b / s
    return None


def fienup_init(n, n_iters=200, alpha=0.7, seed=0):
    """Initialize using Fienup/Gerchberg-Saxton spectral flattening."""
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n) * 0.5)
    fft_size = 1 << (2 * n - 2).bit_length()

    for _ in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        PSD = np.abs(A) ** 2
        absA = np.abs(A)
        phase = A / np.maximum(absA, 1e-15)

        PSD_max = PSD.max()
        if PSD_max < 1e-15:
            break
        PSD_norm = PSD / PSD_max
        PSD_flat = PSD_norm ** alpha + 1e-8

        A_new = np.sqrt(PSD_flat) * phase
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = rng.dirichlet(np.ones(n) * 0.5)
        else:
            a = a / s

    return a


def lp_iterate(a, n_iters=100, t=0.3, verbose=False, time_limit=None, t_start=None, C_margin=1.0):
    """LP-based best response iteration with adaptive step size."""
    n = len(a)
    a = normalize_simplex(a.copy())

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    cur_obj = best_obj
    no_improve = 0
    max_no_improve = 80

    for it in range(n_iters):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break

        conv = fast_convolve(a)
        C = conv.max() * C_margin
        if C < 1e-15:
            break

        b = solve_lp_response(a, C)
        if b is None:
            b = solve_lp_response(a, C * 1.1)
            if b is None:
                break

        a_new = (1.0 - t) * a + t * b
        a = normalize_simplex(a_new)

        cur_obj = 2.0 * n * fast_convolve(a).max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            # Reduce step size
            t *= 0.5
            if t < 0.005:
                break
            no_improve = 0
            C_margin = max(0.95, C_margin * 0.97)

        if verbose and it % 100 == 0:
            print("    LP iter %d: obj=%.8f, t=%.4f" % (it, cur_obj, t))

            n_iter = max(15, int(sub / 0.08))
            a_cur, _ = lp_iterate(a_cur, n_iters=n_iter, t=t_val,
                                  time_limit=sub, t_start=time.monotonic())
            score = safe_evaluate(a_cur)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_cur.copy()
                print("    New best after LP t=%.2f: %.8f" % (t_val, score))

        # Coordinate descent on the refined point
        if time.monotonic() < deadline - total_budget * 0.15:
            cd_time = min(30.0, (deadline - time.monotonic() - total_budget * 0.15) * 0.4)
            if cd_time > 5.0:
                a_cd, _ = cd_fast(a_cur, time_limit=cd_time, seed=seed + idx * 777)
                score_cd = safe_evaluate(a_cd)
                if np.isfinite(score_cd) and score_cd < best_score:
                    best_score = score_cd
                    best_sequence = a_cd.copy()
                    print("    New best after CD: %.8f" % score_cd)

    # === Phase 4: Final polish ===
    print("Phase 4: Final polish...")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 5
        a_best = normalize_simplex(best_sequence.copy())

        # Very fine LP steps
        half = remaining * 0.45
        n_iter = max(10, int(half / 0.08))
        a_opt, _ = lp_iterate(a_best, n_iters=n_iter, t=0.008,
                              time_limit=half, t_start=time.monotonic(), C_margin=0.99)
        score = safe_evaluate(a_opt)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print("  Polish LP: %.8f" % score)

        # Final coordinate descent
        if time.monotonic() < deadline - 5:
            rem = deadline - time.monotonic() - 5
            if rem > 5:
                a_cd, _ = cd_fast(best_sequence, time_limit=rem, seed=seed + 4242)
                score_cd = safe_evaluate(a_cd)
                if np.isfinite(score_cd) and score_cd < best_score:
                    best_score = score_cd
                    best_sequence = a_cd.copy()
                    print("  Polish CD: %.8f" % score_cd)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = safe_evaluate(best_sequence)

    print("FINAL: score=%.8f, n=%d, time=%.0fs" %
          (best_score, len(best_sequence), time.monotonic() - t_start))

    return [float(x) for x in best_sequence]
