LP-based alternating optimization for minimizing the evaluation score of non-negative sequences.
Uses the state-of-the-art LP approach (alternating between fixing one sequence and optimizing

polishing and coordinate descent for final refinement. Explores multiple initializations and

import numpy as np
import time
import cvxpy as cp


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast evaluation of the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:

    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def normalize_one(a):
    """Normalize sequence to sum 1."""
    S = np.sum(a)
    if S < 1e-15:
        return np.ones(len(a)) / len(a)

    """Build matrix A such that conv(a, b) = A @ b."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))

        A[start:end, j] = a[:end - start]

    """
    LP-based alternating optimization.
    a_init: initial sequence (will be normalized to sum=1)
    Returns best sequence found (normalized to sum=1) and its score.
    """
    a = normalize_one(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * np.max(fft_convolve(a, a))

    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    patience = 200

    print(f"    LP optimize: n={n}, initial score={best_score:.8f}")

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        # Solve LP: find b minimizing peak of conv(a, b)
        b_val, t_lp = solve_lp_min_peak(a)

        if b_val is None:
            # Fallback: use old a
            b_val = a.copy()
            t_lp = best_score / (2.0 * n)

        # The LP gives us a direction. Try different blend factors.
        # With t=1, we get peak = t_lp which should be <= current peak.
        # Try t=1 first, then blends.
        t_candidates = [1.0, 0.8, 0.6, 0.4, 0.2, 0.5, 0.9]

        old_peak = np.max(fft_convolve(a, a))
        best_new_score = 2.0 * n * old_peak
        best_new_a = a.copy()

        for t_blend in t_candidates:
            a_new = (1.0 - t_blend) * a + t_blend * b_val
            a_new = np.clip(a_new, 0, None)
            S_new = np.sum(a_new)
            if S_new < 1e-10:
                continue
            a_new = a_new / S_new
            score_new = 2.0 * n * np.max(fft_convolve(a_new, a_new))
            if score_new < best_new_score:
                best_new_score = score_new
                best_new_a = a_new.copy()

        a = best_new_a

        if best_new_score < best_score:
            best_score = best_new_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            # Try a perturbation to escape
            r = rng.random()
            if r < 0.5:
                # Add noise
                a = a + 0.01 * rng.standard_normal(n)
                a = np.clip(a, 1e-6, None)
                a = normalize_one(a)
            else:
                # Random restart blended with current
                a_rand = np.abs(rng.standard_normal(n)) + 0.1
                a = 0.7 * a + 0.3 * normalize_one(a_rand)
                a = normalize_one(a)
            no_improve = 0

        if iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iter_count}, score={best_score:.8f}, t_lp={t_lp:.6f}, t={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def lse_gradient_descent(a_init, time_limit, rng):
    """
    Log-sum-exp gradient descent for polishing.
    Works directly in the non-negative space with projection.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    a = a / S  # normalize to sum=1
    best_a = a.copy()
    best_score = 2.0 * n * np.max(fft_convolve(a, a))

    # Adam parameters
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    step = 0

    t0 = time.monotonic()
    no_improve = 0
    patience = 500

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit

        # Temperature schedule: start large, decrease
        tau = max(0.001, 1.0 * (1.0 - elapsed_frac) ** 2)

        # Compute log-sum-exp of convolution
        c = fft_convolve(a, a)
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        sum_exp = np.sum(exp_shifted)
        w = exp_shifted / sum_exp  # weights (softmax of conv)
        lse_value = c_max + tau * np.log(sum_exp)

        # Gradient of LSE w.r.t. a: 2 * (a_rev * w) centered
        # d/da[j] of conv(a,a)[k] = a[k-j] (if valid) + a[j+k-n+1] (if valid) ... 
        # Actually: d/da[j] of sum_k w_k * conv(a,a)[k] 
        # = sum_k w_k * (a[k-j] * I[0<=k-j<n] + a[k-j] * I[0<=k-j<n])
        # Wait, conv(a,a)[k] = sum_i a[i]*a[k-i]
        # d/da[j] conv(a,a)[k] = a[k-j]*I[0<=k-j<n] + a[k-j]*I[j<=k<n+j]
        # Hmm, let me think more carefully.
        # conv(a,a)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]
        # d/da[j] conv(a,a)[k] = a[k-j]*I[j in range] + a[k-j]*I[k-j in range]
        # = a[k-j] * (I[0<=j<n and 0<=k-j<n] + I[0<=k-j<n and 0<=j<n])
        # Since 0<=j<n always (j is our index), this is:
        # = a[k-j] * (I[0<=k-j<n] + I[0<=k-j<n]) = 2*a[k-j]*I[0<=k-j<n]
        # So gradient[j] = 2 * sum_k w[k] * a[k-j] * I[0<=k-j<n]
        # = 2 * sum_{k: 0<=k-j<n} w[k] * a[k-j]
        # = 2 * sum_{i: 0<=k<n, k=j+i} w[j+i] * a[i]
        # = 2 * (w_shifted * a) where w_shifted[i] = w[j+i]
        # This is a cross-correlation of w and a.

        # More simply: gradient = 2 * fft_convolve(a_reversed, w)[n-1 : 2n-1]
        a_rev = a[::-1]
        corr = fft_convolve(a_rev, w)
        grad = 2.0 * corr[n - 1:2 * n]

        # Normalize gradient (since we're constraining sum=1)
        inner = np.dot(a, grad)
        grad = grad - inner * np.ones(n) / n  # project onto sum=0 hyperplane
        # Actually for sum constraint, the projected gradient is grad - mean(grad)
        grad = grad - np.mean(grad)

        # Adam step
        step += 1
        base_lr = 0.5 / np.sqrt(step + 1)
        if elapsed_frac < 0.3:
            lr = base_lr * 2.0
        elif elapsed_frac < 0.7:
            lr = base_lr
        else:
            lr = base_lr * 0.3

        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * grad ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.clip(a_new, 0, None)
        a_new = normalize_one(a_new)

        # Check score every 10 steps
        if step % 10 == 0:
            score = 2.0 * n * np.max(fft_convolve(a_new, a_new))
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 10

            if no_improve > patience:
                # Escape
                a = best_a + 0.05 * rng.standard_normal(n)
                a = np.clip(a, 1e-6, None)
                a = normalize_one(a)
                no_improve = 0

        if step % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD step={step}, score={best_score:.8f}, tau={tau:.5f}, t={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def coordinate_descent_polish(a_init, time_limit, rng):
    """
    Coordinate descent: for each element, try to adjust it to improve the score.
    Uses the fact that changing a[j] affects conv(a,a)[k] for k in [j, j+n-1] (approximately).
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    a = a / S
    best_a = a.copy()
    best_score = 2.0 * n * np.max(fft_convolve(a, a))

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        improved = False

        # Random order
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            base = a[j]
            if base < 1e-8:
                # Try turning on this element
                for frac in [0.01, 0.05, 0.1]:
                    trial = a.copy()
                    trial[j] = frac / n
                    S_t = np.sum(trial)
                    trial = trial / S_t
                    s = 2.0 * n * np.max(fft_convolve(trial, trial))
                    if s < best_score:
                        best_score = s
                        best_a = trial.copy()
                        a = trial.copy()
                        improved = True
                        break
                    elif s < 2.0 * n * np.max(fft_convolve(a, a)):
                        continue
                    else:
                        break
                continue

            # Try scaling this element
            for frac in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                trial = a.copy()
                trial[j] = base * frac
                if trial[j] < 0:
                    continue
                S_t = np.sum(trial)
                if S_t < 1e-10:
                    continue
                trial = trial / S_t
                s = 2.0 * n * np.max(fft_convolve(trial, trial))
                if s < 2.0 * n * np.max(fft_convolve(a, a)):
                    a = trial.copy()
                    improved = True
                    if s < best_score:
                        best_score = s
                        best_a = a.copy()
                else:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 5:
                # Random perturbation
                j = int(rng.integers(0, n))
                a[j] = max(0, a[j] * np.exp(rng.normal(0, 0.2)))
                a = normalize_one(a)
                no_improve = 0
        else:
            no_improve = 0

        if iteration % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iteration}, score={best_score:.8f}, t={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def make_structured_init(n, rng, method='wide'):
    """Create structured initial sequences with good properties."""
    t = np.arange(n, dtype=np.float64)

    if method == 'wide':
        # Wide-band: sum of many incommensurate frequencies
        a = np.ones(n)
        for k in range(1, 20):
            freq = k * (1.0 + 0.5 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.12 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'tapered':
        # Tapered ends (Hann-like window)
        a = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
        a += 0.1 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'two_gap':
        # Two high blocks separated by a gap
        a = np.ones(n)
        a[:n//4] = 1.3
        a[n//4:n//2] = 0.7
        a[n//2:3*n//4] = 1.3
        a[3*n//4:] = 0.7
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)


def peak_shave_step(a, rng, n_top=8, n_bot=16):
    """
    One peak-shaving step: reduce top contributors to the peak, increase others.
    """
    n = len(a)
    c = fft_convolve(a, a)
    peak_idx = int(np.argmax(c))

    # Compute contribution of each element to the peak
    contribs = np.zeros(n)
    for i in range(n):
        if 0 <= peak_idx - i < n:
            contribs[i] = a[i] * a[peak_idx - i]

    top_idx = np.argsort(contribs)[::-1][:n_top]
    bottom_idx = np.argsort(contribs)[:n_bot]

    reduction = rng.uniform(0.01, 0.04)
    a_new = a.copy()
    a_new[top_idx] *= (1.0 - reduction)
    a_new[bottom_idx] *= (1.0 + reduction * n_top / n_bot)
    a_new = np.clip(a_new, 0, None)
    return a_new


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimizing the evaluation score.
    Phase 1: LP-based alternating optimization (primary)
    Phase 2: LP from different initializations
    Phase 3: Log-sum-exp gradient polishing
    Phase 4: Coordinate descent polish
    Phase 5: Peak shaving
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 40.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Determine main sequence length
    n_main = 256

    # Prepare initial sequence for main optimization
    if start_seq is not None and len(start_seq) == n_main:
        a0 = start_seq.copy()
    elif start_seq is not None:
        x_old = np.linspace(0, 1, len(start_seq))
        x_new = np.linspace(0, 1, n_main)
        a0 = np.interp(x_new, x_old, start_seq)
    else:
        a0 = make_structured_init(n_main, rng, 'wide')

    # === PHASE 1: LP-based alternating optimization (55% of time) ===
    print(f"\n{'='*60}")
    print(f"PHASE 1: LP alternating optimization (n={n_main})")
    print(f"{'='*60}")

    t_phase1 = 0.55 * time_budget
    a_norm = normalize_one(a0)
    lp_best, lp_score = lp_alternate_optimize(a_norm, t_phase1, rng)

    if lp_score < best_score:
        best_score = lp_score
        best_seq = lp_best * np.sum(a0)  # rescale
        print(f"  Phase 1 result: score={best_score:.8f} *** NEW BEST ***")
    else:
        print(f"  Phase 1 result: score={lp_score:.8f} (no improvement over {best_score:.8f})")

    # === PHASE 2: LP from different initializations (15% of time) ===
    print(f"\n{'='*60}")
    print(f"PHASE 2: LP from different initializations")
    print(f"{'='*60}")

    t_phase2 = 0.15 * time_budget
    t_per_init = t_phase2 / 3

    for init_method in ['tapered', 'two_gap', 'wide']:
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(t_per_init, max(10.0, deadline - time.monotonic() - 30))
        if t_alloc < 10:
            break

        a_try = make_structured_init(n_main, rng, init_method)
        a_try_norm = normalize_one(a_try)
        lp_b, lp_s = lp_alternate_optimize(a_try_norm, t_alloc, rng)

        if lp_s < best_score:
            best_score = lp_s
            best_seq = lp_b.copy()
            print(f"  Init '{init_method}': score={best_score:.8f} *** NEW BEST ***")
        else:
            print(f"  Init '{init_method}': score={lp_s:.8f} (no improvement)")

    # === PHASE 3: Log-sum-exp gradient polishing (15% of time) ===
    print(f"\n{'='*60}")
    print(f"PHASE 3: LSE gradient polishing")
    print(f"{'='*60}")

    if best_seq is not None:
        t_phase3 = min(0.15 * time_budget, max(30.0, deadline - time.monotonic() - 60))
        if t_phase3 > 15:
            a_to_polish = normalize_one(best_seq)
            polished, pol_score = lse_gradient_descent(a_to_polish, t_phase3, rng)
            if pol_score < best_score:
                best_score = pol_score
                best_seq = polished.copy()
                print(f"  After LSE-GD: score={best_score:.8f} *** NEW BEST ***")
            else:
                print(f"  After LSE-GD: score={pol_score:.8f} (no improvement)")

    # === PHASE 4: Coordinate descent (10% of time) ===
    print(f"\n{'='*60}")
    print(f"PHASE 4: Coordinate descent polish")
    print(f"{'='*60}")

    if best_seq is not None:
        t_phase4 = min(0.10 * time_budget, max(20.0, deadline - time.monotonic() - 30))
        if t_phase4 > 10:
            cd_result, cd_score = coordinate_descent_polish(best_seq, t_phase4, rng)
            if cd_score < best_score:
                best_score = cd_score
                best_seq = cd_result.copy()
                print(f"  After CD: score={best_score:.8f} *** NEW BEST ***")
            else:
                print(f"  After CD: score={cd_score:.8f} (no improvement)")

    # === PHASE 5: Peak shaving (5% of time) ===
    print(f"\n{'='*60}")
    print(f"PHASE 5: Peak shaving")
    print(f"{'='*60}")

    if best_seq is not None:
        t_phase5 = max(10.0, deadline - time.monotonic() - 20)
        if t_phase5 > 10:
            a_cur = best_seq.astype(np.float64).copy()
            cur_score = local_score(a_cur)
            t0 = time.monotonic()
            step = 0
            while time.monotonic() - t0 < t_phase5:
                step += 1
                a_try = peak_shave_step(a_cur, rng)
                s_try = local_score(a_try)
                if s_try < cur_score:
                    a_cur = a_try
                    cur_score = s_try
                    if s_try < best_score:
                        best_score = s_try
                        best_seq = a_try.copy()
                        print(f"    Peak shave step {step}: score={best_score:.8f} *** NEW BEST ***")
                elif step % 500 == 0:
                    # Try a gentler version
                    a_try2 = a_cur.copy()
                    c = fft_convolve(a_try2, a_try2)
                    peak_idx = int(np.argmax(c))
                    contribs = np.zeros(len(a_try2))
                    for i in range(len(a_try2)):
                        if 0 <= peak_idx - i < len(a_try2):
                            contribs[i] = a_try2[i] * a_try2[peak_idx - i]
                    top_idx = np.argsort(contribs)[::-1][:4]
                    a_try2[top_idx] *= 0.98
                    a_try2 = np.clip(a_try2, 0, None)
                    s_try2 = local_score(a_try2)
                    if s_try2 < cur_score:
                        a_cur = a_try2
                        cur_score = s_try2
                        if s_try2 < best_score:
                            best_score = s_try2
                            best_seq = a_try2.copy()

    # === FINAL: Try a different sequence length if time remains ===
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\n{'='*60}")
        print(f"TRYING n=384")
        print(f"{'='*60}")
        t_try = min(30.0, deadline - time.monotonic() - 20)
        if t_try > 15:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, 384)
            a_384 = np.interp(x_new, x_old, best_seq)
            a_384_norm = normalize_one(a_384)
            lp_384, s_384 = lp_alternate_optimize(a_384_norm, t_try, rng)
            if s_384 < best_score:
                best_score = s_384
                best_seq = lp_384.copy()
                print(f"  n=384: score={best_score:.8f} *** NEW BEST ***")
            else:
                print(f"  n=384: score={s_384:.8f} (no improvement)")

    # === FINAL VERIFICATION ===
    print(f"\n{'='*60}")
    print(f"FINAL RESULT")
    print(f"{'='*60}")

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"  n={len(final_list)}, local_score={best_score:.8f}, evaluate_sequence={final_eval:.8f}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
