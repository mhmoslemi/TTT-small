from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

    a = np.clip(a, 0.0, 1000.0)

    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

    Returns the optimal b vector.

For k=0: only j=0 contributes (since k-j = -j < 0 for j > 0). A_ub[0, 0] = f[0]. ✓
For k=n-1: j ranges from 0 to n-1 (all valid since k-j ranges from n-1 to 0). A_ub[n-1, j] = f[n-1-j]. ✓
For k=2n-2: only j=n-1 contributes (k-j = 2n-2-j >= 0 requires j <= 2n-2, and k-j < n requires j > n-2, so j = n-1). A_ub[2n-2, n-1] = f[2n-2-(n-1)] = f[n-1]. ✓

Now, one efficiency concern: building A_ub in a double loop is O(n^2) per LP call. For n=384, that's ~147K operations per iteration, which is fine.

But actually, I can build A_ub more efficiently using the structure. The matrix A_ub[:, :n] is a convolution matrix. I can build it using:

```python
# Build the convolution matrix
# A[k, j] = f[k-j] if 0 <= k-j < n
A = np.zeros((2*n-1, n))
for j in range(n):
    # f[k-j] is valid when 0 <= k-j < n, i.e., j <= k < j+n
    A[j:j+n, j] = f
```

This is O(n^2) but with better cache behavior. For n=384, it's 384 assignments of 384 elements each = 147K operations, very fast.

Let me now think about the overall algorithm flow:

1. Start from the provided sequence (n=384) or a good initialization.
2. Run LP2 (minimize cross-correlation peak) iteratively.
3. After each iteration, check if the actual peak (max of g*g) decreased.
4. If not, try a convex combination: f_new = (1-t)f + t*g for small t.
5. Periodically try different n values and different initializations.
6. Final polish with subgradient or coordinate descent.

Let me also think about what happens at a fixed point of LP2. At a fixed point, g* = f, so the LP is minimizing max(f*g) subject to sum(g) = S, g >= 0, and the solution is g = f itself (since f is already optimal). This means max(f*f) is at a local minimum of the "LP relaxation."

The quality of this local minimum depends on the initial point. So multiple restarts are important.

"""
LP-based alternating minimization for minimizing the peak autocorrelation
of non-negative sequences. Uses linear programming to find optimal update
directions, combined with spectral analysis and multi-scale exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def normalized_score(a):
    """Score with sum normalized to sqrt(2n)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return np.max(c)  # When sum = sqrt(2n), score = max_c


def lp_min_cross_peak(f, S_target):
    """
    LP: minimize t subject to (f*g)_k <= t for all k, sum(g) = S_target, g >= 0.
    Returns (g_optimal, t_optimal).
    """
    n = len(f)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Build constraint matrix
    A_ub = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S_target])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[-1]
    else:
        # Fallback
        g = np.full(n, S_target / n)
        t = np.max(fft_convolve(f, g))
        return g, t


def lp_max_sum(f, M):
    """
    LP: maximize sum(g) subject to (f*g)_k <= M for all k, g >= 0.
    Returns g_optimal.
    """
    n = len(f)
    
    c_obj = -np.ones(n)
    
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return np.ones(n)


def exact_subgradient(a):
    """
    Compute exact subgradient of max(a*a) at the argmax.
    Returns (peak_value, subgradient, peak_index).
    """
    n = len(a)
    c = fft_convolve(a, a)
    k_star = int(np.argmax(c))
    peak_val = c[k_star]
    
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= k_star - j < n:
            val += a[k_star - j]
        if 0 <= j - k_star < n:
            val += a[j - k_star]
        g[j] = 2.0 * val
    
    return peak_val, g, k_star


def subgradient_descent(a_init, time_limit, rng):
    """
    Subgradient descent with adaptive step size.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S_target = np.sqrt(2.0 * n)
    a = a * S_target / np.sum(a)
    
    best_a = a.copy()
    best_peak = normalized_score(a)
    
    t0 = time.monotonic()
    step = 0
    lr = S_target / np.sqrt(n)  # Initial step size
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        peak_val, g, k_star = exact_subgradient(a)
        
        # Projected subgradient step
        a_new = a - lr * g / np.linalg.norm(g)
        a_new = np.clip(a_new, 0.0, None)
        a_new = a_new * S_target / max(np.sum(a_new), 1e-10)
        
        new_peak = normalized_score(a_new)
        
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
        
        if new_peak < peak_val:
            a = a_new
        else:
            # Reduce step size
            lr *= 0.9
        
        # Decay step size
        lr = max(lr * (1 - 1e-5 * step), 0.001 * S_target / np.sqrt(n))
        
        if step % 500 == 0:
            print(f"    SG step={step}, peak={peak_val:.4f}, best={best_peak:.4f}, lr={lr:.4f}")
    
    return best_a, best_peak


def lp_alternating(f_init, time_limit, use_lp2=True, verbose=True):
    """
    LP-based alternating minimization.
    """
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S_target = np.sqrt(2.0 * n)
    f = f * S_target / np.sum(f)
    
    best_f = f.copy()
    best_peak = normalized_score(f)
    
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        if use_lp2:
            g, t_lp = lp_min_cross_peak(f, S_target)
        else:
            M = normalized_score(f)
            g = lp_max_sum(f, M)
            if np.sum(g) < 1e-10:
                g = np.full(n, S_target / n)
            g = g * S_target / np.sum(g)
        
        # Try direct replacement first
        g_peak = normalized_score(g)
        
        if g_peak < best_peak:
            best_peak = g_peak
            best_f = g.copy()
            f = g.copy()
            no_improve = 0
        else:
            # Try convex combinations
            improved = False
            for t_mix in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                f_trial = (1 - t_mix) * f + t_mix * g
                f_trial = np.clip(f_trial, 0.0, None)
                f_trial = f_trial * S_target / max(np.sum(f_trial), 1e-10)
                trial_peak = normalized_score(f_trial)
                
                if trial_peak < best_peak:
                    best_peak = trial_peak
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    improved = True
                    no_improve = 0
                    break
                elif trial_peak < normalized_score(f):
                    f = f_trial.copy()
                    improved = True
            
            if not improved:
                no_improve += 1
                # Reset to best if stuck for too long
                if no_improve > 5:
                    f = best_f.copy()
                    no_improve = 0
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            cur_peak = normalized_score(f)
            print(f"    LP iter={iter_count}, cur_peak={cur_peak:.6f}, best={best_peak:.6f}, t={elapsed:.0f}s")
    
    return best_f, best_peak


def make_initialization(n, rng, method='random'):
    """Create initial sequences with different methods."""
    if method == 'random':
        a = 1.0 + 0.5 * rng.standard_normal(n)
    elif method == 'broadband':
        t = np.arange(n, dtype=np.float64)
        a = np.ones(n)
        for k in range(1, 12):
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.2 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    elif method == 'random_sparse':
        a = np.zeros(n)
        n_nonzero = int(n * rng.uniform(0.5, 0.8))

        a[idx] = rng.uniform(0.5, 2.0, n_nonzero)
        a[idx] += 0.3 * rng.standard_normal(n_nonzero)
    else:
        a = np.ones(n) + 0.3 * rng.standard_normal(n)
    
    return np.clip(a, 0.01, 100.0)

Hmm. Let me try to think about what minimizes the peak of the convolution.

For the convolution b[k] = sum_j a[j]*a[k-j], the peak is typically at k = n-1 (full overlap). At k = n-1:
b[n-1] = sum_{j=0}^{n-1} a[j]*a[n-1-j]

This is the "palindromic dot product" of a with itself. To minimize this, we want a[j] and a[n-1-j] to be "oppositely correlated" - when one is large, the other should be small.

But we also want sum(a) to be large (to keep the denominator large). And we want b[k] to be small for all k, not just k=n-1.

A good strategy might be: make the sequence "anti-palindromic" on average, while keeping the overall sum high.
