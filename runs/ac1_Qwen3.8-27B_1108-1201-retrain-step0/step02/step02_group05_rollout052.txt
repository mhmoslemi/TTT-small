Uses multi-phase approach: frequency-inspired initialization, L-BFGS with

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)

    return e / s


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def pnorm_obj_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p.
    Returns (score, grad_theta).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_clipped = np.maximum(c, 1e-30)
    sum_cp = np.sum(c_clipped ** p)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    # Gradient of pnorm w.r.t. c
    grad_c = c_clipped ** (p - 1) / (pnorm ** (p - 1))

    # Gradient of pnorm w.r.t. a via conv(a,a)
    # d(pnorm)/da[i] = 2 * conv(a_rev, grad_c)[n-1+i]
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, grad_c)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    # Chain rule: theta -> softmax -> a
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def softk_obj_grad(theta, k):
    """
    Objective: 2n * sum of top-k values of conv(softmax(theta), softmax(theta)).
    A smoothed approximation to the max that's differentiable.
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    # Use soft top-k via sorting and thresholding
    c_sorted = np.sort(c)[::-1]
    threshold = c_sorted[min(k, len(c) - 1)]
    weight = np.exp(50.0 * (c - threshold))
    weight = weight / np.sum(weight)

    score = 2.0 * n * np.dot(weight, c)

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, weight)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def spectral_flatten(a, n_iter=50, mu=0.5):
    """
    Spectral flattening: iterate between time and frequency domain
    to flatten the magnitude spectrum of the zero-padded sequence.
    """
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for _ in range(n_iter):
        F = np.fft.fft(a, N)
        mag = np.abs(F)
        phase = np.angle(F)
        avg_mag = np.mean(mag)
        # Blend magnitude toward flat
        new_mag = (1 - mu) * mag + mu * avg_mag
        F_new = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(F_new))[:n]
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / np.sum(a_new)
        a = a_new

    return a


def make_flat_spectrum_init(n, rng, n_harmonics=20):
    """
    Initialize from a sequence with near-flat magnitude spectrum.
    Start with random phases, compute time-domain signal, take absolute value.
    """
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    # Random phases
    phases = rng.uniform(0, 2 * np.pi, N)
    # Slightly non-uniform magnitude to avoid exact flat (which might give negative values)
    mags = 1.0 + 0.1 * rng.standard_normal(N)
    mags = np.maximum(mags, 0.5)

    F = mags * np.exp(1j * phases)
    a = np.real(np.fft.ifft(F))[:n]
    a = np.abs(a) + 0.01
    a = a / np.sum(a)
    return a


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization with non-negativity."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization with controlled spread."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_multi_sine_init(n, rng, n_sines=5):
    """Multi-sinusoidal initialization."""
    a = np.ones(n)
    t = np.arange(n, dtype=np.float64)
    for _ in range(n_sines):
        freq = rng.integers(1, max(2, n // 3)) + 1
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.sin(2 * np.pi * freq * t / n + phase)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p, deadline):
    """Run L-BFGS with annealed p-norm schedule."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))

    for p in p_schedule:
        if time.monotonic() > deadline - 10:
            break

        result = minimize(
            pnorm_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(p,),
            options={'maxiter': maxiter_per_p, 'ftol': 1e-16,
                     'gtol': 1e-14, 'maxfun': maxiter_per_p * 5}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)

        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        theta = best_theta

    return best_theta, best_score


def run_lbgfs_softk(theta, k_schedule, maxiter_per_k, deadline):
    """Run L-BFGS with annealed soft-top-k objective."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))

    for k in k_schedule:
        if time.monotonic() > deadline - 10:
            break

        result = minimize(
            softk_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(k,),
            options={'maxiter': maxiter_per_k, 'ftol': 1e-16,
                     'gtol': 1e-14, 'maxfun': maxiter_per_k * 5}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)

        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        theta = best_theta

    return best_theta, best_score


def peak_targeted_search(a_init, time_limit, rng):
    """
    Local search that targets the peak of the convolution.
    Uses gradient information to identify which elements to adjust.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.001
    cooling = 0.9995
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))

        # Compute contribution of each pair to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = current[i] * current[j]
            j2 = i - peak_idx
            if 0 <= j2 < n:
                contribs[i] += current[i] * current[j2]

        # Also compute "second peak" pressure
        c_sorted_idx = np.argsort(c)[::-1]
        top_lags = c_sorted_idx[:max(3, n // 32)]

        # Pressure from all top lags
        pressure = np.zeros(n)
        for lag in top_lags:
            w = c[lag] / max(c[peak_idx], 1e-10)
            for i in range(n):
                j = lag - i
                if 0 <= j < n:
                    pressure[i] += w * current[i] * current[j]
                j2 = i - lag
                if 0 <= j2 < n:
                    pressure[i] += w * current[i] * current[j2]

        r = rng.random()

        if r < 0.40:
            # Targeted peak reduction: reduce high-pressure elements, increase low-pressure
            top_k = max(5, n // 8)
            high_idx = np.argsort(pressure)[-top_k:]
            low_idx = np.argsort(pressure)[:top_k]

            candidate = current.copy()
            candidate[high_idx] *= np.exp(rng.normal(-0.02, 0.01, size=len(high_idx)))
            candidate[low_idx] *= np.exp(rng.normal(0.02, 0.01, size=len(low_idx)))

        elif r < 0.55:
            # Gradient-guided step: move in negative gradient direction of soft-max
            c_max = np.max(c)
            tau = 0.005 * c_max
            shifted = (c - c_max) / tau
            exp_shifted = np.exp(np.minimum(shifted, 0))
            w = exp_shifted / np.sum(exp_shifted)

            a_rev = current[::-1]
            cross = fft_convolve(a_rev, w)
            grad_a = 2.0 * cross[n - 1:2 * n - 1]
            grad_a = grad_a / (np.max(np.abs(grad_a)) + 1e-10)

            step_size = 0.005 * rng.random() + 0.001
            candidate = current - step_size * grad_a
            candidate = np.maximum(candidate, 1e-10)

        elif r < 0.70:
            # Symmetric perturbation: perturb a and a_rev together
            pert = rng.normal(0, 0.02, n)
            pert_sym = 0.5 * (pert + pert[::-1])
            candidate = current * np.exp(pert_sym)

        elif r < 0.80:
            # Block modification
            width = max(1, int(rng.integers(2, max(3, n // 10))))
            start = int(rng.integers(0, n - width))
            candidate = current.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, size=width))

        elif r < 0.90:
            # Swap-based: exchange content between two blocks
            w1 = max(2, n // 16)
            w2 = max(2, n // 16)
            s1 = int(rng.integers(0, n - w1))
            s2 = int(rng.integers(0, n - w2))
            if s1 != s2 and abs(s1 - s2) > max(w1, w2):
                candidate = current.copy()
                candidate[s1:s1 + w1] = current[s2:s2 + w2] * rng.uniform(0.8, 1.2)
                candidate[s2:s2 + w2] = current[s1:s1 + w1] * rng.uniform(0.8, 1.2)
            else:
                candidate = current * np.exp(rng.normal(0, 0.01, n))

        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0, 0.01, n))

        candidate = np.maximum(candidate, 1e-10)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 1e-6)

        if time.monotonic() - last_print > 10.0:
            last_print = time.monotonic()
            print(f"    peak_search: score={current_score:.6f}, best={best_score:.6f}, "
                  f"t={time.monotonic()-t0:.0f}s, T={temperature:.6f}")

    return best, best_score


def coordinate_descent(a_init, time_limit, rng):
    """
    Coordinate descent: iteratively adjust individual or small groups of
    coordinates to minimize the score.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))

        # Try adjusting the top contributors to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = current[i] * current[j]

        # Try several random subsets
        for _ in range(50):
            k = int(rng.integers(1, 6))
            if rng.random() < 0.5:
                # Reduce top contributors
                idx = np.argsort(contribs)[-k:]
            else:
                idx = rng.choice(n, size=k, replace=False)

            candidate = current.copy()
            scale = rng.uniform(0.9, 1.1, size=k)
            candidate[idx] *= scale
            candidate = np.maximum(candidate, 1e-10)
            candidate = candidate / np.sum(candidate)

            new_score = local_score(candidate)
            if new_score < current_score:
                current = candidate
                current_score = new_score
                if current_score < best_score:
                    best = current.copy()
                    best_score = current_score

        if time.monotonic() - last_print > 5.0:
            last_print = time.monotonic()
            print(f"    coord_descent: best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for optimal non-negative sequence.
    Phase 1: L-BFGS with p-norm and soft-top-k, multiple inits and lengths.
    Phase 2: Spectral flattening refinement.
    Phase 3: Peak-targeted local search.
    Phase 4: Coordinate descent polish.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        best_score = s
        best_seq = start_seq.copy()

    # Phase 1: L-BFGS optimization
    print("\n=== Phase 1: L-BFGS optimization ===")

    lengths = [128, 192, 256, 320, 384]

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break

        time_remaining = deadline - time.monotonic()
        # Allocate time proportionally, favoring n=256
        if n == 256:
            time_alloc = min(500.0, time_remaining * 0.35)
        else:
            time_alloc = max(40.0, time_remaining * 0.15)

        n_deadline = time.monotonic() + time_alloc

        # Generate initializations
        inits = []
        if start_seq is not None:
            x_old = np.linspace(0, 1, len(start_seq))
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, start_seq)
            a_interp = np.maximum(a_interp, 1e-6)
            a_interp = a_interp / np.sum(a_interp)
            inits.append(('interp', a_interp))

        if best_seq is not None and abs(len(best_seq) - n) / n < 0.3:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n)
            a_best = np.interp(x_new, x_old, best_seq)
            a_pert = a_best * np.exp(rng.normal(0, 0.08, n))
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert = a_pert / np.sum(a_pert)
            inits.append(('best_pert', a_pert))

        inits.append(('chirp1', make_chirp_init(n, 0.2 + 0.4 * rng.random(), rng)))
        inits.append(('chirp2', make_chirp_init(n, 0.3 + 0.4 * rng.random(), rng)))
        inits.append(('random', make_random_init(n, rng, spread=0.2 + 0.4 * rng.random())))
        inits.append(('sine', make_multi_sine_init(n, rng, n_sines=3 + int(rng.integers(0, 5))))
        inits.append(('flat_spec', make_flat_spectrum_init(n, rng)))

        p_schedule = [8, 16, 32, 64, 128, 256]
        k_schedule = [n // 4, n // 8, n // 16, n // 32, 3, 2, 1]

        for init_name, a0 in inits:
            if time.monotonic() > n_deadline - 20:
                break

            theta = np.log(a0 + 1e-30)

            # Try p-norm
            remaining = n_deadline - time.monotonic()
            if remaining > 30:
                maxiter = min(300, int(remaining * 5))
                theta_best, s = run_lbgfs_pnorm(theta, p_schedule,
                                                 maxiter_per_p=max(50, maxiter // len(p_schedule)),
                                                 deadline=n_deadline - 5)
                if s < best_score:
                    best_score = s
                    best_seq = softmax(theta_best).copy()
                    print(f"  n={n}, init='{init_name}', pnorm: score={s:.6f} *** NEW BEST ***")
                else:
                    print(f"  n={n}, init='{init_name}', pnorm: score={s:.6f}")
                    theta = theta_best  # Use improved theta for softk

            # Try soft-top-k from the best theta so far
            if time.monotonic() < n_deadline - 10:
                theta_best2, s2 = run_lbgfs_softk(theta, k_schedule,
                                                    maxiter_per_k=200,
                                                    deadline=n_deadline)
                if s2 < best_score:
                    best_score = s2
                    best_seq = softmax(theta_best2).copy()
                    print(f"  n={n}, init='{init_name}', softk: score={s2:.6f} *** NEW BEST ***")

        print(f"  Best after n={n}: {best_score:.6f}, seq_len={len(best_seq) if best_seq is not None else 0}")

    # Phase 2: Spectral flattening on best
    print(f"\n=== Phase 2: Spectral flattening (score={best_score:.6f}) ===")
    if best_seq is not None:
        n = len(best_seq)
        a_spec = spectral_flatten(best_seq, n_iter=100, mu=0.3)
        s_spec = local_score(a_spec)
        if s_spec < best_score:
            best_score = s_spec
            best_seq = a_spec.copy()
            print(f"  Spectral improved: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Spectral: {s_spec:.6f} (no improvement)")

        # Blend spectral with best
        for alpha in [0.3, 0.5, 0.7]:
            a_blend = (1 - alpha) * best_seq + alpha * spectral_flatten(best_seq, 200, 0.5)
            a_blend = np.maximum(a_blend, 1e-10)
            a_blend = a_blend / np.sum(a_blend)
            s_blend = local_score(a_blend)
            if s_blend < best_score:
                best_score = s_blend
                best_seq = a_blend.copy()
                print(f"  Blend alpha={alpha}: {best_score:.6f} *** NEW BEST ***")

    # Phase 3: Peak-targeted local search
    print(f"\n=== Phase 3: Peak-targeted local search (score={best_score:.6f}) ===")
    if best_seq is not None:
        polish_time = min(400.0, max(60.0, deadline - time.monotonic() - 120))
        print(f"  Time for peak search: {polish_time:.0f}s")
        polished, p_score = peak_targeted_search(best_seq, polish_time, rng)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After peak search: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  After peak search: {p_score:.6f} (no improvement)")

    # Phase 4: Coordinate descent polish
    print(f"\n=== Phase 4: Coordinate descent (score={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        cd_time = min(200.0, max(30.0, deadline - time.monotonic() - 30))
        print(f"  Time for coord descent: {cd_time:.0f}s")
        cd_result, cd_score = coordinate_descent(best_seq, cd_time, rng)
        if cd_score < best_score:
            best_score = cd_score
            best_seq = cd_result.copy()
            print(f"  After coord descent: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  After coord descent: {cd_score:.6f} (no improvement)")

    # Final: One more round of L-BFGS on the best
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Final L-BFGS polish (score={best_score:.6f}) ===")
        n = len(best_seq)
        theta = np.log(best_seq + 1e-30)
        theta_final, s_final = run_lbgfs_pnorm(theta, [64, 128, 256],
                                                maxiter_per_p=200,
                                                deadline=deadline - 5)
        if s_final < best_score:
            best_score = s_final
            best_seq = softmax(theta_final).copy()
            print(f"  Final L-BFGS: {best_score:.6f} *** NEW BEST ***")

    # Verify with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        # Update best if evaluation differs (shouldn't, but safety)
        if final_score < best_score:
            best_score = final_score
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
