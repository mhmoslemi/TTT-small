    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_step(a, max_iters=10, line_search_steps=20):
    """Do one LP-based step: find direction, then line search."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    
    g = solve_lp_direction(a, M)
    if g is None or g.sum() <= 0:
        return a.copy(), 2*n*M
    
    # Line search: a_new = (1-t)*a + t*g, find best t
    best_t = 0.0
    best_obj = 2*n*M
    for i in range(1, line_search_steps+1):
        t = i / line_search_steps
        a_new = (1-t)*a + t*g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        conv_new = fast_convolve(a_new)
        obj = 2*n*conv_new.max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
        if t >= 1.0:
            break
    
    if best_t == 0:
        return a.copy(), best_obj
    a_new = (1-best_t)*a + best_t*g
    a_new = np.maximum(a_new, 0)
    a_new /= a_new.sum()
    return a_new, best_obj

def gradient_descent(a0, time_limit, t_smooth=10.0, lr_init=0.05):
    """Smoothed gradient descent on the simplex."""
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    lr = lr_init
    patience = 0
    max_pat = 30
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        obj, grad, conv = smooth_objective_grad_fast(a, t_smooth)
        
        # Gradient descent step
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        actual_obj = 2.0 * n * conv_new.max()
        
        if actual_obj < best_obj - 1e-9:
            best_obj = actual_obj
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        conv = conv_new
        
        if patience >= max_pat:
            patience = 0
            lr *= 0.5
            t_smooth *= 1.5
            if lr < 0.001:
                lr = lr_init * 0.3
                t_smooth = 5.0
    
    return best_a, best_obj

def lsd_optimize(a0, time_limit):
    """Larger-scale descent: alternate LP steps with gradient steps."""
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # LP step
        a_new, lp_obj = lp_step(a, line_search_steps=10)
        
        if lp_obj < best_obj - 1e-9:
            best_obj = lp_obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Try gradient descent for a few steps
            a = a_new if a_new.sum() > 1e-10 else a
            a = np.maximum(a, 0)
            a /= a.sum()
            
            # Short gradient descent burst
            for _ in range(5):
                if time.monotonic() - t0 >= time_limit:
                    break
                obj, grad, _ = smooth_objective_grad_fast(a, t=5.0)
                a_new2 = a - 0.03 * grad
                a_new2 = project_simplex(a_new2)
                conv2 = fast_convolve(a_new2)
                obj2 = 2.0 * n * conv2.max()
                if obj2 < best_obj:
                    best_obj = obj2
                    best_a = a_new2.copy()
                a = a_new2
        
        iter_count += 1
        if iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LSD iter %d: obj=%.6f, elapsed=%.1fs" % (iter_count, best_obj, elapsed))
    
    return best_a, best_obj

def peak_targeted_search(a0, time_limit, rng):
    """Target peak positions directly with local moves."""
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        frac = (time.monotonic() - t0) / max(time_limit, 1.0)
        T = 0.02 * (1.0 - frac)**2 + 0.0001
        
        # Find peak positions
        max_c = conv.max()
        peak_thresh = max_c * 0.9
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        
        # For each peak, find the most "expensive" pair
        k = int(rng.choice(peak_positions))
        j_min = max(0, k - n + 1)
        j_max = min(n-1, k)
        
        # Contribution of each j to conv[k]
        contributions = np.array([a[j] * a[k-j] for j in range(j_min, j_max+1)])
        total = contributions.sum()
        if total < 1e-15:
            continue
        
        # Pick a j with high contribution to move mass away from
        prob = contributions / total
        top_j = j_min + int(rng.choice(j_max - j_min + 1, p=prob))
        
        # Find a target position that's not contributing to peaks
        target_j = None
        for _ in range(10):
            candidate = int(rng.integers(n))
            if candidate == top_j:
                continue
            # Check if candidate contributes to any peak
            contrib_score = 0.0
            for pk in peak_positions[:10]:
                if 0 <= pk - candidate <= n-1:
                    contrib_score += a[pk - candidate]
                if 0 <= candidate - pk <= n-1:  # This would be for the other direction
                    pass
            if contrib_score < np.mean(a) * 3:
                target_j = candidate
                break
        if target_j is None:
            target_j = int(rng.integers(n))
            if target_j == top_j:
                continue
        
        # Move mass
        delta = a[top_j] * rng.uniform(0.02, 0.15)
        a[top_j] -= delta
        a[target_j] += delta
        
        if a[top_j] < 0:
            a[top_j] = 0
            a[target_j] -= (-a[top_j] + delta)
            continue
        
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            a[top_j] += delta
            a[target_j] -= delta
    
    return best_a, best_obj

"""
Step function optimizer combining LP-based direction finding, smoothed
gradient descent with convolution flattening, and peak-targeted local search.
Targets score below 1.5027 by minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build constraint matrix A where A[k,m] = f[k-m] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        k_min = m
        k_max = min(2 * n - 2, m + n - 1)
        A[k_min:k_max + 1, m] = f[0:k_max - k_min + 1]
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    n = len(f)
    A = build_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_step(a, line_search_steps=15):
    """One LP-based step: find direction, line search."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    if M < 1e-15:
        return a.copy(), 2.0 * n * M
    g = solve_lp_direction(a, M)
    if g is None or g.sum() <= 1e-15:
        return a.copy(), 2.0 * n * M
    best_t = 0.0
    best_obj = 2.0 * n * M
    best_a = a.copy()
    for i in range(1, line_search_steps + 1):
        t = i / line_search_steps
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-12:
            break
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj = 2.0 * n * conv_try.max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
            best_a = a_try.copy()
    return best_a, best_obj


def smooth_grad(a, t=10.0):
    """Smoothed max objective and gradient via log-sum-exp."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = conv / t - max_c / t
    log_sum_exp = np.log(np.exp(shifted).sum()) + max_c / t
    obj = 2.0 * n * log_sum_exp
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= wsum
    cross = np.correlate(weights, a, mode='valid')
    grad = 2.0 * cross[:n]
    obj_grad = 2.0 * n * grad
    return obj, obj_grad, conv


def flatten_grad(a):
    """Grad of sum((conv - mean)^2) for convolution flattening."""
    n = len(a)
    conv = fast_convolve(a)
    mu = 1.0 / (2 * n - 1)  # mean when sum(a)=1
    residual = conv - mu
    # d/d(a_i) sum_k residual_k^2 = 2 * sum_k residual_k * d(conv_k)/d(a_i)
    # = 2 * sum_k residual_k * 2*a[k-i] = 4 * (residual * a)_i
    cross = np.correlate(residual, a, mode='valid')
    grad = 4.0 * cross[:n]
    return grad, conv


def gradient_descent(a0, time_limit, t_smooth=8.0, lr_init=0.04, use_flatten=True):
    """Hybrid smoothed gradient + flatten gradient descent."""
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    lr = lr_init
    patience = 0
    max_pat = 25
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        obj_s, grad_s, conv = smooth_grad(a, t_smooth)
        total_grad = grad_s
        if use_flatten:
            grad_f, _ = flatten_grad(a)
            total_grad = grad_s + 0.3 * grad_f
        a_new = a - lr * total_grad
        a_new = project_simplex(a_new)
        conv_new = fast_convolve(a_new)
        actual_obj = 2.0 * n * conv_new.max()
        if actual_obj < best_obj - 1e-10:
            best_obj = actual_obj
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        a = a_new
        if patience >= max_pat:
            patience = 0
            lr *= 0.5
            t_smooth *= 1.4
            if lr < 0.0008:
                lr = lr_init * 0.4
                t_smooth = 5.0
    return best_a, best_obj


def lsd_optimize(a0, time_limit):
    """Larger-scale descent: alternate LP steps with gradient bursts."""
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        a_new, lp_obj = lp_step(a)
        if lp_obj < best_obj - 1e-10:
            best_obj = lp_obj
            best_a = a_new.copy()
            a = a_new
        else:
            a = np.maximum(a_new, 0)
            ss = a.sum()
            if ss < 1e-12:
                a = best_a.copy()
            else:
                a = a / ss
            for _ in range(8):
                if time.monotonic() - t0 >= time_limit:
                    break
                _, grad, _ = smooth_grad(a, t=5.0)
                _, grad_f, _ = flatten_grad(a)
                total_grad = grad + 0.5 * grad_f
                a_try = a - 0.02 * total_grad
                a_try = project_simplex(a_try)
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                a = a_try
        iter_count += 1
        if iter_count % 40 == 0:
            elapsed = time.monotonic() - t0
            print("    LSD iter %d: best=%.6f, elapsed=%.1fs" % (iter_count, best_obj, elapsed))
    return best_a, best_obj


def peak_targeted_search(a0, time_limit, rng):
    """Aggressive peak-targeted local search."""
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.025 * (1.0 - frac) ** 2 + 0.0001
        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.85
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        k = int(rng.choice(peak_positions))
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        contributions = np.array([a[j] * a[k - j] for j in range(j_min, j_max + 1)])
        total = contributions.sum()
        if total < 1e-15:
            continue
        prob = contributions / total
        top_j = j_min + int(rng.choice(j_max - j_min + 1, p=prob))
        target_j = None
        for _ in range(15):
            candidate = int(rng.integers(n))
            if candidate == top_j:
                continue
            contrib_score = 0.0
            for pk in peak_positions[:12]:
                if 0 <= pk - candidate <= n - 1:
                    contrib_score += a[pk - candidate]
            if contrib_score < np.mean(a) * 2.5:
                target_j = candidate
                break
        if target_j is None:
            target_j = int(rng.integers(n))
            if target_j == top_j:
                continue
        delta = a[top_j] * rng.uniform(0.03, 0.2)
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < -1e-12:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            a[top_j] += delta
            a[target_j] -= delta
    return best_a, best_obj

            if t_alloc < 5:
                break
            a_opt, _ = gradient_descent(a_start, time_limit=t_alloc,
                                        t_smooth=t_s, lr_init=lr_s)
            record(a_opt)
            a_start = best_sequence if best_sequence is not None else a_start
            a_start /= a_start.sum()

    print("  Phase 3 done in %.1fs. Best: %.6f" % (time.monotonic() - t3, best_score))

    # Phase 4: Peak-targeted local search
    print("Phase 4: Peak-targeted local search...")
    t4 = time.monotonic()
    if best_sequence is not None and time.monotonic() < deadline - 50:
        ps_budget = min(400.0, deadline - time.monotonic() - 50)
        n_ps = 4
        ps_per = ps_budget / n_ps
        for i in range(n_ps):
            if time.monotonic() > deadline - 50:
                break
            rng_ps = np.random.default_rng(seed + 10000 + i * 333)
            a_opt, _ = peak_targeted_search(best_sequence.copy(),
                                            time_limit=ps_per, rng=rng_ps)
            record(a_opt)

    print("  Phase 4 done in %.1fs. Best: %.6f" % (time.monotonic() - t4, best_score))

    # Phase 5: Final alternating polish
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("Phase 5: Final polish...")
        a_best = best_sequence.copy()
        a_best = np.maximum(a_best, 0)
        a_best /= a_best.sum()
        remaining = deadline - time.monotonic() - 10
        if remaining > 40:
            half = remaining * 0.5
            a_opt, _ = lsd_optimize(a_best, time_limit=half)
            record(a_opt)
            if time.monotonic() < deadline - 10:
                rem2 = deadline - time.monotonic() - 10
                if rem2 > 15:
                    rng_f = np.random.default_rng(seed + 55555)
                    a_opt2, _ = peak_targeted_search(a_best, time_limit=rem2, rng=rng_f)
                    record(a_opt2)

    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = safe_evaluate(best_sequence)

    print("FINAL: score=%.6f, n=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
