LP-guided optimization for minimizing the peak of self-convolution.
Uses CVXPY LP oracle to find optimal descent directions, combined with
gradient-based polishing and multi-start strategy.
"""

import numpy as np
import time
import cvxpy as cp

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

    """Normalize a to have the target sum, maintaining non-negativity."""
    S = np.sum(a)
    if S < 1e-12:
        return np.full(len(a), target_sum / len(a))
    return a * (target_sum / S)

def solve_lp_direction(a, M):
    """
    Solve LP: maximize sum(b) subject to conv(a,b) <= M, b >= 0.
    Returns the optimal b.
    """
    n = len(a)
    b = cp.Variable(n, nonneg=True)

    # conv(a,b)[k] = sum_j a[j] * b[k-j] (with valid indices)
    # This is: T @ b <= M * ones(2n-1)
    # where T is the (2n-1) x n Toeplitz matrix

    # We can build this using a circulant-like structure

    # T[k, j] = a[k-j] if 0 <= k-j < n else 0
    
    # For efficiency, use the fact that T is Toeplitz

    # Last column: T[0,n-1]=0, ..., T[n-2,n-1]=0, T[n-1,n-1]=a[0], T[n,n-1]=a[1], ...
    
    # Let's just build it with a loop (it's only 511x256)

    for k in range(2*n-1):
        for j in range(n):

                T[k, j] = a[k-j]

Multi-strategy optimization for minimizing self-convolution peak.
Combines SOCP-guided direction finding, Nesterov accelerated projected
subgradient descent, and local peak-shaving polish.

import numpy as np
import time
import cvxpy as cp

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the evaluation score for a sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)
    j_idx = np.arange(n)
    diff = k_idx[:, None] - j_idx[None, :]
    valid = (diff >= 0) & (diff < n)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    T[valid] = a[diff[valid].astype(int)]
    return T
