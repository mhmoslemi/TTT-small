    LP-guided + L-BFGS optimization for minimum peak convolution of non-negative sequences.
    Uses LP feasible direction search as the primary method, with L-BFGS log-sum-exp
    refinement and targeted local search. Explores n=192 to n=512.
    """
    
    import numpy as np
    import time
    from scipy.optimize import minimize, linprog
    from scipy.fft import rfft, irfft, next_fast_len
    
    def fft_convolve(a, b):
        n, m = len(a), len(b)
        size = next_fast_len(n + m - 1)
        fa = rfft(a, size)
        fb = rfft(b, size)
        return irfft(fa * fb, size)[:n + m - 1]
    
    def compute_conv(a):
        return fft_convolve(a, a)
    
    def peak_score(a):
        n = len(a)
        c = compute_conv(a)
        return 2.0 * n * np.max(c) / (np.sum(a) ** 2)
    
    def logsumexp_obj_grad(theta, tau):
        """L-BFGS objective: 2n * soft_max(conv(softmax(theta), softmax(theta)), tau)"""
        n = len(theta)
        t = theta - np.max(theta)
        e = np.exp(t)
        a = e / np.sum(e)
        c = compute_conv(a)
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        sum_exp = np.sum(exp_shifted)
        soft_val = c_max + tau * np.log(sum_exp)
        w = exp_shifted / sum_exp
        score = 2.0 * n * soft_val
        # Gradient
        a_rev = a[::-1]
        cross = fft_convolve(a_rev, w)
        grad_a = 2.0 * cross[n-1:2*n-1]
        inner = np.dot(grad_a, a)
        grad_theta = a * (grad_a - inner)
        return score, grad_theta
    
    def lp_direction(f, M, n):
        """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M, g >= 0"""
        # Build constraint matrix A: A[k,j] = f[k-j] if 0<=k-j<n
        k_idx = np.arange(2*n-1)[:, None]

        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
        return None

    def lbgfs_refine(a, time_limit, deadline):

        n = len(a)

            theta_new = res.x
            a_new = softmax_safe(theta_new)

    def softmax_safe(theta):

        e = np.exp(t)
        return e / np.sum(e)

        n = len(a)

"""
LP-guided + L-BFGS optimization for minimum peak convolution of non-negative sequences.
Combines LP-based feasible direction search, L-BFGS log-sum-exp refinement,
gradient peak-peeling, and targeted local search. Explores multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft, next_fast_len


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = next_fast_len(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_conv(a):
    return fft_convolve(a, a)


def peak_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = compute_conv(a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax_safe(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def logsumexp_obj_grad(theta, tau):
    n = len(theta)
    a = softmax_safe(theta)
    c = compute_conv(a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score = 2.0 * n * soft_val
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def build_conv_matrix(f, n):
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.where(valid, f[np.clip(diff, 0, n - 1)], 0.0)
    return A


def lp_direction(f, M, n):
    A = build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(f, time_limit=120.0, max_iters=200):
    f = f.copy().astype(np.float64)
    n = len(f)
    s = np.sum(f)
    best_f = f.copy()
    best_peak = np.max(compute_conv(f))
    t0 = time.monotonic()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = compute_conv(f)
        M = np.max(c)

        if M < best_peak * (1 - 1e-8):
            best_peak = M
            best_f = f.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 20:
            break

        g = lp_direction(f, M, n)
        if g is None or np.sum(g) < 1e-10:
            break

        g = g / np.sum(g) * s

        improved = False
        for t in [0.3, 0.5, 0.1, 0.7, 0.2, 0.4, 0.6, 0.15, 0.25, 0.35, 0.45, 1.0]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            sn = np.sum(f_new)
            if sn < 1e-10:
                continue
            f_new = f_new / sn * s
            c_new = compute_conv(f_new)
            peak_new = np.max(c_new)
            if peak_new < M * (1 - 1e-7):
                f = f_new
                improved = True
                break

        if not improved:
            for t in [0.05, 0.02, 0.01]:
                f_new = (1 - t) * f + t * g
                f_new = np.maximum(f_new, 0.0)
                sn = np.sum(f_new)
                if sn < 1e-10:
                    continue
                f_new = f_new / sn * s
                c_new = compute_conv(f_new)
                peak_new = np.max(c_new)
                if peak_new < M * (1 - 1e-7):
                    f = f_new
                    improved = True
                    break

        if not improved:
            break

        if (it + 1) % 20 == 0:
            cur_score = peak_score(f)
            best_score = peak_score(best_f)
            print(f"    LP iter {it+1}: cur={cur_score:.6f}, best={best_score:.6f}")

    return best_f, peak_score(best_f)


def gradient_peak_peel(a, time_limit=60.0, max_iters=500):
    a = a.copy().astype(np.float64)
    n = len(a)
    s = np.sum(a)
    best_a = a.copy()
    best_score = peak_score(a)
    t0 = time.monotonic()
    alpha = 0.01

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = compute_conv(a)
        c_max = np.max(c)
        peak_lags = np.where(c >= c_max * 0.85)[0]

        grad = np.zeros(n)
        for lag in peak_lags[:8]:
            k = int(lag)
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                partner = k - j
                if 0 <= partner < n:
                    grad[j] += 2.0 * a[partner]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        grad = grad / grad_norm * c_max

        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / np.sum(a_new) * s

        new_score = peak_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        if new_score < peak_score(a):
            a = a_new
        else:
            alpha *= 0.95

    return best_a, best_score


def lbgfs_refine(a, time_limit=200.0):
    n = len(a)
    theta = np.log(a + 1e-30)
    best_a = a.copy()
    best_score = peak_score(a)
    t0 = time.monotonic()

    tau_schedule = [0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001, 0.00005]

    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        try:
            res = minimize(
                logsumexp_obj_grad, theta.copy(),
                method='L-BFGS-B', jac=True, args=(tau,),
                options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12}
            )
            theta_new = res.x
            a_new = softmax_safe(theta_new)
            a_new = np.maximum(a_new, 1e-12)
            a_new = a_new / np.sum(a_new)
            s = peak_score(a_new)
            if s < best_score:
                best_score = s
                best_a = a_new.copy()
                theta = theta_new.copy()
        except Exception as e:
            break

    return best_a, best_score


def local_polish(a, time_limit=120.0, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    a = a.copy().astype(np.float64)
    n = len(a)
    s = np.sum(a)
    best_a = a.copy()
    best_score = peak_score(a)
    t0 = time.monotonic()
    temp = 0.0005

    while time.monotonic() - t0 < time_limit:
        c = compute_conv(a)
        c_max = np.max(c)
        threshold = c_max * 0.92
        peak_lags = np.where(c >= threshold)[0]

        candidate = a.copy()
        for lag in peak_lags[:4]:
            k = int(lag)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1)
            partners = k - idx
            contribs = a[idx] * a[partners]
            top_k = max(2, len(idx) // 5)
            top_local = idx[np.argsort(contribs)[-top_k:]]
            candidate[top_local] *= np.exp(rng.normal(-0.04, 0.015, size=len(top_local)))

        r = rng.random()
        if r < 0.3:
            width = max(1, n // 16)
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, size=width))
        elif r < 0.4:
            sym = 0.5 * (a + a[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.01, n))

        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate) * s
        new_score = peak_score(candidate)

        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
        cur_score = peak_score(a)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-15)):
            a = candidate
        temp = max(temp * 0.9995, 1e-8)

    return best_a, best_score

        time_alloc = min(200.0, max(60.0, (deadline - time.monotonic() - 120.0) / max(1, len(lengths_to_try))))
        inits = make_diverse_inits(n, rng, ref_seq=best_seq)

        for init_i, a0 in enumerate(inits):
            if time.monotonic() > deadline - 150:
                break

            theta = np.log(a0 + 1e-30)
            t_alloc_init = min(60.0, time_alloc / max(1, len(inits) - init_i))

            try:
                res = minimize(
                    logsumexp_obj_grad, theta.copy(),
                    method='L-BFGS-B', jac=True, args=(0.01,),
                    options={'maxiter': 150, 'ftol': 1e-14, 'gtol': 1e-11}
                )
                a_new = softmax_safe(res.x)
                a_new = np.maximum(a_new, 1e-12)
                a_new = a_new / np.sum(a_new)

                res2 = minimize(
                    logsumexp_obj_grad, np.log(a_new + 1e-30),
                    method='L-BFGS-B', jac=True, args=(0.003,),
                    options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-12}
                )
                a_new = softmax_safe(res2.x)
                a_new = np.maximum(a_new, 1e-12)
                a_new = a_new / np.sum(a_new)

                sc = peak_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_new.copy()
                    best_n = n
                    print(f"  n={n}, init={init_i}: {sc:.6f} *** NEW BEST ***")
                else:
                    print(f"  n={n}, init={init_i}: {sc:.6f}")
            except Exception as e:
                print(f"  n={n}, init={init_i}: error {e}")

    print(f"\n=== Phase 5: LP on best found (score={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 180:
        t_lp2 = min(250.0, deadline - time.monotonic() - 120.0)
        lp_res, lp_score = lp_optimize(best_seq, time_limit=t_lp2, max_iters=200)
        if lp_score < best_score:
            best_score = lp_score
            best_seq = lp_res.copy()
            best_n = len(best_seq)
            print(f"  LP2 improved: {best_score:.6f} ***")
        else:
            print(f"  LP2 best: {lp_score:.6f}")

    print(f"\n=== Phase 6: L-BFGS final refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        t_lb2 = min(150.0, deadline - time.monotonic() - 60.0)
        lb_res, lb_score = lbgfs_refine(best_seq, time_limit=t_lb2)
        if lb_score < best_score:
            best_score = lb_score
            best_seq = lb_res.copy()
            best_n = len(best_seq)
            print(f"  L-BFGS2 improved: {best_score:.6f} ***")
        else:
            print(f"  L-BFGS2 best: {lb_score:.6f}")

    print(f"\n=== Phase 7: Local polish (score={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        t_polish = max(30.0, deadline - time.monotonic() - 15.0)
        pol_res, pol_score = local_polish(best_seq, time_limit=t_polish, rng=rng)
        if pol_score < best_score:
            best_score = pol_score
            best_seq = pol_res.copy()
            best_n = len(best_seq)
            print(f"  Polish improved: {best_score:.6f} ***")
        else:
            print(f"  Polish best: {pol_score:.6f}")

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
