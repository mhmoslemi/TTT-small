"""
Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses iterative LP-based approach inspired by the state-of-the-art method.
Theoretical minimum is 2.0 (achieved by uniform sequences), verified by Cauchy-Schwarz.

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def solve_lp(f, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns optimal g or None if infeasible.
    """
    n = len(f)
    num_constraints = 2 * n - 1
    
    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraint matrix A_ub: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_j f[k-j] * g[j] for valid j
    A_ub = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]
    
    b_ub = np.full(num_constraints, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return np.maximum(result.x, 0)
    return None

    best_score = current_score

    t_mix = 0.1  # Mixing parameter

            current_score = new_score

"""
Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses iterative LP-based approach (solving dual LP for each step) combined with
coordinate descent and structured random searches.
By Cauchy-Schwarz, the provable minimum is 2.0, achieved by uniform sequences.
This program verifies this and searches exhaustively for any non-uniform improvement.

import numpy as np

from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2 efficiently."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_constraint_matrix(n):
    """
    Build the constraint matrix A for the LP where:
    (f*g)[k] = sum_j A[k,j] * g[j] <= M
    A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    This is a Toeplitz matrix. We build it once per f.
    """
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k + 1, n)
        for j in range(j_start, j_end):
            A[k, j] = 1  # placeholder, actual value depends on f
    return A


def solve_lp_for_direction(f, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k=0,...,2n-2.
    Returns optimal g or None.
    """
    n = len(f)
    num_constraints = 2 * n - 1

    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)

    # Build constraint matrix: (f*g)[k] = sum_j f[k-j]*g[j] for valid j
    A_ub = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        j_lo = max(0, k - n + 1)
        j_hi = min(k + 1, n)
        for j in range(j_lo, j_hi):
            A_ub[k, j] = f[k - j]

    b_ub = np.full(num_constraints, M)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None

def optimize_gradient_descent(a_init, time_limit, rng):

        max_c = np.max(c)
        # Subgradient: for each argmax k, add current[k-i] to grad[i]
        max_indices = np.where(np.isclose(c, max_c, atol=1e-10))[0]

        for k in max_indices:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += current[j]

        grad /= len(max_indices)

        # But we also need to handle the normalization constraint
