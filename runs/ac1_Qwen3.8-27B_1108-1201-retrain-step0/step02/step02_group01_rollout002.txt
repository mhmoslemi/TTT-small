"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multi-start exploration with diverse initializations
2. Iterative LP-based direction finding (from SoTA) for principled refinement
3. Multiplicative (mirror descent) updates for hard-max regime
4. Soft-max Adam for smooth transitions
5. Coordinate descent polish

Key innovation: Heavy use of LP-based refinement that directly exploits
the constraint structure, combined with adaptive active-set sizing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def compute_grad_a(w, a):
    """Compute grad[m] = 2 * sum_j w[m+j] * a[j] for m=0..n-1 using FFT."""
    n = len(a)
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    # full_conv has length len(w) + n - 1
    # We need full_conv[n-1 : 2n-1]
    grad = np.zeros(n)
    end_idx = min(2 * n - 1, len(full_conv))
    grad[:] = 2.0 * full_conv[n - 1: end_idx][:n]
    return grad


def a_from_theta(theta, S):
    """Convert log-parameters to sequence: a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def softmax_obj_grad(theta, n, T, S):
    """Soft-max objective and gradient w.r.t. theta."""
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    if c_max < 1e-15:
        return c_max, np.zeros(n)
    shifted = T * (c - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T
    # Soft weights
    w = np.exp(shifted)
    w /= w.sum()
    grad_a = compute_grad_a(w, a)
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    return obj, grad_theta


def multiplicative_update(a, c, S, top_k, eta, min_ratio=0.3, max_ratio=3.0):
    """
    Multiplicative (mirror descent) update for hard top-k max.
    Reduces elements that contribute to the top-k lags.
    """
    n = len(a)
    # Find top-k lags
    top_lags = np.argsort(c)[::-1][:top_k]
    # Build weight vector
    w = np.zeros(2 * n - 1)
    w[top_lags] = 1.0 / len(top_lags)
    # Compute subgradient
    grad = compute_grad_a(w, a)
    # Multiplicative update in log space
    # log(a_new) = log(a) - eta * (grad - mean_weighted)
    # Use a scaled version for stability
    grad_max = grad.max()
    if grad_max < 1e-15:
        return a.copy()
    # Normalize gradient to [0, 1] range approximately
    grad_norm = grad / grad_max
    # Update
    a_new = a * np.exp(-eta * grad_norm)
    # Clip ratios
    ratio = a_new / np.maximum(a, 1e-30)
    ratio = np.clip(ratio, min_ratio, max_ratio)
    a_new = a * ratio
    # Normalize
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new * (S / s)
    else:
        a_new = np.full(n, S / n)
    return a_new


def solve_lp_direction(f, M, n, n_constraints=50, time_limit=15):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for top active lags, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    c = fft_convolve_self(f)
    active_lags = np.argsort(c)[::-1][:n_constraints]
    
    # Build constraint matrix A: A[i,j] = f[k_i - j] if 0 <= k_i-j < n
    A = np.zeros((n_constraints, n))
    for i, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[i, j] = f[k - j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(n_constraints, M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None and result.x.sum() > 1e-10:
            g = result.x
            return g / g.sum()
    except Exception:
        pass
    return None


def run_lbfgs_smooth(theta_init, n, T, S, time_budget, max_iter=200):
    """Run L-BFGS-B on soft-max objective."""
    state = {'start': time.monotonic(), 'budget': time_budget,
             'best_obj': np.inf, 'best_a': None}
    a0 = a_from_theta(theta_init, S)
    c0 = fft_convolve_self(a0)
    state['best_obj'] = c0.max()
    state['best_a'] = a0.copy()
    
    def lbfgs_cb(theta):
        elapsed = time.monotonic() - state['start']
        if elapsed > state['budget']:
            return state['best_obj'], np.zeros(len(theta))
        obj, grad = softmax_obj_grad(theta, n, T, S)
        a = a_from_theta(theta, S)
        c = fft_convolve_self(a)
        if c.max() < state['best_obj']:
            state['best_obj'] = c.max()
            state['best_a'] = a.copy()
        return obj, grad
    
    try:
        minimize(lbfgs_cb, theta_init, jac=True, method='L-BFGS-B',
                 options={'maxiter': max_iter, 'maxfun': 1000, 'gtol': 1e-8})
    except Exception:
        pass
    return state['best_a'], state['best_obj']


def run_adam_anneal(theta_init, n, S, time_budget, T_start=3.0, T_end=200.0, lr=0.03):
    """Adam with exponential temperature annealing."""
    theta = theta_init.copy().astype(np.float64)
    n_par = len(theta)
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    start = time.monotonic()
    
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * np.exp(np.log(max(T_end / T_start, 1.01)) * progress)
        
        a = a_from_theta(theta, S)
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()
        
        obj, grad = softmax_obj_grad(theta, n, T, S)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr_t = lr * (0.6 ** (progress * 2.0))
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
    
    return theta, best_a, best_obj


def coordinate_descent_polish(a, n, S, time_budget, max_sweeps=30):
    """Coordinate descent polish with binary search per coordinate."""
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            original = a[i]
            if original <= 1e-12:
                continue
            # Try reducing or slightly increasing a[i]
            best_local = original
            best_local_obj = best_obj
            # Binary search in [0, original*2]
            lo, hi = 0.0, original * 2.0
            for _ in range(10):
                mid = (lo + hi) / 2.0
                a[i] = mid
                s = a.sum()
                if s < 1e-10:
                    a[i] = original
                    break
                a_norm = a * (S / s)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_local_obj:
                    best_local_obj = obj_check
                    best_local = mid
                if obj_check < best_obj * 1.005:
                    lo = mid
                else:
                    hi = mid
            a[i] = best_local
            if abs(best_local - original) > 1e-8:
                improved = True
            s = a.sum()
            if s > 1e-10:
                a = a * (S / s)
            c = fft_convolve_self(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()
        if not improved:
            break
    return best_a, best_obj


def get_initialization(name, n, rng, S):
    """Generate various initializations."""
    if name == "uniform":
        a = np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        a = a * (S / a.sum())
    elif name == "gaussian":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n / 2) / (n / 6)) ** 2)
        a = a * (S / a.sum())
    elif name == "cosine":
        i = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * i / n)
        a = np.maximum(a, 0.01)
        a = a * (S / a.sum())
    elif name == "anti_peak":
        i = np.arange(n)
        a = 1.0 - 0.4 * np.cos(2 * np.pi * (n - 1) * i / n)
        a = np.maximum(a, 0.01)
        a = a * (S / a.sum())
    elif name == "two_humps":
        i = np.arange(n)
        a = (np.exp(-0.5 * ((i - n * 0.25) / (n / 10)) ** 2) +
             np.exp(-0.5 * ((i - n * 0.75) / (n / 10)) ** 2))
        a = a * (S / a.sum())
    elif name == "raised_cos":
        i = np.arange(n)
        t = i / (n - 1)
        a = np.sin(np.pi * t) ** 2
        a = a * (S / a.sum())
    else:
        a = np.full(n, S / n)
    return a
