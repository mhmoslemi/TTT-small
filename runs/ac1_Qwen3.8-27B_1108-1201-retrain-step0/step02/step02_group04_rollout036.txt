import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_and_grad(a, T=10.0):
    """Compute score, max conv, and gradient of smooth max."""
    n = len(a)
    c = fft_convolve(a, a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf, c, np.zeros(n)
    max_c = np.max(c)
    score = 2.0 * n * max_c / (sum_a ** 2)
    
    # Softmax for smooth max
    c_shifted = c - np.max(c)  # numerical stability
    p = np.exp(T * c_shifted) / np.exp(T * c_shifted).sum()
    
    # Gradient: g[i] = 2 * sum_m p[i+m] * a[m]
    g = np.zeros(n)
    for i in range(n):
        g[i] = 2.0 * np.dot(p[i:i+n], a)
    
    return score, c, g

def optimize_sequence(a_init, time_limit, T=10.0, lr=1e-4):
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0, 1000)
    n = len(a)
    a = a / np.sum(a)  # normalize to sum = 1
    
    current_score = 2.0 * n * np.max(fft_convolve(a, a))
    best = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    step = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        score, c, g = score_and_grad(a, T)
        
        # Gradient descent
        a_new = a - lr * g
        a_new = np.clip(a_new, 0, 1000)
        if np.sum(a_new) > 1e-10:
            a_new = a_new / np.sum(a_new)
        
        new_score = 2.0 * n * np.max(fft_convolve(a_new, a_new))
        
        if new_score < current_score:
            a = a_new
            current_score = new_score
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            no_improve += 1
            # Reduce learning rate
            lr *= 0.5
        
        step += 1
        if step % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    step={step}, score={current_score:.6f}, best={best_score:.6f}, time={elapsed:.1f}s, lr={lr:.2e}")
    
    return best, best_score
