Minimizes peak linear self-convolution of non-negative sequences.

import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    n = len(a)
    S = np.sum(a)

        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)

def binomial_init(n):
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2.0*k - 1.0) / (2.0*k)
    return a

def soft_max_conv(a, tau):
    """Returns (smoothed_max_value, gradient_wrt_a, conv)."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 0)

    w = exp_vals / np.sum(exp_vals)
