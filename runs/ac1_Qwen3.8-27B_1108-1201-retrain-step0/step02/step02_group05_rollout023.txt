    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1

    fb = np.fft.rfft(b, size)

    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)

    grad_a = 2.0 * cross[n-1:2*n-1]

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def soft_max_obj_grad(theta, tau):
    """Objective and gradient for 2n * soft_max(conv(softmax(theta)), tau)."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def topk_obj_grad(theta, k):
    """Objective: 2n * sum of top-k convolution values. Piecewise differentiable."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    top_idx = np.argsort(c)[-k:]
    score = 2.0 * n * np.sum(c[top_idx])

    weighted_c = np.zeros_like(c)
    weighted_c[top_idx] = 1.0

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, weighted_c)
    grad_a = 2.0 * n * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def lp_project(a, M):
    """Solve LP: maximize sum(g) s.t. conv(a,g) <= M for all lags, g >= 0."""
    n = len(a)
    # Build constraint matrix: A[k, j] = a[j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[j]

    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)

    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        g = res.x
        s = np.sum(g)
        if s > 1e-10:
            return g / s
    return a.copy()


def lp_iteration(a, n_iters=30, step_size=0.3):

    best = a.copy()

        M = np.max(c)

        g = lp_project(current, M)
        lp_time = time.monotonic() - t0

        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / np.sum(a_new)

        s = local_score(a_new)

            best_score = s
            best = a_new.copy()

            remaining_for_lp = deadline - time.monotonic() - 300
            iters = max(5, int(remaining_for_lp / 10))
            a_lp, s_lp = lp_iteration(best_seq, n_iters=iters, step_size=0.25)
            if s_lp < best_score:
                best_score = s_lp
                best_seq = a_lp.copy()
                print(f"  LP round {round_idx}: score={s_lp:.6f} *** NEW BEST ***")
            else:
                print(f"  LP round {round_idx}: score={s_lp:.6f}")

            # Polish with L-BFGS after LP
            if time.monotonic() > deadline - 250:
                break
            theta = np.log(best_seq + 1e-30)
            theta, s = run_lbgfs(theta, soft_max_obj_grad, 0.0005, 300, deadline - 200)
            a_new = softmax(theta)
            if s < best_score:
                best_score = s
                best_seq = a_new.copy()
                print(f"  Post-LP polish: score={s:.6f} *** NEW BEST ***")

    # === Phase 3: Top-k focusing ===
    print(f"\n=== Phase 3: Top-k L-BFGS focusing (score={best_score:.6f}) ===")
    if best_seq is not None:
        theta = np.log(best_seq + 1e-30)
        for k in [5, 3, 2, 1]:
            if time.monotonic() > deadline - 250:
                break
            theta, s = run_lbgfs(theta, topk_obj_grad, k, 400, deadline - 200)
            a_new = softmax(theta)
            if s < best_score:
                best_score = s
                best_seq = a_new.copy()
                print(f"  top-k={k}: score={s:.6f} *** NEW BEST ***")
            else:
                print(f"  top-k={k}: score={s:.6f}")

    # === Phase 4: P-norm with high p ===
    print(f"\n=== Phase 4: High p-norm (score={best_score:.6f}) ===")
    if best_seq is not None:
        theta = np.log(best_seq + 1e-30)
        for p in [64, 128, 256, 512]:
            if time.monotonic() > deadline - 200:
                break
            theta, s = run_lbgfs(theta, pnorm_obj_grad, p, 300, deadline - 150)
            a_new = softmax(theta)
            if s < best_score:
                best_score = s
                best_seq = a_new.copy()
                print(f"  p={p}: score={s:.6f} *** NEW BEST ***")
            else:
                print(f"  p={p}: score={s:.6f}")

    # === Phase 5: Multi-restart ===
    print(f"\n=== Phase 5: Multi-restart (score={best_score:.6f}) ===")
    if best_seq is not None:
        n = len(best_seq)
        n_restarts = 12
        time_per_restart = min(35.0, (deadline - time.monotonic() - 150) / max(1, n_restarts))

        for r in range(n_restarts):
            if time.monotonic() > deadline - 150:
                break

            scale = 0.03 + 0.07 * rng.random()
            a_pert = best_seq * np.exp(rng.normal(0, scale, n))
            # Also try asymmetric perturbation (break symmetry)
            if r % 3 == 0:
                a_pert = a_pert * np.exp(rng.normal(0, 0.02, n) + 0.01 * np.linspace(-1, 1, n))
            a_pert = np.maximum(a_pert, 1e-8)
            a_pert = a_pert / np.sum(a_pert)

            theta = np.log(a_pert + 1e-30)
            sub_dl = time.monotonic() + time_per_restart

            # Quick optimization: soft-max then top-k
            theta, s1 = run_lbgfs(theta, soft_max_obj_grad, 0.002, 200, sub_dl - 5)
            if time.monotonic() < sub_dl - 10:
                theta, s2 = run_lbgfs(theta, topk_obj_grad, 2, 200, sub_dl)
                s = s2
            else:
                s = s1

            a_new = softmax(theta)
            s = local_score(a_new)
            if s < best_score:
                best_score = s
                best_seq = a_new.copy()
                print(f"  Restart {r} (scale={scale:.3f}): score={s:.6f} *** NEW BEST ***")
            else:
                print(f"  Restart {r} (scale={scale:.3f}): score={s:.6f}")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length exploration (score={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        extra_lengths = sorted(set([n_best - 12, n_best - 8, n_best - 4, n_best + 4,
                                     n_best + 8, n_best + 12, n_best + 16]))
        extra_lengths = [l for l in extra_lengths if l != n_best and 128 <= l <= 512]

        for n in extra_lengths:
            if time.monotonic() > deadline - 80:
                break

            # Interpolate from best
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.maximum(a_interp, 1e-6)
            a_interp = a_interp / np.sum(a_interp)

            theta = np.log(a_interp + 1e-30)
            sub_dl = time.monotonic() + 30
            theta, s = run_lbgfs(theta, soft_max_obj_grad, 0.002, 300, sub_dl)
            if time.monotonic() < sub_dl - 5:
                theta, s = run_lbgfs(theta, topk_obj_grad, 2, 200, sub_dl)
            a_new = softmax(theta)
            s = local_score(a_new)
            if s < best_score:
                best_score = s
                best_seq = a_new.copy()
                print(f"  n={n}: score={s:.6f} *** NEW BEST ***")
            else:
                print(f"  n={n}: score={s:.6f}")

    # === Phase 7: Final SA polish ===
    print(f"\n=== Phase 7: Final SA polish (score={best_score:.6f}) ===")
    if best_seq is not None:
        polish_time = max(20.0, deadline - time.monotonic() - 10)
        print(f"  Polish time: {polish_time:.0f}s")
        polished, p_score = peak_targeted_sa(best_seq, polish_time, rng)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  After polish: {p_score:.6f} (no improvement)")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
