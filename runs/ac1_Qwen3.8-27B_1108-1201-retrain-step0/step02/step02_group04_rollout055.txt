```python
    """Minimize max(conv(a,a)) subject to a >= 0, sum(a) = 1."""
    a = a.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a /= s  # normalize to sum = 1
    
    t0 = time.monotonic()
    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break

        max_c = np.max(c)

        weights /= np.sum(weights)
```

```python
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def fft_cross_correlate(w, a):
    """Compute (w ⋆ a)[k] = sum_j w[j+k] * a[j] for k = 0, ..."""
    n = len(a)
    L1 = len(w)
    # Result length is L1 + n - 1
    size = 1
    while size < L1 + n - 1:
        size <<= 1
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fw * np.conj(fa), size)
    return result[:L1 + n - 1]

    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
```

```python
"""
Minimize 2n*max(conv(a,a))/sum(a)^2 for non-negative a.
Uses smoothed gradient descent with adaptive temperature and multiple initializations.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score_fast(a):
    """Fast score computation."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def simple_projection(b):
    """Project b onto {x >= 0, sum(x) = 1}."""
    b = b.copy()
    n = len(b)
    # Sort in decreasing order
    sorted_b = np.sort(b)[::-1]
    cumsum = np.cumsum(sorted_b)
    # Find largest m such that sorted_b[m-1] > (cumsum[m-1] - 1) / m
    rho = np.arange(1, n + 1)
    condition = sorted_b > (cumsum - 1) / rho
    m = np.max(rho[condition]) if np.any(condition) else 0
    if m == 0:
        return np.ones(n) / n
    theta = (cumsum[m-1] - 1) / m
    return np.maximum(b - theta, 0)

def optimize_gradient(a_init, time_budget, T_init=0.05, verbose=True):
    """
    Minimize max(conv(a,a)) subject to a >= 0, sum(a) = 1
    using smoothed gradient descent with momentum.
    """
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 1e-10, None)
    a = a / np.sum(a)  # normalize to sum = 1
    n = len(a)
    
    t0 = time.monotonic()
    lr = 0.1
    momentum = np.zeros_like(a)
    beta = 0.85
    T = T_init
    
    best = a.copy()
    best_score = compute_score_fast(a)
    
    if verbose:
        print(f"    Initial score: {best_score:.8f}")
    
    stuck_count = 0
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_budget:
            break
        
        c = fft_convolve(a, a)
        sum_a = np.sum(a)
        
        # Softmax weights (smoothed max approximation)
        c_max = np.max(c)
        exp_c = np.exp((c - c_max) / T)
        total_exp = np.sum(exp_c)
        weights = exp_c / total_exp

            continue
        score_grad_norm = score_grad / grad_norm
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * score_grad_norm
        a_new = a - lr * momentum
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a
        else:
            a_new /= s
```
