import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_val(a):
    """Compute objective 2*n*max(conv(a,a))/sum(a)^2 assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        if sweep % 5 == 0 and sweep <= 10:
            elapsed = time.monotonic() - t0
            print("    CD sweep %d, obj=%.6f, elapsed=%.0fs" % (sweep, best_obj, elapsed))

    return best_a, best_obj


def peak_anneal(a0, time_limit, rng):
    """Peak-targeted simulated annealing."""
    a = normalize(a0.copy())
    n = len(a)
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.02 * (1.0 - frac) ** 2 + 0.0003
        max_c = conv.max()
        if max_c < 1e-15:
            break

        peak_thresh = max_c * 0.80
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])

        k = int(rng.choice(peak_positions[:15]))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))

        target_j = int(rng.integers(n))
        for _ in range(8):
            candidate = int(rng.integers(n))
            contrib = 0.0
            for pk in peak_positions[:6]:
                d = pk - candidate
                if 0 <= d < n:
                    contrib += a[d]
            if contrib < a.mean() * 1.5:
                target_j = candidate
                break

        if target_j == top_j:
            continue
        delta = a[top_j] * rng.uniform(0.02, 0.08)
        if delta < 1e-12:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue

        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g = np.maximum(g, 0.01)
        g /= g.sum()
        inits['gauss_%s' % sf] = g

    for freq in [1, 2, 3, 5]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        s = np.maximum(s, 0.05)
        s /= s.sum()
        inits['sin_%d' % freq] = s

    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    center = np.zeros(n)
    w2 = max(1, n // 4)
    center[n//2 - w2:n//2 + w2] = 1.0
    center /= center.sum()
    inits['center'] = center

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    for alpha in [0.2, 0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['dir_%s' % alpha] = r

    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    rc = np.maximum(rc, 0.05)
    rc /= rc.sum()
    inits['raised_cos'] = rc

    if parent is not None and len(parent) > 0:
        p = np.asarray(parent, dtype=np.float64)
        p = np.maximum(p, 0.0)
        if p.sum() > 1e-10:
            if len(p) >= n:
                pp = p[:n].copy()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp

    # Thinned periodic
    period = max(2, n // 8)
    thin = np.zeros(n)
    for j in range(0, n, period):
        thin[j] = 1.0
    thin = np.maximum(thin, 0.01)
    thin /= thin.sum()
    inits['thinned'] = thin

    # Spectral init: inverse FFT of flat-ish spectrum
    spectrum = np.ones(n // 2 + 1)
    spectrum[np.random.default_rng(777).integers(1, max(2, n // 4))] = 0.5
    sp = np.zeros(n)
    sp[:len(spectrum)] = spectrum
    sp_init = np.fft.irfft(sp * np.exp(1j * rng.uniform(0, 2 * np.pi, len(spectrum))), n)
    sp_init = np.maximum(sp_init, 0.0)
    if sp_init.sum() > 1e-10:
        sp_init /= sp_init.sum()
        inits['spectral'] = sp_init

    return inits


def pgd_optimize(a0, time_limit, t_init=30.0, lr_init=0.06):
    """Projected gradient descent."""
    a = normalize(a0.copy())
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    lr = lr_init
    patience = 0
    max_pat = 35
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        sg = np.zeros(n)
        thresh = w.max() * 0.003
        active = np.where(w > thresh)[0]
        for ki in active:
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
        a_new = a - lr * sg
        a_new = project_simplex(a_new)
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            lr = max(0.003, lr * 0.75)
            t *= 1.3
    return best_a, best_obj


def get_parent_seq():
    """Try to get parent sequence from global variable."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multi-scale scan, coordinate descent, mirror descent,
    and peak-targeted annealing to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)

    parent = get_parent_seq()

    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  NEW BEST: score=%.6f, n=%d, t=%.0fs" % (score, len(a), time.monotonic() - t_start))

    # Phase 1: Multi-scale scan
    print("Phase 1: Multi-scale scan")
    scan_lengths = [128, 256, 384, 512, 768, 1024, 1536, 2048]
    scan_results = []
    phase1_end = t_start + min(400.0, (deadline - t_start) * 0.2)

    for n in scan_lengths:
        if time.monotonic() > phase1_end - 60:
            break
        inits = generate_initializations(n, rng, parent)
        n_configs = len(inits)
        time_per = max(1.0, (phase1_end - 60 - time.monotonic()) / max(1, (len(scan_lengths) - scan_lengths.index(n)) * n_configs))

        for name, a0 in list(inits.items())[:6]:
            if time.monotonic() > phase1_end - 60:
                break
            t_alloc = min(time_per, phase1_end - 60 - time.monotonic())
            if t_alloc < 1.0:
                break
            a_opt, o = mirror_descent(a0, time_limit=t_alloc, t_init=30.0, eta_init=0.4)
            score = safe_evaluate(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt))
            record(a_opt)
            a_opt2, o2 = coordinate_descent(a_opt, time_limit=min(3.0, t_alloc * 0.3))
            record(a_opt2)

    print("  Scan done. Best so far: %.6f" % best_score)
    scan_results.sort(key=lambda x: x[0])

    # Phase 2: Deep optimization on top candidates
    print("Phase 2: Deep optimization")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 10:
            top_configs.append((n, name, a_opt))
            seen.add(key)

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    phase2_end = t_start + min(1800.0, (deadline - t_start) * 0.7)
    n_deep = len(top_configs)
    time_per_deep = max(15.0, (phase2_end - time.monotonic()) / max(1, n_deep))

    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > phase2_end - 30:
            break
        t_alloc = min(time_per_deep, phase2_end - 30 - time.monotonic())
        if t_alloc < 5.0:
            break
        print("  [%d/%d] n=%d, %s, budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))

        sub = t_alloc / 6.0
        a_cur = a0.copy()

        a_cur, _ = mirror_descent(a_cur, time_limit=sub * 1.2, t_init=40.0, eta_init=0.35)
        record(a_cur)

        a_cur, _ = coordinate_descent(a_cur, time_limit=sub * 1.5)
        record(a_cur)

        a_cur, _ = pgd_optimize(a_cur, time_limit=sub, t_init=50.0, lr_init=0.05)
        record(a_cur)

        a_cur, _ = coordinate_descent(a_cur, time_limit=sub * 1.5)
        record(a_cur)

        a_cur, _ = mirror_descent(a_cur, time_limit=sub, t_init=150.0, eta_init=0.12)
        record(a_cur)

        a_cur, _ = coordinate_descent(a_cur, time_limit=max(2.0, t_alloc - 5 * sub), n_sweeps=5)
        record(a_cur)

    print("  Deep opt done. Best: %.6f" % best_score)

    # Phase 3: Peak-targeted annealing
    if best_sequence is not None and time.monotonic() < deadline - 60:
        print("Phase 3: Peak-targeted annealing")
        phase3_budget = min(400.0, deadline - time.monotonic() - 60)
        n_anneal = min(6, max(2, int(phase3_budget / 60)))
        anneal_per = phase3_budget / n_anneal

        for i in range(n_anneal):
            if time.monotonic() > deadline - 60:
                break
            a_start = best_sequence.copy()
            a_start = normalize(a_start)
            rng_a = np.random.default_rng(seed + 10000 + i * 7919)
            a_opt, _ = peak_anneal(a_start, time_limit=anneal_per, rng=rng_a)
            record(a_opt)
            a_opt2, _ = coordinate_descent(a_opt, time_limit=min(5.0, anneal_per * 0.2))
            record(a_opt2)

        print("  Annealing done. Best: %.6f" % best_score)

    # Phase 4: Final coordinate descent polish
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("Phase 4: Final polish")
        a_best = normalize(best_sequence.copy())
        remaining = deadline - time.monotonic() - 5
        a_opt, _ = coordinate_descent(a_best, time_limit=remaining * 0.6, n_sweeps=50)
        record(a_opt)

        if time.monotonic() < deadline - 5:
            rem2 = deadline - time.monotonic() - 5
            a_opt2, _ = mirror_descent(a_opt, time_limit=rem2 * 0.4, t_init=500.0, eta_init=0.03)
            record(a_opt2)

            if time.monotonic() < deadline - 5:
                rem3 = deadline - time.monotonic() - 5
                a_opt3, _ = coordinate_descent(a_opt2, time_limit=rem3, n_sweeps=10)
                record(a_opt3)

        print("  Polish done. Best: %.6f" % best_score)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f, length=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
