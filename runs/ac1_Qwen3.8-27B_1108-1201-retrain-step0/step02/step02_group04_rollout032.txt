import numpy as np
import time
import math

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a**2)

def compute_gradient(a, T):
    n = len(a)
    c = fft_convolve(a, a)
    sum_a = np.sum(a)
    
    c_max = np.max(c)
    c_shifted = (c - c_max) / T
    exp_c = np.exp(c_shifted)
    sum_exp = np.sum(exp_c)
    p = exp_c / sum_exp
    g = T * (math.log(sum_exp) + c_max / T)
    
    # Cross-correlation H[i] = sum_j p[i+j] * a[j]
    L = 1
    while L < (2*n-1) + n - 1:
        L <<= 1
    Fp = np.fft.rfft(p, L)
    Fa = np.fft.rfft(a, L)
    H = np.fft.irfft(Fp * np.conj(Fa), L)[:n]
    
    dg_da = (2.0 / T) * H
    grad = 2.0 * n * (dg_da / (sum_a**2) - 2.0 * g / (sum_a**3))
    score_smooth = 2.0 * n * g / (sum_a**2)
    
    return grad, score_smooth

def lp_update(a, M, max_iter=1000):
    """Solve the LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0"""
    import scipy.optimize as so
    n = len(a)
    
    # Build constraint matrix A: A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(k+1, n)):
            A[k, j] = a[k-j]
    
    # max c^T b s.t. A b <= M, b >= 0
    c_obj = -np.ones(n)  # minimize -sum(b)
    b_constraint = np.full(2*n-1, M)
    
    res = so.linprog(c_obj, A_ub=A, b_ub=b_constraint, bounds=[(0, None)]*n, method='highs')

"""
Optimization of non-negative step function minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses gradient descent with smoothed max, FFT-based gradients, LP refinement,
and multiple restarts to search for sequences with low peak auto-correlation.

import numpy as np
import time
import math


def fft_convolve(a, b=None):
    """Compute linear convolution of two 1-D arrays using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the exact evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_smoothed_score_and_gradient(a, T):
    """
    Compute smoothed score and its gradient using log-sum-exp approximation of max.
    a should be normalized to sum=1 for stability.
    """
    n = len(a)
    c = fft_convolve(a, a)
    sum_a = np.sum(a)

    # Softmax for smooth max
    c_max = np.max(c)
    c_shifted = (c - c_max) / T
    # Clip for numerical stability
    c_shifted = np.clip(c_shifted, -500, 500)
    exp_c = np.exp(c_shifted)
    sum_exp = np.sum(exp_c)
    if sum_exp < 1e-300:
        p = np.full_like(c, 1.0 / len(c))
        g = c_max
    else:
        p = exp_c / sum_exp
        g = T * (math.log(sum_exp) + c_max / T)

    # Cross-correlation: H[i] = sum_j p[i+j] * a[j] for j=0..n-1
    # Using FFT: H = ifft(fft(p) * conj(fft(a))) with zero-padding
    L = 1
    while L < (2 * n - 1) + n - 1:
        L <<= 1
    Fp = np.fft.rfft(p, L)
    Fa = np.fft.rfft(a, L)
    H = np.fft.irfft(Fp * np.conj(Fa), L)[:n]

    # dg/da[i] = (2/T) * H[i]
    dg_da = (2.0 / T) * H

    # f = 2n * g / sum_a^2
    # df/da[i] = 2n * (dg_da[i]/sum_a^2 - 2*g/sum_a^3)
    grad = 2.0 * n * (dg_da / (sum_a ** 2) - 2.0 * g / (sum_a ** 3))
    smooth_score = 2.0 * n * g / (sum_a ** 2)

    return grad, smooth_score


def build_conv_matrix(a):
    """Build the convolution constraint matrix A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k + 1, n)
        for j in range(j_start, j_end):
            A[k, j] = a[k - j]
    return A


def lp_update(a, M, time_budget=30.0):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Returns optimal b or None.
    """
    import scipy.optimize as so
    n = len(a)
    if n > 2000:
        return None

    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    t0 = time.monotonic()
    try:
        res = so.linprog(c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_budget})
    except Exception:
        return None

    if time.monotonic() - t0 > time_budget:
        return None

    if res.success and res.x is not None:
        return np.clip(res.x, 0, None)
    return None

            # Try random coordinate changes
            improved = False
            for _ in range(n):
                i = rng.integers(0, n)
                a_trial = a.copy()
                a_trial[i] = max(0.0, a_trial[i] + rng.standard_normal() * 0.1)
                s_trial = np.sum(a_trial)
                if s_trial < 1e-10:
                    continue
                a_trial = a_trial / s_trial
                trial_score = compute_score(a_trial)
                if trial_score < a_score:
                    a = a_trial
                    a_score = trial_score
                    improved = True
                    if a_score < best_local_score:
                        best_local_score = a_score
                        best_local = a.copy()
        if num_passes > 5:
            # Add noise to escape
            a_noisy = a + 0.05 * rng.standard_normal(n)
            a_noisy = np.clip(a_noisy, 0.0, None)
            s_noisy = np.sum(a_noisy)
            if s_noisy > 1e-10:
                a_noisy = a_noisy / s_noisy
                if compute_score(a_noisy) < a_score * 1.05:
                    a = a_noisy
                    a_score = compute_score(a)
            num_passes = 0

    return best_local, best_local_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Find the optimal non-negative sequence minimizing the evaluation score.
    Uses multi-phase optimization: gradient descent, LP refinement, coordinate descent.
    """
    t_global = time.monotonic()
    deadline = t_global + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def update_best(seq, score):
        nonlocal best_score, best_seq
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"    *** NEW BEST: {best_score:.8f} (n={len(seq)})")

    # Phase 0: Collect starting candidates
    candidates = []

    # Provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            hs = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            if len(hs) > 0 and np.sum(hs) > 0.01:
                s = compute_score(hs)
                print(f"Provided sequence: n={len(hs)}, score={s:.8f}")
                candidates.append(hs.copy())
                update_best(hs, s)
        except Exception as e:
            print(f"Error with provided sequence: {e}")

    # Generate diverse random starting points
    for n_trial in [128, 192, 256, 320, 512]:
        for _ in range(2):
            a = np.abs(rng.standard_normal(n_trial)) + 0.1
            candidates.append(a.copy())

    # Asymmetric structured initializations
    for n_trial in [128, 256, 512]:
        i = np.arange(n_trial, dtype=np.float64)
        # Exponentially decaying (asymmetric)
        a1 = np.exp(-i / (n_trial * 0.3))
        candidates.append(a1.copy())
        # Quadratic bump off-center
        center = n_trial * 0.35
        a2 = np.exp(-((i - center) ** 2) / (n_trial * 0.1))
        candidates.append(a2.copy())
        # Linear ramp (maximally asymmetric)
        a3 = np.linspace(0.1, 1.0, n_trial)
        candidates.append(a3.copy())

    # Sort candidates by initial score (best first)
    candidate_scores = []
    for c in candidates:
        s = compute_score(c)
        candidate_scores.append((s, c))
    candidate_scores.sort(key=lambda x: x[0])

    print(f"\nTop 5 candidates:")
    for s, c in candidate_scores[:5]:
        print(f"  n={len(c)}, score={s:.6f}")

    # Phase 1: Gradient descent on top candidates
    print(f"\n=== Phase 1: Gradient Descent ===")
    time_per_gd = 120.0
    num_gd = min(5, len(candidate_scores))

    for ci in range(num_gd):
        if time.monotonic() > deadline - 120:
            break
        s_init, a_init = candidate_scores[ci]
        time_left = deadline - time.monotonic()
        t_alloc = min(time_per_gd, time_left * 0.4)
        if t_alloc < 10:
            break
        print(f"\n  GD on candidate {ci}: n={len(a_init)}, init_score={s_init:.6f}, budget={t_alloc:.0f}s")
        result, score = gradient_descent_optimize(a_init, t_alloc, rng)
        update_best(result, score)

    # Phase 2: LP refinement on best
    print(f"\n=== Phase 2: LP Refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        time_left = deadline - time.monotonic()
        t_alloc = min(240.0, time_left * 0.35)
        print(f"  LP on best: n={len(best_seq)}, current_best={best_score:.8f}, budget={t_alloc:.0f}s")
        result, score = lp_refine(best_seq, t_alloc)
        update_best(result, score)

        # Also try LP on a few other good candidates
        for ci in range(num_gd, min(num_gd + 3, len(candidate_scores))):
            if time.monotonic() > deadline - 30:
                break
            s_init, a_init = candidate_scores[ci]
            t_alloc = min(60.0, (deadline - time.monotonic()) * 0.3)
            if t_alloc < 10:
                break
            print(f"\n  LP on candidate {ci}: n={len(a_init)}, init_score={s_init:.6f}")
            result, score = lp_refine(a_init, t_alloc)
            update_best(result, score)

    # Phase 3: Coordinate descent refinement
    print(f"\n=== Phase 3: Coordinate Descent ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        time_left = deadline - time.monotonic()
        t_alloc = min(180.0, time_left * 0.5)
        print(f"  CD on best: n={len(best_seq)}, current_best={best_score:.8f}, budget={t_alloc:.0f}s")
        result, score = coordinate_descent_refine(best_seq, t_alloc, rng)
        update_best(result, score)

    # Phase 4: More gradient descent with different params
    print(f"\n=== Phase 4: Additional Optimization ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        time_left = deadline - time.monotonic()
        t_alloc = min(120.0, time_left * 0.6)
        if t_alloc > 20:
            # Try with different temperature schedule
            print(f"  Additional GD: budget={t_alloc:.0f}s")
            result, score = gradient_descent_optimize(best_seq, t_alloc, rng)
            update_best(result, score)

        # Try coordinate descent again
        if time.monotonic() < deadline - 15:
            t_alloc2 = min(60.0, (deadline - time.monotonic()) * 0.5)
            if t_alloc2 > 10:
                result, score = coordinate_descent_refine(best_seq, t_alloc2, rng)
                update_best(result, score)

    # Phase 5: Final LP polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_alloc = min(45.0, max(5.0, deadline - time.monotonic() - 10))
        result, score = lp_refine(best_seq, t_alloc, max_iterations=20)
        update_best(result, score)
        result, score = coordinate_descent_refine(result, min(20, max(5, deadline - time.monotonic() - 10)), rng)
        update_best(result, score)

    # Final answer
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = 2.0

    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(final_list)
    print(f"\n{'='*60}")
    print(f"=== FINAL: score={final_score:.8f}, n={len(final_list)} ===")
    print(f"{'='*60}")

    return final_list
