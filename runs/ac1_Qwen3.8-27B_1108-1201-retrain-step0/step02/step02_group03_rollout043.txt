```
def self_conv(a):
```

```
def lp_solve(f, M, timeout=None):

    rows=[]; cols=[]; data=[]
```

```python
"""
Optimizer for non-negative step functions minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2.

The program uses several complementary strategies:
  - quick broad candidate generation,
  - coarse optimization with piecewise-constant / bin parameters,
  - Adam-style mirror descent with annealed top-k smoothing,
  - targeted local search using active-convolution-lag gradient transfers,
  - optional LP direction moves inspired by the previous SOTA,
  - final polishing and safety checks.

The returned sequence is normalized to sum 1, which is harmless because the
objective is scale invariant.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    SCIPY_OK = True
except Exception:
    SCIPY_OK = False


def clean_array(a):
    """Convert to float64 array, remove NaN/inf, clip negatives to zero."""
    a = np.asarray(a, dtype=np.float64)
    if a.ndim == 0:
        a = a.reshape(1)
    a = np.where(np.isfinite(a), a, 0.0)
    return np.maximum(a, 0.0)


def normalize(a):
    """Return a non-negative array with sum 1."""
    a = clean_array(a)
    n = len(a)
    if n == 0:
        return a
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def self_conv(a):
    """Linear self-convolution of a 1D array using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    size = 1
    while size < L:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:L]
    c[c < 0.0] = 0.0
    return c


def lin_conv(a, b):
    """Linear convolution of two 1D arrays using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([], dtype=np.float64)
    L = n + m - 1
    size = 1
    while size < L:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:L]


def gradient_wrap(a, w):
    """
    Gradient of sum_t w[t] * (a*a)[t] with respect to a.
    For c_t = sum_i a_i a_{t-i}, derivative is 2 * sum_k a_k w[t+k].
    """
    n = len(a)
    L = 2 * n - 1
    wr = w[::-1]
    h = lin_conv(a, wr)
    # Valid output is h[L-n : L], reversed.
    return 2.0 * h[L - n:L][::-1]


def normalized_max_conv(a):
    """
    Return (M, normalized_a) where M = max(conv(a,a)) after scaling sum(a)=1.
    """
    a = clean_array(a)
    n = len(a)
    if n == 0:
        return np.inf, a
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.inf, a
    a = a / s
    c = self_conv(a)
    mval = float(np.max(c))
    if not np.isfinite(mval):
        mval = np.inf
    return mval, a


def score(a):
    """Objective value, scale invariant."""
    a = clean_array(a)
    n = len(a)
    if n == 0:
        return np.inf
    mval, _ = normalized_max_conv(a)
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval)


def resize_sequence(a, new_n):
    """Piecewise-linear resampling to length new_n."""
    a = clean_array(a)
    n = len(a)
    if new_n <= 0:
        return np.array([], dtype=np.float64)
    if n == 0:
        return np.ones(new_n, dtype=np.float64)
    if n == new_n:
        return a.copy()
    if n == 1:
        x_old = np.array([0.5], dtype=np.float64)
    else:
        x_old = np.linspace(0.0, 1.0, n)
    if new_n == 1:
        x_new = np.array([0.5], dtype=np.float64)
    else:
        x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def get_score_tuple(item):
    return item[0]


def update_best(best_seq, best_score, a):
    """Update best sequence if a has a lower score."""
    if a is None:
        return best_seq, best_score
    try:
        sc = score(a)
    except Exception:
        return best_seq, best_score
    if np.isfinite(sc) and sc < best_score:
        return clean_array(a).copy(), float(sc)
    return best_seq, best_score


def generate_random_candidates(n, rng, count):
    """Generate diverse nonnegative random smooth-ish sequences of length n."""
    out = []
    if n <= 0:
        return out
    x = (np.arange(n) + 0.5) / float(n)
    for _ in range(count):
        typ = rng.random()
        if typ < 0.22:
            alpha = rng.uniform(-2.0, 2.0)
            a = 1.0 + alpha * (x - 0.5)
        elif typ < 0.45:
            c1 = rng.uniform(-1.0, 1.0)
            c2 = rng.uniform(-1.0, 1.0)
            c3 = rng.uniform(-1.0, 1.0)
            y = x - 0.5
            a = 1.0 + c1 * y + 0.35 * c2 * y * y + 0.12 * c3 * y * y * y
        elif typ < 0.72:
            a = np.ones(n, dtype=np.float64)
            ntones = int(rng.integers(2, 6))
            for j in range(ntones):
                freq = int(rng.integers(1, 7))
                amp = rng.uniform(0.10, 0.60) / float(j + 1)
                phase = rng.uniform(0.0, 2.0 * np.pi)
                a += amp * np.cos(2.0 * np.pi * freq * x + phase)
            a += 0.05 * rng.standard_normal(n)
        elif typ < 0.88:
            a = np.exp(rng.normal(0.0, rng.uniform(0.2, 1.2), n))
            # mild low-pass / growth smoothing
            den = max(1, n // 8)
            a = a[0:1] + np.cumsum(a) / float(den)
        else:
            # three-block pattern plus strong smoothing
            first = max(1, n // 3)
            last = max(1, n // 3)
            b = np.ones(n, dtype=np.float64)
            b[:first] = 0.3
            b[-last:] = 0.3
            b[first:n - last] = 0.4
            a = b + 0.25 * rng.standard_normal(n)
        a = clean_array(a) + 1e-9
        a = normalize(a)
        out.append(a)
    return out


def optimize_adam(a_init, time_limit, rng, lr_base=0.8, print_label=""):
    """
    Adam-style mirror descent on softmax-parameterized nonnegative sequence.
    Minimizes a top-k smoothed version of max self-convolution.
    """
    if time_limit <= 0:
        a = clean_array(a_init)
        return a, score(a)

    a = normalize(a_init)
    n = len(a)
    if n == 0:
        return a, np.inf
    L = 2 * n - 1

    best = a.copy()
    best_score = score(a)

    theta = np.log(a + 1e-15)
    mom = np.zeros(n, dtype=np.float64)
    var = np.zeros(n, dtype=np.float64)

    t0 = time.monotonic()
    step = 0
    stall = 0
    last_print = t0
    topk_start = 0.25
    topk_end = 0.005

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-12))

        if L <= 2:
            K = 1
        else:
            K = int(np.ceil(L * topk_start * (topk_end / topk_start) ** (frac ** 0.7)))
            K = max(2, min(K, max(2, L // 2)))

        shifted = theta - np.max(theta)
        shifted = np.clip(shifted, -50.0, 0.0)
        a = np.exp(shifted)
        a = np.maximum(a, 1e-15)
        sa = float(np.sum(a))
        if sa <= 1e-12 or not np.isfinite(sa):
            a = best.copy()
            theta = np.log(a + 1e-15)
            mom = np.zeros(n, dtype=np.float64)
            var = np.zeros(n, dtype=np.float64)
            step = 0
        else:
            a = a / sa
            c = self_conv(a)
            if K < L:
                top_idx = np.argpartition(c, -K)[-K:]
            else:
                top_idx = np.arange(L)
            w = np.zeros(L, dtype=np.float64)
            w[top_idx] = 1.0 / float(max(1, len(top_idx)))

            ga = gradient_wrap(a, w)
            gtheta = a * (ga - float(np.dot(a, ga)))
            gtheta -= float(np.mean(gtheta))
            if not np.all(np.isfinite(gtheta)):
                gtheta = np.zeros(n, dtype=np.float64)

            step += 1
            b1 = 0.85
            b2 = 0.999
            mom = b1 * mom + (1.0 - b1) * gtheta
            var = b2 * var + (1.0 - b2) * (gtheta * gtheta)
            mom_hat = mom / (1.0 - b1 ** step)
            var_hat = var / (1.0 - b2 ** step)

            lr = lr_base * (1.0 - 0.80 * frac)
            theta = theta - lr * mom_hat / (np.sqrt(var_hat) + 1e-8)

            if not np.all(np.isfinite(theta)):
                theta = np.log(a + 1e-15)
                mom = np.zeros(n, dtype=np.float64)
                var = np.zeros(n, dtype=np.float64)
                step = 0

        if step % 200 == 0:
            shifted = theta - np.max(theta)
            shifted = np.clip(shifted, -50.0, 0.0)
            a_check = np.exp(shifted)
            a_check = np.maximum(a_check, 1e-15)
            sa = float(np.sum(a_check))
            if sa > 1e-12:
                a_check = a_check / sa
                sc = score(a_check)
                if np.isfinite(sc) and sc < best_score:
                    best = a_check.copy()
                    best_score = float(sc)
                    stall = 0
                else:
                    stall += 200

                if stall > 4500:
                    a_noise = clean_array(best * np.exp(rng.normal(0.0, 0.22, n))) + 1e-12
                    a_noise = normalize(a_noise)
                    theta = np.log(a_noise + 1e-15)
                    mom = np.zeros(n, dtype=np.float64)
                    var = np.zeros(n, dtype=np.float64)
                    step = 0
                    stall = 0

        if print_label and time.monotonic() - last_print > 30.0:
            print(f"    {print_label}: best={best_score:.6f} t={time.monotonic()-t0:.1f}s")
            last_print = time.monotonic()

    return best, best_score


def optimize_bin(w_init, R, time_limit, rng, print_label=""):

    Optimize a piecewise-constant sequence made by repeating m bin values R times.
    w_init: bin weights, sum normalized to 1.
    Returns full length m*R sequence and its score.

        a = np.repeat(w, R) / float(max(1, R))
        return a, score(a)

    if R <= 0:
        R = 1
    w = normalize(w_init)
    m_len = len(w)
    if m_len == 0:
        return w, np.inf
    n = m_len * R
    L = 2 * n - 1

    best_w = w.copy()
    best_score = score(np.repeat(best_w, R) / float(R))
    theta = np.log(best_w + 1e-15)
    mom = np.zeros(m_len, dtype=np.float64)
    var = np.zeros(m_len, dtype=np.float64)
    starts = np.arange(0, n, R, dtype=np.int64)

    t0 = time.monotonic()
    step = 0
    stall = 0
    last_print = t0
    topk_start = 0.25
    topk_end = 0.005

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-12))

        if L <= 2:
            K = 1
        else:
            K = int(np.ceil(L * topk_start * (topk_end / topk_start) ** (frac ** 0.7)))
            K = max(2, min(K, max(2, L // 2)))

        shifted = np.clip(shifted, -50.0, 0.0)
        w = np.exp(shifted)
        w = np.maximum(w, 1e-15)

        if sw <= 1e-12 or not np.isfinite(sw):
            w = best_w.copy()
            theta = np.log(w + 1e-15)
            mom = np.zeros(m_len, dtype=np.float64)
            var = np.zeros(m_len, dtype=np.float64)
            step = 0
        else:
            w = w / sw
            a = np.repeat(w, R) / float(R)
            c = self_conv(a)
            if K < L:
                top_idx = np.argpartition(c, -K)[-K:]
            else:
                top_idx = np.arange(L)
            wg = np.zeros(L, dtype=np.float64)
            wg[top_idx] = 1.0 / float(max(1, len(top_idx)))

            ga = gradient_wrap(a, wg)
            # Pull back gradient to bin weights.
            gw = np.add.reduceat(ga, starts, axis=0) / float(R)
            gtheta = w * (gw - float(np.dot(w, gw)))
            gtheta -= float(np.mean(gtheta))
            if not np.all(np.isfinite(gtheta)):
                gtheta = np.zeros(m_len, dtype=np.float64)

            step += 1
            b1 = 0.85
            b2 = 0.999
            mom = b1 * mom + (1.0 - b1) * gtheta
            var = b2 * var + (1.0 - b2) * (gtheta * gtheta)
            mom_hat = mom / (1.0 - b1 ** step)
            var_hat = var / (1.0 - b2 ** step)

            lr = 1.0 * (1.0 - 0.75 * frac)
            theta = theta - lr * mom_hat / (np.sqrt(var_hat) + 1e-8)

            if not np.all(np.isfinite(theta)):
                theta = np.log(w + 1e-15)
                mom = np.zeros(m_len, dtype=np.float64)
                var = np.zeros(m_len, dtype=np.float64)
                step = 0

        if step % 80 == 0:
            shifted = theta - np.max(theta)
            shifted = np.clip(shifted, -50.0, 0.0)
            w_check = np.exp(shifted)
            w_check = np.maximum(w_check, 1e-15)
            swc = float(np.sum(w_check))
            if swc > 1e-12:
                w_check = w_check / swc
                a_check = np.repeat(w_check, R) / float(R)
                sc = score(a_check)
                if np.isfinite(sc) and sc < best_score:
                    best_score = float(sc)
                    best_w = w_check.copy()
                    stall = 0
                else:
                    stall += 80

                if stall > 3500:
                    w_noise = clean_array(best_w * np.exp(rng.normal(0.0, 0.25, m_len))) + 1e-12
                    w_noise = normalize(w_noise)
                    theta = np.log(w_noise + 1e-15)
                    mom = np.zeros(m_len, dtype=np.float64)
                    var = np.zeros(m_len, dtype=np.float64)
                    step = 0
                    stall = 0

        if print_label and time.monotonic() - last_print > 20.0:
            print(f"    {print_label}: best={best_score:.6f} t={time.monotonic()-t0:.1f}s")
            last_print = time.monotonic()

    best_a = np.repeat(best_w, R) / float(R)
    best_a = normalize(best_a)
    best_score = score(best_a)
    return best_a, best_score


def coarse_search(n, ref, time_limit, rng):
    """
    Coarse-to-fine search by optimizing bin weights and replicating each bin.
    """
    if n <= 2 or time_limit <= 10:
        return None, np.inf

    m_target = 32
    R = max(1, n // m_target)
    m = n // R
    if m < 4:
        return None, np.inf

    inits = []

    if ref is not None:
        aa = resize_sequence(ref, n)
        try:
            block = aa.reshape(m, R).sum(axis=1)
            inits.append(normalize(block))
        except Exception:
            pass

    # Smooth three-block low-high-low initialization.
    w3 = np.ones(m, dtype=np.float64)
    third = max(1, m // 3)
    w3[:third] = 0.30
    w3[-third:] = 0.30
    w3[third:m - third] = 0.40
    w3 += 0.03 * rng.standard_normal(m)
    inits.append(normalize(w3))

    # Linear ramp.
    x = np.linspace(0.0, 1.0, m)
    inits.append(normalize(np.maximum(1.0 + 1.2 * (x - 0.5), 1e-6)))

    # Random smooth bin weights.
    wrand = np.exp(rng.normal(0.0, 0.7, m))
    inits.append(normalize(wrand))

    inits = inits[:4]
    n_init = max(1, len(inits))
    time_per = max(8.0, (time_limit - 15.0) / float(n_init))

    best_a = None
    best_sc = np.inf
    for w in inits:
        if time.monotonic() + time_per > time_limit + time.monotonic() - time_limit:
            pass
        a_opt, sc_opt = optimize_bin(w, R, time_per, rng)
        if a_opt is not None:
            a_opt = normalize(a_opt)
            sc_opt = score(a_opt)
            if np.isfinite(sc_opt) and sc_opt < best_sc:
                best_a = a_opt.copy()
                best_sc = float(sc_opt)

    return best_a, best_sc


def choose_top_candidates(candidates, max_count):
    """Pick top candidates, with at most two per length."""
    scored = []
    for a in candidates:
        if a is None:
            continue
        try:
            sc = score(a)
        except Exception:
            continue
        if np.isfinite(sc):
            scored.append((sc, clean_array(a).copy()))
    if not scored:
        return []
    scored.sort(key=get_score_tuple)

    selected = []
    len_count = {}
    for sc, a in scored:
        n = len(a)
        cnt = len_count.get(n, 0)
        if cnt >= 2:
            continue
        selected.append((sc, a))
        len_count[n] = cnt + 1
        if len(selected) >= max_count:
            break
    return selected


def local_search_active(a_init, time_limit, rng, print_label=""):
    """
    Targeted local search with simulated-annealing acceptance.
    Moves are driven by gradients of the highest convolution lags.
    """
    if time_limit <= 0:
        a = clean_array(a_init)
        return a, score(a)

    a = normalize(a_init)
    n = len(a)
    if n == 0:
        return a, np.inf

    cur_M, a = normalized_max_conv(a)
    best_M = cur_M
    best = a.copy()
    best_score = float(2.0 * n * best_M)

    L = 2 * n - 1
    t0 = time.monotonic()
    last_print = t0
    stall = 0
    topk_start = 64
    topk_end = 8
    n_trials = 3 if n <= 600 else 1

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-12))
        temp = best_M * 1e-4 * (1e-4 ** frac)
        if temp < best_M * 1e-8:
            temp = best_M * 1e-8

        c = self_conv(a)
        cur_M = float(np.max(c))
        if cur_M < best_M:
            best_M = cur_M
            best = a.copy()
            best_score = float(2.0 * n * best_M)

        K = int(topk_start * (topk_end / topk_start) ** (frac ** 0.6))
        if L <= 4:
            K = max(1, min(K, L))
        else:
            K = max(4, min(K, max(4, L // 4)))

        top_idx = np.argpartition(c, -K)[-K:]
        w = np.zeros(L, dtype=np.float64)
        w[top_idx] = 1.0 / float(max(1, len(top_idx)))
        gtop = gradient_wrap(a, w)
        if not np.all(np.isfinite(gtop)):
            gtop = np.zeros(n, dtype=np.float64)
        move_score = a * gtop

        r = rng.random()
        candidate = None
        cand_M = np.inf

        if r < 0.32:
            # Dense projected gradient step.
            d = -gtop - float(np.mean(gtop))
            nd = float(np.linalg.norm(d))
            na = float(np.linalg.norm(a))
            if nd > 1e-14:
                if n <= 600:
                    etas = (0.05, 0.15, 0.35, 0.70)
                else:
                    etas = (0.10, 0.25, 0.50)
                best_ct_m = np.inf
                best_ct_a = None
                for eta in etas:
                    step_size = eta * na / nd
                    cand_raw = a + step_size * d
                    mval, cand_norm = normalized_max_conv(cand_raw)
                    if np.isfinite(mval) and mval < best_ct_m:
                        best_ct_m = mval
                        best_ct_a = cand_norm
                if best_ct_a is not None:
                    candidate = best_ct_a
                    cand_M = best_ct_m

        elif r < 0.68:
            # Sparse transfer from high-gradient mass to low-gradient mass.
            n_pick = min(10, n)
            best_ct_m = np.inf
            best_ct_a = None
            for _ in range(n_trials):
                if n_pick > 1:
                    src_order = np.argpartition(move_score, -n_pick)[-n_pick:]
                    tgt_order = np.argpartition(move_score, n_pick - 1)[:n_pick]
                else:
                    src_order = np.array([0], dtype=np.int64)
                    tgt_order = np.array([0], dtype=np.int64)

                src = int(src_order[rng.integers(len(src_order))])
                tgt = int(tgt_order[rng.integers(len(tgt_order))])

                if src == tgt:
                    changed = False
                    for tidx in tgt_order:
                        tval = int(tidx)
                        if tval != src:
                            tgt = tval
                            changed = True
                            break
                    if not changed:
                        continue

                frac_move = rng.uniform(0.05, 0.45)
                delta = frac_move * a[src]
                if delta <= 1e-300:
                    continue
                cand_raw = a.copy()
                cand_raw[src] -= delta
                cand_raw[tgt] += delta
                mval, cand_norm = normalized_max_conv(cand_raw)
                if np.isfinite(mval) and mval < best_ct_m:
                    best_ct_m = mval
                    best_ct_a = cand_norm
            if best_ct_a is not None:
                candidate = best_ct_a
                cand_M = best_ct_m

        elif r < 0.82:
            # Block transfer.
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            nblocks = max(1, n // (2 * width))
            if nblocks > 0:
                block_scores = np.sum(move_score[:nblocks * width].reshape(nblocks, width), axis=1)
                bpick = min(6, nblocks)
                src_b = int(np.argpartition(block_scores, -bpick)[-bpick:][rng.integers(bpick)])
                tgt_b = int(np.argpartition(block_scores, bpick - 1)[:bpick][rng.integers(bpick)])
                if src_b != tgt_b:
                    src_slice = slice(src_b * width, (src_b + 1) * width)
                    tgt_slice = slice(tgt_b * width, (tgt_b + 1) * width)
                    amt = rng.uniform(0.05, 0.25) * float(np.sum(a[src_slice])) / float(width)
                    cand_raw = a.copy()
                    cand_raw[src_slice] -= amt
                    cand_raw[tgt_slice] += amt
                    mval, cand_norm = normalized_max_conv(cand_raw)
                    if np.isfinite(mval):
                        candidate = cand_norm
                        cand_M = mval

        elif r < 0.92:
            # Partial reversal / symmetrization.
            alpha = rng.uniform(-0.30, 0.30)
            cand_raw = a + alpha * (a[::-1] - a) + 0.01 * rng.standard_normal(n)
            mval, cand_norm = normalized_max_conv(cand_raw)
            if np.isfinite(mval):
                candidate = cand_norm
                cand_M = mval

        else:
            # Multiplicative noise.
            sigma = rng.uniform(0.01, 0.08)
            cand_raw = a * np.exp(rng.normal(0.0, sigma, n))
            mval, cand_norm = normalized_max_conv(cand_raw)
            if np.isfinite(mval):
                candidate = cand_norm
                cand_M = mval

        if candidate is not None and np.isfinite(cand_M):
            delta = cand_M - cur_M
            accept = False
            if delta < 0.0:
                accept = True
            else:
                if temp > 1e-300:
                    ratio = delta / temp
                    if ratio < 700.0:
                        accept = rng.random() < np.exp(-ratio)
                    else:
                        accept = False
            if accept:
                a = candidate
                cur_M = cand_M
                if cur_M < best_M:
                    best_M = cur_M
                    best = a.copy()
                    best_score = float(2.0 * n * cur_M)
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        if stall > 7000:
            a_noise = clean_array(best * np.exp(rng.normal(0.0, 0.05, n))) + 1e-12
            a = normalize(a_noise)
            cur_M, a = normalized_max_conv(a)
            temp = best_M * 1e-4
            stall = 0

        if print_label and time.monotonic() - last_print > 30.0:
            print(f"    {print_label}: best_score={best_score:.6f} t={time.monotonic()-t0:.1f}s")
            last_print = time.monotonic()

    final_score = score(best)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = float(final_score)
    return best, best_score


def lp_solve(f, M, max_iters=1):
    """
    Solve max sum(g) subject to g>=0 and f*g <= M.
    Returns g or None.
    """
    if not SCIPY_OK:
        return None
    n = len(f)
    if n == 0 or M <= 0:
        return None
    L = 2 * n - 1

    rows = []
    cols = []
    data = []

    for t in range(L):
        j0 = max(0, t - n + 1)
        j1 = min(n - 1, t)
        for j in range(j0, j1 + 1):
            i = t - j
            rows.append(t)
            cols.append(j)
            data.append(float(f[i]))

    if not data:
        return None

    try:
        A = coo_matrix((data, (rows, cols)), shape=(L, n)).tocsr()
        cobj = -np.ones(n, dtype=np.float64)
        b = np.full(L, float(M), dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = linprog(cobj, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if res is not None and res.success and res.x is not None:
            x = clean_array(np.asarray(res.x, dtype=np.float64))
            sx = float(np.sum(x))
            if sx > 1e-12 and np.isfinite(sx):
                return x
    except Exception:
        pass
    return None


def lp_polish(a_init, time_limit, rng, print_label=""):
    """
    Optional LP-driven direction polishing.
    Uses the current sequence f, solves a linear program for a direction g
    with low cross-convolution f*g, and then line-searches f -> (1-t)f + t g.
    """
    if not SCIPY_OK:
        a = clean_array(a_init)
        return a, score(a)

    a = normalize(a_init)
    n = len(a)
    if n == 0 or time_limit < 30 or n > 768:
        return a, score(a)

    best = a.copy()
    best_score = score(a)
    cur_score = best_score

    t0 = time.monotonic()
    max_iters = max(1, int(time_limit / 70.0))
    last_print = t0

    for _ in range(max_iters):
        if time.monotonic() - t0 >= time_limit:
            break

        M = float(np.max(self_conv(a)))
        if M <= 0:
            break

        g = None
        for red in (0.020, 0.010, 0.000, 0.000):
            mc = M * (1.0 - red)
            if mc <= 1e-14:
                break
            g = lp_solve(a, mc)
            if g is not None:
                sg = float(np.sum(g))
                if sg > 0.75:
                    break
                if red == 0.0:
                    break
        if g is None:
            break

        sg = float(np.sum(g))
        if sg <= 1e-12 or not np.isfinite(sg):
            break
        g = g / sg

        best_ct_a = a.copy()
        best_ct_score = cur_score
        alphas = (0.05, 0.10, 0.20, 0.35, 0.55, 0.80, 1.00)
        for alpha in alphas:
            cand = (1.0 - alpha) * a + alpha * g
            mval, cand_norm = normalized_max_conv(cand)
            if np.isfinite(mval):
                sc = float(2.0 * n * mval)
                if sc < best_ct_score:
                    best_ct_score = sc
                    best_ct_a = cand_norm

        if best_ct_score < cur_score:
            a = best_ct_a
            cur_score = best_ct_score
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            break

        if print_label and time.monotonic() - last_print > 30.0:
            print(f"    {print_label}: best={best_score:.6f} t={time.monotonic()-t0:.1f}s")
            last_print = time.monotonic()

    final_score = score(best)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = float(final_score)
    return best, best_score


def remaining_time(deadline):
    return max(0.0, deadline - time.monotonic())


def propose_candidate(budget_s=2400.0, seed=42, **kwargs):
    """
    Main entry point. Searches for a good non-negative sequence within budget.
    """
    t_start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0

    # Leave time for final construction and validation.
    total_budget = max(120.0, budget - 60.0)
    deadline = t_start + total_budget
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf
    all_cands = []

    # Load provided reference sequence if available.
    ref = None
    if "height_sequence_1" in globals():
        hs1 = globals().get("height_sequence_1")
        if hs1 is not None:
            try:
                arr = np.asarray(hs1, dtype=np.float64)
                arr = clean_array(arr)
                if arr.size > 0 and float(np.sum(arr)) > 1e-12:
                    ref = arr
                    best_seq, best_score = update_best(best_seq, best_score, ref)
                    print(f"Initial reference: n={len(ref)}, score={best_score:.6f}")
            except Exception:
                ref = None

    # ---------------- Quick broad candidate generation ----------------
    if remaining_time(deadline) > 30:
        quick_t = min(60.0, remaining_time(deadline) * 0.05)
        if quick_t < 20:
            quick_t = 20.0 if remaining_time(deadline) > 20 else 0.0

        if remaining_time(deadline) < 600:
            lengths = (256,)
        elif remaining_time(deadline) < 1200:
            lengths = (256, 512)
        else:
            lengths = (256, 512, 1024)

        for n in lengths:
            if remaining_time(deadline) < 20:
                break
            cands = []
            if ref is not None:
                cands.append(resize_sequence(ref, n))
            count = 18 if n <= 256 else 12
            cands.extend(generate_random_candidates(n, rng, count))
            for a in cands:
                a = normalize(a)
                all_cands.append(a)
                best_seq, best_score = update_best(best_seq, best_score, a)
            print(f"Quick n={n}: global best={best_score:.6f}")

    # ---------------- Initial polish of reference ----------------
    if ref is not None and remaining_time(deadline) > 120:
        init_t = min(300.0, remaining_time(deadline) * 0.15)
        if init_t > 60:
            a0 = resize_sequence(ref, len(ref))
            t1 = max(10.0, 0.5 * init_t)
            t1 = min(t1, remaining_time(deadline) - 30)
            if t1 > 20:
                a1, s1 = optimize_adam(a0, t1, rng, print_label="ref-adam")
                best_seq, best_score = update_best(best_seq, best_score, a1)
                if a1 is not None:
                    all_cands.append(a1)

                t2 = max(10.0, 0.5 * init_t)
                t2 = min(t2, remaining_time(deadline) - 30)
                if t2 > 20:
                    a2, s2 = local_search_active(a1, t2, rng, print_label="ref-local")
                    best_seq, best_score = update_best(best_seq, best_score, a2)
                    if a2 is not None:
                        all_cands.append(a2)

    # ---------------- Coarse bin optimization ----------------
    if remaining_time(deadline) > 250:
        coarse_t = min(300.0, remaining_time(deadline) * 0.15)
        n_list = (256, 512, 1024)
        for n in n_list:
            if remaining_time(deadline) < 90:
                break
            t_alloc = coarse_t / float(len(n_list))
            t_alloc = min(t_alloc + 60.0, remaining_time(deadline) - 60)
            if t_alloc < 25:
                continue
            a_coarse, sc_coarse = coarse_search(n, ref, t_alloc, rng)
            best_seq, best_score = update_best(best_seq, best_score, a_coarse)
            if a_coarse is not None:
                all_cands.append(a_coarse)
            print(f"Coarse n={n}: global best={best_score:.6f}")

    # ---------------- Fine Adam optimization on top candidates ----------------
    if remaining_time(deadline) > 300 and best_seq is not None:
        fine_t = min(700.0, remaining_time(deadline) * 0.30)
        max_cand = 5 if fine_t > 500 else 3
        selected = choose_top_candidates(all_cands, max_cand)
        if best_seq is not None:
            selected.append((score(best_seq), best_seq.copy()))
        if selected:
            selected.sort(key=get_score_tuple)
            selected = selected[:max_cand]

        for sc0, a0 in selected:
            if remaining_time(deadline) < 80:
                break
            time_per = fine_t / float(max(1, len(selected)))
            time_per = max(35.0, min(180.0, time_per))
            time_per = min(time_per, remaining_time(deadline) - 50)
            if time_per < 25:
                break
            a_opt, s_opt = optimize_adam(a0, time_per, rng, print_label=f"fine n={len(a0)}")
            best_seq, best_score = update_best(best_seq, best_score, a_opt)
            if a_opt is not None:
                all_cands.append(a_opt)

    # ---------------- Optional LP polish ----------------
    if SCIPY_OK and remaining_time(deadline) > 180 and best_seq is not None:
        lp_t = min(300.0, remaining_time(deadline) * 0.15)
        if len(best_seq) <= 768 and lp_t > 50:
            a_lp, s_lp = lp_polish(best_seq, lp_t, rng, print_label="LP")
            best_seq, best_score = update_best(best_seq, best_score, a_lp)
            if a_lp is not None:
                all_cands.append(a_lp)

    # ---------------- Final local search polish ----------------
    if remaining_time(deadline) > 80 and best_seq is not None:
        local_t = remaining_time(deadline) - 60
        if local_t > 60:
            a_final, s_final = local_search_active(best_seq, local_t, rng, print_label="final-local")
            best_seq, best_score = update_best(best_seq, best_score, a_final)

    # ---------------- Final safety and validation ----------------
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64)

    best_seq = clean_array(best_seq)
    if best_seq.size == 0 or float(np.sum(best_seq)) <= 1e-12:
        best_seq = np.ones(256, dtype=np.float64)

    # Scale to sum 1. The objective is scale invariant, and this keeps numbers well-conditioned.
    sfinal = float(np.sum(best_seq))
    if sfinal > 0:
        best_seq = best_seq / sfinal
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_list = [float(x) for x in best_seq]

    # If the evaluator is available, validate. This also ensures the exact
    # returned list is sane under the harness.
    try:
        ev = evaluate_sequence(final_list)
        if np.isfinite(ev):
            print(f"Final internal score={score(np.asarray(final_list)):.6f}, "
                  f"evaluate_sequence score={float(ev):.6f}, n={len(final_list)}")
        else:
            print("evaluate_sequence returned non-finite; using fallback")
            final_list = [1.0] * 256
    except Exception:
        # If evaluate_sequence is not actually available in the runtime, internal score is enough.
        try:
            print(f"Final internal score={score(np.asarray(final_list)):.6f}, n={len(final_list)}")
        except Exception:
            print(f"Final n={len(final_list)}")

    return final_list
```
