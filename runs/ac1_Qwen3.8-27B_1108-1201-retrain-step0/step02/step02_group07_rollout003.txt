"""
LP-driven optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal perturbation directions,
combined with coordinate descent and random restarts.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                C[k, j] = a[i]
    return C


def lp_direction(a, M):
    """Solve LP to find best direction g."""
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    C = build_conv_matrix(a)
    constraints = [C @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLPK, verbose=False)
    if g.value is None:
        return a.copy()
    g_val = np.maximum(g.value, 0)
    if g_val.sum() < 1e-10:
        return a.copy()
    return g_val


def lp_optimize(a0, time_limit, max_iters=1000, t_step=0.3):
    """LP-based optimization loop."""
    a = np.maximum(a0.copy(), 0.0).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        g = lp_direction(a, M)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Check if g is essentially the same as a
        if np.allclose(g, a, atol=1e-6):
            break
        
        # Update with step size t
        a_new = (1 - t_step) * a + t_step * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()
        obj_new = 2.0 * n * M_new

"""
LP-driven optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines linear programming for direction finding, gradient-based refinement,
and multi-scale search to find near-optimal non-negative step functions.
"""

import numpy as np
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]


def build_conv_matrix(a):
    """Build matrix C (shape (2n-1, n)) such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            C[k, j] = a[k - j]
    return C


def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    C = build_conv_matrix(a)
    constraints = [C @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            return a.copy()
    if g.value is None:
        return a.copy()
    g_val = np.maximum(g.value, 0.0)
    if g_val.sum() < 1e-10:
        return a.copy()
    return g_val


def lp_optimize(a0, time_limit, max_iters=5000, t_init=0.4):
    """LP-based optimization loop with adaptive step size."""
    a = np.maximum(a0.copy(), 0.0).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    t = t_init
    iter_count = 0
    stall = 0
    last_best = best_obj

    while time.monotonic() - t0 < time_limit:
        g = lp_direction(a, M)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Check if direction is essentially the same as current
        cos_sim = np.dot(g, a) / (np.linalg.norm(g) * np.linalg.norm(a) + 1e-15)
        if cos_sim > 0.9999:
            stall += 1
            if stall > 20:
                break
        else:
            stall = 0

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            t *= 0.5
            continue
        a_new = a_new / s

        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()
        obj_new = 2.0 * n * M_new

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            M = M_new
            t = min(0.8, t * 1.05)
            stall = 0
        elif obj_new < last_best + 1e-8:
            a = a_new
            conv = conv_new
            M = M_new
            t = max(0.05, t * 0.8)
            stall += 1
        else:
            t = max(0.02, t * 0.5)
            stall += 1

        iter_count += 1
        if iter_count % 200 == 0:

        if stall > 100:
            break
        if best_obj < last_best - 1e-6:

            stall = 0

def gradient_descent(a0, time_limit, lr_init=0.05, t_smooth=30.0):
    """Gradient descent with softmax-smoothed max for smoothness."""
    a = np.maximum(a0.copy(), 0.0).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    lr = lr_init
    t = t_smooth
    t0 = time.monotonic()
    patience = 0

    while time.monotonic() - t0 < time_limit:
        # Compute smoothed max gradient
        max_c = conv.max()
        if max_c < 1e-15:
            break
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Gradient: for each j, grad_j = sum_k w[k] * 2 * a[k-j] (if valid)
        grad = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-6:
                continue
            wk = w[k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * wk * a[k - j]

        # Projected gradient step (maintain sum=1, a>=0)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new / s

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            patience = 0
        else:
            a = a_new
            conv = conv_new
            patience += 1

        if patience > 30:
            lr *= 0.7
            t *= 1.3
            patience = 0
            if lr < 0.001:
                break

    return best_a, best_obj


def coordinate_refine(a0, time_limit):

    a = np.maximum(a0.copy(), 0.0).astype(np.float64)
    n = len(a)

            if time.monotonic() - t0 > time_limit:

                        best_obj = obj_j
                        best_a = a.copy()
                        improved = True
                        break
                a[j] = old_val
            else:
                # Try scaling
                for frac in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 3.0]:
                    a[j] = old_val * frac

                    if s < 1e-10:

                        best_obj = obj_j

                a[j] = old_val

        if not improved:

            no_improve_count = 0

    w = max(1, n // 8)
    e = np.zeros(n)
    e[:w] = 1.0
    e[-w:] = 1.0
    if e.sum() > 0:
        starts.append(e / e.sum())

    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    starts.append(r / r.sum())
    starts.append(r[::-1] / r.sum())

    # Random
    for _ in range(3):
        r = rng.dirichlet(np.ones(n) * 0.5)
        starts.append(r)

    # Parent (interpolated to length n)
    if parent is not None and parent.size > 0:
        if parent.size == n:
            p = np.maximum(parent, 0).copy()
            p_sum = p.sum()
            if p_sum > 1e-10:
                starts.append(p / p_sum)
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
            p = np.maximum(p, 0)
            p_sum = p.sum()
            if p_sum > 1e-10:
                starts.append(p / p_sum)

    # Chirp
    chirp = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * x / (2.0 * n)))
    chirp = np.maximum(chirp, 0)
    if chirp.sum() > 1e-10:
        starts.append(chirp / chirp.sum())

    # Two-peak
    for ratio in [0.3, 0.4]:
        tp = np.zeros(n)
        tp[int(n * ratio)] = 1.0
        tp[int(n * (1 - ratio))] = 1.0
        tp += 0.1 * np.ones(n) / n
        starts.append(tp / tp.sum())

    return starts


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Uses LP-based direction finding, gradient descent,
    and coordinate refinement to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)

    parent = get_parent()

    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  NEW BEST: score=%.6f, n=%d, time=%.0fs" % (score, len(a), time.monotonic() - t_start))

    # Phase 1: Scan different lengths with quick LP optimization
    print("Phase 1: Quick scan of lengths...")
    scan_lengths = [128, 192, 256, 320, 384, 512, 768]
    scan_results = []
    phase1_budget = min(400.0, (deadline - time.monotonic()) * 0.2)

    for n in scan_lengths:
        if time.monotonic() > deadline - 1600:
            break
        time_per_n = max(5.0, phase1_budget / len(scan_lengths))
        starts = generate_starts(n, rng, parent)
        # Limit number of starts
        max_starts = max(3, min(8, int(time_per_n / 3.0)))
        starts = starts[:max_starts]

        for a0 in starts:
            if time.monotonic() > deadline - 1600:
                break
            t_alloc = max(2.0, min(time_per_n / max_starts, deadline - time.monotonic() - 1600))
            if t_alloc < 1.0:
                break
            a_opt, obj = lp_optimize(a0, time_limit=t_alloc, t_init=0.4)
            score = safe_evaluate(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, a_opt))
            record(a_opt)

    print("  Scan done. Best so far: %.6f" % best_score)
    scan_results.sort(key=lambda x: x[0])

    # Phase 2: Deep LP optimization on top candidates
    print("Phase 2: Deep LP optimization...")
    top_configs = []
    seen_n = set()
    for score, n, a_opt in scan_results:
        if n not in seen_n and len(top_configs) < 8:
            top_configs.append((n, a_opt))
            seen_n.add(n)

    # Always include parent-based
    if parent is not None:
        p = np.maximum(parent, 0).copy()
        p /= p.sum()
        np_parent = len(p)
        if np_parent not in seen_n:
            top_configs.append((np_parent, p))

    if best_sequence is not None:
        top_configs.append((len(best_sequence), best_sequence.copy()))

    phase2_budget = min(1400.0, (deadline - time.monotonic()) * 0.55)
    n_deep = len(top_configs)
    time_per_deep = max(10.0, phase2_budget / max(1, n_deep))

    for idx, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 700:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 700)
        if t_alloc < 5.0:
            break
        print("  [%d/%d] n=%d, budget=%.0fs" % (idx + 1, n_deep, n, t_alloc))

        a_cur = a0.copy()
        a_cur = np.maximum(a_cur, 0.0)
        a_cur /= a_cur.sum()

        # Multiple LP passes with decreasing step
        sub1 = t_alloc * 0.4
        a_cur, _ = lp_optimize(a_cur, time_limit=sub1, t_init=0.5)
        record(a_cur)

        sub2 = t_alloc * 0.25
        a_cur, _ = lp_optimize(a_cur, time_limit=sub2, t_init=0.3)
        record(a_cur)

        # Gradient refinement
        sub3 = t_alloc * 0.2
        a_cur, _ = gradient_descent(a_cur, time_limit=sub3, lr_init=0.03, t_smooth=50.0)
        record(a_cur)

        # LP again
        sub4 = max(1.0, t_alloc - sub1 - sub2 - sub3)
        if sub4 > 1.0:
            a_cur, _ = lp_optimize(a_cur, time_limit=sub4, t_init=0.2)
            record(a_cur)

    print("  Deep LP done. Best: %.6f" % best_score)

    # Phase 3: Coordinate refinement and fine-tuning
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("Phase 3: Coordinate refinement...")
        phase3_budget = min(400.0, deadline - time.monotonic() - 80)
        n_refine = 4
        refine_per = phase3_budget / n_refine

        for i in range(n_refine):
            if time.monotonic() > deadline - 80:
                break
            a_start = best_sequence.copy()
            if i % 2 == 0:
                a_opt, _ = coordinate_refine(a_start, time_limit=refine_per)
            else:
                a_opt, _ = gradient_descent(a_start, time_limit=refine_per * 0.5, lr_init=0.02, t_smooth=100.0)
                if time.monotonic() < deadline - 80:
                    rem = deadline - time.monotonic() - 80
                    if rem > 1.0:
                        a_opt, _ = coordinate_refine(a_opt, time_limit=rem)
            record(a_opt)

        print("  Refinement done. Best: %.6f" % best_score)

    # Phase 4: Try larger n with LP
    if best_sequence is not None and time.monotonic() < deadline - 200:
        print("Phase 4: Trying larger lengths...")
        for n_try in [512, 768, 1024]:
            if time.monotonic() > deadline - 150:
                break
            # Interpolate best to new length
            if len(best_sequence) != n_try:
                a_interp = np.interp(
                    np.linspace(0, 1, n_try),
                    np.linspace(0, 1, len(best_sequence)),
                    best_sequence
                )
                a_interp = np.maximum(a_interp, 0)
                a_interp /= a_interp.sum()
            else:
                a_interp = best_sequence.copy()

            t_alloc = min(150.0, deadline - time.monotonic() - 150)
            if t_alloc < 10.0:
                break
            a_opt, _ = lp_optimize(a_interp, time_limit=t_alloc * 0.6, t_init=0.4)
            record(a_opt)
            if time.monotonic() < deadline - 150:
                rem = deadline - time.monotonic() - 150
                if rem > 5.0:
                    a_opt, _ = coordinate_refine(a_opt, time_limit=rem)
                    record(a_opt)

    # Phase 5: Final polish
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("Phase 5: Final polish...")
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            a_best = best_sequence.copy()
            a_best = np.maximum(a_best, 0.0)
            a_best /= a_best.sum()

            half = remaining * 0.5
            a_opt, _ = gradient_descent(a_best, time_limit=half, lr_init=0.01, t_smooth=200.0)
            record(a_opt)

            if time.monotonic() < deadline - 5:
                rem2 = deadline - time.monotonic() - 5
                if rem2 > 3:
                    a_opt2, _ = coordinate_refine(a_opt, time_limit=rem2)
                    record(a_opt2)

    # Final evaluation
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = safe_evaluate(best_sequence)

    final_score = safe_evaluate(best_sequence)
    print("FINAL SCORE: %.6f" % final_score)
    print("Sequence length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
