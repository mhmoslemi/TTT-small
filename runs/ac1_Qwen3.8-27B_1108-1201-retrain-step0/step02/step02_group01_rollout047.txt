The inner problem (fixing w) is a quadratic program! The objective is a^T Q w a where Q_w = sum_k w[k] P_k. If Q_w is positive definite (or PSD), this is a convex QP.

Is Q_w PSD? Q_w = sum_k w[k] P_k. Each P_k is a (0,1) matrix that's a permutation of a submatrix of the identity. The sum of such matrices with non-negative weights... it's not clear that it's PSD.

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using multi-scale optimization with
soft-max smoothing, Adam optimizer, adaptive temperature, and coordinate
descent polish.

Key improvements over previous:
1. Multi-scale: optimize at small n first, upsample to larger n
2. Adaptive temperature based on peak spread

4. Momentum reset to escape local minima
5. Longer runtime allocation to fine-tuning

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def compute_grad_from_weights(w, a):
    """Compute grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j.
    This is the gradient of sum_k w[k] * c[k] w.r.t. a[j].
    """
    n = len(a)
    # grad[j] = 2 * sum_k w[k] * a[k-j] where 0 <= k-j < n
    # = 2 * (conv(w, a_flipped))[n-1 + j] ... let me think
    # Actually: sum_k w[k] * a[k-j] for 0<=k-j<n
    # Let m = k-j, so k = m+j. sum_m w[m+j] * a[m] for 0<=m<n
    # This is (correlation of w and a)[j] = (conv(w_rev, a))[n-1+j]
    # Hmm, let me just use FFT directly.
    # 
    # Alternative: grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]
    # (where w is defined on [0, 2n-2] and w[x] = 0 for x >= 2n-1)
    #
    # This is a "sliding dot product" of a with w.
    # = (a * w_flipped_full)[n-1+j] where w_flipped_full is w reversed and padded
    #
    # Actually, let me think of it as:
    # grad[j]/2 = sum_m a[m] * w[m+j] = cross-correlation of a and w at lag j
    # = (a * w_rev)[n-1+j] where w_rev is w reversed
    #
    # w has length 2n-1. w_rev has length 2n-1.
    # conv(a, w_rev) has length n + 2n-1 - 1 = 3n-2
    # We want elements [n-1, n-1+n-1] = [n-1, 2n-2]
    
    w_rev = w[::-1]
    full = fft_convolve(a, w_rev)
    # full has length n + (2n-1) - 1 = 3n-2
    # We want full[n-1 : 2n-1] (n elements)
    end = min(2*n-1, len(full))
    return 2.0 * full[n-1:end]


def compute_soft_weights(c, T):
    """Soft-max weights for the convolution peaks."""
    c_max = c.max()
    if c_max < 1e-15:
        return np.ones(len(c)) / len(c)
    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        return np.ones(len(c)) / len(c)
    return exp_scaled / total


def a_from_theta(theta, S):
    """a = S * softmax(theta)"""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def softmax_obj_and_grad(theta, T, S):
    """Compute soft-max objective and gradient w.r.t. theta."""
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    
    shifted = T * (c - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T
    
    w = compute_soft_weights(c, T)
    grad_a = compute_grad_from_weights(w, a)
    
    # Chain rule for softmax parameterization
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    
    return obj, grad_theta, a, c


def adam_optimize(theta_init, S, time_budget, initial_T=5.0, final_T=500.0,
                  lr=0.05, name=""):
    """Adam optimizer with temperature annealing and momentum reset."""
    theta = theta_init.copy().astype(np.float64)
    n = len(theta)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    start = time.monotonic()
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    best_theta = theta.copy()
    
    no_improve_count = 0
    patience = 200
    
    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        
        # Temperature schedule: exponential increase
        T = initial_T * np.exp(np.log(final_T / initial_T) * progress)
        
        a = a_from_theta(theta, S)
        c = fft_convolve_self(a)
        cur = c.max()
        
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()
            best_theta = theta.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Momentum reset to escape local minima
        if no_improve_count > patience:
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve_count = 0
            # Small random perturbation
            theta += 0.01 * np.random.randn(n)
            patience = int(patience * 1.5)
        
        # Compute gradient
        if T < 1e6:
            obj, grad, _, _ = softmax_obj_and_grad(theta, T, S)
        else:
            w = np.zeros(len(c))
            idx = np.argsort(c)[::-1][:16]
            w[idx] = 1.0 / len(idx)
            grad_a = compute_grad_from_weights(w, a)
            dot = np.dot(grad_a, a)
            grad = grad_a * a - (a / S) * dot
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        # Adaptive learning rate: decrease over time
        lr_t = lr * (1.0 - progress * 0.7)
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
    
    return best_theta, best_a, best_obj


def upsample(a_small, n_large):
    """Upsample a sequence using cubic interpolation."""
    from scipy.interpolate import interp1d
    n_small = len(a_small)
    x_small = np.arange(n_small)
    x_large = np.linspace(0, n_small - 1, n_large)
    f = interp1d(x_small, a_small, kind='cubic', fill_value='extrapolate')
    a_large = f(x_large)
    a_large = np.maximum(a_large, 0.0)
    if a_large.sum() < 1e-10:
        a_large = np.ones(n_large)
    return a_large


def coordinate_descent_polish(a, S, time_budget, n_top=200):
    """Coordinate descent on the top-contributing positions."""
    a = a.copy().astype(np.float64)
    n = len(a)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Identify top contributing positions (those with highest a[j] * peak_contribution)
    # The contribution of a[j] to the peak is related to a[j] * a[n-1-j] (center lag)
    # More generally, use the "pressure" from the gradient
    w = np.zeros(len(c))
    peak_idx = np.argsort(c)[::-1][:32]
    w[peak_idx] = 1.0 / 32
    grad_a = compute_grad_from_weights(w, a)
    
    # Sort positions by |gradient| (most pressure first)
    order = np.argsort(np.abs(grad_a))[::-1]
    top_positions = order[:min(n_top, n)]
    
    for sweep in range(100):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in top_positions:
            if time.monotonic() - start > time_budget:
                break
            original = a[i]
            if original < 1e-10:
                continue
            
            # Try several perturbations
            best_val = original
            best_local = best_obj
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
                a[i] = original * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = original
                    break
                a_norm = a * (S / s)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_local:
                    best_local = obj_check
                    best_val = original * factor
            a[i] = best_val
            # Renormalize
            s = a.sum()
            if s > 1e-10:
                a = a * (S / s)
            c = fft_convolve_self(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_from_weights(w, a):
    """Compute grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j in [0,n-1]."""
    n = len(a)
    w_rev = w[::-1]
    out_len = n + len(w) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(w_rev, fsize)
    full = np.real(ifft(A * B))[:out_len]
    end = min(2 * n - 1, out_len)
    return 2.0 * full[n - 1:end]


def compute_soft_weights(c, T):
    """Soft-max weights emphasizing peaks."""
    c_max = c.max()
    if c_max < 1e-15:
        return np.ones(len(c)) / len(c)
    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        return np.ones(len(c)) / len(c)
    return exp_scaled / total


def a_from_theta(theta, S):
    """a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def effective_num_peaks(c, T):
    """Estimate the effective number of active peaks."""
    c_max = c.max()
    if c_max < 1e-15:
        return 1.0
    w = compute_soft_weights(c, T)
    entropy = -np.sum(w * np.log(np.maximum(w, 1e-30)))
    return float(np.exp(entropy))

def upsample_sequence(a_small, n_large):
    """Upsample using cubic interpolation, ensure non-negative."""
    n_small = len(a_small)

    f = interp1d(x_small, a_small, kind='cubic', fill_value='extrapolate')
    a_large = f(x_large)
    a_large = np.maximum(a_large, 0.0)
    if a_large.sum() < 1e-10:

    return a_large
