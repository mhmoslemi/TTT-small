"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
- Normalize to sum = sqrt(2n), reducing objective to max of self-convolution.
- Phase 1: FISTA with adaptive log-sum-exp from diverse initializations.
- Phase 2: Multi-start exploration with restarts for escaping local minima.
- Phase 3: Aggressive polishing with high-beta FISTA, top-k subgradient,
  and active-set equalization.
- Phase 4: Coordinate-style fine-tuning.
- Multiple sequence lengths explored for diversity.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_lse(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a.
    Uses FFT cross-correlation.
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_toppk(a, c, top_k):
    """Gradient using top-k active lags via FFT cross-correlation."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total

    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]

    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_threshold(a, c, threshold_frac=0.99):
    """Gradient using all lags above a threshold fraction of the max."""
    n = len(a)
    c_max = c.max()
    threshold = c_max * threshold_frac
    active = c >= threshold
    if not np.any(active):
        return np.zeros(n)
    w = np.where(active, c, 0.0)
    total = w.sum()
    if total < 1e-30:
        return np.zeros(n)
    w = w / total

    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=2.0, beta_max=500.0,
              beta_rate=0.003, restart_interval=2000, restart_prob=0.3,
              noise_scale=0.01):
    """FISTA with adaptive log-sum-exp, simplex projection, and restarts."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    last_improve_it = 0
    no_improve_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            last_improve_it = it
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count >= restart_interval and rng.random() < restart_prob:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            a = y.copy()
            t = 1.0
            no_improve_count = 0

        grad = fft_grad_lse(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta * 2.0 * (1.0 + 0.0002 * it))
        step = max(step, 1e-7)
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    return best_local, best_score


def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=256,
                       step_decay=0.003, use_threshold=False, threshold_frac=0.995):
    """Nesterov-accelerated top-k/threshold subgradient method."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        if use_threshold:
            grad = fft_grad_threshold(y, c, threshold_frac)
        else:
            grad = fft_grad_toppk(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + step_decay * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 5000 == 0 and rng.random() < 0.1:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0

    return best_local, best_score


def run_active_set_equalize(a_init, n, S, max_time, rng, n_outer=100):
    """
    Active set equalization: iteratively identify active lags and try to
    equalize them using a directional step.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Identify active set (top lags)
        c_max = c.max()
        # Use lags within 1% of max
        active_mask = c >= c_max * 0.99
        active_lags = np.where(active_mask)[0]
        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(c)])

        # Compute gradient from active set
        w = np.zeros(2 * n - 1)
        for k in active_lags:
            w[k] = c[k]
        total = w.sum()
        w = w / total

        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Adaptive step size with backtracking
        step = 0.5 * S / (n * (1.0 + 0.01 * it))
        for _ in range(20):
            candidate = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(candidate)
            if c_new.max() < cur:
                a = candidate
                break
            step *= 0.7
        else:
            # No improvement found, try smaller step
            step *= 0.5
            a = project_simplex(a - step * grad, S)

        # Occasionally add noise to escape local minima
        if it % 50 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, eta_init=0.5, top_k=20):
    """Mirror descent (multiplicative weights) with subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        eta = eta_init / (1.0 + 0.005 * it)
        grad = fft_grad_toppk(a, c, top_k)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_new * (S / s_new)

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a):
    """Compute the actual objective value for a sequence."""
    n = len(a)
    a_clean = np.clip(a, 0.0, 1000.0)
    s = a_clean.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a_clean)
    return 2.0 * n * c.max() / (s * s)

                print("  [ASE %s] obj=%.6f" % (name, obj_ase))
            remaining = deadline - time.monotonic()

    print("=== After Phase 2: global_best=%.6f ===" % global_best)

    # --- Phase 3: Focused polish on the single best ---
    if global_a is not None:
        print("=== Phase 3: Focused Polish ===")
        n_b = len(global_a)
        S_b = np.sqrt(2.0 * float(n_b))
        remaining = deadline - time.monotonic()

        # Iterative high-beta FISTA with decreasing noise
        for round_idx in range(3):
            if time.monotonic() > deadline - 30:
                break
            seg = max(3.0, remaining * 0.25)
            noise = 0.003 / (round_idx + 1)
            a_pol, _ = run_fista(global_a, n_b, S_b, seg, rng,
                                  beta_init=100.0 + 50 * round_idx,
                                  beta_max=2000.0,
                                  beta_rate=0.02,
                                  restart_interval=5000,
                                  restart_prob=0.15,
                                  noise_scale=noise)
            a_pol = safe_normalize(a_pol, n_b)
            obj_pol = compute_objective(a_pol)
            if obj_pol < global_best:
                global_best = obj_pol
                global_a = a_pol.copy()
                print("  [Polish round %d] obj=%.6f" % (round_idx, obj_pol))
            remaining = deadline - time.monotonic()

        # Final top-k with very high k
        if remaining > 5:
            seg = max(2.0, remaining * 0.3)
            a_fin, _ = run_nesterov_toppk(global_a, n_b, S_b, seg, rng,
                                           top_k_init=100, top_k_max=1024,
                                           use_threshold=True, threshold_frac=0.999)
            a_fin = safe_normalize(a_fin, n_b)
            obj_fin = compute_objective(a_fin)
            if obj_fin < global_best:
                global_best = obj_fin
                global_a = a_fin.copy()
                print("  [Final TopK] obj=%.6f" % obj_fin)

    # --- Phase 4: Verification and fallback ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal obj=%.6f" %
          (len(global_a), global_best))
    return [float(x) for x in global_a]
