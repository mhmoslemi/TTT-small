"""
Optimization of non-negative sequences to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS-B on log-parameterized variables with annealed log-sum-exp objective,
combined with targeted peak-shaving refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import next_fast_len


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = next_fast_len(n + m - 1)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def logsumexp_conv(a, tau):
    """
    Compute the log-sum-exp based smooth max of conv(a,a), and its gradient.
    Returns (value, grad_a) where value ≈ tau * log(sum(exp(conv/tau)))
    which approaches max(conv) as tau -> 0.
    """
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    # Clip to avoid overflow
    shifted = np.clip(shifted, -100, 100)
    exp_sum = np.sum(np.exp(shifted))
    value = max_c + tau * np.log(exp_sum)
    
    # Gradient: grad_a[j] = (1/a_scale) * sum_k w[k] * a[k-j] ... 
    # Actually, d(conv[k])/da[j] = a[k-j] (if 0 <= k-j < n)
    # So d(logsumexp)/da[j] = sum_k w[k] * a[k-j] where w[k] = exp((conv[k]-max)/tau) / sum
    w = np.exp(shifted) / exp_sum
    
    # grad[j] = sum_{k: 0<=k-j<n} w[k] * a[k-j]
    # = cross-correlation of w and a, evaluated at specific lags
    # More precisely: grad[j] = sum_k w[k] * a[k-j] (with a out-of-bounds = 0)
    # This is the convolution of w with reversed a, or equivalently:
    # grad = np.convolve(w, a[::-1], mode='same') ... no that's not right either.
    
    # Let me think again. conv[k] = sum_i a[i] * a[k-i] (valid i).
    # d(conv[k])/da[j] = a[k-j] if 0 <= k-j < n, else 0.
    # grad[j] = sum_k w[k] * d(conv[k])/da[j] = sum_k w[k] * a[k-j] (valid)
    # = sum_{m: 0<=m<n} w[j+m] * a[m]  (where m = k-j, so k = j+m)
    # = (w * a)[j] (cross-correlation, or convolution with flipped a)
    
    # Actually: grad[j] = sum_m w[j+m] * a[m] for valid j+m
    # This is the cross-correlation (w ⊛ a)[j] = sum_m w[j+m] * a[m]
    # = sum_m a[m] * w[j+m]
    
    # In terms of convolution: if we define b[m] = a[m], then
    # cross_corr[j] = sum_m b[m] * w[j+m]
    # = (b * flip(w))[n-1-j] ... this is getting complicated.
    
    # Let me just use FFT:
    # grad[j] = sum_m a[m] * w[j+m]
    # This is the cross-correlation of a and w.
    # cross_correlation(a, w)[d] = sum_m a[m+d] * w[m]  ... no.
    
    # Standard definition: (f ⊛ g)[n] = sum_m f*[m] * g[m+n]
    # For real: (f ⊛ g)[n] = sum_m f[m] * g[m+n]
    
    # So grad[j] = sum_m a[m] * w[j+m] = (a ⊛ w)[j] ... 
    # Wait: (a ⊛ w)[j] = sum_m a[m] * w[m+j]. Yes! That's what we want.
    
    # We can compute cross-correlation via FFT:
    # DFT(f ⊛ g) = conj(DFT(f)) * DFT(g)
    
    size = 2 * n
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    grad = np.fft.irfft(np.conj(fa) * fw, size)[:n]
    
    return value, grad, conv


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def optimize_single(n, theta_init, time_limit, rng):
    """
    Optimize using L-BFGS-B with annealed log-sum-exp.
    theta parameterizes a = softmax(theta) * sqrt(2n)
    """
    S_target = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    
    best_score = np.inf
    best_a = None
    t0 = time.monotonic()
    
    # Annealing schedule for tau
    tau_start = 10.0 / n
    tau_end = 0.01 / n
    
    # Multiple phases with decreasing tau
    n_phases = 5
    phase_times = []
    # Allocate more time to later phases (finer tau)
    for i in range(n_phases):
        if i < n_phases - 1:
            phase_times.append(time_limit * 0.15)
        else:
            phase_times.append(time_limit * 0.25)
    
    for phase in range(n_phases):
        if time.monotonic() - t0 >= time_limit:
            break
        
        phase_t0 = time.monotonic()
        phase_time = min(phase_times[phase], time_limit - (phase_t0 - t0))
        
        frac = (phase + 1) / n_phases
        tau = tau_start * (tau_end / tau_start) ** (frac ** 1.5)
        
        def objective(theta_vec):
            a = softmax_normalize(theta_vec) * S_target
            val, _, _ = logsumexp_conv(a, tau)
            # The actual score we care about
            # score = 2n * max_conv / (sum a)^2 = 2n * max_conv / (2n) = max_conv (when sum = sqrt(2n))
            # With logsumexp approximation: score ≈ 2n * logsumexp_val / (2n) = logsumexp_val
            # Wait: score = 2n * max_conv / (sum a)^2. If sum a = S_target = sqrt(2n),
            # then (sum a)^2 = 2n, so score = max_conv.
            # And logsumexp_conv approximates max_conv.
            return val
        
        def gradient(theta_vec):
            a = softmax_normalize(theta_vec) * S_target
            val, grad_a, _ = logsumexp_conv(a, tau)
            # d(val)/d(theta_j) = S_target * p_j * (grad_a - dot(p, grad_a))
            # where p = softmax(theta) = a / S_target
            p = a / S_target
            inner = np.dot(p, grad_a)
            grad_theta = S_target * p * (grad_a - inner)
            return grad_theta
        
        # Run L-BFGS-B
        result = minimize(
            objective, theta, method='L-BFGS-B',
            jac=gradient, options={
                'maxiter': 500,
                'maxfun': 10000,
                'ftol': 1e-12,
                'gtol': 1e-8,
                'maxls': 50,
            }
        )
        
        theta = result.x
        a_final = softmax_normalize(theta) * S_target
        sc = score_fast(a_final)
        
        if sc < best_score:
            best_score = sc
            best_a = a_final.copy()
        
        elapsed = time.monotonic() - t0
        print(f"        Phase {phase+1}/{n_phases}: tau={tau:.2e}, "
              f"score={sc:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_score


def peak_shave(a_init, time_limit, rng):
    """
    Targeted peak-shaving: iteratively identify the max lag and reduce
    contributions to it while redistributing mass.
    """
    a = a_init.copy()
    n = len(a)
    S = np.sum(a)
    a = a / S  # Normalize to sum = 1
    best_a = a.copy()
    best_max_conv = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        current_max = conv[peak_idx]
        
        if current_max < best_max_conv:
            best_max_conv = current_max
            best_a = a.copy()
        
        # Decompose the peak: conv[peak_idx] = sum_i a[i]*a[peak_idx - i]
        # Find the top contributors
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        # Reduce top contributors, redistribute to random positions
        k_top = max(3, n // 16)
        top_idx = np.argsort(contribs)[-k_top:]
        
        # Shrink top contributors
        shrink = rng.uniform(0.02, 0.08)
        total_reduced = 0
        for idx in top_idx:
            amt = a[idx] * shrink
            a[idx] -= amt
            total_reduced += amt
        
        # Redistribute to random positions (avoiding positions that would increase the peak)
        others = np.setdiff1d(np.arange(n), top_idx)
        n_add = min(len(others), max(3, n // 32))
        add_idx = rng.choice(others, size=n_add, replace=False)
        
        # For each add position, check if it would increase the peak
        # conv[peak_idx] += a[add_idx_new] * a[peak_idx - add_idx] + old_contribution
        # We just add proportionally
        a[add_idx] += total_reduced / n_add
        
        a = np.clip(a, 0, None)
        S_new = np.sum(a)
        if S_new < 1e-15:
            break
        a = a / S_new
    
    return best_a * np.sqrt(2.0 * n), 2.0 * n * best_max_conv


def make_geometric_init(n, ratio, rng):
    """Create a geometric sequence initialization."""
    a = ratio ** np.arange(n, dtype=np.float64)
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def make_ramp_init(n, power, rng):
    """Create a power-law ramp initialization."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = (1.0 - t) ** power
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Create a bell-shaped (Gaussian) initialization."""
    t = np.arange(n, dtype=np.float64)
    center = n * 0.3  # Off-center to break symmetry
    a = np.exp(-((t - center) / width) ** 2)
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def make_inverse_flat_init(n, rng):
    """
    Create initialization by inverting a target flat spectrum.
    We want DFT(a)^2 to be flat, so DFT(a) ≈ sqrt(2n-1) * exp(i*phase).
    For a >= 0, the best we can do is minimize variance of |DFT(a)|^2.
    Start with inverse of a slightly perturbed flat spectrum.
    """
    size = next_fast_len(2 * n - 1)
    # Target: |A(f)|^2 as flat as possible
    # Start with a and iterate: compute conv, subtract from flat, inverse FFT, project
    a = np.ones(n) / n
    for _ in range(50):
        conv = fft_convolve(a, a)
        target = np.sum(conv) / (2 * n - 1)  # uniform target
        error = conv - target
        # Go back to frequency domain
        A = np.fft.rfft(a, size)
        # Gradient of max_conv w.r.t. A (approximately): 
        # We want to reduce the peak, so subtract a fraction of the error spectrum
        E = np.fft.rfft(error, size)
        A -= 0.1 * E / (np.max(np.abs(A)) + 1e-10)
        a_new = np.fft.irfft(A, size)[:n]
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        if np.allclose(a, a_new, atol=1e-8):
            break
        a = a_new
    return a


def upsample_sequence(a, n_new):
    """Upsample a sequence to a new length using interpolation."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.clip(a_new, 0.001, None)
    return a_new / np.sum(a_new)
