def spectral_flat_init(n, rng):
    """
    Initialize toward flat autocorrelation:
    Compute a_ideal = IDFT(sqrt(FFT(ones(2n-1)))) and take its non-negative part.
    """
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    # DFT of a ones vector of length 2n-1, zero-padded to N
    ones = np.zeros(N)
    ones[:2 * n - 1] = 1.0
    F_ones = np.fft.fft(ones)
    # a_ideal has FFT = sqrt(F_ones) (since F_ones is real and >= 0 for symmetric ones)
    # Take sqrt of the magnitude (handle any tiny negative parts)
    a_ideal = np.fft.ifft(np.sqrt(np.maximum(F_ones, 0))).real[:n]
    a = np.clip(a_ideal, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    # Add small noise for diversity
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 1e-6, None)
    a /= np.sum(a)
    return a


def ramp_init(n, style, rng):
    """Generate various structured initializations."""
    t = np.linspace(0, 1, n)

def parent_init(n, parent_seq):
    """Resample parent sequence to length n."""
    x_old = np.linspace(0, 1, len(parent_seq))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, np.asarray(parent_seq, dtype=np.float64))
    a = np.clip(a, 1e-4, None)
    a /= np.sum(a)
    return a

                    'ftol': 1e-12,
                    'gtol': 1e-8,
                    'iprint': -1,
                }
            )
            theta = result.x
        except Exception:
            pass

        # Evaluate actual (non-smooth) score
        a = softmax_fn(theta)
        actual_score = 2.0 * n * np.max(fft_convolve(a, a))
        if actual_score < best_score:
            best_score = actual_score
            best_theta = theta.copy()

        elapsed = time.monotonic() - t0
        print(f"        Stage {stage+1}/{len(tau_schedule)} tau={tau:.4f}: "
              f"score={actual_score:.6f} best={best_score:.6f} t={elapsed:.1f}s")

        if time.monotonic() - t0 > time_limit - 10:
            break

    return best_theta, best_score


def coordinate_polish(a_in, time_limit, rng, n_sweeps_max=200):
    """
    Coordinate descent polish: targeted peak shaving.
    Works on simplex (sum a = 1, a >= 0).
    """
    a = a_in.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-15:
        return a, np.inf
    a /= s
    n = len(a)

    conv = fft_convolve(a, a)
    current_score = 2.0 * n * np.max(conv)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    patience = 0
    max_patience = 300

    while time.monotonic() - t0 < time_limit and patience < max_patience:
        conv = fft_convolve(a, a)
        max_val = np.max(conv)
        # Find top peaks
        peak_threshold = max_val * 0.95
        peak_indices = np.where(conv >= peak_threshold)[0]

        if len(peak_indices) == 0:
            break

        # Pick the worst peak
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]

        # Find contributing pairs
        # conv[peak_idx] = sum_i a[i]*a[peak_idx - i]
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        if lo > hi:
            patience += 1
            continue

        contribs = a[lo:hi + 1] * a[peak_idx - hi:peak_idx - lo + 1][::-1]
        # The pair with max product is the most harmful
        max_contrib_pos = int(np.argmax(contribs))
        i = lo + max_contrib_pos
        j = peak_idx - i

        # Try to reduce the larger of a[i], a[j]
        if i == j:
            # Square term: reduce a[i]
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                cand = a.copy()
                removed = cand[i] * frac
                cand[i] -= removed
                if cand[i] < 0:
                    continue
                # Distribute to random non-peaking elements
                others = np.where(conv < peak_threshold)[0]
                if len(others) == 0:
                    others = np.arange(n)
                n_add = min(5, len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                cand[add_idx] += removed / n_add
                new_score = 2.0 * n * np.max(fft_convolve(cand, cand))
                if new_score < current_score:
                    a = cand
                    current_score = new_score
                    patience = 0
                    break
        else:
            # Reduce the larger element
            if a[i] >= a[j]:
                reduce_idx = i
                other_idx = j
            else:
                reduce_idx = j
                other_idx = i
            for frac in [0.01, 0.02, 0.05, 0.1, 0.15]:
                cand = a.copy()
                removed = cand[reduce_idx] * frac
                cand[reduce_idx] -= removed
                if cand[reduce_idx] < 0:
                    continue
                # Add to elements that don't contribute to peaks
                # Find elements that are NOT in any peak's contributing set
                safe = np.ones(n, dtype=bool)
                for p in peak_indices:
                    p_lo = max(0, p - n + 1)
                    p_hi = min(n - 1, p)
                    safe[p_lo:p_hi + 1] = False
                    safe[p - p_hi:p - p_lo + 1] = False
                safe_indices = np.where(safe)[0]
                if len(safe_indices) == 0:
                    safe_indices = np.arange(n)
                n_add = min(3, len(safe_indices))
                add_idx = rng.choice(safe_indices, size=n_add, replace=False)
                cand[add_idx] += removed / n_add
                new_score = 2.0 * n * np.max(fft_convolve(cand, cand))
                if new_score < current_score:
                    a = cand
                    current_score = new_score
                    patience = 0
                    break
        else:
            patience += 1

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

    return best_a, best_score


def random_perturbation_polish(a_in, time_limit, rng):
    """
    Random perturbation local search with acceptance.
    Mix of: pair transfers, block scaling, peak-targeted moves.
    """
    a = a_in.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    a /= s
    n = len(a)
    current_score = 2.0 * n * np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    patience = 0
    max_patience = 2000

    while time.monotonic() - t0 < time_limit and patience < max_patience:
        r = rng.random()
        if r < 0.35:
            # Pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.02, 0.12)
            amt = a[i] * frac
            cand = a.copy()
            cand[i] -= amt
            cand[j] += amt
            cand = np.clip(cand, 0.0, None)
            cs = np.sum(cand)
            if cs < 1e-15:
                continue
            cand /= cs
        elif r < 0.55:
            # Single element scale
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.08))
            cand = a.copy()
            cand[j] *= factor
            cand = np.clip(cand, 0.0, None)
            cs = np.sum(cand)
            if cs < 1e-15:
                continue
            cand /= cs
        elif r < 0.75:
            # Block scaling
            width = max(2, int(n * 0.02))
            start = int(rng.integers(0, n - width))
            factors = np.exp(rng.normal(0, 0.04, size=width))
            cand = a.copy()
            cand[start:start + width] *= factors
            cand = np.clip(cand, 0.0, None)
            cs = np.sum(cand)
            if cs < 1e-15:
                continue
            cand /= cs
        else:
            # Peak-targeted: reduce top contributors to worst peak
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            if lo > hi:
                patience += 1
                continue
            contribs = a[lo:hi + 1] * a[peak_idx - hi:peak_idx - lo + 1][::-1]
            top_k = max(2, min(10, len(contribs) // 4))
            top_local = np.argsort(contribs)[-top_k:]
            top_global = lo + top_local
            cand = a.copy()
            reductions = rng.uniform(0.02, 0.08, size=len(top_global))
            total_removed = 0.0
            for idx, red in zip(top_global, reductions):
                rem = cand[idx] * red
                cand[idx] -= rem
                total_removed += rem
            # Redistribute to safe positions
            safe = np.where(conv < np.max(conv) * 0.9)[0]
            if len(safe) == 0:
                safe = np.arange(n)
            n_add = min(5, len(safe))
            add_idx = rng.choice(safe, size=n_add, replace=False)
            cand[add_idx] += total_removed / n_add
            cand = np.clip(cand, 0.0, None)
            cs = np.sum(cand)
            if cs < 1e-15:
                continue
            cand /= cs

        new_score = 2.0 * n * np.max(fft_convolve(cand, cand))
        if new_score < current_score:
            a = cand
            current_score = new_score
            patience = 0
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        else:
            patience += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for optimal non-negative sequence minimizing
    2n * max(conv(a,a)) / (sum(a))^2.
    Uses spectral init, L-BFGS with LSE smoothing, and coordinate polish.
    """
    t_global = time.monotonic()
    deadline = t_global + max(10.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Get parent sequence if available
    parent_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        parent_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.001, 1000.0)
        ps = score_fast(parent_seq)
        print(f"Parent sequence: n={len(parent_seq)}, score={ps:.6f}")
        if ps < best_score:
            best_score = ps
            best_seq = parent_seq.copy()

    # Define search configurations: (n, list of init types)
    n_candidates = [512, 768, 1024, 1536, 2048, 3072]
    init_types = ['spectral', 'ramp_linear', 'ramp_quad', 'ramp_sin',
                  'ramp_cos', 'ramp_asym', 'parent', 'uniform']

    def make_init(n, init_type):
        if init_type == 'spectral':
            return spectral_flat_init(n, rng)
        elif init_type == 'ramp_linear':
            return ramp_init(n, 'linear', rng)
        elif init_type == 'ramp_quad':
            return ramp_init(n, 'quad', rng)
        elif init_type == 'ramp_sin':
            return ramp_init(n, 'sin', rng)
        elif init_type == 'ramp_cos':
            return ramp_init(n, 'cos', rng)
        elif init_type == 'ramp_asym':
            return ramp_init(n, 'asym', rng)
        elif init_type == 'parent':
            if parent_seq is not None:
                return parent_init(n, parent_seq)
            else:
                return ramp_init(n, 'linear', rng)
        elif init_type == 'uniform':
            a = np.ones(n) + 0.05 * rng.standard_normal(n)
            a = np.clip(a, 1e-4, None)
            return a / np.sum(a)
        else:
            return np.ones(n) / n

    # Allocate time: more time for smaller n (faster convergence), less for large n
    # Weight allocation
    time_weights = {512: 1.5, 768: 1.2, 1024: 1.0, 1536: 0.8, 2048: 0.6, 3072: 0.4}
    total_weight = sum(time_weights.values())
    time_budget = deadline - t_global - 240  # reserve 240s for final polish

    # Phase 1: L-BFGS optimization for each n
    print("\n=== Phase 1: L-BFGS optimization ===")
    for n in n_candidates:
        if time.monotonic() > deadline - 240:
            break
        n_time = time_budget * time_weights[n] / total_weight
        n_time = max(30.0, min(300.0, n_time))

        # Choose 1-2 initializations for this n
        n_inits = 2 if n <= 1024 else 1
        per_init_time = n_time / n_inits

        for init_i in range(n_inits):
            if time.monotonic() > deadline - 240:
                break
            init_type = init_types[(init_i + n // 128) % len(init_types)]
            a0 = make_init(n, init_type)
            theta_init = np.log(a0 + 1e-12)

            # Temperature schedule: start smooth, end sharp
            # tau in units of typical conv value
            # Typical conv value ~ 1/n for sum(a)=1
            tau_start = 10.0 / n
            tau_mid = 2.0 / n
            tau_end = 0.3 / n

            # Use 3 stages
            tau_schedule = [tau_start, tau_mid, tau_end]
            if per_init_time > 100:
                tau_schedule = [tau_start, tau_start * 0.5, tau_mid, tau_mid * 0.5, tau_end]

            max_iter = int(max(100, 300 * per_init_time / 60))

            theta_best, lbfgs_score = optimize_lbfgs(
                n, theta_init, tau_schedule, per_init_time, max_iters_per_stage=max_iter
            )

            a_best = softmax_fn(theta_best)
            actual_score = 2.0 * n * np.max(fft_convolve(a_best, a_best))
            print(f"  n={n}, init='{init_type}': score={actual_score:.6f}, "
                  f"t={time.monotonic()-t_global:.1f}s")

            if actual_score < best_score:
                best_score = actual_score
                best_seq = (a_best * np.sqrt(2.0 * n)).copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Phase 2: Coordinate descent polish on best sequence
    print(f"\n=== Phase 2: Coordinate descent polish ===")
    print(f"  Best before polish: {best_score:.6f}")
    if best_seq is not None:
        polish_time = min(180.0, max(30.0, deadline - time.monotonic() - 60))
        print(f"  Polish time: {polish_time:.0f}s")
        n_best = len(best_seq)
        a_norm = best_seq / np.sum(best_seq)

        # Coordinate polish
        polished, pol_score = coordinate_polish(a_norm, polish_time * 0.5, rng)
        scaled = polished * np.sqrt(2.0 * len(polished))
        pol_score_actual = score_fast(scaled)
        print(f"  After coordinate polish: {pol_score_actual:.6f}")
        if pol_score_actual < best_score:
            best_score = pol_score_actual
            best_seq = scaled.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 3: Random perturbation polish
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\n=== Phase 3: Random perturbation polish ===")
        rtime = min(120.0, deadline - time.monotonic() - 60)
        print(f"  Time: {rtime:.0f}s")
        a_norm = best_seq / np.sum(best_seq)
        polished2, pol2_score = random_perturbation_polish(a_norm, rtime, rng)
        scaled2 = polished2 * np.sqrt(2.0 * len(polished2))
        pol2_actual = score_fast(scaled2)
        print(f"  After random polish: {pol2_actual:.6f}")
        if pol2_actual < best_score:
            best_score = pol2_actual
            best_seq = scaled2.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 4: Quick L-BFGS restart from best with different inits if time allows
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\n=== Phase 4: Final restart attempts ===")
        remaining = deadline - time.monotonic() - 60
        n_best = len(best_seq)
        # Try a slightly perturbed version
        a_norm = best_seq / np.sum(best_seq)
        perturbed = a_norm * np.exp(rng.normal(0, 0.1, n_best))
        perturbed = np.clip(perturbed, 1e-8, None)
        perturbed /= np.sum(perturbed)
        theta = np.log(perturbed + 1e-12)
        tau = 0.5 / n_best
        try:
            result = scipy.optimize.minimize(
                lambda th: lse_score_and_grad(th, n_best, tau),
                theta, method='L-BFGS-B', jac=True,
                options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-8, 'iprint': -1}
            )
            a_final = softmax_fn(result.x)
            sc = 2.0 * n_best * np.max(fft_convolve(a_final, a_final))
            print(f"  Final restart: score={sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = (a_final * np.sqrt(2.0 * n_best)).copy()
                print(f"    *** New best: {best_score:.6f} ***")
        except Exception as e:
            print(f"  Final restart failed: {e}")

    # Final verification and output
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Ensure sum is reasonable
        s = sum(final_list)
        if s < 0.01:
            # Scale up
            final_list = [x * 10 for x in final_list]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    # Fallback
    if parent_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in parent_seq]
    return [1.0] * 256
