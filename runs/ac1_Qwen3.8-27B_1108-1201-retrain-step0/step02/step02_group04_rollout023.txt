    n = len(a)

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses LP-based direction finding + log-sum-exp gradient descent + subgradient polishing.
Key insight: flatten the auto-convolution; theoretical minimum approaches 1 as n grows.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.signal import fftconvolve


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fftconvolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def build_c_matrix(a):
    """Build convolution constraint matrix: C[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            C[k, js] = a[k - js]
    return C


def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)<=elementwise M bound."""
    n = len(a)
    b = cp.Variable(n)
    obj = cp.Maximize(cp.sum(b))
    constraints = [b >= 0]
    C = build_c_matrix(a)
    constraints.append(C @ b <= M)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=200000, verbose=False)
        if b.value is not None and np.all(np.isfinite(b.value)):
            return np.clip(b.value, 0, None)
    except Exception:
        pass
    try:
        prob.solve(solver=cp.SCS, max_iters=100000, verbose=False)
        if b.value is not None and np.all(np.isfinite(b.value)):
            return np.clip(b.value, 0, None)
    except Exception:
        pass
    return None


def lse_gradient(a, temp=0.05):
    """Gradient of log-sum-exp(conv(a,a)/temp)."""
    n = len(a)
    c = fftconvolve(a, a)
    max_c = np.max(c)
    c_shifted = (c - max_c) / temp
    exp_c = np.exp(c_shifted)
    wsum = np.sum(exp_c)
    if wsum < 1e-30:
        return np.zeros(n)
    weights = exp_c / wsum

    g = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            g[js] += weights[k] * 2.0 * a[k - js]

    return g / temp


def subgradient_polish(a, time_limit, rng):
    """Polish using subgradient descent on max(conv(a,a))."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, local_score(a)
    a = a / s
    best = a.copy()
    best_score = local_score(a)
    lr = 0.15
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fftconvolve(a, a)
        max_val = np.max(c)

        k_top = min(20, len(c))
        top_k = np.argsort(c)[-k_top:]

        g = np.zeros(n)
        for ki in top_k:
            j_min = max(0, ki - n + 1)
            j_max = min(n - 1, ki)
            if j_min <= j_max:
                js = np.arange(j_min, j_max + 1)
                g[js] += 2.0 * a[ki - js]

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = lr * factor
            a_new = a - step * g / g_norm
            a_new = np.clip(a_new, 0, None)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc = local_score(a_new)
            if sc < best_score:
                a = a_new
                best = a_new.copy()
                best_score = sc
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            lr *= 0.95
            if stall > 50:
                break

    return best, best_score


def lse_gradient_descent(a, time_limit, rng):
    """Gradient descent on softened max via log-sum-exp."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, local_score(a)
    a = a / s
    best = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    temp = 0.1

    while time.monotonic() - t0 < time_limit:
        g = lse_gradient(a, temp)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        improved = False
        for step in [0.3, 0.2, 0.1, 0.05, 0.02]:
            a_new = a - step * g
            a_new = np.clip(a_new, 0, None)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc = local_score(a_new)
            if sc < best_score:
                a = a_new
                best = a_new.copy()
                best_score = sc
                improved = True
                break
        if not improved:
            temp *= 0.7
            if temp < 0.001:
                break

    return best, best_score


def lp_optimize(a0, time_limit):
    """LP-based iterative optimization."""
    n = len(a0)
    a = a0.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, local_score(a)
    a = a / s
    best = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()

    it = 0
    while time.monotonic() - t0 < time_limit:
        c = fftconvolve(a, a)
        M = np.max(c)
        current_score = 2.0 * n * M / np.sum(a) ** 2

        if current_score < best_score:
            best_score = current_score
            best = a.copy()

        b = solve_lp_direction(a, M)
        if b is None or np.sum(b) < 1e-10:
            break

        b = b / np.sum(b)

        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_new = (1 - t) * a + t * b
            a_new = np.clip(a_new, 0, None)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc = local_score(a_new)
            if sc < current_score:
                a = a_new
                current_score = sc
                improved = True
                if sc < best_score:
                    best_score = sc
                    best = a.copy()
                break

        if not improved:
            break

        it += 1
        if it % 5 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, M={M:.6f}")

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30
    rng = np.random.default_rng(seed)

    best_score = np.inf

    def try_update(a):

        if a is None:

            return
        sc = local_score(a)

    # Check provided sequence

    # Try uniform and random inits for various n
    for n in [256, 512, 1024]:
        a_uni = np.ones(n)
        print(f"Uniform n={n}: score={local_score(a_uni):.8f}")
        a_rand = rng.exponential(1.0, size=n)

        start_points.append((f"rand_norm_{n}", np.abs(rng.standard_normal(n)) + 0.1))
        start_points.append((f"bump_{n}", np.sin(np.linspace(0, np.pi, n)) + 0.5))

    # Phase 1: LP-based optimization
    print("\n=== Phase 1: LP-based optimization ===")
    for name, a0 in start_points:
        if time.monotonic() > deadline:
            break
        a0 = np.clip(a0, 0, 1000)
        s0 = np.sum(a0)
        if s0 < 0.01:
            continue
        a0 = a0 / s0
        n = len(a0)
        print(f"\n  [{name}] n={n}, initial score={local_score(a0):.8f}")

        time_left = deadline - time.monotonic()
        if time_left < 10:
            break

        # Allocate time: more iterations for smaller n
        if n <= 256:
            lp_time = min(120, time_left * 0.4)

        a0 = a0 / s0

        # Use the best result from LP phase if available
        if best_a is not None and len(best_a) == len(a0):
