    """Compute linear convolution using FFT."""

    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:

    """

    # But a appears on both sides, so dc[k]/da[j] = a[k-j] (from first) + a[j-k] (from second, if j-k valid)
    # Hmm no. c[k] = sum_i a[i] * a[k-i]. Taking derivative wrt a[j]:
    # The term where i=j gives a[k-j]. The term where k-i=j, i.e., i=k-j, gives a[k-j].
    # But these are the same term when i=j and i=k-j, i.e., j = k-j, i.e., k=2j.
    # Actually no: c[k] = sum_i a[i] * a[k-i]. The derivative wrt a[j] is:
    # sum_i [delta_{i,j} * a[k-i] + a[i] * delta_{k-i,j}]
    # = a[k-j] + a[k-j] (if both terms are valid, which they are since k-i=j means i=k-j)
    # Wait: the second sum is over i where k-i = j, i.e., i = k-j.
    # So dc[k]/da[j] = a[k-j] (from i=j) + a[k-j] (from i=k-j) = 2*a[k-j] (if 0<=k-j<n)
    # UNLESS k-j = j (i.e., k=2j), in which case it's just a[j] (one term).
    # Actually no, even when k=2j, both terms give a[j], so it's 2*a[j].
    # Wait, let me be more careful.
    # c[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * a[k-i]
    # dc[k]/da[j] = sum_i [d(a[i])/da[j] * a[k-i] + a[i] * d(a[k-i])/da[j]]
    # = a[k-j] (if j is in range) + a[j*(k-j)] ... 
    # First term: i=j, gives a[k-j] (valid if 0<=k-j<n)
    # Second term: k-i=j, i.e., i=k-j, gives a[k-j] (valid if 0<=k-j<n)
    # These are the same value a[k-j], so total = 2*a[k-j] if 0<=k-j<n
    # (When k-j=j, both terms are a[j], still giving 2*a[j])
    
    # So d(value)/da[j] = 2n * sum_k w[k] * 2 * a[k-j] * I(0<=k-j<n)
    # = 4n * sum_k w[k] * a[k-j] * I(0<=k-j<n)
    # = 4n * (cross_correlation of w and a)[j]
    
    grad_a = 4.0 * n * grad_from_weights_fft(w, a)
    
    # Convert to theta gradient: a = softmax(theta)
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return value, grad_theta


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def make_asymmetric_init(n, asym, rng):
    """Asymmetric initialization - large at one edge, small at other."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 + asym * (2.0 * t - 1.0)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_u_shaped_init(n, depth, rng):
    """U-shaped: large at edges, small in middle."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 + depth * np.cos(np.pi * t)
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def fast_local_search(a_init, time_limit, rng, tau=0.005):
    """
    Fast gradient-informed local search.
    Uses the soft-max gradient to propose moves that target peaks.
    Much faster than random perturbation approach.
    """
    a = a_init.astype(np.float64).copy()
    a = a / np.sum(a)
    n = len(a)
    current_score = seq_score(a)
    best = a.copy()
    best_score = current_score

    t_start = time.monotonic()
    temperature = 0.001
    cooling = 0.99995
    base_lr = 0.05

    iters = 0
    last_report = t_start

    while time.monotonic() - t_start < time_limit:
        iters += 1

        # Compute gradient direction
        c = fft_conv(a, a)
        sm_val, w = soft_max_weights(c, tau)

        # Gradient of 2n * soft_max
        grad = 4.0 * n * grad_from_weights_fft(w, a)

        # Propose move: move against the gradient (to reduce the objective)
        # a_new = a * exp(-lr * normalized_grad)
        # This respects non-negativity automatically
        
        grad_norm = np.max(np.abs(grad))
        if grad_norm < 1e-15:
            break
        grad_normed = grad / grad_norm

        # Create move
        lr = base_lr * (0.5 + 0.5 * rng.random())
        # Add some noise to escape local minima
        noise_scale = 0.02 * temperature
        move = -lr * grad_normed + noise_scale * rng.standard_normal(n)

        candidate = a * np.exp(move)
        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)

        new_score = seq_score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            a = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 1e-10)

        # Occasionally try a different kind of move
        if iters % 50 == 0:
            r = rng.random()
            if r < 0.3:
                # Peak-targeted reduction
                c = fft_conv(a, a)
                peak_idx = int(np.argmax(c))
                # Find top contributors to the peak
                contribs = np.zeros(n)
                for i in range(n):
                    k = peak_idx - i
                    if 0 <= k < n:
                        contribs[i] = a[i] * a[k]
                top_k = max(5, n // 8)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate = a.copy()
                candidate[top_idx] *= np.exp(-0.05 * rng.random(top_k))
                # Boost non-peak contributors
                other = np.setdiff1d(np.arange(n), top_idx)
                if len(other) > 0:
                    n_boost = min(15, len(other))
                    boost_idx = rng.choice(other, size=n_boost, replace=False)
                    candidate[boost_idx] *= np.exp(0.05 * rng.random(n_boost))
                candidate = np.maximum(candidate, 1e-12)
                candidate = candidate / np.sum(candidate)
                new_score = seq_score(candidate)
                if new_score < current_score:
                    a = candidate
                    current_score = new_score
                    if current_score < best_score:
                        best_score = current_score
                        best = a.copy()

        if time.monotonic() - last_report > 5.0:
            last_report = time.monotonic()
            elapsed = time.monotonic() - t_start
            print(f"    local_search: score={current_score:.6f}, best={best_score:.6f}, "
                  f"iter={iters}, t={elapsed:.0f}s")

    return best, best_score


def run_lbgfs(a0, tau_schedule, maxiter_per_tau, deadline):
    """Run L-BFGS with annealed soft-max temperature."""
    n = len(a0)
    theta = np.log(a0 + 1e-30)
    best_theta = theta.copy()
    best_score = seq_score(a0)

    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break

        result = minimize(
            soft_max_obj_grad_theta, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(tau,),
            options={'maxiter': maxiter_per_tau, 'ftol': 1e-15,
                     'gtol': 1e-12, 'maxfun': maxiter_per_tau * 10}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = seq_score(a_new)

        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        theta = best_theta

    return softmax(best_theta), best_score


def multi_resolution_optimize(a_target, n_target, deadline, rng, tau_schedule=[0.01, 0.005, 0.002, 0.001]):
    """
    Multi-resolution zoom-in: optimize at n/4, upsample, optimize at n/2, upsample, optimize at n.
    """
    resolutions = []
    # Build resolution ladder
    res = n_target
    while res > 16:
        resolutions.append(res)
        res = max(16, res // 2)
    resolutions.append(max(16, n_target // 4))
    resolutions = sorted(set(resolutions), key=lambda x: x)
    # Remove duplicates and ensure n_target is last
    if resolutions[-1] != n_target:
        resolutions.append(n_target)

    current = None
    for i, n_res in enumerate(resolutions):
        if time.monotonic() > deadline:
            break

        if current is None:
            # Interpolate from target or create new
            if n_res == n_target:
                current = a_target.copy()
            else:
                current = a_target[:n_res].copy()
                current = current / np.sum(current)
        else:
            # Upsample from previous resolution
            old_n = len(current)
            if n_res == old_n:
                continue
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n_res)
            current = np.interp(x_new, x_old, current)
            current = np.maximum(current, 1e-8)
            current = current / np.sum(current)

        # Add small perturbation for diversity
        if i > 0:
            current = current * np.exp(rng.normal(0, 0.05, n_res))
            current = np.maximum(current, 1e-8)
            current = current / np.sum(current)

        # Optimize at this resolution
        sub_deadline = min(deadline, time.monotonic() + max(20, (deadline - time.monotonic()) * (1.0 / (len(resolutions) - i))))
        a_opt, s = run_lbgfs(current, tau_schedule, maxiter_per_tau=100, deadline=sub_deadline)
        current = a_opt
        print(f"    multi_res: n={n_res}, score={s:.6f}")

    return current, seq_score(current)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimum peak autocorrelation sequence.
    Phase 1: L-BFGS soft-max on multiple lengths and initializations
    Phase 2: Multi-resolution zoom-in on best candidate
    Phase 3: Fast gradient-informed local search
    Phase 4: Final L-BFGS polish
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        best_score = seq_score(start_seq)
        best_seq = start_seq.copy()
        print(f"Starting sequence: n={len(best_seq)}, score={best_score:.6f}")

    # Phase 1: L-BFGS optimization on multiple lengths
    print("\n=== Phase 1: L-BFGS soft-max optimization ===")
    lengths = [128, 192, 256, 320, 384]

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break

        # Time allocation
        remaining = deadline - time.monotonic()
        if n == 256 or (best_seq is not None and n == len(best_seq)):
            time_alloc = min(400.0, remaining * 0.30)
        else:
            time_alloc = min(200.0, remaining * 0.15)

        sub_deadline = time.monotonic() + time_alloc

        # Generate initializations
        inits = []

        # Interpolate from best
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.maximum(a_interp, 1e-6)
            a_interp = a_interp / np.sum(a_interp)
            inits.append(('interp_best', a_interp))

        # Diverse inits
        inits.append(('chirp', make_chirp_init(n, 0.3 + 0.4 * rng.random(), rng)))
        inits.append(('random', make_random_init(n, rng, spread=0.2 + 0.4 * rng.random())))
        inits.append(('bell', make_bell_init(n, n * (0.1 + 0.2 * rng.random()), rng)))
        inits.append(('asym', make_asymmetric_init(n, 0.2 + 0.3 * rng.random(), rng)))
        inits.append(('u_shape', make_u_shaped_init(n, 0.2 + 0.3 * rng.random(), rng)))

        # Perturbed best
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n)
            a_pert = np.interp(x_new, x_old, best_seq)
            a_pert = a_pert * np.exp(rng.normal(0, 0.15, n))
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert = a_pert / np.sum(a_pert)
            inits.append(('pert_best', a_pert))

        tau_schedule = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

        for init_name, a0 in inits:
            if time.monotonic() > sub_deadline:
                break

            a_opt, s = run_lbgfs(a0, tau_schedule, maxiter_per_tau=150, deadline=sub_deadline)

            if s < best_score:
                best_score = s
                best_seq = a_opt.copy()
                print(f"  n={n}, init='{init_name}': score={s:.6f} *** NEW BEST ***")
            else:
                print(f"  n={n}, init='{init_name}': score={s:.6f}")

        print(f"  Phase 1 best after n={n}: {best_score:.6f} (n={len(best_seq)})")

    # Phase 2: Multi-resolution zoom-in
    if best_seq is not None and time.monotonic() < deadline - 200:
        print(f"\n=== Phase 2: Multi-resolution zoom-in ===")
        n_best = len(best_seq)
        mr_deadline = min(time.monotonic() + 300, deadline - 150)
        a_mr, s_mr = multi_resolution_optimize(best_seq, n_best, mr_deadline, rng)
        if s_mr < best_score:
            best_score = s_mr
            best_seq = a_mr.copy()
            print(f"  Multi-res improved: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Multi-res best: {s_mr:.6f}")

    # Phase 3: Fast gradient-informed local search
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Phase 3: Fast local search ===")
        polish_time = min(400.0, max(60.0, deadline - time.monotonic() - 30))
        print(f"  Polish time: {polish_time:.0f}s")

        # Run local search with different tau values
        for tau_ls in [0.01, 0.003, 0.001]:
            if time.monotonic() > deadline - 30:
                break
            a_pol, s_pol = fast_local_search(best_seq, min(polish_time / 3, deadline - time.monotonic() - 30), rng, tau=tau_ls)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = a_pol.copy()
                print(f"  Local search (tau={tau_ls}): {best_score:.6f} *** NEW BEST ***")
            else:
                print(f"  Local search (tau={tau_ls}): {s_pol:.6f}")

    # Phase 4: Final L-BFGS polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Phase 4: Final L-BFGS polish ===")
        final_tau = [0.001, 0.0005, 0.0002, 0.0001]
        a_fin, s_fin = run_lbgfs(best_seq, final_tau, maxiter_per_tau=300, deadline=deadline - 5)
        if s_fin < best_score:
            best_score = s_fin
            best_seq = a_fin.copy()
            print(f"  Final polish: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Final polish: {s_fin:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if best_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return [1.0] * 256
