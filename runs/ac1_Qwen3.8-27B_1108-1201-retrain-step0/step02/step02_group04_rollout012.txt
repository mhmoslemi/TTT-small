"""
Optimize non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses multiplicative gradient descent (mirror descent on simplex) with a
softmax-smoothed max objective, combined with LP-based perturbations and
multiple restarts from diverse initializations.
Key insight: the previous program used an incorrect gradient (for sum(a^2)
rather than max(conv(a,a))). This implementation uses the correct gradient
of the smoothed maximum of the full convolution.
"""

import numpy as np
import time
from scipy.optimize import linprog


def conv_full(a):
    """Full linear convolution of a with itself."""
    return np.convolve(a, a)


def fast_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2 using FFT for speed."""
    a = np.maximum(a, 0.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    # FFT-based convolution
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    return float(2.0 * n * np.max(c) / (s * s))


def compute_gradient(a, beta):
    """
    Compute gradient of the softmax-smoothed max of conv(a,a) w.r.t. a.
    grad[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    where w = softmax(beta * (c - max(c)))
    """
    c = conv_full(a)
    n = len(a)
    c_shift = c - np.max(c)
    w = np.exp(beta * c_shift)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # Build the n x n matrix W where W[i,j] = w[i+j]
    # Then grad = W @ a
    idx = np.arange(n)
    W = w[idx[:, None] + idx[None, :]]
    grad = W @ a
    return grad, c, np.max(c)


def mul_update(a, grad, lr):
    """Multiplicative (mirror descent) update maintaining the simplex."""
    g = grad - grad.mean()
    log_step = -lr * g
    log_step = np.clip(log_step, -50.0, 50.0)
    a_new = a * np.exp(log_step)
    s = a_new.sum()
    if s < 1e-12:
        a_new = a.copy()
    else:
        a_new = a_new / s
    return a_new

        a_new = mul_update(a, grad, lr)

def lp_perturb(a, time_budget, rng):
    """
    Use LP to find a better direction.
    Solve: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0.
    Then mix current a with the LP solution.
    """
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    attempt = 0

    while time.monotonic() - t0 < time_budget:
        attempt += 1
        c = conv_full(a)
        M = np.max(c) * (1.0 + 0.1 * rng.random())

        # Build constraint matrix: A[k,j] = a[k-j] if 0 <= k-j < n
        # A is (2n-1) x n
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1) + 1
            if j_start < j_end:
                A[k, j_start:j_end] = a[k - j_end + 1:k - j_start + 1][::-1]

        # Solve LP: minimize -sum(b) s.t. A @ b <= M*e, b >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': 30})
            if res.success and res.x is not None:
                b = np.maximum(res.x, 0)
                b = b / b.sum()
                # Mix

                    a_mix = (1 - t_mix) * a + t_mix * b
                    a_mix = np.maximum(a_mix, 1e-10)
                    a_mix = a_mix / a_mix.sum()
                    s = fast_score(a_mix)
                    if s < best_score:
                        best_score = s
                        best_a = a_mix.copy()
                a = best_a.copy()
        except Exception:
            pass

    return best_a, best_score

        a = 1.0 + 0.5 * np.cos(2 * np.pi * alpha * i)
        a = np.maximum(a, 1e-6)
        inits[f'quasi_{n}'] = a

        # Random smooth (Gaussian process-like)
        for k in range(2):
            a = np.abs(rng.standard_normal(n)) + 0.2
            # Smooth it
            kernel = np.exp(-np.arange(-5, 6)**2 / 4.0)
            kernel /= kernel.sum()
            a = np.convolve(a, kernel, mode='same')
            inits[f'smooth_{n}_{k}'] = a

        # Two-scale pattern
        for period in [8, 16, 32]:
            if period < n:
                base = np.ones(n)
                for j in range(n):
                    base[j] = 1.0 + 0.4 * np.cos(2 * np.pi * j / period)
                base = np.maximum(base, 1e-6)
                inits[f'twoscale_{n}_{period}'] = base

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Find the optimal non-negative sequence minimizing the evaluation score.
    Uses multiplicative gradient descent with correct smoothed-max gradient,
    LP-based perturbations, and multiple diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 60.0  # Reserve 60s for final eval
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent is not None and len(parent) > 0:
            parent = np.maximum(parent, 0)
            ps = fast_score(parent)
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            if ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except (NameError, TypeError, ValueError):
        pass

    # Generate initializations
    lengths = [256, 512, 1024]
    inits = generate_initializations(rng, lengths)
    print(f"Generated {len(inits)} initializations")

    # First pass: quick evaluation of all inits
    for name, a0 in inits.items():
        a0 = np.maximum(a0, 1e-6)
        s = fast_score(a0)
        if s < best_score:
            best_score = s
            best_seq = a0.copy()
            print(f"  Better init found: {name}, n={len(a0)}, score={s:.6f}")

    print(f"After init scan: best={best_score:.8f}")

    # Allocate remaining time for optimization
    elapsed = time.monotonic() - t_start
    remaining = deadline - time.monotonic()
    print(f"Remaining time for optimization: {remaining:.1f}s")

    # Sort inits by initial score (try best first)
    scored_inits = []
    for name, a0 in inits.items():
        a0 = np.maximum(a0, 1e-6)
        s = fast_score(a0)
        scored_inits.append((s, name, a0))
    scored_inits.sort(key=lambda x: x[0])

    # Phase 1: Optimize top inits with limited time each
    top_k = min(15, len(scored_inits))
    time_per_phase1 = max(10.0, remaining / (top_k + 5))

    for rank in range(top_k):
        if time.monotonic() > deadline:
            break
        s_init, name, a0 = scored_inits[rank]
        n = len(a0)
        a0 = a0 / a0.sum()

        # Skip if already very good
        if s_init < best_score * 0.9:
            continue

        time_left = deadline - time.monotonic()
        budget = min(time_per_phase1, time_left * 0.5)
        if budget < 5:
            break

        print(f"\n  Phase 1 [{rank+1}/{top_k}]: {name}, n={n}, init_score={s_init:.6f}, budget={budget:.0f}s")
        a_opt, score_opt = optimize_mirror(a0, budget, rng,
                                            beta_start=30.0, beta_end=200.0, lr_start=0.3)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"    ** NEW BEST: {score_opt:.8f}")

    # Phase 2: Focus on the best sequence found, with longer budget
    if best_seq is not None and time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        print(f"\n=== Phase 2: Refine best (score={best_score:.8f}), budget={remaining:.0f}s ===")

        # Try multiple refinement passes
        a_cur = best_seq.copy()
        a_cur = a_cur / a_cur.sum()

        split = remaining / 4.0
        for p in range(4):
            if time.monotonic() > deadline:
                break
            # Vary hyperparameters
            b0 = 50.0 + p * 30
            b1 = 200.0 + p * 100
            lr0 = 0.3 / (1 + 0.3 * p)
            print(f"  Pass {p+1}: beta=[{b0},{b1}], lr={lr0:.4f}")
            a_opt, score_opt = optimize_mirror(a_cur, min(split, deadline - time.monotonic()),
                                                rng, beta_start=b0, beta_end=b1, lr_start=lr0)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"    ** NEW BEST: {score_opt:.8f}")
            a_cur = a_opt.copy()

    # Phase 3: Try LP perturbations on the best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"\n=== Phase 3: LP perturbations, budget={remaining:.0f}s ===")
        a_opt, score_opt = lp_perturb(best_seq, remaining, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: {score_opt:.8f}")

    # Phase 4: One more round of diverse restarts if time remains
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"\n=== Phase 4: Diverse restarts, budget={remaining:.0f}s ===")
        for k in range(5):
            if time.monotonic() > deadline:
                break
            # Perturb the best sequence significantly
            a0 = best_seq.astype(np.float64).copy()
            n = len(a0)
            # Random subset replacement
            n_replace = max(1, n // 3)
            idx = rng.choice(n, size=n_replace, replace=False)
            a0[idx] = np.abs(rng.standard_normal(n_replace)) + 0.05
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            s0 = fast_score(a0)
            budget = min(remaining / (5 - k) * 0.8, 60.0)
            if budget < 5:
                break
            print(f"  Restart {k+1}: init_score={s0:.6f}, budget={budget:.0f}s")
            a_opt, score_opt = optimize_mirror(a0, budget, rng,
                                                beta_start=50.0, beta_end=300.0, lr_start=0.4)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"    ** NEW BEST: {score_opt:.8f}")

    # Final answer
    if best_seq is None:
        best_seq = np.ones(256)

    # Ensure non-negative and valid
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    final_list = [float(x) for x in best_seq]
    final_fast = fast_score(best_seq)
    print(f"\n=== FINAL: n={len(final_list)}, fast_score={final_fast:.8f} ===")

    # Verify with actual evaluator
    eval_score = evaluate_sequence(final_list)
    print(f"  evaluate_sequence confirms: {eval_score:.8f}")

    return final_list
