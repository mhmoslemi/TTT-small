import numpy as np
import time

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b) if b is not a else n
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (S * S), np.max(c), c

def make_ramp(n, p=1.0, side='right'):
    """Create a power-law ramp."""
    t = np.arange(1, n+1, dtype=np.float64)
    if side == 'right':
        a = t**p
    else:
        a = t**p[::-1]
    # Normalize to have sum sqrt(2n)
    a = a * np.sqrt(2.0 * n) / np.sum(a)
    return a
