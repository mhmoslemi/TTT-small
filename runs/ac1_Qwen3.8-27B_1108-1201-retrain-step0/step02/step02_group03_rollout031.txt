   - Keep the DC component (mean)
   - Reduce the AC components
   - Or: set `b_flattened = b * (1 - α * (b - mean(b)) / max(|b - mean(b)|))` for some α
6. Compute `B_flattened = FFT(b_flattened)`
7. We need `B_flattened = A' * A'` for some `A'`. So set `A' = sqrt(B_flattened)` (complex square root)
8. `a' = IFFT(A').real`
9. Clip to non-negative: `a' = max(0, a')`
10. Go to step 2

This is a variant of the GS algorithm adapted for our problem. The complex square root in step 7 is the key non-trivial step.

Let me think about whether this will work well. The issue is that `B_flattened` might not be a valid "squared spectrum" in the sense that there might not exist a signal `A'` such that `A'^2 = B_flattened`. But we can always take the complex square root, and the result will be a valid signal (by the properties of the DFT).

The main concern is that after clipping to non-negative, we lose information and the iteration might not converge well. But with a few iterations, it should give a decent starting point.

"""
Multi-phase optimization for minimizing peak self-convolution of non-negative
sequences. Uses spectral flattening initialization, adaptive gradient descent
(variance → soft-max), coarse-to-fine upsampling, and targeted local search.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def spectral_init(n, rng, n_iter=30, alpha=0.7):
    """
    GS-style spectral flattening initialization.
    Tries to make conv(a,a) as flat as possible.
    """
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    # Target: flat convolution of length 2n-1
    target = np.zeros(N)
    target[:2*n-1] = 1.0
    
    # Target spectrum
    C_target = np.fft.rfft(target)
    
    # Initialize
    a = np.ones(n)
    a = a + rng.standard_normal(n) * 0.1
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    
    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        B = A * A  # spectrum of the (circular) convolution
        
        b = np.fft.irfft(B, N)[:2*n-1]
        
        # Flatten b: reduce deviations from mean
        b_mean = np.mean(b)
        b_ac = b - b_mean
        # Scale down the AC component
        b_flattened = b_mean + alpha * b_ac
        
        # Make sure b_flattened corresponds to a valid squared spectrum
        B_flat_spec = np.fft.rfft(b_flattened, N)
        # Take complex square root
        # B_flat_spec should be A^2, so A = sqrt(B_flat_spec)
        # Handle the phase carefully
        phase = np.angle(B_flat_spec)
        mag = np.abs(B_flat_spec)
        A_new = np.sqrt(mag) * np.exp(0.5j * phase)
        
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new / s
        a = a_new
    
    return a


def chirp_init(n, rng, alpha=0.5):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def multi_tone_init(n, rng, n_tones=4):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(n_tones):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.4 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def upsample(a, new_n, rng):
    """Upsample a sequence to a longer length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.02 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)


def grad_soft_max_conv(a, tau, N=None):
    """Compute soft-max(conv(a,a)) and its gradient w.r.t. a."""
    n = len(a)

        N = 1
        while N < 2 * n - 1:
            N <<= 1
    c = np.fft.irfft(np.fft.rfft(a, N) ** 2, N)[:2*n-1]
    
    c_max = np.max(c)

    e = np.exp(shifted)

    w = e / s  # weights for soft-max

    # d/d a_i conv(a,a)[k] = a[k-i] (when valid)

    # = (a ⋆ w)[i] = conv(a_rev, w)[i + n - 1]

    # grad[i] = sum_k w[k] * a[k-i]
    # This is the cross-correlation of a and w at lag i
    # = (a * w_rev)[i] where * denotes convolution... 
    # Let me think more carefully.

    # Let j = k-i: f(i) = sum_j w[j+i] * a[j]
    # This is the cross-correlation of w and a, evaluated at lag i
    # = sum_j a[j] * w[j+i]

    # In terms of convolution: (a ⋆ w)[i] = conv(a, w_rev)[i + len(w) - 1]...

    # grad = 2 * (w ⋆ a) truncated to length n

    # (w ⋆ a)[i] = sum_j w[j+i] * a[j]

    # cross_corr = ifft(fft(w, size) * conj(fft(a, size)))
    # But we need the right alignment.
    
    # Simpler: just compute it directly using the convolution
    # sum_k w[k] * a[k-i] for i = 0, ..., n-1
    # = reverse of conv(w, a_rev) shifted
    # conv(w, a_rev)[k] = sum_j w[j] * a_rev[k-j] = sum_j w[j] * a[n-1-k+j]

    A = np.fft.rfft(a, size)

    # grad[i] = sum_k w[k] * a[k-i]
    # = sum_k w[k] * a_rev[n-1-k+i]

    # Let v = [a, 0, 0, ..., 0] (length 2n-1, a in first n positions)

    # So we can use 2 * sum_k w[k] * a[k-i] as an approximation.
    
    # Actually, it IS exactly 2 * a[k-i] for the gradient of sum a[j]*a[k-j] with respect to a[i],

    # If j=i: term is a[k-i]
    # If k-j=i, i.e., j=k-i: term is a[k-i]
    # These are the same value, and they're the same term only when i = k-i, i.e., k=2i.
    # So the derivative is a[k-i] + a[k-i] = 2*a[k-i] when k != 2i, and a[k-i] when k=2i.

def grad_variance_conv(a, N=None):
    """Compute variance of conv(a,a) and its gradient."""
    n = len(a)
    if N is None:
        N = 1
        while N < 2 * n - 1:
            N <<= 1
    c = np.fft.irfft(np.fft.rfft(a, N) ** 2, N)[:2*n-1]
    
    c_mean = np.mean(c)
    variance = np.mean((c - c_mean) ** 2)
    
    # Gradient of variance:
    # d/d a_i [mean((c - c_mean)^2)] 
    # = (2/(2n-1)) * sum_k (c[k] - c_mean) * d c[k]/d a[i] - (2/(2n-1)) * c_mean * sum_k d c[k]/d a[i]
    # sum_k d c[k]/d a[i] = sum_k 2*a[k-i] = 2*sum(a) = 2*S (independent of i)
    # So: grad[i] = (2/(2n-1)) * [sum_k (c[k]-c_mean) * 2*a[k-i] - c_mean * 2*S]
    # = (4/(2n-1)) * [sum_k (c[k]-c_mean)*a[k-i] - c_mean*S]
    
    c_ac = c - c_mean  # zero-mean convolution
    S = np.sum(a)
    
    # sum_k c_ac[k] * a[k-i] = cross-correlation of c_ac and a at lag i
    W_full = np.fft.fft(c_ac, N)
    A_full = np.fft.fft(a, N)
    cross = np.fft.ifft(np.conj(W_full) * A_full).real[:n]
    
    grad = (4.0 / (2*n-1)) * (cross - c_mean * S)
    
    return variance, grad, c


def optimize_gradient(n, a_init, time_limit, rng, 
                      tau_start_mult=30.0, tau_end_mult=0.2,
                      use_var_phase=True, var_fraction=0.3):
    """
    Adam gradient descent with adaptive objective:
    - First phase: minimize variance of convolution
    - Second phase: minimize soft-max of convolution with annealed temperature
    """
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S
    theta = np.log(a / S + 1e-12)
    
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr0 = 2.0
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    tau_start = tau_start_mult / n
    tau_end = tau_end_mult / n
    
    var_phase_end = var_fraction * time_limit
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        step += 1
        
        a = np.exp(theta - np.max(theta))
        a = a / np.sum(a) * S
        
        if use_var_phase and elapsed < var_phase_end:
            value, g_a, c = grad_variance_conv(a, N)
        else:
            elapsed_frac = min(1.0, max(0.0, (elapsed - var_phase_end) / max(1, time_limit - var_phase_end)))
            tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)
            value, g_a, c = grad_soft_max_conv(a, tau, N)
        
        # Chain rule: g_theta = a * (g_a - <a, g_a> / S) / S
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S) / S
        
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        if step % 200 == 0:
            a_check = np.exp(theta - np.max(theta))
            a_check = a_check / np.sum(a_check) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
            if step % 1000 == 0:
                print(f"        step={step}, score={score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")
    
    a_final = np.exp(best_theta - np.max(best_theta))
    a_final = a_final / np.sum(a_final) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
    return a_final, best_score


def local_search(a_init, time_limit, rng, top_k=8):
    """
    Simulated annealing with convolution-peak-directed perturbations.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    # Adaptive temperature
    c = fft_convolve(current, current)
    c_std = np.std(c)
    temp = 0.002 * c_std
    cooling = 0.9997
    iter_count = 0
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        iter_count += 1
        
        if r < 0.3:
            # Peak-directed: find top convolution lags and reduce contributing elements
            c = np.fft.irfft(np.fft.rfft(current, N) ** 2, N)[:2*n-1]
            n_top = min(top_k, len(c))
            top_lags = np.argsort(c)[-n_top:]
            
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            
            n_reduce = max(2, n // 12)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.02, size=len(bot_idx)))
        elif r < 0.5:
            # Block shift: reduce one block, increase adjacent
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - 2 * width))
            candidate[start:start + width] *= np.exp(rng.normal(-0.07, 0.02, size=width))
            candidate[start + width:start + 2 * width] *= np.exp(rng.normal(0.07, 0.02, size=width))
        elif r < 0.65:
            # Multi-block random
            n_blocks = int(rng.integers(2, 6))
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 12))))
                bs = int(rng.integers(0, n - bw))

"""
Multi-phase optimization for minimizing peak self-convolution of non-negative
sequences. Combines spectral flattening initialization, adaptive gradient descent
(variance then soft-max with annealed temperature), coarse-to-fine upsampling,
and targeted local search with convolution-peak-directed mutations.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def get_fft_size(n, for_cross_corr=False):
    """Get appropriate FFT size."""
    if for_cross_corr:
        need = 3 * n - 2
    else:
        need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    return size


def spectral_init(n, rng, n_iter=30, alpha=0.7):
    """
    GS-style spectral flattening initialization.
    Iteratively modifies the spectrum to flatten the self-convolution.
    """
    N = get_fft_size(n, for_cross_corr=False)
    
    # Target: flat convolution
    target = np.zeros(N)
    target[:2 * n - 1] = 1.0
    C_target = np.fft.rfft(target)
    
    # Initialize
    a = 1.0 + 0.1 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    
    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        B = A * A  # spectrum of circular convolution
        
        b = np.fft.irfft(B, N)[:2 * n - 1]
        
        # Flatten: reduce AC component
        b_mean = np.mean(b)
        b_ac = b - b_mean
        b_flattened = b_mean + alpha * b_ac
        
        # Convert back to spectrum
        B_flat = np.fft.rfft(b_flattened, N)
        
        # Complex square root to get A from B = A^2
        phase = np.angle(B_flat)
        mag = np.abs(B_flat)
        A_new = np.sqrt(np.maximum(mag, 1e-15)) * np.exp(0.5j * phase)
        
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
    
    return a


def chirp_init(n, rng, alpha=0.5):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def multi_tone_init(n, rng, n_tones=4):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(n_tones):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.4 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def random_init(n, rng, spread=0.3):
    """Random initialization with Gaussian perturbation."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def upsample(a, new_n, rng):
    """Upsample a sequence to a longer length via interpolation + noise."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.015 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)


def compute_conv(a, N):
    """Compute circular convolution (length 2n-1 part) using pre-computed FFT size."""
    n = len(a)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    return c


def grad_soft_max_conv(a, tau, N):
    """Compute soft-max(conv(a,a)) and gradient w.r.t. a."""
    n = len(a)
    c = compute_conv(a, N)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    value = c_max + tau * np.log(s)
    w = e / s
    
    # Gradient: 2 * cross-correlation of w and a
    # (w *shift* a)[i] = sum_j w[j+i] * a[j]
    N2 = get_fft_size(n, for_cross_corr=True)
    W_full = np.fft.fft(w, N2)
    A_full = np.fft.fft(a, N2)
    cross = np.fft.ifft(np.conj(W_full) * A_full).real[:n]
    
    grad = 2.0 * cross
    return value, grad, c


def grad_variance_conv(a, N):
    """Compute variance of conv(a,a) and its gradient."""
    n = len(a)
    c = compute_conv(a, N)
    
    c_mean = np.mean(c)
    c_ac = c - c_mean
    variance = np.mean(c_ac ** 2)
    
    # Gradient: (4/(2n-1)) * (c_ac cross-correlated with a)
    N2 = get_fft_size(n, for_cross_corr=True)
    W_full = np.fft.fft(c_ac, N2)
    A_full = np.fft.fft(a, N2)
    cross = np.fft.ifft(np.conj(W_full) * A_full).real[:n]
    
    grad = (4.0 / (2 * n - 1)) * cross
    return variance, grad, c


def optimize_gradient(n, a_init, time_limit, rng,
                      tau_start_mult=25.0, tau_end_mult=0.15,
                      use_var_phase=True, var_fraction=0.25,
                      lr0=2.0):
    """
    Adam gradient descent with adaptive objective:
    - Early phase: minimize variance of convolution (smooth)
    - Late phase: minimize soft-max with annealed temperature (sharp)
    Normalizes to sum = sqrt(2n).
    """
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S
    theta = np.log(a / S + 1e-12)
    
    N = get_fft_size(n, for_cross_corr=False)
    
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)
    
    tau_start = tau_start_mult / n
    tau_end = tau_end_mult / n
    var_phase_end = var_fraction * time_limit if use_var_phase else 0.0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        step += 1
        
        # Normalize a from theta
        a = np.exp(theta - np.max(theta))
        a = a / np.sum(a) * S
        
        if use_var_phase and elapsed < var_phase_end:
            value, g_a, c = grad_variance_conv(a, N)
        else:
            if elapsed > var_phase_end:
                frac = min(1.0, (elapsed - var_phase_end) / max(1e-6, time_limit - var_phase_end))
            else:
                frac = 0.0
            tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
            value, g_a, c = grad_soft_max_conv(a, tau, N)
        
        # Chain rule for softmax parameterization
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S) / S
        
        # Adam update
        m1 = b1 * m1 + (1.0 - b1) * g_theta
        m2 = b2 * m2 + (1.0 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Periodic evaluation
        if step % 150 == 0:
            a_check = np.exp(theta - np.max(theta))
            a_check = a_check / np.sum(a_check) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
            if step % 1000 == 0:
                print(f"        step={step}, score={score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")
    
    a_final = np.exp(best_theta - np.max(best_theta))
    a_final = a_final / np.sum(a_final) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
    return a_final, best_score

    """
    Simulated annealing with convolution-peak-directed perturbations.

    current = a_init.astype(np.float64).copy()

    cooling = 0.99975
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:

        if r < 0.30:
            # Peak-directed reduction
            c = compute_conv(current, N)
            n_top = min(top_k, len(c))
            top_lags = np.argsort(c)[-n_top:]
            
            contrib = np.zeros(n)
            for lag in top_lags:

                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):

            n_reduce = max(2, n // 12)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.015, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.015, size=len(bot_idx)))
        elif r < 0.50:

            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - 2 * width))
