"""
Optimization of step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-space mirror descent, coordinate descent, and adapted annealing.
Explores multiple sequence lengths with diverse initializations.
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_value(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def conv_and_obj(a):
    """Return both convolution and objective."""
    n = len(a)
    conv = fast_convolve(a)
    obj = 2.0 * n * conv.max()
    return conv, obj

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_weights(conv, temperature):
    """Compute softmax weights for smoothing the max."""
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    return w

def smoothed_subgradient(conv, a, temperature, thresh_frac=0.005):
    """Compute smoothed subgradient of max(conv) w.r.t. a."""
    n = len(a)
    w = softmax_weights(conv, temperature)

Step function optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-space mirror descent, peak-targeted coordinate descent,
and informed simulated annealing. Explores multiple sequence lengths
with diverse initializations, focusing on asymmetric structures.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

    n = len(a)
    conv = fast_convolve(a)
    obj = 2.0 * n * conv.max()
    return conv, obj


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad(conv, a, tau):
    """Compute smoothed gradient of max(conv) w.r.t. a using softmax."""
    n = len(a)
    s = conv / tau
    s -= s.max()
    w = np.exp(s)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    thresh = w.max() * 0.01
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            partner = a[k - je:k - js + 1][::-1]
            sg[js:je + 1] += w[k] * 2.0 * partner
    return sg


def mirror_descent(a0, time_limit, tau_init=10.0, lr_init=0.5, patience_max=80):
    """Log-space mirror descent on simplex."""
    a = np.maximum(a0, 1e-12).copy()
    n = len(a)
    a /= a.sum()
    x = np.log(a)
    conv, cur_obj = conv_and_obj(a)
    best_obj = cur_obj
    best_a = a.copy()
    tau = tau_init
    lr = lr_init
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        sg = smoothed_grad(conv, a, tau)
        sg_c = sg - sg.mean()
        x = x - lr * sg_c
        x_shifted = x - x.max()
        a_new = np.exp(x_shifted)
        a_new /= a_new.sum()
        conv, new_obj = conv_and_obj(a_new)

        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new

        if patience >= patience_max:
            patience = 0
            lr *= 0.65
            tau *= 1.4
            if lr < 0.05:
                break

    return best_a, best_obj


def pgd_simplex(a0, time_limit, tau_init=10.0, lr_init=0.1):
    """Projected gradient descent on simplex."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    a /= a.sum()
    conv, cur_obj = conv_and_obj(a)
    best_obj = cur_obj
    best_a = a.copy()
    tau = tau_init
    lr = lr_init
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        sg = smoothed_grad(conv, a, tau)
        a_new = a - lr * sg
        a_new = project_simplex(a_new)
        conv, new_obj = conv_and_obj(a_new)

        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new

        if patience >= 60:
            patience = 0
            lr *= 0.7
            tau *= 1.3
            if lr < 0.005:
                break

    return best_a, best_obj


def coord_descent_peak(a0, time_limit, seed=42):
    """Peak-targeted coordinate descent: transfer mass from peak contributors to safe positions."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    a /= a.sum()
    conv, cur_obj = conv_and_obj(a)
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        k_star = int(np.argmax(conv))
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        if js > je:
            break

        contributions = a[js:je + 1] * a[k_star - je:k_star - js + 1][::-1]
        if contributions.max() < 1e-15:
            break

        # Find top contributors to the peak
        order = np.argsort(contributions)[::-1]
        top_k = min(15, len(order))
        top_sources = order[:top_k]

        # Compute "safety" score for each target position
        # A target t is safe if a[k_star - t] is small (won't increase the peak much)
        partner_for_peak = np.zeros(n)
        for j in range(n):
            pk = k_star - j
            if 0 <= pk < n:
                partner_for_peak[j] = a[pk]

        improved = False
        for _ in range(top_k):
            src = js + int(top_sources[rng.integers(top_k)])
            if a[src] < 1e-12:
                continue

            # Choose target: prefer positions with small partner_for_peak
            # Sample from low-partner positions
            sorted_targets = np.argsort(partner_for_peak)
            top_safe = sorted_targets[:min(50, n)]
            tgt = int(top_safe[rng.integers(len(top_safe))])
            if tgt == src:
                tgt = int(sorted_targets[len(sorted_targets) // 2])
                if tgt == src:
                    continue

            # Try different fractions
            for frac in [0.3, 0.15, 0.08, 0.5]:
                delta = a[src] * frac
                if delta < 1e-13:
                    continue
                a[src] -= delta
                a[tgt] += delta
                new_conv = fast_convolve(a)
                new_obj = 2.0 * n * new_conv.max()

                if new_obj < cur_obj - 1e-12:
                    cur_obj = new_obj
                    conv = new_conv
                    improved = True
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                    break
                else:
                    a[src] += delta
                    a[tgt] -= delta

        if not improved:
            stall_count += 1
            if stall_count > 50:
                break
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                delta = a[i] * 0.2
                a[i] -= delta
                a[j] += delta
                new_conv = fast_convolve(a)
                new_obj = 2.0 * n * new_conv.max()
                if new_obj < best_obj:
                    cur_obj = new_obj
                    conv = new_conv
                    best_obj = new_obj
                    best_a = a.copy()
                    stall_count = 0
                else:
                    a[i] += delta
                    a[j] -= delta
        else:
            stall_count = 0

    return best_a, best_obj


def anneal_peak(a0, time_limit, seed=0):
    """Informed simulated annealing with peak-targeted moves."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    a /= a.sum()
    conv, cur_obj = conv_and_obj(a)
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.02 * (1.0 - frac) ** 2 + 0.0005

        # Get top peaks
        top_peaks = np.argsort(conv)[::-1][:16]
        k_star = int(top_peaks[rng.integers(min(16, len(top_peaks)))])

        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        if js > je:
            k_star = int(np.argmax(conv))
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            if js > je:
                break

        contributions = a[js:je + 1] * a[k_star - je:k_star - js + 1][::-1]
        total = contributions.sum()
        if total < 1e-15:
            break

        # Source: proportional to contribution
        probs = contributions / total
        src = js + int(rng.choice(je - js + 1, p=probs))

        # Target: prefer positions with low contribution to all top peaks
        best_tgt = int(rng.integers(n))
        best_score = np.inf
        for _ in range(8):
            cand = int(rng.integers(n))
            if cand == src:
                continue
            score = 0.0
            for pk in top_peaks[:8]:
                d1 = pk - cand
                if 0 <= d1 < n:
                    score += a[d1]
                d2 = cand - (2 * n - 2 - pk)
                if 0 <= d2 < n:
                    score += a[d2]
            if score < best_score:
                best_score = score
                best_tgt = cand

        tgt = best_tgt
        if tgt == src:
            continue

        if a[src] < 1e-12:
            continue

        delta = a[src] * rng.uniform(0.05, 0.3)
        a[src] -= delta
        a[tgt] += delta

        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj

        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
                no_improve += 1
            else:
                a[src] += delta
                a[tgt] -= delta
                no_improve += 1

        if no_improve > max_no_improve:
            max_no_improve = int(max_no_improve * 0.7)
            if max_no_improve < 50:
                break

    return best_a, best_obj


def anneal_swap(a0, time_limit, seed=0):
    """Simulated annealing with swap-style moves."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    a /= a.sum()
    conv, cur_obj = conv_and_obj(a)
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.03 * (1.0 - frac) ** 1.5 + 0.001

        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or a[i] < 1e-10:
            continue

        # Transfer move
        delta = a[i] * rng.uniform(0.02, 0.2)
        a[i] -= delta
        a[j] += delta

        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj

        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[i] += delta
                a[j] -= delta
            no_improve += 1

        if no_improve > 300:
            break

    return best_a, best_obj


def resample_sequence(parent, new_n):
    """Resample a sequence to a new length via interpolation."""
    old_n = len(parent)
    if old_n == new_n:
        return parent.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    resampled = np.interp(x_new, x_old, parent)
    resampled = np.maximum(resampled, 0.0)
    s = resampled.sum()
    if s > 1e-10:
        resampled /= s
    else:
        resampled = np.ones(new_n) / new_n
    return resampled


def gen_init_ramp(n, reverse=False):
    """Linearly increasing (or decreasing) initialization."""
    x = np.arange(1, n + 1, dtype=np.float64)
    if reverse:
        x = x[::-1]
    x /= x.sum()
    return x


def gen_init_gaussian(n, center_frac):
    """Gaussian-shaped initialization."""
    x = np.arange(n, dtype=np.float64)
    center = center_frac * n
    sigma = n * 0.15
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    s = g.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return g / s


def gen_init_bimodal(n, pos1, pos2):
    """Two-bump initialization."""
    x = np.arange(n, dtype=np.float64)
    g1 = np.exp(-0.5 * ((x - pos1 * n) / (n * 0.08)) ** 2)
    g2 = np.exp(-0.5 * ((x - pos2 * n) / (n * 0.08)) ** 2)
    g = g1 + g2
    s = g.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return g / s


def gen_init_random(n, rng):
    """Random Dirichlet initialization."""
    alpha = rng.choice([0.5, 1.0, 2.0])
    return rng.dirichlet(np.ones(n) * alpha)


def gen_init_chirp(n):
    """Chirp-like initialization (asymmetric)."""
    x = np.arange(n, dtype=np.float64)
    phase = 2.0 * np.pi * x * x / (2.0 * n)
    c = 0.5 * (1.0 + np.cos(phase))
    c = np.maximum(c, 0.0)
    s = c.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return c / s


def safe_evaluate(a):
    """Safely call evaluate_sequence."""
    try:
        a_clean = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a_clean = np.minimum(a_clean, 1000.0)
        s = a_clean.sum()
        if s < 0.01:
            if s > 1e-15:
                a_clean = a_clean * (1.0 / s)
            else:
                return float('inf')
        if not np.all(np.isfinite(a_clean)):
            return float('inf')
        return float(evaluate_sequence(a_clean.tolist()))
    except Exception:
        return float('inf')


def get_parent():
    """Try to access parent sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None
