```python
import numpy as np
import time

def compute_score_fast(p):
    """Compute the score for a probability vector p (sum=1)."""
    n = len(p)
    c = np.convolve(p, p)
    return 2.0 * n * np.max(c)

def smooth_max_gradient(p, tau):
    """Compute the gradient of the smooth max of conv(p,p)."""
    n = len(p)
    c = np.convolve(p, p)  # length 2n-1
    
    # Softmax weights
    c_shifted = c - c.max()
    w = np.exp(c_shifted / tau)
    w = w / w.sum()
    
    # Gradient: grad[i] = 2 * sum_j w[i+j] * p[j] for valid i+j in [0, 2n-2]
    grad = np.zeros(n)
    for i in range(n):
        # j ranges from 0 to min(n-1, 2n-2-i)
        j_max = min(n-1, 2*n-2-i)
        if j_max >= 0:
            grad[i] = 2.0 * np.dot(p[:j_max+1], w[i:i+j_max+1])
    
    return grad, c, w

def optimize_smooth_max(p_init, time_limit, seed=42):
    """Optimize using smooth max gradient descent."""
    rng = np.random.default_rng(seed)
    n = len(p_init)
    p = np.clip(p_init.astype(np.float64), 0, None).copy()
    p = p / p.sum()
    
    best_p = p.copy()
    best_score = compute_score_fast(p)
    
    t0 = time.monotonic()
    tau = 0.1  # initial temperature
    lr = 0.1  # initial learning rate
    step = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient
        grad, c, w = smooth_max_gradient(p, tau)
        
        # Gradient descent
        p_new = p - lr * grad
        
        # Project onto probability simplex
        p_new = np.clip(p_new, 0, None)
        s = p_new.sum()
        if s < 1e-10:
            p_new = np.ones(n) / n
        else:
            p_new = p_new / s
        
        # Evaluate
        new_score = compute_score_fast(p_new)
        
        if new_score < best_score:
            best_score = new_score
            best_p = p_new.copy()
            no_improve = 0
        elif new_score < best_score * 1.01:
            # Accept if close (for exploration)
            pass
        else:
            no_improve += 1
        
        # Accept if it improves the smooth objective
        smooth_new = tau * np.log(np.sum(np.exp((np.convolve(p_new, p_new) - np.convolve(p_new, p_new).max()) / tau)))
        smooth_cur = tau * np.log(np.sum(np.exp((c - c.max()) / tau)))
        if smooth_new <= smooth_cur or rng.random() < 0.1:
            p = p_new
        else:
            lr *= 0.5
            if lr < 1e-6:
                # Perturb and restart
                p = p + 0.01 * rng.standard_normal(n)
                p = np.clip(p, 0, None)
                p = p / p.sum()
                lr = 0.1
        
        # Anneal temperature
        if step % 100 == 0:
            tau *= 0.95
            tau = max(tau, 0.01)
        
        step += 1
        
        if step % 10000 == 0:
            print(f"    step={step}, score={best_score:.8f}, tau={tau:.4f}, lr={lr:.6f}")
    
    return best_p, best_score
```
