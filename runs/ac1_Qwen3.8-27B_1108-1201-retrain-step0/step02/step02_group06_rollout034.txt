import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_of_a(a):
    """Score = 2n * max(conv(a,a)) / (sum a)^2. Scale invariant, so we can use sum=1."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)

def smooth_max_grad(a, tau):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    value = max_c + tau * np.log(np.sum(exp_vals))
    # Gradient: grad[j] = 2 * sum_m a[m] * w[j+m]
    # This is cross-correlation of a with w, evaluated at j
    fa = np.fft.rfft(a, size)

def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)

def project_simplex(v):
    """Project onto {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

    new_a = np.clip(new_a, 0, None)
    new_a /= np.sum(new_a)
    return new_a

"""
Multi-scale optimization for minimizing peak linear convolution of non-negative sequences.
Uses soft-max gradient descent on the simplex with Adam, spectral initialization,
multi-resolution upscaling, and aggressive coordinate descent polishing.
Key insight: the score 2n*max(conv(a,a))/(sum a)^2 equals 2n*max(conv) when sum=1,
and the theoretical lower bound is 2n/(2n-1), approached by flattening the convolution.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_of_a(a):
    """Score = 2n * max(conv(a,a)) / (sum a)^2. Scale invariant."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def smooth_max_grad(conv, tau):
    """Compute soft-max of conv and gradient weights."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    value = max_c + tau * np.log(np.sum(exp_vals))
    return value, w


def compute_grad_a(a, w):
    """Compute d(soft_max(conv(a,a)))/da using cross-correlation."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(fa) * fw, size)[:n]
    return 2.0 * corr


def softmax(z):
    """Stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def init_uniform(n):
    return np.ones(n) / n


def init_u_shape(n, rng, depth=0.5):
    """U-shaped: high at edges, low in middle."""
    t = np.arange(n) / max(n - 1, 1)
    a = 1.0 - depth * np.sin(np.pi * t) ** 2
    a = np.clip(a, 0.01, None)
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def init_two_hump(n, rng, pos=0.25):
    """Two peaks at pos and 1-pos fractions of the sequence."""
    t = np.arange(n) / max(n - 1, 1)
    a = 1.0 + 0.5 * (np.exp(-((t - pos) ** 2) / 0.01) + np.exp(-((t - (1 - pos)) ** 2) / 0.01))
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def init_fourier_flat(n, rng):
    """Initialize with flat power spectrum (approximately)."""
    # Create a random phase spectrum with uniform magnitude
    size = n
    A = np.ones(size // 2 + 1, dtype=np.complex128)
    phases = rng.uniform(0, 2 * np.pi, size // 2 + 1)
    A *= np.exp(1j * phases)
    # Make it Hermitian
    A_full = np.zeros(size, dtype=np.complex128)
    A_full[:size // 2 + 1] = A
    for k in range(1, size // 2):
        A_full[size - k] = np.conj(A[k])
    a = np.fft.irfft(A_full, size)
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_ramp(n, rng):
    """Linear ramp with noise."""
    t = np.arange(n) / max(n - 1, 1)
    a = 0.5 + t
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def init_cosine_mod(n, rng, freq=1):
    """Cosine modulation for spectral shaping."""
    t = np.arange(n) / max(n - 1, 1)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def optimize_adam_simplex(a0, time_limit, rng, verbose=True):
    """
    Adam optimization on softmax parameterization over the simplex.
    Minimizes 2n * max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    n = len(a0)
    # Initialize theta from a0
    a0_clip = np.clip(a0, 1e-15, None)
    a0_norm = a0_clip / np.sum(a0_clip)
    theta = np.log(a0_norm)
    
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    t0 = time.monotonic()
    best_a = a0_norm.copy()
    best_score = 2.0 * n * np.max(fft_convolve(a0_norm, a0_norm))
    
    # Temperature: start high (smooth), end low (sharp)
    tau_start = 5.0 / np.sqrt(n)
    tau_end = 0.05 / np.sqrt(n)
    
    # Learning rate
    lr_base = 1.0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = min(1.0, (time.monotonic() - t0) / time_limit)
        # Exponential temperature decay
        tau = tau_start * (tau_end / tau_start) ** (elapsed ** 0.6)
        
        a = softmax(theta)
        conv = fft_convolve(a, a)
        value, w = smooth_max_grad(conv, tau)
        grad_a = compute_grad_a(a, w)
        
        # Gradient w.r.t. theta (Riemannian: project gradient onto tangent space of simplex)
        inner = np.dot(a, grad_a)
        grad_theta = a * (grad_a - inner)
        
        step += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * (grad_theta * grad_theta)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        
        # Adaptive learning rate
        lr_eff = lr_base / (1.0 + 0.005 * step)
        theta = theta - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate periodically
        if step % 50 == 0:
            a_check = softmax(theta)
            sc = 2.0 * n * np.max(fft_convolve(a_check, a_check))
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                if verbose and step % 500 == 0:
                    print(f"      step={step}, score={sc:.6f}, best={best_score:.6f}, "
                          f"tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")
    
    return best_a, best_score


def optimize_projected_grad(a0, time_limit, rng, verbose=True):
    """
    Projected gradient descent with FISTA acceleration on the simplex.
    Directly minimizes max(conv(a,a)).
    """
    n = len(a0)
    a = a0.copy()
    a = np.clip(a, 0, None)
    a = a / np.sum(a)
    a_prev = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_val = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    step = 0
    
    tau_start = 5.0 / np.sqrt(n)
    tau_end = 0.1 / np.sqrt(n)
    lr = 1.0 / (2.0 * np.sqrt(n))
    
    while time.monotonic() - t0 < time_limit:
        elapsed = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed ** 0.7)
        
        # FISTA momentum
        t_prev = t_mom
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        beta = (t_prev - 1.0) / t_mom
        
        y = a + beta * (a - a_prev)
        y = project_simplex(y)
        
        conv = fft_convolve(y, y)
        value, w = smooth_max_grad(conv, tau)
        grad = compute_grad_a(y, w)
        
        # Gradient step + projection
        candidate = y - lr * grad
        candidate = project_simplex(candidate)
        
        a_prev = a
        a = candidate
        step += 1
        
        if step % 100 == 0:
            actual_val = np.max(fft_convolve(a, a))
            if actual_val < best_val:
                best_val = actual_val
                best_a = a.copy()
                if verbose and step % 1000 == 0:
                    print(f"      FISTA step={step}, score={2.0*n*best_val:.6f}, "
                          f"t={time.monotonic()-t0:.1f}s")
    
    return best_a, 2.0 * n * best_val


def coordinate_descent(a0, time_limit, rng, verbose=True):
    """
    Aggressive coordinate descent on the simplex.
    Uses peak-targeted moves and annealed step sizes.
    """
    n = len(a0)
    a = a0.copy()
    a = np.clip(a, 0, None)
    a = a / np.sum(a)
    
    current_score = 2.0 * n * np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 20000:
        elapsed = (time.monotonic() - t0) / time_limit
        # Anneal the perturbation size
        scale = 0.15 * (1.0 - 0.5 * elapsed)
        
        r = rng.random()
        
        if r < 0.35:
            # Pair transfer
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            frac = rng.uniform(0.02, scale)
            amt = a[i] * frac
            if amt < 1e-15:
                continue
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate /= cs
            
        elif r < 0.55:
            # Single element scale
            j = rng.integers(0, n)
            factor = np.exp(rng.normal(0, 0.08))
            candidate = a.copy()
            candidate[j] *= factor
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate /= cs
            
        elif r < 0.75:
            # Block perturbation
            width = max(1, int(n * rng.uniform(0.01, 0.05)))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate /= cs
            
        else:
            # Peak-targeted reduction
            conv = fft_convolve(a, a)
            # Get top 3 peak locations
            peak_indices = np.argsort(conv)[-3:][::-1]
            for peak_idx in peak_indices:
                candidate = a.copy()
                # Find top contributors to this peak
                contribs = np.zeros(n)
                p = int(peak_idx)
                for i in range(n):
                    j = p - i
                    if 0 <= j < n:
                        contribs[i] = a[i] * a[j]
                top_k = max(3, n // 64)
                top_idx = np.argsort(contribs)[-top_k:]
                reduction = rng.uniform(0.01, 0.05)
                candidate[top_idx] *= (1.0 - reduction)
                # Redistribute removed mass to random non-peak locations
                removed = np.sum(a[top_idx] - candidate[top_idx])
                if removed > 1e-15:
                    others = np.setdiff1d(np.arange(n), top_idx)
                    if len(others) > 0:
                        n_add = min(10, len(others))
                        add_idx = rng.choice(others, n_add, replace=False)
                        candidate[add_idx] += removed / n_add
                candidate = np.clip(candidate, 0, None)
                cs = np.sum(candidate)
                if cs < 1e-10:
                    break
                candidate /= cs
                break
            
            if cs < 1e-10:
                continue
        
        new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
        if new_score < current_score:
            a = candidate
            current_score = new_score
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
                if verbose and (int(best_score * 1000000) % 100 == 0):
                    print(f"      CD: score={best_score:.6f}, t={time.monotonic()-t0:.1f}s")
        else:
            no_improve += 1
    
    return best_a, best_score


def upsample(a, new_n, rng):
    """Upsample sequence to new length with smooth interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_a = np.interp(x_new, x_old, a)
    # Add small noise to break symmetry
    noise_level = 0.005 * np.mean(new_a)
    new_a += noise_level * rng.standard_normal(new_n)
    new_a = np.clip(new_a, 0, None)
    s = np.sum(new_a)
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return new_a / s


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search minimizing 2n*max(conv(a,a))/(sum a)^2 for non-negative a.
    Uses spectral init, Adam on simplex, FISTA, coordinate descent, and multi-scale.
    """

    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None

    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            if len(start_seq) > 0 and np.sum(start_seq) > 0.01:
                s0 = score_of_a(start_seq)
                print(f"Provided sequence: n={len(start_seq)}, score={s0:.6f}")
                if s0 < best_score:
                    best_score = s0
                    best_seq = start_seq.copy()
        except Exception as e:
            print(f"Error reading provided sequence: {e}")
    
    def try_sequence(a, label=""):
        nonlocal best_score, best_seq
        if a is None or len(a) == 0:
            return
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            return
        sc = score_of_a(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** New best ({label}): score={best_score:.6f}, n={len(a)}, "

        inits.append(('cosine3', init_cosine_mod(n, rng, freq=3)))
        # U-shaped with different depths

        inits.append(('u_shallow', init_u_shape(n, rng, depth=0.2)))
        return inits
    
    # Phase 1: Adam optimization at multiple scales
    print("\n=== Phase 1: Adam optimization (multi-scale) ===")
    
    # Configure: (n, n_inits_to_try, method)
    phases = [

                break

            if method == 'adam':

                a_best, sc = optimize_projected_grad(a0, t_left, rng, verbose=False)

    # Phase 2: Coordinate descent polish on best sequence

    if best_seq is not None:

        a_norm = best_seq / np.sum(best_seq)
        polished, sc = coordinate_descent(a_norm, polish_time, rng, verbose=True)
        polished_scaled = polished * np.sum(best_seq)
        try_sequence(polished_scaled, "polish")
    
    # Phase 3: Second round of optimization with different inits

    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic() - 60
        n3 = len(best_seq) if best_seq is not None else 512

        for trial in range(3):

            a0 = init_u_shape(n_try, rng, depth=rng.uniform(0.2, 0.8))
            t_alloc = min(80.0, remaining / (4 - trial))
            a_best, sc = optimize_adam_simplex(a0, t_alloc, rng, verbose=False)
            try_sequence(a_best, f"phase3_trial{trial},n={n_try}")
        
        # Also try FISTA on the current best
        if best_seq is not None:
            a_norm = best_seq / np.sum(best_seq)
            t_alloc = min(60.0, remaining / 3)
            a_best, sc = optimize_projected_grad(a_norm, t_alloc, rng, verbose=False)
            a_scaled = a_best * np.sum(best_seq)
            try_sequence(a_scaled, "phase3_fista")
    
    # Phase 4: Final coordinate descent polish
    print(f"\n=== Phase 4: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        final_time = max(10.0, deadline - time.monotonic() - 30)
        a_norm = best_seq / np.sum(best_seq)
        polished, sc = coordinate_descent(a_norm, final_time, rng, verbose=True)
        polished_scaled = polished * np.sum(best_seq)

    # Final verification

        final_score = evaluate_sequence(final_list)

        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
