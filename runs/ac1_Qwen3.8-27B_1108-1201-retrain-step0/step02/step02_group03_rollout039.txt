```python
"""
Multi-scale L-BFGS optimization with annealed soft-max for minimizing
peak autocorrelation of non-negative sequences. Uses FFT-based initialization,
scipy L-BFGS for smooth optimization, and peak-directed local search for
fine-tuning. Explores multiple sequence lengths and diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def obj_and_grad(theta, tau, n):
    """
    Compute soft-max of conv(a,a) and its gradient w.r.t. theta,
    where a = softmax(theta) (sums to 1).
    Returns (value, gradient).
    """
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    # Soft-max
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / tau)
    sum_exp = np.sum(exp_c)
    value = c_max + tau * np.log(sum_exp)
    w = exp_c / sum_exp
    
    # Gradient of soft-max wrt a: G[i] = 2 * sum_j a[j] * w[i+j]
    # Computed via: G[i] = 2 * conv(a[::-1], w)[n-1+i]
    a_flipped = a[::-1]
    N = 1
    while N < n + (2*n - 1) - 1:
        N <<= 1
    A = np.fft.rfft(a_flipped, N)
    W = np.fft.rfft(w, N)
    conv_aw = np.fft.irfft(A * W, N)[:n + 2*n - 2]
    G = 2.0 * conv_aw[n-1:2*n-1]
    
    # Gradient w.r.t. theta: a * (G - dot(a, G))
    inner = np.dot(a, G)
    grad = a * (G - inner)
    
    return value, grad


def fft_phase_init(n, rng, n_iter=200):
    """Gerchberg-Saxton: flatten magnitude spectrum, enforce non-negativity."""
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    
    # Start with random phases
    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        # Add random phase perturbation (decaying)
        noise_scale = 0.01 * (1 - it / n_iter)
        phase *= np.exp(1j * noise_scale * rng.standard_normal(N_fft))
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def chirp_init(n, rng, n_harmonics=3):
    """Initialize with sum of cosines + noise."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(n_harmonics):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.3 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def random_init(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / np.sum(a)


def upsample(a, new_n, rng):
    """Upsample a sequence via linear interpolation + small noise."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)

    a_init is the initial sequence (will be normalized to sum=1).

    """
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a)  # normalize to sum=1
    theta = np.log(a + 1e-15)
    # Remove mean to reduce degeneracy (adding a constant to theta doesn't change softmax)
    theta -= np.mean(theta)
    
    best_score = 2.0 * n * np.max(fft_convolve(a, a))  # score when sum=1
    best_a = a.copy()
    t0 = time.monotonic()
```

Let me also think about the **local search** more carefully. The key insight is that the peak of the autocorrelation is determined by a few "important" lags. If I can identify which pairs (i, j) contribute most to the peak and adjust them, I can make targeted improvements.

A better local search algorithm:
1. Compute `c = conv(a, a)`
2. Find the top K lags (e.g., K=5)
3. For each of these lags, compute the contribution of each pair
4. Find the "bottleneck" pairs (those with high contribution to peak lags)
5. Propose moves that reduce the total contribution to peak lags while maintaining sum constraint

This is essentially a **convex optimization** subproblem:
- Minimize: sum of contributions to top-K lags
- Subject to: a >= 0, sum(a) = 1
- (Approximately, since we're only looking at K lags)

But solving this exactly is complex. A simpler heuristic:
- For the peak lag l: `c[l] = sum_i a[i]*a[i+l]`
- The gradient of c[l] w.r.t. a is `2 * (a shifted by -l) + 2 * (a shifted by +l)` (approximately)
- Take a small step in the negative gradient direction, project onto {a >= 0, sum(a) = 1}

This is a **projected gradient step** on the simplex. It's simpler and might work well.

```python
def projected_grad_step(a, l, step_size):
    """Take a step to reduce c[l] while staying on the simplex."""
    n = len(a)
    # Gradient of c[l] w.r.t. a
    grad = np.zeros(n)
    i_min = max(0, l - n + 1)
    i_max = min(n - 1, l)
    for i in range(i_min, i_max + 1):
        j = l - i
        grad[i] += 2 * a[j]
        grad[j] += 2 * a[i]
    
    # Negative gradient step
    candidate = a - step_size * grad
    # Project onto {x >= 0, sum(x) = 1}
    # This is the Euclidean projection onto the probability simplex
    # Algorithm: find t such that sum(max(x_i - t, 0)) = 1
    x_sorted = np.sort(candidate)[::-1]
    css = np.cumsum(x_sorted)
    rho = np.where(x_sorted > 0)[0][0] + 1  # first positive index + 1
    # Bisection for t
    lo, hi = 0, x_sorted[0]
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.sum(np.maximum(candidate - mid, 0)) > 1:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    candidate = np.maximum(candidate - t, 0)
    candidate /= np.sum(candidate)
    return candidate
```

```
sort x in decreasing order: x_(1) >= x_(2) >= ... >= x_(n)
find rho = max{j: x_(j) - 1/j * sum_{i<=j} x_(i) > 0}
```

The correct algorithm (Duchi et al. 2008):
1. Sort x in decreasing order
2. Compute `v(j) = (sum_{i=1}^j x_(i) - 1) / j` for j = 1, ..., n
3. Find `rho = max{j: v(j) > 0}`
4. `theta = v(rho)`
5. `result = max(x - theta, 0)`

```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, n + 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]  # largest valid rho
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

This is O(n log n) due to sorting, which is fine for our purposes.

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence (scale-invariant)."""
    n = len(a)
