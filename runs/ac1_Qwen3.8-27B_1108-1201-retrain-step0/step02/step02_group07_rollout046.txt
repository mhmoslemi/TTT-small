    k_min = max(0, m)  # smallest k where k-m >= 0
    k_max = min(2*n-2, m+n-1)  # largest k where k-m < n

    if 0 <= 2*m < 2*n-1:

        F = None

            return -np.inf

Coordinate descent with exact 1D subproblem solving for minimizing

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_optimal_x(a, m, conv, n):
    """Find x >= 0 minimizing max(conv_new) where a[m]=x."""
    x_old = a[m]
    conv_len = len(conv)
    
    k_start = max(0, m)
    k_end = min(conv_len - 1, m + n - 1)
    
    # Unaffected max
    f_const = -1e30
    if k_start > 0:
        f_const = max(f_const, conv[:k_start].max())
    if k_end < conv_len - 1:
        f_const = max(f_const, conv[k_end + 1:].max())
    
    # Quadratic term at k=2m
    if 0 <= 2 * m < conv_len:
        D = conv[2 * m] - x_old * x_old
    else:
        D = -1e30

        ks_lin = ks[mask]
    else:
        ks_lin = ks

        F = conv[ks_lin] - E * x_old
    else:
        E = None
    
    # Ternary search
    lo = 0.0
    hi = max(2.0 * x_old + 0.01, 0.05)

        val = f_const

            q = D + x * x

    return x_opt
