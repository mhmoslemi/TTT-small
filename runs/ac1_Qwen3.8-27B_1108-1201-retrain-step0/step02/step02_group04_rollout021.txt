import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):

    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s ** 2)

"""
Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses projected gradient descent with log-sum-exp smoothing of the max term,
FFT-based fast convolution, multiple initializations, and coordinate descent.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the score for a given sequence (scale-invariant)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s ** 2))


def optimize_gradient(a_init, beta, lr, n_steps, time_limit, rng):
    """
    Projected gradient descent with log-sum-exp smoothing.
    Works with sum(a)=1 normalization. Returns best sequence and its score.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s

    c = fft_convolve(a, a)
    current_score = 2.0 * n * np.max(c)
    best = a.copy()
    best_score = current_score

    t0 = time.monotonic()
    step = 0
    no_improve = 0
    patience = 5000

    while time.monotonic() - t0 < time_limit and step < n_steps:
        c = fft_convolve(a, a)
        c_max = np.max(c)

        # Compute softmax weights for log-sum-exp gradient
        scaled = beta * (c - c_max)
        np.clip(scaled, -100, 100, out=scaled)
        exp_c = np.exp(scaled)
        w = exp_c / np.sum(exp_c)

        # Gradient: 2 * correlate(w, a, 'valid')
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # Adaptive learning rate with decay
        lr_t = lr / (1.0 + 0.0005 * step)

        # Gradient step
        a_new = a - lr_t * grad
        a_new = np.clip(a_new, 0.0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new /= s_new

        # Evaluate
        c_new = fft_convolve(a_new, a_new)
        score_new = 2.0 * n * np.max(c_new)

        if score_new < current_score - 1e-12:
            a = a_new
            current_score = score_new
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            no_improve += 1
            if no_improve > patience:
                # Perturb to escape local minimum
                a = a + 0.05 * rng.standard_normal(n)
                a = np.clip(a, 0.0, None)
                s = np.sum(a)
                if s > 1e-10:
                    a /= s
                else:
                    a = np.ones(n) / n
                c = fft_convolve(a, a)
                current_score = 2.0 * n * np.max(c)
                no_improve = 0

        step += 1
        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      step={step}, score={current_score:.8f}, best={best_score:.8f}, "
                  f"lr={lr_t:.6f}, time={elapsed:.1f}s")

    return best, best_score


def optimize_topk(a_init, k, lr, n_steps, time_limit, rng):
    """
    Optimizer that focuses on the top-k peaks of the convolution.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:

    else:
        a /= s

    c = fft_convolve(a, a)
    current_score = 2.0 * n * np.max(c)
    best = a.copy()
    best_score = current_score

    t0 = time.monotonic()
    step = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and step < n_steps:
        c = fft_convolve(a, a)
        c_max_val = np.max(c)

        # Get top-k indices
        if k < len(c):
            top_idx = np.argpartition(c, -k)[-k:]
        else:
            top_idx = np.arange(len(c))

        # Weighted gradient from top peaks
        top_vals = c[top_idx]
        top_max = np.max(top_vals)
        weights = np.exp(50.0 * (top_vals - top_max))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for idx, wt in zip(top_idx, weights):
            idx = int(idx)
            # Gradient of c[idx] w.r.t. a: 2*a[idx-i] for valid i
            lo = max(0, idx - n + 1)
            hi = min(n, idx + 1)
            for i in range(lo, hi):
                j = idx - i
                if 0 <= j < n:
                    grad[i] += 2.0 * wt * a[j]

        # Scale gradient
        grad_scale = np.max(np.abs(grad))
        if grad_scale > 0:
            grad /= grad_scale

        lr_t = lr / (1.0 + 0.001 * step)
        a_new = a - lr_t * grad
        a_new = np.clip(a_new, 0.0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new /= s_new

        c_new = fft_convolve(a_new, a_new)
        score_new = 2.0 * n * np.max(c_new)

        if score_new < current_score - 1e-12:
            a = a_new
            current_score = score_new
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            no_improve += 1
            if no_improve > 3000:
                a = a + 0.03 * rng.standard_normal(n)
                a = np.clip(a, 0.0, None)
                s = np.sum(a)
                if s > 1e-10:
                    a /= s
                c = fft_convolve(a, a)
                current_score = 2.0 * n * np.max(c)
                no_improve = 0

        step += 1
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      topk step={step}, score={current_score:.8f}, best={best_score:.8f}, "
                  f"time={elapsed:.1f}s")

    return best, best_score


def coordinate_descent(a_init, time_limit, rng):
    """
    Coordinate descent: try adjusting each element to improve the score.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a /= s

    c = fft_convolve(a, a)
    current_score = 2.0 * n * np.max(c)
    best = a.copy()
    best_score = current_score

    t0 = time.monotonic()
    sweep = 0
    indices = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        rng.shuffle(indices)
        improved_this_sweep = 0

        for i in indices:
            if time.monotonic() - t0 > time_limit:
                break

            i = int(i)
            # Try several multiplicative factors
            for factor in [0.8, 0.6, 1.2, 1.5]:
                a_new = a.copy()
                a_new[i] *= factor
                a_new = np.clip(a_new, 0.0, None)
                s_new = np.sum(a_new)
                if s_new < 1e-10:
                    continue
                a_new /= s_new
                c_new = fft_convolve(a_new, a_new)
                score_new = 2.0 * n * np.max(c_new)
                if score_new < current_score - 1e-12:
                    a = a_new
                    current_score = score_new
                    improved_this_sweep += 1
                    if current_score < best_score:
                        best_score = current_score
                        best = a.copy()
                    break

        sweep += 1
        if sweep % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep}, score={current_score:.8f}, best={best_score:.8f}, "
                  f"improved={improved_this_sweep}, time={elapsed:.1f}s")

        # Early stop if no improvement for several sweeps
        if improved_this_sweep == 0 and sweep > 5:
            break

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):

            return

        if s < best_score:
            best_score = s
            best_seq = seq.copy()

        else:
            print(f"  {label}: n={len(seq)}, score={s:.8f}")

    # Try the parent sequence
    parent = None
    if 'height_sequence_1' in kwargs and kwargs['height_sequence_1'] is not None:
        parent = np.asarray(kwargs['height_sequence_1'], dtype=np.float64)

    elif 'height_sequence_1' in dir() or 'height_sequence_1' in globals():
        try:
            parent = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
