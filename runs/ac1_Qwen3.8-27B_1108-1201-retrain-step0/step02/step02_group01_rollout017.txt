Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative refinement as the primary method,
with adaptive active constraint selection, 1D line search for blending, power-mean gradient
descent as a secondary smooth method, and coordinate descent polish.

Key differences from previous approach:
1. LP-based iteration is the PRIMARY optimization method (not just periodic restarts)
2. Adaptive active constraint selection (all lags above 95% of peak)
3. Proper 1D line search for the blend parameter (convex minimization in t)
4. Power-mean objective (instead of soft-max) for smoother gradient refinement
5. Monotonic improvement guarantee via conservative step acceptance
6. Multiple LP constraint strategies tried per iteration
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog, minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def compute_objective(a):
    """Compute the true objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (S * S))


def power_mean_and_grad(c, p, a, n):
    """Compute p-power mean of c and its gradient w.r.t. a.
    c is the self-convolution of a (length 2n-1).
    grad[m] = 2 * sum_j a[j] * w[j+m] where w[k] = c[k]^(p-1) * scale.
    """
    c_safe = np.maximum(c, 1e-30)
    if p == np.inf:
        return c.max(), None
    cp = c_safe ** p
    sum_cp = cp.sum()
    if sum_cp < 1e-300:
        return 0.0, np.zeros(n)
    pm = sum_cp ** (1.0 / p)
    scale = sum_cp ** (1.0 / p - 1.0)
    w = c_safe ** (p - 1) * scale
    # grad[m] = 2 * sum_j a[j] * w[j+m] for valid j+m < 2n-1
    # This is cross-correlation: grad[m] = 2 * (a_rev * w_full)[m + n - 1]
    a_rev = a[::-1]
    w_full = np.zeros(2 * n - 1)
    w_full[:len(w)] = w
    conv_result = fft_convolve(a_rev, w_full)
    grad = 2.0 * conv_result[n - 1: 2 * n - 1]
    return pm, grad


def solve_lp_direction(f, M, active_lags, time_limit=5.0):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0.
    Returns optimal g vector or None."""
    n = len(f)
    K = len(active_lags)
    if K == 0:
        return np.ones(n) / n
    # Build constraint matrix: A[idx, j] = f[k-j] if 0 <= k-j < n
    A = np.zeros((K, n))
    for idx in range(K):
        k = active_lags[idx]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            A[idx, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    # Maximize sum(g) = minimize -sum(g)
    c_obj = -np.ones(n)
    b = np.full(K, M)
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                         method='highs',
                         options={'presolve': True, 'time_limit': time_limit})
        if result.success and result.x is not None:
            g = result.x
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def find_best_blend_t(c_a, c_ag, c_gg, max_t=0.5, n_try=30):
    """Find t in (0, max_t) minimizing max_k [(1-t)^2*c_a[k] + 2t(1-t)*c_ag[k] + t^2*c_gg[k]].
    Uses ternary search since the function is convex in t (max of quadratics)."""
    def peak_at_t(t):
        val = (1.0 - t) ** 2 * c_a + 2.0 * t * (1.0 - t) * c_ag + t ** 2 * c_gg
        return val.max()

    # Ternary search on [0, max_t]
    a_lo = 0.0
    a_hi = max_t
    for _ in range(40):
        m1 = a_lo + (a_hi - a_lo) / 3.0
        m2 = a_hi - (a_hi - a_lo) / 3.0
        f1 = peak_at_t(m1)
        f2 = peak_at_t(m2)
        if f1 < f2:
            a_hi = m2
        else:
            a_lo = m1
    t_opt = (a_lo + a_hi) / 2.0
    return t_opt, peak_at_t(t_opt)


def lp_iteration(a, S, max_active=500, time_limit=5.0):
    """One LP-based iteration. Returns (a_new, improved) or (a, False) if no improvement."""
    n = len(a)
    c = fft_convolve_self(a)
    M = c.max()

    # Select active lags: all lags above 95% of peak, capped at max_active
    threshold = 0.95 * M
    active_mask = c >= threshold
    active_lags = np.where(active_mask)[0]
    if len(active_lags) > max_active:
        # Keep the top max_active
        top_idx = np.argsort(c)[::-1][:max_active]
        active_lags = top_idx
    elif len(active_lags) < 10:
        # Ensure we have at least 10 constraints
        active_lags = np.argsort(c)[::-1][:max(10, min(50, 2 * n - 1))]

    # Solve LP
    g_raw = solve_lp_direction(a, M, active_lags, time_limit=time_limit)
    if g_raw is None:
        return a, False

    g_sum = g_raw.sum()
    if g_sum <= a.sum() * (1 + 1e-6):
        # LP didn't find a direction with larger sum; no improvement possible
        return a, False

    # Normalize g to have sum S
    g = g_raw * (S / g_sum)
    g = np.maximum(g, 0.0)

    # Compute cross-convolutions for line search
    c_ag = fft_convolve(a, g)
    c_gg = fft_convolve_self(g)

    # Find best blend
    t_opt, peak_opt = find_best_blend_t(c, c_ag, c_gg, max_t=0.5)

    if t_opt < 1e-4:
        return a, False

    # Build new sequence
    a_new = (1.0 - t_opt) * a + t_opt * g
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-10:
        return a, False
    a_new = a_new * (S / s_new)

    # Verify improvement
    c_new = fft_convolve_self(a_new)
    if c_new.max() < M * (1 - 1e-8):
        return a_new, True
    else:
        # Try smaller t
        for t_try in [t_opt * 0.5, t_opt * 0.25, 0.05, 0.02]:
            a_try = (1.0 - t_try) * a + t_try * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try * (S / s_try)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < M * (1 - 1e-8):
                return a_try, True
        return a, False


def gradient_step(a, S, n, p, lr, n_steps=100):
    """Several steps of gradient descent on the power-mean objective."""
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = fft_convolve_self(a).max()

    for step in range(n_steps):
        c = fft_convolve_self(a)
        pm, grad = power_mean_and_grad(c, p, a, n)
        if grad is None:
            break
        # Normalize gradient to maintain sum constraint
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        a_new = a - lr * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new * (S / s_new)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < best_obj:
            best_obj = c_new.max()
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce learning rate
            lr *= 0.5
            if lr < 1e-10:
                break
    return best_a, best_obj


def coordinate_descent_polish(a, S, n, time_budget, max_sweeps=100):
    """Coordinate descent: try reducing each element that contributes to peak lags."""
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        # Focus on elements that contribute to near-peak lags
        threshold = 0.9 * best_obj
        peak_lags = np.where(c >= threshold)[0]
        # For each peak lag k, the elements a[j] with a[k-j] > 0 contribute
        # Sample a subset of indices to try
        indices_to_try = np.random.default_rng(sweep).choice(n, size=min(n, 200), replace=False)

        for i in indices_to_try:
            if time.monotonic() - start > time_budget:
                break
            original = a[i]
            if original < 1e-10:
                continue
            # Try reducing a[i]
            for factor in [0.5, 0.7, 0.85]:
                a[i] = original * factor
                c_check = fft_convolve_self(a)
                if c_check.max() < best_obj * 0.9999:
                    best_obj = c_check.max()
                    best_a = a.copy()
                    improved = True
                    c = c_check
                    break
            else:
                a[i] = original  # Restore if no improvement

        if not improved:
            # Try increasing some elements
            for i in indices_to_try[:50]:
                if time.monotonic() - start > time_budget:
                    break
                original = a[i]
                for factor in [1.3, 1.5, 2.0]:
                    a[i] = original * factor
                    s = a.sum()
                    a_norm = a * (S / s)
                    c_check = fft_convolve_self(a_norm)
                    if c_check.max() < best_obj * 0.9999:
                        best_obj = c_check.max()
                        best_a = a_norm.copy()
                        improved = True
                        a = a_norm
                        c = c_check
                        break
                else:
                    a[i] = original
        if not improved:
            break

    return best_a, best_obj
