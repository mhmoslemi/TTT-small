def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_conv_grad(a, k):
    """Gradient of conv(a,a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    for i in range(max(0, k - n + 1), min(n, n - k)):

        g[i] += a[i + k] if i + k < n else 0

    return g

"""
Optimized step function coefficient search using adaptive smoothed gradient descent,
coordinate descent with peak targeting, and diverse multi-length exploration.
Combines log-sum-exp smoothing with backtracking line search, KKT-guided
coordinate updates, and iterative peak reduction for efficient minimization
of 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:2 * n - 1]


def softmax_weights(conv, t, threshold=1e-8):
    """Compute softmax weights for the peaks of conv."""
    max_c = conv.max()
    if max_c < 1e-15:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    scaled = t * (conv - max_c)
    w = np.exp(scaled)
    total = w.sum()
    if total < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= total
    # Sparsify: zero out tiny weights
    w[w < threshold * w.max()] = 0.0
    if w.sum() > 0:
        w /= w.sum()
    else:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    return w


def conv_grad_smoothed(a, conv, t, threshold=1e-8):
    """Gradient of smoothed max of conv(a,a) w.r.t. a."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    w = softmax_weights(conv, t, threshold)
    g = np.zeros(n)
    active = np.where(w > 0)[0]
    for k in active:
        wk = w[k]
        if wk < 1e-15:
            continue
        k = int(k)
        if k == 0:
            g += 2.0 * wk * a
        else:
            g[:n - k] += wk * a[k:]
            g[k:] += wk * a[:n - k]
    return g


def objective_normalized(a):
    """Objective 2*n*max(conv)/sum^2 for a (assumed non-negative)."""
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smooth_descent(a0, time_limit, t_init=30.0, t_max=5000.0, eta_init=0.5,
                   min_eta=0.01, patience_max=60, backtracking_max=30):
    """
    Smoothed gradient descent with adaptive temperature and backtracking line search.
    Minimizes 2*n*max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        grad = conv_grad_smoothed(a, conv, t)
        # Center the gradient (since we're on the simplex)
        grad_centered = grad - grad.mean()

        # Backtracking line search
        alpha = eta
        success = False
        for _ in range(backtracking_max):
            a_candidate = a - alpha * grad_centered
            a_candidate = np.maximum(a_candidate, 0.0)
            sc = a_candidate.sum()
            if sc < 1e-12:
                break
            a_candidate /= sc
            new_conv = fast_convolve(a_candidate)
            new_obj = 2.0 * n * new_conv.max()
            if new_obj < cur_obj - 1e-12:
                a = a_candidate
                conv = new_conv
                cur_obj = new_obj
                success = True
                break
            alpha *= 0.5

        if not success:
            # Try a smaller fixed step
            alpha = eta * 0.1
            a_candidate = a - alpha * grad_centered
            a_candidate = np.maximum(a_candidate, 0.0)
            sc = a_candidate.sum()
            if sc < 1e-12:
                patience += 1
            else:
                a_candidate /= sc
                new_conv = fast_convolve(a_candidate)
                new_obj = 2.0 * n * new_conv.max()
                if new_obj < cur_obj:
                    a = a_candidate
                    conv = new_conv
                    cur_obj = new_obj
                    success = True
                else:
                    patience += 1
        else:
            patience = 0

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()

        # Adaptive parameter updates
        if patience >= patience_max:
            patience = 0
            eta *= 0.7
            t *= 1.5
            if t > t_max:
                t = t_max
            if eta < min_eta:
                # Reset with different parameters
                eta = eta_init * 0.3
                t = min(t_init * 2.0, t_max)

        # Gradually increase temperature
        t = min(t * 1.01, t_max)

    return best_a, best_obj


def coordinate_descent(a0, time_limit, max_sweeps=5000, peak_threshold=0.8):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes the max convolution value.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        max_c = conv.max()
        if max_c < 1e-15:
            break

        for j in range(n):
            # Current contribution of a[j] to each conv[k]
            # conv[k] contains a[j]*a[j+k] (if j+k<n) and a[j-k]*a[j] (if j-k>=0)
            # If we change a[j] from val to val+delta:
            #   conv[k] changes by delta*(a[j+k]+a[j-k]) for relevant k
            #   sum changes by delta
            val = a[j]
            if val < 1e-15:
                # Try setting it to a small positive value
                delta = a.mean() * 0.1
                a[j] = delta
                new_conv = fast_convolve(a)
                new_obj = 2.0 * n * new_conv.max()
                if new_obj < best_obj:
                    improved = True
                    conv = new_conv
                    if new_obj < 2.0 * n * conv.max():
                        best_obj = 2.0 * n * conv.max()
                        best_a = a.copy()
                else:
                    a[j] = 0.0
                continue

            # Try reducing a[j]
            # The peak conv values that involve a[j]:
            # conv[0] has a[j]^2
            # conv[k] has a[j]*a[j+k] for k=1..n-1-j-1... 
            # Actually conv[k] = sum_i a[i]*a[i+k]
            # a[j] appears in conv[k] when j is in [0, n-1-k] as a[j] (first factor)
            #   contributing a[j]*a[j+k], AND when j-k is in [0,n-1-k] i.e. j in [k, n-1]
            #   contributing a[j-k]*a[j]
            
            # Try halving a[j]
            delta = val * 0.5
            a[j] = val - delta
            new_conv = fast_convolve(a)
            new_obj = 2.0 * n * new_conv.max()
            if new_obj < best_obj:
                conv = new_conv
                best_obj = new_obj
                best_a = a.copy()
                improved = True
            else:
                # Try moving mass to a random position with low contribution
                a[j] = val
                # Find position with lowest "autocorrelation contribution"
                # Approximate: position where a is smallest and far from peaks
                target = int(np.argmin(a))
                if target == j:
                    target = (j + 1) % n
                move = val * 0.3
                a[j] = val - move
                a[target] += move
                new_conv = fast_convolve(a)
                new_obj = 2.0 * n * new_conv.max()
                if new_obj < best_obj:
                    conv = new_conv
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = val
                    a[target] -= move

        if not improved:
            # Try a more aggressive perturbation
            peak_k = int(np.argmax(conv))
            # Reduce a at positions contributing most to this peak
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            seg_a = a[js:je + 1]
            seg_b = a[peak_k - je:peak_k - js + 1][::-1]
            contribs = seg_a * seg_b
            total = contribs.sum()
            if total > 1e-15:
                # Reduce the largest contributor
                max_local = int(np.argmax(contribs))
                pos = js + max_local
                delta = a[pos] * 0.2
                a[pos] -= delta
                # Distribute to positions with low contribution
                low_positions = np.where(a < a.mean())[0]
                if len(low_positions) > 0:
                    dst = low_positions[np.random.randint(len(low_positions))]
                    a[dst] += delta
                new_conv = fast_convolve(a)
                new_obj = 2.0 * n * new_conv.max()
                if new_obj < best_obj * 1.02:  # Accept small worsening for escape
                    conv = new_conv
                    best_obj = min(best_obj, new_obj)
                    best_a = a.copy()

    return best_a, best_obj


def peak_reduction(a0, time_limit, iterations=2000):
    """
    Iteratively target and reduce the peak of the convolution.
    At each step, identify the peak and redistribute mass from
    positions contributing to it.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(iterations):
        if time.monotonic() - t0 > time_limit:
            break

        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Find all peaks above 90% of max
        thresh = max_c * 0.9
        peak_positions = np.where(conv >= thresh)[0]

        for pk in peak_positions[:5]:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            seg_a = a[js:je + 1]
            seg_b = a[pk - je:pk - js + 1][::-1]
            contribs = seg_a * seg_b
            total = contribs.sum()
            if total < 1e-15:
                continue

            # Find the position with highest contribution
            probs = contribs / total
            # Pick top contributor with some randomness
            top_idx = int(np.argmax(contribs))
            if rng.random() < 0.7:
                src_local = top_idx
            else:
                src_local = int(rng.choice(len(contribs), p=probs))
            src = js + src_local

            # Find target: position that minimizes impact on all peaks
            best_target = -1
            best_target_score = np.inf
            candidates = np.where(a < a.mean() * 1.5)[0]
            if len(candidates) == 0:
                candidates = np.argmin(a)
                candidates = [int(candidates)]
            # Sample a few candidates
            sample_size = min(10, len(candidates))
            sampled = rng.choice(candidates, size=sample_size, replace=False)

            for tgt in sampled:
                tgt = int(tgt)
                if tgt == src:
                    continue
                # Estimate impact: how much would moving delta to tgt affect peaks
                impact = 0.0
                for p2 in peak_positions[:8]:
                    p2 = int(p2)
                    # Effect on conv[p2]: delta * a[tgt+p2] + delta * a[tgt-p2] (approximately)
                    effect = 0.0
                    if tgt + p2 < n:
                        effect += a[tgt + p2]
                    if tgt - p2 >= 0:
                        effect += a[tgt - p2]
                    impact += effect
                if impact < best_target_score:
                    best_target_score = impact
                    best_target = tgt

            if best_target < 0:
                best_target = int(np.argmin(a))
                if best_target == src:
                    best_target = (src + 1) % n

            # Determine how much to move
            delta = a[src] * rng.uniform(0.05, 0.25)
            a[src] -= delta
            a[best_target] += delta

            if a[src] < 0:
                a[src] = 0
                a[best_target] -= a[src]  # shouldn't happen
                continue

            new_conv = fast_convolve(a)
            new_obj = 2.0 * n * new_conv.max()
            if new_obj <= best_obj:
                conv = new_conv
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_a = a.copy()
            else:
                # Revert
                a[src] += delta
                a[best_target] -= delta
                if a[best_target] < 0:
                    a[best_target] = 0

    return best_a, best_obj


def diverse_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Gaussian
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        inits['gauss_%.2f' % sf] = g / g.sum()

    # Sinusoidal
    for freq in [1, 2, 3, 5]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        inits['sin_%d' % freq] = s / s.sum()

    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Random (various Dirichlet concentrations)
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['dir_%.1f' % alpha] = r

    # Binary random
    b = rng.integers(0, 2, n).astype(float)
    if b.sum() > 0:
        inits['binary'] = b / b.sum()

    # Sparse binary
    sp = np.zeros(n)
    k = max(1, n // 4)
        idx = rng.choice(n, k, replace=False)
    sp[idx] = 1.0
    inits['sparse_bin'] = sp / sp.sum()

    # Chirp
    chirp = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * x / (2.0 * n)))
    chirp = np.maximum(chirp, 0)
    if chirp.sum() > 1e-10:
        inits['chirp'] = chirp / chirp.sum()

    # Power law
    pw = (x + 1) ** 0.5
    inits['powlaw'] = pw / pw.sum()

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0).astype(float)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0).astype(float))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits['parent_interp'] = pp / pp.sum()

    # Perturb parent
    if parent is not None:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0).astype(float)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                pp_pert = pp + rng.normal(0, 0.05, n)
                pp_pert = np.maximum(pp_pert, 0)
                if pp_pert.sum() > 1e-10:
                    inits['parent_pert'] = pp_pert / pp_pert.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0).astype(float))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                pp_pert = pp + rng.normal(0, 0.05, n)
                pp_pert = np.maximum(pp_pert, 0)
                if pp_pert.sum() > 1e-10:
                    inits['parent_pert'] = pp_pert / pp_pert.sum()

    return inits

            record(a_opt)

    scan_results.sort(key=lambda x: x[0])
    print("  Scan best: %.6f" % best_score)

    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 16:
            top_configs.append((n, name, a_opt))
            seen.add(key)

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best_so_far', best_sequence.copy()))

    phase2_budget = min(1500.0, (deadline - time.monotonic()) * 0.50)
    n_deep = len(top_configs)
    time_per_deep = max(10.0, phase2_budget / max(1, n_deep))

    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 800:
            break
        t_alloc = min(time_per_deep * 1.2, deadline - time.monotonic() - 800)
        if t_alloc < 8.0:
            break
        print("  [%d/%d] n=%d, %s, budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))

        a_cur = a0.copy()
        sub = t_alloc / 6.0

        # Round 1: Smooth descent (warm)
        a_cur, _ = smooth_descent(a_cur, time_limit=sub, t_init=50.0, eta_init=0.4)
        record(a_cur)

        # Round 2: Peak reduction
        a_cur, _ = peak_reduction(a_cur, time_limit=sub)
        record(a_cur)

        # Round 3: Smooth descent (cooler)
        a_cur, _ = smooth_descent(a_cur, time_limit=sub, t_init=200.0, eta_init=0.15)
        record(a_cur)

        # Round 4: Coordinate descent
        a_cur, _ = coordinate_descent(a_cur, time_limit=sub)
        record(a_cur)

        # Round 5: Peak reduction again
        a_cur, _ = peak_reduction(a_cur, time_limit=sub)
        record(a_cur)

        # Round 6: Final smooth descent (very cool)
        a_cur, _ = smooth_descent(a_cur, time_limit=sub, t_init=1000.0, eta_init=0.05)
        record(a_cur)

    print("  Deep opt best: %.6f" % best_score)

    # Phase 3: Iterative refinement on the best
    print("=== Phase 3: Refinement ===")
    if best_sequence is not None and time.monotonic() < deadline - 100:
        phase3_budget = min(600.0, deadline - time.monotonic() - 100)
        a_best = best_sequence.copy()
        n_best = len(a_best)
        n_rounds = 5
        per_round = phase3_budget / n_rounds

        for r in range(n_rounds):
            if time.monotonic() > deadline - 100:
                break
            print("  Refinement round %d/%d" % (r + 1, n_rounds))
            # Slightly perturb
            a_try = a_best.copy() + rng.normal(0, 0.01, n_best)
            a_try = np.maximum(a_try, 0)
            a_try /= a_try.sum()

            a_opt, _ = smooth_descent(a_try, time_limit=per_round * 0.4,
                                       t_init=100.0, eta_init=0.1)
            record(a_opt)

            a_opt, _ = peak_reduction(a_opt, time_limit=per_round * 0.4)
            record(a_opt)

            a_opt, _ = coordinate_descent(a_opt, time_limit=per_round * 0.2)
            record(a_opt)

            if best_sequence is not None:
                a_best = best_sequence.copy()

    print("  Refinement best: %.6f" % best_score)

    # Phase 4: Final polish
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_final = best_sequence.copy()
            n_f = len(a_final)

            # Coordinate descent for a long time
            a_opt, _ = coordinate_descent(a_final, time_limit=remaining * 0.5)
            record(a_opt)

            if time.monotonic() < deadline - 5 and best_sequence is not None:
                rem = deadline - time.monotonic() - 5
                if rem > 5:
                    a_opt, _ = peak_reduction(best_sequence, time_limit=rem * 0.6)
                    record(a_opt)

            if time.monotonic() < deadline - 5 and best_sequence is not None:
                rem = deadline - time.monotonic() - 5
                if rem > 3:
                    a_opt, _ = smooth_descent(best_sequence, time_limit=rem,
                                               t_init=2000.0, eta_init=0.02,
                                               patience_max=100)
                    record(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = safe_evaluate(best_sequence)

    print("=== FINAL: score=%.6f, n=%d ===" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
