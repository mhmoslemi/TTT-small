    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def compute_soft_max_conv_and_grad(a, tau):
    """
    Compute soft-max of conv(a,a) and its gradient w.r.t. a.
    Returns: (soft_max_value, gradient_array, full_convolution)
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    w = e / np.sum(e)
    value = c_max + tau * np.log(np.sum(e))

    # Gradient: grad[j] = 2 * sum_m a[m] * w[m+j] = 2 * (a_rev conv w)[j+n-1]
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    grad = 2.0 * corr[n - 1:2 * n - 1]

    return value, grad, c


def fft_phase_init(n, rng, n_iter=80):
    """Gerchberg-Saxton style initialization to flatten magnitude spectrum."""
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft <<= 1

    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-14] = 1e-14
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def chirp_init(n, rng, alpha=0.5):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def multi_tone_init(n, rng, n_tones=4):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(n_tones):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.4 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def optimize_adam(n, a_init, time_limit, rng, verbose=True):
    """
    Adam gradient descent on softmax-parameterized sequence.
    Normalizes to sum = sqrt(2n) so score = max(conv).
    Uses adaptive temperature with plateau at the end.
    """
    S = math.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S
    theta = np.log(a / S + 1e-15) * 10.0  # Scale theta for better conditioning
    theta_scale = 10.0

    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr0 = 3.0
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = compute_score(a)
    patience = 0
    PATIENCE_LIMIT = 3000  # Steps without improvement before restart

    # Temperature schedule: start high, anneal, then plateau
    tau_start = 50.0 / n
    tau_mid = 5.0 / n
    tau_end = 0.2 / n
    # Phase fractions: 0-0.4: start->mid, 0.4-0.8: mid->end, 0.8-1.0: plateau at end

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        elapsed_frac = min(1.0, elapsed / time_limit)

        # Adaptive temperature
        if elapsed_frac < 0.4:
            f = elapsed_frac / 0.4
            tau = tau_start * (tau_mid / tau_start) ** f
        elif elapsed_frac < 0.8:
            f = (elapsed_frac - 0.4) / 0.4
            tau = tau_mid * (tau_end / tau_mid) ** f
        else:
            tau = tau_end

        a = softmax_forward(theta / theta_scale) * S
        value, g_a, c = compute_soft_max_conv_and_grad(a, tau)

        # Gradient w.r.t. theta (with normalization constraint)
        a_soft = softmax_forward(theta / theta_scale)
        inner = np.dot(a_soft, g_a)
        g theta = a_soft * (g_a - inner / S) * theta_scale

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 200 == 0:
            a_check = softmax_forward(theta / theta_scale) * S
            score = compute_score(a_check)
            if score < best_score - 1e-8:
                best_score = score
                best_theta = theta.copy()
                patience = 0
            else:
                patience += 200

            if verbose and step % 2000 == 0:
                print(f"      step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={elapsed:.1f}s")

            # Restart if stuck
            if patience > PATIENCE_LIMIT:
                if verbose:
                    print(f"      Restart at step {step}, best={best_score:.6f}")
                theta = best_theta + rng.normal(0, 0.5, size=n)
                m1 = np.zeros(n)
                m2 = np.zeros(n)
                patience = 0
                lr0 = max(0.5, lr0 * 0.7)

    a_final = softmax_forward(best_theta / theta_scale) * S
    score_final = compute_score(a_final)
    if score_final < best_score:
        best_score = score_final
        best_theta = theta.copy()
        a_final = softmax_forward(best_theta / theta_scale) * S
    return a_final, best_score


def peak_deflation(a, n_peaks=10, n_shrink=30, n_grow=30, strength=0.03, iterations=50, rng=None):
    """
    Explicit peak deflation: identify lags with high convolution values,
    reduce elements contributing most to those peaks, and redistribute to
    elements contributing least.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a)
    current = a.copy()
    current_score = compute_score(current)
    S = np.sum(current)

    for it in range(iterations):
        c = fft_convolve(current)
        # Get top lags
        top_lags = np.argsort(c)[-n_peaks:]

        # Compute contribution of each element to the peak lags
        contrib = np.zeros(n)
        for lag in top_lags:
            weight = c[lag] / (np.max(c) + 1e-10)
            i_min = max(0, lag - n + 1)
            i_max = min(n - 1, lag)
            for i in range(i_min, i_max + 1):
                j = lag - i
                contrib[i] += weight * current[i] * current[j]

        # Elements with high contribution get reduced, low contribution get increased
        top_idx = np.argsort(contrib)[-n_shrink:]
        bot_idx = np.argsort(contrib)[:n_grow]

        candidate = current.copy()
        candidate[top_idx] *= np.exp(rng.normal(-strength, strength * 0.3, size=len(top_idx)))
        candidate[bot_idx] *= np.exp(rng.normal(strength, strength * 0.3, size=len(bot_idx)))
        candidate = np.clip(candidate, 0.0, None)

        s = np.sum(candidate)
        if s > 1e-10:
            candidate = candidate / s * S

        new_score = compute_score(candidate)
        if new_score < current_score:
            current = candidate
            current_score = new_score
        else:
            # Try smaller step
            candidate2 = current.copy()
            candidate2[top_idx] *= np.exp(rng.normal(-strength * 0.3, strength * 0.1, size=len(top_idx)))
            candidate2[bot_idx] *= np.exp(rng.normal(strength * 0.3, strength * 0.1, size=len(bot_idx)))
            candidate2 = np.clip(candidate2, 0.0, None)
            s2 = np.sum(candidate2)
            if s2 > 1e-10:
                candidate2 = candidate2 / s2 * S
            new_score2 = compute_score(candidate2)
            if new_score2 < current_score:
                current = candidate2
                current_score = new_score2

    return current, current_score


def local_search_anneal(a_init, time_limit, rng, verbose=True):

            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.02, size=len(bot_idx)))
        elif r < 0.45:
            # Adjacent block swap
            width = max(1, min(n // 25, int(rng.integers(1, max(2, n // 15)))))
            if n > 2 * width:
                start = int(rng.integers(0, n - 2 * width))
                candidate[start:start + width] *= np.exp(rng.normal(-0.06, 0.02, size=width))
                candidate[start + width:start + 2 * width] *= np.exp(rng.normal(0.06, 0.02, size=width))
        elif r < 0.60:
            # Multi-block perturbation
            for _ in range(int(rng.integers(2, 4))):
                bw = max(1, int(rng.integers(1, max(2, n // 20))))

            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.1 * (sym - current)
            candidate += 0.005 * rng.standard_normal(n)
        elif r < 0.90:
            # Localized perturbation around peak-contributing regions
            c = fft_convolve(current)
            peak_lag = np.argmax(c)
            center = peak_lag // 2
            radius = max(5, n // 10)
            lo = max(0, center - radius)
            hi = min(n, center + radius)
            candidate[lo:hi] *= np.exp(rng.normal(0.0, 0.03, size=hi - lo))
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:

def upsample_seq(a, new_n, rng):
    """Upsample via interpolation + noise."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.02 * rng.standard_normal(new_n) * np.mean(a_interp)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)

"""
Optimized step function search minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses gradient descent with adaptive soft-max temperature, active peak tracking,
peak-deflation heuristics, and multi-restart strategy to improve upon 1.512372.

def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)

        return np.inf
    c = fft_convolve(a)

    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def multi_tone_init(n, rng, n_tones=4):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(n_tones):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.4 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def optimize_adam(n, a_init, time_limit, rng, verbose=True):
    """
    Adam gradient descent on softmax-parameterized sequence.
    Normalizes to sum = sqrt(2n) so score = max(conv).
    Uses adaptive temperature with plateau at the end and restart mechanism.
    """
    S = math.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S
    theta = np.log(a / S + 1e-15)
    theta_scale = 8.0

    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr0 = 2.5
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = compute_score(a)
    patience = 0
    PATIENCE_LIMIT = 5000

    tau_start = 60.0 / n
    tau_mid = 5.0 / n
    tau_end = 0.15 / n

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        elapsed_frac = min(1.0, elapsed / time_limit)

        if elapsed_frac < 0.35:
            f = elapsed_frac / 0.35
            tau = tau_start * (tau_mid / tau_start) ** f
        elif elapsed_frac < 0.75:
            f = (elapsed_frac - 0.35) / 0.40
            tau = tau_mid * (tau_end / tau_mid) ** f
        else:
            tau = tau_end

        a_norm = softmax_forward(theta * theta_scale)
        a = a_norm * S
        value, g_a, c = compute_soft_max_conv_and_grad(a, tau)

        inner = np.dot(a_norm, g_a)
        g_theta = a_norm * (g_a - inner / S) * theta_scale

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 300 == 0:
            a_check = softmax_forward(theta * theta_scale) * S
            score = compute_score(a_check)
            if score < best_score - 1e-9:
                best_score = score
                best_theta = theta.copy()
                patience = 0
            else:
                patience += 300

            if verbose and step % 3000 == 0:
                print(f"      step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={elapsed:.1f}s")

            if patience > PATIENCE_LIMIT:
                if verbose:
                    print(f"      Restart at step {step}, best={best_score:.6f}")
                theta = best_theta + rng.normal(0, 0.3 / theta_scale, size=n)
                m1 = np.zeros(n)
                m2 = np.zeros(n)
                patience = 0
                lr0 = max(0.3, lr0 * 0.6)

    a_final = softmax_forward(best_theta * theta_scale) * S
    score_final = compute_score(a_final)
    if score_final < best_score:
        best_score = score_final
    return a_final, best_score


def peak_deflation(a, time_limit, rng):
    """
    Explicit peak deflation with time constraint.
    """
    n = len(a)
    current = a.copy()
    current_score = compute_score(current)
    S = np.sum(current)
    t0 = time.monotonic()
    n_shrink = max(5, n // 12)
    n_grow = max(5, n // 12)

    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current)
        n_peaks = min(12, len(c))
        top_lags = np.argsort(c)[-n_peaks:]

        contrib = np.zeros(n)
        max_c = np.max(c)
        for lag in top_lags:
            weight = c[lag] / (max_c + 1e-10)
            i_min = max(0, lag - n + 1)
            i_max = min(n - 1, lag)
            for i in range(i_min, i_max + 1):
                j = lag - i
                contrib[i] += weight * current[i] * current[j]

        top_idx = np.argsort(contrib)[-n_shrink:]
        bot_idx = np.argsort(contrib)[:n_grow]

        strength = 0.02 * max(0.5, 1.0 - (time.monotonic() - t0) / time_limit)
        candidate = current.copy()
        candidate[top_idx] *= np.exp(rng.normal(-strength, strength * 0.4, size=len(top_idx)))
        candidate[bot_idx] *= np.exp(rng.normal(strength, strength * 0.4, size=len(bot_idx)))
        candidate = np.clip(candidate, 0.0, None)

        s = np.sum(candidate)
        if s > 1e-10:
            candidate = candidate / s * S

        new_score = compute_score(candidate)
        if new_score < current_score:
            current = candidate
            current_score = new_score

        iter_count += 1

    if verbose_getattr := True:
        pass
    return current, current_score, iter_count


def local_search_anneal(a_init, time_limit, rng, verbose=True):
    """
    Simulated annealing with diverse perturbation strategies.
    """
    current = a_init.astype(np.float64).copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9998
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        iter_count += 1

        if r < 0.30:
            # Peak-directed
            c = fft_convolve(current)
            n_top = min(8, len(c))
            top_lags = np.argsort(c)[-n_top:]
            contrib = np.zeros(n)
            max_c = np.max(c)
            for lag in top_lags:
                w = c[lag] / (max_c + 1e-10)
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += w * current[i] * current[j]
            n_red = max(3, n // 15)
            top_idx = np.argsort(contrib)[-n_red:]
            bot_idx = np.argsort(contrib)[:n_red]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.04, 0.015, size=len(bot_idx)))
        elif r < 0.45:
            # Adjacent block
            width = max(1, min(n // 25, int(rng.integers(1, max(2, n // 15)))))
            if n > 2 * width:
                start = int(rng.integers(0, n - 2 * width))
                candidate[start:start + width] *= np.exp(rng.normal(-0.05, 0.02, size=width))
                candidate[start + width:start + 2 * width] *= np.exp(rng.normal(0.05, 0.02, size=width))
        elif r < 0.60:
            # Multi-block
            for _ in range(int(rng.integers(2, 4))):
                bw = max(1, int(rng.integers(1, max(2, n // 20))))
                bs = int(rng.integers(0, n - bw))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0.0, 0.035, size=bw))
        elif r < 0.75:
            # Symmetrization
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.08 * (sym - current)
            candidate += 0.004 * rng.standard_normal(n)
        elif r < 0.90:
            # Peak-lag localized
            c = fft_convolve(current)
            peak_lag = int(np.argmax(c))
            center = peak_lag // 2
            radius = max(5, n // 10)
            lo = max(0, center - radius)
            hi = min(n, center + radius)
            candidate[lo:hi] *= np.exp(rng.normal(0.0, 0.025, size=hi - lo))
        else:
            candidate = current * np.exp(rng.normal(0.0, 0.008, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current
        else:
            candidate = candidate / s * np.sum(current)

        new_score = compute_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 30:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score

        temp *= cooling

    if verbose and iter_count > 0:
        elapsed = time.monotonic() - t0
        print(f"    Local search: {iter_count} iters in {elapsed:.0f}s, best={best_score:.6f}")
    return best, best_score


def upsample_seq(a, new_n, rng):
    """Upsample via interpolation + noise."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    noise_scale = 0.02 * np.mean(a_interp)
    a_interp += noise_scale * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 1e-6, None)
    return a_interp / np.sum(a_interp)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for optimal non-negative step function minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    Phases: diverse init -> gradient descent -> peak deflation -> 
            local search -> restarts -> upsample attempts.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = compute_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # === Phase 1: Diverse initialization ===
    print("\n=== Phase 1: Initialization (remaining={:.0f}s) ===".format(deadline - time.monotonic()))
    candidates = []

    lengths_to_try = [256, 384, 512]
    for n in lengths_to_try:
        if time.monotonic() > deadline - 600:
            break
        inits = []
        inits.append(('fft_a', fft_phase_init(n, rng, n_iter=100)))
        inits.append(('fft_b', fft_phase_init(n, rng, n_iter=200)))
        inits.append(('chirp', chirp_init(n, rng, alpha=0.3 + 0.4 * rng.random())))
        inits.append(('multi', multi_tone_init(n, rng, n_tones=4 + rng.integers(0, 3))))

        if start_seq is not None:
            if len(start_seq) == n:
                inits.append(('given', start_seq.copy() / np.sum(start_seq)))
            else:
                inits.append(('given_up', upsample_seq(start_seq, n, rng)))

        for name, a in inits:
            score = compute_score(a)
            print(f"  n={n}, {name}: score={score:.6f}")
            candidates.append((score, n, a.copy(), name))

    candidates.sort(key=lambda x: x[0])
    print(f"  Top 5: {[(f'{c[0]:.4f}', c[1], c[3]) for c in candidates[:5]]}")

    # === Phase 2: Gradient optimization ===
    print("\n=== Phase 2: Gradient optimization ===")
    n_cand = min(5, len(candidates))
    time_for_grads = max(300.0, (deadline - time.monotonic() - 500) / 1.0)
    time_per_cand = time_for_grads / n_cand

    optimized = []
    for i in range(n_cand):
        if time.monotonic() > deadline - 300:
            break
        score0, n, a, name = candidates[i]
        t_alloc = time_per_cand * min(1.3, 384.0 / max(n, 1))
        t_alloc = max(40.0, min(t_alloc, deadline - time.monotonic() - 80))

        print(f"  [{i+1}/{n_cand}] n={n}, init={name}, score0={score0:.6f}, t={t_alloc:.0f}s")
        a_opt, s_opt = optimize_adam(n, a, t_alloc, rng, verbose=(i < 2))
        print(f"    -> score={s_opt:.6f}")

        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")

        optimized.append((s_opt, n, a_opt))

    # === Phase 3: Peak deflation ===
    print(f"\n=== Phase 3: Peak deflation (best={best_score:.6f}) ===")
    if best_seq is not None:
        time_left = deadline - time.monotonic() - 200
        if time_left > 30:
            t_def = min(150.0, time_left)
            print(f"  Deflating, t={t_def:.0f}s")
            deflated, d_score, n_iter = peak_deflation(best_seq, t_def, rng)
            print(f"  After deflation: score={d_score:.6f} ({n_iter} iterations)")
            if d_score < best_score:
                best_score = d_score
                best_seq = deflated.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: Local search ===
    print(f"\n=== Phase 4: Local search (best={best_score:.6f}) ===")
    if best_seq is not None:
        time_left = deadline - time.monotonic() - 100
        if time_left > 30:
            t_ls = min(250.0, time_left)
            print(f"  Polishing n={len(best_seq)}, t={t_ls:.0f}s")
            polished, p_score = local_search_anneal(best_seq, t_ls, rng)
            print(f"  After polish: score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = polished.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 5: Additional optimization rounds ===
    time_left = deadline - time.monotonic() - 40
    if time_left > 80 and best_seq is not None:
        print(f"\n=== Phase 5: Additional rounds (remaining={time_left:.0f}s) ===")
        n_best = len(best_seq)
        n_rounds = min(3, int(time_left // 100))
        for r in range(n_rounds):
            if time.monotonic() > deadline - 40:
                break
            a_new = best_seq * np.exp(rng.normal(0, 0.06, size=n_best))
            a_new = np.clip(a_new, 1e-6, None)
            t_alloc = min(70.0, (deadline - time.monotonic() - 35) / max(1, n_rounds - r))
            if t_alloc < 20:
                break
            a_opt, s_opt = optimize_adam(n_best, a_new, t_alloc, rng, verbose=False)
            print(f"  Round {r+1}: score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"  *** New best: {best_score:.6f} ***")
            # Quick local search
            if deadline - time.monotonic() > 50:
                p2, s2 = local_search_anneal(best_seq, min(40.0, deadline - time.monotonic() - 35), rng, verbose=False)
                if s2 < best_score:
                    best_score = s2
                    best_seq = p2.copy()
                    print(f"    After polish: {best_score:.6f}")

    # === Phase 6: Try larger n ===
    time_left = deadline - time.monotonic() - 30
    if time_left > 80 and best_seq is not None:
        n_cur = len(best_seq)
        for target_n in [n_cur * 2, n_cur + 128]:
            if target_n > 4096 or target_n <= n_cur:
                continue
            if time.monotonic() > deadline - 50:
                break
            a_up = upsample_seq(best_seq, target_n, rng)
            t_alloc = min(80.0, deadline - time.monotonic() - 40)
            if t_alloc < 30:
                break
            print(f"\n  Upsampling to n={target_n}, t={t_alloc:.0f}s")
            a_opt, s_opt = optimize_adam(target_n, a_up, t_alloc, rng, verbose=False)
            print(f"    Score at n={target_n}: {s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} (n={target_n}) ***")

    # Final
    print(f"\n{'='*60}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
