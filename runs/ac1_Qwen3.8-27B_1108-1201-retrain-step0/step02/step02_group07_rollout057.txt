```python
import numpy as np
import time
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_value(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()

def lp_iteration(f, M, solver=cp.GLOP):
    """
    Solve the LP: maximize sum(g) subject to g >= 0 and conv(f,g) <= M (componentwise).
    Returns the rescaled g with sum(g) = sum(f).
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    
    # Build convolution constraints
    # conv(f,g)[k] = sum_j f[j]*g[k-j] for valid j
    # We need conv(f,g)[k] <= M for all k
    
    constraints = []
    for k in range(2 * n - 1):
        # conv(f,g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j]*g[k-j]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # terms: f[j] * g[k-j] for j in [js, je]
        # g indices: k-j for j in [js, je], i.e., g[k-je] ... g[k-js]
        coeffs = f[js:je+1]  # f[js], f[js+1], ..., f[je]
        g_indices = np.arange(k - je, k - js + 1)  # k-je, k-je+1, ..., k-js
        constraints.append(cp.sum(coeffs * g[g_indices]) <= M)
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver)
    
    if g.value is None:
        return f.copy()  # Return f as fallback
    
    g_val = np.maximum(g.value, 0)
    s_g = g_val.sum()
    if s_g < 1e-10:
        return f.copy()
    
    # Rescale to match sum(f)
    target_sum = f.sum()
    g_val = g_val * (target_sum / s_g)
    return g_val
```

```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

    result = np.fft.irfft(A * B, fft_size)
    return result[:size]
```

```python
"""
LP-based optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal mixing directions that respect
correlation budget constraints, combined with diverse initializations,
perturbation-restart cycles, and gradient-based polishing.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def build_lp_matrix(f):
    """Build sparse (2n-1) x n matrix A such that A @ g = conv(f, g)."""
    n = len(f)
    nrows = 2 * n - 1
    row_idx = []
    col_idx = []
    data = []
    for j in range(n):
        length = min(n, nrows - j)
        for k in range(length):
            row_idx.append(j + k)
            col_idx.append(j)
            data.append(f[k])
    return csr_matrix((data, (row_idx, col_idx)), shape=(nrows, n))


def lp_step(f, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, conv(f,g) <= M (componentwise).
    Returns rescaled g with sum(g) = sum(f).
    """
    n = len(f)
    c = -np.ones(n)
    A_ub = build_lp_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return f.copy()

    if res.status != 0 or res.x is None:
        return f.copy()

    g = np.maximum(res.x, 0.0)
    s_g = g.sum()
    if s_g < 1e-10:
        return f.copy()

    target_sum = f.sum()
    g = g * (target_sum / s_g)
    return g


def lp_iteration(f, max_iters=1000, tol=1e-10, time_limit=None):
    """
    Run LP-based iterations to reduce max(conv(f,f)).
    f should be non-negative with positive sum.
    """
    n = len(f)
    f = np.maximum(f, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_M = M
    t0 = time.monotonic()
    stagnation = 0

    for it in range(max_iters):
        if time_limit is not None and (time.monotonic() - t0) > time_limit:
            break

        g = lp_step(f, M)
        if g is None or not np.all(np.isfinite(g)):
            break

        # Line search
        best_t = 0.0
        best_new_M = M
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.01]:
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            if f_new.sum() < 1e-10:
                continue
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            if M_new < best_new_M:
                best_new_M = M_new
                best_t = t

        if best_t == 0.0:
            stagnation += 1
            if stagnation > 5:
                break
            # Perturb and retry
            f = f + 0.001 * np.random.standard_normal(n)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            continue

        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 20:
                break

        if best_M - M < tol:
            break

    return best_f


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(conv, a, t=50.0):
    """Smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = t * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    thresh = w.max() * 0.002
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    return sg

def anneal_polish(a, time_limit, seed=123):
    """Informed annealing polish targeting peak reduction."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    n = len(a)

    best_a = a.copy()

    while (time.monotonic() - t0) < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.01 * (1.0 - frac) ** 2 + 0.0003
        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.8
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        target_j = int(rng.integers(n))
        if target_j == top_j:
            continue
        delta = a[top_j] * rng.uniform(0.01, 0.05)
        if delta < 1e-10:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:

                a[target_j] -= delta

    return best_a

def generate_init(name, n, rng, parent=None):
    """Generate a specific initialization."""
    x = np.arange(n, dtype=np.float64)
    if name == 'uniform':
        return np.ones(n) / n
    elif name == 'gauss':
        g = np.exp(-0.5 * ((x - n * 0.5) / (0.3 * n)) ** 2)
        return g / g.sum()
    elif name == 'sin1':
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
        return s / s.sum()
    elif name == 'sin2':
        s = 0.5 * (1.0 + np.cos(4.0 * np.pi * x / n))
        s = np.maximum(s, 0.0)
        if s.sum() < 1e-10:
            return np.ones(n) / n
        return s / s.sum()
    elif name == 'ramp':
        r = np.arange(1, n + 1, dtype=np.float64)
        return r / r.sum()
    elif name == 'bimodal':
        bi = np.zeros(n)
        w = max(1, n // 6)
        bi[:w] = 1.0
        bi[-w:] = 1.0
        return bi / bi.sum()
    elif name == 'rand':
        return rng.dirichlet(np.ones(n) * 0.5)
    elif name == 'chirp':
        chirp = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * x / (2.0 * n)))
        chirp = np.maximum(chirp, 0.0)
        if chirp.sum() < 1e-10:
            return np.ones(n) / n
        return chirp / chirp.sum()
    elif name == 'parent':
        if parent is not None and len(parent) == n:
            pp = np.maximum(parent, 0.0).astype(np.float64)
            pp = pp / pp.sum()
            return pp
        return None
    elif name == 'parent_interp':
        if parent is not None:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                return pp / pp.sum()
        return None
    else:
        return np.ones(n) / n

    scan_results.sort(key=lambda x: x[0])

    # Phase 2: Deep LP refinement of top candidates
    print("\n=== Phase 2: Deep LP refinement ===")
    top_configs = []
    seen = set()
    for obj, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 8:
            top_configs.append((n, name, a_opt))
            seen.add(key)

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    phase2_time = min(800.0, (deadline - time.monotonic()) * 0.4)
    n_deep = len(top_configs)
    time_per_deep = max(5.0, phase2_time / max(1, n_deep))

    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 300:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 300)
        if t_alloc < 3.0:
            break
        print("  [%d/%d] n=%d, %s, budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))

        a_cur = a0.copy()

        # Multiple LP passes with perturbation
        sub = t_alloc / 4.0
        for p in range(4):
            if time.monotonic() > deadline - 300:
                break
            a_cur = lp_iteration(a_cur, max_iters=5000, tol=1e-12, time_limit=sub)
            record(a_cur)
            # Perturb for next pass
            a_cur = a_cur + 0.002 * rng.standard_normal(n)
            a_cur = np.maximum(a_cur, 0.0)
            a_cur = a_cur / a_cur.sum()

    print("  Phase 2 done. Best: %.6f" % best_score)

    # Phase 3: Gradient polishing
    print("\n=== Phase 3: Gradient polishing ===")
    if best_sequence is not None and time.monotonic() < deadline - 100:
        remaining = deadline - time.monotonic() - 100
        a_best = np.maximum(best_sequence, 0.0).astype(np.float64)
        a_best = a_best / a_best.sum()
        n_best = len(a_best)

        polish_time = remaining / 4.0

        # Gradient polish
        a_opt = gradient_polish(a_best, time_limit=polish_time, t_init=80.0, lr_init=0.04)
        record(a_opt)

        if time.monotonic() < deadline - 80:
            a_opt = mirror_polish(a_best, time_limit=polish_time, eta_init=0.08)
            record(a_opt)

        if time.monotonic() < deadline - 60:
            a_opt = anneal_polish(a_best, time_limit=polish_time, seed=seed + 777)
            record(a_opt)

        # LP + polish combo
        if time.monotonic() < deadline - 40:
            a_pert = a_best + 0.003 * rng.standard_normal(n_best)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            a_opt = lp_iteration(a_pert, max_iters=3000, tol=1e-12,
                                 time_limit=min(polish_time, deadline - time.monotonic() - 30))
            record(a_opt)

    print("  Phase 3 done. Best: %.6f" % best_score)

    # Phase 4: Final perturbation-restart cycles
    print("\n=== Phase 4: Final refinement ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            n_cycles = min(6, int(remaining / 20))
            cycle_time = remaining / max(1, n_cycles)
            for i in range(n_cycles):
                if time.monotonic() > deadline - 10:
                    break
                a_start = np.maximum(best_sequence, 0.0).astype(np.float64)
                a_start = a_start / a_start.sum()
                # Different perturbation strengths
                strengths = [0.01, 0.02, 0.005, 0.015, 0.03, 0.008]
                a_pert = a_start + strengths[i % len(strengths)] * rng.standard_normal(len(a_start))
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                t_alloc = min(cycle_time, deadline - time.monotonic() - 10)
                if t_alloc < 3:
                    break
                a_opt = lp_iteration(a_pert, max_iters=5000, tol=1e-12, time_limit=t_alloc)
                record(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = safe_evaluate(best_sequence)

    print("\nFINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))
    print("Total time: %.1fs" % (time.monotonic() - t_start))

    return [float(x) for x in best_sequence]
```
