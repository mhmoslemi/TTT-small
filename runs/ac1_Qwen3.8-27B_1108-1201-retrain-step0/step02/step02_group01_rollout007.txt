Minimize 2*n*max(conv(a,a))/sum(a)^2 using:

2. Barzilai-Borwein adaptive step sizes with backtracking
3. Periodic LP-based direction computation and blending

6. Periodic random perturbations (annealed) to escape local minima

Key innovations over previous approach:

- Nesterov momentum for faster convergence
- BB step sizes for better convergence rate
- Reduced-constraint LP for periodic "jumps"

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]

def fft_grad_from_weights(w, a):
    """grad[j] = 2 * sum_k w[k] * a[k-j] for j=0..n-1."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1 - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    cc = np.real(ifft(np.conj(A) * W))[:n]
    return 2.0 * cc

def softmax_weights(c, T):
    c_max = c.max()
    if c_max < 1e-30:
        return np.ones(len(c)) / len(c)
    if T > 1e8:
        idx = np.where(c > c_max * 0.999)[0]
        w = np.zeros(len(c))
        w[idx] = 1.0 / len(idx)
        return w
    x = T * (c - c_max) / c_max
    x = x - x.max()
    e = np.exp(x)
    total = e.sum()
    if total < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / total

def compute_objective(a):
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_conv_self(a)
    return 2.0 * len(a) * c.max() / (S * S)

    if s < 1e-30:

def run_subgradient(a_init, time_budget, T_start=3.0, T_end=3000.0,
                    momentum_beta=0.85, initial_lr=0.5, name=""):

    Subgradient descent with Nesterov momentum, BB step sizes, and backtracking.
    Optimizes max(conv(a,a)) subject to a>=0, sum(a)=S.
    """

    S = np.sqrt(2.0 * n)
    a = normalize_a(a_init.copy(), S)
    c = fft_conv_self(a)
    best_M = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    momentum = np.zeros(n)
    look-ahead = a.copy()  # Nesterov look-ahead
    
    # BB state
    prev_grad = None
    prev_a = a.copy()
    bb_lr = initial_lr
    
    # Backtracking state
    bt_factor = 1.0
    stagnant = 0

        T = T_base * (0.5 + 0.5 * cycle)
        T = max(T, T_start)

        a_la = normalize_a(look-ahead, S)
        c_la = fft_conv_self(a_la)

        grad_la = fft_grad_from_weights(w, a_la)
        grad_la_c = grad_la - grad_la.mean()

            s = a - prev_a
            y = grad_la_c - prev_grad

                    else:
                        bb_lr = sy / yy

def run_lp_direction(f, M, n, max_constraints=200, time_limit=8):
    """Solve reduced LP: max sum(g) s.t. conv(f,g)[k] <= M for top-k lags, g>=0."""
    from scipy.optimize import linprog
    
    c = fft_conv_self(f)
    # Select active constraints
    if max_constraints < len(c):
        thresh = np.sort(c)[::-1][min(max_constraints, len(c)) - 1]
        active = np.where(c >= thresh)[0]
    else:
        active = np.arange(len(c))
    
    # Build constraint matrix
    A = np.zeros((len(active), n))
    for i, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[k - lo: k - hi + 1]
    
    c_obj = -np.ones(n)
    b = np.full(len(active), M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def coordinate_polish(a, time_budget, max_inner=20):

    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_a(a.copy(), S)
    c = fft_conv_self(a)
    best_M = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    # FPS (fast peak search): identify peaks and their contributing indices
    peak_thresh = 0.95 * best_M
    peak_lags = np.where(c >= peak_thresh)[0]
    
    for sweep in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        # Focus on indices that contribute to peaks
        for lag in peak_lags[:50]:  # top peaks only
            lo = max(0, lag - n + 1)
            hi = min(n - 1, lag)
            for j in range(lo, hi + 1):
                if time.monotonic() - start > time_budget:
                    break
                orig = a[j]
                if orig < 1e-12:
                    continue
                # Try reducing a[j]
                for frac in [0.5, 0.1, -0.1, -0.5]:
                    a[j] = max(0.0, orig * (1 + frac))
                    a_temp = normalize_a(a, S)
                    c_temp = fft_conv_self(a_temp)
                    if c_temp.max() < best_M - 1e-12:
                        best_M = c_temp.max()
                        best_a = a_temp.copy()
                        a = a_temp
                        c = c_temp
                        improved = True
                        break
                    a[j] = orig
                if improved:
                    peak_thresh = 0.95 * best_M
                    peak_lags = np.where(c >= peak_thresh)[0]
        if not improved:
            break
    
    return best_a, best_M

    elif name == "gaussian":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n / 2) / (n / 5)) ** 2) + 0.1
        return normalize_a(a, S)
    elif name == "cosine":
        i = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * 2 * i / n)
        a = np.maximum(a, 0.05)
        return normalize_a(a, S)
    elif name == "alternating":
        i = np.arange(n)
        a = 1.0 + 0.3 * np.cos(np.pi * i)
        a = np.maximum(a, 0.05)
        return normalize_a(a, S)
    elif name == "two_tone":
        i = np.arange(n)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 3 * i / n) + 0.2 * np.cos(2 * np.pi * 7 * i / n)
        a = np.maximum(a, 0.05)
        return normalize_a(a, S)
    elif name == "sawtooth":
        i = np.arange(n)
        a = np.abs(np.sin(np.pi * i / n))
        a = np.maximum(a, 0.05)
        return normalize_a(a, S)
    else:
        return np.full(n, S / n)

    """Main search function."""
    rng = np.random.default_rng(seed)

            given_a = given_raw.copy()
            n_g = len(given_a)
            S_g = np.sqrt(2.0 * n_g)
            given_a = normalize_a(given_a, S_g)
            obj_g = compute_objective(given_a)
            print("Given: n=%d obj=%.6f" % (n_g, obj_g))
            if obj_g < global_best_obj:
