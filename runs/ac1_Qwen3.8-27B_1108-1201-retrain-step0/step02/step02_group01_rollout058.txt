Actually, here's another idea: **use the "Frank-Wolfe" (conditional gradient) method**. The Frank-Wolfe method for convex minimization over a polytope works by:
1. Compute the gradient (or subgradient) `g` at the current point
2. Find the vertex `v` of the polytope that minimizes `<g, v>` (the "linear oracle")

For our problem, the "polytope" is the simplex `{a >= 0, sum(a) = S}`, and the vertices are `S * e_j` (unit vectors scaled by S). The linear oracle would choose `v = S * e_j` where `j = argmin g[j]`.

The step size for Frank-Wolfe is `gamma_t = 2/(t+2)` (for the "standard" variant) or `gamma_t = 1/(t+1)` (for the "pairwise" variant).

The advantage of Frank-Wolfe is that it's very simple and doesn't require a projection step (the "vertex" is always a basis vector, so the update is just adding a spike to one coordinate).

def solve_lp_direction(a, M, n, top_k_constraints=50, time_limit=15):
    """LP: max sum(g) s.t. (a*g)[k] <= M for top-k lags, g >= 0.
    Returns normalized g or None.
    """
    c = fft_convolve_self(a)
    if top_k_constraints < len(c):
        active_lags = np.argsort(c)[::-1][:top_k_constraints]
    else:
        active_lags = np.arange(len(c))

    # Build constraint matrix
    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[idx, j] = a[k - j]

    c_obj = -np.ones(n)
    b_ub = np.full(len(active_lags), M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

            # (where a[k-i] is the value with a[i]=0, and k-i != i for the linear term)

            # Build the "affine" part: for each k, contribution = alpha_k * x + beta_k * x^2
            # alpha_k = 2*a[k-i] if 0<=k-i<n and k-i != i
            #         = a[k-i] if k-i == i (i.e., k==2i, but then we also have the x^2 term)
            # Actually: with a[i]=0, b_base[k] = sum_{j != i} a[j]*a[k-j] (with valid indices)
            # Setting a[i]=x: b[k] = b_base[k] + x*a[k-i] + a[i]*x + (x^2 if 2i==k and k-i==i)
            # Wait: b[k] = sum_j a[j]*a[k-j]. Terms with a[i]:
            #   j=i: a[i]*a[k-i] = x*a[k-i] (if 0<=k-i<n and k-i != i, since a[i]=0 in base)
            #   k-j=i, j=k-i: a[k-i]*a[i] = a[k-i]*x (if 0<=k-i<n and k-i != i)
            #   j=i and k-j=i (i.e., 2i=k): a[i]*a[i] = x^2
            #
            # So: b[k] = b_base[k] + 2*x*a[k-i] (if 0<=k-i<n and k-i != i)
            #             + x*a[k-i] + a[k-i]*x (if 0<=k-i<n and k-i == i, i.e. k==2i)
            #             Wait, if k-i==i, then the term j=i gives x*a[i]=x^2, and the term
            #             k-j=i gives a[i]*x=x^2. But these are the SAME term (j=i, k-j=i),
            #             so it's just x^2, not 2x^2.
            #
            # Let me redo: b[k] = sum_j a[j]*a[k-j]
            # With a[i]=x, a[j]=0 for j=i (in the base), the additional terms are:
            #   j=i: x * a[k-i] (where a[k-i] is the value in base, i.e., with a[i]=0)
            #   j=k-i (if k-i != i and 0<=k-i<n): a[k-i] * x
            # If k-i == i (k==2i): only one term x*x = x^2
            # If k-i != i: two terms x*a[k-i] + a[k-i]*x = 2x*a[k-i]
            #
            # So: b[k] = b_base[k] + 2*x*a_base[k-i] if 0<=k-i<n and k-i!=i
            #              b_base[k] + x^2 if k==2i
            #              b_base[k] otherwise

            # For the objective max_k b[k], we need to find x >= 0 minimizing
            # max_k (b_base[k] + delta_k(x))
            # where delta_k(x) = 2*x*a_base[k-i] (linear) or x^2 (if k==2i) or 0

            # The function is convex in x. Find x* where the subgradient contains 0.
            # Subgradient at x: {2*x (if k==2i is active) + 2*sum_{k active, k!=2i} w_k * a_base[k-i]}
            # At optimum: 2*x + 2*sum_{k in active, k!=2i} a_base[k-i] * w_k = 0 (if x>0)
            # But this is hard to solve directly. Use bisection.

            # Binary search for optimal x in [0, 3*original]
            lo_x = 0.0
            hi_x = 3.0 * original
            best_x = original
            best_x_obj = np.inf

            for _ in range(14):
                mid_x = (lo_x + hi_x) / 2.0
                # Compute objective at x = mid_x
                # b[k] = b_base[k] + 2*mid_x*a_base[k-i] (for k where 0<=k-i<n, k!=2i)
                #       + b_base[k] + mid_x^2 (for k=2i)
                b_test = b_base.copy()
                # Add linear terms
                lo_k = max(0, i)
                hi_k = min(2 * n - 2, i + n - 1)
                # k-i ranges from max(0-i, 0) to min(n-1-i, 2n-2-i)
                # Valid k: 0 <= k-i < n and k-i != i, i.e., k != 2i
                # k ranges from max(0, i) to min(2n-2, i+n-1)
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                valid_ks = np.arange(k_lo, k_hi + 1)
                # Exclude k = 2i if in range
                mask = valid_ks != 2 * i
                if np.any(mask):
                    ks = valid_ks[mask]
                    b_test[ks] += 2.0 * mid_x * a_base[ks - i]
                # Add quadratic term at k=2i
                if 0 <= 2 * i < 2 * n - 1:
                    b_test[2 * i] += mid_x * mid_x

                obj_test = b_test.max()
                if obj_test < best_x_obj:
                    best_x_obj = obj_test
                    best_x = mid_x

                # Determine direction: is the objective increasing or decreasing?
                # Check at a slightly larger x
                if lo_x < hi_x - 1e-12:
                    # Evaluate at mid_x * 1.01 to determine direction
                    x_check = min(mid_x * 1.5, hi_x) if mid_x > 0 else hi_x * 0.5
                    b_chk = b_base.copy()
                    ks2 = valid_ks[mask]
                    if np.any(mask):
                        b_chk[ks2] += 2.0 * x_check * a_base[ks2 - i]
                    if 0 <= 2 * i < 2 * n - 1:
                        b_chk[2 * i] += x_check * x_check
                    obj_chk = b_chk.max()
                    if x_check > mid_x:
                        if obj_chk > obj_test:
                            hi_x = x_check
                        else:
                            lo_x = mid_x
                    else:
                        if obj_chk < obj_test:
                            lo_x = x_check
                        else:
                            hi_x = mid_x

            a[i] = best_x

            # Renormalize
            s_now = a.sum()
            if s_now < 1e-10:
                a = np.full(n, S / n)
            else:
                a = a * (S / s_now)

            c = fft_convolve_self(a)
            if c.max() < best_obj - 1e-10:
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        if not improved and sweep > 2:
            break

    return best_a, best_obj


def get_initialization(name, n, rng, S):
    """Generate various initializations."""
    i = np.arange(n)
    if name == "uniform":
        a = np.ones(n)
    elif name == "random":
        a = rng.random(n) + 0.01
    elif name == "gaussian":
        a = np.exp(-0.5 * ((i - n / 2) / (n / 6)) ** 2)
        a = np.maximum(a, 1e-6)
    elif name == "sinc":
        center = (n - 1) / 2.0
        sigma = n / 8.0
        a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
        a = np.maximum(a, 1e-6)
    elif name == "u_shape":
        t = i / max(n - 1, 1)
        a = 1.0 + 2.0 * (2 * t - 1) ** 2
        a = np.maximum(a, 0.01)
    elif name == "w_shape":
        t = i / max(n - 1, 1)
        a = 1.0 + 0.5 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 0.01)
    elif name == "cosine_taper":
        t = i / max(n - 1, 1)
        a = 0.5 * (1 + np.cos(np.pi * t)) ** 0.5
        a = np.maximum(a, 0.01)
    elif name == "bell":
        t = i / max(n - 1, 1)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 0.01)
    elif name == "two_gauss":
        a = np.exp(-0.5 * ((i - n * 0.3) / (n / 10)) ** 2) + \
            np.exp(-0.5 * ((i - n * 0.7) / (n / 10)) ** 2)
        a = np.maximum(a, 1e-6)
    else:
        a = np.ones(n)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    global_best = np.inf
    global_a = None

    # Try to load the given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw.copy()
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given sequence: n=%d, obj=%.8f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given sequence: %s" % str(e))

    # Phase 1: Mirror descent exploration (35% of time)
    explore_budget = 0.30 * (deadline - time.monotonic())
    print("Phase 1: Exploration, budget=%.1fs" % explore_budget)

    lengths_to_try = [4096]
    if given_a is not None:
        lengths_to_try.insert(0, len(given_a))
    # Add a couple more sizes
    if 2048 not in lengths_to_try:
        lengths_to_try.append(2048)
    if 8192 not in lengths_to_try:
        lengths_to_try.append(8192)

    inits = ["sinc", "u_shape", "gaussian", "cosine_taper", "uniform", "w_shape",
             "bell", "two_gauss", "random"]

    per_attempt = max(2.0, explore_budget / (len(lengths_to_try) * len(inits)))
    per_attempt = min(per_attempt, 8.0)

    for n in lengths_to_try:
        if time.monotonic() > deadline - 60.0:
            break
        S = np.sqrt(2.0 * float(n))

        for init_name in inits:
            if time.monotonic() > deadline - 60.0:
                break
            if per_attempt < 1.5:
                break

            a_init = get_initialization(init_name, n, rng, S)
            a_best, obj_best = run_mirror_descent(
                a_init, S, per_attempt, eta0=0.3 / np.sqrt(n),
                beta_mom=0.95, top_k=1, use_avg=True
            )
            True_obj = eval_obj(a_best)
            if True_obj < global_best:
                global_best = True_obj
                global_a = a_best.copy()
                print("  Explore: n=%d init=%s obj=%.8f" % (n, init_name, True_obj))

    print("After Phase 1: best=%.8f" % global_best)

    # Phase 2: Mirror descent with more iterations on best (25% of remaining)
    if global_a is not None:
        exploit_budget = 0.25 * (deadline - time.monotonic())
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a = global_a.copy()
        a = np.maximum(a, 1e-15)
        a = a * (S / a.sum())
        print("Phase 2: Mirror descent refinement, budget=%.1fs, n=%d" % (exploit_budget, n))

        # Try different momentum and top_k settings
        configs = [
            {'eta0': 0.2 / np.sqrt(n), 'beta': 0.97, 'top_k': 1},
            {'eta0': 0.5 / np.sqrt(n), 'beta': 0.93, 'top_k': 3},
            {'eta0': 0.1 / np.sqrt(n), 'beta': 0.98, 'top_k': 1},
            {'eta0': 0.3 / np.sqrt(n), 'beta': 0.95, 'top_k': 5},
        ]
        per_config = exploit_budget / len(configs)

        for cfg in configs:
            if time.monotonic() > deadline - 30.0:
                break
            a_best, obj_best = run_mirror_descent(
                a, S, per_config, eta0=cfg['eta0'],
                beta_mom=cfg['beta'], top_k=cfg['top_k'], use_avg=True
            )
            True_obj = eval_obj(a_best)
            if True_obj < global_best:
                global_best = True_obj
                global_a = a_best.copy()
                a = a_best.copy()
                a = np.maximum(a, 1e-15)
                a = a * (S / a.sum())
                print("  MD refine: eta0=%.4f beta=%.2f top_k=%d obj=%.8f" %
                      (cfg['eta0'], cfg['beta'], cfg['top_k'], True_obj))

    # Phase 3: L-BFGS on soft-max (20% of remaining)
    if global_a is not None:
        lbfgs_budget = 0.20 * (deadline - time.monotonic())
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a = global_a.copy()
        a = np.maximum(a, 1e-15)
        a = a * (S / a.sum())
        theta = np.log(a / a.sum())
        print("Phase 3: L-BFGS soft-max refinement, budget=%.1fs" % lbfgs_budget)

        temps = [50, 100, 200, 500]
        per_temp = lbfgs_budget / len(temps)

        for T in temps:
            if time.monotonic() > deadline - 20.0:
                break
            a_best, obj_best = run_lbfgs_refine_impl(
                theta, n, T, S, per_temp, max_iter=200
            )
            if a_best is not None:
                True_obj = eval_obj(a_best)
                if True_obj < global_best:
                    global_best = True_obj
                    global_a = a_best.copy()
                    print("  L-BFGS T=%d: obj=%.8f" % (T, True_obj))

        lp_budget = 0.10 * (deadline - time.monotonic())
        if lp_budget > 5.0:
            n = len(global_a)
            S = np.sqrt(2.0 * float(n))
            a = global_a.copy()
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            M = c.max()
            print("Phase 4: LP direction, budget=%.1fs" % lp_budget)

            g = solve_lp_direction(a, M, n, top_k_constraints=40,
                                   time_limit=min(10, lp_budget - 2))
            if g is not None:
                for t_blend in [0.2, 0.4, 0.6, 0.8]:
                    a_new = (1 - t_blend) * a + t_blend * g * S
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 0.01:
                        a_new = a_new * (S / s_new)
                        obj_new = eval_obj(a_new)
                        if obj_new < global_best:
                            global_best = obj_new
                            global_a = a_new.copy()
