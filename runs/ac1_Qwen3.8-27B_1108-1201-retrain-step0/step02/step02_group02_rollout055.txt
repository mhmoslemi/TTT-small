import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.sum(np.arange(1, n+1) * u > cssv)
    # Actually, use the standard algorithm
    theta = (cssv[rho-1] - 1) / rho if rho > 0 else v.max()
    return np.maximum(v - theta, 0)

def soft_max_value_and_grad(a, tau):
    """Compute soft-max value and gradient of the self-convolution."""
    n = len(a)
    c = fft_convolve(a, a)
    # Soft-max
    c_max = np.max(c)
    shifted = tau * (c - c_max)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + np.log(sum_exp) / tau
    
    # Gradient: grad_j = (4n/tau) * sum_m w[j+m] * a[m]
    # This is cross-correlation of w and a
    L = 1
    while L < 2*n:
        L <<= 1
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    # cross_corr[j] = sum_m w[j+m] * a[m] = ifft(FFT(w) * conj(FFT(a)))[j]
    cross = np.fft.irfft(W * np.conj(A), L)[:n]
    grad = (4.0 * n / tau) * cross
    
    return value, grad, c, w

Hybrid spectral-gradient optimization for minimizing peak autocorrelation ratio.
Uses soft-max smoothed objectives with L-BFGS and projected gradient descent,
multi-n exploration, diverse initializations (geometric, linear, spectral),
and intensive local polish (coordinate descent + block moves + peak shaving).

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Fast score computation."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def soft_max_obj_grad(theta, n, tau, S_target):
    """
    Compute soft-max smoothed objective and gradient in theta-space.
    theta: parameter vector, a = S_target * softmax(theta)
    Returns (value, grad_theta).
    """
    t = theta - np.max(theta)
    e = np.exp(t)
    a = e / np.sum(e) * S_target
    
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = tau * (c - c_max)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + np.log(sum_exp) / tau
    
    # Gradient of convolution w.r.t. a
    L = 1
    while L < 2 * n:
        L <<= 1
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    cross = np.fft.irfft(W * np.conj(A), L)[:n]
    grad_a = (4.0 * n / tau) * cross
    
    # Chain rule to theta-space
    inner = np.dot(a, grad_a)
    grad_theta = a * (grad_a - inner / S_target)
    
    return value, grad_theta


def softmax_normalize(theta, S_target):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e) * S_target


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u - cssv / np.arange(1, n + 1) > 0
    rho = np.sum(cond)
    if rho == 0:
        theta = v.max()
    else:
        theta = cssv[rho - 1] / rho
    return np.maximum(v - theta, 0.0)


def pgd_a_space(a_init, n, time_limit, rng):
    """
    Projected gradient descent in a-space with soft-max smoothing.
    Projects onto the probability simplex at each step.
    """
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    a = a / S  # normalize to sum=1
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(fft_convolve(a, a))
    
    t0 = time.monotonic()
    step = 0
    momentum = np.zeros(n)
    beta = 0.85
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        
        # Temperature schedule: ramp up from moderate to very high
        c = fft_convolve(a, a)
        c_max = np.max(c)
        # tau * c_max: from ~3 to ~30
        tau_cmax = 3.0 + 27.0 * (elapsed ** 0.7)
        tau = tau_cmax / max(c_max, 1e-10)
        
        # Compute gradient
        L = 1
        while L < 2 * n:
            L <<= 1
        c_shifted = tau * (c - c_max)
        exp_c = np.exp(c_shifted)
        w = exp_c / np.sum(exp_c)
        
        W = np.fft.rfft(w, L)
        A = np.fft.rfft(a, L)
        cross = np.fft.irfft(W * np.conj(A), L)[:n]
        grad = (4.0 * n / tau) * cross
        
        # Nesterov momentum
        a_nest = a + beta * momentum
        # Recompute gradient at Nesterov point (optional, for accuracy)
        # For speed, use gradient at current point
        
        # Step
        lr = 0.05 / (1.0 + 0.05 * step)
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        
        # Project to simplex (sum=1, a>=0)
        a_new = project_simplex(a_new)
        
        # Evaluate
        new_c = fft_convolve(a_new, a_new)
        new_score = 2.0 * n * np.max(new_c)
        
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        step += 1
        
        # Restart if stuck
        if no_improve > 2000:
            r = rng.random()
            if r < 0.5:
                a = project_simplex(best_a + 0.01 * rng.standard_normal(n))
            else:
                # Block perturbation
                blen = max(1, n // 16)
                start = int(rng.integers(0, n - blen))
                a[start:start+blen] *= np.exp(0.05 * rng.standard_normal(blen))
                a = project_simplex(a)
            no_improve = 0
        
        if step % 2000 == 0:
            print(f"    PGD step={step}, score={new_score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score


def l-bfgs_optimize(theta_init, n, tau_schedule, time_limit):
    """
    L-BFGS optimization in theta-space with annealed temperature.
    tau_schedule: list of (tau, max_iter) pairs.
    """
    S_target = np.sqrt(2.0 * n)
    theta = theta_init.copy()
    t0 = time.monotonic()
    
    for tau, max_iter in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        result = minimize(
            fun=lambda th: soft_max_obj_grad(th, n, tau, S_target)[0],
            x0=theta,
            jac=lambda th: soft_max_obj_grad(th, n, tau, S_target)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'gtol': 1e-8, 'maxfun': max_iter * 2}
        )
        theta = result.x
    
    return theta


def make_initializations(n, rng, provided_seq=None):
    """Generate diverse initializations."""
    inits = []
    
    # 1. Uniform
    inits.append(np.ones(n))
    
    # 2. Geometric (tilted) - multiple ratios
    for r in [0.7, 0.85, 0.95, 1.05, 1.2, 1.5]:
        t = np.arange(n, dtype=np.float64)
        a = r ** (t / n)
        inits.append(a)
    
    # 3. Linear ramp
    for alpha in [0.3, 0.5, 1.0, 2.0]:
        t = np.arange(n, dtype=np.float64) / (n - 1)
        a = 1.0 + alpha * t
        inits.append(a)
    
    # 4. Sine-modulated
    for freq in [1, 2, 3]:
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * t / n)
        inits.append(np.clip(a, 0.1, None))
    
    # 5. Random
    for _ in range(3):
        a = np.abs(rng.standard_normal(n)) + 0.1
        inits.append(a)
    
    # 6. Quadratic
    t = np.arange(n, dtype=np.float64) / (n - 1)
    inits.append(1.0 + t ** 2)
    inits.append(1.0 + (1 - t) ** 2)
    
    # 7. From provided sequence
    if provided_seq is not None and len(provided_seq) > 0:
        x_old = np.linspace(0, 1, len(provided_seq))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, provided_seq)
        inits.append(np.clip(a, 0.01, None))
    
    # 8. "Spectral" initialization: based on sqrt of triangular
    # The ideal convolution is flat, so a should have "flat spectrum" in some sense
    # Approximate: a_j proportional to sqrt(1 - (2j/(n-1) - 1)^2) (semicircle)
    t = np.linspace(-1, 1, n)
    a = np.sqrt(np.maximum(1 - t**2, 0.01))
    inits.append(a)
    
    # 9. Logarithmic
    t = np.arange(1, n + 1, dtype=np.float64)
    inits.append(1.0 / np.log(t + 1))
    
    return inits


def coordinate_descent_polish(a_init, n, time_limit, rng):
    """
    Improved coordinate descent: try multiple perturbation sizes,
    use golden section search for 1D optimization.
    """
    a = a_init.astype(np.float64).copy()
    current_score = compute_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    # Pre-compute: for each coordinate, the "influence" on the peak
    c = fft_convolve(a, a)
    peak_idx = int(np.argmax(c))
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        improved = False
        
        # Random order
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-8:
                continue
            
            base = a[j]
            # Try several multiplicative factors
            best_local = base
            best_local_score = current_score
            
            # Geometric search: try factors from 0.3 to 3.0
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                a[j] = base * factor
                s = compute_score(a)
                if s < best_local_score:
                    best_local_score = s
                    best_local = base * factor
            
            if best_local != base:
                a[j] = best_local
                current_score = best_local_score
                improved = True
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
        
        # If no improvement, try random perturbations
        if not improved:
            for _ in range(5):
                j = int(rng.integers(0, n))
                a[j] = max(0.0, a[j] * np.exp(rng.normal(0, 0.1)))
                s = compute_score(a)
                if s < current_score:
                    current_score = s
                    if s < best_score:
                        best_score = s
                        best_a = a.copy()
                else:
                    a[j] = a[j] / np.exp(rng.normal(0, 0.1))  # revert (approximately)
        
        if iteration % 10 == 0:
            print(f"    CD iter={iteration}, score={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score


def block_move_polish(a_init, n, time_limit, rng):
    """
    Block-move local search: move mass between blocks to reduce peak.
    """
    a = a_init.astype(np.float64).copy()
    current_score = compute_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    step = 0
    temp = 0.005
    cooling = 0.9999
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))
        
        # Compute contribution to peak
        contribs = np.zeros(n)
        for i in range(n):
            k = peak_idx - i
            if 0 <= k < n:
                contribs[i] = a[i] * a[k]
        
        r = rng.random()
        
        if r < 0.4:
            # Reduce top contributors, increase bottom
            top_k = max(3, n // 32)
            bottom_k = max(3, n // 16)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            bottom_idx = np.argsort(contribs)[:bottom_k]
            
            amt = 0.03 * a[top_idx].sum() / bottom_k
            a[top_idx] *= (1.0 - 0.03)
            a[bottom_idx] += amt / bottom_k * rng.uniform(0.5, 1.5, size=bottom_k)
        
        elif r < 0.6:
            # Block scale
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.05))
            a[start:start + width] *= factor
        
        elif r < 0.8:
            # Pair swap
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amt = 0.02 * min(a[i], a[j])
                a[i] -= amt
                a[j] += amt
        
        else:
            # Small global noise
            a *= np.exp(rng.normal(0, 0.003, size=n))
        
        a = np.maximum(a, 0.0)
        new_score = compute_score(a)
        
        if new_score < current_score or rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-10)):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        
        temp *= cooling
    
    return best_a, best_score
