"""
Peak-autocorrelation minimization using antisymmetric-structure-aware gradient
descent with adaptive temperature annealing and convolution-peak surgery.
Exploits the decomposition of midpoint convolution into symmetric/antisymmetric
components for initialization, and uses FFT-based soft-max gradients with
Adam optimization for the main search.

import numpy as np
import time


def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax_vals(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def compute_grad(a, tau):
    """
    Compute soft-max(conv(a,a), tau) and its gradient w.r.t. a.
    Returns (value, grad_a) where grad_a has same shape as a.
    """
    n = len(a)
    c = fft_convolve(a)
    c_shift = c - np.max(c)
    exp_c = np.exp(c_shift / tau)
    p = exp_c / np.sum(exp_c)
    value = np.max(c) + tau * np.log(np.sum(exp_c))
    # grad_a[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # = 2 * conv(p[::-1], a)[2n-2-i]
    full_conv = fft_convolve(p[::-1], a)
    grad_a = 2.0 * full_conv[2 * n - 2 : n - 2 : -1]
    return value, grad_a, c


def gs_init(n, rng, n_iter=100, flatten_power=1.0):
    """Gerchberg-Saxton with adjustable flattening power."""
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    a = np.ones(n, dtype=np.float64) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-14] = 1e-14
        # Flatten: mag_new = mag^(-flatten_power) / sum, then adjust
        target = 1.0 / np.sqrt(mag)
        target = target / np.sum(target) * n
        A_new = target * (A / mag)
        a_full = np.fft.ifft(A_new).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def antisymmetric_init(n, rng, method='ramp'):
    """Initialize with strong antisymmetric component."""
    t = np.arange(n, dtype=np.float64) / n
    if method == 'ramp':
        a = 1.0 + 0.8 * (2.0 * t - 1.0)
    elif method == 'sinusoid':
        a = 1.0 + 0.6 * np.sin(np.pi * t * (1 + rng.integers(0, 3)))
    elif method == 'power':
        p = 1.0 + 2.0 * rng.random()
        a = 1.0 + 0.5 * (t ** p - (1.0 - t) ** p)
    else:
        a = 1.0 + 0.4 * (2.0 * t - 1.0)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def random_structure_init(n, rng):
    """Random structured initialization with block patterns."""
    n_blocks = int(rng.integers(3, 10))
    block_size = n // n_blocks
    a = np.zeros(n)
    for i in range(n_blocks):
        start = i * block_size
        end = min(start + block_size, n)
        a[start:end] = 0.3 + 0.7 * rng.random()
    # Add antisymmetric perturbation
    t = np.arange(n, dtype=np.float64) / n
    a *= (1.0 + 0.3 * (2.0 * t - 1.0))
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)

    b1, b2, eps = 0.85, 0.999, 1e-8

    eval_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Annealing: fast decay initially, slow at end
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)

        a = softmax_vals(theta) * S
        value, grad_a, c = compute_grad(a, tau)

        # Project gradient to tangent space of simplex (sum constraint)
        inner = np.dot(a, grad_a)
        g_theta = a * (grad_a - inner / S)

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # Periodic evaluation
        if step % 200 == 0:
            a_check = softmax_vals(theta) * S
            score = local_score(a_check)
            eval_count += 1
            if score < best_score:
                best_score = score
                best_theta = theta.copy()

        # Periodic random perturbation to escape local minima
        if step % 5000 == 0 and elapsed_frac < 0.8:
            theta += 0.1 * rng.standard_normal(n)

    a_final = softmax_vals(best_theta) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
    return a_final, best_score


def convolution_surgery(a, time_limit, rng, n_top=5):
    """
    Local search that specifically targets convolution peaks.
    Identifies the pairs contributing most to peak lags and
    reduces their products.
    """
    current = a.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9997

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.4:
            # Peak-directed: reduce contributors to top peaks
            conv = fft_convolve(current)
            top_lags = np.argsort(conv)[-n_top:]
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            n_modify = max(3, n // 15)
            top_idx = np.argsort(contrib)[-n_modify:]
            bot_idx = np.argsort(contrib)[:n_modify]
            candidate[top_idx] *= np.exp(rng.normal(-0.08, 0.03, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.08, 0.03, size=len(bot_idx)))
        elif r < 0.55:
            # Antisymmetric perturbation: push a[i] up, a[n-1-i] down (or vice versa)
            n_pairs = max(2, n // 20)
            indices = rng.choice(n // 2, size=n_pairs, replace=False)
            for idx in indices:
                i = idx
                j = n - 1 - idx
                if rng.random() < 0.5:
                    candidate[i] *= np.exp(rng.normal(0.1, 0.03))
                    candidate[j] *= np.exp(rng.normal(-0.1, 0.03))
                else:
                    candidate[i] *= np.exp(rng.normal(-0.1, 0.03))
                    candidate[j] *= np.exp(rng.normal(0.1, 0.03))
        elif r < 0.7:
            # Block swap
            width = max(1, int(rng.integers(1, max(2, n // 15))))
            start = int(rng.integers(0, n - 2 * width - 1))
            swap_start = int(rng.integers(0, n - width - 1))
            if abs(start - swap_start) > width:
                candidate[start:start + width], candidate[swap_start:swap_start + width] = \
                    candidate[swap_start:swap_start + width].copy(), \
                    candidate[start:start + width].copy()
                # Smooth the transitions
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))
                candidate[swap_start:swap_start + width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.85:
            # Smoothed antisymmetric nudge
            t = np.arange(n, dtype=np.float64) / n
            anti = (2.0 * t - 1.0)
            alpha = rng.normal(0.02, 0.01)
            candidate = current * (1.0 + alpha * anti)
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / s * np.sum(current)

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 5:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score

        temp *= cooling

    return best, best_score


def upsample_sequence(a, new_n, rng):
    """Upsample via interpolation with small noise."""
    n = len(a)
    if new_n <= n:
        return a[:new_n].copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search minimizing 2*n*max(conv)/sum^2 for non-negative sequences.
    Uses antisymmetric-aware initialization, adaptive-temperature gradient descent,
    and convolution-peak surgery local search.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Phase 1: Diverse initialization across multiple lengths
    print("\n=== Phase 1: Initialization ===")
    candidates = []
    lengths_to_try = [256, 384, 512, 768, 1024, 1536, 2048]

    for n in lengths_to_try:
        if time.monotonic() > deadline - 300:
            break
        inits = []

        # GS variants
        a_gs1 = gs_init(n, rng, n_iter=80, flatten_power=0.8)
        inits.append(('gs1', a_gs1))

        a_gs2 = gs_init(n, rng, n_iter=120, flatten_power=1.2)
        inits.append(('gs2', a_gs2))

        # Antisymmetric inits
        a_anti = antisymmetric_init(n, rng, method='ramp')
        inits.append(('ramp', a_anti))

        a_anti2 = antisymmetric_init(n, rng, method='sinusoid')
        inits.append(('sin', a_anti2))

        # Random structure
        a_rnd = random_structure_init(n, rng)
        inits.append(('struct', a_rnd))

        for name, a in inits:
            score = local_score(a)
            print(f"  n={n}, {name}: score={score:.6f}")
            candidates.append((score, n, a.copy(), name))

    candidates.sort(key=lambda x: x[0])
    print(f"\n  Top 5: {[(f'{c[0]:.4f}', c[1], c[3]) for c in candidates[:5]]}")

    # Phase 2: Gradient optimization on top candidates
    print("\n=== Phase 2: Gradient optimization ===")
    n_cand = min(10, len(candidates))
    time_left = deadline - time.monotonic() - 400  # reserve for later phases
    time_per = max(10.0, time_left / n_cand)

    optimized = []
    for i in range(n_cand):
        if time.monotonic() > deadline - 300:
            break
        score0, n, a, name = candidates[i]
        # Allocate time based on n (larger n = slower per step)
        t_alloc = time_per * min(1.0, 512.0 / n)
        t_alloc = max(8.0, t_alloc)
        t_alloc = min(t_alloc, deadline - time.monotonic() - 200)
        if t_alloc < 5:
            break

        print(f"  [{i+1}/{n_cand}] n={n}, init={name}, score0={score0:.6f}, t_alloc={t_alloc:.1f}s")
        a_opt, s_opt = gradient_optimize(n, a, t_alloc, rng)
        print(f"    -> score={s_opt:.6f}")

        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")

        optimized.append((s_opt, n, a_opt))

    # Phase 3: Convolution surgery local search
    print(f"\n=== Phase 3: Local search (score={best_score:.6f}) ===")
    if best_seq is not None:
        polish_time = min(300.0, max(20.0, deadline - time.monotonic() - 200))
        print(f"  Surgery on n={len(best_seq)}, time={polish_time:.1f}s")
        polished, p_score = convolution_surgery(best_seq, polish_time, rng, n_top=6)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After surgery: {best_score:.6f}")

    # Phase 4: Upsample and re-optimize
    print(f"\n=== Phase 4: Upsample and re-optimize ===")
    if best_seq is not None and optimized:
        optimized.sort(key=lambda x: x[0])
        for target_n in [1024, 2048, 3072, 4096]:
            if time.monotonic() > deadline - 100:
                break
            if len(best_seq) >= target_n:
                continue
            a_up = upsample_sequence(best_seq, target_n, rng)
            # Also try antisymmetric nudge on the upsampled version
            t = np.arange(target_n, dtype=np.float64) / target_n
            a_up2 = a_up * (1.0 + 0.1 * (2.0 * t - 1.0))
            a_up2 = np.clip(a_up2, 0.01, None)
            a_up2 = a_up2 / np.sum(a_up2)

            s_up = local_score(a_up)
            s_up2 = local_score(a_up2)
            a_best_up = a_up if s_up < s_up2 else a_up2
            s_best_up = min(s_up, s_up2)

            t_alloc = min(120.0, deadline - time.monotonic() - 60)
            if t_alloc < 10:
                break
            print(f"  Upsample to n={target_n}, score={s_best_up:.6f}, t_alloc={t_alloc:.1f}s")
            a_opt, s_opt = gradient_optimize(target_n, a_best_up, t_alloc, rng)
            print(f"    -> score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} (n={target_n}) ***")

                # Quick surgery on the new best
                if deadline - time.monotonic() > 30:
                    p2, s2 = convolution_surgery(best_seq, min(40.0, deadline - time.monotonic() - 30), rng)
                    if s2 < best_score:
                        best_score = s2
                        best_seq = p2.copy()
                        print(f"    After quick surgery: {best_score:.6f}")
            break  # Only try one upsample target to save time

    # Phase 5: Additional gradient passes on the best
    remaining = deadline - time.monotonic() - 20
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Phase 5: Additional optimization ===")
        n_best = len(best_seq)
        n_passes = min(3, int(remaining // 30))
        for p in range(n_passes):
            if time.monotonic() > deadline - 25:
                break
            # Perturb and re-optimize
            t = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_seq * (1.0 + 0.05 * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.01, None)
            a_pert = a_pert / np.sum(a_pert) * np.sum(best_seq)

            t_alloc = max(5.0, (deadline - time.monotonic() - 20) / (n_passes - p))
            a_opt, s_opt = gradient_optimize(n_best, a_pert, t_alloc, rng)
            print(f"  Pass {p+1}: score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} ***")
                # Quick local search
                if deadline - time.monotonic() > 20:
                    p2, s2 = convolution_surgery(best_seq, min(25.0, deadline - time.monotonic() - 15), rng)
                    if s2 < best_score:
                        best_score = s2
                        best_seq = p2.copy()
                        print(f"    After surgery: {best_score:.6f}")

    # Final verification
    print(f"\n{'='*50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
