def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:

def peak_gradient_vec(a, k_star):
    n = len(a)
    g = np.zeros(n)
    # First term: a[k_star - j] for 0 <= k_star - j < n
    # j ranges from max(0, k_star-n+1) to min(n-1, k_star)
    j_start = max(0, k_star - n + 1)
    j_end = min(n - 1, k_star)
    if j_start <= j_end:
        # For j in [j_start, j_end], contribution is a[k_star - j]
        idx = k_star - np.arange(j_start, j_end + 1)
        g[j_start:j_end+1] += a[idx]
    # Second term: a[j - k_star] for 0 <= j - k_star < n
    # j ranges from max(0, k_star) to min(n-1, k_star+n-1)
    j_start2 = max(0, k_star)
    j_end2 = min(n - 1, k_star + n - 1)
    if j_start2 <= j_end2:
        idx2 = np.arange(j_start2, j_end2 + 1) - k_star
        g[j_start2:j_end2+1] += a[idx2]
    return 2.0 * g

"""
Direct subgradient optimization with exponential (softmax) parameterization,
backtracking line search, heavy-ball momentum, and adaptive exploration.
Works directly on the peak subgradient rather than soft-max approximations.
Targets n=384 (primary, leveraging height_sequence_1) and n=512 (secondary).
Includes symmetrization, multi-restart, and a fine-grained polish phase.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax_stable(theta):
    """Stable softmax returning values summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def compute_score_c(a):
    """Compute score and full convolution. Assumes sum(a)=1, a>=0."""
    n = len(a)
    c = fft_convolve(a, a)
    peak_idx = int(np.argmax(c))
    peak = c[peak_idx]
    score = 2.0 * n * peak
    return score, c, peak_idx


def peak_subgrad_a(a, k_star):
    """
    Subgradient of max_k (a*a)[k] w.r.t. a, evaluated at peak index k_star.
    Returns gradient vector of length n.
    """
    n = len(a)
    g = np.zeros(n)
    # Term 1: a[k_star - j] for 0 <= k_star - j < n
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n - 1, k_star)
    if j_lo <= j_hi:
        g[j_lo:j_hi + 1] += a[k_star - np.arange(j_lo, j_hi + 1)]
    # Term 2: a[j - k_star] for 0 <= j - k_star < n
    j_lo2 = max(0, k_star)
    j_hi2 = min(n - 1, k_star + n - 1)
    if j_lo2 <= j_hi2:
        g[j_lo2:j_hi2 + 1] += a[np.arange(j_lo2, j_hi2 + 1) - k_star]
    return 2.0 * g


def grad_theta(a, g_a):
    """
    Gradient of score w.r.t. theta where a = softmax(theta), sum(a)=1.
    score = 2n * peak(a*a)
    g_theta[j] = 2n * a[j] * (g_a[j] - dot(g_a, a))
    """
    n = len(a)
    inner = np.dot(g_a, a)
    return 2.0 * n * a * (g_a - inner)


def optimize_direct(a_init, time_limit, rng, verbose=True):
    """
    Direct subgradient descent with backtracking line search and momentum.
    Works in theta-space with a = softmax(theta).
    """
    n = len(a_init)
    a_init = np.clip(a_init, 1e-10, None)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init + 1e-15).astype(np.float64)

    # State
    velocity = np.zeros(n)
    momentum = 0.85
    best_theta = theta.copy()
    best_score = compute_score_c(softmax_stable(theta))[0]
    current_lr = 1.0

    t0 = time.monotonic()
    step = 0
    no_improve = 0
    patience = 500
    stagnation_count = 0

    if verbose:
        print(f"    Init score: {best_score:.6f}")

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = elapsed / time_limit

        a = softmax_stable(theta)
        score, c, k_star = compute_score_c(a)
        g_a = peak_subgrad_a(a, k_star)
        g_theta = grad_theta(a, g_a)

        # Backtracking line search with momentum
        # Direction: d = g_theta + momentum * velocity
        direction = g_theta + momentum * velocity

        # Normalize direction for consistent step size
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            break
        direction = direction / dir_norm

        # Initial step size
        if frac < 0.3:
            lr = current_lr * 2.0 / (1.0 + step * 0.001)
        elif frac < 0.7:
            lr = current_lr * 1.0 / (1.0 + step * 0.0005)
        else:
            lr = current_lr * 0.5 / (1.0 + step * 0.0002)

        # Backtracking
        alpha = lr
        improved = False
        for _ in range(12):
            theta_try = theta + alpha * direction
            a_try = softmax_stable(theta_try)
            score_try = compute_score_c(a_try)[0]
            if score_try < score - 1e-12:
                improved = True
                break
            alpha *= 0.6

        if improved:
            velocity = momentum * velocity + (1 - momentum) * direction
            theta = theta + alpha * direction
            current_lr = min(current_lr * 1.02, 50.0)
            current_score = score_try
        else:
            # No improvement via gradient; try perturbation
            if no_improve > patience:
                r = rng.random()
                if r < 0.3:
                    # Random kick
                    theta = best_theta + 0.2 * rng.standard_normal(n)
                elif r < 0.5:
                    # Targeted: perturb peak contributors
                    contribs = np.zeros(n)
                    j_lo = max(0, k_star - n + 1)
                    j_hi = min(n - 1, k_star)
                    contribs[j_lo:j_hi + 1] = a[j_lo:j_hi + 1] * a[k_star - np.arange(j_lo, j_hi + 1)]
                    j_lo2 = max(0, k_star)
                    j_hi2 = min(n - 1, k_star + n - 1)
                    contribs[j_lo2:j_hi2 + 1] += a[j_lo2:j_hi2 + 1] * a[np.arange(j_lo2, j_hi2 + 1) - k_star]
                    top_k = np.argsort(contribs)[-n // 4:]
                    theta[top_k] += 0.1 * rng.standard_normal(len(top_k))
                    theta = theta - 0.05 * np.sum(theta) / n  # renormalize
                elif r < 0.7:
                    # Symmetrize
                    a_cur = softmax_stable(theta)
                    a_sym = (a_cur + a_cur[::-1]) / 2.0
                    a_sym = a_sym / np.sum(a_sym)
                    theta = np.log(a_sym + 1e-15)
                elif r < 0.85:
                    # Block perturbation
                    w = max(4, n // 16)
                    s = int(rng.integers(0, n - w))
                    theta[s:s+w] += 0.1 * rng.standard_normal(w)
                else:
                    # Global scale perturbation
                    theta += 0.05 * rng.standard_normal(n)

                a_check = softmax_stable(theta)
                score_check = compute_score_c(a_check)[0]
                stagnation_count = 0
                no_improve = 0
                current_lr = max(0.1, current_lr * 0.5)
                current_score = score_check
            else:
                # Small random step
                theta += 0.01 * rng.standard_normal(n)
                current_score = score

        step += 1
        no_improve += 1

        if current_score < best_score - 1e-8:
            best_score = current_score
            best_theta = theta.copy()
            no_improve = 0
            stagnation_count = 0

        if step % 500 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    step={step}, t={elapsed:.0f}s, cur={current_score:.6f}, best={best_score:.6f}, lr={current_lr:.2f}")

        # Adaptive momentum: reduce in later phase
        if frac > 0.8:
            momentum = 0.7
        elif frac > 0.5:
            momentum = 0.8

    a_best = softmax_stable(best_theta)
    final_score = compute_score_c(a_best)[0]
    if final_score < best_score:
        best_score = final_score
        best_theta = theta.copy()
        a_best = softmax_stable(best_theta)

    return a_best, best_score


def polish_fine(a_init, time_limit, rng, verbose=True):
    """
    Fine-grained polish: very small steps, try multiple peak indices,
    and coordinate-wise adjustments.
    """
    n = len(a_init)
    a_init = np.clip(a_init, 1e-10, None)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init + 1e-15).astype(np.float64)

    best_theta = theta.copy()
    best_score = compute_score_c(softmax_stable(theta))[0]
    t0 = time.monotonic()
    step = 0

    if verbose:
        print(f"    Polish init: {best_score:.6f}")

    while time.monotonic() - t0 < time_limit:
        a = softmax_stable(theta)
        score, c, k_star = compute_score_c(a)

        # Try the top few peak indices for gradient direction
        c_sorted = np.argsort(c)[::-1][:3]
        best_dir = None
        best_dir_score = score

        for k_cand in c_sorted:
            g_a = peak_subgrad_a(a, int(k_cand))
            g_theta = grad_theta(a, g_a)
            # Try small step
            for lr in [0.5, 0.2, 0.1, 0.05, 0.02]:
                theta_try = theta - lr * g_theta / (np.linalg.norm(g_theta) + 1e-15)
                a_try = softmax_stable(theta_try)
                s_try = compute_score_c(a_try)[0]
                if s_try < best_dir_score:
                    best_dir_score = s_try
                    best_dir = theta_try.copy()
                    break
            if best_dir is not None:
                break

        if best_dir is not None:
            theta = best_dir
            step += 1
            if best_dir_score < best_score:
                best_score = best_dir_score
                best_theta = theta.copy()
        else:
            # Try coordinate-wise: find best single-coordinate change
            j = int(rng.integers(0, n))
            for delta in [0.05, -0.05, 0.1, -0.1, 0.2, -0.2]:
                theta_try = theta.copy()
                theta_try[j] += delta
                a_try = softmax_stable(theta_try)
                s_try = compute_score_c(a_try)[0]
                if s_try < score:
                    theta = theta_try
                    break

            # Occasional symmetrization
            if step % 100 == 0:
                a_cur = softmax_stable(theta)
                a_sym = (a_cur + a_cur[::-1]) / 2.0
                a_sym = a_sym / np.sum(a_sym)
                s_sym = compute_score_c(a_sym)[0]
                if s_sym < best_score:
                    best_score = s_sym
                    theta = np.log(a_sym + 1e-15)
                    best_theta = theta.copy()

            step += 1

        if step % 200 == 0 and verbose:
            print(f"    Polish step={step}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")

    a_best = softmax_stable(best_theta)
    return a_best, best_score


def make_init(n, rng, method='broadband', base_seq=None):
    """Create initial sequences."""
    if base_seq is not None and len(base_seq) != n:
        x_old = np.linspace(0, 1, len(base_seq))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, base_seq)
        return np.clip(a, 1e-4, None)
    if base_seq is not None and len(base_seq) == n:
        return np.clip(base_seq, 1e-4, None)

    t = np.arange(n, dtype=np.float64)
    if method == 'broadband':
        a = np.ones(n) * 0.5
        for k in range(1, 20):
            freq = k * (1.0 + 0.5 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.1 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.01 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'smooth':
        a = 1.0 + 0.3 * np.sin(2 * np.pi * t / n * 3 + rng.uniform(0, 2))
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    else:
        a = 1.0 + 0.2 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
