"""
Optimized step function search using FFT-based gradient descent with 
annealed soft-max, flat-spectrum initialization, and peak-targeting 
local search. Focuses on larger sequence lengths for better theoretical bounds.
"""

import numpy as np
import time
import math


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def gs_init(n, rng, n_iter=200, L=None):
    """
    Gerchberg-Saxton initialization targeting flat power spectrum.
    Projects to non-negativity at each step.
    """
    if L is None:
        L = 1
        while L < 2 * n - 1:
            L <<= 1
    
    # Target magnitude: flat spectrum
    # sum(a) should be such that |A(0)| = sum(a)
    # For flat |A(f)| = C, we have C = sum(a) (at f=0)
    # So target |A(f)| = sum(a) for all f
    # This means the "energy" in the frequency domain is C^2 * L
    # And the "energy" in the time domain is C^2 (Parseval: sum |A|^2 / L = sum |a|^2)
    # So sum |a|^2 = C^2 * L / L = C^2... hmm that doesn't work out.
    
    # Actually, Parseval: sum_{f=0}^{L-1} |A(f)|^2 = L * sum_{k=0}^{L-1} |a[k]|^2
    # If |A(f)| = C for all f, then L * C^2 = L * sum |a[k]|^2, so sum |a[k]|^2 = C^2
    # And A(0) = sum a[k] = C (real and positive)
    # So sum a[k] = C and sum a[k]^2 = C^2
    # This means a[k] are all 0 except one which is C (delta function)!
    # That's not what we want.
    
    # The issue is that perfectly flat spectrum with non-negative time domain is impossible
    # (except for the trivial delta case).
    
    # Instead, let's target a "nearly flat" spectrum. We'll use the iterative approach:
    # 1. Start with some a
    # 2. Compute A = DFT(a padded to L)
    # 3. Set |A| to a target that's "as flat as possible" given the constraint
    #    that A(0) = sum(a) and the inverse must be non-negative in [0, n-1]
    # 4. Simplest: just set |A(f)| = |A(0)| for all f (perfectly flat)
    #    and hope that the projection to non-negative gives something reasonable
    
    a = np.ones(n) / n  # Start uniform
    scale = np.sqrt(2.0 * n)  # Target sum
    
    for it in range(n_iter):
        a_pad = np.zeros(L)
        a_pad[:n] = a * scale
        A = np.fft.fft(a_pad)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        # Target: flat magnitude
        target_mag = np.abs(A[0])  # = sum(a_pad) = scale
        A_new = target_mag * phase
        a_full = np.fft.ifft(A_new).real
        a = a_full[:n]
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
            a = a * scale
        else:
            a = a / s * scale
    
    return a


def gs_init_relaxed(n, rng, n_iter=200, L=None, beta=0.5):
    """
    Relaxed GS: target magnitude is a mix of flat and current magnitude.
    This helps avoid getting stuck.
    """
    if L is None:
        L = 1
        while L < 2 * n - 1:
            L <<= 1
    
    scale = np.sqrt(2.0 * n)
    a = np.ones(n) / n * scale
    
    for it in range(n_iter):
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        
        # Relaxed target: interpolate between flat and current
        target_flat = np.abs(A[0])
        target_mag = (1 - beta) * mag + beta * target_flat
        
        A_new = target_mag * phase
        a_full = np.fft.ifft(A_new).real
        a = a_full[:n]
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n * scale
        else:
            a = a / s * scale
    
    return a

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        lr *= lr_decay
        
        # Periodic check
        if step % n_check == 0:
            a_check = np.exp(theta - np.max(theta))
            a_check = a_check / np.sum(a_check) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
            if step % (n_check * 10) == 0:
                elapsed = time.monotonic() - t0
                print(f"      step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.4f}, lr={lr:.4f}, t={elapsed:.1f}s")
    
    # Final check
    a_final = np.exp(best_theta - np.max(best_theta))
    a_final = a_final / np.sum(a_final) * S
    final_score = local_score(a_final)
    if final_score < best_score:
        best_score = final_score
    
    return a_final, best_score


def peak_targeting_search(a_init, time_limit, rng, L=None):
    """
    Local search that specifically targets the peak of the convolution.
    Uses a multi-scale approach: identify contributing elements and adjust them.
    """
    if L is None:
        L = 1
        while L < 2 * len(a_init) - 1:
            L <<= 1
    
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    temp = 0.01  # Starting temperature for acceptance
    cooling = 0.9997
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        A = np.fft.rfft(current, L)
        C = np.fft.irfft(A * A, L)[:2 * n - 1]
        peak_val = np.max(C)
        peak_pos = np.argmax(C)
        
        # Find contributing elements to the peak
        # C[peak_pos] = sum_k current[k] * current[peak_pos - k]
        k_range = range(max(0, peak_pos - n + 1), min(n, peak_pos + 1))
        pairs = [(k, peak_pos - k) for k in k_range if 0 <= peak_pos - k < n]
        
        # Find the top contributing pairs
        if len(pairs) > 0:
            contribs = [(current[k] * current[peak_pos - k], k, peak_pos - k) for k, _ in pairs]
            contribs.sort(reverse=True)
            
            r = rng.random()
            candidate = current.copy()
            
            if r < 0.4:
                # Reduce top contributors, increase bottom
                n_top = max(3, n // 20)
                top_pairs = contribs[:n_top]
                bot_pairs = contribs[-n_top:]
                
                # Reduce
                reduce_strength = rng.uniform(0.02, 0.08)
                for _, k, j in top_pairs:
                    # Reduce the larger of the pair
                    if current[k] >= current[j]:
                        candidate[k] *= (1 - reduce_strength * rng.random())
                    else:
                        candidate[j] *= (1 - reduce_strength * rng.random())
                
                # Increase non-contributing elements
                contrib_set = set()
                for _, k, j in contribs:
                    contrib_set.add(k)
                    contrib_set.add(j)
                non_contrib = [i for i in range(n) if i not in contrib_set]
                if non_contrib:
                    n_inc = min(5, len(non_contrib))
                    inc_idx = rng.choice(non_contrib, size=n_inc, replace=False)
                    inc_strength = rng.uniform(0.02, 0.08)
                    candidate[inc_idx] *= (1 + inc_strength)
                    
            elif r < 0.6:
                # Gaussian perturbation on contributing elements
                sigma = rng.uniform(0.02, 0.06)
                contrib_indices = set()
                for _, k, j in contribs:
                    contrib_indices.add(k)
                    contrib_indices.add(j)
                for idx in contrib_indices:
                    candidate[idx] *= np.exp(rng.normal(0, sigma))
            
            elif r < 0.75:
                # Block perturbation: reduce a block of contributing elements
                block_size = max(2, n // 30)
                # Find a contiguous block of high contributors
                scores = np.zeros(n)
                for _, k, j in contribs:
                    scores[k] += current[k] * current[j]
                    scores[j] += current[j] * current[k]
                top_block = np.argsort(scores)[-block_size:]
                candidate[top_block] *= np.exp(rng.normal(-0.05, 0.02, size=block_size))
                # Increase a random other block
                other = np.setdiff1d(np.arange(n), top_block)
                if len(other) > block_size:
                    inc_block = rng.choice(other, size=block_size, replace=False)
                    candidate[inc_block] *= np.exp(rng.normal(0.05, 0.02, size=block_size))
            
            elif r < 0.85:
                # Symmetrization step (helps when peak is off-center)
                # If peak is at position p, the "center" of correlation is p
                # Symmetrize around the midpoint
                sym = 0.5 * (current + current[::-1])
                alpha = rng.uniform(0.05, 0.2)
                candidate = current + alpha * (sym - current)
            
            else:
                # Small global noise
                candidate = current * np.exp(rng.normal(0, 0.01, size=n))
            
            candidate = np.clip(candidate, 0.0, None)
            s = np.sum(candidate)
            if s < 1e-10:
                candidate = current.copy()
            else:
                candidate = candidate / s * np.sum(current)
            
            new_score = local_score(candidate)
            
            if new_score < current_score:
                current = candidate
                current_score = new_score
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
            else:
                delta = new_score - current_score
                if delta < temp * 5:
                    accept_prob = np.exp(-delta / max(temp, 1e-12))
                    if rng.random() < accept_prob:
                        current = candidate
                        current_score = new_score
        
        temp *= cooling
    
    return best, best_score


def upsample_refine(a, new_n, rng, time_for_grad=30.0):
    """Upsample and refine a sequence."""
    n = len(a)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    S = np.sqrt(2.0 * new_n)
    a_interp = a_interp / np.sum(a_interp) * S
    return a_interp

"""
Multi-strategy optimization for minimizing peak autocorrelation of non-negative
step functions. Uses FFT-based gradient descent with annealed soft-max, 
flat-spectrum (Gerchberg-Saxton) initialization, peak-targeting local search,
and coarse-to-fine upsampling. Focuses on larger sequence lengths where the
theoretical minimum 2n/(2n-1) approaches 1, giving more room for improvement.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def fft_convolve_inplace(a, L):
    """Compute a*a convolution using precomputed FFT size L."""
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:2 * len(a) - 1]


def gs_init(n, rng, n_iter=200, L=None):
    """Gerchberg-Saxton initialization targeting flat power spectrum with non-negativity."""
    if L is None:
        L = 1
        while L < 2 * n - 1:
            L <<= 1

    scale = np.sqrt(2.0 * n)
    a = np.ones(n) * scale / n

    for it in range(n_iter):
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        target_mag = np.abs(A[0])
        A_new = target_mag * phase
        a_full = np.fft.ifft(A_new).real
        a = a_full[:n]
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) * scale / n
        else:
            a = a / s * scale

    return a


def gs_init_diverse(n, rng, n_iter=150, L=None, beta=0.4):
    """Relaxed GS with interpolation for diversity."""
    if L is None:
        L = 1
        while L < 2 * n - 1:
            L <<= 1

    scale = np.sqrt(2.0 * n)
    a = np.ones(n) * scale / n
    a += 0.1 * rng.standard_normal(n) * scale / n
    a = np.clip(a, 0.01 * scale / n, None)
    a = a / np.sum(a) * scale

    for it in range(n_iter):
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        target_flat = np.abs(A[0])
        target_mag = (1.0 - beta) * mag + beta * target_flat
        A_new = target_mag * phase
        a_full = np.fft.ifft(A_new).real
        a = a_full[:n]
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) * scale / n
        else:
            a = a / s * scale

    return a


def compute_grad_soft_max(a, tau, L):
    """Compute soft-max(conv(a,a)) and gradient w.r.t. a, using FFT."""
    n = len(a)
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)[:2 * n - 1]

    c_max = np.max(C)
    shifted = (C - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    w = e / s
    value = c_max + tau * np.log(s)

    # Gradient: g[k] = (2/tau) * sum_l a[l] * w[l+k]
    # Use cross-correlation via FFT
    N = 1
    while N < n + (2 * n - 1):
        N <<= 1

    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)
    R = np.fft.irfft(A_fft * np.conj(W_fft), N)

    g = (2.0 / tau) * R[:n]
    return value, g, C


def optimize_gradient(n, a_init, time_limit, rng, L=None):
    """Adam gradient descent with annealed soft-max on log-parameterized sequence."""
    if L is None:
        L = 1
        while L < 2 * n - 1:
            L <<= 1

    S = np.sqrt(2.0 * n)
    a = np.clip(a_init, 0, None).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a / s * S

    theta = np.log(a + 1e-12)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    step = 0

    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = local_score(a)

    # Temperature schedule (absolute values, since conv values are O(1) in this normalization)
    tau_start = 0.15
    tau_end = 0.003
    anneal_power = 0.4

    lr = 2.0
    lr_decay = 0.9997

    # Adaptive check frequency
    n_check = max(50, n // 64)

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (frac ** anneal_power)

        # Reconstruct a from theta
        a = np.exp(theta - np.max(theta))
        a = a / np.sum(a) * S

        value, g_a, C = compute_grad_soft_max(a, tau, L)

        # Project gradient to maintain sum constraint
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        step += 1
        m1 = b1 * m1 + (1.0 - b1) * g_theta
        m2 = b2 * m2 + (1.0 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)

        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        lr *= lr_decay

        if step % n_check == 0:
            a_check = np.exp(best_theta - np.max(best_theta))
            a_check = a_check / np.sum(a_check) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = (theta - np.max(theta)).copy()
            if step % (n_check * 5) == 0:
                print(f"        step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.4f}, lr={lr:.4f}, t={elapsed:.0f}s")

    a_final = np.exp(best_theta - np.max(best_theta))
    a_final = a_final / np.sum(a_final) * S
    final_score = local_score(a_final)
    if final_score < best_score:
        best_score = final_score

    return a_final, best_score


def peak_target_search(a_init, time_limit, rng, L=None):
    """Local search targeting convolution peaks with multi-strategy mutations."""
    if L is None:
        L = 1
        while L < 2 * len(a_init) - 1:
            L <<= 1

    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score

    t0 = time.monotonic()
    temp = 0.005
    cooling = 0.9998

    while time.monotonic() - t0 < time_limit:
        A = np.fft.rfft(current, L)
        C = np.fft.irfft(A * A, L)[:2 * n - 1]

        # Find top peak positions
        n_peaks = min(5, 2 * n - 1)
        top_positions = np.argsort(C)[-n_peaks:]

        r = rng.random()
        candidate = current.copy()

        if r < 0.35:
            # Multi-peak reduction: reduce elements contributing to top peaks
            contrib = np.zeros(n)
            for pos in top_positions[:3]:
                for k in range(max(0, pos - n + 1), min(n, pos + 1)):
                    j = pos - k
                    if 0 <= j < n:
                        contrib[k] += current[k] * current[j]
                        contrib[j] += current[j] * current[k]

            n_reduce = max(3, n // 25)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]

            red_str = rng.uniform(0.03, 0.1)
            candidate[top_idx] *= np.exp(rng.normal(-red_str, 0.02, size=n_reduce))
            candidate[bot_idx] *= np.exp(rng.normal(red_str, 0.02, size=n_reduce))

        elif r < 0.5:
            # Block-level perturbation on high-contribution region
            block = max(2, n // 40)
            start = int(rng.integers(0, n - 2 * block))
            candidate[start:start + block] *= np.exp(rng.normal(-0.06, 0.02, size=block))
            candidate[start + block:start + 2 * block] *= np.exp(rng.normal(0.06, 0.02, size=block))

        elif r < 0.65:
            # Partial symmetrization (helps when peaks are asymmetric)
            sym = 0.5 * (current + current[::-1])
            alpha = rng.uniform(0.03, 0.15)
            candidate = current + alpha * (sym - current)
            candidate += 0.005 * rng.standard_normal(n)

        elif r < 0.8:
            # Gaussian noise on a subset
            n_sub = max(5, n // 10)
            idx = rng.choice(n, size=n_sub, replace=False)
            candidate[idx] *= np.exp(rng.normal(0, 0.04, size=n_sub))

        elif r < 0.9:
            # Sparsify: reduce random subset, boost complement
            n_sparse = max(1, n // 8)
            sparse_idx = rng.choice(n, size=n_sparse, replace=False)
            candidate[sparse_idx] *= np.exp(rng.normal(-0.15, 0.05, size=n_sparse))
            dense_idx = np.setdiff1d(np.arange(n), sparse_idx)
            if len(dense_idx) > 3:
                n_inc = min(8, len(dense_idx))
                inc_idx = rng.choice(dense_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.15, 0.05, size=n_inc))

        else:
            # Global small noise
            candidate = current * np.exp(rng.normal(0, 0.012, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / s * np.sum(current)

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 8:
                if np.exp(-delta / max(temp, 1e-12)) > rng.random():
                    current = candidate
                    current_score = new_score

        temp *= cooling

    return best, best_score


def upsample_sequence(a, new_n, rng):
    """Upsample via interpolation with small noise."""
    n = len(a)
    x_old = np.linspace(0, 1, n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.005 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.0001, None)
    S = np.sqrt(2.0 * new_n)
    return a_interp / np.sum(a_interp) * S


def propose_candidate(budget_s=2400, **kwargs):
    """
    Multi-phase search: diverse initialization, gradient optimization,
    peak-targeting local search, and coarse-to-fine upsampling.
    Focuses on n=1024 to n=4096 for best theoretical bounds.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 35.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(777)

    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, None)
        if len(start_seq) > 0:
            s = local_score(start_seq)
            print(f"Provided: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()

    # Main search over sequence lengths
    # Allocate time proportionally: longer sequences get more time
    lengths_and_weights = [(512, 0.08), (1024, 0.22), (2048, 0.35), (4096, 0.35)]

    for n, weight in lengths_and_weights:
        if time.monotonic() > deadline - 80:
            break

        time_left = deadline - time.monotonic() - 50
        if time_left < 40:
            break

        t_alloc = time_left * weight / (sum(w for _, w in lengths_and_weights))
        t_alloc = max(30.0, min(time_left * 0.4, t_alloc))

        print(f"\n{'='*60}")
        print(f"n={n}, allocated={t_alloc:.0f}s, remaining={time_left:.0f}s")
        print(f"{'='*60}")

        # Precompute FFT size
        L = 1
        while L < 2 * n - 1:
            L <<= 1

        # Try 2-3 initializations
        n_inits = 2 if t_alloc > 100 else 1
        for init_i in range(n_inits):
            if time.monotonic() > deadline - 60:
                break

            t_remaining = deadline - time.monotonic() - 40
            if t_remaining < 20:
                break

            # Time allocation within this init
            t_gs = min(15.0, t_remaining * 0.1)
            t_grad = t_remaining * 0.55
            t_peak = t_remaining * 0.35

            # Initialization
            if init_i == 0:
                print(f"  [{init_i+1}] GS init...")
                a_init = gs_init(n, rng, n_iter=200, L=L)
            else:
                print(f"  [{init_i+1}] Diverse GS init...")
                a_init = gs_init_diverse(n, rng, n_iter=150, L=L, beta=0.3 + 0.2 * rng.random())

            s_init = local_score(a_init)
            print(f"    init score: {s_init:.6f}")

            if s_init < best_score:
                best_score = s_init
                best_seq = a_init.copy()

            # Gradient optimization
            t_g = min(t_grad, (deadline - time.monotonic() - 40) * 0.5)
            if t_g < 10:
                break
            print(f"    gradient descent: {t_g:.0f}s")
            a_opt, s_opt = optimize_gradient(n, a_init, t_g, rng, L=L)
            print(f"    grad score: {s_opt:.6f}")

            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** NEW BEST: {best_score:.6f} (n={n}) ***")

            # Peak targeting
            t_p = min(t_peak, (deadline - time.monotonic() - 25) * 0.5)
            if t_p < 8:
                break
            print(f"    peak search: {t_p:.0f}s")
            a_pk, s_pk = peak_target_search(a_opt, t_p, rng, L=L)
            print(f"    peak score: {s_pk:.6f}")

            if s_pk < best_score:
                best_score = s_pk
                best_seq = a_pk.copy()
                print(f"    *** NEW BEST: {best_score:.6f} (n={n}) ***")

        # Try upsampling the best sequence found so far to this length
        if best_seq is not None and len(best_seq) < n and time.monotonic() < deadline - 50:
            t_up = min(25.0, (deadline - time.monotonic() - 30) * 0.3)
            if t_up > 10:
                print(f"  Upsample {len(best_seq)} -> {n}")
                a_up = upsample_sequence(best_seq, n, rng)
                a_up_opt, s_up = optimize_gradient(n, a_up, t_up * 0.7, rng, L=L)
                print(f"    upsampled score: {s_up:.6f}")
                if s_up < best_score:
                    best_score = s_up
                    best_seq = a_up_opt.copy()
                    print(f"    *** NEW BEST: {best_score:.6f} (n={n}) ***")

    # Final polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        t_pol = min(50.0, deadline - time.monotonic() - 15)
        n_final = len(best_seq)
        L_final = 1
        while L_final < 2 * n_final - 1:
            L_final <<= 1
        print(f"\nFinal polish: n={n_final}, score={best_score:.6f}, t={t_pol:.0f}s")
        a_pol, s_pol = peak_target_search(best_seq, t_pol, rng, L=L_final)
        if s_pol < best_score:
            best_score = s_pol
            best_seq = a_pol.copy()
            print(f"  After polish: {best_score:.6f}")

        # One more gradient pass if time allows
        if time.monotonic() < deadline - 20 and best_score > 1.50:
            t_mid = min(30.0, deadline - time.monotonic() - 15)
            if t_mid > 10:
                print(f"  Final gradient: t={t_mid:.0f}s")
                a_gr, s_gr = optimize_gradient(n_final, best_seq, t_mid, rng, L=L_final)
                print(f"    score: {s_gr:.6f}")
                if s_gr < best_score:
                    best_score = s_gr
                    best_seq = a_gr.copy()
                    print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
