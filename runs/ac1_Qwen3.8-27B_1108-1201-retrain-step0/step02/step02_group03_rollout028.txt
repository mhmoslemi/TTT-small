"""
Multi-strategy optimization for minimizing 2n*max(self-convolution)/sum^2 of non-negative sequences.
Combines FFT-based gradient descent with adaptive soft-max, window-function initialization,
spectral shaping, symmetry exploitation, and peak-directed local search.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def soft_max_value_and_weights(c, tau):
    """Compute soft max of c and the softmax weights."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    return c_max + tau * np.log(s), e / s


def gradient_soft_max_conv(a, tau):
    """
    Compute soft_max(conv(a,a), tau) and its gradient w.r.t. a.
    Returns (value, grad) where grad has same length as a.
    """
    n = len(a)
    c = fft_convolve(a, a)  # length 2n-1
    value, w = soft_max_value_and_weights(c, tau)
    
    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j in [0, n-1]
    # This is 2 * cross_correlation(w, a)[:n]
    # cross_correlation(x, y)[j] = sum_k x[k+j] * y[k]
    # = IFFT(FFT(x) * conj(FFT(y)))[:len(y)]
    
    L = 1
    while L < 3 * n - 2:
        L <<= 1
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    r = np.fft.irfft(W * np.conj(A), L)[:n]
    grad = 2.0 * r
    return value, grad, c


def softmax_normalize(theta):
    """Stable softmax."""
    theta = theta - np.max(theta)
    e = np.exp(theta)
    return e / np.sum(e)


def init_window(n, window_type='hann', rng=None):
    """Initialize with a window function."""
    t = np.arange(n)
    if window_type == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
    elif window_type == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t / (n - 1)) + 0.08 * np.cos(4 * np.pi * t / (n - 1))
    elif window_type == 'flattop':
        a = (0.21557895 - 0.41663158 * np.cos(2*np.pi*t/(n-1)) + 
             0.277263158 * np.cos(4*np.pi*t/(n-1)) - 0.083578947 * np.cos(6*np.pi*t/(n-1)) +
             0.006947368 * np.cos(8*np.pi*t/(n-1)))
    elif window_type == 'gaussian':
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif window_type == 'sinc_clipped':
        center = (n - 1) / 2.0
        a = np.sinc((t - center) / (n * 0.3))
        a = np.clip(a, 0.01, None)
    else:
        a = np.ones(n)
    
    if rng is not None:
        a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 1e-3, None)
    return a / np.sum(a)


def init_fft_phase(n, rng, n_iter=100):
    """GS-style initialization targeting flat spectrum."""
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    
    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        # Target: flat magnitude
        A_new = mag.mean() * A / mag
        a_full = np.fft.irfft(A_new, N_fft)
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def init_random(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def gradient_optimize(n, a_init, time_limit, rng, tau_start=None, tau_end=None):
    """
    Adam gradient descent on softmax(theta) parameterization.
    Minimizes soft_max(conv(a,a), tau) with annealed tau.
    """
    if tau_start is None:
        tau_start = 50.0 / n
    if tau_end is None:
        tau_end = 0.1 / n
    
    a = a_init.copy()
    a = a / np.sum(a)
    theta = np.log(a + 1e-15)
    
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr0 = 2.0
    step = 0
    t0 = time.monotonic()
    
    best_theta = theta.copy()
    best_score = compute_score(a)
    patience = 0
    best_value = None
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        # Anneal temperature
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        
        a = softmax_normalize(theta)
        value, grad_a, c = gradient_soft_max_conv(a, tau)
        
        # Project gradient to theta space
        inner = np.dot(a, grad_a)
        grad_theta = a * (grad_a - inner)
        
        step += 1
        m1 = b1 * m1 + (1 - b1) * grad_theta
        m2 = b2 * m2 + (1 - b2) * (grad_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.1 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Periodic evaluation
        if step % 500 == 0:
            a_check = softmax_normalize(theta)
            s = compute_score(a_check)

                best_theta = theta.copy()
                patience = 0
            else:

    a_final = softmax_normalize(best_theta)
    score_final = compute_score(a_final)
    if score_final < best_score:

def local_search_peak(a_init, time_limit, rng, top_k=5):
    """
    Local search that specifically targets reducing the peak lag of the convolution.
    Uses peak-directed perturbations with simulated annealing.
    """
    current = a_init.astype(np.float64).copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9997
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        iter_count += 1
        
        if r < 0.4:
            # Peak-directed: find the top lags and reduce contributing elements
            conv = fft_convolve(current, current)
            n_top = min(top_k, len(conv))
            top_lags = np.argsort(conv)[-n_top:]
            
            # Find which elements contribute most to the peak lags
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            
            # Reduce top contributors, increase bottom contributors
            n_red = max(2, n // 15)
            top_idx = np.argsort(contrib)[-n_red:]
            bot_idx = np.argsort(contrib)[:n_red]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=n_red))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_red))
        elif r < 0.6:
            # Block perturbation: reduce a block, increase another
            n_blocks = rng.integers(2, 5)
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 10))))
                bs = int(rng.integers(0, n - bw))
                sign = rng.choice([-1, 1])
                candidate[bs:bs+bw] *= np.exp(sign * rng.uniform(0.03, 0.1))
        elif r < 0.75:
            # Spectral shaping: modify in frequency domain
            N_fft = 1
            while N_fft < 2 * n - 1:
                N_fft <<= 1
            A = np.fft.rfft(current, N_fft)
            # Add small random phase perturbation
            A *= np.exp(1j * rng.normal(0, 0.05, size=len(A)))
            # Flatten magnitude slightly
            mag = np.abs(A)
            mag_target = mag ** 0.95 * np.max(A[0:1].real) ** 0.05
            A = mag_target * A / mag
            candidate = np.clip(np.fft.irfft(A, N_fft)[:n], 0, None)
        elif r < 0.85:
            # Symmetry push: move toward symmetric
            sym = 0.5 * (current + current[::-1])
            alpha = rng.uniform(0.05, 0.2)
            candidate = (1 - alpha) * current + alpha * sym
            candidate += 0.005 * rng.standard_normal(n)
        else:
            # Global noise
            candidate *= np.exp(rng.normal(0, 0.02, size=n))
        
        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / s * np.sum(current)
        
        new_score = compute_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 5:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    current = candidate
                    current_score = new_score
        
        temp = max(temp * cooling, 1e-6)
    
    return best, best_score


def upsample(a, new_n, rng):
    """Upsample via interpolation + small noise."""
    n = len(a)
    if new_n <= n:
        return a[:new_n].copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new += 0.01 * rng.standard_normal(new_n)
    a_new = np.clip(a_new, 1e-4, None)
    return a_new / np.sum(a_new)

"""
Multi-strategy optimization for minimizing 2n*max(self-convolution)/sum^2 of non-negative sequences.
Uses FFT-based gradient descent with annealed soft-max, window-function initialization,
spectral shaping, and peak-directed local search. Key innovation: cross-correlation-based
gradient computation and adaptive temperature scheduling with spectral exploration.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def soft_max_value_and_weights(c, tau):
    """Compute soft max of c and the softmax weights."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    return c_max + tau * np.log(s), e / s


def gradient_soft_max_conv(a, tau):
    """
    Compute soft_max(conv(a,a), tau) and its gradient w.r.t. a.
    Uses cross-correlation via FFT for efficient gradient computation.
    Returns (value, grad, conv).
    """
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_value_and_weights(c, tau)

    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid (0 <= k-j < n)
    # = 2 * cross_correlation(w, a)[j]
    L = 1
    while L < 3 * n - 2:
        L <<= 1
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    r = np.fft.irfft(W * np.conj(A), L)[:n]
    grad = 2.0 * r
    return value, grad, c


def softmax_normalize(theta):
    """Stable softmax."""
    theta = theta - np.max(theta)
    e = np.exp(theta)
    return e / np.sum(e)


def init_window(n, window_type, rng):
    """Initialize with various window functions."""
    t = np.arange(n)
    if n > 1:
        t_norm = t / (n - 1)
    else:
        t_norm = t
    if window_type == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t_norm))
    elif window_type == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t_norm) + 0.08 * np.cos(4 * np.pi * t_norm)
    elif window_type == 'flattop':
        a = (0.21557895 - 0.41663158 * np.cos(2 * np.pi * t_norm) +
             0.27726316 * np.cos(4 * np.pi * t_norm) - 0.08357895 * np.cos(6 * np.pi * t_norm) +
             0.00694737 * np.cos(8 * np.pi * t_norm))
    elif window_type == 'gaussian':
        center = (n - 1) / 2.0
        sigma = max(1.0, n / 4.0)
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif window_type == 'sinc_clipped':
        center = (n - 1) / 2.0
        a = np.sinc((t - center) / max(1.0, n * 0.25))
        a = np.clip(a, 0.01, None)
    elif window_type == 'raised_cos_sym':
        # Raised cosine with symmetry emphasis
        a = 0.5 * (1 + np.cos(np.pi * t_norm))
        a = a ** 1.5
    else:
        a = np.ones(n)

    if rng is not None:
        a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 1e-3, None)
    return a / np.sum(a)


def init_fft_phase(n, rng, n_iter=100, target_exp=1.0):
    """GS-style initialization targeting flat |spectrum|^target_exp."""
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1

    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-12)
        # Target: |A|^2 flat => |A| constant
        target_mag = np.mean(mag) ** (1.0 - target_exp * 0.1) * mag_safe ** (target_exp * 0.1)
        A_new = target_mag * A / mag_safe
        a_full = np.fft.irfft(A_new, N_fft)
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def init_random(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def init_symmetric(n, base, rng):
    """Make a sequence symmetric."""
    a = base.copy()
    a = 0.5 * (a + a[::-1])
    a = np.clip(a, 1e-4, None)
    a += 0.005 * rng.standard_normal(n)
    a = np.clip(a, 1e-4, None)
    a = 0.5 * (a + a[::-1])
    return a / np.sum(a)


def gradient_optimize(n, a_init, time_limit, rng, tau_start=None, tau_end=None, lr0=2.0):
    """
    Adam gradient descent on softmax(theta) parameterization.
    Minimizes soft_max(conv(a,a), tau) with annealed tau.
    Returns (best_a, best_score).
    """
    if tau_start is None:
        tau_start = 80.0 / n
    if tau_end is None:
        tau_end = 0.05 / n

    a = a_init.copy()
    a = a / np.sum(a)
    theta = np.log(a + 1e-15)

    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    step = 0
    t0 = time.monotonic()

    best_theta = theta.copy()
    best_score = compute_score(a)
    last_improve_step = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.35)

        a = softmax_normalize(theta)
        value, grad_a, c = gradient_soft_max_conv(a, tau)

        # Project to theta gradient
        inner = np.dot(a, grad_a)
        grad_theta = a * (grad_a - inner)

        # Check for gradient explosion
        g_norm = np.linalg.norm(grad_theta)
        if np.isfinite(g_norm) and g_norm > 1e6:
            grad_theta *= 1e6 / g_norm

        step += 1
        m1 = b1 * m1 + (1 - b1) * grad_theta
        m2 = b2 * m2 + (1 - b2) * (grad_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.05 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # Periodic evaluation
        if step % 300 == 0:
            a_check = softmax_normalize(theta)
            s = compute_score(a_check)
            if s < best_score:
                best_score = s
                best_theta = theta.copy()
                last_improve_step = step
            if step % 1500 == 0:
                print(f"          step={step}, score={s:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={elapsed:.1f}s")

            # If stagnated, add small perturbation
            if step - last_improve_step > 6000 and step % 3000 == 0:
                theta += 0.5 * rng.standard_normal(n)
                m1 *= 0.5
                m2 *= 0.5

    a_final = softmax_normalize(best_theta)
    score_final = compute_score(a_final)
    if score_final < best_score:
        best_score = score_final
    return a_final, best_score


def local_search_peak(a_init, time_limit, rng, top_k=5):
    """
    Simulated annealing with peak-directed and spectral perturbations.
    """
    current = a_init.astype(np.float64).copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9998
    iter_count = 0
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        iter_count += 1

        if r < 0.35:
            # Peak-directed reduction
            conv = fft_convolve(current, current)
            n_top = min(top_k, len(conv))
            top_lags = np.argsort(conv)[-n_top:]
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            n_red = max(2, n // 12)
            top_idx = np.argsort(contrib)[-n_red:]
            bot_idx = np.argsort(contrib)[:n_red]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=n_red))
            candidate[bot_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_red))
        elif r < 0.55:
            # Multi-block perturbation
            n_blocks = int(rng.integers(2, 5))
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 8))))
                bs = int(rng.integers(0, n - bw))
                sign = rng.choice([-1, 1])
                mag = rng.uniform(0.02, 0.08)
                candidate[bs:bs + bw] *= np.exp(sign * mag)
        elif r < 0.70:
            # Spectral shaping
            A = np.fft.rfft(current, N_fft)
            # Slightly compress dynamic range of magnitude
            mag = np.abs(A)
            mag_min = np.percentile(mag[mag > 0], 5) if np.any(mag > 0) else 1e-10
            mag_target = mag ** 0.92 * (mag_min / np.min(mag[mag > 0] + 1e-10)) ** 0.08 if np.any(mag > 0) else mag
            A_new = mag_target * A / (mag + 1e-15)
            # Small random phase
            A_new *= np.exp(1j * rng.normal(0, 0.03, size=len(A_new)))
            candidate = np.clip(np.fft.irfft(A_new, N_fft)[:n], 0, None)
        elif r < 0.82:
            # Symmetry push + noise
            sym = 0.5 * (current + current[::-1])
            alpha = rng.uniform(0.05, 0.15)
            candidate = (1 - alpha) * current + alpha * sym
            candidate += 0.003 * rng.standard_normal(n)
        elif r < 0.92:
            # Sparse modification: pick a few positions and modify significantly
            n_mod = int(rng.integers(3, max(4, n // 10)))
            idx = rng.choice(n, size=n_mod, replace=False)
            candidate[idx] *= np.exp(rng.normal(0, 0.15, size=n_mod))
        else:
            # Global small noise
            candidate *= np.exp(rng.normal(0, 0.01, size=n))

        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / s * np.sum(current)

        new_score = compute_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 5:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    current = candidate
                    current_score = new_score

        temp = max(temp * cooling, 1e-7)

    return best, best_score


def upsample(a, new_n, rng):
    """Upsample via interpolation + noise."""
    n = len(a)
    if new_n <= n:
        return a[:new_n].copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new += 0.008 * rng.standard_normal(new_n)
    a_new = np.clip(a_new, 1e-4, None)
    return a_new / np.sum(a_new)


def spectral_refinement(a, n_iter=50, rng=None):
    """
    Iterative spectral flattening with non-negativity projection.
    Aims to make |FFT(a)|^2 as flat as possible.
    """
    n = len(a)
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    a_cur = a.copy()

    for it in range(n_iter):
        A = np.fft.rfft(a_cur, N_fft)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-12)
        # Compress magnitude toward flat
        target = mag ** 0.7 * np.mean(mag) ** 0.3
        A_new = target * (A / mag_safe)
        a_full = np.fft.irfft(A_new, N_fft)
        a_cur = np.clip(a_full[:n], 0, None)
        s = np.sum(a_cur)
        if s < 1e-10:
            a_cur = a.copy()
        else:
            a_cur = a_cur / s * np.sum(a)

    return a_cur

    lengths = [256, 384, 512, 768, 1024]

    for n in lengths:
        if time.monotonic() > deadline - 150:
            break

        # Window inits
        for wt in ['hann', 'blackman', 'flattop', 'gaussian', 'raised_cos_sym']:
            a = init_window(n, wt, rng)
            s = compute_score(a)
            candidates.append((s, n, a.copy(), f'win_{wt}'))

        # FFT phase init (two variants)
        a = init_fft_phase(n, rng, n_iter=80)
        s = compute_score(a)
        candidates.append((s, n, a.copy(), 'fft_phase1'))

        a = init_fft_phase(n, rng, n_iter=150, target_exp=0.5)
        s = compute_score(a)
        candidates.append((s, n, a.copy(), 'fft_phase2'))

        # Symmetric variants
        a_base = init_window(n, 'hann', rng)
        a_sym = init_symmetric(n, a_base, rng)
        s = compute_score(a_sym)
        candidates.append((s, n, a_sym.copy(), 'sym_hann'))

        # Random
        a = init_random(n, rng, spread=0.2 + 0.3 * rng.random())
        s = compute_score(a)
        candidates.append((s, n, a.copy(), 'random'))

    # Add provided sequence
    if start_seq is not None:
        candidates.append((best_score, len(start_seq), start_seq.copy(), 'provided'))

    # Spectral refinement on top few
    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates")
    print(f"  Top 5 before spectral: {[(f'{c[0]:.5f}', c[1], c[3]) for c in candidates[:5]]}")

    # Apply spectral refinement to top candidates
    refined = []
    for i in range(min(5, len(candidates))):
        if time.monotonic() > deadline - 120:
            break
        s0, n, a, name = candidates[i]
        a_ref = spectral_refinement(a, n_iter=30, rng=rng)
        s_ref = compute_score(a_ref)
        refined.append((s_ref, n, a_ref.copy(), name + '_spec'))
        if s_ref < best_score:
            best_score = s_ref
            best_seq = a_ref.copy()

    candidates.extend(refined)
    candidates.sort(key=lambda x: x[0])
    print(f"  Top 5 after spectral: {[(f'{c[0]:.5f}', c[1], c[3]) for c in candidates[:5]]}")

    # Phase 2: Gradient optimization on best candidates
    print("\n=== Phase 2: Gradient optimization ===")
    n_opt = min(6, len(candidates))
    remaining_total = deadline - time.monotonic() - 120

    for i in range(n_opt):
        if time.monotonic() > deadline - 90:
            break
        s0, n, a, name = candidates[i]

        # Allocate time
        remaining = deadline - time.monotonic() - 60
        n_left = n_opt - i
        t_alloc = remaining / max(1, n_left) * 0.65
        # Scale for computation cost
        t_alloc *= min(1.0, 384.0 / n)
        t_alloc = max(15.0, min(150.0, t_alloc))

        print(f"  [{i+1}/{n_opt}] n={n}, init={name}, score0={s0:.6f}, t={t_alloc:.0f}s")

        a_opt, s_opt = gradient_optimize(n, a, t_alloc, rng)
        print(f"    -> score={s_opt:.6f}")

        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")

    # Phase 3: Upsample best and optimize at higher resolution
    print("\n=== Phase 3: Upsampling ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        target_lengths = [l for l in [512, 1024, 2048] if l > n_cur]
        for target_n in target_lengths:
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic() - 50
            t_alloc = min(100.0, remaining * 0.4)
            if t_alloc < 15:
                break
            a_up = upsample(best_seq, target_n, rng)
            print(f"  Upsample n={n_cur} -> {target_n}, t={t_alloc:.0f}s")
            a_opt, s_opt = gradient_optimize(target_n, a_up, t_alloc, rng)
            print(f"    -> score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                n_cur = target_n
                print(f"    *** New best: {best_score:.6f} (n={target_n}) ***")

    # Phase 4: Spectral refinement + local search
    print("\n=== Phase 4: Spectral + Local search polish ===")
    if best_seq is not None:
        # First try spectral refinement
        if time.monotonic() < deadline - 60:
            a_spec = spectral_refinement(best_seq, n_iter=40, rng=rng)
            s_spec = compute_score(a_spec)
            print(f"  After spectral refinement: {s_spec:.6f}")
            if s_spec < best_score:
                best_score = s_spec
                best_seq = a_spec.copy()

        # Then local search
        remaining = deadline - time.monotonic() - 20
        polish_time = min(180.0, remaining * 0.5)
        if polish_time > 10:
            print(f"  Local search: n={len(best_seq)}, score={best_score:.6f}, t={polish_time:.0f}s")
            polished, p_score = local_search_peak(best_seq, polish_time, rng, top_k=6)
            if p_score < best_score:
                best_score = p_score
                best_seq = polished.copy()
                print(f"  After local search: {best_score:.6f}")

    # Phase 5: Additional optimization rounds
    remaining = deadline - time.monotonic() - 25
    if remaining > 40 and best_seq is not None:
        print(f"\n=== Phase 5: Additional rounds ===")
        n_best = len(best_seq)
        n_extra = min(4, int(remaining // 35))
        for r in range(n_extra):
            if time.monotonic() > deadline - 25:
                break
            # Different perturbation strategies
            if r == 0:
                a_new = best_seq * np.exp(rng.normal(0, 0.03, size=n_best))
            elif r == 1:
                a_new = best_seq.copy()
                a_new[:] = np.exp(rng.normal(np.log(np.mean(a_new)), 0.3, size=n_best))
                a_new = 0.5 * (a_new + a_new[::-1])
            elif r == 2:
                a_new = spectral_refinement(best_seq, n_iter=20, rng=rng)
                a_new += 0.02 * rng.standard_normal(n_best)
                a_new = np.clip(a_new, 0, None)
            else:
                a_new = init_window(n_best, 'flattop', rng)
                a_new = 0.5 * a_new + 0.5 * best_seq

            a_new = np.clip(a_new, 1e-4, None)
            a_new = a_new / np.sum(a_new)

            t_alloc = (deadline - time.monotonic() - 20) / max(1, n_extra - r) * 0.6
            t_alloc = max(10.0, min(60.0, t_alloc))
            print(f"  Round {r+1}/{n_extra}: t={t_alloc:.0f}s")
            a_opt, s_opt = gradient_optimize(n_best, a_new, t_alloc, rng)
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} ***")
                # Quick polish
                if deadline - time.monotonic() > 30:
                    p2, s2 = local_search_peak(best_seq, min(25.0, deadline - time.monotonic() - 20), rng)
                    if s2 < best_score:
                        best_score = s2
                        best_seq = p2.copy()

    # Final: verify and return
    print(f"\n{'='*60}")
    if best_seq is not None:
        # Ensure values are in valid range
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        local_s = compute_score(np.array(final_list))
        print(f"=== FINAL: evaluate_sequence={final_score:.6f}, local={local_s:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
