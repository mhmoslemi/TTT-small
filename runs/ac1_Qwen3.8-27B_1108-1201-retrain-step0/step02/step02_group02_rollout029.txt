    # Second term: let i = j-k, so k = j-i. Sum over i in [0, min(j, n-1)] of w[j-i] * a[i] (with k=j-i in [0,2n-2]).
    # = sum_i a[i] * w[j-i] * I[0<=j-i<=2n-2]
    # This is the convolution (a * w)[j] (restricted to valid range).
    
    # First term: let i = k-j, so k = i+j. Sum over i in [0, min(n-1, 2n-2-j)] of w[i+j] * a[i].
    # = sum_i a[i] * w[i+j]
    # This is the "anti-convolution" or (w * a_reversed) evaluated at some index.
    
    # Let me define a_rev[i] = a[n-1-i]. Then:
    # sum_i a[i] * w[i+j] = sum_i a_rev[n-1-i] * w[i+j]
    # Let t = n-1-i: = sum_t a_rev[t] * w[n-1-t+j] = (a_rev * w)[n-1+j]
    # where (a_rev * w) is the standard convolution.
    
    # So: First term at j = (a_rev * w)[n-1+j]
    # Second term at j = (a * w)[j]
    
    # Therefore: grad = 2 * (a * w)[0:n] ... no wait, the two terms are different.
    # grad[j] = (a_rev * w)[n-1+j] + (a * w)[j]
    
    # Let me verify with a small example.
    # a = [a0, a1, a2], w = [w0, w1, w2, w3, w4] (convolution of a with a has length 5)
    # grad[0] = w[0]*a[0] + w[0]*a[0] + w[1]*a[-1](invalid) + ... 
    # Using formula: (a_rev * w)[n-1+0] + (a * w)[0] = (a_rev * w)[2] + (a * w)[0]
    # a_rev * w: a_rev = [a2, a1, a0], convolved with w = [w0,w1,w2,w3,w4]
    # (a_rev * w)[2] = a_rev[0]*w[2] + a_rev[1]*w[1] + a_rev[2]*w[0] = a2*w2 + a1*w1 + a0*w0
    # (a * w)[0] = a0*w0
    # grad[0] = a2*w2 + a1*w1 + a0*w0 + a0*w0 = a0*2*w0 + a1*w1 + a2*w2
    
    # Direct computation: grad[0] = sum_k w[k] * (a[k-0] * I[0<=k-0<3] + a[0-k] * I[0<=0-k<3])
    # = sum_k w[k] * (a[k] * I[0<=k<3] + a[-k] * I[0<=-k<3])
    # I[0<=-k<3] means k in {-2,-1,0} but k>=0, so k=0.
    # = w[0]*(a[0]*1 + a[0]*1) + w[1]*(a[1]*1 + 0) + w[2]*(a[2]*1 + 0)
    # = 2*w[0]*a[0] + w[1]*a[1] + w[2]*a[2]
    
    # My formula gives: a0*2*w0 + a1*w1 + a2*w2. ✓
    
    # Great, so the gradient is:
    # grad[j] = (a_rev * w)[n-1+j] + (a * w)[j]
    
    a_rev = a[::-1]
    conv1 = fft_convolve(a_rev, w)  # (a_rev * w), length n + (2n-1) - 1 = 3n-2
    conv2 = fft_convolve(a, w)      # (a * w), length n + (2n-1) - 1 = 3n-2
    
    # First term at j: conv1[n-1+j], for j = 0, ..., n-1
    # Second term at j: conv2[j], for j = 0, ..., n-1
    grad = conv1[n-1:2*n-1] + conv2[0:n]
    
    return value, grad, c, w


def score_and_grad(a, tau):
    """
    Compute the score (soft-max smoothed) and gradient.
    Normalizes a to sum=1 first.
    """
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return 1e10, np.zeros_like(a)
    b = a / S  # normalize to sum=1
    
    value, grad, c, w = lse_smoothed_convolution(b, tau)
    
    # The objective is 2n * value (since score = 2n * max(b*b) when sum(b)=1)
    # But we need the gradient w.r.t. a (not b).
    # b = a/S, so db_i/da_j = (delta_ij * S - a_i) / S^2 = (delta_ij - b_i)/S
    # d_objective/da_j = 2n * sum_i (db_i/da_j) * grad_b_i = 2n/S * (grad_j - b_j * sum(grad))
    
    grad_a = (grad - b * np.sum(grad)) / S
    score = 2.0 * len(b) * value  # This is the smoothed score
    
    return score, grad_a


def lse_objective_and_grad(theta, tau, n):
    """
    Objective and gradient for L-BFGS, parameterized by theta.
    a = softmax(theta) normalized, so sum(a) = 1.
    """
    # Stable softmax
    theta_shifted = theta - np.max(theta)
    exp_theta = np.exp(theta_shifted)
    a = exp_theta / np.sum(exp_theta)
    
    # Compute convolution and soft-max
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    lse_value = c_max + tau * np.log(sum_exp)
    
    # The objective: 2n * lse_value (since sum(a)=1, score = 2n * max(a*a))
    # We want to minimize this.
    # But actually, since score = 2n * max(a*a) / (sum(a))^2 and sum(a)=1,
    # score = 2n * max(a*a). So minimizing the soft-max of a*a is equivalent.
    
    obj = 2.0 * n * lse_value
    
    # Gradient w.r.t. a:
    # d(obj)/da_j = 2n * sum_k w_k * (a_{k-j} + a_{j-k}) (with bounds)
    a_rev = a[::-1]
    conv1 = fft_convolve(a_rev, w)
    conv2 = fft_convolve(a, w)
    grad_a = conv1[n-1:2*n-1] + conv2[0:n]
    grad_a = 2.0 * n * grad_a
    
    # Gradient w.r.t. theta: grad_theta = J^T * grad_a where J is softmax Jacobian
    # J[i,j] = a_i * (delta_ij - a_j)
    # grad_theta_j = sum_i grad_a_i * a_i * (delta_ij - a_j) = grad_a_j * a_j - a_j * sum_i grad_a_i * a_i
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return obj, grad_theta

def peak_shave_polish(a_init, time_limit, rng):

        num_peaks = min(5, 2*n-1)
        top_indices = np.argsort(c)[::-1][:num_peaks]

        influence = np.zeros(n)
        for k in top_indices:
            for j in range(n):

                    influence[j] += c[k] * current[j] * current[k-j]
