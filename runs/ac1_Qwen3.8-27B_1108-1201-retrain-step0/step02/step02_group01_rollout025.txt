"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative refinement.

Core algorithm:
1. Start from given sequence or random initialization
2. Iteratively solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0
3. Blend LP solution with current sequence
4. Add randomization and coordinate descent for escape from local minima
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def build_sparse_A(f):
    """Build sparse constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    # Column j has f[0], f[1], ..., f[n-1] starting at row j
    rows = []
    cols = []
    data = []
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        data.extend(f.tolist())
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def lp_solve(f, M, n_constraints=None, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0."""
    n = len(f)
    total_lags = 2 * n - 1
    
    # Build constraint matrix
    A_full = build_sparse_A(f)
    
    if n_constraints is not None and n_constraints < total_lags:
        c_self = fft_convolve_self(f)
        active = np.argsort(c_self)[::-1][:n_constraints]
        A = A_full[active, :]
        b = np.full(len(active), M)
    else:
        A = A_full
        b = np.full(total_lags, M)
    
    c_obj = -np.ones(n)
    result = linprog(c_obj, A_ub=A, b_ub=b,
                     bounds=[(0, None)] * n, method='highs',
                     options={'time_limit': time_limit})
    if result.success:
        return result.x
    return None

def objective_value(a):
    """Compute the true objective: 2*n*max(conv)/sum^2."""
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * len(a) * c.max() / (S * S)

def run_lp_iteration(f, t_blend, n_constraints=None, time_limit=30):
    """Run one LP iteration and return updated sequence."""
    n = len(f)
    S = f.sum()
    c = fft_convolve_self(f)
    M = c.max()
    
    g = lp_solve(f, M, n_constraints, time_limit)
    if g is None:
        return None, M
    
    g_sum = g.sum()
    if g_sum < 1e-10:
        return None, M
    
    # Normalize g to have same sum as f
    g_norm = g * (S / g_sum)
    
    # Blend
    f_new = (1 - t_blend) * f + t_blend * g_norm
    f_new = np.maximum(f_new, 0.0)
    f_sum = f_new.sum()
    if f_sum < 1e-10:
        return None, M
    f_new = f_new * (S / f_sum)
    
    return f_new, M

def coordinate_descent(a, S, max_sweeps=50, time_budget=60):
    """Simple coordinate descent: try reducing each element."""
    n = len(a)
    a = a.copy()
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            # Try reducing a[i]
            for frac in [0.9, 0.8, 0.7, 0.5]:
                a[i] = orig * frac
                s = a.sum()
                a_norm = a * (S / s)
                c_check = fft_convolve_self(a_norm)
                obj = 2.0 * n * c_check.max() / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_norm.copy()
                else:
                    break
            a[i] = orig  # restore (we'll use best_a at the end)
    
    return best_a, best_obj

def gradient_step(a, S, lr, top_k=16, n_steps=100):
    """Subgradient/gradient step using soft-max weights."""
    n = len(a)
    a = a.copy()
    for step in range(n_steps):
        c = fft_convolve_self(a)
        c_max = c.max()
        
        # Soft-max weights
        T = 50.0
        shifted = T * (c - c_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        
        # Gradient: grad[i] = 2 * sum_j w[k] * a[k-i] for valid k
        # = 2 * (w * a_rev)[n-1 + i] where a_rev = a[::-1]
        # Using FFT:
        a_rev = a[::-1]
        fsize = 1 << (2*n - 1).bit_length()
        W = fft(w, fsize)
        Ar = fft(a_rev, fsize)
        grad_full = np.real(ifft(W * Ar))
        # grad[i] = 2 * grad_full[n-1+i]
        grad = 2.0 * grad_full[n-1:n-1+n]
        
        # Normalize to maintain sum
        grad = grad - grad.mean()  # tangent to simplex
        # Also ensure we don't go negative: use log-barrier-like projection
        a = a - lr * grad
        a = np.maximum(a, 1e-10)
        a = a * (S / a.sum())
    
    return a

**Spectral approach:**

The objective is 2n * max(a*a) / S^2. By Wiener-Khinchin, a*a = IFFT(|FFT(a)|^2). So we want to minimize the max of IFFT(|FFT(a)|^2).

The max of IFFT(P) is bounded by max(P) * sum(P) / (2n) ... no, that's not right.

Actually, IFFT(P)[k] = (1/(2n)) * sum_m P[m] * exp(2πimk/(2n)) (with appropriate normalization).

The peak of IFFT(P) is at most (1/(2n)) * sum_m P[m] = (1/(2n)) * ||P||_1.

But P[m] = |FFT(a)[m]|^2, and sum_m P[m] = (2n) * ||a||^2 (by Parseval).

So the peak is at most ||a||^2, which is a trivial bound.

A tighter bound: the peak of IFFT(P) at k=0 is (1/(2n)) * sum_m P[m] = ||a||^2. But the peak is typically not at k=0 for our problem (it's at k=n-1 for the linear convolution).

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    global_best_obj = np.inf
    global_best_a = None

    # --- Load given sequence ---
    given_a = None
    given_n = 0
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        raw = np.clip(raw, 0.0, 1000.0)
        if raw.size >= 32 and raw.sum() > 0.01:
            given_n = len(raw)
            S_g = np.sqrt(2.0 * given_n)
            given_a = raw * (S_g / raw.sum())
            obj_g = compute_obj(given_a)
            global_best_obj = obj_g
            global_best_a = given_a.copy()
            print("Given: n=%d, obj=%.6f" % (given_n, obj_g))
    except Exception as e:
        print("No given sequence: %s" % str(e))

    # --- Phase 1: LP iterations from given sequence (main workhorse) ---
    if given_a is not None:
        n = given_n
        S = np.sqrt(2.0 * n)
        f = given_a.copy()
        print("Phase 1: LP from given, n=%d" % n)
        it = 0
        phase1_end = deadline - 500.0

        while time.monotonic() < phase1_end:
            it += 1
            elapsed_frac = it / max(1, it + 50)  # rough progress
            if it <= 40:
                t = 1.0
                n_act = 2048
                tl = 20
            elif it <= 100:
                t = 0.5
                n_act = 4096
                tl = 20
            elif it <= 200:
                t = 0.3
                n_act = 4096
                tl = 15
            else:
                t = 0.2
                n_act = 4096
                tl = 15

            f_new, M, ok = lp_step(f, t, n_act, time_limit=tl)
            if ok:
                obj = compute_obj(f_new)
                if obj < global_best_obj:
                    global_best_obj = obj
                    global_best_a = f_new.copy()
                    if it % 10 == 1 or obj < global_best_obj - 1e-6:
                        print("  LP it=%d: obj=%.6f (t=%.1f, act=%d)" %
                              (it, obj, t, n_act))
                f = f_new

            # Periodic random perturbation to escape local minima
            if it % 25 == 0:
                noise_scale = 0.03 * f.mean()
                f = f + rng.random(n) * noise_scale
                f = np.maximum(f, 0.0)
                f = f * (S / f.sum())

        print("Phase 1 done: %d LP iters, best=%.6f" % (it, global_best_obj))

    # --- Phase 2: Try different lengths ---
    if time.monotonic() < deadline - 400.0:
        remaining_explore = min(300.0, deadline - time.monotonic() - 200.0)
        print("Phase 2: Multi-length exploration, %.0fs" % remaining_explore)

        # Try n=8192 with top-k LP (full LP would be too large)
        if remaining_explore > 60:
            n_try = 8192
            S_try = np.sqrt(2.0 * n_try)
            # Start from upsampled version of best_a or uniform
            if global_best_a is not None:
                f_try = np.interp(
                    np.linspace(0, len(global_best_a) - 1, n_try),
                    np.arange(len(global_best_a)),
                    global_best_a
                )
                f_try = np.maximum(f_try, 0) * (S_try / f_try.sum())
            else:
                f_try = np.full(n_try, S_try / n_try)

            obj_try = compute_obj(f_try)
            print("  n=8192 start: obj=%.6f" % obj_try)

            t0 = time.monotonic()
            it2 = 0
            while time.monotonic() - t0 < remaining_explore * 0.5:
                it2 += 1
                f_try, M2, ok2 = lp_step(f_try, 0.5, n_active=4096, time_limit=15)
                if ok2:
                    obj2 = compute_obj(f_try)
                    if obj2 < global_best_obj:
                        global_best_obj = obj2
                        global_best_a = f_try.copy()
                        print("  n=8192 it=%d: obj=%.6f" % (it2, obj2))

            print("  n=8192 done: %d iters" % it2)

        # Try a cosine-modulated initial for n=4096
        if remaining_explore > 120 and time.monotonic() < deadline - 200.0:
            n_cos = 4096
            S_cos = np.sqrt(2.0 * n_cos)
            i = np.arange(n_cos)
            # Anti-peak modulation: reduce center, boost edges
            a_cos = 1.0 + 0.3 * np.cos(2 * np.pi * i / n_cos)
            a_cos = np.maximum(a_cos, 0.01) * (S_cos / a_cos.sum())
            obj_cos = compute_obj(a_cos)
            if obj_cos < global_best_obj:
                global_best_obj = obj_cos
                global_best_a = a_cos.copy()
                print("  Cosine init: obj=%.6f" % obj_cos)

            # Run a few LP iters from this
            t1 = time.monotonic()
            for _ in range(10):
                if time.monotonic() - t1 > 60:
                    break
                a_cos, M3, ok3 = lp_step(a_cos, 0.7, n_active=2048, time_limit=15)
                if ok3:
                    obj3 = compute_obj(a_cos)
                    if obj3 < global_best_obj:
                        global_best_obj = obj3
                        global_best_a = a_cos.copy()

    # --- Phase 3: Gradient refinement ---
    if global_best_a is not None and time.monotonic() < deadline - 150.0:
        n = len(global_best_a)
        S = np.sqrt(2.0 * n)
        a = global_best_a * (S / global_best_a.sum())
        print("Phase 3: Gradient refinement, n=%d" % n)

        # Multiple passes with decreasing learning rate
        for lr, Tmax in [(0.003, 200), (0.001, 500), (0.0005, 1000)]:
            if time.monotonic() > deadline - 150.0:
                break
            t0 = time.monotonic()
            # Run for up to 30s per pass
            max_time = min(30.0, deadline - time.monotonic() - 150.0)
            if max_time < 5:
                break
            # Estimate steps
            n_steps = 200
            a = grad_step(a, S, lr, n_iters=n_steps,
                          T_schedule=lambda s: Tmax)
            obj = compute_obj(a)
            if obj < global_best_obj:
                global_best_obj = obj
                global_best_a = a.copy()
                print("  Grad lr=%.5f: obj=%.6f (%.1fs)" %
                      (lr, obj, time.monotonic() - t0))

    # --- Phase 4: Coordinate descent polish ---
    if global_best_a is not None and time.monotonic() < deadline - 20.0:
        n = len(global_best_a)
        S = np.sqrt(2.0 * n)
        a = global_best_a * (S / global_best_a.sum())
        remaining = min(120.0, deadline - time.monotonic() - 10.0)
        if remaining > 10:
            print("Phase 4: Coord descent, %.0fs" % remaining)
            a_best, obj_best = spectral_coord_descent(a, S, time_budget=remaining * 0.8)
            if obj_best < global_best_obj:
                global_best_obj = obj_best
                global_best_a = a_best
                print("  Coord: obj=%.6f" % obj_best)

            # Final tiny gradient step
            if time.monotonic() < deadline - 15.0:
                a = global_best_a * (S / global_best_a.sum())
                a = grad_step(a, S, 0.0002, n_iters=100,
                              T_schedule=lambda s: 2000.0)
                obj = compute_obj(a)
                if obj < global_best_obj:
                    global_best_obj = obj
                    global_best_a = a
                    print("  Final grad: obj=%.6f" % obj)

    # --- Final validation ---
    if global_best_a is None:
        global_best_a = np.ones(256, dtype=np.float64)

    global_best_a = np.clip(global_best_a, 0.0, 1000.0)
    if global_best_a.sum() < 0.01:
        global_best_a = np.ones(256, dtype=np.float64)

    result = global_best_a.tolist()
    ver = evaluate_sequence(result)
    print("FINAL: internal=%.6f, verified=%.6f, n=%d" %
          (global_best_obj, ver, len(result)))

    if not np.isfinite(ver):
        result = [1.0] * 256
        ver = evaluate_sequence(result)
        print("Fallback uniform: obj=%.6f" % ver)

    return result
