"""
Minimize 2*n*max(self_conv(a))/sum(a)^2 using Nesterov-accelerated Adam with
adaptive soft-max temperature, fast O(n) coordinate descent, and LP-based
directional refinement. Multi-start with diverse initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def softmax_obj_and_grad_a(a, T):
    """
    Compute soft-max approximation of max(self_conv(a)) and gradient w.r.t. a.
    Returns (objective, gradient, convolution_values, weights).
    """
    c = fft_convolve_self(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[c == c_max] = 1.0 / (c == c_max).sum()
        obj = c_max
    else:
        w = exp_shifted / total
        obj = c_max + np.log(total) / T
    
    # Gradient: grad[m] = 2 * sum_j w[m+j] * a[j]
    a_rev = a[::-1]
    n = len(a)
    full = fft_convolve(w, a_rev)
    grad = 2.0 * full[n - 1: 2 * n - 1]
    return obj, grad, c, w


def a_from_theta(theta, S):
    """a = S * softmax(theta)."""
    t = theta - theta.max()
    e = np.exp(t)
    z = e.sum()
    if z < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * e / z


def grad_theta_from_grad_a(grad_a, a, S):
    """Chain rule for log-parameterization."""
    dot = np.dot(grad_a, a)
    return grad_a * a - (a / S) * dot


def nesterov_adam(theta_init, n, S, time_budget, T_start=10.0, T_end=500.0,
                   lr_base=0.05, momentum=0.9, beta2=0.999):
    """Nesterov-accelerated Adam on log-parameters with adaptive temperature."""
    theta = theta_init.copy().astype(np.float64)
    n_par = len(theta)
    
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    beta1 = momentum
    
    start = time.monotonic()
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    best_theta = theta.copy()
    
    # Nesterov look-ahead state
    prev_m_hat = np.zeros(n_par)
    prev_v_hat = np.zeros(n_par)
    
    while time.monotonic() - start < time_budget:
        t += 1
        progress = (time.monotonic() - start) / time_budget
        
        # Adaptive temperature: exponential growth
        T = T_start * (T_end / T_start) ** (progress ** 0.5)
        
        # Nesterov look-ahead
        if t > 1:
            lr_prev = lr_base * (0.5 ** ((t-1) / 300.0))
            theta_look = theta - lr_prev * prev_m_hat / (np.sqrt(prev_v_hat) + 1e-8)
        else:
            theta_look = theta
        
        a_look = a_from_theta(theta_look, S)
        obj, grad_a, c, w = softmax_obj_and_grad_a(a_look, T)
        grad = grad_theta_from_grad_a(grad_a, a_look, S)
        
        # Update momentum
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        t_corr1 = 1 - beta1 ** t
        t_corr2 = 1 - beta2 ** t
        m_hat = m / t_corr1
        v_hat = v / t_corr2
        
        # Update
        lr = lr_base * (0.5 ** (t / 300.0))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Track best
        a = a_from_theta(theta, S)
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()
            best_theta = theta.copy()
        
        prev_m_hat = m_hat
        prev_v_hat = v_hat
    
    return best_theta, best_a, best_obj

"""
Minimize 2*n*max(self_conv(a))/sum(a)^2 using Nesterov-accelerated Adam with
adaptive soft-max temperature, fast O(n) coordinate descent with incremental
convolution updates, and LP-based directional refinement.
Multi-start with diverse structured initializations.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def softmax_obj_and_grad_a(a, T):
    """
    Compute soft-max approximation of max(self_conv(a)) and gradient w.r.t. a.
    Returns (objective, gradient, convolution_values, weights).
    """
    c = fft_convolve_self(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        mask = (c == c_max)
        cnt = mask.sum()
        w = mask.astype(np.float64) / cnt
        obj = c_max
    else:
        w = exp_shifted / total
        obj = c_max + np.log(total) / T
    a_rev = a[::-1]
    n = len(a)
    full = fft_convolve(w, a_rev)
    grad = 2.0 * full[n - 1: 2 * n - 1]
    return obj, grad, c, w


def a_from_theta(theta, S):
    """a = S * softmax(theta)."""
    t = theta - theta.max()
    e = np.exp(t)
    z = e.sum()
    if z < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * e / z


def grad_theta_from_grad_a(grad_a, a, S):
    """Chain rule for log-parameterization."""
    dot = np.dot(grad_a, a)
    return grad_a * a - (a / S) * dot


def nesterov_adam(theta_init, n, S, time_budget, T_start=5.0, T_end=1000.0,
                   lr_base=0.05, beta1=0.9, beta2=0.999, lr_decay=300.0):
    """Nesterov-accelerated Adam on log-parameters with adaptive temperature."""
    theta = theta_init.copy().astype(np.float64)
    n_par = len(theta)
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    start = time.monotonic()

    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    best_theta = theta.copy()

    prev_m_hat = np.zeros(n_par)
    prev_v_hat = np.zeros(n_par)
    prev_lr = lr_base

    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** (progress ** 0.7)

        if t > 1:
            theta_look = theta - prev_lr * prev_m_hat / (np.sqrt(prev_v_hat) + 1e-8)
        else:
            theta_look = theta

        a_look = a_from_theta(theta_look, S)
        obj, grad_a, c, w = softmax_obj_and_grad_a(a_look, T)
        grad = grad_theta_from_grad_a(grad_a, a_look, S)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_base * (0.5 ** (t / lr_decay))

        cur = c.max()

        prev_v_hat = v_hat

        if time.monotonic() - start > time_budget:

        for i in order:

            ai = a[i]

            best_local_val = c.max()

            # Generate trial values

            # Also try values based on the top lags

            for k in top_idx:

                if 0 <= k_idx < n:
                    slope = 2.0 * a[k_idx]
                    if abs(slope) > 1e-15:
                        # x that makes c[k] equal to average
                        target = c.mean()
                        x_trial = ai + (target - c[k]) / slope
                        if x_trial > 0:
                            trials.append(x_trial)

            for x in trials:
                if x <= 1e-15 or x > 50.0 * ai + 1.0:
                    continue
                dx = x - ai
                # Update c incrementally (on a copy for trial)
                k_start = i
                k_end = min(i + n, c_len)
                if k_start >= k_end:
                    continue
                a_slice = a[0:k_end - k_start]
                new_c_affected = c[k_start:k_end] + 2.0 * dx * a_slice
                if k_start + i < c_len and 2 * i >= k_start and 2 * i < k_end:
                    new_c_affected[2 * i - k_start] += dx * dx
                # Compute max
                max_aff = new_c_affected.max()
                max_left = c[:k_start].max() if k_start > 0 else 0.0
                max_right = c[k_end:].max() if k_end < c_len else 0.0
                val = max(max_aff, max_left, max_right)
                if val < best_local_val:
                    best_local_val = val
                    best_local = x

            if abs(best_local - ai) > 1e-10:
                # Apply update with incremental convolution
                dx = best_local - ai
                k_start = i
                k_end = min(i + n, c_len)
                c[k_start:k_end] += 2.0 * dx * a[0:k_end - k_start]
                if 2 * i < c_len:
                    c[2 * i] += dx * dx
                a[i] = best_local
                # Renormalize: a /= (1+dx/S_cur) approximately
                # Actually: new_sum = S + dx (since we worked with sum=S)
                new_sum = a.sum()
                if new_sum > 1e-10 and abs(new_sum - S) > 1e-8:

                    a *= scale
                    c *= scale * scale

                cur_obj = c.max()
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()

                a *= S / s

def solve_lp_direction(f, M, n, top_k_constraints=100, time_limit=15):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    from scipy.optimize import linprog

    c = fft_convolve_self(f)
    if top_k_constraints < len(c):
        active_lags = np.argsort(c)[::-1][:top_k_constraints]
    else:
        active_lags = np.arange(len(c))

    # Build constraint matrix: A[k_idx, j] = f[k-j] if 0<=k-j<n
    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        # j ranges from max(0, k-n+1) to min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[idx, j_lo:j_hi + 1] = f[k - j_hi:k - j_lo + 1][::-1]

    c_obj = -np.ones(n)
    b = np.full(len(active_lags), M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return None, None

Minimize 2*n*max(self_conv(a))/sum(a)^2 using Nesterov-accelerated Adam with

import numpy as np

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def softmax_obj_and_grad_a(a, T):
    """
    Compute soft-max approximation of max(self_conv(a)) and gradient w.r.t. a.
    Returns (objective, gradient, convolution_values, weights).
    """
    c = fft_convolve_self(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        mask = (c == c_max)
        cnt = mask.sum()
        w = mask.astype(np.float64) / cnt
        obj = c_max
    else:
        w = exp_shifted / total
        obj = c_max + np.log(total) / T
    a_rev = a[::-1]
    n = len(a)
    full = fft_convolve(w, a_rev)
    grad = 2.0 * full[n - 1: 2 * n - 1]
    return obj, grad, c, w


def a_from_theta(theta, S):

    t = theta - theta.max()

def grad_theta_from_grad_a(grad_a, a, S):
    """Chain rule for log-parameterization."""
    dot = np.dot(grad_a, a)

def solve_lp_direction(f, M, n, top_k_constraints=100, time_limit=15):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    from scipy.optimize import linprog

    c = fft_convolve_self(f)
    if top_k_constraints < len(c):
        active_lags = np.argsort(c)[::-1][:top_k_constraints]
    else:
        active_lags = np.arange(len(c))

    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[idx, j_lo:j_hi + 1] = f[k - j_hi:k - j_lo + 1][::-1]

    c_obj = -np.ones(n)
    b = np.full(len(active_lags), M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return None, None
