```python
def compute_segment_convolution(n, k):
    """
    For a step function with k equal-length segments of length L = n/k,
    compute the self-convolution at all breakpoints m = cL for c = 0, ..., 2k-2.
    Return a list of 2k-1 matrices, each of size k x k, such that
    b[cL] = sum_{j,j'} h_j * N[c][j][j'] * h_{j'}
    where h is the vector of segment heights.
    """
    L = n // k  # assuming n is divisible by k
    # Actually, let's require n = k * L exactly
    
    breakpoints = []
    for c in range(2*k - 1):
        m = c * L
        # b[m] = sum_{i=0}^{n-1} a[i] * a[m-i]
        # For each pair of segments (j, j'), count the number of i
        # such that i in seg j and m-i in seg j'
        N = np.zeros((k, k))
        for j in range(k):
            i_start_j = j * L
            i_end_j = (j+1) * L - 1
            for jp in range(k):
                # m - i in seg jp means:
                # jp * L <= m - i <= (jp+1) * L - 1
                # m - (jp+1)*L + 1 <= i <= m - jp*L
                i_start = max(i_start_j, m - (jp+1)*L + 1, 0)

                if i_end >= i_start:
                    N[j, jp] = i_end - i_start + 1
        breakpoints.append(N)
    return breakpoints
```

```python
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_smooth_max(c, beta):
    """Log-sum-exp smooth approximation to max."""
    c_max = c.max()
    return c_max + np.log(np.sum(np.exp(beta * (c - c_max)))) / beta

def lse_gradient(a, c, beta):
    """Gradient of LSE smooth max of self-convolution w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def fista_optimize(a_init, n, S, max_time, rng, beta_start=1.0, beta_end=200.0,
                    beta_growth=0.003, noise_interval=5000, noise_prob=0.1, noise_scale=0.003):
    """FISTA with adaptive LSE."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = beta_start * (1.0 + beta_growth * it)
        beta = min(beta, beta_end)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        grad = lse_gradient(y, c, beta)
        
        # Backtracking line search for step size
        step = 1.0 / (beta * 2.0)
        x_new = project_simplex(y - step * grad, S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-project y to simplex for numerical stability
        if it % 100 == 0:
            y = project_simplex(y, S)
        a = x_new
        t = t_new
        
        # Periodic noise for escape from local minima
        if it % noise_interval == 0 and rng.random() < noise_prob:
            y = project_simplex(y + rng.normal(0, noise_scale * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score

def subgradient_polish(a_init, n, S, max_time, rng, top_k=10):
    """Subgradient method targeting the max directly."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = 2.0 * S / (np.linalg.norm(grad) * (1.0 + 0.01 * it))
        step = min(step, 2.0 * S / n * 10)  # Cap the step
        a = project_simplex(a - step * grad, S)
        
        # Occasional restart with noise
        if it % 3000 == 0 and rng.random() < 0.05:
            a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
    
    return best_local, best_score
```
