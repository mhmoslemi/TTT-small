    n = len(a)
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b) => minimize -sum(b)
    # Constraints: for k = 0, ..., 2n-2: sum_i a[i]*b[k-i] <= M
    #             b[j] >= 0
    
    c = np.zeros(n)  # minimize -sum(b)
    c[:] = -1.0
    
    # A_ub * b <= bullet_b
    # For each k, the constraint is: sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*b[k-i] <= M
    A_ub = np.zeros((2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            A_ub[k, j] = a[i]
    
    # b >= 0 is handled by bounds
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x

def lp_iteration(a, M, rng):
    """One iteration of the LP-based optimization."""
    n = len(a)
    
    # Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0
    c = -np.ones(n)  # minimize -sum(b)
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n-1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            A_ub[k, j] = a[i]
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.status != 0:
        return None
    
    b = result.x
    sum_b = np.sum(b)
    if sum_b < 1e-10:
        return None
    
    # Rescale to match sum of a
    sum_a = np.sum(a)
    b = b * (sum_a / sum_b)
    
    return b

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

In fact, this is very analogous to the "Golomb ruler" problem or the "optimal placement" problem.

A (v, k, lambda) difference set is a k-subset D of a group of order v such that every non-identity element can be expressed as a difference of exactly lambda pairs of elements of D.

If we use the indicator of a difference set (padded to length v), then the autocorrelation has:
- Peak at lag 0: k
- Off-peak values: lambda (constant!)

So max autocorrelation = k, and sum = k, and score = 2v*k/k^2 = 2v/k.

For a (11, 5, 2) difference set: score = 2*11/5 = 4.4. Worse than uniform.

For a (7, 3, 1) difference set: score = 2*7/3 = 4.67. Also worse.

So difference sets don't directly help. The issue is that for binary sequences, the peak is always at least as large as the sum (since conv[0] = sum of squares = number of 1s = sum for binary).

For non-binary sequences, we can do better because the "effective peak" can be spread out.

"""
LP-based optimization for minimizing peak autocorrelation of non-negative sequences.
Uses the alternating LP method: fix current sequence, solve LP for next direction,
convex combine, repeat.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local score computation."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s ** 2))


def solve_lp(a, M):
    """Solve the LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    
    # Build constraint matrix efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # A_ub[k, j] = a[k - j] for j in [j_start, j_end]
            idx = np.arange(j_start, j_end + 1)
            A_ub[k, idx] = a[k - idx]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    
    if res.status == 0:
        return res.x
    return None

    """LP-based iterative optimization."""
    a = a_init.astype(np.float64).copy()

    best_score = local_score(a)
    
    t0 = time.monotonic()
    t = 0.15
    it = 0
    stall = 0
    current_M = np.max(fft_convolve(a, a))
    
    while time.monotonic() - t0 < time_limit:
        M = current_M
        
        b = solve_lp(a, M)
        if b is None:
            b = solve_lp(a, M * 1.05)
            if b is None:
                break
        
        sum_b = np.sum(b)
        if sum_b < 1e-10:
            break
        
        b = b * (sum_target / sum_b)
        
        # Try the convex combination
        a_new = (1.0 - t) * a + t * b
        a_new = np.clip(a_new, 0.0, 1000.0)
        new_sum = np.sum(a_new)
        if new_sum < 1e-6:
            break
        a_new = a_new * (sum_target / new_sum)
        
        new_M = np.max(fft_convolve(a_new, a_new))
        
        if new_M <= M + 1e-9:
            a = a_new
            current_M = new_M
