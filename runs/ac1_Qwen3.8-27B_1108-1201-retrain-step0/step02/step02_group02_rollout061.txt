```python
"""
Multi-scale search for low-peak nonnegative step functions.

Strategy:
- The objective is scale invariant, so all internal search uses sum(a)=1.
- Maintain the exact convolution vector and use a fast one-coordinate update
  rule that changes one coefficient and renormalizes the whole vector exactly.
- Alternate between:
  * smooth mirror-descent in log-space with an adaptive entropic peak weight;
  * exact-in-one-dimensional coordinate descent (with vectorized candidate tries);
- Use coarse-to-fine restarts from small lengths to escape local minima;
- Polish several promising sequences with the coordinate search.
"""

import numpy as np
import time


def next_pow2(k: int) -> int:
    if k <= 1:
        return 1
    return 1 << (int(k) - 1).bit_length()


def fft_convolve(a, b):
    a = np.ascontiguousarray(a, dtype=np.float64)
    b = np.ascontiguousarray(b, dtype=np.float64)
    out_len = len(a) + len(b) - 1
    size = next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    out = np.fft.irfft(fa * fb, size)
    return out[:out_len]


def self_conv(a):
    a = np.ascontiguousarray(a, dtype=np.float64)
    n = len(a)
    out_len = 2 * n - 1
    size = next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    out = np.fft.irfft(fa * fa, size)
    return np.maximum(out[:out_len], 0.0)


def normalize_sum(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-14:
        a = np.ones_like(a)
    else:
        a = a / s
    return a


def local_score(a):
    a = normalize_sum(a)
    n = len(a)
    if n == 0:
        return np.inf
    c = self_conv(a)
    return float(2.0 * n * np.max(c))


def derivative_from_weights(w, a):
    """
    If c_k = sum_i a_i a_{k-i}, then
    d(sum_k w_k c_k)/da_i = 2 sum_k w_k a_{k-i}
    (where a_{k-i} must be in range).
    """
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    n = len(a)
    return 2.0 * corr[n - 1:2 * n - 1]


def resample_sequence(a, n):
    a = normalize_sum(a)
    m = len(a)
    if m == n:
        return a.copy()
    if m == 1:
        return np.ones(n, dtype=np.float64) / n
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    b = np.interp(x_new, x_old, a)
    return normalize_sum(b)


def make_structured_init(n, rng, method):
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64) / max(1, n)
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0

    if method == "geometric":
        r = float(rng.uniform(0.55, 1.55))
        a = np.exp(np.arange(n, dtype=np.float64) * float(np.log(r)))
        if rng.random() < 0.5:
            a = a[::-1]
    elif method == "smooth_skew":
        a = 1.0 + float(rng.uniform(0.2, 1.2)) * t
        a += float(rng.uniform(0.0, 0.9)) * t * t
        a += 0.05 * rng.standard_normal(n)
    elif method == "two_block":
        a = np.ones(n)
        w1 = max(2, int(rng.integers(4, max(5, n // 3))))
        p1 = int(rng.integers(0, max(1, n // 4)))
        a[p1:p1 + w1] = float(rng.uniform(1.5, 2.5))
        a += 0.07 * rng.standard_normal(n)
    elif method == "random_log":
        a = np.exp(rng.normal(0.0, float(rng.uniform(0.5, 1.2)), size=n))
    elif method == "periodic_jitter":
        period = max(3, int(np.sqrt(n)))
        pattern = 1.0 + 0.5 * rng.standard_normal(period)
        period = max(3, int(period))
        a = np.tile(pattern, n // period + 2)[:n]
        a += 0.08 * rng.standard_normal(n)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)

    a = np.clip(a, 1e-5, None)
    return normalize_sum(a)

    n = len(a)
    if n <= 1:
        return a.copy(), local_score(a)
    if time_limit <= 0.0:
        return a.copy(), local_score(a)

    u = np.log(np.clip(a, 1e-300, None))
    u -= np.mean(u)

    c = self_conv(a)
    best_score = float(2.0 * n * np.max(c))
    best_a = a.copy()

    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.90, 0.999, 1e-8
    step = 0
    no_improve = 0
    beta = None

    k_start = max(3, min(int(k_start), n))
    k_end = max(2, min(int(k_end), k_start))

    t0 = time.monotonic()
    limit = float(time_limit)

    while time.monotonic() - t0 < limit:
        step += 1
        elapsed = time.monotonic() - t0

        score = float(2.0 * n * M)

            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            beta = 20.0 / M

        k_target = int(round(k_start + (k_end - k_start) * progress))
        k_target = max(2, min(k_target, n))

        t = (c - M) * beta
        t = np.clip(t, -60.0, 0.0)
        e = np.exp(t)
        se = float(np.sum(e))
        if not np.isfinite(se) or se <= 0.0:
            beta *= 0.5
            continue

        w = e / se
        sum_w2 = float(np.dot(w, w))
        if sum_w2 <= 1e-300:
            n_eff = 1.0
        else:
            n_eff = 1.0 / sum_w2

        # Adapt beta so the entrobic peak set has roughly k_target members.
        if n_eff < 0.5 * k_target:
            beta *= 0.60
        elif n_eff > 2.5 * k_target:
            beta *= 1.40

        beta_min = 1.0 / (200.0 * M)
        beta_max = 1.0 / (1e-4 * M)
        if beta < beta_min:
            beta = beta_min
        elif beta > beta_max:
            beta = beta_max

        g = derivative_from_weights(w, a)
        grad = g - float(np.sum(a * g))

        # Occasional rotation/perturbation if the search appears stuck.
        if no_improve > 2500 and step % 97 == 0:
            k = max(1, n // 6)
            idx = rng.choice(n, size=k, replace=False)
            u[idx] += 0.25 * rng.standard_normal(k)
            no_improve = 300

        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)

        # Learning rate: modest, with early boost and slow decay.
        base_lr = lr * (1.0 + 0.4 * np.exp(-step / 70.0)) / np.sqrt(1.0 + step / 150.0)
        u0 = u.copy()
        u = u - base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # Keep the log-coordinates numerically safe.
        u -= np.max(u)
        e = np.exp(np.clip(u, -60.0, 0.0))
        a_new = e / float(np.sum(e))
        if not np.all(np.isfinite(a_new)):
            u = u0
            a = a.copy()
            continue
        a = a_new

        c = self_conv(a)

        # Very periodic exact objective check / recovery.
        if step % 64 == 0:
            s = float(2.0 * n * np.max(c))
            if s < best_score - 1e-14:
                best_score = s
                best_a = a.copy()
                no_improve = 0

    best_score = min(best_score, local_score(best_a))
    return best_a, best_score


def cd_polish(a_init, time_limit, rng, max_candidates=14):
    """
    Coordinate descent with exact one-coordinate rescaling.

    Changing coordinate j by d changes the unnormalized convolution at indices
    j..j+n-1.  After division by (1+d)^2 to restore sum 1, the new score can
    be evaluated in O(n * D) for D candidate deltas.
    """
    a = normalize_sum(a_init)
    n = len(a)
    if n <= 1:
        return a.copy(), local_score(a)
    if time_limit <= 0.0:
        return a.copy(), local_score(a)

    c = self_conv(a)
    score = float(2.0 * n * np.max(c))
    best_a = a.copy()
    best_score = score

    rel_neg = np.array([-0.95, -0.80, -0.60, -0.40, -0.20, -0.10, -0.05], dtype=np.float64)
    rel_pos = np.array([0.05, 0.12, 0.25, 0.50, 0.90, 1.60], dtype=np.float64)
    base = 1.0 / (1.2 * n)
    abs_pos = np.array([0.02 * base, 0.08 * base, 0.25 * base, 0.60 * base, 1.2 * base, 2.5 * base],
                       dtype=np.float64)
    abs_neg = np.array([-0.05 * base, -0.20 * base, -0.60 * base], dtype=np.float64)

    t0 = time.monotonic()
    limit = float(time_limit)
    passes = 0

    while time.monotonic() - t0 < limit:
        passes += 1
        if passes % 4 == 1:
            c = self_conv(a)
            score = float(2.0 * n * np.max(c))
            if score < best_score - 1e-15:
                best_score = score
                best_a = a.copy()

        order = rng.permutation(n)
        pass_improved = False

        for j in order:
            if time.monotonic() - t0 > limit:
                break

            x = float(a[j])
            cands = []

            if x > 1e-12:
                for f in rel_neg:
                    d = float(f) * x
                    if d > -x + 1e-14 and (1.0 + d) > 0.02:
                        cands.append(d)
                for f in rel_pos:
                    d = float(f) * x
                    if d < 8.0 and (1.0 + d) > 0.02:
                        cands.append(d)

            for d in abs_pos:
                cands.append(float(d))

            if x > 3.0 * base:
                for d in abs_neg:
                    d = float(d)
                    if d > -x + 1e-14 and (1.0 + d) > 0.02:
                        cands.append(d)

            # One mild random proposal.
            if x > 1e-12:
                cands.append(float(x * rng.normal(0.0, 0.45)))
            else:
                cands.append(float(base * rng.exponential(1.0)))

            deltas = np.asarray(cands, dtype=np.float64)
            deltas = deltas[np.isfinite(deltas)]
            deltas = deltas[deltas > -x + 1e-14]
            deltas = deltas[1.0 + deltas > 0.02]
            deltas = deltas[np.abs(deltas) < 8.0]

            # De-duplicate.
            unique = []
            for d in deltas:
                d = float(d)
                ok = True
                for u in unique:
                    if abs(d - u) <= 1e-12 * max(1.0, abs(d), abs(u)):
                        ok = False
                        break
                if ok:
                    unique.append(d)
            if not unique:
                continue
            deltas = np.asarray(unique, dtype=np.float64)
            if len(deltas) > max_candidates:
                # Keep only the most interesting candidates.
                keep = min(max_candidates, len(deltas))
                # Keep small absolute ones plus extreme relative ones by simple truncation.
                deltas = deltas[:keep]

            cidx = c[j:j + n]
            old_other = 2.0 * a
            S = 1.0 + deltas
            S2 = S * S

            # Affected coefficients: (cidx + old_other[m] * d) / S^2, plus d^2
            # on the diagonal offset m=j.
            aff = (cidx[:, None] + old_other[:, None] * deltas[None, :]) / S2[None, :]
            if 0 <= j < n:
                aff[j, :] += deltas * deltas

            aff_max = np.max(aff, axis=0)

            # Unaffected coefficients merely get scaled by 1/S^2.
            left_max = float(np.max(c[:j])) if j > 0 else 0.0
            right_start = j + n
            right_max = float(np.max(c[right_start:])) if right_start < len(c) else 0.0
            unmax = max(left_max, right_max)

            cand_score = 2.0 * n * np.maximum(aff_max, unmax / S2)
            bi = int(np.argmin(cand_score))

            if cand_score[bi] < score - 1e-14:
                d = float(deltas[bi])
                bak_a = a.copy()
                bak_c = c.copy()

                a[j] = x + d
                a /= (1.0 + d)

                lin = old_other * d
                if 0 <= j < n:
                    lin[j] = 2.0 * x * d + d * d

                c[j:j + n] += lin
                c *= 1.0 / ((1.0 + d) * (1.0 + d))
                c = np.maximum(c, 0.0)

                new_score = float(2.0 * n * np.max(c))
                if new_score < score - 1e-15:
                    score = new_score
                    pass_improved = True
                    if score < best_score - 1e-15:
                        best_score = score
                        best_a = a.copy()
                else:
                    a = bak_a
                    c = bak_c

        # If a whole random-order pass gave no improvement, try a few stochastic
        # single-coordinate moves, but revert if they worsen the exact score.
        if not pass_improved and passes % 5 == 0:
            for _ in range(6):
                if time.monotonic() - t0 > limit:
                    break
                j = int(rng.integers(0, n))
                x = float(a[j])
                if x > 1e-12:
                    d = float(x * rng.normal(0.0, 0.35))
                else:
                    d = float(base * rng.exponential(1.0))

                if d <= -x + 1e-14 or (1.0 + d) <= 0.02 or abs(d) > 5.0:
                    continue

                bak_a = a.copy()
                bak_c = c.copy()
                old_other = 2.0 * a

                a[j] = x + d
                a /= (1.0 + d)
                lin = old_other * d
                if 0 <= j < n:
                    lin[j] = 2.0 * x * d + d * d
                c[j:j + n] += lin
                c *= 1.0 / ((1.0 + d) * (1.0 + d))
                c = np.maximum(c, 0.0)

                new_score = float(2.0 * n * np.max(c))
                if new_score < score:
                    score = new_score
                    if score < best_score - 1e-15:
                        best_score = score
                        best_a = a.copy()
                else:
                    a = bak_a
                    c = bak_c

    best_a = normalize_sum(best_a)
    best_score = min(best_score, local_score(best_a))
    return best_a, best_score


def top_candidates(candidates, k):
    """Return up to k best candidates as a list of (score, sequence)."""
    if not candidates:
        return []
    idx = list(range(len(candidates)))
    # Avoid lambda by simple insertion into a small list.
    chosen = []
    for i in idx:
        sc = candidates[i][0]
        seq = candidates[i][1]
        if not np.isfinite(sc):
            continue
        if len(chosen) < k:
            chosen.append((sc, seq, i))
        else:
            worst = 0
            for q in range(1, len(chosen)):
                if chosen[q][0] > chosen[worst][0]:
                    worst = q
            if sc < chosen[worst][0]:
                chosen[worst] = (sc, seq, i)
    chosen.sort(key=lambda z: z[0])
    return [(float(sc), seq) for sc, seq, _ in chosen]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_global = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 45.0)
    deadline = t_global + total_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None
    candidates = []

    def add_candidate(seq):
        nonlocal best_score, best_seq
        if seq is None:
            return
        a = normalize_sum(seq)
        if a.size == 0:
            return
        try:
            s = local_score(a)
        except Exception:
            s = np.inf
        if not np.isfinite(s):
            return
        candidates.append((float(s), a.copy()))
        if s < best_score:
            best_score = float(s)
            best_seq = a.copy()
            print(f"new best local score = {best_score:.9f}, n={len(best_seq)}", flush=True)

    # ------------------------------------------------------------------
    # Load any previously found sequence.
    # ------------------------------------------------------------------
    provided = None
    try:
        provided = globals().get("height_sequence_1", None)
    except Exception:
        provided = None

    if provided is not None:
        try:
            p = np.asarray(provided, dtype=np.float64).reshape(-1)
            if p.size > 0:
                p = np.clip(p, 0.0, 1000.0)
                add_candidate(p)
        except Exception:
            provided = None

    # Determine primary lengths to explore.
    lengths = []
    if provided is not None and len(provided) > 0:
        lp = int(len(provided))
        if lp >= 16 and lp <= 1024:
            lengths.append(lp)
    for x in (384, 512, 256, 768):
        if x not in lengths:
            lengths.append(x)

    # Keep enough time for polish.
    main_budget = 0.62 * total_budget
    polish_budget = 0.28 * total_budget
    reserve = total_budget - main_budget - polish_budget
    if reserve < 0:
        main_budget = max(0.0, main_budget + reserve)
        polish_budget = max(0.0, polish_budget - max(0.0, reserve))

    # ------------------------------------------------------------------
    # Quick baseline coordinate polish from the best available sequence.
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline - 120:
        tl = min(150.0, max(30.0, 0.07 * total_budget))
        try:
            polished, sc = cd_polish(best_seq, tl, rng)
            add_candidate(polished)
        except Exception as e:
            print("baseline cd failed:", e, flush=True)

    # ------------------------------------------------------------------
    # Build starting points.
    # ------------------------------------------------------------------
    starts = {}
    for n in lengths:
        starts[n] = []
        if best_seq is not None:
            starts[n].append(("provided", resample_sequence(best_seq, n)))
        for method in ["geometric", "smooth_skew", "random_log", "two_block", "periodic_jitter"]:
            starts[n].append((method, make_structured_init(n, rng, method)))
        for (name, seq) in list(starts[n]):
            add_candidate(seq)

    # ------------------------------------------------------------------
    # Coarse-to-fine exploration.
    # ------------------------------------------------------------------
    coarse_budget = min(160.0, max(60.0, 0.065 * total_budget))
    if time.monotonic() < deadline - coarse_budget - 120:
        coarse_lengths = [16, 32, 48]
        per_run = coarse_budget / max(1.0, len(coarse_lengths) * 3.0)
        coarse_runs = 3
        run_count = 0
        for nco in coarse_lengths:
            if time.monotonic() > deadline - coarse_budget / 4.0 - 120:
                break
            for r in range(coarse_runs):
                if time.monotonic() > deadline - 120:
                    break
                run_count += 1
                tl = min(per_run, max(2.0, time.monotonic() and (deadline - time.monotonic()) / max(1.0, run_count + 1)))
                if tl <= 0.5:
                    continue
                method = ["geometric", "smooth_skew", "random_log"][r % 3]
                a0 = make_structured_init(nco, rng, method)
                try:
                    ac, sc = smooth_optimize(a0, tl, rng, k_start=min(16, nco), k_end=max(2, nco // 12), lr=0.13)
                    for n in lengths[:4]:
                        if n <= 2 * nco or n <= 512:
                            add_candidate(resample_sequence(ac, n))
                except Exception as e:
                    print("coarse smooth failed:", e, flush=True)

    # ------------------------------------------------------------------
    # Main smooth refinement on promising starts per length.
    # ------------------------------------------------------------------
    main_time = min(max(30.0, main_budget), max(0.0, deadline - time.monotonic() - 120))
    if main_time > 10.0:
        # Weight 384 and provided length higher, but keep 512/768 active.
        weights = {}
        for i, n in enumerate(lengths):
            w = 1.0 / (i + 1.2)
            if provided is not None and int(len(provided)) == n:
                w *= 2.2
            if n == 384:
                w *= 1.8
            if n == 512:
                w *= 1.25
            if n == 256:
                w *= 0.85
            if n == 768:
                w *= 0.75
            weights[n] = weights.get(n, 0.0) + w

        total_w = sum(weights.values())
        for n in lengths:
            if time.monotonic() > deadline - 120:
                break
            w = weights.get(n, 0.0)
            if w <= 0:
                continue
            tl = main_time * (w / total_w)
            tl = min(tl, max(0.0, deadline - time.monotonic() - 120))
            if tl < 20.0:
                continue

            # Pick a few starting sequences for this length.
            st = starts.get(n, [])
            # Evaluate current local scores to sort approximately without a nested function.
            scored = []
            for name, seq in st:
                try:
                    s = local_score(seq)
                except Exception:
                    s = np.inf
                if np.isfinite(s):
                    scored.append((s, seq))
            scored.sort(key=lambda z: z[0])
            picked = scored[:2]

            for kst, (sn, seq) in enumerate(picked):
                if time.monotonic() > deadline - 120:
                    break
                alloc = tl if kst == 0 else max(10.0, tl * 0.35)
                alloc = min(alloc, max(0.0, deadline - time.monotonic() - 120))
                if alloc < 8.0:
                    continue
                try:
                    res, sc = smooth_optimize(seq, alloc, rng,
                                              k_start=min(120, max(12, n // 4)),
                                              k_end=min(20, max(6, n // 32)),
                                              lr=0.16)
                    add_candidate(res)
                except Exception as e:
                    print("main smooth failed for n=%d: %s" % (n, e), flush=True)

    # ------------------------------------------------------------------
    # Coordinate-descent polishing on the best candidates.
    # ------------------------------------------------------------------
    polish_time = min(max(60.0, polish_budget), max(0.0, deadline - time.monotonic() - 75))
    if polish_time > 30.0:
        top = top_candidates(candidates, 8)
        # Deduplicate by length and rough shape? Keep top few globally but limit memory.
        seen = set()
        polished_picks = []
        for sc, seq in top:
            key = (len(seq), int(round(np.round(np.array(seq, dtype=np.float64), 5)).tobytes(), 2))) if len(seq) <= 1024 else (len(seq),)
            # Simpler: keep only distinct lengths plus multiple best if same length.
            # We avoid hashing large bytes because it can be slow; use a signature instead.
            sig_vals = seq[::max(1, len(seq) // 16)]
            sig = (len(seq), int(round(float(np.mean(sig_vals))), 2))
            if sig in seen:
                continue
            seen.add(sig)
            polished_picks.append((sc, seq))
            if len(polished_picks) >= 5:
                break

        # Fallback to global best if none selected.
        if best_seq is not None and not polished_picks:
            polished_picks.append((best_score, best_seq))

        for i, (sc, seq) in enumerate(polished_picks):
            if time.monotonic() > deadline - 75:
                break
            frac = 1.0 / (i + 1.4)
            tl = polish_time * frac
            tl = min(tl, max(0.0, deadline - time.monotonic() - 75))
            if tl < 20.0:
                continue
            try:
                res, sc2 = cd_polish(seq, tl, rng, max_candidates=16 if len(seq) <= 512 else 10)
                add_candidate(res)
            except Exception as e:
                print("polish cd failed for n=%d: %s" % (len(seq), e), flush=True)

    # ------------------------------------------------------------------
    # Final very short coordinate search on the current global best.
    # ------------------------------------------------------------------
    if best_seq is not None:
        tl = max(0.0, deadline - time.monotonic() - 60.0)
        if tl > 25.0:
            try:
                res, sc = cd_polish(best_seq, tl, rng, max_candidates=12)
                add_candidate(res)
            except Exception as e:
                print("final cd failed:", e, flush=True)

    # ------------------------------------------------------------------
    # Return a valid plain list.
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: a simple valid sequence.
        best_seq = normalize_sum(np.ones(256, dtype=np.float64))

    out = [float(min(1000.0, max(0.0, x))) for x in np.asarray(best_seq, dtype=np.float64).reshape(-1)]
    final_eval = np.inf
    try:
        final_eval = evaluate_sequence(out)
    except Exception:
        final_eval = np.inf

    if final_eval < best_score:
        best_score = float(final_eval)

    print("FINAL propose_candidate score (local/eval): %.9f, n=%d" % (best_score, len(out)), flush=True)
    return out
```
