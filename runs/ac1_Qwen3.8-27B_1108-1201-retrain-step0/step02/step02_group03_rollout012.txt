Peak-suppression optimization for non-negative step functions minimizing
2*n*max(autocorr)/sum^2. Uses GS initialization, gradient descent with
annealed soft-max, iterative peak-directed suppression, and L-BFGS-B polish.

import numpy as np

from scipy.optimize import minimize as scipy_minimize

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def compute_gradient(a, tau=None):
    """Compute gradient of soft-max or hard-max of convolution."""
    n = len(a)
    c = fft_convolve(a, a)
    if tau is None or tau < 1e-10:
        # Hard max
        max_val = np.max(c)
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        e = np.exp(shifted)
        w = e / np.sum(e)
    
    # Gradient: (2/tau) * convolve(a_rev, w)[n-1:2n-1] for soft-max
    # For hard max (tau -> 0), gradient is 2 * a shifted

        g = 2.0 * corr[n-1:2*n-1]
    return g, c

def gs_initialize(n, rng, n_iter=100, target='flat', n_random=1):

    while N_fft < 2*n - 1:

    best_a = None
    best_score = np.inf

        else:

            mag = np.abs(A)
            mag[mag < 1e-15] = 1e-15

            elif target == 'raised':

def chirp_initialize(n, rng):

        e = np.exp(shifted)
        w = e / np.sum(e)
        
        a_rev = a[::-1]
        corr = fft_convolve(a_rev, w)
        g_a = (2.0 / tau) * corr[n-1:2*n-1]
        
        # Normalize gradient by sum constraint
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)
        
        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1**step)
        m2_hat = m2 / (1.0 - b2**step)
        
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        if step % 200 == 0:
            a_check = softmax_normalize(theta) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
    
    a_final = softmax_normalize(best_theta) * S
    final_score = local_score(a_final)
    if final_score < best_score:
        best_score = final_score

def peak_suppression(a, n_iter=50, rng=None):
    """
    Iteratively identify convolution peaks and apply targeted
    amplitude adjustments to reduce them.
    """
    n = len(a)
    a = a.copy()
    
    for it in range(n_iter):
        c = fft_convolve(a, a)
        max_val = np.max(c)
        max_idx = np.argmax(c)

        # conv(a,a)[max_idx] = sum_i a[i] * a[max_idx - i]
        contrib = np.zeros(n)
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
            # Also the symmetric contribution
            j2 = max_idx - i  # same thing for autocorrelation
        
        # Actually, for np.convolve, the gradient at position i is:
        # d/d(a[i]) conv[a,b][k] = b[k-i] if valid
        # For a=b: it's a[k-i] + a[i-(k-n+1)]... 
        # Let me just use the formula: grad[i] = a[max_idx - i] if 0 <= max_idx - i < n
        # Plus the symmetric: grad[i] += a[i - (max_idx - n + 1)] if valid
        
        grad = np.zeros(n)
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                grad[i] += a[j]
            # Symmetric (for the other factor)
            j2 = i - (max_idx - n + 1)  # This is for when i is the second index
            # Actually for convolve(a,a), d/d(a[i]) = a[max_idx-i] (first factor)
            # + a[i-(max_idx-n+1)] (second factor, if the convolution is symmetric)
            # For autocorrelation, it's just 2*a[max_idx-i] when max_idx >= n-1
            # and a[max_idx-i] when max_idx < n-1 (only one factor valid)
        
        # Simpler: just use the FFT-based gradient
        a_rev = a[::-1]
        w = np.zeros(2*n-1)
        w[max_idx] = 1.0
        corr = fft_convolve(a_rev, w)
        grad = 2.0 * corr[n-1:2*n-1]
        
        # Scale by 1/tau (with tau=1 for hard max, this is the subgradient)
        
        # Find top contributors
        n_reduce = max(3, n // 20)
        top_idx = np.argsort(grad)[-n_reduce:]
        bot_idx = np.argsort(grad)[:n_reduce]
        
        # Reduce top contributors, increase bottom
        alpha = 0.02 + 0.03 * (1 - it / n_iter)
        a[top_idx] *= np.exp(-alpha)
        a[bot_idx] *= np.exp(alpha)
        
        # Renormalize
        a = np.clip(a, 1e-10, None)
        a = a / np.sum(a) * np.sum(a)  # Keep same sum
    
    return a

def local_search(a_init, time_limit, rng, top_k=10):
    """Targeted local search with peak-directed mutations."""
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9997
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        c = fft_convolve(current, current)
        
        if r < 0.4:
            # Peak-directed: reduce elements contributing most to top peaks
            n_top = min(top_k, len(c))
            top_lags = np.argsort(c)[-n_top:]
            contrib = np.zeros(n)
            for lag in top_lags:
                a_rev = current[::-1]
                w = np.zeros(2*n-1)
                w[int(lag)] = 1.0
                corr = fft_convolve(a_rev, w)
                contrib += np.abs(corr[n-1:2*n-1])
            
            n_reduce = max(3, n // 15)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:max(3, n // 15)]
            
            alpha = 0.03 + 0.04 * rng.random()
            candidate[top_idx] *= np.exp(-alpha + 0.01*rng.standard_normal(len(top_idx)))
            candidate[bot_idx] *= np.exp(alpha + 0.01*rng.standard_normal(len(bot_idx)))
        elif r < 0.6:
            # Structured perturbation: modify a contiguous block
            width = max(2, int(rng.integers(2, max(3, n // 10))))
            start = int(rng.integers(0, n - width))
            direction = rng.choice([-1, 1])
            magnitude = 0.03 + 0.05 * rng.random()
            candidate[start:start+width] *= np.exp(direction * magnitude)
            # Compensate elsewhere
            comp_width = max(2, width)
            comp_start = int(rng.integers(0, n - comp_width))
            candidate[comp_start:comp_start+comp_width] *= np.exp(-direction * magnitude * width // max(1, comp_width))
        elif r < 0.75:
            # Pairwise amplitude swap (maintain sum approximately)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                ei = 0.05 + 0.1 * rng.random()
                ei = np.clip(ei, -0.5, 0.5)
                candidate[i] *= np.exp(ei)
                candidate[j] *= np.exp(-ei * current[i] / max(current[j], 1e-10))
        elif r < 0.85:
            # Symmetry nudge
            sym = 0.5 * (current + current[::-1])
            t = 0.1 + 0.2 * rng.random()
            candidate = (1-t) * current + t * sym
            candidate += 0.005 * rng.standard_normal(n)
        else:
            # Global perturbation
            candidate = current * np.exp(rng.normal(0, 0.02, size=n))
        
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current
        else:
            candidate = candidate / s * np.sum(current)
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * current_score:
                prob = np.exp(-delta / max(temp * current_score, 1e-12))
                if rng.random() < prob:
                    current = candidate
                    current_score = new_score
        
        temp *= cooling
    
    return best, best_score

def upsample(a, new_n, rng):
    n = len(a)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)

Actually, wait. Let me think about this differently.

The autocorrelation of a sequence is related to its power spectrum by the Wiener-Khinchin theorem. Specifically:
- If A(f) = DFT(a), then the autocorrelation r(k) = IDFT(|A(f)|^2)
- The convolution conv(a,a) is NOT the same as the autocorrelation (it's the correlation with one sequence reversed)

Actually, for real sequences:
- Autocorrelation: R(k) = sum_i a[i] * a[i+k]
- Convolution: (a*a_conv)[k] = sum_i a[i] * a[k-i]

These are related: (a*a_conv)[k] = R(k-n+1) (or something like that, depending on indexing).

Let me verify: for a = [a_0, a_1, ..., a_{n-1}]:
(a*a_conv)[k] = sum_{i=0}^{n-1} a[i] * a[k-i] (where a[j] = 0 for j outside [0,n-1])

The "correlation" is: (a star a)[k] = sum_i a[i+k] * conj(a[i]) = sum_i a[i+k] * a[i]

So (a*a_conv)[k] = (a star a)[k-n+1] (shifting index).

The power spectrum is |A(f)|^2 where A(f) = DFT(a). And by Wiener-Khinchin:
R(k) = IDFT(|A(f)|^2)[k]

So the convolution is a shifted version of the autocorrelation, and the autocorrelation is the IDFT of the power spectrum.

This means: to minimize the peak of the convolution, we need to minimize the peak of the IDFT of |A(f)|^2. This is equivalent to making |A(f)|^2 as flat as possible (which is what the GS algorithm does).

The improvement needed is small (about 0.6%). This suggests that the solution is near-optimal and we need very careful local optimization. The main things that could help:

1. **Better local search**: More targeted mutations that specifically address the convolution peaks
2. **Longer optimization time**: More iterations of gradient descent
3. **Different parameterization**: Maybe a different way to parameterize the sequence

Let me focus on making the local search much more effective, and also try a different gradient approach.

**New idea: "Convolution equalization"**

The idea is:
1. Compute the convolution c = conv(a, a)
2. The target is a flat convolution: c_target[k] = mean(c) for all k
3. The "error" is e[k] = c[k] - c_target[k]
4. We want to minimize ||e||_inf
5. The gradient of c[k] with respect to a is: 2 * (shifted a) for valid k
6. So the "force" on a[i] from lag k is proportional to a[k-i] * sign(e[k])
7. Apply a small update: a[i] -= eta * sum_k sign(e[k]) * a[k-i] * w[k]

This is like a "convolution equalizer" that tries to flatten the output.

Actually, this is essentially what the gradient descent is already doing (with soft-max weights instead of hard signs). The key difference would be in the step size schedule and the temperature annealing.

Let me try a different approach: **Use scipy.optimize with a smooth approximation of the max**.

The function f(a) = max_k conv(a,a)_k can be approximated by the log-sum-exp:
f_smooth(a) = tau * log(sum_k exp(conv(a,a)_k / tau))

As tau -> 0, this approaches the true max. The gradient is smooth and well-defined.

We can minimize f_smooth(a) subject to sum(a) = 1, a >= 0 using scipy.optimize.minimize with L-BFGS-B.

The key insight for making this work well:
- Start with a moderate tau (so the problem is smooth)
- Gradually decrease tau
- At each stage, run L-BFGS to convergence
- This is essentially "homotopy continuation"

Let me also think about the **number of elements n**. The previous best used n=256. Should we try different n?

For a given n, we have n degrees of freedom (minus 1 for the normalization). The convolution has 2n-1 values. We want to make them as equal as possible. With n degrees of freedom, we can control at most n-1 of the 2n-1 convolution values independently (the rest are determined). So for larger n, we have more control, but the problem is also harder to optimize.

I think n=256 to n=1024 is a good range to try.

"""
Homotopy-continuation optimization for minimizing peak autocorrelation
of non-negative step functions. Uses log-sum-exp smooth approximation
with decreasing temperature, GS initialization, and targeted local search.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def fft_convolve(a, b=None):
    """Compute linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute the evaluation score."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def logsumexp_conv(a, tau):
    """Compute log-sum-exp of convolution values (smooth max approximation)."""
    c = fft_convolve(a, a)
    c_max = np.max(c)
    return c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau)))

def logsumexp_conv_grad(a, tau):
    """Compute log-sum-exp and its gradient."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    w = e / np.sum(e)
    value = c_max + tau * np.log(np.sum(e))
    
    # Gradient: sum_k w[k] * d(conv[a,a][k])/d(a)
    # d(conv[a,a][k])/d(a[i]) = a[k-i] + a[k-i] (symmetric) = 2*a[k-i] for valid k-i
    # More precisely, use: conv(a_rev, w)[n-1:2n-1] * 2
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = 2.0 * corr[n-1:2*n-1]
    return value, g, c, w

def gs_init(n, rng, n_iter=100, target_type='flat', n_trials=3):
    """Gerchberg-Saxton initialization with multiple trials."""
    N_fft = 1
    while N_fft < 2*n - 1:
        N_fft <<= 1
    
    best_a = None
    best_score = np.inf
    
    for trial in range(n_trials):
        if trial == 0:
            a = np.ones(n) / n
        elif trial == 1:
            a = np.abs(rng.standard_normal(n)) + 0.05
            a /= np.sum(a)
        else:
            # Geometric decay
            a = np.exp(-np.arange(n) * 0.01 * (1 + rng.random()))
            a /= np.sum(a)
        
        for it in range(n_iter):
            A = np.fft.fft(a, N_fft)
            mag = np.abs(A)
            mag[mag < 1e-15] = 1e-15
            
            if target_type == 'flat':
                target = np.ones(N_fft) / np.sqrt(N_fft)
            elif target_type == 'tapered':
                # Tapered: lower at edges
                k = np.arange(N_fft)
                dist = np.minimum(np.abs(k - N_fft//2), N_fft//2 - np.abs(k - N_fft//2))
                target = 0.7 + 0.3 * np.cos(np.pi * dist / (N_fft//2))
                target /= np.mean(target)
            else:
                target = np.ones(N_fft) / np.sqrt(N_fft)
            
            phase = A / mag
            new_spectrum = target * phase
            a_full = np.fft.ifft(new_spectrum).real
            a = np.clip(a_full[:n], 0.0, None)
            s = np.sum(a)
            if s < 1e-10:
                a = np.ones(n) / n
            else:
                a /= s
        
        score = local_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

def homotopy_optimize(a_init, time_limit, rng, tau_stages=None):
    """
    Homotopy continuation: minimize log-sum-exp of convolution
    with decreasing temperature.
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    
    if tau_stages is None:
        # Default: geometric sequence from high to low
        tau_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
        tau_stages = [t / np.sqrt(n) for t in tau_stages]
    
    a = a_init.copy()
    a = a / np.sum(a) * S_target
    
    # Parameterize with log for positivity
    theta = np.log(a / S_target + 1e-15)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = local_score(a)
    
    for stage, tau in enumerate(tau_stages):
        stage_time_left = deadline - time.monotonic() if 'deadline' in dir() else None
        
        # Run L-BFGS-B at this temperature
        def obj(x):
            raw = np.exp(x)
            a_local = raw / np.sum(raw) * S_target
            val, _, _, _ = logsumexp_conv(a_local, tau)
            return val
        
        def obj_grad(x):
            raw = np.exp(x)
            a_local = raw / np.sum(raw) * S_target
            val, g_a, _, _ = logsumexp_conv(a_local, tau)
            # Chain rule: da_local/dx = a_local * (I - a_local/S_target * 1^T) ... 
            # Actually: a_local[i] = exp(x[i]) * S_target / sum(exp(x))
            # da_local[i]/dx[j] = a_local[i] * (delta[i,j] - a_local[j]/S_target) * S_target
            # Wait: a_local[i] = S_target * exp(x[i]) / S where S = sum(exp(x))
            # da_local[i]/dx[j] = S_target * (delta[i,j]*exp(x[j])*S - exp(x[i])*exp(x[j])) / S^2
            # = a_local[i] * (delta[i,j] - exp(x[j])/S)
            # = a_local[i] * (delta[i,j] - a_local[j]/S_target)
            
            inner = np.dot(g_a, a_local) / S_target
            g_x = g_a * a_local / S_target - a_local * inner / S_target
            # Hmm let me redo this.
            # Let e = exp(x), S = sum(e), a = S_target * e / S
            # da[i]/dx[j] = S_target * (delta[i,j]*e[j]*S - e[i]*e[j]) / S^2
            # = S_target * e[i]/S * (delta[i,j] - e[j]/S)
            # = a[i] * (delta[i,j] - e[j]/S)
            # = a[i] * (delta[i,j] - a[j]/S_target)
            
            # grad_x[j] = sum_i g_a[i] * da[i]/dx[j]
            # = sum_i g_a[i] * a[i] * (delta[i,j] - a[j]/S_target)
            # = g_a[j]*a[j] - a[j]*sum_i(g_a[i]*a[i])/S_target
            
            inner2 = np.dot(g_a, a_local)
            g_x = g_a * a_local - a_local * inner2 / S_target
            
            return g_x
        
        # Time budget for this stage
        elapsed = time.monotonic() - t0
        total_stages = len(tau_stages)
        remaining = time_limit - elapsed
        if remaining <= 0:
            break
        
        # Allocate more time to later (lower tau) stages
        weight = (stage + 1) ** 0.5
        stage_budget = remaining * weight / sum((s+1)**0.5 for s in range(stage, total_stages))
        stage_budget = min(stage_budget, remaining)
        
        maxiter = max(20, int(stage_budget * 5))
        
        try:
            result = scipy_minimize(
                obj, theta.copy(), jac=obj_grad,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8}
            )
            theta = result.x
        except Exception as e:
            print(f"    Stage {stage} (tau={tau:.4f}) failed: {e}")
            break
        
        a = np.exp(theta)
        a = a / np.sum(a) * S_target
        score = local_score(a)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        elapsed = time.monotonic() - t0
        print(f"    Stage {stage}: tau={tau:.4f}, score={score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_score

def targeted_local_search(a_init, time_limit, rng):
    """
    Very targeted local search that specifically addresses convolution peaks.
    Uses the gradient structure to make informed moves.
    """
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    
    # Multi-scale temperature
    base_temp = 0.002
    patience = 0
    max_patience = 500
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        peak_lag = np.argmax(c)
        peak_val = c[peak_lag]
        
        # Compute gradient at the peak
        a_rev = current[::-1]
        w = np.zeros(2*n-1)
        w[peak_lag] = 1.0
        corr = fft_convolve(a_rev, w)
        grad = 2.0 * corr[n-1:2*n-1]
        
        r = rng.random()
        candidate = current.copy()
        
        if r < 0.5:
            # Gradient step (deterministic)
            # Normalize gradient
            g_norm = grad / (np.linalg.norm(grad) + 1e-10)
            # Step size adapts to current score
            step_size = 0.001 * n / (np.sum(current) + 1e-10)
            candidate -= step_size * g_norm
            # Also add small random component
            candidate += 0.001 * step_size * rng.standard_normal(n)
        elif r < 0.7:
            # Top-k reduction
            n_top = max(5, n // 20)
            top_idx = np.argsort(grad)[-n_top:]
            bot_idx = np.argsort(grad)[:n_top]
            
            # Check if top and bottom overlap
            overlap = np.intersect1d(top_idx, bot_idx)
            if len(overlap) > n_top // 2:
                # Degenerate case, use random
                idx = rng.choice(n, size=2*n_top, replace=False)
                top_idx = idx[:n_top]
                bot_idx = idx[n_top:]
            
            alpha = 0.01 + 0.02 * rng.random()
            candidate[top_idx] *= np.exp(-alpha)
            candidate[bot_idx] *= np.exp(alpha)
        elif r < 0.8:
            # Ridge modification: modify elements along the "ridge" of the peak
            # The ridge at lag k involves pairs (i, k-i)
            offset = peak_lag - (n-1)  # How far from center
            for i in range(max(0, -offset), min(n, n - offset)):
                j = i + offset
                # This pair (i, j) contributes to the peak
                # Randomly adjust
                if rng.random() < 0.3:
                    delta = rng.normal(0, 0.01)
                    candidate[i] *= np.exp(delta)
                    candidate[j] *= np.exp(-delta)
        elif r < 0.9:
            # Block rotation: shift amplitudes within a block
            block_size = max(4, n // 32)
            start = int(rng.integers(0, n - block_size))
            shift = int(rng.integers(1, block_size // 2 + 1))
            block = current[start:start+block_size].copy()
            candidate[start:start+block_size] = np.roll(block, shift)
            # Smooth the transition
            blend = 0.3
            candidate[start:start+block_size] = (1-blend)*candidate[start:start+block_size] + blend*current[start:start+block_size]
        else:
            # Full gradient step with random restart component
            g_norm = grad / (np.linalg.norm(grad) + 1e-10)
            step_size = 0.002 * n / (np.sum(current) + 1e-10)
            candidate -= step_size * g_norm
            candidate += 0.01 * step_size * rng.standard_normal(n)
        
        # Project to non-negative
        candidate = np.maximum(candidate, 0.0)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current
        else:
            # Renormalize to keep same sum
            candidate = candidate / s * np.sum(current)
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            patience = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if patience == 0:
                    pass  # Reset patience on improvement
        else:
            patience += 1
            # Accept with probability based on "temperature"
            delta = new_score - current_score
            temp = base_temp * max(0.1, 1.0 - patience / max_patience)
            if delta < temp:
                prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < prob:
                    current = candidate
                    current_score = new_score
    
    return best, best_score

def upsample(a, new_n, rng):
    """Upsample via interpolation + noise."""
    n = len(a)
    x_old = np.linspace(0, 1, n + 1)
    x_new = np.linspace(0, 1, new_n + 1)
    a_padded = np.concatenate([a[:1], a, a[-1:]])
    a_interp = np.interp(x_new, x_old, a_padded)
    a_interp += 0.005 * rng.standard_normal(new_n + 1)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)
