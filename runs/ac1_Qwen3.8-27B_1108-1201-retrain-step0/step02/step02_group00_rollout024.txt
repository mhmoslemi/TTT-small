import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_lse_smooth_grad(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def compute_toppk_grad(a, c, top_k):
    """Subgradient using top-k active lags."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)

    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    if fsize < size:
        fsize <<= 1
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_smooth_grad(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a via FFT cross-correlation."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (3 * n - 3).bit_length()
    if fsize < 3 * n - 2:
        fsize <<= 1
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def toppk_subgrad(a, c, top_k):
    """Subgradient using top-k active lags, weighted by value."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)

    fsize = 1 << (3 * n - 3).bit_length()
    if fsize < 3 * n - 2:

    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

    n_tries = 0
    while time.monotonic() - start < max_time:

        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:

        # Try moving a small fraction of mass from i to j

        else:

            a[j] -= move

        a_res, _ = run_fista(init, n_c, S_c, per_explore, rng, beta_init=5.0, beta_max=300.0)
        a_res = safe_normalize(a_res, n_c)
        obj = score_seq(a_res)
        if obj < global_best:
            global_best = obj
            global_a = a_res.copy()
            print("  FISTA [%s]: %.6f" % (name, global_best))

    print("  After exploration: best=%.6f" % global_best)

    # === Phase 2: Intensive polishing (70% of time) ===
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        polish_time = 0.65 * budget_s
        print("=== Phase 2: Polishing (%.0f sec) ===" % polish_time)

        # 2a: FISTA with high beta
        seg_a = polish_time * 0.3
        t0 = time.monotonic()
        a_p, _ = run_fista(global_a, n_best, S_best, seg_a, rng, beta_init=50.0, beta_max=800.0)
        a_p = safe_normalize(a_p, n_best)
        obj = score_seq(a_p)
        if obj < global_best:
            global_best = obj
            global_a = a_p.copy()
            print("  FISTA high-beta: %.6f" % global_best)

        # 2b: Active-set Nesterov with line search
        remaining = deadline - time.monotonic() - 0.2 * budget_s
        seg_b = max(2.0, remaining * 0.4)
        a_p, _ = run_active_nesterov(global_a, n_best, S_best, seg_b, rng)
        a_p = safe_normalize(a_p, n_best)
        obj = score_seq(a_p)
        if obj < global_best:
            global_best = obj
            global_a = a_p.copy()
            print("  ActiveSet Nesterov: %.6f" % global_best)

        # 2c: Another FISTA pass with even higher beta
        remaining = deadline - time.monotonic() - 0.15 * budget_s
        if remaining > 5:
            a_p, _ = run_fista(global_a, n_best, S_best, remaining, rng, beta_init=100.0, beta_max=1000.0)
            a_p = safe_normalize(a_p, n_best)
            obj = score_seq(a_p)
            if obj < global_best:
                global_best = obj
                global_a = a_p.copy()
                print("  FISTA ultra-beta: %.6f" % global_best)

    # === Phase 3: Coordinate swap polish (15% of time) ===
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        remaining = deadline - time.monotonic() - 15
        if remaining > 3:
            print("=== Phase 3: Coord swap polish (%.0f sec) ===" % remaining)
            a_p, _ = coord_swap_polish(global_a, n_best, S_best, remaining, rng)
            a_p = safe_normalize(a_p, n_best)
            obj = score_seq(a_p)
            if obj < global_best:
                global_best = obj
                global_a = a_p.copy()
                print("  CoordSwap: %.6f" % global_best)

    # === Final verification ===
    if global_a is None:
        global_a = np.ones(256) * 0.5

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 0.5

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Final verification: %.6f" % ver if np.isfinite(ver) else "Final: inf")

    if not np.isfinite(ver):
        print("  Rejected, using fallback")
        final_list = [float(0.5)] * 256

    print("Returning length=%d, score=%.6f" % (len(final_list), ver))
    return final_list
