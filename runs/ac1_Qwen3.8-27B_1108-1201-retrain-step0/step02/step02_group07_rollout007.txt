2. Find the top contributors to conv[k*]: pairs (j, k*-j) where a[j]*a[k*-j] is large
3. For the top contributor (j1, j2) where j1 + j2 = k*:

   - Add δ to a position p that minimizes the increase in max conv
   - The position p should be one where a[p] * a[k*-p] and a[p] * a[p-p'] are small for all peak positions k'
4. Try multiple candidate positions p and pick the best

This is essentially a coordinate descent on the peak.

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)
    return result[:size]


def obj_from_conv(conv, n):
    """Objective given precomputed convolution (assumes sum(a)=1)."""
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project vector onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def delta_convolve_shift(a, j1, p, delta):
    """
    Compute the change in full convolution when a[j1] -= delta and a[p] += delta.
    Returns an array of length 2n-1 with the incremental change.
    """
    n = len(a)
    size = 2 * n - 1
    dc = np.zeros(size)
    # Contribution from a[p] += delta: changes conv[q] by delta * a[q-p] for valid q-p
    lo_p = max(0, p)
    hi_p = min(size - 1, p + n - 1)
    if lo_p <= hi_p:
        dc[lo_p:hi_p + 1] += delta * a[lo_p - p:hi_p - p + 1]
    # Contribution from a[j1] -= delta: changes conv[q] by -delta * a[q-j1] for valid q-j1
    lo_j = max(0, j1)
    hi_j = min(size - 1, j1 + n - 1)
    if lo_j <= hi_j:
        dc[lo_j:hi_j + 1] -= delta * a[lo_j - j1:hi_j - j1 + 1]
    return dc

def walsh_hadamard_init(n, rng):
    """Generate a zero-mean pseudo-random pattern using Walsh-Hadamard basis."""
    # Find next power of 2 >= n
    m = 1
    while m < n:
        m *= 2
    # Generate random Walsh-Hadamard coefficients
    coeffs = rng.standard_normal(m)
    coeffs[0] = 0  # Remove DC component
    # Inverse Hadamard (unnormalized)
    # Use FFT-based approach for large m
    h = np.zeros(m)
    # Simple approach: use random signs
    r = np.zeros(m)
    # Apply random Hadamard transform
    # For efficiency, use the fact that H*H = m*I
    # Generate random vector, apply H (which is its own inverse up to scaling)
    x = rng.standard_normal(m)
    # Hadamard transform via butterfly
    h = x.copy()
    h_len = 1
    while h_len < m:
        new_h = np.zeros_like(h)
        for i in range(0, m, 2 * h_len):
            for j in range(h_len):
                new_h[i + j] = h[i + j] + h[i + j + h_len]
                new_h[i + j + h_len] = h[i + j] - h[i + j + h_len]
        h = new_h
        h_len *= 2
    h /= np.sqrt(m)
    # Truncate to n
    r = h[:n]
    r -= r.mean()
    std = r.std()
    if std > 1e-10:
        r /= std
    return r


def quadrature_init(n, rng):
    """Initialize using quadratic residue structure."""
    # Use a simple pseudo-quadratic pattern
    x = np.arange(n)
    # Multiple frequency components
    r = np.zeros(n)
    for freq in [1, 2, 3, 5, 7, 11, 13]:
        r += np.cos(2 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
    r -= r.mean()
    std = r.std()
    if std > 1e-10:
        r /= std
    return r

def mirror_descent_opt(a0, time_limit, t_init=50.0, eta_init=0.3):
    """Mirror descent on simplex with adaptive parameters."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = obj_from_conv(conv, n)
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    max_pat = 40
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Compute smoothed subgradient
        max_c = conv.max()
        if max_c < 1e-15:
            break
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Compute subgradient: sg[j] = sum_k w[k] * a[k-j] for valid pairs
        sg = np.zeros(n)
        thresh = w.max() * 0.001
        active = np.where(w > thresh)[0]
        for ki in active:
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]

        # Mirror descent update
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s

        conv = fast_convolve(a)
        cur_obj = obj_from_conv(conv, n)
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_pat:
            patience = 0
            eta *= 0.6
            t *= 1.5
            if eta < 0.003:
                break

    return best_a, best_obj

            je = min(n - 1, k)
            if js <= je:
                sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]

        a_new = a - lr * sg
        a_new = project_simplex(a_new)
        a = a_new
        conv = fast_convolve(a)
        cur_obj = obj_from_conv(conv, n)
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_pat:
            patience = 0
            lr *= 0.7
            t *= 1.3
            if lr < 0.001:
                break

    return best_a, best_obj


def anneal_peak_targeted(a0, time_limit, rng):
    """Simulated annealing that targets peak reduction."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = obj_from_conv(conv, n)
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.01 * (1.0 - frac) ** 2 + 0.0003

        # Find peak
        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_k = int(np.argmax(conv))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if js > je:
            break

        j_arr = np.arange(js, je + 1)
        paired = peak_k - j_arr
        products = a[j_arr] * a[paired]
        total = products.sum()
        if total < 1e-15:
            break

        # Probabilistically pick a source from top contributors
        top_frac = 0.3
        n_top = max(1, int(len(j_arr) * top_frac))
        top_idx = np.argsort(products)[::-1][:n_top]
        probs = products[top_idx] / products[top_idx].sum()
        ti = int(rng.choice(n_top, p=probs))
        src = j_arr[top_idx[ti]]

        # Pick target: prefer low-contribution positions
        target = int(rng.integers(n))
        for _ in range(3):
            cand = int(rng.integers(n))
            # Check contribution to peaks
            contrib = 0.0
            d1 = peak_k - cand
            if 0 <= d1 < n:
                contrib += a[d1]
            d2 = cand - peak_k + 2 * n - 2
            if 0 <= d2 < n:
                contrib += a[d2]
            if contrib < a.mean() * 1.5:
                target = cand
                break

        if target == src:
            continue

        delta = a[src] * rng.uniform(0.02, 0.08)
        if delta < 1e-12:
            continue

        a[src] -= delta
        a[target] += delta
        if a[src] < 0:
            a[src] += delta
            a[target] -= delta
            continue

        s = a.sum()
        if s < 1e-12:
            a[src] += delta
            a[target] -= delta
            continue
        a /= s
        conv = fast_convolve(a)
        new_obj = obj_from_conv(conv, n)
        delta_obj = new_obj - cur_obj

        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
            cur_obj = new_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            a[src] += delta
            a[target] -= delta
            a /= a.sum() if a.sum() > 1e-12 else 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400):
    """
    Main entry point. Multi-phase optimizer combining:
    1. Quick scan over sizes and initializations
    2. Deep optimization with mirror descent + PGD
    3. Efficient peak-shaving local search
    4. Targeted simulated annealing
    Explores larger sequence lengths for more optimization freedom.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)

    parent = get_parent()

    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  NEW BEST: score=%.6f, n=%d" % (score, len(a)))

    # Phase 1: Quick scan
    print("Phase 1: Quick scan...")
    scan_lengths = [256, 512, 1024, 2048]
    scan_results = []
    phase1_budget = min(400.0, (deadline - time.monotonic()) * 0.18)

    for n in scan_lengths:
        if time.monotonic() > deadline - 1800:
            break
        inits = generate_inits(n, rng, parent)
        time_per = max(5.0, phase1_budget / max(1, len(scan_lengths) * len(inits)))

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 1800:
                break
            t_alloc = min(time_per, deadline - time.monotonic() - 1800)
            if t_alloc < 2.0:
                break
            a_opt, obj_opt = mirror_descent_opt(a0, time_limit=t_alloc, t_init=30.0, eta_init=0.5)
            score = safe_evaluate(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt))
                record(a_opt)
        print("  n=%d scanned, best so far: %.6f" % (n, best_score))

    print("  Scan complete. Best: %.6f" % best_score)
    scan_results.sort(key=lambda x: x[0])

    # Phase 2: Deep optimization of top candidates
    print("Phase 2: Deep optimization...")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 10:
            top_configs.append((n, name, a_opt))
            seen.add(key)

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    phase2_budget = min(1200.0, (deadline - time.monotonic()) * 0.50)
    n_deep = len(top_configs)
    time_per_deep = max(15.0, phase2_budget / max(1, n_deep))

    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 700:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 700)
        if t_alloc < 10.0:
            break
        print("  Deep [%d/%d] n=%d, %s, budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))

        a_cur = a0.copy()
        sub = t_alloc / 6.0

        # Cycle of optimization methods
        a_cur, _ = mirror_descent_opt(a_cur, time_limit=sub, t_init=40.0, eta_init=0.4)
        record(a_cur)
        a_cur, _ = pgd_optimize(a_cur, time_limit=sub, t_init=20.0, lr_init=0.05)
        record(a_cur)

        # Peak shaving
        conv_cur = fast_convolve(a_cur)
        t_ps = min(sub, deadline - time.monotonic() - 700)
        if t_ps > 3:
            n_iters = min(500, int(t_ps * 50))
            rng_ps = np.random.default_rng(seed + idx * 100)
            a_cur, conv_cur, _ = peak_shave_iter(a_cur, conv_cur, n_iters=n_iters,
                                                   n_trials=20, delta_frac=0.04, rng=rng_ps)
            record(a_cur)

        a_cur, _ = mirror_descent_opt(a_cur, time_limit=sub, t_init=100.0, eta_init=0.15)
        record(a_cur)
        a_cur, _ = pgd_optimize(a_cur, time_limit=sub, t_init=60.0, lr_init=0.03)
        record(a_cur)

        # More peak shaving
        conv_cur = fast_convolve(a_cur)
        t_ps2 = min(sub, deadline - time.monotonic() - 700)
        if t_ps2 > 3:
            n_iters2 = min(500, int(t_ps2 * 50))
            rng_ps2 = np.random.default_rng(seed + idx * 200)
            a_cur, conv_cur, _ = peak_shave_iter(a_cur, conv_cur, n_iters=n_iters2,
                                                  n_trials=20, delta_frac=0.02, rng=rng_ps2)
            record(a_cur)

    print("  Deep opt complete. Best: %.6f" % best_score)

    # Phase 3: Focused optimization on the best
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("Phase 3: Focused optimization on best...")
        a_best = best_sequence.copy()
        n_best = len(a_best)
        phase3_budget = min(600.0, deadline - time.monotonic() - 50)

        sub = phase3_budget / 5.0

        # Fine mirror descent
        a_cur, _ = mirror_descent_opt(a_best, time_limit=sub, t_init=200.0, eta_init=0.08)
        record(a_cur)

        # Fine PGD
        a_cur, _ = pgd_optimize(a_cur, time_limit=sub, t_init=100.0, lr_init=0.02)
        record(a_cur)

        # Extended peak shaving
        conv_cur = fast_convolve(a_cur)
        t_ps3 = min(sub * 1.5, deadline - time.monotonic() - 50)
        if t_ps3 > 5:
            n_iters3 = min(2000, int(t_ps3 * 80))
            rng_ps3 = np.random.default_rng(seed + 7777)
            a_cur, conv_cur, _ = peak_shave_iter(a_cur, conv_cur, n_iters=n_iters3,
                                                  n_trials=40, delta_frac=0.03, rng=rng_ps3)
            record(a_cur)

        # Annealing
        t_ann = min(sub, deadline - time.monotonic() - 50)
        if t_ann > 5:
            rng_ann = np.random.default_rng(seed + 9999)
            a_cur, _ = anneal_peak_targeted(a_cur, time_limit=t_ann, rng=rng_ann)
            record(a_cur)

        # Final peak shave
        if time.monotonic() < deadline - 5:
            conv_cur = fast_convolve(a_cur)
            t_final = deadline - time.monotonic() - 5
            n_iters_f = min(3000, int(t_final * 100))
            rng_f = np.random.default_rng(seed + 12345)
            a_cur, conv_cur, _ = peak_shave_iter(a_cur, conv_cur, n_iters=n_iters_f,
                                                  n_trials=50, delta_frac=0.01, rng=rng_f)
            record(a_cur)

    print("Phase 3 complete. Best: %.6f" % best_score)

    # Phase 4: Try larger sequences if time remains
    if time.monotonic() < deadline - 200:
        print("Phase 4: Trying larger sequences...")
        for n_large in [4096]:
            if time.monotonic() > deadline - 100:
                break
            inits = generate_inits(n_large, rng, parent)
            for name in ['uniform', 'walsh', 'multi_freq']:
                if name not in inits:
                    continue
                if time.monotonic() > deadline - 60:
                    break
                a0 = inits[name]
                t_alloc = min(80.0, deadline - time.monotonic() - 50)
                if t_alloc < 10:
                    break
                print("  Trying n=%d, init=%s, budget=%.0fs" % (n_large, name, t_alloc))
                a_opt, _ = mirror_descent_opt(a0, time_limit=t_alloc * 0.4, t_init=30.0, eta_init=0.4)
                conv_opt = fast_convolve(a_opt)
                t_ps = t_alloc * 0.3
                n_iters = min(300, int(t_ps * 40))
                rng_ps = np.random.default_rng(seed + n_large)
                a_opt, _, _ = peak_shave_iter(a_opt, conv_opt, n_iters=n_iters,
                                               n_trials=15, delta_frac=0.04, rng=rng_ps)
                a_opt, _ = pgd_optimize(a_opt, time_limit=t_alloc * 0.3, t_init=50.0, lr_init=0.03)
                record(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
