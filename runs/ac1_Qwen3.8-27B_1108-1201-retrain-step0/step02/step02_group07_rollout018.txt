import numpy as np

def solve_lp(f, M):
    """Solve: max sum(b) s.t. (f*b)(k) <= M for all k, b >= 0."""
    n = len(f)
    b = cp.Variable(n, nonnegative=True)
    
    # (f*b)(k) = sum_j f_j * b_{k-j} for valid j
    # This is a matrix-vector product: (F @ b)(k) <= M
    # where F is a (2n-1) x n matrix
    
    # Build F: F[k, j] = f_j if 0 <= k-j < n, else 0
    # i.e., F[k, j] = f_j if max(0, k-n+1) <= j <= min(n-1, k), else 0
    
    F = np.zeros((2*n-1, n))
    for j in range(n):
        for k in range(j, j+n):
            F[k, j] = f[j]
    
    constraints = [F @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve()
    
    if b.value is None:
        return None
    return np.maximum(b.value, 0)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_conv_matrix(f):
    """Build matrix F such that (F @ b)[k] = (convolve(f, b, 'full'))[k]."""
    n = len(f)
    F = np.zeros((2 * n - 1, n))
    for j in range(n):
        F[j:j + n, j] = f[j]
    return F


def solve_lp_direction(f, M, solver='ECOS'):
    """
    Solve: max sum(b) s.t. (f*b)(k) <= M for all k, b >= 0.
    Returns optimal b and sum(b).
    """
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    F = build_conv_matrix(f)
    constraints = [F @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=solver)
    if b.value is None:
        return None, 0.0
    b_val = np.maximum(b.value, 0)
    return b_val, b_val.sum()


def lp_iteration(f, t_list=None, solver='ECOS', max_iters=100, time_limit=None):
    """
    LP-based fixed-point iteration.
    f is normalized (sum=1).
    """
    if t_list is None:
        t_list = [1.0, 0.8, 0.5, 0.3, 0.2, 0.1]
    
    n = len(f)
    f = np.maximum(f, 0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(n) / n
    f = f / s
    
    conv = fast_convolve(f)
    M = conv.max()
    best_M = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        b, S = solve_lp_direction(f, M, solver=solver)
        if b is None or S < 1e-10:
            break
        
        g = b / S  # sum(g) = 1
        
        # Try different step sizes
        improved = False
        for t in t_list:
            f_trial = (1 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0)
            s_trial = f_trial.sum()
            if s_trial < 1e-15:
                continue
            f_trial = f_trial / s_trial
            conv_trial = fast_convolve(f_trial)
            M_trial = conv_trial.max()
            if M_trial < best_M * (1 - 1e-10):
                f = f_trial
                M = M_trial
                best_M = M_trial
                best_f = f.copy()
                conv = conv_trial
                improved = True
                break
        
        if not improved:
            # Try a larger M to get a better direction
            M *= 1.05
            conv_max = conv.max()
            if M < conv_max:
                M = conv_max * 1.01
    
    return best_f


def mirror_descent(a0, time_limit, t_init=50.0, eta_init=0.3):
    """Multiplicative mirror descent on the simplex."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    max_pat = 60
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        # Compute smoothed subgradient
        max_c = conv.max()
        if max_c < 1e-15:
            break
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient
        sg = np.zeros(n)
        for ki in range(len(conv)):
            if w[ki] < 1e-10:
                continue
            k = ki
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
        
        # Mirror descent update
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_pat:
            patience = 0
            eta *= 0.7
            t *= 1.4
            if eta < 0.003:
                eta = eta_init * 0.3
                t = t_init * 2.0
    
    return best_a, best_obj


def coord_descent_informed(a0, time_limit, rng, n_deltas=8):
    """
    Informed paired coordinate descent.
    Focuses on pairs that contribute to the current peak.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 500
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # Find peak positions
        peak_thresh = max_c * 0.9
        peak_positions = np.where(conv >= peak_thresh)[0]
        k_star = int(rng.choice(peak_positions))
        
        # Find contributing positions to peak k_star
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        contributors = a[js:je + 1]
        partner = a[k_star - je:k_star - js + 1][::-1]
        products = contributors * partner
        total = products.sum()
        if total < 1e-15:
            no_improve += 1
            continue
        
        # Probability proportional to contribution
        probs = products / total
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        
        # Choose target: prefer low values
        # Sample a few candidates and pick the one with lowest "pressure"
        target_j = int(rng.integers(n))
        for _ in range(10):
            cand = int(rng.integers(n))
            # Estimate pressure: sum of a[d] for d where cand interacts with peaks
            pressure = 0.0
            for pk in peak_positions[:5]:
                d1 = pk - cand
                if 0 <= d1 < n:
                    pressure += a[d1]
                d2 = cand - pk + 2 * n - 2
                if 0 <= d2 < n:
                    pressure += a[d2]
            if pressure < a[target_j] * 3 + 0.01:
                target_j = cand
                break
        
        if target_j == top_j:
            continue
        
        # Try several deltas
        a_top = a[top_j]
        if a_top < 1e-15:
            no_improve += 1
            continue
        
        best_delta = 0
        best_new_obj = best_obj
        
        for d_frac in np.linspace(0.01, 0.5, n_deltas):
            delta = a_top * d_frac
            a[top_j] -= delta
            a[target_j] += delta
            new_conv = fast_convolve(a)
            new_obj = 2.0 * n * new_conv.max()
            if new_obj < best_new_obj:
                best_new_obj = new_obj
                best_delta = delta
            a[top_j] += delta
            a[target_j] -= delta
        
        if best_delta > 0:
            a[top_j] -= best_delta
            a[target_j] += best_delta
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_a, best_obj


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def safe_evaluate(a):
    """Safely evaluate a sequence."""
    try:
        a_clean = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a_clean = np.minimum(a_clean, 1000.0)
        if a_clean.sum() < 0.01:
            return float('inf')
        if not np.all(np.isfinite(a_clean)):
            return float('inf')
        score = float(evaluate_sequence(a_clean.tolist()))
        return score
    except Exception:
        return float('inf')
