    cp = np.power(c, p-1)
    sum_cp = np.sum(c * cp)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_cp ** (1.0/p)
    score = 2.0 * n * pnorm
    factor = 4.0 * n * sum_cp ** (1.0/p - 1.0)
    a_rev = a[::-1]

    grad_a = factor * cross[n-1:2*n-1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta

# Direct max approximation using the top-K terms
def topk_obj_grad(theta, K, tau):
    """
    Approximate the max using a weighted sum of top-K terms (soft top-K).
    """

    a = softmax(theta)
    c = fft_convolve(a, a)
    # Use soft-max over all elements (simpler than top-K)
    c_shift = (c - c.max()) / tau
    w = np.exp(c_shift) / np.exp(c_shift).sum()
    soft_max_val = c.max() + tau * np.log(np.exp(c_shift).sum())
    score = 2.0 * n * soft_max_val
    
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n-1:2*n-1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta

# LP projection
def lp_project(f, M):
    """
    Find g >= 0 maximizing sum(g) subject to conv(f, g)[k] <= M for all k.
    Returns the optimal g.
    """
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) -> minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: for each k in 0..2n-2, conv(f, g)[k] <= M
    # conv(f, g)[k] = sum_i f[i] * g[k-i] where 0<=i<n and 0<=k-i<n
    # = sum_{i=max(0,k-n+1)}^{min(k, n-1)} f[i] * g[k-i]
    
    # Build constraint matrix A (shape: (2n-1, n))
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(k+1, n)):
            j = k - i  # index in g
            if 0 <= j < n:
                A[k, j] = f[i]
    
    b = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return f.copy()  # fallback

# Coordinate descent on peaks
def peak_coord_descent(a, n_iters, rng):
    """
    For each iteration, find the peak lag, identify top contributors,
    and reduce them while increasing others.
    """
    n = len(a)
    current = a.copy()
    current_score = local_score(current)
    
    for it in range(n_iters):
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]
        
        # Find top contributors to the peak
        contribs = np.zeros(n)
        for i in range(max(0, peak_idx - n + 1), min(peak_idx + 1, n)):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = current[i] * current[j]
        
        # Reduce top contributors, increase bottom ones
        top_k = max(5, n // 16)
        top_idx = np.argsort(contribs)[-top_k:]
        bot_idx = np.argsort(contribs)[:top_k]
        
        # Reduce top by factor, increase bottom by factor
        reduce_factor = np.exp(rng.uniform(0.02, 0.08, size=len(top_idx)))
        increase_factor = np.exp(rng.uniform(0.02, 0.08, size=len(bot_idx)))
        
        candidate = current.copy()
        candidate[top_idx] *= reduce_factor
        candidate[bot_idx] *= increase_factor
        candidate = np.maximum(candidate, 1e-10)
        candidate = candidate / candidate.sum()
        
        new_score = local_score(candidate)
        if new_score < current_score:
            current = candidate
            current_score = new_score
    
    return current, current_score

import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / e.sum()


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    cp = np.power(c, p - 1)
    sum_cp = np.sum(c * cp)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    factor = 4.0 * n * sum_cp ** (1.0 / p - 1.0)
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def soft_max_obj_grad(theta, tau):
    """Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau)."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score = 2.0 * n * soft_max_val
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def make_asymmetric_init(n, decay, rng):

def peak_coord_descent(a, n_iters, rng):

    At each step: identify the peak lag, reduce top contributors, increase bottom ones.
    """
    n = len(a)
    current = a.copy() / a.sum()
    current_score = local_score(current)

    for it in range(n_iters):

        # Get top 3 peaks

        # Compute contribution of each a[i] to the main peak

            j = int(main_peak) - i
            if 0 <= j < n:
                contribs[i] += current[i] * current[j]

        # Also check other peaks

            for i in range(max(0, int(pk) - n + 1), min(int(pk) + 1, n)):
                j = int(pk) - i
                if 0 <= j < n:
                    contribs[i] += 0.5 * current[i] * current[j]

        # Reduce top contributors, increase bottom
        top_k = max(8, n // 12)
        sorted_idx = np.argsort(contribs)
        top_idx = sorted_idx[-top_k:]
        bot_idx = sorted_idx[:top_k]

        # Multiple step sizes
        best_candidate = None
        best_new_score = current_score
        for step in [0.03, 0.05, 0.08, 0.12]:
            candidate = current.copy()
            candidate[top_idx] *= np.exp(-step)

        else:

        # Shift to get proper alignment

        P[mask] *= 0.85

        d2 = min(t_global + 350, deadline - 600)
        theta_best2, s2 = run_softmax_anneal(theta_best, tau_schedule, d2, maxiter_per_tau=250)
        a2 = softmax(theta_best2)
        print(f"  After soft-max: score={s2:.6f}")
        if s2 < best_score:
            best_score = s2
            best_seq = a2.copy()

    # ============================================================
    # Phase 2: Multi-start with diverse initializations
    # ============================================================
    print(f"\n=== Phase 2: Multi-start exploration (best={best_score:.6f}) ===")

    n_main = 256
    inits = []
    # Asymmetric inits
    inits.append(('exp_decay_1', make_asymmetric_init(n_main, 1.0, rng)))
    inits.append(('exp_decay_2', make_asymmetric_init(n_main, 2.0, rng)))
    inits.append(('exp_decay_3', make_asymmetric_init(n_main, 3.0, rng)))
    inits.append(('power_1', make_power_init(n_main, 0.5, rng)))
    inits.append(('power_2', make_power_init(n_main, 1.0, rng)))
    inits.append(('power_3', make_power_init(n_main, 1.5, rng)))
    inits.append(('chirp_1', make_chirp_init(n_main, 0.5, rng)))
    inits.append(('chirp_2', make_chirp_init(n_main, 0.8, rng)))
    inits.append(('ramp_1', make_ramp_init(n_main, rng, direction=1)))
    inits.append(('ramp_2', make_ramp_init(n_main, rng, direction=-1)))
    inits.append(('gauss_asym', make_gaussian_asym(n_main, n_main * 0.3, n_main * 0.2, rng)))

    # If we have a best, add perturbed versions
    if best_seq is not None and len(best_seq) == n_main:
        for scale in [0.05, 0.1, 0.2]:
            pert = best_seq * np.exp(rng.normal(0, scale, n_main))
            pert = np.maximum(pert, 1e-8)
            pert = pert / pert.sum()
            inits.append((f'pert_{scale}', pert))

    p_schedule_short = [8, 16, 32, 64]
    time_per_init = 25.0

    for init_name, a0 in inits:
        if time.monotonic() > deadline - 500:
            break
        d = min(time.monotonic() + time_per_init, deadline - 500)
        theta = np.log(a0 + 1e-30)
        theta_best, s = run_pnorm_anneal(theta, p_schedule_short, d, maxiter_per_p=100)
        a = softmax(theta_best)
        s_final = local_score(a)
        print(f"  n={n_main}, init='{init_name}': score={s_final:.6f}")
        if s_final < best_score:
            best_score = s_final
            best_seq = a.copy()
            print(f"    *** NEW BEST: {best_score:.6f} ***")

    # ============================================================
    # Phase 3: LP projection refinement
    # ============================================================
    print(f"\n=== Phase 3: LP projection refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        d3 = min(time.monotonic() + 150, deadline - 300)
        a_lp, s_lp = lp_refine(best_seq, n_iters=50, rng=rng)
        print(f"  After LP: score={s_lp:.6f}")
        if s_lp < best_score:
            best_score = s_lp
            best_seq = a_lp.copy()
            print(f"    *** NEW BEST: {best_score:.6f} ***")

    # ============================================================
    # Phase 4: Peak-directed coordinate descent
    # ============================================================
    print(f"\n=== Phase 4: Peak-directed coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_cd = 2000
        a_cd, s_cd = peak_coord_descent(best_seq, n_cd, rng)
        print(f"  After coord descent: score={s_cd:.6f}")
        if s_cd < best_score:
            best_score = s_cd
            best_seq = a_cd.copy()
            print(f"    *** NEW BEST: {best_score:.6f} ***")

        # Another LP pass after coordinate descent
        if time.monotonic() < deadline - 100:
            a_lp2, s_lp2 = lp_refine(best_seq, n_iters=30, rng=rng)
            print(f"  After 2nd LP: score={s_lp2:.6f}")
            if s_lp2 < best_score:
                best_score = s_lp2
                best_seq = a_lp2.copy()
                print(f"    *** NEW BEST: {best_score:.6f} ***")

    # ============================================================
    # Phase 5: Fourier-domain exploration
    # ============================================================
    print(f"\n=== Phase 5: Fourier exploration (best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        d5 = min(time.monotonic() + 120, deadline - 50)
        # Run multiple Fourier iterations
        a_four, s_four = fourier_explore(best_seq, 200, rng)
        print(f"  After Fourier: score={s_four:.6f}")
        if s_four < best_score:
            best_score = s_four
            best_seq = a_four.copy()
            print(f"    *** NEW BEST: {best_score:.6f} ***")

    # ============================================================
    # Phase 6: Try other lengths
    # ============================================================
    print(f"\n=== Phase 6: Other length exploration (best={best_score:.6f}) ===")
    if time.monotonic() < deadline - 200:
        for n_try in [200, 320, 384, 512]:
            if time.monotonic() > deadline - 100:
                break
            d = min(time.monotonic() + 60, deadline - 100)
            # Use best sequence interpolated to new length + perturbation
            if best_seq is not None:
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, n_try)
                a_interp = np.interp(x_new, x_old, best_seq)
                a0 = a_interp * np.exp(rng.normal(0, 0.1, n_try))
            else:
                a0 = make_asymmetric_init(n_try, 2.0, rng)
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            theta = np.log(a0 + 1e-30)
            # Also try pure asymmetric init
            a0b = make_asymmetric_init(n_try, 2.0 + rng.random(), rng)
            theta_b = np.log(a0b + 1e-30)

            best_local = (a0, local_score(a0))
            best_local_b = (a0b, local_score(a0b))

            for th in [theta, theta_b]:
                if time.monotonic() > d:
                    break
                theta_best, s = run_pnorm_anneal(th, [8, 16, 32, 64], d, maxiter_per_p=80)
                a = softmax(theta_best)
                s = local_score(a)
                if s < best_local[1]:
                    best_local = (a, s)

            a_loc, s_loc = best_local
            print(f"  n={n_try}: score={s_loc:.6f}")
            if s_loc < best_score:
                best_score = s_loc
                best_seq = a_loc.copy()
                print(f"    *** NEW BEST: {best_score:.6f} ***")

    # ============================================================
    # Phase 7: Final intensive refinement
    # ============================================================
    print(f"\n=== Phase 7: Final refinement (best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        # Very fine soft-max
        n = len(best_seq)
        theta = np.log(best_seq + 1e-30)
        tau_fine = [0.001, 0.0005, 0.0002, 0.0001]
        d7 = deadline - 10
        theta_best, s7 = run_softmax_anneal(theta, tau_fine, d7, maxiter_per_tau=200)
        a7 = softmax(theta_best)
        s7 = local_score(a7)
        print(f"  After final soft-max: score={s7:.6f}")
        if s7 < best_score:
            best_score = s7
            best_seq = a7.copy()

        # Final coordinate descent
        if time.monotonic() < deadline - 5:
            a_cd2, s_cd2 = peak_coord_descent(best_seq, 500, rng)
            print(f"  After final coord descent: score={s_cd2:.6f}")
            if s_cd2 < best_score:
                best_score = s_cd2
                best_seq = a_cd2.copy()

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if best_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return [1.0] * 256
