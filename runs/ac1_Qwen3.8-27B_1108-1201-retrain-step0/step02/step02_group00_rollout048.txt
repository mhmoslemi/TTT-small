"""
Minimize 2*n*max(conv) / sum^2 using adaptive Nesterov with:
- Backtracking line search
- Adaptive log-sum-exp sharpening
- Structured perturbations
- Multi-start with diverse initializations
- Power-sum smoothing alternative

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_grad(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), 0.0, 1.0
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    entropy = -np.sum(w * np.log(w + 1e-300))
    eff_support = np.exp(entropy)
    return grad, eff_support, 1.0


def power_sum_grad(a, c, p):
    """Gradient of (sum c^p)^(1/p) w.r.t. a, for even p >= 2."""
    n = len(a)
    c_pos = np.maximum(c, 0)  # c should be non-negative for non-negative a
    cp = c_pos ** (p - 1)
    total = (c_pos ** p).sum()
    if total < 1e-300:
        return np.zeros(n), 1.0
    val = total ** (1.0 / p)
    # d/dai [(sum c^p)^(1/p)] = (1/p) * (sum c^p)^(1/p - 1) * p * c^(p-1) * dc/dai
    # = (sum c^p)^(1/p - 1) * c^(p-1) * dc/dai
    # dc/dai = 2 * sum_k w_k * a_{k-i} where w_k is the "weight" at lag k
    # Actually, d(conv)[k]/da_i = a_{i+k} + a_{i-k} (for valid indices)
    # So grad_i = (sum c^p)^(1/p-1) * sum_k c[k]^(p-1) * (a_{i+k} + a_{i-k})
    # This is a cross-correlation of c^(p-1) with a
    w = c_pos ** (p - 1)
    w = w / total  # normalize
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = (total ** (1.0/p - 1.0)) * 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, 1.0


def nesterov_optimize(a_init, n, S, max_time, rng, beta_init=3.0, beta_max=500.0):
    """
    Nesterov-optimized log-sum-exp with backtracking and adaptive beta.
    Returns (best_sequence, best_score).
    """
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    base_step = 1.0 / (beta_init * 2.0)
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.003 * it + 0.00001 * it * it))
        
        c = fft_conv(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            last_improve = it
        
        grad, eff_support, _ = lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        step = base_step / (1.0 + 0.001 * it)
        step = max(step, 1e-8)
        
        # Backtracking: try this step and half
        x_try1 = project_simplex(y - step * grad, S)
        c_try1 = fft_conv(x_try1)
        score_try1 = c_try1.max()
        
        if score_try1 < cur:
            x_new = x_try1
        else:
            x_try2 = project_simplex(y - 0.5 * step * grad, S)
            c_try2 = fft_conv(x_try2)
            score_try2 = c_try2.max()
            if score_try2 < cur:
                x_new = x_try2
            else:
                x_new = project_simplex(y - 0.25 * step * grad, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic perturbation
        if it % 2000 == 0:
            cooling = max(0.1, 1.0 - it * 0.0001)
            scale = 0.008 * cooling * S / np.sqrt(n)
            # Mix of Gaussian and structured perturbation
            noise = rng.normal(0, 1, n)
            freq = rng.integers(1, n // 2, size=2)
            phase = rng.uniform(0, 2 * np.pi, size=2)
            structured = sum(np.sin(2 * np.pi * f * np.arange(n) / n + p) 
                            for f, p in zip(freq, phase))
            y = project_simplex(y + scale * (noise + 0.5 * structured), S)
            t = 1.0
            # Reset beta if stuck
            if it - last_improve > 5000:
                t = 1.0
                y = project_simplex(y + 0.02 * S / np.sqrt(n) * rng.normal(0, 1, n), S)
    
    return best_local, best_score


def power_sum_optimize(a_init, n, S, max_time, rng, p_init=8, p_max=64):

        it += 1
        p = min(p_max, p_init + it // 200)

def topk_subgrad_optimize(a_init, n, S, max_time, rng, k_init=5, k_max=256):
    """Nesterov with top-k subgradient (no smoothing)."""
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n

    t = 1.0

    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        k = min(k_max, k_init + it // 100)
        
        c = fft_conv(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        top_vals = c[top_idx]
        total = top_vals.sum()

            break

        w_full[top_idx] = weights
        
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = 2.0 * S / (n * gnorm * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)

            y = project_simplex(y + scale * rng.normal(0, 1, n), S)
            t = 1.0
    
    return best_local, best_score


def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
