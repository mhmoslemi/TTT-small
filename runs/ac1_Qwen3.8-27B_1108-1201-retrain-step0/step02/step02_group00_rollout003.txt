Key improvements over previous version:

4. Backtracking line search for step size adaptation
5. Periodic random perturbations to escape local minima
6. Multiple polishing phases with increasing active set size

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def subgradient_at_lag(a, k):
    """Compute db[k]/da where b = convolve(a, a). O(n) operation."""
    n = len(a)
    g = np.zeros(n)
    # Term 1: a[k-j] for j where 0 <= k-j < n => k-n+1 <= j <= k
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if lo1 < hi1:
        g[lo1:hi1] += a[k - np.arange(lo1, hi1)]
    # Term 2: a[j-k] for j where 0 <= j-k < n => k <= j <= k+n-1
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if lo2 < hi2:
        g[lo2:hi2] += a[np.arange(lo2, hi2) - k]
    return g


def subgradient_active_set(a, active_lags):
    """Average subgradient over a set of active lags."""
    n = len(a)
    g = np.zeros(n)

    g /= len(active_lags)
    return g
