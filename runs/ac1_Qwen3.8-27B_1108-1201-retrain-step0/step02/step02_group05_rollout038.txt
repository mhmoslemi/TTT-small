Optimized peak autocorrelation minimization for non-negative sequences.
Uses multi-stage approach: L-BFGS p-norm annealing, adaptive subgradient refinement,
peak-shaving local search, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def pnorm_obj_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p.
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def soft_max_obj_grad(theta, tau):

    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).

    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def subgradient_refine(a, time_limit, rng, n_top=5, initial_lr=0.005, decay=0.9995, min_lr=1e-7):
    """
    Adaptive subgradient method with momentum for the max-conv objective.
    Uses multiplicative updates in log-space.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    current_score = local_score(a)
    best = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    lr = initial_lr
    momentum = np.zeros(n)
    beta = 0.7
    frame = 100  # restart every frame iterations if no improvement
    frame_count = 0
    frame_best = current_score
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        c_max = np.max(c)
        
        # Find top positions
        if n_top >= len(c):
            top_positions = np.arange(len(c))
        else:
            top_positions = np.argpartit ion(c, -n_top)[-n_top:]
            # Actually use argpartition properly
            idx = np.argsort(c)[-n_top:]
            top_positions = idx
        
        # Compute average gradient of top peaks
        grad = np.zeros(n)
        for k in top_positions:
            # grad[i] += a[k-i] for valid i
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += a[k - i]
        grad /= len(top_positions)
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * grad
        momentum -= np.mean(momentum)  # center to preserve sum
        
        # Multiplicative update
        a_new = a * np.exp(-lr * momentum)
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / np.sum(a_new)
        
        new_score = local_score(a_new)
        
        if new_score < current_score:
            a = a_new
            current_score = new_score
            frame_count = 0
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            frame_count += 1
            if frame_count >= frame:
                # Restart from best with perturbation
                a = best * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-12)
                a = a / np.sum(a)
                current_score = local_score(a)
                momentum = np.zeros(n)
                frame_count = 0
                frame /= 1.2
        
        lr = max(lr * decay, min_lr)
    
    return best, best_score


def peak_shave_search(a_init, time_limit, rng):
    """
    Intensive local search focused on reducing the convolution peak.
    Uses targeted perturbations of elements contributing to the peak.
    """
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    current = current / np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    temperature = 0.001
    cooling = 0.9999
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        c_max = np.max(c)
        peak_idx = int(np.argmax(c))
        
        # Find contributing elements to the peak
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            contrib[i] = current[i] * current[peak_idx - i]
        
        r = rng.random()
        candidate = current.copy()
        
        if r < 0.35:
            # Reduce top contributors to peak, increase others
            top_k = max(3, n // 8)
            top_idx = np.argsort(contrib)[-top_k:]
            other = np.setdiff1d(np.arange(n), top_idx)
            
            # Decrease top contributors
            dec_factor = np.exp(rng.normal(-0.02, 0.01, size=len(top_idx)))
            candidate[top_idx] *= dec_factor
            
            # Increase some others
            if len(other) > 0:
                n_inc = min(8, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                inc_factor = np.exp(rng.normal(0.02, 0.01, size=n_inc))
                candidate[inc_idx] *= inc_factor
        elif r < 0.55:
            # Targeted: reduce elements with highest a[k* - i] * a[i] at peak
            # and also check nearby peaks
            nearby_peaks = np.where(c > 0.95 * c_max)[0]
            for pk in nearby_peaks:
                lo2 = max(0, pk - n + 1)
                hi2 = min(n - 1, pk)
                for i in range(lo2, hi2 + 1):
                    # Gradient contribution
                    candidate[i] *= (1 - 0.005 * current[pk - i] / c_max * n)
            
            # Normalize
            candidate = np.maximum(candidate, 1e-12)
            candidate = candidate / np.sum(candidate)
        elif r < 0.70:
            # Pair swap: move mass from peak-contributing to non-contributing
            top_k = max(2, n // 16)
            top_idx = np.argsort(contrib)[-top_idx] if False else np.argsort(contrib)[-top_k:]
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                src = rng.choice(top_idx)
                dst = rng.choice(other)
                amount = current[src] * 0.02 * rng.random()
                candidate[src] -= amount
                candidate[dst] += amount
                candidate = np.maximum(candidate, 1e-12)
                candidate = candidate / np.sum(candidate)
        elif r < 0.85:
            # Block perturbation around peak contributors
            width = max(1, int(rng.integers(2, max(3, n // 12))))
            # Pick a block near a top contributor
            top_k = max(3, n // 10)
            top_idx = np.argsort(contrib)[-top_k:]
            center = int(rng.choice(top_idx))
            start = max(0, center - width // 2)
            end = min(n, start + width)
            start = max(0, end - width)
            candidate[start:end] *= np.exp(rng.normal(-0.01, 0.015, size=end - start))
            # Compensate elsewhere
            other = np.setdiff1d(np.arange(n), np.arange(start, end))
            if len(other) > 0:
                comp_idx = rng.choice(other, size=min(10, len(other)), replace=False)
                candidate[comp_idx] *= np.exp(rng.normal(0.01, 0.015, size=len(comp_idx)))
        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.01, n))
        
        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score
        
        temperature = max(temperature * cooling, 1e-10)
    
    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Bell-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p, deadline):
    """Run L-BFGS with annealed p-norm."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))
    
    for p in p_schedule:
        if time.monotonic() > deadline - 10:
            break
        result = minimize(
            pnorm_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(p,),
            options={'maxiter': maxiter_per_p, 'ftol': 1e-15, 'gtol': 1e-13, 'maxfun': maxiter_per_p * 3}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)
        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        theta = best_theta
    
    return best_theta, best_score


def run_lbgfs_softmax(theta, tau_schedule, maxiter_per_tau, deadline):
    """Run L-BFGS with annealed soft-max temperature."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))
    
    for tau in tau_schedule:
        if time.monotonic() > deadline - 10:
            break
        result = minimize(
            soft_max_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(tau,),
            options={'maxiter': maxiter_per_tau, 'ftol': 1e-15, 'gtol': 1e-13, 'maxfun': maxiter_per_tau * 3}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)
        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        theta = best_theta
    
    return best_theta, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization for minimum peak autocorrelation.
    Phases: L-BFGS p-norm, soft-max refinement, subgradient, peak-shave, multi-length.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Load starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        if np.sum(start_seq) > 0.01:
            start_seq = start_seq / np.sum(start_seq)
            s = local_score(start_seq)
            print(f"Starting sequence: n={len(start_seq)}, score={s:.6f}")
            best_score = s
            best_seq = start_seq.copy()
    
    # ============ Phase 1: L-BFGS p-norm on starting length ============
    print("\n=== Phase 1: L-BFGS p-norm annealing ===")
    if best_seq is not None:
        n0 = len(best_seq)
        theta = np.log(best_seq + 1e-30)
        p_schedule = [8, 16, 32, 64, 128, 256]
        phase1_deadline = min(t_global + 200, deadline - 600)
        theta_best, s = run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p=300, deadline=phase1_deadline)
        if s < best_score:
            best_score = s
            best_seq = softmax(theta_best).copy()
            print(f"  After p-norm L-BFGS: {best_score:.6f} *** IMPROVED ***")
        else:
            print(f"  After p-norm L-BFGS: {s:.6f} (no improvement)")
    else:
        # Generate initial sequence
        n0 = 256
        a0 = make_chirp_init(n0, 0.3, rng)
        best_seq = a0
        best_score = local_score(a0)
        print(f"  Generated initial: n={n0}, score={best_score:.6f}")
    
    # ============ Phase 2: Soft-max L-BFGS refinement ============
    print(f"\n=== Phase 2: Soft-max L-BFGS (score={best_score:.6f}) ===")
    n = len(best_seq)
    theta = np.log(best_seq + 1e-30)
    tau_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    phase2_deadline = min(t_global + 500, deadline - 400)
    theta_best, s = run_lbgfs_softmax(theta, tau_schedule, maxiter_per_tau=400, deadline=phase2_deadline)
    if s < best_score:
        best_score = s
        best_seq = softmax(theta_best).copy()
        print(f"  After soft-max: {best_score:.6f} *** IMPROVED ***")
    else:
        print(f"  After soft-max: {s:.6f} (no improvement)")
    
    # ============ Phase 3: Subgradient refinement ============
    print(f"\n=== Phase 3: Subgradient refinement (score={best_score:.6f}) ===")
    sub_time = min(300.0, max(60.0, deadline - time.monotonic() - 300))
    sub_best, sub_score = subgradient_refine(best_seq, sub_time, rng, n_top=5, initial_lr=0.003, decay=0.9997)
    if sub_score < best_score:
        best_score = sub_score
        best_seq = sub_best.copy()
        print(f"  After subgradient: {best_score:.6f} *** IMPROVED ***")
    else:
        print(f"  After subgradient: {sub_score:.6f} (no improvement)")
    
    # ============ Phase 4: Peak-shave local search ============
    print(f"\n=== Phase 4: Peak-shave local search (score={best_score:.6f}) ===")
    shave_time = min(400.0, max(60.0, deadline - time.monotonic() - 100))
    shave_best, shave_score = peak_shave_search(best_seq, shave_time, rng)
    if shave_score < best_score:
        best_score = shave_score
        best_seq = shave_best.copy()
        print(f"  After peak-shave: {best_score:.6f} *** IMPROVED ***")
    else:
        print(f"  After peak-shave: {shave_score:.6f} (no improvement)")
    
    # ============ Phase 5: Try other lengths ============
    print(f"\n=== Phase 5: Multi-length exploration (score={best_score:.6f}) ===")
    if time.monotonic() < deadline - 120:
        # Try lengths near the best
        n_best = len(best_seq)
        candidate_lengths = sorted(set([
            n_best - 16, n_best - 8, n_best + 8, n_best + 16,
            n_best - 32, n_best + 32
        ] & set(range(64, 513))))
        
        for n_try in candidate_lengths:
            if time.monotonic() > deadline - 60:
                break
            
            # Interpolate from best
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.maximum(a_interp, 1e-6)
            a_interp = a_interp / np.sum(a_interp)
            
            theta_t = np.log(a_interp + 1e-30)
            p_sched = [32, 64, 128]
            theta_t_best, s_t = run_lbgfs_pnorm(theta_t, p_sched, maxiter_per_p=150, deadline=deadline - 30)
            
            if s_t < best_score:
                best_score = s_t
                best_seq = softmax(theta_t_best).copy()
                print(f"  n={n_try}: {best_score:.6f} *** NEW BEST ***")
            else:
                print(f"  n={n_try}: {s_t:.6f}")
    
    # ============ Phase 6: Final intensive refinement ============
    print(f"\n=== Phase 6: Final refinement (score={best_score:.6f}) ===")
    if time.monotonic() < deadline - 30:
        n_final = len(best_seq)
        theta = np.log(best_seq + 1e-30)
        # Very tight soft-max
        tau_final = [0.0002, 0.0001, 0.00005]
        theta_f, s_f = run_lbgfs_softmax(theta, tau_final, maxiter_per_tau=500, deadline=deadline - 15)
        if s_f < best_score:
            best_score = s_f
            best_seq = softmax(theta_f).copy()
            print(f"  After final soft-max: {best_score:.6f} *** IMPROVED ***")
        
        # Quick peak shave
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic() - 5
            if remaining > 20:
                ps_best, ps_score = peak_shave_search(best_seq, remaining, rng)
                if ps_score < best_score:
                    best_score = ps_score
                    best_seq = ps_best.copy()
                    print(f"  After final shave: {best_score:.6f} *** IMPROVED ***")
    
    # Final verification
    print(f"\n=== FINAL: n={len(best_seq)}, local_score={best_score:.6f} ===")
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(final_list)
    print(f"=== evaluate_sequence score: {final_score:.6f} ===")
    return final_list
