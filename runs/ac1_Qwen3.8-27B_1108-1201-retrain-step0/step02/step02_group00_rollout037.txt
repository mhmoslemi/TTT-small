2. Compute their gradients
3. Find weights w that make the weighted gradient as uniform as possible (close to a constant)
4. Take a step in the negative direction of the non-uniform part

def find_optimal_subgradient(a, c, top_k):
    """
    Find the subgradient (convex combination of active gradients) that is
    closest to being constant (i.e., closest to the KKT condition).
    """
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    
    # Build gradient matrix: each row is the gradient of c_k for k in top_indices
    # grad_k[j] = 2 * a[k-j] (with appropriate bounds)
    G = np.zeros((top_k, n))
    for i, k in enumerate(top_indices):
        k = int(k)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                G[i, j] = 2.0 * a[idx]
    
    # Find weights w >= 0, sum(w) = 1, such that G^T w is as constant as possible
    # i.e., minimize ||G^T w - mean(G^T w) * 1||^2
    # This is equivalent to: minimize ||G^T w||^2 - (1^T G^T w)^2 / n
    # = w^T (G G^T - (1^T G)(1^T G)^T / n) w
    # Subject to w >= 0, 1^T w = 1
    
    # This is a QP! We can solve it with CVXPY.
    # But for speed, let's use a simpler approach: iterative proportional fitting.
    
    # Start with uniform weights
    w = np.ones(top_k) / top_k
    
    # Iterate: adjust w to make G^T w more uniform
    for _ in range(50):
        grad = G.T @ w  # current gradient vector (length n)
        mean_grad = grad.mean()
        # We want to reduce the variance of grad
        # The "descent direction" for w is: for each i, if (G[i] - mean_component) is large,
        # reduce w[i]. 
        # Simple approach: w_i proportional to exp(-eta * ||G[i] - projection||^2)
        # Or: w_i proportional to 1 / (std of G[i])
        
        # Actually, let's use a simpler heuristic:
        # The contribution of gradient i to the "non-uniformity" is related to how
        # much G[i] deviates from the mean of G (across rows).
        row_sums = G.sum(axis=1)  # sum of each gradient row
        # Gradients with larger row sums contribute more to the "peak"
        # We want to downweight them
        dev = np.abs(row_sums - row_sums.mean())
        w = np.exp(-0.5 * dev / max(dev.max(), 1e-10))
        w = w / w.sum()
    
    return w, G.T @ w

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- Normalize to sum = sqrt(2n), reducing objective to max of self-convolution.
- Phase 1: FISTA with log-sum-exp smoothing (coarse to fine beta).
- Phase 2: Exact subgradient descent with backtracking line search.
- Phase 3: Coordinate descent polish.
- Phase 4: Multi-start with perturbations.
- Try multiple sequence lengths for diversity.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_corr(a, w, n):
    """
    Compute cross-correlation: result[m] = sum_k w[k] * a[k-m]
    (This gives the gradient of the weighted sum of convolution values.)
    w has length 2n-1, a has length n.
    """
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    result = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * result


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=200.0):
    """FISTA with adaptive log-sum-exp and simplex projection."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.003 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Log-sum-exp gradient
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            grad = np.zeros(n)
        else:
            w = exp_bc / total
            grad = fft_cross_corr(y, w, n)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta * 2.0)
        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        step = 1.0 / (beta * 2.0 * (1.0 + 0.001 * it))
        step = max(step, 1e-7)

        if it % 5000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def run_exact_subgrad(a_init, n, S, max_time, rng, top_k=20):
    """
    Exact subgradient descent with backtracking line search.
    Uses the top-k active constraints for the subgradient.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (n * 0.1)  # initial step size

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Compute subgradient using top-k lags
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        grad = fft_cross_corr(a, w_full, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Gradient descent step with backtracking
        g_normed = grad / gnorm
        alpha = step
        for _ in range(20):
            a_trial = project_simplex(a - alpha * gnorm * g_normed, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() <= cur + 1e-12:
                break
            alpha *= 0.5
        else:
            alpha = step * 0.01  # fallback

        a = a_trial
        step = min(alpha * 1.5, S / (n * 0.01))  # cautiously increase step

        if it % 3000 == 0 and rng.random() < 0.05:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

    return best_local, best_score


def coordinate_descent_polish(a_init, n, S, max_time, n_passes=3):
    """
    Coordinate descent: for each coordinate, try a few values and keep the best.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    c = fft_convolve_self(a)
    if c.max() < best_score:
        best_score = c.max()
        best_local = a.copy()

    per_pass = max_time / n_passes
    rng = np.random.default_rng(12345)

    for pass_idx in range(n_passes):
        c_start = time.monotonic()
        order = rng.permutation(n)

        for m in order:
            if time.monotonic() - start > max_time:
                return best_local, best_score

            old_val = a[m]
            if old_val < 1e-15:
                continue

            # Try several multipliers of the current value
            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]
            best_trial_score = c.max()
            best_trial_val = old_val

            remainder = S - old_val
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 1000:
                    continue
                new_remainder = S - new_val
                if new_remainder < 0:
                    continue
                # Scale other coordinates
                scale = new_remainder / max(remainder, 1e-30)
                a_trial = a.copy()
                a_trial[m] = new_val
                mask = np.arange(n) != m
                a_trial[mask] = a[mask] * scale
                if (a_trial < 0).any():
                    a_trial = np.maximum(a_trial, 0)
                    # Renormalize
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * S / s_t

                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < best_trial_score:
                    best_trial_score = c_trial.max()
                    best_trial_val = new_val
                    # Update c for the next trial (save computation)
                    # Actually, let's just keep computing fresh for correctness

            a[m] = best_trial_val
            # Renormalize
            s_now = a.sum()
            if s_now > 1e-15:
                a = a * S / s_now
            a = np.maximum(a, 0)

        # End of one pass
        c = fft_convolve_self(a)
        if c.max() < best_score:
            best_score = c.max()
            best_local = a.copy()
            print("    CD pass %d: %.6f" % (pass_idx + 1, best_score))

    return best_local, best_score

Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- Normalize to sum = sqrt(2n), reducing objective to max of self-convolution.
- Phase 1: FISTA with log-sum-exp smoothing (beta 5→80).
- Phase 2: FISTA with high beta (80→600) for refinement.
- Phase 3: Exact top-k subgradient with backtracking line search.
- Phase 4: Coordinate descent polish (randomized order, multiple multipliers).
- Phase 5: Multi-start with perturbations, repeating phases 3-4.
- Phase 6: Final coordinate descent + verification.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient(a, c, beta=None, top_k=None):
    """
    Compute gradient of smoothed or top-k objective w.r.t. a.
    Uses FFT cross-correlation.
    """
    n = len(a)
    if beta is not None:
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            return np.zeros(n)
        w = exp_bc / total
    else:
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            return np.zeros(n)
        weights = top_vals / total
        w = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w[int(k)] = weights[i]

    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=200.0):
    """FISTA with adaptive log-sum-exp and simplex projection."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    step_init = 1.0 / (beta_init * 2.0)
    step = step_init
    it = 0
    last_best_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init + (beta_max - beta_init) * min(1.0, it / max(1, int(max_time * 80.0))))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            last_best_it = it

        grad = compute_gradient(y, c, beta=beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta * 2.0)
        x_new = project_simplex(y - step * grad, S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
                last_best_it = it

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic perturbation to escape plateaus
        if it - last_best_it > 5000 and rng.random() < 0.2:
            scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, scale, n), S)
            a = y.copy()
            t = 1.0

    return best_local, best_score


def run_exact_subgrad(a_init, n, S, max_time, rng, top_k=30):
    """
    Exact subgradient descent with backtracking line search.
    Directly minimizes the max of self-convolution.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    alpha = S / (2.0 * np.sqrt(n))
    last_best = 0

    # Precompute FFT size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            last_best = it

        # Top-k subgradient
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        gsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, gsize)
        A = fft(a, gsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        a_trial = a.copy()
        alpha_try = alpha

    Coordinate descent polish: for each coordinate, try several multipliers.

    pass_count = 0

    while time.monotonic() - start < max_time:
        pass_count += 1
        order = rng.permutation(n)
        pass_start = time.monotonic()

        for m in order:
            if time.monotonic() - start > max_time:
                break

            old_val = a[m]
            if old_val < 1e-18:
                # Try a small positive value
                new_val = S / n * 0.1
                a[m] = new_val
                s_now = a.sum()
                if s_now > 1e-15:
                    a = a * S / s_now
                continue

            best_trial = cur_max
            best_new_val = old_val

            remainder = S - old_val
            for mult in multiplier_list:
                new_val = old_val * mult
                if new_val < 0 or new_val > 1000:
                    continue
                new_rem = S - new_val
                if new_rem < 0:
                    continue
                scale = new_rem / max(remainder, 1e-30)
                if scale < 0.1 or scale > 10:
                    continue

                a[m] = new_val
                # Scale others
                mask = np.arange(n) != m
                a[mask] = a[mask] * scale
                a = np.maximum(a, 0.0)
                s_now = a.sum()
                if s_now < 1e-15:
                    a[m] = old_val
                    a = a * S / max(old_val, 1e-15)
                    break
                a = a * S / s_now

                c_trial = fft_convolve_self(a)
                trial_max = c_trial.max()
                if trial_max < best_trial:
                    best_trial = trial_max
                    best_new_val = new_val
                    c = c_trial.copy()

            # Restore if no improvement
            if best_new_val != old_val:
                # Already at best position
                cur_max = best_trial
            else:
                # Restore original
                a[m] = old_val
                s_now = a.sum()
                if s_now > 1e-15:
                    a = a * S / s_now

        # End of pass
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        elapsed = time.monotonic() - pass_start
        if elapsed < 0.5:
            break  # Pass was too fast, no more time anyway

    return best_local, best_score

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):

    else:

    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)
