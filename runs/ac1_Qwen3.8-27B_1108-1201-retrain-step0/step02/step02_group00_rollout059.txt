```python
def fourier_sequence(c, n, m):
    """Generate sequence from Fourier coefficients."""
    t = np.arange(n, dtype=np.float64) / n
    a = c[0] * np.ones(n)
    for j in range(1, m+1):
        a += c[2*j-1] * np.cos(2*np.pi*j*t)
        a += c[2*j] * np.sin(2*np.pi*j*t)
    return np.maximum(a, 0.0)

    if S < 1e-10:
        return 1e10
    c_conv = fft_convolve_self(a)
    c_max = c_conv.max()
    lse = c_max + np.log(np.sum(np.exp(beta * (c_conv - c_max)))) / beta
    return 2*n*lse / (S**2)

def fourier_grad(c, n, m, beta):
    """Gradient of the log-sum-exp objective w.r.t. Fourier coefficients."""
    a = fourier_sequence(c, n, m)
    S = a.sum()
    if S < 1e-10:
        return np.zeros_like(c)
    c_conv = fft_convolve_self(a)
    c_max = c_conv.max()
    exp_bc = np.exp(beta * (c_conv - c_max))
    w = exp_bc / exp_bc.sum()
    
    # Gradient of lse w.r.t. a
    # lse = (1/beta) * log(sum exp(beta * conv_k))
    # d(lse)/da[j] = (1/beta) * sum_k w[k] * d(conv_k)/da[j]
    # d(conv_k)/da[j] = 2*a[k-j] (if valid)
    # So d(lse)/da[j] = (2/beta) * sum_k w[k] * a[k-j]
    
    # This is (2/beta) * (w * a)[j] where * is cross-correlation
    # Actually: sum_k w[k] * a[k-j] for each j
    # = cross-correlation of w and a, evaluated at j
    
    fsize = 1 << (2*n-1+n-2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    cross_corr = np.real(ifft(W * np.conj(A)))[:n]
    # cross_corr[j] = sum_k w[k] * a[k-j] (approximately)
    
    da_le = 2.0 / beta * cross_corr
    
    # Gradient of objective w.r.t. a
    # obj = 2n * lse / S^2
    # d(obj)/da = 2n * (da_le / S^2 - 2 * lse * da_S / S^3)
    # where da_S = 1 (derivative of sum a w.r.t. each a_j is 1)
    da_S = np.ones(n)
    da_obj = 2*n * (da_le / S**2 - 2*lse / S**3 * da_S)
    # Simplify: = 2n/S^2 * (da_le - 2*lse/S)
    da_obj = 2*n/S**2 * (da_le - 2*lse/S)
    
    # Gradient w.r.t. c: chain rule
    # a = max(0, c_0 + sum c_j * basis_j)
    # da/dc_0 = 1 (where a > 0)
    # da/dc_j = basis_j (where a > 0)
    
    # For simplicity, ignore the max(0,.) clipping (assume a > 0 everywhere)
    # and just compute the inner product with the basis functions.
    
    t = np.arange(n, dtype=np.float64) / n
    mask = a > 0  # where the max is active
    grad_c = np.zeros(2*m+1)
    grad_c[0] = np.sum(da_obj[mask])
    for j in range(1, m+1):
        grad_c[2*j-1] = np.sum(da_obj[mask] * np.cos(2*np.pi*j*t[mask]))
        grad_c[2*j] = np.sum(da_obj[mask] * np.sin(2*np.pi*j*t[mask]))
    
    return grad_c
```
