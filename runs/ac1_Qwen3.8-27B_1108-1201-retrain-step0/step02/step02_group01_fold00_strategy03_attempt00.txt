def fft_convolve(seq):
    # Convolve using FFT for efficiency
    n = len(seq)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.fft(seq, fsize)
    B = np.conj(A)
    C = np.real(np.fft.ifft(A * B))
    return C[:size]

    return [seq[i] * seq[j] for i in range(n) for j in range(n)]
