**Key insight:** The sum of all elements of `conv(a,a)` equals `(sum(a))^2`, and there are `2n-1` elements. So `max_b >= (sum(a))^2/(2n-1)`, giving a lower bound of `2n/(2n-1)` which approaches 1 for large n. This means longer sequences can achieve much lower scores than the uniform sequence (which gives 2.0).

The goal is to make `conv(a,a)` as flat as possible. By the convolution theorem, this is equivalent to making `|FFT(a)|^2` as flat as possible. I'll use:
1. **Spectral method** (iterative projection): alternate between flattening the power spectrum and projecting to non-negativity
2. **Subgradient descent** for refinement
3. Multiple sequence lengths to exploit the `2n/(2n-1)` lower bound

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses spectral projection method (flat power spectrum) + subgradient descent.
Key: sum of conv elements = (sum a)^2, so min possible max is (sum a)^2/(2n-1),
giving lower bound 2n/(2n-1) -> 1 for large n. Longer sequences are better.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Fast score computation using FFT."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s ** 2)


def spectrally_project(a, n_iters=500, rng=None):
    """
    Iterative projection: alternate between flattening |FFT(a)|^2
    and projecting to non-negative. Converges to flat power spectrum.
    """
    n = len(a)
    a_work = a.copy().astype(np.float64)
    if np.sum(a_work) < 1e-10:
        a_work = np.ones(n)
    
    best_a = a_work.copy()
    best_score = fast_score(a_work)
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for it in range(n_iters):
        # Forward FFT
        A = np.fft.rfft(a_work, size)
        
        # Compute current power spectrum
        power = np.abs(A) ** 2
        
        # Flattening: set target magnitude to geometric mean
        # (we want flat power spectrum, so flat magnitude)
        target_mag = np.sqrt(np.mean(power))
        
        # Create new spectrum with flat magnitude, preserved phase
        phase = A / (np.abs(A) + 1e-30)
        A_new = target_mag * phase
        
        # Inverse FFT
        a_work = np.fft.irfft(A_new, size)[:n]
        
        # Project to non-negative
        a_work = np.maximum(a_work, 0.0)
        
        # Normalize (score is homogeneous, but keep scale reasonable)
        s = np.sum(a_work)
        if s < 1e-10:
            a_work = np.ones(n)
            continue
        a_work = a_work / s  # normalize to sum=1
        
        # Track best
        sc = fast_score(a_work)
        if sc < best_score:
            best_score = sc
            best_a = a_work.copy()
    
    return best_a, best_score


def subgradient_descent(a_init, n_iters=5000, lr0=0.01, rng=None):
    """
    Subgradient descent on the normalized simplex to minimize max(conv(a,a)).
    Works with sum(a)=1, minimizing max_b directly (score = 2n * max_b).
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_maxb = np.max(fft_convolve(a, a))
    best_score = 2.0 * n * best_maxb
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for it in range(n_iters):
        # Compute convolution via FFT
        A = np.fft.rfft(a, size)
        c = np.fft.irfft(A * A, size)[:2 * n - 1]
        
        max_b = np.max(c)
        
        # Find all indices where conv achieves max
        K_star = np.where(np.isclose(c, max_b, atol=1e-12))[0]
        
        # Compute subgradient: sg[m] = sum_{k in K*} 2*a[k-m] (with a[j]=0 outside [0,n-1])
        sg = np.zeros(n)
        for k in K_star:
            # a[k-m] for m in [0, n-1], valid when 0 <= k-m < n, i.e., k-n+1 <= m <= k
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                sg[lo:hi+1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        
        # Learning rate: lr_t = lr0 / (it + 1)^0.5
        lr = lr0 / np.sqrt(it + 1)
        
        # Update
        a_new = a - lr * sg
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = _project_simplex(a_new)
        
        # Check score
        A_new = np.fft.rfft(a_new, size)
        c_new = np.fft.irfft(A_new * A_new, size)[:2 * n - 1]
        max_b_new = np.max(c_new)
        
        if max_b_new < best_maxb:
            best_maxb = max_b_new
            best_score = 2.0 * n * best_maxb
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score


def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Binary search for threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def combined_optimize(n, rng, time_budget=100.0):
    """Run spectral method + subgradient for a given n."""
    t0 = time.monotonic()
    
    # Initialize with random non-negative
    a_init = rng.random(n) + 0.1

    n_spectral = min(1000, max(100, int(500 * time_left / time_budget)))
    
    a_spec, score_spec = spectrally_project(a_init, n_iters=n_spectral, rng=rng)
    print(f"    n={n}, after spectral: score={score_spec:.6f}", flush=True)
    
    # Phase 2: Subgradient refinement
    time_left = time_budget - (time.monotonic() - t0)
    if time_left > 5:
        # Estimate iterations based on n
        # Each iteration is O(n log n) for FFT + O(n * |K*|) for subgradient
        start = time.monotonic()
        # Test how fast one iteration is
        a_test = a_spec.copy()
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        A = np.fft.rfft(a_test, size)
        c = np.fft.irfft(A * A, size)[:2 * n - 1]
        _ = np.max(c)
        _ = np.where(np.isclose(c, np.max(c), atol=1e-12))[0]
        iter_time = time.monotonic() - start
        n_sub = min(20000, max(500, int(time_left / max(iter_time * 2, 1e-6))))
        
        a_sub, score_sub = subgradient_descent(a_spec, n_iters=n_sub, lr0=0.05, rng=rng)
        print(f"    n={n}, after subgradient ({n_sub} iters): score={score_sub:.6f}", flush=True)
        
        if score_sub < score_spec:
            return a_sub, score_sub
    
    return a_spec, score_spec


def random_restart_optimize(n, rng, time_budget=100.0, n_restarts=3):
    """Try multiple random initializations."""
    best_a = None
    best_score = np.inf
    t0 = time.monotonic()
    
    for r in range(n_restarts):
        time_left = time_budget - (time.monotonic() - t0)
        if time_left < 10:
            break
        per_restart = time_left / (n_restarts - r)
        
        # Different random initializations
        init_type = r % 4
        if init_type == 0:
            a_init = rng.random(n) + 0.01
        elif init_type == 1:
            a_init = np.exp(-np.linspace(0, 2, n) * rng.uniform(0.5, 2.0))
        elif init_type == 2:
            a_init = np.abs(rng.standard_normal(n)) + 0.01
        else:
            a_init = np.ones(n) + 0.5 * rng.standard_normal(n)
            a_init = np.maximum(a_init, 0.01)
        
        a_init = a_init / np.sum(a_init)
        
        a_r, s_r = combined_optimize(n, rng, time_budget=per_restart)
        if s_r < best_score:
            best_score = s_r
            best_a = a_r.copy()
    
    return best_a, best_score
```
