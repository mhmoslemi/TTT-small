import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_subgradient(a, c, top_k=None):
    """Compute the projected subgradient of max(c) w.r.t. a."""
    n = len(a)
    M = c.max()
    
    if top_k is None:
        # Use all positions where c is within 1% of max
        threshold = 0.99 * M
        peaks = np.where(c >= threshold)[0]
    else:
        peaks = np.argsort(c)[::-1][:top_k]
    
    # Weights for peaks
    w = np.zeros(2*n - 1)
    peak_vals = c[peaks]
    # Weight by excess over average
    avg = (a.sum()**2) / (2*n - 1)
    excess = np.maximum(peak_vals - avg, 0)
    total_excess = excess.sum()
    if total_excess < 1e-15:
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        weights = excess / total_excess
    
    # Compute gradient: g[i] = 2 * sum_k w[k] * a[k-i] (if valid)
    # This is 2 * (w_conv * a_rev)[offset]
    w_conv = np.zeros(2*n - 1)
    w_conv[peaks] = weights
    
    a_rev = a[::-1]
    # (w_conv * a_rev) has length (2n-1) + n - 1 = 3n-2
    out_len = len(w_conv) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w_conv, fsize)
    AR = fft(a_rev, fsize)
    conv = np.real(ifft(W * AR))[:out_len]
    
    # g[i] = 2 * conv[n-1 + i] for i = 0..n-1
    g = 2.0 * conv[n-1 : n-1+n]
    
    # Project onto tangent space: subtract mean so sum(g) = 0
    g = g - g.mean()
    
    return g, peaks

Minimize 2*n*max(conv(a,a))/sum(a)^2 using:

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_projected_subgradient(a, c, intensity=0.05):
    """
    Compute the projected subgradient of max(c) with respect to a.
    Uses adaptive peak weighting based on excess over average.
    Returns the gradient (sum = 0) and the peak information.
    """
    n = len(a)
    M = c.max()
    avg = (a.sum() ** 2) / (2 * n - 1)
    
    # Select peaks: those with excess > 0
    excess = np.maximum(c - avg, 0)
    # Focus on peaks that are significantly above average
    threshold = avg + intensity * (M - avg)
    peak_mask = c >= threshold
    peaks = np.where(peak_mask)[0]
    
    if len(peaks) == 0:
        peaks = np.array([np.argmax(c)])
    
    # Weight by excess
    peak_excess = c[peaks] - avg
    total_excess = peak_excess.sum()
    if total_excess < 1e-15:
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        weights = peak_excess / total_excess
    
    # Build the weight vector
    w = np.zeros(2 * n - 1)
    w[peaks] = weights
    
    # Compute gradient via FFT
    a_rev = a[::-1]
    out_len = (2 * n - 1) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    AR = fft(a_rev, fsize)
    conv = np.real(ifft(W * AR))[:out_len]
    
    # g[i] = 2 * conv[n-1+i]
    g = 2.0 * conv[n - 1: n - 1 + n]
    
    # Project: make sum(g) = 0
    g = g - g.mean()
    
    return g, peaks, M


def polyak_step_size(max_c, S, n, grad_norm, damping=1.0):
    """Compute Polyak step size."""
    lower_bound = S ** 2 / (2 * n - 1)
    gap = max_c - lower_bound
    if grad_norm < 1e-15:
        return 0.0
    return damping * gap / (grad_norm ** 2)


def project_simplex(a, S):
    """Project onto {a >= 0, sum(a) = S} using clip-and-renormalize."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

5. If peak_contribution[i] < valley_contribution[i], increase a[i]

The key insight is that the "equation" c[k] = target for all k is a system of 8191 quadratic equations in 4096 unknowns. The solution (if it exists) would give a perfectly flat convolution. In general, no exact solution exists, but we can find the least-squares solution or the minimum-max solution.

**New idea: Use a "water-filling" approach**

Think of it this way: we have a fixed amount of "mass" S to distribute among n positions. The convolution at each position k is a "bowl" that gets filled by the mass. We want to minimize the maximum fill level.

The water-filling algorithm:
1. Start with a uniform distribution
2. Find the maximum peak
3. "Drain" mass from the contributors to that peak
4. "Pour" the drained mass into positions that don't fill any peak
5. Repeat until no more improvement is possible

This is essentially the subgradient method, but with a more careful "drain and pour" strategy.

**The key to improvement**: The careful "pouring" step. When we drain mass from position i (to reduce peak k), we should pour it into positions j that:
- Don't contribute to any current peak
- Minimize the creation of new peaks

This is a combinatorial optimization problem, but we can approximate it with a greedy strategy.

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Projected subgradient with Polyak step size and Nesterov momentum on the TRUE max
2. Adaptive exponential peak weighting (sharpens over time to focus on worst peaks)
3. Vectorized 1D coordinate optimization (ternary search with FFT-freeUpdates)
4. LP-based directional improvements as periodic restarts
5. Multi-start exploration with structured initializations
Key difference from prior: direct subgradient on true max (no soft-max bias),
faster coordinate optimization via incremental convolution updates, and
greedy "drain-and-pour" peak reduction strategy.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, c, sharpness=5.0):
    """
    Compute projected subgradient of max(c) w.r.t. a.
    Uses exponential weighting on peak excess for focus.
    Returns gradient (sum zero) and max value.
    """
    n = len(a)
    M = c.max()
    S = a.sum()
    avg = S * S / (2 * n - 1)

    # Excess above average
    excess = np.maximum(c - avg, 0.0)
    max_excess = excess.max()
    if max_excess < 1e-15:
        # All flat, just use the argmax
        peaks = np.array([np.argmax(c)])
        weights = np.array([1.0])
    else:
        # Exponential weighting: focus on highest peaks
        w_exp = np.exp(sharpness * excess / max_excess)
        # Zero out negligible contributions
        w_exp[excess < 0.01 * max_excess] = 0.0
        total_w = w_exp.sum()
        if total_w < 1e-15:
            peaks = np.array([np.argmax(c)])
            weights = np.array([1.0])
            w_exp = np.zeros(2 * n - 1)
            w_exp[peaks[0]] = 1.0
        else:
            w_exp = w_exp / total_w
            peaks = np.where(w_exp > 0)[0]
            weights = w_exp[peaks]

    # Build weight vector for FFT
    w = np.zeros(2 * n - 1)
    if 'peaks' in dir() and len(peaks) > 0:
        w[peaks] = weights

    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (valid indices)
    a_rev = a[::-1]
    out_len = (2 * n - 1) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    Wf = fft(w, fsize)
    Af = fft(a_rev, fsize)
    conv = np.real(ifft(Wf * Af))[:out_len]

    g = 2.0 * conv[n - 1: n - 1 + n]
    g = g - g.mean()  # Project to tangent space

    return g, M


def optimize_coordinate(i, a, c, S, n):
    """
    Optimize a single coordinate a[i] using vectorized ternary search.
    Uses incremental update: c_new = c + delta * influence + delta^2 * correction
    """
    original = a[i]
    if original < 1e-14:
        return original, c.max()

    # Precompute influence vector
    # c[k] changes by 2*(new-old)*a[k-i] for k in [i, i+n-1] (k != 2i)
    # c[2i] changes by (new^2 - old^2) = (new-old)^2 + 2*(new-old)*old
    lo_k = i
    hi_k = min(i + n - 1, 2 * n - 2)

    # a[k-i] for k in [lo_k, hi_k] is a[0, 1, ..., hi_k-i]
    a_slice = a[0: hi_k - lo_k + 1]
    influence = np.zeros(2 * n - 1)
    influence[lo_k: hi_k + 1] = 2.0 * a_slice

    # At k = 2i: the linear part gives 2*a[i] which equals influence[2i],
    # but the true change is (new-old)^2 + 2*(new-old)*a[i]
    # So the "extra" quadratic term at 2i is (new-old)^2
    quad_idx = 2 * i
    has_quad = (0 <= quad_idx <= 2 * n - 2)

    best_obj = c.max()
    best_val = original

    # Ternary search
    lo, hi = 0.0, original * 2.0
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0

        d1 = m1 - original
        c_m1 = c + influence * d1
        if has_quad:
            c_m1[quad_idx] += d1 * d1
        obj1 = c_m1.max()

        d2 = m2 - original
        c_m2 = c + influence * d2
        if has_quad:
            c_m2[quad_idx] += d2 * d2
        obj2 = c_m2.max()

        if obj1 < obj2:
            hi = m2
        else:
            lo = m1

        if hi - lo < 1e-10 * max(original, 1e-8):
            break

    opt = (lo + hi) / 2.0
    d = opt - original
    c_opt = c + influence * d
    if has_quad:
        c_opt[quad_idx] += d * d
    opt_obj = c_opt.max()

    if opt_obj < best_obj - 1e-14:
        best_val = opt
        best_obj = opt_obj

    return best_val, best_obj


def solve_lp_direction_simple(a, M, n, num_constraints=40, time_limit=5.0):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0.
    Returns optimal g normalized to sum 1, or None.
    """
    from scipy.optimize import linprog

    c = fft_convolve_self(a)
    active_lags = np.argsort(c)[::-1][:num_constraints]

    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[idx, lo:hi + 1] = f[k - np.arange(lo, hi + 1)] if False else a[np.clip(k - np.arange(lo, hi + 1), 0, n - 1)]
            # More carefully: A[idx, j] = a[k-j] for j in [lo, hi]
            A[idx, lo:hi + 1] = a[k - np.arange(lo, hi + 1)]

    c_obj = -np.ones(n)
    b = np.full(len(active_lags), M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                         method='highs', options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def subgradient_main(a_init, S, n, time_budget, verbose=True):
    """
    Main subgradient optimization with:
    - Adaptive sharpness (increases over time)
    - Polyak step size with damping
    - Nesterov momentum
    - Periodic coordinate optimization on critical coordinates
    - Periodic LP direction
    """
    start = time.monotonic()
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    v = np.zeros(n)  # momentum velocity
    beta = 0.85  # momentum
    t = 0
    no_improve = 0
    cooldown = 0

    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget

        c = fft_convolve_self(a)
        cur_obj = c.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            cooldown = max(cooldown, 5)
        else:
            no_improve += 1

        if no_improve > 3000:
            break

        # Adaptive sharpness: start broad, focus narrow
        sharpness = 3.0 + 12.0 * progress  # from 3 to 15

        # Compute subgradient
        g, M = compute_subgradient(a, c, sharpness=sharpness)
        gnorm = np.linalg.norm(g)

        if gnorm < 1e-14:
            break

        # Polyak step size with damping
        lower_bound = S * S / (2 * n - 1)
        gap = M - lower_bound
        if gap > 1e-12:
            eta = 0.3 * gap / (gnorm * gnorm)
        else:
            eta = 0.0

        # Cap step size for stability
        max_eta = 2.0 * S / n / (gnorm + 1e-15)
        eta = min(eta, max_eta)

        # Decay momentum over time
        beta_t = beta * (1.0 - 0.3 * progress)

        # Nesterov momentum update
        v = beta_t * v + g
        step = -eta * v
        step = step - step.mean()  # maintain zero sum

        a = a + step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        # Periodic coordinate optimization
        if t % 30 == 0 and cooldown <= 0 and time.monotonic() - start < time_budget - 10.0:
            c = fft_convolve_self(a)
            coord_time_budget = min(5.0, time_budget - (time.monotonic() - start) - 10.0)
            c_obj = c.max()

            # Find top peaks and their critical coordinates
            top_peak_idx = np.argsort(c)[::-1][:15]

            critical_coords = []
            seen = set()
            for kp in top_peak_idx[:8]:
                # For peak at position kp, contribution of a[i] is 2*a[i]*a[kp-i]
                lo = max(0, kp - n + 1)
                hi = min(n - 1, kp)
                if lo > hi:
                    continue
                i_range = np.arange(lo, hi + 1)
                contribs = 2.0 * a[i_range] * a[kp - i_range]
                # Get top contributors
                top_local = i_range[np.argsort(contribs)[::-1][:5]]
                for ii in top_local:
                    if ii not in seen:
                        seen.add(ii)
                        critical_coords.append(ii)

            # Optimize critical coordinates
            for ci in critical_coords[:30]:
                if time.monotonic() - start > time_budget - 10.0:
                    break
                opt_val, opt_obj = optimize_coordinate(ci, a, c, S, n)
                if opt_val < a[ci] - 1e-12:
                    a[ci] = opt_val
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a * (S / s)
                    c = fft_convolve_self(a)
                    if c.max() < best_obj - 1e-10:
                        best_obj = c.max()
                        best_a = a.copy()
                        no_improve = 0
                        cooldown = 3

            cooldown = 2

        # Periodic LP direction
        if t % 500 == 0 and time.monotonic() - start < time_budget - 30.0:
            c = fft_convolve_self(a)
            M_cur = c.max()
            g_lp = solve_lp_direction_simple(a, M_cur, n, num_constraints=30,
                                             time_limit=3.0)
            if g_lp is not None:
                # Try blending
                for blend in [0.2, 0.4, 0.6, 0.8]:
                    a_try = (1.0 - blend) * a + blend * g_lp * S
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try > 1e-15:
                        a_try = a_try * (S / s_try)
                        c_try = fft_convolve_self(a_try)
                        if c_try.max() < best_obj - 1e-10:
                            best_obj = c_try.max()
                            best_a = a_try.copy()
                            a = a_try.copy()
                            print("  LP blend=%.1f improved to %.6f" % (blend, best_obj))
                            break

    if verbose:
        print("  Subgradient done: t=%d, best_obj=%.6f" % (t, best_obj))

    return best_a, best_obj


def cyclic_coordinate_descent(a, S, n, time_budget, max_sweeps=100):
    """
    Cyclic coordinate descent: go through all coordinates in order,
    optimizing each one. Multiple sweeps until time runs out.
    """
    start = time.monotonic()
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        # Use a pseudo-random order for diversity
        rng = np.random.default_rng(42 + sweep)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget - 2.0:
                break

            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best_obj - 1e-12:
                best_obj = cur
                best_a = a.copy()

            opt_val, opt_obj = optimize_coordinate(i, a, c, S, n)
            if opt_val < a[i] - 1e-10 or (opt_val > a[i] + 1e-10 and opt_obj < cur):
                a[i] = opt_val
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a = np.full(n, S / n)
                else:
                    a = a * (S / s)
                improved = True

        if not improved:
            break

    return best_a, best_obj


def get_initialization(name, n, rng, S):
    """Generate various structured initializations."""
    i = np.arange(n)
    if name == "uniform":
        a = np.full(n, 1.0)
    elif name == "random":
        a = rng.random(n)
    elif name == "gaussian":
        a = np.exp(-0.5 * ((i - n / 2) / (n / 8)) ** 2)
    elif name == "cosine":
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * i / n)
    elif name == "anti_peak":
        a = 1.0 - 0.3 * np.cos(2 * np.pi * (n - 1) * i / n)
    elif name == "two_humps":
        a = np.exp(-0.5 * ((i - n * 0.25) / (n / 12)) ** 2) + \
            np.exp(-0.5 * ((i - n * 750) / (n / 12)) ** 2)
    elif name == "spike":
        a = np.ones(n)
        a[n // 4] = 3.0
        a[3 * n // 4] = 3.0
    else:
        a = np.full(n, 1.0)

    a = np.maximum(a, 0.01)
    return a * (S / a.sum())
