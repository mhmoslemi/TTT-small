```python
"""
LP-based iterative optimization for minimizing peak autocorrelation of
non-negative sequences. Uses linear programming to find the best direction
at each step, followed by coordinate descent polish.
"""

import numpy as np
import time
import cvxpy as cp


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A where A[k,j] = a[k-j] if valid, else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
```

```python
def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # j must satisfy 0 <= j < n and 0 <= k-j < n
        # => max(0, k-n+1) <= j <= min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            # A[k, j] = a[k-j] for j in [j_start, j_end]
            # k-j ranges from k-j_start to k-j_end
            A[k, j_start:j_end+1] = a[k - j_end : k - j_start + 1][::-1]
    return A
```
