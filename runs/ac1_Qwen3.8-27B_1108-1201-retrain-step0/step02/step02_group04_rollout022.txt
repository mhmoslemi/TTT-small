import time
import math

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)

    conv = fft_convolve(a, a)

    max_conv = np.max(conv)

def optimize_sequence(a_init, max_time, t_schedule=(1.0, 5.0, 20.0, 100.0), rng=None):
    """Optimize a sequence to minimize the score."""
    if rng is None:
        rng = np.random.default_rng()
    
    a = a_init.astype(np.float64).copy()
    n = len(a)
    t0 = time.monotonic()
    
    # Normalize to have sum = sqrt(2n) for better conditioning
    target_sum = math.sqrt(2 * n)
    current_sum = np.sum(a)
    if current_sum > 0:
        a = a * target_sum / current_sum
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    # Schedule: start with small t (smooth), increase t (sharper)
    t_values = t_schedule
    lr = 0.1
    patience = 500
    
    for t in t_values:
        if time.monotonic() - t0 > max_time:
            break
            
        no_improve = 0
        lr_current = lr
        while time.monotonic() - t0 < max_time:
            score, grad = compute_grad_softmax(a, t)
            if np.isinf(score):
                break
            
            # Gradient step
            a_new = a - lr_current * grad
            a_new = np.maximum(a_new, 0)
            
            # Renormalize
            s = np.sum(a_new)
            if s > 0:
                a_new = a_new * target_sum / s
            
            new_score = compute_score(a_new)
            if new_score < score:
                a = a_new
                no_improve = 0
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
            else:
                no_improve += 1
                lr_current *= 0.9
                if no_improve > patience:
                    break
        
        lr = lr_current
    
    return best_a, best_score

"""
Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses gradient descent with soft-max smoothing of the peak, multiple initializations,
annealing of the soft-max temperature, and multi-length search.
"""

import numpy as np
import time
import math


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def compute_grad_softmax(a, t):
    """Compute score and gradient using soft-max smoothing with temperature t."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, np.zeros(n)
    conv = fft_convolve(a, a)
    max_conv = np.max(conv)

    shifted = t * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = np.sum(exp_shifted)
    p = exp_shifted / Z
    soft_max = max_conv + np.log(Z) / t

    # Gradient of soft_max w.r.t. a[i] = sum_k p[k] * d(conv[k])/d(a[i])
    # d(conv[k])/d(a[i]) = a[k-i] (if valid) + a[i-k] (if valid)
    # term1[i] = sum_j a[j] * p[i+j]  (cross-correlation)
    # term2[i] = sum_j a[j] * p[i-j]  (convolution)
    term1 = np.correlate(p, a, 'full')[:n]
    term2 = fft_convolve(a, p)[:n]
    grad_soft_max = term1 + term2

    # d(score)/d(a[i]) = (2n/S^2) * grad_soft_max[i] - (2n*soft_max)/S^3
    grad = (2.0 * n / (S * S)) * grad_soft_max - (2.0 * n * soft_max) / (S ** 3)

    score = 2.0 * n * max_conv / (S * S)
    return score, grad


def optimize_sequence(a_init, max_time, rng, verbose=False):
    """Optimize a sequence to minimize the score using soft-max gradient descent."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0.0)

    target_sum = math.sqrt(2.0 * n)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) * target_sum / n
    else:
        a = a * target_sum / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Annealing: start with smooth (low t), sharpen (high t)
    t = 1.0
    lr = 2.0 / n
    no_improve = 0
    step = 0
    consecutive_failures = 0

    while time.monotonic() - t0 < max_time:
        score, grad = compute_grad_softmax(a, t)
        if not np.isfinite(score):
            break

        # Gradient step with projected non-negativity and renormalization
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = np.ones(n) * target_sum / n
        else:
            a_new = a_new * target_sum / s

        new_score = compute_score(a_new)

        if np.isfinite(new_score) and new_score < best_score:
            a = a_new
            best_score = new_score
            best_a = a.copy()
            no_improve = 0
            consecutive_failures = 0

        else:
            no_improve += 1
            consecutive_failures += 1
            lr *= 0.95

                # Increase temperature (sharper soft-max)

def two_level_sequence(n, m, alpha, beta):
    """Create a two-level sequence: alpha for first m, beta for rest."""
    a = np.zeros(n)

def structured_init(n, rng, seed_type):
    """Generate structured initial sequences."""
    if seed_type == 0:

    elif seed_type == 1:

    elif seed_type == 3:

                pos += run
            else:

        time_per_init = max(5.0, min(60.0, (deadline - time.monotonic()) * 0.5 / (n_inits * max(1, len(set(lengths_to_try)) - lengths_to_try.index(n) - 1))))
        
        for init_type in range(n_inits):
            if time.monotonic() > deadline - 30:
                break
            a_init = structured_init(n, rng, init_type)
            # Also try perturbations of the current best if same length
            if best_seq is not None and len(best_seq) == n:
                a_init = best_seq + 0.02 * np.sqrt(2 * n) / n * rng.standard_normal(n)
                a_init = np.maximum(a_init, 0)
            
            time_left = deadline - time.monotonic() - 20
            if time_left < 5:
                break
            alloc_time = min(time_per_init, time_left / max(1, n_inits - init_type))
            a_opt, score = optimize_sequence(a_init, alloc_time, rng)
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  n={n}, init={init_type}: score={score:.8f} (NEW BEST)")

    print(f"After Phase 2: best_score={best_score:.8f}, best_len={len(best_seq) if best_seq is not None else 0}")

    # Phase 3: Polishing the best sequence with extended time
    print("\n=== Phase 3: Polishing ===")
    if best_seq is not None:
        time_left = deadline - time.monotonic() - 10
        if time_left > 30:
            # Polish with multiple restarts from the best
            for restart in range(5):
                if time.monotonic() > deadline - 10:
                    break
                n = len(best_seq)
                # Small perturbation
                scale = max(0.01, 0.1 / (restart + 1)) * math.sqrt(2 * n) / n
                a_init = best_seq + scale * rng.standard_normal(n)
                a_init = np.maximum(a_init, 0.0)
                s = np.sum(a_init)
                if s > 0:
                    a_init = a_init * math.sqrt(2 * n) / s

                alloc = min(120.0, time_left / (5 - restart))
                a_opt, score = optimize_sequence(a_init, alloc, rng, verbose=True)
                if score < best_score:
                    best_score = score
                    best_seq = a_opt.copy()
                    print(f"  Polish restart {restart}: score={score:.8f} (NEW BEST)")

    # Phase 4: Try optimizing the parent sequence specifically
    print("\n=== Phase 4: Parent optimization ===")
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        time_left = deadline - time.monotonic() - 10
        if time_left > 30:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            n = len(start_seq)
            target_sum = math.sqrt(2.0 * n)
            s = np.sum(start_seq)
            if s > 0:
                start_seq = start_seq * target_sum / s
            alloc = min(180.0, time_left)
            a_opt, score = optimize_sequence(start_seq, alloc, rng, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  Parent optimized: score={score:.8f} (NEW BEST)")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = 2.0

    # Ensure all values are valid
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = compute_score(best_seq)
    print(f"\n=== FINAL: local_score={final_score:.8f}, n={len(best_seq)} ===")

    # Validate with evaluate_sequence
    final_list = [float(x) for x in best_seq]
    eval_score = evaluate_sequence(final_list)
    print(f"evaluate_sequence score={eval_score:.8f}")

    if not np.isfinite(eval_score) or eval_score > final_score + 0.01:
        # Fallback to a safe uniform sequence
        print("WARNING: Evaluation mismatch, using fallback")
        final_list = [1.0] * 256
        eval_score = evaluate_sequence(final_list)
        print(f"Fallback uniform score={eval_score:.8f}")

    return final_list
