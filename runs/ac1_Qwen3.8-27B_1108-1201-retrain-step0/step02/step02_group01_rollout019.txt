"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Mirror descent (multiplicative weights) with subgradient on log-parameters
2. LP-based direction finding with all constraints
3. Coordinate descent polish

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_convolve(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]

def compute_subgradient(a, h):
    """Compute subgradient g[j] = sum_k h[k]*(a[k-j] + a[j-k]) using FFT."""
    n = len(a)
    fsize = 1 << (3 * n - 2 + 1).bit_length()
    H = fft(h, fsize)
    A = fft(a, fsize)
    # term1[j] = sum_k h[k]*a[k-j] = ifft(H * conj(A))[j]
    term1 = np.real(ifft(H * np.conj(A)))[:n]
    # term2[j] = sum_k h[k]*a[j-k] = ifft(H * A)[j]
    term2 = np.real(ifft(H * A))[:n]
    return term1 + term2

def mirror_descent_step(a, g, eta):
    """Multiplicative weights update: a_new = a * exp(-eta*g) / sum(...)"""
    log_update = -eta * g
    # Shift for numerical stability
    log_update -= log_update.min()
    weights = np.exp(log_update)
    a_new = a * weights
    s = a_new.sum()
    if s < 1e-300:
        return np.full(len(a), 1.0 / len(a))
    return a_new / s

def solve_lp_direction(f, M, n, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns g (unnormalized) or None."""
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[k, j] = f[k - j]
    
    c_obj = -np.ones(n)
    b = np.full(num_constraints, M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                         method='highs', options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            if result.x.sum() > 1e-10:
                return result.x
    except Exception:
        pass
    return None

def coordinate_descent_polish(a, S, time_budget):
    """Polish using coordinate descent."""
    n = len(a)
    a = a.copy()
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Get indices of top peaks
    peak_lags = np.argsort(c)[::-1][:32]
    
    max_sweeps = 200
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        # Focus on coordinates that contribute to peak lags
        important_coords = set()
        for k in peak_lags[:16]:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                if a[j] > 0.01 * S / n:
                    important_coords.add(j)
        
        for i in sorted(important_coords)[:200]:
            if time.monotonic() - start > time_budget:
                break
            original = a[i]
            if original <= 1e-15:
                continue
            # Try reducing a[i] by various amounts
            best_local = original
            best_local_obj = c.max()
            for frac in [0.7, 0.5, 0.3, 0.1]:
                a[i] = original * frac
                s = a.sum()
                if s < 1e-10:
                    a[i] = original
                    break
                a_norm = a * (S / s)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_local_obj:
                    best_local_obj = obj_check
                    best_local = original * frac
                    c = c_check
                    a[:] = a_norm
                else:
                    a[i] = original
                    break
            if best_local != original:
                improved = True
                a[:] = a * (S / a.sum())
                c = fft_convolve_self(a)
                if c.max() < best_obj:
                    best_obj = c.max()
                    best_a = a.copy()
        if not improved:
            break
    
    return best_a, best_obj
