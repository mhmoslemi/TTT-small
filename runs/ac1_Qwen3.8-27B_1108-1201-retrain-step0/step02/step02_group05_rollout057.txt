```python
def subgradient_step(a, num_steps=100):
    n = len(a)
    for _ in range(num_steps):
        c = fft_convolve(a, a)
        k_star = np.argmax(c)
        # Subgradient of conv[k*] with respect to a
        # d(conv[k*])/da[i] = a[k*-i] (if 0<=k*-i<n) + a[i-k*] (if 0<=i-k*<n)
        grad = np.zeros(n)
        for i in range(n):
            if 0 <= k_star - i < n:
                grad[i] += a[k_star - i]
            if 0 <= i - k_star < n:
                grad[i] += a[i - k_star]
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        # Step in negative gradient direction
        # Use adaptive step size
        step = 0.1 / grad_norm  # or some adaptive rule
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        a_new = a_new / np.sum(a_new)
        # Check if it improved
        c_new = fft_convolve(a_new, a_new)
        if np.max(c_new) < np.max(c):
            a = a_new
    return a
```

Now the algorithm:
1. Start with a (normalized, sum=1)
2. Compute c = conv(a,a)
3. Compute weights w = softmax(c / tau) for some temperature tau
4. Solve QP: min a^T W a s.t. a >= 0, sum(a) = 1, where W[i,j] = w[i+j]
5. Update a
6. Anneal tau (decrease over time)
7. Repeat

As tau -> 0, the weights concentrate on the peak, and the QP becomes:

which is exactly what we want (minimize the peak).

```python
def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def projected_gradient_step(a, W, lr):
    grad = W @ a
    a_new = a - lr * grad
    a_new = project_simplex(a_new)
    return a_new
```

This is much faster than SLSQP for each step, and we can do many more steps.

```python
"""
IWL (Iteratively Weighted Least Squares) optimization for minimum peak autocorrelation
non-negative sequences. Combines spectral weighting with projected gradient descent
on the simplex, with adaptive temperature annealing and restart strategies.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def compute_Wa(w, a):
    """Compute W @ a where W[i,j] = w[i+j]. O(n^2) but with small constant."""
    n = len(a)
    Wa = np.zeros(n)
    for i in range(n):
        Wa[i] = np.dot(w[i:i+n], a)
    return Wa


def iwl_optimize(a0, time_limit, tau_start=0.3, tau_end=0.001, inner_steps=100, lr=0.85):
    """
    IWL optimization: iteratively reweight the convolution peaks and solve
    the resulting QP via projected gradient descent.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    no_improve_count = 0
    patience = 50
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        tau = tau_start * (tau_end / tau_start) ** progress
        
        c = fft_convolve(a, a)
        # Compute weights
        w = np.exp(c / tau)
        w = w / np.sum(w)
        
        # Inner projected gradient steps
        for _ in range(inner_steps):
            Wa = compute_Wa(w, a)
            a_new = a - lr * Wa
            a_new = project_simplex(a_new)
            
            # Check if score improved
            # (Do this every few steps to save time)
            a = a_new
        
        # Evaluate
        score = local_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Adaptive: if stuck, perturb
        if no_improve_count >= patience:
            # Add random perturbation
            a = a * np.exp(np.random.standard_normal(n) * 0.05)
            a = np.maximum(a, 1e-10)
            a = a / np.sum(a)
            no_improve_count = 0
    
    return best_a, best_score
```

```python
        #                  = a[k-m] (if 0<=k-m<n) + a[k-m] (if 0<=k-m<n and m != k-m)
        #                  = 2*a[k-m] if m = k-m (i.e., k=2m)
        #                  = a[k-m] if m != k-m
        
        grad = np.zeros(n)
        for m in range(n):
            if 0 <= k_star - m < n:
                if k_star - m == m:  # k_star = 2m
                    grad[m] = 2.0 * a[m]
                else:
                    grad[m] = a[k_star - m]
        
        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step
        a_new = a - (step / gnorm) * grad
        a_new = np.maximum(a_new, 0)
        a_new = a_new / np.sum(a_new)
        
        # Accept if better (or with probability for noise)
        if local_score(a_new) < local_score(a):
            a = a_new
            step = min(step * 1.02, 0.5)
        else:
            step = max(step * 0.95, 0.01)
    
    return a
```

```python
    n, m = len(a), len(b)

        size <<= 1

    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
IWL (Iteratively Weighted Least Squares) + L-BFGS hybrid optimization
for minimum peak autocorrelation non-negative sequences.
Combines spectral peak weighting with projected gradient descent (with momentum)
on the probability simplex, L-BFGS p-norm refinement, and multi-start exploration.

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def Wa_product(w, a):
    """Compute W @ a where W[i,j] = w[i+j] (i,j in [0,n-1])."""
    n = len(a)
    result = np.empty(n)
    for i in range(n):
        result[i] = np.dot(w[i:i + n], a)
    return result

    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    n = len(theta)
    a = softmax_fn(theta)

    center = (n - 1) * bias
    a = np.exp(-0.3 * ((t - center) / (n * 0.3)) ** 2)
    a = np.maximum(a + 0.03 * rng.standard_normal(n), 0.01)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search combining IWL, multi-start exploration, L-BFGS refinement,
    and final polishing for minimum peak autocorrelation non-negative sequences.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 80.0
    rng = np.random.default_rng(seed)
    n_target = 256

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        start = start / np.sum(start)
        s = local_score(start)
        print(f"Provided: n={len(start)}, score={s:.6f}")
        best_score = s
        best_seq = start.copy()

    # Phase 1: IWL from provided sequence
    print(f"\n=== Phase 1: IWL from start ===")
    if best_seq is not None:
        a_start = make_init_interp(best_seq, n_target, rng, noise=0.0) if len(best_seq) != n_target else best_seq.copy()
        t1_limit = min(500.0, deadline - t_global - 1500.0)
        a1, s1 = iwl_optimize(a_start, t1_limit, tau_start=1.0, tau_end=0.01,
                              inner_steps=200, lr=0.45, momentum=0.85, seed=seed)
        print(f"  IWL from start: score={s1:.6f}")
        if s1 < best_score:
            best_score, best_seq = s1, a1
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 2: Multi-start IWL with diverse initializations
    print(f"\n=== Phase 2: Multi-start IWL ===")
    inits = []
    # Chirp variants
    for alpha in [0.15, 0.3, 0.5]:
        for phase in [0.1, 0.3, 0.7]:
            inits.append(('chirp_a{}_p{}'.format(alpha, phase), make_init_chirp(n_target, alpha, phase, rng)))
    # Bell variants
    for width_frac in [0.12, 0.2, 0.3]:
        inits.append(('bell_w{}'.format(width_frac), make_init_bell(n_target, width_frac * n_target, rng)))
    # Asymmetric
    for bias in [0.25, 0.35, 0.45]:
        inits.append(('asym_b{}'.format(bias), make_init_asym(n_target, bias, rng))
    # Perturbed best
    if best_seq is not None:
        for noise in [0.05, 0.1, 0.15, 0.25]:
            inits.append(('pert_n{}'.format(noise), make_init_interp(best_seq, n_target, rng, noise=noise)))

    t2_start = time.monotonic()
    t2_budget = min(700.0, max(300.0, deadline - t2_start - 800.0))
    per_init = t2_budget / len(inits)

    for i, (name, a0) in enumerate(inits):
        if time.monotonic() > deadline - 600.0:
            break
        t_limit = min(per_init * 1.5, deadline - time.monotonic() - 600.0)
        if t_limit < 15:
            break
        a_i, s_i = iwl_optimize(a0, t_limit, tau_start=0.5, tau_end=0.01,
                                inner_steps=150, lr=0.45, momentum=0.85, seed=seed + i * 7)
        if s_i < best_score:
            best_score, best_seq = s_i, a_i
            print(f"  [{name}] score={s_i:.6f} *** NEW BEST ***")
        else:
            print(f"  [{name}] score={s_i:.6f}")

    # Phase 3: L-BFGS p-norm refinement
    print(f"\n=== Phase 3: L-BFGS refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        t3_deadline = min(deadline - 400.0, time.monotonic() + 500.0)
        p_schedule = [4, 8, 16, 32, 64, 128]
        a3, s3 = lbgfs_pnorm(best_seq, p_schedule, maxiter_per_p=300, deadline=t3_deadline)
        print(f"  L-BFGS: score={s3:.6f}")
        if s3 < best_score:
            best_score, best_seq = s3, a3
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 4: Final intensive IWL polish
    print(f"\n=== Phase 4: Final IWL polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        t4_time = max(100.0, deadline - time.monotonic() - 30.0)
        # Run with lower tau for fine-tuning
        a4, s4 = iwl_optimize(best_seq, t4_time, tau_start=0.2, tau_end=0.003,
                              inner_steps=300, lr=0.35, momentum=0.9, seed=seed + 500)
        print(f"  Final IWL: score={s4:.6f}")
        if s4 < best_score:
            best_score, best_seq = s4, a4
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 5: Quick L-BFGS on final best
    if best_seq is not None and time.monotonic() < deadline - 30:
        t5_deadline = deadline - 10
        a5, s5 = lbgfs_pnorm(best_seq, [32, 64, 128], maxiter_per_p=300, deadline=t5_deadline)
        if s5 < best_score:
            best_score, best_seq = s5, a5
            print(f"  Phase 5 L-BFGS: score={s5:.6f} *** NEW BEST ***")

    # Final output
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if best_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return [1.0] * 256
```
