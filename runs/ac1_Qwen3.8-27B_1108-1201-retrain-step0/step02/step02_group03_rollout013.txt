"""
Optimized step function search using gradient descent with annealed soft-max,
peak-directed local refinement, and multi-length exploration.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def soft_max_and_weights(x, tau):
    """Compute soft-max value and normalized weights."""
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    return x_max + tau * np.log(s), e / s


def grad_soft_max_conv(a, tau):
    """Compute soft-max(conv(a,a)) and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_and_weights(c, tau)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = (2.0 / tau) * corr[n - 1:2 * n - 1]
    return value, g, c


def fft_phase_init(n, rng, n_iter=100):
    """Gerchberg-Saxton style initialization."""
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1

    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def random_structured_init(n, rng):
    """Structured random initialization."""
    a = np.ones(n)
    # Add several random sinusoidal components
    n_components = rng.integers(3, 8)
    for _ in range(n_components):
        freq = rng.uniform(0.02, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        a += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def optimize_gradient(n, a_init, time_limit, rng, tau_start_mult=60.0, tau_end_mult=0.2):
    """
    Adam gradient descent on softmax-parameterized sequence with annealed
    soft-max temperature.
    """
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S
    theta = np.log(a / S + 1e-12)
    
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr0 = 3.0
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)
    
    tau_start = tau_start_mult / n
    tau_end = tau_end_mult / n
    
    # Cosine annealing for learning rate
    min_lr = lr0 * 0.01
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        elapsed_frac = min(1.0, elapsed / time_limit)
        
        # Annealing: cosine schedule
        tau = tau_end + 0.5 * (tau_start - tau_end) * (1 + np.cos(np.pi * elapsed_frac))
        
        # Learning rate: cosine decay
        lr = min_lr + 0.5 * (lr0 - min_lr) * (1 + np.cos(np.pi * elapsed_frac))
        
        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)

        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 200 == 0:
            a_check = softmax_normalize(theta) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
            if step % 1000 == 0:
                print(f"        step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, lr={lr:.3f}, t={time.monotonic()-t0:.1f}s")
    
    a_final = softmax_normalize(best_theta) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
        best_theta = theta.copy()
    
    a_final = softmax_normalize(best_theta) * S
    return a_final, best_score

    cooling = 0.9998

        iter_count += 1
        
        if r < 0.5:
            # Peak-directed: reduce elements that contribute most to the peak lag
            conv = fft_convolve(current, current)
            peak_lag = np.argmax(conv)
            
            # Compute contribution of each element to the peak
            i_min = max(0, peak_lag - n + 1)
            i_max = min(n - 1, peak_lag)
            contrib = np.zeros(n)
            for i in range(i_min, i_max + 1):
                j = peak_lag - i
                contrib[i] += current[i] * current[j]
            
            # Also consider second and third peaks
            top_lags = np.argsort(conv)[-3:]
            for lag in top_lags[1:]:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += 0.5 * current[i] * current[j]
            
            # Reduce top contributors, increase bottom contributors
            n_reduce = max(3, n // 20)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            
            scale = np.exp(rng.normal(-0.04, 0.02, size=len(top_idx)))
            candidate[top_idx] *= scale
            candidate[bot_idx] *= np.exp(rng.normal(0.04, 0.02, size=len(bot_idx)))
            
        elif r < 0.7:

            n_blocks = rng.integers(2, 6)

                factor = np.exp(rng.normal(-0.05, 0.03, size=bw))

                # Compensate elsewhere
                comp_idx = rng.integers(0, n)
                candidate[comp_idx] /= factor.mean()

def upsample(a, new_n, rng):
    """Upsample a sequence to a longer length via interpolation + noise."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def local_score(a):
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / e.sum()

    x_max = x.max()
    shifted = (x - x_max) / tau
    e = np.exp(shifted)
    s = e.sum()
    return x_max + tau * np.log(s), e / s

    n = len(a)
    c = fft_convolve(a, a)
    val, w = soft_max_val_and_weights(c, tau)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = (2.0 / tau) * corr[n-1:2*n-1]
    return val, g, c

The sum doubles: sum_new = 2 * sum_old
Score_new = 2*(2n) * 2*max_conv_old / (2*sum_old)^2 = 8n * max_conv / (4 sum_old^2) = 2 * (2n * max_conv / sum_old^2) = 2 * S_old

Wait, that's worse. Let me redo this more carefully.

If a_new = [a_0, a_0, a_1, a_1, ..., a_{n-1}, a_{n-1}] (length 2n):
sum_new = 2 * sum_old
conv_new at lag 0 = 2 * sum(a_i^2) = 2 * conv_old(0)
conv_new at other lags are also approximately doubled

So score_new = 2*(2n) * 2*max_conv_old / (2*sum_old)^2 = 8n * max_conv_old / (4 sum_old^2) = 2 * (2n * max_conv_old / sum_old^2) = 2 * S_old

So naive doubling makes the score worse! This is because the `2n` factor in the numerator grows.

But if we upsample smoothly (interpolation), the situation is different. The convolution of the interpolated sequence will have a different peak structure.

Actually, the key insight is: the score is `2n * max_conv / sum^2`. If we can keep `max_conv / sum^2` roughly constant while increasing n, the score increases. So we NEED `max_conv / sum^2` to decrease as n increases for the score to improve.

For a "flat" convolution, `max_conv ≈ sum^2 / (2n-1)`, so `max_conv / sum^2 ≈ 1/(2n-1)`, and score ≈ `2n/(2n-1) ≈ 1`. So for perfectly flat convolution, larger n is better.

For a constant sequence, `max_conv / sum^2 = 1/n`, and score = 2. The ratio doesn't depend on n.

So the question is: can we make the convolution flatter for larger n? If the "flatness improvement" from going from n to 2n is more than a factor of 2, the score improves.
