Optimization of non-negative step function to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses LP-based iteration (alternating between fixing a and optimizing b via LP,
then updating a towards b) combined with multiple initializations and local search.

import numpy as np

from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_score(a):
    """Compute score = 2*n*max(convolve(a,a))/sum(a)^2 using FFT for large n."""
    a = np.maximum(a, 0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    # Use FFT for convolution when n is large
    if n > 128:
        size = 1
        while size < 2*n - 1:
            size <<= 1
        fa = np.fft.rfft(a, size)
        c = np.fft.irfft(fa * fa, size)[:2*n-1]
        c = np.maximum(c, 0)  # numerical errors
    else:
        c = np.convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def build_conv_matrix_indices(n):
    """Precompute the sparsity pattern of the convolution matrix."""
    rows = []
    cols = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)

def solve_lp(a, M, rows, cols, n):
    """Solve: max sum(b) s.t. B*b <= M, b >= 0."""
    # Build sparse matrix
    vals = np.zeros(len(rows))
    for idx in range(len(rows)):
        k = rows[idx]
        j = cols[idx]
        vals[idx] = a[k - j]
    
    B = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    
    res = linprog(c_obj, A_ub=B, b_ub=b_ub, bounds=[(0, None)] * n, 
                  method='highs', options={'presolve': True})
    
    if res.status != 0:
        return None
    return res.x

    else:
        c = np.convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def build_conv_indices(n):
    """Precompute sparsity pattern indices for the convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_for_b(a, M, rows, cols, n):
    """Solve: max sum(b) s.t. B*b <= M, b >= 0. Returns optimal b or None."""
    vals = np.empty(len(rows), dtype=np.float64)
    for idx in range(len(rows)):
        vals[idx] = a[rows[idx] - cols[idx]]
    B = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=B, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 5})
    if res.status != 0:
        return None
    return res.x


def lp_optimize(a_init, time_limit, t=1.0, verbose=False):
    """LP-based iteration to minimize peak self-convolution."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return a_init.copy(), fast_score(a_init)
    a = a / S * math.sqrt(2.0 * n)
    S = math.sqrt(2.0 * n)

    current_score = fast_score(a)
    best_a = a.copy()
    best_score = current_score
    rows, cols = build_conv_indices(n)

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        c = np.convolve(a, a)
        M = np.max(c)

        b = solve_lp_for_b(a, M, rows, cols, n)
        if b is None:
            break

        sum_b = np.sum(b)
        if sum_b < 1e-12:
            break

        b = b * S / sum_b

        improved = False
        for t_try in [t, 0.5 * t, 0.25 * t, 0.1 * t]:
            a_new = (1.0 - t_try) * a + t_try * b
            a_new = np.maximum(a_new, 0.0)
            s_new = np.sum(a_new)
            if s_new < 0.01:
                continue
            a_new = a_new / s_new * S
            sc = fast_score(a_new)
            if sc < current_score - 1e-12:
                a = a_new
                current_score = sc
                t = t_try
                improved = True
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                break

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"        LP it={it}, score={current_score:.10f}, best={best_score:.10f}, t={t:.3f}, time={elapsed:.1f}s")

    return best_a, best_score


def local_search(a, time_limit, rng):
    """Coordinate-wise local search to polish the solution."""
    a = a.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return a, fast_score(a)
    a = a / S * math.sqrt(2.0 * n)
    S = math.sqrt(2.0 * n)
    current_score = fast_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()

    base_step = S / np.sqrt(n)
    step_mult = 1.0

    while time.monotonic() - t0 < time_limit:
        improved = False
        step = base_step * step_mult
        n_trials = min(n, 200)
        indices = rng.choice(n, size=n_trials, replace=False) if n_trials < n else np.arange(n)
        for i in indices:
            for sign in [1, -1]:
                a_try = a.copy()
                a_try[i] = max(0.0, a_try[i] + sign * step)
                s_try = np.sum(a_try)
                if s_try < 0.01:
                    continue
                a_try = a_try / s_try * S
                sc = fast_score(a_try)
                if sc < current_score - 1e-12:
                    a = a_try
                    current_score = sc
                    improved = True
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                    break
        if not improved:
            step_mult *= 0.5
            if step_mult < 1e-6:
                break
        else:
            step_mult = max(step_mult, 0.5)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Find a non-negative sequence minimizing 2*n*max(convolve(a,a))/sum(a)^2.
    Uses LP-based iteration from multiple diverse starting points,
    followed by local search polishing.
    """
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def try_update(seq):
        nonlocal best_score, best_seq
        if seq is None or len(seq) == 0:
            return
        s = fast_score(seq)
        if s < best_score:
            best_score = s
            best_seq = seq.copy()

    # Generate diverse initial candidates
    print("=== Generating initial candidates ===")
    candidates = []

    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        seq = np.array(height_sequence_1, dtype=np.float64)
        seq = np.clip(seq, 0, 1000)
        if np.sum(seq) > 0.01:
            candidates.append(('provided', seq))

    for n in [128, 256, 512]:
        candidates.append((f'uniform_{n}', np.ones(n)))
        i = np.arange(n, dtype=np.float64)
        candidates.append((f'cosine_{n}', 0.5 * (1.0 + np.cos(2.0 * np.pi * i / (n - 1)))))
        # Truncated central binomial coefficients (from sqrt(1/(1-z)) power series)
        p = np.array([comb(2 * k, k) / (4.0 ** k) for k in range(min(n, 1024))], dtype=np.float64)
        if len(p) < n:
            p = np.concatenate([p, np.zeros(n - len(p))])
        candidates.append((f'binom_{n}', p))
        # Random
        candidates.append((f'rand_{n}', rng.random(n) + 0.01))
        # Asymmetric: high at one end
        asym = np.linspace(1.0, 0.1, n)
        candidates.append((f'asym_{n}', asym))
        # Two peaks
        idx = np.arange(n)
        peak1 = np.exp(-8.0 * ((idx - n // 4) / n) ** 2)
        peak2 = np.exp(-8.0 * ((idx - 3 * n // 4) / n) ** 2)
        candidates.append((f'twopk_{n}', peak1 + peak2))

    for name, seq in candidates:
        try_update(seq)
    print(f"  Best initial score: {best_score:.8f}")

    # Sort candidates by initial score (best first)
    scored_candidates = []
    for name, seq in candidates:
        s = fast_score(seq)
        scored_candidates.append((s, name, seq))
    scored_candidates.sort(key=lambda x: x[0])

    # Run LP optimization on each candidate
    print("\n=== LP Optimization ===")
    total_start = time.monotonic()

    for s_init, name, seq in scored_candidates:
        if time.monotonic() > deadline:
            print("  Time budget exhausted.")
            break
        time_left = deadline - time.monotonic()
        if time_left < 5:
            break

        n = len(seq)
        # Estimate time per LP solve
        if n <= 128:
            est_time_per_lp = 0.005
        elif n <= 256:
            est_time_per_lp = 0.015
        elif n <= 512:
            est_time_per_lp = 0.06
        else:
            est_time_per_lp = 0.25

        # Allocate time: at least 30s, at most 60% of remaining
        alloc = min(max(30.0, time_left * 0.4), time_left * 0.7)
        # Cap number of iterations
        max_iters_est = int(alloc / est_time_per_lp)
        if max_iters_est > 5000:
            alloc /= (max_iters_est / 5000.0)

        print(f"  {name} (n={n}, init_score={s_init:.6f}, alloc={alloc:.0f}s)")
        a_opt, s_opt = lp_optimize(seq.copy(), alloc, t=1.0, verbose=False)
        try_update(a_opt)
        if s_opt < s_init:
            print(f"  -> {s_opt:.10f} (improved from {s_init:.6f})")
        else:
            print(f"  -> {s_opt:.10f} (no improvement)")

    # Second pass: optimize the best sequence more aggressively
    if best_seq is not None and time.monotonic() < deadline:
        time_left = deadline - time.monotonic()
        n = len(best_seq)
        print(f"\n=== Deep optimization of best (n={n}, score={best_score:.8f}, time_left={time_left:.0f}s) ===")

        # Re-run LP with smaller step for finer convergence
        alloc1 = min(120.0, time_left * 0.3)
        a2, s2 = lp_optimize(best_seq.copy(), alloc1, t=0.5, verbose=False)
        try_update(a2)
        print(f"  LP t=0.5: {s2:.10f}")

        # Random restarts
        for restart in range(8):
            if time.monotonic() > deadline:
                break
            time_left = deadline - time.monotonic()
            if time_left < 20:
                break

            scale = [0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.5][restart]
            a_pert = best_seq * (1.0 + scale * rng.standard_normal(n))
            a_pert = np.maximum(a_pert, 0.0)
            s_pert = np.sum(a_pert)
            if s_pert < 0.01:
                continue
            a_pert = a_pert / s_pert * math.sqrt(2.0 * n)

            alloc = min(60.0, time_left * 0.3)
            a_opt, s_opt = lp_optimize(a_pert, alloc, t=1.0, verbose=False)
            try_update(a_opt)
            print(f"  Restart {restart} (scale={scale}): {s_opt:.10f} {'*' if s_opt < best_score else ''}")

    # Local search polishing
    if best_seq is not None and time.monotonic() < deadline:
        time_left = deadline - time.monotonic()
        if time_left > 20:
            print(f"\n=== Local search polishing (time_left={time_left:.0f}s) ===")
            loc_time = min(120.0, time_left * 0.5)
            a_ls, s_ls = local_search(best_seq.copy(), loc_time, rng)
            try_update(a_ls)
            print(f"  After local search: {s_ls:.10f}")

            # One more LP pass after local search
            if time.monotonic() < deadline:
                time_left2 = deadline - time.monotonic()
                if time_left2 > 20:
                    a3, s3 = lp_optimize(best_seq.copy(), min(60.0, time_left2 * 0.5), t=1.0, verbose=False)
                    try_update(a3)
                    print(f"  Final LP pass: {s3:.10f}")

    # Verify and return
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(final_list)
    our_score = fast_score(np.array(final_list))
    print(f"\n=== FINAL: evaluate_score={final_score:.10f}, our_score={our_score:.10f}, n={len(final_list)} ===")

    return final_list
