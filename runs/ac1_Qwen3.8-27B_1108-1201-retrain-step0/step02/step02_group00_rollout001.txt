"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key insight: Optimal sequences must be ASYMMETRIC (symmetric sequences have
center convolution value = sum(a^2) >= (sum a)^2/n, giving objective >= 2).

Algorithm:
- Diverse asymmetric initializations (exponential, power-law, step, front/back loaded)
- Phase 1: FISTA with adaptive soft-max for broad exploration
- Phase 2: Aggressive peak-pincer optimization (collect top peaks, combined subgradient)
- Phase 3: High-precision polish with large-k top subgradient
- Backtracking line search for robust step selection
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_soft(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a.
    grad[m] = (2/beta) * sum_k softmax_k * a[k-m] (cross-correlation).
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # grad[m] = 2 * sum_k w[k] * a[k - m] for valid k, m
    # This is the cross-correlation of w and a, computed via FFT
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_toppk(a, c, top_k):
    """Gradient using top-k active lags."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total

    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_val(a):
    """Compute the objective value 2n*max_conv/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def run_fista(a_init, n, S, max_time, rng, beta_init=2.0, beta_max=300.0, noise_scale=0.005):
    """FISTA with adaptive soft-max and simplex projection."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.003 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = fft_grad_soft(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta * 2.0 * (1.0 + 0.0003 * it))
        step = max(step, 1e-7)

        x_new = project_simplex(y - step * grad, S)

        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 2000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0

    return best_local, best_score


def run_peak_pincer(a_init, n, S, max_time, rng, top_k=10, noise_scale=0.003):
    """
    Peak-pincer method: repeatedly identify top peaks, compute combined
    subgradient, take projected steps. More aggressive peak reduction.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (2.0 * np.sqrt(n))

    while time.monotonic() - start < max_time:
        it += 1
        k = min(256, top_k + it // 50)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = fft_grad_toppk(y, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: decrease step as we make progress
        step = 2.0 * S / (n * (1.0 + 0.003 * it))

        # Verify improvement, backtrack if needed
        c_new = fft_convolve_self(x_new)
        if c_new.max() > cur * 1.01:

def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
