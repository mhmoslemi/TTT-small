def obj_normalized(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def solve_lp_direction(a, M):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (a*g)_k <= M for all k.
    Returns normalized g (sum=1) or None if infeasible.
    """
    n = len(a)
    n_cons = 2 * n - 1
    
    # Build constraint matrix
    A_ub = np.zeros((n_cons, n))
    for k in range(n_cons):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, k - j] += a[j]
    
    b_ub = np.full(n_cons, M)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not result.success:
        return None
    
    g = result.x.copy()
    s = g.sum()
    if s < 1e-10:
        return None
    g /= s
    return g

def lp_optimize(a0, time_limit, max_iters=5000):
    """LP-based optimization with adaptive step size."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step_size = 0.4
    patience = 0
    max_patience = 100
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = conv.max()
        g = solve_lp_direction(a, M)
        if g is None:
            # Try with slightly larger M
            g = solve_lp_direction(a, M * 1.01)
            if g is None:
                break
        
        a_new = (1 - step_size) * a + step_size * g
        a_new = np.maximum(a_new, 0.0)
        ns = a_new.sum()
        if ns < 1e-12:
            break
        a_new /= ns
        
        new_conv = fast_convolve(a_new)
        new_obj = 2.0 * n * new_conv.max()
        
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        conv = new_conv
        
        if patience >= max_patience:
            step_size *= 0.7
            patience = 0
            if step_size < 0.005:
                break
    
    return best_a, best_obj

def smooth_obj_and_grad(x, n, t):
    """
    Smooth objective and gradient in log-parametrization.
    x is the unconstrained parameter, a = softmax(x).
    Returns (f, dx) where dx = df/dx.
    """
    # Compute a from x
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a /= a.sum()
    
    # Compute conv and objective
    conv = fast_convolve(a)
    conv_scaled = conv / t
    conv_scaled_shifted = conv_scaled - conv_scaled.max()
    w = np.exp(conv_scaled_shifted)
    w_sum = w.sum()
    w /= w_sum
    f = 2.0 * n * t * (conv_scaled.max() + np.log(w_sum))
    
    # Compute g = df/da = 4n * correlate(w, a, 'valid')
    g = 4.0 * n * np.correlate(w, a, mode='valid')
    
    # Compute df/dx = a * (g - dot(g, a))
    ga = np.dot(g, a)
    dx = a * (g - ga)
    
    return f, dx

def lbb_optimize(a0, time_limit, t=1.0):
    """L-BFGS-B optimization in log-parametrization."""
    a = np.maximum(a0, 1e-300).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    # Initialize x from a
    x0 = np.log(a)
    
    t0 = time.monotonic()
    best_obj = obj_normalized(a)
    best_a = a.copy()
    
    # Use L-BFGS-B with time-based stopping
    nfev_counter = [0]
    
    def fun(x):
        nfev_counter[0] += 1
        f, dx = smooth_obj_and_grad(x, n, t)
        # Periodically update best
        if nfev_counter[0] % 10 == 0:
            a_curr = np.exp(x - x.max())
            a_curr /= a_curr.sum()
            curr_obj = obj_normalized(a_curr)
            if curr_obj < best_obj:
                best_obj = curr_obj
                best_a = a_curr.copy()
        return f, dx
    
    result = minimize(fun, x0, method='L-BFGS-B', 
                      options={'maxiter': 10000, 'maxfun': 100000,
                               'factr': 1e8})
    
    # Final check
    x_final = result.x
    a_final = np.exp(x_final - x_final.max())
    a_final /= a_final.sum()
    final_obj = obj_normalized(a_final)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a_final.copy()
    
    return best_a, best_obj

def anneal_informed(a0, time_limit, rng):
    """Informed simulated annealing."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.02 * (1.0 - frac) ** 2 + 0.0003
        
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # Find peak positions
        peak_thresh = max_c * 0.8
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        
        k = int(rng.choice(peak_positions))
        
        # Find which a values contribute most to this peak
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        seg_a = a[j_start:j_end + 1]
        seg_b = a[k - j_end:k - j_start + 1][::-1]
        products = seg_a * seg_b
        total = products.sum()
        if total < 1e-15:
            continue
        
        probs = products / total
        top_j = j_start + int(rng.choice(j_end - j_start + 1, p=probs))
        
        # Choose target (prefer low-contribution positions)
        target_j = int(rng.integers(n))
        
        if top_j == target_j:
            continue
        
        delta = a[top_j] * rng.uniform(0.01, 0.08)
        if delta < 1e-10:
            continue
        
        a[top_j] -= delta
        a[target_j] += delta
        
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
    
    return best_a, best_obj

def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian shapes
    for sf in [0.15, 0.25, 0.4, 0.6]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_' + str(sf)] = g
    
    # Cosine (raised)
    c = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1))) if n > 1 else np.ones(1)
    c = np.maximum(c, 1e-10)
    c /= c.sum()
    inits['cosine'] = c
    
    # Parabolic
    p = 1.0 - (2.0 * x / (n - 1) - 1.0) ** 2 if n > 1 else np.ones(1)
    p = np.maximum(p, 1e-10)
    p /= p.sum()
    inits['parabola'] = p
    
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 8)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        inits['rand_' + str(alpha)] = rng.dirichlet(np.ones(n) * alpha)
    
    # Parent (resized)
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0).copy()
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits['parent_interp'] = pp / pp.sum()
    
    # Symmetric random
    half_n = (n + 1) // 2
    half = rng.dirichlet(np.ones(half_n))
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[:n - half_n][::-1]
    sym = np.maximum(sym, 1e-15)
    sym /= sym.sum()
    inits['sym_rand'] = sym
    
    return inits

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) for a with sum(a)=1."""

def build_conv_matrix(a):
    """
    Build the convolution constraint matrix.
    Row k of the matrix gives coefficients for (a*g)_k = sum_j A[k,j] * g_j.
    Shape: (2*n-1, n)
    """
    n = len(a)
    n_cons = 2 * n - 1
    A = np.zeros((n_cons, n))
    for k in range(n_cons):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, k - j] += a[j]
    return A


def solve_lp_dir_maxsum(a, M):
    """
    LP: maximize sum(g) s.t. g >= 0, (a*g)_k <= M for all k.
    Returns normalized g or None.
    """
    n = len(a)
    n_cons = 2 * n - 1
    A_ub = build_conv_matrix(a)
    b_ub = np.full(n_cons, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success:
        return None
    g = result.x.copy()
    s = g.sum()
    if s < 1e-12:
        return None
    return g / s


def solve_lp_dir_mint(a):
    """
    LP: minimize t s.t. (a*g)_k <= t for all k, g >= 0, sum(g) = 1.
    Returns (g, t_opt) or (None, None).
    """
    n = len(a)
    n_cons = 2 * n - 1
    A_conv = build_conv_matrix(a)
    
    # Variables: g_0,...,g_{n-1}, t (n+1 total)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A_conv, -np.ones((n_cons, 1))])
    b_ub = np.zeros(n_cons)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    if not result.success:
        return None, None
    g = result.x[:n].copy()
    t_opt = result.x[n]
    return g, t_opt

def generate_inits(n, rng, parent):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g = np.maximum(g, 1e-15)
        g /= g.sum()
        inits['gauss_' + str(sf)] = g
    
    # Raised cosine
    c = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(n - 1, 1)))
    c = np.maximum(c, 1e-10)

    # Parabolic
    p = 1.0 - (2.0 * x / max(n - 1, 1) - 1.0) ** 2

    inits['parabola'] = p
    
    # Bimodal (edges)

    w = max(1, n // 8)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Triangular (peak at center)

    # === Phase 3: Annealing and further refinement ===
    if best_sequence is not None and time.monotonic() < deadline - 50:
        print("Phase 3: Annealing + refinement...")
        phase3_budget = min(500.0, deadline - time.monotonic() - 50)
        
        # Multiple annealing runs from best
        n_anneal = 3
        anneal_per = phase3_budget * 0.35 / n_anneal
        
        for i in range(n_anneal):
            if time.monotonic() > deadline - 50:
                break
            rng_a = np.random.default_rng(seed + 5000 + i * 13)
            a_start = normalize(best_sequence)
            a_opt, _ = anneal_informed(a_start, time_limit=anneal_per, rng=rng_a)
            record(a_opt)
        
        # LP polish on the best found
        if time.monotonic() < deadline - 40:
            remaining = deadline - time.monotonic() - 40
            if remaining > 10:
                a_best = normalize(best_sequence)
                a_opt, _ = lp_optimize(a_best, time_limit=remaining * 0.5, max_iters=50000)
                record(a_opt)
        
        # L-BFGS-B with different temperature
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic() - 30
            if remaining > 5:
                a_opt, _ = lbb_optimize(normalize(best_sequence), time_limit=min(remaining, 200), t=0.3)
                record(a_opt)
    
    # === Phase 4: Final exploration with different lengths ===
    if time.monotonic() < deadline - 30:
        print("Phase 4: Final exploration...")
        remaining = deadline - time.monotonic() - 30
        
        # Try a few more lengths that might be optimal
        extra_lengths = [192, 384, 768, 1536]
        if best_sequence is not None:
            bn = len(best_sequence)
            for en in [bn // 2, bn * 2]:
                if en not in extra_lengths and 32 <= en <= 4096:
                    extra_lengths.append(en)
        
        for n in extra_lengths[:4]:
            if time.monotonic() > deadline - 30:
                break
            t_alloc = min(remaining / 4.0, deadline - time.monotonic() - 30)
            if t_alloc < 5.0:
                break
            
            # Use parent or uniform init
            if parent is not None and parent.size >= n:
                a0 = normalize(parent[:n])
            else:
                a0 = np.ones(n) / n
            
            a_opt, _ = lp_optimize(a0, time_limit=t_alloc, max_iters=10000)
            record(a_opt)
            
            # Quick L-BFGS polish
            if time.monotonic() < deadline - 20:
                rem = min(deadline - time.monotonic() - 20, 60)
                if rem > 3:
                    a_opt2, _ = lbb_optimize(a_opt, time_limit=rem, t=1.0)
                    record(a_opt2)
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = safe_evaluate(best_sequence)
    
    print("=" * 60)
    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))
    print("Total time: %.1fs" % (time.monotonic() - t_start))
    
    return [float(x) for x in best_sequence]
