import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    C = fft_convolve(a, a)
    return 2.0 * n * np.max(C) / (S * S)

def solve_lp(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)(k) <= M for all k."""
    n = len(a)
    num_vars = n
    num_constraints = 2 * n - 1
    
    # conv(a, b)(k) = sum_i a[i] * b[k-i] for valid i
    # This is a linear function of b.
    # For constraint k: sum_i a[i] * b[k-i] <= M
    # The coefficient of b[j] in constraint k is a[k-j] (if valid) or 0.
    
    # Build the constraint matrix A (num_constraints x num_vars)
    # A[k, j] = coefficient of b[j] in constraint k
    # conv(a,b)(k) = sum_j b[j] * a[k-j] (where a[k-j] is 0 if k-j out of range)
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    
    # This is a Toeplitz-like matrix.
    # For efficiency, let's build it row by row.
    
    A = np.zeros((num_constraints, num_vars))
    for k in range(num_constraints):
        # b[j] appears with coefficient a[k-j] if 0 <= k-j < n
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    
    # Objective: max sum(b) => min -sum(b)
    c = -np.ones(num_vars)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * num_vars
    
    # Solve
    result = linprog(c, A_ub=A, b_ub=np.full(num_constraints, M), bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return None

def fft_autocorr(a):
    """Self-convolution (autocorrelation) via FFT."""
    n = len(a)
    size = 1

    """Build Toeplitz matrix A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)

    for k in range(ncols):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        A[k, lo:hi + 1] = a[k - hi:k - lo:-1]

def subgradient_step(a, t, rng=None):
    """One subgradient step on the max autocorrelation peak."""
    n = len(a)
    C = fft_autocorr(a)
    k_star = int(np.argmax(C))
    # Gradient of C(k_star) w.r.t. a[i] is 2*a[k_star - i] if valid
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    # Project onto sum(a)=1: subtract mean
    grad = grad - np.mean(grad)
    a_new = a - t * grad
    a_new = np.maximum(a_new, 0.0)
    S_new = a_new.sum()
    if S_new < 1e-10:
        return a
    return a_new / S_new

                stall = 0

    return best_a, 2.0 * n * best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Find the optimal non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP-based iterative refinement with multiple initializations and
    subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)

    best_score = np.inf
    best_seq = None

    configs = []

    # Try parent sequence if available
    try:
        if 'height_sequence_1' in dir() and height_sequence_1 is not None:
            parent = np.asarray(height_sequence_1, dtype=np.float64)
            parent = np.clip(parent, 0, 1000)
            if parent.sum() > 0.01:
                configs.append((len(parent), parent, 350, 'parent'))
    except Exception as e:
        print(f"  Parent load error: {e}")

    # Random initializations
    for n in [256, 512, 1024]:
        configs.append((n, rng.exponential(1.0, size=n), 200, f'exp_{n}'))

    # Cosine window
    for n, period in [(256, 7.0), (512, 11.0), (256, 13.0)]:
        x = np.arange(n)
        init = 0.5 + 0.4 * np.cos(2.0 * np.pi * x / period)
        configs.append((n, init, 120, f'cos_{n}_p{int(period)}'))

    # Sine with offset
    n = 512
    x = np.arange(n)
    init = 0.5 + 0.3 * np.sin(2.0 * np.pi * x / 5.0) + 0.1 * np.sin(2.0 * np.pi * x / 17.0)
    configs.append((n, init, 120, 'sin512'))

    # Ramp
    configs.append((256, np.linspace(0.3, 2.0, 256), 80, 'ramp'))

    # Two-bump
    n = 512
    x = np.arange(n)
    init = np.exp(-((x - 128) ** 2) / 2000.0) + np.exp(-((x - 384) ** 2) / 2000.0) + 0.05
    configs.append((n, init, 120, 'twobump'))

    # Process configs
    for n, init, time_alloc, name in configs:
        if time.monotonic() >= deadline:
            break
        init = np.clip(init, 0.01, None)
        print(f"  [{name}] n={n}, alloc={time_alloc}s")

        # LP optimization
        a_opt, score = optimize_lp(init, time_alloc, t_init=0.5, rng=rng, verbose=True)
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"    LP result: {score:.6f} (new best)")
        else:
            print(f"    LP result: {score:.6f} (not better than {best_score:.6f})")

        # Subgradient polishing on the LP result
        time_left = deadline - time.monotonic()
        polish_time = min(60.0, time_left * 0.3)
        if polish_time > 10:
            a_pol, score_pol = refine_subgradient(a_opt, polish_time, rng=rng)
            if score_pol < best_score:
                best_score = score_pol
                best_seq = a_pol.copy()
                print(f"    After polish: {score_pol:.6f} (new best)")

    # If we have time, try a few more random restarts
    time_left = deadline - time.monotonic()
    if time_left > 60:
        print(f"\n  Extra time: {time_left:.0f}s, trying more random restarts")
        for attempt in range(5):
            if time.monotonic() >= deadline:
                break
            n = rng.choice([256, 512])
            init = rng.dirichlet(np.ones(n) * 0.5)  # diverse random
            time_alloc = min(60.0, (deadline - time.monotonic()) * 0.4)
            if time_alloc < 20:
                break
            print(f"  [restart_{attempt}] n={n}")
            a_opt, score = optimize_lp(init, timealloc if False else time_alloc, t_init=0.4, rng=rng, verbose=False)
            # Quick polish
            if time.monotonic() < deadline:
                polish_t = min(20.0, (deadline - time.monotonic()) * 0.5)
                if polish_t > 10:
                    a_pol, score_pol = refine_subgradient(a_opt, polish_t, rng=rng)
                    score = min(score, score_pol)
                    if score_pol < score:
                        a_opt = a_pol
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"    Score: {score:.6f} (new best)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = 2.0

    # Return at a reasonable scale (score is scale-invariant)
    final = np.clip(best_seq * 100.0, 0.0, 1000.0)
    final_list = [float(x) for x in final]

    # Validate
    vs = evaluate_sequence(final_list)
    print(f"\n  Validated score: {vs:.8f}")
    if np.isfinite(vs) and vs < best_score:
        best_score = vs

    print(f"=== FINAL: score={best_score:.8f}, n={len(final_list)} ===")
    return final_list
