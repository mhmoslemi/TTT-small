"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions. Uses L-BFGS-B with log-sum-exp smoothing on softmax
parameterization, peak-directed local search, and progressive length refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max(conv) / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def softmax_stable(theta):
    """Numerically stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_and_grad(conv, tau):
    """Compute log-sum-exp smoothed max and gradient w.r.t. conv elements."""
    M = np.max(conv)
    shifted = (conv - M) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    Z = np.sum(exp_vals)
    w = exp_vals / Z
    value = M + tau * np.log(Z)
    return value, w


def objective_and_grad(theta, n, tau):
    """
    Objective and gradient for L-BFGS-B.
    theta parameterizes a = softmax(theta) * sqrt(2n).
    We minimize 2n * max(conv(a,a)) / (sum(a))^2 = max(conv(a_normalized, a_normalized)) * 2n
    where a_normalized has sum 1.
    """
    p = softmax_stable(theta)  # sum = 1
    conv = fft_convolve(p, p)
    V, w = smooth_max_and_grad(conv, tau)
    score = 2.0 * n * V  # since sum(p) = 1, score = 2n * max(conv) / 1 = 2n * V

    # Gradient of V w.r.t. p:
    # dV/dp[j] = 2 * sum_k w[k] * p[k+j] (valid range) = 2 * cross_corr(p, w)[j]
    size = 2 * n
    fp = np.fft.rfft(p, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(fp) * fw, size)[:n]
    dV_dp = 2.0 * corr

    # Gradient w.r.t. theta (softmax chain rule):
    # dV/dtheta[j] = p[j] * (dV/dp[j] - dot(p, dV_dp))
    inner = np.dot(p, dV_dp)
    grad_theta = 2.0 * n * p * (dV_dp - inner)

    return score, grad_theta


def parametric_init(n, style, rng):
    """Generate diverse parametric initializations (normalized to sum=1)."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    center = (n - 1) / 2.0

    if style == 'uniform':
        a = np.ones(n)
    elif style == 'cosine':
        a = np.cos(np.pi * (t - 0.5)) ** 2
    elif style == 'gaussian':
        a = np.exp(-0.5 * ((t - 0.5) / 0.2) ** 2)
    elif style == 'triangle':
        a = 1.0 - np.abs(t - 0.5) * 2.0
    elif style == 'cosine_cubed':
        a = np.cos(np.pi * (t - 0.5)) ** 6
    elif style == 'exp_taper':
        a = np.exp(-3.0 * np.abs(t - 0.5))
    elif style == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(2 * np.pi * t))
    elif style == 'bell':
        a = np.exp(-4.0 * (t - 0.5) ** 2) * (1.0 + 0.3 * np.cos(4 * np.pi * t))
    elif style == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 10)
        a[:edge] = np.linspace(0.3, 1.0, edge)
        a[-edge:] = np.linspace(1.0, 0.3, edge)
    elif style == 'random_smooth':
        a = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 5) * t + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)

    else:

    a = np.maximum(a, 1e-6)
    a = a / np.sum(a)
    return a


def spectral_init(n, rng, n_iter=40):
    """Spectral iterative initialization."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)

    for it in range(n_iter):
        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        corr = np.fft.irfft(power, 2 * n)[:n]
        alpha = 0.5 / (n + 1.0)
        a = a - alpha * corr
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def lbfgs_optimize(n, a_init, tau_schedule, time_limit, rng):
    """
    Run L-BFGS-B optimization with temperature schedule.
    a_init is normalized (sum=1).
    Returns best a found (normalized) and its score.
    """
    theta = np.log(a_init + 1e-15)
    theta -= np.mean(theta)  # center for numerical stability
    best_theta = theta.copy()
    best_score = score_fast(a_init)
    t0 = time.monotonic()

    # Phase 1: Coarse optimization with larger tau
    for phase, (tau, max_iter) in enumerate(tau_schedule):
        if time.monotonic() - t0 > time_limit * 0.85:
            break

        def obj_grad(th):
            return objective_and_grad(th, n, tau)

        result = minimize(
            obj_grad, theta, jac=True, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8,
                     'maxfun': max_iter * 3}
        )

        theta = result.x
        a = softmax_stable(theta)
        sc = 2.0 * n * np.max(fft_convolve(a, a))
        if sc < best_score:
            best_score = sc
            best_theta = theta.copy()

        print(f"        phase {phase}: tau={tau:.2e}, score={sc:.6f}, "
              f"iter={result.nit}, time={time.monotonic()-t0:.1f}s")

    # Return best
    best_a = softmax_stable(best_theta)
    return best_a, best_score


def peak_shaving(a, time_limit, rng):
    """
    Peak-directed local search.
    Targets the maximum of conv(a,a) specifically.
    Works on normalized sequence (sum=1).
    """
    a = a.copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a, score_fast(a)
    a = a / s
    n = len(a)

    current_score = 2.0 * n * np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 3000

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))

        # Method 1: Reduce elements contributing to peak
        if rng.random() < 0.5:
            # Find contributors to conv[peak_idx]
            # conv[peak_idx] = sum_i a[i] * a[peak_idx - i] for valid i
            contrib = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contrib[i] = a[i] * a[j]
                    if i != j:
                        pass  # each pair counted once through i
            # Actually conv[peak_idx] = sum_{i: 0<=i<n, 0<=peak_idx-i<n} a[i]*a[peak_idx-i]
            # Top contributors
            top_k = max(2, n // 16)
            top_idx = np.argsort(contrib)[-top_k:]

            candidate = a.copy()
            # Reduce top contributors
            reduce_frac = rng.uniform(0.02, 0.08)
            reduction = candidate[top_idx] * reduce_frac
            candidate[top_idx] -= reduction
            total_red = np.sum(reduction)

            # Redistribute to random non-top elements
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_add = min(max(3, n // 32), len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                candidate[add_idx] += total_red / n_add

            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                no_improve += 1
                continue
            candidate = candidate / cs

            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                no_improve = 0
            else:
                no_improve += 1

        # Method 2: Pair swap (move mass from high-contrib to low-contrib)
        elif rng.random() < 0.5:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.1)
            amt = a[i] * frac
            if amt > a[i]:
                continue
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                no_improve += 1
                continue
            candidate = candidate / cs

            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                no_improve = 0
            else:
                no_improve += 1

        # Method 3: Small Gaussian perturbation on a random block
        else:
            width = max(1, int(rng.integers(2, max(3, n // 10))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] += rng.normal(0, 0.01 * np.mean(a), size=width)
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                no_improve += 1
                continue
            candidate = candidate / cs

            new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
            if new_score < current_score:
                a = candidate
                current_score = new_score
                no_improve = 0
            else:
                no_improve += 1

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

    return best_a, best_score


def simulated_annealing_polish(a, time_limit, rng):
    """SA-style polish that accepts some worse moves to escape local minima."""
    a = a.copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a, score_fast(a)
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_score = current_score

    temp = 0.001 / n
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        elapsed_frac = min(1.0, (time.monotonic() - t0) / total_time)
        temp = 0.001 / n * (0.01) ** (elapsed_frac ** 1.5)

        r = rng.random()
        if r < 0.4:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.005, 0.05)
            amt = a[i] * frac
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
        elif r < 0.7:
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.03))
            candidate = a.copy()
            candidate[j] *= factor
        else:
            width = max(1, int(rng.integers(1, max(2, n // 20))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.02, size=width))

        candidate = np.clip(candidate, 0, None)
        cs = np.sum(candidate)
        if cs < 1e-10:
            continue
        candidate = candidate / cs

        new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
        delta = new_score - current_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-15)):
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()

def upsample_sequence(a, new_n):
    """Upsample sequence a to new_n using cubic interpolation."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    # Use scipy or numpy interpolation
    a_new = np.interp(x_new, x_old, a, period=1) if False else np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

                best_seq = (a_opt * np.sqrt(2.0 * n_try)).copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Final: one more polish on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_final = len(best_seq)
        a_final_norm = best_seq / np.sum(best_seq)
        final_time = min(20.0, deadline - time.monotonic() - 5)
        if final_time > 3:
            a_pol, sc_pol = peak_shaving(a_final_norm, final_time, rng)
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = (a_pol * np.sqrt(2.0 * n_final)).copy()
                print(f"    *** Final polish: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
