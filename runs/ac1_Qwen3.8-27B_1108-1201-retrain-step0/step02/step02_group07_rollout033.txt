import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_normalized(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def project_simplex(v):
    """Project vector onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    if s < 1e-15:

    best_obj = obj_normalized(a)
    
    for it in range(iterations):
        A = fft(a, 2 * n)
        power = np.abs(A) ** 2
        avg_power = power.mean()
        
        # Try to reduce high-power frequencies and boost low-power ones
        # This creates a "flattening" force
        target = avg_power
        # Compute gradient: if we increase |A[k]|^2 at frequency k,
        # it increases certain peaks in the time domain

        # But this might make a negative...
        
        # Alternative: use the spectrum to guide adjustments
        # Frequencies with high power contribute to peaks
        # We want to reduce the contribution of high-power frequencies
        
        # Compute the "excess" at each frequency
        excess = (power - avg_power) / (avg_power + 1e-15)
        
        # Modulate the spectrum
        A_new = A * np.exp(-step * excess.astype(np.float64) / 2.0)
        
        # But we need to keep the DC component (sum) fixed
        A_new[0] = A[0]

def peak_directed_search(a, time_limit, rng, n_trials_per_peak=3):

    Local search that directs modifications toward reducing convolution peaks.
    """
    a = np.maximum(a, 0).copy()

            total = products.sum()
            
            if total < 1e-15:

            # to this peak

            # Pre-compute contribution of each position to the max peak
            # For position j, its contribution to c[k] is a[j] * a[k-j]
            # We want to find j that has low contribution to peak positions
            
            # Sample a few random targets

            best_move = None
            
            for trial in range(n_trials_per_peak):
                # Choose source: weighted by contribution to this peak

                    delta = a[src] * delta_frac

                        continue

                        continue

                    a[tgt] += delta

                    new_max = new_conv.max()

                        conv = new_conv

                        a[src] += delta

def gradient_pegd(a, time_limit, t_init=50.0, lr_init=0.1):
    """
    Projected Gradient Descent with peak-weighted subgradient.
    """
    a = np.maximum(a, 0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    t = t_init
    lr = lr_init
    patience = 0
    max_pat = 30
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        # Compute smoothed subgradient
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Subgradient: sg[j] = sum_k w[k] * a[k-j] (for valid k-j)
        sg = np.zeros(n)
        for ki in range(len(conv)):
            if w[ki] < 1e-6:
                continue
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
        
        # Gradient step with projection
        a_new = a - lr * sg
        a_new = project_simplex(a_new)
        
        if a_new.sum() < 0.1:
            a_new = 0.5 * a + 0.5 * project_simplex(a)
        
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_pat:
            patience = 0
            lr *= 0.7
            t *= 1.3
            if lr < 0.003:
                break
    
    return best_a, best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    result = np.fft.irfft(A * A, fft_size)

def obj_normalized(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""

def project_simplex(v):
    """Project vector onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(conv, a, t):
    """Smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)

    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    # Only process significant weights
    thresh = w.max() * 0.001
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)

Multi-stage optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses efficient coordinate descent with best-improvement, mirror descent,
PGD, and simulated annealing. Focuses on fine-grained peak reduction
starting from the parent construction.

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def project_simplex(v):
    """Project vector onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_sg(conv, a, t):
    """Smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = t * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    thresh = w.max() * 0.001
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    return sg

def mirror_descent(a0, time_limit, t_init=50.0, eta_init=0.3):
    """Mirror descent on the simplex with adaptive parameters."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    max_pat = 50
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        sg = compute_sg(conv, a, t)
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            eta *= 0.6
            t *= 1.4
            if eta < 0.003:
                break
    return best_a, best_obj

def pgd_step(a0, time_limit, t_init=30.0, lr_init=0.08):
    """Projected gradient descent with adaptive parameters."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    lr = lr_init
    patience = 0
    max_pat = 40
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        sg = compute_sg(conv, a, t)
        a_new = a - lr * sg
        a_new = project_simplex(a_new)
        if a_new.sum() < 0.1:
            a_new = 0.5 * a + 0.5 * project_simplex(a)
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            lr *= 0.7
            t *= 1.3
            if lr < 0.002:
                break
    return best_a, best_obj

def coord_descent(a0, time_limit, rng):
    """
    Best-improvement coordinate descent via pairwise mass transfers.
    Uses subgradient to identify source/target pairs and tries multiple
    transfer sizes to find the best move.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_max = conv.max()
    best_max = cur_max
    best_a = a.copy()
    t0 = time.monotonic()
    stall = 0
    max_stall = 300
    t_sm = 50.0
    while time.monotonic() - t0 < time_limit:
        if stall > max_stall:
            break
        # Compute subgradient
        sg = compute_sg(conv, a, t_sm)
        # Source: highest subgradient
        src = int(np.argmax(sg))
        # Target: lowest subgradient (excluding source)
        sg_work = sg.copy()
        sg_work[src] = 1e10
        tgt = int(np.argmin(sg_work))
        if src == tgt:
            stall += 1
            continue
        # Try multiple transfer fractions
        best_delta = 0.0
        best_new_max = cur_max
        fracs = [0.1, 0.05, 0.2, 0.02, 0.3, 0.15]
        for frac in fracs:
            delta = a[src] * frac
            if delta < 1e-14:
                continue
            a[src] -= delta
            a[tgt] += delta
            new_conv = fast_convolve(a)
            new_max = new_conv.max()
            if new_max < best_new_max:
                best_new_max = new_max
                best_delta = delta
            a[src] += delta
            a[tgt] -= delta
        if best_delta > 0:
            a[src] -= best_delta
            a[tgt] += best_delta
            conv = fast_convolve(a)
            cur_max = conv.max()
            if cur_max < best_max - 1e-14:
                best_max = cur_max
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            # Adapt temperature
            if stall % 50 == 0:
                t_sm *= 0.9
        else:
            stall += 1
            if stall % 100 == 0:
                t_sm = max(5.0, t_sm * 0.5)
    return best_a, 2.0 * n * best_max
