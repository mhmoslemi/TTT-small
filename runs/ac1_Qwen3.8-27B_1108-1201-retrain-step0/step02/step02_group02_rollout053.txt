import numpy as np
import time
import cvxpy as cp

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def solve_lp(f, M, n):
    """
    Solve: maximize sum(g) subject to g >= 0, ||f*g||_inf <= M
    """
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    
    # Build constraint matrix A where A[k, j] = f[k-j] if 0 <= k-j < n
    constraints = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                row[j] = f[idx]
        constraints.append(row @ g <= M)
    
    prob = cp.Problem(objective, constraints)
    prob.solve()
    
    if g.value is None:
        return None
    return np.clip(g.value, 0, None)

def lp_iteration(f, n, alpha=0.1):
    """
    One LP-based iteration:
    1. Compute M = ||f*f||_inf
    2. Solve LP to find optimal g
    3. Rescale g to have same sum as f
    4. Blend: f_new = (1-alpha)*f + alpha*g

    S = np.sum(f)
    M = np.max(fft_convolve(f, f))
    
    g = solve_lp(f, M, n)

"""
LP-based peak minimization for non-negative step functions.
Uses alternating LP optimization (fix one argument, optimize the other),
combined with spectral shaping and gradient-based polishing.
"""

import numpy as np
import time
import cvxpy as cp
import scipy.optimize as so

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def normalize_sequence(a, target_sum=None):
    """Normalize sequence to have target_sum (default sqrt(2n))."""
    n = len(a)
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)
    S = np.sum(a)
    if S > 1e-10:
        a = a * (target_sum / S)
    return a

def solve_lp_min_peak(f, S, n):
    """
    Solve LP: minimize t subject to (f*g)[k] <= t for all k, g >= 0, sum(g) = S.
    Returns (g_opt, t_opt) or (None, None) on failure.
    """
    g = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    
    # Build constraint: A @ g <= t * ones
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # This is a banded matrix
    constraints = [cp.sum(g) == S]
    
    # Add convolution constraints
    # For efficiency, build the matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    
    constraints.append(A @ g <= t * np.ones(2*n-1))
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iter=5000)
    except:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except:
            return None, None
    
    if g.value is None:
        return None, None
    
    g_val = np.clip(g.value, 0, None)
    t_val = t.value if t.value is not None else np.max(fft_convolve(f, g_val))
    return g_val, t_val

def lp_optimize(a0, time_limit, rng, verbose=True):
    """
    Main LP-based optimization loop.
    """
    n = len(a0)
    a = normalize_sequence(a0.astype(np.float64).copy())
    S = np.sum(a)
    
    best_a = a.copy()
    best_score = local_score(a)
    
    t0 = time.monotonic()
    alpha = 0.5
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:

        g, t_opt = solve_lp_min_peak(a, S, n)
        
        if g is None:

            a = np.clip(a + 0.01 * rng.standard_normal(n), 0.01, None)
            a = normalize_sequence(a)
            no_improve += 1
            iteration += 1
            continue

            iteration += 1

                best_score = score_try

                no_improve = 0

                a = a_try
                no_improve += 1

        if not improved and no_improve > 0:

        else:

def spectral_polish(a0, time_limit, rng, verbose=False):
    """
    Spectral shaping: flatten the power spectrum to reduce peak.
    """
    n = len(a0)
    a = normalize_sequence(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = local_score(a)
    
    t0 = time.monotonic()
    step = 0
    lr = 0.1
    temp = 0.01
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        
        # Compute power spectrum
        A = np.fft.rfft(a, 2*n)  # Use zero-padded FFT for linear conv
        power = np.abs(A)**2
        
        # Target: flat power spectrum
        target_power = np.mean(power) * np.ones_like(power)
        
        # Gradient: push power toward target
        # Modify the phase/amplitude in frequency domain
        # Simple approach: adjust amplitude
        # a_new = ifft(sqrt(lerp(power, target, lr)) * exp(i*phase))
        phase = np.angle(A)
        new_amp = np.sqrt((1-lr) * power + lr * target_power)
        new_A = new_amp * np.exp(1j * phase)
        a_new = np.fft.irfft(new_A, 2*n)[:n]
        
        # Project to non-negative
        a_new = np.clip(a_new, 0, None)
        a_new = normalize_sequence(a_new)
        
        score_new = local_score(a_new)
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            lr = min(0.5, lr * 1.05)
        else:
            # Accept with probability (simulated annealing)
            if rng.random() < np.exp(-(score_new - best_score) / max(temp, 1e-10)):
                a = a_new
            lr *= 0.95
        
        temp *= 0.9999
        
        if verbose and step % 500 == 0:
            print(f"  Spectral step {step}: best={best_score:.6f}, lr={lr:.4f}")
    
    return best_a, best_score


def gradient_polish(a0, time_limit, rng, verbose=False):
    """
    Gradient-based polishing using soft-max of the peak.
    """
    n = len(a0)
    theta = np.log(a0 + 1e-10)
    S_target = np.sqrt(2.0 * n)
    
    best_a = a0.copy()
    best_score = local_score(a0)
    
    t0 = time.monotonic()
    step = 0
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        elapsed_frac = (time.monotonic() - t0) / time_limit
        
        # Soft-max temperature
        tau = (50.0 / n) * (1.0 / n) ** (elapsed_frac)
        
        a = softmax_normalize(theta) * S_target
        
        # Compute soft-max gradient
        c = fft_convolve(a, a)
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        sum_exp = np.sum(exp_shifted)
        w = exp_shifted / sum_exp
        
        # Gradient
        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g_a = 2.0 * corr[n-1:2*n-1]
        
        # Convert to theta-space
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S_target)
        
        # Adam update
        m_buf = beta1 * m_buf + (1-beta1) * g_theta
        v_buf = beta2 * v_buf + (1-beta2) * g_theta**2
        m_hat = m_buf / (1 - beta1**step)
        v_hat = v_buf / (1 - beta2**step)
        
        lr = 1.0 / np.sqrt(step + 1)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Check score
        if step % 100 == 0:
            a_check = softmax_normalize(theta) * S_target
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_a = a_check.copy()
    
    return best_a, best_score


def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def normalize_sequence(a, target_sum=None):
    """Normalize sequence to have target_sum (default sqrt(2n))."""
    n = len(a)
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)
    S = np.sum(a)
    if S > 1e-10:
        a = a * (target_sum / S)
    return a


def build_conv_matrix(f, n):
    """Build matrix A where A[k,j] = f[k-j] if 0<=k-j<n, else 0."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def solve_lp_min_peak(f, S, n):
    """
    Solve LP: minimize t subject to (f*g)[k] <= t for all k, g >= 0, sum(g) = S.
    Returns (g_opt, t_opt) or (None, None) on failure.
    """
    g = cp.Variable(n, nonneg=True)
    t = cp.Variable()

    A = build_conv_matrix(f, n)

    constraints = [
        cp.sum(g) == S,
        A @ g <= t * np.ones(2 * n - 1)
    ]

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iter=10000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=10000)
        except Exception:
            return None, None

    if g.value is None:
        return None, None

    g_val = np.clip(g.value, 0, None)
    # Ensure sum constraint is met numerically
    gs = np.sum(g_val)
    if gs > 1e-10:
        g_val = g_val * (S / gs)
    t_val = t.value if t.value is not None else np.max(fft_convolve(f, g_val))
    return g_val, t_val


def greedy_peak_shave(a, n_shaves, rng):
    """
    Fast greedy peak reduction: identify peak contributors and redistribute mass.
    """
    a = a.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)

    for _ in range(n_shaves):
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]

        # Find contributors to peak
        contribs = np.zeros(n)
        for i in range(n):
            idx = peak_idx - i
            if 0 <= idx < n:
                contribs[i] = a[i] * a[idx]

        # Find top contributor pair
        top_i = int(np.argmax(contribs))
        partner = peak_idx - top_i
        if not (0 <= partner < n):
            partner = peak_idx + top_i - 2 * (n - 1) if peak_idx >= n else peak_idx + top_i

        # Reduce the larger element in the pair
        if a[top_i] >= a.get(partner, 0) if 0 <= partner < n else True:
            reduce_idx = top_i
        else:
            reduce_idx = partner if 0 <= partner < n else top_i

        # Amount to reduce
        amt = a[reduce_idx] * rng.uniform(0.01, 0.04)
        a[reduce_idx] -= amt

        # Redistribute to a low-contributing element
        low_idx = int(np.argmin(contribs))
        a[low_idx] += amt

        a = np.clip(a, 0.0, None)
        # Renormalize
        a = a * (S / max(np.sum(a), 1e-10))

    return a

def gradient_polish(a0, time_limit, rng, verbose=False):
    """Soft-max gradient descent in log-space for fine polishing."""
    n = len(a0)
    theta = np.log(np.maximum(a0, 1e-10))
    S_target = np.sqrt(2.0 * n)

    best_a = a0.astype(np.float64).copy()

    t0 = time.monotonic()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        frac = (time.monotonic() - t0) / time_limit

        # Temperature schedule
        tau = (30.0 / n) * (0.01 / n) ** (frac ** 0.5)

        a = softmax_normalize(theta) * S_target

        c = fft_convolve(a, a)
        c_max = np.max(c)
        shifted = (c - c_max) / max(tau, 1e-10)
        exp_s = np.exp(shifted)
        sum_exp = np.sum(exp_s)
        w = exp_s / sum_exp

        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g_a = 2.0 * corr[n - 1:2 * n - 1]

        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S_target)

        m_buf = b1 * m_buf + (1 - b1) * g_theta
        v_buf = b2 * v_buf + (1 - b2) * g_theta ** 2
        m_hat = m_buf / (1.0 - b1 ** step)
        v_hat = v_buf / (1.0 - b2 ** step)

        lr = 1.5 / np.sqrt(step + 1) * (1.0 - frac * 0.5)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if step % 200 == 0:

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s

            a += amp2 * np.sin(2 * np.pi * freq * t / n + phase + 0.5)
        a += 0.01 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'chirp':
        a = np.ones(n)
        f0 = rng.uniform(1, 5)
        bw = rng.uniform(5, 20)
        a += 0.25 * np.cos(2 * np.pi * (f0 * t / n + bw * (t / n) ** 2))
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'two_scale':
        # Two-scale: large-scale smooth + small-scale rough
        a = np.ones(n)
        a += 0.3 * np.cos(2 * np.pi * rng.uniform(2, 5) * t / n)
        a += 0.15 * np.cos(2 * np.pi * rng.uniform(10, 30) * t / n + rng.uniform(0, 6))
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'noise_floor':
        # Uniform + structured noise
        a = np.ones(n)
        noise = np.cumsum(rng.standard_normal(n))
        noise = (noise - np.mean(noise)) / (np.std(noise) + 1e-10)
        a += 0.15 * noise
        return np.clip(a, 0.01, None)

    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization: LP-based peak minimization (primary),
    greedy shaving, spectral reshaping, and gradient polishing.
    Explores multiple initializations and sequence lengths.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            if len(start_seq) > 0:
                s = local_score(start_seq)
                print(f"Provided: n={len(start_seq)}, score={s:.6f}")
                if s < best_score:
                    best_score = s
                    best_seq = start_seq.copy()
        except Exception as e:
            print(f"Warning: could not load provided sequence: {e}")

    n_main = len(best_seq) if best_seq is not None else 256

    # === PHASE 1: LP optimization on main sequence (45% time) ===
    print(f"\n{'=' * 50}")
    print(f"PHASE 1: LP optimization (n={n_main})")
    print(f"{'=' * 50}")

    t_p1 = 0.45 * time_budget
    a_start = best_seq.copy() if best_seq is not None else make_good_init(n_main, rng)
    a_start = normalize_sequence(a_start)

    a_best, score1 = lp_optimize(a_start, t_p1, rng, verbose=True)
    print(f"  Phase 1 result: score={score1:.6f}")
    if score1 < best_score:
        best_score = score1
        best_seq = a_best.copy()
        print(f"  *** New best: {best_score:.6f} ***")

    # === PHASE 2: Fresh LP from different initializations (15% time) ===
    if time.monotonic() < deadline - 45:
        print(f"\n{'=' * 50}")
        print(f"PHASE 2: Fresh LP explorations")
        print(f"{'=' * 50}")

        methods = ['broadband', 'chirp', 'two_scale', 'noise_floor']
        t_per = max(5.0, (0.15 * time_budget) / len(methods))

        for m in methods:
            if time.monotonic() > deadline - 30:
                break
            t_avail = min(t_per, deadline - time.monotonic() - 20)
            if t_avail < 5:
                break

            a_fresh = make_good_init(n_main, rng, m)
            a_fresh = normalize_sequence(a_fresh)
            s_init = local_score(a_fresh)
            print(f"  Init '{m}': initial score={s_init:.6f}")

            a_try, s_try = lp_optimize(a_fresh, t_avail, rng, verbose=False)
            print(f"  Init '{m}': final score={s_try:.6f}")

            if s_try < best_score:
                best_score = s_try
                best_seq = a_try.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === PHASE 3: Spectral reshaping (12% time) ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n{'=' * 50}")
        print(f"PHASE 3: Spectral reshaping")
        print(f"{'=' * 50}")

        t_p3 = min(0.12 * time_budget, deadline - time.monotonic() - 20)
        if t_p3 > 10:
            a_spec, score3 = spectral_reshape(best_seq, t_p3, rng, verbose=True)
            print(f"  Phase 3 result: score={score3:.6f}")
            if score3 < best_score:
                best_score = score3
                best_seq = a_spec.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === PHASE 4: Gradient polish (10% time) ===
    if best_seq is not None and time.monotonic() < deadline - 25:
        print(f"\n{'=' * 50}")
        print(f"PHASE 4: Gradient polish")
        print(f"{'=' * 50}")

        t_p4 = min(0.10 * time_budget, deadline - time.monotonic() - 15)
        if t_p4 > 10:
            a_grad, score4 = gradient_polish(best_seq, t_p4, rng, verbose=True)
            print(f"  Phase 4 result: score={score4:.6f}")
            if score4 < best_score:
                best_score = score4
                best_seq = a_grad.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === PHASE 5: Try other sequence lengths (10% time) ===
    if time.monotonic() < deadline - 40:
        print(f"\n{'=' * 50}")
        print(f"PHASE 5: Other lengths")
        print(f"{'=' * 50}")

        candidates = [n_main // 2, n_main * 3 // 2, n_main + 64, n_main - 64]
        candidates = [c for c in candidates if 64 <= c <= 1024]
        # Add some standard sizes
        for alt in [192, 320, 384, 448, 512]:
            if alt not in candidates:
                candidates.append(alt)

        t_per_alt = max(8.0, (0.10 * time_budget) / max(len(candidates), 1))

        for n_alt in sorted(candidates):
            if time.monotonic() > deadline - 30:
                break
            t_avail = min(t_per_alt, deadline - time.monotonic() - 20)
            if t_avail < 8:
                break

            if best_seq is not None:
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, n_alt)
                a0_alt = np.interp(x_new, x_old, best_seq)
            else:
                a0_alt = make_good_init(n_alt, rng)

            a0_alt = normalize_sequence(a0_alt)
            a_try, s_try = lp_optimize(a0_alt, t_avail, rng, verbose=False)
            print(f"  n={n_alt}: score={s_try:.6f}")

            if s_try < best_score:
                best_score = s_try
                best_seq = a_try.copy()
                print(f"  *** New best: {best_score:.6f} (n={n_alt}) ***")

    # === FINAL: Polish with remaining time ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_final = deadline - time.monotonic() - 10
        if t_final > 15:
            print(f"\n{'=' * 50}")
            print(f"FINAL POLISH (t={t_final:.0f}s, n={len(best_seq)})")
            print(f"{'=' * 50}")
            a_final, score_final = lp_optimize(best_seq, t_final, rng, verbose=False)
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"  Final: {best_score:.6f}")

    # Verify and return
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            final_eval = evaluate_sequence(final_list)
        except Exception:
            final_eval = best_score
        print(f"\n{'=' * 50}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'=' * 50}")
        return final_list

    # Fallback
    print("Warning: using fallback uniform sequence")
    return [1.0] * 256
