def fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = fft(a, size)
    c = np.real(ifft(A * A))

def seg_conv_max(bounds, heights):
    """Exact max of self-convolution for k-segment sequence."""
    k = len(heights)
