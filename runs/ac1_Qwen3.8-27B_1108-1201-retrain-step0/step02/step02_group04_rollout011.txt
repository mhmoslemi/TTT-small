   ```python
   import numpy as np
   import time
   from scipy.optimize import linprog

   def fft_convolve(a, b):
       n = len(a)
       m = len(b)
       size = 1
       while size < n + m - 1:
           size <<= 1
       fa = np.fft.rfft(a, size)
       fb = np.fft.rfft(b, size)
       return np.fft.irfft(fa * fb, size)[:n + m - 1]

   def score_fast(a):
       a = np.clip(a, 0, 1000)
       n = len(a)
       s = np.sum(a)
       if s < 0.01:
           return np.inf
       c = fft_convolve(a, a)
       return 2.0 * n * np.max(c) / (s ** 2)

   def solve_lp(f, M):
       """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0"""
       n = len(f)
       # conv(f,g)[k] = sum_{i: valid} f[i] * g[k-i] <= M
       # Variables: g[0], ..., g[n-1]
       # Constraints: for k = 0, ..., 2n-2: sum_{i: 0<=i<n, 0<=k-i<n} f[i] * g[k-i] <= M
       
       # Build constraint matrix A (each row is one constraint)
       # For constraint k, the coefficient of g[j] is f[k-j] if 0 <= k-j < n, else 0
       
       A_ub = np.zeros((2*n-1, n))
       for k in range(2*n-1):
           for j in range(n):
               idx = k - j
               if 0 <= idx < n:
                   A_ub[k, j] = f[idx]
       
       b_ub = np.full(2*n-1, M)
       c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
       
       res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
       if res.success:
           return res.x
       return None

   def lp_iteration(f, n_iter=100, t=0.1):
       """Run LP-based iterations"""
       current = f.copy()
       best = f.copy()
       best_score = score_fast(best)
       
       for it in range(n_iter):
           c = fft_convolve(current, current)
           M = np.max(c)
           
           g = solve_lp(current, M)
           if g is None:
               break
           
           sum_g = np.sum(g)
           if sum_g <= 0:
               break
           
           # Normalize g to have same sum as current
           g_norm = g * np.sum(current) / sum_g
           
           # Update
           new_f = (1-t) * current + t * g_norm
           new_f = np.clip(new_f, 0, 1000)
           
           new_score = score_fast(new_f)
           if new_score < best_score:
               best_score = new_score
               best = new_f.copy()
           
           current = new_f
       
       return best, best_score
   ```

```python
import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT for speed."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):

    a = np.clip(a.astype(np.float64), 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s ** 2))


def build_lp_constraints(f, n):

    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            idxs = np.arange(j_start, j_end + 1)
            A[k, idxs] = f[k - idxs]
    return A

    A_eq[0, :n] = 1.0

        c_shifted = c - c_max
        weights = np.exp(alpha * c_shifted)
        weights /= np.sum(weights)

                idxs = np.arange(j_start, j_end + 1)
                grad[idxs] += w * 2.0 * current[k - idxs]

        score = 2.0 * n * c_max / (sum_a ** 2)
        grad_scale = 2.0 * n / (sum_a ** 2)
        grad = grad * grad_scale - (4.0 * n * c_max / (sum_a ** 3))  # gradient of score

        new_f = np.clip(new_f, 0.0, 1000.0)

def random_init(n, rng, style='uniform'):
    """Generate a random initial sequence."""
    if style == 'uniform':

    elif style == 'exponential':
        a = rng.exponential(1.0, n)
    elif style == 'sine':
        t = np.arange(n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * rng.integers(1, 5) * t / n)
        a = np.abs(a) + 0.1

        a = np.abs(a) + 0.05
    else:
        a = np.ones(n)
    return np.clip(a, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Find the optimal non-negative sequence minimizing the evaluation score.
    Uses LP-based iterative optimization with multiple restarts and
    gradient refinement.
    """
    t_global = time.monotonic()
    deadline = t_global + max(0, int(budget_s) - 20)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def update_best(seq, score):
        nonlocal best_score, best_seq
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  NEW BEST: score={score:.8f}, n={len(seq)}", flush=True)

    # Check provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            s = score_fast(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.8f}", flush=True)
            update_best(start_seq, s)
        except Exception:
            pass

    # Define optimization runs to try
    runs = []

    # Primary: optimize the provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            runs.append(('provided_lp_minpeak', np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0), 'min_peak'))
            runs.append(('provided_lp_maxsum', np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0), 'max_sum'))
        except Exception:
            pass

    # Random starts with n=256
    n_main = 256
    for i, style in enumerate(['noise', 'exponential', 'sine', 'two_bump', 'cosine_weighted', 'uniform']):
        a = random_init(n_main, rng, style)
        runs.append((f"rand256_{style}", a, 'min_peak'))

    # Try n=512
    for style in ['noise', 'exponential', 'sine']:
        a = random_init(512, rng, style)
        runs.append((f"rand512_{style}", a, 'min_peak'))

    # Try n=128
    for style in ['noise', 'two_bump']:
        a = random_init(128, rng, style)
        runs.append((f"rand128_{style}", a, 'min_peak'))

    # Run optimizations with time allocation
    time_per_run = 60.0  # base time per run
    print(f"Total runs planned: {len(runs)}", flush=True)

    for run_idx, (name, init_seq, mode) in enumerate(runs):
        if time.monotonic() > deadline:
            break

        time_left = deadline - time.monotonic()
        if time_left < 15:
            break

        # Allocate time: more for earlier/promising runs
        if run_idx < 3:
            alloc = min(time_per_run * 2, time_left * 0.4)
        else:
            alloc = min(time_per_run, time_left * 0.2)
        alloc = max(alloc, 10.0)

        print(f"\nRun {run_idx}: {name}, n={len(init_seq)}, mode={mode}, alloc={alloc:.0f}s", flush=True)

        try:
            # LP optimization
            seq, sc = lp_optimize(init_seq, alloc * 0.6, mode=mode, verbose=False)
            update_best(seq, sc)

            # Gradient refinement on the LP result
            time_left2 = deadline - time.monotonic()
            if time_left2 > 20 and sc < best_score + 0.1:
                alloc2 = min(40.0, time_left2 * 0.3)
                seq2, sc2 = gradient_refine(seq, alloc2)
                update_best(seq2, sc2)

            # Also try gradient refine directly on the LP output with different alpha
            time_left3 = deadline - time.monotonic()
            if time_left3 > 20 and sc < best_score + 0.1:
                alloc3 = min(30.0, time_left3 * 0.3)
                seq3, sc3 = gradient_refine(seq, alloc3, alpha_init=200.0)
                update_best(seq3, sc3)

        except Exception as e:
            print(f"  Error in run {name}: {e}", flush=True)
            continue

    # If we still have time, do more focused optimization on the best found
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Focused optimization on best (score={best_score:.8f}) ===", flush=True)
        time_left = deadline - time.monotonic() - 30

        # Try LP with min_peak on the best
        alloc = min(120.0, time_left * 0.4)
        seq, sc = lp_optimize(best_seq, alloc, mode='min_peak')
        update_best(seq, sc)

        # Try LP with max_sum on the best
        if time.monotonic() < deadline - 30:
            alloc = min(120.0, (deadline - time.monotonic() - 30) * 0.4)
            seq, sc = lp_optimize(best_seq, alloc, mode='max_sum')
            update_best(seq, sc)

        # Gradient refinement with various alphas
        if time.monotonic() < deadline - 30:
            for alpha in [50, 200, 500]:
                if time.monotonic() > deadline - 30:
                    break
                alloc = min(60.0, (deadline - time.monotonic() - 30) * 0.3)
                seq, sc = gradient_refine(best_seq, alloc, alpha_init=alpha)
                update_best(seq, sc)

        # Final LP pass
        if time.monotonic() < deadline - 30:
            alloc = min(90.0, deadline - time.monotonic() - 30)
            seq, sc = lp_optimize(best_seq, alloc, mode='min_peak')
            update_best(seq, sc)

    # Final: try a few more random restarts if time allows
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        for i in range(5):
            if time.monotonic() > deadline - 30:
                break
            # Perturb the best sequence
            noise = 0.01 * rng.standard_normal(n_best)
            perturbed = np.clip(best_seq + noise, 0.0, 1000.0)
            alloc = min(30.0, (deadline - time.monotonic() - 30) * 0.5)
            seq, sc = lp_optimize(perturbed, alloc, mode='min_peak')
            update_best(seq, sc)

    # Final answer
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = 2.0

    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    final_score = evaluate_sequence(final_list)
    print(f"\n=== FINAL: score={final_score:.8f}, n={len(final_list)} ===", flush=True)

    if not np.isfinite(final_score):
        # Fallback to uniform
        final_list = [1.0] * 256
        final_score = evaluate_sequence(final_list)
        print(f"Fallback uniform: score={final_score:.8f}", flush=True)

    return final_list
```
