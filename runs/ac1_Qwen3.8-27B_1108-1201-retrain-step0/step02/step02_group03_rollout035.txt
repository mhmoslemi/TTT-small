peak-directed local search. Exploits scale invariance by normalizing sum=1.
"""

import numpy as np
import time


def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_of(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    b = fft_convolve(a, a)
    return 2.0 * n * np.max(b) / (s * s)


def top_k_gradient(a, k):
    """
    Compute the average gradient of the top-k autocorrelation lags.
    Returns (gradient, b_array, top_lags, max_val).
    """
    n = len(a)
    b = fft_convolve(a, a)
    # Get top-k lag indices
    k_eff = min(k, len(b))
    top_idx = np.argsort(b)[-k_eff:]
    max_val = b[np.argmax(b)]
    
    g = np.zeros(n)
    for lag in top_idx:
        # d conv(a,a)[lag] / d a[i] = a[lag-i] (if valid) + a[i-lag] (if valid)
        # a[lag-i] valid when 0 <= lag-i < n => lag-n+1 <= i <= lag
        lo1 = max(0, lag - n + 1)
        hi1 = min(n - 1, lag)
        if lo1 <= hi1:
            g[lo1:hi1+1] += a[lag - np.arange(lo1, hi1+1)]
        
        # a[i-lag] valid when 0 <= i-lag < n => lag <= i <= lag+n-1
        lo2 = max(0, lag)
        hi2 = min(n - 1, lag + n - 1)
        if lo2 <= hi2:
            g[lo2:hi2+1] += a[np.arange(lo2, hi2+1) - lag]
    
    return g, b, top_idx, max_val


def mirror_descent(a_init, time_limit, rng, k_start=64, k_end=2, eta_init=0.5):
    """
    Mirror descent with top-k subgradient and backtracking line search.
    Operates on a normalized to sum=1.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_score = score_of(a)
    
    eta = eta_init
    t0 = time.monotonic()
    step = 0
    stall_count = 0
    stall_limit = 200
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        
        # Anneal k
        k = max(k_end, int(k_start * (1 - frac) + k_end * frac))
        
        # Compute gradient
        g, b, top_lags, max_val = top_k_gradient(a, k)
        
        # Project gradient (remove mean to maintain sum=1)
        g -= np.mean(g)
        
        # Mirror descent update: a *= exp(-eta * g)
        # Use line search if needed
        improved = False
        for ls_iter in range(10):
            a_new = a * np.exp(-eta * g)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)
            
            s_new = score_of(a_new)
            if s_new < best_score * (1 + 1e-9):  # Allow tiny improvements
                a = a_new
                if s_new < best_score:
                    best_score = s_new
                    best_a = a.copy()
                    stall_count = 0
                improved = True
                eta = min(eta * 1.05, 2.0)  # Slightly increase step
                break
            else:
                eta *= 0.5
        
        if not improved:
            stall_count += 1
            # If stuck, try a random perturbation
            if stall_count > stall_limit:
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                eta = eta_init * 0.5
                stall_count = 0
        
        step += 1
        if step % 500 == 0:
            print(f"      MD step={step}, k={k}, eta={eta:.4f}, score={best_score:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_score


def local_search_peaked(a_init, time_limit, rng, n_top=5):
    """
    Local search focused on reducing the top convolution peaks.
    Uses exponential multiplicative updates.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    current_score = score_of(a)
    best = a.copy()
    best_score = current_score
    
    temp = 0.003
    cooling = 0.9997
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fft_convolve(a, a)
        top_lags = np.argsort(b)[-n_top:]
        
        # Compute contribution to each position from top lags
        contrib = np.zeros(n)
        for lag in top_lags:
            lo1 = max(0, lag - n + 1)
            hi1 = min(n - 1, lag)
            if lo1 <= hi1:
                contrib[lo1:hi1+1] += a[lag - np.arange(lo1, hi1+1)] * a[np.arange(lo1, hi1+1)]
            lo2 = max(0, lag)
            hi2 = min(n - 1, lag + n - 1)
            if lo2 <= hi2:
                contrib[lo2:hi2+1] += a[np.arange(lo2, hi2+1)] * a[lag - np.arange(lo2, hi2+1)]
        
        candidate = a.copy()
        r = rng.random()
        it += 1
        
        if r < 0.4:
            # Reduce top contributors, increase bottom
            n_change = max(2, n // 12)
            top_idx = np.argsort(contrib)[-n_change:]
            bot_idx = np.argsort(contrib)[:n_change]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.02, size=len(bot_idx)))
        elif r < 0.6:
            # Block swap
            width = max(1, n // 32)
            start = rng.integers(0, n - 2 * width)
            candidate[start:start+width] *= np.exp(rng.normal(-0.06, 0.02, width))
            candidate[start+width:start+2*width] *= np.exp(rng.normal(0.06, 0.02, width))
        elif r < 0.75:
            # Multiplicative noise
            candidate *= np.exp(rng.normal(0, 0.02, n))
        elif r < 0.9:
            # Targeted: reduce specific positions that appear in peak lags
            n_reduce = max(3, n // 8)
            peak_positions = np.argsort(contrib)[-n_reduce:]
            candidate[peak_positions] *= np.exp(rng.normal(-0.1, 0.03, size=n_reduce))
            # Compensate elsewhere
            other = np.setdiff1d(np.arange(n), peak_positions)
            n_comp = min(n_reduce, len(other))
            comp_idx = rng.choice(other, size=n_comp, replace=False)
            candidate[comp_idx] *= np.exp(rng.normal(0.1, 0.03, size=n_comp))
        else:
            # Symmetrization nudge
            sym = 0.5 * (a + a[::-1])
            candidate = a + 0.1 * (sym - a)
            candidate *= np.exp(rng.normal(0, 0.01, n))
        
        candidate = np.maximum(candidate, 1e-15)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = a.copy()
        else:
            candidate = candidate / s
        
        new_score = score_of(candidate)
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 5:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    a = candidate
                    current_score = new_score
        
        temp *= cooling
    
    return best, best_score


def fft_phase_init(n, rng, n_iter=80):
    """Gerchberg-Saxton style initialization."""
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft <<= 1
    
    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def chirp_init(n, rng, alpha=0.5):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def multi_tone_init(n, rng, n_tones=4):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(n_tones):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.4 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def upsample(a, new_n, rng, noise_scale=0.02):
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += noise_scale * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)

"""
Optimized step function search using top-k subgradient mirror descent with
adaptive step sizing, soft-top-k gradient blending, and targeted peak-directed
local search. Exploits scale invariance by normalizing to sum=1 throughout.
"""

import numpy as np
import time


def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_of(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    b = fft_convolve(a, a)
    return 2.0 * n * np.max(b) / (s * s)


def compute_gradient_soft_topk(a, k, tau):
    """
    Compute gradient using soft weighting over top-k lags.
    Weight for lag l: exp(b[l]/tau) restricted to top-k, normalized.
    Gradient of b[l] w.r.t. a[j] = 2*a[l-j] if 0<=l-j<n else 0.
    """
    n = len(a)
    b = fft_convolve(a, a)
    k_eff = min(k, len(b))
    # Get top-k indices
    top_idx = np.argsort(b)[-k_eff:]
    top_vals = b[top_idx]
    # Soft weights within top-k
    w = np.exp(top_vals / tau)
    w = w / np.sum(w)
    max_val = np.max(b)

    g = np.zeros(n)
    for i, lag in enumerate(top_idx):
        wi = 2.0 * w[i]
        if wi < 1e-12:
            continue
        # g[j] += wi * a[lag-j] if 0 <= lag-j < n
        lo = max(0, lag - n + 1)
        hi = min(n - 1, lag)
        if lo <= hi:
            g[lo:hi + 1] += wi * a[lag - np.arange(lo, hi + 1)]

    return g, b, max_val


def mirror_descent_optimize(a_init, time_limit, rng, k_start=48, k_end=3,
                            tau_start=None, tau_end=None, eta_init=1.0):
    """
    Nesterov-accelerated mirror descent with soft-top-k gradient.
    Normalizes to sum=1. Uses backtracking line search.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / np.sum(a)

    if tau_start is None:
        tau_start = n / 8.0
    if tau_end is None:
        tau_end = n / 200.0

    best_a = a.copy()
    best_score = score_of(a)

    eta = eta_init
    t0 = time.monotonic()
    step = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Anneal k and tau
        k = max(k_end, int(k_start * (1 - frac ** 0.5) + k_end * frac ** 0.5))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)

        # Compute gradient
        g, b, max_val = compute_gradient_soft_topk(a, k, tau)
        # Project: remove mean to preserve sum=1
        g -= np.mean(g)

        # Backtracking line search
        improved = False
        eta_try = eta
        for ls in range(8):
            a_new = a * np.exp(-eta_try * g)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)
            s_new = score_of(a_new)
            if s_new < best_score:
                a = a_new
                best_score = s_new
                best_a = a.copy()
                stall = 0
                eta = min(eta * 1.02, 3.0)
                improved = True
                break
            elif s_new < score_of(a):
                # Accept even if not new best (progress)
                a = a_new
                eta = min(eta * 1.01, 3.0)
                improved = True
                break
            else:
                eta_try *= 0.5

        if not improved:
            stall += 1
            eta = max(eta * 0.5, 0.01)
            if stall > 150:
                # Kick: add noise
                a = a * np.exp(rng.normal(0, 0.15, n))
                a = np.maximum(a, 1e-12)
                a = a / np.sum(a)
                eta = eta_init * 0.5
                stall = 0

        step += 1
        if step % 800 == 0:
            print(f"      MD step={step}, k={k}, tau={tau:.1f}, eta={eta:.3f}, "
                  f"best={best_score:.6f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def local_search_focused(a_init, time_limit, rng, n_top=6):
    """
    Intensive local search targeting the top convolution peak positions.
    Identifies which elements contribute most to peak lags and redistributes.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / np.sum(a)
    current_score = score_of(a)
    best = a.copy()
    best_score = current_score

    temp = 0.002
    cooling = 0.9998
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        b = fft_convolve(a, a)
        top_lags = np.argsort(b)[-n_top:]

        # Compute per-position contribution to top lags
        contrib = np.zeros(n)
        for lag in top_lags:
            lo = max(0, lag - n + 1)
            hi = min(n - 1, lag)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                contrib[idx] += a[idx] * a[lag - idx]

        candidate = a.copy()
        r = rng.random()
        it += 1

        if r < 0.35:
            # Reduce top contributors to peaks, boost low contributors
            n_ch = max(2, n // 10)
            top_pos = np.argsort(contrib)[-n_ch:]
            bot_pos = np.argsort(contrib)[:n_ch]
            candidate[top_pos] *= np.exp(rng.normal(-0.04, 0.015, n_ch))
            candidate[bot_pos] *= np.exp(rng.normal(0.04, 0.015, n_ch))
        elif r < 0.50:
            # Adjacent block swap
            w = max(1, n // 40)
            s = rng.integers(0, max(1, n - 2 * w))
            candidate[s:s + w] *= np.exp(rng.normal(-0.05, 0.02, w))
            candidate[s + w:s + 2 * w] *= np.exp(rng.normal(0.05, 0.02, w))
        elif r < 0.65:
            # Multi-point perturbation at peak positions
            n_pts = max(3, n // 6)
            pts = rng.choice(n, size=n_pts, replace=False)
            candidate[pts] *= np.exp(rng.normal(0, 0.04, n_pts))
        elif r < 0.78:
            # Targeted: strongly reduce worst offenders
            n_worst = max(2, n // 15)
            worst = np.argsort(contrib)[-n_worst:]
            candidate[worst] *= np.exp(rng.normal(-0.08, 0.03, n_worst))
            rest = np.setdiff1d(np.arange(n), worst)
            n_comp = min(n_worst, len(rest))
            if n_comp > 0:
                comp = rng.choice(rest, size=n_comp, replace=False)
                candidate[comp] *= np.exp(rng.normal(0.08, 0.03, n_comp))
        elif r < 0.90:
            # Symmetry nudge + noise
            sym = 0.5 * (a + a[::-1])
            candidate = a + 0.08 * (sym - a)
            candidate *= np.exp(rng.normal(0, 0.01, n))
        else:
            # Global small noise
            candidate *= np.exp(rng.normal(0, 0.012, n))

        candidate = np.maximum(candidate, 1e-15)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = a.copy()
        else:
            candidate = candidate / s

        new_score = score_of(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 8:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    a = candidate
                    current_score = new_score

        temp *= cooling

    return best, best_score


def fft_phase_init(n, rng, n_iter=100):
    """Gerchberg-Saxton initialization for flat spectrum."""
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft <<= 1
    a = np.ones(n) / n
    for _ in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-14] = 1e-14
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def chirp_init(n, rng, alpha=None):
    if alpha is None:
        alpha = 0.3 + 0.5 * rng.random()
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def multi_tone_init(n, rng, n_tones=5):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(n_tones):
        freq = rng.integers(1, max(2, n // 2))
        amp = 0.5 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def random_init(n, rng, spread=0.35):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def upsample(a, new_n, rng, noise=0.02):
    n = len(a)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_i = np.interp(x_new, x_old, a)
    a_i += noise * rng.standard_normal(new_n)
    a_i = np.clip(a_i, 0.001, None)
    return a_i / np.sum(a_i)

        if time.monotonic() > deadline - 180:
            break
        inits = []
        a = fft_phase_init(n, rng, n_iter=100)
        inits.append(('fft', a))
        a = chirp_init(n, rng)
        inits.append(('chirp', a))
        a = multi_tone_init(n, rng, n_tones=5)
        inits.append(('mt', a))
        a = random_init(n, rng, spread=0.4)
        inits.append(('rand', a))

        for name, a in inits:
            s = score_of(a)
            print(f"  n={n} {name}: {s:.6f}")
            candidates.append((s, n, a.copy(), name))

    if start_seq is not None:
        candidates.append((score_of(start_seq), len(start_seq), start_seq.copy(), 'given'))

    candidates.sort(key=lambda x: x[0])
    print(f"  Top 4: " + ", ".join(f"{c[3]}(n={c[1]}):{c[0]:.4f}" for c in candidates[:4]))

    # Phase 2: Mirror descent on top candidates
    print("\n=== Phase 2: Mirror Descent ===")
    n_cand = min(5, len(candidates))

    for i in range(n_cand):
        if time.monotonic() > deadline - 90:
            break
        s0, n, a0, name = candidates[i]
        remaining = deadline - time.monotonic() - 45
        t_alloc = remaining / (n_cand - i)
        # Scale down for larger n (slower per step)
        t_alloc *= min(1.0, 384.0 / n)
        t_alloc = max(15.0, min(t_alloc, 200.0))
        print(f"  [{i+1}/{n_cand}] n={n} {name} s0={s0:.6f} t={t_alloc:.0f}s")

        a_opt, s_opt = mirror_descent_optimize(a0, t_alloc, rng,
                                                k_start=48, k_end=3,
                                                eta_init=0.8)
        print(f"    MD -> {s_opt:.6f}")

        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 3: Focused local search on best
    print("\n=== Phase 3: Local Search Polish ===")
    if best_seq is not None:
        polish_t = min(300.0, max(10.0, deadline - time.monotonic() - 20))
        print(f"  n={len(best_seq)}, score={best_score:.6f}, t={polish_t:.0f}s")
        polished, p_s = local_search_focused(best_seq, polish_t, rng, n_top=6)
        if p_s < best_score:
            best_score = p_s
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")

    # Phase 4: Upsample & optimize
    print("\n=== Phase 4: Upsample ===")
    if best_seq is not None and len(best_seq) <= 384:
        target = len(best_seq) * 2
        if target <= 768:
            a_up = upsample(best_seq, target, rng)
            t_alloc = min(80.0, max(10.0, deadline - time.monotonic() - 15))
            if t_alloc > 10:
                print(f"  Upsample to {target}, t={t_alloc:.0f}s")
                a_opt, s_opt = mirror_descent_optimize(a_up, t_alloc, rng,
                                                        k_start=32, k_end=3,
                                                        eta_init=0.5)
                print(f"    -> {s_opt:.6f}")
                if s_opt < best_score:
                    best_score = s_opt
                    best_seq = a_opt.copy()
                    # Quick polish
                    if deadline - time.monotonic() > 30:
                        p, ps = local_search_focused(best_seq,
                                                     min(40.0, deadline - time.monotonic() - 10),
                                                     rng)
                        if ps < best_score:
                            best_score = ps
                            best_seq = p.copy()

    # Phase 5: Restarts from best
    remaining = deadline - time.monotonic() - 10
    if remaining > 50 and best_seq is not None:
        print(f"\n=== Phase 5: Restarts ({remaining:.0f}s left) ===")
        n_rst = min(4, int(remaining // 50))
        for r in range(n_rst):
            if time.monotonic() > deadline - 12:
                break
            a_new = best_seq * np.exp(rng.normal(0, 0.12, len(best_seq)))
            a_new = np.maximum(a_new, 1e-12)
            a_new = a_new / np.sum(a_new)
            t_alloc = min(40.0, (deadline - time.monotonic() - 8) / max(1, n_rst - r))
            a_opt, s_opt = mirror_descent_optimize(a_new, t_alloc, rng,
                                                    k_start=24, k_end=2,
                                                    eta_init=0.6)
            print(f"  RST{r+1}: {s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} ***")
                if deadline - time.monotonic() > 25:
                    p, ps = local_search_focused(best_seq,
                                                 min(20.0, deadline - time.monotonic() - 8),
                                                 rng)
                    if ps < best_score:
                        best_score = ps
                        best_seq = p.copy()

    # Final
    print(f"\n{'='*50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        fs = evaluate_sequence(final_list)
        print(f"=== FINAL: score={fs:.6f}, n={len(final_list)} ===")
        return final_list

    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
