import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog

def fft_convolve(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]

def fft_convolve_self(a):
    return fft_convolve(a, a)

def a_from_theta(theta, S):
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    result = S * exp_theta / total
    return result

def softmax_obj_grad(theta, n, T, S):
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    
    # Softmax value
    shifted = T * (c - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T
    
    # Weights
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient w.r.t. a
    # Using the formula: grad_a = 2 * conv(w, a_rev)[n-1:2n-1]
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    grad_a = 2.0 * full_conv[n-1:2*n-1]
    
    # Chain rule for theta parameterization
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    
    return obj, grad_theta

def true_obj(a):
    c = fft_convolve_self(a)
    return c.max()

def subgradient_step(a, S, step_size, num_max=4):
    """Take a subgradient step at the max point(s)."""
    n = len(a)
    c = fft_convolve_self(a)
    # Find the max point(s)
    max_val = c.max()
    # Get indices near the max
    thresh = max_val - 1e-6 * max_val
    active = np.where(c >= thresh)[0]
    if len(active) > num_max:
        active = active[np.argsort(c[active])[::-1][:num_max]]
    
    # Subgradient at active points: for k in active, dc[k]/da[i] = a[k-i] + a[i-k]
    # Use all active points with equal weight
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            # dc[k]/da[i] = a[k-i] if 0<=k-i<n, plus a[i-k] if 0<=i-k<n and i!=k-i
            val = 0.0
            if 0 <= k-i < n:
                val += a[k-i]
            if 0 <= i-k < n and i != k-i:
                val += a[i-k]
            elif 0 <= i-k < n and i == k-i:
                val += a[i-k]  # same as a[k-i] since k-i = i-k = 0
            grad[i] += val
    grad /= len(active)

    grad = grad - np.mean(grad)
    
    # Line search
    for alpha in [step_size, step_size*0.5, step_size*0.25, step_size*0.1]:
        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new * (S / s)
        obj_new = true_obj(a_new)
        if obj_new < true_obj(a):
            return a_new
    return a
