def mirror_descent_step(a, c, eta, top_k=8):
    """Multiplicative weight update."""
    n = len(a)
    c_max = c.max()
    threshold = c_max * 0.98
    hot_lags = np.where(c >= threshold)[0]
    if len(hot_lags) > top_k:
        hot_lags = hot_lags[np.argsort(c[hot_lags])[::-1][:top_k]]
    if len(hot_lags) == 0:
        hot_lags = np.array([np.argmax(c)])
    
    g = np.zeros(n)
    for k in hot_lags:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            if k - i == i:
                g[i] += a[k - i]
            else:
                g[i] += 2.0 * a[k - i]
    
    # Normalize gradient
    g_max = g.max()
    if g_max < 1e-30:
        return a.copy()
    
    # Multiplicative update: a_new[i] = a[i] * exp(-eta * g[i] / g_max)
    exponent = -eta * g / g_max
    # Clip to avoid underflow
    exponent = np.clip(exponent, -20, 20)
    a_new = a * np.exp(exponent)
    
    # Renormalize
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new = a_new * (a.sum() / s)
    
    return np.maximum(a_new, 0)

def block_coordinate_descent(a, n, S, time_budget, block_size=8, max_sweeps=100):
    """Block coordinate descent with FFT-based evaluation."""
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for start_idx in range(0, n, block_size):
            if time.monotonic() - start > time_budget:
                break
            end_idx = min(start_idx + block_size, n)
            block_idx = slice(start_idx, end_idx)
            block = a[block_idx].copy()
            block_orig = block.copy()
            
            # Try small perturbations
            for _ in range(3):
                # Random perturbation
                if time.monotonic() - start > time_budget:
                    break
                perturbation = np.random.randn(end_idx - start_idx) * 0.1 * block_orig.max()
                a[block_idx] = np.maximum(block_orig + perturbation, 0)
                s = a.sum()
                if s < 1e-10:
                    a[block_idx] = block_orig
                    continue
                a = a * (S / s)
                c_new = fft_convolve_self(a)
                obj_new = c_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    break
            else:
                a[block_idx] = block_orig
                s = a.sum()
                if s > 1e-10:
                    a = a * (S / s)
        
        if not improved:
            # Try larger perturbations
            for _ in range(5):
                if time.monotonic() - start > time_budget:
                    break
                idx = np.random.randint(0, n, block_size)
                for i in idx:
                    orig = a[i]
                    for factor in [0.5, 0.7, 1.3, 1.5]:
                        a[i] = orig * factor
                        s = a.sum()
                        if s < 1e-10:
                            a[i] = orig
                            continue
                        a = a * (S / s)
                        c_new = fft_convolve_self(a)
                        obj_new = c_new.max()
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_a = a.copy()
                            improved = True
                        else:
                            break
                    # Renormalize back
                    s = a.sum()
                    if s > 1e-10:
                        a = a * (S / s)
            
            if not improved:
                break
    
    return best_a, best_obj

def run_mirror_descent(a_init, time_budget, eta_init=0.1, eta_min=0.001, top_k=8, 
                       shuffle_interval=50):
    """Mirror descent with decreasing step size."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t = 0
    while time.monotonic() - start < time_budget:
        t += 1
        eta = max(eta_min, eta_init * (0.995 ** t))
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()
        
        a_new = mirror_descent_step(a, c, eta, top_k=top_k)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a = best_a.copy()
            continue
        a = a_new * (S / s)
        
        # Periodic shuffle / escape
        if t % shuffle_interval == 0:
            noise = np.random.randn(n) * 0.01 * a.max()
            a = a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 0:
                a = a * (S / s)
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using power-mean surrogate L-BFGS,
mirror descent (multiplicative weights), targeted hot-lag reduction,
and block coordinate descent with FFT.

Key innovations over previous:
1. Power-mean objective (sharper focus on max than log-sum-exp)
2. Multiplicative weight / mirror descent for natural non-negativity
3. Targeted hot-lag subgradient with weighted contribution analysis
4. "Shake and re-optimize" to escape local minima
5. Structured initializations (ramp, anti-symmetric, low-pass)
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def true_objective_value(a):
    """Compute the actual objective 2n*maxconv/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def compute_power_mean_grad(c, a, p, S):
    """
    Compute gradient of power-mean objective w.r.t. log-parameters.
    a = S * softmax(theta), objective = (sum c_k^p / K)^(1/p)
    Returns (objective_value, grad_theta).
    """
    n = len(a)
    K = len(c)
    c_clipped = np.maximum(c, 1e-20)
    
    c_p_sum = np.sum(c_clipped ** p)
    M_p = (c_p_sum / K) ** (1.0 / p)
    
    # dM_p/dc_k = (1/K) * M_p^(p-1) * c_k^(p-1)
    weights = c_clipped ** (p - 1)
    w_sum = weights.sum()
    
    if w_sum < 1e-30:
        return M_p, np.zeros(n)
    
    # grad_a[i] = (1/K) * M_p^(p-1) * sum_k weights[k] * d(conv[k])/d(a[i])
    # d(conv[k])/d(a[i]) = 2*a[k-i] (approx, ignoring diagonal)
    # sum_k weights[k] * a[k-i] = cross-correlation of weights and a at lag i
    
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    
    # corr[i] = sum_k weights[k] * a[(k-i) mod fsize] -- but since fsize >= 2n-1,
    # and both weights and a have support [0, n-1], this gives the linear cross-corr
    # corr[i] for i in [0, n-1] = sum_k weights[k] * a[k-i] for valid k-i
    
    grad_a = 2.0 * corr[:n]
    
    # Diagonal correction: for even k where k/2 is in [0,n-1], subtract a[k/2]
    # This is a small effect, skip for efficiency
    
    scale = (1.0 / K) * (M_p ** (p - 1)) * w_sum
    grad_a *= scale
    
    # Chain rule for log-parameters: a[i] = S*exp(theta[i])/sum(exp(theta))
    # d a[i]/d theta[j] = a[i]*(delta_ij - a[j]/S)
    # dM/dtheta[j] = grad_a[j]*a[j] - (a[j]/S)*sum_i grad_a[i]*a[i]
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    
    return M_p, grad_theta


def softmax_from_theta(theta, S):
    """Compute a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def run_lbfgs_pmean(theta_init, n, S, p, time_budget, max_iter=300):
    """Run L-BFGS on power-mean objective with log-parameterization."""
    state = {
        'start': time.monotonic(),
        'budget': time_budget,
        'best_obj': np.inf,
        'best_a': softmax_from_theta(theta_init, S).copy(),
        'n_calls': 0

    a0 = softmax_from_theta(theta_init, S)
    c0 = fft_convolve_self(a0)
    state['best_obj'] = c0.max()
    state['best_a'] = a0.copy()
    
    def obj_grad(theta):
        state['n_calls'] += 1
        elapsed = time.monotonic() - state['start']
        if elapsed > state['budget']:
            return state['best_obj'], np.zeros(len(theta))
        
        a = softmax_from_theta(theta, S)

        true_max = c.max()

            state['best_obj'] = true_max
            state['best_a'] = a.copy()
        
        obj, grad = compute_power_mean_grad(c, a, p, S)
        return obj, grad
    
    try:

def run_mirror_descent(a_init, time_budget, eta_init=0.05, eta_decay=0.999, 
                       top_k=16, shake_every=200, shake_amp=0.02):
    """Mirror descent (multiplicative weights) with periodic shaking."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t = 0
    while time.monotonic() - start < time_budget:
        t += 1
        eta = eta_init * (eta_decay ** t)
        if eta < 1e-6:
            eta = 1e-6
        
        # Identify hot lags
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()
        
        # Find hot lags (within 2% of max)
        c_max = c.max()
        threshold = c_max * 0.98
        hot_mask = c >= threshold
        hot_lags = np.where(hot_mask)[0]
        
        if len(hot_lags) > top_k:
            # Keep only the top_k hottest
            hot_vals = c[hot_lags]
            top_idx = np.argsort(hot_vals)[::-1][:top_k]
            hot_lags = hot_lags[top_idx]
        elif len(hot_lags) == 0:
            hot_lags = np.array([np.argmax(c)])
        
        # Compute subgradient (weighted by contribution)
        g = np.zeros(n)
        for k in hot_lags:
            weight = c[k] / c_max  # weight by how hot this lag is
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                ki = k - i
                if ki == i:
                    g[i] += weight * a[ki]
                else:
                    g[i] += weight * 2.0 * a[ki]
        
        g_max = g.max()
        if g_max < 1e-30:
            break
        
        # Multiplicative update
        exponent = -eta * g / g_max
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)
        a_new = np.maximum(a_new, 0)
        
        s = a_new.sum()
        if s < 1e-10:
            a = best_a.copy()
            continue
        a = a_new * (S / s)
        
        # Periodic shake to escape local minima
        if t % shake_every == 0:
            noise = np.random.randn(n) * shake_amp * a.max()
            a = a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 0:
                a = a * (S / s)
            # Reduce shake amplitude over time
            shake_amp *= 0.9
    
    return best_a, best_obj


def coordinate_descent_polish(a_init, time_budget, max_sweeps=200):
    """Coordinate descent: try reducing each element that contributes to hot lags."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        # Find hot lags
        c = fft_convolve_self(a)
        c_max = c.max()
        hot_threshold = c_max * 0.95
        hot_lags = np.where(c >= hot_threshold)[0]
        
        if len(hot_lags) > 10:
            hot_lags = hot_lags[np.argsort(c[hot_lags])[::-1][:10]]
        
        # For each hot lag, find the positions contributing most
        active_positions = set()
        for k in hot_lags:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # Find top contributors
            contribs = np.array([a[i] * a[k - i] for i in range(lo, hi + 1)])
            if len(contribs) <= 5:
                active_positions.update(range(lo, hi + 1))
            else:
                top_idx = np.argsort(contribs)[::-1][:3]
                active_positions.update(range(lo + top_idx[0], lo + top_idx[0] + 1))
                active_positions.update(range(lo + top_idx[1], lo + top_idx[1] + 1))
                active_positions.update(range(lo + top_idx[2], lo + top_idx[2] + 1))
        
        improved = False
        for i in sorted(active_positions):
            if time.monotonic() - start > time_budget:

            orig = a[i]

def block_optimize(a_init, time_budget, block_size=16, max_iters=500):
    """Optimize blocks of coordinates using local perturbation."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = a * (S / a.sum())
    
    start = time.monotonic()

    best_a = a.copy()
    
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Pick a random block
        start_idx = rng.integers(0, n - block_size)
        idx = np.arange(start_idx, start_idx + block_size)
        
        # Generate perturbation: move mass within the block
        block_orig = a[idx].copy()
        block_sum = block_orig.sum()
        
        if block_sum < 1e-15:
            continue
        
        # Random re-distribution
        t = rng.dirichlet(np.ones(block_size) * 2.0)
        a[idx] = block_sum * t
        
        s = a.sum()
        if s < 0.01:
            a[idx] = block_orig
            continue
        a_scaled = a * (S / s)
        
        c_new = fft_convolve_self(a_scaled)
        obj_new = c_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_scaled.copy()
            a = a_scaled.copy()
        else:
            a[idx] = block_orig
            s = a.sum()
            if s > 0:
                a = a * (S / s)
    
    return best_a, best_obj


def get_structured_init(n, S, rng, init_type="ramp"):
    """Generate structured initializations."""
    if init_type == "uniform":

    elif init_type == "ramp":
        a = np.linspace(0.01, 3.0, n)
    elif init_type == "anti_sym":
        # Anti-symmetric: large on left, small on right
        a = np.linspace(3.0, 0.01, n)
    elif init_type == "quadratic_ramp":
        x = np.linspace(0, 1, n)
        a = x ** 2 + 0.01
    elif init_type == "sqrt_ramp":
        x = np.linspace(0, 1, n)
        a = np.sqrt(x) + 0.01

        x = np.arange(n)

    elif init_type == "cosine_window":
        x = np.linspace(0, np.pi, n)

        x = np.arange(n)

        a = np.maximum(a, 0.01)
    elif init_type == "random":
        a = rng.random(n) + 0.01
    else:
        a = np.full(n, S / n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a * (S / s)

        a_for_log = np.maximum(a_for_log, 1e-15)
        theta = np.log(a_for_log) - np.log(S)
        
        for p in [4, 8]:
            if time.monotonic() > deadline - 60:
                break
            per_p_time = phase1_budget / 2
            t0 = time.monotonic()
            a_best, obj_best = run_lbfgs_pmean(theta, n, S, p, per_p_time, max_iter=200)
            elapsed = time.monotonic() - t0
            obj_eval = true_objective_value(a_best)
            print("L-BFGS p=%d: maxconv=%.6f, obj=%.6f (t=%.1fs, calls=)" %
                  (p, obj_best, obj_eval, elapsed))
            
            if obj_eval < global_best:
                global_best = obj_eval
                global_a = a_best.copy()
            
            # Update theta from best
            a_best = np.maximum(a_best, 1e-15)
            s = a_best.sum()
            a_best = a_best * (S / s)
            theta = np.log(a_best) - np.log(S)
    
    # Phase 2: Mirror descent
    # Budget: 25% of time
    phase2_budget = 0.25 * (deadline - time.monotonic())
    
    if global_a is not None and phase2_budget > 5:
        a_init = global_a.copy()
        s = a_init.sum()
        a_init = a_init * (S / s)
        
        t0 = time.monotonic()
        a_best, obj_best = run_mirror_descent(
            a_init, phase2_budget, eta_init=0.03, eta_decay=0.9995,
            top_k=16, shake_every=100, shake_amp=0.01
        )
        elapsed = time.monotonic() - t0
        obj_eval = true_objective_value(a_best)
        print("Mirror descent: obj=%.6f (t=%.1fs)" % (obj_eval, elapsed))
        
        if obj_eval < global_best:
            global_best = obj_eval
            global_a = a_best.copy()
    
    # Phase 3: L-BFGS with higher p (more focused on max)
    # Budget: 20% of time
    phase3_budget = 0.20 * (deadline - time.monotonic())
    
    if global_a is not None and phase3_budget > 5:
        a_for_log = global_a.copy()
        a_for_log = np.maximum(a_for_log, 1e-15)
        s = a_for_log.sum()
        a_for_log = a_for_log * (S / s)
        theta = np.log(a_for_log) - np.log(S)
        
        for p in [16, 32]:
            if time.monotonic() > deadline - 30:
                break
            per_p_time = phase3_budget / 2
            a_best, obj_best = run_lbfgs_pmean(theta, n, S, p, per_p_time, max_iter=150)
            obj_eval = true_objective_value(a_best)
            print("L-BFGS p=%d (refine): obj=%.6f" % (p, obj_eval))
            
            if obj_eval < global_best:
                global_best = obj_eval
                global_a = a_best.copy()
            
            a_best = np.maximum(a_best, 1e-15)
            s = a_best.sum()
            a_best = a_best * (S / s)
            theta = np.log(a_best) - np.log(S)
    
    # Phase 4: Coordinate descent polish
    # Budget: 15% of time
    phase4_budget = 0.15 * (deadline - time.monotonic())
    
    if global_a is not None and phase4_budget > 3:
        a_init = global_a.copy()
        s = a_init.sum()
        a_init = a_init * (S / s)
        
        t0 = time.monotonic()
        a_best, obj_best = coordinate_descent_polish(a_init, phase4_budget)
        elapsed = time.monotonic() - t0
        obj_eval = true_objective_value(a_best)
        print("Coord descent: obj=%.6f (t=%.1fs)" % (obj_eval, elapsed))
        
        if obj_eval < global_best:
            global_best = obj_eval
            global_a = a_best.copy()
    
    # Phase 5: Block optimization + final mirror descent
    remaining = deadline - time.monotonic() - 10.0
    
    if global_a is not None and remaining > 5:
        a_init = global_a.copy()
        s = a_init.sum()
        a_init = a_init * (S / s)
        
        # Block optimization
        block_time = min(remaining * 0.5, 120.0)
        if block_time > 3:
            a_best, obj_best = block_optimize(a_init, block_time, block_size=16)
            obj_eval = true_objective_value(a_best)
            print("Block opt: obj=%.6f" % obj_eval)
            if obj_eval < global_best:
                global_best = obj_eval
                global_a = a_best.copy()
        
        # Final mirror descent polish
        remaining2 = deadline - time.monotonic() - 8.0
        if remaining2 > 3:
            a_init = global_a.copy()
            s = a_init.sum()
            a_init = a_init * (S / s)
            a_best, obj_best = run_mirror_descent(
                a_init, remaining2, eta_init=0.01, eta_decay=0.9999,
                top_k=8, shake_every=500, shake_amp=0.005
            )
            obj_eval = true_objective_value(a_best)
            print("Final MD: obj=%.6f" % obj_eval)
            if obj_eval < global_best:
                global_best = obj_eval
                global_a = a_best.copy()
    
    # Phase 6: Try alternative initializations if time permits
    remaining = deadline - time.monotonic() - 10.0
    if remaining > 60:
        # Try a few structured inits with quick optimization
        for init_type in ["quadratic_ramp", "bimodal", "chirp_magnitude"]:
            if time.monotonic() > deadline - 30:
                break
            a_init = get_structured_init(n, S, rng, init_type)
            
            # Quick mirror descent
            t0 = time.monotonic()
            a_best, obj_best = run_mirror_descent(
                a_init, min(30.0, remaining / 3), eta_init=0.02, top_k=12
            )
            elapsed = time.monotonic() - t0
            obj_eval = true_objective_value(a_best)
            print("Init %s: obj=%.6f (t=%.1fs)" % (init_type, obj_eval, elapsed))
            
            if obj_eval < global_best:
                global_best = obj_eval
                global_a = a_best.copy()
                
            # Quick L-BFGS from this
            if time.monotonic() > deadline - 30:
                break
            remaining2 = deadline - time.monotonic() - 20.0
            if remaining2 > 10:
                a_for_log = np.maximum(a_best, 1e-15)
                s = a_for_log.sum()
                a_for_log = a_for_log * (S / s)
                theta = np.log(a_for_log) - np.log(S)
                a_best2, obj_best2 = run_lbfgs_pmean(theta, n, S, 8, remaining2, max_iter=100)
                obj_eval2 = true_objective_value(a_best2)
                if obj_eval2 < global_best:
                    global_best = obj_eval2
                    global_a = a_best2.copy()
    
    # Final verification
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("FINAL: obj=%.6f (verified=%.6f)" % (global_best, ver))
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.6f" % ver)
    
    return final_a
