"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.
Uses FISTA with adaptive log-sum-exp, subgradient polishing, and local search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_lse_grad(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_bc = np.exp(shifted)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), 0.0
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, lse_val


def compute_toppk_grad(a, c, top_k):
    """Gradient using top-k active lags."""
    n = len(a)
    if top_k >= len(c):
        top_indices = np.arange(len(c))
        top_vals = c.copy()
    else:
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    
    w_full = np.zeros(len(c))
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fista_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=500.0, L_init=None):
    """FISTA with log-sum-exp."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    # Estimate L (Lipschitz constant of gradient)
    if L_init is None:
        # L = beta * 2 * max_b (rough estimate)
        c = fft_convolve_self(a)
        L_init = beta_init * 2.0 * c.max()
    
    step = 1.0 / max(L_init, 1e-10)
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.001 * it))
        
        # Recompute gradient at y
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()
        
        grad, lse_val = compute_lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Update step size based on current Lipschitz estimate
        L = beta * 2.0 * c.max()
        step = 1.0 / max(L, 1e-10)
        
        x_new = project_simplex(y - step * grad, S)
        
        # Check improvement periodically
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic perturbation
        if it % 2000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
    
    return best_local, best_score


def nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=256):
    """Nesterov top-k subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    # Lower bound for Polyak step
    lower_bound = S * S / (2 * n - 1)  # = 2n/(2n-1) when S = sqrt(2n)
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()
        
        grad = compute_toppk_grad(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Polyak step size
        gap = cur_max - lower_bound
        if gap > 1e-10:
            step = min(2.0 * gap / (gnorm * gnorm), 1.0)
        else:
            step = 0.0
            break
        
        step = min(step, 2.0 * S / (n * gnorm + 1e-10))
        
        x_new = project_simplex(y - step * grad, S)
        
        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 3000 == 0 and rng.random() < 0.15:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
    
    return best_local, best_score


def local_refine(a, n, S, max_time, rng):
    """Try small local perturbations to improve the exact objective."""
    a = np.clip(a, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    
    # Find the lag where max occurs
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_local = a.copy()
        
        # Find top few peaks
        top_lags = np.argsort(c)[::-1][:5]
        peak_lag = top_lags[0]
        
        # The gradient of b[peak_lag] w.r.t. a is:
        # g[m] = a[peak_lag - m] + a[m - peak_lag] (with 0 for out of range)
        g = np.zeros(n)
        for m in range(n):
            idx1 = int(peak_lag) - m
            idx2 = m - int(peak_lag)
            if 0 <= idx1 < n:
                g[m] += a[idx1]
            if 0 <= idx2 < n:
                g[m] += a[idx2]
        
        # Move in direction that reduces b[peak_lag]
        # We want to decrease a[m] where g[m] is large, and increase where g[m] is small
        # But we must maintain sum = S
        # So: delta[m] = -eta * (g[m] - mean(g))
        eta = 0.01 * best_score / (np.linalg.norm(g) + 1e-10)
        delta = -eta * (g - g.mean())
        
        # Also add small random component
        delta += rng.normal(0, eta * 0.1, n)
        
        a_new = project_simplex(a + delta, S)
        
        c_new = fft_convolve_self(a_new)
        if c_new.max() < best_score:
            a = a_new
        else:
            # Try smaller step
            a_new = project_simplex(a + 0.1 * delta, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < best_score:
                a = a_new
            else:
                # Stuck, try random perturbation
                a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
    
    return best_local, best_score

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.
Uses FISTA with adaptive log-sum-exp, scipy L-BFGS refinement,
Nesterov top-k subgradient polishing, and targeted local search.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_lse_grad(a, c, beta):
    """Gradient and value of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -700, 700)
    exp_bc = np.exp(shifted)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, lse_val


def compute_toppk_grad(a, c, top_k):
    """Gradient using top-k active lags with exponential weighting."""
    n = len(a)
    if top_k >= len(c):
        top_indices = np.arange(len(c))
        top_vals = c.copy()
    else:
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    # Exponential weighting: prioritize the peak more
    rank = np.arange(len(top_vals), 0, -1, dtype=np.float64)
    weights = top_vals / top_vals.max() * np.exp(-0.1 * (rank - 1))
    weights = weights / weights.sum()
    w_full = np.zeros(len(c))
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_toppk_grad_uniform(a, c, top_k):
    """Gradient using top-k active lags with uniform weighting."""
    n = len(a)
    if top_k >= len(c):
        top_indices = np.arange(len(c))
        top_vals = c.copy()
    else:
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(len(c))
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fista_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=500.0):
    """FISTA with log-sum-exp and adaptive step size."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    # Initial Lipschitz estimate
    c0 = fft_convolve_self(a)
    L = max(beta_init * 2.0 * c0.max(), 1e-10)
    step = 1.0 / L

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.0008 * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()

        grad, lse_val = compute_lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: use current Lipschitz estimate
        L_new = beta * 2.0 * c.max()
        step = 1.0 / max(L_new, 1e-10)

        # Backtracking: ensure sufficient decrease
        x_new = project_simplex(y - step * grad, S)
        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        # Periodic perturbation to escape local minima
        if it % 3000 == 0 and rng.random() < 0.25:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0

    return best_local, best_score


def nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=300):
    """Nesterov top-k subgradient with Polyak step size."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    lower_bound = S * S / (2.0 * n - 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 80)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()

        grad = compute_toppk_grad_uniform(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Polyak step size with cap
        gap = cur_max - lower_bound
        if gap > 1e-12:
            step = min(2.0 * gap / (gnorm * gnorm + 1e-30), 1.0)
        else:
            break
        step = min(step, 2.0 / (gnorm + 1e-15))

        x_new = project_simplex(y - step * grad, S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 2000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0

    return best_local, best_score


def scipy_lbfgs_optimize(a_init, n, S, max_time, beta=50.0):
    """Use scipy L-BFGS-B on softmax-reparameterized log-sum-exp objective."""
    from scipy.optimize import minimize

    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    # Initialize log-params
    x0 = np.log(np.maximum(a, 1e-10))
    x0 = x0 - x0.mean()  # Remove gauge freedom

    def objective_and_grad(x):
        # Softmap reparametrization
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        Z = exp_x.sum()
        a_curr = S * exp_x / Z

        # Compute objective (log-sum-exp approximation to max)
        c = fft_convolve_self(a_curr)
        c_max = c.max()
        shifted = beta * (c - c_max)
        shifted = np.clip(shifted, -700, 700)
        exp_bc = np.exp(shifted)
        total = exp_bc.sum()
        lse_val = c_max + np.log(total) / beta

        # Gradient w.r.t. a
        w = exp_bc / total
        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a_curr, fsize)
        grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Chain rule: d(a_i)/dx_j = a_i*(delta_ij - a_j/S)
        # d f / dx_j = sum_i grad_a[i] * a_i * (delta_ij - a_j/S)
        #             = grad_a[j] * a_j - (a_j/S) * sum_i grad_a[i] * a_i
        ga_dot_a = np.dot(grad_a, a_curr)
        grad_x = grad_a * a_curr - (a_curr / S) * ga_dot_a

        return lse_val, grad_x

    start = time.monotonic()
    best_x = x0.copy()
    best_val = np.inf

    result = minimize(
        objective_and_grad, x0, method='L-BFGS-B',
        jac=True,
        options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-8,
                 'maxfun': 500, 'disp': False}
    )

    if result.fun < best_val:
        best_x = result.x
        best_val = result.fun

    # Convert back
    x_final = best_x
    x_shifted = x_final - x_final.max()
    exp_x = np.exp(x_shifted)
    Z = exp_x.sum()
    a_final = S * exp_x / Z
    a_final = np.clip(a_final, 0, 1000)

    c = fft_convolve_self(a_final)
    score = c.max()

    return a_final, score


def local_refine(a_init, n, S, max_time, rng):
    """Targeted local search focusing on peak lags."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)

    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    stall_count = 0

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score - 1e-13:
            best_score = cur_max
            best_local = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 500:
            # Add random perturbation
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            stall_count = 0
            continue

        # Find top lags
        top_k = min(10, len(c))
        top_lags = np.argsort(c)[::-1][:top_k]
        peak_val = c[top_lags[0]]

        # Compute composite subgradient from top lags
        g = np.zeros(n)
        for lag in top_lags:
            lag_int = int(lag)
            weight = c[lag] / peak_val
            for m in range(n):
                idx1 = lag_int - m
                if 0 <= idx1 < n:
                    g[m] += weight * a[idx1]
                idx2 = m - lag_int
                if 0 <= idx2 < n:
                    g[m] += weight * a[idx2]

        # Direction: reduce the peak
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        eta = 0.05 * best_score / (gnorm + 1e-10)
        delta = -eta * g_centered / gnorm

        # Try the step
        a_new = project_simplex(a + delta, S)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < best_score:
            a = a_new
        else:
            # Try smaller step
            a_new = project_simplex(a + 0.2 * delta, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < best_score:
                a = a_new
            else:
                # Try opposite direction on some elements
                small_delta = rng.normal(0, 0.002 * S / np.sqrt(n), n)
                a = project_simplex(a + small_delta, S)

    return best_local, best_score

        s4_actual = a4.sum()
        if s4_actual > 1e-10:
            a4 = a4 * (S_best / s4_actual)
        c4 = fft_convolve_self(a4)
        obj4 = 2.0 * n_best * c4.max() / (a4.sum() ** 2)
        if obj4 < global_best:
            global_best = obj4
            global_a = a4.copy()
            print("  Nesterov: %.6f" % global_best)

    # Phase 5: Local refinement
    if global_a is not None and time.monotonic() < deadline - 15:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = deadline - time.monotonic()

        print("=== Phase 5: Local refinement (%.0fs) ===" % (remaining * 0.2))
        seg5 = remaining * 0.2
        a5, s5 = local_refine(global_a, n_best, S_best, seg5, rng)
        a5 = np.clip(a5, 0, 1000)
        s5_actual = a5.sum()
        if s5_actual > 1e-10:
            a5 = a5 * (S_best / s5_actual)
        c5 = fft_convolve_self(a5)
        obj5 = 2.0 * n_best * c5.max() / (a5.sum() ** 2)
        if obj5 < global_best:
            global_best = obj5
            global_a = a5.copy()
            print("  Local: %.6f" % global_best)

    # Phase 6: Final Nesterov with high top-k
    if global_a is not None and time.monotonic() < deadline - 5:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = deadline - time.monotonic()

        if remaining > 5:
            print("=== Phase 6: Final polish (%.0fs) ===" % remaining)
            a6, s6 = nesterov_toppk(global_a, n_best, S_best, remaining, rng,
                                     top_k_init=30, top_k_max=400)
            a6 = np.clip(a6, 0, 1000)
            s6_actual = a6.sum()
            if s6_actual > 1e-10:
                a6 = a6 * (S_best / s6_actual)
            c6 = fft_convolve_self(a6)
            obj6 = 2.0 * n_best * c6.max() / (a6.sum() ** 2)
            if obj6 < global_best:
                global_best = obj6
                global_a = a6.copy()
                print("  Final: %.6f" % global_best)

    # Verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        try:
            ver = evaluate_sequence(final_list)
            if np.isfinite(ver):
                print("Verification: %.6f" % ver)
                if ver < global_best:
                    global_best = ver
            else:
                print("Verification failed (inf), using fallback")
                global_a = np.ones(256, dtype=np.float64) * 0.1
        except Exception as e:
            print("Verification error: %s" % str(e))
            global_a = np.ones(256, dtype=np.float64) * 0.1
    else:
        global_a = np.ones(256, dtype=np.float64) * 0.1

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * 0.1

    print("Returning length=%d, internal_score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
