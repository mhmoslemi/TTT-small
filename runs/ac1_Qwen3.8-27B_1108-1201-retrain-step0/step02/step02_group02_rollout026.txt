def solve_lp_update(f, M, n):
    # Maximize sum(g) subject to (f*g)[k] <= M, g >= 0
    # In standard form: minimize -sum(g) subject to A@g <= b, g >= 0
    
    # A has 2n-1 rows, n columns
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g)
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if result.success:
        return result.x
    return None

from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def compute_conv_peak_and_index(a):
    """Compute convolution, its peak value, and peak index."""
    c = fft_convolve(a, a)
    idx = int(np.argmax(c))
    return c, c[idx], idx

    """
    Solve LP: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g vector, or None if infeasible.
    """

    A = np.zeros((2*n-1, n))

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)

    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)

    For the LP: (f*g)[k] = sum_j A[k,j]*g[j] where A[k,j] = f[k-j] if valid.
    We precompute the index pairs for efficiency.

        row_data = []
        for j in range(j_min, j_max + 1):

        pairs.append(row_data)
    return pairs

    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)

    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')

    if res.success:
        return res.x
    return None


def lp_optimize(f_init, time_limit, rng, verbose=False):
    """LP-based iterative optimization with adaptive step sizes and restarts."""
    n = len(f_init)
    f = normalize_sqrt2n(f_init.astype(np.float64).copy(), n)
    pairs = build_constraint_matrix(n)

    best_f = f.copy()
    best_score = local_score(f)
    if verbose:
        print(f"    LP init: n={n}, score={best_score:.6f}")

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    patience = 40
    step_size = 0.3

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 >= time_limit:
            break

        c = fft_convolve(f, f)
        M = float(np.max(c))

        g = solve_lp(f, M, n, pairs)
        if g is None:
            f = best_f * np.exp(rng.normal(0, 0.1, n))
            f = normalize_sqrt2n(np.clip(f, 0, None), n)
            no_improve += 1
            continue

        S_g = float(np.sum(g))
        if S_g < 1e-10:
            no_improve += 1
            continue
        g = g * np.sqrt(2.0 * n) / S_g

        # Line search with multiple step sizes
        improved = False
        for t in [0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.02]:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.clip(f_trial, 0.0, None)
            f_trial = normalize_sqrt2n(f_trial, n)
            s = local_score(f_trial)
            if s < best_score:
                best_score = s
                best_f = f_trial.copy()
                f = f_trial.copy()
                improved = True
                step_size = t
                break
            elif s < local_score(f) * 0.999:
                f = f_trial.copy()

        if improved:
            no_improve = 0
            # Gradually increase step size
            step_size = min(step_size * 1.1, 0.5)
        else:
            no_improve += 1
            step_size = max(step_size * 0.9, 0.05)

        if no_improve >= patience:
            # Restart strategy
            r = rng.random()
            if r < 0.4:
                # Gaussian perturbation around best
                f = best_f * np.exp(rng.normal(0, 0.2, n))
            elif r < 0.7:
                # Targeted: perturb peak contributors
                c_best = fft_convolve(best_f, best_f)
                peak_idx = int(np.argmax(c_best))
                contribs = np.zeros(n)
                for i in range(n):
                    if 0 <= peak_idx - i < n:
                        contribs[i] = best_f[i] * best_f[peak_idx - i]
                top_k = np.argsort(contribs)[::-1][:max(4, n // 8)]
                f = best_f.copy()
                f[top_k] *= np.exp(rng.normal(-0.15, 0.1, len(top_k)))
            else:
                # Block fluctuation
                f = best_f.copy()
                block = max(4, n // 6)
                start = int(rng.integers(0, n - block))
                f[start:start + block] *= np.exp(rng.normal(0, 0.25, block))

            f = normalize_sqrt2n(np.clip(f, 0, None), n)
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    LP iter={it}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")

    return best_f, best_score


def peak_shave_polish(a_init, time_limit, rng):
    """Targeted peak-directed polishing with mass redistribution."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = local_score(a)
    cur_score = best_score
    t0 = time.monotonic()
    temp = 0.005
    cooling = 0.9999

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))

        # Find top contributors to peak
        contribs = np.zeros(n)
        for i in range(n):
            if 0 <= peak_idx - i < n:
                contribs[i] = a[i] * a[peak_idx - i]

        r = rng.random()
        if r < 0.4:
            # Reduce top contributors, boost bottom
            top_k = max(3, n // 20)
            bot_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            bot_idx = np.argsort(contribs)[:bot_k]
            red = rng.uniform(0.03, 0.08)
            a[top_idx] *= (1.0 - red)
            a[bot_idx] *= (1.0 + red * len(top_idx) / len(bot_idx))
        elif r < 0.65:
            # Pair transfer
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            j_red = top_idx[int(rng.integers(0, top_k))]
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red:
                amt = a[j_red] * rng.uniform(0.03, 0.08)
                a[j_red] = max(0.0, a[j_red] - amt)
                a[j_inc] += amt
        elif r < 0.8:
            # Block scaling
            w = max(2, int(rng.integers(2, max(3, n // 5))))
            start = int(rng.integers(0, n - w))
            a[start:start + w] *= np.exp(rng.normal(0, 0.04))
        else:
            # Global small perturbation
            a *= np.exp(rng.normal(0, 0.008, n))

        a = np.clip(a, 0.0, None)
        a = normalize_sqrt2n(a, n)
        s = local_score(a)

        if s < cur_score:
            cur_score = s
            if s < best_score:
                best_score = s
                best_a = a.copy()
        elif rng.random() < np.exp(-(s - cur_score) / max(temp, 1e-12)):
            cur_score = s

        temp *= cooling

    return best_a, best_score


def coordinate_polish(a_init, time_limit, rng):
    """Coordinate descent with geometric search."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = local_score(a)
    cur_score = best_score
    t0 = time.monotonic()
    round_num = 0

    while time.monotonic() - t0 < time_limit:
        round_num += 1
        order = rng.permutation(n)
        improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            base = a[j]
            if base < 1e-10:
                continue

            best_factor = 1.0
            for factor in [0.6, 0.75, 0.85, 0.92, 1.0, 1.08, 1.15, 1.25, 1.4]:
                a[j] = base * factor
                s = local_score(a)
                if s < cur_score:
                    cur_score = s
                    best_factor = factor

            a[j] = base * best_factor
            if best_factor != 1.0:
                improved = True
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()

        if not improved and round_num > 1:
            # Perturb a random coordinate
            j = int(rng.integers(0, n))
            a[j] = max(0.0, a[j] * np.exp(rng.normal(0, 0.3)))
            cur_score = local_score(a)

    return best_a, best_score


def make_structured_init(n, rng, method='auto'):
    """Create structured initial sequences for diversity."""
    t = np.arange(n, dtype=np.float64)

    if method == 'auto':
        method = ['broadband', 'sine', 'two_peak', 'random', 'raised_cos'][int(rng.integers(0, 5))]

    if method == 'broadband':
        a = np.ones(n)
        for k in range(1, 12):
            freq = rng.uniform(0.3, 4.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.12 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    elif method == 'sine':
        a = np.sin(np.pi * t / n) ** 2 + 0.1
    elif method == 'two_peak':
        a = np.exp(-((t - n * 0.2) / (n * 0.1)) ** 2) + np.exp(-((t - n * 0.8) / (n * 0.1)) ** 2) + 0.1
    elif method == 'raised_cos':
        a = 0.5 * (1 + np.cos(2 * np.pi * t / n)) + 0.1
    else:
        a = np.abs(rng.standard_normal(n)) + 0.1

    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase LP-based search for minimizing the evaluation score.
    Phase 1: LP optimization from provided/best sequence
    Phase 2: LP optimization from diverse initializations
    Phase 3: Peak-shave polish
    Phase 4: Coordinate descent polish
    Phase 5: Final LP refinement
    """
    t_start = time.monotonic()
    total_budget = max(1.0, float(budget_s) - 25.0)
    deadline = t_start + total_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Try to load provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            a_prov = np.asarray(height_sequence_1, dtype=np.float64)
            a_prov = np.clip(a_prov, 0.0, 1000.0)
            if len(a_prov) > 0 and np.sum(a_prov) > 0.01:
                s_prov = local_score(a_prov)
                print(f"Provided sequence: n={len(a_prov)}, score={s_prov:.6f}")
                if s_prov < best_score:
                    best_score = s_prov
                    best_seq = a_prov.copy()
        except Exception as e:
            print(f"Warning: could not load provided sequence: {e}")

    # === PHASE 1: LP optimization on the best available sequence ===
    print(f"\n{'='*50}")
    print(f"PHASE 1: LP optimization (primary)")
    print(f"{'='*50}")

    if best_seq is not None:
        n_main = len(best_seq)
        t_alloc = 0.45 * total_budget
        t_alloc = min(t_alloc, deadline - time.monotonic() - 30)
        if t_alloc > 20:
            print(f"  n={n_main}, time={t_alloc:.0f}s, start_score={best_score:.6f}")
            res, sc = lp_optimize(best_seq, t_alloc, rng, verbose=True)
            if sc < best_score:
                best_score = sc
                best_seq = res.copy()
                print(f"  Phase 1 result: {best_score:.6f}")
            else:
                print(f"  Phase 1: no improvement (best remains {best_score:.6f})")
    else:
        # No provided sequence - start from scratch
        n_main = 384
        a0 = make_structured_init(n_main, rng, 'broadband')
        t_alloc = 0.3 * total_budget
        res, sc = lp_optimize(a0, t_alloc, rng, verbose=True)
        best_score = sc
        best_seq = res.copy()
        print(f"  Phase 1 (from scratch): {best_score:.6f}")

    # === PHASE 2: LP from diverse initializations ===
    print(f"\n{'='*50}")
    print(f"PHASE 2: LP from diverse starts")
    print(f"{'='*50}")

    n_candidates = [len(best_seq)] if best_seq is not None else [384]
    # Add a couple other lengths
    for nl in [256, 512]:
        if nl not in n_candidates:
            n_candidates.append(nl)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(0.12 * total_budget, deadline - time.monotonic() - 30)
        if t_alloc < 15:
            break

        # Initialize from best if possible, else random
        if best_seq is not None and len(best_seq) == n_try:
            a0 = best_seq + 0.05 * rng.standard_normal(n_try)
            a0 = np.clip(a0, 0.01, None)
        elif best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a0 = np.interp(x_new, x_old, best_seq)
        else:
            a0 = make_structured_init(n_try, rng, 'auto')

        print(f"  n={n_try}, time={t_alloc:.0f}s")
        res, sc = lp_optimize(a0, t_alloc, rng, verbose=False)
        if sc < best_score:
            best_score = sc
            best_seq = res.copy()
            print(f"  *** New best: {best_score:.6f} (n={n_try}) ***")
        else:
            print(f"  n={n_try}: best={sc:.6f} (no improvement)")

    # === PHASE 3: Peak-shave polish ===
    print(f"\n{'='*50}")
    print(f"PHASE 3: Peak-shave polish")
    print(f"{'='*50}")

    if best_seq is not None:
        t_alloc = min(0.12 * total_budget, deadline - time.monotonic() - 25)
        if t_alloc > 10:
            print(f"  time={t_alloc:.0f}s, start={best_score:.6f}")
            polished, sc = peak_shave_polish(best_seq, t_alloc, rng)
            if sc < best_score:
                best_score = sc
                best_seq = polished.copy()
                print(f"  After peak-shave: {best_score:.6f}")
            else:
                print(f"  Peak-shave: no improvement ({best_score:.6f})")

    # === PHASE 4: Coordinate descent ===
    print(f"\n{'='*50}")
    print(f"PHASE 4: Coordinate descent")
    print(f"{'='*50}")

    if best_seq is not None:
        t_alloc = min(0.08 * total_budget, deadline - time.monotonic() - 20)
        if t_alloc > 10:
            print(f"  time={t_alloc:.0f}s, start={best_score:.6f}")
            cd_res, sc = coordinate_polish(best_seq, t_alloc, rng)
            if sc < best_score:
                best_score = sc
                best_seq = cd_res.copy()
                print(f"  After CD: {best_score:.6f}")
            else:
                print(f"  CD: no improvement ({best_score:.6f})")

    # === PHASE 5: Final LP refinement ===
    print(f"\n{'='*50}")
    print(f"PHASE 5: Final LP refinement")
    print(f"{'='*50}")

    if best_seq is not None:
        t_alloc = max(5.0, deadline - time.monotonic() - 10)
        if t_alloc > 10:
            print(f"  time={t_alloc:.0f}s, start={best_score:.6f}")
            res, sc = lp_optimize(best_seq, t_alloc, rng, verbose=False)
            if sc < best_score:
                best_score = sc
                best_seq = res.copy()
                print(f"  Final LP: {best_score:.6f}")

    # === Verify and return ===
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'='*50}")
        return final_list

    # Fallback
    print("WARNING: No valid sequence found, returning uniform")
    return [1.0] * 256
