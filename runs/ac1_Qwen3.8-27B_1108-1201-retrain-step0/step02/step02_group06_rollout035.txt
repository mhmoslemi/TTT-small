Optimization for minimizing peak autocorrelation of non-negative sequences.
Uses Fienup spectral initialization, Adam with smooth max, and coordinate descent.

import time

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_of_normalized(a):
    """Score assuming sum(a) = 1. Returns 2n * max(conv)."""
    n = len(a)
    return 2.0 * n * np.max(fft_convolve(a, a))

def full_score(a):
    """Full score: 2n * max(conv) / sum^2."""
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    return 2.0 * len(a) * np.max(fft_convolve(a, a)) / (S * S)

def fienup_init(n, rng, n_iter=300):
    """Fienup/Gerchberg-Saxton initialization with flat target spectrum."""
    a = rng.uniform(0.5, 1.5, n)
    a = a / np.sum(a)
    
    for it in range(n_iter):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        target = np.sqrt(np.sum(a * a))  # Flat target: |A(f)| = sqrt(sum(a^2))
        # Update magnitude, keep phase
        with np.errstate(divide='ignore', invalid='ignore'):

    return a

    a = a_init.copy()

"""
Multi-phase optimization for minimizing peak convolution of non-negative sequences.
Uses Fienup spectral initialization, Adam gradient descent with soft-max smoothing,
and aggressive peak-targeted coordinate descent. Explores multiple sequence lengths.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_normalized(a):
    """Score = 2n * max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv)


def score_full(a):
    """Full score: 2n * max(conv) / (sum a)^2."""
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    n = len(a)
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def softmax(x):
    """Stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)


def fienup_init(n, rng, n_iter=250):
    """Fienup/Gerchberg-Saxton initialization targeting flat power spectrum."""
    a = rng.uniform(0.3, 1.5, n)
    a = a / np.sum(a)

    for it in range(n_iter):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        # Target: flat magnitude = sqrt(sum(a^2)) (will adjust after projection)
        target_mag = np.full_like(mag, np.sqrt(np.sum(a * a)))
        with np.errstate(divide='ignore', invalid='ignore'):
            phase = np.where(mag > 1e-15, A / mag, 1.0)
        A_new = phase * target_mag
        a_new = np.fft.irfft(A_new, n)
        a = np.maximum(a_new, 0.0)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def fienup_init_v2(n, rng, n_iter=200, floor=0.0):
    """Fienup with a floor to prevent too many zeros."""
    a = rng.uniform(0.5, 1.0, n)
    a = a / np.sum(a)

    for it in range(n_iter):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        target_mag = np.full_like(mag, np.sqrt(np.sum(a * a)))
        # Reduce target at DC to allow more variance
        target_mag[0] = np.sqrt(np.sum(a * a)) * 1.0
        with np.errstate(divide='ignore', invalid='ignore'):
            phase = np.where(mag > 1e-15, A / mag, 1.0)
        A_new = phase * target_mag
        a_new = np.fft.irfft(A_new, n).real
        a = np.maximum(a_new, floor)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def grad_soft_max(conv, a, tau):
    """
    Compute gradient of soft_max(conv) w.r.t. a.
    soft_max = max(conv) + tau * log(sum(exp((conv - max)/tau)))
    grad = 2 * cross_correlation(w, a) where w are soft-max weights.
    """
    n = len(a)
    c = conv / tau
    c -= np.max(c)
    exp_c = np.exp(c)
    w = exp_c / np.sum(exp_c)

    # cross_correlation(w, a)[j] = sum_k w[k] * a[k-j]
    # = IFFT(conj(FFT(w)) * FFT(a)) with zero-padding
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    W = np.fft.rfft(w, size)
    A_f = np.fft.rfft(a, size)
    cross = np.fft.irfft(np.conj(W) * A_f, size)[:n]
    return 2.0 * cross, w


def optimize_adam(n, a_init, time_limit, rng, verbose=True):
    """
    Adam gradient descent on simplex (softmax parameterization).
    Minimizes soft_max of linear convolution.
    """
    a = a_init.copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    theta = np.log(a + 1e-15)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.8

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_normalized(a)

    # Temperature schedule
    tau_start = 1.0 / n
    tau_end = 0.02 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Faster decay early, slower late
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.6)

        a = softmax(theta)
        conv = fft_convolve(a, a)
        grad_a, w = grad_soft_max(conv, a, tau)

        # Gradient w.r.t. theta via softmax chain rule
        inner = np.dot(a, grad_a)
        grad_theta = a * (grad_a - inner)

        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * grad_theta
        v_buf = beta2 * v_buf + (1 - beta2) * (grad_theta * grad_theta)
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)

        # Cosine LR schedule
        lr = base_lr * 0.5 * (1.0 + np.cos(np.pi * elapsed_frac))
        # Add small exploration noise early
        if elapsed_frac < 0.2:
            noise_scale = 0.01 * (1.0 - elapsed_frac / 0.2)
            grad_theta = grad_theta + noise_scale * rng.standard_normal(n)

        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if step % 30 == 0:
            a_eval = softmax(theta)
            sc = score_normalized(a_eval)
            if sc < best_score:
                best_score = sc
                best_a = a_eval.copy()
        if verbose and step % 3000 == 0:
            a_eval = softmax(theta)
            sc = score_normalized(a_eval)
            print(f"      Adam n={n} step={step} score={sc:.6f} best={best_score:.6f} tau={tau:.2e}")

    return best_a, best_score


def coordinate_descent(a_init, time_limit, rng, verbose=True):
    """
    Aggressive coordinate descent with peak-targeted moves.
    Works on simplex (sum=1, a>=0).
    """
    a = a_init.copy()
    a = np.clip(a, 0, None)
    a = a / np.sum(a)
    n = len(a)

    current_score = score_normalized(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    patience = 0

    # Precompute FFT size
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    while time.monotonic() - t0 < time_limit and patience < 8000:
        r = rng.random()

        if r < 0.35:
            # Peak-targeted: reduce elements contributing most to peak
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n, peak_idx + 1)
            for i in range(lo, hi):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(4, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate = a.copy()
            red = rng.uniform(0.03, 0.12)
            candidate[top_idx] *= (1.0 - red)
            removed = np.sum(a[top_idx] - candidate[top_idx])
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_add = min(15, len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                candidate[add_idx] += removed / n_add
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs

        elif r < 0.55:
            # Multi-peak targeted: reduce top 2 peaks' contributors
            conv = fft_convolve(a, a)
            peaks = np.argsort(conv)[-2:]
            candidate = a.copy()
            for pk in peaks:
                contribs = np.zeros(n)
                lo = max(0, pk - n + 1)
                hi = min(n, pk + 1)
                for i in range(lo, hi):
                    j = pk - i
                    if 0 <= j < n:
                        contribs[i] = candidate[i] * candidate[j]
                top_k = max(3, n // 20)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= 0.92
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs

        elif r < 0.75:
            # Pair transfer
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            amt = a[i] * rng.uniform(0.02, 0.15)
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
            candidate = np.clip(candidate, 0, None)
            candidate = candidate / np.sum(candidate)

        elif r < 0.90:
            # Block perturbation
            width = max(1, n // int(rng.integers(10, 40)))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.06, width))
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs

        else:
            # Sparse Gaussian perturbation
            candidate = a.copy()
            n_pert = max(3, n // 24)
            idx = rng.choice(n, size=n_pert, replace=False)
            candidate[idx] *= np.exp(rng.normal(0, 0.04, n_pert))
            candidate = np.clip(candidate, 0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate = candidate / cs

        new_score = score_normalized(candidate)
        if new_score < current_score:
            a = candidate
            current_score = new_score
            patience = 0
        else:
            patience += 1

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

        if verbose and (best_score < 1.51 or patience % 5000 == 0):
            print(f"      CD best={best_score:.6f} patience={patience} t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def spectral_refine(n, a_init, time_limit, rng):
    """
    Iterative spectral refinement: alternate between time-domain projection
    and frequency-domain amplitude adjustment. More aggressive than Fienup.
    """
    a = a_init.copy()
    a = a / np.sum(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_normalized(a)

    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a)
        mag = np.abs(A)
        phase = np.where(mag > 1e-15, A / (mag + 1e-30), 1.0)

        # Target: slightly below flat to reduce DC peak
        target = np.sqrt(np.sum(a * a))
        # Reduce DC component target
        target[0] /= 1.1

        A_new = phase * target
        a_new = np.fft.irfft(A_new, n).real
        a = np.maximum(a_new, 0.0)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
            break
        a = a / s

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: Fienup init -> spectral refine -> Adam -> coordinate descent.
    Tries multiple lengths and starting points.
    """
    t_start = time.monotonic()

    rng = np.random.default_rng(seed)

    best_seq = None

    def update_best(a, label=""):

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** New best: {sc:.6f} ({label}) ***")

    # Check provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s0 = score_full(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s0:.6f}")
        if s0 < best_score:
            best_score = s0
            best_seq = start_seq.copy()
            print(f"  *** New best from provided: {best_score:.6f} ***")

    # Define configurations: (length, n_fienup_iter, adam_frac, cd_frac)
    configs = [
        (512, 200, 0.40, 0.30),
        (512, 300, 0.35, 0.35),
        (768, 200, 0.40, 0.30),
        (1024, 200, 0.40, 0.30),
        (1536, 150, 0.45, 0.25),
        (2048, 100, 0.50, 0.20),
    ]

    total_configs = len(configs)
    for ci, (n, nf_iter, adam_frac, cd_frac) in enumerate(configs):
        if time.monotonic() > deadline - 45:
            break

        remaining = deadline - time.monotonic() - 30
        if remaining < 30:
            break

        print(f"\n=== Config {ci+1}/{total_configs}: n={n} ===")

        # Allocation
        spec_time = min(30.0, remaining * 0.05)
        adam_time = remaining * adam_frac * 0.6
        cd_time = remaining * cd_frac * 0.6
        spec_time = min(spec_time, remaining * 0.1)

        # Phase 1: Fienup + spectral refine
        a_init = fienup_init(n, rng, n_iter=min(nf_iter, 300))
        sc_init = score_normalized(a_init)
        print(f"  Fienup init: {sc_init:.6f}")

        if spec_time > 5:
            a_spec, sc_spec = spectral_refine(n, a_init, spec_time, rng)
            print(f"  Spectral refine: {sc_spec:.6f}")
            update_best(a_spec, f"spec n={n}")
            a_init = a_spec

        # Phase 2: Adam optimization
        if time.monotonic() < deadline - 30:
            adam_time = min(adam_time, deadline - time.monotonic() - 30)
            a_opt, sc_opt = optimize_adam(n, a_init, adam_time, rng, verbose=(n <= 1024))
            print(f"  Adam: {sc_opt:.6f}")
            update_best(a_opt, f"adam n={n}")

        # Phase 3: Coordinate descent
        if time.monotonic() < deadline - 20:
            cd_time = min(cd_time, deadline - time.monotonic() - 20)
            a_cd, sc_cd = coordinate_descent(a_opt, cd_time, rng, verbose=(n <= 1024))
            print(f"  CD: {sc_cd:.6f}")
            update_best(a_cd, f"cd n={n}")

    # Final polish on best
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n=== Final polish ===")
        polish_time = min(deadline - time.monotonic() - 15, 120.0)
        if polish_time > 5:
            a_final, sc_final = coordinate_descent(best_seq, polish_time, rng, verbose=True)
            if sc_final < best_score:
                best_score = sc_final

                print(f"  *** Final polish improved to: {best_score:.6f} ***")

    # Convert to output format

        # Scale so values are in reasonable range (score is scale-invariant)
        a_out = best_seq * np.sqrt(2.0 * len(best_seq))
        result = [float(max(0.0, min(1000.0, x))) for x in a_out]
        final_score = evaluate_sequence(result)
        print(f"\n=== FINAL: n={len(result)}, score={final_score:.6f} ===")
        return result

    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
