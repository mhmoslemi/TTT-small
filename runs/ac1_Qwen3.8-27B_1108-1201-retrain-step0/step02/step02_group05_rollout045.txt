import time

from scipy.optimize import minimize

    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)

    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

"""
LP-based optimization for minimum peak autocorrelation non-negative sequences.
Combines linearized LP iterations (with line search) and L-BFGS p-norm annealing.
The LP approach directly targets the max autocorrelation constraint by linearizing
the quadratic form around the current point, solving a linear program, and using
damped updates with line search for stability.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def compute_ka(a, k):
    """(C_k a)_i = a[k-i] if 0 <= k-i < n, else 0."""
    n = len(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    result = np.zeros(n)
    result[mask] = a[j[mask]]
    return result


def solve_lp(a, active_set):
    """Solve the linearized LP: min t s.t. 2*(C_k a*)^T a <= t + conv(a*,a*)[k]."""
    n = len(a)
    a_new = cp.Variable(n)
    t = cp.Variable()
    constraints = [cp.sum(a_new) == 1, a_new >= 0]

    for k in active_set:
        ka = compute_ka(a, k)
        conv_k = float(np.dot(ka, a))
        constraints.append(2.0 * (ka @ a_new) <= t + conv_k)

    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue

    if a_new.value is not None:
        result = np.maximum(a_new.value, 0.0)
        s = np.sum(result)
        if s > 1e-10:
            result /= s
            return result
    return None


def gradient_step(a, active_set, eta):
    """Gradient descent step on the simplex for the max autocorrelation."""
    n = len(a)
    g = np.zeros(n)
    for k in active_set:
        g += compute_ka(a, k)
    g -= np.mean(g)
    a_new = a - eta * g
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s

        a_try[bottom_idx] *= np.exp(rng.normal(0.01, 0.005, size=len(bottom_idx)))
        a_try = np.maximum(a_try, 1e-10)
        a_try /= np.sum(a_try)

        new_score = local_score(a_try)
        if new_score < best_score:
            best_score = new_score
            best_a = a_try.copy()
            a = a_try
        elif rng.random() < np.exp(-(new_score - local_score(a)) / max(temp, 1e-15)):
            a = a_try
            a /= np.sum(a)

        temp *= 0.9995

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: L-BFGS p-norm initialization, LP-based optimization
    with linearized constraints and line search, fine-tuning local search.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 60.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        best_score = s
        best_seq = start_seq.copy()
    else:
        n = 256
        best_seq = np.ones(n) / n
        best_score = local_score(best_seq)
        print(f"Starting from uniform: n={n}, score={best_score:.6f}")

    # Phase 1: L-BFGS with p-norm annealing
    print(f"\n=== Phase 1: L-BFGS p-norm (score={best_score:.6f}) ===")
    t1_deadline = min(t_global + 400, deadline - 300)
    a_lb, s_lb = run_lbgfs(best_seq, t1_deadline)
    if s_lb < best_score:
        best_score = s_lb
        best_seq = a_lb.copy()
        print(f"  L-BFGS improved: {best_score:.6f} ***")
    else:
        print(f"  L-BFGS: {s_lb:.6f} (no improvement)")

    # Phase 2: LP optimization (main workhorse)
    print(f"\n=== Phase 2: LP optimization (score={best_score:.6f}) ===")
    t2_limit = min(900.0, deadline - time.monotonic() - 300)
    a_lp, s_lp = lp_optimize(best_seq, max_iters=2000, active_frac=0.995,
                             time_limit=t2_limit)
    if s_lp < best_score:
        best_score = s_lp
        best_seq = a_lp.copy()
        print(f"  LP improved: {best_score:.6f} ***")
    else:
        print(f"  LP: {s_lp:.6f} (no improvement)")

    # Phase 3: LP with tighter active set (focus on top peaks only)
    print(f"\n=== Phase 3: LP tight active set (score={best_score:.6f}) ===")
    t3_limit = min(400.0, deadline - time.monotonic() - 150)
    a_lp2, s_lp2 = lp_optimize(best_seq, max_iters=1000, active_frac=0.999,
                               time_limit=t3_limit)
    if s_lp2 < best_score:
        best_score = s_lp2
        best_seq = a_lp2.copy()
        print(f"  LP tight improved: {best_score:.6f} ***")
    else:
        print(f"  LP tight: {s_lp2:.6f} (no improvement)")

    # Phase 4: Another L-BFGS pass from LP result
    if time.monotonic() < deadline - 200:
        print(f"\n=== Phase 4: L-BFGS refinement (score={best_score:.6f}) ===")
        t4_deadline = min(time.monotonic() + 200, deadline - 80)
        a_lb2, s_lb2 = run_lbgfs(best_seq, t4_deadline)
        if s_lb2 < best_score:
            best_score = s_lb2
            best_seq = a_lb2.copy()
            print(f"  L-BFGS ref improved: {best_score:.6f} ***")
        else:
            print(f"  L-BFGS ref: {s_lb2:.6f} (no improvement)")

        # Another LP pass
        if time.monotonic() < deadline - 100:
            t5_limit = min(150.0, deadline - time.monotonic() - 50)
            a_lp3, s_lp3 = lp_optimize(best_seq, max_iters=500, active_frac=0.995,
                                       time_limit=t5_limit)
            if s_lp3 < best_score:
                best_score = s_lp3
                best_seq = a_lp3.copy()
                print(f"  Final LP improved: {best_score:.6f} ***")

    # Phase 5: Fine-tune
    if time.monotonic() < deadline - 30:
        print(f"\n=== Phase 5: Fine-tune (score={best_score:.6f}) ===")
        ft_limit = min(60.0, deadline - time.monotonic() - 20)
        a_ft, s_ft = fine_tune(best_seq, time_limit=ft_limit)
        if s_ft < best_score:
            best_score = s_ft
            best_seq = a_ft.copy()
            print(f"  Fine-tune improved: {best_score:.6f} ***")
        else:
            print(f"  Fine-tune: {s_ft:.6f} (no improvement)")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    return [1.0] * 256
