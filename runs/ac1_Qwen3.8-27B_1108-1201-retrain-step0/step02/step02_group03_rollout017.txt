LP-guided iterative optimization for minimizing peak autocorrelation of
non-negative step functions. Combines linear programming descent directions,

The core innovation is using LP to find guaranteed descent directions that
reduce the maximum autocorrelation lag while preserving non-negativity.

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_constraint_matrix(a):
    """Build the (2n-1) x n constraint matrix for the LP.
    M[j, k] = a[j-k] if 0 <= j-k < n, else 0.

    n = len(a)
    M = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):
        k_min = max(0, j - n + 1)
        k_max = min(n - 1, j)
        if k_min <= k_max:
            M[j, k_min:k_max + 1] = a[j - k_max:j - k_min + 1][::-1]
    return M


def solve_direction_lp(a, M_val, active_frac=1.0):
    """Solve LP to find direction g maximizing sum(g) s.t. (a*g) <= M_val, g >= 0.
    Uses active constraint subset for speed when active_frac < 1.
    Returns the direction g or None if infeasible.
    """

    conv = fft_convolve(a, a)
    M_actual = np.max(conv)

    if active_frac < 1.0:
        threshold = M_actual * (1.0 - active_frac)
        active = np.where(conv >= threshold)[0]
        if len(active) < 2:

    else:

        if active_frac < 1.0:

        # Try the step
        a_new = (1.0 - current_t) * a + current_t * g
        a_new = np.clip(a_new, 0.0, None)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new * S

        score_new = local_score(a_new)

        if score_new < best_score - 1e-10:
            best_score = score_new
            best_a = a_new.copy()
            stale = 0
        else:
            stale += 1

        a = a_new
        iter_count += 1

        # Adaptive step size
        if stale > max_stale:
            current_t = max(t_min, current_t * 0.7)
            active_frac = min(1.0, active_frac + 0.1)
            # Add small perturbation to escape
            a = a * np.exp(np.random.default_rng(iter_count).normal(0, 0.01, n))
            a = np.clip(a, 0.0, None)
            a = a / np.sum(a) * S
            stale = 0
            if current_t <= t_min * 1.1:
                break
        elif stale > max_stale // 2:
            current_t = max(t_min, current_t * 0.85)

        if iter_count % 100 == 0:
            cur_score = local_score(a)
            elapsed = time.monotonic() - t0
            print(f"      LP iter {iter_count}: score={cur_score:.6f}, "
                  f"best={best_score:.6f}, M={M:.4f}, t={current_t:.4f}, "
                  f"elapsed={elapsed:.1f}s")

    final_score = local_score(best_a)
    if final_score < best_score:
        best_score = final_score
    return best_a, best_score


def fft_phase_init(n, rng, n_iter=80):
    """Gerchberg-Saxton style initialization for flat autocorrelation."""
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1

    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def chirp_init(n, rng, alpha=None):
    """Chirp-like initialization."""
    if alpha is None:
        alpha = 0.3 + 0.4 * rng.random()
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def multitone_init(n, rng, n_tones=5):
    """Multi-tone sinusoidal initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(n_tones):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.5 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def smooth_init(n, rng):
    """Initialize with a smooth bump function."""
    t = np.linspace(0, 1, n)
    center = 0.3 + 0.4 * rng.random()
    width = 0.2 + 0.3 * rng.random()
    a = np.exp(-((t - center) ** 2) / (2 * width ** 2))
    a += 0.1 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def soft_max_and_weights(x, tau):
    """Compute soft-max value and normalized weights."""
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    return x_max + tau * np.log(s), e / s


def grad_soft_max_conv(a, tau):
    """Compute soft-max(conv(a,a)) and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_and_weights(c, tau)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = (2.0 / tau) * corr[n - 1:2 * n - 1]
    return value, g, c


def gradient_optimize(n, a_init, time_limit, rng):
    """Adam gradient descent on log-parameterized sequence."""
    S = np.sqrt(2.0 * n)
    a = np.clip(a_init, 1e-10, None)
    a = a / np.sum(a) * S
    theta = np.log(a / S + 1e-12)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr0 = 3.0
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)
    tau_start = 50.0 / n
    tau_end = 0.2 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)

        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)

        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 500 == 0:
            a_check = softmax_normalize(theta) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
            if step % 2000 == 0:
                print(f"        grad step={step}, score={score:.6f}, "
                      f"best={best_score:.6f}, tau={tau:.2e}")

    a_final = softmax_normalize(best_theta) * S
    score_final = local_score(a_final)
    return a_final, min(best_score, score_final)


def local_search_polish(a_init, time_limit, rng):
    """Simulated annealing with peak-directed perturbations."""
    current = np.clip(a_init.astype(np.float64).copy(), 0.0, None)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    S = np.sum(current)
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9997
    top_k = 6

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.4:
            # Peak-directed: reduce contributions to top lags
            conv = fft_convolve(current, current)
            top_lags = np.argsort(conv)[-top_k:]
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            n_reduce = max(1, n // 12)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.015, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.015, size=len(bot_idx)))
        elif r < 0.6:
            # Block perturbation
            n_blocks = rng.integers(1, 4)
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 15))))
                bs = int(rng.integers(0, n - bw))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0.0, 0.04, size=bw))
        elif r < 0.8:
            # Symmetry nudge
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.1 * (sym - current)
            candidate += 0.005 * rng.standard_normal(n)
        else:
            # Global small noise
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / s * S

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 5:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    current = candidate
                    current_score = new_score

        temp *= cooling

    return best, best_score


def upsample(a, new_n, rng):
    """Upsample a sequence via interpolation + noise."""
    n = len(a)
    if new_n <= n:
        return a[:new_n].copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-method search for optimal non-negative step function minimizing
    2*n*max(conv(a,a))/sum(a)^2. Uses LP descent, gradient optimization,
    and local search with diverse initializations.
    """
    t_global = time.monotonic()
    deadline = t_global + max(0.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        start_seq = start_seq[~np.isnan(start_seq)]
        s_prov = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = start_seq.copy()

    # === Phase 1: Generate diverse initializations ===
    print("\n=== Phase 1: Initialization ===")
    candidates = []
    lengths_to_try = [192, 256, 320, 384, 512, 768, 1024]

    for n in lengths_to_try:
        if time.monotonic() > deadline - 600:
            break
        inits = []
        inits.append(('fft60', fft_phase_init(n, rng, n_iter=60)))
        inits.append(('fft120', fft_phase_init(n, rng, n_iter=120)))
        inits.append(('chirp', chirp_init(n, rng)))
        inits.append(('chirp2', chirp_init(n, rng, alpha=0.1 + 0.8 * rng.random())))
        inits.append(('multitone', multitone_init(n, rng, n_tones=3 + rng.integers(0, 4))))
        inits.append(('smooth', smooth_init(n, rng)))
        inits.append(('random', random_init(n, rng, spread=0.2 + 0.4 * rng.random())))

        for name, a in inits:
            score = local_score(a)
            candidates.append((score, n, a.copy()))
            if n == lengths_to_try[0]:
                print(f"  n={n}, {name}: {score:.6f}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Best init: score={candidates[0][0]:.6f}, n={candidates[0][1]}")

    # === Phase 2: LP-based optimization on top candidates ===
    print("\n=== Phase 2: LP Optimization ===")

    # Build list of sequences to optimize
    opt_seqs = []
    if best_seq is not None:
        opt_seqs.append(('provided', best_seq.copy(), len(best_seq)))
    for i, (sc, n, a) in enumerate(candidates[:6]):
        opt_seqs.append((f'cand{i}', a.copy(), n))

    # Deduplicate by length (keep best per length)
    seen_lengths = {}
    deduped = []
    for name, a, n in opt_seqs:
        if n not in seen_lengths or local_score(a) < seen_lengths[n]:
            seen_lengths[n] = local_score(a)
            deduped.append((name, a, n))
    opt_seqs = deduped

    # Sort by initial score
    opt_seqs.sort(key=lambda x: local_score(x[1]))

    n_opt = len(opt_seqs)
    lp_time_total = min(1200.0, (deadline - time.monotonic() - 300.0) * 0.6)
    lp_time_per = max(20.0, lp_time_total / max(1, n_opt))

    for i, (name, a, n) in enumerate(opt_seqs):
        if time.monotonic() > deadline - 200:
            break
        s0 = local_score(a)
        t_alloc = min(lp_time_per * 1.3, deadline - time.monotonic() - 150)
        # Scale time for larger n (LP is slower)
        t_alloc *= min(1.0, 512.0 / max(1, n))
        t_alloc = max(15.0, t_alloc)

        print(f"  [{i+1}/{n_opt}] {name}, n={n}, score0={s0:.6f}, t={t_alloc:.0f}s")
        a_opt, s_opt = lp_optimize(a, t_alloc)
        print(f"    LP result: {s_opt:.6f}")

        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")

    # === Phase 3: Gradient optimization on best LP results ===
    print("\n=== Phase 3: Gradient Optimization ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Try gradient on the best sequence and a few variants
        grad_candidates = [best_seq.copy()]
        # Add perturbed versions
        for _ in range(2):
            pert = best_seq * np.exp(rng.normal(0, 0.05, n_best))
            pert = np.clip(pert, 0.001, None)
            grad_candidates.append(pert / np.sum(pert) * np.sum(best_seq))

        # Also try gradient on best init
        if candidates:
            grad_candidates.append(candidates[0][2].copy())

        grad_time_total = min(400.0, (deadline - time.monotonic() - 200.0) * 0.5)
        grad_time_per = max(15.0, grad_time_total / len(grad_candidates))

        for i, a in enumerate(grad_candidates):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(grad_time_per, deadline - time.monotonic() - 100)
            t_alloc = max(10.0, t_alloc * min(1.0, 512.0 / len(a)))
            print(f"  Grad [{i+1}/{len(grad_candidates)}], n={len(a)}, t={t_alloc:.0f}s")
            a_grad, s_grad = gradient_optimize(len(a), a, t_alloc, rng)
            print(f"    Grad result: {s_grad:.6f}")
            if s_grad < best_score:
                best_score = s_grad
                best_seq = a_grad.copy()
                print(f"    *** New best: {best_score:.6f} (n={len(a)}) ***")

    # === Phase 4: Coarse-to-fine upsample ===
    print("\n=== Phase 4: Upsampling ===")
    if best_seq is not None:
        n_curr = len(best_seq)
        target_n = n_curr * 2
        if target_n <= 2048 and time.monotonic() < deadline - 180:
            a_up = upsample(best_seq, target_n, rng)
            t_alloc = min(120.0, deadline - time.monotonic() - 120)
            if t_alloc > 20:
                print(f"  Upsample n={n_curr} -> {target_n}, t={t_alloc:.0f}s")
                a_opt, s_opt = lp_optimize(a_up, t_alloc * 0.5)
                a_opt2, s_opt2 = gradient_optimize(target_n, a_opt, t_alloc * 0.5, rng)
                s_final = min(s_opt, s_opt2)
                a_final = a_opt if s_opt <= s_opt2 else a_opt2
                print(f"    Upsample result: {s_final:.6f}")
                if s_final < best_score:
                    best_score = s_final
                    best_seq = a_final.copy()
                    print(f"    *** New best: {best_score:.6f} (n={target_n}) ***")

    # === Phase 5: Local search polish ===
    print("\n=== Phase 5: Local Search Polish ===")
    if best_seq is not None:
        polish_time = max(30.0, min(300.0, deadline - time.monotonic() - 30.0))
        print(f"  Polishing n={len(best_seq)}, score={best_score:.6f}, t={polish_time:.0f}s")
        polished, p_score = local_search_polish(best_seq, polish_time, rng)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")

    # === Phase 6: Final LP pass if time remains ===
    remaining = deadline - time.monotonic() - 40.0
    if remaining > 60 and best_seq is not None:
        print(f"\n=== Phase 6: Final LP Pass ({remaining:.0f}s) ===")
        a_final_lp, s_final_lp = lp_optimize(best_seq, min(remaining * 0.7, 300.0))
        if s_final_lp < best_score:
            best_score = s_final_lp
            best_seq = a_final_lp.copy()
            print(f"  After final LP: {best_score:.6f}")
        # Quick polish
        if deadline - time.monotonic() > 30:
            p3, s3 = local_search_polish(best_seq, min(60.0, deadline - time.monotonic() - 30.0), rng)
            if s3 < best_score:
                best_score = s3
                best_seq = p3.copy()
                print(f"  After final polish: {best_score:.6f}")

    # === Final verification and return ===
    print(f"\n{'='*50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Verify no NaN/inf
        final_list = [x for x in final_list if np.isfinite(x)]
        if not final_list:
            final_list = [1.0] * 256
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        if final_score < best_score:
            best_score = final_score
        return final_list
    else:
        # Fallback
        fallback = [1.0] * 256
        print(f"=== FALLBACK: returning uniform n=256 ===")
        return fallback
