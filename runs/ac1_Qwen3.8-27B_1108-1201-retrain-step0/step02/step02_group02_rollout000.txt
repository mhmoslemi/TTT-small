"""
LP-guided peak optimization for non-negative step functions.
Uses a linear programming approach to find directions that reduce the
peak autocorrelation, combined with gradient-based local polish and
multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def build_conv_matrix(f):
    """
    Build matrix A such that A @ g = f * g (linear convolution).
    A has shape (2n-1, n), where A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return A


def solve_lp(A, M, n):
    """
    Solve: maximize sum(g) subject to A@g <= M, g >= 0.
    Returns g or None if infeasible.
    """
    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success and result.x is not None:
        return np.maximum(result.x, 0.0)
    return None


def lp_optimize(f_init, n, time_limit, rng, verbose=True):
    """
    Main LP-based optimization loop.
    Repeatedly finds LP direction and takes line-searched steps.
    """
    S_target = np.sqrt(2.0 * n)
    f = f_init.astype(np.float64).copy()
    f = np.clip(f, 0.0, 1000.0)
    S = np.sum(f)
    if S < 0.01:
        f = np.ones(n) / n * S_target
    else:
        f = f / S * S_target

    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    patience = 50

    # Pre-compute matrix (will be rebuilt each iteration since f changes)
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break

        c = fft_convolve(f, f)
        M = np.max(c)

        # Periodically try tighter constraint for stronger direction
        if iteration % 20 == 0:
            M_eff = M * 0.995
        else:
            M_eff = M

        # Solve LP
        A = build_conv_matrix(f)
        g = solve_lp(A, M_eff, n)

        if g is None or np.sum(g) < 1e-10:
            if verbose and iteration % 20 == 0:
                print(f"    LP iter={iteration}, infeasible, score={best_score:.6f}, t={elapsed:.0f}s")
            no_improve += 1
            if no_improve > patience:
                # Perturb and reset
                f = best_f + 0.01 * rng.standard_normal(n) * S_target
                f = np.clip(f, 0, 1000)
                f = f / np.sum(f) * S_target
                no_improve = 0
            continue

        # Rescale g to have same sum as f
        g = g / np.sum(g) * S_target

        # Line search
        best_t = 0.0
        best_f_trial = f.copy()
        best_trial_score = best_score

        # Try a range of step sizes
        t_candidates = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]
        for t in t_candidates:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.clip(f_trial, 0.0, 1000.0)
            s = local_score(f_trial)
            if s < best_trial_score:
                best_trial_score = s
                best_t = t
                best_f_trial = f_trial.copy()

        if best_t > 0 and best_trial_score < best_score:
            f = best_f_trial.copy()
            if best_trial_score < best_score:
                best_score = best_trial_score
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Try a random perturbation to escape
            if no_improve > patience // 2:
                f = best_f * (1 + 0.05 * rng.standard_normal(n))
                f = np.clip(f, 0, 1000)
                f = f / np.sum(f) * S_target
                no_improve = 0

        if verbose and iteration % 25 == 0:
            print(f"    LP iter={iteration}, score={best_score:.6f}, t_step={best_t:.3f}, "
                  f"elapsed={elapsed:.0f}s, remaining={remaining:.0f}s")

    return best_f, best_score


def gradient_polish(f_init, time_limit, rng, verbose=True):
    """
    Gradient-based polish using softmax parameterization and Adam.
    Focuses on fine-tuning the sequence to reduce peak further.
    """
    n = len(f_init)
    S_target = np.sqrt(2.0 * n)

    # Initialize theta from f_init
    f_safe = np.maximum(f_init, 1e-10)
    theta = np.log(f_safe / np.sum(f_safe))

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step_count = 0

    best_f = f_init.astype(np.float64).copy()
    best_score = local_score(best_f)
    no_improve = 0
    patience = 2000

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        elapsed_frac = 1.0 - remaining / time_limit

        # Adaptive temperature
        if elapsed_frac < 0.5:
            tau = (20.0 / n) * (0.01 / n) ** (elapsed_frac * 2)
        else:
            tau = (0.01 / n) * (0.001 / n) ** ((elapsed_frac - 0.5) * 2)

        # Compute softmax
        f = np.exp(theta - np.max(theta))
        f = f / np.sum(f) * S_target

        # Compute soft-max of convolution
        c = fft_convolve(f, f)
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted - np.max(shifted))
        sum_exp = np.sum(exp_shifted)
        w = exp_shifted / sum_exp

        # Gradient in f-space
        rev_f = f[::-1]
        corr = fft_convolve(rev_f, w)
        g_f = 2.0 * corr[n - 1:2 * n - 1]

        # Gradient in theta-space (simple approximation)
        g_theta = f * (g_f - np.dot(f, g_f) / S_target)

        # Adam step
        step_count += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g_theta
        v_buf = beta2 * v_buf + (1 - beta2) * g_theta ** 2
        m_hat = m_buf / (1.0 - beta1 ** step_count)
        v_hat = v_buf / (1.0 - beta2 ** step_count)

        lr = 1.0 / np.sqrt(step_count + 1) * (1.0 if elapsed_frac < 0.5 else 0.3)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic evaluation
        if step_count % 100 == 0:
            f_check = np.exp(theta - np.max(theta))
            f_check = f_check / np.sum(f_check) * S_target
            f_check = np.clip(f_check, 0, 1000)
            score = local_score(f_check)
            if score < best_score:
                best_score = score
                best_f = f_check.copy()
                no_improve = 0
            else:
                no_improve += 100

            if no_improve > patience:
                # Escape: perturbation
                theta = np.log(np.maximum(best_f, 1e-10) / np.sum(best_f)) + 0.1 * rng.standard_normal(n)
                no_improve = 0

            if verbose and step_count % 2000 == 0:
                print(f"    Grad step={step_count}, score={best_score:.6f}, "
                      f"tau={tau:.2e}, elapsed={time.monotonic()-t0:.0f}s")

    return best_f, best_score


def make_structured_init(n, rng, method='broadband'):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)

    if method == 'broadband':
        a = np.ones(n)
        n_freqs = min(20, n // 4)
        for k in range(1, n_freqs + 1):
            freq = k * (1.0 + 0.5 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.12 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'cosine_mod':
        # Cosine modulation with multiple frequencies
        a = np.ones(n)
        for _ in range(3):
            f1 = rng.uniform(1, 5)
            f2 = rng.uniform(5, 15)
            a += 0.1 * np.cos(2 * np.pi * f1 * t / n) * np.cos(2 * np.pi * f2 * t / n)
        return np.clip(a, 0.01, None)
    else:
        a = 1.0 + 0.2 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)


def coordinate_descent_polish(f_init, time_limit, rng):
    """
    Coordinate descent: adjust individual elements to reduce peak.
    """
    current = f_init.astype(np.float64).copy()
    n = len(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    round_num = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break
        round_num += 1
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit - 1:
                break
            base = current[j]
            if base < 1e-8:
                continue

            # Try multiplicative adjustments
            found_better = False
            for frac in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                trial = base * frac
                current[j] = trial
                new_score = local_score(current)
                if new_score < current_score:
                    current_score = new_score
                    found_better = True
                    improved = True
                    if current_score < best_score:
                        best_score = current_score
                        best = current.copy()
                else:
                    current[j] = base
                    break

            if not found_better:
                current[j] = base

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase LP-guided search for minimizing the evaluation score.
    Phase 1: LP optimization at multiple lengths
    Phase 2: Gradient polish
    Phase 3: Coordinate descent
    Phase 4: Final exploration
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    def try_sequence(f, label=""):
        nonlocal best_score, best_seq
        s = local_score(f)
        if s < best_score:
            best_score = s
            best_seq = f.copy()
            print(f"  *** New best ({label}): n={len(f)}, score={best_score:.6f} ***")
        else:
            print(f"  ({label}): n={len(f)}, score={s:.6f} (no improvement)")

    # === PHASE 1: LP optimization at primary lengths ===
    print(f"\n{'='*60}")
    print(f"PHASE 1: LP-guided optimization")
    print(f"{'='*60}")

    # Primary: n=384 (from provided sequence)
    n_primary = len(start_seq) if start_seq is not None else 384

    # Use provided sequence as starting point
    if start_seq is not None and len(start_seq) == n_primary:
        f_start = start_seq.copy()
    else:
        f_start = make_structured_init(n_primary, rng, 'broadband')

    t_lp1 = 0.30 * time_budget
    print(f"\n  LP optimization at n={n_primary}, budget={t_lp1:.0f}s")
    f_lp1, s_lp1 = lp_optimize(f_start, n_primary, t_lp1, rng)
    try_sequence(f_lp1, "LP-primary")

    # Secondary: try n=512
    if time.monotonic() < deadline - 120:
        n_sec = 512
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_sec)
            f_sec = np.interp(x_new, x_old, best_seq)
        else:
            f_sec = make_structured_init(n_sec, rng, 'broadband')

        t_lp2 = min(0.20 * time_budget, deadline - time.monotonic() - 100)
        if t_lp2 > 30:
            print(f"\n  LP optimization at n={n_sec}, budget={t_lp2:.0f}s")
            f_lp2, s_lp2 = lp_optimize(f_sec, n_sec, t_lp2, rng)
            try_sequence(f_lp2, "LP-sec512")

    # Tertiary: try n=256 with different init
    if time.monotonic() < deadline - 100:
        n_ter = 256
        f_ter = make_structured_init(n_ter, rng, 'cosine_mod')
        t_lp3 = min(0.15 * time_budget, deadline - time.monotonic() - 80)
        if t_lp3 > 20:
            print(f"\n  LP optimization at n={n_ter}, budget={t_lp3:.0f}s")
            f_lp3, s_lp3 = lp_optimize(f_ter, n_ter, t_lp3, rng)
            try_sequence(f_lp3, "LP-ter256")

    # === PHASE 2: Gradient polish at best length ===
    print(f"\n{'='*60}")
    print(f"PHASE 2: Gradient polish")
    print(f"{'='*60}")

    if best_seq is not None:
        t_grad = min(0.15 * time_budget, max(60.0, deadline - time.monotonic() - 100))
        if t_grad > 30:
            print(f"\n  Gradient polish at n={len(best_seq)}, budget={t_grad:.0f}s, "
                  f"start score={best_score:.6f}")
            f_grad, s_grad = gradient_polish(best_seq, t_grad, rng)
            try_sequence(f_grad, "grad-polish")

    # === PHASE 3: Coordinate descent ===
    print(f"\n{'='*60}")
    print(f"PHASE 3: Coordinate descent")
    print(f"{'='*60}")

    if best_seq is not None:
        t_cd = min(0.10 * time_budget, max(30.0, deadline - time.monotonic() - 50))
        if t_cd > 20:
            print(f"\n  CD at n={len(best_seq)}, budget={t_cd:.0f}s")
            f_cd, s_cd = coordinate_descent_polish(best_seq, t_cd, rng)
            try_sequence(f_cd, "CD")

    # === PHASE 4: Additional exploration ===
    print(f"\n{'='*60}")
    print(f"PHASE 4: Additional exploration")
    print(f"{'='*60}")

    # Try more LP rounds at best length
    if best_seq is not None and time.monotonic() < deadline - 60:
        t_extra_lp = min(0.08 * time_budget, deadline - time.monotonic() - 40)
        if t_extra_lp > 15:
            print(f"\n  Extra LP at n={len(best_seq)}, budget={t_extra_lp:.0f}s")
            f_el, s_el = lp_optimize(best_seq, len(best_seq), t_extra_lp, rng)
            try_sequence(f_el, "LP-extra")

    # Try n=448 or n=640
    for n_try in [448, 640, 768]:
        if time.monotonic() > deadline - 40:
            break
        t_try = min(30.0, deadline - time.monotonic() - 30)
        if t_try < 10:
            break

        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            f_try = np.interp(x_new, x_old, best_seq)
        else:
            f_try = make_structured_init(n_try, rng, 'broadband')

        print(f"\n  LP at n={n_try}, budget={t_try:.0f}s")
        f_res, s_res = lp_optimize(f_try, n_try, t_try, rng)
        try_sequence(f_res, f"LP-n{n_try}")

    # === FINAL: Quick gradient + CD polish if time remains ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        t_final = deadline - time.monotonic() - 12
        if t_final > 30:
            print(f"\n{'='*60}")
            print(f"FINAL POLISH (t={t_final:.0f}s)")
            print(f"{'='*60}")

            # Split time between gradient and CD
            t_g = t_final * 0.5
            t_c = t_final * 0.5

            f_g, s_g = gradient_polish(best_seq, t_g, rng, verbose=False)
            try_sequence(f_g, "final-grad")

            if time.monotonic() < deadline - 15:
                t_c2 = deadline - time.monotonic() - 13
                if t_c2 > 10:
                    f_c, s_c = coordinate_descent_polish(best_seq, t_c2, rng)
                    try_sequence(f_c, "final-CD")

    # Verify with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'='*60}")
        # Guard: if evaluate_sequence fails, use a fallback
        if np.isinf(final_eval):
            print("  WARNING: evaluate_sequence returned inf, using fallback")
            if start_seq is not None:
                return [float(max(0.0, min(1000.0, x))) for x in start_seq]
            return [1.0] * n_primary
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 384
