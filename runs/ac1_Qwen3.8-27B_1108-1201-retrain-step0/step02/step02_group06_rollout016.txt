"""
Optimizes non-negative step function to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothed gradient descent (Adam), LP-based refinement,
and coordinate descent polishing. Key insight: gradient of smooth max of
self-convolution equals 2*cross_correlation(weights, a), computed via FFT.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    b = fft_convolve(a, a)
    return float(2.0 * n * np.max(b) / (S * S))


def smooth_max_grad(a, tau):
    """Log-sum-exp of conv(a,a) with gradient. a should sum to 1."""
    n = len(a)
    b = fft_convolve(a, a)
    max_b = np.max(b)
    shifted = (b - max_b) / tau
    exp_vals = np.exp(shifted)
    max_exp = np.max(exp_vals)
    if max_exp > 0:
        exp_vals = exp_vals / max_exp
    Z = np.sum(exp_vals)
    if Z < 1e-300:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
        return max_b, np.zeros(n), max_b, w
    w = exp_vals / Z
    L = max_b + tau * (np.log(Z) + np.log(max_exp) if max_exp > 1 else np.log(Z))
    # Gradient: grad[j] = 2 * sum_m w[m] * a[m-j] = 2 * cross_correlation(a, w)[j]
    size = 2 * n
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(fa * np.conj(fw), size)[:n]
    grad = 2.0 * corr
    return L, grad, max_b, w


def softmax_stable(theta):
    """Numerically stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_init(n, rng, n_iter=80, alpha_decay=0.95):
    """Spectral iterative init: alternate between time/freq domain peak reduction."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    alpha = 1.0 / (n * 0.5)
    for _ in range(n_iter):
        b = fft_convolve(a, a)
        a = a - alpha * b[:n]
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        alpha *= alpha_decay
    return a


def raised_cosine_init(n, rng, width=4.0, power=2.0):
    """Raised cosine bump centered in the sequence."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    center = 0.5
    x = (t - center) * width
    a = np.cos(np.pi * x / 2.0) ** 2 * (np.abs(x) < 1.0)
    a = a ** (1.0 / power)
    a = np.maximum(a, 0.001)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def ramp_init(n, rng, style='linear'):
    """Various smooth ramp initializations."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if style == 'linear':
        a = 1.0 + 0.5 * t
    elif style == 'quad':
        a = 1.0 + 0.6 * t ** 2
    elif style == 'sin':
        a = 1.0 + 0.4 * np.sin(np.pi * t)
    elif style == 'cos_center':
        a = np.cos(np.pi * (t - 0.5)) ** 2 + 0.3
    elif style == 'inv_cos':
        a = 1.0 / (np.cos(np.pi * (t - 0.5)) ** 2 + 0.5)
    elif style == 'edge':
        a = 1.0 + 0.5 * (2.0 * t - 1.0) ** 4
    else:
        a = 1.0 + 0.3 * t
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.001, None) / np.sum(np.clip(a, 0.001, None))


def interpolate_seq(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    result = np.interp(x_new, x_old, a)
    return np.clip(result, 0.001, None) / np.sum(np.clip(result, 0.001, None))


def adam_optimize(n, a_init, time_limit, seed):
    """Adam on softmax parameterization. Returns best (a, score) with sum(a)=1."""
    rng = np.random.default_rng(seed)
    a0 = np.clip(a_init, 1e-10, None)
    a0 = a0 / np.sum(a0)
    theta = np.log(a0)
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.8
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * np.max(fft_convolve(a0, a0))
    tau_init = max(0.5 / n, 1e-6)
    tau_final = max(0.01 / n, 1e-8)

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_init * (tau_final / tau_init) ** (frac ** 0.6)

        a = softmax_stable(theta)
        L, grad_a, max_b, w = smooth_max_grad(a, tau)

        p = a
        inner = np.dot(p, grad_a)
        g_theta = p * (grad_a - inner)

        step += 1
        m = b1 * m + (1 - b1) * g_theta
        v = b2 * v + (1 - b2) * (g_theta * g_theta)
        m_hat = m / (1.0 - b1 ** step)
        v_hat = v / (1.0 - b2 ** step)
        lr_eff = lr / (1.0 + 0.01 * step ** 0.5)
        theta = theta - lr_eff * m_hat / (np.sqrt(v_hat) + eps)

        if step % 20 == 0:
            a_chk = softmax_stable(theta)
            sc = 2.0 * n * np.max(fft_convolve(a_chk, a_chk))
            if sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

    return best_a, best_score


def lp_refine(a, time_limit, seed):
    """LP-based refinement: find b maximizing sum(b) s.t. a*b <= M, b >= 0."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a = a.copy()
    a = np.clip(a, 1e-10, None)
    a = a / np.sum(a)
    b_conv = fft_convolve(a, a)
    M = np.max(b_conv)
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    t_step = 0.1
    stagnation = 0

    while time.monotonic() - t0 < time_limit and stagnation < 30:
        b_conv = fft_convolve(a, a)
        M = np.max(b_conv)
        # Build constraint matrix: C[m, j] = a[m-j] if 0 <= m-j < n
        n_lags = 2 * n - 1
        C = np.zeros((n_lags, n))
        for j in range(n):
            start = j
            end = min(j + n, n_lags)
            C[start:end, j] = a[:end - start]
        # LP: max sum(b) s.t. C @ b <= M, b >= 0
        try:
            res = linprog(-np.ones(n), A_ub=C, b_ub=np.full(n_lags, M),
                          bounds=[(0, None)] * n, method='highs')
            if res.success:
                b_new = res.x
                b_sum = np.sum(b_new)
                if b_sum > 1e-10:
                    b_new = b_new / b_sum
                else:
                    stagnation += 5
                    continue
                # Blend
                a_new = (1.0 - t_step) * a + t_step * b_new
                a_new = np.clip(a_new, 1e-10, None)
                a_new = a_new / np.sum(a_new)
                new_M = np.max(fft_convolve(a_new, a_new))
                new_score = 2.0 * n * new_M
                if new_score < best_score - 1e-10:
                    best_score = new_score
                    best_a = a_new.copy()
                    a = a_new
                    stagnation = 0
                    if t_step > 0.02:
                        t_step *= 0.9
                else:
                    stagnation += 1
                    t_step *= 0.85
            else:
                stagnation += 5
        except Exception:
            stagnation += 10

    return best_a, best_score


def coord_descent(a_init, time_limit, seed):
    """Coordinate descent with pair transfers and block moves."""
    rng = np.random.default_rng(seed)
    a = np.clip(a_init, 0.0, None)
    s = np.sum(a)
    a = a / s
    n = len(a)
    current_score = 2.0 * n * np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    patience = 0
    max_patience = 3000

    while time.monotonic() - t0 < time_limit and patience < max_patience:
        r = rng.random()
        if r < 0.35:
            # Pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.02, 0.12)
            amt = a[i] * frac
            cand = a.copy()
            cand[i] = a[i] - amt
            cand[j] = a[j] + amt
            cand = np.clip(cand, 0.0, None)
            cs = np.sum(cand)
            if cs < 1e-10:
                patience += 1
                continue
            cand = cand / cs
            ns = 2.0 * n * np.max(fft_convolve(cand, cand))
            if ns < current_score:
                a = cand
                current_score = ns
                patience = 0
            else:
                patience += 1
        elif r < 0.55:
            # Single scale
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.04))
            cand = a.copy()
            cand[j] *= factor
            cand = np.clip(cand, 0.0, None)
            cs = np.sum(cand)
            if cs < 1e-10:
                patience += 1
                continue
            cand = cand / cs
            ns = 2.0 * n * np.max(fft_convolve(cand, cand))
            if ns < current_score:
                a = cand
                current_score = ns
                patience = 0
            else:
                patience += 1
        elif r < 0.75:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, max(1, n - width)))
            cand = a.copy()
            cand[start:start + width] *= np.exp(rng.normal(0, 0.03, size=width))
            cand = np.clip(cand, 0.0, None)
            cs = np.sum(cand)
            if cs < 1e-10:
                patience += 1
                continue
            cand = cand / cs
            ns = 2.0 * n * np.max(fft_convolve(cand, cand))
            if ns < current_score:
                a = cand
                current_score = ns
                patience = 0
            else:
                patience += 1
        else:
            # Peak-directed: reduce contributors to peak
            b = fft_convolve(a, a)
            peak_idx = int(np.argmax(b))
            # Contributors to b[peak_idx]: a[i]*a[peak_idx-i]
            contribs = np.zeros(n)
            for i in range(peak_idx + 1):
                j = peak_idx - i
                if 0 <= j < n and 0 <= i < n:
                    contribs[i] = max(contribs[i], a[i] * a[j])
                    contribs[j] = max(contribs[j], a[i] * a[j])
            top_k = max(5, n // 24)
            top_idx = np.argsort(contribs)[-top_k:]
            cand = a.copy()
            reduction = rng.uniform(0.02, 0.06, size=len(top_idx))
            removed = np.sum(cand[top_idx] * reduction)
            cand[top_idx] *= (1.0 - reduction)
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_add = min(len(top_idx), len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                cand[add_idx] += removed / n_add
            cand = np.clip(cand, 0.0, None)
            cs = np.sum(cand)
            if cs < 1e-10:
                patience += 1
                continue
            cand = cand / cs
            ns = 2.0 * n * np.max(fft_convolve(cand, cand))
            if ns < current_score:
                a = cand
                current_score = ns
                patience = 0
            else:
                patience += 1

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

    return best_a, best_score


def projected_grad_descent(n, a_init, time_limit, seed):
    """Nesterov accelerated projected gradient on simplex."""
    rng = np.random.default_rng(seed)
    a = np.clip(a_init, 0, None)
    s = np.sum(a)
    a = a / s if s > 1e-15 else np.ones(n) / n
    a_prev = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_val = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    step = 0
    lr = 0.5 / n
    tau_init = 3.0 / n
    tau_final = 0.05 / n

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_init * (tau_final / tau_init) ** (frac ** 0.7)

        t_prev = t_mom
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        beta = (t_prev - 1.0) / t_mom

        y = a + beta * (a - a_prev)
        y = project_simplex(y)

        L, grad, max_b, w = smooth_max_grad(y, tau)
        # Project gradient onto tangent space of simplex
        grad_proj = grad - np.mean(grad)
        lr_eff = lr / (1.0 + 0.005 * step ** 0.3)
        cand = y - lr_eff * grad_proj
        cand = project_simplex(cand)

        a_prev = a
        a = cand
        step += 1

        actual = np.max(fft_convolve(a, a))
        if actual < best_val:
            best_val = actual
            best_a = a.copy()

        if step % 5000 == 0:
            print(f"        PGD step={step}, score={2*n*best_val:.6f}, t={time.monotonic()-t0:.1f}s")

    return best_a, 2.0 * n * best_val

        if time.monotonic() > deadline - 180:
            break
        inits = make_inits(n, n_init)
        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 180:
                break
            t_alloc = base_time * (1.0 + 0.3 * np.log2(n / 256))
            t_alloc = max(5.0, min(60.0, t_alloc))
            a_opt, sc = adam_optimize(n, a0, t_alloc, seed + n * 100 + idx)
            actual = 2.0 * n * np.max(fft_convolve(a_opt, a_opt))
            print(f"  n={n}, init={idx}, score={actual:.6f}, t={time.monotonic()-t_global:.0f}s")
            if actual < best_score:
                best_score = actual
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} (n={n}) ***")

    # Phase 2: LP refinement on best sequences
    print(f"\n=== Phase 2: LP refinement ===")
    print(f"  Best before LP: {best_score:.6f}")
    if best_seq is not None:
        lp_time = min(200.0, max(30.0, (deadline - time.monotonic()) * 0.30))
        print(f"  LP time budget: {lp_time:.0f}s")
        a_norm = best_seq / np.sum(best_seq)
        a_lp, lp_score = lp_refine(a_norm, lp_time, seed + 999)
        actual_lp = 2.0 * len(a_lp) * np.max(fft_convolve(a_lp, a_lp))
        print(f"  After LP: {actual_lp:.6f}")
        if actual_lp < best_score:
            best_score = actual_lp
            best_seq = a_lp.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 3: Projected gradient refinement
    print(f"\n=== Phase 3: Projected gradient ===")
    if best_seq is not None:
        pgd_time = min(150.0, max(20.0, (deadline - time.monotonic()) * 0.20))
        print(f"  PGD time budget: {pgd_time:.0f}s")
        a_norm = best_seq / np.sum(best_seq)
        a_pgd, pgd_score = projected_grad_descent(len(a_norm), a_norm, pgd_time, seed + 888)
        actual_pgd = 2.0 * len(a_pgd) * np.max(fft_convolve(a_pgd, a_pgd))
        print(f"  After PGD: {actual_pgd:.6f}")
        if actual_pgd < best_score:
            best_score = actual_pgd
            best_seq = a_pgd.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 4: Coordinate descent polish
    print(f"\n=== Phase 4: Coordinate descent polish ===")
    if best_seq is not None:
        cd_time = min(300.0, max(30.0, (deadline - time.monotonic()) * 0.35))
        print(f"  CD time budget: {cd_time:.0f}s")
        a_norm = best_seq / np.sum(best_seq)
        a_cd, cd_score = coord_descent(a_norm, cd_time, seed + 777)
        actual_cd = 2.0 * len(a_cd) * np.max(fft_convolve(a_cd, a_cd))
        print(f"  After CD: {actual_cd:.6f}")
        if actual_cd < best_score:
            best_score = actual_cd
            best_seq = a_cd.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 5: Multi-scale exploration with remaining time
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print(f"\n=== Phase 5: Multi-scale exploration ===")
        remaining = deadline - time.monotonic() - 30
        # Try interpolating best to different lengths and refining
        target_ns = [384, 512, 640, 768, 1024, 1280]
        target_ns = [nn for nn in target_ns if nn != len(best_seq)]
        rng2 = np.random.default_rng(seed + 555)
        for nn in target_ns:
            if time.monotonic() > deadline - 30:
                break
            a_interp = interpolate_seq(best_seq, nn)
            # Quick Adam
            t_alloc = min(40.0, remaining / max(1, len(target_ns)))
            a_opt, _ = adam_optimize(nn, a_interp, t_alloc, seed + nn)
            sc = 2.0 * nn * np.max(fft_convolve(a_opt, a_opt))
            print(f"  n={nn}, score={sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Phase 6: Final CD polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        final_cd_time = min(60.0, deadline - time.monotonic() - 10)
        if final_cd_time > 5:
            a_norm = best_seq / np.sum(best_seq)
            a_final, _ = coord_descent(a_norm, final_cd_time, seed + 666)
            sc_final = 2.0 * len(a_final) * np.max(fft_convolve(a_final, a_final))
            print(f"\n  Final CD: {sc_final:.6f}")
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            final_score = evaluate_sequence(final_list)
        except Exception:
            final_score = np.inf
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    # Default: uniform sequence
    return [1.0] * 512
