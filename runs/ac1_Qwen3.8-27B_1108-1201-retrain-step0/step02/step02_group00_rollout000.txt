    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    # BB step size state
    x_prev = a.copy()
    grad_prev = None
    bb_step = 1.0 / (4.0 * beta_start)

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: exponential ramp
        progress = min(1.0, it / (max_time * 100.0))
        beta = beta_start * (beta_end / beta_start) ** progress
        beta = min(beta, beta_end)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # BB step size
        if grad_prev is not None:
            s_vec = y - x_prev
            y_vec = grad - grad_prev
            sy = np.dot(s_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            yy = np.dot(y_vec, y_vec)
            if sy > 1e-12:
                if yy > 1e-12:
                    bb_step = np.clip(sy / yy, 1e-8, 10.0 / beta)
                else:
                    bb_step = np.clip(ss / sy, 1e-8, 10.0 / beta)

        x_prev = y.copy()
        grad_prev = grad.copy()

        x_new = project_simplex(y - bb_step * grad, S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic restart
        if it % 5000 == 0 and rng.random() < 0.2:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0
            bb_step = 1.0 / (4.0 * beta)

        # Periodic evaluation
        if it % 1000 == 0:
            c_chk = fft_convolve_self(x_new)
            if c_chk.max() < best_score:
                best_score = c_chk.max()
                best_local = x_new.copy()

    return best_local, best_score


def run_toppk_polish(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=512):
    """Nesterov-accelerated top-k subgradient with adaptive step."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 2.0 / (n ** 0.5)

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = toppk_gradient(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: decay slowly
        step = step * 0.9999 + (1.0 - 0.9999) * (2.0 * S) / (n * (1.0 + 0.001 * it))

        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 3000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def active_set_polish(a_init, n, S, max_time, rng):
    """
    Polish using exact active set: identify which lags are at the maximum,
    then do coordinate-descent-style updates targeting those lags.
    */
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Find active set: lags within 1% of max
        threshold = cur * 0.99
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Use these active lags for gradient
        top_k = min(len(active), 256)
        active_top = active[np.argsort(c[active])[::-1][:top_k]]

        grad = toppk_gradient(a, c, len(active_top))
        # But weight only by active lags
        # Actually just use toppk_gradient with the right k
        # Re-compute with exact active set
        active_vals = c[active_top]
        total = active_vals.sum()
        if total < 1e-30:
            break
        weights = active_vals / total
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(active_top):
            w_full[int(k)] = weights[i]
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 4.0 / (n * (1.0 + 0.01 * (it_counter := 0)))
        # Use a fixed small step
        step = 2.0 / n
        x_new = project_simplex(a - step * grad / max(gnorm, 1e-15), S)

        c_new = fft_convolve_self(x_new)
        if c_new.max() < best_score:
            best_score = c_new.max()
            best_local = x_new.copy()
            a = x_new
        else:
            # Line search: reduce step
            step *= 0.5
            x_new = project_simplex(a - step * grad / max(gnorm, 1e-15), S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < best_score:
                best_score = c_new.max()
                best_local = x_new.copy()
                a = x_new

        if it % 5000 == 0 and isinstance(it, (int, np.integer)) and rng.random() < 0.05:
            a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    Returns a list of floats.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None

    # Determine main sequence length
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        main_len = len(given)
        if main_len < 64:
            main_len = 8192
    except Exception:
        main_len = 8192

    n = main_len
    n = min(n, 16384)  # Cap length for speed
    S = np.sqrt(2.0 * float(n))

    print("Main length: %d, S = %.4f" % (n, S))

    # --- Generate initializations ---
    inits = []

    # Given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if len(given) == n and np.sum(np.clip(given, 0, 1000)) > 0.01:
            inits.append(("given", np.clip(given, 0, 1000)))
    except Exception:
        pass

    # Segment-based initializations
    for k in [3, 5, 9]:
        seq_seg, score_seg = optimize_segment_count(n, k, rng, max_iter=500)
        inits.append(("seg%d" % k, seq_seg))
        print("  Segment k=%d: score=%.6f" % (k, score_seg))

    # Random
    inits.append(("rand", rng.random(n) + 0.1))

    # Sinusoidal
    tt = np.linspace(0, 1, n, endpoint=False)
    inits.append(("sin3", 1.0 + 0.4 * np.sin(2 * np.pi * 3 * tt) + 0.1))
    inits.append(("sin7", 1.0 + 0.3 * np.sin(2 * np.pi * 7 * tt) + 0.2 * np.cos(2 * np.pi * 2 * tt) + 0.1))

    # Triangular
    tri = 1.0 - 2.0 * np.abs(tt - 0.5)
    inits.append(("tri", tri + 0.1))

    # Two-step
    two_step = np.where(np.arange(n) < n // 2, 1.5, 0.5)
    inits.append(("twostep", two_step))

    # Three-step asymmetric
    three_step = np.where(np.arange(n) < n // 3, 2.0, np.where(np.arange(n) < 2 * n // 3, 0.5, 1.0))
    inits.append(("threestep", three_step))

    n_inits = len(inits)
    print("Total initializations: %d" % n_inits)

    # --- Phase 1: FISTA continuation on each init ---
    usable = max(1.0, deadline - time.monotonic())
    per_init = usable * 0.5 / n_inits
    print("=== Phase 1: FISTA continuation (%.1f sec each) ===" % per_init)

    for name, arr in inits:
        if time.monotonic() > deadline - 120:
            break
        arr = safe_normalize(arr, n)
        t0 = time.monotonic()

        a_fista, score_fista = run_fista_continuation(
            arr, n, S, seg_time := per_init, rng,
            beta_start=2.0, beta_end=200.0
        )
        a_norm = safe_normalize(a_fista, n)
        c = fft_convolve_self(a_norm)
        score = c.max()
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            print("  FISTA [%s]: %.6f (obj=%.6f)" % (name, score, full_objective(a_norm)))

    print("After Phase 1: best=%.6f" % global_best)

    # --- Phase 2: Top-k Nesterov polish on best ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        seg2 = max(5.0, remaining * 0.25)
        print("=== Phase 2: Top-k Nesterov polish (%.1f sec) ===" % seg2)

        a_pk, score_pk = run_toppk_polish(
            global_a, n, S, seg2, rng,
            top_k_init=5, top_k_max=256
        )
        a_pk = safe_normalize(a_pk, n)
        c_pk = fft_convolve_self(a_pk)
        if c_pk.max() < global_best:
            global_best = c_pk.max()
            global_a = a_pk.copy()
            print("  Top-k polish: %.6f" % global_best)

    # --- Phase 3: High-beta FISTA on best ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        seg3 = max(5.0, remaining * 0.2)
        print("=== Phase 3: High-beta FISTA (%.1f sec) ===" % seg3)

        a_hb, score_hb = run_fista_continuation(
            global_a, n, S, seg3, rng,
            beta_start=50.0, beta_end=500.0
        )
        a_hb = safe_normalize(a_hb, n)
        c_hb = fft_convolve_self(a_hb)
        if c_hb.max() < global_best:
            global_best = c_hb.max()
            global_a = a_hb.copy()
            print("  High-beta FISTA: %.6f" % global_best)

    # --- Phase 4: Active set polish ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        seg4 = max(5.0, remaining * 0.2)
        print("=== Phase 4: Active set polish (%.1f sec) ===" % seg4)

        it = 0
        a_as = global_a.copy()
        while time.monotonic() < time.monotonic() + seg4:
            c = fft_convolve_self(a_as)
            cur = c.max()
            if cur < global_best:
                global_best = cur
                global_a = a_as.copy()

            threshold = cur * 0.995
            active = np.where(c >= threshold)[0]
            if len(active) == 0:
                active = np.array([np.argmax(c)])
            top_k = min(len(active), 128)
            active_sorted = active[np.argsort(c[active])[::-1][:top_k]]

            active_vals = c[active_sorted]
            total = active_vals.sum()
            if total < 1e-30:
                break
            weights = active_vals / total
            w_full = np.zeros(2 * n - 1)
            for i_idx, k_idx in enumerate(active_sorted):
                w_full[int(k_idx)] = weights[i_idx]
            fsize = 1 << (len(w_full) + n - 2).bit_length()
            W = fft(w_full, fsize)
            A = fft(a_as, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break

            step = 3.0 / n
            for ls in range(10):
                x_new = project_simplex(a_as - step * grad / max(gnorm, 1e-15), S)
                c_new = fft_convolve_self(x_new)
                if c_new.max() < cur:
                    a_as = x_new
                    break
                step *= 0.7
            else:
                break

            it += 1

        print("  Active set polish: %.6f" % global_best)

    # --- Phase 5: Final verification and micro-polish ---
    if global_a is not None:
        # Verify
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")

        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(512.0) / 256.0)

    # Fallback
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(512.0) / 256.0)

    # Final clamp
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(512.0) / 256.0)

    final_score = full_objective(global_a)
    print("Final: length=%d, internal_score=%.6f" % (len(global_a), final_score))

    result = [float(x) for x in global_a]
    # Final verification
    final_ver = evaluate_sequence(result)
    print("Final evaluate_sequence: %.6f" % final_ver if np.isfinite(final_ver) else "Final: inf")
    if not np.isfinite(final_ver):
        result = [1.0] * 256
        print("Falling back to ones(256)")

    return result
