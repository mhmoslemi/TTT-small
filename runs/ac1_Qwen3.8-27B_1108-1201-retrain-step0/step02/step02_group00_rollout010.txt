"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.
Uses targeted coordinate descent guided by peak autocorrelation lags,
combined with brief FISTA refinements and stochastic perturbation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(seq):
    """Compute the actual objective value (without normalization)."""
    n = len(seq)
    c = fft_convolve_self(seq)
    max_c = c.max()
    sum_s = seq.sum()
    if sum_s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_s ** 2)


def targeted_cd(f, n, S, max_iters, rng, verbose=False):
    """
    Targeted coordinate descent: use peak lag to guide mass transfers.
    """
    f = f.copy().astype(np.float64)
    best_score = np.inf
    best_f = f.copy()
    
    for it in range(max_iters):
        c = fft_convolve_self(f)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_f = f.copy()
        
        # Find top K peak lags
        top_k = min(5, 2 * n - 1)
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[top_k - 1]
        peak_lags = np.where(c >= threshold)[0]
        
        # Compute "pressure" for each index based on peak lags
        # pressure[i] = sum over peak lags k of f[k-i] (valid indices only)
        pressure = np.zeros(n)
        for k in peak_lags:
            for i in range(max(0, k - n + 1), min(k + 1, n)):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += f[j]
        
        # Find indices with high pressure (to decrease) and low pressure (to increase)
        # But we need f[i] > 0 to decrease
        candidates_high = np.where((pressure > np.median(pressure)) & (f > 0.01 * S / n))[0]
        candidates_low = np.where((pressure < np.median(pressure)))[0]
        
        if len(candidates_high) == 0 or len(candidates_low) == 0:
            break
        
        # Try several transfers
        best_delta_score = cur
        best_move = None
        
        n_tries = min(50, len(candidates_high) * len(candidates_low))
        
        for _ in range(n_tries):
            i = candidates_high[rng.integers(len(candidates_high))]
            j = candidates_low[rng.integers(len(candidates_low))]
            if i == j:
                continue
            # Try transferring some fraction of f[i] to f[j]
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                delta = frac * f[i]
                if delta <= 0 or delta >= f[i]:
                    continue
                f_trial = f.copy()
                f_trial[i] -= delta
                f_trial[j] += delta
                c_trial = fft_convolve_self(f_trial)
                score = c_trial.max()
                if score < best_delta_score:
                    best_delta_score = score
                    best_move = (i, j, delta)
        
        if best_move is not None:
            i, j, delta = best_move
            f[i] -= delta
            f[j] += delta
            # Re-normalize (shouldn't be needed since we transferred)
        else:
            # Stagnation: try random perturbation
            idx = rng.integers(n, size=10)
            delta = rng.uniform(-0.01, 0.01) * S / n
            f[idx] += delta
            f = np.maximum(f, 0)
            f = f * (S / f.sum()) if f.sum() > 0 else np.ones(n) * S / n
    
    return best_f, best_score

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.
Uses peak-lag guided coordinate descent with stochastic perturbation,
combined with brief FISTA refinements for escaping local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    v = np.maximum(v, 0)
    s = v.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    # Binary search for theta
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_grad_lse(a, c, beta):
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_peak(a):
    c = fft_convolve_self(a)
    return c.max()


def run_fista_short(a_init, n, S, max_time, rng, beta=50.0):
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    a = a * (S / s) if s > 1e-15 else np.ones(n) * S / n
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = a.copy()
    best_score = compute_peak(y)
    step = 1.0 / (beta * 2.0)
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best = y.copy()
        grad = fft_grad_lse(y, c, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        if it % 200 == 0:
            step *= 0.995
    return best, best_score

    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_grad_lse(a, c, beta):
    n = len(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_bc = np.exp(shifted)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=20.0, beta_max=300.0):
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    a = a * (S / s) if s > 1e-15 else np.ones(n) * S / n
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = a.copy()
    best_score = fft_convolve_self(y).max()
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))
        step = 1.0 / (beta * 2.0 * (1.0 + 0.001 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best = y.copy()
        
        grad = fft_grad_lse(y, c, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return best, best_score


def peak_guided_cd(f, n, S, max_time, rng):
    """
    Coordinate descent guided by peak autocorrelation lags.
    
    Key insight: at peak lag k*, the subgradient component for index i
    is proportional to f[k*-i]. So we transfer mass from indices with
    high f[k*-i] (high "pressure") to indices with low f[k*-i].
    """
    f = np.clip(f, 0.0, 1000.0).copy().astype(np.float64)
    s = f.sum()
    f = f * (S / s) if s > 1e-15 else np.ones(n) * S / n
    
    start = time.monotonic()
    best_score = fft_convolve_self(f).max()
    best_f = f.copy()
    no_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(f)
        cur = c.max()
        
        if cur < best_score - 1e-10:
            best_score = cur
            best_f = f.copy()
            no_improve = 0
            if it % 200 == 1:
                print("      CD[%d]: %.8f" % (it, best_score))
        
        if time.monotonic() - start > max_time - 5:
            break
        
        # Find top peak lags (up to 10)
        # Efficient: use argpartition for top-k
        k_top = min(10, 2 * n - 1)
        top_idx = np.argpartition(c, -k_top)[-k_top:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        
        # Compute pressure: for each index i, sum of f[k-i] over peak lags k
        pressure = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k + 1, n)
            # Vectorized: pressure[lo:hi] += f[k-hi+1:k-lo+1][::-1]
            # f[k-i] for i in [lo, hi) => f[k-lo down to k-hi+1]
            seg = f[k - hi + 1: k - lo + 1][::-1]
            pressure[lo:hi] += seg
        
        # Source: high pressure, has mass to give
        # Dest: low pressure
        p75 = np.percentile(pressure, 75)
        p25 = np.percentile(pressure, 25)
        f_mean = S / n
        
        sources = np.where((pressure >= p75) & (f >= f_mean))[0]
        dests = np.where((pressure <= p25))[0]
        
        if len(sources) < 5:
            sources = np.where(f >= 2 * f_mean)[0]
        if len(dests) < 5:
            dests = np.where(f <= f_mean)[0]
        
        if len(sources) == 0 or len(dests) == 0:
            no_improve += 10
            if no_improve > 200:
                break
            # Random perturbation
            idx = rng.integers(0, n, size=rng.integers(10, 50))
            f[idx] += rng.normal(0, 0.01 * f_mean, len(idx))
            f = np.maximum(f, 0)
            fs = f.sum()
            f = f * (S / fs)
            continue
        
        # Try transfers
        improved = False
        n_attempts = min(40, 5 * max(1, len(sources) // 100))
        
        for _ in range(n_attempts):
            i = sources[rng.integers(len(sources))]
            j = dests[rng.integers(len(dests))]
            if i == j:
                continue
            
            # Try a range of transfer fractions
            max_transfer = 0.6 * f[i]
            for frac in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                delta = frac * f[i]
                if delta < 1e-12 or delta > max_transfer:
                    continue
                
                f[i] -= delta
                f[j] += delta
                
                c2 = fft_convolve_self(f)
                score = c2.max()
                
                if score < cur - 1e-10:
                    improved = True
                    cur = score
                    break
                else:
                    # Revert
                    f[i] += delta
                    f[j] -= delta
            
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 100:
                # Larger perturbation
                n_pert = rng.integers(20, 100)
                idx = rng.choice(n, size=n_pert, replace=False)
                scale = 0.05 * f_mean
                f[idx] += rng.normal(0, scale, n_pert)
                f = np.maximum(f, 0)
                fs = f.sum()
                f = f * (S / fs)
                no_improve = 0
            elif no_improve > 50:
                # Medium perturbation
                n_pert = rng.integers(5, 20)
                idx = rng.choice(n, size=n_pert, replace=False)
                scale = 0.02 * f_mean
                f[idx] += rng.normal(0, scale, n_pert)
                f = np.maximum(f, 0)
                fs = f.sum()
                f = f * (S / fs)
    
    return best_f, best_score
