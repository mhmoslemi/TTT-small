import numpy as np

from scipy.optimize import minimize

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_of(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (s ** 2)

def gs_init(n, rng, n_iter=100):
    a = np.ones(n) / n
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft <<= 1
    for _ in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        target_mag = np.ones(N_fft) / np.sqrt(N_fft)
        # Blend: target is flat, current is mag
        # Use GS: set magnitude to target, keep phase
        A_new = target_mag * (A / mag)
        a_full = np.fft.ifft(A_new).real
        a = np.clip(a_full[:n], 0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a

    N_fft = 1

        # Target: flat magnitude

        a_full = np.fft.ifft(A_new).real

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def score_of(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * np.max(c) / (s * s))

def gs_init(n, rng, n_iter=200, target='flat'):
    a = np.ones(n) / n
    N = 2 * n - 1
    Nf = 1
    while Nf < N:
        Nf <<= 1
    for it in range(n_iter):
        A = np.fft.fft(a, Nf)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        if target == 'flat':
            target_mag = np.full(Nf, 1.0 / np.sqrt(Nf))
        elif target == 'tapered':
            # Slightly tapered target to reduce center peak
            w = np.linspace(1.0, 0.8, Nf)
            target_mag = w / np.sqrt(np.sum(w**2))
        A_new = target_mag * (A / mag)
        a = np.clip(np.fft.ifft(A_new).real[:n], 0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a /= s
    return a

def soft_max_obj_and_grad(theta, tau):
    """Compute soft-max score and gradient for log-parameterized sequence."""
    n = len(theta)
    # a = softmax(theta) (normalized to sum 1)
    t = theta - np.max(theta)
    e = np.exp(t)
    a = e / np.sum(e)
    
    # Compute convolution
    c = fft_convolve(a)
    
    # Soft-max
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    w = np.exp(shifted)
    w /= np.sum(w)
    val = c_max + tau * np.log(np.sum(np.exp(shifted)))
    
    # Gradient w.r.t. a: g_a[m] = 2 * sum_j a[j] * w[m+j]
    u = a[::-1]
    sz = 1
    while sz < n + 2*n - 1:
        sz <<= 1
    g_a = 2.0 * np.fft.irfft(np.fft.rfft(u, sz) * np.fft.rfft(w, sz), sz)
    g_a = g_a[n-1:2*n-1]
    
    # Gradient w.r.t. theta (chain rule for softmax)
    inner = np.dot(g_a, a)
    g_theta = a * (g_a - inner)
    
    return val, g_theta

def lbfgs_optimize(a_init, tau, tol=1e-8, maxiter=500, time_limit=60):
    """Optimize using L-BFGS with given soft-max temperature."""
    n = len(a_init)
    theta0 = np.log(a_init + 1e-15)
    t0 = time.monotonic()
    
    def func_and_grad(theta):
        if time.monotonic() - t0 > time_limit:
            raise RuntimeError("time limit")
        val, g = soft_max_obj_and_grad(theta, tau)
        return val, g
    
    result = minimize(func_and_grad, theta0, method='L-BFGS', 
                      jac=True, options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-8})
    
    # Convert back
    t = result.x - np.max(result.x)
    e = np.exp(t)
    a = e / np.sum(e)
    return a, result.fun

def subgradient_polish(a, time_limit, lr=0.01, decay=0.999):
    """Polish using subgradient of the exact max."""
    n = len(a)
    current = a.copy()
    current /= np.sum(current)
    best = current.copy()
    best_score = score_of(current)
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current)
        max_idx = np.argmax(c)

        # dc[k]/da[m] = 2*a[k-m] if 0<=k-m<n
        m = max_idx
        # g_a[m'] = 2 * a[m-m'] for m' in range(n) such that 0 <= m-m' < n
        g_a = np.zeros(n)
        valid = np.arange(n)
        offsets = m - valid  # k-m for each m'
        valid_mask = (offsets >= 0) & (offsets < n)
        g_a[valid_mask] = 2.0 * a[offsets[valid_mask]]
        
        # Normalize gradient
        g_norm = np.linalg.norm(g_a)
        if g_norm < 1e-15:
            break
        
        # Update: decrease a in direction of g_a
        # We want to reduce c[max_idx], so move opposite to gradient
        # But we need to maintain sum(a) = 1 and a >= 0
        lr_eff = lr / g_norm
        candidate = current - lr_eff * g_a
        # Project
        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            break
        candidate /= s
        
        # Check score
        new_score = score_of(candidate)
        if new_score < best_score:
            best_score = new_score
            best = candidate.copy()
        current = candidate
        lr *= decay
        step += 1
        
        if step % 1000 == 0:
            print(f"    subgrad step {step}, score={current_score:.6f}, best={best_score:.6f}")

"""
Spectral-continuation L-BFGS optimization for minimizing peak autocorrelation
of non-negative step functions. Uses enhanced Gerchberg-Saxton initialization,
L-BFGS with soft-max continuation, top-k subgradient polishing, and spectral
repair. Designed to beat 1.5027 target.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def score_of(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * np.max(c) / (s * s))


def gs_init(n, rng, n_iter=200, mode='flat', taper=0.0):
    """Gerchberg-Saxton style initialization with optional spectral tapering."""
    a = np.ones(n) / n
    N = 2 * n - 1
    Nf = 1
    while Nf < N:
        Nf <<= 1

    # Build target magnitude
    if mode == 'flat':
        target_mag = np.full(Nf, 1.0 / np.sqrt(Nf))
    elif mode == 'tapered':
        # Slightly suppress center frequencies to reduce center-lag peak
        idx = np.arange(Nf)
        center = Nf // 2
        freq_dist = np.abs(idx - center) / center
        target_mag = (1.0 - taper * freq_dist) / np.sqrt(Nf)
        target_mag[target_mag < 0.1 / np.sqrt(Nf)] = 0.1 / np.sqrt(Nf)
    elif mode == 'raised':
        # Slightly boost center frequencies
        idx = np.arange(Nf)
        center = Nf // 2
        freq_dist = np.abs(idx - center) / center
        target_mag = (1.0 + taper * (1.0 - freq_dist)) / np.sqrt(Nf)
    elif mode == 'deep':
        # More aggressive flatting with oscillation
        target_mag = np.full(Nf, 1.0 / np.sqrt(Nf))
    else:
        target_mag = np.full(Nf, 1.0 / np.sqrt(Nf))

    for it in range(n_iter):
        A = np.fft.fft(a, Nf)
        mag = np.abs(A)
        mag_safe = np.where(mag < 1e-15, 1e-15, mag)
        phase = A / mag_safe
        A_new = target_mag * phase
        a_full = np.fft.ifft(A_new).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def random_smooth_init(n, rng, bandwidth=0.2):
    """Random smooth initialization via low-pass filtered noise."""
    a = 1.0 + 0.5 * rng.standard_normal(n)
    # Apply low-pass filter
    A = np.fft.rfft(a)
    freq = np.arange(len(A))
    cutoff = int(bandwidth * n)
    A[cutoff:] *= 0.1
    a = np.fft.irfft(A, n=n).real
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def chirp_init(n, rng, alpha=0.3):
    """Quadratic chirp initialization."""
    t = np.arange(n, dtype=np.float64)
    freq = rng.uniform(0.3, 2.0)
    a = 1.0 + alpha * np.cos(np.pi * freq * t * t / n)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def soft_max_score_grad_theta(theta, tau):
    """
    Compute soft-max of autocorrelation and its gradient w.r.t. theta
    where a = softmax(theta) (normalized to sum 1).
    Returns (value, gradient).
    """
    n = len(theta)
    # Compute a = softmax(theta)
    t_shifted = theta - np.max(theta)
    e = np.exp(t_shifted)
    a = e / np.sum(e)

    # Compute convolution c = conv(a, a), length 2n-1
    c = fft_convolve(a)

    # Soft-max
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    val = c_max + tau * np.log(sum_exp)

    # Gradient w.r.t. a: g_a[m] = 2 * sum_{j=0}^{n-1} a[j] * w[m+j]
    # Computed via: g_a = 2 * (rev(a) * w)[n-1 : 2n-1]
    u = a[::-1]
    sz = 1
    while sz < n + 2 * n - 1:
        sz <<= 1
    conv_full = np.fft.irfft(np.fft.rfft(u, sz) * np.fft.rfft(w, sz), sz)
    g_a = 2.0 * conv_full[n - 1:2 * n - 1]

    # Chain rule for softmax: g_theta = a * (g_a - dot(g_a, a))
    inner = np.dot(g_a, a)
    g_theta = a * (g_a - inner)

    return val, g_theta


def soft_max_obj_callable(theta, tau):
    """Top-level function for scipy minimize. Returns (f, fprime)."""
    return soft_max_score_grad_theta(theta, tau)


class SoftMaxObj:
    """Callable wrapper for scipy.optimize.minimize."""
    def __init__(self, tau):
        self.tau = tau

    def __call__(self, theta):
        return soft_max_score_grad_theta(theta, self.tau)


def lbfgs_optimize(a_init, tau, maxiter=500, time_limit=60.0):
    """Run L-BFGS optimization with given soft-max temperature."""
    n = len(a_init)
    theta0 = np.log(a_init + 1e-15)
    t0 = time.monotonic()

    # We need a callable that returns (f, fprime) for L-BFGS
    # scipy's L-BFGS expects func to return (f, fprime) when jac=True
    # But our function already returns a tuple, so we use wrap=True trick
    # Actually scipy L-BFGS expects func to return f (float) and jac to return gradient
    # Let's use the two-arg form

    obj_wrapper = SoftMaxObj(tau)

    def func_only(x):
        val, _ = obj_wrapper(x)
        return val

    def jac_only(x):
        _, g = obj_wrapper(x)
        return g

    # Check time
    if time.monotonic() - t0 > time_limit:
        return a_init, score_of(a_init)

    try:
        result = minimize(
            func_only, theta0, method='L-BFGS', jac=jac_only,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 3}
        )
    except Exception:
        return a_init, score_of(a_init)

    # Convert back from theta to a
    t_shifted = result.x - np.max(result.x)
    e = np.exp(t_shifted)
    a = e / np.sum(e)
    return a, score_of(a)


def subgradient_topk(a, time_limit, k=5, lr0=0.03, decay=0.9995):
    """
    Subgradient polish targeting top-k peak lags simultaneously.
    Uses projected gradient with adaptive step size.
    """
    n = len(a)
    current = a.astype(np.float64).copy()
    current = current / np.sum(current)
    best = current.copy()
    best_score = score_of(current)
    lr = lr0
    t0 = time.monotonic()
    step = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current)
        # Get top-k lags
        top_k = np.argsort(c)[-k:]

        # Compute combined gradient: sum of gradients of top-k lags
        # dc[l]/da[m] = 2*a[l-m] if 0<=l-m<n
        g_a = np.zeros(n)
        for lag in top_k:
            for m in range(n):
                offset = lag - m
                if 0 <= offset < n:
                    g_a[m] += 2.0 * current[offset]

        # Normalize
        g_norm = np.linalg.norm(g_a)
        if g_norm < 1e-15:
            break

        # Adaptive step: use line search-like approach
        candidate = current - lr * g_a / g_norm
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            lr *= 0.5
            continue
        candidate = candidate / s

        new_score = score_of(candidate)
        if new_score < best_score:
            best_score = new_score
            best = candidate.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.5
                no_improve = 0

        # Accept if it improves current or with some probability
        if new_score <= score_of(current) + 0.001:
            current = candidate
        elif np.random.random() < 0.1:
            current = candidate
        else:
            current = best.copy()

        lr *= decay
        step += 1

        if step % 5000 == 0:
            print(f"    subgrad step={step}, cur={score_of(current):.6f}, best={best_score:.6f}, lr={lr:.4f}")

    return best, best_score

        Nf <<= 1

    best_a = a.copy()

        # Ensure hermitian symmetry

def continuation_optimize(a_init, n, time_limit, rng):
    """
    Run L-BFGS with continuation (decreasing soft-max temperature).

    a = a_init.copy()

    time_per_tau = time_limit / n_tau

    for i, alpha in enumerate(alphas):
        if time.monotonic() > time_limit + time.monotonic() - time_limit:
            break
        tau = peak0 * alpha
        t_remaining = time_limit - (time.monotonic() - (time_limit - time_per_tau * (n_tau - i)))
        # Simpler: just track elapsed
        maxiter = 200 + int(200 * (1 - i / n_tau))
        a, sc = lbfgs_optimize(a, tau, maxiter=maxiter, time_limit=time_per_tau * 1.2)

    return a, score_of(a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for optimal non-negative step function minimizing
    2*n*max(conv(a,a))/sum(a)^2.

    Strategy:
    1. Generate diverse initializations (GS variants, random smooth, chirp)
    2. Optimize each with L-BFGS continuation (soft-max temperature annealing)
    3. Polish best candidates with top-k subgradient method
    4. Apply spectral repair
    5. Return best found sequence
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided starting sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        sc = score_of(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = start_seq.copy()

    # Lengths to try (longer = potentially better, but slower)
    lengths = [256, 384, 512, 768, 1024, 1536, 2048, 3072, 4096]

    # Time allocation: more time for promising (medium-large) lengths
    # We'll iterate through lengths and allocate time based on remaining budget

    all_candidates = []  # (score, n, sequence)

    for n in lengths:
        # Check if we have enough time left for this length
        time_left = deadline - time.monotonic()
        if time_left < 30:
            break

        # Allocate time for this length
        # Smaller lengths get less time, larger get more
        frac = min(1.0, time_left / 600.0)
        n_inits = max(2, int(4 * frac + 2))
        time_per_init = max(5.0, min(40.0, (time_left - 10) / (n_inits * 3)))
        time_for_length = min(time_left * 0.7, n_inits * time_per_init * 3 + 20)

        print(f"\n=== n={n}, time_alloc={time_for_length:.0f}s, n_inits={n_inits} ===")

        # Generate initializations
        inits = []

        # GS flat
        a1 = gs_init(n, rng, n_iter=150, mode='flat')
        inits.append(('gs_flat', a1))

        # GS tapered (suppress center freq)
        a2 = gs_init(n, rng, n_iter=150, mode='tapered', taper=0.3)
        inits.append(('gs_taper', a2))

        # GS raised (boost center)
        a3 = gs_init(n, rng, n_iter=150, mode='raised', taper=0.3)
        inits.append(('gs_raise', a3))

        # Random smooth
        a4 = random_smooth_init(n, rng, bandwidth=0.15)
        inits.append(('rand_smooth', a4))

        # Chirp
        a5 = chirp_init(n, rng, alpha=0.2 + 0.4 * rng.random())
        inits.append(('chirp', a5))

        # GS with more iterations (deeper optimization)
        if n <= 1024:
            a6 = gs_init(n, rng, n_iter=300, mode='flat')
            inits.append(('gs_deep', a6))

        # Use provided sequence as init (resampled if needed)
        if best_seq is not None and len(best_seq) != n:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.clip(a_interp, 0.01, None)
            a_interp = a_interp / np.sum(a_interp)
            inits.append(('prev_best', a_interp))

        # Limit number of inits
        inits = inits[:n_inits + 2]

        # Evaluate and optimize each init
        for name, a_init in inits:
            if time.monotonic() > deadline - 20:
                break

            sc_init = score_of(a_init)
            t_init = time.monotonic()

            # Continuation L-BFGS
            t_alloc = min(time_per_init * 2.5, deadline - time.monotonic() - 15)
            if t_alloc < 3:
                break

            a_opt, sc_opt = continuation_optimize(a_init, n, t_alloc, rng)

            # Subgradient polish
            if time.monotonic() < deadline - 10:
                t_polish = min(10.0, deadline - time.monotonic() - 10)
                if t_polish > 2:
                    a_pol, sc_pol = subgradient_topk(a_opt, t_polish, k=5, lr0=0.02)
                    if sc_pol < sc_opt:
                        a_opt, sc_opt = a_pol, sc_pol

            # Spectral repair
            if time.monotonic() < deadline - 5:
                a_rep, sc_rep = spectral_repair(a_opt)
                if sc_rep < sc_opt:
                    a_opt, sc_opt = a_rep, sc_rep

            elapsed = time.monotonic() - t_init
            print(f"  {name}: init={sc_init:.5f} -> opt={sc_opt:.5f} ({elapsed:.1f}s)")

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  *** NEW BEST: {best_score:.6f} (n={n}, {name}) ***")

            all_candidates.append((sc_opt, n, a_opt.copy()))

    # Final phase: intensive polish on the best candidate
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final intensive polish (n={len(best_seq)}) ===")
        t_polish = min(120.0, deadline - time.monotonic() - 10)
        print(f"  Polish time: {t_polish:.0f}s")

        # Multiple subgradient rounds with decreasing top-k
        a_final = best_seq.copy()
        for k_val in [8, 5, 3, 1]:
            if time.monotonic() > deadline - 10:
                break
            t_round = min(30.0, deadline - time.monotonic() - 10)
            a_round, sc_round = subgradient_topk(a_final, t_round, k=k_val, lr0=0.015, decay=0.9998)
            if sc_round < best_score:
                best_score = sc_round
                best_seq = a_round.copy()
                a_final = a_round.copy()
                print(f"  k={k_val}: score={best_score:.6f}")

        # One more L-BFGS round with very small tau
        if time.monotonic() < deadline - 15:
            a_final = best_seq / np.sum(best_seq)
            c = fft_convolve(a_final)
            tau_small = np.max(c) * 0.001
            t_lbg = min(30.0, deadline - time.monotonic() - 10)
            a_lopt, scl = lbfgs_optimize(a_final, tau_small, maxiter=300, time_limit=t_lbg)
            if scl < best_score:
                best_score = scl
                best_seq = a_lopt.copy()
                print(f"  Final L-BFGS: score={best_score:.6f}")

    # Multi-restart from best with perturbations if time remains
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        n_restarts = min(4, int((deadline - time.monotonic() - 20) / 30))
        print(f"\n=== Multi-restart phase ({n_restarts} restarts, n={n_best}) ===")
        for r in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            # Perturb the best sequence
            a_pert = best_seq * np.exp(rng.normal(0, 0.05, size=n_best))
            a_pert = np.clip(a_pert, 0.0, None)
            s = np.sum(a_pert)
            if s < 1e-10:
                continue
            a_pert = a_pert / s

            t_alloc = min(25.0, deadline - time.monotonic() - 15)
            if t_alloc < 5:
                break

            a_opt, sc_opt = continuation_optimize(a_pert, n_best, t_alloc, rng)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  Restart {r+1}: {best_score:.6f}")

    # Final verification
    print(f"\n{'=' * 50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if best_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return [1.0] * 256
