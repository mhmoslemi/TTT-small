```python
from scipy.optimize import linprog

def lp_direction(f, M):
    """Find the LP direction: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    # Constraint: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_i f[i]*g[k-i] (valid i)
    # This is linear in g.
    # A @ g <= M
    # A[k, j] = f[j] if k-j is in [0, n-1], else 0
    # i.e., A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Wait, (f*g)[k] = sum_j f[j] * g[k-j]
    # For a fixed k, the terms involving g[m] (where m = k-j, so j = k-m):
    # (f*g)[k] = sum_m f[k-m] * g[m] (for valid m, k-m)
    # So A[k, m] = f[k-m] if 0 <= k-m < n, else 0
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A[k, m] = f[j]
    
    c_coef = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    res = linprog(c_coef, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
```
