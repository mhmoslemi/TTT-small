    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]

def a_from_theta(theta, S):
    """Convert log-parameters to sequence: a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total

def gs_iterate(a, S, N, relaxation=0.8, iters=200, verbose=False):
    """Gerchberg-Saxton spectral flattening algorithm.
    
    Alternates between:
    - Frequency domain: flatten |FFT(a)|
    - Time domain: project to non-negative, normalize sum to S
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = fft_convolve_self(a).max()
    
    for it in range(iters):
        # Compute FFT
        A = fft(a, N)
        
        # Compute target magnitudes
        norm_a = np.linalg.norm(a)
        dc = S  # A[0] should be S
        # Target for other frequencies: make |A[k]|^2 as flat as possible
        # Total energy: sum|A[k]|^2 = N * ||a||^2
        # If |A[0]| = S and |A[k]| = T for k != 0:
        # S^2 + (N-1)*T^2 = N * norm_a^2
        # T = sqrt((N * norm_a^2 - S^2) / (N-1))
        T2 = (N * norm_a**2 - S**2) / (N - 1)
        if T2 < 0:
            T2 = 0
        T = np.sqrt(T2)
        
        # Create target magnitude spectrum
        target_mag = np.full(N, T)
        target_mag[0] = S
        
        # Replace magnitudes, keep phases
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        
        # IFFT and take first n elements
        a_new = np.real(ifft(A_new, N))[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        
        # Normalize
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new * (S / s)
        
        # Relaxation
        a = (1 - relaxation) * a + relaxation * a_new
        
        # Track best
        c = fft_convolve_self(a)
        obj = c.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if verbose and it % 50 == 0:
            print("  GS iter %d: obj=%.6f" % (it, best_obj))
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Gerchberg-Saxton (GS) spectral flattening for fast exploration
2. L-BFGS and Adam with soft-max surrogate for exploitation
3. FFT-based gradient computation exploiting Toeplitz structure
4. Multi-start with diverse initializations, sequence lengths, and GS refinement
5. Adaptive peak-cycling and coordinate descent for final polish
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def a_from_theta(theta, S):
    """Convert log-parameters to sequence: a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def theta_from_a(a, S):
    """Convert sequence to log-parameters."""
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return np.log(a * S)


def compute_grad(w, a):
    """Compute grad[j] = 2 * sum_k w[k] * a[k-j] using FFT convolution.
    
    w: weights on convolution lags (length 2n-1)
    a: sequence (length n)
    Returns: gradient (length n)
    """
    n = len(a)
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    start = n - 1
    end = start + n
    grad = 2.0 * full_conv[start:end]
    return grad


def soft_weights(c, T, top_k=0):
    """Compute soft-max weights on convolution peaks.
    
    If top_k > 0, only consider top_k peaks (uniform weight among them).
    Otherwise use full softmax with temperature T.
    """
    if top_k > 0 and top_k < len(c):
        w = np.zeros(len(c))
        idx = np.argsort(c)[::-1][:top_k]
        w[idx] = 1.0 / top_k
        return w
    c_max = c.max()
    if c_max < 1e-15:
        return np.ones(len(c)) / len(c)
    scaled = T * (c - c_max) / c_max
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        return np.ones(len(c)) / len(c)
    return exp_scaled / total


def softmax_objective_and_grad(theta, n, T, S, top_k=0):
    """Soft-max surrogate objective and gradient w.r.t. theta."""
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()

    # Soft-max objective
    if T < 1e6:
        w = soft_weights(c, T, top_k)
        if top_k > 0 and top_k < len(c):
            obj = np.max(c[np.argsort(c)[::-1][:top_k]])
        else:
            scaled = T * (c - c_max) / c_max
            obj = c_max + (1.0 / T) * c_max * np.log(np.sum(np.exp(scaled - scaled.max())))
    else:
        w = np.zeros(len(c))
        idx = np.argsort(c)[::-1][:max(1, top_k)]
        w[idx] = 1.0 / len(idx)
        obj = c_max

    # Gradient
    grad_a = compute_grad(w, a)
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot

    return obj, grad_theta


def gs_search(a_init, S, N, iters=300, relaxation=0.7, verbose=False):
    """Gerchberg-Saxton spectral flattening.
    
    Alternates between flattening the magnitude spectrum and projecting
    to non-negative time-domain signals.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a_init, fft_convolve_self(a_init).max()
    a = a * (S / s)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    for it in range(iters):
        # Compute FFT
        A = fft(a, N)

        # Target magnitudes: flat for k != 0
        norm_sq = np.dot(a, a)
        T2 = (N * norm_sq - S * S) / (N - 1)
        if T2 < 0:
            T2 = 0.0
        T = np.sqrt(T2)

        target_mag = np.full(N, T)
        target_mag[0] = S

        # Keep phase, replace magnitude
        phase = np.angle(A)
        A_new = target_mag * np.cos(phase) + 1j * target_mag * np.sin(phase)

        # IFFT
        a_new = np.real(ifft(A_new, N))[:n]

        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)

        # Normalize
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new * (S / s_new)

        # Relaxation
        a = (1.0 - relaxation) * a + relaxation * a_new

        # Track best
        c = fft_convolve_self(a)
        obj = c.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if verbose and it % 100 == 0:
            print("    GS iter %d: obj=%.6f" % (it, best_obj))

    return best_a, best_obj


def run_adam_anneal(theta_init, n, S, time_budget, T_start=5.0, T_end=500.0,
                    lr=0.05, top_k_switch=50.0):
    """Adam optimizer with temperature annealing and adaptive top-k."""
    theta = theta_init.copy().astype(np.float64)
    n_par = len(theta)
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    start = time.monotonic()
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    best_theta = theta.copy()

    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (t) / max(1, int(time_budget / 0.003)))

        # Temperature schedule
        T = T_start * np.exp(np.log(T_end / T_start) * progress)

        # Adaptive top-k: start with all, switch to top-k as T increases
        if T > top_k_switch:
            k = max(4, int(20 * (T / top_k_switch) ** 0.5))
            k = min(k, n)
        else:
            k = 0

        a = a_from_theta(theta, S)
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()
            best_theta = theta.copy()

        # Compute gradient
        if T < 1e6:
            w = soft_weights(c, T, k)
        else:
            w = np.zeros(2 * n - 1)
            idx = np.argsort(c)[::-1][:max(1, k)]
            w[idx] = 1.0 / len(idx)

        grad_a = compute_grad(w, a)
        dot = np.dot(grad_a, a)
        grad = grad_a * a - (a / S) * dot

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr_t = lr * (0.7 ** progress)
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)

    return best_theta, best_a, best_obj

    theta = theta_init.copy().astype(np.float64)

def lbfgs_wrapper(theta, n, T, S, state, top_k=0):
    """L-BFGS objective with state tracking."""
    elapsed = time.monotonic() - state['start']
    if elapsed > state['time_budget']:
        return state['best_obj'], np.zeros(len(theta))
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    true_max = c.max()
    if true_max < state['best_obj']:
        state['best_obj'] = true_max
        state['best_a'] = a.copy()

    w = soft_weights(c, T, top_k)
    grad_a = compute_grad(w, a)
    dot = np.dot(grad_a, a)
    grad = grad_a * a - (a / S) * dot

    if T < 1e6:
        scaled = T * (c - true_max) / max(true_max, 1e-15)
        obj = true_max + (true_max / T) * np.log(np.sum(np.exp(scaled - scaled.max())))
    else:
        obj = true_max

    return obj, grad


def run_lbfgs(theta_init, n, T, S, time_budget, top_k=0, max_iter=500):
    """Run L-BFGS-B on soft-max objective."""
    state = {
        'n': n, 'T': T, 'S': S,
        'start': time.monotonic(), 'time_budget': time_budget,
        'best_obj': np.inf, 'best_a': None, 'top_k': top_k
    }
    a0 = a_from_theta(theta_init, S)
    c0 = fft_convolve_self(a0)
    state['best_obj'] = c0.max()
    state['best_a'] = a0.copy()

    try:
        minimize(
            lbfgs_wrapper, theta_init.copy(),
            args=(n, T, S, state, top_k),
            jac=True, method='L-BFGS-B',
            options={'maxiter': max_iter, 'maxfun': 5000, 'maxls': 100,
                     'gtol': 1e-10}
        )
    except Exception as e:
        print("    L-BFGS error: %s" % str(e))

    return state['best_a'], state['best_obj']


def coordinate_descent(a, S, time_budget, max_sweeps=100):
    """Coordinate descent polish: try reducing each element."""
    n = len(a)
    a = a.copy().astype(np.float64)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False

        # Randomize order for diversity
        order = np.random.default_rng(sweep).permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig <= 1e-12:
                continue

            # Try a range of reductions
            best_local = orig
            best_local_obj = best_obj
            for frac in [0.5, 0.7, 0.85, 0.95, 1.1, 1.3]:
                a[i] = orig * frac
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_tmp = a * (S / s)
                c_tmp = fft_convolve_self(a_tmp)
                obj_tmp = c_tmp.max()
                if obj_tmp < best_local_obj:
                    best_local_obj = obj_tmp
                    best_local = orig * frac

            a[i] = best_local
            if abs(best_local - orig) > 1e-10 * orig:
                improved = True
                s = a.sum()
                if s > 1e-10:
                    a = a * (S / s)
                    c = fft_convolve_self(a)
                    if c.max() < best_obj:
                        best_obj = c.max()
                        best_a = a.copy()

        if not improved:
            break

    return best_a, best_obj


def peak_cycling(a, S, time_budget, max_iters=5000):
    """Iteratively reduce the peak autocorrelation by cycling through peaks."""
    n = len(a)
    a = a.copy().astype(np.float64)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    start = time.monotonic()

    lr = 0.01
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        c = fft_convolve_self(a)
        obj = c.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Find the peak
        k_peak = np.argmax(c)

        # Gradient of c[k_peak]
        grad_a = compute_grad(np.zeros(2 * n - 1) + (np.arange(2 * n - 1) == k_peak), a)

        # Take a step (decrease in direction of gradient, projected to maintain sum)
        # grad_a points in direction of increase; we want to decrease
        # Project: a_new = a - lr * (grad_a - mean(grad_a)) to maintain sum
        grad_centered = grad_a - np.mean(grad_a)
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new * (S / s)

        # Check if this helps
        c_new = fft_convolve_self(a_new)
        if c_new.max() < obj:
            a = a_new
            # Decay learning rate slowly
            lr = max(0.001, lr * 0.999)
        else:
            # Try smaller step
            a_new2 = a - 0.1 * lr * grad_centered
            a_new2 = np.maximum(a_new2, 0.0)
            s2 = a_new2.sum()
            if s2 > 1e-10:
                a_new2 = a_new2 * (S / s2)
                c_new2 = fft_convolve_self(a_new2)
                if c_new2.max() < obj:
                    a = a_new2
                else:
                    lr *= 0.9
            else:
                lr *= 0.9

    return best_a, best_obj


def get_initialization(name, n, rng, S):
    """Generate various initializations."""
    i = np.arange(n)
    if name == "uniform":
        a = np.ones(n)
    elif name == "random":
        a = rng.random(n)
    elif name == "gaussian":
        a = np.exp(-0.5 * ((i - n / 2) / (n / 6)) ** 2)
    elif name == "cosine":
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * i / n)
        a = np.maximum(a, 0.01)
    elif name == "anti_peak":
        a = 1.0 - 0.4 * np.cos(2 * np.pi * (n - 1) * i / n)
        a = np.maximum(a, 0.01)
    elif name == "two_humps":
        a = np.exp(-0.5 * ((i - n * 0.25) / (n / 10)) ** 2) + \
            np.exp(-0.5 * ((i - n * 0.75) / (n / 10)) ** 2)
    elif name == "flat_cos":
        a = 1.0 + 0.1 * np.cos(2 * np.pi * i / n) + 0.05 * np.cos(4 * np.pi * i / n)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    global_best = np.inf
    global_a = None

    # Try to load the given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            n_given = len(given_raw)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_raw * (S_given / given_raw.sum())
            c_given = fft_convolve_self(given_a)
            obj_given = 2.0 * n_given * c_given.max() / (given_a.sum() ** 2)
            print("Given sequence: n=%d, obj=%.6f" % (n_given, obj_given))
            if np.isfinite(obj_given) and obj_given < global_best:
                global_best = obj_given
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given sequence: %s" % str(e))

    # ============ Phase 1: GS Exploration (25% time) ============
    explore_budget = 0.25 * (deadline - time.monotonic())
    print("Phase 1: GS exploration, budget=%.0fs" % explore_budget)

    lengths = [2048, 4096, 8192]
    if given_a is not None:
        lengths.insert(0, len(given_a))
    # Remove duplicates while preserving order
    seen = set()
    unique_lengths = []
    for ln in lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)
    lengths = unique_lengths

    init_names = ["uniform", "random", "cosine", "two_humps", "flat_cos", "gaussian"]

    for n in lengths:
        if time.monotonic() > deadline - 120.0:
            break
        S = np.sqrt(2.0 * float(n))
        N = 1 << (2 * n - 1).bit_length()

        # Limit GS iterations based on time
        per_init_budget = min(8.0, explore_budget / (len(lengths) * len(init_names)))
        gs_iters = max(50, int(per_init_budget / 0.02))

        for init_name in init_names:
            if time.monotonic() > deadline - 120.0:
                break
            if per_init_budget < 0.5:
                break

            a_init = get_initialization(init_name, n, rng, S)

            t0 = time.monotonic()
            a_gs, obj_gs = gs_search(a_init, S, N, iters=gs_iters,
                                     relaxation=0.7, verbose=False)
            elapsed = time.monotonic() - t0

            # Compute true objective
            obj_true = 2.0 * n * obj_gs / S ** 2

            if obj_true < global_best:
                global_best = obj_true
                global_a = a_gs.copy()
                print("  GS: n=%d init=%s obj=%.6f (t=%.1fs)" %
                      (n, init_name, obj_true, elapsed))

    # Also run GS on the given sequence
    if given_a is not None:
        remaining = deadline - time.monotonic() - 120.0
        if remaining > 5.0:
            n_g = len(given_a)
            S_g = np.sqrt(2.0 * float(n_g))
            N_g = 1 << (2 * n_g - 1).bit_length()
            gs_iters = max(50, int(min(10.0, remaining * 0.5) / 0.02))

            t0 = time.monotonic()
            a_gs, obj_gs = gs_search(given_a, S_g, N_g, iters=gs_iters,
                                     relaxation=0.5, verbose=False)
            elapsed = time.monotonic() - t0
            obj_true = 2.0 * n_g * obj_gs / S_g ** 2
            if obj_true < global_best:
                global_best = obj_true
                global_a = a_gs.copy()
                print("  GS on given: obj=%.6f (t=%.1fs)" % (obj_true, elapsed))

    # ============ Phase 2: Gradient Exploitation (50% time) ============
    exploit_budget = 0.50 * (deadline - time.monotonic())
    print("Phase 2: Gradient exploitation, budget=%.0fs" % exploit_budget)

                                         time_budget=lbfgs_time, max_iter=300)

            if obj_true < global_best:
                global_best = obj_true
                global_a = a_best.copy()
                print("  L-BFGS T=30: obj=%.6f (t=%.1fs)" % (obj_true, elapsed))
            # Update theta
            a_best = np.maximum(a_best, 1e-10)
            a_best = a_best * (S / a_best.sum())
            theta = theta_from_a(a_best, S)
