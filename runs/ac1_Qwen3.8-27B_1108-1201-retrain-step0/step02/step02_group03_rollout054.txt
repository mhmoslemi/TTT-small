def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def grad_soft_max_conv(a, tau):
    """Compute soft-max(conv(a,a)) and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    w = e / np.sum(e)
    value = c_max + tau * np.log(np.sum(e))
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return value, g, c, w

Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions.

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def grad_soft_max_conv(a, tau):
    """Compute soft-max(conv(a,a)) and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    w = e / np.sum(e)
    value = c_max + tau * np.log(np.sum(e))
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return value, g


def optimize_gradient(a_init, time_limit, rng, lr0=3.0):
    """
    Adam gradient descent in log-parameterization with oscillating
    soft-max temperature. Normalizes sum to sqrt(2n).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(a_init, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a / s * S

    theta = np.log(a / S + 1e-12)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)

    tau_high = 50.0 / n
    tau_low = 0.2 / n
    period = 1500

    while time.monotonic() - t0 < time_limit:
        frac = (step % period) / period
        tau = tau_low + (tau_high - tau_low) * (1 + np.cos(np.pi * frac)) / 2

        a = np.exp(theta - np.max(theta))
        a = a / np.sum(a) * S

        value, g_a = grad_soft_max_conv(a, tau)

        inner = np.dot(a, g_a) / S
        g_theta = a * (g_a - inner)

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.2 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 300 == 0:
            a_check = np.exp(theta - np.max(theta))
            a_check = a_check / np.sum(a_check) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
            if step % 3000 == 0:
                elapsed = time.monotonic() - t0
                print(f"      step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={elapsed:.1f}s")

    a_final = a_final / np.sum(a_final) * S
    score_final = local_score(a_final)

def upsample_seq(a, new_n):
    """Upsample a sequence to a new length via linear interpolation."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp = np.clip(a_interp, 1e-6, None)
    return a_interp
