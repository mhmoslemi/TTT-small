"""
Optimized step function search combining adaptive gradient descent with
staircase temperature, LP-based Frank-Wolfe refinement, peak-targeted
mutations, and multi-length exploration. Targets improvement below 1.5027.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax normalization."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def soft_max_and_weights(x, tau):
    """Compute soft-max value and normalized softmax weights."""
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    return x_max + tau * np.log(s), e / s


def grad_soft_max_conv(a, tau):
    """Compute soft-max(conv(a,a)) and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_and_weights(c, tau)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = (2.0 / tau) * corr[n - 1:2 * n - 1]
    return value, g, c


def grad_peak_weighted_conv(a, tau, peak_weight=2.0, top_k=5):
    """Gradient with extra weight on top-k peaks."""
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_and_weights(c, tau)
    # Boost weights on top peaks
    top_idx = np.argsort(c)[-top_k:]
    w_boosted = w.copy()
    w_boosted[top_idx] *= peak_weight
    w_boosted /= np.sum(w_boosted)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w_boosted)
    g = (2.0 / tau) * corr[n - 1:2 * n - 1]
    return value, g, c


def lp_refinement(a, n_max_iters=100):
    """
    LP-based Frank-Wolfe refinement.
    Given a, find b maximizing sum(b) s.t. b>=0, (a*b)[k] <= M for all k.
    Returns the direction b (normalized to same sum as a).
    """
    n = len(a)
    c = fft_convolve(a, a)
    M = np.max(c)
    n_c = 2 * n - 1

    # LP: max sum(b) s.t. A @ b <= M, b >= 0
    # A[k, i] = coefficient of b[i] in (a*b)[k]
    # (a*b)[k] = sum_i a[max(0,k-n+1):min(k+1,n)] ... 
    # Actually (a*b)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * b[k-i]
    # So for fixed k, (a*b)[k] = sum_{j: 0<=j<n, 0<=k-j<n} b[j] * a[k-j]
    # Coefficient of b[j] in (a*b)[k] is a[k-j] if 0<=k-j<n, else 0

    # Build constraint matrix A (n_c x n)
    A = np.zeros((n_c, n))
    for k in range(n_c):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            A[k, i] = a[k - i]

    # Objective: maximize sum(b) -> minimize -sum(b)
    c_obj = -np.ones(n)

    # Bounds: b >= 0
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=np.full(n_c, M), bounds=bounds,
                  method='highs', options={'presolve': True})

    if res.status == 0 and res.x is not None:
        b = np.maximum(0, res.x)
        s = np.sum(b)
        if s > 1e-10:
            return b / s  # normalize to sum=1
    return None


def fft_phase_init(n, rng, n_iter=80):
    """Gerchberg-Saxton style initialization."""
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft <<= 1

    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def chirp_init(n, rng, alpha=0.5):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def cosine_taper_init(n, rng):
    """Cosine-tapered initialization (Bell-shaped)."""
    t = np.linspace(0, 1, n)
    a = 0.3 + 0.7 * np.sin(np.pi * t) ** 2
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def upsample(a, new_n, rng):
    """Upsample a sequence to a longer length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.015 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)

        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current
        else:
            candidate = candidate / s

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            no_improve += 1
            delta = new_score - current_score
            if delta < temp * 5:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score

        temp *= cooling
        if no_improve > max_no_improve:
            # Restart from best
            current = best.copy() * np.exp(rng.normal(0, 0.05, size=n))
            current = np.clip(current, 0, None)
            current = current / np.sum(current)
            current_score = local_score(current)
            no_improve = 0
            temp = 0.002

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: gradient descent with staircase temperature,
    LP Frank-Wolfe refinement, peak-targeted local search, and
    multi-length exploration to minimize 2*n*max(conv)/sum^2.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Phase 1: Generate diverse initial candidates
    print("\n=== Phase 1: Initialization ===")
    t_phase1_start = time.monotonic()
    phase1_budget = min(120.0, time_budget * 0.06)
    candidates = []

    lengths_to_try = [192, 256, 320, 384, 512]

    for n in lengths_to_try:
        if time.monotonic() - t_phase1_start > phase1_budget:
            break

        inits = []

        # FFT phase init
        a1 = fft_phase_init(n, rng, n_iter=60)
        inits.append(('fft60', a1))

        a2 = fft_phase_init(n, rng, n_iter=120)
        inits.append(('fft120', a2))

        # Chirp
        a3 = chirp_init(n, rng, alpha=0.3 + 0.3 * rng.random())
        inits.append(('chirp', a3))

        # Cosine taper
        a4 = cosine_taper_init(n, rng)
        inits.append(('cosine', a4))

        # Random
        t = np.arange(n, dtype=np.float64)
        a5 = 1.0 + 0.3 * rng.standard_normal(n)
        a5 = np.clip(a5, 0.01, None)
        a5 = a5 / np.sum(a5)
        inits.append(('random', a5))

        for name, a_init in inits:
            score = local_score(a_init)
            print(f"  n={n}, {name}: score={score:.6f}")
            candidates.append((score, n, a_init.copy(), name))

    # Also add the provided sequence as a candidate
    if best_seq is not None:
        candidates.append((best_score, len(best_seq), best_seq.copy(), 'provided'))

    candidates.sort(key=lambda x: x[0])
    print(f"\n  Top 5: {[(f'{c[0]:.4f}', c[1], c[3]) for c in candidates[:5]]}")

    # Phase 2: Gradient optimization on best candidates
    print("\n=== Phase 2: Gradient optimization ===")
    t_phase2_start = time.monotonic()

    # Select top candidates, ensuring diversity in length
    selected = []
    used_lengths = set()
    for c in candidates:
        if len(selected) >= 6:
            break
        # Allow at most 2 per length
        count_for_len = sum(1 for s in selected if s[1] == c[1])
        if count_for_len < 2:
            selected.append(c)

    # Reserve time for phases 3 and 4
    reserve = 300.0
    phase2_budget = max(60.0, (deadline - time.monotonic()) - reserve)
    time_per_cand = phase2_budget / len(selected)

    optimized = []
    for i, (score0, n, a, name) in enumerate(selected):
        if time.monotonic() > deadline - reserve - 30:
            break
        t_alloc = time_per_cand * (1.0 if n <= 320 else 0.7)
        t_alloc = max(30.0, min(t_alloc, deadline - time.monotonic() - reserve - 30))

        print(f"  [{i+1}/{len(selected)}] n={n}, init={name}, score0={score0:.6f}, t_alloc={t_alloc:.1f}s")
        a_opt, s_opt = optimize_gradient_v2(n, a, t_alloc, rng, use_lp=True)
        print(f"    -> score={s_opt:.6f}")

        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")

        optimized.append((s_opt, n, a_opt))

    # Phase 3: Coarse-to-fine upsample from best short sequence
    print("\n=== Phase 3: Coarse-to-fine upsample ===")
    if optimized:
        optimized.sort(key=lambda x: x[0])
        best_short_score, best_short_n, best_short_a = optimized[0]

        for target_n in [384, 512, 768]:
            if time.monotonic() > deadline - 120:
                break
            if best_short_n >= target_n:
                continue
            a_up = upsample(best_short_a, target_n, rng)
            t_alloc = min(80.0, deadline - time.monotonic() - 100)
            if t_alloc < 20:
                break
            print(f"  Upsample to n={target_n} (from n={best_short_n}), t_alloc={t_alloc:.1f}s")
            a_opt, s_opt = optimize_gradient_v2(target_n, a_up, t_alloc, rng, use_lp=True)
            print(f"    -> score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} (n={target_n}) ***")
                optimized.append((s_opt, target_n, a_opt))

    # Phase 4: Local search polish on best candidate
    print(f"\n=== Phase 4: Local search polish ===")
    if best_seq is not None:
        polish_time = min(300.0, max(30.0, deadline - time.monotonic() - 30))
        print(f"  Polishing n={len(best_seq)}, score={best_score:.6f}, time={polish_time:.1f}s")
        polished, p_score = local_search_v2(best_seq, polish_time, rng, top_k=5)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
        else:
            print(f"  Polish didn't improve: {p_score:.6f} vs {best_score:.6f}")

    # Phase 5: Additional restarts if time remains
    remaining = deadline - time.monotonic() - 25
    if remaining > 60 and best_seq is not None:
        print(f"\n=== Phase 5: Additional restarts (remaining={remaining:.1f}s) ===")
        n_best = len(best_seq)
        n_restarts = min(4, int(remaining // 50))

        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break

            # Perturb best sequence
            perturbation = rng.normal(0, 0.08, size=n_best)
            a_new = best_seq * np.exp(perturbation)
            a_new = np.clip(a_new, 1e-6, None)
            a_new = a_new / np.sum(a_new)

            t_alloc = min(40.0, (deadline - time.monotonic() - 25) / (n_restarts - r))
            a_opt, s_opt = optimize_gradient_v2(n_best, a_new, t_alloc, rng, use_lp=False)
            print(f"  Restart {r+1}: score={s_opt:.6f}")

            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} ***")
                # Quick polish
                if deadline - time.monotonic() > 30:
                    p2, s2 = local_search_v2(best_seq, min(25.0, deadline - time.monotonic() - 25), rng)
                    if s2 < best_score:
                        best_score = s2
                        best_seq = p2.copy()
                        print(f"    After quick polish: {best_score:.6f}")

    # Final verification
    print(f"\n{'='*50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
