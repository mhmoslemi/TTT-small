    # So T2 = (w * a)[0:n]
    
    conv_wa = fft_convolve(w, a)  # length (2n-1) + n - 1 = 3n-2
    T2 = conv_wa[:n]
    
    g = 2.0 * (T1 + T2)
    
    return value, g, c, w

def exact_subgrad(a):
    """Compute exact subgradient at the peak."""
    n = len(a)
    c = fft_convolve(a, a)
    k_star = int(np.argmax(c))
    
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= k_star - j < n:
            val += a[k_star - j]
        if 0 <= j - k_star < n:
            val += a[j - k_star]
        g[j] = 2.0 * val
    
    return c[k_star], g, k_star

def water_filling(a, target_sum):
    """Project a onto {x >= 0, sum(x) = target_sum}."""
    n = len(a)
    # Sort in decreasing order
    sorted_a = np.sort(a)[::-1]
    cumsum = np.cumsum(sorted_a)
    # Find the largest k such that sorted_a[k-1] > (target_sum - cumsum[k-1]) / k
    rho = np.where(sorted_a > (target_sum - np.arange(1, n+1) * sorted_a + np.arange(1, n+1) * sorted_a) / np.arange(1, n+1))[0]
    
    # Actually, let me use the standard water-filling:
    # We want max(a - t, 0) to sum to target_sum.
    # Sort a in decreasing order: a_(1) >= a_(2) >= ... >= a_(n)
    # Find the largest k such that a_(k) > (target_sum - sum_{i=1}^{k-1} a_(i)) / k
    # Then t = (target_sum - sum_{i=1}^{k-1} a_(i)) / k
    
    sorted_a = np.sort(a)[::-1]
    cumsum = np.concatenate([[0], np.cumsum(sorted_a)])
    # Condition: sorted_a[k-1] > (target_sum - cumsum[k-1]) / k for k = 1, ..., n
    ks = np.arange(1, n+1)
    thresholds = (target_sum - cumsum[:n]) / ks
    valid = sorted_a > thresholds
    if not np.any(valid):
        t = target_sum / n
        return np.full(n, target_sum / n)
    k = np.max(ks[valid])
    t = (target_sum - cumsum[k-1]) / k
    return np.maximum(a - t, 0.0)

def optimize_gradient(n, a_init, time_limit, rng, S_target=None):
    """
    Main gradient optimization loop with soft-max smoothing.
    """
    if S_target is None:
        S_target = np.sqrt(2.0 * n)
    
    a = a_init.astype(np.float64).copy()
    a = (a / np.sum(a)) * S_target  # normalize
    n = len(a)
    
    # Adam parameters
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.985, 1e-10
    step = 0
    
    best_a = a.copy()
    best_score = compute_score(a)
    no_improve = 0
    patience = 2000
    
    t0 = time.monotonic()
    print(f"  Init: score={best_score:.6f}, n={n}")
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        
        # Adaptive temperature
        if elapsed_frac < 0.5:
            tau = (50.0 / n) * (0.1 / n) ** (elapsed_frac / 0.5)
        elif elapsed_frac < 0.85:
            tau = (0.1 / n) * (0.01 / n) ** ((elapsed_frac - 0.5) / 0.35)
        else:
            tau = 0.005 / n
        
        # Compute gradient
        value, g, c, w = soft_max_grad(a, tau)
        
        # Normalize gradient by S_target (since we're working in a-space, not probability space)
        # The gradient of max(c) w.r.t. a is g.
        # We want to minimize max(c) subject to sum(a) = S_target and a >= 0.
        # Lagrangian: max(c) - lambda * (sum(a) - S_target)
        # Gradient: g - lambda * 1
        # At optimum: (g - lambda) is in the normal cone of {a >= 0}
        # So: for a[j] > 0, g[j] = lambda; for a[j] = 0, g[j] >= lambda.
        # lambda = weighted average of g over support: lambda = sum(a*g) / sum(a)
        inner = np.dot(a, g) / S_target
        g_projected = g - inner  # Remove the component along the constraint
        
        # Adam update
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g_projected
        v_buf = beta2 * v_buf + (1 - beta2) * g_projected ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)
        
        # Learning rate
        base_lr = 1.0 / np.sqrt(step + 1)
        if elapsed_frac < 0.2:
            lr = base_lr * 2.0
        elif elapsed_frac < 0.6:
            lr = base_lr
        else:
            lr = base_lr * 0.3
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = water_filling(a_new, S_target)
        
        # Check score
        if step % 100 == 0:
            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 100
                a = a_new.copy()  # Still accept (for exploration)
            
            if no_improve > patience:
                # Escape: random perturbation
                r = rng.random()
                if r < 0.4:
                    a = best_a + 0.1 * S_target * rng.standard_normal(n)
                    a = water_filling(a, S_target)
                elif r < 0.7:
                    # Targeted perturbation: reduce peak contributors
                    c_cur = fft_convolve(a, a)
                    peak_idx = int(np.argmax(c_cur))
                    contribs = np.zeros(n)
                    for i in range(n):
                        if 0 <= peak_idx - i < n:
                            contribs[i] += a[i] * a[peak_idx - i]
                    top_k = np.argsort(contribs)[::-1][:max(4, n//8)]
                    a[top_k] *= (1 - 0.1 * rng.uniform(0.5, 1.5, len(top_k)))
                    a = water_filling(a, S_target)
                else:
                    # Block scaling
                    block_size = max(2, n // 12)
                    start = int(rng.integers(0, n - block_size))
                    a[start:start+block_size] *= np.exp(rng.normal(0, 0.1, block_size))
                    a = water_filling(a, S_target)
                no_improve = 0
            
            if step % 2000 == 0:
                cur_score = compute_score(a)
                print(f"    step={step}, cur={cur_score:.6f}, best={best_score:.6f}, tau={tau:.2e}")
        else:
            a = a_new.copy()
    
    # Final check
    final_score = compute_score(best_a)
    return best_a, final_score

def lp_direction(f, M, n):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and (f*g)[k] <= M for all k.
    Returns g (the LP solution) or None if infeasible.
    """
    # Variables: g[0], ..., g[n-1] >= 0
    # Objective: maximize sum(g) -> minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_i f[i] * g[k-i] (with g[j]=0 if j out of [0,n-1])
    # For each k, the constraint is: sum_i f[i] * g[k-i] <= M
    # Coefficient of g[j] in constraint k: f[k-j] (if 0 <= k-j < n)
    
    # Build A_ub: (2n-1) x n matrix
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]
    
    b_ub = np.full(2*n-1, M)
    
    # Non-negativity: g >= 0 (handled by bounds)
    bounds = [(0, None)] * n
    
    try:
        from scipy.optimize import linprog
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
        else:
            return None
    except:
        return None

def make_init(n, rng, method='default'):
    """Create initial sequence."""
    if method == 'default':
        a = np.ones(n)
        a += 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'edge_heavy':
        t = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * (t - 0.5)**2 * 4  # parabolic, heavy at edges
        a += 0.1 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'periodic':
        period = max(3, int(np.sqrt(n)))
        a = np.ones(n)
        for i in range(n):
            a[i] = 1.0 + 0.5 * np.cos(2 * np.pi * i / period)
        a += 0.1 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'chirp':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.5 * np.abs(np.cos(np.pi * t * t / n))
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    else:
        a = np.ones(n) + 0.2 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

def local_polish(a_init, time_limit, rng):
    """
    Local polishing using coordinate descent and line search.
    """
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    n = len(a)
    a = a / S * np.sqrt(2*n)  # normalize
    
    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    t0 = time.monotonic()
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        # Pick a random coordinate
        j = int(rng.integers(0, n))
        
        # Try scaling this coordinate
        base = a[j]
        if base < 1e-8:
            a[j] = 1e-8
            new_score = compute_score(a)
            if new_score < cur_score:
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
            else:
                a[j] = base
            continue
        
        # Geometric search
        improved = False
        for factor in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
            a[j] = base * factor
            a = water_filling(a, np.sqrt(2*n))
            new_score = compute_score(a)
            if new_score < cur_score:
                cur_score = new_score
                improved = True
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
            else:
                a[j] = base
                a = water_filling(a, np.sqrt(2*n))
                break
        
        if not improved and iteration % n == 0:
            # Random perturbation
            a *= np.exp(rng.normal(0, 0.02, n))
            a = water_filling(a, np.sqrt(2*n))
            new_score = compute_score(a)
            if new_score < cur_score * 1.01:
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
        
        iteration += 1
        if iteration % 5000 == 0:
            print(f"    Polish iter={iteration}, cur={cur_score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score

"""
Peak-balancing optimization for non-negative step functions minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses soft-max gradient descent with water-filling projection, LP-based direction finding,
peak shaving mutations, and multi-phase search with diverse initializations.

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def soft_max_grad(a, tau):
    """
    Compute soft-max smoothed value and gradient of max(convolution).
    Returns (smoothed_value, gradient, convolution, weights).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)

    # T1[j] = sum_l a[l] * w[l+j] = conv(a, w_flipped)[2n-2 + j]
    # T2[j] = sum_l w[l] * a[j-l] = conv(w, a)[j]
    w_flipped = w[::-1]
    conv_aw = fft_convolve(a, w_flipped)
    T1 = conv_aw[2*n-2 : 3*n-2]
    conv_wa = fft_convolve(w, a)
    T2 = conv_wa[:n]

    g = T1 + T2
    return value, g, c, w


def exact_subgrad(a):
    """Compute exact subgradient at the peak of the convolution."""
    n = len(a)
    c = fft_convolve(a, a)
    k_star = int(np.argmax(c))
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= k_star - j < n:
            val += a[k_star - j]
        if 0 <= j - k_star < n:
            val += a[j - k_star]
        g[j] = 2.0 * val
    return c[k_star], g, k_star


def water_filling(a, target_sum):
    """Project a onto {x >= 0, sum(x) = target_sum} via water-filling."""
    n = len(a)
    sorted_a = np.sort(a)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(sorted_a)])
    ks = np.arange(1, n + 1, dtype=np.float64)
    thresholds = (target_sum - cumsum[:n]) / ks
    valid = sorted_a > thresholds
    if not np.any(valid):
        return np.full(n, target_sum / n)
    k = int(np.max(ks[valid]))
    t = (target_sum - cumsum[k-1]) / k
    return np.maximum(a - t, 0.0)


def optimize_gradient(n, a_init, time_limit, rng, S_target=None):
    """Main gradient optimization with soft-max smoothing and water-filling."""
    if S_target is None:
        S_target = np.sqrt(2.0 * n)

    a = a_init.astype(np.float64).copy()
    a = (a / max(np.sum(a), 1e-10)) * S_target
    n = len(a)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.985, 1e-10
    step = 0

    best_a = a.copy()
    best_score = compute_score(a)
    no_improve = 0
    patience = 3000

    t0 = time.monotonic()
    print(f"  Init: score={best_score:.6f}, n={n}, S={S_target:.4f}")

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit

        # Adaptive temperature schedule
        if elapsed_frac < 0.4:
            tau = (80.0 / n) * (0.2 / n) ** (elapsed_frac / 0.4)
        elif elapsed_frac < 0.8:
            tau = (0.2 / n) * (0.02 / n) ** ((elapsed_frac - 0.4) / 0.4)
        else:
            tau = 0.005 / n

        # Compute gradient
        value, g, c, w = soft_max_grad(a, tau)

        # Remove component along sum constraint
        inner = np.dot(a, g) / S_target
        g_proj = g - inner

        # Adam update
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g_proj
        v_buf = beta2 * v_buf + (1 - beta2) * g_proj ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)

        base_lr = 1.5 / np.sqrt(step + 1)
        if elapsed_frac < 0.15:
            lr = base_lr * 2.0
        elif elapsed_frac < 0.6:
            lr = base_lr
        else:
            lr = base_lr * 0.4

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = water_filling(a_new, S_target)

        if step % 50 == 0:
            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 50

            a = a_new.copy()

            if no_improve > patience:
                r = rng.random()
                if r < 0.35:
                    # Gaussian kick from best
                    a = best_a + 0.08 * S_target * rng.standard_normal(n)
                    a = water_filling(np.maximum(a, 0), S_target)
                elif r < 0.6:
                    # Peak-directed: reduce top contributors to peak
                    c_cur = fft_convolve(a, a)
                    peak_idx = int(np.argmax(c_cur))
                    contribs = np.zeros(n)
                    for i in range(n):
                        if 0 <= peak_idx - i < n:
                            contribs[i] = a[i] * a[peak_idx - i]
                    top_k = np.argsort(contribs)[::-1][:max(4, n // 10)]
                    a[top_k] *= (1.0 - 0.08 * rng.uniform(0.5, 1.5, len(top_k)))
                    a = water_filling(np.maximum(a, 0), S_target)
                elif r < 0.8:
                    # Block rebalancing
                    block_size = max(2, n // 10)
                    start = int(rng.integers(0, n - block_size))
                    a[start:start+block_size] *= np.exp(rng.normal(0, 0.15, block_size))
                    a = water_filling(np.maximum(a, 0), S_target)
                else:
                    # Mass transfer: reduce high, increase low
                    sorted_idx = np.argsort(a)
                    n_move = max(4, n // 12)
                    amt = 0.02 * S_target / n_move
                    a[sorted_idx[-n_move:]] -= amt
                    a[sorted_idx[:n_move]] += amt
                    a = water_filling(np.maximum(a, 0), S_target)
                no_improve = 0

            if step % 2000 == 0:
                cur_score = compute_score(a)
                print(f"    step={step}, cur={cur_score:.6f}, best={best_score:.6f}, tau={tau:.2e}")

    final_score = compute_score(best_a)
    return best_a, final_score


def lp_direction(f, M, n):
    """Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'dual_infeasibility_tolerance': 1e-7})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def local_polish(a_init, time_limit, rng):
    """Coordinate descent and peak-shaving local polish."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    a = (a / max(np.sum(a), 1e-10)) * S_target

    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    t0 = time.monotonic()
    iteration = 0
    temp = 0.01
    cooling = 0.9999

    while time.monotonic() - t0 < time_limit:
        r = rng.random()

        if r < 0.5:
            # Coordinate scaling
            j = int(rng.integers(0, n))
            base = a[j]
            if base < 1e-10:
                a[j] = 1e-10
                new_score = compute_score(a)
                if new_score <= cur_score:
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                else:
                    a[j] = base
                iteration += 1
                continue
            for factor in [0.6, 0.8, 0.9, 1.1, 1.2, 1.4]:
                a[j] = base * factor
                a = water_filling(np.maximum(a, 0), S_target)
                new_score = compute_score(a)
                if new_score < cur_score:
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                else:
                    a[j] = base
                    a = water_filling(np.maximum(a, 0), S_target)
                    break
        elif r < 0.75:
            # Peak shaving: reduce top contributors, boost bottom
            c_cur = fft_convolve(a, a)
            peak_idx = int(np.argmax(c_cur))
            contribs = np.zeros(n)
            for i in range(n):
                if 0 <= peak_idx - i < n:
                    contribs[i] = a[i] * a[peak_idx - i]
            top_k = max(3, n // 15)
            bot_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            bot_idx = np.argsort(contribs)[:bot_k]
            red = rng.uniform(0.02, 0.06)
            a[top_idx] *= (1.0 - red)
            a[bot_idx] *= (1.0 + red * top_k / bot_k)
            a = water_filling(np.maximum(a, 0), S_target)
            new_score = compute_score(a)
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
        elif r < 0.9:
            # Mass transfer
            n_move = max(2, n // 20)
            sorted_idx = np.argsort(a)
            amt = 0.01 * S_target / n_move
            a[sorted_idx[-n_move:]] -= amt
            a[sorted_idx[:n_move]] += amt
            a = water_filling(np.maximum(a, 0), S_target)
            new_score = compute_score(a)
            if new_score < cur_score:
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
            else:
                a[sorted_idx[-n_move:]] += amt
                a[sorted_idx[:n_move]] -= amt
                a = water_filling(np.maximum(a, 0), S_target)
        else:
            # Small global perturbation
            a *= np.exp(rng.normal(0, 0.01, n))
            a = water_filling(np.maximum(a, 0), S_target)
            new_score = compute_score(a)
            if new_score < cur_score * 1.005:
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()

        temp *= cooling
        iteration += 1
        if iteration % 3000 == 0:
            print(f"    Polish iter={iteration}, cur={cur_score:.6f}, best={best_score:.6f}")

    return best_a, best_score


def make_init(n, rng, method='edge_heavy'):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if method == 'edge_heavy':
        # Heavier at edges to counteract center peak in convolution
        norm_t = t / (n - 1)
        a = 1.0 + 1.5 * (norm_t - 0.5) ** 2 * 4
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'broadband':
        a = np.ones(n)
        for k in range(1, 12):
            freq = k * (1.0 + 0.5 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.2 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'chirp':
        a = 1.0 + 0.4 * np.abs(np.cos(np.pi * t * t / (2.0 * n)))
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'two_hump':
        centers = [n * 0.2, n * 0.8]
        widths = [n * 0.15, n * 0.15]
        a = np.ones(n) * 0.5
        for c_, w_ in zip(centers, widths):
            a += 1.5 * np.exp(-0.5 * ((t - c_) / w_) ** 2)
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    else:
        a = 1.0 + 0.2 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

        else:
            a0_try = make_init(n_try, rng, 'edge_heavy')
        a0_try = a0_try / max(np.sum(a0_try), 1e-10) * S_try

        print(f"  Trying n={n_try} (t={t_try:.0f}s)...")
        a_res, s_try = optimize_gradient(n_try, a0_try, t_try, rng, S_try)
        actual_try = compute_score(a_res)
        print(f"  n={n_try}: score={actual_try:.6f}")

        if actual_try < best_score:
            best_score = actual_try
            best_seq = a_res.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === PHASE 4: Final polish ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_final = deadline - time.monotonic() - 10
        if t_final > 15:
            print(f"\n{'='*50}")
            print(f"PHASE 4: Final polish (t={t_final:.0f}s)")
            print(f"{'='*50}")
            final_polished, final_score = local_polish(best_seq, t_final, rng)
            if final_score < best_score:
                best_score = final_score
                best_seq = final_polished.copy()
            print(f"  Final local: {best_score:.6f}")

    # Also try a quick LP-guided improvement on the best
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        S_t = np.sqrt(2.0 * n)
        a = (best_seq / max(np.sum(best_seq), 1e-10)) * S_t
        c = fft_convolve(a, a)
        M = np.max(c)
        g_lp = lp_direction(a, M * 1.01, n)
        if g_lp is not None and np.sum(g_lp) > 0:
            for t_blend in [0.05, 0.1, 0.2, 0.3]:
                a_blend = (1 - t_blend) * a + t_blend * g_lp
                a_blend = water_filling(np.maximum(a_blend, 0), S_t)
                s_blend = compute_score(a_blend)
                if s_blend < best_score:
                    best_score = s_blend
                    best_seq = a_blend.copy()
                    print(f"  LP blend t={t_blend}: {best_score:.6f}")

    # Verify with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'='*50}")
        if final_eval != np.inf and final_eval < best_score:
            best_score = final_eval
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
