"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.
Multi-phase approach: multi-scale init, FISTA with adaptive LSE, 
ramp-weighted top-k subgradient, spectral diversity, local search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_lse(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_ramp(a, c, top_k, decay=0.1):
    """Gradient using top-k lags with exponential ramp weighting."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    ranks = np.arange(top_k, dtype=np.float64)
    weights = np.exp(-decay * ranks)
    weights /= weights.sum()
    
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_topk(a, c, top_k):
    """Gradient using top-k active lags with uniform weights."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    
    w_full = np.zeros(2 * n - 1)
    for k in top_indices:
        w_full[int(k)] = 1.0 / top_k
    
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista_adaptive(a_init, n, S, max_time, rng, 
                        beta_init=3.0, beta_max=500.0, 
                        beta_sched='linear', noise_period=5000, noise_level=0.003):
    """FISTA with adaptive log-sum-exp and simplex projection."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        if beta_sched == 'linear':
            beta = min(beta_max, beta_init + (beta_max - beta_init) * it / (max_time * 50))
        elif beta_sched == 'exponential':
            beta = min(beta_max, beta_init * (1.0 + 0.001 * it))
        else:
            beta = min(beta_max, beta_init * (1.0 + 0.003 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        grad = fft_grad_lse(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (beta * 2.0)
        x_new = project_simplex(y - step * grad, S)
        
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it > 0 and it % noise_period == 0 and rng.random() < 0.2:
            y = project_simplex(y + rng.normal(0, noise_level * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score


def run_nesterov_ramp(a_init, n, S, max_time, rng, 
                       top_k_init=5, top_k_max=256, decay=0.05):
    """Nesterov-accelerated with ramp-weighted top-k subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        grad = fft_grad_ramp(y, c, top_k, decay)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 2.0 / (it + 10)
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))

        y = np.maximum(y, 0.0)

        t = t_new
    
    return best_local, best_score


def run_nesterov_topk(a_init, n, S, max_time, rng,
                       top_k_init=5, top_k_max=256):
    """Nesterov-accelerated top-k subgradient method."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        grad = fft_grad_topk(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 2.0 / (it + 10)
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return best_local, best_score


def spectral_init(n, rng, S):
    """Generate initialization by randomizing DFT phases."""
    # Start with a shape that has good spectral properties
    tt = np.linspace(0, 1, n, endpoint=False)
    base = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt)
    
    # Get its DFT
    A = fft(base)
    
    # Randomize phases (keep magnitudes)
    mags = np.abs(A)
    phases = rng.uniform(0, 2 * np.pi, n)
    A_new = mags * np.exp(1j * phases)
    
    # Ensure Hermitian symmetry for real result
    if n % 2 == 0:
        A_new[n//2] = np.abs(A_new[n//2])
    for i in range(1, n // 2 + 1):
        A_new[n - i] = np.conj(A_new[i])
    
    a = np.real(ifft(A_new))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    return a


def multiscale_init(target_n, S, max_time, rng):
    """Build initialization by solving at increasing scales."""
    scales = [64, 256, 1024, 4096]
    scales = [s for s in scales if s < target_n]
    if target_n <= 4096:
        scales = [s for s in scales if s < target_n]
        if not scales:
            scales = [max(64, target_n // 2)]
    
    current_n = scales[0] if scales else 64
    S_cur = np.sqrt(2.0 * current_n)
    
    # Start with random
    a = rng.random(current_n) + 0.1
    a = a * (S_cur / a.sum())
    
    start = time.monotonic()
    time_per_scale = max_time / len(scales)
    
    for n_s in scales:
        S_s = np.sqrt(2.0 * n_s)
        # Upsample current solution
        if len(a) != n_s:
            # Use linear interpolation to resample
            old_idx = np.linspace(0, 1, len(a), endpoint=False)
            new_idx = np.linspace(0, 1, n_s, endpoint=False)
            a = np.interp(new_idx, old_idx, a)
        
        a = np.maximum(a, 0.0)
        a = a * (S_s / max(a.sum(), 1e-15))
        
        # Run FISTA for a short time
        remaining = time.monotonic() - start
        if remaining > max_time:
            break
        t_alloc = min(time_per_scale, max_time - remaining)
        a, score = run_fista_adaptive(a, n_s, S_s, t_alloc, rng,
                                        beta_init=5.0, beta_max=50.0)
    
    # Upsample to target_n
    if len(a) != target_n:
        old_idx = np.linspace(0, 1, len(a), endpoint=False)
        new_idx = np.linspace(0, 1, target_n, endpoint=False)
        a = np.interp(new_idx, old_idx, a)
    
    a = np.maximum(a, 0.0)
    S = np.sqrt(2.0 * target_n)
    a = a * (S / max(a.sum(), 1e-15))
    return a


def local_search(a, n, S, max_time, rng):
    """Local search: random coordinate perturbations + gradient steps."""
    a = a.copy()
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    c = fft_convolve_self(a)
    best_score = c.max()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        # Pick a few random coordinates to perturb
        n_perturb = max(1, n // 100)
        indices = rng.choice(n, n_perturb, replace=False)
        
        # Try adding/subtracting small amounts
        best_trial = a.copy()
        best_trial_score = best_score
        
        for idx in indices:
            for delta in [0.01, 0.005, -0.01, -0.005, 0.001, -0.001]:
                trial = best_trial.copy()
                trial[idx] += delta * S / n
                if trial[idx] < 0:
                    continue
                s = trial.sum()
                trial = trial * (S / s)
                c = fft_convolve_self(trial)
                sc = c.max()
                if sc < best_trial_score:
                    best_trial_score = sc
                    best_trial = trial.copy()
        
        if best_trial_score < best_score - 1e-10:
            a = best_trial
            best_score = best_trial_score
            best_local = a.copy()
            print("    Local search improvement: %.6f (iter %d)" % (best_score, it))
    
    return best_local, best_score

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Multi-phase approach:
- Multi-scale initialization (solve at small n, upscale)
- Spectral initialization (randomized DFT phases)
- FISTA with adaptive log-sum-exp (smooth -> sharp)
- Ramp-weighted top-k subgradient for polishing
- Nesterov acceleration with plateau detection and restarts
- Local coordinate search for final refinement
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_lse(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a_j.
    Uses FFT: grad[j] = 2 * sum_k w[k] * c_cross[k,j]
    where c_cross is the cross-correlation of w with a.
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_ramp(a, c, top_k, decay=0.05):
    """Gradient using top-k lags with exponential ramp weighting."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    ranks = np.arange(top_k, dtype=np.float64)
    weights = np.exp(-decay * ranks)
    weights /= weights.sum()

    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]

    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_topk(a, c, top_k):
    """Gradient using top-k active lags with uniform weights."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]

    w_full = np.zeros(2 * n - 1)
    for k in top_indices:
        w_full[int(k)] = 1.0 / top_k

    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista(a_init, n, S, max_time, rng,
              beta_init=3.0, beta_max=300.0,
              noise_period=4000, noise_prob=0.15, noise_level=0.004):
    """FISTA with adaptive log-sum-exp and simplex projection."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stall_count = 0
    stall_threshold = 8000
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: exponential growth capped at beta_max
        beta = min(beta_max, beta_init * (1.001 ** it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            last_improve_it = it
            stall_count = 0
        else:
            stall_count += 1

        grad = fft_grad_lse(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: base step / (2*beta) with slight decay for stability
        step = 1.0 / (beta * 2.0 * (1.0 + 0.0001 * it))
        step = max(step, 1e-8)

        x_new = project_simplex(y - step * grad, S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score - 1e-12:
                best_score = c_x.max()
                best_local = x_new.copy()
                last_improve_it = it
                stall_count = 0

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Noise injection for exploration
        if it % noise_period == 0 and rng.random() < noise_prob:
            y = project_simplex(y + rng.normal(0, noise_level * S / np.sqrt(n), n), S)
            t = 1.0

        # Plateau restart
        if stall_count > stall_threshold:
            y = project_simplex(y + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            t = 1.0
            stall_count = 0

    return best_local, best_score


def run_nesterov_ramp(a_init, n, S, max_time, rng,
                      top_k_init=5, top_k_max=256, decay=0.05):
    """Nesterov-accelerated with ramp-weighted top-k subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 80)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()

        grad = fft_grad_ramp(y, c, top_k, decay)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 / (it + 20)
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score - 1e-12:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    return best_local, best_score


def run_nesterov_topk(a_init, n, S, max_time, rng,
                      top_k_init=5, top_k_max=512):
    """Nesterov-accelerated top-k subgradient method."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)

        a = a * (S / s)

    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 80)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))

        y = np.maximum(y, 0.0)
        a = x_new

    return best_local, best_score


def spectral_init(n, rng, S):
    """Generate initialization by randomizing DFT phases of a base shape."""
    tt = np.linspace(0, 1, n, endpoint=False)
    base = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt)
    A = fft(base)
    mags = np.abs(A)
    phases = rng.uniform(0, 2 * np.pi, n)
    A_new = mags * np.exp(1j * phases)
    # Enforce Hermitian symmetry
    A_new[0] = np.abs(A_new[0])
    if n % 2 == 0:
        A_new[n // 2] = np.abs(A_new[n // 2])
    for i in range(1, n // 2 + 1):
        A_new[n - i] = np.conj(A_new[i])
    a = np.real(ifft(A_new))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def multiscale_init(target_n, S, max_time, rng):
    """Build initialization by solving at increasing scales then upscaling."""
    scales = [64, 256, 1024]
    scales = [s for s in scales if s < target_n]
    if not scales:
        scales = [max(64, target_n // 2)]

    # Start with a bump-like shape at smallest scale
    n0 = scales[0]
    S0 = np.sqrt(2.0 * n0)
    tt0 = np.linspace(0, 1, n0, endpoint=False)
    a = np.exp(-0.5 * ((tt0 - 0.5) * 4.0) ** 2) + 0.1
    a = a * (S0 / a.sum())

    start = time.monotonic()
    time_per = max_time / len(scales)

    for n_s in scales:
        S_s = np.sqrt(2.0 * n_s)
        # Upsample via interpolation
        if len(a) != n_s:
            old_idx = np.linspace(0, 1, len(a), endpoint=False)
            new_idx = np.linspace(0, 1, n_s, endpoint=False)
            a = np.interp(new_idx, old_idx, a)
        a = np.maximum(a, 0.0)
        a = a * (S_s / max(a.sum(), 1e-15))

        elapsed = time.monotonic() - start
        t_alloc = min(time_per * 1.5, max_time - elapsed)
        if t_alloc < 1.0:
            break
        a, _ = run_fista(a, n_s, S_s, t_alloc, rng,
                          beta_init=5.0, beta_max=80.0,
                          noise_period=2000, noise_prob=0.1, noise_level=0.005)

    # Final upscale to target_n
    if len(a) != target_n:
        old_idx = np.linspace(0, 1, len(a), endpoint=False)
        new_idx = np.linspace(0, 1, target_n, endpoint=False)
        a = np.interp(new_idx, old_idx, a)
    a = np.maximum(a, 0.0)
    a = a * (S / max(a.sum(), 1e-15))
    return a


def local_search(a, n, S, max_time, rng):
    """Local search: gradient-guided coordinate perturbations."""
    a = a.copy()
    start = time.monotonic()

    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Compute gradient to guide which coordinates to try
        top_k = max(3, min(32, it // 5))
        grad = fft_grad_topk(a, c, top_k)

        # Pick coordinates with largest gradient magnitude
        n_perturb = max(2, n // 2000)
        magnitudes = np.abs(grad)
        indices = np.argsort(magnitudes)[::-1][:n_perturb]

        improved = False
        for idx in indices:
            # Try moving in the -grad direction (which should decrease max)
            for frac in [0.01, 0.005, 0.002]:
                delta = -np.sign(grad[idx]) * frac * S / n
                trial = a.copy()
                trial[idx] += delta
                if trial[idx] < 0:
                    continue
                s_tr = trial.sum()
                trial = trial * (S / s_tr)
                c_tr = fft_convolve_self(trial)
                sc = c_tr.max()
                if sc < best_score - 1e-10:
                    a = trial
                    best_score = sc
                    best_local = a.copy()
                    c = c_tr
                    improved = True
                    break
            if time.monotonic() - start > max_time * 0.9:
                break

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.

    Multi-phase:
    1. Build diverse initializations (given, multi-scale, spectral, random)
    2. Explore with FISTA + Nesterov on each init
    3. Polish best with high-beta FISTA, ramp-weighted Nesterov
    4. Local coordinate search for final refinement
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # --- Determine main length ---
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            main_len = len(given)
        else:
            main_len = 8192
    except Exception:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * n)

    # --- Phase 1: Build initializations ---
    print("=== Phase 1: Initializations ===")
    t0 = time.monotonic()

    inits = []

    # Given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            g = np.clip(given, 0, 1000)
            g = g * (S / g.sum())
            inits.append(("given", g))
    except Exception:
        pass

    # Multi-scale init
    ms_time = 30.0
    ms_init = multiscale_init(n, S, ms_time, rng)
    inits.append(("multiscale", ms_init))

    # Spectral inits (random phase)
    for i in range(2):
        spec = spectral_init(n, rng, S)
        inits.append(("spec%d" % i, spec))

    # Random with slight structure
    tt = np.linspace(0, 1, n, endpoint=False)
    rand_struct = rng.random(n) * 0.5 + 0.5 * (1.0 + 0.2 * np.sin(2 * np.pi * 5 * tt))
    inits.append(("rand_struct", rand_struct))

    # Pure random
    inits.append(("rand", rng.random(n) + 0.05))

    # Normalize all
    normalized_inits = []
    for name, arr in inits:
        arr = safe_normalize(arr, n)
        # Quick score check
        c = fft_convolve_self(arr)
        sc = c.max()
        normalized_inits.append((name, arr, sc))
        print("  Init [%s]: initial score = %.6f" % (name, sc))

    # Sort by initial score (best first)
    normalized_inits.sort(key=lambda x: x[2])
    print("  Built %d inits in %.1fs" % (len(normalized_inits), time.monotonic() - t0))

    # --- Phase 2: Exploration ---
    remaining = deadline - time.monotonic()
    n_inits = len(normalized_inits)
    # Allocate more time to better inits
    weights = np.array([1.0 / (i + 1) for i in range(n_inits)])
    weights /= weights.sum()
    times_alloc = weights * remaining * 0.65
    times_alloc = np.maximum(times_alloc, 10.0)
    # Rescale to fit
    total_alloc = times_alloc.sum()
    if total_alloc > remaining * 0.65:
        times_alloc = times_alloc * (remaining * 0.65 / total_alloc)

    print("=== Phase 2: Exploration ===")

    for i, (name, arr, init_score) in enumerate(normalized_inits):
        if time.monotonic() > deadline - 300:
            break
        t_avail = times_alloc[i]
        t_start = time.monotonic()

        # FISTA with adaptive beta
        seg_fista = t_avail * 0.55
        a_f, score_f = run_fista(arr, n, S, seg_fista, rng,
                                  beta_init=3.0, beta_max=250.0,
                                  noise_period=3000, noise_prob=0.12, noise_level=0.004)
        a_f = safe_normalize(a_f, n)
        c = fft_convolve_self(a_f)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_f.copy()
            print("  FISTA [%s]: %.6f (t=%.0fs)" % (name, global_best, time.monotonic() - t_start))

        # Nesterov ramp on FISTA result
        rem = t_avail - (time.monotonic() - t_start)
        if rem > 5:
            a_r, _ = run_nesterov_ramp(a_f, n, S, rem * 0.4, rng,
                                        top_k_init=3, top_k_max=128, decay=0.05)
            a_r = safe_normalize(a_r, n)
            c_r = fft_convolve_self(a_r)
            if c_r.max() < global_best:
                global_best = c_r.max()
                global_a = a_r.copy()
                print("  Ramp [%s]: %.6f" % (name, global_best))

            rem2 = t_avail - (time.monotonic() - t_start)
            if rem2 > 5:
                a_t, _ = run_nesterov_topk(a_r, n, S, rem2, rng,
                                            top_k_init=5, top_k_max=200)
                a_t = safe_normalize(a_t, n)
                c_t = fft_convolve_self(a_t)
                if c_t.max() < global_best:
                    global_best = c_t.max()
                    global_a = a_t.copy()
                    print("  TopK [%s]: %.6f" % (name, global_best))

    print("=== After exploration: global_best=%.6f ===" % global_best)

    # --- Phase 3: Intensive polishing ---
    if global_a is not None:
        remaining = max(0.0, deadline - time.monotonic())
        print("=== Phase 3: Polishing (%.0f sec) ===" % remaining)

        # 3a: High-beta FISTA
        seg = remaining * 0.25
        a_p, _ = run_fista(global_a, n, S, seg, rng,
                           beta_init=30.0, beta_max=600.0,
                           noise_period=5000, noise_prob=0.08, noise_level=0.002)
        a_p = safe_normalize(a_p, n)
        c_p = fft_convolve_self(a_p)
        if c_p.max() < global_best:
            global_best = c_p.max()
            global_a = a_p.copy()
            print("  Polish FISTA: %.6f" % global_best)

        # 3b: Nesterov top-k with high k
        rem = max(0.0, deadline - time.monotonic())
        seg = rem * 0.25
        a_pk, _ = run_nesterov_topk(global_a, n, S, seg, rng,
                                     top_k_init=32, top_k_max=512)
        a_pk = safe_normalize(a_pk, n)
        c_pk = fft_convolve_self(a_pk)
        if c_pk.max() < global_best:
            global_best = c_pk.max()
            global_a = a_pk.copy()
            print("  Polish TopK: %.6f" % global_best)

        # 3c: Nesterov ramp with different decay
        rem = max(0.0, deadline - time.monotonic())
        seg = rem * 0.2
        a_rm, _ = run_nesterov_ramp(global_a, n, S, seg, rng,
                                     top_k_init=16, top_k_max=256, decay=0.02)
        a_rm = safe_normalize(a_rm, n)
        c_rm = fft_convolve_self(a_rm)
        if c_rm.max() < global_best:
            global_best = c_rm.max()
            global_a = a_rm.copy()
            print("  Polish Ramp(low): %.6f" % global_best)

        # 3d: Local search
        rem = max(0.0, deadline - time.monotonic())
        seg = rem * 0.15
        a_ls, _ = local_search(global_a, n, S, seg, rng)
        a_ls = safe_normalize(a_ls, n)
        c_ls = fft_convolve_self(a_ls)
        if c_ls.max() < global_best:
            global_best = c_ls.max()
            global_a = a_ls.copy()
            print("  Local Search: %.6f" % global_best)

        # 3e: Final FISTA with very high beta
        rem = max(0.0, deadline - time.monotonic())
        if rem > 10:
            a_fp, _ = run_fista(global_a, n, S, rem, rng,
                                 beta_init=80.0, beta_max=1000.0,
                                 noise_period=8000, noise_prob=0.05, noise_level=0.001)
            a_fp = safe_normalize(a_fp, n)
            c_fp = fft_convolve_self(a_fp)
            if c_fp.max() < global_best:
                global_best = c_fp.max()
                global_a = a_fp.copy()
                print("  Final FISTA: %.6f" % global_best)

    # --- Phase 4: Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
