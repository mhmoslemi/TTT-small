    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

"""
Optimizes non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses projected gradient descent with power-sum smoothing of the max,
projecting onto the probability simplex. Tries multiple initializations
and sequence lengths, with adaptive learning rate and p-annealing.

import numpy as np
import time


def fft_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_score(a):
    """Compute the evaluation score directly."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (S * S)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    w = np.arange(1, n + 1, dtype=np.float64)
    sv = np.cumsum(v_sorted)
    cond = v_sorted * w + 1.0 > sv
    if not np.any(cond):
        theta = (sv[-1] - 1.0) / n
        return np.maximum(v - theta, 0.0)
    rho = int(np.max(np.where(cond)[0])) + 1
    theta = (sv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def optimize_sequence(a_init, deadline, p_start=4, p_end=64, lr_init=0.1, max_iters=500000):
    """Optimize a sequence using projected gradient descent with power-sum smoothing."""
    a = a_init.copy()
    S = np.sum(a)
    if S < 0.01:
        a = np.ones(len(a))
        S = len(a)
    a = a / S

    n = len(a)
    best_a = a.copy()
    best_score = compute_score(best_a)

    c = fft_convolve(a)
    c = np.maximum(c, 0)
    best_smooth = np.sum(c ** p_start) ** (1.0 / p_start)

    lr = lr_init
    patience = 0
    max_patience = 2000
    p = p_start

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal p
        progress = min(it / max_iters, 1.0)
        p = p_start * (p_end / p_start) ** progress

        # Compute convolution
        c = fft_convolve(a)
        c = np.maximum(c, 0)

        # Power-sum gradient
        c_pow = c ** (p - 1)
        c_pow_sum = np.sum(c ** p)
        if c_pow_sum < 1e-300:
            break
        smooth_max = c_pow_sum ** (1.0 / p)
        scale = c_pow_sum ** (1.0 / p - 1.0)

        # Gradient via cross-correlation
        corr = np.correlate(c_pow, a, 'full')[:n]
        grad = scale * corr

        # Center gradient to maintain sum = 1
        grad -= np.mean(grad)

        # Gradient step
        new_a = project_simplex(a - lr * grad)

        # Evaluate
        c_new = fft_convolve(new_a)
        c_new = np.maximum(c_new, 0)
        new_score = 2.0 * n * np.max(c_new)  # sum is 1 after projection
        new_smooth = np.sum(c_new ** p) ** (1.0 / p)

        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
            patience = 0

        if new_smooth < best_smooth * 1.001:
            a = new_a
            best_smooth = min(best_smooth, new_smooth)
            lr = min(lr * 1.005, 1.0)
        else:
            a = new_a
            lr *= 0.95

        if lr < 1e-7:
            lr = lr_init * 0.1
            patience = 0

        if patience > max_patience and it % 5000 == 0:
            # Perturb to escape local minimum
            a = project_simplex(a + 0.01 * np.random.standard_normal(n))
            lr = lr_init * 0.5
            patience = 0

    return best_a, best_score


def optimize_with_top_peaks(a_init, deadline, n_peaks=20):
    """Optimize using top-peak subgradient method."""
    a = a_init.copy()
    S = np.sum(a)
    if S < 0.01:
        a = np.ones(len(a))
        S = len(a)
    a = a / S

    n = len(a)
    best_a = a.copy()
    best_score = compute_score(best_a)

    lr = 0.05
    c = fft_convolve(a)
    c = np.maximum(c, 0)
    current_max = np.max(c)

    for it in range(100000):
        if time.monotonic() > deadline:
            break

        c = fft_convolve(a)
        c = np.maximum(c, 0)
        current_max = np.max(c)

        # Find top peaks
        threshold = current_max * 0.95
        peak_indices = np.where(c >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(c)])

        # Compute subgradient
        grad = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                grad[lo:hi + 1] += a[k - np.arange(lo, hi + 1)]
        grad /= len(peak_indices)

        # Center to maintain sum
        grad -= np.mean(grad)

        new_a = project_simplex(a - lr * grad)
        c_new = fft_convolve(new_a)
        c_new = np.maximum(c_new, 0)
        new_max = np.max(c_new)
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()

        if new_max < current_max:
            a = new_a
            lr = min(lr * 1.01, 0.5)
        else:
            a = new_a
            lr *= 0.9

        if lr < 1e-7:
            lr = 0.05
