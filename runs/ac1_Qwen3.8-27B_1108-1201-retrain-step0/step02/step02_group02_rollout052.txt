Multi-phase optimization for minimizing the step function evaluation score.
Uses softmax-parameterized gradient descent with adaptive temperature,
peak-directed local search, and multi-scale exploration across different
sequence lengths. Targets n in [256, 512] with chirp and structured

import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score(a):
    """Fast score computation."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def chirp_init(n, rng, type='quadratic'):
    """Generate chirp-like initializations."""
    t = np.arange(n, dtype=np.float64)
    if type == 'quadratic':
        # a[j] proportional to 1 + A*cos(2*pi*beta*j^2/n)
        beta = rng.uniform(0.5, 3.0)

        a = 1.0 + A * np.cos(2 * np.pi * beta * t**2 / n)
    elif type == 'linear':
        beta = rng.uniform(0.5, 5.0)
        A = rng.uniform(0.3, 0.8)
        a = 1.0 + A * np.cos(2 * np.pi * beta * t / n)
    elif type == 'multi':
        a = np.ones(n)
        for k in range(3, 10):
            freq = rng.uniform(0.5, k * 2.0)
            phase = rng.uniform(0, 2 * np.pi)
            a += 0.2 / k * np.cos(2 * np.pi * freq * t / n + phase)
    else:
        a = 1.0 + 0.5 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

    # dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    # Using FFT: gradient[j] = 2 * (a_rev * w)[j + n - 1]
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    grad_a = 2.0 * corr[n-1:2*n-1]
    
    # score = 2n * value / S^2 where value = c_max + tau*log(sum_exp)
    value = c_max + tau * np.log(np.sum(exp_s))
    # d(score)/d(theta[j]) = (2n/S^2) * S_target * (grad_a[j] - <a, grad_a>/S_target)
    # where the second term comes from the softmax constraint
    inner = np.dot(a, grad_a) / S_target
    grad_theta = S_target * (grad_a - inner)
    
    return value, grad_theta, c, w

def optimize_gradient(n, a_init, time_limit, rng, seed_offset=0):

def peak_shave(a_init, time_limit, rng):
    """
    Peak-directed local search: iteratively reduce peak contributors
    and redistribute mass.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = score(current)
    best = current.copy()
    best_score_val = current_score
    
    t0 = time.monotonic()
    step = 0
    lr = 0.03
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))
        
        # Compute contributions to peak
        contribs = np.zeros(n)
        for i in range(n):
            if 0 <= peak_idx - i < n:
                contribs[i] += current[i] * current[peak_idx - i]
        
        r = rng.random()
        
        if r < 0.4:
            # Reduce top contributors, boost bottom
            top_k = max(3, n // 20)
            bot_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            bot_idx = np.argsort(contribs)[:bot_k]
            
            red_amt = current[top_idx].sum() * lr
            add_amt = red_amt * rng.uniform(0.5, 1.0)
            
            for i in top_idx:
                current[i] = max(0.0, current[i] - red_amt / top_k)
            for i in bot_idx:
                current[i] += add_amt / bot_k
        
        elif r < 0.6:
            # Single element modification
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            j = top_idx[int(rng.integers(0, top_k))]
            current[j] *= np.exp(-rng.uniform(0.05, 0.2))
            # Compensate elsewhere
            k = int(rng.integers(0, n))
            if k != j:
                current[k] += current[j] * 0.1
                current[k] = min(current[k], 1000.0)
        
        elif r < 0.8:
            # Pair swap
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            j1 = top_idx[int(rng.integers(0, top_k))]
            j2 = int(rng.integers(0, n))
            if j1 != j2:
                amt = min(current[j1], current[j2]) * rng.uniform(0.05, 0.15) * rng.choice([1, -1])
                current[j1] = max(0.0, current[j1] - amt)
                current[j2] = min(1000.0, current[j2] + amt)
        
        else:
            # Small global noise
            current *= np.exp(rng.normal(0, 0.01, size=n))
        
        current = np.clip(current, 0.0, 1000.0)
        
        new_score = score(current)
        if new_score < current_score:
            current_score = new_score
            if new_score < best_score_val:
                best_score_val = new_score
                best = current.copy()
        elif rng.random() < 0.05:
            # Accept worse with small probability (simulated annealing)
            current_score = new_score
        
        # Adaptive learning rate
        if step % 500 == 0:
            lr *= 0.95
        
        if step % 2000 == 0:
            print(f"      PS: step={step}, cur={current_score:.6f}, best={best_score_val:.6f}")
    
    return best, best_score_val

def coordinate_polish(a_init, time_limit, rng):
    """
    Coordinate descent: for each element, try a few multipliers.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = score(current)
    best = current.copy()
    best_score_val = current_score
    t0 = time.monotonic()
    
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]
    
    while time.monotonic() - t0 < time_limit - 5:
        order = rng.permutation(n)
        improved = False
        
        for j in order:
            if time.monotonic() - t0 > time_limit - 5:
                break
            if current[j] < 1e-8:
                continue
            
            save = current[j]
            for mu in multipliers:
                trial = save * mu
                if trial < 0 or trial > 1000:
                    continue
                current[j] = trial
                s = score(current)
                if s < current_score:
                    current_score = s
                    improved = True
                    if s < best_score_val:
                        best_score_val = s
                        best = current.copy()
                    break
                else:
                    current[j] = save
                    # If going up, stop (assuming unimodal in log space)
                    if mu > 1.0:
                        break
                    elif mu < 1.0 and save * mu > save:
                        break
        
        if not improved and rng.random() < 0.3:
            # Random kick
            j = int(rng.integers(0, n))
            current[j] = max(0.0, current[j] * np.exp(rng.normal(0, 0.3)))
            current = np.clip(current, 0.0, 1000.0)
            current_score = score(current)
    
    return best, best_score_val

"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score(a):
    """Fast score computation matching evaluate_sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def soft_max_grad(a, tau):
    """
    Compute soft-max approximation of max(convolution) and its gradient w.r.t. a.
    Returns (soft_max_value, gradient_wrt_a, convolution).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    value = c_max + tau * np.log(np.sum(exp_s))

    # Gradient: d(soft_max)/da[j] = 2 * sum_k w[k] * a[k-j] (if valid)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    grad_a = 2.0 * corr[n - 1:2 * n - 1]

    return value, grad_a, c, w


def exact_subgrad(a):
    """
    Exact subgradient of max(convolution) at the argmax.
    Returns (peak_value, subgradient, peak_index).
    """
    n = len(a)
    c = fft_convolve(a, a)
    peak_val = np.max(c)
    # Find all indices achieving the max (within tolerance)
    tol = 1e-10 * max(1.0, peak_val)
    peak_indices = np.where(np.abs(c - peak_val) < tol)[0]

    g = np.zeros(n)
    for k in peak_indices:
        for j in range(n):
            if 0 <= k - j < n:
                g[j] += a[k - j]
        # Handle the case where k-j == j (self-term counted once in convolution
        # but appears twice in derivative since c[k] = sum a[i]*a[k-i])
        # Actually dc[k]/da[j] = a[k-j] + a[j-k] if both valid, else one term
        # For k in [0, 2n-2]: dc[k]/da[j] = a[k-j]*(valid1) + a[k-j]*(valid2)
        # where valid1: 0<=j<n and 0<=k-j<n; valid2: same condition
        # So it's always 2*a[k-j] if 0<=k-j<n
    g *= 2.0 / len(peak_indices)

    return peak_val, g, int(peak_indices[0])


def chirp_init(n, rng):
    """Generate chirp-like initializations with good autocorrelation properties."""
    t = np.arange(n, dtype=np.float64)
    method = rng.integers(0, 4)

    if method == 0:
        # Quadratic chirp
        beta = rng.uniform(1.0, 4.0)
        A = rng.uniform(0.3, 0.7)
        a = 1.0 + A * np.cos(2 * np.pi * beta * t ** 2 / (2.0 * n))
    elif method == 1:
        # Multi-frequency
        a = np.ones(n)
        for k in range(3, 12):
            freq = rng.uniform(0.5, k * 1.5)
            phase = rng.uniform(0, 2 * np.pi)
            a += (0.25 / np.sqrt(k)) * np.cos(2 * np.pi * freq * t / n + phase)
    elif method == 2:
        # Log-chirp
        beta = rng.uniform(0.5, 3.0)
        A = rng.uniform(0.3, 0.7)
        a = 1.0 + A * np.cos(2 * np.pi * beta * np.log(t + 1) / np.log(n + 1) * n)
    else:
        # Random smooth
        a = 1.0 + 0.4 * rng.standard_normal(n)
        # Smooth it
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')

    return np.clip(a, 0.01, None)

        elif frac < 0.75:
            tau = S_target / n * (0.4 - (frac - 0.4) * 0.8)
        elif frac < 0.9:
            tau = S_target / n * 0.12 * (1.0 - (frac - 0.75) / 0.15)
        else:
            tau = S_target / n * 0.01

        v_buf = beta2 * v_buf + (1 - beta2) * grad_theta ** 2

def peak_shave_v2(a_init, time_limit, rng, verbose=True):
    """
    Improved peak-directed local search with multi-peak awareness
    and adaptive step sizes.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    S = np.sum(current)
    current_score = score(current)
    best = current.copy()
    best_score_val = current_score

    t0 = time.monotonic()
    step = 0
    alpha = 0.05  # base step size
    t_anim = 0.01  # annealing temperature

    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(current, current)
        
        # Multi-peak: find top peaks
        sorted_idx = np.argsort(c)[::-1]
        num_peaks = min(3, 2 * n - 1)

        r = rng.random()

        if r < 0.35:
            # Multi-peak reduction: reduce contributors to all top peaks
            contribs = np.zeros(n)
            for pk in sorted_idx[:num_peaks]:
                weight = c[pk] / np.sum(c[sorted_idx[:num_peaks]])
                for i in range(n):
                    if 0 <= pk - i < n:
                        contribs[i] += weight * current[i] * current[pk - i]

            top_k = max(4, n // 16)
            bot_k = max(8, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            bot_idx = np.argsort(contribs)[:bot_k]

            red_frac = alpha * rng.uniform(0.5, 1.5)
            red_amt = current[top_idx].sum() * red_frac
            add_frac = red_frac * rng.uniform(0.6, 1.2)
            add_amt = current[bot_idx].sum() * add_frac

            current[top_idx] -= red_amt / top_k
            current[bot_idx] += add_amt / bot_k

        elif r < 0.55:
            # Single-element targeted reduction
            pk = sorted_idx[0]
            contribs = np.zeros(n)
            for i in range(n):
                if 0 <= pk - i < n:
                    contribs[i] = current[i] * current[pk - i]

            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            j = top_idx[int(rng.integers(0, top_k))]
            
            # Find best redistribution target (min contribution to any peak)
            min_contrib = np.inf
            best_target = -1
            for k in range(n):
                max_c = 0.0
                for pk2 in sorted_idx[:num_peaks]:
                    if 0 <= pk2 - k < n:
                        max_c = max(max_c, current[k] * current[pk2 - k])
                if max_c < min_contrib:
                    min_contrib = max_c
                    best_target = k
            
            if best_target >= 0 and best_target != j:
                amt = current[j] * alpha * rng.uniform(0.3, 1.0)
                current[j] -= amt
                current[best_target] += amt

        elif r < 0.7:
            # Geometric perturbation of a random subset
            k_elems = max(3, n // 20)
            idx = rng.choice(n, size=k_elems, replace=False)
            factors = np.exp(rng.normal(0, alpha * 2.0, size=k_elems))
            current[idx] *= factors

        elif r < 0.85:
            # Pair swap: move mass from high-contrib to low-contrib
            pk = sorted_idx[0]
            contribs = np.zeros(n)
            for i in range(n):
                if 0 <= pk - i < n:
                    contribs[i] = current[i] * current[pk - i]
            
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]

            j_dst = int(rng.integers(0, n))
            if j_src != j_dst:
                amt = current[j_src] * alpha * rng.uniform(0.1, 0.5)
                current[j_src] -= amt
                current[j_dst] += amt

        else:
            # Small global perturbation
            current *= np.exp(rng.normal(0, alpha * 0.3, size=n))

        current = np.clip(current, 0.0, 1000.0)

        # Maintain sum (approximately)
        S_new = np.sum(current)
        if S_new > 0.01:
            current *= S / S_new

        new_score = score(current)

        # Acceptance: always accept improvement, sometimes accept worse (SA)

            current_score = new_score
            if new_score < best_score_val:
                best_score_val = new_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if rng.random() < np.exp(-delta / max(t_anim, 1e-15)):
                current_score = new_score

        t_anim *= 0.99999

        if verbose and step % 3000 == 0:
            print(f"        PS: step={step}, cur={current_score:.6f}, "
                  f"best={best_score_val:.6f}, alpha={alpha:.4f}")

        if step % 2000 == 0:
            alpha *= 0.9

    return best, best_score_val


def coordinate_polish(a_init, time_limit, rng, verbose=True):
    """
    Coordinate descent with geometric search in log-space.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = score(current)
    best = current.copy()
    best_score_val = current_score
    t0 = time.monotonic()
    round_num = 0

    while time.monotonic() - t0 < time_limit - 5:
        round_num += 1
        order = rng.permutation(n)
        improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit - 5:
                    break
            if current[j] < 1e-10:
                continue

            save = current[j]
            # Try a range of multipliers in log-space
            log_steps = np.linspace(-1.5, 1.5, 9)
            found_better = False

            for ls in log_steps:
                if ls == 0:
                    continue
                trial = save * np.exp(ls)
                if trial < 0 or trial > 1000:
                    continue
                current[j] = trial
                s = score(current)
                if s < current_score:
                    current_score = s
                    improved = True
                    found_better = True
                    if s < best_score_val:
                        best_score_val = s
                        best = current.copy()
                    # Continue searching in same direction
                else:
                    current[j] = save
                    break

            if not found_better:
                current[j] = save

        # Random kick if no improvement
        if not improved:
            k = int(rng.integers(0, n))
            current[k] = max(0.0, current[k] * np.exp(rng.normal(0, 0.4)))
            current = np.clip(current, 0.0, 1000.0)
            current_score = score(current)

        if verbose and round_num % 10 == 0:
            print(f"        CD: round={round_num}, cur={current_score:.6f}, "
                  f"best={best_score_val:.6f}, t={time.monotonic()-t0:.0f}s")

    return best, best_score_val

    # Second run with different initialization
    if time.monotonic() < deadline - 60:
        t_p1b = min(0.15 * time_budget, deadline - time.monotonic() - 40)
        if t_p1b > 20:
            print(f"\n  Phase 1b: Different init, t={t_p1b:.0f}s")
            a0b = chirp_init(n_primary, rng)
            # Mix with best
            a0b = 0.5 * a0b / np.sum(a0b) + 0.5 * best_seq / np.sum(best_seq)
            a1b, s1b = optimize_softmax(n_primary, a0b, t_p1b, rng, verbose=False)
            if s1b < best_score_val:
                best_score_val = s1b
                best_seq = a1b.copy()
                print(f"  Phase 1b result: {best_score_val:.6f} *** NEW BEST ***")
            else:
                print(f"  Phase 1b result: {s1b:.6f} (no improvement)")

    # === PHASE 2: Explore other n values ===
    print(f"\n{'='*60}")
    print(f"PHASE 2: Multi-n exploration")
    print(f"{'='*60}")

    n_tries = [n_primary - 16, n_primary - 32, n_primary + 16, n_primary + 32,
               n_primary + 48, 256, 320]
    n_tries = [n for n in n_tries if n >= 128 and n <= 640 and n != n_primary]
    # Remove duplicates
    seen = set()
    n_tries_unique = []
    for n_try in n_tries:
        if n_try not in seen:
            seen.add(n_try)
            n_tries_unique.append(n_try)
    n_tries = n_tries_unique[:5]

    for n_try in n_tries:
        if time.monotonic() > deadline - 40:
            break
        t_try = min(0.06 * time_budget, deadline - time.monotonic() - 30)
        if t_try < 10:
            break

        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a0_try = np.interp(x_new, x_old, best_seq)
        else:
            a0_try = chirp_init(n_try, rng)

        a_try, s_try = optimize_softmax(n_try, a0_try, t_try, rng, verbose=False)
        print(f"  n={n_try}: score={s_try:.6f}")

        if s_try < best_score_val:
            best_score_val = s_try
            best_seq = a_try.copy()
            print(f"  *** NEW BEST at n={n_try}: {best_score_val:.6f} ***")

    # === PHASE 3: Peak-shave local search ===
    print(f"\n{'='*60}")
    print(f"PHASE 3: Peak-shave local search")
    print(f"{'='*60}")

    if best_seq is not None:
        t_p3 = min(0.20 * time_budget, deadline - time.monotonic() - 30)
        t_p3 = max(t_p3, 20.0)
        t_p3 = min(t_p3, deadline - time.monotonic() - 20)
        print(f"  Time: {t_p3:.0f}s, starting: {best_score_val:.6f}")
        a_ps, s_ps = peak_shave_v2(best_seq, t_p3, rng, verbose=True)
        if s_ps < best_score_val:
            best_score_val = s_ps
            best_seq = a_ps.copy()
            print(f"  After peak-shave: {best_score_val:.6f} *** NEW BEST ***")
        else:
            print(f"  After peak-shave: {s_ps:.6f} (best={best_score_val:.6f})")

    # === PHASE 4: Coordinate descent ===
    print(f"\n{'='*60}")
    print(f"PHASE 4: Coordinate descent polish")
    print(f"{'='*60}")

    if best_seq is not None:
        t_p4 = min(0.12 * time_budget, deadline - time.monotonic() - 20)
        t_p4 = max(t_p4, 15.0)
        t_p4 = min(t_p4, deadline - time.monotonic() - 10)
        if t_p4 > 10:
            print(f"  Time: {t_p4:.0f}s, starting: {best_score_val:.6f}")
            a_cd, s_cd = coordinate_polish(best_seq, t_p4, rng, verbose=True)
            if s_cd < best_score_val:
                best_score_val = s_cd
                best_seq = a_cd.copy()
                print(f"  After CD: {best_score_val:.6f} *** NEW BEST ***")
            else:
                print(f"  After CD: {s_cd:.6f} (best={best_score_val:.6f})")

    # === PHASE 5: Final peak-shave ===
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 15:
            print(f"\n{'='*60}")
            print(f"PHASE 5: Final polish (t={remaining:.0f}s)")
            print(f"{'='*60}")
            a_fin, s_fin = peak_shave_v2(best_seq, remaining, rng, verbose=True)
            if s_fin < best_score_val:
                best_score_val = s_fin
                best_seq = a_fin.copy()
                print(f"  Final: {best_score_val:.6f} *** NEW BEST ***")

    # Final verification and return
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"=== FINAL RESULT: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'='*60}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
