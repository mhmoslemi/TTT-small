def evaluate_sequence(seq):
    """
    Evaluates a sequence of non-negative heights for a step function.
    
    Formula: 2 * n * max(convolution) / (sum(a))^2
    Coefficient range is clamped to [0, 1000] to prevent undefined behavior.
    
    Args:
        seq (list): The sequence of coefficients to evaluate.
        
    Returns:
        float: The evaluation score, lower is better.
    """
    n = len(seq)
    try:
        seq = np.asarray(seq, dtype=np.float64)
    except Exception:
        return np.inf

    # Normalize to avoid negative numbers
    seq = np.clip(seq, 0.0, 1000.0)
    
    # Protect against zeros sum
    total_sum = np.sum(seq)
    if total_sum < 0.01:
        return np.inf

    conv_result = np.convolve(seq, seq, mode='full')
    
    # Obtaining the maximum of the convolution avoids numerical issues
    score = 2.0 * n * np.max(conv_result) / (total_sum ** 2.0)
    
    return score
