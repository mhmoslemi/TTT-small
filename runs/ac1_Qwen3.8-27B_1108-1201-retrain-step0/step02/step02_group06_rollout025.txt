def fft_convolve(a, b):

    m = len(b)
    size = 1
    while size < n + m - 1:

    fb = np.fft.rfft(b, size)

LP-based peak shaving for minimizing peak autocorrelation of non-negative step functions.
Uses linear programming to find descent directions that reduce the peak of the
self-convolution, combined with targeted coordinate descent for final polish.

import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score computation: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def build_conv_matrix(a, n):
    """
    Build matrix A (shape (2n-1, n)) such that A @ b = convolve(a, b).
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        A[k_start:k_end + 1, j] = a[0:k_end - k_start + 1]
    return A

    """
    Spectral initialization: alternate between time-domain non-negativity
    and frequency-domain flat power spectrum.
    """
    a = 1.0 + 0.3 * rng.standard_normal(n)

    a = a / np.sum(a)

        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        
        # Target: flat power spectrum (but non-zero at DC)

        s = np.sum(a)
        if s < 1e-10:

"""
LP-based peak shaving for minimizing peak autocorrelation of non-negative step functions.
Uses linear programming to find descent directions that reduce the peak of the self-convolution,
combined with targeted coordinate descent for final polish. Explores multiple sequence lengths
and initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score computation: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def build_conv_matrix(a, n):
    """Build matrix A (shape (2n-1, n)) such that A @ b = convolve(a, b)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        A[k_start:k_end + 1, j] = a[0:k_end - k_start + 1]
    return A


def lp_peak_shave(a, n, time_limit, rng, verbose=True):
    """
    LP-based iterative peak shaving.
    a: normalized sequence (sum=1), length n.
    Returns improved sequence.
    """
    a = a.copy()
    a = a / np.sum(a)
    t0 = time.monotonic()
    
    # Momentum buffer
    b_momentum = np.zeros(n)
    momentum_alpha = 0.3
    momentum_count = 0
    max_momentum = 10
    
    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)
        
        # Build constraint matrix and solve LP
        A = build_conv_matrix(a, n)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not res.success:
            # Try with slightly relaxed constraint
            res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M * 1.01), bounds=bounds, method='highs')
            if not res.success:
                break
        
        b = res.x.copy()
        b_sum = np.sum(b)
        if b_sum < 1e-10:
            break
        b = b / b_sum
        
        # Check if b is substantially different from a
        if np.max(np.abs(b - a)) < 1e-10:
            # Add small perturbation to escape
            b += 0.01 * rng.standard_normal(n)
            b = np.clip(b, 0, None)
            b_sum = np.sum(b)
            if b_sum < 1e-10:
                break
            b = b / b_sum
        
        # Update momentum
        b_momentum = momentum_alpha * b + (1 - momentum_alpha) * b_momentum
        momentum_count += 1
        b_use = b_momentum / np.sum(b_momentum) if momentum_count > 2 else b
        
        # Line search for best mixing parameter
        best_alpha = 0.0
        best_peak = M
        alphas = [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
        for alpha in alphas:
            candidate = (1 - alpha) * a + alpha * b_use
            candidate = np.clip(candidate, 0, None)
            s = np.sum(candidate)
            if s < 1e-10:
                continue
            candidate = candidate / s
            peak = np.max(fft_convolve(candidate, candidate))
            if peak < best_peak - 1e-14:
                best_peak = peak
                best_alpha = alpha
        
        if best_alpha == 0:
            # Try with raw b instead of momentum
            for alpha in alphas:
                candidate = (1 - alpha) * a + alpha * b
                candidate = np.clip(candidate, 0, None)
                s = np.sum(candidate)
                if s < 1e-10:
                    continue
                candidate = candidate / s
                peak = np.max(fft_convolve(candidate, candidate))
                if peak < best_peak - 1e-14:
                    best_peak = peak
                    best_alpha = alpha
            if best_alpha == 0:
                break
        
        a = (1 - best_alpha) * a + best_alpha * b_use
        a = np.clip(a, 0, None)
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s
        
        if verbose and it % 50 == 0:
            current_peak = np.max(fft_convolve(a, a))
            print(f"    LP iter {it}: peak={current_peak:.10f}, score={2*n*current_peak:.6f}")
    
    return a

def coordinate_polish(a, n, time_limit, rng):
    """
    Targeted coordinate descent polish focusing on peak reduction.
    """
    a = a.copy()
    a = a / np.sum(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    no_improve = 0
    max_no_improve = 300
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        conv = fft_convolve(a, a)
        peak_val = np.max(conv)
        peak_indices = np.where(conv > peak_val * 0.99)[0]
        peak_idx = int(np.argmax(conv))
        
        r = rng.random()
        
        if r < 0.4:
            # Strategy 1: Reduce top contributors to all near-peak locations
            candidate = a.copy()
            total_reduced = 0.0
            for pk in peak_indices[:5]:
                contribs = np.zeros(n)
                for i in range(max(0, int(pk) - n + 1), min(n, int(pk) + 1)):
                    j = int(pk) - i
                    if 0 <= j < n:
                        contribs[i] = a[i] * a[j]
                top_k = max(2, n // 32)
                top_idx = np.argsort(contribs)[-top_k:]
                for i in top_idx:
                    if candidate[i] > 1e-10:
                        amt = candidate[i] * 0.05
                        candidate[i] -= amt
                        total_reduced += amt
            # Redistribute
            if total_reduced > 1e-15:
                # Add to elements in valleys
                valley_idx = np.argsort(conv)[:max(5, n // 16)]
                n_add = min(5, len(valley_idx))
                add_idx = rng.choice(valley_idx, size=n_add, replace=False)
                for j in add_idx:
                    candidate[j] += total_reduced / n_add
            candidate = np.clip(candidate, 0, None)
            s = np.sum(candidate)
            if s < 1e-10:
                continue
            candidate = candidate / s
            
        elif r < 0.7:
            # Strategy 2: Pair transfer from high to low
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            # Prefer moving mass from high-value to low-value elements
            if a[i] < a[j]:
                i, j = j, i
            frac = rng.uniform(0.02, 0.1)
            amt = a[i] * frac
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
            candidate = np.clip(candidate, 0, None)
            s = np.sum(candidate)
            if s < 1e-10:
                continue
            candidate = candidate / s
        else:
            # Strategy 3: Block modification
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            factor = np.exp(rng.normal(0, 0.03, size=width))
            candidate[start:start + width] *= factor
            candidate = np.clip(candidate, 0, None)
            s = np.sum(candidate)
            if s < 1e-10:
                continue
            candidate = candidate / s
        
        new_peak = np.max(fft_convolve(candidate, candidate))
        if new_peak < best_peak - 1e-14:
            a = candidate
            best_peak = new_peak
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_a


def spectral_init(n, rng, n_iter=100):
    """Spectral initialization for flat autocorrelation."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    
    for it in range(n_iter):
        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        # Target flat power spectrum
        target = np.sqrt(np.ones_like(power))
        target[0] = np.sqrt(power[0])
        # Mix current and target in frequency domain
        alpha = 0.5
        A_new = (1 - alpha) * A + alpha * target * np.exp(1j * np.angle(A))
        a = np.fft.irfft(A_new, 2 * n)[:n]
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    
    return a


def ramp_init(n, style, rng=None):
    """Create various initializations."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if style == 'linear':
        a = 1.0 + 0.5 * t
    elif style == 'quadratic':
        a = 1.0 + 0.5 * t ** 2
    elif style == 'cosine':
        a = 1.0 + 0.3 * np.cos(np.pi * t)
    elif style == 'exp':
        a = np.exp(0.3 * t)
    elif style == 'two_block':
        a = np.ones(n)
        a[n // 3:] *= 1.2
    elif style == 'three_block':
        a = np.ones(n)
        a[n // 2:] *= 1.15
        a[:n // 4] *= 0.9
    elif style == 'bell':
        a = np.exp(-2.0 * (t - 0.5) ** 2)
    else:
        a = 1.0 + 0.3 * t
    if rng is not None:
        a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
