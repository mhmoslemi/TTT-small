import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def true_objective(a):
    c = fft_convolve_self(a)
    return c.max()

def subgradient(a, active_lags):
    """Compute subgradient of max(c) at point a.
    active_lags: indices of lags achieving (or near) the max.
    """
    n = len(a)
    c = fft_convolve_self(a)
    w = np.zeros(len(c))
    w[active_lags] = 1.0 / len(active_lags)
    # grad[m] = 2 * sum_k w[k] * a[k-m] for 0<=k-m<n
    # This is 2 * (w * rev(a))[m + n - 1] where * is linear convolution
    a_rev = a[::-1]
    full = np.convolve(w, a_rev)  # length (2n-1) + n - 1 = 3n-2
    grad = 2.0 * full[n-1:2*n-1]
    return grad

def project_simplex(v, S):
    """Project v onto {a >= 0, sum(a) = S}."""
    n = len(v)
    # Solve: a = max(0, v - lambda), sum(a) = S
    # Bisection on lambda
    lo = v.min() - S  # if lambda < this, all a > 0
    hi = v.max()  # if lambda > this, all a = 0
    # Actually: a_i = max(0, v_i - lambda), sum = S
    # f(lambda) = sum(max(0, v_i - lambda)) is decreasing in lambda
    # f(lo) >= S (if lo is small enough), f(hi) <= S
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(0, v - mid))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.maximum(0, v - (lo + hi) / 2)

def active_set_subgradient(a, S, time_budget, n_active=20, lr_init=0.1):
    """Subgradient method with projection."""
    a = a.copy().astype(np.float64)
    a = a * (S / a.sum())
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    lr = lr_init
    for step in range(1000000):
        if time.monotonic() - start > time_budget:
            break
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()
        
        # Find active lags
        threshold = cur * (1 - 0.01)  # lags within 1% of max
        active = np.where(c >= threshold)[0]
        
        # Compute subgradient
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        a_rev = a[::-1]
        full = np.convolve(w, a_rev)
        grad = 2.0 * full[n-1:2*n-1]
        
        # Project gradient onto the tangent space of {sum = S}
        # (subtract the mean component)
        grad_proj = grad - np.mean(grad)
        
        # Line search
        for alpha in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05]:
            v = a - alpha * grad_proj
            a_new = project_simplex(v, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur:
                a = a_new
                break
        else:
            lr *= 0.5
            if lr < 1e-10:
                lr = lr_init * 0.1
                # Random perturbation
                a = a + 0.01 * S * np.random.randn(n)
                a = project_simplex(a, S)
        
        # Decay learning rate
        lr = max(lr_init * 0.9999**step, 1e-6 * S / n)
    
    return best_a, best_obj

    c_max = c.max()
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
    else:
        top_idx = np.arange(len(c))
        K = len(c)
    top_vals = c[top_idx]
    shifted = T * (top_vals - top_vals.max())
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w_top = np.ones(K) / K
    else:
        w_top = exp_shifted / total
    w = np.zeros(len(c))
    w[top_idx] = w_top
    grad = compute_grad_from_weights(w, a)
    obj = top_vals.max() + np.log(max(total, 1e-300)) / T
    return obj, grad, c


def true_max(a):
    """True max of self-convolution."""
    return fft_convolve_self(a).max()


def x_to_a(x, S):
    """Convert squared parameters to sequence."""
    x2 = x * x
    s = x2.sum()
    if s < 1e-30:
        return np.full(len(x), S / len(x))
    return S * x2 / s


def a_to_x(a, S):
    """Convert sequence to squared parameters."""
    a = np.maximum(a, 1e-15)
    a = a * (S / a.sum())
    return np.sqrt(a)


def objective_func(x, T, K, S, state):
    """Objective and gradient for L-BFGS-B with squared param.
    All args passed explicitly to avoid closures."""
    if time.monotonic() - state['start'] > state['budget']:
        return state['best_obj'], np.zeros_like(x)

    n = len(x)
    a = x_to_a(x, S)
    obj_a, grad_a, c = topk_softmax_and_grad(a, T, K)

    tm = c.max()
    if tm < state['best_obj']:
        state['best_obj'] = tm
        state['best_a'] = a.copy()

    # Chain rule for squared parameterization:
    # a_i = S * x_i^2 / s where s = sum(x^2)
    # da_i/dx_j = 2*S*x*(delta_ij*s - x_i*x_j) / s^2
    # grad_x_j = 2*S*x_j/s^2 * (s*grad_a_j - x_j*dot(grad_a, x^2))
    x2 = x * x
    s = x2.sum()
    if s < 1e-30:
        return obj_a, np.zeros_like(x)
    dot = np.dot(grad_a, x2)
    grad_x = 2.0 * S * x / (s * s) * (s * grad_a - x * dot)

    # Numerical check
    if not np.all(np.isfinite(grad_x)):
        return obj_a, np.zeros_like(x)

    return obj_a, grad_x


def run_lbfgs(x_init, S, T, K, budget, max_iter=2000):
    """Run L-BFGS-B with squared parameterization."""
    state = {'best_obj': np.inf, 'best_a': None,
             'start': time.monotonic(), 'budget': budget}
    a0 = x_to_a(x_init, S)
    state['best_obj'] = true_max(a0)
    state['best_a'] = a0.copy()

    try:
        minimize(objective_func, x_init.copy(),
                 args=(T, K, S, state),
                 jac=True, method='L-BFGS-B',
                 options={'maxiter': max_iter, 'maxfun': 5000,
                          'gtol': 1e-12, 'maxls': 50})
    except Exception as e:
        print("  L-BFGS error: %s" % str(e))

    return state['best_a'], state['best_obj']


def coord_descent(a, S, budget, max_sweeps=200):
    """Coordinate descent to polish the sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = true_max(a)
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > budget:
            break
        improved = False
        c = fft_convolve_self(a)
        c_max_val = c.max()

        for i in range(n):
            if time.monotonic() - start > budget:
                break
            orig = a[i]
            if orig < 1e-14:
                continue

            best_val = orig
            best_local = c_max_val
            # Try several fractions
            for frac in [0.0, 0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7]:
                a[i] = orig * frac
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_n = a * (S / s)
                obj_test = true_max(a_n)
                if obj_test < best_local:
                    best_local = obj_test
                    best_val = orig * frac
                a[i] = a_n[i] if s >= 1e-10 else orig

            if best_val != orig:
                a[i] = best_val
                s = a.sum()
                if s > 1e-10:
                    a = a * (S / s)
                improved = True
                c = fft_convolve_self(a)
                if c.max() < best_obj:
                    best_obj = c.max()
                    best_a = a.copy()

        if not improved:
            break

    return best_a, best_obj


def perturb_and_renorm(a, S, rng, frac=0.1):
    """Randomly reduce a fraction of entries, then renormalize."""
    n = len(a)
    a_new = a.copy()
    n_perturb = max(1, int(n * frac))
    idx = rng.choice(n, size=n_perturb, replace=False)
    for i in idx:
        a_new[i] *= rng.uniform(0.0, 0.5)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new * (S / s)


def get_initialization(name, n, rng, S):
    """Generate various initializations."""
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "gaussian_center":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n / 2) / (n / 8)) ** 2)
        return a * (S / a.sum())
    elif name == "chirp":
        i = np.arange(n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * i * 3 / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "anti_chirp":
        i = np.arange(n)
        a = 1.0 - 0.5 * np.cos(2 * np.pi * i * 3 / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "bimodal":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n * 0.25) / (n / 12)) ** 2) + \
            np.exp(-0.5 * ((i - n * 0.75) / (n / 12)) ** 2)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    global_best = np.inf
    global_a = None

    # Load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 8 and given_raw.sum() > 0.01:
            n_given = len(given_a) if given_a is not None else len(given_raw)
            S_given = np.sqrt(2.0 * float(n_given))
            given_a = given_raw * (S_given / given_raw.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.6f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: %s" % str(e))

    # Determine which lengths to try
    lengths_to_try = []
    if given_a is not None:
        lengths_to_try.append(len(given_a))
    for n_try in [4096, 8192, 2048]:
        if n_try not in lengths_to_try:
            lengths_to_try.append(n_try)

    # Phase 1: Quick exploration (25% of time)
    explore_time = 0.25 * (deadline - time.monotonic())
    print("Phase 1: Exploration (%.0fs)" % explore_time)

    for n in lengths_to_try[:3]:
        if time.monotonic() > deadline - 60:
            break
        S = np.sqrt(2.0 * float(n))
        time_per_init = min(4.0, explore_time / (3 * len(lengths_to_try)))
        if time_per_init < 1.0:
            break

        inits = ["uniform", "gaussian_center", "chirp", "bimodal"]
        for init_name in inits[:3]:
            if time.monotonic() > deadline - 60:
                break
            a_init = get_initialization(init_name, n, rng, S)
            x_init = a_to_x(a_init, S)
            t0 = time.monotonic()
            a_best, obj_best = run_lbfgs(x_init, S, T=50, K=10,
                                         budget=time_per_init, max_iter=200)
            elapsed = time.monotonic() - t0
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print("  Explore n=%d %s: obj=%.6f (%.1fs)" %
                      (n, init_name, obj_best, elapsed))
            # Continue from this point with higher K
            if obj_best < global_best + 0.01 and elapsed < time_per_init:
                x_cur = a_to_x(a_best, S)
                a_best2, obj_best2 = run_lbfgs(x_cur, S, T=100, K=30,
                                               budget=time_per_init, max_iter=200)
                if obj_best2 < global_best:
                    global_best = obj_best2
                    global_a = a_best2.copy()
                    print("  Explore2 n=%d %s: obj=%.6f" % (n, init_name, obj_best2))

    # Phase 2: Focused refinement (55% of time)
    exploit_time = 0.55 * (deadline - time.monotonic())
    print("Phase 2: Refinement (%.0fs)" % exploit_time)

    if global_a is None:
        n = 4096
        S = np.sqrt(2.0 * n)
        global_a = np.full(n, S / n)
        global_best = 2.0

    n = len(global_a)
    S = np.sqrt(2.0 * float(n))
    a = global_a.copy()
    a = a * (S / a.sum())
    x = a_to_x(a, S)

    # Progressive refinement with increasing K and T
    schedules = [
        # (T, K, fraction_of_time)
        (30, 5, 0.10),
        (50, 10, 0.10),
        (80, 15, 0.10),
        (120, 20, 0.10),
        (200, 30, 0.10),
        (300, 40, 0.10),
        (500, 50, 0.10),
        (800, 60, 0.10),
        (1200, 80, 0.10),
        (2000, 100, 0.10),
    ]

    stall_count = 0
    for (T, K, frac) in schedules:
        if time.monotonic() > deadline - 30:
            break
        t_budget = frac * exploit_time
        if t_budget < 2.0:
            t_budget = 2.0

        obj_before = global_best
        x = a_to_x(a, S)
        a, obj = run_lbfgs(x, S, T, K, t_budget, max_iter=2000)

        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            a = a * (S / a.sum())
            stall_count = 0
            print("  T=%d K=%d: obj=%.6f" % (T, K, obj))
        else:
            stall_count += 1
            if stall_count >= 3 and time.monotonic() < deadline - 60:
                # Perturb and try again
                a = perturb_and_renorm(a, S, rng, frac=0.15)
                x = a_to_x(a, S)
                a2, obj2 = run_lbfgs(x, S, T, K, min(t_budget, 15.0), max_iter=500)
                if obj2 < global_best:
                    global_best = obj2
                    global_a = a2.copy()
                    a = a2 * (S / a2.sum())
                    stall_count = 0
                    print("  Perturb+T=%d K=%d: obj=%.6f" % (T, K, obj2))

    # Phase 3: Random restarts (10% of time)
    restart_time = 0.10 * (deadline - time.monotonic())
    print("Phase 3: Restarts (%.0fs)" % max(0, restart_time))

    if global_a is not None and restart_time > 5.0:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        best_starter = global_a.copy()
        best_starter = best_starter * (S / best_starter.sum())

        n_restarts = max(2, int(restart_time / 15.0))
        for ri in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            a_try = perturb_and_renorm(best_starter, S, rng,
                                       frac=rng.uniform(0.05, 0.25))
            x_try = a_to_x(a_try, S)
            t_b = min(15.0, restart_time / (n_restarts - ri))
            a_r, obj_r = run_lbfgs(x_try, S, T=200, K=40,
                                   budget=t_b, max_iter=500)
            if obj_r < global_best:
                global_best = obj_r
                global_a = a_r.copy()
                best_starter = a_r.copy()
                best_starter = best_starter * (S / best_starter.sum())
                print("  Restart %d: obj=%.6f" % (ri, obj_r))

    # Phase 4: Coordinate descent polish (10% of time)
    polish_time = max(5.0, deadline - time.monotonic() - 10.0)
    print("Phase 4: Polish (%.0fs)" % polish_time)

    if global_a is not None and polish_time > 3.0:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a = global_a.copy()
        a = a * (S / a.sum())

        a_polished, obj_polished = coord_descent(a, S, polish_time)
        if obj_polished < global_best:
            global_best = obj_polished
            global_a = a_polished.copy()
            print("  Coord descent: obj=%.6f" % obj_polished)

        # Final L-BFGS micro-step
        remaining = deadline - time.monotonic() - 8.0
        if remaining > 5.0 and global_a is not None:
            a = global_a.copy()
            a = a * (S / a.sum())
            x = a_to_x(a, S)
            a_f, obj_f = run_lbfgs(x, S, T=500, K=50, budget=remaining, max_iter=500)
            if obj_f < global_best:
                global_best = obj_f
                global_a = a_f.copy()
                print("  Final L-BFGS: obj=%.6f" % obj_f)

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_list = global_a.tolist()
    ver = evaluate_sequence(final_list)
    print("FINAL: best=%.6f, verified=%.6f, n=%d" %
          (global_best, ver, len(final_list)))

    if not np.isfinite(ver):
        final_list = [1.0] * 256
        print("Fallback used")

    return final_list
