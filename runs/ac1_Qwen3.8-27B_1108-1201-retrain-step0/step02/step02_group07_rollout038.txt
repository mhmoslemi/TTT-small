import time

def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_value(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()

def grad_of_max(conv, a, temperature=1.0):
    """
    Smoothed gradient of max(conv) using log-sum-exp.
    Returns gradient of (1/T) * log(sum(exp(T*conv))) approximated.
    For T→∞, this approaches the subgradient of max.
    """
    n = len(a)
    # Softmax weights
    scaled = conv * temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # Gradient: sum_k w_k * grad c_k
    # grad c_k[j] = 2 * a[k-j] if 0 <= k-j < n
    grad = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-10:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # For j in [js, je]: k-j is in [0, n-1], so grad[k][j] = 2*a[k-j]

    return grad

def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_grad(conv_weights, a):
    """
    Compute gradient of sum_k w_k * c_k where c_k = conv(a,a)_k.
    grad[j] = 2 * sum_m w[j+m] * a[m]
    Using FFT: IFFT(FFT(w, N) * conj(FFT(a, N)))[:n]
    """
    n = len(a)
    w_len = len(conv_weights)
    N = 1 << (n + w_len - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(conv_weights, N)
    corr = np.fft.irfft(A * np.conj(W), N)
    return 2.0 * corr[:n]

def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute softmax weights on convolution values."""
    scaled = conv * temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-30:

    best_a = a.copy()

    v = np.zeros(n)

    total_time = time_limit

        grad = compute_grad(w, a)
        
        # Adam update

        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

    step_count = 0
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))

        # Temperature schedule: start moderate, increase over time, with periodic resets
        if elapsed - last_temp_restart > restart_every:
            last_temp_restart = elapsed
            patience = 0
            # Reset with a fresh temperature
            temp_base = temp_init * (1 + 2 * frac)
            patience = 0
        else:
            temp_base = temp_init * (temp_max / temp_init) ** (patience / max_patience)

        temperature = min(temp_base, temp_max)

        # Compute softmax weights and gradient
        w = softmax_weights(conv, temperature)
        grad = compute_grad(w, a)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Adaptive learning rate (decreases with time)
        eff_lr = lr_init * (1.0 + 0.5 * frac) / (1.0 + frac * 3.0)
        a_new = a - eff_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)

        conv = fast_conv(a)
        cur_obj = 2.0 * n * conv.max()
        step_count += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # If stuck for too long, perturb
        if patience >= max_patience:
            patience = 0
            # Perturb: add small noise and re-project
            noise_scale = 0.05 * (1 - frac * 0.5)
            a = a + noise_scale * (np.random.randn(n) * a + 0.01 * np.ones(n))
            a = project_simplex(a)
            conv = fast_conv(a)

        # Speed control for large n
        if n > 2000 and step_count % 5 == 0:
            pass  # Just let it run; FFT is fast enough

    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_targets=3):
    """
    Coordinate descent: iteratively move mass from high-contribution
    positions to low-contribution positions to reduce the peak.
    """
    n = len(a0)
    a = normalize(a0.copy())
    conv = fast_conv(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0
    max_no_improve = 5000

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        # Find peak positions
        max_c = conv.max()
        if max_c < 1e-15:
            break
        # Get top peaks
        threshold = max_c * 0.85
        peaks = np.where(conv >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(conv)])

        # Compute contribution of each position to peaks
        contrib = np.zeros(n)
        for p in peaks[:min(10, len(peaks))]:
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            contrib[js:je + 1] += a[p - je:p - js + 1][::-1]

        # Choose source: high contribution, significant mass
        source_candidates = np.where((contrib > contrib.mean()) & (a > 1e-6))[0]
        if len(source_candidates) == 0:
            source_candidates = np.argsort(contrib)[::-1][:n // 4]
        j = source_candidates[rng.integers(len(source_candidates))]

        # Choose target: low contribution
        target_candidates = np.where(contrib < contrib.mean() * 0.5)[0]
        if len(target_candidates) == 0:
            target_candidates = np.argsort(contrib)[:n // 4]
        k = target_candidates[rng.integers(len(target_candidates))]

        if j == k:
            no_improve += 1
            continue

        # Try a move
        for frac_move in [0.02, 0.05, 0.01]:
            delta = a[j] * frac_move
            if delta < 1e-12:
                break
            a[j] -= delta
            a[k] += delta
            if a[j] < 0:
                a[j] += delta
                a[k] -= delta
                break
            new_conv = fast_conv(a)
            new_obj = 2.0 * n * new_conv.max()
            if new_obj < cur_obj - 1e-12:
                cur_obj = new_obj
                conv = new_conv
                no_improve = 0
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
                break
            else:
                a[j] += delta
                a[k] -= delta

        no_improve += 1

    return best_a, best_obj


def informed_anneal(a0, time_limit, seed=0):
    """
    Simulated annealing with peak-targeting moves.
    Biases moves toward reducing the current peak.
    """
    n = len(a0)
    a = normalize(a0.copy())
    conv = fast_conv(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    rng = np.random.default_rng(seed)

    t0 = time.monotonic()
    iterations = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        T = 0.02 * (1.0 - frac) ** 2 + 0.0003

        # Find peaks
        max_c = conv.max()
        if max_c < 1e-15:
            break
        threshold = max_c * 0.8
        peaks = np.where(conv >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(conv)])

        # Pick a peak
        p = peaks[rng.integers(len(peaks))]
        js = max(0, p - n + 1)
        je = min(n - 1, p)
        seg_a = a[js:je + 1]
        seg_b = a[p - je:p - js + 1][::-1]
        products = seg_a * seg_b
        total = products.sum()
        if total < 1e-15:
            continue
        probs = products / total

        # Source: weighted by contribution to this peak
        j = js + rng.integers(je - js + 1) if rng.random() > 0.5 else \
            js + int(rng.choice(je - js + 1, p=probs))

        # Target: prefer positions with low contribution to all peaks
        target_contrib = np.zeros(n)
        for pk in peaks[:5]:
            js2 = max(0, pk - n + 1)
            je2 = min(n - 1, pk)
            target_contrib[js2:je2 + 1] += a[pk - je2:pk - js2 + 1][::-1]

        # Sample from low-contribution positions
        low_mask = target_contrib < np.median(target_contrib)
        if low_mask.sum() > 0:
            candidates = np.where(low_mask)[0]
        else:
            candidates = np.argsort(target_contrib)[:n // 3]
        k = candidates[rng.integers(len(candidates))]

        if j == k:
            continue

        # Move amount
        delta = a[j] * rng.uniform(0.01, 0.05)
        if delta < 1e-12:
            continue

        a[j] -= delta
        a[k] += delta
        if a[j] < 0:
            a[j] += delta
            a[k] -= delta
            continue

        new_conv = fast_conv(a)
        new_obj = 2.0 * n * new_conv.max()
        d_obj = new_obj - cur_obj

        if d_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            accept_prob = np.exp(-d_obj / max(T, 1e-12))
            if rng.random() < accept_prob:
                cur_obj = new_obj
                conv = new_conv
            else:
                a[j] += delta
                a[k] -= delta

        iterations += 1

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # U-shape (high at edges, low in middle) - good for flat convolution
    u = np.minimum(x, n - 1 - x).astype(float)
    u = u ** 0.5 + 0.1
    inits['u_shape'] = u / u.sum()

    # Inverted Gaussian
    for sigma in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - n / 2) / (sigma * n)) ** 2)
        g_inv = 1.0 / (g + 0.1)
        inits['inv_gauss_' + str(sigma)] = g_inv / g_inv.sum()

    # Cosine window (raised cosine)
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    inits['cosine'] = cos_w / cos_w.sum()

    # Inverse cosine (anti-window)
    inv_cos = 1.0 / (cos_w + 0.01)
    inits['inv_cosine'] = inv_cos / inv_cos.sum()

    # Sine patterns
    for freq in [1, 2, 3, 5]:
        s = 0.5 * (1 + np.cos(2 * np.pi * freq * x / n))
        inits['sin_' + str(freq)] = s / s.sum()

    # Random
    for alpha in [0.1, 0.5, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['rand_' + str(alpha)] = r

    # Two-peak (bimodal at 1/4 and 3/4)
    two = np.exp(-0.5 * ((x - n * 0.25) / (n * 0.1)) ** 2) + \
          np.exp(-0.5 * ((x - n * 0.75) / (n * 0.1)) ** 2)
    inits['bimodal'] = two / two.sum()

    # Parent (if available)
    if parent is not None:
        p = np.maximum(parent, 0.0)
        if p.size >= n:
            p_use = p[:n]
        else:
            p_use = np.interp(np.linspace(0, 1, n),
                              np.linspace(0, 1, p.size), p)
        p_use = np.maximum(p_use, 0.0)
        if p_use.sum() > 1e-10:
            inits['parent'] = p_use / p_use.sum()

    return inits


def get_parent():
    """Try to load parent sequence from global variable."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def safe_eval(a):
    """Safely evaluate a sequence using the global evaluate_sequence."""
    try:
        a_list = [float(max(0.0, x)) for x in a]
        if not a_list or sum(a_list) < 0.01:
            return float('inf')
        score = evaluate_sequence(a_list)
        return float(score) if np.isfinite(score) else float('inf')
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Quick scan over lengths and initializations.
    Phase 2: Deep Adam optimization on top candidates.
    Phase 3: Coordinate descent + annealing for fine-tuning.
    Phase 4: Final polish.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    deadline = t_global + max(1.0, float(budget_s) - 20.0)

    parent = get_parent()

    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_eval(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  [t=%5.0fs] New best: score=%.6f, n=%d" %
                  (time.monotonic() - t_global, score, len(a)))

    # Phase 1: Quick scan
    print("Phase 1: Quick scan over lengths and initializations...")
    print("  Parent available:", parent is not None)
    if parent is not None:
        record(parent)

    scan_lengths = [128, 256, 512, 1024, 2048]
    phase1_budget = min(500.0, (deadline - time.monotonic()) * 0.25)

    scan_results = []
    for n in scan_lengths:
        if time.monotonic() > deadline - 1500:
            break
        inits = generate_inits(n, rng, parent)
        n_inits = len(inits)
        time_per_init = max(3.0, phase1_budget / max(1, len(scan_lengths) * n_inits) * 0.6)

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 1500:
                break
            t_alloc = min(time_per_init, deadline - time.monotonic() - 1500)
            if t_alloc < 2.0:
                break
            a_opt, obj_val = adam_optimize(
                a0, time_limit=t_alloc,
                temp_init=15.0, temp_max=150.0,
                lr_init=0.03, restart_every=10,
                name="scan_n%d_%s" % (n, name)
            )
            score = safe_eval(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt))
            record(a_opt)

    print("  Phase 1 complete. Best: %.6f" % best_score)
    scan_results.sort(key=lambda x: x[0])

    # Phase 2: Deep optimization
    print("Phase 2: Deep optimization on top candidates...")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 10:
            top_configs.append((n, a_opt, score))
            seen.add(key)

    if best_sequence is not None:
        top_configs.append((len(best_sequence), best_sequence.copy(), best_score))

    phase2_budget = min(1200.0, (deadline - time.monotonic()) * 0.50)
    n_deep = max(1, len(top_configs))
    time_per_deep = max(10.0, phase2_budget / n_deep)

    for idx, (n, a0, sc) in enumerate(top_configs):
        if time.monotonic() > deadline - 700:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 700)
        if t_alloc < 5.0:
            break
        print("  [%d/%d] n=%d, prev_score=%.6f, alloc=%.0fs" %
              (idx + 1, n_deep, n, sc, t_alloc))

        # Multi-stage: Adam -> CD -> Adam -> Anneal
        sub = t_alloc
        stages = [
            ('adam1', 0.35, dict(temp_init=30, temp_max=200, lr_init=0.02,
                                 restart_every=20)),
            ('cd1', 0.20, dict()),
            ('adam2', 0.25, dict(temp_init=80, temp_max=400, lr_init=0.01,
                                 restart_every=30)),
            ('anneal', 0.20, dict()),
        ]
        a_cur = a0.copy()
        for stage_name, frac, params in stages:
            t_stage = sub * frac
            if time.monotonic() > deadline - 700:
                break
            if t_stage < 1.0:
                break
            if stage_name.startswith('adam'):
                a_cur, _ = adam_optimize(a_cur, time_limit=t_stage, **params)
            elif stage_name.startswith('cd'):
                a_cur, _ = coordinate_descent(a_cur, time_limit=t_stage)
            elif stage_name.startswith('anneal'):
                a_cur, _ = informed_anneal(a_cur, time_limit=t_stage,
                                           seed=seed + idx * 100)
            record(a_cur)

    print("  Phase 2 complete. Best: %.6f" % best_score)

    # Phase 3: Intensive fine-tuning on best
    if best_sequence is not None and time.monotonic() < deadline - 50:
        print("Phase 3: Intensive fine-tuning...")
        phase3_budget = min(500.0, deadline - time.monotonic() - 30)
        a_best = best_sequence.copy()
        n_best = len(a_best)

        # Multiple annealing runs with different seeds
        n_anneals = max(2, int(phase3_budget / 80))
        time_per_anneal = phase3_budget / n_anneals

        for i in range(n_anneals):
            if time.monotonic() > deadline - 30:
                break
            a_start = a_best.copy()
            # Add small perturbation for diversity
            if i > 0:
                pert = 0.02 * (i / n_anneals)
                a_start = a_start + pert * np.random.randn(n_best)
                a_start = normalize(a_start)
            a_opt, _ = informed_anneal(a_start, time_limit=time_per_anneal,
                                       seed=seed + 10000 + i * 7)
            record(a_opt)
            if best_sequence is not None:
                a_best = best_sequence.copy()

    print("  Phase 3 complete. Best: %.6f" % best_score)

    # Phase 4: Final polish with very fine Adam
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_polish = best_sequence.copy()
            # Fine Adam
            t_half = remaining * 0.6
            a_opt, _ = adam_optimize(a_polish, time_limit=t_half,
                                     temp_init=200, temp_max=1000,
                                     lr_init=0.005, restart_every=50)
            record(a_opt)

            if time.monotonic() < deadline - 5:
                rem = deadline - time.monotonic() - 5
                if rem > 5:
                    a_opt2, _ = coordinate_descent(a_polish, time_limit=rem)
                    record(a_opt2)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = safe_eval(best_sequence)
        print("  WARNING: Using fallback uniform sequence. Score: %.6f" % best_score)

    print("\nFINAL: score=%.6f, length=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
