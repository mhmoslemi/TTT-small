```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Nesterov-accelerated subgradient method with Polyak step size
2. Fast coordinate descent with incremental convolution updates
3. KKT-based refinement
4. Multi-start with diverse initializations

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_nonneg_sum(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(a)
    if S <= 0:
        return np.zeros(n)
    if S / n >= a.max():
        return np.full(n, S / n)
    lo, hi = 0.0, a.max()
    for _ in range(40):
        tau = (lo + hi) / 2.0
        s = np.sum(np.maximum(a - tau, 0.0))
        if s > S:
            lo = tau
        else:
            hi = tau
    return np.maximum(a - (lo + hi) / 2.0, 0.0)


def subgrad_at(c, a, n, k_top=1):
    """Compute subgradient (average of top-k) of max conv(a,a) w.r.t. a."""
    top_idx = np.argpartition(c, -k_top)[-k_top:]
    grad = np.zeros(n)
    for kval in top_idx:
        kval = int(kval)
        lo = max(0, kval - n + 1)
        hi = min(n - 1, kval)
        # grad[m] += a[kval - m] for m in [lo, hi]
        # a[kval-m] for m in [lo,hi] = a[kval-hi : kval-lo+1] reversed
        jac = a[kval - hi : kval - lo + 1]
        grad[lo:hi+1] += jac[::-1]
    grad /= float(k_top)
    return grad


def nesterov_subgrad(a_init, n, S, time_budget, k_top=5, beta=0.9):
    """Nesterov-accelerated subgradient method with Polyak step."""
    a = a_init.copy().astype(np.float64)
    a = project_nonneg_sum(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    y = a.copy()  # Nesterov extrapolation point
    start = time.monotonic()
    t = 0
    
    while time.monotonic() - start < time_budget:
        t += 1
        
        # Evaluate at y
        c_y = fft_convolve_self(y)
        f_y = c_y.max()
        
        # Subgradient at y
        grad = subgrad_at(c_y, y, n, k_top)
        
        # Polyak step size
        gap = f_y - best_obj
        if gap < 1e-10:
            gap = 1e-10
        gnorm2 = np.dot(grad, grad)
        if gnorm2 < 1e-30:
            break
        eta = min(gap / gnorm2, 1.0 / (np.sqrt(t) + 1))
        
        # Nesterov update
        x_new = project_nonneg_sum(y - eta * grad, S)
        # Extrapolate
        y = (1 - beta) * x_new + beta * a
        y = project_nonneg_sum(y, S)
        a = x_new
        
        # Evaluate at a
        c_a = fft_convolve_self(a)
        f_a = c_a.max()
        if f_a < best_obj:
            best_obj = f_a
            best_a = a.copy()
            if t % 50 == 0:
                print("  Nesterov: t=%d obj=%.6f" % (t, best_obj))
    
    return best_a, best_obj

                    idx = 2 * m - k_lo
                    c_aff[idx] = c[2*m] + mid * mid - v * v
                
                local_max = max(c_aff.max(), unaff)
                
                if local_max < best_val:
                    best_val = local_max
                    best_v = mid
                
                # Subgradient sign for binary search direction
                # The max is a convex piecewise-linear function of v'
                # Find the active lag
                k_star = k_lo + np.argmax(c_aff)
                if k_star == 2 * m:
                    deriv = 2.0 * mid
                else:
                    j_star = k_star - m
                    if 0 <= j_star < n:
                        deriv = 2.0 * a[j_star]
                    else:
                        deriv = 0.0
                
                if deriv > 0:
                    lo = mid
                else:
                    hi = mid
            
            a[m] = best_v
            if abs(a[m] - v) > 1e-12:
                improved = True
        
        # Renormalize after each sweep
        a = project_nonneg_sum(a, S)
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_obj
```

Wait, I realize there's an issue with the coordinate descent. When I update `a[m]`, the convolution `c` becomes stale. I can't use the old `c` for the next coordinate. I need to either:
1. Update `c` incrementally after each coordinate change.
2. Recompute `c` from scratch after each coordinate (too slow).
3. Recompute `c` after each sweep (the within-sweep updates are approximate).

Option 3 is the simplest and should work in practice. The coordinate descent with renormalization after each sweep is a standard "relaxed" approach.

But actually, I can do option 1 efficiently. After changing `a[m]` from `v` to `v'`:
- For k in [m, min(m+n-1, 2n-2)]: `c[k] += 2*(v'-v)*a_new[k-m]` (where a_new is the updated a)
  Wait, but a_new[k-m] might be different from a[k-m] if k-m was already updated in this sweep.
  
  This is the issue with in-place coordinate descent: the updates are sequential, so later coordinates see the effect of earlier ones.

  For simplicity, let me just use option 3: recompute the convolution after each sweep. The within-sweep binary search uses the "stale" convolution, which gives an approximate answer. This is fine for a heuristic.

Actually, let me reconsider. The coordinate descent with recompute-after-sweep might not converge well. A better approach:

**Sequential coordinate descent with incremental updates:**
1. Start with `a` and `c = conv(a, a)`.
2. For m = 0, 1, ..., n-1:
   a. Find optimal `a'[m]` using the current `c` (which might be slightly stale for this sweep).
   b. Update `a[m] = a'[m]`.
   c. Update `c` incrementally: `c[k] += 2*(a'[m] - a[m])*a[k-m]` for affected k (and fix the diagonal).
3. After the sweep, renormalize and recompute `c`.

This way, within a sweep, the convolution is always up-to-date (except for the current coordinate being updated). This is more accurate.

Let me implement this:

```python
def coord_descent_incremental(a_init, n, S, time_budget, max_sweeps=100):
    a = a_init.copy().astype(np.float64)
    a = project_nonneg_sum(a, S)
    c = fft_convolve_self(a).copy()
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    n_lags = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        for m in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            v = a[m]
            if v < 1e-15:
                # Skip (or try a small value)
                continue
            
            # Find optimal v' in [0, 2*v] using current c
            k_lo = m
            k_hi = min(m + n - 1, n_lags - 1)
            j_end = k_hi - m + 1
            
            # Unaffected max (using current c, which is up-to-date for all lags)
            unaff = -np.inf
            if m > 0:
                unaff = np.max(c[:m])
            if m + n <= n_lags - 1:
                unaff = max(unaff, np.max(c[m+n:]))
            
            # Binary search
            best_v = v
            best_val = c.max()
            lo, hi = 0.0, v * 2.0
            
            for _ in range(15):
                mid = (lo + hi) / 2.0
                delta = 2.0 * (mid - v)
                c_aff = c[k_lo:k_hi+1] + delta * a[:j_end]
                if 2 * m >= k_lo and 2 * m <= k_hi:
                    idx = 2 * m - k_lo
                    c_aff[idx] = c[2*m] + mid*mid - v*v
                local_max = max(c_aff.max(), unaff)
                if local_max < best_val:
                    best_val = local_max
                    best_v = mid
                # Direction
                k_star = k_lo + np.argmax(c_aff)
                if k_star == 2 * m:
                    deriv = 2.0 * mid
                else:
                    deriv = 2.0 * a[k_star - m] if (k_star - m) < n else 0.0
                if deriv > 0:
                    lo = mid
                else:
                    hi = mid
            
            a[m] = best_v
            
            # Update c incrementally
            dv = best_v - v
            if abs(dv) > 1e-15:
                # c[k] += 2*dv*a[k-m] for k in [m, k_hi], k != 2m
                c[k_lo:k_hi+1] += 2.0 * dv * a[:j_end]
                # Fix diagonal
                if 2 * m >= k_lo and 2 * m <= k_hi:
                    idx = 2 * m - k_lo
                    # We already added 2*dv*a[0] = 2*dv*v (since a[m] = v before update, but a[:j_end] includes a[0] which is a[0])
                    # Wait, a[:j_end] = a[0], a[1], ..., a[j_end-1]
                    # The diagonal term is at k = 2m, so j = k-m = m
                    # c[2m] += 2*dv*a[m] (from the linear term)
                    # But we need c[2m] to increase by dv^2 + 2*dv*v = dv*(2v + dv) = dv*(v + best_v)
                    # The linear term gives 2*dv*a[m_old] = 2*dv*v
                    # We need an additional dv^2
                    c[2*m] += dv * dv
        
        # Renormalize
        a = project_nonneg_sum(a, S)
        # Recompute c from scratch (to avoid accumulated errors)
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
    
    return best_a, best_obj
```

This looks reasonable. The key efficiency gain is that within a sweep, I don't recompute the full convolution; I just update the affected lags.
