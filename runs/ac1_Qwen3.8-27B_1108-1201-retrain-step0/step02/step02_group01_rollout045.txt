import time
from scipy.fft import fft, ifft, next_fast_len

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = next_fast_len(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def true_objective(a):
    """Compute the true evaluation."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)

def spectral_compress(a, beta, n_freqs=None, gamma=0.1):
    """Compress the power spectrum and project back to time domain.
    
    beta > 1: compresses peaks (makes spectrum flatter)
    gamma: blending parameter (0 = no change, 1 = full update)
    """
    n = len(a)
    N = next_fast_len(2 * n - 1)
    A = fft(a, N)
    P = np.abs(A) ** 2
    
    # Compress: P_new = P^(1/beta) * scaling
    # We want to keep P(0) fixed (DC component)
    P0 = P[0]
    
    if n_freqs is not None:
        # Only compress a subset of frequencies
        idx = np.arange(1, N)
        idx = idx[:n_freqs]
    else:
        idx = np.arange(1, N)
    
    P_work = P[idx].copy()
    if P_work.max() < 1e-30:
        return a.copy()
    
    # Compress: raise to power 1/beta (beta > 1 makes it flatter)
    P_compressed = P_work ** (1.0 / beta)
    
    # Scale to maintain same total power (excluding DC)
    total_old = P_work.sum()
    total_new = P_compressed.sum()
    if total_new > 1e-30:
        P_compressed *= total_old / total_new
    
    A_new = A.copy()
    A_new[idx] = np.sqrt(P_compressed) * np.exp(1j * np.angle(A[idx]))
    
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    
    S = a.sum()
    if a_new.sum() < 1e-10:
        return a.copy()
    a_new = a_new * (S / a_new.sum())
    
    return (1 - gamma) * a + gamma * a_new

def topk_subgradient_step(a, k_peaks, step_size):
    """Take a subgradient step based on top-k peaks of the self-convolution."""
    n = len(a)
    c = fft_convolve_self(a)
    
    # Find top-k peaks
    top_idx = np.argsort(c)[::-1][:k_peaks]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / k_peaks
    
    # Gradient of sum_k w[k] * conv(a,a)[k] with respect to a[i]
    # conv(a,a)[k] = sum_j a[j] * a[k-j]
    # d/d a[i] conv(a,a)[k] = a[k-i] + a[i-k] (with out-of-range = 0)
    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    
    grad = np.zeros(n)
    for k in top_idx:
        # a[k-i] for valid i: 0 <= k-i < n => k-n+1 <= i <= k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += w[k] * a[k - i]
        # a[i-k] for valid i: 0 <= i-k < n => k <= i <= k+n-1
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        for i in range(lo2, hi2 + 1):
            grad[i] += w[k] * a[i - k]
    
    # Normalize gradient
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:

    # Step in negative gradient direction

    S = a.sum()
    if a_new.sum() < 1e-10:
        return a.copy()
    a_new = a_new * (S / a_new.sum())
    
    return a_new
