So the constraints are: for each k from 0 to 2n-2:
`sum_{l=max(0,k-n+1)}^{min(n-1,k)} f_{k-l} * g_l <= M`

This is a linear program with n variables and 2n-1 constraints. We can solve it with scipy.optimize.linprog.

For n=256, this is 256 variables and 511 constraints - very tractable.

The optimal solution will be at a vertex, determined by n active constraints. The idea is that the active constraints correspond to the "tight" peaks in the convolution.

After solving, we get `g*`. Rescale to `g = g* / sum(g*)` (so sum is 1). Then update `f <- (1-t) f + t g` for small t.

```python
import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def lp_direction(f, M):
    """Find the LP direction: max sum(g) s.t. g >= 0, conv(f,g) <= M."""
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: max sum(g) => min -sum(g)
    c = -np.ones(n)
    
    # Constraints: conv(f, g)_k <= M for k = 0, ..., 2n-2
    # A_ub @ g <= b_ub
    # A_ub[k, l] = f[k-l] if 0 <= k-l < n, else 0
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        l_start = max(0, k - n + 1)
        l_end = min(n-1, k)
        for l in range(l_start, l_end + 1):
            A_ub[k, l] = f[k - l]
    b_ub = np.full(2*n-1, M)
    
    # bounds: g_i >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        if g.sum() > 1e-10:
            return g / g.sum()
    return None
```

```python
"""
Hybrid optimizer using LP-based direction finding, mirror descent,
and informed annealing to minimize 2*n*max(conv(a,a))/(sum(a))^2.
The key novelty is the LP-based peak reduction: at each step, solve an LP
to find a direction that is "compatible" with the current peak constraint,
then move toward that direction.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def fast_convolve_cross(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]

def obj_value(a):
    """Compute the objective value: 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def project_simplex(v):
    """Project vector onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M):
    """
    Solve: max sum(g) s.t. g >= 0, conv(f,g)_k <= M for all k.
    Returns normalized g (sum=1) or None if infeasible.
    """
    n = len(f)
    c = -np.ones(n)
    
    # Build constraint matrix
    # conv(f,g)_k = sum_l f[k-l] * g_l for valid l
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        l_start = max(0, k - n + 1)
        l_end = min(n - 1, k)
        if l_start <= l_end:
            A_ub[k, l_start:l_end+1] = f[k - l_end : k - l_start + 1][::-1]
```

```python
    Minimize t s.t. b_k + 2*conv(a,d)_k <= t, -a <= d <= delta_box, sum(d)=0.

    c = np.zeros(n + 1)

    # 1) b_k + 2*conv(a,d)_k - t <= 0 for k=0,...,2n-2
```

```python
peak-reducing direction, then take a verified step in that direction.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_norm(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(conv, a, t, thresh_frac=0.003):
    """Smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = t * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    thresh = w.max() * thresh_frac
    active = np.where(w > thresh)[0]
    for ki in active:
        k = int(ki)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    return sg


def solve_lp_peak(a, delta_box):
    """
    Solve LP: minimize t subject to
      b_k + 2*conv(a,d)_k <= t  for all k
      -a_j <= d_j <= delta_box  for all j
      sum(d) = 0
    Returns (d, t_opt) or (None, None).
    """
    n = len(a)
    b = fast_convolve(a)

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_rows = []
    b_rows = []

    for k in range(2 * n - 1):
        row = np.zeros(n + 1)
        l_start = max(0, k - n + 1)
        l_end = min(n - 1, k)
        if l_start <= l_end:
            idx = np.arange(l_start, l_end + 1)
            row[idx] = 2.0 * a[k - idx]
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-b[k])

    for j in range(n):
        row = np.zeros(n + 1)
        row[j] = -1.0
        A_rows.append(row)
        b_rows.append(a[j])

    for j in range(n):
        row = np.zeros(n + 1)
        row[j] = 1.0
        A_rows.append(row)
        b_rows.append(delta_box)

    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  method='highs', options={'time_limit': 8.0})

    if res.success:
        d = res.x[:n]
        t_opt = res.x[n]
        return d, t_opt
    return None, None


def lp_optimize(a0, time_limit, delta_init=0.03, max_iters=500):
    """
    LP-based peak reduction: iteratively solve LP for best linearized
    peak-reducing step, take verified steps.
    """
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()

    t0 = time.monotonic()
    iter_count = 0
    delta = delta_init
    stuck_count = 0

    while time.monotonic() - t0 < time_limit and iter_count < max_iters:
        M = conv.max()
        if M < 1e-15:
            break

        d, t_opt = solve_lp_peak(a, delta)
        if d is None:
            delta *= 0.5
            stuck_count += 1
            if stuck_count > 20:
                break
            continue

        # Check if LP says improvement is possible
        if t_opt >= M * (1.0 - 1e-10):
            stuck_count += 1
            if stuck_count > 5:
                delta *= 0.7
                stuck_count = 0
            continue

        # Try step sizes
        best_t_frac = 0.0
        best_new_obj = cur_obj
        for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_try = a + frac * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try /= s_try
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_new_obj:
                best_new_obj = obj_try
                best_t_frac = frac

        if best_t_frac == 0.0:
            stuck_count += 1
            if stuck_count > 15:
                delta *= 0.6
                stuck_count = 0
            continue

        a = a + best_t_frac * d
        a = np.maximum(a, 0.0)
        a /= a.sum()
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        # Adapt delta
        if stuck_count > 10:
            delta *= 0.7
            stuck_count = 0
        elif best_t_frac >= 0.5 and stuck_count == 0:
            delta = min(delta * 1.1, 0.1)

        iter_count += 1

    return best_a, best_obj


def mirror_descent(a0, time_limit, t_init=50.0, eta_init=0.3):
    """Multiplicative mirror descent on the simplex."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    max_pat = 50
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient(conv, a, t)
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            eta *= 0.65
            t *= 1.3
            if eta < 0.004:
                elapsed = time.monotonic() - t0
                frac = min(1.0, elapsed / max(time_limit, 1.0))
                eta = eta_init * (1.0 - 0.5 * frac)
                t = t_init * (1.0 + 3.0 * frac)
    return best_a, best_obj


def pgd_optimize(a0, time_limit, t_init=30.0, lr_init=0.08):
    """Projected gradient descent with logsumexp smoothing."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    lr = lr_init
    patience = 0
    max_pat = 40
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient(conv, a, t)
        a_new = a - lr * sg
        a_new = project_simplex(a_new)
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            lr *= 0.7
            t *= 1.2
            if lr < 0.002:
                lr = lr_init * 0.5
                t = t_init * 2.0
    return best_a, best_obj


def anneal_informed(a0, time_limit, rng):
    """Informed simulated annealing targeting peak reduction."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.015 * (1.0 - frac) ** 2 + 0.0005
        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.80
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        # Choose target: prefer positions with low contribution to peaks
        target_j = int(rng.integers(n))
        for _ in range(3):
            candidate = int(rng.integers(n))
            contrib = 0.0
            for pk in peak_positions[:6]:
                d = pk - candidate
                if 0 <= d <= n - 1:
                    contrib += a[d]
            if contrib < a.mean() * 1.5:
                target_j = candidate
                break
        if target_j == top_j:
            continue
        delta = a[top_j] * rng.uniform(0.01, 0.05)
        if delta < 1e-9:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for a given length."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n
    for sf in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_' + str(sf)] = g
    for freq in [1, 2, 3, 5]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        s /= s.sum()
        inits['cos_' + str(freq)] = s
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    # Tri-modal
    tri = np.zeros(n)
    w = max(1, n // 7)
    tri[:w] = 1.0
    tri[n//2 - w//2:n//2 + w//2] = 1.0
    tri[-w:] = 1.0
    if tri.sum() > 0:
        tri /= tri.sum()
        inits['trimodal'] = tri
    # Random
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['rand_' + str(alpha)] = r
    # Chirp
    chirp = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * x / (2.0 * n)))
    chirp = np.maximum(chirp, 0.0)
    if chirp.sum() > 1e-10:
        chirp /= chirp.sum()
        inits['chirp'] = chirp
    # Parent interpolation
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0).copy()
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp
    return inits

def safe_evaluate(a):
    """Safely evaluate a sequence, returning inf on failure."""
    try:
        a_clean = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a_clean = np.minimum(a_clean, 1000.0)
        if a_clean.sum() < 0.01:
            a_clean = a_clean * (1.0 / max(a_clean.sum(), 1e-15))
        if not np.all(np.isfinite(a_clean)):
            return float('inf')

        return score
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=2400):
    """
    Main entry point. Uses LP-based peak reduction, mirror descent, PGD,
    and informed annealing to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)

    parent = get_parent()

    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d" % (score, len(a)))

    # Phase 1: Quick scan with mirror descent
    print("Phase 1: Quick scan...")
    scan_lengths = [256, 384, 512, 768, 1024]
    if parent is not None:
        scan_lengths.insert(0, len(parent))
    scan_lengths = list(dict.fromkeys(scan_lengths))  # deduplicate, preserve order
    scan_results = []

    phase1_budget = min(300.0, (deadline - time.monotonic()) * 0.12)
    n_configs_total = 0
    for n in scan_lengths:
        inits = generate_initializations(n, rng, parent)
        n_configs_total += len(inits)

    time_per_scan = max(1.5, phase1_budget / max(1, n_configs_total))

    for n in scan_lengths:
        if time.monotonic() > deadline - 1500:
            break
        inits = generate_initializations(n, rng, parent)
        for name, a0 in inits.items():
            if time.monotonic() > deadline - 1500:
                break
            t_alloc = min(time_per_scan, deadline - time.monotonic() - 1500)
            if t_alloc < 1.0:
                break
            a_opt, _ = mirror_descent(a0, time_limit=t_alloc, t_init=40.0, eta_init=0.4)
            score = safe_evaluate(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt))
            record(a_opt)

    print("  Scan complete. Best: %.6f" % best_score)
    scan_results.sort(key=lambda x: x[0])

    # Phase 2: Deep LP optimization on top candidates
    print("Phase 2: Deep LP optimization...")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 10:
            top_configs.append((n, name, a_opt))
            seen.add(key)

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    phase2_budget = min(1400.0, (deadline - time.monotonic()) * 0.55)
    n_deep = len(top_configs)
    time_per_deep = max(10.0, phase2_budget / max(1, n_deep))

    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 700:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 700)
        if t_alloc < 5.0:
            break
        print("  [%d/%d] n=%d, %s, budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))

        a_cur = a0.copy()

        # LP optimization (main)
        sub = t_alloc * 0.45
        a_cur, _ = lp_optimize(a_cur, time_limit=sub, delta_init=0.03)
        record(a_cur)

        # Mirror descent refinement
        sub = t_alloc * 0.2
        a_cur, _ = mirror_descent(a_cur, time_limit=sub, t_init=100.0, eta_init=0.1)
        record(a_cur)

        # More LP
        if time.monotonic() < deadline - 700:
            sub = t_alloc * 0.2
            a_cur, _ = lp_optimize(a_cur, time_limit=sub, delta_init=0.015)
            record(a_cur)

        # PGD polish
        if time.monotonic() < deadline - 700:
            sub = t_alloc * 0.15
            a_cur, _ = pgd_optimize(a_cur, time_limit=sub, t_init=80.0, lr_init=0.03)
            record(a_cur)

    print("  Deep optimization complete. Best: %.6f" % best_score)

    # Phase 3: Annealing and final LP polish
    if best_sequence is not None and time.monotonic() < deadline - 80:
        print("Phase 3: Annealing + final LP...")
        phase3_budget = min(600.0, deadline - time.monotonic() - 80)

        # Multiple annealing runs
        n_anneal = 3
        anneal_per = phase3_budget * 0.5 / n_anneal
        for i in range(n_anneal):
            if time.monotonic() > deadline - 80:
                break
            a_start = best_sequence.copy()
            rng_a = np.random.default_rng(seed + 7000 + i * 137)
            a_opt, _ = anneal_informed(a_start, time_limit=anneal_per, rng=rng_a)
            record(a_opt)

        # Final LP refinement
        if best_sequence is not None and time.monotonic() < deadline - 40:
            remaining = deadline - time.monotonic() - 40
            if remaining > 20:
                a_best = best_sequence.copy()
                a_best = np.maximum(a_best, 0.0)
                a_best /= a_best.sum()
                a_opt, _ = lp_optimize(a_best, time_limit=remaining * 0.6, delta_init=0.01)
                record(a_opt)

                if time.monotonic() < deadline - 40:
                    rem2 = deadline - time.monotonic() - 40
                    if rem2 > 10:
                        rng_f = np.random.default_rng(seed + 12345)
                        a_opt2, _ = anneal_informed(a_best, time_limit=rem2, rng=rng_f)
                        record(a_opt2)

        print("  Phase 3 complete. Best: %.6f" % best_score)

    # Phase 4: Ultra-fine polish
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("Phase 4: Ultra-fine polish...")
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            a_best = best_sequence.copy()
            a_best = np.maximum(a_best, 0.0)
            a_best /= a_best.sum()
            # Very fine LP steps
            a_opt, _ = lp_optimize(a_best, time_limit=remaining * 0.5, delta_init=0.005)
            record(a_opt)
            if time.monotonic() < deadline - 10:
                rem = deadline - time.monotonic() - 10
                if rem > 5:
                    a_opt, _ = mirror_descent(a_best, time_limit=rem, t_init=500.0, eta_init=0.02)
                    record(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
```
