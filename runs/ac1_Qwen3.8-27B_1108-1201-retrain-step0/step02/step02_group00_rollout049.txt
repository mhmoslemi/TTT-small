"""
Minimize 2*n*max_conv / sum^2 using combined spectral shaping + Nesterov top-k subgradient.

Key innovations over previous approach:
1. Spectral shaping: modify |FFT(a)|^2 in frequency domain to flatten autocorrelation,
   then project back to non-negative simplex. This is a "global" move that affects all
   elements simultaneously and can escape local minima that trap pure time-domain methods.
2. Multi-segment initialization: find good 3/5/9-segment structures analytically,
   then use as starting points for full optimization.
3. Adaptive alternating: switch between spectral (global) and subgradient (local) steps
   based on progress detection.
4. Multi-start with diverse initializations including m-sequence inspired patterns.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_toppk(a, c, top_k):
    """Gradient using top-k active lags, computed via FFT cross-correlation."""
    n = len(a)
    if top_k > len(c):
        top_k = len(c)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_shaping_step(a, N, top_k, alpha, rng=None):
    """
    One step of spectral shaping: modify |FFT(a)|^2 to reduce peak convolution.
    Returns (new_a, c, c_max, improvement).
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    # Compute effective cosine for top-k lags
    m = np.arange(N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a, c, c_max, 0.0
    weights = top_vals / total

    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    # Scale alpha by how far above target we are
    alpha_eff = alpha * min(1.0, c_max / 1.0)
    P_new = np.maximum(P * (1.0 - alpha_eff * cos_eff), 0.0)

    # Preserve phase, modify magnitude
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]

    # Project to simplex
    a_new = project_simplex(a_new_full, a.sum())
    return a_new, c, c_max, 0.0


def spectral_shaping_multi(a, N, top_k, alpha, n_iter, rng, target=1.5):
    """
    Multiple spectral shaping steps with adaptive alpha.
    Returns best a found during the run.
    """
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score = np.inf

    for it in range(n_iter):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = c.max()

        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        if c_max <= target:
            break

        # Adaptive alpha: larger when further from target
        excess = max(0.0, c_max - target) / max(c_max, 1e-10)
        alpha_curr = alpha * (0.3 + 0.7 * excess)

        m = np.arange(N)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

        P_new = np.maximum(P * (1.0 - alpha_curr * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)

        # Occasionally add small noise to escape local minima
        if it % 200 == 100 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

    return best_a, best_score


def nesterov_toppk_polish(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=512):
    """Nesterov-accelerated top-k subgradient with FFT gradient for polishing."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = fft_grad_toppk(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 3000 == 0 and rng.random() < 0.15:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def fista_lse(a_init, n, S, max_time, rng, beta_init=3.0, beta_max=300.0):
    """FISTA with log-sum-exp smooth approximation."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    step = 1.0 / (beta_init * 2.0)
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.003 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        step = 1.0 / (beta * 2.0 * (1.0 + 0.0005 * it))
        step = max(step, 1e-6)

        if it % 2000 == 0 and rng.random() < 0.12:
            y = project_simplex(y + rng.normal(0, 0.004 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def multi_segment_init(n, k, rng):
    """Create a multi-segment initialization with optimized heights."""
    # Use equal-width segments for simplicity
    seg_len = n // k
    remainder = n % k
    boundaries = [0]
    for i in range(k):
        boundaries.append(boundaries[-1] + seg_len + (1 if i < remainder else 0))

    # Try random heights and keep the best
    best_score = np.inf
    best_heights = None
    for trial in range(200):
        heights = rng.random(k) + 0.1
        a = np.zeros(n)
        for i in range(k):
            a[boundaries[i]:boundaries[i + 1]] = heights[i]
        c = fft_convolve_self(a)
        score = c.max() / (a.sum() ** 2 / (2 * n))
        if score < best_score:
            best_score = score
            best_heights = heights.copy()

    # Build final sequence
    a = np.zeros(n)
    for i in range(k):
        a[boundaries[i]:boundaries[i + 1]] = best_heights[i]
    return a, best_score


def make_msequence_init(n, rng):
    """Create an m-sequence inspired initialization (binary with good autocorrelation)."""
    # Generate a pseudo-random binary sequence with balanced 0s and 1s
    a = rng.random(n)
    # Apply a "chirp" modulation to create structure
    t = np.arange(n) / n
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (3 * t + 0.5 * t * t)) + 0.1 * rng.standard_normal(n)
    a = np.clip(a, 0.05, 1.0)
    return a


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses combined spectral shaping + Nesterov subgradient with multi-start.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None

    # Determine main length
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            main_len = len(given)
        else:
            main_len = 8192
    except Exception:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()  # FFT size for full convolution

    print("n=%d, S=%.4f, N(FFT)=%d" % (n, S, N))

    # --- Phase 1: Diverse exploration ---
    print("=== Phase 1: Exploration ===")
    t_phase1 = time.monotonic()
    phase1_budget = (deadline - t_phase1) * 0.35

    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
    except Exception:
        pass

    tt = np.linspace(0, 1, n, endpoint=False)
    inits.append(("rand", rng.random(n) + 0.05))
    inits.append(("chirp", 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)))
    inits.append(("sin_multi", 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt) + 0.1 * np.sin(2 * np.pi * 13 * tt)))
    inits.append(("bump", np.exp(-0.5 * ((tt - 0.5) * 5.0) ** 2) + 0.1))

    # Multi-segment inits
    ms3, _ = multi_segment_init(n, 3, rng)
    inits.append(("seg3", ms3))
    ms5, _ = multi_segment_init(n, 5, rng)
    inits.append(("seg5", ms5))
    ms9, _ = multi_segment_init(n, 9, rng)
    inits.append(("seg9", ms9))

    # M-sequence inspired
    inits.append(("mseq", make_msequence_init(n, rng)))

    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d initializations, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 60:
            break
        a_norm = safe_normalize(arr, n)

        t0 = time.monotonic()
        # Spectral shaping burst
        ss_budget = min(per_init * 0.5, 120.0)
        n_ss_iters = min(5000, int(ss_budget / 0.002))  # ~2ms per SS step
        a_ss, score_ss = spectral_shaping_multi(a_norm, N, top_k=10, alpha=0.3,
                                                 n_iter=n_ss_iters, rng=rng, target=1.45)
        a_ss = safe_normalize(a_ss, n)
        c_ss = fft_convolve_self(a_ss)
        sc = c_ss.max()
        if sc < global_best:
            global_best = sc
            global_a = a_ss.copy()
            print("  [%s] SS: %.6f" % (name, global_best))

        # FISTA burst on the best from SS
        rem = per_init - (time.monotonic() - t0)
        if rem > 2:
            a_f, score_f = fista_lse(a_ss, n, S, rem, rng, beta_init=5.0, beta_max=200.0)
            a_f = safe_normalize(a_f, n)
            c_f = fft_convolve_self(a_f)
            if c_f.max() < global_best:
                global_best = c_f.max()
                global_a = a_f.copy()
                print("  [%s] FISTA: %.6f" % (name, global_best))

    print("=== Phase 1 done. Best: %.6f ===" % global_best)

    # --- Phase 2: Focused optimization on best candidate ---
    if global_a is None:
        global_a = np.ones(n)
        global_best = 2.0

    print("=== Phase 2: Focused optimization ===")
    t_phase2 = time.monotonic()
    phase2_budget = (deadline - t_phase2) * 0.45

    # Combined SS + Nesterov
    a_curr = global_a.copy()
    S_curr = a_curr.sum()
    stall_count = 0
    last_improve = global_best
    it_outer = 0

    while time.monotonic() < t_phase2 + phase2_budget:
        it_outer += 1

        # Spectral shaping block
        ss_time = min(30.0, (t_phase2 + phase2_budget) - time.monotonic())
        if ss_time > 1:
            n_iters = min(3000, int(ss_time / 0.002))
            top_k_ss = 5 + 5 * (it_outer // 10)
            a_curr, sc = spectral_shaping_multi(a_curr, N, top_k=min(top_k_ss, 50),
                                                 alpha=0.25 / (1.0 + 0.1 * it_outer),
                                                 n_iter=n_iters, rng=rng, target=1.45)
            a_curr = safe_normalize(a_curr, n)
            c_curr = fft_convolve_self(a_curr)
            if c_curr.max() < global_best:
                global_best = c_curr.max()
                global_a = a_curr.copy()
                last_improve = global_best
                stall_count = 0
                if it_outer % 5 == 0:
                    print("  [%d] SS: %.6f" % (it_outer, global_best))
            else:
                stall_count += 1

        # Nesterov subgradient block
        rem = (t_phase2 + phase2_budget) - time.monotonic()
        if rem > 3:
            top_k_ng = 10 + 10 * (it_outer // 5)
            a_ng, sc_ng = nesterov_toppk_polish(a_curr, n, S, min(rem, 60.0), rng,
                                                 top_k_init=top_k_ng, top_k_max=top_k_ng + 100)
            a_ng = safe_normalize(a_ng, n)
            c_ng = fft_convolve_self(a_ng)
            if c_ng.max() < global_best:
                global_best = c_ng.max()
                global_a = a_ng.copy()
                a_curr = a_ng.copy()
                last_improve = global_best
                stall_count = 0
                print("  [%d] NG: %.6f" % (it_outer, global_best))
            else:
                stall_count += 1

        # Restart if stalled
        if stall_count >= 4:
            print("  [%d] Stalled, restarting with perturbation" % it_outer)
            a_curr = project_simplex(a_curr + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            stall_count = 0

    print("=== Phase 2 done. Best: %.6f ===" % global_best)

    # --- Phase 3: Polish ---
    print("=== Phase 3: Polish ===")
    t_phase3 = time.monotonic()
    remaining = deadline - t_phase3

    if global_a is not None and remaining > 5:
        # High-k Nesterov
        seg1 = min(remaining * 0.35, 300.0)
        a_p1, _ = nesterov_toppk_polish(global_a, n, S, seg1, rng,
                                         top_k_init=50, top_k_max=300)
        a_p1 = safe_normalize(a_p1, n)
        c1 = fft_convolve_self(a_p1)
        if c1.max() < global_best:
            global_best = c1.max()
            global_a = a_p1.copy()
            print("  Polish NG high-k: %.6f" % global_best)

        # Spectral shaping with very small alpha (fine tuning)
        rem2 = deadline - time.monotonic()
        if rem2 > 5:
            n_iters = min(int(rem2 * 0.4 / 0.002), 10000)
            a_p2, _ = spectral_shaping_multi(global_a, N, top_k=30, alpha=0.05,
                                              n_iter=n_iters, rng=rng, target=1.45)
            a_p2 = safe_normalize(a_p2, n)
            c2 = fft_convolve_self(a_p2)
            if c2.max() < global_best:
                global_best = c2.max()
                global_a = a_p2.copy()
                print("  Polish SS fine: %.6f" % global_best)

        # Final FISTA with high beta
        rem3 = deadline - time.monotonic()
        if rem3 > 5:
            a_p3, _ = fista_lse(global_a, n, S, min(rem3, 120.0), rng,
                                 beta_init=100.0, beta_max=500.0)
            a_p3 = safe_normalize(a_p3, n)
            c3 = fft_convolve_self(a_p3)
            if c3.max() < global_best:
                global_best = c3.max()
                global_a = a_p3.copy()
                print("  Polish FISTA: %.6f" % global_best)

        # Very final Nesterov with very high k
        rem4 = deadline - time.monotonic()
        if rem4 > 3:
            a_p4, _ = nesterov_toppk_polish(global_a, n, S, rem4, rng,
                                             top_k_init=100, top_k_max=1024)
            a_p4 = safe_normalize(a_p4, n)
            c4 = fft_convolve_self(a_p4)
            if c4.max() < global_best:
                global_best = c4.max()
                global_a = a_p4.copy()
                print("  Final polish: %.6f" % global_best)

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
