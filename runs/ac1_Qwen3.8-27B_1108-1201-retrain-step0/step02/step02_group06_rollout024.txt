import time

# Helper functions (top-level)

def fft_convolve(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def score_fast(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a)
    return 2.0 * n * np.max(conv) / (S*S)

def smoothed_max_and_grad(conv, tau):
    """Given conv = fft_convolve(a), compute smoothed max and grad w.r.t. conv."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    exp_vals = np.exp(shifted)
    z = np.sum(exp_vals)
    w = exp_vals / z
    value = max_c + tau * np.log(z)
    return value, w

def grad_a_from_weights(a, w):
    """Compute gradient of smoothed max w.r.t. a, given weights w on conv values."""
    n = len(a)
    # grad[i] = 2 * sum_k w[k] * a[k-i] (valid)
    # = 2 * cross_correlation(w, a)[i]
    size = 1
    while size < len(w) + n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(fw) * fa, size)[:n]
    return 2.0 * corr

def softmax(x):
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_init(n, rng, n_iter=50):
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    for _ in range(n_iter):
        conv = fft_convolve(a)
        a = a - 0.5 / n * conv[:n]  # subtract scaled correlation
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a

def adam_optimize(n, a_init, time_limit, rng):
    """ADAM optimization on simplex, minimizing smoothed max of conv."""
    a = a_init.copy()
    s = np.sum(a)
    a = a / s
    n = len(a)
    
    # Use log-barrier parameterization or direct simplex
    # I'll use direct optimization with projection
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 1.0 / (2 * n)
    step = 0
    
    tau_start = 5.0 / n
    tau_end = 0.01 / n
    
    best_a = a.copy()
    best_val = np.max(fft_convolve(a))
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        
        conv = fft_convolve(a)
        value, w = smoothed_max_and_grad(conv, tau)
        grad = grad_a_from_weights(a, w)
        
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        
        # Adaptive learning rate
        lr_eff = lr / (1 + step * 0.001)
        
        candidate = a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        candidate = project_simplex(candidate)
        
        # Check improvement
        new_conv = fft_convolve(candidate)
        new_val = np.max(new_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = candidate.copy()
        
        a = candidate
        
        if step % 500 == 0:
            sc = 2.0 * n * best_val
            print(f"    ADAM step={step}, score={sc:.6f}, tau={tau:.2e}")
    
    return best_a

"""
Optimized step function search minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses multi-start ADAM with smoothed-max, peak-targeted local search,
and spectral initialization. Explores multiple sequence lengths.
"""

import numpy as np
import time


def fft_convolve(a):
    """Linear autocorrelation via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def score_fast(a):
    """Compute 2n*max(conv)/(sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a)
    return 2.0 * n * np.max(conv) / (S * S)


def softmax_stable(x):
    x = x - np.max(x)
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def project_simplex(v):
    """Project v onto {x>=0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_max_weights(conv, tau):
    """Compute softmax weights on conv values and the smoothed max value."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    exp_vals = np.exp(shifted)
    z = np.sum(exp_vals)
    w = exp_vals / z
    value = max_c + tau * np.log(z)
    return value, w


def grad_from_weights(a, w):
    """Compute gradient of smoothed max w.r.t. a using FFT cross-correlation."""
    n = len(a)
    size = 1
    while size < len(w) + n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(fw) * fa, size)[:n]
    return 2.0 * corr


def spectral_init(n, rng, n_iter=80):
    """Spectral iteration: alternate time-domain clipping and correlation subtraction."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    a = a / np.sum(a)
    for it in range(n_iter):
        conv = fft_convolve(a)
        # Subtract the correlation pattern (scaled)
        alpha = 1.0 / (n * 0.5)
        a_new = a - alpha * conv[:n]
        a = np.clip(a_new, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.clip(1.0 + 0.2 * rng.standard_normal(n), 0.005, None)
            a = a / np.sum(a)
        else:
            a = a / s
    return a


def diverse_init(n, rng, style=0):
    """Generate diverse initializations."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if style == 0:
        a = 1.0 + 0.3 * t
    elif style == 1:
        a = 1.0 + 0.4 * t ** 2
    elif style == 2:
        a = 1.0 + 0.3 * np.sin(np.pi * t)
    elif style == 3:
        a = 1.0 + 0.5 * np.sqrt(t)
    elif style == 4:
        a = 1.0 - 0.3 * t
    elif style == 5:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 0.01)
    elif style == 6:
        a = 1.0 + 0.2 * rng.standard_normal(n)
    elif style == 7:
        # Bi-modal
        center = n // 2
        a = 1.0 + 0.5 * np.exp(-((t - 0.3) ** 2) / 0.05) + 0.3 * np.exp(-((t - 0.7) ** 2) / 0.05)
    elif style == 8:
        # Geometric decay
        a = 0.5 ** (t * 2)
        a = a / np.min(a)  # normalize to have min 1
    else:
        a = 1.0 + 0.2 * rng.standard_normal(n)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a


def interpolate_to_length(a, n):
    """Interpolate sequence a to length n."""
    m = len(a)
    if m == n:
        return a.copy()
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, a)


def adam_optimize_simplex(n, a_init, time_limit, rng):
    """
    ADAM optimization on the simplex (sum a = 1, a >= 0).
    Minimizes smoothed max of conv(a,a).
    """
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # Learning rate schedule
    lr_base = 0.5 / n

    tau_start = 10.0 / n
    tau_end = 0.005 / n

    best_a = a.copy()
    best_val = np.max(fft_convolve(a))
    t0 = time.monotonic()
    step = 0
    no_improve = 0
    patience = 3000

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.4)

        conv = fft_convolve(a)
        value, w = smoothed_max_weights(conv, tau)
        grad = grad_from_weights(a, w)

        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)

        # Adaptive LR: larger early, smaller later
        lr = lr_base * (1.0 + 0.5 * np.exp(-step / 500.0))

        candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        candidate = project_simplex(candidate)

        # Evaluate
        new_conv = fft_convolve(candidate)
        new_val = np.max(new_conv)

        if new_val < best_val:
            best_val = new_val
            best_a = candidate.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = candidate

        # Restart if stuck
        if no_improve > patience and frac < 0.8:
            a = best_a.copy() + 0.1 / n * rng.standard_normal(n)
            a = np.clip(a, 0, None)
            s2 = np.sum(a)
            if s2 > 1e-10:
                a = a / s2
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

        if step % 1000 == 0:
            sc = 2.0 * n * best_val
            print(f"      ADAM n={n} step={step} score={sc:.6f} tau={tau:.2e} t={time.monotonic()-t0:.1f}s")

    return best_a, 2.0 * n * best_val


def peak_cd(a_init, time_limit, rng):
    """
    Peak-targeted coordinate descent.
    Directly attacks the maximum of the convolution.
    """
    a = a_init.copy()
    s = np.sum(a)
    a = a / s
    n = len(a)

    conv = fft_convolve(a)
    current_val = np.max(conv)
    best_a = a.copy()
    best_val = current_val
    t0 = time.monotonic()
    stall = 0
    max_stall = 2000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fft_convolve(a)
        peak_idx = int(np.argmax(conv))

        r = rng.random()

        if r < 0.5:
            # Reduce top contributors to the peak
            # conv[peak_idx] = sum_i a_i * a[peak_idx - i]
            contribs = np.zeros(n)
            valid_i = np.arange(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
            for i in valid_i:
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]

            top_k = max(2, n // 32)
            top_idx = np.argsort(contribs)[-top_k:]

            # Find good targets (low contribution to peak)
            other_mask = np.ones(n, dtype=bool)
            other_mask[top_idx] = False
            others = np.where(other_mask)[0]
            if len(others) == 0:
                continue
            other_contribs = contribs[others]
            n_good = min(5, len(others))
            good_idx = others[np.argsort(other_contribs)[:n_good]]

            # Move mass
            total = 0.0
            for i in top_idx:
                frac = rng.uniform(0.03, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                total += amt
            for j in good_idx:
                a[j] += total / len(good_idx)

        elif r < 0.75:
            # Paired transfer: move from high-contrib to low-contrib element
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            # Prefer moving from peak contributors to non-peak
            amt = a[i] * rng.uniform(0.02, 0.08)
            a[i] -= amt
            a[j] += amt
        else:
            # Block perturbation
            width = max(2, int(rng.integers(2, max(3, n // 16))))
            start = int(rng.integers(0, max(1, n - width)))
            a[start:start + width] *= np.exp(rng.normal(0, 0.03, size=width))

        a = np.clip(a, 0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = best_a.copy()
            continue
        a = a / s

        new_val = np.max(fft_convolve(a))
        if new_val < current_val * (1 - 1e-9):
            current_val = new_val
            stall = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # Revert
            a = best_a.copy()
            current_val = best_val
            stall += 1

    return best_a, 2.0 * n * best_val


def gaussian_cd(a_init, time_limit, rng):
    """
    Gaussian perturbation local search on the simplex.
    Tries small random perturbations and accepts improvements.
    """
    a = a_init.copy()
    s = np.sum(a)
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_val = np.max(fft_convolve(a))
    t0 = time.monotonic()
    current_val = best_val
    step_size = 0.05 / np.sqrt(n)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 5000:
        # Generate perturbation
        perturbation = rng.standard_normal(n) * step_size
        candidate = a + perturbation
        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            stall += 1
            continue
        candidate = candidate / s

        new_val = np.max(fft_convolve(candidate))
        if new_val < current_val:
            a = candidate
            current_val = new_val
            stall = 0
            # Occasionally reduce step size for fine-tuning
            if rng.random() < 0.1:
                step_size *= 0.95
            if new_val < best_val:
                best_val = new_val
                best_a = candidate.copy()
        else:
            stall += 1
            # Increase step size if stuck
            if stall % 100 == 0:
                step_size *= 1.2
                step_size = min(step_size, 0.5 / np.sqrt(n))

    return best_a, 2.0 * n * best_val
