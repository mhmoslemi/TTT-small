import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s), conv

def coord_descent_optimize(a, time_limit, rng):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_a = a.copy()
    best_max = conv.max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # One round of coordinate descent
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            k_lo = j
            k_hi = min(j + n - 1, 2*n-2)
            if k_lo > k_hi:
                continue
            L = k_hi - k_lo + 1  # length of affected range
            
            # Current conv values in affected range
            c = conv[k_lo:k_hi+1]
            # Direction: 2*a[k-j] for k in [j, j+n-1]
            d = 2.0 * a[0:L]
            # Quadratic term at k=2j
            quad_idx = 2*j - k_lo  # = j
            has_quad = (0 <= quad_idx < L)
            
            # M_out
            m_out = 0.0
            if k_lo > 0:
                m_out = max(m_out, conv[:k_lo].max())
            if k_hi < 2*n-2:
                m_out = max(m_out, conv[k_hi+1:].max())
            
            # If current max is outside, skip
            if c.max() <= m_out + 1e-14:
                continue
            
            # Find best u
            # f(u) = max(c + u*d + (u^2 if quad)(at quad_idx), m_out) / (1+u)^2
            # Try candidate u values
            best_u = 0.0
            best_f = max(c.max(), m_out)  # f(0) (ignoring the (1+0)^2 = 1 factor)
            
            # Candidate 1: set a[j] to 0
            u1 = -a[j]
            # Candidate 2: individual minimizers
            # For (c[i] + u*d[i])/(1+u)^2, min at u = 1 - c[i]/d[i] (if d[i] > 0)
            candidates = [u1 * 0.999, u1 * 0.5, -a[j] * 0.3, a[j] * 0.5, a[j] * 2.0]
            # Add minimizers for top peaks
            for i in np.argsort(c)[-5:]:

                    if u_star > -a[j] * 0.99 and u_star < 10 * a[j] + 1:
                        candidates.append(u_star)
            
            for u in candidates:
                if u <= -a[j]:

                new_c = c + u * d
                if has_quad:
                    new_c[quad_idx] += u * u
                new_max = max(new_c.max(), m_out) / denom
                if new_max < best_f:
                    best_f = new_max
                    best_u = u

                a[j] = max(0.0, a[j])

                    a = np.ones(n) / n

                improved = True
        
        cur_max = conv.max()

            best_max = cur_max
            best_a = a.copy()

            pert = rng.normal(0, 0.01, n)
            a = np.maximum(a + pert, 0)
            a /= a.sum()
            conv = fast_convolve(a)
    
    return best_a, 2.0 * n * best_max
