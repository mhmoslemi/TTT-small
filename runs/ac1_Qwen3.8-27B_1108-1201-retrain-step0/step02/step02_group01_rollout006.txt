import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog, minimize


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)

import numpy as np

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def objective_value(a):
    """Compute the true objective: 2*n*max(conv)/sum^2"""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def norm_conv_max(a, S):
    """Max of self-convolution when sum(a) = S. Objective = 2n*M/S^2 = M when S=sqrt(2n)."""
    c = fft_convolve_self(a)
    return c.max()


def normalize_to_sum(a, S):

    Uses constraint generation for speed.
    Returns g or None if failed.

    c_ff = fft_convolve_self(f)
    # Start with top-k most "active" constraints
    k_init = min(initial_k, 2 * n - 1)
    active = np.argsort(c_ff)[::-1][:k_init]

    g = None
    for rnd in range(max_rounds):
        # Build constraint matrix for active constraints
        A = np.zeros((len(active), n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # A[i, j] = f[k-j] for j in [lo, hi]
            # k-j ranges from k-hi to k-lo
            A[i, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

                    return g
                return None
            g = result.x.copy()

            return None

        top_viol = np.argsort(viol_amounts)[::-1][:top_viol_count]

def softmax_obj_grad(theta, n, T, S):
    """Softmax objective and gradient w.r.t. theta parameters."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        a = np.full(n, S / n)
        p = np.full(n, 1.0 / n)
    else:
        p = exp_theta / total
        a = S * p

    c = fft_convolve_self(a)
    c_max = c.max()

    # Softmax approximation of max
    scaled = T * (c - c_max)
    shifted = scaled - scaled.max()
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T

    # Weights
    w = np.exp(shifted) / np.sum(np.exp(shifted))

    # grad_a[m] = 2 * sum_j w[m+j] * a[j] (for the part where m+j < 2n-1)
    # This is the cross-correlation of w and a
    # grad_a[m] = 2 * (w * a_rev)[m + n - 1] where a_rev = a[::-1]
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    grad_a = np.zeros(n)
    lo = n - 1
    hi = min(2 * n - 1, len(full_conv))
    grad_a[:] = 2.0 * full_conv[lo:hi][:n]

    # Chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot

    return obj, grad_theta


def run_adam_anneal(f_init, n, S, time_budget, T_start=5.0, T_end=500.0, lr_init=0.05):
    """Adam optimizer with temperature annealing on log-parameterization."""
    a = np.maximum(f_init, 1e-10)
    a = normalize_to_sum(a, S)
    theta = np.log(a / a.sum() + 1e-30)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    start = time.monotonic()
    best_m = norm_conv_max(a, S)
    best_a = a.copy()

    while time.monotonic() - start < time_budget:
        t += 1
        a = normalize_to_sum(np.maximum(f_init, 1e-10), S)  # placeholder
        # Recompute a from theta
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
        else:
            a = S * exp_theta / total

        cur_m = norm_conv_max(a, S)
        if cur_m < best_m:
            best_m = cur_m
            best_a = a.copy()

        # Temperature schedule
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * np.exp(np.log(T_end / T_start) * progress)

        obj, grad = softmax_obj_grad(theta, n, T, S)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_init * (0.6 ** (progress * 2.0))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_m

                    if m < best_m:
                        best_m = m
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] = 0.0
                        a = normalize_to_sum(a, S)
                continue

            # Try reducing and increasing
            for frac in [0.5, 0.7, 0.9, 1.2, 1.5, 2.0]:
                a[i] = orig * frac
                s = a.sum()
                if s < 1e-12:
                    a[i] = orig
                    break
                a_tmp = a * (S / s)
                m = norm_conv_max(a_tmp, S)
                if m < best_m:
                    best_m = m
                    best_a = a_tmp.copy()
                    a = a_tmp.copy()
                    improved = True
                else:
                    a[i] = orig

        if not improved and sweep > 0:
            break

    return best_a, best_m


def run_lbfgs_phase(f_init, n, S, time_budget):
    """L-BFGS on softmax objective with multiple temperatures."""
    a = np.maximum(f_init, 1e-10)
    a = normalize_to_sum(a, S)
    theta = np.log(a / a.sum() + 1e-30)

    best_a = a.copy()
    best_m = norm_conv_max(a, S)

    temps = [20, 50, 150, 400]
    per_temp_time = time_budget / (len(temps) + 1)

    for T in temps:
        if time.monotonic() > time_budget:
            break
        t_start = time.monotonic()
        t_budget = min(per_temp_time, time_budget - (t_start - 0))

        # Use a closure-free approach with a state dict
        state = {'n': n, 'T': T, 'S': S, 'start': t_start,
                 'budget': t_budget, 'best_m': best_m, 'best_a': best_a.copy()}

        try:
            result = minimize(
                _lbfgs_fn, theta, args=(state,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': 200, 'maxfun': 500, 'gtol': 1e-8}
            )
        except Exception:
            pass

        if state['best_m'] < best_m:
            best_m = state['best_m']
            best_a = state['best_a'].copy()
            theta = np.log(np.maximum(best_a, 1e-10) / best_a.sum() + 1e-30)

    return best_a, best_m


def _lbfgs_fn(theta, state):
    """Objective function for L-BFGS (top-level, no closure)."""
    elapsed = time.monotonic() - state['start']
    if elapsed > state['budget']:
        obj, grad = softmax_obj_grad(theta, state['n'], state['T'], state['S'])
        return obj, np.zeros_like(theta)

    obj, grad = softmax_obj_grad(theta, state['n'], state['T'], state['S'])

    # Track true objective
    n = state['n']
    S = state['S']
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total > 1e-300:
        a = S * exp_theta / total
        m_true = norm_conv_max(a, S)
        if m_true < state['best_m']:
            state['best_m'] = m_true
            state['best_a'] = a.copy()

    return obj, grad


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP-based direction finding, gradient annealing, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)

    global_best_obj = np.inf
    global_best_a = None

    # Try to load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_g = len(given_a)
            S_g = np.sqrt(2.0 * n_g)
            given_a = normalize_to_sum(given_a, S_g)
            obj_g = objective_value(given_a)
            print("Given: n=%d, obj=%.6f" % (n_g, obj_g))
            if np.isfinite(obj_g) and obj_g < global_best_obj:
                global_best_obj = obj_g
                global_best_a = given_a.copy()
    except Exception as e:
        print("No given sequence: %s" % str(e))

    remaining = total_deadline - time.monotonic()
    if remaining < 10.0:
        if global_best_a is not None:
            return global_best_a.tolist()
        return [1.0] * 256

    # === Phase 1: Exploration (25% of time) ===
    explore_budget = 0.25 * remaining
    print("Phase 1: Exploration, budget=%.0fs" % explore_budget)

    if given_a is not None:
        n_main = len(given_a)
        S_main = np.sqrt(2.0 * n_main)
    else:
        n_main = 4096
        S_main = np.sqrt(2.0 * n_main)

    lengths_to_try = [n_main]
    if n_main != 2048:
        lengths_to_try.append(2048)
    if n_main != 8192:
        lengths_to_try.append(8192)

    init_names = ["cosine_edge", "gaussian", "half_cos", "quad"]
    per_attempt = explore_budget / (len(lengths_to_try) * len(init_names))

    for n_try in lengths_to_try:
        if time.monotonic() > total_deadline - 60.0:
            break
        S_try = np.sqrt(2.0 * float(n_try))

        for init_name in init_names:
            if time.monotonic() > total_deadline - 60.0:
                break
            t_budget = min(per_attempt, 15.0)
            if t_budget < 2.0:
                continue

            a_init = get_initialization(init_name, n_try, rng, S_try)

            # Quick Adam anneal
            a_best, m_best = run_adam_anneal(a_init, n_try, S_try, t_budget,
                                              T_start=5.0, T_end=200.0, lr_init=0.05)
            obj_best = 2.0 * n_try * m_best / (S_try * S_try)  # = m_best since S=sqrt(2n)
            if obj_best < global_best_obj:
                global_best_obj = obj_best
                global_best_a = a_best.copy()
                print("  Explore n=%d %s: obj=%.6f" % (n_try, init_name, obj_best))

    # === Phase 2: LP-based refinement (35% of time) ===
    remaining = total_deadline - time.monotonic()
    lp_budget = 0.35 * remaining
    print("Phase 2: LP refinement, budget=%.0fs" % lp_budget)

    if global_best_a is not None and lp_budget > 10.0:
        n = len(global_best_a)
        S = np.sqrt(2.0 * float(n))
        a = normalize_to_sum(global_best_a, S)

        # Run LP phase
        a_lp, m_lp = run_lp_phase(a, S, n, lp_budget * 0.7)
        obj_lp = 2.0 * n * m_lp / (S * S)
        if obj_lp < global_best_obj:
            global_best_obj = obj_lp
            global_best_a = a_lp.copy()
            print("  LP phase: obj=%.6f" % obj_lp)

        # Follow up with gradient polish on LP result
        rem = total_deadline - time.monotonic()
        if rem > 5.0:
            a_grad, m_grad = run_adam_anneal(a_lp, n, S, min(rem * 0.3, 30.0),
                                              T_start=50.0, T_end=500.0, lr_init=0.03)
            obj_grad = 2.0 * n * m_grad / (S * S)
            if obj_grad < global_best_obj:
                global_best_obj = obj_grad
                global_best_a = a_grad.copy()
                print("  Post-LP gradient: obj=%.6f" % obj_grad)

    # === Phase 3: Gradient polishing (30% of time) ===
    remaining = total_deadline - time.monotonic()
    grad_budget = 0.30 * remaining
    print("Phase 3: Gradient polish, budget=%.0fs" % grad_budget)

    if global_best_a is not None and grad_budget > 5.0:
        n = len(global_best_a)
        S = np.sqrt(2.0 * float(n))
        a = normalize_to_sum(global_best_a, S)

        # L-BFGS
        lbfgs_time = grad_budget * 0.4
        if lbfgs_time > 5.0:
            a_lb, m_lb = run_lbfgs_phase(a, n, S, lbfgs_time)
            obj_lb = 2.0 * n * m_lb / (S * S)
            if obj_lb < global_best_obj:
                global_best_obj = obj_lb
                global_best_a = a_lb.copy()
                print("  L-BFGS: obj=%.6f" % obj_lb)

        # High-T Adam (top-k subgradient)
        rem = total_deadline - time.monotonic()
        highT_time = min(grad_budget * 0.3, rem * 0.5)
        if highT_time > 5.0 and global_best_a is not None:
            a = normalize_to_sum(global_best_a, S)
            a_hi, m_hi = run_adam_highT(a, n, S, highT_time, T=2000.0, lr_init=0.02, top_k=16)
            obj_hi = 2.0 * n * m_hi / (S * S)
            if obj_hi < global_best_obj:
                global_best_obj = obj_hi
                global_best_a = a_hi.copy()
                print("  High-T Adam: obj=%.6f" % obj_hi)

        # Another LP round
        rem = total_deadline - time.monotonic()
        if rem > 15.0 and global_best_a is not None:
            a = normalize_to_sum(global_best_a, S)
            a_lp2, m_lp2 = run_lp_phase(a, S, n, rem * 0.5, max_iters=50)
            obj_lp2 = 2.0 * n * m_lp2 / (S * S)
            if obj_lp2 < global_best_obj:
                global_best_obj = obj_lp2
                global_best_a = a_lp2.copy()
                print("  LP round 2: obj=%.6f" % obj_lp2)

    # === Phase 4: Coordinate descent polish (10% of time) ===
    remaining = total_deadline - time.monotonic()
    cd_budget = min(0.10 * (total_deadline - time.monotonic() - 30.0), max(10.0, remaining * 0.5))
    print("Phase 4: Coord descent, budget=%.0fs" % cd_budget)

    if global_best_a is not None and cd_budget > 3.0:
        n = len(global_best_a)
        S = np.sqrt(2.0 * float(n))
        a = normalize_to_sum(global_best_a, S)
        a_cd, m_cd = coordinate_descent(a, n, S, cd_budget)
        obj_cd = 2.0 * n * m_cd / (S * S)
        if obj_cd < global_best_obj:
            global_best_obj = obj_cd
            global_best_a = a_cd.copy()
            print("  Coord descent: obj=%.6f" % obj_cd)

        # Final tiny Adam polish
        rem = total_deadline - time.monotonic() - 5.0
        if rem > 3.0 and global_best_a is not None:
            a = normalize_to_sum(global_best_a, S)
            a_final, m_final = run_adam_highT(a, n, S, rem, T=5000.0, lr_init=0.01, top_k=32)
            obj_final = 2.0 * n * m_final / (S * S)
            if obj_final < global_best_obj:
                global_best_obj = obj_final
                global_best_a = a_final.copy()
                print("  Final polish: obj=%.6f" % obj_final)

    # === Final verification ===
    if global_best_a is None:
        global_best_a = np.ones(256)

    global_best_a = np.clip(np.asarray(global_best_a, dtype=np.float64), 0.0, 1000.0)
    if global_best_a.sum() < 0.01:
        global_best_a = np.ones(256)

    final_list = global_best_a.tolist()
    ver = evaluate_sequence(final_list)
    print("FINAL: computed=%.6f, verified=%s" % (global_best_obj,
          "%.6f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
        print("Fallback: obj=%.6f" % ver)

    return final_list
