"""
Optimizes non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) L-BFGS with log-sum-exp smoothing, (2) LP-based iteration, (3) random restarts.

import numpy as np
import time
import scipy.optimize as sp_opt
from scipy.fft import fft, ifft, rfft, irfft

def fft_convolve_auto(a):
    """Full auto-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    result = irfft(fa * fa, size)[:2*n - 1]
    return result

def score_and_grad_smooth(a, beta=20.0):
    """Compute smooth score and gradient using log-sum-exp."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, np.zeros(n)
    
    c = fft_convolve_auto(a)
    
    # Log-sum-exp of beta * c
    max_bc = beta * np.max(c)
    exp_terms = np.exp(beta * c - max_bc)
    sum_exp = np.sum(exp_terms)
    lse = (max_bc + np.log(sum_exp)) / beta  # smooth max
    score = 2.0 * n * lse / (S * S)
    
    # Weights
    w = exp_terms / sum_exp  # normalized, sums to 1
    
    # sm_grad[i] = 2 * sum_k w[k] * a[k-i]
    # Compute via cross-correlation using FFT
    size = 1
    while size < (2*n-1) + n - 1:
        size <<= 1
    W = fft(w, size)
    A = fft(a, size)
    cross = ifft(np.conj(W) * A, size).real[:n]
    sm_grad = 2.0 * cross
    
    # Gradient of score
    grad = (2.0 * n / S**3) * (sm_grad * S - lse)
    
    return score, grad

def lbfgs_optimize(a_init, time_limit, rng):
    """Optimize using L-BFGS with log-sum-exp smoothing."""
    current = a_init.astype(np.float64).copy()
    n = len(current)
    
    best = current.copy()
    best_score = score_and_grad_smooth(current, beta=20.0)[0]
    
    t0 = time.monotonic()
    
    # Use scipy L-BFGS-B with bounds
    from scipy.optimize import minimize
    
    # We'll do multiple rounds with increasing beta
    betas = [5, 10, 20, 50, 100]
    
    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        
        def obj(x):
            s, g = score_and_grad_smooth(x, beta=beta)
            return s, g
        
        res = minimize(obj, current, method='L-BFGS-B', jac=True,
                       bounds=[(0, 1000)] * n,
                       options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-8})
        
        current = np.clip(res.x, 0.0, 1000.0)
        current_score, _ = score_and_grad_smooth(current, beta=beta)
        # Use the "true" score (max, not smooth max) for comparison
        true_score = 2.0 * n * np.max(fft_convolve_auto(current)) / np.sum(current)**2
        
        if true_score < best_score:
            best_score = true_score
            best = current.copy()
        
        print(f"      L-BFGS beta={beta}: score={current_score:.8f}, true_score={true_score:.8f}, best={best_score:.8f}")
    
    return best, best_score
