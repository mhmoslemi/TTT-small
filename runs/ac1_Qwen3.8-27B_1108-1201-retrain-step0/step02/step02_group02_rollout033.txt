def make_structured_init(n, rng, method='broadband'):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)
    
    if method == 'broadband':
        a = np.ones(n)
        for k in range(1, 20):
            freq = k * (1 + 0.5 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.2 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

"""
Hybrid L-BFGS optimization for minimizing the convolution peak score of non-negative
step functions. Uses log-space parameterization, adaptive soft-max smoothing for the
non-smooth max operator, multi-temperature annealing, peak-directed mutations,
and fast coordinate descent. Exploits FFT-based gradient computation for efficiency.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def smooth_score_grad(a, tau):
    """
    Compute smooth (soft-max) approximation of the score and its gradient.
    score ≈ 2n * soft_max(c) / S^2 where c = convolve(a,a).
    Returns (score, gradient, convolution, weights, peak_index).
    """
    n = len(a)
    S = np.sum(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    peak_idx = int(np.argmax(c))

    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    soft_max = c_max + tau * np.log(sum_exp)

    score = 2.0 * n * soft_max / (S * S)

    # Gradient: g_j = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # Computed via FFT: corr[j] = convolve(w, a_rev)[n-1+j]
    v = a[::-1]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    fw = np.fft.rfft(w, size)
    fv = np.fft.rfft(v, size)
    conv_wv = np.fft.irfft(fw * fv, size)
    corr = conv_wv[n - 1:n - 1 + n]

    grad_softmax = 2.0 * corr
    grad_score = (2.0 * n / (S * S)) * (grad_softmax - soft_max / S)

    return score, grad_score, c, w, peak_idx


def exact_subgrad_score(a):
    """
    Compute exact score and subgradient at the peak.
    Returns (score, subgradient, peak_value, peak_index).
    """
    n = len(a)
    S = np.sum(a)
    c = fft_convolve(a, a)
    peak_idx = int(np.argmax(c))
    peak_val = c[peak_idx]
    score = 2.0 * n * peak_val / (S * S)

    # Subgradient: for the peak index k*, dc_{k*}/da_j = 2*a_{k*-j} if 0<=k*-j<n else 0
    g = np.full(n, -peak_val / S)  # constant part from -soft_max/S
    for j in range(n):
        m = peak_idx - j
        if 0 <= m < n:
            g[j] = 2.0 * a[m] - peak_val / S
    grad_score = (2.0 * n / (S * S)) * g

    return score, grad_score, peak_val, peak_idx


def optimize_l_bfgs(a_init, tau, time_limit, max_iter=2000):
    """
    Run L-BFGS-B in log-space to minimize the soft-max smoothed score.
    """
    n = len(a_init)
    theta0 = np.log(np.maximum(a_init, 1e-10))

    def objective(theta):
        a = np.exp(theta)
        a = np.minimum(a, 1000.0)
        s, g, _, _, _ = smooth_score_grad(a, tau)
        if np.isinf(s) or np.isnan(s):
            return 1e10, np.zeros(n)
        return s, g

    result = minimize(
        objective, theta0, method='L-BFGS-B',
        jac=True,
        options={'maxiter': max_iter, 'maxfun': 50000,
                 'ftol': 1e-14, 'gtol': 1e-10, 'gtol_scalar': 1e-10}
    )

    a_result = np.exp(result.x)
    a_result = np.clip(a_result, 0, 1000)
    score_result = local_score(a_result)
    return a_result, score_result


def optimize_subgrad(a_init, time_limit, max_iter=500):
    """
    L-BFGS-B using exact subgradient (peak-only) for final convergence.
    """
    n = len(a_init)
    theta0 = np.log(np.maximum(a_init, 1e-10))

    def objective(theta):
        a = np.exp(theta)
        a = np.minimum(a, 1000.0)
        s, g, _, _ = exact_subgrad_score(a)
        if np.isinf(s) or np.isnan(s):
            return 1e10, np.zeros(n)
        return s, g

    result = minimize(
        objective, theta0, method='L-BFGS-B',
        jac=True,
        options={'maxiter': max_iter, 'maxfun': 20000,
                 'ftol': 1e-14, 'gtol': 1e-8}
    )

    a_result = np.exp(result.x)
    a_result = np.clip(a_result, 0, 1000)
    score_result = local_score(a_result)
    return a_result, score_result


def make_structured_init(n, rng, method='broadband'):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)

    if method == 'broadband':
        a = np.ones(n)
        for k in range(1, 25):
            freq = k * (0.8 + 0.4 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.25 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    elif method == 'fourier':
        # Random phase spectrum -> take real part + DC offset
        spectrum = np.zeros(n, dtype=np.complex128)
        spectrum[0] = 1.0  # DC
        for k in range(1, n // 2 + 1):
            amp = 1.0 / np.sqrt(k + 1)
            phase = rng.uniform(0, 2 * np.pi)
            spectrum[k] = amp * np.exp(1j * phase)
            spectrum[n - k] = np.conj(spectrum[k])
        a = np.fft.irfft(spectrum, n) + 0.5
        return np.clip(a, 0.01, None)

    elif method == 'm_seq':
        # Pseudo-random binary-like sequence
        a = np.ones(n)
        a[rng.random(n) < 0.3] = 0.2
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)

    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)


def peak_directed_refine(a_init, time_limit, rng):
    """
    Peak-directed refinement: identify peak contributors and perform
    targeted mass redistribution with L-BFGS micro-steps.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    step = 0
    no_improve = 0
    patience = 200

    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]

        # Compute contribution of each element to the peak
        contribs = np.zeros(n)
        for j in range(n):
            m = peak_idx - j
            if 0 <= m < n:
                contribs[j] = 2.0 * current[j] * current[m]

        r = rng.random()

        if r < 0.35:
            # Reduce top contributors, redistribute to low contributors
            top_k = max(4, n // 12)
            bottom_k = max(4, n // 8)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            bottom_idx = np.argsort(contribs)[:bottom_k]

            reduction = rng.uniform(0.01, 0.04)
            total_red = 0.0
            for idx in top_idx:
                amt = current[idx] * reduction
                total_red += amt
                current[idx] -= amt
            # Distribute to bottom
            for idx in bottom_idx:
                current[idx] += total_red * (0.5 + 0.5 * rng.random())
            current = np.clip(current, 1e-6, 1000)

        elif r < 0.55:
            # Micro L-BFGS step with very low temperature
            tau_micro = max(n / 2000.0, 0.01)
            max_iter = max(50, int(min(10.0, time_limit / 20) * 30))
            a_res, s_res = optimize_l_bfgs(current, tau_micro, 10.0, max_iter=max_iter)
            if s_res < current_score:
                current = a_res
                current_score = s_res
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
                    no_improve = 0
            continue

        elif r < 0.75:
            # Pair swap between top and bottom contributors
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            j_red = top_idx[int(rng.integers(0, top_k))]
            # Choose j_inc that doesn't contribute much to the peak
            low_contrib = np.argsort(contribs)[:n // 6]
            j_inc = low_contrib[int(rng.integers(0, len(low_contrib)))]
            if j_inc != j_red:
                amt = current[j_red] * rng.uniform(0.02, 0.06)
                current[j_red] -= amt
                current[j_inc] += amt
                current = np.clip(current, 1e-6, 1000)

        elif r < 0.9:
            # Block scaling
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.04))
            current[start:start + width] *= factor
            current = np.clip(current, 1e-6, 1000)

        else:
            # Small global perturbation
            current *= np.exp(rng.normal(0, 0.008, size=n))
            current = np.clip(current, 1e-6, 1000)

        new_score = local_score(current)
        if new_score < current_score:
            current_score = new_score
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            no_improve += 1
            # Revert if significantly worse
            if new_score > current_score * 1.02:
                current = best.copy()
                current_score = best_score

        # Escape mechanism
        if no_improve > patience:
            r_esc = rng.random()
            if r_esc < 0.5:
                # Gaussian kick in log-space
                theta = np.log(np.maximum(current, 1e-10))
                theta += 0.2 * rng.standard_normal(n)
                current = np.clip(np.exp(theta), 1e-6, 1000)
            else:
                # Restructure: partially reset
                frac = rng.uniform(0.1, 0.4)
                n_reset = max(1, int(frac * n))
                idx = rng.choice(n, n_reset, replace=False)
                current[idx] = np.clip(rng.uniform(0.1, 3.0, n_reset), 1e-6, 1000)
            no_improve = 0

        if step % 500 == 0:
            print(f"    PD step={step}, cur={current_score:.6f}, best={best_score:.6f}")

    return best, best_score


def coordinate_descent_fast(a_init, time_limit, rng):
    """
    Fast coordinate descent: for each coordinate, use gradient-informed
    1D line search (geometric) to find the best value.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    iteration = 0
    order = rng.permutation(n)

    while time.monotonic() - t0 < time_limit:
        improved_this_round = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            base_val = current[j]
            if base_val < 1e-8:
                continue

            # Try geometric sequence of multipliers
            best_local = base_val
            best_local_score = current_score

            for frac in [0.6, 0.75, 0.9, 1.1, 1.3, 1.5]:
                trial = base_val * frac
                current[j] = trial
                new_score = local_score(current)

                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = trial
                else:
                    current[j] = base_val
                    break

            if best_local != base_val:
                current[j] = best_local
                current_score = best_local_score
                improved_this_round = True
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()

        if not improved_this_round:
            # Try random perturbation to escape
            for _ in range(3):
                j = int(rng.integers(0, n))
                current[j] = max(1e-6, current[j] * np.exp(rng.normal(0, 0.3)))
                current = np.clip(current, 1e-6, 1000)
                new_score = local_score(current)
                if new_score < current_score:
                    current_score = new_score
                    improved_this_round = True
                    if current_score < best_score:
                        best_score = current_score
                        best = current.copy()
                    break
            if not improved_this_round:
                # Larger kick
                j = int(rng.integers(0, n))
                current[j] = max(1e-6, rng.uniform(0.1, 5.0))
                current = np.clip(current, 1e-6, 1000)
                current_score = local_score(current)

        iteration += 1
        if iteration % 10 == 0:
            print(f"    CD iter={iteration}, cur={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")

    return best, best_score
