import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(w_sorted)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w_sorted
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    # In signal processing: (w ⋆ a)[i] = sum_m w[m] * conj(a[m-i])
    # For real signals: (w ⋆ a)[i] = sum_m w[m] * a[m-i]

def hankel_quad(w, a):
    """Compute a^T W a where W[i,j] = w[i+j]."""
    return np.dot(a, hankel_mv(w, a))


def fft_grad_smooth(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def solve_qp_fista(a_init, w, n, S, n_iter=100):
    """
    Solve min a^T W a subject to a >= 0, sum(a) = S.
    W is the Hankel matrix with W[i,j] = w[i+j].
    Uses FISTA with backtracking.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    step = 0.5  # 1/L where L=2 (Lipschitz constant of gradient)
    
    for _ in range(n_iter):
        # Compute gradient: 2 * W @ y
        grad = 2.0 * hankel_mv(w, y)
        
        # Backtracking line search
        x_new = project_simplex(y - step * grad, S)
        
        # Check if objective decreased (optional, for line search)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return a


def ao_optimize(a_init, n, S, max_time, rng, w_temperature=1.0):
    """
    Alternating Optimization:
    - a-update: FISTA on QP min a^T W a s.t. a in C
    - w-update: w[k] = exp(c[k]/T) / sum(exp(c[j]/T))
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Compute current convolution
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        
        # w-update: temperature-based
        T = w_temperature * (1.0 + 0.01 * it)
        c_shifted = c - c.max()
        w = np.exp(c_shifted / T)
        w = w / w.sum()
        
        # a-update: FISTA on QP
        n_fista_iter = min(200, max(20, 200000 // max(1, it)))
        a = solve_qp_fista(a, w, n, S, n_iter=n_fista_iter)
        
        # Occasionally add perturbation
        if it % 50 == 0 and rng.random() < 0.1:
            a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
    
    return best_a, best_score


def fista_smooth(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=300.0):
    """FISTA with smooth max (log-sum-exp)."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 1.0 / (beta_init * 2.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.003 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        grad = fft_grad_smooth(y, c, beta)
        x_new = project_simplex(y - step * grad, S)
        
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        step = 1.0 / (beta * 2.0)
        step = max(step, 1e-8)
    
    return best_local, best_score


def safe_normalize(a, n):
