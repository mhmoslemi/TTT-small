def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def fft_phase_init(n, rng, n_iter=100):
    """Gerchberg-Saxton: flatten magnitude spectrum."""
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft <<= 1
    
    a = np.ones(n) / n
    for _ in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a

def chirp_init(n, rng, alpha=0.5):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2 * np.pi * t * t / (2 * n))
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)

def golay_like_init(n, rng):
    """Initialize based on Golay sequence pattern (balanced)."""
    # Generate a random balanced sequence
    a = rng.choice([0.3, 1.0, 1.7], size=n)
    # Smooth it
    kernel = np.ones(5) / 5
    a = np.convolve(a, kernel, mode='same')
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)

def random_structured_init(n, rng):
    """Random init with some structure (low-freq bias)."""
    a = np.ones(n)
    n_components = rng.integers(2, 6)
    for _ in range(n_components):
        freq = rng.uniform(1, n // 4)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.2, 0.8)
        t = np.arange(n)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)

def gradient_optimize(a_init, time_limit, rng, verbose=False):
    """Adam gradient descent with annealed soft-max temperature."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init / np.sum(a_init) * S
    theta = np.log(a / S + 1e-12)
    
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr0 = 3.0
    t0 = time.monotonic()
    step = 0
    best_score = score_of(a)
    best_theta = theta.copy()
    
    tau_start = 50.0 / n
    tau_end = 0.2 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed ** 0.5)
        
        a_curr = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a_curr, tau)
        
        # Normalize gradient: project out the direction that changes the sum
        inner = np.dot(a_curr, g_a)
        g_theta = a_curr * (g_a - inner / S)
        
        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        
        # Adaptive learning rate
        lr = lr0 / np.sqrt(1.0 + 0.3 * step)
        
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        if step % 500 == 0:
            a_check = softmax_normalize(theta) * S
            s = score_of(a_check)
            if s < best_score:
                best_score = s
                best_theta = theta.copy()
            if verbose:
                print(f"      step={step}, score={s:.6f}, best={best_score:.6f}, tau={tau:.2e}")
    
    a_final = softmax_normalize(best_theta) * S
    return a_final, best_score

    #   So conv[2i] = x^2 + sum_{j≠i} a[j]*a[2i-j] (where 2i-j is valid and ≠ i... wait)

    #   
    #   When we vary a[i] = x:
    #   The terms involving x are:

    #       C_k = sum_{j≠i, valid} a[j]*a[k-j] (terms not involving a[i])

    def objective_1d(x, i):
        a_new = a.copy()
        a_new[i] = max(0, x)
        c = fft_convolve(a_new, a_new)

    #   if k-i is in [0, n-1] and k-i != i:

    # The new score depends on both max_b and S_new.

    # This means: a_new = (a with a[i] changed to x) / S_new * S_old
    # which is equivalent to: a_new[j] = a[j] for j≠i, a_new[i] = x

    sweep_count = 0
    while time.monotonic() - t0 < time_limit:

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            a_i = a[i]
            S = np.sum(a)
            
            def obj(x):
                if x < 0:
                    return np.inf
                a[i] = x
                S_new = S - a_i + x
                if S_new < 1e-10:
                    return np.inf
                a_norm = a / S_new * S  # normalize back
                c = fft_convolve(a_norm, a_norm)
                return np.max(c) / S_new  # proportional to score
                # Actually score = 2*n*max_c/S_new^2 * S^2 (since we renormalize)
                # = 2*n*max_c * S^2 / S_new^2
                # For 1D minimization, S is constant, so we minimize max_c / S_new^2 * S^2
                # = max_c / (S_new/S)^2
                # Hmm this is getting complicated. Let me just use the actual score.
            
            def obj_score(x):
                if x < 0:
                    return np.inf
                a[i] = x
                S_new = np.sum(a)
                if S_new < 1e-10:
                    return np.inf
                c = fft_convolve(a, a)
                return 2.0 * n * np.max(c) / (S_new ** 2)
            
            # Use golden section or minimize_scalar
            x_lo = 0.0
            x_hi = max(2.0 * a_i, 0.1) if a_i > 0 else 0.5
            
            # First check current value
            a[i] = a_i  # restore
            current_score = obj_score(a_i)
            
            result = minimize_scalar(obj_score, bounds=(x_lo, x_hi), method='bounded', 
                                     options={'xatol': 1e-8 * max(a_i, 0.01)})
            
            a[i] = a_i  # restore before applying
            if result.fun < current_score - 1e-10:
                a[i] = max(0, result.x)
                new_score = obj_score(a[i])
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
        
        sweep_count += 1
        if sweep_count >= n_sweeps:
            break
    
    return best_a, best_score
