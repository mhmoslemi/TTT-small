Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Eta-space (log-space) parameterization for automatic constraint handling
2. Adaptive temperature soft-max gradient
3. Multi-phase search: exploration -> exploitation -> polish
4. Structured initializations exploiting anti-symmetric bias
5. Block-wise coordinate descent for final refinement

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_h(w, a):
    """Compute h[j] = sum_k w[k] * a[k-j] for j=0..n-1."""
    n = len(a)
    a_rev = a[::-1]
    out_len = (2*n - 1) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    Ar = fft(a_rev, fsize)
    conv = np.real(ifft(W * Ar))
    return conv[n-1:2*n-1]


def theta_to_a(theta, S):
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def objective_value(a):
    """Compute the true objective value."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

            best_obj = obj
            best_a = a_opt.copy()
            print("  Phase1: n=%d init=%s obj=%.6f" % (n, init_name, obj))
    
    return best_a, best_obj


def make_initialization(n, name, alpha, rng, S):
    i = np.arange(n, dtype=np.float64)
    if name == "uniform":
        a = np.ones(n)
    elif name == "ramp":
        a = 1.0 + alpha * (2.0 * i / (n-1) - 1.0)
    elif name == "sine3":
        a = 1.0 + alpha * np.sin(2.0 * np.pi * 3 * i / n)
    elif name == "sine5":
        a = 1.0 + alpha * np.sin(2.0 * np.pi * 5 * i / n)
    elif name == "cos2":
        a = 1.0 + alpha * np.cos(2.0 * np.pi * 2 * i / n)
    elif name == "rand":
        a = 1.0 + alpha * (rng.random(n) - 0.5)
    elif name == "triangle":
        a = 1.0 + alpha * np.abs(2.0 * i / (n-1) - 1.0)
    elif name == "two_ramp":
        a = np.ones(n)
        half = n // 2
        a[:half] = 1.0 + alpha * (i[:half] / half)
        a[half:] = 1.0 - alpha * (i[half:] - half) / (n - half)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def quick_optimize(a, n, time_budget):
    """Quick gradient descent to find a good local optimum."""

    a = a.max(1e-10)

    m = np.zeros(n)

        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
