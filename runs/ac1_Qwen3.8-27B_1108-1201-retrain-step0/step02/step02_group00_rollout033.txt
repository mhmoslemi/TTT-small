One more idea: **coordinate descent**. Since the subgradient at the max lag is a shifted copy of a, we can try coordinate descent:
- Pick the lag k* with the max convolution value.
- For each j, the contribution to b[k*] from a[j] is a[j]*a[k*-j].
- To reduce b[k*], we can reduce a[j] or a[k*-j].
- The optimal coordinate update (for a single coordinate) would be to set a[j] = 0 if it doesn't increase other lag values.

Here's another idea: **alternating minimization**. Split a into "even" and "odd" indexed elements (or first half and second half). Minimize w.r.t. one set while fixing the other. This is like a "ping-pong" approach.

For example:
- Fix a[1], a[3], a[5], ... and optimize a[0], a[2], a[4], ...
- Then fix a[0], a[2], a[4], ... and optimize a[1], a[3], a[5], ...

The optimization w.r.t. one set is a convex problem (since the convolution is linear in each set when the other is fixed). We could solve it with a quadratic program or a simpler method.

Actually, when we fix the "odd" elements and optimize the "even" elements, the convolution becomes:
b[k] = sum_j a[j]*a[k-j]

The terms where both j and k-j are even contribute to the "even-even" part.
The terms where one is even and one is odd contribute to the "even-odd" part (linear in the even elements).
The terms where both are odd contribute to the "odd-odd" part (constant).

So b[k] = "odd-odd"(k) + "even-odd"(k; a_even) + "even-even"(k; a_even)

The "even-even" part is quadratic in a_even, making the problem convex but not linear.

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy:
- Multiple initializations: given sequence, cosine modulations, linear tilts, random.
- Multiple lengths: 4096, 8192, 16384.
- Phase 1: Fast subgradient with momentum for broad exploration.
- Phase 2: FISTA with adaptive log-sum-exp for smooth optimization.
- Phase 3: Hybrid polish alternating subgradient and FISTA.
- Direct spike targeting: explicitly reduce elements contributing to max spikes.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, c, top_k=1):
    """Compute subgradient of max conv: sum of top-k shifted copies of a."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    grad = np.zeros(n)
    for k, w in zip(top_indices, weights):
        k = int(k)
        # b[k] = sum_j a[j]*a[k-j] where j in [max(0,k-n+1), min(n-1,k)]
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        # gradient w.r.t. a[j] is a[k-j]
        js = np.arange(j_lo, j_hi + 1)
        grad[js] += w * a[k - js]
    return 2.0 * grad


def run_subgradient_momentum(a_init, n, S, max_time, rng, top_k=5):
    """Subgradient method with heavy ball momentum."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    v = np.zeros(n)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        grad = compute_subgradient(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (gnorm * np.sqrt(1.0 + 0.1 * it))
        v = 0.9 * v - step * grad
        a_new = project_simplex(a + v, S)

        if it % 10 == 0:
            c_check = fft_convolve_self(a_new)
            if c_check.max() < best_score:
                best_score = c_check.max()
                best_local = a_new.copy()

        a = a_new

        if it % 5000 == 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            v = np.zeros(n)

    return best_local, best_score


def run_fista_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=400.0):
    """FISTA with adaptive log-sum-exp approximation to max."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(0.003 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Log-sum-exp gradient
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            w = np.zeros_like(c)
            w[c.argmax()] = 1.0
        else:
            w = exp_bc / total

        # Gradient: 2 * cross-correlation of w with a
        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta * 2.0 * (1.0 + 0.001 * it))
        step = max(step, 1e-8)
        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 2000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def run_hybrid_polish(a_init, n, S, max_time, rng):
    """Alternate between subgradient (top-k) and FISTA (high beta)."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    # Phase A: subgradient with increasing top_k
    while time.monotonic() - start < max_time * 0.4:
        it += 1
        top_k = min(64, 3 + it // 50)
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        grad = compute_subgradient(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 2.0 * S / (gnorm * np.sqrt(1.0 + 0.05 * it))
        a = project_simplex(a - step * grad, S)

    # Phase B: FISTA with high beta
    remaining = max_time - (time.monotonic() - start)
    if remaining > 1:
        a2, s2 = run_fista_lse(a, n, S, remaining, rng, beta_init=50.0, beta_max=600.0)
        c2 = fft_convolve_self(a2)
        if c2.max() < best_score:
            best_score = c2.max()
            best_local = a2.copy()

    return best_local, best_score


def spike_targeting(a, n, S, n_steps=1000):
    """Directly target the largest spikes by reducing contributing elements."""
    best_a = a.copy()
    best_score = np.inf
    for step in range(n_steps):
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # Find the top few lags
        top_k = 5
        top_indices = np.argsort(c)[::-1][:top_k]

        # For each top lag, find the elements contributing most and reduce them
        for k in top_indices:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            js = np.arange(j_lo, j_hi + 1)
            contributions = a[js] * a[k - js]
            # Reduce the top contributors
            top_contrib = np.argsort(contributions)[::-1][:min(20, len(contributions))]
            for j_idx in top_contrib:
                j = js[j_idx]
                other = k - j
                # Reduce both a[j] and a[other] proportionally
                reduce_factor = 0.01
                a[j] = max(0, a[j] * (1 - reduce_factor))
                if other != j and 0 <= other < n:
                    a[other] = max(0, a[other] * (1 - reduce_factor))

        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a * (S / s)
        else:
            a = best_a.copy()
            break

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
```
