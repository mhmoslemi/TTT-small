Optimization of non-negative sequences to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses gradient-based methods (soft-max + p-norm objectives with L-BFGS),

import numpy as np

from scipy.optimize import minimize

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def local_score(p):
    """Score for probability vector p: 2n * max(conv(p,p))."""
    n = len(p)
    c = fft_convolve(p, p)
    return 2.0 * n * np.max(c)

def make_initializations(n, rng):
    """Generate diverse initial probability vectors."""
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Ramp up
    t = np.linspace(0, 1, n)
    inits['ramp_up'] = (t + 0.01)
    inits['ramp_up'] = inits['ramp_up'] / np.sum(inits['ramp_up'])
    
    # Ramp down
    inits['ramp_down'] = inits['ramp_up'][::-1].copy()
    
    # Gaussian
    center = (n - 1) / 2.0
    sigma = n * 0.2
    t = np.arange(n)
    g = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    inits['gaussian'] = g / np.sum(g)
    
    # Wide Gaussian (closer to uniform)
    sigma = n * 0.4
    g = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    inits['wide_gauss'] = g / np.sum(g)
    
    # V-shape (large at edges, small in middle)
    v = 1.0 - np.abs(t - center) / center
    v = np.clip(v, 0.01, None)
    inits['v_shape'] = (1.0 / v)  # inverted V = U-shape
    inits['u_shape'] = inits['u_shape'] / np.sum(inits['u_shape'])
    
    # Random
    r = rng.random(n) + 0.01
    inits['random'] = r / np.sum(r)
    
    # Bimodal
    center1 = n * 0.25
    center2 = n * 0.75
    sigma2 = n * 0.1
    b = np.exp(-0.5 * ((t - center1) / sigma2) ** 2) + np.exp(-0.5 * ((t - center2) / sigma2) ** 2)
    inits['bimodal'] = b / np.sum(b)
    
    # Chirp-like
    freq = 0.5 + rng.random()
    c = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
    c = np.clip(c, 0.01, None)
    inits['chirp'] = c / np.sum(c)
    
    return inits

def polish_peak_reduction(p_init, time_limit, rng):
    """
    Local search with peak-targeted moves for the convolution.
    """
    current = p_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.001
    cooling = 0.9999
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.40:
            # Targeted peak reduction: reduce values contributing to the peak
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            # Find which pairs contribute to this peak
            # c[peak_idx] = sum_i a[i]*a[peak_idx-i]
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] += current[i] * current[j]
            # Reduce the top contributors
            top_k = max(2, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            # Increase others slightly
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(8, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.60:
            # Spread mass from high-convolution regions
            c = fft_convolve(current, current)
            c_max = np.max(c)
            c_min = np.min(c)
            # Compute "pressure" at each position
            pull = np.zeros(n)
            for i in range(n):
                for k in range(max(0, i - (n-1)), min(2*n-1, i + n)):
                    j = k - i
                    if 0 <= j < n:
                        weight = max(0, c[k] - c_max * 0.8)
                        pull[i] += weight * current[j]
            if np.max(pull) > 1e-15:
                pull = pull / np.max(pull)
                candidate *= np.exp(-0.03 * pull)
        elif r < 0.75:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))
        elif r < 0.85:
            # Partial symmetrization (anti-symmetric for convolution flatness)
            # For convolution flatness, we want a[i]*a[n-1-i] to be small
            # So make a "anti-correlated" with its reverse
            anti = current - current[::-1]
            candidate = current + 0.1 * anti
        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.02, size=n))
        
        candidate = np.maximum(candidate, 1e-15)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score
        
        temperature = max(temperature * cooling, 1e-10)
    
    return best, best_score

def run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p, deadline):
    """Run L-BFGS with annealed p-norm."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))
    
    for p in p_schedule:
        if time.monotonic() > deadline - 10:
            break
        
        result = minimize(
            pnorm_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(p,),
            options={'maxiter': maxiter_per_p, 'ftol': 1e-14, 'gtol': 1e-12, 'maxfun': maxiter_per_p * 5}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)
        
        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        theta = best_theta
    
    return best_theta, best_score
