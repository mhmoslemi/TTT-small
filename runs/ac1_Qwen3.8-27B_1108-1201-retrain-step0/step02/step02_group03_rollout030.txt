"""
Optimized peak-autocorrelation minimizer for non-negative step functions.
Uses improved spectral initialization (adaptive GS), gradient descent with
annealed log-sum-exp, peak-directed local search, and multi-length search.
Key improvements: better gradient computation, adaptive temperature,
peak-deflation mutations, and progressive multi-scale refinement.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax_normalize(theta):
    """Stable softmax normalization."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def gs_init(n, rng, n_iter=300, soft_beta=0.0):
    """
    Improved Gerchberg-Saxton initialization with soft non-negativity.
    Uses a tempered approach for the clipping step.
    """
    N_fft = 1
    while N_fft < 2 * n:
        N_fft <<= 1

    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-15)
        phase = A / mag_safe

        # Target: flat magnitude spectrum
        target_mag = np.full_like(mag, np.mean(mag))
        # Blend: partially keep original magnitude to preserve structure
        blend = 0.7 + 0.3 * (it / n_iter)
        target_mag = blend * np.full_like(mag, np.mean(mag)) + (1 - blend) * mag

        A_new = target_mag * phase
        a_new = np.fft.ifft(A_new).real[:n]

        # Soft non-negativity: use tanh-based soft clip
        if soft_beta > 0:
            a_new = 0.5 * a_new * (1 + np.tanh(a_new / soft_beta))
        else:
            a_new = np.maximum(a_new, 0.0)

        s = np.sum(a_new)
        if s < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new
    return a


def gs_init_v2(n, rng, n_iter=200):
    """
    Alternative GS: random phase initialization with iterative projection.
    """
    N_fft = 1
    while N_fft < 2 * n:
        N_fft <<= 1

    # Random phase initialization
    A = np.exp(2j * np.pi * rng.random(N_fft))
    A[0] = np.abs(A[0])  # Keep DC real positive

    a = np.ones(n) / n
    for it in range(n_iter):
        # Project onto flat magnitude
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-15)
        phase = A / mag_safe
        A = np.full_like(mag, np.mean(mag)) * phase

        # Inverse FFT and project onto non-negative reals of length n
        a_full = np.fft.ifft(A).real
        a_new = np.maximum(a_full[:n], 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = np.ones(n) / n
        a_new = a_new / s

        # Forward FFT back
        a_pad = np.zeros(N_fft)
        a_pad[:n] = a_new
        A = np.fft.fft(a_pad)
        a = a_new

    return a


def chirp_init(n, rng, alpha=0.5, beta=0.3):
    """Chirp-like initialization with random parameters."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1.0))
    a += beta * np.sin(2 * np.pi * rng.integers(1, max(2, n // 2)) * t / n)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def multitone_init(n, rng, n_tones=5):
    """Multi-tone sinusoidal initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(n_tones):
        freq = rng.integers(1, max(2, n // 2))
        amp = 0.3 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def random_smooth_init(n, rng, scale=8.0):
    """Smooth random initialization via low-pass filtered noise."""
    a = np.ones(n) + scale * rng.standard_normal(n)
    # Low-pass filter
    A = np.fft.rfft(a)
    cutoff = max(1, int(np.sqrt(n)))
    A[cutoff:] *= 0.1
    a = np.fft.irfft(A, n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def optimize_gradient(n, a_init, time_limit, rng, tau_start_mult=60.0,
                      tau_end_mult=0.2, lr0=3.0):
    """
    Adam gradient descent on softmax-parameterized sequence.
    Minimizes logsumexp(conv(a,a)/tau) with annealed temperature.
    """
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a) * S
    theta = np.log(a / S)

    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8

    t0 = time.monotonic()
    step = 0
    best_score = local_score(a)
    best_theta = theta.copy()

    tau_start = tau_start_mult / n
    tau_end = tau_end_mult / n

    stall_count = 0
    last_improve_score = best_score
    no_improve_steps = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        # Adaptive temperature: faster cooling if improving, slower if stalled
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)

        a = softmax_normalize(theta) * S
        c = fft_convolve(a, a)

        # Soft-max weights for smooth max approximation
        c_max = c.max()
        shifted = (c - c_max) / tau
        shifted = np.clip(shifted, -50.0, 50.0)
        e = np.exp(shifted)
        w = e / np.sum(e)

        # Gradient: grad[j] = 2 * conv(a_rev, w)[n-1+j]
        a_rev = a[::-1]
        corr = fft_convolve(a_rev, w)
        grad = 2.0 * corr[n - 1:2 * n - 1] / tau

        # Gradient w.r.t. theta (softmax parameterization)
        inner = np.dot(a, grad)
        grad_theta = a * (grad - inner / S)

        # Adam update
        step += 1
        m1 = b1 * m1 + (1 - b1) * grad_theta
        m2 = b2 * m2 + (1 - b2) * (grad_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)

        lr = lr0 / np.sqrt(1.0 + 0.4 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # Periodic evaluation
        if step % 150 == 0:
            a_check = softmax_normalize(theta) * S
            score = local_score(a_check)
            if score < best_score - 1e-8:
                best_score = score
                best_theta = theta.copy()
                no_improve_steps = 0
            else:
                no_improve_steps += 150

            # Adaptive restart: if stuck for too long, add perturbation
            if no_improve_steps > 3000:
                theta += 0.5 * rng.standard_normal(n)
                no_improve_steps = 0

            if step % 1500 == 0:
                print(f"        step={step}, score={score:.6f}, best={best_score:.6f}, "
                      f"tau={tau:.2e}, t={elapsed:.1f}s")

    a_final = softmax_normalize(best_theta) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
    return a_final, best_score


def peak_deflation_search(a_init, time_limit, rng, top_k=6):
    """
    Local search that specifically targets the peak lags of the convolution.
    Uses contribution analysis to identify which elements to modify.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9995
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        iter_count += 1

        conv = fft_convolve(current, current)

        if r < 0.40:
            # Peak-directed: reduce elements contributing to top lags
            n_top = min(top_k, len(conv))
            top_lags = np.argsort(conv)[-n_top:]
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            # Reduce top contributors, increase bottom
            n_adj = max(2, n // 12)
            top_idx = np.argsort(contrib)[-n_adj:]
            bot_idx = np.argsort(contrib)[:n_adj]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.02, size=len(bot_idx)))

        elif r < 0.55:
            # Adjacent block transfer
            width = max(1, int(rng.integers(1, max(2, n // 25))))
            start = int(rng.integers(0, max(1, n - 2 * width)))
            dec = np.exp(rng.normal(-0.07, 0.02, size=width))
            inc = np.exp(rng.normal(0.07, 0.02, size=width))
            candidate[start:start + width] *= dec
            candidate[start + width:start + 2 * width] *= inc

        elif r < 0.70:
            # Multi-block perturbation
            n_blocks = int(rng.integers(2, 5))
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 20))))
                bs = int(rng.integers(0, max(1, n - bw)))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0.0, 0.04, size=bw))

        elif r < 0.80:
            # Symmetrization step
            sym = 0.5 * (current + current[::-1])
            alpha = 0.1 + 0.1 * rng.random()
            candidate = (1 - alpha) * current + alpha * sym
            candidate += 0.005 * rng.standard_normal(n)

        elif r < 0.90:
            # Spectral nudge: small random phase perturbation in frequency domain
            N_fft = 1
            while N_fft < 2 * n:
                N_fft <<= 1
            A = np.fft.fft(current, N_fft)
            mag = np.abs(A)
            mag_safe = np.maximum(mag, 1e-15)
            phase = A / mag_safe
            # Perturb phase slightly
            phase = phase * np.exp(1j * 0.05 * rng.standard_normal(N_fft))
            A_new = mag_safe * phase
            a_new = np.fft.ifft(A_new).real[:n]
            a_new = np.maximum(a_new, 0.0)
            s = np.sum(a_new)
            if s > 1e-10:
                candidate = a_new / s * np.sum(current)
            else:
                candidate = current + 0.01 * rng.standard_normal(n)

        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / s * np.sum(current)

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 20:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    current = candidate
                    current_score = new_score

        temp *= cooling
        if temp < 1e-6:
            temp = 0.001  # Reheat occasionally

    return best, best_score


def upsample(a, new_n, rng, noise_level=0.01):
    """Upsample a sequence to a longer length via interpolation + noise."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += noise_level * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)
