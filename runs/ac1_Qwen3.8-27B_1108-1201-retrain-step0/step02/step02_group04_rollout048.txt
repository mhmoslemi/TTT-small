def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def fft_cross_corr(a, b):

Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses smoothed gradient descent (log-sum-exp approximation of max) with simplex projection.
Multiple restarts with different initializations and sequence lengths.

import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def simplex_project(y, target_sum):
    """Project y onto the simplex {x >= 0, sum(x) = target_sum}."""
    n = len(y)
    if n == 0:
        return y
    # Sort in decreasing order
    w = np.sort(y)[::-1].astype(np.float64)
    cumsum = np.cumsum(w)
    rho = np.arange(1, n + 1)
    # Find the largest i (1-indexed) such that w[i-1] > (cumsum[i-1] - target_sum) / i
    theta = (cumsum - target_sum) / rho
    # Condition: w[i-1] > theta[i-1]
    cond = w > theta
    if not np.any(cond):
        # All y[i] <= 0, uniform distribution
        return np.full(n, target_sum / n)
    rho_val = np.max(np.where(cond)[0]) + 1
    tau = (cumsum[rho_val - 1] - target_sum) / rho_val
    return np.maximum(0.0, y - tau)

def compute_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    max_c = np.max(c)
    return 2.0 * n * max_c / (S * S)

def smooth_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*c))) where c=conv(a,a).
    Returns (grad, c, max_c).
    """
    n = len(a)
    c = fft_convolve(a, a)
    max_c = np.max(c)
    
    # Compute softmax weights
    logits = beta * (c - max_c)
    # numsically stable: logits are all <= 0
    exp_logits = np.exp(logits)
    w = exp_logits / np.sum(exp_logits)
    
    # Compute gradient
    # grad[i] = sum_{k=0}^{n-1} w[i+k]*a[k] + sum_{k=0}^{n-1} w[i-k]*a[k]
    # First term: cross-correlation type = conv(a, w_rev)[2n-2-i]
    # Second term: convolution type = conv(w, a)[i]
    
    size = 1
    while size < 4 * n:
        size <<= 1
    
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    
    # Second term: conv(w, a)
    wa = np.fft.irfft(W * A, size)
    second_term = wa[:n]
    
    # First term: conv(a, w_rev)[2n-2-i] for i=0,...,n-1
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, size)
    aw_rev = np.fft.irfft(A * W_rev, size)
    # first_term[i] = aw_rev[2n-2-i] = aw_rev[n-1:2n-1][::-1]
    first_term = aw_rev[n-1:2*n-1][::-1]
    
    grad = first_term + second_term
    return grad, c, max_c

def optimize_subgradient(a_init, n_iter, lr_start, deadline):
    """
    Subgradient method using the exact max function.
    """
    n = len(a_init)
    target_sum = 1.0
    a = simplex_project(a_init.astype(np.float64), target_sum)
    
    best_score = compute_score(a)
    best_a = a.copy()
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = fft_convolve(a, a)
        max_c = np.max(c)
        
        # Find peak positions (within 1% of max)
        threshold = max_c * 0.99
        peak_mask = c >= threshold
        peak_idx = np.where(peak_mask)[0]
        
        if len(peak_idx) == 0:
            break
        
        # Compute subgradient
        grad = np.zeros(n)
        for m in peak_idx:
            for i in range(n):
                # dc[m]/da[i] = a[m-i] (if 0<=m-i<n) + a[i-m] (if 0<=i-m<n)
                mi = m - i
                im = i - m
                if 0 <= mi < n:
                    grad[i] += a[mi]
                if 0 <= im < n:
                    grad[i] += a[im]
        
        grad /= len(peak_idx)
        
        # Step
        lr = lr_start / (1 + 0.001 * it)
        a_new = a - lr * grad
        a_new = simplex_project(a_new, target_sum)

            best_a = a_new.copy()

            print(f"    subgrad iter={it}, lr={lr:.6f}, score={best_score:.8f}")
    
    return best_a, best_score

Wait, I need to reconsider. The subgradient computation is O(n * |P|) per step, which for n=1024 and |P| potentially large could be slow. Let me optimize that.

Actually, for the subgradient, I can compute it more efficiently:
`grad[i] = (1/|P|) * sum_{m in P} (a[m-i] + a[i-m])`

This is equivalent to: for each peak position m, add `a[|m-i|]` to grad[i] (with the factor of 2 when m=i).

Actually, `a[m-i] + a[i-m]` is not the same as `2*a[|m-i|]` in general. When m >= i: `a[m-i] + a[i-m]` where `a[i-m] = 0` if i-m < 0 (i.e., m > i). So it's just `a[m-i]`. When m = i: both terms give `a[0]`, so it's `2*a[0]`. When m < i: it's `a[i-m]`.

Wait, I defined `a[k] = 0` for k outside [0, n-1]. So:
- If m >= i: `a[m-i]` is valid (m-i >= 0) and `a[i-m]` = 0 (i-m <= 0, and only non-zero if i-m = 0, i.e., m = i)
  - Actually if m = i: a[m-i] = a[0] and a[i-m] = a[0], so sum = 2*a[0]
  - If m > i: a[m-i] is valid (if m-i < n), a[i-m] = 0 (since i-m < 0)
- If m < i: `a[m-i]` = 0 (m-i < 0), `a[i-m]` is valid (if i-m < n)

So: `dc[m]/da[i] = a[|m-i|] + (a[0] if m == i else 0)`

Wait no. If m = i: a[m-i] = a[0] and a[i-m] = a[0], sum = 2*a[0]. So it's `a[0] + a[0] = 2*a[0]`.
If m ≠ i: it's just `a[|m-i|]` (if |m-i| < n).

So `grad[i] = (1/|P|) * [sum_{m in P, m≠i} a[|m-i|] + 2*a[0]*1[i in P]]`

This is a sum of shifted copies of a, which can be computed efficiently using convolution!

`grad[i] = (1/|P|) * sum_{m in P} (a[|m-i|] + a[0]*1[m==i])`
`= (1/|P|) * [sum_{m in P} a[|m-i|] + a[0] * 1[i in P]]`

The term `sum_{m in P} a[|m-i|]` is the auto-correlation of the indicator function of P with a. Let `p[m] = 1[m in P]` (length 2n-1). Then:
`sum_{m in P} a[|m-i|] = (p cross-corr a)[i]` where we need to handle the absolute value.

Actually, `sum_m p[m] * a[|m-i|] = sum_m p[m] * a[m-i] * 1[0<=m-i<n] + sum_m p[m] * a[i-m] * 1[0<=i-m<n]`

This is the same as before: cross-correlation + convolution of p and a.

So I can compute the subgradient using the same FFT trick! Just replace w with p (the indicator of peak positions) and normalize.

def subgradient_fft(a, peak_mask):
    """Compute subgradient of max(c) using FFT."""
    n = len(a)
    # peak_mask has length 2n-1
    p = peak_mask.astype(np.float64)
    
    size = 1
    while size < 4 * n:
        size <<= 1
    
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    
    # conv(p, a)
    pa = np.fft.irfft(P * A, size)
    second_term = pa[:n]
    
    # cross-corr: conv(a, p_rev)[2n-2-i]
    p_rev = p[::-1]
    P_rev = np.fft.rfft(p_rev, size)
    apa_rev = np.fft.irfft(A * P_rev, size)
    first_term = apa_rev[n-1:2*n-1][::-1]
    
    grad = (first_term + second_term) / np.sum(peak_mask)
    return grad

"""
Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses smoothed gradient descent (log-sum-exp approximation of max) with simplex projection,
subgradient refinement, and multiple restarts with different initializations and lengths.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def simplex_project(y, target_sum):
    """Project y onto the simplex {x >= 0, sum(x) = target_sum}."""
    n = len(y)
    if n == 0:
        return y.copy()
    y = np.asarray(y, dtype=np.float64)
    w = np.sort(y)[::-1]
    cumsum = np.cumsum(w)
    rho = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - target_sum) / rho
    cond = w > theta
    if not np.any(cond):
        return np.full(n, target_sum / n, dtype=np.float64)
    rho_val = int(np.max(np.where(cond)[0]) + 1)
    tau = (cumsum[rho_val - 1] - target_sum) / rho_val
    return np.maximum(0.0, y - tau)


def compute_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    max_c = np.max(c)
    return 2.0 * n * max_c / (S * S)


def compute_gradient(a, peak_positions):
    """
    Compute the (sub)gradient of max(c) w.r.t. a, where c = conv(a,a).
    Uses FFT for efficiency. peak_positions is a boolean mask of length 2n-1.
    """
    n = len(a)
    p = peak_positions.astype(np.float64)
    count = np.sum(p)
    if count == 0:
        return np.zeros(n)

    size = 1
    while size < 4 * n:
        size <<= 1

    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)

    # conv(p, a) -> second term
    pa = np.fft.irfft(P * A, size)
    second_term = pa[:n]

    # cross-corr: conv(a, p_rev)[2n-2-i] -> first term
    p_rev = p[::-1]
    P_rev = np.fft.rfft(p_rev, size)
    apa_rev = np.fft.irfft(A * P_rev, size)
    first_term = apa_rev[n - 1:2 * n - 1][::-1]

    grad = (first_term + second_term) / count
    return grad


def optimize_smoothed(a_init, beta_start, beta_end, n_iter, lr_start, lr_end, deadline, verbose=True):
    """
    Smoothed gradient descent with simplex projection.
    Minimizes (1/beta)*log(sum(exp(beta*c))) as approximation to max(c).
    """
    n = len(a_init)
    target_sum = 1.0
    a = simplex_project(a_init.astype(np.float64), target_sum)

    best_score = compute_score(a)
    best_a = a.copy()
    no_improve_count = 0

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, n_iter - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        # Compute smoothed gradient
        c = fft_convolve(a, a)
        max_c = np.max(c)
        logits = beta * (c - max_c)
        exp_logits = np.exp(logits)
        w = exp_logits / np.sum(exp_logits)

        # Gradient via FFT (same as subgradient but with soft weights)
        size = 1
        while size < 4 * n:
            size <<= 1
        W = np.fft.rfft(w, size)
        A = np.fft.rfft(a, size)

        wa = np.fft.irfft(W * A, size)
        second_term = wa[:n]

        w_rev = w[::-1]
        W_rev = np.fft.rfft(w_rev, size)
        aw_rev = np.fft.irfft(A * W_rev, size)
        first_term = aw_rev[n - 1:2 * n - 1][::-1]

        grad = first_term + second_term

        # Normalize gradient for stability
        grad_norm = np.max(np.abs(grad))
        if grad_norm > 1e-12:
            grad = grad / grad_norm * 0.1

        a_new = a - lr * grad
        a_new = simplex_project(a_new, target_sum)

        score = compute_score(a_new)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve_count = 0
        else:
            no_improve_count += 1
            # Try smaller step
            a_new2 = a - lr * 0.05 * grad
            a_new2 = simplex_project(a_new2, target_sum)
            score2 = compute_score(a_new2)
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_a = a_new2.copy()
                a = a_new2
                no_improve_count = 0

        if no_improve_count > 200:
            # Add noise to escape local minimum
            noise = 0.01 * np.random.standard_normal(n)
            a = simplex_project(a + noise, target_sum)
            no_improve_count = 0

        if verbose and it % 200 == 0:
            print(f"      iter={it}, beta={beta:.1f}, score={compute_score(a):.8f}, best={best_score:.8f}")

    return best_a, best_score


def optimize_subgradient(a_init, n_iter, lr_start, deadline, tol_frac=0.01, verbose=True):
    """
    Subgradient method using exact max(c) and FFT-based gradient computation.
    """
    n = len(a_init)
    target_sum = 1.0
    a = simplex_project(a_init.astype(np.float64), target_sum)

    best_score = compute_score(a)
    best_a = a.copy()

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = fft_convolve(a, a)
        max_c = np.max(c)

        # Find peaks within tolerance
        threshold = max_c * (1.0 - tol_frac)
        peak_mask = c >= threshold

        # Compute subgradient via FFT
        grad = compute_gradient(a, peak_mask)

        # Step with diminishing learning rate
        lr = lr_start / (1.0 + 0.005 * it)

        a_new = a - lr * grad
        a_new = simplex_project(a_new, target_sum)

        score = compute_score(a_new)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new

        if verbose and it % 100 == 0:
            print(f"      subgrad iter={it}, lr={lr:.6f}, best={best_score:.8f}")

    return best_a, best_score


def make_initializations(n, rng):
    """Generate various initial sequences of length n, normalized to sum=1."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussian
    sigma = n / 4.0
    g = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    inits['gaussian'] = g / np.sum(g)

    # Exponential
    e = np.exp(-np.abs(x - n / 2) / (n / 6.0))
    inits['exponential'] = e / np.sum(e)

    # Cosine
    c = (1.0 + np.cos(2.0 * np.pi * x / n)) ** 2
    c = np.maximum(c, 1e-6)
    inits['cosine'] = c / np.sum(c)

    # Random
    r = np.abs(rng.standard_normal(n)) + 0.1
    inits['random'] = r / np.sum(r)

    # Two-peak
    peak1 = np.exp(-0.5 * ((x - n * 0.25) / (n / 10.0)) ** 2)
    peak2 = np.exp(-0.5 * ((x - n * 0.75) / (n / 10.0)) ** 2)
    two = peak1 + peak2
    inits['two_peak'] = two / np.sum(two)

    # Linear ramp
    ramp = np.linspace(0.1, 2.0, n)
    inits['ramp'] = ramp / np.sum(ramp)

    return inits
