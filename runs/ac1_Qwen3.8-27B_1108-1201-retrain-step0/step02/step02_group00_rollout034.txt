def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_lse(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a.
    grad[m] = 2 * sum_k w[k] * a[k-m] (valid indices).
    Computed via FFT cross-correlation.
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_toppk(a, c, top_k, ramp=1.0):
    """Gradient using top-k active lags with optional ramp weighting."""
    n = len(a)
    k_eff = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k_eff]
    
    if ramp < 1.0 and k_eff > 1:
        # Ramp: weight the top lag more, others less
        weights = np.ones(k_eff)
        weights[1:] *= ramp
    else:
        weights = np.ones(k_eff)
    
    top_vals = c[top_indices] * weights
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    norm_weights = top_vals / total

    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = norm_weights[i]

    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_exact_topk(a, c, top_k):
    """Exact subgradient: use only the top-k lags with uniform weight.
    grad[m] = 2/k * sum_{k in top} a[k-m] (valid indices).
    """
    n = len(a)
    k_eff = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k_eff]
    
    w_full = np.zeros(2 * n - 1)
    w_full[top_indices] = 1.0 / k_eff

    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista_improved(a_init, n, S, max_time, rng, beta_init=2.0, beta_max=300.0,
                        beta_rate=0.001, restart_interval=5000, restart_prob=0.2):
    """FISTA with improved log-sum-exp schedule and periodic restarts."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    patience = 0
    patience_best = best_score

    while time.monotonic() - start < max_time:
        it += 1
        # Slower beta increase for better convergence
        beta = min(beta_max, beta_init * (1.0 + beta_rate * it))
        step = 1.0 / (2.0 * beta)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            patience = 0
        else:
            patience += 1

        # Adaptive restart
        if patience > restart_interval and rng.random() < restart_prob:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0
            patience = 0
            continue

        grad = fft_grad_lse(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        x_new = project_simplex(y - step * grad, S)

        # Evaluate every few iterations
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
                patience = 0

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    return best_local, best_score


def run_nesterov_exact(a_init, n, S, max_time, rng, top_k_init=3, top_k_max=512,
                        ramp_start=1.0, ramp_end=0.3):
    """Nesterov-accelerated exact subgradient with adaptive top-k and ramp."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive top-k: start small, grow
        top_k = min(top_k_max, top_k_init + it // 100)
        # Ramp: gradually reduce weight of non-top lags
        progress = min(1.0, it / (max_time * 10))
        ramp = ramp_start + (ramp_end - ramp_start) * progress

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Use exact top-k gradient
        grad = fft_grad_toppk(y, c, top_k, ramp=ramp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: Polyak-like with decay
        gap = best_score - S * S / (n * 2)  # crude lower bound
        gap = max(gap, 1e-6)
        step = 2.0 * gap / (gnorm * gnorm + 1e-15)
        step = min(step, 4.0 * S / (n * gnorm + 1e-15))
        step = max(step, 1e-8)

        x_new = project_simplex(y - step * grad, S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic perturbation
        if it % 3000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, eta_init=0.5, top_k=10):
    """Mirror descent (multiplicative weights) with adaptive learning rate."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Adaptive eta
        eta = eta_init / (1.0 + 0.001 * it)
        k_eff = min(top_k + it // 500, 64)
        grad = fft_grad_toppk(a, c, k_eff, ramp=1.0)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_new * (S / s_new)

    return best_local, best_score


def anti_peak_refine(a, n, S, max_time, rng):
    """
    Local refinement that explicitly targets the dominant lag.
    At each step, identify the peak lag and try to reduce it by
    moving mass from positions that contribute to that lag.
    """
    a = np.clip(a, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Find the peak lag
        peak_lag = int(np.argmax(c))
        
        # Compute the gradient of c[peak_lag] w.r.t. a
        # grad[m] = 2*a[peak_lag - m] if 0 <= peak_lag - m < n
        grad = np.zeros(n)
        valid = np.arange(n)
        shifted = peak_lag - valid
        mask = (shifted >= 0) & (shifted < n)
        grad[mask] = 2.0 * a[shifted[mask]]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Also compute gradient of a random off-peak lag for balance
        other_lag = int(rng.integers(0, 2 * n - 1))
        if other_lag == peak_lag:
            other_lag = (other_lag + 1) % (2 * n - 1)
        
        grad2 = np.zeros(n)
        shifted2 = other_lag - valid
        mask2 = (shifted2 >= 0) & (shifted2 < n)
        grad2[mask2] = 2.0 * a[shifted2[mask2]]
        
        # Combined gradient: reduce peak, allow slight increase elsewhere
        combined = grad - 0.3 * grad2
        cgnorm = np.linalg.norm(combined)
        if cgnorm < 1e-15:
            break

        # Small step
        step = 0.01 * S / cgnorm
        a_new = project_simplex(a - step * combined, S)
        
        c_new = fft_convolve_self(a_new)
        if c_new.max() < best_score:
            best_score = c_new.max()
            best_local = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            step = 0.003 * S / cgnorm
            a_new = project_simplex(a - step * combined, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < best_score:
                best_score = c_new.max()
                best_local = a_new.copy()
                a = a_new
            else:
                # Random perturbation
                a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)

    return best_local, best_score


def coarse_to_fine(a_coarse, n_fine, S_fine, max_time, rng):
    """Take a coarse solution and refine it to a finer grid via upsampling."""
    n_c = len(a_coarse)
    # Upsample by repetition
    ratio = max(1, n_fine // n_c)
    a_fine = np.repeat(a_coarse, ratio)[:n_fine]
    a_fine = np.clip(a_fine, 0.0, 1000.0)
    s = a_fine.sum()
    if s < 1e-15:
        a_fine = np.ones(n_fine) * S_fine / n_fine
    else:
        a_fine = a_fine * (S_fine / s)
    
    # Quick FISTA polish on the fine grid
    a_refined, score = run_fista_improved(a_fine, n_fine, S_fine, max_time, rng,
                                           beta_init=10.0, beta_max=100.0,
                                           beta_rate=0.002)
    return a_refined, score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_score(a):
    """Compute the normalized score (max of self-convolution)."""
    c = fft_convolve_self(a)
    return c.max()


def propose_candidate(seed=42, budget_s=2200):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    
    Strategy:
    1. Multi-start FISTA with improved beta schedule
    2. Exact subgradient Nesterov polishing with adaptive top-k ramp
    3. Anti-peak local refinement
    4. Mirror descent for diverse exploration
    5. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None

    # Try to get initial sequence
    given_seq = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given_seq = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Determine lengths to try
    lengths_to_try = []
    if given_seq is not None:
        lengths_to_try.append(len(given_seq))
    # Add other useful lengths
    for n_try in [4096, 8192, 12288, 16384]:
        if n_try not in lengths_to_try:
            lengths_to_try.append(n_try)
    # Sort by preference: given length first, then nearby
    lengths_to_try = sorted(set(lengths_to_try))

    print("Lengths to try: %s" % lengths_to_try)
    print("Given sequence available: %s" % (given_seq is not None))

    # Phase 1: Multi-start exploration
    phase1_time = max(60.0, (deadline - time.monotonic()) * 0.55)
    print("=== Phase 1: Multi-start FISTA (%.0f sec) ===" % phase1_time)
    phase1_deadline = time.monotonic() + phase1_time

    n_starts = 4  # starts per length
    per_start = phase1_time / (len(lengths_to_try) * n_starts)

    for n in lengths_to_try:
        if time.monotonic() > phase1_deadline:
            break
        S = np.sqrt(2.0 * float(n))
        
        # Build initializations for this length
        inits = []
        if given_seq is not None and len(given_seq) == n:
            inits.append(("given", given_seq.copy()))
        
        tt = np.linspace(0, 1, n, endpoint=False)
        inits.append(("rand", rng.random(n) + 0.1))
        inits.append(("sin_mix", 1.0 + 0.3*np.sin(2*np.pi*3*tt) + 
                      0.2*np.cos(2*np.pi*7*tt) + 0.1*np.sin(2*np.pi*13*tt)))
        inits.append(("bump", np.exp(-0.5*((tt-0.3)*5)**2) + 
                      0.5*np.exp(-0.5*((tt-0.7)*5)**2) + 0.05))
        
        for name, arr in inits[:n_starts]:
            if time.monotonic() > phase1_deadline:
                break
            arr = np.clip(arr, 0.0, 1000.0).astype(np.float64)
            s = arr.sum()
            if s < 1e-15:
                arr = np.ones(n) * S / n
            else:
                arr = arr * (S / s)
            
            t0 = time.monotonic()
            # FISTA with improved schedule
            a_fista, score = run_fista_improved(
                arr, n, S, per_start * 0.7, rng,
                beta_init=2.0, beta_max=200.0, beta_rate=0.0008,
                restart_interval=4000, restart_prob=0.25)
            
            a_norm = safe_normalize(a_fista, n)
            sc = compute_score(a_norm)
            if sc < global_best:
                global_best = sc
                global_a = a_norm.copy()
                elapsed = time.monotonic() - t0
                print("  FISTA [%s, n=%d]: %.6f (%.1fs)" % (name, n, global_best, elapsed))

            # Quick Nesterov polish
            rem = per_start - (time.monotonic() - t0)
            if rem > 3:
                a_ne, score_ne = run_nesterov_exact(
                    a_norm, n, S, rem, rng,
                    top_k_init=3, top_k_max=128,
                    ramp_start=1.0, ramp_end=0.5)
                a_ne = safe_normalize(a_ne, n)
                sc_ne = compute_score(a_ne)
                if sc_ne < global_best:
                    global_best = sc_ne
                    global_a = a_ne.copy()
                    print("  Nesterov [%s, n=%d]: %.6f" % (name, n, global_best))

        print("  After n=%d: best=%.6f" % (n, global_best))

    print("=== Phase 1 complete: global_best=%.6f ===" % global_best)

    # Phase 2: Intensive polishing
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        phase2_time = max(30.0, (deadline - time.monotonic()) * 0.35)
        print("=== Phase 2: Polish (%.0f sec, n=%d) ===" % (phase2_time, n))
        phase2_deadline = time.monotonic() + phase2_time

        # 2a: FISTA with high beta
        seg = phase2_time * 0.3
        a_p, s_p = run_fista_improved(global_a, n, S, seg, rng,
                                       beta_init=30.0, beta_max=400.0,
                                       beta_rate=0.001, restart_interval=3000,
                                       restart_prob=0.15)
        a_p = safe_normalize(a_p, n)
        sc_p = compute_score(a_p)
        if sc_p < global_best:
            global_best = sc_p
            global_a = a_p.copy()
            print("  Polish FISTA high-beta: %.6f" % global_best)

        # 2b: Exact Nesterov with growing top-k
        seg = max(10.0, (phase2_deadline - time.monotonic()) * 0.35)
        a_ne, s_ne = run_nesterov_exact(global_a, n, S, seg, rng,
                                         top_k_init=5, top_k_max=512,
                                         ramp_start=1.0, ramp_end=0.3)
        a_ne = safe_normalize(a_ne, n)
        sc_ne = compute_score(a_ne)
        if sc_ne < global_best:
            global_best = sc_ne
            global_a = a_ne.copy()
            print("  Exact Nesterov: %.6f" % global_best)

        # 2c: Anti-peak refinement
        seg = max(5.0, (phase2_deadline - time.monotonic()) * 0.25)
        a_ap, s_ap = anti_peak_refine(global_a, n, S, seg, rng)
        a_ap = safe_normalize(a_ap, n)
        sc_ap = compute_score(a_ap)
        if sc_ap < global_best:
            global_best = sc_ap
            global_a = a_ap.copy()
            print("  Anti-peak: %.6f" % global_best)

        # 2d: Mirror descent
        seg = max(5.0, (phase2_deadline - time.monotonic()))
        a_md, s_md = run_mirror_descent(global_a, n, S, seg, rng,
                                         eta_init=0.3, top_k=15)
        a_md = safe_normalize(a_md, n)
        sc_md = compute_score(a_md)
        if sc_md < global_best:
            global_best = sc_md
            global_a = a_md.copy()
            print("  Mirror descent: %.6f" % global_best)

    print("=== Phase 2 complete: global_best=%.6f ===" % global_best)

    # Phase 3: Final fine-tuning
    if global_a is not None and time.monotonic() < deadline - 10:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        remaining = deadline - time.monotonic() - 5.0
        print("=== Phase 3: Final tuning (%.0f sec) ===" % remaining)

        # Very fine Nesterov with high top-k
        a_final, s_final = run_nesterov_exact(global_a, n, S, remaining, rng,
                                               top_k_init=50, top_k_max=1024,
                                               ramp_start=0.8, ramp_end=0.4)
        a_final = safe_normalize(a_final, n)
        sc_final = compute_score(a_final)
        if sc_final < global_best:
            global_best = sc_final
            global_a = a_final.copy()
            print("  Final Nesterov: %.6f" % global_best)

        # One more anti-peak pass
        if time.monotonic() < deadline - 10:
            rem2 = deadline - time.monotonic() - 5.0
            a_ap2, s_ap2 = anti_peak_refine(global_a, n, S, rem2, rng)
            a_ap2 = safe_normalize(a_ap2, n)
            sc_ap2 = compute_score(a_ap2)
            if sc_ap2 < global_best:
                global_best = sc_ap2
                global_a = a_ap2.copy()
                print("  Final anti-peak: %.6f" % global_best)

    # Verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Verification score: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        print("No valid candidate found, using fallback")
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_obj = 2.0 * len(global_a) * compute_score(
        safe_normalize(global_a, len(global_a))) / (safe_normalize(global_a, len(global_a)).sum()**2)
    print("Final: length=%d, normalized_score=%.6f, objective≈%.6f" % 
          (len(global_a), global_best, final_obj))
    return [float(x) for x in global_a]
