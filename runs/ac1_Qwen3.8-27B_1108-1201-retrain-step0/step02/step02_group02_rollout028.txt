Multi-strategy optimization for minimizing autocorrelation peak of non-negative sequences.
Combines smooth-max gradient descent, LP-based peak shaving, and diverse restarts.

    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def smooth_max_and_grad(a, tau):
    """Compute smooth max of conv(a,a) and gradient."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1:2 * n - 1]
    
    return value, g, c

def lp_peak_shave(a, M):
    """Solve LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)

    # conv(a, g)[k] = sum_i a[i] * g[k-i] for valid i, k-i
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []

        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):

            cols.append(j)
            vals.append(a[k - j])

    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

Multi-strategy hybrid optimization for minimizing the autocorrelation peak of
non-negative step functions. Combines smooth-max gradient descent in log-space,
LP-based peak shaving, diverse restarts, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def smooth_max_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient wrt a.
    Returns (value, gradient, convolution).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)

    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1:2 * n - 1]

    return value, g, c


def power_mean_grad(a, p):
    """
    Compute p-th power mean of conv(a,a) and its gradient.
    value = (sum(c_k^p))^(1/p), which approximates max(c) for large p.
    Gradient: dc_k/da_j = a[k-j] + a[j-k] (bounded).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_clipped = np.maximum(c, 1e-30)

    # Power mean value
    c_p_sum = np.sum(c_clipped ** p)
    value = c_p_sum ** (1.0 / p)

    # Gradient: d(value)/da_j = (1/p) * c_p_sum^(1/p - 1) * p * sum_k c_k^(p-1) * dc_k/da_j
    #           = c_p_sum^(1/p - 1) * sum_k c_k^(p-1) * (a[k-j] + a[j-k])
    # The sum over k of c_k^(p-1) * (a[k-j] + a[j-k]) is a cross-correlation
    weights = c_clipped ** (p - 1)
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, weights)
    g = 2.0 * corr[n - 1:2 * n - 1] * c_p_sum ** (1.0 / p - 1.0)

    return value, g, c


def lp_solve_peak_shave(a, M):
    """
    Solve LP: maximize sum(g) subject to conv(a, g)[k] <= M for all k, g >= 0.
    Returns optimal g or None if infeasible.
    """
    n = len(a)
    c_obj = -np.ones(n)

    # Build sparse constraint matrix: A[k, j] = a[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    """
    S_target = np.sqrt(2.0 * n)
    a = np.clip(a_init.astype(np.float64).copy(), 0.01, None)
    a = a * S_target / np.sum(a)
    theta = np.log(a + 1e-10)

    best_theta = theta.copy()
    best_score = local_score(a)

    # Adam parameters
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    no_improve = 0
    patience = 4000

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit

        if use_power_mean:
            # Power mean with increasing p
            p = 8.0 + 24.0 * elapsed_frac
            a = softmax_normalize(theta) * S_target
            _, g, c = power_mean_grad(a, p)
            tau = 0.0  # not used
        else:
            # Temperature schedule: aggressive annealing
            if elapsed_frac < 0.3:
                tau = (15.0 / n) * (1.0 - elapsed_frac / 0.3 * 0.7)
            elif elapsed_frac < 0.7:
                tau = (15.0 / n) * 0.3 * (0.1 ** ((elapsed_frac - 0.3) / 0.4))
            else:
                tau = (15.0 / n) * 0.03 * (0.3 ** ((elapsed_frac - 0.7) / 0.3))
            tau = max(tau, 0.002 / n)

            a = softmaxNormalize(theta) * S_target
            _, g, c = smooth_max_grad(a, tau)

        # Convert gradient to theta-space
        inner = np.dot(a, g)
        g_theta = a * (g - inner / S_target)

        # Adam step
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g_theta
        v_buf = beta2 * v_buf + (1 - beta2) * g_theta ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)

        # Learning rate schedule
        base_lr = 2.0 / np.sqrt(step + 1)
        if elapsed_frac < 0.2:
            lr = base_lr * 1.5
        elif elapsed_frac < 0.6:
            lr = base_lr
        else:
            lr = base_lr * 0.4

        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic evaluation
        if step % 80 == 0:
            a_check = softmax_normalize(theta) * S_target
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
                no_improve = 0
            else:
                no_improve += 80

            # Escape from local minima
            if no_improve > patience:
                r = rng.random()
                a_best = softmax_normalize(best_theta) * S_target
                if r < 0.3:
                    # Large Gaussian kick
                    theta = best_theta + 0.4 * rng.standard_normal(n)
                elif r < 0.6:
                    # Peak-directed perturbation
                    c_best = fft_convolve(a_best, a_best)
                    peak_idx = int(np.argmax(c_best))
                    # Find elements contributing to peak
                    contrib = np.zeros(n)
                    for i in range(n):
                        ki = peak_idx - i
                        if 0 <= ki < n:
                            contrib[i] = a_best[i] * a_best[ki]
                    top_contrib = np.argsort(contrib)[-max(1, n // 10):]
                    theta[top_contrib] += 0.25 * rng.standard_normal(len(top_contrib))
                elif r < 0.8:
                    # Block perturbation
                    block = max(4, n // 16)
                    start = int(rng.integers(0, n - block))
                    theta[start:start + block] += 0.2 * rng.standard_normal(block)
                else:
                    # Rescale random subset
                    subset = rng.choice(n, size=n // 4, replace=False)
                    theta[subset] += rng.normal(0, 0.3, size=len(subset))
                no_improve = 0

        if step % 2000 == 0:
            a_now = softmax_normalize(theta) * S_target
            print(f"        grad step={step}, score={local_score(a_now):.6f}, best={best_score:.6f}")

    # Final check
    a_final = softmax_normalize(theta) * S_target
    final_score = local_score(a_final)
    if final_score < best_score:
        best_score = final_score
        best_theta = theta.copy()

    return softmax_normalize(best_theta) * S_target, best_score


def make_init(n, rng, method='broadband'):
    """Create diverse initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if method == 'broadband':
        a = np.ones(n)
        for k in range(1, 18):
            freq = k * (1.0 + 0.7 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.18 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'random':
        return np.abs(rng.standard_normal(n)) + 0.05
    elif method == 'asymmetric':
        a = np.ones(n)
        ratio = rng.uniform(0.3, 0.7)
        split = int(n * ratio)
        a[:split] = rng.uniform(1.0, 2.0)
        a[split:] = rng.uniform(0.3, 0.8)
        a += 0.08 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'two_peak':
        a = np.ones(n)
        center1 = n // 4
        center2 = 3 * n // 4
        width = n // 8
        a[max(0, center1 - width):center1 + width] *= 1.8
        a[max(0, center2 - width):center2 + width] *= 0.4
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'fourier_random':
        # Random Fourier series with random phases
        a = np.ones(n)
        for k in range(1, 12):
            freq = rng.uniform(k * 0.5, k * 1.5)
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.15 * rng.uniform(0.5, 1.0)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        return np.clip(a, 0.01, None)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)


def peak_redistribute(a, rng, strength=0.03):
    """
    Targeted peak redistribution: find the dominant peak, identify top
    contributing pairs, reduce them slightly, and increase non-contributors.
    """
    n = len(a)
    c = fft_convolve(a, a)
    peak_idx = int(np.argmax(c))

    # Compute contribution of each element to the peak
    contrib = np.zeros(n)
    for i in range(n):
        ki = peak_idx - i
        if 0 <= ki < n:
            contrib[i] = a[i] * a[ki]

    top_k = max(4, n // 12)
    bottom_k = max(4, n // 8)
    top_idx = np.argsort(contrib)[::-1][:top_k]
    bottom_idx = np.argsort(contrib)[:bottom_k]

    reduction = rng.uniform(strength * 0.5, strength)
    a_new = a.copy()
    a_new[top_idx] *= (1.0 - reduction)
    a_new[bottom_idx] *= (1.0 + reduction * len(top_idx) / len(bottom_idx))
    return np.clip(a_new, 0.0, None)


def local_polish(a_init, time_limit, rng):
    """
    Local polishing combining peak redistribution and small perturbations.
    Uses simulated annealing acceptance.
    """
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score

    t0 = time.monotonic()
    temp = 0.0005
    cooling = 0.9999
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        r = rng.random()

        if r < 0.4:
            # Peak redistribution
            new_a = peak_redistribute(current, rng, strength=rng.uniform(0.02, 0.06))
        elif r < 0.6:
            # Pair swap: move mass from high-contributor to low
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            contrib = np.zeros(n)
            for i in range(n):
                ki = peak_idx - i
                if 0 <= ki < n:
                    contrib[i] = current[i] * current[ki]
            top_idx = np.argsort(contrib)[::-1][:max(2, n // 16)]
            j_red = top_idx[int(rng.integers(0, len(top_idx)))]
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red:
                new_a = current.copy()
                amt = current[j_red] * rng.uniform(0.02, 0.08)
                new_a[j_red] = max(0, current[j_red] - amt)
                new_a[j_inc] = current[j_inc] + amt
            else:
                new_a = current * np.exp(rng.normal(0, 0.01, size=n))
        elif r < 0.8:
            # Block scaling
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.04))
            new_a = current.copy()
            new_a[start:start + width] *= factor
        else:
            # Global small perturbation
            new_a = current * np.exp(rng.normal(0, 0.008, size=n))

        new_a = np.clip(new_a, 0.0, 1000.0)
        new_score = local_score(new_a)

        # Simulated annealing acceptance
        if new_score < current_score:
            current = new_a
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-10)):
            current = new_a
            current_score = new_score

        temp = max(temp * cooling, 1e-10)

        if step % 500 == 0:
            print(f"        polish step={step}, cur={current_score:.6f}, best={best_score:.6f}, temp={temp:.6f}")

    return best, best_score


def softmaxNormalize(theta):
    """Stable softmax (alias to avoid inconsistency)."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase hybrid search:
    1. Primary gradient optimization at n=384 (from provided sequence)
    2. LP-based iterative polishing
    3. Local peak-redistribution polish
    4. Diverse restarts at different lengths
    5. Final micro-optimization
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 45.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # === PHASE 1: Primary gradient optimization ===
    n_primary = len(start_seq) if start_seq is not None else 384
    print(f"\n{'='*60}")
    print(f"PHASE 1: Gradient optimization (n={n_primary})")
    print(f"{'='*60}")

    if start_seq is not None and len(start_seq) == n_primary:
        a0 = start_seq.copy()
    else:
        a0 = make_init(n_primary, rng, 'broadband')

    t_phase1_grad = 0.50 * time_budget
    t_phase1_lp = 0.15 * time_budget
    t_phase1_polish = 0.10 * time_budget

    # Gradient descent
    t0 = time.monotonic()
    a_opt, s_opt = optimize_gradient(a0, t_phase1_grad, rng, n_primary, use_power_mean=False)
    elapsed1 = time.monotonic() - t0
    print(f"  Gradient done in {elapsed1:.0f}s: score={s_opt:.6f}")
    if s_opt < best_score:
        best_score = s_opt
        best_seq = a_opt.copy()
        print(f"  *** New best: {best_score:.6f} ***")

    # LP-based polishing
    if time.monotonic() < deadline - 30:
        t_lp = min(t_phase1_lp, deadline - time.monotonic() - 30)
        print(f"  LP polishing ({t_lp:.0f}s)...")
        a_lp, s_lp = optimize_lp_iteration(a_opt, t_lp)
        if s_lp < best_score:
            best_score = s_lp
            best_seq = a_lp.copy()
            print(f"  *** After LP: {best_score:.6f} ***")

    # Local polish
    if time.monotonic() < deadline - 30:
        t_pol = min(t_phase1_polish, deadline - time.monotonic() - 30)
        print(f"  Local polish ({t_pol:.0f}s)...")
        a_pol, s_pol = local_polish(best_seq, t_pol, rng)
        if s_pol < best_score:
            best_score = s_pol
            best_seq = a_pol.copy()
            print(f"  *** After polish: {best_score:.6f} ***")

    # === PHASE 2: Diverse restarts ===
    print(f"\n{'='*60}")
    print(f"PHASE 2: Diverse restarts")
    print(f"{'='*60}")

    t_phase2 = min(0.15 * time_budget, max(0, deadline - time.monotonic() - 60))
    if t_phase2 > 20:
        methods = ['random', 'asymmetric', 'two_peak', 'fourier_random']
        t_per_method = t_phase2 / len(methods)

        for method in methods:
            if time.monotonic() > deadline - 40:
                break
            t_alloc = min(t_per_method, deadline - time.monotonic() - 30)
            if t_alloc < 5:
                break
            a0_s = make_init(n_primary, rng, method)
            a_opt_s, s_opt_s = optimize_gradient(a0_s, t_alloc, rng, n_primary, use_power_mean=True)
            if s_opt_s < best_score:
                best_score = s_opt_s
                best_seq = a_opt_s.copy()
                print(f"  Method '{method}': {best_score:.6f} ***")
            else:
                print(f"  Method '{method}': {s_opt_s:.6f} (no improvement)")

    # === PHASE 3: Try different length ===
    print(f"\n{'='*60}")
    print(f"PHASE 3: Different length exploration")
    print(f"{'='*60}")

    alt_lengths = []
    for nl in [256, 512, 192]:
        if nl != n_primary:
            alt_lengths.append(nl)

    remaining = deadline - time.monotonic() - 40
    if remaining > 20 and best_seq is not None:
        # Try one alternative length
        n_alt = alt_lengths[0] if alt_lengths else 256
        t_alt = min(40.0, remaining)
        if t_alt > 10:
            # Interpolate best sequence to new length
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_alt)
            a0_alt = np.interp(x_new, x_old, best_seq)
            a0_alt = np.clip(a0_alt, 0.01, None)

            a_opt_alt, s_opt_alt = optimize_gradient(a0_alt, t_alt, rng, n_alt, use_power_mean=False)
            if s_opt_alt < best_score:
                best_score = s_opt_alt
                best_seq = a_opt_alt.copy()
                print(f"  n={n_alt}: {best_score:.6f} ***")
            else:
                print(f"  n={n_alt}: {s_opt_alt:.6f} (no improvement)")

    # === PHASE 4: Final comprehensive polish ===
    print(f"\n{'='*60}")
    print(f"PHASE 4: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 20
        if remaining > 15:
            # LP polish
            t_lp_final = remaining * 0.5
            a_lp2, s_lp2 = optimize_lp_iteration(best_seq, min(t_lp_final, remaining - 10))
            if s_lp2 < best_score:
                best_score = s_lp2
                best_seq = a_lp2.copy()
                print(f"  After final LP: {best_score:.6f} ***")

            # Gradient micro-optimization
            remaining2 = deadline - time.monotonic() - 20
            if remaining2 > 10:
                a_fin, s_fin = optimize_gradient(best_seq, remaining2, rng, len(best_seq), use_power_mean=False)
                if s_fin < best_score:
                    best_score = s_fin
                    best_seq = a_fin.copy()
                    print(f"  After final grad: {best_score:.6f} ***")

            # One more local polish
            remaining3 = deadline - time.monotonic() - 20
            if remaining3 > 5:
                a_pol2, s_pol2 = local_polish(best_seq, remaining3, rng)
                if s_pol2 < best_score:
                    best_score = s_pol2
                    best_seq = a_pol2.copy()
                    print(f"  After final polish: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'='*60}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 384
