"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy:
- Nesterov-accelerated subgradient with FFT-based top-k gradients
- Backtracking line search for adaptive step sizing
- Diverse structured and random initializations

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_toppk(a, c, top_k):
    """Gradient using top-k active lags via FFT cross-correlation."""
    n = len(a)
    size = 2 * n - 1
    if top_k >= size:
        top_k = size
    top_indices = np.argpartition(-c, top_k)[:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    
    w_full = np.zeros(size)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]

    fsize = 1 << (size + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_lse(a, c, beta):
    """Gradient of log-sum-exp smoothing via FFT."""
    n = len(a)
    size = 2 * n - 1
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (size + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_nesterov(a_init, n, S, max_time, rng, k_init=1, k_max=128,
                  k_schedule='linear', step_init=None, perturb_period=5000):
    """Nesterov-accelerated top-k subgradient with backtracking awareness."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    if step_init is None:
        step_init = S / np.sqrt(n)
    
    step = step_init

    while time.monotonic() - start < max_time:
        it += 1
        
        if k_schedule == 'linear':
            top_k = min(k_max, k_init + it // max(1, n // k_max))
        elif k_schedule == 'sqrt':
            top_k = min(k_max, k_init + int(np.sqrt(it)))
        else:
            top_k = k_init
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = fft_grad_toppk(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking: try to ensure decrease
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        # Check if we improved (every 10th iteration to save time)
        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() > cur * 1.01 and it > 100:
                step *= 0.7
            elif c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
            if step < 1e-8:
                step = step_init * 0.1

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Slowly decrease step
        step *= (1.0 - 0.00002)
        step = max(step, S / (np.sqrt(n) * 50))

        # Periodic perturbation to escape local minima
        if it % perturb_period == 0 and rng.random() < 0.3:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_toppk(a, c, top_k):
    """
    Gradient using top-k active lags, computed via FFT cross-correlation.
    grad[m] = 2 * sum_k w[k] * a[k-m] (circular, truncated to n).
    """
    n = len(a)
    size = 2 * n - 1
    if top_k >= size:
        top_k = size
    if top_k < 1:
        top_k = 1
    top_indices = np.argpartition(-c, min(top_k, size) - 1)[:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total

    w_full = np.zeros(size)
    for i in range(len(top_indices)):
        w_full[int(top_indices[i])] = weights[i]

    fsize = 1 << (size + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_lse(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a, via FFT.
    """
    n = len(a)
    size = 2 * n - 1
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (size + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_nesterov_annealed(a_init, n, S, max_time, rng):
    """
    Nesterov-accelerated subgradient with annealed k schedule.
    Start with large k (broad exploration), end with small k (sharp exploitation).
    Includes momentum reset on stagnation.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stagnation_count = 0
    last_improve = 0

    size = 2 * n - 1
    base_step = S / (np.sqrt(n) * 4.0)
    step = base_step

    while time.monotonic() - start < max_time:
        it += 1

        # Annealed k: start at size//8, decrease to 1
        frac = min(1.0, it / max(1, int(0.7 * max_time / 0.03))))
        top_k = max(1, int((1.0 - frac) * min(size // 8, 256) + 1))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1

        grad = fft_grad_toppk(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: decrease slowly, reset if stuck
        step = base_step / (1.0 + 0.001 * it)
        if stagnation_count > 800:
            step *= 0.5
            if stagnation_count > 2000:
                # Reset momentum and add perturbation
                y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
                t = 1.0
                step = base_step
                stagnation_count = 0

        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
                stagnation_count = 0

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    return best_local, best_score


def run_nesterov_exploit(a_init, n, S, max_time, rng, k_init=1, k_max=16):
    """
    Nesterov for exploitation phase: small k, long run, periodic restarts.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    size = 2 * n - 1
    base_step = S / (np.sqrt(n) * 6.0)

    while time.monotonic() - start < max_time:
        it += 1

        # Oscillate k between k_init and k_max for diversity
        top_k = k_init + (k_max - k_init) * abs(np.sin(0.01 * it))
        top_k = max(1, int(top_k))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = fft_grad_toppk(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = base_step / (1.0 + 0.0005 * it)
        step = max(step, base_step * 0.1)

        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic perturbation to escape local minima
        if it % 4000 == 0 and rng.random() < 0.4:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def run_fista_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=400.0):
    """FISTA with adaptive log-sum-exp smoothing."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 1.0 / (beta_init * 2.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.002 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = fft_grad_lse(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        step = 1.0 / (beta * 2.0 * (1.0 + 0.0003 * it))
        step = max(step, 1e-6)

        if it % 5000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def run_multiplicative(a_init, n, S, max_time, rng, top_k=20):
    """
    Multiplicative weights (mirror descent) with top-k subgradient.
    Good for maintaining positivity and exploring different geometry.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    eta = 0.15

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        eta = 0.15 / (1.0 + 0.003 * it)
        grad = fft_grad_toppk(a, c, top_k)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_new * (S / s_new)

    return best_local, best_score


def perturbation_search(a, n, S, max_time, rng, num_trials=50):
    """
    Local search: try random perturbations around the current best and keep improvements.
    Uses structured perturbations (block shifts, frequency modifications).
    """
    a = np.clip(a, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a * (S / s)

    start = time.monotonic()
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    evaluated = 0

    while time.monotonic() - start < max_time and evaluated < num_trials:
        evaluated += 1
        # Try different perturbation types
        trial_type = rng.integers(0, 4)

        if trial_type == 0:
            # Small Gaussian noise
            candidate = a + rng.normal(0, 0.005 * S / np.sqrt(n), n)
        elif trial_type == 1:
            # Block modification: change a random contiguous block
            block_len = rng.integers(n // 64, n // 8)
            start_idx = rng.integers(0, n - block_len)
            candidate = a.copy()
            candidate[start_idx:start_idx + block_len] *= rng.uniform(0.5, 1.5)
        elif trial_type == 2:
            # Frequency domain modification
            A = fft(a, 1 << (2 * n).bit_length())
            # Slightly modify a random frequency bin
            k = rng.integers(1, n)
            A[k] *= rng.uniform(0.9, 1.1)
            A[-k] = np.conj(A[k])
            candidate = np.real(ifft(A))[:n]
        else:
            # Pair swap: swap intensities between two random positions
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            candidate = a.copy()
            candidate[[i, j]] = a[[j, i]]

        candidate = np.maximum(candidate, 0.0)
        s_cand = candidate.sum()
        if s_cand < 1e-15:
            continue
        candidate = candidate * (S / s_cand)

        c = fft_convolve_self(candidate)
        score = c.max()
        if score < best_score - 1e-10:
            best_score = score
            best_a = candidate.copy()

    return best_a


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses phased approach: diverse exploration, focused exploitation, fine polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    inits = []

    # Use parent sequence if available
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0, 1000)))
    except Exception:
        pass

    if inits:
        main_len = len(inits[0][1])
    else:
        main_len = 8192

    n = main_len
    tt = np.linspace(0, 1, n, endpoint=False)
    S = np.sqrt(2.0 * n)

    # Diverse structured initializations
    # Raised cosine (smooth, bounded)
    inits.append(("raised_cos", 1.0 + 0.5 * np.cos(2.0 * np.pi * tt)))

    # Multi-sine with incommensurate frequencies
    inits.append(("multi_sin",
                  1.0 + 0.25 * np.sin(2 * np.pi * 3 * tt)
                  + 0.20 * np.cos(2 * np.pi * 7 * tt + 0.3)
                  + 0.15 * np.sin(2 * np.pi * 13 * tt + 1.1)
                  + 0.10 * np.cos(2 * np.pi * 5 * tt + 2.0)))

    # Gabor-like: modulated Gaussian
    inits.append(("gabor",
                  np.exp(-0.5 * ((tt - 0.5) * 4.0) ** 2) * (1.0 + 0.3 * np.cos(2 * np.pi * 10 * tt))
                  + 0.05))

    # Random with floor
    inits.append(("rand_floor", np.maximum(rng.random(n) * 0.8 + 0.2, 0.01)))

    # "Zeno" pattern: sum of decaying harmonics
    zeno = np.ones(n)
    for h in range(1, 20):
        zeno += (0.5 ** (h // 3)) * np.cos(2 * np.pi * h * tt + h * 0.7)
    inits.append(("zeno", np.maximum(zeno, 0.01)))

    # Bump with oscillation
    inits.append(("bump_osc",
                  np.exp(-0.5 * ((tt - 0.5) * 6.0) ** 2) * (1.0 + 0.4 * np.cos(2 * np.pi * 17 * tt))
                  + 0.05))

    # Alternative length for diversity
    n2 = n * 2 if n <= 8192 else n // 2
    if n2 != n and 2048 <= n2 <= 32768:
        tt2 = np.linspace(0, 1, n2, endpoint=False)
        inits.append(("rand_alt_len", np.maximum(rng.random(n2) * 0.8 + 0.2, 0.01)))
        inits.append(("sin_alt_len",
                      1.0 + 0.3 * np.sin(2 * np.pi * 5 * tt2)
                      + 0.15 * np.cos(2 * np.pi * 11 * tt2 + 0.5)))

    n_inits = len(inits)
    t_start = time.monotonic()

    # === Phase 1: Diverse Exploration (50% of time) ===
    phase1_time = 0.50 * (deadline - t_start)
    per_init = max(2.0, phase1_time / n_inits)

    print("=== Phase 1: Exploration (%d inits, ~%.1f sec each, %.0f sec total) ==="
          % (n_inits, per_init, phase1_time))

    candidates = []

    for name, arr in inits:
        if time.monotonic() > deadline - phase1_time * 0.3:
            break
        n_i = len(arr)
        if n_i < 128:
            continue
        S_i = np.sqrt(2.0 * float(n_i))
        t0 = time.monotonic()

        # Primary: Nesterov annealed (broad to sharp k)
        seg1 = per_init * 0.55
        a1, score1 = run_nesterov_annealed(arr, n_i, S_i, seg1, rng)
        a1 = safe_normalize(a1, n_i)
        c1 = fft_convolve_self(a1)
        s1 = c1.max()
        candidates.append((name + "_nesterov", a1, s1, n_i))
        if s1 < global_best:
            global_best = s1
            global_a = a1.copy()
            print("  Nesterov-ann [%s] n=%d: %.6f" % (name, n_i, global_best))

        # Secondary: FISTA LSE from the Nesterov result
        rem = per_init - (time.monotonic() - t0)
        if rem > 3:
            a2, score2 = run_fista_lse(a1, n_i, S_i, rem, rng,
                                        beta_init=3.0, beta_max=200.0)
            a2 = safe_normalize(a2, n_i)
            c2 = fft_convolve_self(a2)
            s2 = c2.max()
            candidates.append((name + "_fista", a2, s2, n_i))
            if s2 < global_best:
                global_best = s2
                global_a = a2.copy()
                print("  FISTA-LSE    [%s] n=%d: %.6f" % (name, n_i, global_best))

    # Keep top candidates for later
    candidates.sort(key=lambda x: x[2])
    print("=== After Phase 1: best=%.6f, top-3 scores: %s ==="
          % (global_best, ", ".join("%.4f" % c[2] for c in candidates[:3])))

    # === Phase 2: Focused Exploitation (35% of time) ===
    phase2_time = 0.35 * (budget_s - 20.0)
    time_for_phase2 = max(10.0, min(phase2_time, deadline - time.monotonic() - 60))

    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))

        print("=== Phase 2: Exploitation (%.0f sec, n=%d) ===" % (time_for_phase2, n_best))

        # 2a: Long Nesterov exploit with small k
        seg = time_for_phase2 * 0.35
        a_exp, s_exp = run_nesterov_exploit(global_a, n_best, S_best, seg, rng,
                                             k_init=1, k_max=8)
        a_exp = safe_normalize(a_exp, n_best)
        c_exp = fft_convolve_self(a_exp)
        if c_exp.max() < global_best:
            global_best = c_exp.max()
            global_a = a_exp.copy()
            print("  Exploit Nesterov k=1-8: %.6f" % global_best)

        # 2b: FISTA high-beta polish
        seg = time_for_phase2 * 0.30
        a_fis, s_fis = run_fista_lse(global_a, n_best, S_best, seg, rng,
                                      beta_init=20.0, beta_max=500.0)
        a_fis = safe_normalize(a_fis, n_best)
        c_fis = fft_convolve_self(a_fis)
        if c_fis.max() < global_best:
            global_best = c_fis.max()
            global_a = a_fis.copy()
            print("  FISTA high-beta: %.6f" % global_best)

        # 2c: Multiplicative weights
        seg = time_for_phase2 * 0.20
        a_mul, s_mul = run_multiplicative(global_a, n_best, S_best, seg, rng, top_k=20)
        a_mul = safe_normalize(a_mul, n_best)
        c_mul = fft_convolve_self(a_mul)
        if c_mul.max() < global_best:
            global_best = c_mul.max()
            global_a = a_mul.copy()
            print("  Multiplicative: %.6f" % global_best)

        # 2d: Nesterov exploit high-k (smoother, different geometry)
        seg = time_for_phase2 * 0.15
        a_hk, s_hk = run_nesterov_exploit(global_a, n_best, S_best, seg, rng,
                                           k_init=16, k_max=128)
        a_hk = safe_normalize(a_hk, n_best)
        c_hk = fft_convolve_self(a_hk)
        if c_hk.max() < global_best:
            global_best = c_hk.max()
            global_a = a_hk.copy()
            print("  Nesterov high-k: %.6f" % global_best)

    # === Phase 3: Final Polishing and Local Search (15% of time) ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))

        print("=== Phase 3: Final polish (%.0f sec remaining) ===" % remaining)

        # 3a: Very long Nesterov with k=1 (sharpest)
        seg = remaining * 0.35
        a_fin, s_fin = run_nesterov_exploit(global_a, n_best, S_best, seg, rng,
                                             k_init=1, k_max=4)
        a_fin = safe_normalize(a_fin, n_best)
        c_fin = fft_convolve_self(a_fin)
        if c_fin.max() < global_best:
            global_best = c_fin.max()
            global_a = a_fin.copy()
            print("  Final Nesterov k=1-4: %.6f" % global_best)

        # 3b: Perturbation-based local search
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_pert = perturbation_search(global_a, n_best, S_best, remaining * 0.4, rng,
                                          num_trials=100)
            a_pert = safe_normalize(a_pert, n_best)
            c_pert = fft_convolve_self(a_pert)
            if c_pert.max() < global_best:
                global_best = c_pert.max()
                global_a = a_pert.copy()
                print("  Perturbation search: %.6f" % global_best)

        # 3c: One more Nesterov pass from the perturbed result
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_last, s_last = run_nesterov_exploit(global_a, n_best, S_best, remaining * 0.8,
                                                   rng, k_init=1, k_max=3)
            a_last = safe_normalize(a_last, n_best)
            c_last = fft_convolve_self(a_last)
            if c_last.max() < global_best:
                global_best = c_last.max()
                global_a = a_last.copy()
                print("  Final final Nesterov: %.6f" % global_best)

    # === Phase 4: Try alternative-length candidates if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and len(candidates) > 1:
        for name, arr, score, n_i in candidates:
            if n_i == len(global_a):
                continue
            if score < global_best + 0.01 and remaining > 30:
                S_i = np.sqrt(2.0 * float(n_i))
                a_alt, s_alt = run_nesterov_exploit(arr, n_i, S_i, min(30, remaining - 10),
                                                     rng, k_init=1, k_max=8)
                a_alt = safe_normalize(a_alt, n_i)
                c_alt = fft_convolve_self(a_alt)
                if c_alt.max() < global_best:
                    global_best = c_alt.max()
                    global_a = a_alt.copy()
                    print("  Alt-length [%s] n=%d: %.6f" % (name, n_i, global_best))
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break

    # === Verification ===
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb, dtype=np.float64)
        S_fb = np.sqrt(2.0 * n_fb)
        global_a = global_a * (S_fb / n_fb)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 8192

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL: length=%d, internal_score=%.6f, evaluated=%.6f ==="
          % (len(final_list), global_best, ver if np.isfinite(ver) else float('inf')))

    if not np.isfinite(ver):
        print("  WARNING: evaluate_sequence returned inf, using fallback")
        n_fb = 2048
        tt_fb = np.linspace(0, 1, n_fb, endpoint=False)

        S_fb = np.sqrt(2.0 * n_fb)
