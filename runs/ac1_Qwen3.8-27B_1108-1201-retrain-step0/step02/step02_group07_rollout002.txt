"""
LP-based best response dynamics for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal packing directions, combined with
adaptive step sizing, multi-start exploration, and local refinement via
informed annealing. Key insight: the LP finds the direction that maximizes
total mass while respecting the current convolution bound, enabling large
effective steps toward the optimum.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_lp_matrix(a, n):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<=n-1 else 0."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_best_response(a, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (a*g)[k] <= M for all k. Returns g* or None."""
    c = -np.ones(n)
    A_ub = build_lp_matrix(a, n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception:
        pass
    return None


def compute_objective_normalized(a):
    """Compute 2*n*max(conv) for sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def lp_best_response_step(a, t_max=0.5):
    """One LP-based best response step. Returns (new_a, S, M)."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return a.copy(), 1.0, 0.0
    a_norm = a / s
    conv = fast_convolve(a_norm)
    M = conv.max()
    if M < 1e-15:
        return a_norm.copy(), 1.0, M
    g_star = solve_lp_best_response(a_norm, M, n)
    if g_star is None:
        return a_norm.copy(), 1.0, M
    S = g_star.sum()
    if S <= 1.0001:
        return a_norm.copy(), S, M
    # Adaptive step size
    t = min(t_max, 0.3 * (S - 1.0))
    t = max(t, 0.01)
    a_new = (1.0 - t) * a_norm + t * g_star / S
    a_new = np.maximum(a_new, 0.0)
    ns = a_new.sum()
    if ns < 1e-12:
        return a_norm.copy(), S, M
    a_new /= ns
    return a_new, S, M


def run_lp_optimization(a0, time_limit, max_iters=5000, perturb_every=50):
    """Run LP best response iterations with adaptive step and periodic perturbation."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    cur_obj = compute_objective_normalized(a)
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    stall_count = 0
    stall_threshold = 30
    rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        a_new, S, M = lp_best_response_step(a, t_max=0.5)
        new_obj = compute_objective_normalized(a_new)
        
        if new_obj < cur_obj - 1e-10:
            a = a_new
            cur_obj = new_obj
            stall_count = 0
            if cur_obj < best_obj - 1e-8:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            stall_count += 1
            if stall_count >= stall_threshold:
                # Perturb and restart
                stall_count = 0
                stall_threshold = max(20, stall_threshold // 2)
                noise = rng.normal(0, 0.02, n)
                a_pert = np.maximum(a + noise, 0.0)
                ps = a_pert.sum()
                if ps > 1e-10:
                    a_pert /= ps
                    perturb_obj = compute_objective_normalized(a_pert)
                    if perturb_obj < best_obj * 1.05:
                        a = a_pert
                        cur_obj = perturb_obj
                if time.monotonic() - t0 > time_limit * 0.9:
                    break
    
    return best_a, best_obj


def run_gradient_refinement(a0, time_limit, t_init=50.0, eta_init=0.3):
    """Mirror descent refinement on the simplex."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t = t_init
    eta = eta_init
    patience = 0
    max_pat = 40
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        # Softmax subgradient
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        # Compute subgradient
        sg = np.zeros(n)
        thresh = w.max() * 0.005
        active = np.where(w > thresh)[0]
        for ki in active:
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
        
        # Mirror descent update
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            eta *= 0.7
            t *= 1.4
            if eta < 0.005:
                break
    
    return best_a, best_obj


def informed_anneal(a0, time_limit, rng):
    """Informed simulated annealing targeting peak reduction."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.01 * (1.0 - frac) ** 2 + 0.0003
        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.8
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        target_j = int(rng.integers(n))
        if target_j == top_j:
            continue
        delta = a[top_j] * rng.uniform(0.01, 0.05)
        if delta < 1e-10:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
    
    return best_a, best_obj

def generate_initializations(n, rng, parent):
    """Generate diverse initializations for a given length."""
    x = np.arange(n, dtype=np.float64)
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Gaussian
    for sf in [0.15, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits.append(g)
    # Sine patterns
    for freq in [1, 2, 3, 5]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        s /= s.sum()
        inits.append(s)
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits.append(bi)
    # Chirp
    chirp = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * x / (2.0 * n)))
    chirp = np.maximum(chirp, 0.0)
    if chirp.sum() > 1e-10:
        chirp /= chirp.sum()
        inits.append(chirp)
    # Random Dirichlet
    for alpha in [0.5, 1.0, 3.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)
    # Parent interpolation
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0).copy()
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits.append(pp)
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits.append(pp)
    return inits

        if time.monotonic() < deadline - 600:
            a_cur, _ = run_gradient_refinement(a_cur, time_limit=sub4, t_init=200.0, eta_init=0.1)
            record(a_cur)
    
    print("  Phase 2 best: %.6f" % best_score)
    
    # Phase 3: Informed annealing on best candidate
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("Phase 3: Informed annealing...")
        phase3_budget = min(400.0, deadline - time.monotonic() - 80)
        n_anneal = 3
        anneal_per = phase3_budget / n_anneal
        
        for i in range(n_anneal):
            if time.monotonic() > deadline - 80:
                break
            a_start = best_sequence.copy()
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
            rng_a = np.random.default_rng(seed + 7000 + i * 137)
            a_opt, _ = informed_anneal(a_start, time_limit=anneal_per, rng=rng_a)
            record(a_opt)
            # Follow up with gradient
            if time.monotonic() < deadline - 80:
                rem = min(30.0, deadline - time.monotonic() - 80)
                if rem > 5:
                    a_opt2, _ = run_gradient_refinement(a_opt, time_limit=rem, t_init=100.0, eta_init=0.15)
                    record(a_opt2)
        
        print("  Phase 3 best: %.6f" % best_score)
    
    # Phase 4: Try longer sequences if time permits
    if time.monotonic() < deadline - 300:
        print("Phase 4: Longer sequence exploration...")
        for n_try in [2048, 4096]:
            if time.monotonic() > deadline - 200:
                break
            a0 = np.ones(n_try) / n_try
            # Start from parent if available
            if parent is not None:
                pp = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    pp /= pp.sum()
                    a0 = pp
            
            t_alloc = min(250.0, deadline - time.monotonic() - 200)
            if t_alloc < 20:
                break
            print("  Trying n=%d with %.0fs" % (n_try, t_alloc))
            a_opt, _ = run_lp_optimization(a0, time_limit=t_alloc * 0.6, max_iters=1000)
            record(a_opt)
            if time.monotonic() < deadline - 100:
                rem = min(t_alloc * 0.4, deadline - time.monotonic() - 100)
                if rem > 5:
                    a_opt2, _ = run_gradient_refinement(a_opt, time_limit=rem, t_init=30.0, eta_init=0.3)
                    record(a_opt2)
    
    # Phase 5: Final polish
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("Phase 5: Final polish...")
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_best = best_sequence.copy()
            a_best = np.maximum(a_best, 0.0)
            a_best /= a_best.sum()
            
            half1 = remaining * 0.4
            a_opt, _ = run_lp_optimization(a_best, time_limit=half1, max_iters=5000, perturb_every=20)
            record(a_opt)
            
            if time.monotonic() < deadline - 5:
                rem2 = deadline - time.monotonic() - 5
                if rem2 > 5:
                    a_opt2, _ = run_gradient_refinement(a_best, time_limit=rem2, t_init=500.0, eta_init=0.05)
                    record(a_opt2)
    
    # Fallback
    if best_sequence is None:
        a_fb = np.ones(256) / 256
        best_sequence = a_fb
        best_score = safe_evaluate(best_sequence)
    
    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))
    
    return [float(x) for x in best_sequence]
