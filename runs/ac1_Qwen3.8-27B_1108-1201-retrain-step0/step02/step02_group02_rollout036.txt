"""
Multi-phase optimization for minimizing autocorrelation peak of non-negative step functions.
Uses smooth log-sum-exp objective with L-BFGS-B, direct subgradient with projection,
and coordinate descent polish. Focuses on n=384 with secondary exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar, minimize


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score for a sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def smooth_obj_and_grad(theta, tau, n):
    """
    Compute smooth objective and gradient in theta-space.
    a = softmax(theta), score = 2n * smooth_max(a*a)
    """
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = tau * (c - c_max)
    exp_shifted = np.exp(np.clip(shifted, -500, 500))
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    smooth_max = c_max + np.log(max(sum_exp, 1e-300)) / tau
    obj = 2.0 * n * smooth_max
    
    # Gradient of smooth_max w.r.t. a
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w, size)
    part2 = irfft(fa * fw, size)[:n]
    fa_flip = rfft(a[::-1], size)
    part1 = irfft(fa_flip * fw, size)[n-1:2*n-1]
    grad_a = part1 + part2
    
    # Transform to theta-space: grad_theta_j = 2n * a_j * (grad_a_j - a·grad_a)
    mean_grad = np.dot(a, grad_a)
    grad_theta = 2.0 * n * a * (grad_a - mean_grad)
    
    return obj, grad_theta


def make_init(n, rng, method):

            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.05 * rng.standard_normal(n)

        a = np.ones(n)

            end = int((b + 1) * n // n_blocks)

    elif method == 'alternating':
        a = np.ones(n)
        a[::2] = rng.uniform(0.5, 1.5) * np.ones(n // 2 + 1)
        a[1::2] = rng.uniform(0.5, 1.5) * np.ones(n // 2)
    else:
        a = np.ones(n)
    return np.clip(a, 0.01, None)

def subgradient_step(a, S_target, h):
    """
    One step of projected subgradient method.
    Minimize max(a*a) with constraints sum(a) = S_target, a >= 0.
    """
    n = len(a)
    c = fft_convolve(a, a)
    
    # Find peaks (within tolerance)
    c_max = np.max(c)
    tol = 1e-10 * c_max
    peak_indices = np.where(c >= c_max - tol)[0]
    
    # Compute subgradient (sum over active peaks)
    g = np.zeros(n)
    for k in peak_indices:
        for j in range(n):
            val = 0.0
            if 0 <= k - j < n:
                val += a[k - j]
            if 0 <= j - k < n:
                val += a[j - k]
            g[j] += 2.0 * val
    
    # Normalize subgradient (optional, for step size)
    g_norm = np.linalg.norm(g)
    if g_norm < 1e-15:
        return a.copy(), c_max
    
    # Step
    p = a - h * g
    
    # Project onto {x: sum(x) = S_target, x >= 0}
    # Find t such that sum(max(p_i - t, 0)) = S_target
    lo = np.min(p) - S_target - 1
    hi = np.max(p) + 1
    
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(p - mid, 0))
        if s > S_target:
            lo = mid
        else:
            hi = mid
    
    t_proj = (lo + hi) / 2
    a_new = np.maximum(p - t_proj, 0)
    
    # Ensure exact sum (numerical correction)
    diff = S_target - np.sum(a_new)
    if np.sum(a_new) > 0:
        a_new += diff / n  # Simple correction
    
    # Re-clip
    a_new = np.clip(a_new, 0, 1000)
    
    return a_new, c_max

            if time.monotonic() - t0 > time_limit:
                break

                continue
            
            # Try geometric sequence of values
            best_val = base
            best_s = current_score
            
            for frac in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]:

                a[j] = trial
                new_score = compute_score(a)

                    best_s = new_score
                    best_val = trial
            
            if best_val != base:
                a[j] = best_val
                current_score = best_s
                improved = True
                if current_score < best_score:
                    best_score = current_score
                    best = a.copy()
            else:
                a[j] = base
        
        if not improved:
            # Random perturbation
            for _ in range(3):
                j = int(rng.integers(0, n))

                a = a * (S_target / np.sum(a))
                new_score = compute_score(a)
                if new_score < current_score * 1.01:
                    current_score = new_score
        
        iteration += 1
        if iteration % 10 == 0:
            print(f"    CD iter={iteration}, cur={current_score:.6f}, best={best_score:.6f}")
    
    return best, best_score
