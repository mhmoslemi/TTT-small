LP-based block coordinate descent with spectral peak reduction,
mirror descent polishing, and simulated annealing for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_norm(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def compute_conv_matrix(f, n):
    """Build the convolution constraint matrix A where A[k,j] = f[k-j] if valid."""
    # A is (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = f[k-j]
    return A

def solve_lp(f, M, n):
    """Solve the LP: max sum(g) s.t. g>=0, conv(f,g) <= M componentwise."""
    # Build constraint matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je+1] = f[k-je:k-js+1][::-1]
    
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.status != 0:
        return None
    return result.x

def lp_iteration(f, max_iters=200, time_limit=300, t_schedule='adaptive'):

        M = conv.max()
        g = solve_lp(a, M, n)
        if g is None or g.sum() < 1e-12:

        conv = fast_convolve(a)

def pgd_optimize(a0, time_limit, t_init=30.0, lr_init=0.08):
    """Projected gradient descent."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_init
    lr = lr_init
    patience = 0
    max_pat = 40
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        sg = np.zeros(n)
        thresh = w.max() * 0.002
        active = np.where(w > thresh)[0]
        for ki in active:
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                sg[js:je+1] += w[k] * a[k-je:k-js+1][::-1]
        
        a_new = a - lr * sg
        # Project onto simplex
        a_new = project_simplex(a_new)
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            lr *= 0.7
            t *= 1.2
            if lr < 0.002:
                lr = lr_init * 0.5
                t = t_init * 2.0
    
    return best_a, best_obj

def project_simplex(v):
    """Project vector onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def anneal_informed(a0, time_limit, rng):
    """Informed simulated annealing targeting peak reduction."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.015 * (1.0 - frac) ** 2 + 0.0005
        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.75
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])
        k = int(rng.choice(peak_positions))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je+1]
        seg_b = a[k-je:k-js+1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        probs = products / total_prod
        top_j = js + int(rng.choice(je - js + 1, p=probs))
        target_j = int(rng.integers(n))
        for _ in range(5):
            candidate = int(rng.integers(n))
            contrib = 0.0
            for pk in peak_positions[:8]:
                d = pk - candidate
                if 0 <= d <= n - 1:
                    contrib += a[d]
                d2 = candidate - pk + 2*n - 2
                if 0 <= d2 <= n - 1:
                    contrib += a[d2]
            if contrib < a.mean() * 2.0:
                target_j = candidate
                break
        if target_j == top_j:
            continue
        delta = a[top_j] * rng.uniform(0.01, 0.06)
        if delta < 1e-9:
            continue
        a[top_j] -= delta
        a[target_j] += delta
        if a[top_j] < 0:
            a[top_j] += delta
            a[target_j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - cur_obj
        if delta_obj < 0:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            if rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                cur_obj = new_obj
                conv = new_conv
            else:
                a[top_j] += delta
                a[target_j] -= delta
    
    return best_a, best_obj

def spectral_peak_reduction(a0, time_limit, n_iter=500):
    """Reduce peaks in the autocorrelation using spectral domain operations."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Find peak positions
        max_c = conv.max()
        peak_mask = conv > max_c * 0.85
        peak_freq = np.fft.rfft(peak_mask.astype(np.float64))
        
        # In frequency domain, reduce components that contribute to peaks
        A = np.fft.rfft(a, 2*n-1)
        power = np.abs(A) ** 2
        # Reduce power at frequencies that correlate with peaks
        # This is a heuristic: reduce overall power at high frequencies
        # where the peak structure lives
        # Actually, let's do a simpler approach: 
        # Compute the "peak-inducing" frequency components and reduce them
        
        # Alternative: use the subgradient approach in frequency domain
        # grad of peak with respect to a is 2 * sum_k w_k * shift(a, k)
        # In frequency domain, this is 2 * FFT(w) * FFT(a) (approximately)
        
        # Let's just do a simple approach: reduce the peak by scaling down
        # the components of a that contribute most to the peak
        peak_pos = np.argmax(conv)
        js = max(0, peak_pos - n + 1)
        je = min(n - 1, peak_pos)
        contrib_left = a[js:je+1]
        contrib_right = a[peak_pos-je:peak_pos-js+1][::-1]
        
        # Reduce the product at the peak
        # Scale down the larger contributor
        if contrib_left.max() > contrib_right.max():
            idx = js + np.argmax(contrib_left)
        else:
            idx = peak_pos - je + np.argmax(contrib_right)
        
        # Move mass from the peak contributor to a random low-contribution position
        low_positions = np.where(a < a.mean())[0]
        if len(low_positions) == 0:
            low_positions = np.arange(n)
        
        # Do a small adjustment
        amount = a[idx] * 0.01
        target = low_positions[np.random.default_rng(it).integers(len(low_positions))]
        a[idx] -= amount
        a[target] += amount
        a = np.maximum(a, 0)
        a /= a.sum()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n
    
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_' + str(sf)] = g
    
    for freq in [1, 2, 3, 5]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        s /= s.sum()
        inits['cos_' + str(freq)] = s
    
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 8)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Random
    for alpha in [0.5, 1.0, 2.0]:
        inits['rand_' + str(alpha)] = rng.dirichlet(np.ones(n) * alpha)
    
    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0).copy()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            pp /= pp.sum()
            inits['parent'] = pp
    
    return inits

"""
LP-based block coordinate descent with multiple polishers for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key innovation: solve LP to find optimal update direction at each step,
combined with mirror descent, PGD, and annealing for robustness.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_norm(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f, n):
    """Build dense convolution constraint matrix A where A[k,j] = f[k-j] if valid."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je+1] = f[k-je:k-js+1][::-1]
    return A

def solve_lp(f, M, n, time_limit=3.0, peak_indices=None):
    """
    Solve LP: max sum(g) s.t. g>=0, conv(f,g)[k] <= M for k in peak_indices (or all k).
    Returns optimal g or None.
    """
    A = build_conv_matrix(f, n)
    if peak_indices is not None:
        A = A[peak_indices, :]
        b_ub = np.full(len(peak_indices), M)
    else:
        b_ub = np.full(2*n-1, M)
    
    c = -np.ones(n)
    bounds = [(0.0, M * n)] * n  # Safety upper bound
    
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.status == 0:
            return result.x
        elif result.status == 3:  # Time limit
            # Return best found so far (might be infeasible)
            if hasattr(result, 'x') and result.x is not None:
                return np.maximum(result.x, 0)
    except Exception:
        pass
    return None

def lp_iterate(f, time_limit, peak_frac=0.8, use_sparse_constraint=False, n_peak_max=50):
    """
    LP-based block coordinate descent.
    At each step: solve LP for optimal direction, line search, update.
    """
    a = np.maximum(f, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    stall_count = 0
    max_stall = 30
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        
        # Optionally use sparse constraint (only peak positions)
        if use_sparse_constraint:
            thresh = M * peak_frac
            peak_idx = np.where(conv >= thresh)[0]
            if len(peak_idx) > n_peak_max:
                # Keep only the top peaks
                peak_idx = np.argsort(conv)[-n_peak_max:]
        else:
            peak_idx = None
        
        g = solve_lp(a, M, n, time_limit=2.0, peak_indices=peak_idx)
        if g is None or g.sum() < 1e-12:
            break
        
        g = np.maximum(g, 0)
        g /= g.sum()
        
        # Line search
        improved = False
        for t in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
            candidate = (1 - t) * a + t * g
            candidate = np.maximum(candidate, 0)
            cs = candidate.sum()
            if cs < 1e-12:
                continue
            candidate /= cs
            cval = fast_convolve(candidate)
            cobj = 2.0 * n * cval.max()
            if cobj < best_obj - 1e-12:
                a = candidate
                conv = cval
                best_obj = cobj
                best_a = a.copy()
                improved = True
                stall_count = 0
                break
        
        if not improved:
            stall_count += 1
            if stall_count >= max_stall:
                break
    
    return best_a, best_obj
