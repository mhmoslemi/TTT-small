def fft_convolve(a, b):

    n, m = len(a), len(b)
    size = 1

        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

Let me estimate: for a perfect flat spectrum, the time-domain signal would be a sinc function (or Dirichlet kernel for finite FFT). The negative lobes of the sinc are the problem. The Gerchberg-Saxton method tries to minimize these negative lobes while maintaining flatness.

"""
Optimization of non-negative step functions to minimize peak autocorrelation.
Uses FFT-based phase retrieval for initialization, Adam gradient descent with
exp-parameterization and soft-max smoothing, and local search polish.
Key innovation: correct gradient computation accounting for both terms in
d(conv)/da, and larger sequence lengths for better theoretical bounds.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def fft_phase_init(n, n_iter=200, rng=None, soft_alpha=0.3):
    """Gerchberg-Saxton with soft projection."""
    if rng is None:
        rng = np.random.default_rng()
    
    N = 1
    while N < 2*n:
        N <<= 1
    
    # Random initialization
    a = rng.standard_normal(n)
    a = np.maximum(a, 0.1)
    a = a / np.sum(a)
    
    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        
        # Target: flat magnitude
        a_new = np.fft.irfft(phase, N)[:n]
        
        # Soft non-negativity: blend with clipped version
        a_clip = np.maximum(a_new, 0)
        s = np.sum(a_clip)
        if s < 1e-10:
            a_clip = np.maximum(a_new, 0.01)
            s = np.sum(a_clip)
        a_clip = a_clip / s
        
        # Blend (more soft projection as we iterate)
        alpha = soft_alpha * (0.5 + 0.5 * it / n_iter)
        a = alpha * a_clip + (1 - alpha) * a_new
        a = np.maximum(a, 1e-8)
        a = a / np.sum(a)
    
    return a

def gradient_step(theta, tau, N):
    """Compute gradient of the soft-score w.r.t. theta."""
    n = len(theta)
    a = np.exp(theta)
    
    # Soft-max of convolution
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    w = e / np.sum(e)
    H = c_max + tau * np.log(np.sum(e))
    
    # Gradient of H w.r.t. a
    # g_a[i] = (2/tau) * (R[(-i)%N] + C[i])
    # where R = cross-correlation of a and w, C = convolution of w and a
    A_f = np.fft.rfft(a, N)
    W_f = np.fft.rfft(w, N)
    
    # Cross-correlation: R[l] = sum_j a[j]*w[j-l]
    R = np.fft.irfft(A_f * np.conj(W_f), N)
    # Convolution: C[i] = (w*a)[i]
    C = np.fft.irfft(W_f * A_f, N)[:2*n-1]
    
    # g_a[i] = (2/tau) * (R[(-i)%N] + C[i])
    # For i=0: R[0] + C[0]
    # For i>0: R[N-i] + C[i]
    R_part = np.zeros(n)
    R_part[0] = R[0]
    R_part[1:] = R[N-1 : N-1+n-1][::-1]  # R[N-1], R[N-2], ..., R[N-(n-1)]
    
    g_a = (2.0 / tau) * (R_part + C[:n])
    
    # Full objective gradient
    S = np.sum(a)
    # f = 2n * H / S^2
    # df/da[i] = 2n * (g_a[i] * S - 2*H) / S^3
    f_a = 2.0 * n * (g_a * S - 2.0 * H) / (S ** 3)
    
    # Chain rule for theta: a[i] = exp(theta[i])
    # (we normalize theta to have mean 0, so effectively a[i] = exp(theta[i] - mean(theta)))
    # da[i]/dtheta[j] = a[i]*(delta_ij - 1/n)
    inner = np.sum(f_a * a) / n
    g_theta = a * (f_a - inner)
    
    return g_theta, H, S

def optimize_gradient(a_init, time_budget, rng, tau_start=None, tau_end=None):
    """Adam gradient descent with exp parameterization."""
    n = len(a_init)
    N = 1
    while N < 2*n:
        N <<= 1
    
    a0 = np.maximum(a_init, 1e-10)
    theta = np.log(a0)
    theta = theta - np.mean(theta)
    
    if tau_start is None:
        tau_start = n * 0.05
    if tau_end is None:
        tau_end = n * 0.001
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    step = 0
    
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_budget)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        
        g_theta, H, S = gradient_step(theta, tau, N)
        
        step += 1
        m = b1 * m + (1 - b1) * g_theta
        v = b2 * v + (1 - b2) * g_theta ** 2
        m_hat = m / (1 - b1 ** step)
        v_hat = v / (1 - b2 ** step)
        
        # Learning rate schedule
        lr = 0.5 / np.sqrt(1 + 0.05 * step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        theta = theta - np.mean(theta)  # re-normalize
        
        # Periodic evaluation
        if step % 100 == 0:
            a_check = np.exp(theta)
            s_check = compute_score(a_check)
            if s_check < best_score:
                best_score = s_check
                best_a = a_check.copy()
    
    # Final evaluation
    a_final = np.exp(theta)
    s_final = compute_score(a_final)
    if s_final < best_score:
        best_score = s_final
        best_a = a_final.copy()
    
    return best_a, best_score

def local_search_polish(a_init, time_budget, rng, top_k=5):
    """Simulated annealing with peak-directed perturbations."""
    current = a_init.astype(np.float64).copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9995
    
    while time.monotonic() - t0 < time_budget:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.3:
            # Peak-directed: reduce elements contributing to peak lags
            conv = fft_convolve(current, current)
            n_top = min(top_k, len(conv))
            top_lags = np.argsort(conv)[-n_top:]
            contrib = np.zeros(n)
