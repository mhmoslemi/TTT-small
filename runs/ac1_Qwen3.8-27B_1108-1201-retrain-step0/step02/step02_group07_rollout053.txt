def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def conv_grad_weighted(a, weights):
    """
    Compute sum_k weights[k] * grad(conv[k])/grad(a)
    where grad(conv[k])/grad(a)[i] = 2*a[k-i] for valid i.
    """
    n = len(a)
    size = 2 * n - 1
    grad = np.zeros(n)
    for k in range(size):
        if weights[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # grad[i] += weights[k] * 2 * a[k-i] for i in [js, je]
        # a[k-i] for i in [js, je] is a[ks] for ks in [k-je, k-js]
        ks_start = k - je
        ks_end = k - js
        grad[js:je+1] += weights[k] * 2.0 * a[ks_start:ks_end+1]
    return grad

def softmax_from_z(z):

    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B on softmax-parameterized log-sum-exp smoothed auto-convolution peak
2. Exact coordinate descent (exploiting convexity of 1D subproblems)
3. Adaptive active-set refinement and local search
Uses parent sequence as primary starting point with multiple polish stages.
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def conv_grad_weighted(a, weights):
    """
    Compute sum_k weights[k] * grad(conv[k])/grad(a).
    grad(conv[k])/grad(a)[i] = 2*a[k-i] for i in valid range for lag k.
    """
    n = len(a)
    size = 2 * n - 1
    grad = np.zeros(n)
    for k in range(size):
        wk = weights[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        ks_start = k - je
        ks_end = k - js
        grad[js:je + 1] += wk * 2.0 * a[ks_start:ks_end + 1]
    return grad


def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def obj_grad_z(z, K, beta):
    """Objective and gradient w.r.t. z for softmax-parameterized problem."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    n = len(a)

    size = len(conv)
    if K >= size:
        top_idx = np.argsort(-conv)
    else:
        top_idx = np.argpartition(-conv, K)[:K]
    top_vals = conv[top_idx]

    scaled = beta * top_vals
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    logsumexp_val = -(1.0 / beta) * np.log(exp_scaled.sum())

    w = exp_scaled / exp_scaled.sum()
    weights = np.zeros(size)
    weights[top_idx] = w

    grad_a = conv_grad_weighted(a, weights)
    grad_z = grad_a - a * (a @ grad_a)

    obj = n * 2.0 * logsumexp_val
    return obj, grad_z

def lbfgs_optimize(a0, time_limit, K=50, beta=30):
    """
    L-BFGS-B optimization on softmax parameterization with adaptive K and beta.
    Multiple rounds with increasing sharpness.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()

    z = np.log(a + 1e-30)
    z -= z.mean()

    best_a = a.copy()
    best_obj = safe_evaluate(a)
    t0 = time.monotonic()

    n_rounds = 6
    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # Adaptive parameters: sharper max approximation over rounds
        cur_beta = beta * (1.0 + 0.8 * rnd)
        cur_K = min(K + 15 * rnd, 2 * n - 1)
        time_per = min(remaining / max(1, n_rounds - rnd), 150.0)

        def fun(z_local):
            return obj_grad_z(z_local, cur_K, cur_beta)[0]

        def jac(z_local):
            return obj_grad_z(z_local, cur_K, cur_beta)[1]

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': 300, 'maxfun': 10000, 'ftol': 1e-14, 'gtol': 1e-10}
            )
            z = result.x
        except Exception:
            break

        a = softmax_from_z(z)
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_obj:
            best_obj = score
            best_a = a.copy()

    return best_a


def coordinate_descent(a0, time_limit, n_sweeps=200):
    """
    Exact coordinate descent. For each coordinate i, minimize the (convex)
    1D function max_k conv[k](a[i]) exactly using scipy minimize_scalar.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    a = a / a.sum()

    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    # Precompute for each coordinate i, which k values involve a[i]
    # and what the coefficients are
    # conv[k] involves a[i] when 0<=k-i<n (i.e., k in [i, i+n-1] intersect [0, 2n-2])
    # The linear coefficient of a[i] in conv[k] (for k != 2i) is 2*a[k-i]
    # For k = 2i, the quadratic term is a[i]^2

    no_improve_sweeps = 0
    max_no_improve = 10

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        # Random order for diversity
        order = np.random.default_rng(sweep).permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Determine the 1D function: max of {a[i]^2 + c_{2i}, 2*a[i]*a[k-i] + c_k, c_k}
            # For k in [max(0,i), min(2n-2, i+n-1)]:
            #   if k == 2*i: quadratic term a[i]^2 + (conv[k] - a[i]^2)
            #   else: linear term 2*a[i]*a[k-i] + (conv[k] - 2*a[i]*a[k-i])

            # Compute current conv for reference
            # Actually, let's just evaluate the 1D function numerically
            # f(a_i) = max_k conv[k] when we set a[i] = a_i and renormalize
            # But renormalization complicates things. Instead, work with unnormalized.
            # Since the objective is scale-invariant, we can minimize max_k conv[k]
            # with the constraint sum(a)=1. If we change a[i], we need to renormalize.
            # Alternatively, minimize max_k conv[k] / (sum(a))^2 but that's more complex.
            # Simplest: just minimize max_k conv[k] subject to sum(a)=1, a>=0.
            # When we change a[i] to v, we set a_new = a, a_new[i] = v, then renormalize.
            # But this changes all other coordinates slightly. For coordinate descent to work,
            # we should minimize max_k conv[k] with a[i] free and all else fixed,
            # then renormalize. The renormalization is a scaling that doesn't change
            # the argmax of conv (it scales all values equally).
            # So: minimize max_k conv[k](a with a[i]=v) over v >= 0, then renormalize.

            # For efficiency, precompute the "base" conv without a[i] contribution
            # conv[k] = base[k] + a[i]*coeff[k] + (quadratic if k==2i)
            # where coeff[k] = 2*a[k-i] if 0<=k-i<n and k!=2i, else 0
            # and quadratic part: a[i]^2 if k==2i

            a_i = a[i]
            # Remove a[i] contribution from all conv values
            base = fast_convolve(a)  # current conv
            # Subtract a[i]'s contribution
            # a[i] contributes to conv[k] as: 2*a[i]*a[k-i] (or a[i]^2 if k=2i)
            # Actually it's the full conv, so to get "base" we subtract:
            # For each k where a[i] appears:
            #   if k==2i: subtract a[i]^2
            #   elif 0<=k-i<n: subtract 2*a[i]*a[k-i]
            #   else: no contribution

            # Rather than computing base explicitly, just define f(v) directly
            def f_1d(v):
                a_trial = a.copy()
                a_trial[i] = v
                # Don't renormalize yet - just compute max conv
                conv_trial = fast_convolve(a_trial)
                return conv_trial.max()

            # Bounds for v: [0, a_i * 3] (generous upper bound)
            v_upper = max(a_i * 3.0, a.mean() * 5.0)

            # Use bounded scalar minimization
            try:
                res = minimize_scalar(f_1d, bounds=(0.0, v_upper), method='bounded',
                                      options={'xatol': 1e-12, 'maxiter': 100})
                v_opt = res.x
            except Exception:
                v_opt = a_i

            if abs(v_opt - a_i) > 1e-12:
                a[i] = max(0.0, v_opt)
                # Renormalize
                s = a.sum()
                if s > 1e-15:
                    a /= s
                improved = True

        # Check improvement
        if improved:
            cur_conv = fast_convolve(a)
            cur_obj = 2.0 * n * cur_conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve_sweeps = 0
            else:
                no_improve_sweeps += 1
        else:
            no_improve_sweeps += 1

        if no_improve_sweeps >= max_no_improve:
            break

    return best_a


def local_search_refined(a0, time_limit, rng):
    """
    Refined local search: identifies peak-contributing positions and
    performs targeted mass transfers with line search.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    a = a / a.sum()

    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()

    no_improve = 0
    max_no_improve = 30

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        # Compute peak threshold
        max_c = conv.max()
        peak_thresh = max_c * 0.85
        peaks = np.where(conv >= peak_thresh)[0]

        if len(peaks) == 0:
            peaks = np.array([np.argmax(conv)])

        # Compute contribution of each position to peaks
        contrib = np.zeros(n)
        for k in peaks[:25]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            peak_weight = conv[k] / max_c
            for j in range(js, je + 1):
                ks = k - j
                if 0 <= ks < n:
                    contrib[j] += peak_weight * a[j] * a[ks]

        if contrib.max() < 1e-15:
            break

        # Source: high contributor to peaks
        source_probs = contrib ** 1.5
        source_probs = source_probs / (source_probs.sum() + 1e-30)
        i = int(rng.choice(n, p=source_probs))

        # Target: low contributor, prefer positions that DON'T add to peaks
        # Compute "peak_exposure" of each position
        exposure = np.zeros(n)
        for k in peaks[:25]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                exposure[j] += conv[k] / max_c
        target_probs = 1.0 / (exposure + 1e-8)
        target_probs = target_probs / target_probs.sum()
        j = int(rng.choice(n, p=target_probs))

        if i == j:
            no_improve += 1
            continue

        # Line search: try several transfer fractions
        best_delta = None
        best_obj_local = cur_obj

        for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.25]:
            delta = a[i] * frac
            if delta < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i] -= delta
            a_trial[j] += delta
            if a_trial[i] < 0:
                continue
            conv_trial = fast_convolve(a_trial)
            obj_trial = 2.0 * n * conv_trial.max()
            if obj_trial < best_obj_local:
                best_obj_local = obj_trial
                best_delta = delta

        if best_delta is not None:
            a[i] -= best_delta
            a[j] += best_delta
            if a[i] < 0:
                a[i] = 0.0
                a[j] -= a[i]  # shouldn't happen
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_a, best_obj


def multistart_optimize(a0, time_limit, rng, n_starts=3):
    """
    Multi-start: perturb the starting point and run L-BFGS from each.
    """
    n = len(a0)
    best_a = a0.copy()
    best_score = safe_evaluate(a0)

    for s in range(n_starts):
        if time.monotonic() > time.time() + time_limit:
            break

        # Perturb
        a_pert = a0.copy()
        noise_scale = 0.05 / (s + 1)
        a_pert += rng.standard_normal(n) * noise_scale
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()

        time_per = (time_limit - (time.time() - time.time())) / (n_starts - s)
        time_per = min(120.0, time_limit / (n_starts - s))

        a_opt = lbfgs_optimize(a_pert, time_limit=time_per, K=40, beta=30)
        score = safe_evaluate(a_opt)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_opt.copy()

    return best_a
