```python
def project_simplex_exact(v, S):
    """Exact projection onto {a >= 0, sum(a) = S}."""
    n = len(v)
    # Find tau such that sum(max(v[i]-tau, 0)) = S
    # Sort v in decreasing order
    v_sorted = np.sort(v)[::-1]
    # The optimal tau is determined by the "water filling"
    # If v_sorted[k] > tau > v_sorted[k+1], then:
    # sum_{i=1}^{k} (v_sorted[i-1] - tau) = S
    # k * v_sorted[:k].sum() - k * tau = S
    # tau = (v_sorted[:k].sum() - S) / k
    
    cumsum = np.cumsum(v_sorted)
    # For each possible k, compute tau_k = (cumsum[k-1] - S) / k
    k = np.arange(1, n+1)
    taus = (cumsum - S) / k
    # tau is valid for position k if v_sorted[k-1] > tau_k >= v_sorted[k] (with v_sorted[n]=0)
    # Simplified: find the largest k such that v_sorted[k-1] > (cumsum[k-1] - S) / k
    # i.e., k * v_sorted[k-1] > cumsum[k-1] - S
    # i.e., k * v_sorted[k-1] + S > cumsum[k-1]
    
    valid = k * v_sorted + S > cumsum
    # Find the largest k that is valid
    if not valid[0]:
        tau = S / n  # uniform
    else:
        k_max = np.max(k[valid])
        tau = (cumsum[k_max-1] - S) / k_max
    result = np.maximum(v - tau, 0)
    return result
```

```python
TOTAL_BUDGET = 2390  # seconds (leave 10s margin)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]

def compute_subgradient(a, w):
    """
    Compute g[i] = sum_k w[k] * dC[k]/da[i] where C[k] = sum_j a[j]*a[k-j].
    dC[k]/da[i] = a[k-i] + a[i] if 0<=k-i<n, else 0.

def project_simplex(v, S):
    """Exact projection onto {a >= 0, sum(a) = S}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    k = np.arange(1, n + 1)
    # tau_k = (cumsum[k-1] - S) / k
    # Valid if v_sorted[k-1] > tau_k (i.e., v_sorted[k-1]*k > cumsum[k-1] - S)
    valid = v_sorted * k > cumsum - S
    if not np.any(valid):
        return np.full(n, S / n)
    k_max = np.max(k[valid])
    tau = (cumsum[k_max - 1] - S) / k_max
    return np.maximum(v - tau, 0.0)

def nesterov_subgradient_step(a, S, n, active_width=100, step_scale=1.0):
    """
    One Nesterov subgradient step.
    Returns: new_a, g, c, M
    """
    c = fft_convolve_self(a)
    M = c.max()
    
    # Active set: lags where c[k] >= M - active_width * (something)
    # Use a relative threshold: lags within 1% of max
    threshold = M * (1.0 - 0.05)  # lags within 5% of max
    active = c >= threshold
    if not np.any(active):
        active[np.argmax(c)] = True
    
    w = np.zeros(len(c))
    w[active] = 1.0 / active.sum()
    
    g = compute_subgradient(a, w)
    # Project gradient to tangent space of simplex (sum = 0)
    g = g - np.mean(g)
    
    grad_norm = np.linalg.norm(g)
    if grad_norm < 1e-15:
        return a, g, c, M
    
    # Step size: proportional to step_scale / step iteration (handled externally)
    alpha = step_scale / grad_norm
    # Cap the step to avoid too-large moves
    alpha = min(alpha, 0.1)
    
    a_new = project_simplex(a - alpha * g, S)
    return a_new, g, c, M

            no_improve_count = 0
            if verbose and t % 1000 == 0:
                print("  t=%d, obj=%.8f, improved" % (t, M))
        
        if t % 1000 == 0:
            no_improve_count += 1
        
        # Active set
        threshold = M * (1.0 - 0.02)
        active = c >= threshold
        if active.sum() < 1:
            active[np.argmax(c)] = True
        w = np.zeros(len(c))
        w[active] = 1.0 / active.sum()
        
        # Subgradient
        g = compute_subgradient(a, w)
        g = g - np.mean(g)  # tangent to simplex
        
        grad_norm = np.linalg.norm(g)
        if grad_norm < 1e-15:
            break
        
        # Adaptive step size: decrease over time
        # Use alpha_t = alpha_base / sqrt(t)
        alpha = alpha_base / np.sqrt(t)
        # Scale by problem: roughly S^2 / n gives the right scale
        alpha *= (S * S / n) / (grad_norm * n)
        alpha = min(alpha, 0.5 / grad_norm)
        
        a_old = a.copy()
        a = project_simplex(a - alpha * g, S)
        
        # Nesterov extrapolation
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = beta / t  # simplified: ~2/t for large t
        # Actually the standard Nesterov update is:
        # y_{t+1} = a_{t+1} + (t/(t+1)) * (a_{t+1} - a_t)
        # which means the momentum factor is t/(t+1)
        momentum = t / (t + 1.0)
        y = a + momentum * (a - a_old)
        y = project_simplex(y, S)
        # For next iteration, we could evaluate at y (Nesterov) or at a (standard)
        # For simplicity, we evaluate at a (this gives the standard subgradient rate)
        # For Nesterov rate, we'd evaluate at y:
        # a = y (for the next subgradient evaluation)
        # But this can be unstable, so let's use a mix:
        # a_next = (1-gamma)*a + gamma*y
        # For now, just use a (standard subgradient rate O(1/sqrt(T)))
    
    return best_a, best_obj

def frank_wolfe_step(a, S, n, c, g):
    """Perform a Frank-Wolfe step: move towards the best vertex."""
    i_star = np.argmin(g)
    v = np.zeros(n)
    v[i_star] = S
    
    # Line search: try a few step sizes
    best_t = 0.0
    best_val = objective_value(a)
    
    for t_candidate in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        a_trial = (1 - t_candidate) * a + t_candidate * v
        a_trial = project_simplex(a_trial, S)
        val = objective_value(a_trial)
        if val < best_val:
            best_val = val
            best_t = t_candidate
    
    if best_t > 0:
        a_new = (1 - best_t) * a + best_t * v
        return project_simplex(a_new, S)
    return a

def coordinate_descent(a, S, n, time_budget, max_sweeps=100):
    """Coordinate descent: try reducing each element."""
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            
            # Try reducing a[i] and redistributing
            # Binary search for optimal value
            best_local = orig
            best_local_obj = best_obj
            
            # Try a few values
            for frac in [0.5, 0.7, 0.3, 0.8, 0.2]:
                a_trial = a.copy()
                removed = orig * (1 - frac)
                a_trial[i] = orig * frac
                # Redistribute removed mass to neighbors or uniformly
                a_trial = a_trial + removed / n  # simple uniform redistribution
                a_trial = project_simplex(a_trial, S)
                c_trial = fft_convolve_self(a_trial)
                obj_trial = c_trial.max()
                if obj_trial < best_local_obj:
                    best_local_obj = obj_trial
                    best_local = a_trial.copy()
            
            if best_local is not orig and isinstance(best_local, np.ndarray):
                if best_local_obj < best_obj:
                    a = best_local
                    best_obj = best_local_obj
                    best_a = a.copy()
                    improved = True
                    c = fft_convolve_self(a)
        
        if not improved and sweep > 3:
            break
    
    return best_a, best_obj
```

```python
"""
Minimize 2*n*max(self_conv)/sum(a)^2 for non-negative sequences.
Uses: (1) Nesterov-accelerated projected subgradient with FFT-based O(n log n)
subgradient computation, (2) adaptive active-set width that tightens over time,
(3) backtracking line search for step size, (4) Frank-Wolfe steps for
exploration, (5) coordinate descent for final polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def compute_subgradient(a, w):
    """
    Compute g[i] = sum_k w[k] * dC[k]/da[i].
    dC[k]/da[i] = a[k-i] + a[i] if 0<=k-i<n, else 0.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Part 1: sum_k w[k]*a[k-i] for valid (k,i)
    # = (a_rev * w)[n-1+i]
    a_rev = a[::-1]
    conv = fft_convolve(a_rev, w)
    part1 = np.zeros(n)
    idx = np.arange(n - 1, 2 * n - 1)
    valid = idx < len(conv)
    part1[valid] = conv[idx[valid]]

    # Part 2: a[i] * sum_{k=i}^{i+n-1} w[k]
    w_ext = np.zeros(3 * n)
    lw = len(w)
    w_ext[:lw] = w
    cw = np.concatenate([[0.0], np.cumsum(w_ext)])
    lo = np.arange(n)
    hi = lo + n
    s = cw[hi] - cw[lo]
    part2 = a * s

    return part1 + part2


def project_simplex(v, S):
    """Exact projection onto {a >= 0, sum(a) = S}."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    k = np.arange(1, n + 1)
    valid = v_sorted * k > cumsum - S
    if not np.any(valid):
        return np.full(n, S / n)
    k_max = np.max(k[valid])
    tau = (cumsum[k_max - 1] - S) / k_max
    return np.maximum(v - tau, 0.0)


def objective_value(a):
    """Compute max of self-convolution."""
    c = fft_convolve_self(a)
    return c.max()

def fw_refine(a_init, S, n, time_budget, verbose=False):
    """Frank-Wolfe style refinement: alternate between subgradient and vertex steps."""
    a = a_init.copy().astype(np.float64)
    a = project_simplex(a, S)
    best_obj = objective_value(a)
    best_a = a.copy()

    start = time.monotonic()
    t = 0

    while time.monotonic() - start < time_budget:
        t += 1

        c = fft_convolve_self(a)
        M = c.max()
        if M < best_obj:
            best_obj = M
            best_a = a.copy()
            if verbose and t % 500 == 0:
                print("    FW t=%d, obj=%.8f" % (t, M))

        # Subgradient step
        threshold = M * 0.995
        active = c >= threshold
        if active.sum() < 1:
            active[np.argmax(c)] = True
        w = np.zeros(len(c))
        w[active] = 1.0 / active.sum()

        g = compute_subgradient(a, w)
        g = g - np.mean(g)
        grad_norm = np.linalg.norm(g)
        if grad_norm < 1e-14:
            break

        alpha = min(1.0 / grad_norm, 0.1)
        a_new = project_simplex(a - alpha * g, S)

        # FW step: move toward best vertex
        if t % 20 == 0:
            i_star = np.argmin(g)
            v = np.zeros(n)
            v[i_star] = S
            # Try a few blend ratios
            best_blend = a_new
            best_blend_obj = objective_value(a_new)
            for frac in [0.05, 0.1, 0.2, 0.3]:
                a_blend = (1.0 - frac) * a_new + frac * v
                a_blend = project_simplex(a_blend, S)
                obj_blend = objective_value(a_blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_blend = a_blend
            a = best_blend
        else:
            a = a_new

    return best_a, best_obj


def coordinate_descent_polish(a_init, S, n, time_budget):
    """Coordinate descent: try small perturbations of individual elements."""
    a = a_init.copy().astype(np.float64)
    a = project_simplex(a, S)
    best_obj = objective_value(a)
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(123)

    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        M = c.max()
        if M < best_obj:
            best_obj = M
            best_a = a.copy()

        # Find elements that contribute most to peak lags
        peak_lags = np.where(c >= M * 0.99)[0]

        # For each peak lag, find which elements contribute most
        # and try reducing them
        for k in peak_lags[:5]:
            if time.monotonic() - start > time_budget:
                break
            # Elements contributing to C[k]: a[j]*a[k-j] for valid j
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue
            # Try reducing the largest contributor
            contribs = a[lo:hi + 1] * a[k - hi:k - lo + 1][::-1]
            if len(contribs) == 0:
                continue
            j_local = lo + np.argmax(contribs)
            if a[j_local] < 1e-10:
                continue

            # Try reducing a[j_local] and redistributing
            for frac in [0.3, 0.5, 0.7, 0.1]:
                a_trial = a.copy()
                removed = a_trial[j_local] * (1.0 - frac)
                a_trial[j_local] *= frac
                # Redistribute to neighbors
                nbor = [j_local - 1, j_local + 1, j_local - 2, j_local + 2]
                nbor = [x for x in nbor if 0 <= x < n]
                if not nbor:
                    nbor = list(range(n))
                a_trial[nbor] += removed / len(nbor)
                a_trial = project_simplex(a_trial, S)
                M_trial = objective_value(a_trial)
                if M_trial < M:
                    a = a_trial
                    M = M_trial
                    if M_trial < best_obj:
                        best_obj = M_trial
                        best_a = a.copy()
                    break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Multi-phase: subgradient -> FW refine -> coordinate polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    global_best = np.inf
    global_a = None

    # Load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            n_given = len(given_raw)
            S_given = np.sqrt(2.0 * n_given)
            given_a = project_simplex(given_raw, S_given)
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.6f" % (n_given, ver if np.isfinite(ver) else float('inf')))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: %s" % str(e))

    # Choose n
    if given_a is not None:
        n = len(given_a)
    else:
        n = 4096
    S = np.sqrt(2.0 * float(n))

    if global_a is None:
        global_a = np.full(n, S / n)
        global_best = objective_value(global_a)
        print("Uniform start: obj=%.6f" % global_best)

    print("n=%d, S=%.4f, initial obj=%.8f" % (n, S, global_best))

    # Phase 1: Main subgradient optimization on best starting point
    print("Phase 1: Subgradient optimization")
    t0 = time.monotonic()
    p1_budget = min(800.0, 0.40 * (deadline - time.monotonic()))
    if p1_budget > 10.0:
        a = global_a.copy()
        a = project_simplex(a, S)
        a_best, obj_best = run_subgradient(a, S, n, p1_budget, verbose=True)
        elapsed = time.monotonic() - t0
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print("Phase 1: obj=%.8f (improved, %.1fs)" % (global_best, elapsed))
        else:
            print("Phase 1: no improvement (%.1fs)" % elapsed)

    # Phase 2: Multi-start with diverse initializations
    print("Phase 2: Multi-start exploration")
    remaining = deadline - time.monotonic() - 60.0
    if remaining > 30.0:
        p2_budget = 0.25 * remaining
        inits = ["gaussian", "two_humps", "mixed_freq", "periodic", "anti_peak"]
        per_init = max(5.0, p2_budget / len(inits))
        for name in inits:
            if time.monotonic() > deadline - 60.0:
                break
            a_init = get_initialization(name, n, rng, S)
            t0 = time.monotonic()
            a_best, obj_best = run_subgradient(a_init, S, n, per_init, verbose=False)
            el = time.monotonic() - t0
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print("  %s: %.8f (NEW BEST, %.1fs)" % (name, global_best, el))
            else:
                print("  %s: %.8f (%.1fs)" % (name, obj_best, el))

    # Phase 3: FW refinement
    print("Phase 3: FW refinement")
    remaining = deadline - time.monotonic() - 30.0
    if remaining > 20.0 and global_a is not None:
        p3_budget = min(300.0, 0.30 * remaining)
        a = global_a.copy()
        a = project_simplex(a, S)
        t0 = time.monotonic()
        a_best, obj_best = fw_refine(a, S, n, p3_budget, verbose=True)
        el = time.monotonic() - t0
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print("Phase 3: obj=%.8f (improved, %.1fs)" % (global_best, el))
        else:
            print("Phase 3: no improvement (%.1fs)" % el)

    # Phase 4: Coordinate descent polish
    print("Phase 4: Coordinate descent")
    remaining = deadline - time.monotonic() - 10.0
    if remaining > 5.0 and global_a is not None:
        a = global_a.copy()
        a = project_simplex(a, S)
        t0 = time.monotonic()
        a_best, obj_best = coordinate_descent_polish(a, S, n, remaining)
        el = time.monotonic() - t0
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print("Phase 4: obj=%.8f (improved, %.1fs)" % (global_best, el))
        else:
            print("Phase 4: no improvement (%.1fs)" % el)

    # Phase 5: Final subgradient with very tight active set
    remaining = deadline - time.monotonic() - 5.0
    if remaining > 5.0 and global_a is not None:
        a = global_a.copy()
        a = project_simplex(a, S)
        # Use a smaller alpha for fine-tuning
        t0 = time.monotonic()
        c = fft_convolve_self(a)
        M = c.max()
        for _ in range(int(remaining * 100)):  # up to 100 iters/sec
            if time.monotonic() - t0 > remaining:
                break
            c = fft_convolve_self(a)
            M = c.max()
            if M < global_best:
                global_best = M
                global_a = a.copy()
            # Very tight active set
            threshold = M * (1.0 - 0.001)
            active = c >= threshold
            if active.sum() < 1:
                active[np.argmax(c)] = True
            w = np.zeros(len(c))
            w[active] = 1.0 / active.sum()
            g = compute_subgradient(a, w)
            g = g - np.mean(g)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            alpha = min(0.01 / gn, 0.1 / gn)
            a = project_simplex(a - alpha * g, S)

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("FINAL: best=%.8f, verified=%s" % (
        global_best, "%.8f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        print("Fallback to uniform")

    return final_a
```
