def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s

def soft_max_and_weights(c, tau):
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    return c_max + tau * np.log(s), e / s

"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions. Combines adaptive soft-max gradient descent, LP-based directional
refinement, and peak-directed local search. Exploits longer sequences (n=512,1024)
for better theoretical bounds and uses the provided sequence as warm start.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def soft_max_and_weights(c, tau):
    """Compute soft-max value and normalized weights."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    return c_max + tau * np.log(s), e / s


def grad_soft_max_conv(a, tau):
    """Compute soft-max(conv(a,a)) and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_and_weights(c, tau)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = (2.0 / tau) * corr[n - 1:2 * n - 1]
    return value, g, c


def upsample(a, new_n, rng, noise_level=0.015):
    """Upsample a sequence to a longer length via interpolation + noise."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += noise_level * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)


def gradient_descent(a_init, time_limit, rng):
    """
    Adaptive soft-max gradient descent with Adam optimizer.
    Uses adaptive temperature scheduling and periodic restarts.
    Returns (normalized_sequence, best_score).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = a / np.sum(a) * S

    theta = np.log(np.maximum(a / S, 1e-10))
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr0 = 3.5
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)

    tau = 80.0 / n
    tau_min = 0.05 / n
    no_improve = 0
    patience = 3000
    restart_count = 0
    max_restarts = 8

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        elapsed_frac = min(1.0, elapsed / time_limit)

        # Gradually decrease tau
        base_tau = (80.0 / n) * (0.05 / 80.0) ** (elapsed_frac ** 0.5)
        tau = max(base_tau, tau_min)

        # If stuck for too long, decrease tau aggressively and perturb
        if no_improve > patience and restart_count < max_restarts:
            tau = max(tau * 0.5, tau_min)
            theta += 0.05 * rng.standard_normal(n)
            no_improve = 0
            restart_count += 1

        a_cur = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a_cur, tau)

        inner = np.dot(a_cur, g_a)
        g_theta = a_cur * (g_a - inner / S)

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step)
        m2_hat = m2 / (1.0 - b2 ** step)
        lr = lr0 / np.sqrt(1.0 + 0.2 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 200 == 0:
            a_check = softmax_normalize(theta) * S
            score = local_score(a_check)
            if score < best_score - 1e-9:
                best_score = score
                best_theta = theta.copy()
                no_improve = 0
            else:
                no_improve += 200
            if step % 2000 == 0:
                print(f"          step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.3e}, t={elapsed:.1f}s")

    a_final = softmax_normalize(best_theta) * S
    score_final = local_score(a_final)
    return a_final / S, min(best_score, score_final)


def build_lp_matrix(a, stride=1):
    """Build the convolution matrix A for the LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    k_indices = np.arange(0, 2 * n - 1, stride)
    n_constraints = len(k_indices)
    A = np.zeros((n_constraints, n), dtype=np.float64)
    for idx, k in enumerate(k_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            A[idx, j_min:j_max + 1] = a[k - j_min:k - j_max + 1][::-1]
    return A, k_indices


def lp_refine(a_normalized, time_limit, rng, max_iter=5000):
    """
    LP-based directional refinement.
    Input: a_normalized with sum=1.
    Returns: (improved_normalized_sequence, best_score).
    """
    n = len(a_normalized)
    S = np.sqrt(2.0 * n)
    a = a_normalized.copy() * S
    conv = fft_convolve(a, a)
    M = np.max(conv)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()

    # Determine stride based on n
    if n <= 256:
        stride = 1
    elif n <= 512:
        stride = 1
    else:
        stride = 2

    c_obj = -np.ones(n)

    for step in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Build LP
        A, k_indices = build_lp_matrix(a, stride)
        b_ub = np.full(len(k_indices), M, dtype=np.float64)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': 5})
        except Exception:
            break

        if not res.success:
            break

        b = res.x
        sum_b = np.sum(b)
        if sum_b < 1e-10:
            break

        g = b / sum_b * S

        # Try multiple step sizes
        improved = False
        for t in [0.15, 0.1, 0.05, 0.03]:
            a_new = (1 - t) * a + t * g
            new_conv = fft_convolve(a_new, a_new)
            new_M = np.max(new_conv)
            if new_M < M * (1 - 1e-8):
                a = a_new
                M = new_M
                score = local_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                improved = True
                break

        if step % 200 == 0:
            print(f"          LP iter={step}, M={M:.6f}, score={local_score(a):.6f}, t={time.monotonic()-t0:.1f}s")

    return best_a / S, best_score


def local_search_refine(a_normalized, time_limit, rng):
    """
    Peak-directed local search with simulated annealing.
    Input: a_normalized with sum=1.
    Returns: (improved_normalized_sequence, best_score).
    """
    n = len(a_normalized)
    S = np.sqrt(2.0 * n)
    current = a_normalized.copy() * S
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9997

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.35:
            # Peak-directed reduction
            conv = fft_convolve(current, current)
            n_top = min(6, len(conv))
            top_lags = np.argsort(conv)[-n_top:]
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            n_reduce = max(1, n // 15)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.02, size=len(bot_idx)))
        elif r < 0.55:
            # Adjacent block shift
            width = max(1, int(rng.integers(1, max(2, n // 25))))
            start = int(rng.integers(0, max(1, n - 2 * width)))
            candidate[start:start + width] *= np.exp(rng.normal(-0.07, 0.02, size=width))
            candidate[start + width:start + 2 * width] *= np.exp(rng.normal(0.07, 0.02, size=width))
        elif r < 0.75:
            # Multi-block perturbation
            n_blocks = int(rng.integers(1, 4))
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 20))))
                bs = int(rng.integers(0, n - bw))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0.0, 0.04, size=bw))
        elif r < 0.85:
            # Partial symmetrization
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.2 * (sym - current)
            candidate += 0.01 * rng.standard_normal(n)
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0.0, 0.012, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s * S

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 20:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score

        temp *= cooling

    return best / S, best_score


def fft_phase_init(n, rng, n_iter=80):
    """Gerchberg-Saxton style initialization."""
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1

    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for optimal non-negative step function minimizing
    2*n*max(conv)/sum^2. Uses adaptive gradient descent, LP-based refinement,
    and peak-directed local search.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Get provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # === Phase 1: Adaptive gradient descent on provided sequence ===
    print("\n=== Phase 1: Gradient Descent on provided sequence ===")
    if start_seq is not None:
        t_alloc = min(600.0, (deadline - time.monotonic()) * 0.3)
        print(f"  n={len(start_seq)}, t={t_alloc:.0f}s")
        a_norm, s_opt = gradient_descent(start_seq, t_alloc, rng)
        a_scaled = a_norm * np.sqrt(2.0 * len(a_norm))
        print(f"    score={s_opt:.6f}")
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_scaled.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 2: Upsample to n=512 and optimize ===
    print("\n=== Phase 2: Upsample to n=512 ===")
    base = (best_seq / local_score(best_seq) ** 0.5) if best_seq is not None else (start_seq if start_seq is not None else np.ones(256) / 256)
    # Normalize base to sum=1
    base = base / np.sum(base)

    target_n = 512
    if time.monotonic() < deadline - 200:
        a_up = upsample(base, target_n, rng, noise_level=0.01)
        t_alloc = min(500.0, (deadline - time.monotonic()) * 0.35)
        t_alloc = max(60.0, t_alloc)
        print(f"  Upsampling to n={target_n}, t={t_alloc:.0f}s")
        a_norm512, s512 = gradient_descent(a_up, t_alloc, rng)
        a_scaled512 = a_norm512 * np.sqrt(2.0 * target_n)
        print(f"    score={s512:.6f}")
        if s512 < best_score:
            best_score = s512
            best_seq = a_scaled512.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 3: LP-based refinement ===
    print(f"\n=== Phase 3: LP Refinement (current best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 120:
        n_cur = len(best_seq)
        if n_cur <= 512:
            a_norm_cur = best_seq / np.sum(best_seq)
            t_alloc = min(500.0, deadline - time.monotonic() - 120)
            t_alloc = max(30.0, t_alloc)
            print(f"  LP on n={n_cur}, t={t_alloc:.0f}s")
            a_lp_norm, s_lp = lp_refine(a_norm_cur, t_alloc, rng)
            a_lp_scaled = a_lp_norm * np.sqrt(2.0 * n_cur)
            print(f"    score={s_lp:.6f}")
            if s_lp < best_score:
                best_score = s_lp
                best_seq = a_lp_scaled.copy()
                print(f"    *** New best: {best_score:.6f} ***")
        else:
            print(f"  Skipping LP for n={n_cur} (too large)")

    # === Phase 4: Local search polish ===
    print(f"\n=== Phase 4: Local Search (current best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_norm_cur = best_seq / np.sum(best_seq)
        t_alloc = min(250.0, deadline - time.monotonic() - 30)
        t_alloc = max(20.0, t_alloc)
        print(f"  Local search on n={len(best_seq)}, t={t_alloc:.0f}s")
        a_ls_norm, s_ls = local_search_refine(a_norm_cur, t_alloc, rng)
        a_ls_scaled = a_ls_norm * np.sqrt(2.0 * len(a_ls_norm))
        print(f"    score={s_ls:.6f}")
        if s_ls < best_score:
            best_score = s_ls
            best_seq = a_ls_scaled.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 5: Final gradient descent polish ===
    if best_seq is not None and deadline - time.monotonic() > 60:
        t_alloc = min(200.0, deadline - time.monotonic() - 20)
        t_alloc = max(30.0, t_alloc)
        print(f"\n=== Phase 5: Final Gradient Polish (t={t_alloc:.0f}s) ===")
        a_norm_cur = best_seq / np.sum(best_seq)
        a_final_norm, s_final = gradient_descent(a_norm_cur, t_alloc, rng)
        a_final_scaled = a_final_norm * np.sqrt(2.0 * len(a_final_norm))
        print(f"    score={s_final:.6f}")
        if s_final < best_score:
            best_score = s_final
            best_seq = a_final_scaled.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 6: Try n=1024 if time permits ===
    if best_seq is not None and deadline - time.monotonic() > 300:
        print(f"\n=== Phase 6: Try n=1024 ===")
        base = best_seq / np.sum(best_seq)
        a_up1024 = upsample(base, 1024, rng, noise_level=0.01)
        t_alloc = min(300.0, deadline - time.monotonic() - 30)
        print(f"  Optimizing n=1024, t={t_alloc:.0f}s")
        a_norm1024, s1024 = gradient_descent(a_up1024, t_alloc, rng)
        a_scaled1024 = a_norm1024 * np.sqrt(2.0 * 1024)
        print(f"    score={s1024:.6f}")
        if s1024 < best_score:
            best_score = s1024
            best_seq = a_scaled1024.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Final verification
    print(f"\n{'='*50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
