        for k in active:
            for j in range(max(0, k-n+1), min(n, k+1)):

        v *= 2

        # But v depends on a, so this is a fixed point
        # Use relaxation: a_new = (1-t)*a + t*v/sum(v)*S
        v_sum = v.sum()
        if v_sum < 1e-15:

        v_norm = v * (S / v_sum)

        if np.linalg.norm(v_norm - a) / S < tol:
            break
        
        # Relaxed update
        t = 0.5  # relaxation parameter
        a_new = (1 - t) * a + t * v_norm
        a_new = np.maximum(a_new, 0)
        a_new = a_new * (S / a_new.sum())
        
        # Check if objective improved

        if c_new.max() > c_max * 1.01:

            a_new = a_new * (S / a_new.sum())
        
        a = a_new
    
    return a

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size-1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def projection_simplex(v, S):
    """Euclidean projection of v onto {a >= 0, sum(a) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.full(n, S/n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1)
    return np.maximum(v - theta, 0)

def subgradient_at_a(a, K):
    """Compute 2 * sum_{k in K} (P_k a) / |K|."""
    n = len(a)

        hi = min(n - 1, k)
        for j in range(lo, hi + 1):

Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. KKT fixed-point iteration (power iteration on active-set matrix) - novel
2. Nesterov-accelerated projected subgradient with BB step sizes
3. Soft-max (log-sum-exp) refinement with Adam at decreasing temperature
4. Fast coordinate descent with incremental convolution updates
5. Adaptive active set tracking with hysteresis

Key novelty: KKT iteration identifies the active set K, forms W_K = sum P_k,
and iterates a -> normalize(W_K * a), converging to a stationary point where
the subgradient is constant on the support.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def projection_simplex(v, S):
    """Euclidean projection of v onto {a >= 0, sum(a) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - S)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(y, K):
    """Compute 2 * sum_{k in K} (P_k y) / |K| efficiently."""
    n = len(y)
    v = np.zeros(n)
    for k in K:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            v[j] += y[k - j]
    v *= 2.0 / len(K)
    return v


def active_set_from_conv(c, top_k=10, rel_threshold=0.99):
    """Get active set: intersection of top-k and relative threshold."""
    c_max = c.max()
    # Lags within relative threshold
    mask = c >= c_max * rel_threshold
    K_thresh = np.where(mask)[0]
    # Top-k lags
    K_top = np.argsort(c)[::-1][:top_k]
    # Union, capped
    K = np.unique(np.concatenate([K_top, K_thresh[:top_k * 3]]))
    K = np.sort(K)[:top_k * 2]
    return K


def kkt_power_iteration(a, S, max_iter=300, verbose=False):
    """
    KKT fixed-point: a_{t+1} = normalize(W_K * a_t) where W_K = sum_{k in K} P_k.
    This converges to the dominant eigenvector of W_K, which satisfies the KKT
    condition 2*W_K*a = lambda*1 (for a > 0).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    for it in range(max_iter):
        c = fft_convolve_self(a)
        c_max = c.max()

        K = active_set_from_conv(c, top_k=15, rel_threshold=0.995)

        # Compute v = W_K * a (without the factor of 2, we normalize anyway)
        v = np.zeros(n)
        for k in K:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                v[j] += a[k - j]

        v_sum = v.sum()
        if v_sum < 1e-15:
            break
        v = v * (S / v_sum)

        # Check convergence
        diff = np.linalg.norm(v - a) / S
        a = v

        c = fft_convolve_self(a)
        if c.max() < best_obj:
            best_obj = c.max()
            best_a = a.copy()

        if diff < 1e-12:
            if verbose:
                print("  KKT converged at iter %d" % it)
            break

    return best_a, best_obj


def kkt_relaxed(a, S, max_iter=500, alpha_schedule=None, verbose=False):
    """
    Relaxed KKT iteration: a_{t+1} = (1-alpha)*a_t + alpha*normalize(W_K * a_t).
    More stable than pure power iteration.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    for it in range(max_iter):
        c = fft_convolve_self(a)
        c_max = c.max()
        K = active_set_from_conv(c, top_k=12, rel_threshold=0.995)

        v = np.zeros(n)
        for k in K:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                v[j] += a[k - j]

        v_sum = v.sum()
        if v_sum < 1e-15:
            break
        v = v * (S / v_sum)

        # Adaptive relaxation
        if alpha_schedule is not None:
            alpha = alpha_schedule(it, max_iter)
        else:
            alpha = 0.8 * (0.995 ** it) + 0.05

        a_new = (1.0 - alpha) * a + alpha * v
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new * (S / a_new.sum())

        c_new = fft_convolve_self(a_new)
        if c_new.max() < c_max:
            a = a_new
            if c_new.max() < best_obj:
                best_obj = c_new.max()
                best_a = a.copy()
        else:
            # Backtrack
            a = (1.0 - alpha * 0.5) * a + alpha * 0.5 * v
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())

        if np.linalg.norm(a - best_a) / S < 1e-13:
            break

    return best_a, best_obj


def nesterov_subgradient(a, S, time_budget, eta0=1.0, top_k=10, verbose=False):
    """
    Nesterov-accelerated projected subgradient with Barzilai-Borwein step sizes.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)

    y = a.copy()
    a_prev = a.copy()
    g_prev = None

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    t = 0

    while time.monotonic() - start < time_budget:
        t += 1

        # Evaluate at y (Nesterov lookahead)
        c = fft_convolve_self(y)
        c_max = c.max()
        K = active_set_from_conv(c, top_k=top_k, rel_threshold=0.99)
        g = compute_subgradient(y, K)

        # BB step size
        if t > 1 and g_prev is not None:
            s_bb = y - a_prev
            y_diff = g - g_prev
            ss = np.dot(s_bb, s_bb)
            sy = np.dot(s_bb, y_diff)
            if ss > 1e-20 and sy > 1e-15:
                eta_bb = sy / ss
                eta_t = np.clip(eta_bb, 0.01 * eta0, 10.0 * eta0)
            else:
                eta_t = eta0 / (1.0 + 0.1 * np.sqrt(t))
        else:
            eta_t = eta0 / (1.0 + 0.1 * np.sqrt(t))

        # Projected subgradient step
        a_new = projection_simplex(y - eta_t * g, S)

        # Nesterov update
        theta_t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = a_new + (theta_t - 1.0) / theta_t * (a_new - a)
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new * (S / y_new.sum())

        a_prev = a
        a = a_new
        y = y_new
        g_prev = g

        # Check improvement
        c_check = fft_convolve_self(a)
        if c_check.max() < best_obj:
            best_obj = c_check.max()
            best_a = a.copy()

        # Early stopping
        if t % 1000 == 0 and verbose:
            elapsed = time.monotonic() - start
            print("    Nesterov t=%d obj=%.6f (t=%.1fs)" % (t, best_obj, elapsed))

    return best_a, best_obj


def softmax_objective_grad(a, S, T):
    """Compute soft-max (log-sum-exp) of conv and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()

    # Log-sum-exp: c_max + (1/T)*log(sum exp(T*(c - c_max)))
    shifted = T * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T

    # Weights: w[k] = exp(T*(c[k]-c_max)) / sum exp(T*(c-c_max))
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    # Gradient w.r.t. a: g[j] = 2 * sum_k w[k] * a[k-j] (valid)
    # Use FFT for efficiency
    # g[j] = 2 * (w conv a_rev)[j + n - 1]
    a_rev = a[::-1]
    conv_full = fft_convolve_self(np.concatenate([w, np.zeros(n)]))
    # Actually, let me just do it directly for the weighted version
    # g[j] = 2 * sum_k w[k] * a[k-j]
    # This is 2 * cross_correlation(w, a) evaluated appropriately
    # Use FFT: FFT(w) * FFT(a_rev) -> inverse -> shift
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A_rev = fft(a_rev, fsize)
    cross = np.real(ifft(W * A_rev))
    # cross[m] = sum_j w[j] * a_rev[m-j] = sum_j w[j] * a[n-1-m+j]
    # We want g[j] = 2 * sum_k w[k] * a[k-j]
    # Let m = k - j + (n-1): a[k-j] = a_rev[n-1-k+j]
    # cross[n-1-k+j] contains the term w[k]*a_rev[n-1-k+j] = w[k]*a[k-j]
    # So sum_k w[k]*a[k-j] = cross[n-1-j + ... ]
    # Let me just compute directly with the shift
    # cross_corr[m] = sum_k w[k] * a[m-k + n-1] ... this is getting confusing
    # Let me just use the direct O(n*|K|) approach since weights are sparse-ish
    # Actually for the soft-max, weights are spread out. Use FFT properly.
    # g[j] = 2 * sum_k w[k] * a[k-j] for 0<=k-j<n
    # Define b[k] = w[k] for k in [0, 2n-2]
    # g[j] = 2 * (b * a_flipped)[j + 2n - 2 - (n-1)] ... 
    # Simpler: g[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * (cross-correlation of w and a) at lag j
    # cross_corr_w_a[j] = sum_k w[k] * a[k-j]
    # This is the convolution of w with reversed a, shifted.
    # conv(w, a_rev)[m] = sum_k w[k] * a_rev[m-k] = sum_k w[k] * a[n-1-m+k]
    # We want sum_k w[k] * a[k-j]. Set n-1-m+k = k-j => m = n-1+j.
    # So g[j] = 2 * conv(w, a_rev)[n-1+j]

    fsize2 = 1 << (3 * n - 1).bit_length()
    W2 = fft(w, fsize2)
    A2 = fft(a_rev, fsize2)
    conv_w_arev = np.real(ifft(W2 * A2))
    g = 2.0 * conv_w_arev[n - 1: 2 * n - 1][:n]

    return obj, g, c


def adam_softmax(a, S, time_budget, T_init=20.0, T_end=500.0, lr=0.5, verbose=False):
    """Adam optimizer on the soft-max (log-sum-exp) objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())

    # Use log-parameterization: a = S * softmax(theta)
    theta = np.log(a / a.sum())
    theta = theta - theta.mean()  # Center for numerical stability

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (t * 0.002) / time_budget)
        T = T_init * np.exp(np.log(T_end / T_init) * progress)

        a = np.exp(theta - theta.max())
        a = a * (S / a.sum())

        obj, g_a, c = softmax_objective_grad(a, S, T)

        # Chain rule for log-parameterization
        # d/dtheta_j (sum_i g_a[i] * a[i]) = g_a[j]*a[j] - (a[j]/S) * sum_i g_a[i]*a[i]
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot

        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr_t = lr * (0.5 ** (t / 2000.0))
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # Track best (true max)
        a_true = np.exp(theta - theta.max())
        a_true = a_true * (S / a_true.sum())
        c_true = fft_convolve_self(a_true)
        if c_true.max() < best_obj:
            best_obj = c_true.max()
            best_a = a_true.copy()

    return best_a, best_obj


def coordinate_descent_fast(a, S, time_budget, verbose=False):
    """
    Fast coordinate descent: for each coordinate, compute the optimal 1D step
    using the sensitivity of the max convolution entry.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    start = time.monotonic()
    n_trials = 5

    for sweep in range(500):
        if time.monotonic() - start > time_budget:
            break

        c_max = c.max()
        # Active positions
        K_pos = np.where(c >= c_max * 0.999)[0]
        if len(K_pos) == 0:
            K_pos = np.array([np.argmax(c)])
        K_pos = K_pos[:20]  # Cap

        improved = False
        for i in range(0, n, 1):
            if time.monotonic() - start > time_budget:
                break

            orig = a[i]
            if orig <= 1e-14:
                continue

            # Sensitivity: how does c[k] change when a[i] changes by delta?
            # c[k] += delta * (a[k-i] + a[i-k]) for valid indices
            # We want to find delta that minimizes max over K_pos
            h = np.zeros(len(K_pos))
            for idx, k in enumerate(K_pos):
                h[idx] = 0.0
                if 0 <= k - i < n:
                    h[idx] += a[k - i]
                if 0 <= i - k < n:
                    h[idx] += a[i - k]

            # c_new[k] = c[k] + delta * h[idx]
            # Minimize max_k (c[k] + delta * h[idx]) over delta
            # This is: minimize max_k (c[k] + delta * h_k)
            # The optimal delta makes the active constraints balanced
            # Try a few candidate deltas
            best_delta = 0.0
            best_max = c_max

            # Candidate deltas: try reducing a[i]
            for trial in range(n_trials):
                # Try different reductions
                if trial == 0:
                    delta = -orig * 0.5
                elif trial == 1:
                    delta = -orig * 0.3
                elif trial == 2:
                    delta = -orig * 0.7
                elif trial == 3:
                    delta = orig * 0.3
                else:
                    # Optimal: find delta that equalizes the top constraints
                    # c[k] + delta * h[k] = c[k'] + delta * h[k'] for all active
                    # delta = (c[k'] - c[k]) / (h[k] - h[k'])
                    h_max_idx = np.argmax(c[K_pos] + 0)  # most constrained
                    h_min_h = np.argmin(h)
                    if abs(h[h_max_idx] - h[h_min_h]) > 1e-15:
                        delta = (c[K_pos][h_min_h] - c[K_pos][h_max_idx]) / (h[h_max_idx] - h[h_min_h])
                    else:
                        delta = -orig * 0.1
                    delta = np.clip(delta, -orig * 0.9, orig)

                new_val = orig + delta
                if new_val < 0:
                    continue

                # Compute new max
                new_c_active = c[K_pos] + delta * h
                new_max = max(new_max, np.max(new_c_active))

                if new_max < best_max:
                    best_max = new_max
                    best_delta = delta

            if best_delta != 0.0:
                a[i] = orig + best_delta
                a[i] = max(a[i], 0.0)
                s = a.sum()
                if s > 1e-10:
                    a = a * (S / s)
                    c = fft_convolve_self(a)
                    if c.max() < best_obj:
                        best_obj = c.max()
                        best_a = a.copy()
                    improved = True

        if not improved and sweep > 5:
            break

        if verbose and sweep % 50 == 0:
            print("    CoordDescent sweep=%d obj=%.6f" % (sweep, best_obj))

    return best_a, best_obj


def ftrl_method(a, S, time_budget, eta0=0.5, top_k=8, verbose=False):
    """
    FTRL with squared L2 regularization (Euclidean projection on simplex).
    a_t = Proj_{Δ_S}(a_{t-1} - eta_t * g_t) with AdaGrad-like step sizes.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())

    G = np.zeros(n)  # accumulated squared gradients (AdaGrad)
    t = 0
    start = time.monotonic()

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    while time.monotonic() - start < time_budget:
        t += 1
        c = fft_convolve_self(a)
        c_max = c.max()
        K = active_set_from_conv(c, top_k=top_k, rel_threshold=0.99)
        g = compute_subgradient(a, K)

        # AdaGrad step
        G += g * g
        eta_t = eta0 / np.sqrt(G + 1e-8)

        a_new = projection_simplex(a - g * eta_t, S)

        c_new = fft_convolve_self(a_new)
        if c_new.max() <= c_max:
            a = a_new
            if c_new.max() < best_obj:
                best_obj = c_new.max()
                best_a = a.copy()
        else:
            # Halve step
            a_new = projection_simplex(a - g * eta_t * 0.5, S)
            c_new2 = fft_convolve_self(a_new)
            if c_new2.max() <= c_max:
                a = a_new
                if c_new2.max() < best_obj:
                    best_obj = c_new2.max()
                    best_a = a.copy()

        if t % 5000 == 0 and verbose:
            elapsed = time.monotonic() - start
            print("    FTRL t=%d obj=%.6f (t=%.1fs)" % (t, best_obj, elapsed))

    return best_a, best_obj


def get_initialization(name, n, rng, S):
    """Generate various initializations."""
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "gaussian":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n / 2) / (n / 6)) ** 2)
        return a * (S / a.sum())
    elif name == "cosine":
        i = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * i / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "anti_peak":
        i = np.arange(n)
        a = 1.0 - 0.4 * np.cos(2 * np.pi * (n - 1) * i / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "raised":
        i = np.arange(n)
        t_arr = i / (n - 1)
        a = np.sin(np.pi * t_arr) ** 2
        return a * (S / a.sum())
    elif name == "two_humps":
        i = np.arange(n)
        a = (np.exp(-0.5 * ((i - n * 0.25) / (n / 10)) ** 2) +
             np.exp(-0.5 * ((i - n * 0.75) / (n / 10)) ** 2))
        return a * (S / a.sum())
    elif name == "inverse_cosine":
        i = np.arange(n)
        a = 1.0 / (1.0 + 0.5 * np.cos(2 * np.pi * i / n))
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def propose_candidate(seed=42, budget_s=2400, **kwargs):

        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:

            S_given = np.sqrt(2.0 * n_given)
            given_a = given_raw * (S_given / given_raw.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.6f" % (n_given, ver))

                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: %s" % str(e))

    # Define the main optimization pipeline for a given starting point
    def optimize_from(a_start, n, S, time_budget, label=""):
        nonlocal global_best, global_a
        if time_budget < 5.0:
            return

        # Phase A: KKT Power Iteration (fast, ~1s)

            cur_obj = obj_rel
            a = a_rel
            if cur_obj < global_best:
                global_best = cur_obj
                global_a = a.copy()
                print("  %s KKT-relaxed: obj=%.6f" % (label, cur_obj))

        elapsed = time.monotonic() - t0
        remaining = time_budget - elapsed
        if remaining < 5.0:
            return

        # Phase B: Nesterov Subgradient (main workhorse)
        nest_time = remaining * 0.35
        a_nest, obj_nest = nesterov_subgradient(a, S, nest_time, eta0=0.8, top_k=10)
        if obj_nest < cur_obj:
            cur_obj = obj_nest
            a = a_nest
            if cur_obj < global_best:
                global_best = cur_obj
                global_a = a.copy()
                print("  %s Nesterov: obj=%.6f" % (label, cur_obj))

        elapsed = time.monotonic() - t0
        remaining = time_budget - elapsed
        if remaining < 5.0:
            return

        # Phase C: Soft-max Adam refinement
        adam_time = remaining * 0.30
        a_adam, obj_adam = adam_softmax(a, S, adam_time, T_init=10.0, T_end=300.0, lr=0.3)
        if obj_adam < cur_obj:
            cur_obj = obj_adam
            a = a_adam
            if cur_obj < global_best:
                global_best = cur_obj
                global_a = a.copy()
                print("  %s Adam-softmax: obj=%.6f" % (label, cur_obj))

        elapsed = time.monotonic() - t0
        remaining = time_budget - elapsed
        if remaining < 3.0:
            return

        # Phase D: FTRL polish
        ftrl_time = remaining * 0.30
        a_ftrl, obj_ftrl = ftrl_method(a, S, ftrl_time, eta0=0.3, top_k=8)
        if obj_ftrl < cur_obj:
            cur_obj = obj_ftrl
            a = a_ftrl
            if cur_obj < global_best:
                global_best = cur_obj
                global_a = a.copy()
                print("  %s FTRL: obj=%.6f" % (label, cur_obj))

        elapsed = time.monotonic() - t0
        remaining = time_budget - elapsed
        if remaining < 2.0:
            return

        # Phase E: Coordinate descent polish
        cd_time = remaining * 0.7
        a_cd, obj_cd = coordinate_descent_fast(a, S, cd_time)
        if obj_cd < cur_obj:
            cur_obj = obj_cd
            a = a_cd
            if cur_obj < global_best:
                global_best = cur_obj
                global_a = a.copy()
                print("  %s CoordDescent: obj=%.6f" % (label, cur_obj))

        # Phase F: One more round of Nesterov from the polished point
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 5.0 and global_a is not None:
            a_final, obj_final = nesterov_subgradient(
                global_a, S, remaining, eta0=0.3, top_k=12
            )
            if obj_final < global_best:
                global_best = obj_final
                global_a = a_final.copy()
                print("  %s Final Nesterov: obj=%.6f" % (label, obj_final))

    # Main search: try multiple starting points and n values
    # Focus primarily on the given n, but also try others
    if given_a is not None:
        n_given = len(given_a)
        S_given = np.sqrt(2.0 * n_given)

        # Primary: optimize from the given sequence with most of the time
        primary_budget = min(600.0, (deadline - time.monotonic()) * 0.5)
        print("\n--- Primary optimization from given (n=%d) ---" % n_given)
        optimize_from(given_a, n_given, S_given, primary_budget, "primary")

    # Secondary: try other initializations at the same n
    remaining = deadline - time.monotonic() - 30.0
    if remaining > 20.0:
        n = len(given_a) if given_a is not None else 4096
        S = np.sqrt(2.0 * n)

        inits = ["gaussian", "cosine", "anti_peak", "random", "inverse_cosine", "two_humps"]
        per_init_time = remaining / (len(inits) + 2)

        for init_name in inits:
            if time.monotonic() > deadline - 30.0:
                break
            if per_init_time < 3.0:
                break
            a_init = get_initialization(init_name, n, rng, S)
            print("\n--- Secondary: init=%s n=%d ---" % (init_name, n))
            optimize_from(a_init, n, S, per_init_time, init_name)

    # Tertiary: try a different n (smaller for faster convergence)
    remaining = deadline - time.monotonic() - 20.0
    if remaining > 30.0:
        for n_alt in [2048, 8192]:
            if time.monotonic() > deadline - 20.0:
                break
            S_alt = np.sqrt(2.0 * n_alt)
            budget_alt = min(remaining * 0.4, 120.0)
            if budget_alt < 10.0:
                break
            a_alt = get_initialization("cosine", n_alt, rng, S_alt)
            print("\n--- Tertiary: n=%d ---" % n_alt)
            optimize_from(a_alt, n_alt, S_alt, budget_alt, "n=%d" % n_alt)

    # Final: if we have a good result, do one more intensive polish
    remaining = deadline - time.monotonic() - 10.0
    if global_a is not None and remaining > 10.0:
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * n_final)
        print("\n--- Final intensive polish (remaining=%.0fs) ---" % remaining)
        a_p, obj_p = nesterov_subgradient(
            global_a, S_final, remaining * 0.4, eta0=0.2, top_k=15
        )
        if obj_p < global_best:
            global_best = obj_p
            global_a = a_p.copy()
            print("  Final Nesterov2: obj=%.6f" % obj_p)

        remaining2 = deadline - time.monotonic() - 5.0
        if remaining2 > 5.0:
            a_p2, obj_p2 = coordinate_descent_fast(global_a, S_final, remaining2 * 0.6)
            if obj_p2 < global_best:
                global_best = obj_p2
                global_a = a_p2.copy()
                print("  Final CoordDescent2: obj=%.6f" % obj_p2)

    # Verify and return
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_list = global_a.tolist()
    ver = evaluate_sequence(final_list)
    print("\nFINAL: obj=%.6f (verified=%s)" % (
        global_best, "%.6f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
        print("Fallback: obj=%.6f" % ver)

    return final_list
