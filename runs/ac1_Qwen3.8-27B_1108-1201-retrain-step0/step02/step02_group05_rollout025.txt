def run_soft_max_lbgfs(theta, tau_schedule, maxiter_per_tau, deadline, verbose=True):
    """Run L-BFGS with annealed soft-max temperature."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))
    
    for tau in tau_schedule:
        if time.monotonic() > deadline - 10:
            break
        
        result = minimize(
            soft_max_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(tau,),
            options={'maxiter': maxiter_per_tau, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': maxiter_per_tau * 4}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)
        
        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
            if verbose:
                print(f"    soft-max tau={tau}: score={s:.6f} *** improved ***")
        else:
            if verbose:
                print(f"    soft-max tau={tau}: score={s:.6f}")
        theta = best_theta.copy()
    
    return best_theta, best_score

def fine_local_search(a_init, time_limit, rng):
    """
    Very fine local search: small random perturbations targeting peak contributors.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        c_max = np.max(c)
        
        # Find top contributing pairs to the peak
        peak_idx = int(np.argmax(c))
        contribs = np.zeros(n)
        for i in range(min(n, peak_idx + 1)):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = current[i] * current[j]
        
        top_k = max(5, n // 8)
        top_idx = np.argsort(contribs)[-top_k:]
        
        # Strategy: reduce top contributors, increase others
        candidate = current.copy()
        n_reduce = max(2, top_k // 3)
        reduce_idx = top_idx[-n_reduce:]
        increase_idx = np.setdiff1d(np.arange(n), reduce_idx)
        n_increase = min(max(3, n // 10), len(increase_idx))
        increase_idx = rng.choice(increase_idx, size=n_increase, replace=False)
        
        # Reduce
        candidate[reduce_idx] *= np.exp(rng.normal(-0.01, 0.005, size=n_reduce))
        # Increase
        candidate[increase_idx] *= np.exp(rng.normal(0.01, 0.005, size=n_increase))
        
        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
    
    return best, best_score

"""
Multi-phase optimization for minimum peak autocorrelation non-negative sequences.
Combines L-BFGS with soft-max and p-norm objectives, gradient-based local search
with momentum and adaptive step size, targeted peak-pair reduction, and multi-start
perturbation with re-optimization. Focuses on intensive refinement of the best candidate.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def soft_max_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def pnorm_obj_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p.
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def compute_peak_gradient(a, c, threshold_frac=0.95):
    """
    Compute the gradient of the (soft) peak of conv(a,a) with respect to a.
    Returns gradient vector of length n.
    """
    n = len(a)
    c_max = np.max(c)
    peak_mask = (c >= threshold_frac * c_max).astype(np.float64)
    if np.sum(peak_mask) == 0:
        peak_mask[int(np.argmax(c))] = 1.0

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, peak_mask)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    return grad_a


def gradient_local_search(a_init, time_limit, rng, verbose=True):
    """
    Gradient-based local search with adaptive step size, momentum, and restarts.
    Uses the subgradient of the max peak to guide descent in the simplex.
    """
    current = a_init.astype(np.float64).copy()
    current = np.maximum(current, 1e-15)
    current = current / np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()

    step_size = 0.02
    min_step = 1e-6
    max_step = 0.08
    momentum = np.zeros(n)
    momentum_coeff = 0.6

    stall_count = 0
    max_stall = 800
    last_print = t0

    if verbose:
        print(f"  GradLS start: n={n}, score={current_score:.8f}")

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        grad_a = compute_peak_gradient(current, c, threshold_frac=0.93)

        # Project to zero-sum direction
        grad_a = grad_a - np.mean(grad_a)

        # Momentum update
        momentum = momentum_coeff * momentum + (1.0 - momentum_coeff) * grad_a
        momentum = momentum - np.mean(momentum)

        # Step
        candidate = current - step_size * momentum
        candidate = np.maximum(candidate, 1e-15)
        candidate = candidate / np.sum(candidate)

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            stall_count = 0
            step_size = min(step_size * 1.03, max_step)
            momentum_coeff = min(momentum_coeff + 0.005, 0.85)
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            stall_count += 1
            step_size = max(step_size * 0.9, min_step)
            momentum_coeff = max(momentum_coeff - 0.01, 0.3)

            if stall_count > max_stall:
                # Restart with perturbation
                scale = 0.01 + 0.02 * (max_stall / 800.0)
                current = best * np.exp(rng.normal(0, scale, n))
                current = np.maximum(current, 1e-15)
                current = current / np.sum(current)
                current_score = local_score(current)
                step_size = 0.015
                momentum = np.zeros(n)
                momentum_coeff = 0.5
                stall_count = 0
                max_stall = max(200, int(max_stall * 0.7))

        if verbose and time.monotonic() - last_print > 5.0:
            last_print = time.monotonic()
            print(f"    grad_ls: score={current_score:.8f}, best={best_score:.8f}, "
                  f"step={step_size:.5f}, mom={momentum_coeff:.3f}, t={time.monotonic()-t0:.0f}s")

    return best, best_score


def targeted_peak_reduction(a_init, time_limit, rng, verbose=True):
    """
    Targeted local search: specifically reduces the top peak by adjusting
    the contributing pairs and redistributing mass to non-contributing positions.
    """
    current = a_init.astype(np.float64).copy()
    current = np.maximum(current, 1e-15)
    current = current / np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    last_print = t0

    if verbose:
        print(f"  PeakRed start: n={n}, score={current_score:.8f}")

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))

        # Compute contributions to the peak
        contribs = np.zeros(n)
        for i in range(min(n, peak_idx + 1)):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = current[i] * current[j]

        # Find top contributing indices
        top_k = max(4, n // 12)
        top_idx = np.argsort(contribs)[-top_k:]

        # Strategy: reduce top contributors, increase others
        candidate = current.copy()
        n_reduce = max(2, top_k // 3)
        reduce_idx = top_idx[-n_reduce:]

        # Find positions that contribute least to the peak
        other_idx = np.setdiff1d(np.arange(n), reduce_idx)
        # Sort others by their contribution to the peak (ascending)
        other_contribs = np.array([contribs[i] for i in other_idx])
        sorted_other = other_idx[np.argsort(other_contribs)[:max(3, n // 10)]]

        # Apply changes
        red_scale = np.exp(rng.normal(-0.008, 0.003, size=n_reduce))
        inc_scale = np.exp(rng.normal(0.008, 0.003, size=len(sorted_other)))

        candidate[reduce_idx] *= red_scale
        candidate[sorted_other] *= inc_scale

        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < 0.3:
            # Occasionally accept equal or slightly worse for exploration
            if new_score < current_score * 1.001:
                current = candidate
                current_score = new_score

        if verbose and time.monotonic() - last_print > 5.0:
            last_print = time.monotonic()
            print(f"    pk_red: score={current_score:.8f}, best={best_score:.8f}, "
                  f"peak_k={peak_idx}, t={time.monotonic()-t0:.0f}s")

    return best, best_score


def run_soft_max_lbgfs(theta, tau_schedule, maxiter_per_tau, deadline, verbose=True):
    """Run L-BFGS with annealed soft-max temperature."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))

    for tau in tau_schedule:
        if time.monotonic() > deadline - 10:
            break

        result = minimize(
            soft_max_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(tau,),
            options={'maxiter': maxiter_per_tau, 'ftol': 1e-15, 'gtol': 1e-12,
                     'maxfun': maxiter_per_tau * 4}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)

        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
            if verbose:
                print(f"    smax tau={tau:.5f}: {s:.8f} *** improved ***")
        else:
            if verbose:
                print(f"    smax tau={tau:.5f}: {s:.8f}")
        theta = best_theta.copy()

    return best_theta, best_score


def run_pnorm_lbgfs(theta, p_schedule, maxiter_per_p, deadline, verbose=True):
    """Run L-BFGS with annealed p-norm."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))

    for p in p_schedule:

        print(f"  Phase 7: {best_score:.8f} *** NEW BEST ***")
    else:
        print(f"  Phase 7: {p_score:.8f} (no improvement)")

    # ============ Phase 8: Targeted peak reduction (final) ============
    if time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic() - 20
        if remaining > 30:
            polished, p_score = targeted_peak_reduction(best_seq, remaining, rng, verbose=False)
            if p_score < best_score:
                best_score = p_score
                best_seq = polished.copy()
                print(f"  Final peak reduction: {best_score:.8f} *** NEW BEST ***")

    # ============ Phase 9: Try alternative lengths ============
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*60}")
        print(f"Phase 9: Alternative lengths (score={best_score:.8f})")
        print(f"{'='*60}")

        alt_lengths = [240, 272, 288, 224, 320]
        for n_alt in alt_lengths:
            if time.monotonic() > deadline - 50:
                break
            if n_alt == n_best:
                continue

            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_alt)
            a_alt = np.interp(x_new, x_old, best_seq)
            a_alt = np.maximum(a_alt, 1e-8)
            a_alt = a_alt / np.sum(a_alt)

            theta = np.log(a_alt + 1e-30)
            tau_sub = [0.01, 0.003, 0.001, 0.0004]
            sub_deadline = min(time.monotonic() + 50, deadline - 30)
            theta_best, s = run_soft_max_lbgfs(theta, tau_sub, maxiter_per_tau=100,
                                               deadline=sub_deadline, verbose=False)

            # Quick polish
            a_try = softmax(theta_best)
            if time.monotonic() < deadline - 30:
                a_try, s = gradient_local_search(a_try, 15.0, rng, verbose=False)

            if s < best_score:
                best_score = s
                best_seq = a_try.copy()
                print(f"  n={n_alt}: {best_score:.8f} *** NEW BEST ***")
            else:
                print(f"  n={n_alt}: {s:.8f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL SCORE: {final_score:.8f} (n={len(final_list)})")
        print(f"Local score: {local_score(np.array(final_list)/np.sum(final_list)):.8f}")
        print(f"{'='*60}")
        return final_list

    # Ultimate fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
