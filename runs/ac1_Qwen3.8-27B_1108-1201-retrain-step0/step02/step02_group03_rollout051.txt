    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def fft_phase_init(n, rng, n_iter=80):
    """Gerchberg-Saxton initialization."""
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft <<= 1
    a = np.ones(n) / n
    for _ in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a

def gradient_optimize(a_init, time_limit, rng, 
                       tau_start=None, tau_end=None,
                       lr=1.0, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam gradient descent on softmax-parameterized sequence.
    Objective: soft-max approximation of max(conv(a,a)).
    Constraint: sum(a) = sqrt(2n), a >= 0.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    
    if tau_start is None:
        tau_start = n / 20.0
    if tau_end is None:
        tau_end = n / 2000.0
    
    # Initialize theta from a_init
    a_norm = a_init / np.sum(a_init) * S
    theta = np.log(a_norm / S + 1e-15)
    
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    t0 = time.monotonic()
    
    best_a = a_norm.copy()
    best_score = compute_score(best_a)
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        # Annealing: start high, decrease
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        
        # Compute a from theta
        t_shift = theta - np.max(theta)
        e = np.exp(t_shift)
        a = e / e.sum() * S
        
        # Compute convolution
        c = fft_convolve(a, a)
        
        # Soft-max of c
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        e_c = np.exp(shifted)
        w = e_c / e_c.sum()
        soft_val = c_max + tau * np.log(e_c.sum())
        
        # Gradient of soft_val w.r.t. a
        # soft_val = log(sum_k exp(c[k]/tau)) * tau + c_max
        # d(soft_val)/d(a[j]) = sum_k w[k] * d(c[k])/d(a[j])
        # d(c[k])/d(a[j]) = a[k-j] + a[j-k] (with bounds, handling the symmetry)
        # This simplifies to: grad_a = convolve(w, a) + convolve(w_rev, a_rev) [appropriate slices]
        # More precisely: grad_a[j] = sum_k w[k] * (a[k-j] if 0<=k-j<n) + sum_k w[k] * (a[k-j] if 0<=k-j<n)
        # But with the factor of 2 issue at k=2j...
        # Cleanest: grad_a[j] = sum_{k: 0<=k-j<n} w[k] * a[k-j]  [this counts each pair once]
        #           + sum_{k: 0<=k-j<n} w[k] * a[k-j]  [same thing, for the other factor]
        # = 2 * sum_{k: 0<=k-j<n} w[k] * a[k-j]  [but only when j != k-j]
        # The correct formula: grad_a = full_convolve(w, a) + full_convolve(w, a)
        # where we take the appropriate part... 
        
        # Actually, the cleanest derivation:
        # f(a) = sum_k w[k] * (a*a)[k]
        # where (a*a)[k] = sum_i a[i]*a[k-i] (linear conv)
        # df/da[j] = sum_k w[k] * d(a*a)[k]/da[j]
        # d(a*a)[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j] (for 0<=k-j<n, j != k-j)
        #                   = a[k-j] (for k=2j)
        # 
        # This is equivalent to: df/da = (w * a) + (w * a) evaluated appropriately
        # where * denotes linear convolution, taking the "same" part.
        # 
        # More precisely: df/da[j] = sum_k w[k] * a[k-j] (for valid k,j)
        #                           + sum_k w[k] * a[k-j] (for valid k,j)
        # But the second sum is the same as the first (just relabeled), so:
        # df/da[j] = 2 * sum_{k: 0<=k-j<n} w[k] * a[k-j]
        #           - sum_k w[k] * a[k-j] * delta(k, 2j)  [to fix the double count]
        # 
        # This is getting too complicated analytically. Let me use the FFT trick:
        # 
        # The gradient of the quadratic form a^T H a (H symmetric) is 2Ha.
        # Here f(a) = sum_k w[k] (a*a)[k] = a^T H_w a / 2 ... no.
        # f(a) = sum_k w[k] sum_i a[i] a[k-i]
        # Let me index: f(a) = sum_{i,j} a[i] a[j] w[i+j] (when i+j < 2n-1)
        # So H[i,j] = w[i+j]/2 (symmetric)
        # grad_f = (H + H^T) a = 2H a (since H is symmetric)
        # grad_f[j] = 2 * sum_i H[j,i] a[i] = 2 * sum_i w[j+i]/2 * a[i] = sum_i w[j+i] * a[i]
        #           = sum_i a[i] * w[i+j]
        # This is the cross-correlation of a and w!
        # grad_f[j] = sum_i a[i] w[i+j] = (a ⋆ w)[j] where ⋆ denotes cross-correlation
        # = convolve(a, w[::-1])[j + len(w) - 1 - (n-1)] ... 
        # Actually cross-correlation: (a ⋆ w)[j] = sum_i a[i+j] w[i] = sum_i a[i] w[i-j]
        # Hmm, conventions vary. Let me just compute it directly using FFT:
        # grad_f[j] = sum_{i: i+j < 2n-1} a[i] * w[i+j]
        # Let k = i+j: grad_f[j] = sum_{k=j}^{j+n-1} a[k-j] * w[k] (with k < 2n-1)
        # This is a "sliding inner product" which is a correlation.
        # (a corr w)[j] = sum_i a[i] w[i+j] = sum_i a[i+j] w[i]  (depending on convention)
        # Using numpy convention: correlate(a, w, mode='full') ... 
        # Let me just use: grad_f = fft_convolve(a, w) ... no that's not right either.
        
        # OK final answer: I'll use the formula grad_f[j] = sum_i a[i] * w[i+j]
        # which can be computed as: for each j, it's a dot product of a with a shifted w.
        # This is: (a * w_flipped) where we take the right part.
        # Specifically: let full = fft_convolve(a, w), then full has length n + (2n-1) - 1 = 3n-2
        # full[m] = sum_i a[i] w[m-i] (for valid i, m-i)
        # We want grad_f[j] = sum_i a[i] w[i+j] = sum_i a[i] w[(i)+j]
        # Let m = i+j+... no.
        # full[m] = sum_i a[i] w[m-i]. Set m-i = i'+j where i'=i... this doesn't work cleanly.
        
        # Let me just do it with a direct loop-free approach:
        # grad_f[j] = sum_{i=0}^{n-1} a[i] * w[i+j] where i+j < 2n-1
        # For j=0: sum_i a[i] w[i] (first n elements of w)
        # For j=1: sum_i a[i] w[i+1] (elements 1 to n of w)
        # ...
        # For j=n-1: sum_i a[i] w[i+n-1] (elements n-1 to 2n-2 of w)
        # 
        # This is: for each j, dot(a, w[j:j+n]) (the part of w starting at position j)
        # This can be computed as: cross-correlation of a and w, taking the first n lags.
        # np.correlate(w, a, mode='full') gives the cross-correlation.
        # (w corr a)[k] = sum_i w[i+k] a[i] (for the "full" mode, k = -(n-1) to 2n-2)
        # We want: for j = 0, ..., n-1: sum_i w[i+j] a[i]
        # This is (w corr a)[j] for j = 0, ..., n-1
        # In np.correlate(w, a, 'full'), index n-1+j corresponds to lag j
        # So grad_f = np.correlate(w, a, 'full')[n-1 : 2*n-1]
        
        grad_a = np.correlate(w, a, mode='full')[n-1 : 2*n-1]
        
        # Now convert to gradient w.r.t. theta
        # a = softmax(theta) * S
        # da_k/dtheta_j = a_k * (delta[k,j] - a_k/S)
        # d(soft_val)/dtheta_j = sum_k grad_a[k] * a_k * (delta[k,j] - a_k/S)
        # = grad_a[j] * a[j] - (1/S) * sum_k grad_a[k] * a[k]^2
        
        dot_term = np.dot(grad_a, a * a)
        grad_theta = a * (grad_a - dot_term / S)
        
        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * (grad_theta ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        lr_t = lr / np.sqrt(1 + 0.3 * step)
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic check
        if step % 500 == 0:
            a_check = np.exp(theta - np.max(theta))
            a_check = a_check / a_check.sum() * S
            score = compute_score(a_check)
            if score < best_score:
                best_score = score
                best_a = a_check.copy()
            if step % 2000 == 0:
                print(f"          step={step}, soft_val={soft_val:.4f}, score={score:.6f}, best={best_score:.6f}, tau={tau:.1f}")
    
    # Final computation
    a_final = np.exp(theta - np.max(theta))
    a_final = a_final / a_final.sum() * S
    final_score = compute_score(a_final)
    if final_score < best_score:
        best_score = final_score
        best_a = a_final.copy()
    
    return best_a, best_score

def peak_deflation(a, time_limit, rng):
    """
    Targeted local search that specifically tries to reduce the peak of the convolution.
    """
    n = len(a)
    S = np.sum(a)
    current = a.copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        peak_idx = np.argmax(c)
        
        # Strategy: reduce contributions to the peak
        # The peak is at lag peak_idx, so it involves pairs (i, peak_idx-i)
        # We want to reduce c[peak_idx] without increasing other c[k] too much
        
        r = rng.random()
        if r < 0.4:
            # Reduce elements that contribute most to the peak
            i_min = max(0, peak_idx - n + 1)
            i_max = min(n - 1, peak_idx)
            contrib = np.zeros(n)
            for i in range(i_min, i_max + 1):
                j = peak_idx - i
                if 0 <= j < n:
                    contrib[i] += current[i] * current[j]
                    contrib[j] += current[i] * current[j]
            
            # Reduce top contributors, increase bottom contributors
            n_reduce = max(2, n // 8)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            
            factor = rng.uniform(0.85, 0.95)
            candidate = current.copy()
            candidate[top_idx] *= factor
            candidate[bot_idx] *= (1 + (1 - factor) * n_reduce / max(1, np.sum(contrib[bot_idx] > 0) * n))
            # Renormalize
            s = np.sum(candidate)
            if s > 1e-10:
                candidate = candidate / s * S
        elif r < 0.7:
            # Fourier-domain perturbation
            A = np.fft.rfft(current)
            # Add small random phase perturbation
            dA = rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A))
            A += 0.05 * dA
            candidate = np.fft.irfft(A, 2*n-1)[:n]
            candidate = np.clip(candidate, 0, None)
            s = np.sum(candidate)
            if s > 1e-10:
                candidate = candidate / s * S
            else:
                candidate = current.copy()
        elif r < 0.85:
            # Block swap: reduce one block, increase another
            block_size = max(1, n // 16)
            b1 = rng.integers(0, n - 2*block_size)
            b2 = b1 + block_size
            candidate = current.copy()
            reduce_factor = np.exp(-rng.uniform(0.05, 0.15))
            candidate[b1:b1+block_size] *= reduce_factor
            candidate[b2:b2+block_size] *= reduce_factor ** -1
            s = np.sum(candidate)
            if s > 1e-10:
                candidate = candidate / s * S
        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0, 0.02, n))
            s = np.sum(candidate)
            if s > 1e-10:
                candidate = candidate / s * S
        
        new_score = compute_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            # Simulated annealing acceptance
            temp = 0.002 * (1 - (time.monotonic() - t0) / time_limit)
            if temp > 0.0001 and rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-10)):
                current = candidate
                current_score = new_score
    
    return best, best_score
