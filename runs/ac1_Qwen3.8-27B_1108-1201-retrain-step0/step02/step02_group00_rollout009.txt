    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho_candidates = np.arange(1, n + 1, dtype=np.float64)

    else:

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Improved FISTA with exponential beta ramp, backtracking line search,
peak-targeted polishing, and diverse initializations.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a using FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

        # Top-k subgradient
        top_indices = np.argsort(c)[::-1][:k_cur]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 3000 == 0 and rng.random() < 0.15:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def run_local_perturb(a_init, n, S, max_time, rng):
    """Local perturbation / simulated annealing style polish."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    c = fft_convolve_self(a)
    best_score = c.max()

    temp = 0.01
    n_accepted = 0
    n_rejected = 0

    while time.monotonic() - start < max_time:
        # Generate perturbation: modify a random subset of elements
        n_pert = max(1, n // 100)
        indices = rng.integers(0, n, size=n_pert)
        scale = temp * S / np.sqrt(n)
        perturbations = rng.normal(0, scale, size=n_pert)

        a_trial = a.copy()
        a_trial[indices] += perturbations
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            continue
        a_trial = a_trial * (S / s_trial)

        c_trial = fft_convolve_self(a_trial)
        score_trial = c_trial.max()

        if score_trial < best_score:
            a = a_trial.copy()
            best_score = score_trial
            best_local = a_trial.copy()
            n_accepted += 1
        elif rng.random() < np.exp(-(score_trial - best_score) / max(temp * 0.01, 1e-10)):
            a = a_trial.copy()
            n_accepted += 1
        else:
            n_rejected += 1

        # Cool down
        if n_accepted + n_rejected > 200:
            temp *= 0.95
            n_accepted = 0
            n_rejected = 0

        if temp < 1e-6:
            temp = 0.005
            # Restart with small perturbation
            a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses FISTA with adaptive log-sum-exp, backtracking, and peak-targeted polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    total_deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Collect initializations
    inits = []

    # Try to get the given sequence
    given_arr = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given_arr = np.clip(given, 0, 1000).copy()
            inits.append(("given", given_arr.copy()))
    except Exception:
        pass

    # Determine main length
    if given_arr is not None:
        main_len = len(given_arr)
    else:
        main_len = 8192

    n = main_len
    tt = np.linspace(0, 1, n, endpoint=False)

    # Add diverse initializations at main length
    inits.append(("rand", rng.random(n) * 0.5 + 0.1))
    inits.append(("gauss", np.exp(-0.5 * ((tt - 0.5) * 4.0) ** 2) + 0.05))
    inits.append(("sin_mixed", 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) +
                  0.2 * np.cos(2 * np.pi * 7 * tt) +
                  0.1 * np.sin(2 * np.pi * 13 * tt)))
    inits.append(("bimodal", 0.1 + 0.5 * np.exp(-((tt - 0.25) * 8) ** 2) +
                  0.5 * np.exp(-((tt - 0.75) * 8) ** 2)))
    inits.append(("ramp", tt + 0.1 + 0.3 * np.sin(2 * np.pi * 5 * tt)))

    # Also try a longer length for diversity
    n2 = n * 2 if n <= 8192 else n // 2
    if n2 != n and 2048 <= n2 <= 32768:
        tt2 = np.linspace(0, 1, n2, endpoint=False)
        inits.append(("rand_long" if n2 > n else "rand_short", rng.random(n2) * 0.5 + 0.1))
        inits.append(("gauss_alt", np.exp(-0.5 * ((tt2 - 0.5) * 5.0) ** 2) + 0.05))

    n_inits = len(inits)
    usable = max(1.0, total_deadline - time.monotonic())
    per_init = usable * 0.45 / n_inits

    print("=== Phase 1: Exploration (%d inits, %.1f sec each) ===" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > total_deadline - 120:
            break
        n_cur = len(arr)
        if n_cur < 64:
            continue
        S = np.sqrt(2.0 * float(n_cur))
        t_phase = time.monotonic()

        # FISTA with moderate beta
        seg_fista = per_init * 0.7
        a_fista, score_fista = run_fista_v2(arr, n_cur, S, seg_fista, rng,
                                             beta_init=5.0, beta_max=500.0)
        a_norm = safe_normalize(a_fista, n_cur)
        c = fft_convolve_self(a_norm)
        score = c.max()
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            print("  FISTA [%s] n=%d: %.6f" % (name, n_cur, global_best))

        # Peak focus
        rem = per_init - (time.monotonic() - t_phase)
        if rem > 2:
            a_pk, score_pk = run_peak_focus(a_norm, n_cur, S, rem, rng,
                                             top_k=10, k_max=100)
            a_norm2 = safe_normalize(a_pk, n_cur)
            c2 = fft_convolve_self(a_norm2)
            if c2.max() < global_best:
                global_best = c2.max()
                global_a = a_norm2.copy()
                print("  PeakFocus [%s] n=%d: %.6f" % (name, n_cur, global_best))

    print("=== After Phase 1: global_best=%.6f ===" % global_best)

    if global_a is None:
        global_a = np.ones(main_len, dtype=np.float64) * np.sqrt(2.0 * main_len) / main_len
        global_best = compute_objective(global_a)
        print("  Using fallback uniform, best=%.6f" % global_best)

    # Phase 2: Intensive refinement
    n_best = len(global_a)
    S_best = np.sqrt(2.0 * float(n_best))
    remaining = total_deadline - time.monotonic()

    if remaining > 30:
        print("=== Phase 2: Intensive refinement (remaining %.1f sec) ===" % remaining)

        # High-beta FISTA
        seg = max(1.0, remaining * 0.35)
        a_p1, s1 = run_fista_v2(global_a, n_best, S_best, seg, rng,
                                 beta_init=50.0, beta_max=5000.0)
        a_p1 = safe_normalize(a_p1, n_best)
        c1 = fft_convolve_self(a_p1)
        if c1.max() < global_best:
            global_best = c1.max()
            global_a = a_p1.copy()
            print("  High-beta FISTA: %.6f" % global_best)

        # Peak focus with high k
        remaining2 = total_deadline - time.monotonic()
        if remaining2 > 10:
            seg2 = max(1.0, remaining2 * 0.3)
            a_p2, s2 = run_peak_focus(global_a, n_best, S_best, seg2, rng,
                                       top_k=20, k_max=300)
            a_p2 = safe_normalize(a_p2, n_best)
            c2 = fft_convolve_self(a_p2)
            if c2.max() < global_best:
                global_best = c2.max()
                global_a = a_p2.copy()
                print("  PeakFocus high-k: %.6f" % global_best)

    # Phase 3: Local perturbation polish
    remaining3 = total_deadline - time.monotonic()
    if remaining3 > 10:
        print("=== Phase 3: Local polish (remaining %.1f sec) ===" % remaining3)
        a_p3, s3 = run_local_perturb(global_a, n_best, S_best, remaining3, rng)
        a_p3 = safe_normalize(a_p3, n_best)
        c3 = fft_convolve_self(a_p3)
        if c3.max() < global_best:
            global_best = c3.max()
            global_a = a_p3.copy()
            print("  Local perturb: %.6f" % global_best)

    # Phase 4: Final FISTA polish with very high beta
    remaining4 = total_deadline - time.monotonic()
    if remaining4 > 5:
        print("=== Phase 4: Final polish (remaining %.1f sec) ===" % remaining4)
        a_p4, s4 = run_fista_v2(global_a, n_best, S_best, remaining4, rng,
                                 beta_init=200.0, beta_max=10000.0)
        a_p4 = safe_normalize(a_p4, n_best)
        c4 = fft_convolve_self(a_p4)
        if c4.max() < global_best:
            global_best = c4.max()
            global_a = a_p4.copy()
            print("  Final FISTA: %.6f" % global_best)

    # Verification
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    ver = evaluate_sequence(final_list)
    print("Verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        global_a = np.ones(256, dtype=np.float64) * 0.5
        final_list = [float(x) for x in global_a]

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * 0.5

    print("Returning sequence of length %d, internal score=%.6f" %
          (len(global_a), global_best))
    return [float(x) for x in global_a]
