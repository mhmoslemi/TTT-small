def fft_convolve(a, b):

        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def nesterov_smooth_max(a_init, time_limit, tau_schedule='auto'):
    """Nesterov-accelerated proximal gradient for smooth max of autocorrelation."""
    n = len(a_init)
    a = normalize_to_simplex(a_init.copy())
    a_prev = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_val = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    step = 0
    
    # Lipschitz constant of the gradient of smooth max
    # For tau-smoothed max, Lipschitz constant ≈ 2/tau (for autocorrelation)
    # Actually it depends on the problem. Let me use a conservative estimate.
    L = 4.0 / n  # conservative
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        tau = (10.0 / n) * (0.01) ** (frac ** 0.7)  # anneal tau
        
        # Nesterov extrapolation
        t_prev = t_mom
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        beta = (t_prev - 1.0) / t_mom
        y = a + beta * (a - a_prev)
        y = np.clip(y, 0, None)
        sy = np.sum(y)
        if sy > 0:
            y = y / sy
        
        # Compute smooth max and gradient
        value, grad, conv = smooth_max_grad(y, tau)
        
        # Step size: 1/L where L is Lipschitz constant
        # For the smooth max, L ≈ (2/tau) * max_a (roughly)
        # More carefully: L = max ||Hessian of smooth max|| 
        # The Hessian of sum_k w_k * conv[k] involves the w_k and the a values.
        # A safe upper bound: L <= 2 * sum_k w_k * max(a) * 2 = 4 * max(a)
        # But that's very loose. Let me use L = 4 * np.max(a) / tau * tau = 4*max(a)
        # Actually for the soft-max smoothing, the Hessian of the log-sum-exp part is bounded by 1/tau * (diag(exp) - exp exp^T) / sum(exp)
        # This is bounded by 1/(4*tau) for the unnormalized case.
        # Combined with the 2x from the autocorrelation structure...
        # Let me just use a adaptive step size.
        
        # Simple approach: use step = 1 / (4 * max(grad))
        max_g = np.max(np.abs(grad))
        if max_g < 1e-15:
            break
        step_size = 1.0 / (4.0 * max_g)
        # Cap the step size
        step_size = min(step_size, 0.1)
        
        candidate = y - step_size * grad
        candidate = np.clip(candidate, 0, None)
        sc = np.sum(candidate)
        if sc < 1e-10:
            candidate = np.ones(n) / n
        else:
            candidate = candidate / sc
        
        a_prev = a
        a = candidate
        step += 1
        
        # Evaluate

            if actual_val < best_val:
                best_val = actual_val

            if step % 500 == 0:
                print(f"      Nesterov step={step}, val={actual_val:.8f}, "
                      f"score={2*n*actual_val:.6f}, tau={tau:.2e}, t={elapsed:.1f}s")
    
    return best_a, 2.0 * n * best_val

def coordinate_polish(a_init, time_limit, rng):
    """Fast coordinate descent polish."""
    a = normalize_to_simplex(a_init.copy())
    n = len(a)
    current_score = 2.0 * n * np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    patience = 0
    max_patience = 3000
    
    while time.monotonic() - t0 < time_limit and patience < max_patience:
        r = rng.random()
        
        if r < 0.5:
            # Pair transfer
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.2)
            amt = a[i] * frac
            if amt > a[i]:
                amt = a[i] * 0.5
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
        elif r < 0.8:
            # Single scale
            j = int(rng.integers(n))
            factor = np.exp(rng.normal(0, 0.05))
            candidate = a.copy()
            candidate[j] *= factor
        else:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - width))
            candidate = a.copy()
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.03, size=width))
        
        candidate = np.clip(candidate, 0.0, None)
        cs = np.sum(candidate)
        if cs < 1e-10:
            continue
        candidate = candidate / cs
        
        new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
            patience = 0
        else:
            patience += 1
    
    return best_a, best_score

"""
Multi-strategy optimization for minimizing peak autocorrelation of non-negative sequences.
Uses LP-based Frank-Wolfe direction finding with exact step size, Nesterov-accelerated
smooth subgradient, and coordinate descent polishing. Targets score < 1.5027.

import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def normalize_to_simplex(a):
    """Normalize to non-negative simplex."""
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def spectral_init(n, rng, n_iter=50):
    """Spectral iterative initialization."""
    a = 1.0 + 0.5 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = normalize_to_simplex(a)
    for it in range(n_iter):
        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        corr = np.fft.irfft(power, 2 * n)[:n]
        alpha = 1.0 / (n + 1.0)
        a = a - alpha * corr
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def make_varied_init(n, rng, style='spectral'):
    """Create diverse initializations."""
    if style == 'spectral':
        return spectral_init(n, rng, n_iter=40)
    elif style == 'ramp':
        t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = 1.0 + 0.3 * t + 0.02 * rng.standard_normal(n)
        return normalize_to_simplex(np.clip(a, 0.01, None))
    elif style == 'bimodal':
        t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = 1.0 + 0.5 * np.exp(-((t - 0.3) ** 2) / 0.05) + 0.3 * np.exp(-((t - 0.7) ** 2) / 0.05)
        a += 0.02 * rng.standard_normal(n)
        return normalize_to_simplex(np.clip(a, 0.001, None))
    elif style == 'cosine':
        t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a += 0.05 * rng.standard_normal(n)
        return normalize_to_simplex(np.clip(a, 0.001, None))
    elif style == 'triangular':
        t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = 1.0 - 0.4 * np.abs(2 * t - 1)
        a += 0.02 * rng.standard_normal(n)
        return normalize_to_simplex(np.clip(a, 0.001, None))
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return normalize_to_simplex(np.clip(a, 0.01, None))


def build_lp_matrix(f):
    """Build the constraint matrix A for the LP. A[k,j] = f[k-j] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_solve(f, M):
    """Solve LP: max sum(b) s.t. f*b <= M (all lags), b >= 0. Returns b or None."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def find_optimal_t(c_f, d_fg, e_gg):
    """
    Find t in [0,1] minimizing max_k [(1-t)^2 c_k + 2t(1-t) d_k + t^2 e_k].
    Uses ternary search since the objective is convex in t.
    """
    func_a = c_f  # coefficient of (1-t)^2
    func_b = 2.0 * d_fg  # coefficient of t(1-t)
    func_c = e_gg  # coefficient of t^2

    def obj(t):
        val = (1.0 - t) ** 2 * func_a + t * (1.0 - t) * func_b + t ** 2 * func_c
        return np.max(val)

    # Ternary search
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def lp_fw_iterate(f, max_iters=500, time_limit=300.0, verbose=True):
    """
    LP-based Frank-Wolfe iteration with exact step size.
    """
    n = len(f)
    f = normalize_to_simplex(f.copy())
    best_f = f.copy()
    best_M = np.max(fft_convolve(f, f))
    t0 = time.monotonic()
    precompute_A = True

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv_f = fft_convolve(f, f)
        M = np.max(conv_f)

        # Build LP matrix (or reuse if f hasn't changed much)
        A = build_lp_matrix(f)
        c_lp = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c_lp, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

        if not res.success:
            if verbose:
                print(f"    LP failed at iter {it}: {res.message}")
            break

        b = res.x
        sum_b = np.sum(b)
        if sum_b < 1e-10:
            break

        g = b / sum_b  # normalize to simplex

        # Find optimal step size
        conv_fg = fft_convolve(f, g)
        conv_gg = fft_convolve(g, g)

        t_opt = find_optimal_t(conv_f, conv_fg, conv_gg)
        t_opt = np.clip(t_opt, 0.01, 0.5)  # don't go too far in one step

        # Update
        f_new = (1.0 - t_opt) * f + t_opt * g
        f_new = np.clip(f_new, 0.0, None)
        s = np.sum(f_new)
        if s < 1e-10:
            break
        f_new = f_new / s

        conv_new = fft_convolve(f_new, f_new)
        M_new = np.max(conv_new)

        if M_new < best_M:
            best_M = M_new
            best_f = f_new.copy()

        f = f_new

        if verbose and (it + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter {it+1}: M={M_new:.8f}, best={best_M:.8f}, t={t_opt:.3f}, time={elapsed:.1f}s")

        # Convergence check
        if it > 10 and abs(M - M_new) < 1e-14:
            if verbose:
                print(f"    Converged at iter {it+1}")
            break

    return best_f, 2.0 * n * best_M


def smooth_max_and_grad(a, tau):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    shifted = shifted - np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    value = max_c + tau * (np.max(shifted) + np.log(np.sum(exp_vals)))

    # Compute gradient: grad[j] = sum_k w[k] * (a[k-j]_valid + a[j-k]_valid)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        # Term 1: a[k-j] valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            indices = np.arange(lo1, hi1 + 1)
            grad[lo1:hi1 + 1] += wk * a[k - indices]
        # Term 2: a[j-k] valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2 and k < n:
            indices = np.arange(lo2, hi2 + 1)
            grad[lo2:hi2 + 1] += wk * a[indices - k]

    return value, grad, conv


def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_smooth_max(a_init, time_limit=200.0, verbose=True):
    """Nesterov-accelerated proximal gradient on smooth max of autocorrelation."""
    n = len(a_init)
    a = normalize_to_simplex(a_init.copy())
    a_prev = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_val = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        # Anneal tau from large to small
        tau = (10.0 / n) * (0.005) ** (frac ** 0.7)

        # Nesterov extrapolation
        t_prev = t_mom
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        beta = (t_prev - 1.0) / t_mom
        y = a + beta * (a - a_prev)
        y = project_simplex(y)

        # Compute smooth max and gradient
        value, grad, conv = smooth_max_and_grad(y, tau)

        # Adaptive step size
        max_g = np.max(np.abs(grad))
        if max_g < 1e-15:
            break
        step_size = 1.0 / (4.0 * max_g + 1e-10)
        step_size = min(step_size, 0.05)

        candidate = y - step_size * grad
        candidate = project_simplex(candidate)

        a_prev = a
        a = candidate
        step += 1

        if step % 100 == 0:
            actual_val = np.max(fft_convolve(a, a))
            if actual_val < best_val:
                best_val = actual_val
                best_a = a.copy()
            if verbose and step % 500 == 0:
                print(f"    Nesterov step={step}, score={2*n*actual_val:.6f}, "
                      f"best_score={2*n*best_val:.6f}, tau={tau:.2e}, t={elapsed:.1f}s")

    # Final check
    actual_val = np.max(fft_convolve(a, a))
    if actual_val < best_val:
        best_val = actual_val
        best_a = a.copy()

    return best_a, 2.0 * n * best_val


def coordinate_polish(a_init, time_limit=200.0, rng=None):
    """Fast coordinate descent for final polishing."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = normalize_to_simplex(a_init.copy())
    n = len(a)
    current_score = 2.0 * n * np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    patience = 0
    max_patience = 5000

    while time.monotonic() - t0 < time_limit and patience < max_patience:
        r = rng.random()

        if r < 0.4:
            # Pair transfer
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.2)
            amt = min(a[i] * frac, a[i] * 0.5)
            candidate = a.copy()
            candidate[i] -= amt
            candidate[j] += amt
        elif r < 0.7:
            # Single scale
            j = int(rng.integers(n))
            factor = np.exp(rng.normal(0, 0.04))
            candidate = a.copy()
            candidate[j] *= factor
        elif r < 0.9:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - width))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.02, size=width))
        else:
            # Peak-targeted: reduce elements contributing to the peak
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Find elements contributing most to conv[peak_idx]
            contribs = np.zeros(n)
            j_start = max(0, peak_idx - n + 1)
            j_end = min(n - 1, peak_idx)
            for j in range(j_start, j_end + 1):
                i = peak_idx - j
                contribs[j] = a[i]
                contribs[i] += a[j]
            top_k = max(3, n // 32)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate = a.copy()
            candidate[top_idx] *= (1.0 - rng.uniform(0.01, 0.05, size=len(top_idx)))
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_add = min(5, len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                total_removed = np.sum(a[top_idx] - candidate[top_idx])
                candidate[add_idx] += total_removed / max(n_add, 1)

        candidate = np.clip(candidate, 0.0, None)
        cs = np.sum(candidate)
        if cs < 1e-10:
            continue
        candidate = candidate / cs

        new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
            patience = 0
        else:
            patience += 1

    return best_a, best_score


def interp_to_length(a, n_new):
    """Interpolate sequence to new length."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize_to_simplex(np.clip(a_new, 0.001, None))

            a0 = make_varied_init(n, rng, init_style)

        print(f"\n  LP-FW: n={n}, init='{init_style}', iters<={max_iters}, budget={t_left:.0f}s")
        a_best, sc = lp_fw_iterate(a0, max_iters=max_iters, time_limit=t_left, verbose=True)
        a_scaled = a_best * np.sqrt(2.0 * n)
        actual_score = score_fast(a_scaled)
        print(f"  Result: score={actual_score:.6f}, time={time.monotonic()-t_global:.1f}s")

        if actual_score < best_score:
            best_score = actual_score
            best_seq = a_scaled.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # ============ Phase 2: Nesterov smooth subgradient ============
    print(f"\n=== Phase 2: Nesterov optimization ===")
    print(f"  Current best: {best_score:.6f}")

    nest_configs = [
        (512, 'parent', 200),
        (512, 'spectral', 180),
        (768, 'spectral', 150),
        (640, 'bimodal', 150),
    ]

    if start_seq is not None:
        nest_configs.insert(0, (len(start_seq), 'parent', 200))

    for n, init_style, time_alloc in nest_configs:
        if time.monotonic() > deadline - 200:
            break
        t_left = min(time_alloc, deadline - 200 - time.monotonic())
        if t_left < 15:
            break

        if init_style == 'parent' and start_seq is not None:
            if n == len(start_seq):
                a0 = normalize_to_simplex(start_seq)
            else:
                a0 = interp_to_length(start_seq, n)
        else:
            a0 = make_varied_init(n, rng, init_style)

        # If we have a best sequence, also try starting from it
        if best_seq is not None and n == len(best_seq):
            a0 = normalize_to_simplex(best_seq)

        print(f"\n  Nesterov: n={n}, init='{init_style}', budget={t_left:.0f}s")
        a_best, sc = nesterov_smooth_max(a0, time_limit=t_left, verbose=True)
        a_scaled = a_best * np.sqrt(2.0 * n)
        actual_score = score_fast(a_scaled)
        print(f"  Result: score={actual_score:.6f}, time={time.monotonic()-t_global:.1f}s")

        if actual_score < best_score:
            best_score = actual_score
            best_seq = a_scaled.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # ============ Phase 3: Coordinate polish ============
    print(f"\n=== Phase 3: Coordinate polish ===")
    print(f"  Best before polish: {best_score:.6f}")
    if best_seq is not None and time.monotonic() < deadline - 100:
        polish_time = min(300.0, deadline - time.monotonic() - 100)
        a_norm = best_seq / np.sum(best_seq)
        polished, polish_score = coordinate_polish(a_norm, polish_time, rng)
        polished_scaled = polished * np.sqrt(2.0 * len(polished))
        final_score = score_fast(polished_scaled)
        print(f"  After polish: {final_score:.6f}")
        if final_score < best_score:
            best_score = final_score
            best_seq = polished_scaled.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # ============ Phase 4: Extended refinement if time remains ============
    if time.monotonic() < deadline - 50 and best_seq is not None:
        remaining = deadline - time.monotonic() - 50
        if remaining > 60:
            a_norm = best_seq / np.sum(best_seq)
            n_best = len(a_norm)
            print(f"\n=== Phase 4: Extended LP polish (n={n_best}, time={remaining:.0f}s) ===")
            # Do more LP iterations on the current best
            est_iters = int(remaining / 1.0)  # assume ~1s per LP iteration
            a_refined, sc = lp_fw_iterate(a_norm, max_iters=est_iters,
                                          time_limit=min(remaining - 30, 300), verbose=True)
            a_scaled = a_refined * np.sqrt(2.0 * n_best)
            actual_score = score_fast(a_scaled)
            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_scaled.copy()
                print(f"  *** New best: {best_score:.6f} ***")
            else:
                print(f"  No improvement: {actual_score:.6f}")

            # Another round of coordinate polish
            if time.monotonic() < deadline - 30:
                remaining2 = deadline - time.monotonic() - 30
                if remaining2 > 30:
                    a_norm2 = best_seq / np.sum(best_seq)
                    polished2, ps2 = coordinate_polish(a_norm2, min(remaining2, 120), rng)
                    p2_scaled = polished2 * np.sqrt(2.0 * len(polished2))
                    s2 = score_fast(p2_scaled)
                    if s2 < best_score:
                        best_score = s2
                        best_seq = p2_scaled.copy()
                        print(f"  *** New best: {best_score:.6f} ***")

    # ============ Final verification ============
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
