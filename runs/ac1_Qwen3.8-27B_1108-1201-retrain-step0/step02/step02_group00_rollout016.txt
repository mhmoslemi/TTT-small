    """
    n = len(z)
    # Softmax
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    p = exp_z / exp_z.sum()  # p sums to 1, all positive
    
    # Convolution
    c = fft_convolve_self(p)
    
    # Log-sum-exp
    c_max = c.max()
    lse_val = (c_max + np.log(np.sum(np.exp(beta * (c - c_max))))) / beta
    obj = 2.0 * n * lse_val
    
    # Gradient w.r.t. p
    exp_bc = np.exp(beta * (c - c_max))
    w = exp_bc / exp_bc.sum()
    
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    P = fft(p, fsize)
    grad_p = 2.0 * np.real(ifft(W * np.conj(P)))[:n]
    
    # Chain rule: grad_z = Jacobian(softmax)^T * grad_p
    # p = softmax(z), so dp[i]/dz[j] = p[i]*(delta_ij - p[j])
    # grad_z[j] = sum_i grad_p[i] * p[i] * (delta_ij - p[j])
    #            = grad_p[j]*p[j] - p[j]*sum_i grad_p[i]*p[i]
    #            = p[j] * (grad_p[j] - dot(grad_p, p))
    dot = np.dot(grad_p, p)
    grad_z = 2.0 * n * p * (grad_p - dot)
    
    return obj, grad_z
