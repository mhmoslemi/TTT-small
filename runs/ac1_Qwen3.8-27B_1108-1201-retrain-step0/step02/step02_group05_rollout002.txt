import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def local_score(a):
    """Score for sequence: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def soft_max_obj_grad(theta, tau):
    """Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau)."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    score = 2.0 * n * soft_max_val

    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta

def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Bell-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def make_sawtooth_init(n, rng, periods=3):
    """Sawtooth pattern initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.4 * (2.0 * ((t * periods / n) % 1.0) - 1.0)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_multisine_init(n, rng):
    """Multi-sine initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0
    for k in range(3):
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        a += 0.2 * np.sin(2 * np.pi * freq * t / n + phase)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def gradient_peak_chase(a, time_limit, rng, max_iters=5000):
    """
    Gradient-based peak chasing: repeatedly find the peak lag, compute exact gradient,
    and take a line-search step in the negative gradient direction (in theta space).
    """
    current = a.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()

    theta = np.log(current + 1e-15)
    
    step_size = 0.1
    iters_since_improve = 0
    max_stall = 200

    iteration = 0
    while time.monotonic() - t0 < time_limit and iteration < max_iters:
        c = fft_convolve(current, current)
        peak_idx = int(np.argmax(c))

        val, grad_theta = peak_grad_obj(theta, peak_idx)

        if np.max(np.abs(grad_theta)) < 1e-12:
            # Try neighboring lags
            improved = False
            for dk in (-1, 1, -2, 2):
                k2 = peak_idx + dk
                if 0 <= k2 < 2 * n - 1:
                    val2, grad2 = peak_grad_obj(theta, k2)
                    if np.max(np.abs(grad2)) > 1e-12:
                        grad_theta = grad2
                        improved = True
                        break
            if not improved:
                break

        # Line search: try to decrease the actual score
        best_step_val = np.inf
        best_step = None
        base_score = local_score(current)

        for ls in [step_size, step_size * 0.5, step_size * 0.25, step_size * 0.1,
                   step_size * 2, step_size * 0.05]:
            new_theta = theta - ls * grad_theta
            new_a = softmax(new_theta)
            new_score = local_score(new_a)
            if new_score < best_step_val:
                best_step_val = new_score
                best_step = ls

        if best_step is None or best_step_val >= base_score:
            # Gradient step not helping, try a perturbation
            current = current * np.exp(rng.normal(0, 0.01, n))
            current = np.maximum(current, 1e-10)
            current = current / np.sum(current)
            theta = np.log(current + 1e-15)
            current_score = local_score(current)
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
            step_size *= 0.5
            iters_since_improve += 1
        else:
            theta = theta - best_step * grad_theta / step_size * best_step  # already applied
            # Actually recompute:
            theta = theta  # we already modified above? No, we just tried.
            # Let me redo this properly:
            new_theta = theta - best_step * grad_theta
            current = softmax(new_theta)
            theta = new_theta
            current_score = best_step_val
            step_size = min(step_size * 1.05, 1.0)
            iters_since_improve = 0

            if current_score < best_score:
                best_score = current_score
                best = current.copy()

        if iters_since_improve > max_stall:
            # Restart with perturbation
            perturbation_scale = 0.05 + 0.1 * rng.random()
            current = best * np.exp(rng.normal(0, perturbation_scale, n))
            current = np.maximum(current, 1e-10)
            current = current / np.sum(current)
            theta = np.log(current + 1e-15)
            current_score = local_score(current)
            iters_since_improve = 0
            step_size = 0.1

        iteration += 1

    return best, best_score


def water_filling_search(a_init, time_limit, rng):
    """
    Water-filling local search: iteratively identify top-K lags and redistribute
    energy from high-contributing pairs to low-contributing pairs.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()

    temperature = 0.005
    cooling = 0.99995
    target_k = max(3, n // 32)

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        c_sorted_idx = np.argsort(c)[::-1]
        top_lags = c_sorted_idx[:target_k]

        # Compute "responsibility" for the top lags
        pull = np.zeros(n)
        for k in top_lags:
            weight = c[k] - np.mean(c)
            if weight <= 0:
                continue
            # For lag k, the contributing pairs are (i, k-i)
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    pull[i] += weight * current[j]
                    if i != j:
                        pull[j] += weight * current[i]

        candidate = current.copy()
        if np.max(pull) > 1e-10:
            pull_norm = pull / np.max(pull)
            # Reduce high-pull elements, slightly boost low-pull
            reduction = 0.01 + 0.02 * rng.random()
            candidate *= np.exp(-reduction * pull_norm)

        # Add random perturbation
        candidate *= np.exp(rng.normal(0, 0.01, n))
        candidate = np.maximum(candidate, 1e-10)
        candidate = candidate / np.sum(candidate)

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 1e-10)

    return best, best_score

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < phase_dur * 0.5:
            phase_dur = max(5.0, remaining * 0.9)

        r = rng.random()
        if r < 0.4:
            # Gradient peak chase
            phase_time = min(phase_dur, remaining)
            polished, p_score = gradient_peak_chase(current, phase_time, rng, max_iters=2000)
            if p_score < best_score:
                best_score = p_score
                best = polished.copy()
            current = polished
            current_score = p_score
        elif r < 0.7:
            # Water filling
            phase_time = min(phase_dur, remaining)
            polished, p_score = water_filling_search(current, phase_time, rng)
            if p_score < best_score:
                best_score = p_score
                best = polished.copy()
            current = polished
            current_score = p_score
        else:
            # Simulated annealing style
            phase_time = min(phase_dur, remaining)
            t_start = time.monotonic()
            while time.monotonic() - t_start < phase_time:
                candidate = current.copy()
                move_type = rng.random()
                if move_type < 0.4:
                    # Targeted: reduce elements contributing to peak
                    c = fft_convolve(current, current)
                    peak_idx = int(np.argmax(c))
                    contribs = np.zeros(n)
                    lo = max(0, peak_idx - (n - 1))
                    hi = min(n - 1, peak_idx)
                    for i in range(lo, hi + 1):
                        j = peak_idx - i
                        if 0 <= j < n:
                            contribs[i] = 2.0 * current[i] * current[j]
                    top_k = max(5, n // 8)
                    top_idx = np.argsort(contribs)[-top_k:]
                    candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.01, size=len(top_idx)))
                    others = np.setdiff1d(np.arange(n), top_idx)
                    if len(others) > 0:
                        n_inc = min(10, len(others))
                        inc_idx = rng.choice(others, size=n_inc, replace=False)
                        candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.01, size=n_inc))
                elif move_type < 0.7:
                    # Block perturbation
                    width = max(1, int(rng.integers(1, max(2, n // 6))))
                    start = int(rng.integers(0, n - width))
                    candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))
                else:
                    # Global small
                    candidate *= np.exp(rng.normal(0.0, 0.02, n))

                candidate = np.maximum(candidate, 1e-10)
                candidate = candidate / np.sum(candidate)
                new_score = local_score(candidate)

                if new_score < current_score:
                    current = candidate
                    current_score = new_score
                    if current_score < best_score:
                        best_score = current_score
                        best = current.copy()
                elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
                    current = candidate
                    current_score = new_score

                temperature = max(temperature * cooling, 1e-12)

    return best, best_score


def run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p, deadline):
    """Run L-BFGS with annealed p-norm."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))

    for p in p_schedule:
        if time.monotonic() > deadline - 10:
            break
        result = minimize(
            pnorm_obj_grad, theta.copy(),
            method='L-BFGS-B', jac=True, args=(p,),
            options={'maxiter': maxiter_per_p, 'ftol': 1e-14, 'gtol': 1e-12,
                     'maxfun': maxiter_per_p * 5}
        )
        a_new = softmax(result.x)
        s = local_score(a_new)
        if s < best_score:
            best_score = s
            best_theta = result.x.copy()
        theta = best_theta

    return best_theta, best_score


def run_lbgfs_softmax(theta, tau_schedule, maxiter_per_tau, deadline):
    """Run L-BFGS with annealed soft-max temperature."""
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))

    for tau in tau_schedule:
        if time.monotonic() > deadline - 10:
            break
        result = minimize(
            soft_max_obj_grad, theta.copy(),
            method='L-BFGS-B', jac=True, args=(tau,),
            options={'maxiter': maxiter_per_tau, 'ftol': 1e-14, 'gtol': 1e-12,
                     'maxfun': maxiter_per_tau * 5}
        )
        a_new = softmax(result.x)
        s = local_score(a_new)
        if s < best_score:
            best_score = s
            best_theta = result.x.copy()
        theta = best_theta

    return best_theta, best_score


def run_lbgfs_peak(theta, peak_lag, maxiter, deadline):
    """Run L-BFGS targeting a specific peak lag."""
    if time.monotonic() > deadline - 5:
        return theta, local_score(softmax(theta))

    result = minimize(
        peak_grad_obj, theta.copy(),
        method='L-BFGS-B', jac=True, args=(peak_lag,),
        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12,
                 'maxfun': maxiter * 5}
    )
    return result.x, local_score(softmax(result.x))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization combining L-BFGS, gradient peak chasing,
    water-filling local search, and diverse initializations.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0  # safety margin
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        best_score = s
        best_seq = start_seq.copy()

    # Phase 1: L-BFGS optimization over multiple lengths
    print("\n=== Phase 1: L-BFGS optimization ===")
    lengths = [192, 224, 256, 288, 320, 384, 448, 512]

    # Time allocation: more time for lengths near 256 (where we have a good solution)
    total_lbgfs_time = min(900.0, (deadline - t_global) * 0.35)
    time_per_length = total_lbgfs_time / len(lengths)

    for n in lengths:
        if time.monotonic() > deadline - 600:
            break

        time_alloc = time_per_length * (1.5 if n == 256 else 1.0)
        n_sub_deadline = time.monotonic() + min(time_alloc, deadline - t_global - 500)

        # Generate initializations
        inits = []

        # Interpolated from best sequence
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.maximum(a_interp, 1e-6)
            a_interp = a_interp / np.sum(a_interp)
            inits.append(('interp', a_interp))

            # Perturbed interpolation
            a_pert = a_interp * np.exp(rng.normal(0, 0.15, n))
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert = a_pert / np.sum(a_pert)
            inits.append(('interp_pert', a_pert))

        # Structured initializations
        inits.append(('chirp', make_chirp_init(n, 0.3 + 0.4 * rng.random(), rng)))
        inits.append(('random', make_random_init(n, rng, spread=0.2 + 0.3 * rng.random())))
        inits.append(('bell', make_bell_init(n, n * (0.1 + 0.2 * rng.random()), rng)))
        inits.append(('sawtooth', make_sawtooth_init(n, rng, periods=int(rng.integers(2, 7)))))
        inits.append(('multisine', make_multisine_init(n, rng)))

        p_schedule = [4, 8, 16, 32, 64, 128, 256]
        tau_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

        for init_name, a0 in inits:
            if time.monotonic() > n_sub_deadline:
                break

            theta = np.log(a0 + 1e-30)

            # P-norm L-BFGS
            theta_best, s = run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p=150,
                                             deadline=n_sub_deadline)

            # Soft-max refinement
            if s < best_score or time.monotonic() < n_sub_deadline - 20:
                theta_sm, s_sm = run_lbgfs_softmax(theta_best, tau_schedule,
                                                    maxiter_per_tau=200,
                                                    deadline=n_sub_deadline)
                if s_sm < s:
                    s = s_sm
                    theta_best = theta_sm

            # Peak targeting: find the peak and try to reduce it specifically
            if s < best_score and time.monotonic() < n_sub_deadline - 10:
                a_best = softmax(theta_best)
                c = fft_convolve(a_best, a_best)
                peak_lag = int(np.argmax(c))
                theta_pk, s_pk = run_lbgfs_peak(theta_best, peak_lag,
                                                 maxiter=100, deadline=n_sub_deadline)
                if s_pk < s:
                    s = s_pk
                    theta_best = theta_pk

            if s < best_score:
                a_final = softmax(theta_best)
                best_score = s
                best_seq = a_final.copy()
                print(f"  n={n}, init='{init_name}': score={s:.6f} *** NEW BEST ***")
            else:
                print(f"  n={n}, init='{init_name}': score={s:.6f}")

    # Phase 2: Extended soft-max refinement on best
    print(f"\n=== Phase 2: Extended soft-max refinement (score={best_score:.6f}) ===")
    if best_seq is not None:
        n = len(best_seq)
        theta = np.log(best_seq + 1e-30)
        sm_deadline = min(time.monotonic() + 200, deadline - 400)
        tau_ext = [0.005, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        theta_best, s = run_lbgfs_softmax(theta, tau_ext, maxiter_per_tau=300,
                                           deadline=sm_deadline)
        if s < best_score:
            best_score = s
            best_seq = softmax(theta_best).copy()
            print(f"  Soft-max improved: {best_score:.6f} *** NEW BEST ***")

    # Phase 3: Gradient peak chase + local search
    print(f"\n=== Phase 3: Gradient peak chase (score={best_score:.6f}) ===")
    if best_seq is not None:
        chase_time = min(300.0, max(120.0, deadline - time.monotonic() - 300))
        print(f"  Chase time: {chase_time:.0f}s")
        chased, c_score = gradient_peak_chase(best_seq, chase_time, rng, max_iters=10000)
        if c_score < best_score:
            best_score = c_score
            best_seq = chased.copy()
            print(f"  After chase: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  After chase: {c_score:.6f} (no improvement)")

    # Phase 4: Combined local search (water-filling + SA)
    print(f"\n=== Phase 4: Combined local search (score={best_score:.6f}) ===")
    if best_seq is not None:
        ls_time = min(400.0, max(100.0, deadline - time.monotonic() - 30))
        print(f"  Search time: {ls_time:.0f}s")
        polished, p_score = combined_local_search(best_seq, ls_time, rng)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After local search: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  After local search: {p_score:.6f} (no improvement)")

    # Phase 5: Final micro-refinement with peak targeting
    print(f"\n=== Phase 5: Final micro-refinement (score={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n = len(best_seq)
        theta = np.log(best_seq + 1e-30)
        c = fft_convolve(best_seq, best_seq)
        # Try top 3 peak lags
        top_lags = np.argsort(c)[::-1][:3]
        for lag in top_lags:
            if time.monotonic() > deadline - 20:
                break
            theta_new, s_new = run_lbgfs_peak(theta, int(lag), maxiter=200,
                                               deadline=deadline - 15)
            if s_new < best_score:
                best_score = s_new
                best_seq = softmax(theta_new).copy()
                print(f"  Peak lag {lag}: score={s_new:.6f} *** NEW BEST ***")

    # Final verification and return
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
