def segment_sequence(n, k, heights, transition=0.0):
    """Create a sequence with k segments of given heights and smooth transitions."""
    m = n // k
    seq = np.zeros(n)
    for i in range(k):
        start = i * m
        end = start + m
        if i < k - 1:
            end = min(end, n)
        seg = np.full(end - start, heights[i])
        # Add smooth transitions
        if transition > 0 and i > 0:
            t = np.linspace(0, 1, transition)
            seg[:transition] = heights[i-1] + (heights[i] - heights[i-1]) * t
        if transition > 0 and i < k-1:
            t = np.linspace(0, 1, transition)
            seg[-transition:] = heights[i] + (heights[i+1] - heights[i]) * t
        seq[start:end] = seg[:end-start]
    return np.maximum(seq, 0)

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_score(a):
    """Compute the objective value (before the factor 2n/sum^2, assumes normalized)."""
    c = fft_convolve_self(a)
    return c.max()


def compute_full_objective(a):
    """Compute the full objective 2*n*max_conv/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def fft_grad_toppk(a, c, top_k):
    """Gradient using top-k active lags, computed via FFT cross-correlation."""
    n = len(a)
    if top_k >= len(c):
        top_k = len(c)
    top_indices = np.argsort(c)[::-1][:top_k]
    w_full = np.zeros(2 * n - 1)
    for k in top_indices:
        w_full[int(k)] = 1.0 / top_k

    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_lse(a, c, beta):
    """Gradient of log-sum-exp approximation to max."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

            y = project_simplex(y + noise, S)

            stall_count = 0

def peak_shaving(a_init, n, S, max_time, rng, num_shaves=10):

            best_score = cur
            best_local = a.copy()

        # Find top peaks
        top_k = min(num_shaves, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        mean_c = c.mean()

        # Compute "shadow" - the gradient direction from peaks
        shadow = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            if k < n:
                # c[k] = sum_{i=0}^{k} a[i]*a[k-i]
                # d c[k]/d a[i] = a[k-i] (for i <= k) + a[i-(k-n+1)]... 
                # Simpler: use FFT-based gradient
                pass

        # Use FFT gradient for the top peaks
        w_full = np.zeros(2 * n - 1)
        for k in peak_indices:
            w_full[int(k)] = 1.0 / top_k

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Step size: proportional to peak excess
        peak_excess = cur - mean_c
        step = 0.1 * peak_excess / max(np.linalg.norm(grad), 1e-15)
        step = min(step, 0.5 * S / n)

        a_new = project_simplex(a - step * grad, S)

        # Also try "spreading" - move mass from high-gradient to low-gradient indices
        grad_mean = grad.mean()
        high_grad = grad > grad_mean + 0.5 * grad.std()
        low_grad = grad < grad_mean - 0.5 * grad.std()
        if np.any(high_grad) and np.any(low_grad):
            transfer = 0.01 * S * 0.01
            n_high = np.sum(high_grad)
            n_low = np.sum(low_grad)
            a_spread = a.copy()
            a_spread[high_grad] -= transfer / max(n_high, 1)
            a_spread[low_grad] += transfer / max(n_low, 1)
            a_spread = project_simplex(a_spread, S)
            c_spread = fft_convolve_self(a_spread)
            if c_spread.max() < best_score:
                best_score = c_spread.max()
                best_local = a_spread.copy()

        a = a_new

    return best_local, best_score


def segment_optimize(n, k, S, max_time, rng):
    """Optimize k segment heights to minimize max(conv)."""
    m = n // k
    actual_n = m * k  # Use multiples of m

    best_score = np.inf
    best_heights = np.ones(k) * S / k

    # Multiple random initializations
    n_restarts = min(50, max(5, int(max_time / 0.5)))
    per_restart = max_time / n_restarts

    for r in range(n_restarts):
        if time.monotonic() > time.monotonic() + max_time:
            break

        # Random initialization
        h = rng.dirichlet(np.ones(k)) * S
        if h.max() < 1e-10:
            h = np.ones(k) * S / k

        # Simple gradient descent on heights
        lr = S / (k * 10.0)
        for it in range(200):
            # Build sequence from heights
            a = np.repeat(h, m)[:actual_n]
            c = fft_convolve_self(a)
            cur = c.max()

            if cur < best_score:
                best_score = cur
                best_heights = h.copy()

            # Gradient: for each segment j, dh_j = sum of contributions
            # Using numerical gradient (cheap since k is small)
            grad_h = np.zeros(k)
            eps = S * 1e-5 / k
            for j in range(k):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = h_plus / h_plus.sum() * S
                a_plus = np.repeat(h_plus, m)[:actual_n]
                c_plus = fft_convolve_self(a_plus)
                grad_h[j] = (c_plus.max() - cur) / eps

            h_new = h - lr * grad_h
            h_new = np.maximum(h_new, 0.0)
            h_new = h_new / h_new.sum() * S

            # Adaptive learning rate
            if grad_h.std() > 0:
                lr = min(lr * 1.1, S / (k))
            else:
                break
            h = h_new

    # Build final sequence
    a = np.repeat(best_heights, m)[:actual_n]
    # Pad if needed
    if actual_n < n:
        a = np.concatenate([a, np.zeros(n - actual_n)])
    return a, best_score

            a_ps, score_ps = peak_shaving(a_res, n, S, min(5.0, remaining2 * 0.3), rng)
            update_best(a_ps)

    # FISTA with log-sum-exp on best candidate
    if global_a is not None and time.monotonic() < p2_deadline - 5:
        remaining = p2_deadline - time.monotonic()
        alloc = min(30.0, remaining * 0.5)
        a_fista, score_f = fista_lse(global_a, n, S, alloc, rng,
                                      beta_init=10.0, beta_max=400.0)
        update_best(a_fista)

    print("Phase 2 done. Best: %.6f" % global_best)

    # --- Phase 3: Final polishing (remaining time) ---
    print("=== Phase 3: Polishing ===")
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            # Multiple short polishing rounds with different params
            for round_i in range(5):
                rem = deadline - time.monotonic()
                if rem < 3:
                    break
                alloc = min(remaining / (5 - round_i), rem * 0.6)
                top_k_init = 3 + round_i * 3
                a_pol, score_pol = nesterov_toppk(global_a, n, S, alloc, rng,
                                                   top_k_init=top_k_init,
                                                   top_k_max=min(256, 32 + round_i * 32),
                                                   oscillate=(round_i % 2 == 0),
                                                   step_scale=1.5 + 0.3 * round_i)
                update_best(a_pol)

            # Final FISTA with high beta
            rem = deadline - time.monotonic()
            if rem > 5:
                a_fin, score_fin = fista_lse(global_a, n, S, min(20.0, rem * 0.5), rng,
                                              beta_init=100.0, beta_max=800.0)
                update_best(a_fin)

    # --- Phase 4: Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.6f" %
          (len(global_a), global_best))
    return [float(x) for x in global_a]
