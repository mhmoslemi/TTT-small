import numpy as np
import time

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def score(a):
    """Compute the evaluation score."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s ** 2)

def softmax_topk(c, tau):

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    w = e / np.sum(e)
    return c_max + tau * np.log(np.sum(e)), w

def grad_softmax_conv(a, tau, N=None):
    """Compute soft-max(conv(a,a), tau) and gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    value, w = softmax_topk(c, tau)
    # Gradient: 2 * cross_correlation(w, a)
    if N is None:
        N = 1
        while N < 3*n - 2:
            N <<= 1
    Fa = np.fft.fft(a, N)
    Fw = np.fft.fft(w, N)
    grad = 2.0 * np.fft.ifft(Fw * np.conj(Fa)).real[:n]
    return value, grad, c

"""
Optimization of non-negative step functions to minimize peak autocorrelation ratio.
Uses Adam gradient descent with soft-max loss, multiple initializations,
and local search polish.

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def compute_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s ** 2)

def soft_max_value_weights(c, tau):
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    w = e / s
    value = c_max + tau * np.log(s)
    return value, w

def grad_softmax_conv(a, tau, N):
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_value_weights(c, tau)
    Fa = np.fft.fft(a, N)
    Fw = np.fft.fft(w, N)
    grad = 2.0 * np.fft.ifft(Fw * np.conj(Fa)).real[:n]
    return value, grad, c

def init_fft_phase(n, rng, n_iter=100):
    a = np.ones(n) / n
    N = 1
    while N < 2*n - 1:
        N <<= 1
    for _ in range(n_iter):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a

def init_chirp(n, rng, alpha=0.5):
    t = np.arange(n, dtype=np.float64)
    c = rng.uniform(0.1, 0.5)

    return a / np.sum(a)

def adam_optimize(a_init, time_limit, rng, tau_start=1.0, tau_end=0.01):
    n = len(a_init)
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    a = a / np.sum(a)
    
    N = 1
    while N < 3*n - 2:
        N <<= 1
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.05
    step = 0
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    best_sw = best_score
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        
        value, grad, c = grad_softmax_conv(a, tau, N)
        
        # Project gradient (maintain sum constraint)
        grad = grad - np.mean(grad)
        
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new / s

            sc = compute_score(a_new)

        a = a_new
        
        # Adaptive lr
        lr = 0.05 / (1.0 + 0.01 * step) ** 0.5

def lp_complement(f, M):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)[k] <= M for all k."""
    n = len(f)
    # (f*b)[k] = sum_i f[i]*b[k-i] <= M for k=0,...,2n-2
    # Variables: b[0],...,b[n-1]
    # Constraints: for each k, sum_{valid i} f[i]*b[k-i] <= M
    # Objective: maximize sum(b) → minimize -sum(b)
    
    c_obj = -np.ones(n)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            A_ub[k, j] += f[i]
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

"""
Optimization of non-negative step functions to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses Adam gradient descent with annealed soft-max loss on the autocorrelation,
multiple initializations (FFT-phase, chirp, monotone, random), multi-length search,
and local search polish. Key insight: the gradient of soft-max(conv) w.r.t. a
is 2*cross_correlation(weights, a), computed efficiently via FFT.
"""
import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score: 2n*max(conv)/sum^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s ** 2)


def soft_max_value_weights(c, tau):
    """Compute soft-max value and normalized weights."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    w = e / s
    value = c_max + tau * np.log(s)
    return value, w


def grad_softmax_conv(a, tau, N):
    """
    Compute soft-max(conv(a,a), tau) and its gradient w.r.t. a.
    Uses FFT for efficient cross-correlation.
    N must be >= 3*len(a)-2 (next power of 2).
    """
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_value_weights(c, tau)
    Fa = np.fft.fft(a, N)
    Fw = np.fft.fft(w, N)
    grad = 2.0 * np.fft.ifft(Fw * np.conj(Fa)).real[:n]
    return value, grad, c


def init_fft_phase(n, rng, n_iter=100):
    """Gerchberg-Saxton style: flatten magnitude spectrum."""
    a = np.ones(n) / n
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    for _ in range(n_iter):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def init_chirp(n, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    c = rng.uniform(0.1, 0.5)
    alpha = rng.uniform(0.3, 0.8)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * c * t * t / n)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def init_monotone(n, rng):
    """Monotone increasing initialization (reduces center peak)."""
    t = np.arange(n, dtype=np.float64)
    a = t + 1.0
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def init_random(n, rng, spread=0.3):
    """Random perturbation of uniform."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def init_two_peak(n, rng):
    """Two Gaussian peaks at 1/4 and 3/4."""
    t = np.arange(n, dtype=np.float64)
    sigma = n // 6
    a = 1.0 + 0.5 * np.exp(-((t - n * 0.25) ** 2) / (2 * sigma ** 2))
    a += 0.5 * np.exp(-((t - n * 0.75) ** 2) / (2 * sigma ** 2))
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def init_edge_concentrated(n, rng):
    """High at edges, low in middle (V-shape)."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 + 1.5 * np.abs(t - center) / center
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def adam_optimize(a_init, time_limit, rng, tau_start=2.0, tau_end=0.02, lr_init=0.1):
    """
    Adam gradient descent on soft-max(conv(a,a)) with annealed temperature.
    Maintains sum(a)=1 and a>=0.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    N = 1
    while N < 3 * n - 2:
        N <<= 1

    m1 = np.zeros(n)
    m2 = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    step = 0

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    no_improve_steps = 0
    patience = 2000

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        # Two-phase: slow decrease then fast
        if frac < 0.6:
            tau = tau_start * (0.1 * tau_start / tau_start) ** (frac / 0.6)
        else:
            tau_frac = (frac - 0.6) / 0.4
            tau = 0.1 * tau_start * (tau_end / (0.1 * tau_start)) ** (tau_frac ** 0.5)

        value, grad, c = grad_softmax_conv(a, tau, N)

        # Project gradient to maintain sum constraint
        grad = grad - np.mean(grad)

        step += 1
        m1 = beta1 * m1 + (1 - beta1) * grad
        m2 = beta2 * m2 + (1 - beta2) * (grad ** 2)
        m1_hat = m1 / (1.0 - beta1 ** step)
        m2_hat = m2 / (1.0 - beta2 ** step)

        lr = lr_init / (1.0 + 0.005 * step) ** 0.5
        a_new = a - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new / s

        # Check score periodically
        if step % 200 == 0:
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                no_improve_steps = 0
            else:
                no_improve_steps += 200
            if step % 2000 == 0:
                print(f"        step={step}, score={sc:.6f}, best={best_score:.6f}, tau={tau:.4f}, t={elapsed:.1f}s")

        a = a_new

        # Restart mechanism: if stuck, perturb from best
        if no_improve_steps > patience and step > 1000:
            a = best_a * np.exp(rng.normal(0, 0.1, n))
            a = np.clip(a, 0.001, None)
            a = a / np.sum(a)
            m1 = np.zeros(n)
            m2 = np.zeros(n)
            step = 0
            no_improve_steps = 0

    final_score = compute_score(best_a)
    if final_score < best_score:
        best_score = final_score
    return best_a, best_score


def local_search_polish(a_init, time_limit, rng, top_k=5):
    """
    Simulated annealing with peak-directed perturbations.
    """
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    current = np.clip(current, 0, None)
    current = current / np.sum(current)
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score

    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9997
    stall = 0
    stall_limit = 3000

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.35:
            # Peak-directed reduction
            c = fft_convolve(current, current)
            n_top = min(top_k, len(c))
            top_lags = np.argsort(c)[-n_top:]
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            n_reduce = max(2, n // 10)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.06, 0.02, size=n_reduce))
            candidate[bot_idx] *= np.exp(rng.normal(0.06, 0.02, size=n_reduce))
        elif r < 0.55:
            # Adjacent block swap-like
            width = max(1, int(rng.integers(1, max(2, n // 20))))
            start = int(rng.integers(0, max(1, n - 2 * width)))
            candidate[start:start + width] *= np.exp(rng.normal(-0.08, 0.02, size=width))
            candidate[start + width:start + 2 * width] *= np.exp(rng.normal(0.08, 0.02, size=width))
        elif r < 0.7:
            # Multi-block perturbation
            n_blocks = int(rng.integers(2, 5))
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 15))))
                bs = int(rng.integers(0, max(1, n - bw)))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0.0, 0.05, size=bw))
        elif r < 0.85:
            # Partial symmetrization
            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.15 * (sym - current)
            candidate += 0.005 * rng.standard_normal(n)
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0.0, 0.015, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current
        else:
            candidate = candidate / s

        new_score = compute_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            stall = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < 0.01:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score
            stall += 1

        temp *= cooling

        # Restart if stuck too long
        if stall > stall_limit:
            current = best * np.exp(rng.normal(0, 0.05, n))
            current = np.clip(current, 0.001, None)
            current = current / np.sum(current)
            current_score = compute_score(current)
            stall = 0
            temp = 0.002

    return best, best_score


def lp_complement_step(f, M, time_limit=30.0):
    """
    Solve LP: max sum(b) s.t. b>=0, (f*b)[k] <= M for all k.
    Returns the optimal b or None.
    """
    n = len(f)
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            A_ub[k, j] += f[i]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    t0 = time.monotonic()
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def propose_candidate(budget_s=2400, **kwargs):
    """
    Multi-phase search for optimal non-negative step function minimizing
    2*n*max(convolve(a,a))/sum(a)^2.
    """

    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(42)

    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = compute_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Phase 1: Generate diverse initial candidates
    print("\n=== Phase 1: Initialization ===")
    candidates = []

    lengths_to_try = [256, 384, 512, 768, 1024]

    for n in lengths_to_try:
        if time.monotonic() > deadline - 600:
            break
        print(f"  n={n}:")
        inits = []
        inits.append(('fft', init_fft_phase(n, rng, n_iter=80)))
        inits.append(('chirp', init_chirp(n, rng)))
        inits.append(('mono', init_monotone(n, rng)))
        inits.append(('rand', init_random(n, rng)))
        inits.append(('edge', init_edge_concentrated(n, rng)))
        inits.append(('2peak', init_two_peak(n, rng)))

        for name, a in inits:
            s = compute_score(a)
            candidates.append((s, n, a.copy(), name))
            print(f"    {name}: {s:.6f}")

    # Sort and keep top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"\n  Top 5: {[(round(c[0], 4), c[1], c[3]) for c in candidates[:5]]}")

    # Phase 2: Gradient optimization on best candidates
    print("\n=== Phase 2: Adam gradient optimization ===")
    n_cand = min(8, len(candidates))
    time_left = deadline - time.monotonic() - 300
    time_per = max(10.0, time_left / n_cand)

    optimized = []
    for i in range(n_cand):
        if time.monotonic() > deadline - 200:
            break
        s0, n, a, name = candidates[i]
        # Scale time allocation by n (larger n = slower per iteration)
        t_alloc = time_per * min(1.5, 512.0 / n)
        t_alloc = max(15.0, min(t_alloc, deadline - time.monotonic() - 100))

        print(f"  [{i + 1}/{n_cand}] n={n}, init={name}, s0={s0:.6f}, t={t_alloc:.0f}s")
        a_opt, s_opt = adam_optimize(a, t_alloc, rng)
        print(f"    -> score={s_opt:.6f}")

        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")

        optimized.append((s_opt, n, a_opt))

    # Phase 3: Coarse-to-fine upsample
    print("\n=== Phase 3: Coarse-to-fine ===")
    if optimized:
        optimized.sort(key=lambda x: x[0])
        best_short = optimized[0]
        for target_n in [512, 1024, 2048]:
            if time.monotonic() > deadline - 100:
                break
            if len(best_short[2]) >= target_n:
                continue
            x_old = np.linspace(0, 1, len(best_short[2]))
            x_new = np.linspace(0, 1, target_n)
            a_up = np.interp(x_new, x_old, best_short[2])
            a_up += 0.02 * rng.standard_normal(target_n)
            a_up = np.clip(a_up, 0.001, None)
            a_up = a_up / np.sum(a_up)
            t_alloc = min(80.0, max(15.0, deadline - time.monotonic() - 60))
            print(f"  Upsample to n={target_n}, t={t_alloc:.0f}s")
            a_opt, s_opt = adam_optimize(a_up, t_alloc, rng)
            print(f"    -> score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} (n={target_n}) ***")

    # Phase 4: LP complement step
    if best_seq is not None and time.monotonic() < deadline - 50:
        print("\n=== Phase 4: LP complement ===")
        n = len(best_seq)
        if n <= 512:  # LP is expensive for large n
            a = best_seq / np.sum(best_seq)
            c = fft_convolve(a, a)
            M = np.max(c)
            print(f"  n={n}, M={M:.8f}")
            g = lp_complement_step(a, M * 1.001)
            if g is not None:
                g = np.clip(g, 0, None)
                sg = np.sum(g)
                if sg > 1e-10:
                    g = g / sg
                    for t_mix in [0.05, 0.1, 0.2, 0.3, 0.5]:
                        mixed = (1 - t_mix) * a + t_mix * g
                        mixed = np.clip(mixed, 0, None)
                        sm = np.sum(mixed)
                        if sm > 1e-10:
                            mixed = mixed / sm
                        s_mix = compute_score(mixed)
                        print(f"    mix={t_mix}: {s_mix:.6f}")
                        if s_mix < best_score:
                            best_score = s_mix
                            best_seq = mixed.copy()
                            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 5: Local search polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        print("\n=== Phase 5: Local search polish ===")
        polish_time = min(180.0, max(15.0, deadline - time.monotonic() - 15))
        print(f"  Polishing n={len(best_seq)}, score={best_score:.6f}, t={polish_time:.0f}s")

        n_restarts = min(3, int(remaining // 50))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 25:

            a_new = best_seq.copy() * np.exp(rng.normal(0, 0.15, size=n_best))
            a_new = np.clip(a_new, 0.001, None)
            a_new = a_new / np.sum(a_new)
            t_alloc = min(40.0, (deadline - time.monotonic() - 20) / (n_restarts - r))
            t_alloc = max(10.0, t_alloc)
            a_opt, s_opt = adam_optimize(a_new, t_alloc, rng)
            print(f"  Restart {r + 1}: score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()

                    if s2 < best_score:
                        best_score = s2
                        best_seq = p2.copy()
                        print(f"  After quick polish: {best_score:.6f}")

    # Final: verify with evaluate_sequence

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: evaluate_sequence score={final_score:.6f}, n={len(final_list)} ===")
        if final_score < best_score:
            best_score = final_score
        return final_list

    # Fallback
    if start_seq is
