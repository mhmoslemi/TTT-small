import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_val(a):
    """Objective with sum(a) normalized to 1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_weighted_grad(conv, a, t):
    """Compute weighted gradient using softmax on conv peaks."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = t * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-10:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg = a[k - je:k - js + 1][::-1]
        grad[js:je + 1] += w[k] * seg
    return grad


def compute_topk_grad(conv, a, topk=10):
    """Compute gradient using top-k peak weighting (sharper)."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    sorted_idx = np.argsort(conv)[::-1]
    top_positions = sorted_idx[:topk]
    weights = np.zeros(len(conv))
    for pos in top_positions:
        if conv[pos] > 0.5 * max_c:
            weights[pos] = conv[pos] / max_c
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= w_sum
    grad = np.zeros(n)
    for k in range(len(weights)):
        if weights[k] < 1e-10:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg = a[k - je:k - js + 1][::-1]
        grad[js:je + 1] += weights[k] * seg
    return grad


def project_simplex(v):
    """Project vector onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def pair_transfer_search(a0, time_limit, rng, n_iter_per_iter=None):
    """Informed pair transfer with line search."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    frac = 0.0

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.02 * (1.0 - frac) ** 2 + 0.0003
        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Find peak positions
        peak_thresh = max_c * 0.8
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])

        # Pick a peak
        k = int(rng.choice(peak_positions))

        # Find source: position i that contributes most to this peak
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg_a = a[js:je + 1]
        seg_b = a[k - je:k - js + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-15:
            continue
        probs = products / total_prod
        top_i = js + int(rng.choice(je - js + 1, p=probs))

        # Find target: position j that contributes least to peaks
        # Compute "danger" for each position
        danger = np.zeros(n)
        for pk in peak_positions[:min(10, len(peak_positions))]:
            pjs = max(0, pk - n + 1)
            pje = min(n - 1, pk)
            seg_d = a[pk - pje:pk - pjs + 1][::-1]
            danger[pjs:pje + 1] += seg_d
        # Prefer low danger positions
        danger_sorted = np.argsort(danger)
        n_candidates = max(5, n // 20)
        candidates = danger_sorted[:n_candidates]
        target_j = int(candidates[rng.integers(len(candidates))])

        if target_j == top_i:
            continue

        # Line search for delta
        # Try several delta values
        base_delta = a[top_i] * 0.3
        best_delta = 0.0
        best_delta_obj = cur_obj
        for frac_d in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            delta = base_delta * frac_d
            if delta < 1e-12 or delta > a[top_i]:
                continue
            a[top_i] -= delta
            a[target_j] += delta
            new_conv = fast_convolve(a)
            new_obj = 2.0 * n * new_conv.max()
            if new_obj < best_delta_obj:
                best_delta_obj = new_obj
                best_delta = delta
            a[top_i] += delta
            a[target_j] -= delta

        if best_delta > 0:
            a[top_i] -= best_delta
            a[target_j] += best_delta
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            # Random perturbation for exploration
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                delta = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= delta
                a[j] += delta
                new_conv = fast_convolve(a)
                new_obj = 2.0 * n * new_conv.max()
                delta_obj = new_obj - cur_obj
                if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
                    cur_obj = new_obj
                    conv = new_conv
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                else:
                    a[i] += delta
                    a[j] -= delta
    return best_a, best_obj

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%d' % int(sf * 100)] = g

    for freq in [1, 2, 3, 5]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        s /= s.sum()
        inits['sin_%d' % freq] = s

    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Chirp
    chirp = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * x / (2.0 * n)))
    chirp = np.maximum(chirp, 0.0)
    if chirp.sum() > 1e-10:
        chirp /= chirp.sum()
        inits['chirp'] = chirp

    # Random
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['rand_%s' % alpha] = r

    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0).copy()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0.0)
        if pp.sum() > 1e-10:
            pp /= pp.sum()
            inits['parent'] = pp

    return inits

        rng_p = np.random.default_rng(seed + 1000 + idx)
        a_cur, _ = pair_transfer_search(a_cur, time_limit=sub, rng=rng_p)
        record(a_cur)

        a_cur, _ = mga_momentum(a_cur, time_limit=sub, t_init=100.0, lr_init=0.04)
        record(a_cur)

    print("  Deep opt done. Best: %.6f" % best_score)

    # Phase 3: Annealing / pair transfer on best
    if best_sequence is not None and time.monotonic() < deadline - 50:
        print("Phase 3: Intensive pair transfer + annealing...")
        phase3_budget = min(600.0, deadline - time.monotonic() - 50)
        n_rounds = 5
        per_round = phase3_budget / n_rounds

        for i in range(n_rounds):
            if time.monotonic() > deadline - 50:
                break
            a_start = best_sequence.copy()
            a_start /= a_start.sum()
            # Add small perturbation for diversity
            if i > 0:
                pert = rng.normal(0, 0.01, len(a_start))
                a_start = np.maximum(a_start + pert, 0)
                a_start /= a_start.sum()

            rng_a = np.random.default_rng(seed + 5000 + i * 7)
            if i % 2 == 0:
                a_opt, _ = pair_transfer_search(a_start, time_limit=per_round, rng=rng_a)
            else:
                a_opt, _ = mga_sharp(a_start, time_limit=per_round * 0.4, t_init=80.0, lr_init=0.03, topk=25)
                if time.monotonic() < deadline - 50:
                    rem = deadline - time.monotonic() - 50
                    a_opt, _ = pair_transfer_search(a_opt, time_limit=min(per_round * 0.6, rem), rng=rng_a)
            record(a_opt)

        print("  Phase 3 done. Best: %.6f" % best_score)

    # Phase 4: Final polish
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            a_best = best_sequence.copy()
            a_best /= a_best.sum()

            # Alternate between different optimizers
            third = remaining / 3.0
            a_opt, _ = mga_momentum(a_best, time_limit=third, t_init=200.0, lr_init=0.02, momentum=0.8)
            record(a_opt)

            if time.monotonic() < deadline - 10:
                rem2 = deadline - time.monotonic() - 10
                if rem2 > 10:
                    a_opt, _ = mirror_descent_v2(a_best, time_limit=rem2 * 0.5, t_init=300.0, eta_init=0.05)
                    record(a_opt)
                    if time.monotonic() < deadline - 10:
                        rem3 = deadline - time.monotonic() - 10
                        if rem3 > 5:
                            rng_f = np.random.default_rng(seed + 99999)
                            a_opt, _ = pair_transfer_search(a_best, time_limit=rem3, rng=rng_f)
                            record(a_opt)

    # Fallback
    if best_sequence is None:
        fallback = np.ones(256) / 256.0
        best_sequence = fallback
        best_score = safe_evaluate(fallback)
        print("  Using fallback uniform(256)")

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
