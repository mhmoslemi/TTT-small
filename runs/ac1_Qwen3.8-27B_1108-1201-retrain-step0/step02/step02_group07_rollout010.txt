import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def lp_optimize_step(f, M, n):
    """
    Solve LP: maximize sum(g) subject to g >= 0, (f*g)(k) <= M for all k.
    
    f: current sequence (sum=1, length n)
    M: current max of f*f
    Returns: g (the LP solution, not yet rescaled)
    """
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)(k) <= M for k = 0, ..., 2n-2
    # (f*g)(k) = sum_j f(k-j) * g_j for j where 0 <= k-j <= n-1
    # = sum_j g_j * f(k-j)
    
    # A_ub has shape (2n-1, n)
    # A_ub[k, j] = f(k-j) if 0 <= k-j <= n-1, else 0
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g_j >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return np.ones(n) / n  # fallback

def lp_iterative(f, n, max_iters=500, t_init=0.3, t_min=0.01):
    """
    Iterative LP-based optimization.
    f: initial sequence (sum=1)
    """
    f = np.maximum(f, 0).copy()
    f /= f.sum()
    
    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_M = M
    best_obj = 2.0 * n * M
    
    t = t_init
    
    for it in range(max_iters):
        g = lp_optimize_step(f, M, n)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum  # normalize to sum=1
        
        # Update
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        f_new /= f_new.sum()
        
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        obj_new = 2.0 * n * M_new
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_M = M_new
            best_f = f_new.copy()
            f = f_new
            M = M_new
            t = max(t_min, t * 0.95)
        else:
            t = max(t_min, t * 0.7)
            if t < t_min:
                break
    
    return best_f, best_obj

"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal update directions, combined with
line search and coordinate descent for fine-tuning.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def build_lp_matrix(f, n):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def solve_lp(f, M, n):
    c = -np.ones(n)
    A_ub = build_lp_matrix(f, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
    if result.success:
        return result.x
    return None

def line_search(f, g, C_f, C_fg, C_g, t_max=1.0):
    """Find t in [0, t_max] minimizing max_k [(1-t)^2 C_f[k] + 2t(1-t) C_fg[k] + t^2 C_g[k]]"""
    def obj(t):
        val = (1-t)**2 * C_f + 2*t*(1-t) * C_fg + t**2 * C_g
        return val.max()
    
    # Ternary search
    a, b = 0.0, t_max
    for _ in range(50):
        m1 = a + (b - a) / 3
        m2 = b - (b - a) / 3
        if obj(m1) < obj(m2):
            b = m2
        else:
            a = m1
    return (a + b) / 2

def lp_optimize(f, n, time_limit, verbose=False):
    """LP-based iterative optimization."""
    f = np.maximum(f, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f /= s
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_no_improve = 100
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        
        g = solve_lp(f, M, n)
        if g is None:
            if verbose:
                print("  LP failed at iter %d" % it)
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Compute convolution terms for line search
        C_f = fast_convolve(f)
        C_g = fast_convolve(g)
        # C_fg = cross convolution f * g
        fft_size = 1 << (2 * n - 1 - 1).bit_length() if n > 1 else 1
        # Actually let's just use np.convolve for cross or FFT
        # For efficiency, use FFT
        F = np.fft.rfft(f, fft_size)
        G = np.fft.rfft(g, fft_size)
        C_fg = np.fft.irfft(F * G, fft_size)[:2*n-1]
        
        t_opt = line_search(f, g, C_f, C_fg, C_g)
        
        f_new = (1 - t_opt) * f + t_opt * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new /= s_new
        
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        obj_new = 2.0 * n * M_new
        
        it += 1
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            M = M_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        
        if verbose and it % 50 == 0:
            print("    iter %d: M=%.8f, obj=%.8f" % (it, M, obj_new))
    
    return best_f, best_obj

"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal update directions with line search,
combined with coordinate descent and diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]


def build_lp_matrix(f, n):
    """Build sparse A_ub where A_ub[k,j] = f[k-j] if 0<=k-j<n."""
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. A_ub*g <= M*ones, g >= 0."""
    c = -np.ones(n)
    A_ub = build_lp_matrix(f, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True})
    if result.success:
        return np.maximum(result.x, 0.0)
    return None


def line_search_max_conv(C_f, C_fg, C_g, t_max=1.0, n_iter=30):
    """Find t minimizing max_k [(1-t)^2 C_f[k] + 2t(1-t)C_fg[k] + t^2 C_g[k]]."""
    def obj(t):
        val = (1.0 - t) ** 2 * C_f + 2.0 * t * (1.0 - t) * C_fg + t * t * C_g
        return val.max()

    a, b = 0.0, t_max
    for _ in range(n_iter):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if obj(m1) < obj(m2):
            b = m2
        else:
            a = m1
    t_opt = (a + b) / 2.0
    return t_opt


def lp_optimize(f, n, time_limit, verbose=False):
    """LP-based iterative optimization with line search."""
    f = np.maximum(f.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f /= s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        g = solve_lp(f, M, n)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        C_f = fast_convolve(f)
        C_g = fast_convolve(g)
        C_fg = cross_convolve(f, g)

        t_opt = line_search_max_conv(C_f, C_fg, C_g)
        if t_opt < 1e-8:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        f_new = (1.0 - t_opt) * f + t_opt * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new /= s_new

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        obj_new = 2.0 * n * M_new

        it += 1
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            M = M_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and it % 20 == 0:
            print("      LP iter %d: M=%.8f, obj=%.8f, t=%.4f" % (it, M, obj_new, t_opt))

    if verbose:
        print("    LP done: %d iters, best_obj=%.8f" % (it, best_obj))
    return best_f, best_obj


def coordinate_descent(f, n, time_limit, max_rounds=50):
    """Coordinate descent: optimize each a[j] to reduce max convolution."""
    f = np.maximum(f.astype(np.float64), 0.0).copy()
    f /= f.sum()
    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    round_num = 0

    while time.monotonic() - t0 < time_limit and round_num < max_rounds:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            conv = fast_convolve(f)
            M_cur = conv.max()

            # Compute weights for coordinate j
            # conv[k] = base[k] + 2*a[j]*a[k-j] (k != 2j) or a[j]^2 (k=2j)
            a_j = f[j]
            # Remove contribution of a[j]
            f[j] = 0.0
            conv_base = fast_convolve(f)

            # Now conv[k] = conv_base[k] + 2*a[j]*a[k-j] for k where 0<=k-j<n and k!=2j
            # and conv[2j] += a[j]^2
            # So conv[k] = conv_base[k] + w[k]*a[j] where:
            #   w[k] = 2*a[k-j] if 0<=k-j<n and k!=2j
            #   w[2j] = a[j] (quadratic term)
            # More precisely: conv[k] = conv_base[k] + 2*a[j]*a[k-j] (if valid, k!=2j) + a[j]^2 (if k=2j)

            # For 1D search: conv[k] as function of x (=a[j]):
            # k=2j: conv_base[2j] + x^2
            # k!=2j, 0<=k-j<n: conv_base[k] + 2*x*a[k-j]
            # otherwise: conv_base[k]

            # We want to minimize max over k
            # This is convex in x, use ternary search on [0, some_upper]
            upper = a_j * 3.0 + f.max()
            if upper < 1e-10:
                upper = 1.0 / n

            # Quick ternary search
            lo, hi = 0.0, upper
            for _ in range(25):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                # Compute max conv for x = m1
                f[j] = m1
                c1 = fast_convolve(f).max()
                f[j] = m2
                c2 = fast_convolve(f).max()

                if c1 < c2:
                    hi = m2
                else:
                    lo = m1

            f[j] = (lo + hi) / 2.0
            f[j] = max(0.0, f[j])

            # Renormalize
            s = f.sum()
            if s < 1e-15:
                f[j] = 1.0 / n
                continue
            f /= f.sum()

            conv_new = fast_convolve(f)
            M_new = conv_new.max()
            obj_new = 2.0 * n * M_new

            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                improved = True

        round_num += 1
        if not improved:
            if round_num > 2:
                break

    return best_f, best_obj

def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    for sf in [0.1, 0.15, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g = np.maximum(g, 0)
        if g.sum() > 1e-10:
            g /= g.sum()
            inits['gauss_%.2f' % sf] = g

    for freq in [1, 2, 3, 5]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        s /= s.sum()
        inits['cos_%d' % freq] = s

    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 8)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random
    for alpha in [0.1, 0.5, 1.0, 2.0, 5.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['rand_a%.1f' % alpha] = r

    # Comb pattern
    comb = np.zeros(n)
    step = max(1, n // 32)
    comb[::step] = 1.0
    comb /= comb.sum()
    inits['comb'] = comb

    #Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0).copy()
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp

    # Perturbed parent
    if parent is not None and parent.size == n:
        pp = np.maximum(parent, 0.0).copy()
        pp /= pp.sum()
        for noise_level in [0.05, 0.1, 0.2]:
            pn = pp + noise_level * rng.standard_normal(n)
            pn = np.maximum(pn, 0)
            if pn.sum() > 1e-10:
                pn /= pn.sum()
                inits['parent_noisy_%.2f' % noise_level] = pn

    return inits

    print("Sequence length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
