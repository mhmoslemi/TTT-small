def fft_convolve_self(a):

    n = len(a)

    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fast_correlate_valid(w, a):
    """Compute v[m] = sum_i w[i+m] * a[i] for m=0,...,len(w)-len(a) using FFT.
