"""
Multi-strategy optimization for minimizing peak autocorrelation of non-negative
sequences. Uses LP-based iteration (SOTA approach), spectral methods, and
coordinate descent polish.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
    """Fast score: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)

def build_conv_matrix(a):
    """Build the convolution constraint matrix C where (C@g)[k] = (a*g)[k]."""
    n = len(a)
    C = np.zeros((2*n-1, n))
    for i in range(n):
        C[i:i+n, i] = a
    return C

def lp_update(a, M, method='highs'):
    """Solve the LP: maximize sum(g) s.t. a*g <= M (elementwise), g >= 0."""
    n = len(a)
    c = -np.ones(n)
    C = build_conv_matrix(a)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method=method)
    if res.success:
        return res.x
    return None

def lp_iteration(a, max_iter=100, time_limit=60.0):
    """Iterative LP-based optimization."""
    a = np.clip(a, 0, None).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    current_m = np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_m = current_m
    alpha = 0.5
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        g = lp_update(a, current_m)
        if g is None:
            break
        
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Try different blend alphas
        improved = False
        for trial_alpha in [alpha, alpha*0.5, alpha*0.25]:
            a_new = (1 - trial_alpha) * a + trial_alpha * g
            a_new = np.clip(a_new, 0, None)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            m_new = np.max(fft_convolve(a_new, a_new))
            if m_new <= current_m:
                a = a_new
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                current_m = m_new
                improved = True
                break
        
        if not improved:
            alpha *= 0.5
            if alpha < 0.01:
                break
    
    return best_a

def spectral_init(n, rng, n_iter=100):
    """Spectral method: alternate between time and frequency domain."""
    a = 1.0 + np.abs(rng.standard_normal(n)) * 0.3
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    
    for it in range(n_iter):
        conv = fft_convolve(a, a)
        peak = np.max(conv)
        # Flatten the correlation
        alpha = 0.5 / (it + 1)
        a = a - alpha * conv / peak
        a = np.clip(a, 0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    
    return a

def gs_init(n, rng, n_iter=200):
    """Gerchberg-Saxton style initialization for flat spectrum."""
    # Target: flat power spectrum
    a = 1.0 + 0.2 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    
    for it in range(n_iter):
        # Frequency domain: flatten |A|^2
        A = np.fft.rfft(a, 2*n)
        power = np.abs(A)**2
        # Target power: constant
        target = np.mean(power)
        # Update phase to reduce peaks
        # Actually, let's just work in time domain
        conv = fft_convolve(a, a)
        # Reduce the peak
        peak_idx = np.argmax(conv)
        # Damp elements that contribute to peak
        for k in range(min(n, 10)):
            i = peak_idx - k
            j = peak_idx - (peak_idx - i)  # = i... this is getting complicated
            # Just damp all elements slightly proportional to their contribution
        # Simpler: just do the iterative peak reduction
        alpha = 1.0 / (n + it)
        a = a - alpha * (conv / np.max(conv) - 1.0/n)
        a = np.clip(a, 0, None)
        s = np.sum(a)

def coordinate_descent(a_init, time_limit, rng, n_steps_per_iter=50):
    """Coordinate descent polish."""
    a = np.clip(a_init, 0, None).astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a), np.inf
    a = a / s
    n = len(a)
    
    current_m = np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_m = current_m
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Try pairs
        for _ in range(n_steps_per_iter):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.2)
            amt = a[i] * frac
            
            candidate = a.copy()
            candidate[i] = a[i] - amt
            candidate[j] = a[j] + amt
            if candidate[i] < 0:
                continue
            s_c = np.sum(candidate)
            candidate = candidate / s_c
            
            m_c = np.max(fft_convolve(candidate, candidate))
            if m_c < current_m:
                a = candidate
                current_m = m_c
                improved = True
                if m_c < best_m:
                    best_m = m_c
                    best_a = a.copy()
        
        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0
    
    return best_a, best_m

def nelder_mead_2d(a_init, time_limit, rng):
    """Local search on pairs of coordinates."""
    a = np.clip(a_init, 0, None).astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    a = a / s
    n = len(a)
    
    current_m = np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_m = current_m
    t0 = time.monotonic()
    step_size = 0.1
    
    while time.monotonic() - t0 < time_limit:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        
        delta = rng.uniform(-step_size, step_size) * min(a[i], a[j])
        candidate = a.copy()
        candidate[i] += delta
        candidate[j] -= delta
        if candidate[i] < 0 or candidate[j] < 0:
            step_size *= 0.7
            continue
        s_c = np.sum(candidate)
        candidate = candidate / s_c
        
        m_c = np.max(fft_convolve(candidate, candidate))
        if m_c < current_m:
            a = candidate
            current_m = m_c
            step_size = min(step_size * 1.2, 0.3)
            if m_c < best_m:
                best_m = m_c
                best_a = a.copy()
        else:
            step_size *= 0.8
    
    return best_a

"""
Multi-strategy optimization for minimizing peak autocorrelation of non-negative
step functions. Uses LP-based iteration (SOTA approach), diverse initializations,
adaptive blend schedules, coordinate descent polish, and random-restart mutation
to escape local minima. Targets sequences of length 256-2048.

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def max_conv(a):
    """Compute max of autocorrelation."""
    conv = fft_convolve(a, a)
    return np.max(conv)


def build_conv_matrix(a):
    """Build convolution matrix C where (C@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp(a, M):
    """Solve LP: maximize sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    C = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8,
                               'primal_feasibility_tolerance': 1e-8})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    best_a = a.copy()
    best_m = current_m
    alpha = 0.5
    stall_count = 0

        if time.monotonic() - t0 > time_limit:
            break

        g = solve_lp(a, current_m)
        if g is None:

        if g_sum < 1e-10:

def spectral_init(n, rng, n_iter=80):
    """Iterative peak-reduction initialization."""
    a = 1.0 + 0.3 * np.abs(rng.standard_normal(n))
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)

    for it in range(n_iter):

        peak = np.max(conv)
        if peak < 1e-12:

        # Subtract correlation scaled to reduce peaks

    inits = []

    # Linear increasing
    a1 = 1.0 + 0.8 * t + 0.03 * rng.standard_normal(n)
    inits.append(np.clip(a1, 0.01, None))

    # Linear decreasing
    a2 = 1.0 + 0.8 * (1.0 - t) + 0.03 * rng.standard_normal(n)
    inits.append(np.clip(a2, 0.01, None))

    # Quadratic
    a3 = 1.0 + 0.6 * t ** 2 + 0.03 * rng.standard_normal(n)
    inits.append(np.clip(a3, 0.01, None))

    # U-shaped (high at ends, low in middle)
    a4 = 1.0 + 0.8 * (2.0 * t - 1.0) ** 2 + 0.03 * rng.standard_normal(n)
    inits.append(np.clip(a4, 0.01, None))

    # Exponential decay
    a5 = np.exp(-0.5 * t) + 0.03 * rng.standard_normal(n)
    inits.append(np.clip(a5, 0.01, None))

    # Exponential growth
    a6 = np.exp(0.5 * t) + 0.03 * rng.standard_normal(n)
    inits.append(np.clip(a6, 0.01, None))

    # Sinusoidal modulation
    freq = rng.uniform(0.5, 2.5)
    a7 = 1.0 + 0.4 * np.sin(2.0 * np.pi * freq * t) + 0.03 * rng.standard_normal(n)
    inits.append(np.clip(a7, 0.01, None))

    for i in range(len(inits)):
        inits[i] = inits[i] / np.sum(inits[i])

    return inits


def coordinate_descent(a_init, time_limit, rng):
    """
    Coordinate descent / local search polish.
    Works on sequences normalized to sum=1.
    """
    a = np.clip(a_init, 0, None).astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    a = a / s
    n = len(a)

    current_m = max_conv(a)
    best_a = a.copy()
    best_m = current_m
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        improved = False
        r = rng.random()

        if r < 0.35:
            # Pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.02, 0.15)
            amt = a[i] * frac
            candidate = a.copy()
            candidate[i] = a[i] - amt
            candidate[j] = a[j] + amt
            if candidate[i] < 0:
                no_improve += 1
                continue
            candidate = candidate / np.sum(candidate)
            m_c = max_conv(candidate)
            if m_c < current_m:
                a = candidate
                current_m = m_c
                improved = True
                no_improve = 0
                if m_c < best_m:
                    best_m = m_c
                    best_a = a.copy()
            else:
                no_improve += 1

        elif r < 0.55:
            # Single element scale
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.08))
            candidate = a.copy()
            candidate[j] *= factor
            if candidate[j] < 0:
                no_improve += 1
                continue
            candidate = candidate / np.sum(candidate)
            m_c = max_conv(candidate)
            if m_c < current_m:
                a = candidate
                current_m = m_c
                improved = True
                no_improve = 0
                if m_c < best_m:
                    best_m = m_c
                    best_a = a.copy()
            else:
                no_improve += 1

        elif r < 0.75:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))
            candidate = np.clip(candidate, 0.001, None)
            candidate = candidate / np.sum(candidate)
            m_c = max_conv(candidate)
            if m_c < current_m:
                a = candidate
                current_m = m_c
                improved = True
                no_improve = 0
                if m_c < best_m:
                    best_m = m_c
                    best_a = a.copy()
            else:
                no_improve += 1

        else:
            # Peak-targeted: find which elements contribute most to the peak
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Compute contributions to peak
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            for i in range(lo, hi + 1):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
                    if i != j:
                        contribs[j] = a[i] * a[j]

            # Reduce top contributors
            top_k = max(3, min(n // 16, 20))
            top_idx = np.argsort(contribs)[-top_k:]
            candidate = a.copy()
            reduction = rng.uniform(0.02, 0.08, size=len(top_idx))
            removed = np.zeros(n)
            for idx, red in zip(top_idx, reduction):
                removed[idx] = candidate[idx] * red
                candidate[idx] -= removed[idx]

            # Redistribute to random other positions
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_add = min(5, len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                total_removed = np.sum(removed)
                candidate[add_idx] += total_removed / n_add

            candidate = np.clip(candidate, 0.001, None)
            candidate = candidate / np.sum(candidate)
            m_c = max_conv(candidate)
            if m_c < current_m:
                a = candidate
                current_m = m_c
                improved = True
                no_improve = 0
                if m_c < best_m:
                    best_m = m_c
                    best_a = a.copy()
            else:
                no_improve += 1

    return best_a, best_m


def mutate_and_reoptimize(a, n_mutate, rng, time_limit_lp):
    """Apply random mutation then re-run LP."""
    a = a.copy()
    n = len(a)

    # Mutation: multiply random subset by random factors
    n_change = max(1, n // n_mutate)
    idx = rng.choice(n, size=n_change, replace=False)
    factors = np.exp(rng.normal(0, 0.5, size=n_change))
    a[idx] *= factors
    a = np.clip(a, 0.001, None)
    a = a / np.sum(a)

    # Re-run LP
    a_new = lp_iteration(a, max_iter=100, time_limit=time_limit_lp)
    return a_new


def propose_candidate(seed=42, **kwargs):
    """
    Multi-strategy search for optimal non-negative sequence.
    Uses LP iteration as primary engine with diverse inits and polish.
    """
    t_global = time.monotonic()
    time_budget = 2370.0
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.001, 1000.0)
        s0 = score_fast(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s0:.6f}")
        if s0 < best_score:
            best_score = s0
            best_seq = start_seq.copy()

    # Search configurations: (n, num_attempts, description)
    configs = [
        (256, 3, "n=256"),
        (384, 2, "n=384"),
        (512, 4, "n=512"),
        (768, 3, "n=768"),
        (1024, 3, "n=1024"),
        (1536, 2, "n=1536"),
        (2048, 2, "n=2048"),
    ]

    total_attempts = sum(c[1] for c in configs)
    time_per_attempt = (time_budget - 300) / total_attempts  # Reserve 300s for polish

    print(f"\n=== Starting search: {len(configs)} configs, {total_attempts} attempts ===")
    print(f"Estimated time per attempt: {time_per_attempt:.1f}s")

    for n, num_attempts, desc in configs:
        if time.monotonic() > deadline - 180:
            break

        for attempt in range(num_attempts):
            if time.monotonic() > deadline - 180:
                break

            t_start = time.monotonic()
            t_alloc = min(time_per_attempt * 1.5, deadline - time.monotonic() - 150)
            if t_alloc < 10:
                break

            # Choose initialization
            if attempt == 0 and start_seq is not None:
                # Resample provided sequence
                x_old = np.linspace(0, 1, len(start_seq))
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, start_seq)
                a0 = np.clip(a0, 0.001, None)
                a0 = a0 / np.sum(a0)
                init_name = "resampled"
            elif attempt == 0:
                a0 = spectral_init(n, rng, n_iter=60)
                init_name = "spectral"
            elif attempt == 1:
                inits = ramp_inits(n, rng)
                a0 = inits[attempt - 1 % len(inits)]
                init_name = f"ramp_{attempt-1}"
            else:
                # Random with structure
                style = attempt % 4
                if style == 0:
                    a0 = 1.0 + 0.5 * rng.random(n)
                elif style == 1:
                    t = np.arange(n) / n
                    a0 = 1.0 + 0.7 * np.cos(np.pi * t)
                elif style == 2:
                    # Two-bump
                    centers = rng.uniform(0, 1, 2)
                    widths = rng.uniform(0.05, 0.2, 2)
                    a0 = np.ones(n)
                    for c, w in zip(centers, widths):
                        a0 += 0.5 * np.exp(-0.5 * ((np.arange(n) / n - c) / w) ** 2)
                else:
                    a0 = 1.0 + 0.3 * rng.standard_normal(n)
                a0 = np.clip(a0, 0.01, None)
                a0 = a0 / np.sum(a0)
                init_name = f"rand_{style}"

            # LP iteration
            lp_time = min(t_alloc * 0.65, 180.0)
            a_lp = lp_iteration(a0, max_iter=300, time_limit=lp_time)
            score_lp = score_fast(a_lp)
            elapsed = time.monotonic() - t_global
            print(f"  {desc}, attempt={attempt}, init={init_name}, "
                  f"score={score_lp:.6f}, best_so_far={best_score:.6f}, "
                  f"time={elapsed:.0f}s")

            if score_lp < best_score:
                best_score = score_lp
                best_seq = a_lp.copy()
                print(f"    *** New best: {best_score:.6f} ***")

            # Coordinate descent polish
            if time.monotonic() < deadline - 120:
                cd_time = min(t_alloc * 0.3, 60.0)
                a_cd, m_cd = coordinate_descent(a_lp, cd_time, rng)
                score_cd = 2.0 * n * m_cd
                if score_cd < best_score:
                    best_score = score_cd
                    best_seq = a_cd.copy()
                    print(f"    *** New best (CD): {best_score:.6f} ***")

            # Mutation + re-optimization (if time allows)
            if time.monotonic() < deadline - 120:
                rem = deadline - time.monotonic() - 120
                if rem > 30:
                    a_mut = mutate_and_reoptimize(a_lp, 4, rng, min(rem * 0.5, 60.0))
                    score_mut = score_fast(a_mut)
                    if score_mut < best_score:
                        best_score = score_mut
                        best_seq = a_mut.copy()
                        print(f"    *** New best (mut): {best_score:.6f} ***")

    # === Phase 2: Polish the best sequence found ===
    print(f"\n=== Phase 2: Final polish ===")
    print(f"Best before polish: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        a_norm = best_seq / np.sum(best_seq)
        polish_time = min(300.0, deadline - time.monotonic() - 20)

        # Extended LP iteration
        lp_time = polish_time * 0.4
        a_lp2 = lp_iteration(a_norm, max_iter=200, time_limit=lp_time)
        sc_lp2 = score_fast(a_lp2)

        # Coordinate descent on LP result
        cd_time = polish_time * 0.35
        a_cd2, m_cd2 = coordinate_descent(a_lp2, cd_time, rng)
        sc_cd2 = 2.0 * n_best * m_cd2

        # More mutations
        if time.monotonic() < deadline - 30:
            rem = deadline - time.monotonic() - 20
            for mut_i in range(3):
                if time.monotonic() > deadline - 30:
                    break
                a_m = mutate_and_reoptimize(a_cd2, 8, rng, min(rem / 4, 40.0))
                sc_m = score_fast(a_m)
                if sc_m < min(sc_cd2, sc_lp2):
                    a_cd2 = a_m
                    sc_cd2 = sc_m

        final_candidates = [(sc_lp2, a_lp2), (sc_cd2, a_cd2)]
        final_score = min(sc_lp2, sc_cd2)
        final_seq = a_lp2 if sc_lp2 < sc_cd2 else a_cd2
        print(f"  After polish: {final_score:.6f}")

        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"    *** Final best: {best_score:.6f} ***")

    # === Final verification ===
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: verified score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
