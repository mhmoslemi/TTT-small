    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e10
    conv = fft_convolve(a)
    # Soft max
    max_conv = np.max(conv)
    shifted = beta * (conv - max_conv)
    log_sum = max_conv + np.log(np.sum(np.exp(shifted))) / beta
    return 2 * n * log_sum / (S * S)

def score_exact(a):
    """Exact score."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e10
    conv = fft_convolve(a)
    return 2 * n * np.max(conv) / (S * S)
