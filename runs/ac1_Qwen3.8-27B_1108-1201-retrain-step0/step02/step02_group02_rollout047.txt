Hybrid optimization for minimizing autocorrelation peak of non-negative sequences.
Combines LP-based local improvement, soft-max gradient descent, and peak-directed
simulated annealing. Focuses on exploiting the LP structure from the problem statement
for exact local improvements, combined with stochastic search for global exploration.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Fast computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def lp_improve(f, M, max_iter=100):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and (f*g)[k] <= M for all k.
    Returns (g, sum_g) where g is the optimal solution.
    """
    n = len(f)
    conv_len = 2 * n - 1
    
    # c[k] = sum_j f[k-j] * g[j] (with f out of range = 0)
    # This is a matrix-vector product: c = A @ g
    # where A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # Build constraint matrix A (conv_len x n)
    A = np.zeros((conv_len, n))
    for k in range(conv_len):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    # LP: maximize c^T g subject to A @ g <= M * ones, g >= 0
    # linprog minimizes, so: minimize -sum(g) s.t. A @ g <= M, g >= 0
    c_obj = -np.ones(n)
    A_ub = A  # A @ g <= M
    b_ub = M * np.ones(conv_len)
    
    # Bounds: g >= 0 (no upper bound)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        sum_g = np.sum(g)
        return g, sum_g
    return None, 1.0

"""
Hybrid LP + gradient optimization for minimizing autocorrelation peak of non-negative sequences.
Key innovation: uses linear programming to find the maximal "growth direction" that maintains
or reduces the convolution peak, combined with soft-max gradient descent and peak-directed
simulated annealing for global exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Fast computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def build_conv_matrix(f):
    """Build the convolution matrix A where c = A @ g gives convolution f*g."""
    n = len(f)
    conv_len = 2 * n - 1
    A = np.zeros((conv_len, n), dtype=np.float64)
    for k in range(conv_len):
        # c[k] = sum_j f[k-j] * g[j] where 0 <= k-j < n
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = f[k - j]
    return A


def lp_improve(f, M, solver='highs'):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and (f*g)[k] <= M for all k.
    Returns (g, sum_g) or (None, 1.0) on failure.
    """
    n = len(f)
    conv_len = 2 * n - 1

    A = build_conv_matrix(f)

    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    A_ub = A
    b_ub = np.full(conv_len, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method=solver, options={'time_limit': 10})
        if result.success:
            g = result.x
            sum_g = np.sum(g)
            return g, sum_g
    except Exception:
        pass
    return None, 1.0


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def soft_max_value_grad(c, tau):
    """
    Compute soft-max approximation of max(c) and its gradient weights.
    Returns (value, weights) where weights sum to 1.
    """
    c_shifted = (c - np.max(c)) / tau
    exp_c = np.exp(c_shifted)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    value = np.max(c) + tau * np.log(sum_exp)
    return value, w


def compute_subgradient(a, k_star):
    """
    Compute subgradient of c[k_star] with respect to a.
    c[k] = sum_j a[j] * a[k-j]
    dc[k_star]/da[j] = a[k_star - j] + a[j - k_star] (with bounds)
    """
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= k_star - j < n:
            val += a[k_star - j]
        if 0 <= j - k_star < n:
            val += a[j - k_star]
        g[j] = val
    return 2.0 * g


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def gradient_step(a, tau, lr, rng):
    """
    One gradient step in the log-parameterization space.
    a is normalized (sum=1), theta = log(a).
    Returns updated a.
    """
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_value_grad(c, tau)

    # Gradient of soft_max(c) with respect to a
    # d(soft_max)/da[j] = 2 * sum_k w[k] * (a[k-j] + a[j-k]) (with bounds)

    # Let's use: g[j] = 2 * sum_k w[k] * (a[k-j] * I[0<=k-j<n] + a[j-k] * I[0<=j-k<n])

    # T_w[j,m] = w[j-m] + w[m-j] (with bounds)
    # So g[j] = 2 * sum_m (w[j-m] + w[m-j]) * a[m]
    #          = 2 * (sum_m w[j-m] * a[m] + sum_m w[m-j] * a[m])
    # First sum: cross-correlation of w and a at lag j
    # Second sum: cross-correlation of a and w at lag j (same thing by symmetry if w is symmetric)
    
    # Using FFT: cross_correlation(w, a)[j] = sum_m w[m+j] * a[m] = irfft(conj(fft(w)) * fft(a))
    # But we need linear, not circular. Let's zero-pad.
    size = 1
    while size < 2 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)

    # = irfft(conj(W) * A)[j] (with appropriate sign convention)

    # In FFT domain: corr = irfft(conj(X) * Y) where X=fft(x), Y=fft(y)
    # So corr(w, a) = irfft(conj(W) * A)
    
    corr_wa = irfft(np.conj(W) * A, size)[:2*n-1]
    corr_aw = irfft(np.conj(A) * W, size)[:2*n-1]

    # We want: sum_k w[k] * a[k-j] = sum_m a[m] * w[m+j] = corr_wa[j] (if w is indexed from 0)
    # Hmm, this is getting confusing. Let me just use the direct O(n^2) approach for gradient
    # Since n=256, O(n^2) = 65536 operations which is fine
    
    g = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k in range(2*n-1):
            wk = w[k]

                continue
            # a[k-j] valid if 0 <= k-j < n
            if 0 <= k - j < n:
                s += wk * a[k - j]
            # a[j-k] valid if 0 <= j-k < n, i.e., j-n+1 <= k <= j
            if 0 <= j - k < n:
                s += wk * a[j - k]

    inner = np.dot(g, a)
    g_theta = a * (g - inner)
    
    # Gradient step in theta space
    theta = np.log(a + 1e-15)
    theta = theta - lr * g_theta
    a_new = softmax_normalize(theta)

def gradient_descent_direct(a, tau, lr, n_steps, rng):
    """
    Direct gradient descent on a (with projection to simplex).
    More efficient than log-parameterization for this problem.
    """
    current = a.copy()
    best = a.copy()
    current_score = compute_score(current)
    best_score = current_score
    
    for step in range(n_steps):
        c = fft_convolve(current, current)
        value, w = soft_max_value_grad(c, tau)
        
        # Compute gradient
        n = len(current)
        g = np.zeros(n)
        for j in range(n):
            s = 0.0
            for k in range(2 * n - 1):
                wk = w[k]
                if wk < 1e-12:
                    continue
                if 0 <= k - j < n:
                    s += wk * current[k - j]
                if 0 <= j - k < n:
                    s += wk * current[j - k]
            g[j] = 2.0 * s
        
        # Projected gradient step: move in negative gradient direction, project to simplex
        a_trial = current - lr * g
        a_trial = project_simplex(a_trial)
        
        new_score = compute_score(a_trial)
        if new_score < current_score:
            current = a_trial
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
    
    return best, best_score


def peak_directed_perturbation(a, rng, strength=0.05):
    """
    Perturb the sequence by reducing peak contributors and increasing others.
    """
    n = len(a)
    c = fft_convolve(a, a)
    peak_idx = int(np.argmax(c))
    
    a_new = a.copy()
    
    # Compute contribution of each element to the peak
    contribs = np.zeros(n)
    for j in range(n):
        if 0 <= peak_idx - j < n:
            contribs[j] = a[j] * a[peak_idx - j]
    
    # Strategy: reduce top contributors, boost random others
    top_k = max(3, n // 16)
    bottom_k = max(3, n // 8)
    
    top_idx = np.argsort(contribs)[::-1][:top_k]
    # Pick from elements NOT in top
    other_indices = np.setdiff1d(np.arange(n), top_idx)
    bottom_idx = rng.choice(other_indices, size=min(bottom_k, len(other_indices)), replace=False)
    
    total_top = a_new[top_idx].sum()
    reduction = strength * total_top
    
    a_new[top_idx] -= reduction / top_k
    a_new[bottom_idx] += reduction / len(bottom_idx)
    
    a_new = np.clip(a_new, 0, None)
    a_new = a_new / np.sum(a_new)
    
    return a_new

            shift = int(rng.integers(1, n // 4))
            blend = rng.uniform(0.05, 0.2)
            a_new = (1 - blend) * current + blend * np.roll(current, shift)
            a_new = np.clip(a_new, 0, None)
            
        else:
            # Global perturbation
            a_new = current * np.exp(rng.normal(0, 0.02, n))
        
        a_new = np.clip(a_new, 0, None)
        if np.sum(a_new) < 1e-10:
            continue
        a_new = a_new / np.sum(a_new)
        
        new_score = compute_score(a_new)
        
        # Acceptance criterion
        if new_score < current_score:
            current = a_new
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-10)):
            current = a_new
            current_score = new_score
        
        temp *= cooling
        
        if step % 2000 == 0:
            print(f"  SA step={step}, cur={current_score:.8f}, best={best_score:.8f}, temp={temp:.5f}")
    
    return best, best_score


def make_initialization(n, rng, method='broadband'):
    """Create various initializations."""
    if method == 'broadband':
        t = np.arange(n, dtype=np.float64)
        a = np.ones(n)
        for k in range(1, 20):
            freq = k * (0.5 + rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.2 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.001, None)
    
    elif method == 'raised_cosine':
        t = np.arange(n, dtype=np.float64)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t / n)
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.001, None)
    
    elif method == 'two_peak':
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((t - n * 0.3) / (n * 0.1)) ** 2) + \
            np.exp(-0.5 * ((t - n * 0.7) / (n * 0.1)) ** 2)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.001, None)
    
    elif method == 'fourier_random':
        # Random Fourier features (guarantees some structure)
        a = np.ones(n)
        n_freqs = max(5, n // 10)
        freqs = rng.integers(1, n, size=n_freqs)
        for f in freqs:
            amp = rng.uniform(0.1, 0.3)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.001, None)
    
    else:
        return 1.0 + 0.2 * rng.standard_normal(n)


def coordinate_polish(a_init, time_limit, rng):
    """
    Coordinate descent polish: optimize one element at a time.
    """
    current = a_init / np.sum(a_init)
    best = current.copy()
    current_score = compute_score(current)
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    # Precompute which k values are affected by changing a[j]
    # Changing a[j] affects c[k] for k in [j, j+n-1] (since c[k] includes a[j]*a[k-j])
    # and also through the "other side": c[k] includes a[k-j]*a[j] when k-j is in range
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            if current[j] < 1e-8:
                continue
            
            # Try scaling a[j] by different factors
            base = current[j]
            best_j_val = base
            best_j_score = current_score
            
            # Geometric search around current value
            for frac in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                trial = base * frac
                if trial < 0:
                    continue
                old_val = current[j]
                current[j] = trial
                # Renormalize
                current = current / np.sum(current)
                
                s = compute_score(current)
                if s < best_j_score:
                    best_j_score = s
                    best_j_val = trial
                
                current[j] = old_val
            
            if best_j_val != base:
                current[j] = best_j_val
                current = current / np.sum(current)
                current_score = best_j_score
                improved = True
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
        
        if not improved:
            # Try a random perturbation to escape
            j = int(rng.integers(0, n))
            current[j] = max(0, current[j] * np.exp(rng.normal(0, 0.2)))
            current = current / np.sum(current)
            current_score = compute_score(current)
        
        iteration += 1
    
    return best, best_score


def multi_length_exploration(best_seq, best_score, time_limit, rng):
    """Try different sequence lengths."""
    results = []
    t0 = time.monotonic()
    
    for n_try in [192, 256, 320, 384, 448]:
        if time.monotonic() - t0 > time_limit - 10:
            break
        
        t_remaining = time_limit - (time.monotonic() - t0)
        t_alloc = min(30.0, t_remaining * 0.4)
        if t_alloc < 5:
            break
        
        # Interpolate from best sequence
        if best_seq is not None and len(best_seq) > 1:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a0 = np.interp(x_new, x_old, best_seq)
        else:
            a0 = make_initialization(n_try, rng, 'broadband')
        
        a0 = np.clip(a0, 0.001, None)
        a0 = a0 / np.sum(a0)
        
        # Quick LP optimization
        a_res, s_res = lp_optimization_phase(a0, t_alloc, rng, verbose=False)
        
        actual = compute_score(a_res)
        print(f"  n={n_try}: score={actual:.8f}")
        
        if actual < best_score:
            best_score = actual
            best_seq = a_res.copy()
            print(f"  *** New best: {best_score:.8f} at n={n_try} ***")
    
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search combining LP optimization, gradient descent,
    simulated annealing, and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Load provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start_seq) > 0 and np.sum(start_seq) > 0.01:
            s = compute_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.8f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()
    
    n_primary = len(best_seq) if best_seq is not None else 256
    
    # === PHASE 1: LP-based optimization on primary length ===
    print(f"\n{'='*60}")
    print(f"PHASE 1: LP optimization (n={n_primary}), budget=55%")
    print(f"{'='*60}")
    
    t_phase1 = 0.55 * time_budget
    a0 = best_seq / np.sum(best_seq) if best_seq is not None else make_initialization(n_primary, rng, 'broadband')
    a0 = np.clip(a0, 0.001, None)
    a0 = a0 / np.sum(a0)
    
    a_best, s_best = lp_optimization_phase(a0, t_phase1, rng, verbose=True)
    
    actual = compute_score(a_best)
    if actual < best_score:
        best_score = actual
        best_seq = a_best.copy()
        print(f"  Phase 1 result: {best_score:.8f}")
    else:
        print(f"  Phase 1: no improvement (still {best_score:.8f})")
    
    # === PHASE 2: Simulated annealing polish ===
    print(f"\n{'='*60}")
    print(f"PHASE 2: Simulated annealing, budget=20%")
    print(f"{'='*60}")
    
    t_phase2 = min(0.20 * time_budget, max(10.0, deadline - time.monotonic() - 60))
    if t_phase2 > 10 and best_seq is not None:
        a_sa, s_sa = simulated_annealing_phase(best_seq, t_phase2, rng)
        actual_sa = compute_score(a_sa)
        if actual_sa < best_score:
            best_score = actual_sa
            best_seq = a_sa.copy()
            print(f"  SA result: {best_score:.8f}")
        else:
            print(f"  SA: no improvement (still {best_score:.8f})")
    
    # === PHASE 3: Coordinate descent polish ===
    print(f"\n{'='*60}")
    print(f"PHASE 3: Coordinate polish, budget=10%")
    print(f"{'='*60}")
    
    t_phase3 = min(0.10 * time_budget, max(10.0, deadline - time.monotonic() - 30))
    if t_phase3 > 10 and best_seq is not None:
        a_cd, s_cd = coordinate_polish(best_seq, t_phase3, rng)
        actual_cd = compute_score(a_cd)
        if actual_cd < best_score:
            best_score = actual_cd
            best_seq = a_cd.copy()
            print(f"  CD result: {best_score:.8f}")
        else:
            print(f"  CD: no improvement (still {best_score:.8f})")
    
    # === PHASE 4: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"PHASE 4: Multi-length exploration")
    print(f"{'='*60}")
    
    t_phase4 = max(10.0, deadline - time.monotonic() - 20)
    if t_phase4 > 15:
        best_seq, best_score = multi_length_exploration(best_seq, best_score, t_phase4, rng)
    
    # === PHASE 5: Final LP polish on best ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        t_final = deadline - time.monotonic() - 10
        if t_final > 15:
            print(f"\n{'='*60}")
            print(f"PHASE 5: Final LP polish (t={t_final:.0f}s)")
            print(f"{'='*60}")
            a_final, s_final = lp_optimization_phase(
                best_seq / np.sum(best_seq), t_final, rng, verbose=True)
            actual_final = compute_score(a_final)
            if actual_final < best_score:
                best_score = actual_final
                best_seq = a_final.copy()
                print(f"  Final: {best_score:.8f}")
    
    # === Final verification ===
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"=== FINAL: evaluate_sequence score={final_eval:.8f}, n={len(final_list)} ===")
        print(f"{'='*60}")
        
        # Use the locally-computed score for comparison (might differ slightly)
        local_s = compute_score(np.array(final_list))
        print(f"    Local score: {local_s:.8f}")
        
        return final_list
    
    # Fallback
    if best_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return [1.0] * 256
