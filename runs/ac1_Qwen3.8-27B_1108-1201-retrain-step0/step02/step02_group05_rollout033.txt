Hybrid optimization for minimum peak autocorrelation non-negative sequences.
Combines soft-max gradient descent (L-BFGS), LP-based updates, 
multiplicative weight updates, and coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Score for sum-normalized sequence: 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def soft_max_obj_and_grad(theta, tau):
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score = 2.0 * n * soft_max_val
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta

def pnorm_obj_and_grad(theta, p):
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)

    """
    LP-based update: find g >= 0 maximizing sum(g) subject to conv(a,g) <= M.
    Returns the normalized g.
    """
    n = len(a)
    M = np.max(fft_convolve(a, a))
    
    # Build constraint matrix A_ub such that A_ub @ g <= b_ub
    # conv(a, g)[k] = sum_i a_i * g_{k-i} for valid i, k-i
    # For lag k (0 to 2n-2):
    #   conv[a,g][k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * g[k-i]
    
    # We need conv(a,g)[k] <= M for all k
    # This gives 2n-1 constraints
    
    # A_ub has shape (2n-1, n), b_ub = M * ones(2n-1)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            if 0 <= j < n:
                A_ub[k, j] = a[i]
    
    b_ub = np.full(2*n-1, M)
    
    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        g = res.x
        if np.sum(g) > 1e-10:
            g = g / np.sum(g)
            return g
    
    return a.copy()

    Multiplicative weight update to reduce peak autocorrelation.

"""
Hybrid optimization for minimum peak autocorrelation non-negative sequences.
Combines LP-based direction finding, soft-max L-BFGS refinement with fine
continuation schedule, projected subgradient method, and targeted local search.
Exploits the scale invariance of the objective and uses FFT-based convolution.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score for sum-normalized sequence: 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def soft_max_obj_and_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score = 2.0 * n * soft_max_val
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def pnorm_obj_and_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p.
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    cp = c ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def lp_direction(a, M_scale=1.0):
    """
    Find direction g via LP: maximize sum(g) s.t. g>=0, conv(a,g) <= M_scale * max(conv(a,a)).
    Returns normalized g, or None if LP fails.
    """
    n = len(a)
    M = M_scale * np.max(fft_convolve(a, a))

    # Build constraint matrix: conv(a,g)[k] <= M for all k
    # conv(a,g)[k] = sum_{j: 0<=k-j<n} a[k-j] * g[j]
    # So A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A_ub[k, lo:hi + 1] = a[0:hi - lo + 1][::-1]

    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        s = np.sum(g)
        if s > 1e-10:
            return g / s
    return None


def projected_subgradient(a_init, tau_schedule, steps_per_tau, deadline):
    """
    Projected subgradient method on the simplex with soft-max smoothing.
    Works directly on a (not through softmax parameterization).
    """
    current = a_init.astype(np.float64).copy()
    current = np.maximum(current, 1e-15)
    current = current / np.sum(current)
    best = current.copy()
    best_score = local_score(current)
    n = len(current)

    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break
        for step in range(steps_per_tau):
            if time.monotonic() > deadline:
                break
            # Compute soft-max gradient
            c = fft_convolve(current, current)
            c_max = np.max(c)
            shifted = (c - c_max) / tau
            exp_shifted = np.exp(shifted)
            sum_exp = np.sum(exp_shifted)
            w = exp_shifted / sum_exp

            # Gradient of soft_max w.r.t. a (not theta)
            a_rev = current[::-1]
            cross = fft_convolve(a_rev, w)
            grad_a = 2.0 * cross[n - 1:2 * n - 1]

            # Polyak-like step size
            obj_val = 2.0 * n * (c_max + tau * np.log(sum_exp))
            lower_bound = 2.0 * n / (2 * n - 1)  # theoretical minimum
            gap = obj_val - lower_bound
            grad_norm = np.linalg.norm(grad_a)
            if grad_norm < 1e-15:
                break
            eta = 2.0 * gap / (grad_norm ** 2 + 1e-15)
            eta = min(eta, 0.5)

            # Gradient step
            candidate = current - eta * grad_a

            # Project onto simplex {x >= 0, sum(x) = 1}
            candidate = np.maximum(candidate, 0)
            s = np.sum(candidate)
            if s < 1e-15:
                candidate = np.ones(n) / n
            else:
                candidate = candidate / s

            new_score = local_score(candidate)
            if new_score < best_score:
                best_score = new_score
                best = candidate.copy()
            current = candidate

    return best, best_score


def run_lbgfs(theta_init, tau_schedule, maxiter_per_tau, deadline):
    """Run L-BFGS with annealed soft-max temperature."""
    best_theta = theta_init.copy()
    best_score = local_score(softmax(theta_init))

    for tau in tau_schedule:
        if time.monotonic() > deadline - 5:
            break
        result = minimize(
            soft_max_obj_and_grad, best_theta.copy(),
            method='L-BFGS-B', jac=True, args=(tau,),
            options={'maxiter': maxiter_per_tau, 'ftol': 1e-16, 'gtol': 1e-14,
                     'maxfun': maxiter_per_tau * 10}
        )
        if result.x is not None:
            a_new = softmax(result.x)
            s = local_score(a_new)
            if s < best_score:
                best_score = s
                best_theta = result.x.copy()

    return best_theta, best_score

            pressure = np.zeros(n)
            for k in top_lags:
                if c[k] > c_mean * 1.05:
                    weight = (c[k] - c_mean) / c_max
                    for j in range(max(0, k - n + 1), min(n, k + 1)):
                        ij = k - j
                        if 0 <= ij < n:
                            pressure[j] += weight * current[ij]
            if np.max(pressure) > 1e-15:
                pressure /= np.max(pressure)
                step = 0.02 + 0.03 * rng.random()
                candidate *= np.exp(-step * pressure)
        elif r < 0.6:
            # Reduce top contributors at the single peak
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for j in range(max(0, peak_idx - n + 1), min(n, peak_idx + 1)):
                ij = peak_idx - j
                if 0 <= ij < n:
                    contribs[j] = current[j] * current[ij]
            top_k = max(5, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=len(top_idx)))
            # Compensate by increasing random others
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(15, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.75:
            # Smoothed redistribution: pull down high-contributing, push up low
            contribs = np.zeros(n)
            for k in range(2 * n - 1):
                w = max(0, c[k] - c_mean)
                for j in range(max(0, k - n + 1), min(n, k + 1)):
                    ij = k - j
                    if 0 <= ij < n:
                        contribs[j] += w * current[ij]
            if np.max(contribs) > 1e-15:
                contribs /= np.max(contribs)
                candidate *= np.exp(-0.015 * contribs)
        elif r < 0.9:
            # Block perturbation
            width = max(2, int(rng.integers(2, max(3, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.maximum(candidate, 1e-15)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score

        temperature = max(temperature * 0.9999, 1e-8)

    return best, best_score

    """Generate diverse initializations."""
    if style == 'chirp':
        t = np.arange(n, dtype=np.float64)
        freq = 0.5 + rng.random()

    elif style == 'bell':
        t = np.arange(n, dtype=np.float64)
        center = n * (0.3 + 0.4 * rng.random())

        a = np.exp(-0.5 * ((t - center) / width) ** 2)
    elif style == 'two_peak':
