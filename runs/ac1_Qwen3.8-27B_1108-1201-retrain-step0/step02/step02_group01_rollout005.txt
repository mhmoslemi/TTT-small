Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Subgradient descent with Nesterov momentum on softmax parameterization
2. Adaptive soft-max temperature (annealing)

import numpy as np

from scipy.fft import fft, ifft
import sys

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_convolve(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]

def compute_grad_a(w, a):
    """g[j] = 2 * sum_l a[l] * w[j+l] = 2 * conv(flip(a), w)[j+n-1]"""
    n = len(a)
    a_flip = a[::-1]
    full_conv = fft_convolve(a_flip, w)
    return 2.0 * full_conv[n-1:2*n-1]

def softmax_obj_grad(theta, T):
    """Compute soft-max objective and gradient w.r.t. theta.
    a = softmax(theta), objective = 2n * soft_max_T(conv(a,a))."""
    n = len(theta)
    # Compute a = softmax(theta)
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    a = exp_theta / exp_theta.sum()
    
    c = fft_convolve_self(a)
    c_max = c.max()
    
    # Soft-max
    if T > 1e10:
        # Hard max
        w = np.zeros(len(c))
        max_idx = np.where(c == c_max)[0]
        w[max_idx] = 1.0 / len(max_idx)
        obj = 2.0 * n * c_max
    else:
        scaled = T * c
        shifted = scaled - scaled.max()
        exp_shifted = np.exp(shifted)
        total = exp_shifted.sum()
        obj = (2.0 * n / T) * (scaled.max() + np.log(total))
        w = exp_shifted / total
    
    grad_a = compute_grad_a(w, a)
    # Softmax gradient: grad_theta = a * (grad_a - a^T grad_a)
    dot = np.dot(a, grad_a)
    grad_theta = a * (grad_a - dot)
    
    return obj, grad_theta, a, c, c_max

def run_nesterov_subgrad(theta_init, n, T_start, T_end, time_budget, lr=0.5, momentum=0.9):
    """Nesterov accelerated subgradient descent with temperature annealing."""
    theta = theta_init.copy().astype(np.float64)
    v = np.zeros_like(theta)
    
    start = time.monotonic()
    best_obj = np.inf
    best_a = None
    
    # Initial evaluation
    obj, grad, a, c, c_max = softmax_obj_grad(theta, T_start)
    true_obj = 2.0 * n * c_max
    if true_obj < best_obj:
        best_obj = true_obj
        best_a = a.copy()
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Temperature schedule: exponential
        T = T_start * (T_end / T_start) ** progress
        
        # Nesterov: evaluate at theta - v
        theta_v = theta - momentum * v
        obj, grad, a, c, c_max = softmax_obj_grad(theta_v, T)
        
        true_obj = 2.0 * n * c_max
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a.copy()
        
        # Update velocity
        v = momentum * v + lr * grad
        theta = theta - v
        
        # Periodic re-normalization check
        if true_obj < best_obj * 0.999:
            # Good improvement, reset momentum
            v *= 0.5
    
    return best_a, best_obj, theta

def run_adaptive_subgrad(theta_init, n, time_budget):
    """Subgradient with adaptive temperature and learning rate."""
    theta = theta_init.copy().astype(np.float64)
    m = np.zeros_like(theta)  # First moment
    v = np.zeros_like(theta)  # Second moment
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_obj = np.inf
    best_a = None
    best_theta = theta.copy()

    T = 5.0
    lr_base = 1.0
    stall_count = 0
    best_so_far = np.inf
    stall_threshold = 200
    
    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Temperature schedule
        T = 5.0 * np.exp(5.0 * progress)  # 5 to ~1240

            obj, grad, a, c, c_max = softmax_obj_grad(theta, T)
        else:

            c_max = c.max()

            max_idx = np.where(c >= c_max - 1e-10)[0]

            obj = 2.0 * n * c_max

            best_a = a.copy()
            best_theta = theta.copy()
            stall_count = 0

            T *= 5.0
            lr_base *= 1.5
            stall_count = 0

        m = beta1 * m + (1 - beta1) * grad

def upsample(a, factor):
    """Upsample by inserting (factor-1) zeros between each element."""
    n = len(a)
    new_a = np.zeros(n * factor)
    new_a[::factor] = a
    return new_a

            # Try reducing a[i]

                if s < 0.01:

            s = a.sum()

def solve_lp_direction(a, M, n, top_k=50):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for top-k lags, g >= 0."""

    # Build constraint matrix
    A_mat = np.zeros((len(active_lags), n))

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_mat[idx, j] = a[k - j]
    
    c_obj = -np.ones(n)  # Maximize sum = minimize -sum
    b_vec = np.full(len(active_lags), M)
    
    try:
        result = linprog(c_obj, A_ub=A_mat, b_ub=b_vec, 
                        bounds=[(0, None)] * n,
                        method='highs', options={'time_limit': 30})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:

        pass
    return None, None

If `W` is positive definite, the minimum is achieved at the point where the gradient is zero (interior solution). If not, the minimum is on the boundary.

The KKT conditions are:
- `2W a + lambda * 1 + mu = 0` (where `mu >= 0` is the Lagrange multiplier for `a >= 0`)
- `mu[i] * a[i] = 0` (complementary slackness)
- `sum(a) = 1`, `a >= 0`, `mu >= 0`

For the interior solution (all `a[i] > 0`): `2W a = -lambda * 1`, so `a = -lambda * (2W)^{-1} * 1`.

For this to be valid, we need `a[i] > 0` for all `i`, which means `((2W)^{-1} * 1)[i] > 0` (with a negative lambda to make it positive).

This gives us a way to solve the inner problem! And the outer problem is to find the optimal `w`.

But the outer problem is also hard (it's a maximization over the simplex of dimension 2n-2).

However, we can use a mirror descent / multiplicative weights approach for the outer problem:
- Start with `w = uniform`
- Solve the inner problem to get `a(w)`
- Update: `w[k] <- w[k] * (1 + eta * (c_k(a) - max(c(a))))` (or something similar)
- Renormalize `w`

This is essentially the "multiplicative weights" algorithm for the outer maximization, combined with solving the quadratic program for the inner minimization.
