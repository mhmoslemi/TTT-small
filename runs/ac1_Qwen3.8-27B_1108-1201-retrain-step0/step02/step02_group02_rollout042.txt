```python
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def build_lp_matrix(f, n):
    """Build the (2n-1) x n constraint matrix A where A[k,j] = f[k-j] if valid."""
    # A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    # For efficiency, build as sparse
    rows, cols, vals = [], [], []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_step(f, M, maxiter=10000):
    """Solve the LP and return the optimal g."""
    n = len(f)
    A = build_lp_matrix(f, n)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'maxiter': maxiter, 'dual_feasibility_tolerance': 1e-8})
    
    if res.status == 0:
        return res.x
    return None
```

```python
"""
LP-guided optimization for minimizing the peak-to-mean autocorrelation ratio.
Primary method: Linear programming to find feasible descent directions.
Secondary: Soft-max gradient descent and local coordinate polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve(a, b):
    """Fast linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score locally."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_sparse_conv_matrix(f):
    """Build sparse matrix A (2n-1 x n) where A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        for j in range(jmin, jmax + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M, timeout=3.0):
    """
    Solve LP: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g array or None on failure.
    """
    n = len(f)
    A = build_sparse_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(
            c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
            method='highs',
            options={'time_limit': timeout, 'presolve': True}
        )
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_descent(f0, time_budget, rng, verbose=True):
    """
    LP-based descent: iteratively solve LP to find directions that keep peak bounded.
    """
    f = f0.astype(np.float64).copy()
    n = len(f)
    S = f.sum()
    best_f = f.copy()
    best_s = compute_score(f)
    t0 = time.monotonic()
    
    if verbose:
        print(f"  LP Descent: n={n}, initial score={best_s:.6f}, budget={time_budget:.0f}s")
    
    step_size = 0.1
    it = 0
    no_improve = 0
    patience = 50
    
    while time.monotonic() - t0 < time_budget:
        elapsed = time.monotonic() - t0
        if elapsed >= time_budget:
            break
        
        c = fft_convolve(f, f)
        M = float(np.max(c))
        
        g = solve_lp(f, M)
        if g is None:
            if verbose:
                print(f"  LP failed at iter {it}, breaking")
            break
        
        Sg = g.sum()
        if Sg < 1e-10:
            break
        
        # Rescale g to match current sum
        g_scaled = g * (S / Sg)
        
        # Adaptive step size: try multiple values
        candidates = [step_size, step_size * 0.5, step_size * 1.5]
        best_new_s = np.inf
        best_new_f = None
        best_trial_t = step_size
        
        for trial_t in candidates:
            trial_t = min(max(trial_t, 0.01), 0.8)
            f_trial = (1.0 - trial_t) * f + trial_t * g_scaled
            f_trial = np.clip(f_trial, 0.0, 1000.0)
            s_trial = compute_score(f_trial)
            if s_trial < best_new_s:
                best_new_s = s_trial
                best_new_f = f_trial
                best_trial_t = trial_t

            f = best_new_f
            best_s = best_new_s
            best_f = f.copy()
            no_improve = 0
            # Gradually increase step size when improving
            step_size = min(step_size * 1.05, 0.5)
        else:
            no_improve += 1
            step_size = max(step_size * 0.8, 0.01)
            if no_improve > patience:
                # Try a larger jump

                f = np.clip(f, 0.0, None)
                f = f * (S / f.sum())

def softmax_norm(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / e.sum()


def peak_gradient(a):
    """Compute exact subgradient of max(conv(a,a)) wrt a."""
    n = len(a)
    c = fft_convolve(a, a)
    k_star = int(np.argmax(c))
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= k_star - j < n:
            val += a[k_star - j]
        if 0 <= j - k_star < n:
            val += a[j - k_star]
        g[j] = 2.0 * val
    return float(c[k_star]), g, k_star


def gradient_descent(a0, time_budget, rng):
    """
    Gradient descent on the log-softmax parameterization.
    Minimizes the peak of autocorrelation.
    """
    n = len(a0)
    S_target = np.sqrt(2.0 * n)
    a = a0.copy()
    a = a / a.sum() * S_target
    theta = np.log(a + 1e-10)
    
    best_a = a.copy()
    best_s = compute_score(a)
    t0 = time.monotonic()
    
    # Adam params
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        elapsed_frac = (time.monotonic() - t0) / time_budget
        tau = max(1.0 / n, (1.0 / (2*n)) * (0.1) ** (elapsed_frac))
        
        c = fft_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        
        # Gradient of soft-max peak wrt a
        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g_a = 2.0 * corr[n - 1:2 * n - 1]
        
        # Convert to theta space (Jacobian of softmax)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S_target)
        
        # Adam update
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g_theta
        v_buf = beta2 * v_buf + (1 - beta2) * g_theta ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)
        
        lr = 1.0 / np.sqrt(step + 1)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if step % 100 == 0:
            a_new = softmax_norm(theta) * S_target
            s_new = compute_score(a_new)
            if s_new < best_s:
                best_s = s_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 100
            a = a_new
            
            if no_improve > 500:
                theta = theta + 0.2 * rng.standard_normal(n)
                no_improve = 0
        
        a = softmax_norm(theta) * S_target
    
    # Final check
    s_final = compute_score(a)
    if s_final < best_s:
        best_s = s_final
        best_a = a.copy()
```

"""
LP-guided optimization for minimizing peak-to-mean autocorrelation ratio.
Primary method: Linear programming to find feasible descent directions that keep
the cross-convolution peak bounded, enabling iterative peak reduction.
Secondary: Soft-max gradient descent and coordinate-level local polish.
Key insight: Asymmetric two-level structures avoid center-peak concentration.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve(a, b):
    """Fast linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score locally (matches evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_sparse_conv_matrix(f):
    """Build sparse matrix A (2n-1 x n) where A[k,j] = f[k-j] if 0<=k-j<len(f)."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        count = jmax - jmin + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = range(jmin, jmax + 1)
        vals[idx:idx + count] = f[k - np.arange(jmin, jmax + 1)]
        idx += count
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp(f, M, timeout=2.0):
    """
    Solve LP: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g array or None on failure.
    """
    n = len(f)
    A = build_sparse_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
            method='highs',
            options={'time_limit': timeout, 'presolve': True}
        )
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_descent(f0, time_budget, rng, verbose=True):

    LP-based descent: iteratively solve LP to find feasible directions.
    Each step finds g maximizing sum(g) with (f*g) <= peak(f*f),
    then blends f towards g to reduce the peak while preserving sum.

    n = len(f)
    S = f.sum()
    best_f = f.copy()
    best_s = compute_score(f)

        if g is None:
            consecutive_lp_fail += 1
            if consecutive_lp_fail > 5:
                if verbose:
                    print(f"  LP repeatedly failed at iter {it}, breaking")
                break
            continue
        consecutive_lp_fail = 0

        Sg = g.sum()
        if Sg < 1e-10:

def softmax_norm(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / e.sum()


def gradient_descent(a0, time_budget, rng, verbose=True):
    """
    Soft-max gradient descent in log-parameterization.
    Anneals temperature from broad to sharp (exact peak).
    """
    n = len(a0)
    S_target = np.sqrt(2.0 * n)
    a = a0.copy()
    a = np.clip(a, 1e-10, None)
    theta = np.log(a / a.sum())

    best_a = a.copy()
    best_s = compute_score(a)
    t0 = time.monotonic()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    no_improve = 0

    while time.monotonic() - t0 < time_budget:
        elapsed_frac = min((time.monotonic() - t0) / time_budget, 1.0)
        # Temperature: starts wide, narrows to near-exact peak
        tau = (5.0 / n) * (0.01 / n) ** (elapsed_frac)
        tau = max(tau, 0.5 / n)

        a = softmax_norm(theta) * S_target
        c = fft_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        # Gradient via correlation
        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g_a = 2.0 * corr[n - 1:2 * n - 1]

        # Jacobian: theta-space gradient
        inner = np.dot(a, g_a) / S_target
        g_theta = a * (g_a - inner)

        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g_theta
        v_buf = beta2 * v_buf + (1 - beta2) * g_theta ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)

        lr = 1.5 / np.sqrt(step + 1)
        if elapsed_frac > 0.8:
            lr *= 0.5
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if step % 50 == 0:
            a_check = softmax_norm(theta) * S_target
            s_check = compute_score(a_check)
            if s_check < best_s - 1e-10:
                best_s = s_check
                best_a = a_check.copy()
                no_improve = 0
            else:
                no_improve += 50
            if no_improve > 400:
                theta = theta + 0.15 * rng.standard_normal(n)
                no_improve = 0
            if verbose and step % 500 == 0:
                print(f"  GD step {step}: score={best_s:.6f}, tau={tau:.1e}, t={time.monotonic()-t0:.0f}s")

    a_final = softmax_norm(theta) * S_target
    s_final = compute_score(a_final)
    if s_final < best_s:
        best_s = s_final
        best_a = a_final.copy()

    if verbose:
        print(f"  GD done: score={best_s:.6f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_a, best_s


def coordinate_polish(a0, time_budget, rng, verbose=True):
    """Coordinate descent with geometric line search per coordinate."""
    a = a0.astype(np.float64).copy()
    best_a = a.copy()
    best_s = compute_score(a)
    n = len(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_budget:
        order = rng.permutation(n)
        improved = False

        for j in order:
            if time.monotonic() - t0 > time_budget:
                break
            if a[j] < 1e-8:
                continue

            base = a[j]
            # Geometric search: try reducing first (peak reduction), then increasing
            for factor in [0.6, 0.75, 0.9, 0.95, 1.05, 1.1, 1.25, 1.5]:
                a[j] = base * factor
                s = compute_score(a)
                if s < best_s - 1e-10:
                    best_s = s
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = base
                    break

        if not improved:
            # Random exploration
            n_perturb = max(1, n // 20)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a[idx] *= np.exp(rng.normal(0, 0.1, size=n_perturb))
            a = np.clip(a, 0.0, None)

        it += 1
        if verbose and it % 30 == 0:
            print(f"  CD iter {it}: score={best_s:.6f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_s


def make_two_level(n, ratio, rng):

def make_broadband(n, rng):

def peak_shave(a0, time_budget, rng):
    """Targeted peak reduction: identify top contributors and redistribute mass."""
    a = a0.astype(np.float64).copy()
    best_a = a.copy()
    best_s = compute_score(a)
    n = len(a)
    t0 = time.monotonic()
    temp = 0.005
    cooling = 0.9999

    for step in range(1, 1000000):
        if time.monotonic() - t0 > time_budget:
            break

        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))

        # Compute contribution to peak
        contribs = np.zeros(n)
        for i in range(n):
            if 0 <= peak_idx - i < n:
                contribs[i] = a[i] * a[peak_idx - i]

        r = rng.random()
        if r < 0.4:
            # Reduce top contributors, boost bottom
            k = max(3, n // 16)
            top_idx = np.argsort(contribs)[::-1][:k]
            bottom_idx = np.argsort(contribs)[:k]
            red = rng.uniform(0.02, 0.06)
            a[top_idx] *= (1.0 - red)
            a[bottom_idx] *= (1.0 + red * len(top_idx) / len(bottom_idx))
        elif r < 0.6:
            # Pair swap
            k = max(3, n // 12)
            top_idx = np.argsort(contribs)[::-1][:k]
            j_red = top_idx[int(rng.integers(0, k))]
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red and a[j_red] > 1e-6:
                amt = a[j_red] * rng.uniform(0.02, 0.04)
                a[j_red] -= amt
                a[j_inc] += amt
        elif r < 0.8:
            # Block scale
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, max(1, n - width)))
            a[start:start + width] *= np.exp(rng.normal(0, 0.04))
        else:
            # Global micro-perturbation
            a *= np.exp(rng.normal(0, 0.005, size=n))

        a = np.clip(a, 0.0, None)
        if a.sum() < 0.01:
            a = best_a.copy()
            continue

        s_new = compute_score(a)
        if s_new < best_s:
            best_s = s_new
            best_a = a.copy()
        elif rng.random() < np.exp(-(s_new - best_s) / max(temp, 1e-12)):
            pass  # accept with Metropolis

        temp = max(temp * cooling, 1e-6)

    return best_a, best_s
