def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

Minimization of self-convolution peak using adaptive-temperature gradient descent
with FFT-accelerated gradients, line search, and multi-scale exploration.

import numpy as np
import time

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def compute_score(a):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def softmax_grad(a, tau):
    """Compute (lse_value, gradient, c, weights) using softmax approximation of max."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    lse = c_max + tau * np.log(sum_exp)
    
    # Gradient via FFT
    flip_w = w[::-1]
    conv = fft_convolve(flip_w, a)
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    
    return lse, grad, c, w

    n = len(a)
    
    # Normalize to sum = sqrt(2n)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n)
        S = n
    a = a / S * np.sqrt(2.0 * n)
    
    best_a = a.copy()
    best_score = compute_score(a)
    if verbose:
        print(f"    Init: n={n}, score={best_score:.6f}")
    
    t0 = time.monotonic()
    step = 0
    no_improve = 0
    
    # Adam state
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    
    # Temperature
    tau_init = 10.0 / n
    tau_end = 0.005 / n
    
    eval_count = 0
    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        
        # Temperature schedule: exponential decay
        tau = tau_init * (tau_end / tau_init) ** elapsed
        
        lse, g, c, w = softmax_grad(a, tau)
        
        # Normalize gradient (remove component along sum direction)
        mean_g = np.mean(g)
        g_centered = g - mean_g  # This preserves sum(a) to first order
        
        step += 1
        m_buf = b1 * m_buf + (1 - b1) * g_centered
        v_buf = b2 * v_buf + (1 - b2) * g_centered ** 2
        m_hat = m_buf / (1 - b1 ** step)
        v_hat = v_buf / (1 - b2 ** step)
        
        # Adaptive step size

            lr = base_lr * factor
            trial = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            if S_t < 0.01:

import numpy as np
import time


def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score: 2n * max(a*a) / (sum(a))^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def compute_objective(b):
    """minimize max(b*b) where sum(b)=1, b>=0. Returns the score."""
    n = len(b)
    c = fft_convolve(b, b)
    return 2.0 * n * np.max(c)


def softmax_grad(b, tau):
    """
    Compute (lse_value, gradient, c, weights) for the softmax approximation of max(b*b).
    b has sum 1, length n.
    """
    n = len(b)
    c = fft_convolve(b, b)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    lse = c_max + tau * np.log(sum_exp)

    # Gradient: grad[m] = 2 * sum_k w[k] * b[k-m] * I(0<=k-m<n)
    # = 2 * (flip(w) * b)[(2n-2)-m]
    flip_w = w[::-1]
    conv = fft_convolve(flip_w, b)
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]

    return lse, grad, c, w

def coordinate_polish(b_init, time_limit, rng, verbose=False):
    """
    Coordinate descent polish: for each element, try geometric sequence of values.
    """
    b = b_init.astype(np.float64).copy()
    n = len(b)
    b = b / np.sum(b)

    best_b = b.copy()
    best_score = compute_objective(b)
    t0 = time.monotonic()
    round_num = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if b[j] < 1e-10:
                # Occasionally try activating a zero
                if rng.random() < 0.05:
                    trial = b.copy()
                    trial[j] = 0.01 / n
                    trial = trial / np.sum(trial)
                    s = compute_objective(trial)
                    if s < best_score:
                        best_score = s
                        best_b = trial.copy()
                        b = trial
                        improved = True
                continue

            base = b[j]
            # Try geometric ratios
            for frac in [0.6, 0.8, 0.9, 1.1, 1.25, 1.5]:
                trial = b.copy()
                trial[j] = base * frac
                trial = np.clip(trial, 0, None)
                S_t = np.sum(trial)
                if S_t < 1e-10:
                    continue
                trial = trial / S_t
                s = compute_objective(trial)
                if s < best_score:
                    best_score = s
                    best_b = trial.copy()
                if s <= compute_objective(b):
                    b = trial
                    improved = True
                else:
                    break

        round_num += 1
        if verbose and round_num % 10 == 0:
            print(f"      CD round={round_num}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")

        if not improved and round_num > 5:
            # Try a small perturbation to escape
            b *= np.exp(rng.normal(0, 0.005, n))
            b = np.clip(b, 0, None)
            b = b / np.sum(b)

    return best_b, best_score


def peak_shave_polish(b_init, time_limit, rng, verbose=False):
    """
    Peak-directed polishing: identify top peaks and perform targeted
    mass redistribution from peak contributors to non-peak positions.
    """
    b = b_init.astype(np.float64).copy()
    n = len(b)
    b = b / np.sum(b)

    best_b = b.copy()
    best_score = compute_objective(b)
    t0 = time.monotonic()
    temp = 0.001
    cooling = 0.9999

    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(b, b)

        # Find top peaks
        top_k = max(3, n // 10)
        top_peaks = np.argsort(c)[::-1][:top_k]

        # Strategy: reduce elements that contribute to top peaks
        r = rng.random()

        if r < 0.4:
            # Reduce top peak contributors
            k_star = top_peaks[0]
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            # Compute contributions
            contribs = np.zeros(n)
            contribs[lo:hi + 1] = b[lo:hi + 1] * b[k_star - np.arange(lo, hi + 1)]
            # Reduce top contributors
            n_reduce = max(3, (hi - lo + 1) // 4)
            top_contrib = np.argsort(contribs)[::-1][:n_reduce]
            reduction = rng.uniform(0.02, 0.08)
            b[top_contrib] *= (1.0 - reduction)
            # Redistribute to bottom contributors
            bottom_contrib = np.argsort(contribs)[:n_reduce * 2]
            total_red = n_reduce * b[top_contrib].max() * reduction
            b[bottom_contrib] += total_red / max(1, len(bottom_contrib))

        elif r < 0.7:
            # Pair swap: take from peak contributor, give to random
            k_star = top_peaks[int(rng.integers(0, top_k))]
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            j_red = int(rng.integers(lo, hi + 1))
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red and b[j_red] > 1e-6:
                amt = b[j_red] * rng.uniform(0.02, 0.08)
                b[j_red] -= amt
                b[j_inc] += amt

        elif r < 0.85:
            # Block scaling
            width = max(3, int(rng.integers(3, max(4, n // 5))))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.04))
            b[start:start + width] *= factor

        else:
            # Global small perturbation
            b *= np.exp(rng.normal(0, 0.008, size=n))

        b = np.clip(b, 0, None)
        S_b = np.sum(b)
        if S_b < 1e-10:
            b = np.ones(n) / n
            S_b = 1.0
        b = b / S_b

        new_score = compute_objective(b)

        if new_score < best_score:
            best_score = new_score
            best_b = b.copy()
        elif rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12) * 50):
            pass  # Accept in place (already updated)

        temp *= cooling

        if verbose and step % 2000 == 0:
            print(f"      PS step={step}, best={best_score:.6f}, temp={temp:.5f}")

    return best_b, best_score


def make_init(n, rng, method='broadband'):
    """Create structured initial probability vectors (sum=1)."""
    t = np.arange(n, dtype=np.float64)

    if method == 'broadband':
        b = np.ones(n)
        for k in range(1, 18):
            freq = rng.uniform(0.3, n * 0.4)
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.12 / np.sqrt(k)
            b += amp * np.cos(2 * np.pi * freq * t / n + phase)
        b += 0.03 * rng.standard_normal(n)
        b = np.clip(b, 0.001, None)
        return b / np.sum(b)

    elif method == 'two_bump':
        # High at edges, low in middle
        center = n / 2.0
        b = 1.0 + 0.6 * np.abs(t - center) / center
        b += 0.05 * rng.standard_normal(n)
        b = np.clip(b, 0.001, None)
        return b / np.sum(b)

    elif method == 'sinusoidal':
        k = int(rng.integers(2, max(3, n // 4)))
        b = 1.0 + 0.3 * np.cos(2 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
        b += 0.04 * rng.standard_normal(n)
        b = np.clip(b, 0.001, None)
        return b / np.sum(b)

    elif method == 'periodic_sparse':
        # Periodic pattern with some structure
        period = max(3, int(np.sqrt(n) + 0.5))
        pattern = np.ones(period)
        pattern[0] = 1.4
        pattern[period // 2] = 0.6
        b = np.tile(pattern, n // period + 1)[:n]
        b += 0.04 * rng.standard_normal(n)
        b = np.clip(b, 0.001, None)
        return b / np.sum(b)

    elif method == 'random':
        b = np.abs(rng.standard_normal(n)) + 0.05
        return b / np.sum(b)

    else:
        b = np.ones(n) + 0.2 * rng.standard_normal(n)
        b = np.clip(b, 0.01, None)
        return b / np.sum(b)


def interpolate_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    if len(src) == target_n:
        return src.copy()
    x_old = np.linspace(0, 1, len(src))
    x_new = np.linspace(0, 1, target_n)
    return np.interp(x_new, x_old, src)

        # Scale to nice values (sum = sqrt(2n) gives score = 2n * max(b*b))
        n = len(best_b)
        a_final = best_b * np.sqrt(2.0 * n)
        final_list = [float(max(0.0, min(1000.0, x))) for x in a_final]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*55}")
        print(f"  FINAL: score={final_eval:.6f}, n={len(final_list)}")
        print(f"{'='*55}")
        return final_list

    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
