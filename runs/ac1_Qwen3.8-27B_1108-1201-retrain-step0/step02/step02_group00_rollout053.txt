"""
Minimize 2*n*max_conv / sum^2 using segment parameterization.

Key insight: For a piecewise-constant sequence with k segments, the self-convolution
is a quadratic form in the k segment values. This reduces the optimization to
a k-dimensional problem that can be solved very efficiently.

Algorithm:
1. For various k and boundary configurations, optimize segment values using
   log-sum-exp smoothed objective with Nesterov acceleration or L-BFGS.
2. Expand best segment solution to full sequence.
3. Polish with full-sequence FISTA.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_segment_coefficients(lengths, n):
    """Compute C[j,l,m] for all segment pairs and all convolution lags."""
    k = len(lengths)
    result = np.zeros((k, k, 2 * n - 1), dtype=np.float64)
    starts = np.zeros(k, dtype=np.int64)
    for i in range(1, k):
        starts[i] = starts[i-1] + lengths[i-1]
    
    for j in range(k):
        for l in range(k):
            s_j = starts[j]
            L_j = lengths[j]
            s_l = starts[l]
            L_l = lengths[l]
            # For each lag k (0 to 2n-2):
            # c[j,l,k] = number of i where i in [s_j, s_j+L_j-1] and k-i in [s_l, s_l+L_l-1]
            # i in [s_j, s_j+L_j-1] AND i in [k-s_l-L_l+1, k-s_l]
            # i in [max(s_j, k-s_l-L_l+1), min(s_j+L_j-1, k-s_l)]
            
            # Vectorized computation:
            ks = np.arange(2 * n - 1)
            lo = np.maximum(s_j, ks - s_l - L_l + 1)
            hi = np.minimum(s_j + L_j - 1, ks - s_l)
            c = np.maximum(hi - lo + 1, 0)
            result[j, l, :] = c
    return result


def eval_segment_obj(v, C, lengths, n, beta):
    """Evaluate log-sum-exp smoothed objective and gradient for segment values."""
    k = len(v)
    # Compute full convolution: conv[m] = sum_{j,l} v_j * v_l * C[j,l,m]
    # Efficiently: conv = sum_j v_j * (sum_l v_l * C[j,l,:])
    # = sum_j v_j * (C[:,:,m] @ v)[j]
    
    # conv as matrix product:
    # Let M = v.reshape(1,k) @ (C @ v.reshape(k,1,1) broadcast) ... 
    # Actually: conv[m] = v^T C[:,:,m] v
    # We can compute this as: for each m, v @ C[:,:,m] @ v
    # More efficiently: (v*C) @ C conveyor...
    
    # conv = sum_j v_j * sum_l v_l * C[j,l,m] for each m
    # = (v * C) summed over j, then dotted with v over l... 
    # Let's do: temp = np.einsum('j,jlm->m', v, np.einsum('l,jlm->jlm', v, C))
    # Simpler: temp = np.einsum('l,jlm->jm', v, C) gives [j,m]
    #         conv = np.einsum('j,jm->m', v, temp)
    
    temp = np.einsum('l,jlm->jm', v, C)  # shape (k, 2n-1)
    conv = np.einsum('j,jm->m', v, temp)  # shape (2n-1,)
    
    # Log-sum-exp smoothing
    c_max = conv.max()
    exp_bc = np.exp(beta * (conv - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, np.zeros(k)
    w = exp_bc / total  # weights on each lag
    
    # Gradient: d(conv[m])/dv_j = 2 * sum_l v_l * C[j,l,m]
    # grad[j] = sum_m w[m] * 2 * sum_l v_l * C[j,l,m]
    # temp2 = 2 * C @ v -> [j, m]
    temp2 = 2.0 * np.einsum('l,jlm->jm', v, C)  # wait, this is the same as temp*2
    # Actually grad_j = sum_m w[m] * d(conv[m])/dv_j = sum_m w[m] * 2 * (C[:,:,m] @ v)[j]
    # = 2 * (C[:,:,m] @ v)[j] summed over m with weights w[m]
    # = 2 * sum_m w[m] * temp[j, m]
    grad = 2.0 * np.einsum('m,jm->j', w, temp)
    
    # The smoothed objective value (for tracking):
    obj = c_max + np.log(total) / beta
    
    return obj, grad


def optimize_segments(values_init, C, lengths, n, target_sum, max_time, rng, 
                       beta_init=5.0, beta_max=200.0):
    """Optimize segment values using Nesterov accelerated gradient on log-sum-exp."""
    k = len(values_init)
    v = np.clip(values_init, 0, 1000).astype(np.float64).copy()
    s = v @ np.array(lengths, dtype=np.float64)
    if s < 1e-15:
        v = np.ones(k) * target_sum / sum(lengths)
    else:
        v = v * (target_sum / s)
    
    y = v.copy()
    t = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_v = v.copy()
    step = 1.0 / (beta_init * 4.0)
    it = 0
    
    # Constraint: v >= 0 and v @ lengths = target_sum
    # We'll use a simple projection
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.001 * it))
        
        obj, grad = eval_segment_obj(y, C, lengths, n, beta)
        
        # Check actual (unsmoothed) objective
        temp = np.einsum('l,jlm->jm', y, C)
        conv = np.einsum('j,jm->m', y, temp)
        actual_max = conv.max()
        if actual_max < best_obj:
            best_obj = actual_max
            best_v = y.copy()
        
        # Gradient step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        x_new = y - step * grad
        
        # Project onto {v >= 0, v @ lengths = target_sum}
        # This is projection onto intersection of non-negative orthant and a hyperplane
        # Simple approach: iterative projection
        for _ in range(50):
            # Project onto sum constraint
            current_sum = x_new @ np.array(lengths, dtype=np.float64)
            if abs(current_sum - target_sum) > 1e-10:
                x_new = x_new + (target_sum - current_sum) / (np.array(lengths, dtype=np.float64) @ np.array(lengths, dtype=np.float64)) * np.array(lengths, dtype=np.float64)
            # Project onto non-negativity
            x_new = np.maximum(x_new, 0)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - v)
        y = np.maximum(y, 0)
        v = x_new
        t = t_new
        
        step = 1.0 / (beta * 4.0) * 0.99
        step = max(step, 1e-10)
        
        if it % 500 == 0 and rng.random() < 0.1:
            y = np.maximum(y + rng.normal(0, 0.01 * np.ones(k), k), 0)
            s = y @ np.array(lengths, dtype=np.float64)
            if s > 1e-15:
                y = y * (target_sum / s)
            t = 1.0
    
    return best_v, best_obj


def run_fista_full(a_init, n, S, max_time, rng, beta_init=2.0, beta_max=300.0):
    """FISTA with adaptive log-sum-exp for full sequence polishing."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    step = 1.0 / (beta_init * 2.0)
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.002 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Gradient of log-sum-exp
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        step = 1.0 / (beta * 2.0 * (1.0 + 0.0005 * it))
        step = max(step, 1e-6)

        if it % 2000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

"""
Minimize 2*n*max_conv / sum^2 using segment parameterization + FISTA polishing.

Key insight: For a piecewise-constant sequence with k segments, the self-convolution
is a quadratic form in the k segment values: conv[m] = sum_{j,l} v_j*v_l*C[j,l,m].
This reduces optimization to k dimensions, enabling very fast evaluation and
access to different basins of attraction than full-sequence optimization.

Algorithm:
1. Segment optimization: For various k (3-30) and boundary configurations,
   optimize segment values using Nesterov-accelerated gradient on log-sum-exp
   smoothed objective, with normalization constraint handled by rescaling.
2. Full-sequence FISTA: Polish best segment solution with FFT-accelerated FISTA.
3. Iterate: Alternate between segment and full-sequence optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_segment_coefficients(lengths, n):
    """Compute C[j,l,m] for all segment pairs and all convolution lags.
    
    C[j,l,m] = number of integer i such that i is in segment j and (m-i) is in segment l.
    Shape: (k, k, 2n-1)
    """
    k = len(lengths)
    result = np.zeros((k, k, 2 * n - 1), dtype=np.float64)
    starts = np.zeros(k, dtype=np.int64)
    runs = 0
    for i in range(k):
        starts[i] = runs
        runs += lengths[i]
    
    ks = np.arange(2 * n - 1, dtype=np.int64)
    
    for j in range(k):
        s_j = starts[j]
        L_j = lengths[j]
        for l in range(k):
            s_l = starts[l]
            L_l = lengths[l]
            lo = np.maximum(s_j, ks - s_l - L_l + 1)
            hi = np.minimum(s_j + L_j - 1, ks - s_l)
            c = np.maximum(hi - lo + 1, 0)
            result[j, l, :] = c.astype(np.float64)
    return result


def seg_convolve(v, C):
    """Compute full convolution from segment values: conv = sum_{j,l} v_j*v_l*C[j,l,m]."""
    temp = np.einsum('l,jlm->jm', v, C)
    conv = np.einsum('j,jm->m', v, temp)
    return conv


def seg_objective_smoothed(v, C, beta):
    """Log-sum-exp smoothed max of convolution, plus gradient."""
    temp = np.einsum('l,jlm->jm', v, C)
    conv = np.einsum('j,jm->m', v, temp)
    c_max = conv.max()
    exp_bc = np.exp(beta * (conv - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        val = c_max
        grad = np.zeros(len(v))
    else:
        w = exp_bc / total
        val = c_max + np.log(total) / beta
        grad = 2.0 * np.einsum('m,jm->j', w, temp)
    return val, grad


def optimize_segments_nv(values_init, C, lengths, n, target_sum, max_time, rng,
                         beta_init=5.0, beta_max=300.0):
    """Nesterov-accelerated optimization of segment values with normalization."""
    k = len(values_init)
    lengths_f = np.array(lengths, dtype=np.float64)
    
    v = np.clip(values_init, 0, 1000).astype(np.float64).copy()
    s = v @ lengths_f
    if s < 1e-15:
        v = np.ones(k) * target_sum / n
    else:
        v = v * (target_sum / s)
    
    y = v.copy()
    t = 1.0
    start = time.monotonic()
    best_v = v.copy()
    best_obj = np.inf
    step = 1.0 / (2.0 * beta_init)
    it = 0
    stall_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.005 ** it))
        
        # Evaluate current
        temp = np.einsum('l,jlm->jm', y, C)
        conv = np.einsum('j,jm->m', y, temp)
        c_max = conv.max()
        
        if c_max < best_obj * (1.0 - 1e-8):
            best_obj = c_max
            best_v = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Compute smoothed gradient
        exp_bc = np.exp(beta * (conv - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total
        grad = 2.0 * np.einsum('m,jm->j', w, temp)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Gradient step with normalization
        x_new = y - step * grad
        x_new = np.maximum(x_new, 0.0)
        s_new = x_new @ lengths_f
        if s_new < 1e-15:
            x_new = np.ones(k) * target_sum / n
        else:
            x_new = x_new * (target_sum / s_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - v)
        y = np.maximum(y, 0.0)
        v = x_new
        t = t_new
        
        # Adaptive step size
        step = 1.0 / (2.0 * beta)
        step *= max(0.95, 1.0 - 0.0003 * it)
        step = max(step, 1e-12)
        
        # Restart with perturbation if stalled
        if stall_count > 5000 or (it % 3000 == 0 and rng.random() < 0.2):
            noise_scale = 0.01 * target_sum / np.sqrt(n)
            y = np.maximum(y + rng.normal(0, noise_scale, k), 0)
            s_y = y @ lengths_f
            if s_y > 1e-15:
                y = y * (target_sum / s_y)
            v = y.copy()
            t = 1.0
            stall_count = 0
            step = 1.0 / (2.0 * min(beta_max, beta * 0.5))
    
    return best_v, best_obj


def run_fista_full(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=400.0):
    """FISTA with adaptive log-sum-exp for full sequence. Returns best sequence and score."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    step = 1.0 / (beta_init * 2.0)
    it = 0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.003 ** it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score * (1.0 - 1e-8):
            best_score = cur
            best_local = y.copy()
            stall = 0
        else:
            stall += 1

        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = project_simplex(y - step * grad, S)

        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        step = 1.0 / (beta * 2.0)
        step *= max(0.95, 1.0 - 0.0002 * it)
        step = max(step, 1e-8)

            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            a = y.copy()
            t = 1.0
            stall = 0

    return best_local, best_score

def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def generate_boundaries(k, n, rng, mode):
    """Generate segment boundaries."""
    if mode == 0:
        # Evenly spaced
        boundaries = np.linspace(0, n, k + 1).astype(np.int64)
    elif mode == 1:
        # Random
        boundaries = np.sort(rng.integers(0, n + 1, k + 1))
        boundaries[0] = 0
        boundaries[-1] = n
    elif mode == 2:
        # Perturbed even
        even = np.linspace(0, n, k + 1, dtype=np.int64)
        seg_len = n // k
        perturbation = rng.integers(-seg_len // 5, seg_len // 5 + 1, k + 1)
        boundaries = even + perturbation
        boundaries = np.sort(boundaries)
        boundaries[0] = 0
        boundaries[-1] = n
    elif mode == 3:
        # Geometric
        ratios = rng.uniform(0.5, 2.0, k)
        ratios = ratios / ratios.sum() * n
        boundaries = np.zeros(k + 1, dtype=np.int64)
        for i in range(k):
            boundaries[i + 1] = boundaries[i] + max(1, int(round(ratios[i])))
        boundaries[-1] = n
    else:
        boundaries = np.linspace(0, n, k + 1).astype(np.int64)
    
    # Ensure valid
    for i in range(1, k + 1):
        boundaries[i] = max(boundaries[i], boundaries[i - 1] + 1)
    if boundaries[k] > n:
        boundaries[k] = n
        for i in range(k, 0, -1):
            if boundaries[i] <= boundaries[i - 1]:
                boundaries[i] = boundaries[i - 1] + 1
        if boundaries[k] > n:
            return None
    
    lengths = np.diff(boundaries)
    if np.any(lengths < 1):
        return None
    
    return boundaries

        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
    
    print("Final: length=%d, internal=%.6f" % (len(final_list), global_best))
    return final_list
