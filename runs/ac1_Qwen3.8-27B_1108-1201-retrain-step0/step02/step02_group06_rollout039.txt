"""
Optimized peak-autocorrelation minimization for non-negative step functions.
Uses adaptive smoothed-max gradient descent with multiple initializations,
gradient-informed local search, and multi-scale exploration.
"""

import numpy as np
import time
import math


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score computation."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def smoothed_max_and_grad(a, tau):
    """
    Compute smoothed max of conv(a,a) and gradient w.r.t. a.
    Uses log-sum-exp for smooth approximation of max.
    """
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    value = max_c + tau * np.log(np.sum(exp_vals))
    
    # Gradient: grad[j] = 2 * sum_m w[m] * a[m-j]
    # = 2 * (w cross_correlate a)[j]
    # Computed via FFT: irfft(rfft(w, N) * conj(rfft(a, N)), N)
    N = 2 * n
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    xcorr = np.fft.irfft(fw * np.conj(fa), N)
    grad = 2.0 * xcorr[:n]
    
    return value, grad, conv


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta_val = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta_val, 0.0)


def spectral_init(n, rng, n_iter=50):
    """Spectral iterative initialization."""
    a = 1.0 + 0.5 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    
    for it in range(n_iter):
        conv = fft_convolve(a, a)
        alpha = 1.0 / (n + 1.0)
        a = a - alpha * conv[:n]  # Use first n elements (lags 0 to n-1)
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def random_init(n, rng):
    """Random positive initialization."""
    a = rng.random(n) + 0.1
    return a / np.sum(a)


def ramp_init(n, rng, style='linear'):
    """Ramp-like initialization."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if style == 'linear':
        a = 1.0 + 0.3 * t
    elif style == 'quadratic':
        a = 1.0 + 0.4 * t ** 2
    elif style == 'sinusoidal':
        a = 1.0 + 0.3 * np.sin(np.pi * t)
    elif style == 'sqrt':
        a = 1.0 + 0.5 * np.sqrt(t)
    elif style == 'bump':
        a = 1.0 + 0.5 * np.exp(-((t - 0.5) ** 2) / 0.05)
    else:
        a = 1.0 + 0.3 * t
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def interpolate_sequence(a_old, n_new):
    """Interpolate a sequence to a new length."""
    x_old = np.linspace(0, 1, len(a_old))
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a_old)


def optimize_adam(n, a_init, time_limit, rng, tau_factor=5.0):
    """
    Adam optimization on softmax parameterization.
    Minimizes smoothed max of conv(a,a) where a = softmax(theta) * scale.
    """
    scale = np.sqrt(2.0 * n)
    p_init = a_init / np.sum(a_init)
    theta = np.log(p_init + 1e-10).copy()
    
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 2.0
    
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf
    
    # Adaptive tau: start large, decrease
    tau_max = tau_factor / n
    tau_min = 0.01 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        # Smooth tau schedule: fast decrease initially, slow later
        tau = tau_max * (tau_min / tau_max) ** (frac ** 0.5)
        
        a = softmax_normalize(theta) * scale
        value, g_a, conv = smoothed_max_and_grad(a, tau)
        
        p = a / scale
        inner = np.dot(p, g_a)
        g_theta = scale * p * (g_a - inner)
        
        step += 1
        m_vec = beta1 * m_vec + (1 - beta1) * g_theta
        v_vec = beta2 * v_vec + (1 - beta2) * (g_theta * g_theta)
        m_hat = m_vec / (1.0 - beta1 ** step)
        v_hat = v_vec / (1.0 - beta2 ** step)
        lr_eff = lr / np.sqrt(step + 1)
        theta = theta - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate periodically
        if step % 100 == 0 or step < 10:
            a_check = softmax_normalize(theta) * scale
            sc = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (scale * scale)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()
        
        if step % 2000 == 0:
            a_check = softmax_normalize(theta) * scale
            sc = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (scale * scale)
            print(f"      Adam: step={step}, score={sc:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={elapsed:.1f}s")
    
    return best_theta, best_score


def gradient_based_local_search(a_init, time_limit, rng):
    """
    Gradient-informed local search on the simplex.
    Uses the smoothed max gradient to guide moves.
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a), np.inf
    a = a / s
    n = len(a)
    
    current_max = np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_max = current_max
    t0 = time.monotonic()
    step = 0
    patience = 0
    max_patience = 3000
    
    tau = 5.0 / n
    
    while time.monotonic() - t0 < time_limit and patience < max_patience:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        tau = (5.0 / n) * (0.01 / 5.0) ** (frac ** 0.7)
        
        value, grad, conv = smoothed_max_and_grad(a, tau)
        
        # Gradient descent step with adaptive learning rate
        # The gradient points in the direction that increases the smoothed max
        # We want to decrease it, so we go against the gradient
        # But we need to stay on the simplex (sum=1, a>=0)
        
        # Find the most negative gradient component (most "safe" to decrease)
        # and the most positive (most "dangerous" to increase)
        
        r = rng.random()
        
        if r < 0.5:
            # Gradient-informed: move mass from high-grad to low-grad positions
            # High grad means increasing a[j] would increase the max, so we DECREASE it
            # Low grad means decreasing a[j] would decrease the max, so we INCREASE it
            # Wait, the gradient of the smoothed max w.r.t. a[j] tells us:
            # if grad[j] > 0, increasing a[j] increases the max → we should decrease a[j]
            # if grad[j] < 0, increasing a[j] decreases the max → we should increase a[j]
            
            # Pick source (high grad, decrease) and target (low grad, increase)
            top_k = max(3, n // 16)
            source_idx = np.argsort(grad)[-top_k:]  # highest grad
            target_idx = np.argsort(grad)[:top_k]    # lowest grad
            
            # Only pick from nonzero sources
            source_idx = source_idx[a[source_idx] > 1e-8]
            if len(source_idx) == 0:
                continue
            
            n_source = min(3, len(source_idx))
            n_target = min(3, len(target_idx))
            
            src = source_idx[rng.integers(n_source)]
            tgt = target_idx[rng.integers(n_target)]
            
            if src == tgt:
                continue
            
            # Transfer amount proportional to gradient difference
            frac = rng.uniform(0.02, 0.1)
            amt = a[src] * frac
            candidate = a.copy()
            candidate[src] -= amt
            candidate[tgt] += amt
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate /= cs
            
        elif r < 0.7:
            # Block gradient step
            block_size = min(8, n // 8)
            start = int(rng.integers(0, max(1, n - block_size)))
            block = slice(start, start + block_size)
            
            # Move the block in the negative gradient direction
            lr_local = 0.01 / max(np.max(np.abs(grad[block])), 1e-10)
            candidate = a.copy()
            candidate[block] -= lr_local * grad[block]
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate /= cs
            
        elif r < 0.85:
            # Peak-targeted: reduce elements at the peak lag
            peak_idx = int(np.argmax(conv))
            # The peak is at conv[peak_idx] = sum_i a[i]*a[i+peak_idx]
            # Reduce the elements that contribute most
            contribs = np.zeros(n)
            for i in range(n):
                j = i + peak_idx
                if j < n:
                    contribs[i] = a[i] * a[j]
                    contribs[j] += a[i] * a[j]  # both contribute
            top_k = max(3, n // 32)
            top_idx = np.argsort(contribs)[-top_k:]
            
            candidate = a.copy()
            reduce_frac = rng.uniform(0.01, 0.05)
            candidate[top_idx] *= (1.0 - reduce_frac)
            # Redistribute
            removed = np.sum(a[top_idx] - candidate[top_idx])
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0 and removed > 0:
                n_add = min(5, len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                candidate[add_idx] += removed / n_add
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate /= cs
            
        else:
            # Random perturbation (exploration)
            candidate = a * np.exp(rng.normal(0, 0.02, size=n))
            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 1e-10:
                continue
            candidate /= cs
        
        new_max = np.max(fft_convolve(candidate, candidate))
        new_score = 2.0 * n * new_max
        
        # Accept if better (or with small probability if worse, for exploration)
        if new_max < current_max:
            a = candidate
            current_max = new_max
            patience = 0
        else:
            # Simulated annealing: accept with probability exp(-delta / T)
            delta = new_max - current_max
            T = max(1e-6, (1.0 - patience / max_patience) * 1e-4)
            if rng.random() < np.exp(-delta / T):
                a = candidate
                current_max = new_max
            else:
                patience += 1
        
        if current_max < best_max:
            best_max = current_max
            best_a = a.copy()
        
        step += 1
        if step % 5000 == 0:
            print(f"      Local: step={step}, score={2*n*best_max:.6f}, "
                  f"t={time.monotonic()-t0:.1f}s")
    
    return best_a, 2.0 * n * best_max


def block_coordinate_descent(a_init, time_limit, rng):
    """
    Block coordinate descent: optimize groups of consecutive elements.
    """
    a = a_init / np.sum(a_init)
    a = np.clip(a, 0.0, None)
    a = a / np.sum(a)
    n = len(a)
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    
    # Try different block sizes
    block_sizes = [2, 4, 8, 16, 32]
    
    while time.monotonic() - t0 < time_limit:
        bs = block_sizes[rng.integers(len(block_sizes))]
        # Pick a random starting position
        start = int(rng.integers(0, max(1, n - bs + 1)))
        end = min(start + bs, n)
        block = slice(start, end)
        
        # Try modifying this block
        # Strategy: scale the block, which changes the relative weights
        new_vals = a[block] * np.exp(rng.normal(0, 0.1, size=end - start))
        new_vals = np.clip(new_vals, 0, None)
        
        # Also try: equalize the block
        # Or: shift mass within the block
        
        candidate = a.copy()
        candidate[block] = new_vals
        candidate = np.clip(candidate, 0.0, None)
        cs = np.sum(candidate)
        if cs < 1e-10:
            continue
        candidate /= cs
        
        new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
        if new_score < best_score:
            a = candidate
            best_score = new_score
            best_a = a.copy()
    
    return best_a, best_score


def optimize_with_restarts(n, time_limit, rng, provided_seq=None):
    """
    Run optimization with multiple restarts and keep the best.
    """
    best_a = None
    best_score = np.inf
    t0 = time.monotonic()
    
    # Create diverse initializations
    inits = []
    
    if provided_seq is not None:
        a_interp = interpolate_sequence(provided_seq, n)
        inits.append(('provided', a_interp / np.sum(a_interp)))
    
    inits.append(('spectral', spectral_init(n, rng, n_iter=30)))
    inits.append(('ramp_lin', ramp_init(n, rng, 'linear')))
    inits.append(('ramp_quad', ramp_init(n, rng, 'quadratic')))
    inits.append(('random', random_init(n, rng)))
    inits.append(('sinusoidal', ramp_init(n, rng, 'sinusoidal')))
    inits.append(('bump', ramp_init(n, rng, 'bump')))
    
    n_inits = len(inits)
    time_per_init = time_limit / (n_inits * 2)  # Leave time for local search
    
    for i, (name, a0) in enumerate(inits):
        if time.monotonic() - t0 > time_limit - 30:
            break
        
        t_remaining = time_limit - (time.monotonic() - t0)
        t_adam = min(time_per_init, t_remaining * 0.6)
        t_local = min(time_per_init, t_remaining * 0.35)
        
        if t_adam < 5:
            continue
        
        # Phase A: Adam optimization
        theta_best, score_adam = optimize_adam(n, a0, t_adam, rng)
        a_adam = softmax_normalize(theta_best) * np.sqrt(2.0 * n)
        
        # Phase B: Local search polish
        a_norm = a_adam / np.sum(a_adam)
        a_local, score_local = gradient_based_local_search(a_norm, t_local, rng)
        a_local_scaled = a_local * np.sqrt(2.0 * n)
        
        actual_score = score_fast(a_local_scaled)
        print(f"    Init '{name}': adam={score_adam:.6f}, local={score_local:.6f}, "
              f"actual={actual_score:.6f}")
        
        if actual_score < best_score:
            best_score = actual_score
            best_a = a_local_scaled.copy()
            print(f"    *** New best for n={n}: {best_score:.6f} ***")
    
    return best_a, best_score

"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions. Uses adaptive smoothed-max gradient descent with corrected
gradient formula, gradient-informed local search, multi-length exploration,
and curriculum-style progressive length increase.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max_conv / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def smoothed_max_and_grad(a, tau):
    """
    Compute smoothed max of conv(a,a) and its exact gradient w.r.t. a.
    Uses log-sum-exp for smooth approximation of max.
    """
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    value = max_c + tau * np.log(np.sum(exp_vals))

    # grad[j] = 2 * sum_m w[m] * a[m-j] = 2 * (w cross_correlate a)[j]
    # Computed as: ifft(FFT(w, N) * conj(FFT(a, N)))[j] for j=0..n-1
    N = 2 * n
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    xcorr = np.fft.irfft(fw * np.conj(fa), N)
    grad = 2.0 * xcorr[:n]

    return value, grad, conv


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta_val = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta_val, 0.0)


def spectral_init(n, rng, n_iter=50):
    """Spectral iterative initialization (alternating projections)."""
    a = 1.0 + 0.5 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)

    for it in range(n_iter):
        conv = fft_convolve(a, a)
        alpha = 1.0 / (n * 0.5 + 1.0)
        a = a - alpha * conv[:n]
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def random_init(n, rng):
    """Random positive initialization."""
    a = rng.random(n) + 0.05
    return a / np.sum(a)


def ramp_init(n, rng, style='linear'):
    """Ramp-like initialization with noise."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if style == 'linear':
        a = 1.0 + 0.3 * t
    elif style == 'quadratic':
        a = 1.0 + 0.4 * t ** 2
    elif style == 'sinusoidal':
        a = 1.0 + 0.3 * np.sin(np.pi * t)
    elif style == 'sqrt':
        a = 1.0 + 0.5 * np.sqrt(t)
    elif style == 'bump':
        a = 1.0 + 0.5 * np.exp(-((t - 0.5) ** 2) / 0.03)
    elif style == 'cosine':
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t)
    elif style == 'two_hump':
        a = 1.0 + 0.4 * (np.exp(-((t - 0.3) ** 2) / 0.02) + np.exp(-((t - 0.7) ** 2) / 0.02))
    else:
        a = 1.0 + 0.3 * t
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def interpolate_sequence(a_old, n_new):
    """Interpolate a sequence to a new length."""
    x_old = np.linspace(0, 1, len(a_old))
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a_old)


def optimize_adam(n, a_init, time_limit, rng):
    """
    Adam optimization on softmax parameterization.
    Minimizes smoothed max of conv(a,a).
    """
    scale = np.sqrt(2.0 * n)
    p_init = a_init / np.sum(a_init)
    theta = np.log(p_init + 1e-12).copy()

    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 3.0

    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf
    eval_count = 0

    tau_max = 10.0 / n
    tau_min = 0.005 / n

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        # Exponential decay of tau
        tau = tau_max * (tau_min / tau_max) ** (frac ** 0.6)

        a = softmax_normalize(theta) * scale
        value, g_a, conv = smoothed_max_and_grad(a, tau)

        p = a / scale
        inner = np.dot(p, g_a)
        g_theta = scale * p * (g_a - inner)

        step += 1
        m_vec = beta1 * m_vec + (1 - beta1) * g_theta
        v_vec = beta2 * v_vec + (1 - beta2) * (g_theta * g_theta)
        m_hat = m_vec / (1.0 - beta1 ** step)
        v_hat = v_vec / (1.0 - beta2 ** step)
        lr_eff = base_lr / np.sqrt(step + 1)
        theta = theta - lr_eff * m_hat / (np.sqrt(v_hat) + eps)

        if step % 50 == 0 or step < 5:
            a_check = softmax_normalize(theta) * scale
            sc = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (scale * scale)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()
            eval_count += 1

        if step % 5000 == 0:
            a_check = softmax_normalize(theta) * scale
            sc = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (scale * scale)
            print(f"        Adam: step={step}, sc={sc:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.2e}, t={elapsed:.0f}s, evals={eval_count}")

    return best_theta, best_score
