                    best_local_c = c_full
            
            if best_local_val != ai:
                a[i] = best_local_val
                c = best_local_c
                improved = True
                if c.max() < best_obj:
                    best_obj = c.max()
                    best_a = a.copy()
        
        if not improved:
            break
    
    # Renormalize
    s = best_a.sum()
    if s > 1e-10:
        best_a = best_a * (S / s)
    return best_a


def mirror_descent_loop(a_init, time_budget, initial_eta=1.5, top_k=8, spectral_interval=150):
    """Main optimization: multiplicative weight with adaptive step size."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    
    start = time.monotonic()
    c = fft_convolve_self_full(a)
    best_obj = c.max()
    best_a = a.copy()
    
    eta = initial_eta
    step = 0
    no_improve = 0
    eta_history = []
    
    while time.monotonic() - start < time_budget:
        step += 1
        
        # Find active lags
        top_lags = np.argsort(c)[::-1][:top_k]
        c_max = c[top_lags[0]]
        
        # Compute subgradient
        g = compute_subgradient_fft(a, top_lags)
        g_max = np.abs(g).max()
        if g_max < 1e-15:
            break
        g_norm = g / g_max
        
        # Multiplicative weight update with clipping
        log_step = -eta * g_norm
        log_step = np.clip(log_step, -5.0, 0.0)  # Don't increase any element
        a_new = a * np.exp(log_step)
        
        # Also try a symmetric version (reduce peaks, allow slight increases elsewhere)
        log_step2 = -eta * g_norm
        log_step2 = np.clip(log_step2, -3.0, 0.5)  # Allow small increases
        a_new2 = a * np.exp(log_step2)
        
        a_new = np.maximum(a_new, 0.0)
        a_new2 = np.maximum(a_new2, 0.0)
        
        s1 = a_new.sum()
        s2 = a_new2.sum()
        
        best_trial_obj = c_max
        best_trial_a = a.copy()
        
        if s1 > 1e-10:
            a_new_norm = a_new * (S / s1)
            c1 = fft_convolve_self_full(a_new_norm)
            if c1.max() < best_trial_obj:
                best_trial_obj = c1.max()
                best_trial_a = a_new_norm.copy()
        
        if s2 > 1e-10:
            a_new2_norm = a_new2 * (S / s2)
            c2 = fft_convolve_self_full(a_new2_norm)
            if c2.max() < best_trial_obj:
                best_trial_obj = c2.max()
                best_trial_a = a_new2_norm.copy()
        
        # Also try spectral shaping periodically
        if step % spectral_interval == 0:
            a_spectral = spectral_shape(a, gamma=0.3 + 0.3 * (step % 1000) / 1000.0)
            c_spec = fft_convolve_self_full(a_spectral)
            if c_spec.max() < best_trial_obj:
                best_trial_obj = c_spec.max()
                best_trial_a = a_spectral.copy()
        
        if best_trial_obj < best_obj - 1e-12:
            a = best_trial_a
            c = fft_convolve_self_full(a)
            best_obj = c.max()
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.03, 8.0)
        else:
            no_improve += 1
            eta *= 0.75
            if no_improve > 30:
                # Try a more aggressive reset
                eta = max(eta * 3.0, 0.5)
                no_improve = 0
                # Also try a random perturbation
                if rng_local.random() < 0.3:
                    perturbation = 0.05 * rng_local.random(n)
                    a_pert = a + perturbation * a
                    a_pert = np.maximum(a_pert, 0)
                    s_pert = a_pert.sum()
                    if s_pert > 1e-10:
                        a_pert = a_pert * (S / s_pert)
                        c_pert = fft_convolve_self_full(a_pert)
                        if c_pert.max() < best_obj:
                            a = a_pert
                            c = c_pert
                            best_obj = c_pert.max()
                            best_a = a.copy()
    
    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    global rng_local
    rng = np.random.default_rng(seed)
    rng_local = np.random.default_rng(seed + 999)
    
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    global_best = np.inf
    global_a = None
    
    def update_global(a_candidate, verbose=True):
        nonlocal global_best, global_a
        c = fft_convolve_self_full(a_candidate)
        obj = c.max()
        n = len(a_candidate)
        S = a_candidate.sum()
        actual_obj = 2.0 * n * obj / (S * S)
        if actual_obj < global_best:
            global_best = actual_obj
            global_a = a_candidate.copy()
            if verbose:
                print("  New best: obj=%.6f (n=%d, max_c=%.4f, S=%.2f)" % (actual_obj, n, obj, S))
        return actual_obj
    
    # Try to load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            print("Loaded given sequence: n=%d, sum=%.2f" % (len(given_a), given_a.sum()))
            ver = evaluate_sequence(given_a.tolist())
            if np.isfinite(ver):
                print("  Given sequence objective: %.6f" % ver)
                if ver < global_best:
                    global_best = ver
                    global_a = given_a.copy()
    except Exception as e:
        print("Could not load given sequence: %s" % str(e))
    
    # Phase 1: Exploration with different lengths and initializations (25% of time)
    explore_time = 0.25 * (deadline - time.monotonic())
    print("Phase 1: Exploration (%.0f seconds)" % explore_time)
    
    lengths = []
    if given_a is not None:
        lengths.append(len(given_a))
    for n_try in [2048, 4096, 8192]:
        if n_try not in lengths:
            lengths.append(n_try)
    
    # Try a subset of lengths based on available time
    max_explores = max(3, int(explore_time / 30.0))
    
    explore_attempts = []
    for n in lengths[:max_explores]:
        S_curve = np.sqrt(2.0 * float(n))
        # Different initializations
        inits = []
        
        # Asymmetric Gaussian (peak at 1/3)
        i = np.arange(n)
        a_gauss = np.exp(-0.5 * ((i - n * 0.33) / (n / 8.0)) ** 2)
        a_gauss[n * 0.7:] *= 0.3  # Make it asymmetric
        a_gauss = np.maximum(a_gauss, 1e-10)
        inits.append(("asym_gauss", a_gauss * (S_curve / a_gauss.sum())))
        
        # Ramped sequence
        a_ramp = np.linspace(0.1, 3.0, n) * np.exp(-i / n)
        a_ramp = np.maximum(a_ramp, 1e-10)
        inits.append(("ramp", a_ramp * (S_curve / a_ramp.sum())))
        
        # Random
        a_rand = rng.random(n) ** 2  # Skewed distribution
        a_rand = np.maximum(a_rand, 1e-10)
        inits.append(("random", a_rand * (S_curve / a_rand.sum())))
        
        # Two-hump asymmetric
        a_2h = np.exp(-((i - n * 0.2) / (n / 12.0)) ** 2) + 0.7 * np.exp(-((i - n * 0.6) / (n / 8.0)) ** 2)
        a_2h = np.maximum(a_2h, 1e-10)
        inits.append(("two_hump", a_2h * (S_curve / a_2h.sum())))
        
        # Binary-like (rounded random)
        a_bin = (rng.random(n) > 0.4).astype(float)
        a_bin = np.maximum(a_bin, 0.01)
        inits.append(("binary", a_bin * (S_curve / a_bin.sum())))
        
        for name, a_init in inits:
            if time.monotonic() > deadline - 60.0:
                break
            per_time = min(25.0, (deadline - time.monotonic() - 60.0) / max(1, len(inits) - explore_attempts.count((n, name))))
            if per_time < 5.0:
                break
            
            t0 = time.monotonic()
            a_best, obj_best = mirror_descent_loop(a_init, per_time, initial_eta=2.0, top_k=10)
            elapsed = time.monotonic() - t0
            res = update_global(a_best, verbose=False)
            print("  Explore n=%d %s: obj=%.6f (%.1fs)" % (n, name, res, elapsed))
            explore_attempts.append((n, name))
    
    # Phase 2: Main optimization from best found (50% of time)
    main_time = 0.50 * (deadline - time.monotonic())
    print("Phase 2: Main optimization (%.0f seconds)" % main_time)
    
    if global_a is not None:
        a_work = global_a.copy()
        n = len(a_work)
        S = a_work.sum()
        
        # Split main time into chunks for restarting
        n_chunks = max(2, int(main_time / 120.0))
        chunk_time = main_time / n_chunks
        
        for chunk in range(n_chunks):
            if time.monotonic() > deadline - 30.0:
                break
            remaining = deadline - time.monotonic() - 30.0
            t_use = min(chunk_time, remaining)
            
            if chunk > 0:
                # Add slight perturbation for restart
                pert = 0.02 * rng_local.random(n)
                a_work = a_work * (1.0 + pert)
                a_work = np.maximum(a_work, 0)
                a_work = a_work * (S / a_work.sum())
            
            top_k = 8 + chunk * 2  # Increase top_k for later chunks
            eta_init = 2.0 / (1.0 + 0.3 * chunk)
            
            a_best, obj_best = mirror_descent_loop(a_work, t_use, initial_eta=eta_init, top_k=min(top_k, 20))
            res = update_global(a_best, verbose=False)
            print("  Main chunk %d: obj=%.6f" % (chunk + 1, res))
            a_work = a_best
    
    # Phase 3: Polish with coordinate descent (25% of time)
    polish_time = deadline - time.monotonic() - 20.0
    print("Phase 3: Polish (%.0f seconds)" % polish_time)
    
    if global_a is not None and polish_time > 5.0:
        a_work = global_a.copy()
        n = len(a_work)
        
        # Multiple coordinate descent sweeps
        n_cd_sweps = max(1, int(polish_time / 40.0))
        cd_time_each = min(30.0, polish_time / n_cd_sweps)
        
        for cd_i in range(n_cd_sweps):
            if time.monotonic() > deadline - 15.0:
                break
            a_cd = coord_descent_sweep(a_work, max_sweeps=2, time_limit=cd_time_each)
            res = update_global(a_cd, verbose=False)
            print("  CD sweep %d: obj=%.6f" % (cd_i + 1, res))
            a_work = a_cd
        
        # Final short mirror descent
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 5.0:
            a_final, obj_final = mirror_descent_loop(a_work, remaining, initial_eta=0.5, top_k=15, spectral_interval=50)
            res = update_global(a_final, verbose=False)
            print("  Final MD: obj=%.6f" % res)
    
    # Final verification
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("FINAL RESULT: internal=%.6f, verified=%.6f" % (global_best, ver if np.isfinite(ver) else float('inf')))
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("FALLBACK: obj=%.6f" % ver)
    
    return final_a
