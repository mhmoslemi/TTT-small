def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def local_score(a):
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (np.sum(a) ** 2)

def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def lp_direction(a, M):

    n = len(a)

    # A is (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

LP-guided iterative optimization for minimum peak autocorrelation non-negative sequences.
Key innovation: Uses linear programming to find optimal descent directions,
combined with L-BFGS p-norm refinement and targeted local search.

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]


def local_score(a):
    n = len(a)
    c = fft_convolve(a, a)
    s = np.sum(a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def build_conv_matrix(a):
    """Build the convolution matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A


def lp_find_direction(a, target):
    """Solve LP: max sum(g) s.t. conv(a,g) <= target, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, target)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success and np.sum(res.x) > 1e-12:
        return res.x
    return None

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]


def local_score(a):
    """Score: 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = fft_convolve(a, a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    return 2.0 * n * np.max(c) / (s * s)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def build_conv_matrix(a):
    """Build convolution matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A


def solve_lp(a, A, target):
    """Solve LP: max sum(g) s.t. A@g <= target, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, target)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n, method='highs')
        if res.success and np.sum(res.x) > 1e-12:
            return res.x
    except Exception:
        pass
    return None

    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    if p > 1000:
        # Effectively max
        c_max = np.max(c)
        score = 2.0 * n * c_max
        a_rev = a[::-1]
        cross = fft_convolve(a_rev, (c == c_max).astype(float))
        grad_a = 2.0 * cross[n-1:2*n-1]
        inner = np.dot(grad_a, a)
        grad_theta = a * (grad_a - inner)
        return score, grad_theta
    
    cp = np.maximum(c, 1e-30) ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    
    factor = 4.0 * n * (sum_cp ** (1.0/p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n-1:2*n-1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta

    """Run L-BFGS with p-norm annealing."""
    theta = np.log(a_init + 1e-30)
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))
    t0 = time.monotonic()
    
    for p in p_schedule:
        if time.monotonic() - t0 > time_limit - 10:
            break
        
        try:
            result = minimize(
                pnorm_obj_grad, theta.copy(),
                method='L-BFGS-B', jac=True, args=(p,),

                         'maxfun': maxiter_per_p * 10}

            theta_new = result.x

            s = local_score(a_new)

                best_theta = theta_new.copy()
            theta = best_theta
        except Exception as e:
            print(f"        L-BFGS error at p={p}: {e}")
            break
    
    return softmax(best_theta), best_score

    """Solve LP: max sum(g) s.t. A@g <= target*1, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, target)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n, method='highs',
                      options={'time_limit': 10.0})
        if res.success and np.sum(res.x) > 1e-12:
            return res.x
    except Exception:
        pass
    return None


def lp_phase(a_init, time_limit=120.0, epsilon=0.005, verbose=False):
    """LP-based iterative peak reduction with adaptive epsilon."""
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return a, local_score(a)
    a = a / s

    best = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    n = len(a)

    it = 0
    stale = 0
    for eps in [epsilon, epsilon * 0.5, epsilon * 0.2, 0.0]:
        if time.monotonic() - t0 > time_limit:
            break
        while time.monotonic() - t0 < time_limit:
            it += 1
            c = fft_convolve(a, a)
            M = np.max(c)
            A = build_conv_matrix(a)

            target = M * (1.0 - eps) if eps > 0 else M
            g = solve_lp(a, A, target)

            if g is None:
                if target < M:
                    target = M
                    g = solve_lp(a, A, target)
                if g is None:
                    break

            g_sum = np.sum(g)
            if g_sum < 1e-12:
                break
            g = g / g_sum

            if np.abs(g - a).max() < 1e-7:
                stale += 1
                if stale > 3:
                    break
            else:
                stale = 0

            # Line search
            best_t = 0.0
            best_s = best_score
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                candidate = (1.0 - t) * a + t * g
                candidate = np.maximum(candidate, 0)
                cs = np.sum(candidate)
                if cs < 1e-10:
                    continue
                candidate = candidate / cs
                sc = local_score(candidate)
                if sc < best_s - 1e-13:
                    best_s = sc
                    best_t = t

            if best_t == 0.0:
                break

            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)

            cur_score = local_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                if verbose and it % 30 == 0:
                    print(f"          LP iter {it}: score={best_score:.8f}")

    if verbose:
        print(f"        LP done: {it} iters, score={best_score:.8f}")
    return best, best_score


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)

    if p > 500:
        c_max = np.max(c)
        score = 2.0 * n * c_max
        mask = (c >= c_max - 1e-12).astype(float)
        a_rev = a[::-1]
        cross = fft_convolve(a_rev, mask)
        grad_a = 2.0 * cross[n-1:2*n-1]
        inner = np.dot(grad_a, a)
        grad_theta = a * (grad_a - inner)
        return score, grad_theta

    cp = np.maximum(c, 1e-30) ** (p - 1)
    sum_cp = np.sum(cp * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)

    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm

    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n-1:2*n-1]

    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)

    return score, grad_theta


def run_lbgfs(a_init, p_schedule, time_limit=120.0, maxiter_per_p=300):
    """Run L-BFGS with p-norm annealing."""
    theta = np.log(a_init + 1e-30)
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))
    t0 = time.monotonic()

    for p in p_schedule:
        if time.monotonic() - t0 > time_limit - 10:
            break
        try:
            result = minimize(
                pnorm_obj_grad, theta.copy(),
                method='L-BFGS-B', jac=True, args=(p,),
                options={'maxiter': maxiter_per_p, 'ftol': 1e-15,
                         'gtol': 1e-13, 'maxfun': maxiter_per_p * 8}
            )
            theta_new = result.x
            a_new = softmax(theta_new)
            s = local_score(a_new)
            if s < best_score:
                best_score = s
                best_theta = theta_new.copy()
            theta = best_theta
        except Exception:
            break

    return softmax(best_theta), best_score


def soft_max_obj_grad(theta, tau):
    """Objective and gradient for soft-max smoothed peak."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score = 2.0 * n * soft_max_val
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n-1:2*n-1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def run_lbgfs_softmax(a_init, tau_schedule, time_limit=60.0, maxiter_per_tau=300):
    """Run L-BFGS with soft-max temperature annealing."""
    theta = np.log(a_init + 1e-30)
    best_theta = theta.copy()
    best_score = local_score(softmax(theta))
    t0 = time.monotonic()

    for tau in tau_schedule:

                         'gtol': 1e-13, 'maxfun': maxiter_per_tau * 8}

            if s < best_score:

    return softmax(best_theta), best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width, rng):
    """Gaussian bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def make_edge_init(n, rng):
    """Edge-concentrated initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / (n * 0.15)) ** 2)
    a = 1.0 / (a + 0.1)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def local_search_polish(a_init, time_limit=120.0, rng=None):
    """Targeted local search for final polish."""
    if rng is None:
        rng = np.random.default_rng(0)

    a = a_init.astype(np.float64).copy()
    a = a / np.sum(a)
    best = a.copy()
    best_score = local_score(a)
    n = len(a)
    t0 = time.monotonic()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        candidate = a.copy()
        r = rng.random()

        if r < 0.35:
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=top_k))
            other = np.setdiff1d(np.arange(n), top_idx)
            n_inc = min(12, len(other))
            if n_inc > 0:
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.55:
            # Peak-aware global: reduce where convolution is high
            c = fft_convolve(a, a)
            c_norm = c / np.max(c)
            pull = np.zeros(n)
            for k in range(min(n, len(c))):
                w = max(0.0, c_norm[k] - 0.5)
                for i in range(min(n, k + 1)):
                    j = k - i
                    if 0 <= j < n:
                        pull[i] += w * a[j]
            if np.max(pull) > 1e-12:
                pull = pull / np.max(pull)
                candidate *= np.exp(-0.03 * pull)
        elif r < 0.70:
            width = max(1, int(rng.integers(1, max(2, n // 12))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        elif r < 0.85:
            sym = 0.5 * (a + a[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.015, size=n))
        else:
            candidate = a * np.exp(rng.normal(0.0, 0.012, size=n))

        candidate = np.maximum(candidate, 1e-10)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)

        if new_score < best_score:
            best_score = new_score
            best = candidate.copy()
            a = candidate.copy()
        elif rng.random() < np.exp(-(new_score - best_score) / 0.002):
            a = candidate.copy()

    return best, best_score
