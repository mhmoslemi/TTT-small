LP-based iteration for minimum peak autocorrelation non-negative sequences.
Uses the state-of-the-art LP direction-finding approach combined with
gradient-based refinement and local search.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for sequence a: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def score_normalized(a):
    """Score assuming sum(a) = 1: 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def lp_find_direction(a, M):
    """
    Solve LP: maximize sum(b) subject to b >= 0, conv(a,b) <= M.
    Returns the optimal b.
    """
    n = len(a)
    
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = a[k-j] if valid, else 0
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    
    c_obj = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')

        return None
    
    return res.x

import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def score_normalized(a):
    """Score assuming sum(a)=1: 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def build_conv_matrix(a):
    """Build the convolution constraint matrix A where (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_find_direction(a, M):
    """
    Solve LP: maximize sum(b) subject to b >= 0, conv(a,b) <= M.
    Returns optimal b or None if LP fails.
    """

    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)

        result = minimize(
            pnorm_obj_grad, theta.copy(),
            method='L-BFGS-B', jac=True, args=(p,),
            options={'maxiter': 100, 'ftol': 1e-15, 'gtol': 1e-13}
        )
        if result.x is not None:
            a_new = softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / np.sum(a_new)
            s = score_normalized(a_new)
            if s < best_score:
                best_score = s
                best_a = a_new.copy()
                theta = result.x
                if verbose:
                    print(f"    p-norm p={p}: score={s:.8f} *** improved ***")
    
    # Phase B: soft-max refinement
    tau_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_budget:
            break
        result = minimize(
            soft_max_obj_grad, theta.copy(),
            method='L-BFGS-B', jac=True, args=(tau,),
            options={'maxiter': 150, 'ftol': 1e-15, 'gtol': 1e-13}
        )
        if result.x is not None:
            a_new = softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / np.sum(a_new)
            s = score_normalized(a_new)
            if s < best_score:
                best_score = s
                best_a = a_new.copy()
                theta = result.x
                if verbose:
                    print(f"    soft-max tau={tau}: score={s:.8f} *** improved ***")
    
    return best_a, best_score


def local_search(a_init, time_budget, t0, rng, verbose=True):
    """Local search with targeted peak reduction and diverse moves."""
    current = a_init.copy()
    current_score = score_normalized(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    temperature = 0.0008
    cooling = 0.99995
    count = 0
    
    if verbose:
        print(f"  Local search starting: score={current_score:.8f}, n={n}")
    
    while time.monotonic() - t0 < time_budget:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.35:
            # Targeted: reduce top contributors to the peak
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(5, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.015, 0.008, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 2:
                n_inc = min(8, len(other) // 2)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.015, 0.008, size=n_inc))
        elif r < 0.55:
            # Reduce secondary peaks
            c = fft_convolve(current, current)
            c_mean = np.mean(c)
            # Find positions where c exceeds mean by more than a threshold
            hot = c > c_mean * 1.3
            if np.any(hot):
                hot_idx = np.where(hot)[0]
                # Pick a random hot position
                h = int(rng.choice(hot_idx))
                contribs_h = np.zeros(n)
                for i in range(n):
                    j = h - i
                    if 0 <= j < n:
                        contribs_h[i] = current[i] * current[j]
                top_k_h = max(3, n // 15)
                top_h = np.argsort(contribs_h)[-top_k_h:]
                candidate[top_h] *= np.exp(rng.normal(-0.012, 0.006, size=len(top_h)))
        elif r < 0.75:
            # Block perturbation
            width = max(2, int(rng.integers(2, max(3, n // 12))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.85:
            # Partial symmetrization
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.01, size=n))
        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.012, size=n))
        
        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)
        new_score = score_normalized(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if verbose and int((best_score - 1.5) * 1e5) % 5 == 0:
                    print(f"    LS: score={best_score:.8f} (iter {count})")
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score
        
        temperature = max(temperature * cooling, 1e-12)
        count += 1
        
        if verbose and count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LS iter {count}: current={current_score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * t * t / (n * 1.5))
    a = np.clip(a, 0.01, None)
    a += 0.08 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_random_init(n, rng, spread=0.4):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def make_bell_init(n, width_frac, rng):
    """Bell-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * width_frac
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.005, None)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase LP-based optimization:
    1. LP direction-finding iterations (state-of-the-art)
    2. Gradient-based refinement (p-norm + soft-max)
    3. More LP iterations from improved point
    4. Local search polish
    5. Explore other sequence lengths
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 60.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Load starting sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-6, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        s = score_normalized(start_seq)
        print(f"Starting sequence: n={len(start_seq)}, score={s:.8f}")
        best_score = s
        best_seq = start_seq.copy()
    
    # === Phase 1: LP iteration on starting sequence ===
    print("\n" + "=" * 60)
    print("=== Phase 1: LP-based direction finding ===")
    print("=" * 60)
    
    if best_seq is not None:
        time_alloc_1 = min(800.0, (deadline - t_global) * 0.35)
        a1, s1 = run_lp_iterations(best_seq, max_iter=200, time_limit=time_alloc_1, t0=t_global, verbose=True)
        if s1 < best_score:
            best_score = s1
            best_seq = a1.copy()
            print(f"  Phase 1 best: {best_score:.8f}")
    
    # === Phase 2: Gradient polish ===
    print("\n" + "=" * 60)
    print("=== Phase 2: Gradient-based refinement ===")
    print("=" * 60)
    
    if best_seq is not None:
        time_alloc_2 = min(400.0, deadline - time.monotonic() - 400)
        if time_alloc_2 > 60:
            a2, s2 = gradient_polish(best_seq, time_alloc_2, t_global, verbose=True)
            if s2 < best_score:
                best_score = s2
                best_seq = a2.copy()
                print(f"  Phase 2 best: {best_score:.8f}")
    
    # === Phase 3: LP iteration round 2 ===
    print("\n" + "=" * 60)
    print("=== Phase 3: LP iteration (round 2) ===")
    print("=" * 60)
    
    if best_seq is not None:
        time_alloc_3 = min(400.0, deadline - time.monotonic() - 300)
        if time_alloc_3 > 60:
            a3, s3 = run_lp_iterations(best_seq, max_iter=100, time_limit=time_alloc_3, t0=t_global, verbose=True)
            if s3 < best_score:
                best_score = s3
                best_seq = a3.copy()
                print(f"  Phase 3 best: {best_score:.8f}")
    
    # === Phase 4: Local search polish ===
    print("\n" + "=" * 60)
    print("=== Phase 4: Local search polish ===")
    print("=" * 60)
    
    if best_seq is not None:
        time_alloc_4 = max(120.0, deadline - time.monotonic() - 120)
        a4, s4 = local_search(best_seq, time_alloc_4, t_global, rng, verbose=True)
        if s4 < best_score:
            best_score = s4
            best_seq = a4.copy()
            print(f"  Phase 4 best: {best_score:.8f}")
    
    # === Phase 5: Try alternative starting points ===
    print("\n" + "=" * 60)
    print("=== Phase 5: Alternative starting points ===")
    print("=" * 60)
    
    if time.monotonic() < deadline - 200 and best_seq is not None:
        n_best = len(best_seq)
        # Try perturbing the best sequence
        a_pert = best_seq * np.exp(rng.normal(0, 0.1, n_best))
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert = a_pert / np.sum(a_pert)
        
        time_remaining = deadline - time.monotonic() - 100
        if time_remaining > 100:
            t_split = time_remaining / 3.0
            a5, s5 = run_lp_iterations(a_pert, max_iter=50, time_limit=t_split, t0=t_global, verbose=False)
            if s5 < best_score:
                best_score = s5
                best_seq = a5.copy()
                print(f"  Perturbed LP: {best_score:.8f} *** NEW BEST ***")
            
            if time.monotonic() < deadline - 100:
                a5b, s5b = gradient_polish(a5, min(120.0, deadline - time.monotonic() - 100), t_global, verbose=False)
                if s5b < best_score:
                    best_score = s5b
                    best_seq = a5b.copy()
                    print(f"  Perturbed + grad: {best_score:.8f} *** NEW BEST ***")
    
    # === Phase 6: Try other lengths ===
    print("\n" + "=" * 60)
    print("=== Phase 6: Other sequence lengths ===")
    
    if time.monotonic() < deadline - 150:
        other_lengths = [192, 320, 256, 384]
        for n_try in other_lengths:
            if time.monotonic() > deadline - 80:
                break
            # Skip if same as best
            if best_seq is not None and n_try == len(best_seq):
                continue
            
            a0 = make_chirp_init(n_try, 0.4 + 0.3 * rng.random(), rng)
            time_rem = deadline - time.monotonic() - 80
            if time_rem < 60:
                break
            
            # Quick LP
            t_alloc = min(120.0, time_rem * 0.6)
            a6, s6 = run_lp_iterations(a0, max_iter=30, time_limit=t_alloc, t0=t_global, verbose=False)
            
            # Quick gradient
            if s6 < best_score and time.monotonic() < deadline - 60:
                a6b, s6b = gradient_polish(a6, min(100.0, deadline - time.monotonic() - 50), t_global, verbose=False)
                if s6b < s6:
                    s6 = s6b
                    a6 = a6b
            
            print(f"  n={n_try}: score={s6:.8f} {'*** NEW BEST ***' if s6 < best_score else ''}")
            if s6 < best_score:
                best_score = s6
                best_seq = a6.copy()
    
    # === Phase 7: Final LP + gradient squeeze ===
    print("\n" + "=" * 60)
    print("=== Phase 7: Final squeeze ===")
    print("=" * 60)
    
    if best_seq is not None and time.monotonic() < deadline - 50:
        time_final = deadline - time.monotonic() - 30
        # Quick LP
        a7, s7 = run_lp_iterations(best_seq, max_iter=30, time_limit=time_final * 0.5, t0=t_global, verbose=False)
        if s7 < best_score:
            best_score = s7
            best_seq = a7.copy()
            print(f"  Final LP: {best_score:.8f} *** NEW BEST ***")
        
        if time.monotonic() < deadline - 40:
            a7b, s7b = gradient_polish(best_seq, time_final * 0.3, t_global, verbose=False)
            if s7b < best_score:
                best_score = s7b
                best_seq = a7b.copy()
                print(f"  Final grad: {best_score:.8f} *** NEW BEST ***")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"=== FINAL: evaluate_sequence score={final_score:.8f}, n={len(final_list)} ===")
        print(f"{'=' * 60}")
        return final_list
    
    # Fallback
    return [1.0] * 256
