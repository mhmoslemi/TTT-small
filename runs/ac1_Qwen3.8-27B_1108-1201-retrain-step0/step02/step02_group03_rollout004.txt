"""
LP-guided optimization for minimizing peak convolution of non-negative sequences.
Uses linear programming to find descent directions, combined with gradient
descent polish and multiple initializations at different sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Fast linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def build_constraint_matrix(a):
    """Build the P matrix where P[k,m] = a[k-m] if valid."""
    n = len(a)
    P = np.zeros((2 * n - 1, n))
    for m in range(n):
        P[m:m + n, m] = a
    return P


def solve_lp_direction(a, M):
    """Solve the LP to find the direction b that maximizes sum(b) s.t. conv(a,b) <= M."""
    n = len(a)
    P = build_constraint_matrix(a)
    c = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=P, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_iteration(a, M, alpha_candidates=None):
    """One LP-based iteration: find direction, do line search, update."""
    n = len(a)
    S = np.sum(a)
    b_star = solve_lp_direction(a, M)
    if b_star is None or np.sum(b_star) < 1e-15:
        return a, 0.0
    
    # Rescale b_star to have same sum as a
    b_norm = b_star * (S / np.sum(b_star))
    
    # Line search
    if alpha_candidates is None:
        alpha_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    
    best_score = local_score(a)
    best_a = a.copy()
    best_alpha = 0.0
    
    for alpha in alpha_candidates:
        candidate = (1 - alpha) * a + alpha * b_norm
        if np.any(candidate < -1e-10):
            continue
        candidate = np.maximum(candidate, 0)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        score = local_score(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate
            best_alpha = alpha
    
    return best_a, best_score


def fft_phase_init(n, rng, n_iter=80):
    """Gerchberg-Saxton initialization."""
    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, 2 * n)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def chirp_init(n, rng, alpha=0.5):
    """Chirp-based initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def multi_tone_init(n, rng, n_tones=5):

    a = a / S
    theta = np.log(a + 1e-15)

    m2 = np.zeros(n)

    best_a = a.copy()

def local_search_refine(a_init, time_alloc, rng):

    a = a_init.astype(np.float64).copy()
    current_score = local_score(a)
    best = a.copy()
    best_score = current_score
    n = len(a)
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_alloc:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.4:
            # Peak-directed: find peak of conv, reduce contributing elements
            c = fft_convolve(a, a)
            peak_idx = np.argmax(c)
            # Elements contributing to this peak
            i_min = max(0, peak_idx - n + 1)
            i_max = min(n - 1, peak_idx)
            contrib = np.zeros(n)
            for i in range(i_min, i_max + 1):
                j = peak_idx - i
                contrib[i] += a[i] * a[j]
                contrib[j] += a[i] * a[j]
            # Reduce top contributors, increase bottom
            n_mod = max(2, n // 8)
            top = np.argsort(contrib)[-n_mod:]
            bot = np.argsort(contrib)[:n_mod]
            candidate[top] *= np.exp(rng.normal(-0.05, 0.02, n_mod))
            candidate[bot] *= np.exp(rng.normal(0.05, 0.02, n_mod))
        elif r < 0.6:
            # Block perturbation
            n_blocks = rng.integers(2, 5)
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 20))))
                bs = int(rng.integers(0, n - bw))
                candidate[bs:bs+bw] *= np.exp(rng.normal(0, 0.04, bw))
        elif r < 0.8:
            # Symmetry nudge
            sym = 0.5 * (a + a[::-1])
            candidate = a + 0.1 * (sym - a)
            candidate += 0.005 * rng.standard_normal(n)
        else:
            # Small global noise
            candidate = a * np.exp(rng.normal(0, 0.01, n))
        
        candidate = np.maximum(candidate, 0)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s * np.sum(a)
        
        new_score = local_score(candidate)
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 5:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    a = candidate
                    current_score = new_score
        
        temp *= cooling
    
    return best, best_score
