def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def gs_init(n, rng, n_iter=60, n_seed=16):
    """Gerchberg-Saxton with random phase seeds for diversity."""

        # Periodic evaluation
        if step % 300 == 0:
            a_check = np.exp(best_theta - np.max(best_theta))
            a_check = a_check / np.sum(a_check) * S
            score_best = local_score(a_check)
            
            a_curr = np.exp(theta - np.max(theta))
            a_curr = a_curr / np.sum(a_curr) * S
            score_curr = local_score(a_curr)
            
            if score_curr < best_score:
                best_score = score_curr
                best_theta = theta.copy()
            
            if step % 1500 == 0:
                print(f"            step={step}, curr={score_curr:.6f}, best={best_score:.6f}, tau={tau:.4f}, t={elapsed:.1f}s")
    
    a_final = np.exp(best_theta - np.max(best_theta))
    a_final = a_final / np.sum(a_final) * S
    score_final = local_score(a_final)
    return a_final, score_final


def directed_local_search(a_init, time_limit, rng, top_k=10):
    """
    Directed local search: identify top contributing lags and redistribute mass.
    Uses simulated annealing acceptance.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.9997
    iter_count = 0
    stagnant = 0
    max_stagnant = 5000
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        
        candidate = current.copy()
        r = rng.random()
        iter_count += 1
        
        if r < 0.40:
            # Peak-directed: reduce top contributors, boost bottom
            conv = fft_convolve(current, current)
            n_top = min(top_k, len(conv))
            top_lags = np.argsort(conv)[-n_top:]
            contrib = np.zeros(n)
            for lag in top_lags:
                i_min = max(0, lag - n + 1)
                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            n_reduce = max(3, n // 12)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.03, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.03, size=len(bot_idx)))
        elif r < 0.55:
            # Block swap: move mass from one region to another
            w = max(1, int(rng.integers(2, max(3, n // 15))))
            s1 = int(rng.integers(0, n - 2 * w))
            s2 = int(rng.integers(0, n - 2 * w))
            if abs(s1 - s2) < w:
                s2 = (s2 + n // 2) % (n - 2 * w)
            amt = rng.uniform(0.05, 0.15)
            candidate[s1:s1+w] *= (1.0 - amt)
            candidate[s2:s2+w] *= (1.0 + amt)
        elif r < 0.70:
            # Gaussian-shaped perturbation
            center = int(rng.integers(0, n))
            width = max(2, int(rng.integers(2, n // 6)))
            t = np.arange(n)
            pert = np.exp(-0.5 * ((t - center) / width) ** 2)
            amp = rng.normal(0, 0.04)
            candidate *= (1.0 + amp * pert)
        elif r < 0.80:
            # Anti-correlation: reduce where convolution is high, based on local structure
            conv = fft_convolve(current, current)
            # Find the peak lag
            peak_lag = np.argmax(conv)
            # Elements that contribute to this lag
            i_min = max(0, peak_lag - n + 1)
            i_max = min(n - 1, peak_lag)
            # Reduce endpoints more (they only contribute to one lag)
            if r < 0.75:
                # Reduce the element with highest individual contribution
                contribs = []
                for i in range(i_min, i_max + 1):
                    j = peak_lag - i
                    contribs.append(current[i] * current[j])
                if contribs:
                    max_ci = np.argmax(contribs)
                    idx = i_min + max_ci
                    candidate[idx] *= np.exp(rng.normal(-0.1, 0.03))
                    # Boost a random other element
                    other = int(rng.integers(0, n))
                    if other != idx:
                        candidate[other] *= np.exp(rng.normal(0.1, 0.03))
            else:
                # Reduce all contributors slightly
                for i in range(i_min, i_max + 1):
                    candidate[i] *= np.exp(rng.normal(-0.02, 0.01))
        elif r < 0.90:
            # Symmetrization step
            sym = 0.5 * (current + current[::-1])
            alpha = rng.uniform(0.05, 0.2)
            candidate = current + alpha * (sym - current)
            candidate += 0.005 * rng.standard_normal(n)
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0, 0.01, size=n))
        
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / s * np.sum(current)
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            stagnant = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if iter_count % 1000 == 0:
                    print(f"          iter={iter_count}, new best={best_score:.6f}, t={time.monotonic()-t0:.1f}s")
        else:
            stagnant += 1
            delta = new_score - current_score
            if delta < temp * 10:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score
        
        temp *= cooling
        
        # If stagnant for too long, restart from best with perturbation
        if stagnant > max_stagnant and time.monotonic() - t0 < time_limit - 30:
            print(f"          Restarting after {stagnant} stagnant iters, score={best_score:.6f}")
            current = best * np.exp(rng.normal(0, 0.15, size=n))
            current = np.clip(current, 0.01, None)
            s = np.sum(current)
            if s > 0:
                current = current / s * np.sum(best)
            else:
                current = best.copy()
            current_score = local_score(current)
            stagnant = 0
            temp = 0.003
    
    return best, best_score


def upsample_refine(a_short, target_n, rng):
    """Upsample a shorter sequence to target_n with noise."""
    n = len(a_short)
    x_old = np.linspace(0, 1, n + 1)[:-1]
    x_new = np.linspace(0, 1, target_n + 1)[:-1]
    a_interp = np.interp(x_new, x_old, a_short)
    a_interp += 0.03 * rng.standard_normal(target_n)
    a_interp = np.clip(a_interp, 0.005, None)
    return a_interp / np.sum(a_interp)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: diverse init -> gradient optimize -> directed local search -> upsample refine.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Evaluate provided sequence if available
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()
    
    # Phase 1: Diverse initialization across multiple lengths
    print("\n=== Phase 1: Initialization ===")
    candidates = []
    
    lengths = [256, 384, 512, 768]
    
    for n in lengths:
        if time.monotonic() > deadline - 180:
            break
        
        # GS init
        a_gs = gs_init(n, rng, n_iter=40, n_seed=8)
        s_gs = local_score(a_gs)
        print(f"  n={n}: GS score={s_gs:.6f}")
        candidates.append((s_gs, n, a_gs, 'gs'))
        
        # Chirp-gaussian
        a_cg = chirp_gaussian_init(n, rng)
        s_cg = local_score(a_cg)
        print(f"  n={n}: ChirpGauss score={s_cg:.6f}")
        candidates.append((s_cg, n, a_cg, 'cg'))
        
        # Multi-scale
        a_ms = multi_scale_init(n, rng)
        s_ms = local_score(a_ms)
        print(f"  n={n}: MultiScale score={s_ms:.6f}")
        candidates.append((s_ms, n, a_ms, 'ms'))
    
    # Add provided sequence as candidate
    if best_seq is not None:
        candidates.append((best_score, len(best_seq), best_seq, 'provided'))
    
    # Sort by score
    candidates.sort(key=lambda x: x[0])
    print(f"\n  Top 5: {[(f'{c[0]:.4f}', c[1], c[3]) for c in candidates[:5]]}")
    
    # Phase 2: Gradient optimization on top candidates
    print("\n=== Phase 2: Gradient optimization ===")
    n_cand = min(6, len(candidates))
    remaining = deadline - time.monotonic() - 200  # reserve for local search
    
    # Allocate time: more time for shorter sequences (faster iterations)
    optimized = []
    for i in range(n_cand):
        if time.monotonic() > deadline - 120:
            break
        
        score0, n, a, name = candidates[i]
        # Time allocation inversely proportional to n
        base_time = remaining / n_cand
        t_alloc = base_time * (400.0 / n)
        t_alloc = max(10.0, min(t_alloc, deadline - time.monotonic() - 60))
        
        if t_alloc < 8:
            break
            
        print(f"  [{i+1}/{n_cand}] n={n}, init={name}, score0={score0:.6f}, t_alloc={t_alloc:.1f}s")
        
        # Try two different var_weights
        a_opt1, s_opt1 = gradient_optimize(n, a, t_alloc * 0.6, rng, var_weight=0.05)
        print(f"    var_w=0.05: score={s_opt1:.6f}")
        
        if deadline - time.monotonic() > 30:
            a_opt2, s_opt2 = gradient_optimize(n, a, t_alloc * 0.4, rng, var_weight=0.15)
            print(f"    var_w=0.15: score={s_opt2:.6f}")
            if s_opt2 < s_opt1:
                a_opt1, s_opt1 = a_opt2, s_opt2
        
        if s_opt1 < best_score:
            best_score = s_opt1
            best_seq = a_opt1.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")
        
        optimized.append((s_opt1, n, a_opt1))
    
    # Phase 3: Directed local search on best
    print(f"\n=== Phase 3: Directed local search ===")
    if best_seq is not None:
        polish_time = min(150.0, max(20.0, deadline - time.monotonic() - 30))
        print(f"  Polishing n={len(best_seq)}, score={best_score:.6f}, time={polish_time:.1f}s")
        polished, p_score = directed_local_search(best_seq, polish_time, rng, top_k=8)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
    
    # Phase 4: Try upsample + refine on best short sequence
    print(f"\n=== Phase 4: Upsample and refine ===")
    if optimized:
        optimized.sort(key=lambda x: x[0])
        for target_n in [512, 768, 1024]:
            if time.monotonic() > deadline - 60:
                break
            best_short = optimized[0]
            if best_short[1] >= target_n:
                continue
            a_up = upsample_refine(best_short[2], target_n, rng)
            s_up = local_score(a_up)
            t_alloc = min(60.0, deadline - time.monotonic() - 40)
            if t_alloc < 15:
                break
            print(f"  Upsample to n={target_n}: initial={s_up:.6f}, t_alloc={t_alloc:.1f}s")
            a_opt, s_opt = gradient_optimize(target_n, a_up, t_alloc * 0.7, rng, var_weight=0.05)
            print(f"    After grad: score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
            # Quick local search
            if deadline - time.monotonic() > 30 and best_seq is not None and s_opt < best_score + 0.01:
                ls_time = min(30.0, deadline - time.monotonic() - 20)
                if ls_time > 10:
                    pol2, s2 = directed_local_search(best_seq, ls_time, rng, top_k=6)
                    if s2 < best_score:
                        best_score = s2
                        best_seq = pol2.copy()
                        print(f"    After LS: score={best_score:.6f}")
    
    # Phase 5: Final polish loop - alternate between grad and local search
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None:
        n_best = len(best_seq)
        while time.monotonic() < deadline - 25:
            remaining_t = deadline - time.monotonic() - 25
            if remaining_t < 15:
                break
            
            # Alternate: gradient then local search
            t_grad = min(remaining_t * 0.5, 40.0)
            t_ls = min(remaining_t - t_grad - 10, 40.0)
            
            if t_grad > 10:
                a_new, s_new = gradient_optimize(n_best, best_seq, t_grad, rng, var_weight=0.03)
                if s_new < best_score:
                    best_score = s_new
                    best_seq = a_new.copy()
                    print(f"  Grad polish: {best_score:.6f}")
            
            if t_ls > 10 and time.monotonic() < deadline - 30:
                pol, s_pol = directed_local_search(best_seq, t_ls, rng, top_k=6)
                if s_pol < best_score:
                    best_score = s_pol
                    best_seq = pol.copy()
                    print(f"  LS polish: {best_score:.6f}")
    
    # Final verification
    print(f"\n{'='*50}")
    print(f"Best score: {best_score:.6f}, n={len(best_seq) if best_seq is not None else 0}")
    
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Verify with evaluate_sequence
        try:
            final_score = evaluate_sequence(final_list)
            print(f"Verified score: {final_score:.6f}")
            if final_score < best_score:
                best_score = final_score
        except Exception as e:
            print(f"Verification error: {e}")
        return final_list
    
    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
