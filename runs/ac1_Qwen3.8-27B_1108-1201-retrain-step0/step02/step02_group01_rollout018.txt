- **Strongly convex regularization**: Add a quadratic regularization to make the problem strongly convex (and thus FW converges linearly).

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def softmax_grad(c, T):
    """Compute soft-max value and weights."""
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = shifted - shifted.max()
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    obj = c_max + np.log(total) / T
    weights = exp_shifted / total
    return obj, weights
