"""
Optimized search for low-peak self-convolution sequences.
Uses projected gradient descent with annealed log-sum-exp,
scipy L-BFGS-B refinement, and coordinate descent polish.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def smooth_score_and_grad(a, tau):
    """
    Compute smooth approximation of 2n*max(conv(a,a))/sum(a)^2
    and its gradient using log-sum-exp.
    Assumes a >= 0 and sum(a) > 0.
    """
    n = len(a)
    S = np.sum(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    lse = c_max + tau * np.log(sum_exp)
    
    score = 2.0 * n * lse / (S * S)
    
    # grad_lse = 2 * cross_correlation(a, w)
    # cross_correlation(a, w)[j] = sum_m a[m] * w[m+j]
    flip_a = a[::-1]
    conv_full = fft_convolve(flip_a, w)
    corr = conv_full[n-1:2*n-1]
    grad_lse = 2.0 * corr
    
    # grad of score = (2n/S^2) * grad_lse - (4n*lse/S^3) * S * ones = (2n/S^2)*(grad_lse - 2*lse/S * ones)
    grad_score = (2.0 * n / (S * S)) * (grad_lse - 2.0 * lse / S * np.ones(n))
    
    return score, grad_score

def adam_step(params, grad, m_buf, v_buf, step, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Standard Adam update."""
    m_buf = beta1 * m_buf + (1 - beta1) * grad
    v_buf = beta2 * v_buf + (1 - beta2) * grad * grad
    m_hat = m_buf / (1.0 - beta1 ** step)
    v_hat = v_buf / (1.0 - beta2 ** step)
    params = params - lr * m_hat / (np.sqrt(v_hat) + eps)
    return params, m_buf, v_buf

def optimize_adam(a_init, time_limit, rng, verbose=True):
    """
    Adam optimization in a-space with projection to {a >= 0, sum(a) = 1}.
    Uses annealed log-sum-exp temperature.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S  # normalize to sum = 1
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    step = 0
    
    best_a = a.copy()
    best_score = compute_score(a)
    no_improve = 0
    patience = 2000
    
    t0 = time.monotonic()
    if verbose:
        print(f"  Start: score={best_score:.6f}")
    
    # Temperature schedule: start high, decrease to near 0
    tau_start = 1.0 / n
    tau_end = 0.001 / n
    
    while time.monotonic() - t0 < patience and time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        
        # Anneal temperature
        tau = tau_start * (tau_end / tau_start) ** frac
        
        step += 1
        # Adaptive learning rate
        lr = 5.0 / np.sqrt(step + 1)
        if frac > 0.8:
            lr *= 0.3
        
        score, grad = smooth_score_and_grad(a, tau)
        
        # Adam step
        a_new, m_buf, v_buf = adam_step(a, grad, m_buf, v_buf, step, lr)
        
        # Project: non-negative and sum = 1
        a_new = np.maximum(a_new, 0.0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / S_new
        
        a = a_new
        
        # Check improvement every 50 steps
        if step % 50 == 0:
            actual = compute_score(a)
            if actual < best_score:
                best_score = actual
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 50
            
            if no_improve >= patience:
                # Restart with perturbation
                pert = rng.standard_normal(n) * 0.1
                a = np.maximum(best_a + pert, 0.0)
                S_p = np.sum(a)
                if S_p > 1e-10:
                    a = a / S_p
                else:
                    a = best_a.copy()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                step = 0
                no_improve = 0
                if verbose:
                    print(f"  Restart at t={elapsed:.0f}s, best={best_score:.6f}")
            
            if verbose and step % 5000 == 0:
                print(f"  t={elapsed:.0f}s, step={step}, score={actual:.6f}, best={best_score:.6f}, tau={tau:.2e}")
    
    # Final check
    final = compute_score(best_a)
    if final < best_score:
        best_score = final
    
    if verbose:
        print(f"  End: best={best_score:.6f} (computed: {final:.6f})")
    
    return best_a, best_score

def lbfgsb_refine(a_init, time_limit, rng):
    """
    Refine using scipy L-BFGS-B.
    We optimize 2n * lse(a, tau) subject to a >= 0, sum(a) = 1.
    Since sum(a)=1 is a linear constraint, we fix a[0] = 1 - sum(a[1:])
    and work with a[1:].
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    S = np.sum(a)
    a = a / S
    
    tau = 0.01 / n
    
    def objective(x):
        """x is a[1:], a[0] = 1 - sum(x)."""
        a_full = np.empty(n)
        a_full[0] = 1.0 - np.sum(x)
        a_full[1:] = x
        if np.any(a_full < -1e-10):
            return 1e6
        a_full = np.maximum(a_full, 0.0)
        s, _ = smooth_score_and_grad(a_full, tau)
        return s
    
    def gradient(x):
        a_full = np.empty(n)
        a_full[0] = 1.0 - np.sum(x)
        a_full[1:] = x
        a_full = np.maximum(a_full, 0.0)
        s, g = smooth_score_and_grad(a_full, tau)
        # Project gradient to sum=1 constraint: subtract mean
        # Actually, the constraint is a[0] + sum(a[1:]) = 1
        # So da[0] = -sum(da[1:])
        # The reduced gradient for x = a[1:] is:
        # dx_j = g_j - g_0 (for j >= 1)
        g_reduced = g[1:] - g[0]
        return g_reduced
    
    x0 = a[1:].copy()
    
    t0 = time.monotonic()
    result = minimize(
        fun=objective,
        x0=x0,
        jac=gradient,
        method='L-BFGS-B',
        bounds=[(0.0, None)] * (n - 1),
        options={'maxiter': 5000, 'maxfun': 50000, 'ftol': 1e-12, 'gtol': 1e-10}
    )
    
    elapsed = time.monotonic() - t0
    a_res = np.empty(n)
    a_res[0] = 1.0 - np.sum(result.x)
    a_res[1:] = result.x
    a_res = np.maximum(a_res, 0.0)
    S_res = np.sum(a_res)
    if S_res > 1e-10:
        a_res = a_res / S_res
    
    score = compute_score(a_res)
    print(f"  L-BFGS-B: score={score:.6f}, time={elapsed:.1f}s, success={result.success}")
    
    return a_res, score

def coordinate_descent(a_init, time_limit, rng):
    """
    Coordinate descent: for each coordinate, try to improve by adjusting its value.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    S = np.sum(a)
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-10:
                continue
            
            # Try scaling a[j] by various factors
            for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                trial = a.copy()
                old_val = trial[j]
                new_val = old_val * factor
                trial[j] = new_val
                # Renormalize
                S_t = np.sum(trial)
                if S_t < 1e-10:
                    continue
                trial = trial / S_t
                
                s = compute_score(trial)
                if s < best_score:
                    a = trial
                    best_score = s
                    best_a = a.copy()
                    improved = True
                    old_val = new_val
                elif s > old_val * 1.5:  # waste
                    break
                # else: keep trying (the landscape might be non-monotone)
        
        if not improved:
            # Small random perturbation
            j = int(rng.integers(0, n))
            a[j] = max(0.0, a[j] * np.exp(rng.normal(0, 0.3)))
            S_t = np.sum(a)
            if S_t > 1e-10:
                a = a / S_t
            s = compute_score(a)
            if s < best_score * 1.01:
                pass  # accept
            else:
                a = best_a.copy()
    
    return best_a, best_score

def make_init(n, rng, method='uniform_perturb'):
    """Create initial sequences."""
    if method == 'uniform_perturb':
        a = np.ones(n) * (1.0 + 0.3 * rng.standard_normal(n))
        return np.clip(a, 0.01, None)
    elif method == 'two_peak':
        a = np.ones(n)
        a[:n//4] = 1.5
        a[3*n//4:] = 1.5
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'raised_cosine':
        t = np.arange(n) / n
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        return a + 0.01 * rng.standard_normal(n)
    elif method == 'broadband':
        a = np.ones(n)
        t = np.arange(n, dtype=np.float64)
        for k in range(1, 10):
            freq = k * (1 + 0.5 * rng.random())
            phase = rng.uniform(0, 2*np.pi)
            a += (0.2/np.sqrt(k)) * np.cos(2*np.pi*freq*t/n + phase)
        return np.clip(a, 0.01, None)
    else:
        return np.ones(n) / n
