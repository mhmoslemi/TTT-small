import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def update_conv(c, a, j, delta):
    """Update convolution c when a[j] changes by delta."""
    n = len(a)
    c = c.copy()
    # For k in [j, j+n-1]: c[k] += 2*delta*a[k-j]

    return c

    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    # Normalize to sum = n
    S = np.sum(a)
    if S > 0:
        a = a * n / S
    c = fft_convolve(a, a)
    current_score = 2.0 * n * np.max(c) / n**2  # = 2*max(c)/n
    best = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    lr = 0.01
    K = 5  # number of active lags to track
    
    step = 0
    while time.monotonic() - t0 < budget:
        # Compute active set
        c = fftConvolve(a, a)  # or maintain incrementally
        S = np.sum(a)
        M = np.max(c)
        current_score = 2.0 * n * M / (S * S)
        
        if current_score < best_score:
            best_score = current_score
            best = a.copy()
        
        # Find top K lags
        top_k = np.argpartition(c, -K)[-K:]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in top_k:
            # grad[j] += 2*a[k-j] for j where 0 <= k-j < n
            # i.e., j in range [max(0, k-n+1), min(n-1, k)]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            # a[k-j] for j in [j_start, j_end] is a[k-j_end : k-j_start+1] in reverse
            # a[k-j] where j goes from j_start to j_end:
            # when j = j_start: a[k - j_start]
            # when j = j_end: a[k - j_end]
            # So the values are a[k-j_end], a[k-j_end+1], ..., a[k-j_start]
            # which is a[k-j_end : k-j_start+1]
            grad[j_start:j_end+1] += 2.0 * a[k-j_end:k-j_start+1]
        
        # Scale by 1/K (average subgradient)
        grad /= K
        
        # Full gradient of f = 2n*M/S^2
        # df/da[j] = (2n/S^2)*grad[j] - 4n*M/S^3
        full_grad = (2.0 * n / S**2) * grad - 4.0 * n * M / S**3
        
        # Take step
        a -= lr * full_grad
        a = np.clip(a, 0, 1000)
        
        # Re-normalize (optional, since problem is scale-invariant)
        S = np.sum(a)
        if S < 0.01:
            a = a_init.copy()  # restart
            c = fft_convolve(a, a)
            continue
        
        step += 1
        if step % 5000 == 0:
            lr = max(0.001, lr * 0.95)
    
    return best, best_score

"""
Optimization of non-negative step functions to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses subgradient descent with active-set tracking, coordinate descent refinement,
and multiple initializations including structured (Gaussian, chirp) and random seeds.
Key insight: non-uniform sequences can beat the uniform score of 2.0 by flattening
the autocorrelation peak away from c[0]=a[0]^2 and the center lag.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(c, n, S):
    """Compute score from precomputed convolution."""
    if S < 0.01:
        return np.inf
    return 2.0 * n * np.max(c) / (S * S)


def subgradient_of_maxc(a, c, n, K):
    """Compute the averaged subgradient of max(c) using top-K active lags."""
    num_c = len(c)
    if num_c <= K:
        top_k = np.arange(num_c)
    else:
        top_k = np.argpartition(c, num_c - K)[num_c - K:]
    
    grad = np.zeros(n)
    for k in top_k:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        # a[k-j] for j in [j_start, j_end] is a[k-j_end:k-j_start+1] reversed
        grad[j_start:j_end + 1] += 2.0 * a[k - j_end:k - j_start + 1][::-1]
    grad /= K
    return grad


def full_score_gradient(a, c, n, K):
    """Compute full gradient of score = 2n*max(c)/S^2."""
    S = np.sum(a)
    M = np.max(c)
    if S < 1e-10:
        return np.zeros(n)
    grad_M = subgradient_of_maxc(a, c, n, K)
    grad = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M / (S ** 3))
    return grad


def coordinate_descent_refine(a, c, n, budget, rng):
    """Coordinate descent: for each j, try to find better a[j]."""
    S = np.sum(a)
    current_score = compute_score(c, n, S)
    best = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < budget:
        improved = False
        for j in range(n):
            old_val = a[j]
            if old_val < 1e-15:
                continue
            # Try reducing a[j]: set to 0, half, quarter
            for frac in [0.0, 0.5, 0.25]:
                delta = frac * old_val - old_val
                if abs(delta) < 1e-15:
                    continue
                # Update c
                c_new = c.copy()
                k_start = j
                k_end = min(j + n, 2 * n - 1)
                c_new[k_start:k_end] += 2.0 * delta * a[k_start - j:k_end - j]
                if 2 * j < 2 * n - 1:
                    c_new[2 * j] += delta * delta
                S_new = S + delta
                if S_new < 0.01:
                    continue
                score_new = compute_score(c_new, n, S_new)
                if score_new < current_score - 1e-10:
                    # Accept
                    a[j] = frac * old_val
                    c = c_new
                    S = S_new
                    current_score = score_new
                    improved = True
                    if current_score < best_score:
                        best_score = current_score
                        best = a.copy()
                    break
        
        if not improved:
            # Try small random perturbations
            for _ in range(20):
                j = rng.integers(n)
                a_try = a.copy()
                a_try[j] = max(0.0, a_try[j] * (1.0 + 0.1 * rng.standard_normal()))
                S_try = np.sum(a_try)
                if S_try < 0.01:
                    continue
                c_try = fft_convolve(a_try, a_try)
                s_try = compute_score(c_try, n, S_try)
                if s_try < current_score:
                    a = a_try
                    c = c_try
                    S = S_try
                    current_score = s_try
                    improved = True
                    if s_try < best_score:
                        best_score = s_try
                        best = a.copy()
            if not improved:
                break
    
    return best, best_score


def optimize_sequence(a_init, budget, rng, K=5):
    """Main optimization: subgradient descent with adaptive step + coord descent."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) * 0.1
        S = np.sum(a)
    # Normalize to sum = sqrt(2n) for numerical stability
    target_S = np.sqrt(2.0 * n)
    a = a * (target_S / S)
    
    c = fft_convolve(a, a)
    current_score = compute_score(c, n, np.sum(a))
    best = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    lr = 0.1
    patience = 0
    phase = 'sgd'  # or 'coord'
    
    while time.monotonic() - t0 < budget:
        if phase == 'sgd':
            c = fft_convolve(a, a)
            S = np.sum(a)
            M = np.max(c)
            current_score = compute_score(c, n, S)
            
            if current_score < best_score - 1e-12:
                best_score = current_score
                best = a.copy()
                patience = 0
            else:
                patience += 1
            
            # Switch to coordinate descent after some patience
            if patience > 3000 and time.monotonic() - t0 > budget * 0.3:
                phase = 'coord'
                time_left = budget - (time.monotonic() - t0)
                a, best_score = coordinate_descent_refine(a, c, n, time_left, rng)
                best = a.copy() if best_score == compute_score(fft_convolve(a, a), n, np.sum(a)) else best
                # Continue with SGD from the refined point
                phase = 'sgd'
                a = best.copy()
                patience = 0
                lr = 0.05
                continue
            
            if patience > 10000:
                # Kick
                a = a * (1.0 + 0.03 * rng.standard_normal(n))
                a = np.clip(a, 0, 1000)
                patience = 0
                lr = max(lr, 0.03)
                continue
            
            # Compute gradient
            grad = full_score_gradient(a, c, n, K)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                lr *= 0.5
                continue
            
            # Step with normalization constraint
            a_new = a - lr * grad
            a_new = np.clip(a_new, 0, 1000)
            S_new = np.sum(a_new)
            if S_new < 0.01:
                lr *= 0.5
                continue
            # Re-normalize
            a_new = a_new * (target_S / S_new)
            
            c_new = fft_convolve(a_new, a_new)
            score_new = compute_score(c_new, n, np.sum(a_new))
            
            if score_new <= current_score:
                a = a_new
                c = c_new
                current_score = score_new
                lr = min(lr * 1.005, 0.5)
            else:
                lr *= 0.7
        
        if (time.monotonic() - t0) % 10 < 0.1 and patience % 5000 < 10:
            print(f"    t={time.monotonic()-t0:.0f}s, score={current_score:.8f}, best={best_score:.8f}, phase={phase}")
    
    # Final coordinate descent refinement
    if time.monotonic() - t0 < budget - 5:
        c = fft_convolve(a, a)
        a, s = coordinate_descent_refine(a, c, n, min(30, budget - (time.monotonic() - t0)), rng)
        if s < best_score:
            best_score = s
            best = a.copy()
    
    return best, best_score


def generate_initializations(n, rng, provided_seq=None):
    """Generate diverse initializations."""
    inits = []
    
    # Provided sequence
    if provided_seq is not None and len(provided_seq) == n:
        inits.append(np.clip(provided_seq.astype(np.float64), 0, 1000))
    
    center = (n - 1) / 2.0
    j = np.arange(n, dtype=np.float64)
    
    # Gaussian bumps
    for sigma_frac in [0.15, 0.25, 0.35, 0.5, 0.7]:
        sigma = sigma_frac * n
        a = np.exp(-((j - center) ** 2) / (2.0 * sigma ** 2))
        inits.append(a)
    
    # Raised cosine / Hann window
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * j / n))
    inits.append(a)
    
    # Power-law windows
    for p in [0.5, 1.0, 2.0, 3.0]:
        x = 1.0 - np.abs(j - center) / (n / 2.0)
        a = x ** p
        inits.append(a)
    
    # Chirp-modulated (real, made non-negative)
    for freq in [1, 2, 4]:
        a = 1.0 + 0.7 * np.cos(2.0 * np.pi * freq * j / n + np.pi * j ** 2 / (2.0 * n))
        a = np.maximum(a, 0.0)
        if np.sum(a) > 0.1:
            inits.append(a)
    
    # Two-bump (ring)
    for gap_frac in [0.1, 0.2, 0.3]:
        gap = int(gap_frac * n)
        a = np.ones(n)
        a[n // 2 - gap // 2:n // 2 + gap // 2] = 0.0
        inits.append(a)
    
    # Random smooth (low-pass noise)
    for _ in range(4):
        a = rng.standard_normal(n)
        fa = np.fft.rfft(a, 4 * n)
        cutoff = max(1, n // 8)
        fa[cutoff:] = 0.0
        a = np.fft.irfft(fa, 4 * n)[:n]
        a = np.abs(a) + 0.01
        inits.append(a)
    
    # Random smooth (medium-pass)
    for _ in range(3):
        a = rng.standard_normal(n)
        fa = np.fft.rfft(a, 4 * n)
        cutoff = max(1, n // 4)
        fa[cutoff:] = 0.0
        a = np.fft.irfft(fa, 4 * n)[:n]
        a = np.abs(a) + 0.01
        inits.append(a)
    
    # Pure random
    for _ in range(2):
        a = np.abs(rng.standard_normal(n)) + 0.05
        inits.append(a)
    
    # Binary-like (but continuous: rounded to {0, 0.5, 1})
    for _ in range(2):
        a = (rng.random(n) * 3).astype(int).astype(float) * 0.5
        a = np.maximum(a, 0.05)
        inits.append(a)
    
    return inits
