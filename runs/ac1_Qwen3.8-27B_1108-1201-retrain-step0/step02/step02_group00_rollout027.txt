```python
import numpy as np
import time
from scipy.fft import fft, ifft

# Helper functions (top-level, no closures)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def excess_gradient(a, c):
    """Gradient weighted by excess over mean convolution value."""
    n = len(a)
    excess = np.maximum(c - c.mean(), 0)
    total = excess.sum()
    if total < 1e-30:
        return np.zeros(n)
    w = excess / total
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def topk_ramped_gradient(a, c, k, alpha=1.0):
    """Top-k gradient with power-law ramp weighting."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    # Power-law weighting: top lag gets weight 1, next gets 1/2^alpha, etc.
    ranks = np.arange(1, k+1, dtype=np.float64)
    weights = np.power(ranks, -alpha)
    weights /= weights.sum()
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad
```

```python
def fista_lse_backtrack(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=500.0):

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    step = 1.0 / (2.0 * beta_init)
    it = 0
    
    while time.monotonic() - start < max_time:
        if time.monotonic() - start > max_time:
            break
        it += 1
        beta = min(beta_max, beta_init * np.exp(0.003 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:

            best_local = y.copy()

        gnorm = np.linalg.norm(grad)

            break

        step = 1.0 / (beta * 2.0)
        x_new = project_simplex(y - step * grad, S)

            obj_x = c_x.max()

                # Reduce step
                step *= 0.5
                x_new = project_simplex(y - step * grad, S)
```

```python
def fft_convolve_self(a):

    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

    return a * (S/s)
```

```python
def fista_hybrid(a_init, n, S, max_time, rng):
    a = normalize(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    beta = 5.0
    exact_prob = 0.0  # probability of using exact subgradient
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(1000.0, 5.0 * np.exp(0.005 * it))
        exact_prob = min(0.5, 0.5 * np.exp(0.002 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
        
        # Choose gradient
        if rng.random() < exact_prob:
            grad = exact_subgrad(y, c)
            step = 2.0 * S / (np.linalg.norm(grad) * (1.0 + 0.01 * np.sqrt(it)))
        else:
            grad = lse_grad(y, c, beta)
            step = 1.0 / (2.0 * beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t-1.0)/t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Annealing noise
        if it % 2000 == 0:
            noise_level = 0.002 * S / np.sqrt(n) * np.exp(-0.0001 * it)
            y = project_simplex(y + rng.normal(0, noise_level, n), S)
            t = 1.0
```

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Diverse FISTA exploration with exponential beta annealing,
hybrid LSE/exact-subgradient, multi-scale warm starts, and structured
initializations from window functions. Progressive polishing of the
best candidate found during exploration.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def power_mean_gradient(a, c, p):
    """Gradient of power mean (sum c^p)^(1/p)."""
    n = len(a)
    cp = np.power(np.maximum(c, 1e-30), p)
    tot = cp.sum()
    if tot < 1e-30:
        return np.zeros(n)
    w = np.power(np.maximum(c, 1e-30), p - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag (one-hot)."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def topk_gradient(a, c, k, power=1.0):
    """Top-k gradient with power-law weighting on values."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista_diverse(a_init, n, S, max_time, rng, mode="lse",
                      beta_init=5.0, beta_rate=0.004, beta_max=800.0):
    """FISTA with adaptive smoothing. mode: 'lse', 'power', 'hybrid'."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:

        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        # Choose gradient type
        if mode == "hybrid" and it > 500 and rng.random() < min(0.4, 0.001 * it):
            grad = exact_subgradient(y, c)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            step = S / (gnorm * (1.0 + 0.02 * np.log1p(it / 100.0)))
            step = min(step, 1.0 / (2.0 * beta * 0.1))
        elif mode == "power":

            grad = power_mean_gradient(y, c, p)

        else:

def downsample_and_upsample(a, n_small):
    """Downsample a to n_small by averaging, then back to n by interpolation."""
    n = len(a)
    if n_small >= n:
        return a.copy()
    nblocks = n // n_small
    a_small = a[:nblocks * n_small].reshape(n_small, nblocks).mean(axis=1)
    # Upsample by linear interpolation
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n)
    a_up = np.interp(t_new, t_old, a_small)
    return a_up

        if time.monotonic() > deadline - usable_time * 0.30 - 30:
            break
        arr_n = len(arr)
        if arr_n < 64:
            continue
        S_i = np.sqrt(2.0 * arr_n)
        t_init = per_init_times[i]
        if t_init < 3:
            continue

        # Use different modes for diversity
        modes = ["lse", "hybrid", "power"]
        mode = modes[i % len(modes)]

        a_opt, score = run_fista_diverse(arr, arr_n, S_i, t_init, rng, mode=mode,
                                          beta_init=5.0, beta_rate=0.004, beta_max=600.0)
        a_norm = normalize_sum(a_opt, arr_n)
        c_chk = fft_convolve_self(a_norm)
        real_score = c_chk.max()

        if real_score < global_best:
            global_best = real_score
            global_a = a_norm.copy()
            print("  [%s] n=%d: %.6f (time %.0fs)" % (name, arr_n, global_best, time.monotonic() - p1_start))

    print("=== After exploration: best=%.6f, elapsed=%.0fs ===" % (global_best, time.monotonic() - p1_start))

    # Phase 2: Polish the best
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        remaining = deadline - time.monotonic()
        print("=== Phase 2: Polishing (remaining %.0fs) ===" % remaining)

        if remaining > 10:
            a_pol, s_pol = run_polish(global_a, n_best, S_best, remaining * 0.6, rng)
            a_pol = normalize_sum(a_pol, n_best)
            c_pol = fft_convolve_self(a_pol)
            if c_pol.max() < global_best:
                global_best = c_pol.max()
                global_a = a_pol.copy()
                print("  After polish: %.6f" % global_best)

            # Final micro-polish with very high beta
            remaining2 = deadline - time.monotonic()
            if remaining2 > 5:
                a_fin, s_fin = run_fista_diverse(global_a, n_best, S_best, remaining2 * 0.8,
                                                  rng, mode="hybrid",
                                                  beta_init=200.0, beta_rate=0.001,
                                                  beta_max=5000.0)
                a_fin = normalize_sum(a_fin, n_best)
                c_fin = fft_convolve_self(a_fin)
                if c_fin.max() < global_best:
                    global_best = c_fin.max()
                    global_a = a_fin.copy()
                    print("  After micro-polish: %.6f" % global_best)

    # Phase 3: If we have time, try a different length
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alt_n = n // 2 if n > 4096 else n * 2
        if 1024 <= alt_n <= 32768 and alt_n != n:
            S_alt = np.sqrt(2.0 * alt_n)
            print("=== Phase 3: Trying n=%d (remaining %.0fs) ===" % (alt_n, remaining))
            alt_inits = generate_initializations(alt_n, rng)
            for name2, arr2 in [("alt_rand", alt_inits[0]), ("alt_sin", alt_inits[6])]:
                a_alt, s_alt = run_fista_diverse(arr2, alt_n, S_alt, remaining * 0.3,
                                                  rng, mode="hybrid",
                                                  beta_init=10.0, beta_rate=0.005,
                                                  beta_max=1000.0)
                a_alt = normalize_sum(a_alt, alt_n)
                c_alt = fft_convolve_self(a_alt)
                if c_alt.max() < global_best:
                    global_best = c_alt.max()
                    global_a = a_alt.copy()
                    print("  Alt n=%d [%s]: %.6f" % (alt_n, name2, global_best))

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("Returning sequence of length %d, internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
```
